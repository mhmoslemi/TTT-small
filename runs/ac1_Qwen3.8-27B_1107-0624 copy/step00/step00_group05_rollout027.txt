```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_convolution_matrix(a):
    """Build the (2n-1) x n convolution matrix A where (Ab)_k = (a*b)_k."""
    n = len(a)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # Row k has non-zeros at columns j where max(0, k-(n-1)) <= j <= min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a):
    """Solve the LP: max 1^T b s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    c_conv = np.convolve(a, a, mode='full')
    M = c_conv.max()
    
    # Build constraint matrix
    A = build_convolution_matrix(a)
    
    # Objective: max 1^T b -> min -1^T b
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_ub = np.full(2*n - 1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        b_opt = result.x
        # Normalize to have same sum as a
        sum_a = a.sum()
        sum_b = b_opt.sum()
        if sum_b > 0:
            b_opt = b_opt * (sum_a / sum_b)
        return b_opt, M
    else:
        return a.copy(), M
```

```python
def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    # Use water-filling
    # Find tau such that sum(max(0, v_j - tau)) = target_sum
    n = len(v)
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Try all possible "active sets"
    cumsum = np.cumsum(v_sorted)
    # tau = (cumsum[k] - target_sum) / (k+1) for the right k
    k = np.arange(1, n+1)
    tau_candidates = (cumsum - target_sum) / k
    # We need: tau < v_sorted[k-1] and tau >= v_sorted[k] (for interior k)
    # And the active set is the first k elements
    
    # Efficient approach: binary search for tau
    lo = 0.0
    hi = max(v) + 1
    
    for _ in range(60):  # sufficient for float64 precision
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(0, v - mid))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2
    result = np.maximum(0, v - tau)
    # Ensure exact sum (floating point)
    s = result.sum()
    if s > 0:
        result *= target_sum / s
    return result
```
