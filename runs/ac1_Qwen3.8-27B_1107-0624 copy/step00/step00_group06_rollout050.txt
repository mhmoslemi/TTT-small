def compute_smooth_grad(f, ac, beta=1.0):
    """Compute gradient of smooth max of auto-correlation using FFT."""
    n = len(f)
    
    # Softmax weights
    max_ac = np.max(ac)
    shifted = beta * (ac - max_ac)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # grad[j] = 2 * sum_{i=0}^{n-1} weights[j+i] * f[i]
    # This is cross-correlation of weights and f at lag j
    # Use FFT: IFFT(FFT(weights, N) * conj(FFT(f, N)))[:n] * 2
    # N >= len(weights) + n - 1 = 2n-1 + n - 1 = 3n-2
    N = 1 << (3*n-2 - 1).bit_length()
    W = fft(weights, N)
    F = fft(f, N)
    cross = ifft(W * np.conj(F), N).real[:n]
    
    return 2.0 * cross

            inits.append((g / np.sum(g), "given", len(g)))

    # 2. Random with various lengths
    for n_try in [2048, 4096, 6144, 8192]:
        r = np.random.default_rng(seed + 7919 * n_try)
        arr = r.random(n_try) + 0.02
        inits.append((arr / arr.sum(), f"rand_{n_try}", n_try))

    # 3. Quadratic-residue-like patterns (good autocorrelation properties)
    for n_try in [4096, 8192]:
        p = 1
        while p < n_try + 10:
            p *= 2
            if p % 4 == 1:
                break
        p = n_try + 1
        while True:
            if p % 4 == 1:
                is_prime = True
                for d in range(2, int(p**0.5) + 1):
                    if p % d == 0:
                        is_prime = False
                        break
                if is_prime:
                    break
            p += 4
        if p < 100000:
            qr = np.zeros(n_try)
            for i in range(min(n_try, p - 1)):
                if pow(i, (p - 1) // 2, p) == 1:
                    qr[i] = 1.0
                else:
                    qr[i] = 0.15
            if np.sum(qr) > 0:
                inits.append((qr / qr.sum(), f"qr_{n_try}", n_try))

    # 4. Smoothly varying (sinusoidal modulation)
    for n_try in [4096, 8192]:
        t = np.arange(n_try) / n_try
        arr = 0.5 + 0.45 * np.sin(2 * np.pi * 3 * t) + 0.05
        arr = np.maximum(arr, 0.01)
        inits.append((arr / arr.sum(), f"sin_{n_try}", n_try))

    print(f"Total initializations: {len(inits)}")

    # Optimize each initialization
    for idx, (f, name, n) in enumerate(inits):
        if time.monotonic() > deadline - 60:
            break

        elapsed = time.monotonic() - (deadline - budget_s + 10)
        print(f"[{idx}/{len(inits)}] Starting '{name}' (n={n}), elapsed={elapsed:.0f}s")

        f_opt, sc_opt = _optimize_sequence(f.copy(), deadline, rng, best_score)

        if sc_opt < best_score:
            best = f_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.6f} from '{name}'")
        else:
            print(f"  Done '{name}': score={sc_opt:.6f} (best remains {best_score:.6f})")

    # Final polish on the best solution found
    if best is not None and time.monotonic() < deadline - 10:
        print(f"Polishing best solution (score={best_score:.6f})...")
        best, best_score = _optimize_sequence(best.copy(), deadline, rng, best_score)

    if best is None:
        best = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best.tolist()))

    # Scale to a reasonable range (objective is scale-invariant)
    target_sum = np.sqrt(2.0 * len(best))
    best = best * target_sum / np.sum(best)
    best = np.clip(best, 0.0, 1000.0)

    final_score = float(evaluate_sequence(best.tolist()))
    print(f"Final score: {final_score:.6f} (length={len(best)})")

    return [float(x) for x in best]
