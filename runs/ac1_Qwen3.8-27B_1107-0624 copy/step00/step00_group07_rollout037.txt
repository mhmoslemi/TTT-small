def compute_gradient_max_conv(a, top_k=1):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a.
    
    For a single peak at k*, the gradient of conv(a,a)[k*] w.r.t. a[j] is:
    2*a[k*-j] if 0 <= k*-j <= n-1, else 0.
    
    For multiple peaks, we average the gradients.
    """

    C = np.convolve(a, a)
    M = np.max(C)
    
    # Find all indices where C is close to max (within tolerance)
    tol = 1e-10
    active = np.where(C >= M - tol)[0]
    
    if len(active) > top_k:
        # Use only the top k peaks
        sort_idx = np.argsort(C[active])[::-1][:top_k]
        active = active[sort_idx]

        j_max = min(n - 1, k)

            grad[j_range] += 2.0 * a[k - j_range]

import numpy as np

def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    return np.convolve(a, a)


def compute_gradient_from_conv(C, a, tau):
    """
    Compute gradient of soft_max(C, tau) w.r.t. a, where C = conv(a,a).
    
    The soft-max is: tau * log(sum_k exp(C[k]/tau))
    Its gradient w.r.t. C[k] is: exp(C[k]/tau) / sum_j exp(C[j]/tau) = softmax(C/tau)[k]
    
    Then by chain rule: d(soft_max)/da[j] = sum_k softmax(C/tau)[k] * dC[k]/da[j]

"""
Subgradient descent with softmax-smoothed peak minimization for low-autocorrelation sequences.

Minimizes O(a) = 2n * max(conv(a,a)) / (sum(a))² by normalizing to sum=1 and
using a smooth (log-sum-exp) approximation of the max, with exact FFT-based
gradients, momentum, adaptive step size, temperature annealing, and smart restarts.
"""

import numpy as np
import time


def compute_gradient_from_conv(C, a, tau):
    """
    Compute gradient of soft_max(C, tau) w.r.t. a, where C = conv(a,a).
    
    Uses: grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    computed via FFT cross-correlation.
    """
    n = len(a)
    M = np.max(C)

    # Numerically stable softmax
    C_shift = C - M
    w = np.exp(C_shift / tau)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        peaks = np.where(C >= M - 1e-14)[0]
        w = np.zeros(len(C))
        w[peaks] = 1.0 / len(peaks)
    else:
        w /= w_sum

    # FFT-based cross-correlation: xcorr[j] = sum_m a[m] * w[j+m]
    N = 1
    while N < 3 * n:
        N *= 2

    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(fa * np.conj(fw), N)

    grad = 2.0 * xcorr[:n]
    return grad


def normalize_to_sum1(a):
    """Normalize a to be non-negative with sum exactly 1."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def objective_normalized(a):
    """Compute 2n * max(conv(a,a)) for a sequence with sum=1."""
    n = len(a)
    C = np.convolve(a, a)
    return 2.0 * n * np.max(C)
