import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(f*f)/(sum f)^2 using LP-based iterative refinement
    with FFT-accelerated convolution and adaptive scheduling.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    # Get starting sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base = np.ones(256, dtype=np.float64)
    
    best = np.clip(base, 0.0, 1000.0).copy()
    n = best.size
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n}, score={best_score:.6f}")
    
    def full_convolve(f):
        """Full linear convolution via FFT."""
        size = 2 * len(f) - 1
        fft_size = 1 << (size - 1).bit_length()  # next power of 2
        F = np.fft.rfft(f, fft_size)
        return np.fft.irfft(F * F, fft_size)[:size]
    
    def compute_score(f):
        conv = full_convolve(f)
        s = np.sum(f)
        if s < 0.01:
            return np.inf, conv, s
        return float(2 * n * np.max(conv) / (s * s)), conv, s
    
    def lp_step(f, peak_lags, M):
        """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_lags, g >= 0."""
        n = len(f)
        # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
        # (f*g)[k] = sum_j f[k-j] * g[j]
        A = np.zeros((len(peak_lags), n))
        for idx, k in enumerate(peak_lags):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A[idx, j] = f[k - j]
        
        c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(len(peak_lags), M),
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            return None
        return res.x
    
    # Main loop
    while time.monotonic() < deadline:
        # Compute current state
        score, conv, s = compute_score(best)
        if score < best_score:
            best_score = score
            print(f"  Improved: score={best_score:.6f}, time={time.monotonic() - (deadline - budget_s + 5.0):.1f}s")
        
        # Find peak lags
        max_val = np.max(conv)
        # Include all lags within 5% of max for robustness
        threshold = max_val * 0.95
        peak_lags = np.where(conv >= threshold)[0]
        
        # LP step
        g = lp_step(best, peak_lags, max_val)
        if g is not None and np.sum(g) > 0:
            g_norm = g / np.sum(g) * np.sum(best)  # match normalization
            # Blend
            t = 0.3
            candidate = (1 - t) * best + t * g_norm
            candidate = np.clip(candidate, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(candidate.tolist()))
            if cand_score < best_score:
                best = candidate
                best_score = cand_score
                print(f"  LP step: score={best_score:.6f}")
        
        # Local perturbation (every few iterations or if LP doesn't help)
        for _ in range(5):
            candidate = best.copy()
            # Block perturbation
            width = rng.integers(max(1, n // 100), max(2, n // 10))
            start = rng.integers(0, n - width)
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.1))
            candidate = np.clip(candidate, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(candidate.tolist()))
            if cand_score < best_score:
                best = candidate
                best_score = cand_score
        
    return [float(x) for x in best]

    size = 2 * n - 1

    F = np.fft.rfft(f, fft_len)
    return np.fft.irfft(F * F, fft_len)[:size]

def _subgrad_max_conv(f, conv, frac=0.9):
    """Compute subgradient of max(conv) w.r.t. f."""
    n = len(f)
    M = np.max(conv)
    active = np.where(conv >= frac * M)[0]
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        f_start = k - (hi - 1)
        f_end = k - lo + 1
        subgrad[lo:hi] += 2.0 * f[max(0, f_start):min(n, f_end)][::-1]
    subgrad /= len(active)
    return subgrad

def _fft_convolve(f):
    n = len(f)
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_len)
    return np.fft.irfft(F * F, fft_len)[:size]

def _compute_subgrad(f, conv, frac=0.95):
    """Subgradient of max(conv) w.r.t. f."""
    n = len(f)
    M = conv.max()
    active = np.where(conv >= frac * M)[0]
    subgrad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        f_start = k - (hi - 1)
        f_end = k - lo + 1
        subgrad[lo:hi] += 2.0 * f[f_start:f_end][::-1]
    subgrad /= len(active)
    return subgrad
