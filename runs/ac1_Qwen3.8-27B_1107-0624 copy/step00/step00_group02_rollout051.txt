```python
def frank_wolfe_step(a, M, n):
    """
    Solve the LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Returns the optimal b (normalized to sum=1).
    """
    # Use scipy or cvxpy
    from scipy.optimize import linprog
    
    # Variables: b[0], ..., b[n-1]
    # Objective: max sum(b) => min -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)[k] <= M for k = 0, ..., 2n-2
    # (a*b)[k] = sum_{j} a[j] * b[k-j] for valid j
    # b[j] appears in (a*b)[k] when j is in [max(0, k-n+1), min(k, n-1)]
    # So (a*b)[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * b[k-j]
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j] * b[min(n-1, k-j)... 
    
    # This is getting complicated to formulate as a dense matrix.
    # Let me use a different formulation.
    
    # Actually, for the LP, each constraint (a*b)[k] <= M involves at most n variables of b.
    # The constraint matrix is a (2n-1) x n matrix where row k has a[j] in position (k-j) for valid j.
    
    # This is a banded matrix, so we can construct it efficiently.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]  # a[k-j] * b[j]
    
    b_ub = np.full(2*n-1, M)
    
    # Non-negativity: b >= 0 (handled by bounds)
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        b = result.x
        s = b.sum()
        if s > 0:
            b = b / s
        return b
    else:
        return np.ones(n) / n

def find_best_t(a, b, n):
    """
    Find t in [0,1] that minimizes max(((1-t)a+tb) * ((1-t)a+tb)).
    """
    # The convolution ((1-t)a+tb) * ((1-t)a+tb) is a quadratic in t:
    # = (1-t)^2 * (a*a) + 2t(1-t) * (a*b) + t^2 * (b*b)
    # = (a*a) + 2t(1-t)(a*b - a*a) + t^2(b*b - 2a*b + a*a)
    # = (a*a) + 2t(a*b - a*a) - 2t^2(a*b - a*a) + t^2(b*b - 2a*b + a*a)
    
    # Let's compute the three convolutions:
    ca = np.convolve(a, a)
    cab = np.convolve(a, b)
    cb = np.convolve(b, b)
    
    # f(t) = (1-t)^2 * ca + 2t(1-t) * cab + t^2 * cb
    # = ca + 2t(cab - ca) + t^2(cb - 2cab + ca)
    
    A_coef = cb - 2*cab + ca  # coefficient of t^2
    B_coef = 2*(cab - ca)     # coefficient of t
    C_coef = ca                # constant (not needed)
    
    # f(t) = C_coef + B_coef * t + A_coef * t^2
    # We want to minimize max_k f_k(t) for t in [0,1]
    
    # This is a minimax of quadratic functions, which is complicated.
    # Simple approach: try a grid of t values.
    
    best_t = 0
    best_max = np.inf
    for t in np.linspace(0, 1, 50):
        f = (1-t)**2 * ca + 2*t*(1-t) * cab + t**2 * cb
        m = f.max()
        if m < best_max:
            best_max = m
            best_t = t
    
    return (1-best_t)*a + best_t*b
```
