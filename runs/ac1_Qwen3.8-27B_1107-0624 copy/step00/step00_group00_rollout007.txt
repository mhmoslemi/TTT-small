def fast_self_convolve(a):
    """Compute self-convolution using FFT for speed."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2*n-1]
