    cols = []

"""
LP-based iterative search for non-negative sequences minimizing 2n*max(conv)/sum^2.
Uses geometric sequence initialization and successive LP solves to flatten autocorrelation.
"""

import numpy as np
import time
import scipy.optimize as opt
import scipy.sparse as sp


def make_geometric(n, r):
    """Create normalized geometric sequence of length n with ratio r."""
    if abs(r - 1.0) < 1e-10:
        return np.ones(n) / n
    idx = np.arange(n, dtype=np.float64)
    p = np.power(r, idx)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return p / s


def compute_obj(p):
    """Compute 2n*max(conv) for normalized sequence (sum=1)."""
    n = len(p)
    c = np.convolve(p, p)
    return 2.0 * n * float(c.max())


def build_A(p):
    """Build sparse cross-correlation constraint matrix A of shape (2n-1, n)."""
    n = len(p)
    rows = np.empty(3 * n, dtype=np.int32)
    cols = np.empty(3 * n, dtype=np.int32)
    vals = np.empty(3 * n, dtype=np.float64)
    cnt = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        if length > 0:
            rows[cnt:cnt+length] = k
            cols[cnt:cnt+length] = np.arange(j_start, j_end + 1)
            vals[cnt:cnt+length] = p[j_start:j_end+1]
            cnt += length
    A = sp.csr_matrix((vals[:cnt], (rows[:cnt], cols[:cnt])), shape=(2 * n - 1, n))
    return A


def solve_lp(p, M, timeout_s=30.0):
    """Solve max sum(b) s.t. A*b <= M*1, b >= 0. Returns (b, sum_b) or (None, 0)."""
    n = len(p)
    A = build_A(p)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = opt.linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs',
                             options={'time_limit': timeout_s, 'presolve': True})
        if result.success:
            return result.x, float(-result.fun)
        elif result.status == 9:  # time limit
            return None, 0.0
        return None, 0.0
    except Exception:
        return None, 0.0

    Returns (best_seq, best_score)."""

    if s < 1e-15:
        return best_seq_initial, best_score_initial
    p = p / s
    n = len(p)

    best_seq = best_seq_initial.copy()
    best_score = best_score_initial
    t = 0.25
    it = 0
    stall_count = 0

    while time.monotonic() < deadline:
        c = np.convolve(p, p)
        M = float(c.max())

        # Adaptive LP timeout based on remaining time and n

            break
        lp_timeout = min(20.0, max(2.0, remaining / 10.0))

        b, sum_b = solve_lp(p, M, timeout_s=lp_timeout)

        if b is None:

            n_new = max(20, n // 3)
            print(f"  LP failed/slow (n={n}). Restarting with n={n_new}.", flush=True)
            r = 0.95
            p = make_geometric(n_new, r)
            n = n_new
            t = 0.25
            stall_count = 0
            continue

        if sum_b < 1.0 + 1e-6:
            # No improvement direction found - perturb or restart
