```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP to find g that maximizes sum(g) subject to (f*g) <= M elementwise and g >= 0."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    # Build A_ub such that A_ub @ g <= M
    # (f*g)[k] = sum_i f[i] * g[k-i] <= M
    # For each k, the row of A_ub has f[i] in position (k-i) for valid i
    
    # The constraint matrix is (2n-1) x n
    # Row k: for i from max(0, k-n+1) to min(n-1, k), A[k, k-i] = f[i]
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            A_ub[k, k - i] = f[i]
    
    b_ub = np.full(2*n-1, M)
    
    # bounds: g >= 0
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```

```python
def solve_lp_sparse(f, M):
    n = len(f)
    c = -np.ones(n)
    
    # Build sparse A_ub
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)
            vals.append(f[i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```

Uses a combination of subgradient descent, LP-based direction finding,

import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def compute_conv(p):
    """Compute full linear convolution p*p using FFT."""
    n = len(p)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    P = fft(p, size=size)
    conv = np.real(ifft(P * P))
    return conv[:2*n-1]

def get_score(p):
    """Compute the score of a probability vector p."""
    n = len(p)
