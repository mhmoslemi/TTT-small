"""
Search for low-score nonnegative step-function heights.

Strategy:
- Work with sequences normalized to sum 1; the score is scale-invariant.
- Use fast FFT convolutions for internal scoring.
- Use Frank-Wolfe-like LP directions:
    max sum(b) s.t. b >= 0 and conv(a, b) <= current_max_conv(a, a).
- Run full LP at moderate size, then active-constraint LP at larger size.
- Polish with softened-max gradient descent and random local moves.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog, Bounds
    from scipy.sparse import csr_matrix
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False


def _next_pow2(m):
    f = 1
    while f < m:
        f <<= 1
    return f


def _fft_conv_same(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    f = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, f)
    return np.fft.irfft(A * A, f)[:2 * n - 1]


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = a.sum()
    if s <= 0.0:
        return np.inf
    c = _fft_conv_same(a)
    mc = np.max(c)
    if not np.isfinite(mc):
        return np.inf
    if mc < 0.0:
        mc = 0.0
    return float(2.0 * a.size * mc / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    # Tiny positive floor keeps LP matrices finite and avoids exact zeros.
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / a.size
    a = a / s
    return a


def _resize(a, n):
    a = np.asarray(a, dtype=np.float64)
    if n <= 0:
        n = 1
    if a.size == n:
        return _normalize(a)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / n
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    y = np.interp(x_new, x_old, a)
    return _normalize(y)


def _initial(n, code, rng):
    n = max(1, int(n))
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    if code == 0:
        a = np.ones(n, dtype=np.float64)
    elif code == 1:
        a = 0.05 + 0.95 * np.power(1.0 - x, 1.5)
    elif code == 2:
        a = 0.05 + 0.95 * np.power(x, 1.5)
    elif code == 3:
        a = 0.05 + 0.95 * np.power(np.sin(np.pi * x), 1.5)
    elif code == 4:
        a = (
            0.70 * np.exp(-0.5 * ((x - 0.22) / 0.16) ** 2) +
            0.30 * np.exp(-0.5 * ((x - 0.78) / 0.18) ** 2) +
            0.02
        )
    elif code == 5:
        a = (
            0.60 * np.exp(-0.5 * ((x - 0.78) / 0.18) ** 2) +
            0.40 * np.exp(-0.5 * ((x - 0.22) / 0.16) ** 2) +
            0.02
        )
    elif code == 6:
        a = 0.05 + 0.95 * np.exp(-2.5 * x)
    elif code == 7:
        a = 0.05 + 0.95 * np.exp(-2.5 * (1.0 - x))
    elif code == 8:
        a = rng.dirichlet(np.ones(n, dtype=np.float64))
    else:
        a = np.ones(n, dtype=np.float64)
    return _normalize(a)


def _select_rows(c, n, K):
    n = int(n)
    L = 2 * n - 1
    if K is None or K <= 0 or K >= L:
        return np.arange(L, dtype=np.int64)
    c = np.asarray(c, dtype=np.float64)
    order = np.argsort(c, kind="stable")[::-1]
    used = np.zeros(L, dtype=bool)
    center = n - 1
    used[center] = True
    rows_list = [center]

    for idx in order:
        idx = int(idx)
        if 0 <= idx < L:
            if not used[idx]:
                used[idx] = True
                rows_list.append(idx)
            comp = L - 1 - idx
            if 0 <= comp < L and not used[comp]:
                used[comp] = True
                rows_list.append(comp)
        if len(rows_list) >= K:
            break
    return np.unique(np.asarray(rows_list, dtype=np.int64))


def _lp_matrix(a, rows):
    if not _HAVE_SCIPY:
        return None
    rows = np.asarray(rows, dtype=np.int64)
    n = a.size
    m = rows.size
    if m == 0 or n == 0:
        return None

    datal = []
    rowl = []
    coll = []

    for k in rows:
        k = int(k)
        i0 = max(0, k - n + 1)
        i1 = min(n - 1, k)
        cnt = i1 - i0 + 1
        if cnt <= 0:
            continue
        datal.append(a[i0:i1 + 1].astype(np.float64, copy=False))
        rowl.append(np.full(cnt, k, dtype=np.int32))
        iarr = np.arange(i0, i1 + 1, dtype=np.int64)
        coll.append((k - iarr).astype(np.int64, copy=False))

    if not datal:
        return None

    data = np.concatenate(datal)
    row_arr = np.concatenate(rowl)
    col_arr = np.concatenate(coll)

    if data.size == 0:
        return None

    col_arr = col_arr.astype(np.int32, copy=False)
    return csr_matrix((data, (row_arr, col_arr)), shape=(m, n))


def _solve_lp(a, M, rows):
    if not _HAVE_SCIPY:
        return None
    if M is None or not np.isfinite(M) or M <= 0.0:
        return None
    A = _lp_matrix(a, rows)
    if A is None or A.nnz == 0:
        return None

    n = a.size
    cobj = np.full(n, -1.0, dtype=np.float64)
    b = (M * (1.0 + 1e-10) + 1e-15) * np.ones(rows.size, dtype=np.float64)

    res = None
    try:
        bounds = Bounds(0.0, np.inf)
        res = linprog(cobj, A_ub=A, b_ub=b, bounds=bounds, method="highs",
                      options={"disp": False})
    except Exception:
        try:
            res = linprog(cobj, A_ub=A, b_ub=b,
                          bounds=[(0.0, None)] * n,
                          method="highs", options={"disp": False})
        except Exception:
            try:
                res = linprog(cobj, A_ub=A, b_ub=b,
                              bounds=[(0.0, None)] * n,
                              options={"disp": False})
            except Exception:
                return None

    if res is None:
        return None
    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64)
    if x.size != n or not np.all(np.isfinite(x)):
        return None
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s <= 1e-15:
        return None
    return x / s


def _line_search_convex_step(a, g):
    a = np.asarray(a, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    ts = (0.02, 0.05, 0.10, 0.20, 0.30, 0.40, 0.50,
          0.60, 0.70, 0.80, 0.90, 0.95, 1.00,
          1.10, 1.25, 1.50, 2.00)
    best_a = None
    best_s = np.inf

    for t in ts:
        f = a + float(t) * (g - a)
        if float(t) > 1.0:
            if f.min() < -1e-12:
                continue
        if f.min() < 0.0:
            f = np.maximum(f, 0.0)
        s = f.sum()
        if s <= 0.0:
            continue
        f = f / s
        sc = _score_fast(f)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_a = f
    return best_a, best_s


def _random_perturb(a, rng, sigma=0.08):
    a = _normalize(a)
    n = a.size
    repeat = max(1, min(5, n // 4096))
    for _ in range(repeat):
        lo = max(1, n // 400)
        hi = max(lo + 1, n // 20)
        w = int(np.clip(rng.integers(lo, hi), 1, n))
        s = int(np.clip(rng.integers(0, max(1, n - w + 1)), 0, max(0, n - w)))
        e = min(n, s + w)
        if e > s:
            f = np.exp(rng.normal(0.0, sigma, size=e - s, dtype=np.float64))
            a[s:e] *= f
    return _normalize(a)


def _run_lp_frank(a0, budget, full=False, K=64, label="", rng=None):
    if rng is None:
        rng = np.random.default_rng(12345)
    budget = max(0.0, float(budget))
    if budget <= 0.0:
        return _normalize(a0), np.inf

    a = _normalize(a0)
    best_a = a.copy()
    best_s = _score_fast(a)
    n = a.size
    L = 2 * n - 1
    K_current = int(K) if (not full and K is not None and K > 0) else 64

    start = time.monotonic()
    it = 0
    stall = 0

    print(f"[{label}] start n={n} score={best_s:.7f} full={full} K={K_current}")

    while time.monotonic() - start < budget:
        c = _fft_conv_same(a)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        if full:
            rows = np.arange(L, dtype=np.int64)
        else:
            rows = _select_rows(c, n, K_current)

        g = _solve_lp(a, M, rows)

        if g is None and full:
            rows = _select_rows(c, n, min(64, L))
            g = _solve_lp(a, M, rows)

        if g is None:
            stall += 3
        else:
            cand, sc = _line_search_convex_step(a, g)
            if cand is not None and np.isfinite(sc) and sc < best_s - 1e-13:
                a = _normalize(cand)
                best_a = a.copy()
                best_s = sc
                it += 1
                stall = 0
                if not full:
                    # Gradually widen active set after progress.
                    if it % 25 == 0:
                        K_current = min(L, max(K_current + 16, K_current * 2))
                if it % 20 == 0:
                    print(f"[{label}] it={it} score={best_s:.7f}")
            else:
                stall += 1
                if stall == 3:
                    K_current = max(16, K_current // 2)
                elif stall == 6:
                    K_current = min(L, K_current * 2)
                elif stall == 10:
                    # Small escape move, keeping best as reference.
                    a = _random_perturb(a, rng, sigma=0.03)
                    sc_now = _score_fast(a)
                    if sc_now > best_s * 1.5 + 0.1:
                        a = best_a.copy()
                if stall > 18:
                    break

    print(f"[{label}] end n={n} it={it} score={best_s:.7f}")
    return best_a, best_s


def _gradient_vector(a, w):
    n = a.size
    f = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, f)
    A = np.fft.rfft(a[::-1], f)
    conv = np.fft.irfft(W * A, f)
    return 2.0 * conv[n - 1:2 * n - 1]


def _gradient_polish(a0, budget, label="grad"):
    budget = max(0.0, float(budget))
    if budget <= 0.0:
        return _normalize(a0), np.inf

    a = _normalize(a0)
    best_a = a.copy()
    best_s = _score_fast(a)
    n = a.size
    beta = max(1.0, 20.0 * n)
    start = time.monotonic()
    it = 0
    stall = 0

    print(f"[{label}] start n={n} score={best_s:.7f}")

    while time.monotonic() - start < budget:
        c = _fft_conv_same(a)
        mc = float(np.max(c))
        if not np.isfinite(mc):
            break

        e = np.exp(beta * (c - mc))
        se = e.sum()
        if se <= 0.0 or not np.isfinite(se):
            beta *= 0.5
            continue
        w = e / se
        g = _gradient_vector(a, w)

        active = a > 1e-12
        if np.count_nonzero(active) > 1:
            g[active] -= np.mean(g[active])

        d = -g
        best_cand = None
        best_sc = best_s
        alphas = (1.0, 0.5, 0.2, 0.1, 0.05, 0.02)

        for alpha in alphas:
            f = a + float(alpha) * d
            if f.min() < 0.0:
                f = np.maximum(f, 0.0)
            s = f.sum()
            if s <= 0.0:
                continue
            f = f / s
            sc = _score_fast(f)
            if np.isfinite(sc) and sc < best_sc - 1e-13:
                best_sc = sc
                best_cand = f

        if best_cand is not None:
            a = _normalize(best_cand)
            best_a = a.copy()
            best_s = best_sc
            it += 1
            stall = 0
            if it % 40 == 0:
                print(f"[{label}] it={it} score={best_s:.7f}")
            beta = min(beta * 1.03, 200.0 * n)
        else:
            stall += 1
            beta *= 0.75
            if stall > 60:
                break

    print(f"[{label}] end n={n} it={it} score={best_s:.7f}")
    return best_a, best_s


def _random_search(a0, budget, rng, label="random"):
    budget = max(0.0, float(budget))
    if budget <= 0.0:
        return _normalize(a0), np.inf

    a = _normalize(a0)
    best_a = a.copy()
    best_s = _score_fast(a)
    start = time.monotonic()
    it = 0
    stall = 0
    sigma = 0.08

    print(f"[{label}] start n={a.size} score={best_s:.7f}")

    while time.monotonic() - start < budget:
        cand = _random_perturb(a, rng, sigma=sigma)
        sc = _score_fast(cand)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            a = cand
            best_a = a.copy()
            best_s = sc
            it += 1
            stall = 0
            if it % 8 == 0:
                print(f"[{label}] it={it} score={best_s:.7f}")
        else:
            stall += 1
            if stall == 12:
                sigma = max(0.02, sigma * 0.5)
            elif stall == 35:
                sigma = min(0.30, sigma * 1.6)
            if stall > 160:
                break

    print(f"[{label}] end n={a.size} it={it} score={best_s:.7f}")
    return best_a, best_s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Return the best nonnegative height sequence found within the time budget."""
    rng = np.random.default_rng(int(seed))
    total_budget = max(0.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 20.0

    best_a = None
    best_s = np.inf

    # Try the supplied previous construction, if present.
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 4 and np.isfinite(p).any() and p.sum() > 1e-8:
            parent = _normalize(p)
            ps = _score_fast(parent)
            if np.isfinite(ps) and ps < best_s:
                best_s = ps
                best_a = parent.copy()
            print(f"[parent] n={parent.size} fast_score={ps:.7f}")
    except Exception:
        pass

    # Phase 1: moderate-size full Frank-Wolfe/LP optimization.
    n1 = 1024
    small_starts = []
    if parent is not None:
        small_starts.append(("parent_small", _resize(parent, n1)))
    small_starts.append(("uniform_small", _initial(n1, 0, rng)))
    small_starts.append(("asym_left_small", _initial(n1, 1, rng)))
    small_starts.append(("asym_right_small", _initial(n1, 2, rng)))

    remaining = deadline - time.monotonic()
    if remaining > 80.0:
        small_budget = max(120.0, min(360.0, remaining * 0.30))
        per = small_budget / max(1, len(small_starts))
        for name, a0 in small_starts:
            if time.monotonic() >= deadline:
                break
            alloc = max(20.0, min(per, deadline - time.monotonic()))
            ra, rs = _run_lp_frank(a0, alloc, full=True, K=0,
                                   label=name, rng=rng)
            if np.isfinite(rs) and rs < best_s:
                best_s = rs
                best_a = ra.copy()
                print(f"[update] {name} best={best_s:.7f}")

    # Phase 2: active LP polish at larger size.
    n2 = 8192
    large_starts = []
    if best_a is not None:
        large_starts.append(("from_small_large", _resize(best_a, n2)))
    if parent is not None:
        large_starts.append(("parent_large", _resize(parent, n2)))
    large_starts.append(("asym_left_large", _initial(n2, 1, rng)))
    large_starts.append(("asym_right_large", _initial(n2, 2, rng)))

    remaining = deadline - time.monotonic()
    if remaining > 80.0 and best_a is not None:
        large_budget = remaining * 0.50
        per = large_budget / max(1, len(large_starts))
        for name, a0 in large_starts:
            if time.monotonic() >= deadline:
                break
            alloc = max(40.0, min(per, (deadline - time.monotonic()) * 0.45))
            ra, rs = _run_lp_frank(a0, alloc, full=False, K=96,
                                   label=name, rng=rng)
            if np.isfinite(rs) and rs < best_s:
                best_s = rs
                best_a = ra.copy()
                print(f"[update] {name} best={best_s:.7f}")

    # Phase 3: wider active-set polish on current best.
    remaining = deadline - time.monotonic()
    if remaining > 50.0 and best_a is not None:
        a0 = _resize(best_a, n2)
        alloc = remaining * 0.28
        ra, rs = _run_lp_frank(a0, alloc, full=False, K=224,
                               label="polish_large", rng=rng)
        if np.isfinite(rs) and rs < best_s:
            best_s = rs
            best_a = ra.copy()
            print(f"[update] polish_large best={best_s:.7f}")

    # Phase 4: smoothed-max gradient polish.
    remaining = deadline - time.monotonic()
    if remaining > 35.0 and best_a is not None:
        alloc = remaining * 0.28
        ra, rs = _gradient_polish(best_a, alloc, label="grad_polish")
        if np.isfinite(rs) and rs < best_s:
            best_s = rs
            best_a = ra.copy()
            print(f"[update] grad_polish best={best_s:.7f}")

    # Phase 5: heuristic random local search.
    remaining = deadline - time.monotonic()
    if remaining > 20.0 and best_a is not None:
        alloc = remaining * 0.55
        ra, rs = _random_search(best_a, alloc, rng, label="random_polish")
        if np.isfinite(rs) and rs < best_s:
            best_s = rs
            best_a = ra.copy()
            print(f"[update] random_polish best={best_s:.7f}")

    if best_a is None:
        best_a = _initial(1024, 0, rng)
        best_s = _score_fast(best_a)

    best_a = _normalize(best_a)
    # Keep the final length modest enough for the evaluator's direct convolution.
    if best_a.size > 9000:
        best_a = _resize(best_a, n2)
        best_s = _score_fast(best_a)

    print(f"[final] n={best_a.size} internal_score={best_s:.7f}")
    return [float(x) for x in best_a]
