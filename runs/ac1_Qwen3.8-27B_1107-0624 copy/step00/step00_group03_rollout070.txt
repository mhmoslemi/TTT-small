```python
"""
Smooth-max gradient descent + LP polishing for minimizing
    F(a) = 2*n*max_k (a*a)[k] / (sum a)^2
subject to a >= 0.

Method summary:
  - The subgradient of max_k c[k] (c = conv(a,a)) is a weighted (softmax)
    average of the per-lag gradients, each of which is a shifted/reversed
    copy of a.  This is computed in O(n log n) via one FFT convolution.
  - Projected (non-negative) backtracking gradient steps, renormalizing the
    sum each step so that F is monotone in max(c).
  - Several asymmetric initializations are optimized; the best is kept.
  - A final LP polish (maximize sum g s.t. g>=0 and ||f*g||_inf <= M) is
    applied to shave the bound further, following the state-of-the-art idea.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

GLOBAL_DEADLINE = 0.0
GLOBAL_N = 0
GLOBAL_S = 0.0


def fast_score(a):
    """Objective value for a non-negative vector, computed directly."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return float("inf")
    c = np.convolve(a, a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def smooth_max_grad(a, beta):
    """Return (grad, c, p) for the smooth max of the self-convolution.

    grad[i] = sum_k p[k] * d(c[k])/da[i] with p = softmax(beta*c), and
    d(c[k])/da[i] = 2*a[k-i] (0 when out of range).  This equals
    2 * convolve(a[::-1], p)[n-1 : 2n-1].
    """
    n = len(a)
    c = np.convolve(a, a)
    cs = c - np.max(c)
    ec = np.exp(beta * cs)
    p = ec / np.sum(ec)
    q = a[::-1]
    conv = np.convolve(q, p)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c, p


def project_normalize(a, s):
    """Clip to non-negative and rescale to the target sum."""
    a = np.maximum(a, 0.0)
    t = float(np.sum(a))
    if t < 1e-12:
        return a
    return a * (s / t)


def gradient_optimize(a, deadline):
    """Run projected backtracking smooth-max gradient steps until deadline.

    Returns the best (score, sequence) seen during the run.
    """
    global GLOBAL_S
    a = np.array(a, dtype=np.float64)
    a = project_normalize(a, GLOBAL_S)
    best_score = fast_score(a)
    best_seq = a.copy()

    beta = 6.0
    base_lr = 1.0 / max(1.0, len(a))
    stall = 0
    loop = 0

    cur_score = best_score
    while time.monotonic() < deadline:
        grad, c, p = smooth_max_grad(a, beta)
        gn = float(np.linalg.norm(grad)) + 1e-12
        if gn < 1e-14:
            break
        t = base_lr * GLOBAL_S / gn
        accepted = False
        last_try = None
        last_try_score = cur_score
        for _ in range(25):
            a_try = project_normalize(a - t * grad, GLOBAL_S)
            s_try = fast_score(a_try)
            if np.isfinite(s_try):
                if s_try < last_try_score:
                    last_try = a_try
                    last_try_score = s_try
                if s_try < cur_score - 1e-10:
                    a = a_try
                    cur_score = s_try
                    accepted = True
                    break
            t *= 0.5

        if accepted and cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()
            stall = 0
            base_lr = min(base_lr * 1.05, 5.0 / max(1.0, len(a)))
        else:
            if last_try is not None:
                a = project_normalize(last_try, GLOBAL_S)
                cur_score = last_try_score
            stall += 1
            base_lr *= 0.85
            beta = min(beta * 1.15, 2000.0)

            if stall > 3000 or base_lr < 1e-6 / max(1.0, len(a)):
                # local escape: small random kick preserving the sum
                kick = np.random.default_rng(loop).standard_normal(len(a)) * 0.05
                a = project_normalize(a + kick, GLOBAL_S)
                cur_score = fast_score(a)
                stall = 0
                base_lr = 1.0 / max(1.0, len(a))
                beta = 6.0
        loop += 1

    return best_score, best_seq


def lp_polish(f, deadline, max_solves=60):
    """Iteratively apply the LP 'gap reducer' to f, keeping finite improvements.

    Returns the best (score, sequence) found, starting from the score of f.
    """
    global GLOBAL_S
    f = np.array(f, dtype=np.float64)
    f = project_normalize(f, GLOBAL_S)
    n = len(f)
    best_score = fast_score(f)
    best_seq = f.copy()
    t_mix = 0.35

    solves = 0
    while time.monotonic() < deadline and solves < max_solves:
        c = np.convolve(f, f)
        M = float(np.max(c))
        rows = []
        cols = []
        data = []
        for l in range(n):
            k0 = l
            k1 = min(2 * n - 2, l + n - 1)
            if k1 < k0:
                continue
            ks = np.arange(k0, k1 + 1)
            js = ks - l
            rows.append(ks)
            cols.append(np.full(ks.shape, l, dtype=np.int64))
            data.append(f[js])
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b = np.full(2 * n - 2 + 1, M)
        cobj = -np.ones(n)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(
            cobj, A_ub=A, b_ub=b, bounds=bounds,
            method="highs",
            options={"time_limit": max(1.0, 120.0 - (time.monotonic() - GLOBAL_DEADLINE) * 0 + 0.0)},
        )
        solves += 1
        if res.status != 0 or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        gs = float(np.sum(g))
        if gs < 1e-9:
            break
        g = g * (GLOBAL_S / gs)
        cand = (1.0 - t_mix) * f + t_mix * g
        cand = project_normalize(cand, GLOBAL_S)
        sc = fast_score(cand)
        if np.isfinite(sc) and sc < best_score - 1e-9:
            best_score = sc
            best_seq = cand.copy()
            f = cand.copy()
            t_mix = min(t_mix * 1.1, 0.7)
        else:
            t_mix *= 0.6
            if t_mix < 0.02:
                t_mix = 0.35
        if time.monotonic() > GLOBAL_DEADLINE - 60.0:
            break

    return best_score, best_seq
```
