import numpy as np
import time

def compute_objective(a):
    """Compute R = 2n * max(c) / (sum(a))^2 for non-negative a with sum(a) > 0."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = np.max(c)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

def peak_deflation(a, n_steps=10000, eta_start=0.1, eta_end=0.001):
    """Run peak deflation (subgradient) method."""
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s  # normalize sum to 1
    
    best_a = a.copy()
    best_max_c = np.max(np.convolve(a, a))
    
    for step in range(n_steps):
        # Compute convolution
        c = np.convolve(a, a)
        max_c = np.max(c)
        
        # Update best
        if max_c < best_max_c:
            best_max_c = max_c
            best_a = a.copy()
        
        # Find peak location(s)
        k_star = np.argmax(c)
        
        # Subgradient: sg[j] = a[k_star - j] if 0 <= k_star - j < n
        n = len(a)
        sg = np.zeros(n)
        # a[k_star - j] for j = 0, ..., n-1
        # k_star - j ranges from k_star to k_star - n + 1
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        # For j in [k_star - hi, k_star - lo]: sg[j] = a[k_star - j]
        j_lo = k_star - hi
        j_hi = k_star - lo
        sg[j_lo:j_hi+1] = a[lo:hi+1][::-1]
        
        # Step size (annealing)
        t = step / max(1, n_steps - 1)
        eta = eta_start * (eta_end / eta_start) ** t
        
        # Update
        a = a - eta * sg
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a = a / s
    
    return best_a

Actually, let me think about this more carefully. The problem is:
min_a max_k c(k, a) subject to a >= 0, sum(a) = 1

where c(k, a) = sum_i a[i]*a[k-i].

This is a minimax problem. We can use the following approach (similar to the "Chebyshev approximation" or "minimax" methods):

Introduce variable t. Minimize t subject to:
c(k, a) <= t for all k = 0, ..., 2n-2
a >= 0
sum(a) = 1

This is a non-convex problem (c(k,a) is quadratic), but we can use a "convex relaxation" or "iterative linearization" approach:

1. Start with some a^0.
2. At iteration t:
   a. Compute c(k, a^t) for all k.
   b. Find t^t = max_k c(k, a^t).
   c. Linearize c(k, a) around a^t: c(k, a) ≈ c(k, a^t) + grad_k(a^t) · (a - a^t)
   d. Solve: min t subject to c(k, a^t) + grad_k(a^t) · (a - a^t) <= t for all k, a >= 0, sum(a) = 1.
   e. This is a linear program (LP)!
   f. Set a^{t+1} = solution.
   g. Repeat.

This is the "convex-concave procedure" or "successive convex approximation" method. Each iteration solves an LP, which can be done efficiently with scipy or cvxpy.

The LP has 2n-1 + 1 variables (a[0], ..., a[n-1], t) and 2n-1 + n + 1 constraints. For n = 6121, that's about 12000 constraints and 6000 variables. This might be slow for an LP solver.

Alternatively, we can use a "cutting plane" method: at each iteration, only add the constraint for the active k (the one with max c). This reduces the LP size.
