from scipy.optimize import linprog
import time

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Asymmetric ramp initialization to break the symmetry barrier.
2. LP-based direction finding (Gilbert-Johnson-Keener style).
3. Subgradient descent refinement.
4. Multiple restarts with different initializations.
"""

def make_ramp(n, alpha=0.1):
    a = np.linspace(alpha, 1.0, n)
    return a / a.sum()

def make_exponential(n, rate=2.0):
    a = np.exp(-rate * np.arange(n) / n)
    return a / a.sum()

def compute_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def solve_lp_direction(a, M):
    """Find g maximizing sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix T where T[k,i] = a[k-i] if valid
    T = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            T[k, i_lo:i_hi+1] = a[k - np.arange(i_lo, i_hi+1)]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bnd = [(0, None)] * n
    
    result = linprog(c, A_ub=T, b_ub=b_ub, bounds=bnd, method='highs')

    return None

        a = (1 - t) * a + t * g
        a = np.clip(a, 0, None)
        a = a / a.sum()
        
        # Decrease t slowly
        t = max(0.1, t * 0.999)
    
    return best_a, float(best_score)

def subgradient_step(a, beta=50):
    """One subgradient step on smoothed max conv."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    exp_conv = np.exp(beta * (conv - max_val))
    weights = exp_conv / exp_conv.sum()
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_k weights[k] * a[k-j] for 0<=k-j<n
        k_lo = j
        k_hi = min(2*n-1, j+n)
        if k_lo < k_hi:
            grad[j] += np.dot(weights[k_lo:k_hi], a[:k_hi-k_lo])
        # Second term: sum_k weights[k] * a[j-k] for 0<=j-k<n
        k_lo2 = max(0, j-n+1)
        k_hi2 = j + 1
        if k_lo2 < k_hi2:
            grad[j] += np.dot(weights[k_lo2:k_hi2], a[j-k_hi2+1:j-k_lo2+1][::-1])
    return grad
