```python
import numpy as np
import cvxpy as cp
import time

def make_convolution_matrix(f):
    """Create the (2n-1) x n matrix A where (A @ g)[k] = (f*g)(k)."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A[k, j] = f[j]
    return A

def solve_lp_for_g(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)(k) <= M for all k."""
    n = len(f)
    g = cp.Variable(n)
    A = make_convolution_matrix(f)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.CLARABEL)
    if g.value is None:
        return None
    return np.maximum(g.value, 0)

def lp_iteration(f, n_iter=100, alpha=0.1):
    """Iteratively improve f using LP-based updates."""
    f = f.astype(float).copy()
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        g = solve_lp_for_g(f, M)
        if g is None:
            break
        # Rescale g to have same sum as f
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g < 1e-10:
            break
        g = g * sum_f / sum_g
        # Blend
        f_new = (1 - alpha) * f + alpha * g
        # Check if it improved
        conv_new = np.convolve(f_new, f_new)
        if conv_new.max() < M or it < 5:
            f = f_new
    return f
```

"""
Optimization of step function coefficients to minimize 2*n*max(f*f)/(sum(f))^2.
Uses LP-based iterative refinement with multiple parametric starting points.
"""

import numpy as np
import cvxpy as cp
import time


def build_conv_matrix(f):
    """Build matrix A of shape (2n-1, n) where A @ g = f * g (convolution)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp(f, M):
    """Solve: max sum(g) s.t. A @ g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def lp_refine(f, max_iters=200, time_budget=300.0):
    """Iteratively refine f using LP-based updates."""
    f = f.astype(np.float64).copy()
    start_time = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start_time > time_budget:
            break
        
        conv = np.convolve(f, f)
        M = conv.max()
        sum_f = f.sum()
        
        g = solve_lp(f, M)
        if g is None or g.sum() < 1e-10:
            break
        
        # Rescale g to match sum of f
        g = g * sum_f / g.sum()
        
        # Compute score of g directly
        conv_g = np.convolve(g, g)
        score_g = 2 * len(g) * conv_g.max() / (g.sum() ** 2)
        
        # Blend
        alpha = 0.5 if it < 20 else 0.3
        f_new = (1 - alpha) * f + alpha * g
        
        # Check improvement
        conv_new = np.convolve(f_new, f_new)
        score_new = 2 * len(f_new) * conv_new.max() / (f_new.sum() ** 2)
        
        if score_new < 2 * len(f) * M / (sum_f ** 2):
            f = f_new
        # Also try g directly
        elif score_g < 2 * len(f) * M / (sum_f ** 2):
            f = g
    
    return f

    return np.ones(n, dtype=np.float64)

    return np.exp(-(x - center) ** 2 / (2 * sigma ** 2))


def init_triangular(n):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.maximum(0, 1 - np.abs(x - center) / (center + 1))


def init_cosine_bell(n):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    t = (x - center) / (center + 0.5)
    return np.maximum(0, np.cos(t * np.pi / 2) ** 2)

    center = (n - 1) / 2.0

"""
Minimize 2*n*max(f*f)/(sum(f))^2 using parametric initializations and LP-based refinement.

import numpy as np
import cvxpy as cp
import time


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        prob.solve(solver=cp.ECOS)
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def lp_refine(f, time_budget=120.0, max_iters=200):
    f = f.astype(np.float64).copy()
    start = time.monotonic()
    best_f = f.copy()
    best_score = 2 * len(f) * np.convolve(f, f).max() / f.sum()**2
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = np.convolve(f, f)
        M = conv.max()
        g = solve_lp(f, M)
        if g is None or g.sum() < 1e-10:
            break
        g = g * f.sum() / g.sum()
        # Try blend and direct
        for alpha in [0.5, 1.0, 0.3]:
            f_try = (1 - alpha) * f + alpha * g
            s = 2 * len(f_try) * np.convolve(f_try, f_try).max() / f_try.sum()**2
            if s < best_score:
                best_score = s
                best_f = f_try.copy()
                f = f_try.copy()
        # Also try g directly
        s_g = 2 * len(g) * np.convolve(g, g).max() / g.sum()**2
        if s_g < best_score:
            best_score = s_g
            best_f = g.copy()
            f = g.copy()
    
    return best_f


def compute_score(f):
    conv = np.convolve(f, f)

    t = np.abs(x - center) / (center + 0.5)
    return np.maximum(0, 1 - t ** exponent)
