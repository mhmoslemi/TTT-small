```python
"""
Search for a non-negative sequence minimizing
    2 n max_k (a*a)[k] / (sum a)^2.

The optimizer works with the normalized probability vector p = a / sum(a),
so the objective is 2 n max_k (p*p)[k].  The p-vector is represented by
logits z_1,...,z_{n-1} with p_0 = 1 / (1 + sum_j exp(z_j)) and
p_j = exp(z_j) / (1 + sum_i exp(z_i)).  A smooth surrogate of the
maximum convolution coefficient is optimized with L-BFGS-B, followed by
active-peak subgradient polishing.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

N_DEFAULT = 4096
X_DEFAULT = (np.arange(N_DEFAULT, dtype=np.float64) + 0.5) / float(N_DEFAULT)

_FFT_SELF = 1 << (2 * N_DEFAULT - 2).bit_length()
_FFT_GRAD = 1 << (3 * N_DEFAULT - 2).bit_length()
_BOUNDS = [(-40.0, 40.0)] * (N_DEFAULT - 1)


_best_score = float("inf")
_best_z = None
_search_deadline = 0.0


class SearchTimeout(Exception):
    pass


def _pow2_at_least(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _sanitized_z(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -40.0, 40.0)
    return x


def _probs_from_z(z):
    z = _sanitized_z(z)
    m = max(0.0, float(np.max(z))) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N_DEFAULT, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _full_conv_padded(a, b):
    fa = np.fft.rfft(a, _FFT_GRAD)
    fb = np.fft.rfft(b, _FFT_GRAD)
    return np.fft.irfft(fa * fb, _FFT_GRAD).real


def _self_conv(p):
    f = np.fft.rfft(p, _FFT_SELF)
    c = np.fft.irfft(f * f, _FFT_SELF).real
    c = c[:2 * N_DEFAULT - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_p_from_weights(p, w):
    # gp_i = 2 * sum_j p_j w_{i+j}
    conv = _full_conv_padded(p[::-1], w)
    h = conv[N_DEFAULT - 1:2 * N_DEFAULT - 1]
    return 2.0 * h


def _score_p(p):
    c = _self_conv(p)
    return 2.0 * float(N_DEFAULT) * float(np.max(c))


def _update_best(x, score):
    global _best_score, _best_z
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_z = np.array(x, copy=True)


def _score_z(z):
    x = _sanitized_z(z)
    p = _probs_from_z(x)
    s = _score_p(p)
    _update_best(x, s)
    return s


def _objective_and_grad(x, alpha):
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout

    x = _sanitized_z(x)
    p = _probs_from_z(x)
    c = _self_conv(p)

    hard = 2.0 * float(N_DEFAULT) * float(np.max(c))
    _update_best(x, hard)

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_p_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz


def _optimize_z(z0, alpha, max_fun, deadline):
    global _search_deadline
    if minimize is None:
        return

    _search_deadline = float(deadline)
    x0 = _sanitized_z(z0)

    max_fun = int(max(80, max_fun))
    max_iter = int(max(50, max_fun // 2))

    try:
        minimize(
            _objective_and_grad,
            x0,
            args=(float(alpha),),
            method="L-BFGS-B",
            jac=True,
            bounds=_BOUNDS,
            options={
                "maxfun": max_fun,
                "maxiter": max_iter,
                "ftol": 1e-13,
                "gtol": 1e-9,
                "maxls": 30,
                "disp": False,
            },
        )
    except SearchTimeout:
        return
    except Exception:
        return


def _density_to_z(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-12)
    logf = np.log(f)
    z = logf[1:] - logf[0]
    return np.clip(z, -40.0, 40.0)


def _logistic_z(beta):
    x = X_DEFAULT
    u = beta * (x - 0.5)
    u = np.clip(u, -50.0, 50.0)
    f = 2.0 / (1.0 + np.exp(-u))
    return _density_to_z(f)


def _linear_z(rho):
    x = X_DEFAULT
    f = 1.0 + rho * (2.0 * x - 1.0)
    f = np.maximum(f, 1e-8)
    return _density_to_z(f)


def _power_z(a, reverse=False):
    x = X_DEFAULT
    if reverse:
        f = np.power(1.0 - x + 1e-4, a)
    else:
        f = np.power(x + 1e-4, a)
    f = np.maximum(f, 1e-12)
    return _density_to_z(f)


def _exp_z(beta):
    x = X_DEFAULT
    f = np.exp(beta * x)
    return _density_to_z(f)


def _poly_random_z(rng):
    x = X_DEFAULT
    t = x - 0.5
    deg = 5
    coef = rng.normal(scale=1.0, size=deg + 1)
    f = np.ones(N_DEFAULT, dtype=np.float64)
    val = np.ones(N_DEFAULT, dtype=np.float64)
    for k in range(deg, 0, -1):
        val = val * t + coef[k]
    f += val
    f = np.maximum(f, 1e-8)
    f /= float(np.sum(f))
    return _density_to_z(f)


def _random_z(rng):
    x = X_DEFAULT[1:]
    t = x - 0.5
    z = np.zeros_like(x)
    z += rng.uniform(-2.5, 2.5) * t
    z += rng.uniform(-1.2, 1.2) * t * t
    for fq in (1, 2, 3, 4, 6):
        amp = rng.normal(0.0, 1.6 / (fq + 0.5))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.sin(2.0 * np.pi * fq * x + ph)
    return np.clip(z, -40.0, 40.0)


def _make_structured_candidates():
    out = []
    for beta in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0, 6.0, 8.0):
        out.append(_logistic_z(beta))
    for rho in (0.3, 0.5, 0.7, 0.85, 0.95):
        out.append(_linear_z(rho))
        out.append(_linear_z(-rho))
    for a in (0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
        out.append(_power_z(a, reverse=False))
        out.append(_power_z(a, reverse=True))
    for beta in (0.5, 1.0, 2.0, 3.0, 4.0, 6.0):
        out.append(_exp_z(beta))
        out.append(_exp_z(-beta))
    return out


def _make_random_candidates(rng, count):
    out = []
    n_poly = max(10, count // 2)
    n_rand = count - n_poly
    for _ in range(n_poly):
        out.append(_poly_random_z(rng))
    for _ in range(n_rand):
        out.append(_random_z(rng))
    return out


def _sort_by_score(cands):
    if not cands:
        return []
    scores = np.empty(len(cands), dtype=np.float64)
    for i, z in enumerate(cands):
        scores[i] = _score_z(z)
    order = np.argsort(scores)
    return [cands[int(j)] for j in order]


def _add_base_candidate(cands):
    try:
        base = height_sequence_1
    except Exception:
        return cands
    try:
        base = np.asarray(base, dtype=np.float64)
        if base.size == 0:
            return cands
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.clip(base, 0.0, None)
        if float(np.sum(base)) <= 0.0:
            return cands
        if base.size != N_DEFAULT:
            old = np.linspace(0.0, 1.0, base.size)
            new = np.linspace(0.0, 1.0, N_DEFAULT)
            base = np.interp(new, old, base)
        base = base / float(np.sum(base))
        cands.append(_density_to_z(base))
    except Exception:
        pass
    return cands


def _choose_alpha(score):
    if not np.isfinite(score):
        return 2.0e5
    if score > 1.90:
        return 2.0e5
    if score > 1.75:
        return 5.0e5
    if score > 1.60:
        return 1.2e6
    if score > 1.52:
        return 3.0e6
    return 8.0e6


def _adaptive_maxfun(rem, run_idx):
    if rem > 900.0:
        base = 850
    elif rem > 450.0:
        base = 600
    elif rem > 180.0:
        base = 380
    else:
        base = 200
    if run_idx < 6:
        base = int(base * 1.25)
    elif run_idx > 70:
        base = int(base * 0.75)
    return int(max(80, base))


def _polish_subgradient(z0, deadline, rng):
    z = _sanitized_z(z0)
    p = _probs_from_z(z)
    score = _score_p(p)
    _update_best(z, score)

    noise = 0.12
    fails = 0

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p)
        cmax = float(np.max(c))

        topk = 8
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros_like(c)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        gp = _grad_p_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-14:
            direction = gz / gnorm
            for step in (0.35, 0.18, 0.09, 0.045, 0.0225):
                zt = z - step * direction
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < score - 1e-12:
                    z = zt
                    p = pt
                    score = st
                    _update_best(z, score)
                    accepted = True
                    break

        if not accepted:
            best_local_score = score
            best_local_z = z
            best_local_p = p
            for _ in range(3):
                zt = z + rng.normal(0.0, noise, N_DEFAULT - 1)
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < best_local_score - 1e-12:
                    best_local_score = st
                    best_local_z = zt
                    best_local_p = pt
            if best_local_score < score - 1e-12:
                z = best_local_z
                p = best_local_p
                score = best_local_score
                _update_best(z, score)
                accepted = True

        if accepted:
            fails = 0
            noise = min(0.25, noise * 1.25)
        else:
            fails += 1
            noise = max(0.015, noise * 0.80)
            if fails > 120:
                break


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_z, _search_deadline

    start_time = time.monotonic()
    budget_s = float(budget_s)
    end_time = start_time + max(1.0, budget_s) - 2.0

    _best_score = float("inf")
    _best_z = None
    _search_deadline = start_time

    rng = np.random.default_rng(seed)

    # Uniform is a valid fallback and gives objective 2.0.
    _score_z(np.zeros(N_DEFAULT - 1, dtype=np.float64))

    struct_cands = _make_structured_candidates()

    if budget_s > 600.0:
        random_count = 160
    elif budget_s > 180.0:
        random_count = 80
    else:
        random_count = 20

    random_cands = _make_random_candidates(rng, random_count)
    struct_cands = _add_base_candidate(struct_cands)

    struct_sorted = _sort_by_score(struct_cands)
    random_sorted = _sort_by_score(random_cands)

    # Reserve time for final polish if the budget is large.
    if budget_s > 400.0:
        main_deadline = start_time + budget_s - 5.0 - max(60.0, 0.06 * budget_s)
    else:
        main_deadline = start_time + max(10.0, budget_s - 10.0)

    max_struct_runs = len(struct_sorted)
    max_rand_runs = min(50, len(random_sorted)) if budget_s > 300.0 else min(10, len(random_sorted))

    run_idx = 0
    last_best = _best_score
    stagnant = 0

    while time.monotonic() < main_deadline:
        rem = main_deadline - time.monotonic()
        if rem < 12.0:
            break

        if run_idx < max_struct_runs:
            z0 = struct_sorted[run_idx]
        elif run_idx < max_struct_runs + max_rand_runs:
            z0 = random_sorted[run_idx - max_struct_runs]
        else:
            if _best_z is None:
                z0 = rng.normal(0.0, 1.0, N_DEFAULT - 1)
            else:
                frac = max(0.0, min(1.0, rem / max(1.0, main_deadline - start_time)))
                noise = 0.05 + 0.35 * frac
                if stagnant > 30:
                    noise *= 2.0
                z0 = _best_z + rng.normal(0.0, min(1.5, noise), N_DEFAULT - 1)

        run_idx += 1
        alpha = _choose_alpha(_best_score)
        max_fun = _adaptive_maxfun(rem, run_idx)
        _optimize_z(z0, alpha, max_fun, main_deadline)

        if _best_score < last_best - 1e-12:
            stale = last_best - _best_score
            last_best = _best_score
            stagnant = 0
            elapsed = time.monotonic() - start_time
            print(f"run {run_idx}: best={_best_score:.6f} delta={stale:.6f} elapsed={elapsed:.1f}s")

    # Final polishing on the global best.
    if _best_z is not None and time.monotonic() < end_time - 2.0:
        polish_deadline = max(time.monotonic() + 5.0, end_time - 3.0)
        _polish_subgradient(_best_z, polish_deadline, rng)

        # One more short L-BFGS pass if time remains.
        if minimize is not None and _best_z is not None:
            rem = end_time - 3.0 - time.monotonic()
            if rem > 25.0:
                z0 = _best_z + rng.normal(0.0, 0.04, N_DEFAULT - 1)
                max_fun = max(120, int(min(600, rem / 0.006)))
                _optimize_z(z0, 8.0e6, max_fun, end_time - 3.0)

    if _best_z is None:
        z = np.zeros(N_DEFAULT - 1, dtype=np.float64)
    else:
        z = _sanitized_z(_best_z)

    p = _probs_from_z(z)
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]

    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        score = float("inf")

    if not np.isfinite(score) or score > 2.0000001:
        seq = [1.0 / float(N_DEFAULT)] * N_DEFAULT
        try:
            score = float(evaluate_sequence(seq))
        except Exception:
            score = 2.0

    # Safety: if the inherited construction is better, keep it.
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < score:
                seq = [float(v) for v in base]
                score = base_score
    except Exception:
        pass

    print(f"final score={score:.6f} length={len(seq)}")
    return seq
```
