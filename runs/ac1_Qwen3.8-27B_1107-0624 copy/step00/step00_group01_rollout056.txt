```python
from scipy.optimize import linprog

def lp_step(a, M):
    """Solve the LP to find a direction b that maximizes sum(b) subject to a*b <= M."""
    n = len(a)
    
    # Build constraint matrix A (2n-1 rows, n columns)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # For efficiency, build row by row
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k-n+1)
        j_max = min(k, n-1)
        if j_min <= j_max:
            A_ub[k, j_min:j_max+1] = a[k-j_max:k-j_min-1:-1]
    
    # Solve: max 1'b s.t. A_ub @ b <= M*ones, b >= 0
    c = -np.ones(n)  # Minimize -sum(b)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```
