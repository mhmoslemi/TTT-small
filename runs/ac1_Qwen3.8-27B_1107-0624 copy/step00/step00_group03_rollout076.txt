Minimizes 2n*max(conv(s,s))/(sum(s))^2 using projected gradient descent
with log-sum-exp smoothing, FFT-based gradient computation,
multi-length/multi-restart search, and perturbation polishing.
"""
import numpy as np
import time


def _proj(v, target):
    """Euclidean projection onto {x >= 0, sum(x) = target} via bisection."""
    lo = np.min(v) - target
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.maximum(v - mid, 0.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - 0.5 * (lo + hi), 0.0)


def _conv(s):
    """Full linear convolution s * s via FFT."""
    n = len(s)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    F = np.fft.rfft(s, sz)
    return np.fft.irfft(F * F, sz)[:2 * n - 1]


def _corr(p, s):
    """corr[k] = sum_j p[k+j]*s[j] for k=0..n-1, via FFT cross-correlation."""
    n = len(s)
    sz = 1
    while sz < 3 * n - 2:
        sz <<= 1
    P = np.fft.rfft(p, sz)
    S = np.fft.rfft(s[::-1], sz)
    c = np.fft.irfft(P * S, sz)
    return c[n - 1:2 * n - 1]


def _grad(s, eps):
    """
    Compute c = conv(s,s), logsumexp(c/eps), and gradient w.r.t. s.
    Returns (c, obj, grad).
    """

    cm = c.max()
    e = np.exp((c - cm) / eps)
    Z = e.sum()
    obj = cm / eps + np.log(Z)
    p = e / Z
    grad = (2.0 / eps) * _corr(p, s)
    return c, obj, grad
