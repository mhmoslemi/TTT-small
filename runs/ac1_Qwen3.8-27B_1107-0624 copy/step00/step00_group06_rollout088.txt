    f=np.asarray(f,float)
    n=f.size

def project_simplex(v):
    v=np.asarray(v,float)
    n=v.size
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1,n+1) > (cssv - 1.0)
    idx = np.nonzero(cond)[0]
    if idx.size == 0:
        return np.full(n, 1.0/n)
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Multi-start, FFT-plus-LP search for nonnegative step-function heights minimizing

    2*n*max(convolve(f, f)) / (sum(f))**2.

The search combines several complementary ideas:
  * normalization using scale invariance,
  * exact fast scoring by FFT,
  * local multiplicative/projected-gradient updates driven by autocorrelation
    peak responsibilities,
  * an active-constraint LP direction similar to the known state of the art,
  * many randomized/parametric starts and short time slices.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except Exception:
    HAS_LINPROG = False

SEARCH_LENGTHS = [16, 32, 64, 128, 256, 512, 1024, 2048, 4096]


def normalize(f):
    """Clip to a finite nonnegative vector and scale to sum 1."""
    f = np.asarray(f, dtype=np.float64)
    if f.ndim != 1:
        f = f.ravel()
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    f = f + 1e-15
    s = float(f.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return f / s


def convolution_max(f):
    """Linear convolution f*f by FFT, clipped nonnegative, and its max."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64), 0.0
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    m = float(np.max(c)) if c.size else 0.0
    return c, m


def score_and_max(f):
    """Internal score using scale invariance."""
    f = normalize(f)
    n = f.size
    if n == 0:
        return np.inf, 0.0
    _, m = convolution_max(f)
    return float(2.0 * n * m), m


def project_simplex(v):
    """Project v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)


def softmax_weights(c, frac):
    """Soft-max weights over the convolution values."""
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c)) if c.size else 0.0
    if c.size == 0:
        return np.array([], dtype=np.float64)
    if m <= 0.0:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    tau = max(1e-18, frac * m)
    z = (c - m) / tau
    z = np.minimum(z, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def top_weights(c, frac):
    """Uniform weights on the top fraction of convolution peaks."""
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    if L == 0:
        return np.array([], dtype=np.float64)
    T = max(1, int(np.ceil(L * frac)))
    if T >= L:
        return np.full(L, 1.0 / L, dtype=np.float64)
    idx = np.argpartition(c, -T)[-T:]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / T
    return w


def responsibility(f, w):
    """
    Compute r_i = sum_k w_k f_{k-i}, i.e. the peak responsibility gradient / 2.
    Uses FFT as circular cross-correlation with no aliasing because w has length 2n-1.
    """
    f = np.asarray(f, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(w, N)
    r = np.fft.irfft(np.conj(F) * W, N)[:n]
    r = np.where(np.isfinite(r), r, 0.0)
    return np.maximum(r, 0.0)


def responsibility_candidates(f, r):
    """
    Several stable local moves that reduce high-responsibility coordinates
    and increase low-responsibility coordinates while staying on the simplex.
    """
    out = []
    f = np.asarray(f, dtype=np.float64)
    r = np.asarray(r, dtype=np.float64)
    if f.size == 0 or r.size != f.size:
        return out

    rm = float(np.mean(r))
    sd = float(np.std(r))
    if sd < 1e-14 or not np.isfinite(sd):
        return out

    d = (rm - r) / sd
    d = np.clip(np.where(np.isfinite(d), d, 0.0), -4.0, 4.0)

    for beta in (0.05, 0.15, 0.4, 1.0):
        out.append(project_simplex(f + beta * d))
        out.append(normalize(f * np.exp(beta * d)))

    rr = r + max(0.1 * rm, 1e-12) + 1e-15
    for beta in (0.1, 0.3, 0.7):
        out.append(normalize(f / np.power(rr, beta)))

    return out


def make_power_shape(n, a, b, rng=None):
    """Asymmetric power shape with optional light smoothing."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-3
    f = np.power(x + eps, a) * np.power(1.0 - x + eps, b)
    if rng is not None:
        f = f * np.exp(rng.normal(0.0, 0.15, size=n))
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1e6)
    return normalize(f)


def make_random_candidate(n, mode, rng):
    """A few cheap randomized initial structures."""
    mode = int(mode) % 4
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if mode == 0:
        f = np.ones(n, dtype=np.float64) + 0.5 * rng.normal(0.0, 1.0, size=n)
    elif mode == 1:
        alpha = float(rng.uniform(0.05, 20.0))
        f = rng.dirichlet(np.ones(n, dtype=np.float64) * alpha)
    elif mode == 2:
        c1 = float(rng.uniform(0.1, 0.9))
        w1 = float(rng.uniform(0.02, 0.35))
        c2 = float(rng.uniform(0.1, 0.9))
        w2 = float(rng.uniform(0.02, 0.35))
        f = np.exp(-((x - c1) / w1) ** 2) + np.exp(-((x - c2) / w2) ** 2) + 1e-9
    else:
        p = float(rng.uniform(0.2, 0.8))
        f = (rng.random(n) < p).astype(np.float64) + 0.1 * rng.random(n)
    return normalize(f)


def blend_line_search(a, b, a_score, ts):
    """Try convex blends (1-t)*a + t*b and return the best."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    best = a.copy()
    bs = a_score
    for t in ts:
        if t <= 0.0:
            continue
        if t >= 1.0:
            cand = b.copy()
        else:
            cand = (1.0 - t) * a + t * b
        cand = normalize(cand)
        sc, _ = score_and_max(cand)
        if sc < bs - 1e-12:
            bs = sc
            best = cand.copy()
    return best, bs


def choose_lp_K(n):
    """Number of active constraints for the approximate LP direction."""
    L = 2 * n - 1
    if L <= 700:
        return L
    if n <= 600:
        return min(450, max(200, L))
    if n <= 1200:
        return 360
    if n <= 2500:
        return 260
    return 180


def lp_direction(f, m, c, K_total):
    """
    Solve a bounded active-conset LP:

        max sum(g)
        s.t. (f/M) * g <= 1 for selected constraints,
             0 <= g <= 1000.

    The selected top constraints plus spaced fallback constraints avoid
    unbounded columns from truncation.
    """
    if not HAS_LINPROG:
        return None
    f = np.asarray(f, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = f.size
    L = c.size
    if n == 0 or L == 0 or m <= 1e-300 or not np.isfinite(m):
        return None

    if K_total >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        K_top = max(1, K_total // 2)
        K_space = max(4, K_total - K_top)
        top = np.argpartition(c, -K_top)[-K_top:]
        space = np.linspace(0, L - 1, K_space, dtype=np.int64)
        idx = np.unique(np.concatenate((top, space)))

    fs = f / m
    A = np.zeros((idx.size, n), dtype=np.float64)

    for row, kk in enumerate(idx):
        k = int(kk)
        i0 = max(0, k - n + 1)
        i1 = min(n, k + 1)
        start = k - i0
        end = k - i1 + 1
        A[row, i0:i1] = fs[end:start + 1][::-1]

    Af = A.dot(f)
    b = np.maximum(1.0 + 1e-8, Af + 1e-8)
    objective = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0)] * n

    try:
        res = linprog(
            objective,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if getattr(res, "success", False) is not True:
        return None

    g = getattr(res, "x", None)
    if g is None:
        return None

    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, 1000.0)
    gs = float(g.sum())
    if gs <= 1e-14:
        return None
    return g / gs


def local_refine(f, rng, deadline, use_lp=True):
    """
    Local improvement loop:
      * occasional LP-direction blends,
      * peak-responsibility multiplicative/projected updates,
      * randomized blends to escape local optima.
    """
    best = normalize(f)
    best_score, _ = score_and_max(best)
    n = best.size
    if n == 0:
        return best, best_score

    no_imp = 0
    lp_cycle = 0
    last_lp = time.monotonic() - 1000.0
    lp_disable = False

    while time.monotonic() < deadline - 1.0:
        c, m = convolution_max(best)
        if c.size == 0 or m <= 0.0:
            break

        cur_score = float(2.0 * n * m)
        if cur_score < best_score - 1e-12:
            best_score = cur_score

        improved = False

        if (
            use_lp
            and HAS_LINPROG
            and not lp_disable
            and (lp_cycle == 0 or lp_cycle % 4 == 0)
            and (time.monotonic() - last_lp > max(0.2, n / 2000.0))
        ):
            t0 = time.monotonic()
            K = choose_lp_K(n)
            g = lp_direction(best, m, c, K)
            dt = time.monotonic() - t0
            last_lp = time.monotonic()
            lp_cycle += 1

            if dt > 120.0:
                lp_disable = True

            if g is not None:
                cand, cs = blend_line_search(
                    best,
                    g,
                    best_score,
                    (1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02)
                )
                if cs < best_score - 1e-12:
                    best = cand
                    best_score = cs
                    no_imp = 0
                    continue

        weight_list = [softmax_weights(c, 0.4)]
        if n > 32:
            weight_list.append(top_weights(c, 0.05))
            if n > 128:
                weight_list.append(top_weights(c, 0.01))
        else:
            weight_list.append(top_weights(c, 0.25))

        for w in weight_list:
            r = responsibility(best, w)
            for cand in responsibility_candidates(best, r):
                sc, _ = score_and_max(cand)
                if sc < best_score - 1e-12:
                    best = cand
                    best_score = sc
                    no_imp = 0
                    improved = True
                    break
            if improved:
                break

        if improved:
            continue

        if no_imp % 5 == 0:
            mode = int(rng.integers(0, 4))
            rand = make_random_candidate(n, mode, rng)
            cand, cs = blend_line_search(
                best,
                rand,
                best_score,
                (0.1, 0.25, 0.5, 0.75, 1.0)
            )
            if cs < best_score - 1e-12:
                best = cand
                best_score = cs
                no_imp = 0
                continue

        no_imp += 1
        if no_imp > 150:
            break

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a nonnegative sequence with low scrolling/convolution peak.
    Returns the best sequence found before the time budget.
    """

    try:
        total = float(budget_s)
    except Exception:
        total = 1800.0
    if not np.isfinite(total) or total <= 0.0:
        total = 1800.0
    total = max(1.0, total)

    rng = np.random.default_rng(seed)

    end_time = start_time + total
    final_deadline = end_time - max(1.0, total * 0.05)
    stage_deadline = start_time + total * 0.60
    if stage_deadline < start_time + 1.0:
        stage_deadline = start_time + 1.0

    start_tasks = []

    parent = globals().get("height_sequence_1", None)
    if parent is not None:
        pf = np.asarray(parent, dtype=np.float64).ravel()
        if pf.size > 0 and pf.size <= 20000:
            pf = normalize(pf)
            if bool(np.all(np.isfinite(pf))):
                start_tasks.append(pf)
                pf2 = normalize(pf * np.exp(rng.normal(0.0, 0.12, size=pf.size)))
                start_tasks.append(pf2)

    lengths = list(SEARCH_LENGTHS)
    if parent is not None:
        try:
            pn = int(np.asarray(parent).shape[0])
        except Exception:
            pn = 0
        if 0 < pn <= 20000:
            if not any(abs(pn - int(l)) <= max(4, int(0.08 * l)) for l in lengths):
                lengths.append(pn)

    for L in sorted(set(int(x) for x in lengths if 2 <= int(x) <= 20000)):
        if len(start_tasks) >= 24:
            break
        f1 = make_random_candidate(L, int(rng.integers(0, 4)), rng)
        start_tasks.append(f1)
        f2 = make_power_shape(
            L,
            float(rng.uniform(-1.5, 2.0)),
            float(rng.uniform(-1.5, 2.0)),
            rng
        )
        start_tasks.append(f2)

    if not start_tasks:
        start_tasks = [make_random_candidate(128, int(rng.integers(0, 4)), rng)]

    max_tasks = 24
    if len(start_tasks) > max_tasks:
        keep = [start_tasks[0]]
        rest = start_tasks[1:]
        keep.extend(rest[:max_tasks - 1])
        start_tasks = keep

    best = None
    best_score = np.inf
    n_tasks = len(start_tasks)
    per_task = max(5.0, min(180.0, (stage_deadline - start_time) / max(1, n_tasks)))

    for i, f in enumerate(start_tasks):
        if time.monotonic() >= stage_deadline - 1.0:
            break
        t0 = time.monotonic()
        task_deadline = min(t0 + per_task, stage_deadline)
        if task_deadline <= t0:
            break
        f2, s2 = local_refine(f, rng, task_deadline, use_lp=True)
        if s2 < best_score:
            best = f2
            best_score = s2
            print(f"[stage1] task {i}, n={f2.size}, score={s2:.9f}, t={t0 - start_time:.2f}")

    if best is None:
        best = normalize(np.ones(128, dtype=np.float64))
        best_score, _ = score_and_max(best)

    print(f"[before-final] n={best.size}, score={best_score:.9f}")

    if time.monotonic() < final_deadline - 1.0:
        f2, s2 = local_refine(best, rng, final_deadline, use_lp=True)
        if s2 < best_score:
            best = f2
            best_score = s2

    rem = final_deadline - time.monotonic()
    if rem > 20.0:
        rand = make_random_candidate(best.size, int(rng.integers(0, 4)), rng)
        cand, cs = blend_line_search(
            best,
            rand,
            best_score,
            (0.05, 0.1, 0.2, 0.35, 0.5)
        )
        if cs < best_score:
            best = cand
            best_score = cs

    final_list = [float(x) for x in normalize(best)]

    try:
        ev = evaluate_sequence(final_list)
        if not np.isfinite(ev):
            fallback = normalize(np.ones(256, dtype=np.float64))
            final_list = [float(x) for x in fallback]
    except Exception:
        pass

    print(f"[final] n={len(final_list)}, internal_score={best_score:.9f}")
    return final_list
