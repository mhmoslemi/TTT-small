Maximize sum(b)
Subject to: b >= 0, (a*b)_k <= M for all k = 0, ..., 2n-2.

The constraint (a*b)_k <= M expands to:
sum_j a[j] * b[k-j] <= M for all k.

This is a linear program with n variables and 2n-1 constraints. For n = 7000, that's 7000 variables and 13999 constraints. This is solvable with modern LP solvers, but might be slow.

The key insight is that the optimal b will have the property that a*b is "flat" at level M (at the binding constraints). This means b is chosen to "fill in" where a's autocorrelation is low.

After solving the LP, we normalize b to have the same sum as a (or the desired normalization), and then blend: a_new = (1-t)*a + t*b.

The blending parameter t should be small (e.g., 0.01 to 0.1) to avoid overshooting.

1. Generate initial candidates from parametric families.
2. For the best candidate, alternate between:
   a. Subgradient descent (fast, O(n log n) per step with FFT)
   b. LP-based update (slow but gives a "global" direction)
3. Use the best result as the final answer.

import time
import math

def fft_convolve(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def compute_score_fast(a):
    """Fast computation of the objective (scale-invariant form)."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a)
    max_b = np.max(c)
    return 2.0 * n * max_b / (S * S)

def subgradient_descent(a, max_iters, lr0, time_limit):
    """Run subgradient descent with the given parameters."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        return a
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_score = compute_score_fast(a)
    start_time = time.monotonic()
    
    for t in range(max_iters):

def fft_self_convolve(a):
    """Compute linear self-convolution of a using FFT."""
    n = len(a)
    if n <= 1:
        return a * a
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:2*n-1]

def fast_score(a):
    """Compute 2n*max_b/S^2 efficiently."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    c = fft_self_convolve(a)
    return 2.0 * n * c.max() / (S * S)

def normalize(a):

    s = a.sum()
    if s < 1e-15:

    return a / s

    """Run subgradient descent. Returns (best_a, best_score)."""
    n = len(a)

    a = normalize(a)
    
    best_a = a.copy()
    best_score = fast_score(a)
    start = time.monotonic()
    
    # Precompute FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for t in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Compute convolution
        fa = np.fft.rfft(a, size)
        c = np.fft.irfft(fa * fa, size)[:2*n-1]
        
        k_star = int(np.argmax(c))
        peak = c[k_star]
        score = 2.0 * n * peak  # since sum(a) = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Subgradient: df/da[i] = 4n*a[k*-i]/S^2 - 4n*peak/S^3 (valid) or -4n*peak/S^3 (invalid)

        # Learning rate

        # Update
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
    
    return best_a, best_score

    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = normalize(a)
    best_score = fast_score(a)
    start = time.monotonic()
    size = 1
    while size < 2 * n - 1:
        size <<= 1
