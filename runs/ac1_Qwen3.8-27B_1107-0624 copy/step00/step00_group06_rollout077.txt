def _project_simplex(v):

    n = len(v)

"""
Coordinate descent optimization to minimize 2n*max(convolve(x,x))/sum(x)^2.
Uses a linear-ramp initialization (which gives score ≈ 4/3 for large n) combined
with golden-section 1D subproblem solvers within coordinate descent.
Multiple sequence lengths and initializations are explored within the time budget.
"""
import numpy as np
import time


def _eval_coord(c_segment, p, unaff_max, S_other, old, quad_idx, t, n):
    """Evaluate the score when coordinate quad_idx within segment is set to t."""
    delta = 2.0 * (t - old) * p
    delta[quad_idx] = t * t - old * old
    new_c = c_segment + delta
    new_max = max(float(np.max(new_c)), unaff_max)
    new_S = S_other + t
    if new_S < 1e-10:
        return 1e30
    return 2.0 * n * new_max / (new_S * new_S)


def _coord_descent(p, deadline, max_sweeps=5000):
    """Run coordinate descent with golden section to minimize the score."""
    n = len(p)
    p = np.asarray(p, dtype=np.float64).copy()
    S = float(np.sum(p))
    if S < 1e-10:
        p = np.ones(n) / n
        S = 1.0
    c = np.convolve(p, p)
    best_score = 2.0 * n * float(np.max(c)) / (S * S)
    best_p = p.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    sweep = 0
    while sweep < max_sweeps and time.monotonic() < deadline:
        sweep += 1
        improved = 0.0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old = p[i]
            S_other = S - old
            if S_other < 1e-10 and old < 1e-10:
                continue
            k_lo = i
            k_hi = i + n - 1
            quad_idx = i  # position of quadratic term within segment
            # Precompute unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = float(np.max(c[:k_lo]))
            if k_hi + 1 <= 2 * n - 2:
                t2 = float(np.max(c[k_hi + 1:]))
                if t2 > unaff_max:
                    unaff_max = t2
            c_segment = c[k_lo:k_hi + 1]
            # Golden section bounds
            a = 0.0
            b = max(S, 4.0 * old + 0.01)
            # Initial evaluations
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            f1 = _eval_coord(c_segment, p, unaff_max, S_other, old, quad_idx, c1, n)
            f2 = _eval_coord(c_segment, p, unaff_max, S_other, old, quad_idx, d1, n)
            for _ in range(18):
                if f1 < f2:
                    b = d1
                    d1 = c1
                    f2 = f1
                    c1 = b - gr * (b - a)
                    f1 = _eval_coord(c_segment, p, unaff_max, S_other, old, quad_idx, c1, n)
                else:
                    a = c1
                    c1 = d1
                    f1 = f2
                    d1 = a + gr * (b - a)
                    f2 = _eval_coord(c_segment, p, unaff_max, S_other, old, quad_idx, d1, n)
            t_opt = (a + b) * 0.5
            s_opt = f1 if f1 < f2 else f2
            if s_opt < best_score - 1e-13:
                delta = 2.0 * (t_opt - old) * p
                delta[quad_idx] = t_opt * t_opt - old * old
                c[k_lo:k_hi + 1] += delta
                p[i] = t_opt
                S = S_other + t_opt
                best_score = s_opt
                best_p = p.copy()
                improved += best_score - s_opt
        if improved < 1e-14:
            break
    return best_p, best_score


def _make_inits(n, rng):
    """Generate diverse initializations for a given length n."""
    inits = []
    t = np.arange(n, dtype=np.float64)
    # Linear ramp (score ~ 4/3 for large n)
    p = t + 1.0
    inits.append(p / np.sum(p))
    # Reverse ramp
    p = n - t
    inits.append(p / np.sum(p))
    # Quadratic bump
    p = 1.0 - (2.0 * t / n - 1.0) ** 2
    p = np.maximum(p, 0.01)
    inits.append(p / np.sum(p))
    # Cosine modulated variants
    for amp in [0.3, 0.6, 0.9]:
        for freq in [1, 2, 3, 5]:
            p = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n)
            p = np.maximum(p, 0.01)
            inits.append(p / np.sum(p))
    # Sinusoidal (full wave)
    for freq in [1, 2, 3]:
        p = np.abs(np.sin(np.pi * freq * t / n)) + 0.05
        inits.append(p / np.sum(p))
    # Random
    for _ in range(4):
        p = np.abs(rng.standard_normal(n)) + 0.01
        inits.append(p / np.sum(p))
    # Power law
    p = (t + 1.0) ** 2.0
    inits.append(p / np.sum(p))
    p = (t + 1.0) ** 0.5
    inits.append(p / np.sum(p))
    # Two-peak
    for center in [0.25, 0.5, 0.75]:
        p = np.exp(-((t - center * n) ** 2) / (2.0 * (0.1 * n) ** 2)) + 0.01
        inits.append(p / np.sum(p))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-start coordinate descent to find sequences minimizing
    2n*max(convolve(x,x))/sum(x)^2 with x >= 0.
    
    Primary initialization: linear ramp (analytically gives score ~4/3).
    Uses golden-section 1D subproblems within each coordinate update.
    Explores multiple lengths and initial shapes within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    best_score = np.inf
    best_seq = None

    # Try multiple lengths
    lengths = [100, 200, 500, 1000, 2000, 4000]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"n={n}: starting", flush=True)
        n_start = time.monotonic()

        inits = _make_inits(n, rng)

        # Also try the provided sequence if it matches
        try:
            if 'height_sequence_1' in globals():
                h = np.asarray(height_sequence_1, dtype=np.float64)
                if len(h) == n:
                    h = np.clip(h, 0, None)
                    sh = np.sum(h)
                    if sh > 0.01:
                        inits.append(h / sh)
                # Also try resizing
                elif len(h) > 0:
                    h_resized = np.interp(
                        np.linspace(0, 1, n),
                        np.linspace(0, 1, len(h)),
                        np.clip(h, 0, None)
                    )
                    sh = np.sum(h_resized)
                    if sh > 0.01:
                        inits.append(h_resized / sh)
        except Exception:
            pass

        for idx, init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            # Allocate time per init: more time for ramp init (index 0)
            if idx == 0:
                sub_budget = 180.0
            elif idx < 4:
                sub_budget = 120.0
            else:
                sub_budget = 60.0
            sub_deadline = min(deadline, time.monotonic() + sub_budget)
            p_opt, score = _coord_descent(init, sub_deadline)
            if score < best_score:
                best_score = score
                best_seq = p_opt.copy()
                elapsed = time.monotonic() - n_start
                print(f"  n={n} init={idx}: {score:.6f} ({elapsed:.0f}s)", flush=True)

        # Noise refinement on best found at this length
        if best_seq is not None and len(best_seq) == n:
            for nl in [0.02, 0.05, 0.1, 0.2]:
                if time.monotonic() > deadline:
                    break
                noisy = best_seq * (1.0 + nl * rng.standard_normal(n))
                noisy = np.maximum(noisy, 0.0)
                sn = np.sum(noisy)
                if sn < 1e-10:
                    continue
                noisy = noisy / sn
                sub_deadline2 = min(deadline, time.monotonic() + 60.0)
                p_opt, score = _coord_descent(noisy, sub_deadline2)
                if score < best_score:
                    best_score = score
                    best_seq = p_opt.copy()
                    print(f"  n={n} noise={nl}: {score:.6f}", flush=True)

        n_elapsed = time.monotonic() - n_start
        print(f"n={n}: done, best so far={best_score:.6f} ({n_elapsed:.0f}s)", flush=True)

    # Final refinement on the overall best
    if best_seq is not None:
        print(f"Final refinement, n={len(best_seq)}...", flush=True)
        sub_deadline3 = min(deadline, time.monotonic() + 300.0)
        p_opt, score = _coord_descent(best_seq, sub_deadline3)
        if score < best_score:
            best_score = score
            best_seq = p_opt.copy()
            print(f"  After refinement: {score:.6f}", flush=True)
        # Try noise on best
        for nl in [0.01, 0.03, 0.08]:
            if time.monotonic() > deadline:
                break
            noisy = best_seq * (1.0 + nl * rng.standard_normal(len(best_seq)))
            noisy = np.maximum(noisy, 0.0)
            sn = np.sum(noisy)
            if sn < 1e-10:
                continue
            noisy = noisy / sn
            sub_deadline4 = min(deadline, time.monotonic() + 120.0)
            p_opt, score = _coord_descent(noisy, sub_deadline4)
            if score < best_score:
                best_score = score
                best_seq = p_opt.copy()
                print(f"  noise={nl}: {score:.6f}", flush=True)

    print(f"FINAL: best_score={best_score:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.arange(100, dtype=np.float64) + 1.0
        best_seq = best_seq / np.sum(best_seq)

    # Ensure valid for evaluator: non-negative, sum >= 0.01, values <= 1000
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    S = np.sum(best_seq)
    if S < 0.1:
        best_seq = best_seq * (1.0 / max(S, 1e-10))
    return [float(x) for x in best_seq]
