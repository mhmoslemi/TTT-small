def frank_wolfe_step(f, M):
    # Find b that maximizes sum(b) subject to conv(f,b) <= M and b >= 0
    n = len(f)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) => minimize -sum(b)
    # Constraints:
    #   For each k = 0, ..., 2n-2: sum_j f[j]*b[k-j] <= M
    #   (where b[k-j] is 0 if k-j is out of range)
    #   b[j] >= 0
    
    # Build constraint matrix A (size (2n-1) x n)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # which is: A[k, j] = f[k-j] if 0 <= k-j < n, else 0 = f[k-j] if j >= k-(n-1) and j <= k and j < n
    
    # This is a banded matrix!
    # For row k: A[k, j] = f[k-j] for j in max(0, k-n+1) to min(k, n-1)
    
    # Use scipy linprog
    from scipy.optimize import linprog
    
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A_ub[k, j] = f[k-j]
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

    a = a / sum(a) * np.sqrt(2*n)  # or any normalization

    t = 0.1  # step size
    
    while time.monotonic() < deadline and score > current_best:

        b = solve_lp(a, M)

            break

        sum_b = sum(b)

            break

            M = new_M
            t = min(1.0, t * 1.05)  # increase step size
        else:
            t = max(0.01, t * 0.7)  # decrease step size

    valid_diag = 2 * diag_indices < len(weights)
