"""
Algorithm: LP-guided local search for minimizing peak self-convolution.
Starts from multiple diverse initializations and iteratively improves using
LP-based direction finding (maximize sum of direction subject to convolution
constraints) combined with mixing. Also uses mirror descent as secondary method.
"""

import numpy as np
from scipy.optimize import linprog
import time


def build_conv_matrix(a):
    """Build the convolution matrix A where A[k,j] represents the coefficient
    of b_j in (a*b)_k."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A


def solve_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def mirror_descent_step(a, hot_threshold=0.85, eta=0.1):
    """Perform one mirror descent step to reduce peak convolution."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    if M < 1e-15:
        return a
    hot = (c >= hot_threshold * M).astype(np.float64)
    pressure = np.convolve(a, hot)[:n]
    p_max = pressure.max()
    if p_max < 1e-15:
        return a
    pressure /= p_max
    a_new = a * np.exp(-eta * pressure)
    a_new = np.clip(a_new, 1e-15, None)
    a_new /= a_new.sum()
    return a_new


def evaluate_safe(a):
    """Evaluate sequence, returning inf if invalid."""
    try:
        return float(evaluate_sequence(a.tolist()))
    except:
        return np.inf


def lp_iterate(a, max_steps=500, deadline=None, verbose=False):
    """Iteratively improve a using LP-based direction finding."""
    n = len(a)
    current = a.copy()
    current_score = evaluate_safe(current)
    t = 0.1
    
    for step in range(max_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = np.convolve(current, current)
        M = c.max()
        
        b = solve_direction(current, M)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-10:
            break
        
        b = b / b_sum
        a_new = (1 - t) * current + t * b
        a_new = np.clip(a_new, 0, None)
        a_new /= a_new.sum()
        
        score = evaluate_safe(a_new)
        if np.isfinite(score) and score < current_score:
            current = a_new
            current_score = score
            t = min(t * 1.02, 0.4)
            if verbose and step % 100 == 0:
                print(f"  LP step {step}: score={current_score:.6f}, t={t:.3f}")
        else:
            t = max(t * 0.85, 0.02)
    
    return current, current_score


def md_iterate(a, max_steps=500, deadline=None):
    """Iteratively improve a using mirror descent."""
    n = len(a)
    current = a.copy()
    current_score = evaluate_safe(current)
    eta = 0.1

            current_score = score
            eta = min(eta * 1.01, 1.0)
        else:
            eta *= 0.9

import numpy as np
from scipy.optimize import linprog

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (convolve(a,b))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            A[k, idx] = a[k - idx]
    return A


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None
