    n = len(a)

    F = np.fft.rfft(a, n=fft_size)

    return b_full[:conv_len]

def _subgradient_and_max(a, b):
    """Compute the subgradient of max(b) w.r.t. a, and return max_b."""
    n = len(a)
    max_b = b.max()

    threshold = max_b - max_b * 1e-10

Actually, I think an even better approach is to use a "soft" objective:
f_soft(a) = 2n * logsumexp(b) / (alpha * S^2)

where alpha controls the "sharpness" of the approximation to max. For large alpha, f_soft ≈ f. For smaller alpha, it's smoother and gives gradient information from all positions, not just the max.

The gradient of logsumexp(b) w.r.t. a_i is:
sum_k (exp(alpha*b_k) / sum_j exp(alpha*b_j)) * (db_k/da_i)

= sum_k softmax_k * a[k-i] (if valid)

This is a weighted average of the "top" positions, with weights given by the softmax. It's smooth and gives a good descent direction.

    """Compute soft objective and its gradient."""
    n = len(a)
    b = _fft_convolve(a)
    S = a.sum()
    
    # Soft max approximation
    b_shifted = alpha * (b - b.max())  # For numerical stability
    exp_b = np.exp(b_shifted)
    weights = exp_b / exp_b.sum()  # Softmax weights
    
    # Gradient: sum_k weights_k * a[k-i] (for valid k-i)
    grad = np.zeros(n)

    significant = weights > 1e-6
    for kp in np.where(significant)[0]:
        lo = max(0, kp - n + 1)
        hi = min(n - 1, kp)
