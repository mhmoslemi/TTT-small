            i_hi = min(n - 1, k_star)
            s = np.zeros(n)
            idx = np.arange(i_lo, i_hi + 1)
            s[idx] = a[k_star - idx]
        else:
            # Use top_k peaks for a more global update
            max_c = float(c.max())
            score = 2.0 * n * max_c / (S * S)
            if score < best_score - improve_threshold:
                best_score = score
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1

            top_indices = np.argsort(c)[-top_k:]
            top_indices = np.sort(top_indices)
            weights = c[top_indices] / c[top_indices].sum()
            s = np.zeros(n)
            for k_idx in range(len(top_indices)):
                k_val = int(top_indices[k_idx])
                i_lo = max(0, k_val - n + 1)
                i_hi = min(n - 1, k_val)
                idx = np.arange(i_lo, i_hi + 1)
                s[idx] += weights[k_idx] * a[k_val - idx]

        max_s = float(s.max())
        if max_s < 1e-30:
            break

        # Adaptive step size: start large, decrease over time
        frac = min(1.0, t / max(1, n_max * 0.8))
        eta = eta_start - (eta_start - eta_end) * frac

        # If stuck, increase eta temporarily
        if no_improve_count > 5000:
            eta *= 1.5
            no_improve_count = 0

        factor = np.exp(-eta * s / max_s)
        a = a * factor
        a = np.clip(a, 1e-12, 1e6)
        new_sum = float(a.sum())
        if new_sum < 1e-10:
            a = best_a.copy()
            break
        a = a * S / new_sum

    return best_a, best_score


def lp_refine_step(a, deadline, n_constraints=300):
    """
    One LP-based refinement step. Solves:
      maximize sum(g) subject to (f*g)[k] <= M for top-K k, g >= 0
    Returns the LP solution g (scaled to match sum of a) or None.
    """
    n = len(a)
    f = a.copy()
    S = float(f.sum())
    if S < 1e-10:
        return None

    c = np.convolve(f, f)
    M = float(np.max(c))

    K = min(n_constraints, len(c))
    top_k = np.argsort(c)[-K:]

    # Build sparse constraint matrix
    A_rows = []
    A_cols = []
    A_vals = []
    for row_idx, k in enumerate(top_k):
        j_lo = max(0, int(k) - n + 1)
        j_hi = min(n - 1, int(k))
        for j in range(j_lo, j_hi + 1):
            A_rows.append(row_idx)
            A_cols.append(j)
            A_vals.append(float(f[int(k) - j]))

    if len(A_rows) == 0:
        return None

    A = csr_matrix((A_vals, (A_rows, A_cols)), shape=(K, n))
    b = np.full(K, M)
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        if time.monotonic() > deadline - 2.0:
            return None
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, deadline - time.monotonic() - 1.0)})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0.0, 1e6)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                g = g * S / g_sum
                return g
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Optimizes step function coefficients using multiplicative subgradient
    descent with adaptive step size, multiple initializations, random
    restarts, and optional LP-based refinement.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Determine n and starting sequence
    try:
        start_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if start_seq.ndim != 1 or start_seq.size == 0:
            raise ValueError
    except (NameError, TypeError, ValueError):
        start_seq = np.ones(6121, dtype=np.float64)
    n = len(start_seq)
    start_seq = np.clip(start_seq, 0.0, 1000.0)

    best = start_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial: n={n}, score={best_score:.6f}", flush=True)

    # Time allocation
    t_subgrad = time.monotonic() + 0.65 * (float(budget_s) - 5.0)
    t_lp = time.monotonic() + 0.88 * (float(budget_s) - 5.0)
    # Remaining 12% for final polish

    # Build list of initializations
    inits = []
    # 1. Small perturbation of provided sequence
    inits.append(np.clip(best * (1.0 + 0.03 * rng.standard_normal(n)), 1e-6, 1000.0))
    # 2. Moderate perturbation
    inits.append(np.clip(best * (1.0 + 0.15 * rng.standard_normal(n)), 1e-6, 1000.0))
    # 3. Uniform with small noise
    inits.append(np.clip(1.0 + 0.05 * rng.standard_normal(n), 1e-6, 1000.0))
    # 4. U-shaped
    x = np.linspace(-1, 1, n)
    inits.append(1.0 + 0.8 * x * x)
    # 5. Exponential
    inits.append(np.clip(rng.exponential(1.0, size=n) + 0.01, 1e-6, 1000.0))
    # 6. Cosine-based chirp-like
    inits.append(np.clip(0.5 + 0.5 * np.cos(2.0 * np.pi * np.arange(n) * 3.0 / n), 0.01, 1000.0))

    # Phase 1: Subgradient optimization
    total_inits = len(inits)
    for ii in range(total_inits):
        if time.monotonic() > t_subgrad:
            break
        # Two restarts per initialization
        for r in range(2):
            if time.monotonic() > t_subgrad:
                break
            if r == 0:
                a_init = inits[ii].copy()
            else:
                a_init = inits[ii] * (1.0 + 0.1 * rng.standard_normal(n))
                a_init = np.clip(a_init, 1e-6, 1000.0)

            # Vary parameters per init
            etas = [(2.0, 0.02), (3.0, 0.05), (1.5, 0.01), (2.5, 0.03), (4.0, 0.1), (2.0, 0.05)]
            eta_s, eta_e = etas[(ii + r) % len(etas)]
            tk = 1 if (ii + r) % 3 != 0 else 3

            a_best_local, s_best_local = subgrad_pass(
                a_init, t_subgrad, eta_start=eta_s, eta_end=eta_e,
                n_max=150000, top_k=tk
            )
            if s_best_local < best_score:
                best = a_best_local.copy()
                best_score = s_best_local
                print(f"Subgrad init={ii} r={r}: score={best_score:.6f}", flush=True)

    print(f"Phase 1 done: score={best_score:.6f}, time_used={time.monotonic() - (total_deadline - 0.65*(float(budget_s)-5.0)):.0f}s", flush=True)

    # Phase 2: LP refinement
    if best_score < 1.99 and time.monotonic() < t_lp:
        for lp_iter in range(100):
            if time.monotonic() > t_lp:
                break

            g = lp_refine_step(best, t_lp, n_constraints=300)
            if g is None:
                break

            # Try several blend ratios
            improved = False
            for t_blend in [0.3, 0.5, 0.7]:
                if time.monotonic() > t_lp:
                    break
                a_blend = (1.0 - t_blend) * best + t_blend * g
                a_blend = np.clip(a_blend, 0.0, 1000.0)
                s_blend = fast_score(a_blend)
                if s_blend < best_score - 1e-8:
                    # Verify with official evaluator
                    s_official = float(evaluate_sequence(a_blend.tolist()))
                    if s_official < best_score:
                        best = a_blend.copy()
                        best_score = s_official
                        improved = True
                        print(f"LP iter {lp_iter} t={t_blend}: score={best_score:.6f}", flush=True)
                        break

            if not improved:
                # Short subgradient cleanup
                a_cl, s_cl = subgrad_pass(best, min(time.monotonic() + 3.0, t_lp),
                                          eta_start=1.0, eta_end=0.05, n_max=5000, top_k=1)
                if s_cl < best_score - 1e-8:
                    s_official = float(evaluate_sequence(a_cl.tolist()))
                    if s_official < best_score:
                        best = a_cl.copy()
                        best_score = s_official
                        print(f"LP cleanup iter {lp_iter}: score={best_score:.6f}", flush=True)
                else:
                    break  # No progress, stop LP phase

    # Phase 3: Final fine-tuning
    print(f"Phase 3 start: score={best_score:.6f}", flush=True)
    a_final, s_final = subgrad_pass(best, total_deadline, eta_start=0.5, eta_end=0.005,
                                    n_max=500000, top_k=1)
    if s_final < best_score - 1e-10:
        s_official = float(evaluate_sequence(a_final.tolist()))
        if s_official < best_score:
            best = a_final.copy()
            best_score = s_official
            print(f"Phase 3 improved: score={best_score:.6f}", flush=True)

    # Final verification
    final_list = [float(max(0.0, min(1000.0, x))) for x in best]
    final_score = float(evaluate_sequence(final_list))
    print(f"FINAL: score={final_score:.6f}, n={len(final_list)}", flush=True)

    return final_list
