import numpy as np
import time

"""
Optimizes a non-negative sequence to minimize 2n*max(convolve(s,s))/(sum(s))^2.
Uses subgradient descent on the max-convolution objective, with L2-smoothing
as a secondary objective to improve convergence. Starts from multiple 
initializations (linear ramp, power laws, etc.) and refines via
projected gradient descent on the simplex.
"""

def _soft_conv_grad(s, n, T=0.1):
    """Compute smooth gradient of soft-max of convolution."""
    conv = np.convolve(s, s)
    # Soft-max weights
    conv_scaled = conv / T
    conv_scaled -= conv_scaled.max()  # for numerical stability
    weights = np.exp(conv_scaled)
    weights /= weights.sum()
    # Gradient: sum_k weights[k] * d(conv[k])/d(s)
    grad = np.zeros(n)
    # Use FFT for efficiency
    # grad[j] = sum_k weights[k] * 2*s[k-j] (for valid k-j)
    # = 2 * cross_correlation(weights, s)[j]
    # cross_correlation(a, b)[j] = sum_k a[k] * b[k-j]
    # = convolution(a, flip(b))[j + len(b) - 1]
    full = np.convolve(weights, s[::-1], mode='full')
    grad = 2.0 * full[n-1:n-1+n]
    return grad

def _subgrad_max(s, n):
    """Compute subgradient of max(convolve(s,s))."""
    conv = np.convolve(s, s)
    max_b = conv.max()
    # Use indices within 1% of max for a "soft" subgradient
    threshold = max_b * 0.99
    max_idx = np.where(conv >= threshold)[0]
    grad = np.zeros(n)
    for k in max_idx:
        L = max(0, k - n + 1)
        U = min(k, n - 1)
        grad[L:U+1] += 2.0 * s[k-U:k-L+1][::-1]
    return grad, max_b

2. Gradient computation: O(m*n) where m is number of max indices → fast
3. evaluate_sequence: O(n^2) for the convolution inside → ~5-10ms

So each iteration takes about 10-20ms, giving us about 50-100 iterations per second, or 90,000-180,000 iterations in 1800 seconds. That should be plenty.
