```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def solve_lp(a, M):
    """Solve the LP to find the best direction g."""
    n = len(a)
    
    # Build constraint matrix A (sparse): A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        for k in range(j, min(2*n-1, j+n)):
            # a[k-j] is the coefficient
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= M (for each k)
    b = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None
```
