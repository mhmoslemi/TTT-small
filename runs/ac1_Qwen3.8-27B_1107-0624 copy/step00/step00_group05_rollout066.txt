```python
"""
Minimize  2*n*max(a*a)/(sum a)^2  (scale-invariant) over non-negative a.

Workhorse: active-set Successive-Convex-Approximation (SCA) step.
  At the current a (fixed) the constraint (a*g)_k <= M is LINEAR in g, so an
  LP over g (only the near-max "active" lags enforced, top-K) gives a convex
  descent direction; blend it into a via a line search on the true objective.
Diversification: a pool of independent candidates (different lengths and random /
shaped initialisations) explored round-robin, plus random restarts on stagnation.
The global best is polished at the end and returned as a plain Python list.

The authoritative `evaluate_sequence` (imported in the harness) is used for all
final validation; a fast local objective is used inside the inner loop.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

FAST_EPS = 0.01          # evaluator rejects sum < 0.01


def fast_obj(a):
    """Fast scale-invariant objective for clean non-neg arrays."""
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = a.size
    h = np.convolve(a, a)
    M = float(h.max())
    S = float(a.sum())
    if S < FAST_EPS:
        return np.inf
    return 2.0 * n * M / (S * S)


def clamp_seq(a, lo=0.0, hi=1000.0):
    return np.clip(np.asarray(a, dtype=np.float64), lo, hi)


def normalize1(a):
    a = clamp_seq(a)
    s = a.sum()
    if s <= FAST_EPS:
        a = np.ones_like(a) / a.size
    else:
        a = a / s
    return a


def simplex_proj(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0} (Euclidean)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if rho.size == 0:
        return np.full(n, s / n)
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w * s


def active_sca_direction(a, n, K):
    """LP over g: max sum(g) s.t. (a*g)_k<=1 on the top-K lags, g>=0.
    Returns a direction g with sum 1, or None on failure."""
    h = np.convolve(a, a)
    if n <= 256:
        idx = np.arange(2 * n - 1)
    else:
        if K >= 2 * n - 1:
            idx = np.arange(2 * n - 1)
        else:
            idx = np.argpartition(h, -K)[-K:]
    rr, cc, dd = [], [], []
    af = a
    for k in idx:
        j0 = k - (n - 1)
        if j0 < 0:
            j0 = 0
        j1 = k
        if j1 > n - 1:
            j1 = n - 1
        js = np.arange(j0, j1 + 1)
        rr.extend([int(k)] * int(js.size))
        cc.extend(js.tolist())
        dd.extend(af[k - js].tolist())
    A = csr_matrix((dd, (rr, cc)), shape=(int(idx.size), n))
    b = np.ones(int(idx.size))
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method="highs",
                      options={"presolve": True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = g.sum()
    if not np.isfinite(gs) or gs <= 1e-12:
        return None
    return g / gs


def sca_step(a, n, K, tset=(0.5, 0.3, 0.2, 0.15, 0.1, 0.06, 0.04, 0.02, 0.012, 0.008)):
    """One SCA iteration with line search on the true objective.
    Returns (new_a, new_obj, improved)."""
    g = active_sca_direction(a, n, K)
    if g is None:
        return a, fast_obj(a), False
    cur = fast_obj(a)
    best_a, best_o = a, cur
    for t in tset:
        cand = (1.0 - t) * a + t * g
        cand = normalize1(cand)
        o = fast_obj(cand)
        if np.isfinite(o) and o < best_o:
            best_o = o
            best_a = cand
    return best_a, best_o, best_o < cur - 1e-12

    except Exception:
        pass
    # 2) random starts for a few lengths
    for n in n_pool:
        f = rng.random(n) + 0.05
        pool.append({"f": normalize1(f), "n": int(n)})
    # 3) shaped (slowly-varying bump) start
    n = max(512, min(2048, n_pool[0]))
    t = np.linspace(0.0, 1.0, n)
    f = 0.5 + 0.5 * np.exp(-0.5 * ((t - 0.3) / 0.22) ** 2)
    f = f + 0.3 * (1.0 + np.sin(2 * np.pi * 7.0 * t)) / 2.0
    pool.append({"f": normalize1(f), "n": int(n)})
    # 4) another shaped, different bump centre / width
    t = np.linspace(0.0, 1.0, n)
    f = 1.0 + 0.6 * np.exp(-0.5 * ((t - 0.65) / 0.30) ** 2) - 0.25 * np.cos(2 * np.pi * 3.0 * t)
    pool.append({"f": normalize1(f), "n": int(n)})
    # de-dup identical shapes is unnecessary; keep all
    return pool


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s) - 8.0)

    n_pool = [4498, 2048, 1024, 512]
    pool = make_inits(n_pool, rng)
    for c in pool:
        c["best"] = c["f"].copy()
        c["best_o"] = fast_obj(c["f"])
        c["stag"] = 0

    best_f = pool[0]["best"].copy()
    best_o = pool[0]["best_o"]

    K = 400          # active-set size
    step = 0
    last_print = t_start

    while time.monotonic() < deadline - 25.0:
        step += 1
        for c in pool:
            if time.monotonic() >= deadline - 25.0:
                break
            f = c["f"]
            n = c["n"]
            new_f, new_o, improved = sca_step(f, n, K)
            if np.isfinite(new_o):
                if new_o < c["best_o"]:
                    c["best"] = new_f.copy()
                    c["best_o"] = new_o
                    c["stag"] = 0
                    if new_o < best_o:
                        best_o = new_o
                        best_f = new_f.copy()
                        if time.monotonic() - last_print > 5.0:
                            last_print = time.monotonic()
                            print("improve global  %.6f  (n=%d, step=%d, t=%.1fs)"
                                  % (best_o, n, step, time.monotonic() - t_start))
                c["f"] = new_f
            if improved:
                c["stag"] = 0
            else:
                c["stag"] += 1
            # stagnation -> restart this candidate (keep its length)
            if c["stag"] > 250:
                f2 = rng.random(n) + 0.05
                # blend with current best-of-this candidate to avoid pure reset
                f2 = 0.5 * f2 + 0.5 * c["best"]
                c["f"] = normalize1(f2)
                c["stag"] = 0

    # ---- final polish on the global best ----
    t_left = deadline - time.monotonic()
    f = best_f.copy()
    n = f.size
    if t_left > 5.0:
        while time.monotonic() < deadline - 2.0:
            new_f, new_o, _ = sca_step(f, n, max(K, 600))
            if np.isfinite(new_o) and new_o < best_o:
                best_o = new_o
                best_f = new_f.copy()
            f = new_f
        print("post-polish global %.6f (n=%d)" % (best_o, n))

    # validate against authoritative evaluator with a few equivalent resales
    out = [float(x) for x in best_f]
    try:
        sc = float(evaluate_sequence(out))
        if np.isfinite(sc):
            best_final_o = sc
        else:
            best_final_o = best_o
    except Exception:
        best_final_o = best_o

    # absolutely safe fallback: all-ones of small length (objective exactly 2)
    if not np.isfinite(best_final_o) or best_final_o >= 2.0 + 1e-6:
        fb = [1.0] * 256
        try:
            if float(evaluate_sequence(fb)) < best_final_o:
                out = fb
        except Exception:
            pass

    print("FINAL returned objective estimate = %.6f, length = %d"
          % (best_final_o, len(out)))
    return out
```
