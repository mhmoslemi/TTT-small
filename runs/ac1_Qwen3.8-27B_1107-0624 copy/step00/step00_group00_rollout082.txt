```python
"""
Multi-scale non-negative coordinate search for flat self-convolution.

The algorithm searches a family of block-constant seeds (two-block, edge,
Fourier, Dirichlet, upsamples), refines the block weights with exact 1-D
coordinate descent for the objective 2*n*max(conv(a,a))/sum(a)^2, expands the
best block candidates to the requested length, and finally applies the same
coordinate descent to the full sequence.  A small linear-program direction
(which is the linearized state-of-the-art update) is also tried on small block
sizes.
"""

import time
import math
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def candidate_key(item):
    return item[0]


def insert_candidate(candidates, score, seq, limit=3):
    if not (isinstance(score, float) or isinstance(score, np.floating)):
        try:
            score = float(score)
        except Exception:
            return
    if not np.isfinite(score):
        return
    arr = np.asarray(seq, dtype=float).copy()
    if arr.ndim != 1 or arr.size == 0:
        return
    candidates.append((float(score), arr))
    candidates.sort(key=candidate_key)
    while len(candidates) > limit:
        candidates.pop()


def safe_normalize(x, max_val=1000.0):
    x = np.asarray(x, dtype=float).copy()
    if x.ndim != 1 or x.size == 0:
        return np.ones(1, dtype=float)
    x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
    x[x < 0.0] = 0.0
    s = float(x.sum())
    if s <= 1e-15:
        x.fill(1.0)
        s = float(x.size)
    x /= s
    x[x > max_val] = max_val
    s2 = float(x.sum())
    if s2 > 0.0:
        x /= s2
    return x


def two_block_weights(m, left_mass=1.0 / 3.0):
    if m <= 0:
        return np.ones(1)
    m = int(m)
    b = np.zeros(m, dtype=float)
    left = max(1, m // 2)
    right = m - left
    if left > 0:
        b[:left] = float(left_mass) / float(left)
    if right > 0:
        b[left:] = (1.0 - float(left_mass)) / float(right)
    return safe_normalize(b)


def three_block_weights(m, masses):
    if m <= 0:
        return np.ones(1)
    m = int(m)
    b = np.zeros(m, dtype=float)
    n1 = max(1, m // 3)
    n2 = max(1, m // 3)
    n3 = max(1, m - n1 - n2)
    if n1 > 0:
        b[:n1] = float(masses[0]) / float(n1)
    if n2 > 0:
        b[n1:n1 + n2] = float(masses[1]) / float(n2)
    if n3 > 0:
        b[n1 + n2:] = float(masses[2]) / float(n3)
    return safe_normalize(b)


def edge_weights(m, p):
    m = int(m)
    if m <= 0:
        return np.ones(1)
    x = (np.arange(m, dtype=float) + 0.5) / float(m)
    eps = 1e-8
    b = np.power(x + eps, float(p)) * np.power(1.0 - x + eps, float(p))
    if not np.all(np.isfinite(b)) or b.sum() <= 0.0:
        b = np.ones(m, dtype=float)
    return safe_normalize(b)


def dirichlet_weights(m, alpha, rng):
    m = int(m)
    if m <= 0:
        return np.ones(1)
    try:
        b = rng.dirichlet(np.full(m, float(alpha)))
    except Exception:
        b = np.ones(m, dtype=float)
    return safe_normalize(b)


def fourier_weights(m, rng):
    m = int(m)
    if m <= 0:
        return np.ones(1)
    idx = np.arange(m, dtype=float)
    b = np.ones(m, dtype=float)
    for k in range(3):
        f = int(rng.integers(1, max(2, m)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.08, 0.85)) / float(k + 1)
        b += amp * np.cos(2.0 * np.pi * f * idx / float(m) + phase)
    mn = float(np.min(b))
    if mn <= 0.0:
        b += -mn + 0.1
    b = np.clip(b, 1e-12, None)
    return safe_normalize(b)


def upsample_block(b, m):
    b = np.asarray(b, dtype=float).copy()
    if b.size == 0:
        return np.ones(max(1, int(m)))
    m = int(m)
    if b.size == m:
        return safe_normalize(b)
    old_x = (np.arange(b.size, dtype=float) + 0.5) / float(b.size)
    new_x = (np.arange(m, dtype=float) + 0.5) / float(m)
    try:
        b2 = np.interp(new_x, old_x, b)
    except Exception:
        b2 = np.resize(b, m)
    return safe_normalize(b2)


def block_score(b):
    b = np.asarray(b, dtype=float).copy()
    s = float(b.sum())
    if s <= 1e-15:
        return np.inf
    b = b / s
    c = np.convolve(b, b)
    if c.size == 0:
        return np.inf
    return float(2.0 * b.size * float(np.max(c)))


def expand_block(b, n_total):
    b = np.asarray(b, dtype=float).copy()
    if b.size == 0:
        return np.ones(max(1, int(n_total)), dtype=float)
    m = int(b.size)
    n_total = int(n_total)
    if m > n_total:
        b = b[:n_total]
        m = n_total
    counts = np.full(m, max(1, n_total // m), dtype=int)
    counts[:m] = n_total // m
    excess = n_total - int(counts.sum())
    if excess > 0:
        counts[:excess] += 1
    elif excess < 0:
        i = m - 1
        while excess < 0 and i >= 0 and counts[i] > 1:
            counts[i] -= 1
            excess += 1
    vals = np.zeros(m, dtype=float)
    for i in range(m):
        if counts[i] > 0:
            vals[i] = float(b[i]) / float(counts[i])
    a = np.repeat(vals, counts)
    return safe_normalize(a)


def eval_one_ratio(z, A, B, U, quad, S_base, old):
    z = float(z)
    denom = float(S_base) - float(old) + z
    if denom <= 1e-15:
        return np.inf
    if A.size > 0:
        lin = float(np.max(A + B * z))
    else:
        lin = -np.inf
    nmax = max(float(U), lin, float(quad) + z * z)
    return float(nmax) / (denom * denom)


def golden_eval_x(lo, hi, A, B, U, quad, S_base, old, iterations=25):
    lo = float(lo)
    hi = float(hi)
    if not (np.isfinite(lo) and np.isfinite(hi)) or hi <= lo:
        return 0.5 * (lo + hi)
    phi = (math.sqrt(5.0) - 1.0) / 2.0
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = eval_one_ratio(x1, A, B, U, quad, S_base, old)
    f2 = eval_one_ratio(x2, A, B, U, quad, S_base, old)
    for _ in range(int(max(0, iterations))):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - phi * (hi - lo)
            f1 = eval_one_ratio(x1, A, B, U, quad, S_base, old)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = eval_one_ratio(x2, A, B, U, quad, S_base, old)
        if hi - lo <= 1e-14 * max(1.0, abs(hi)):
            break
    return 0.5 * (lo + hi)


def compute_coordinate_candidate(x, c, S, j, k_search, use_refine):
    n = int(x.size)
    old = float(x[j])
    S = float(S)
    mean = S / float(n) if n > 0 else 0.0
    if mean <= 0.0 or not np.isfinite(mean):
        mean = 1.0 / float(max(1, n))

    k_lo = j
    k_hi = j + n - 1
    U = 0.0
    if k_lo > 0:
        U = max(U, float(np.max(c[:k_lo])))
    if k_hi < 2 * n - 2:
        U = max(U, float(np.max(c[k_hi + 1:])))

    A = c[k_lo:k_hi + 1].copy()
    A -= 2.0 * old * x
    B = 2.0 * x.copy()
    quad = float(c[2 * j]) - old * old

    if n > 0:
        A[j] = -np.inf
        B[j] = 0.0

    x_max = max(30.0 * max(0.0, old), 30.0 * mean, 1e-12)
    if not np.isfinite(x_max) or x_max <= 0.0:
        x_max = max(30.0 * mean, 1e-12)

    k_search = max(2, int(k_search))
    best_val = old
    best_ratio = eval_one_ratio(old, A, B, U, quad, S, old)
    idx = 0

    for _ in range(3):
        xs = np.linspace(0.0, x_max, k_search)
        if n > 1:
            lin_max = np.max(A[None, :] + B[None, :] * xs[:, None], axis=1)
        else:
            lin_max = np.full(k_search, -np.inf, dtype=float)

        quad_vals = quad + xs * xs
        N = np.maximum(np.maximum(lin_max, quad_vals), U)
        denom = S - old + xs

        ratios = np.full(k_search, np.inf, dtype=float)
        good = denom > 1e-15
        if np.any(good):
            ratios[good] = N[good] / (denom[good] * denom[good])

        idx = int(np.argmin(ratios))
        best_val = float(xs[idx])
        best_ratio = float(ratios[idx])

        if idx < k_search - 1 or xs[idx] < 0.95 * x_max:
            break
        x_max *= 2.0

    if use_refine and k_search >= 2:
        if idx == 0:
            lo = 0.0
            hi = float(xs[1])
        elif idx == k_search - 1:
            lo = float(xs[-2])
            hi = float(x_max)
        else:
            lo = float(xs[idx - 1])
            hi = float(xs[idx + 1])
        if hi > lo and np.isfinite(lo) and np.isfinite(hi):
            rv = golden_eval_x(lo, hi, A, B, U, quad, S, old, iterations=22)
            rf = eval_one_ratio(rv, A, B, U, quad, S, old)
            if np.isfinite(rf) and rf < best_ratio:
                best_val = float(rv)
                best_ratio = float(rf)

    return max(0.0, float(best_val)), float(best_ratio)


def coordinate_pass(x, rng, k_search, use_refine, deadline):
    n = int(x.size)
    x = x.copy()
    S = float(x.sum())
    if S <= 1e-15:
        x.fill(1.0 / float(max(1, n)))
        S = 1.0
        c = np.convolve(x, x)
        return x, float(np.max(c)), False

    c = np.convolve(x, x)
    ratio = float(np.max(c)) / (S * S)
    improved = False
    order = rng.permutation(n)

    for j in order:
        if time.monotonic() >= deadline:
            break
        val, new_ratio = compute_coordinate_candidate(x, c, S, int(j), k_search, use_refine)
        if np.isfinite(new_ratio) and new_ratio < ratio - 1e-13:
            old_val = float(x[j])
            delta = float(val) - old_val
            if abs(delta) > 1e-15:
                k = int(j)
                c[k:k + n] += 2.0 * delta * x
                c[2 * k] += delta * delta
                S += delta
                x[k] = float(val)
            ratio = float(new_ratio)
            improved = True

    S = float(x.sum())
    if S <= 1e-15:
        x.fill(1.0 / float(max(1, n)))
        S = 1.0
    else:
        x /= S
    c = np.convolve(x, x)
    return x, float(np.max(c)), improved


def random_kick(x, rng, scale=0.25):
    x = safe_normalize(x)
    n = int(x.size)
    if n <= 1:
        return x
    y = x.copy()
    for _ in range(4):
        den = int(rng.integers(20, 220))
        width = max(1, n // max(1, den))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        y[start:stop] *= np.exp(rng.normal(0.0, float(scale), size=stop - start))
    if rng.random() < 0.35:
        y *= float(np.exp(rng.normal(0.0, float(scale) * 0.6)))
    return safe_normalize(y)


def local_search_sequence(x, time_limit, rng, k_search=32, use_refine=False,
                          kick_scale=0.20, max_kicks=10, anchor=None):
    x = safe_normalize(x)
    n = int(x.size)
    time_limit = max(0.0, float(time_limit))
    deadline = time.monotonic() + time_limit

    if anchor is not None:
        anchor = np.asarray(anchor, dtype=float).copy()
        if anchor.ndim == 1 and anchor.size == n:
            best = safe_normalize(anchor)
        else:
            best = x.copy()
    else:
        best = x.copy()

    c_best = np.convolve(best, best)
    best_ratio = float(np.max(c_best))

    work = x.copy()
    c_work = np.convolve(work, work)
    work_ratio = float(np.max(c_work))
    if work_ratio > best_ratio + 1e-6 and rng.random() < 0.85:
        work = random_kick(best, rng, kick_scale)

    stagnant = 0
    low_improve = 0
    kick_count = 0

    while time.monotonic() < deadline:
        work_new, ratio_new, improved = coordinate_pass(
            work, rng, k_search, use_refine, deadline
        )
        work = work_new
        if ratio_new < best_ratio - 1e-12:
            best = work.copy()
            best_ratio = float(ratio_new)
            stagnant = 0
            low_improve = 0
        else:
            stagnant += 1
            if not improved:
                low_improve += 1
            else:
                low_improve = 0

        if (stagnant >= 3) or (low_improve >= 5):
            if kick_count >= int(max_kicks):
                break
            work = random_kick(best, rng, kick_scale * (1.0 + 0.12 * float(kick_count)))
            kick_count += 1
            stagnant = 1
            low_improve = 0

        if kick_count >= int(max_kicks) and stagnant >= 6:
            break

    return safe_normalize(best)


def lp_direction(b):
    if linprog is None:
        return None
    b = safe_normalize(b)
    m = int(b.size)
    if m < 2:
        return None
    conv = np.convolve(b, b)
    M = float(np.max(conv))
    if M <= 0.0 or not np.isfinite(M):
        return None

    A = np.zeros((2 * m - 1, m), dtype=float)
    for j in range(m):
        A[j:j + m, j] = b

    c_obj = -np.ones(m, dtype=float)
    bounds = [(0.0, None)] * m
    b_ub = np.full(2 * m - 1, M, dtype=float)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:
        return None

    if res is None or not res.success:
        return None
    d = np.asarray(res.x, dtype=float).copy()
    d = np.clip(d, 0.0, None)
    ds = float(d.sum())
    if ds <= 1.0 + 1e-8:
        return None
    return d / ds


def try_lp_blend(b, max_iters=3, time_limit=10.0):
    b = safe_normalize(b)
    deadline = time.monotonic() + max(0.0, float(time_limit))
    score = block_score(b)
    for _ in range(int(max_iters)):
        if time.monotonic() >= deadline:
            break
        d = lp_direction(b)
        if d is None:
            break
        best_b = b.copy()
        best_score = float(score)
        for t in (0.01, 0.03, 0.08, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0):
            cand = (1.0 - t) * b + t * d
            sc = block_score(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_b = cand.copy()
        if best_score < float(score) - 1e-11:
            b = best_b
            score = best_score
        else:
            break
    return b


def make_start(m, idx, rng, best_block):
    m = int(m)
    idx = int(idx)
    if idx == 0:
        return two_block_weights(m, 1.0 / 3.0)
    if idx == 1:
        return two_block_weights(m, 0.20)
    if idx == 2:
        return two_block_weights(m, 0.50)
    if idx == 3:
        return edge_weights(m, -0.45)
    if idx == 4:
        return edge_weights(m, 0.35)
    if idx == 5:
        return dirichlet_weights(m, 0.35, rng)
    if idx == 6:
        return dirichlet_weights(m, 1.40, rng)
    if idx == 7:
        return fourier_weights(m, rng)
    if best_block is not None:
        return upsample_block(best_block, m)
    return three_block_weights(m, (0.1, 0.8, 0.1))


def refine_block(b, m, time_limit, rng, use_lp=False):
    b = safe_normalize(b)
    time_limit = max(0.0, float(time_limit))
    deadline = time.monotonic() + time_limit

    if use_lp and int(m) <= 80 and time.monotonic() < deadline - 3.0:
        b = try_lp_blend(b, max_iters=2, time_limit=min(8.0, deadline - time.monotonic()))

    remaining = deadline - time.monotonic()
    if remaining > 0.1:
        use_refine = int(m) <= 350
        k_search = 42 if int(m) <= 220 else 26
        b = local_search_sequence(
            b, remaining, rng,
            k_search=k_search,
            use_refine=use_refine,
            kick_scale=0.22,
            max_kicks=10
        )

    if use_lp and int(m) <= 60 and deadline - time.monotonic() > 3.0:
        b = try_lp_blend(b, max_iters=1, time_limit=min(3.0, deadline - time.monotonic()))

    return safe_normalize(b)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a low 2*n*max(conv(a,a))/sum(a)^2 sequence."""
    rng = np.random.default_rng(int(seed))
    budget_s = max(20.0, float(budget_s))
    total_deadline = time.monotonic() + max(1.0, budget_s - 6.0)

    # Determine output length.  Prefer the supplied construction if it is in a sane range.
    n_target = 4806
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        if parent.ndim == 1 and parent.size > 0:
            n_target = int(parent.size)
    except Exception:
        parent = None
    if n_target < 12:
        n_target = 4806
    if n_target > 6000:
        n_target = 4806

    candidates = []

    # Baseline: optimal two-block sequence.
    baseline_full = expand_block(two_block_weights(2, 1.0 / 3.0), n_target)
    baseline_score = float(evaluate_sequence(baseline_full.tolist()))
    insert_candidate(candidates, baseline_score, baseline_full, limit=3)

    # Include parent/height_sequence_1 if valid and not too long.
    if parent is not None and parent.ndim == 1 and parent.size == n_target:
        p = safe_normalize(parent)
        ps = float(evaluate_sequence(p.tolist()))
        insert_candidate(candidates, ps, p, limit=3)

    best_score = candidates[0][0] if candidates else float("inf")
    print(f"n={n_target}, initial best={best_score:.6f}")

    # Quick full-space coordinate polish from the simple two-block baseline.
    if time.monotonic() < total_deadline - 180.0 and n_target <= 6000:
        quick_lim = min(70.0, total_deadline - time.monotonic() - 180.0)
        if quick_lim > 20.0:
            y = local_search_sequence(
                baseline_full, quick_lim, rng,
                k_search=18, use_refine=False,
                kick_scale=0.15, max_kicks=6,
                anchor=baseline_full
            )
            sc = float(evaluate_sequence(y.tolist()))
            insert_candidate(candidates, sc, y, limit=3)
            if len(candidates) and candidates[0][0] < best_score:
                best_score = candidates[0][0]
                print(f"quick full polish best={best_score:.6f}")

    # Block-constant search across scales.
    raw_m = [2, 3, 4, 6, 9, 12, 18, 27, 36, 54, 72, 89, 108, 122,
             178, 270, 402, 534, 801, 1202, 1602, 2403]
    m_list = []
    seen = set()
    for m in raw_m:
        m = int(m)
        if 2 <= m < n_target and m not in seen:
            seen.add(m)
            m_list.append(m)
    if not m_list:
        m_list = [2, 3]

    # Leave room for final full-space polish.
    block_deadline = time.monotonic() + max(
        60.0, min(1250.0, budget_s - 550.0)
    )
    if block_deadline > total_deadline - 120.0:
        block_deadline = total_deadline - 120.0
    if block_deadline < time.monotonic() + 40.0:
        block_deadline = time.monotonic() + 40.0

    best_block = None

    for m in m_list:
        if time.monotonic() >= block_deadline:
            break
        if best_score < 1.495:
            break

        remaining_m = block_deadline - time.monotonic()
        if m <= 12:
            t_m = min(70.0, remaining_m)
            starts = 8
        elif m <= 36:
            t_m = min(55.0, remaining_m)
            starts = 7
        elif m <= 120:
            t_m = min(40.0, remaining_m)
            starts = 6
        elif m <= 300:
            t_m = min(28.0, remaining_m)
            starts = 4
        else:
            t_m = min(18.0, remaining_m)
            starts = 2

        if t_m < 8.0:
            break

        t_per = t_m / float(starts)
        use_lp = (m <= 80)

        for s in range(starts):
            if time.monotonic() >= block_deadline:
                break
            if t_per < 1.0:
                break
            b0 = make_start(m, s, rng, best_block)
            b1 = refine_block(b0, m, t_per, rng, use_lp=use_lp)
            full = expand_block(b1, n_target)
            sc = float(evaluate_sequence(full.tolist()))
            insert_candidate(candidates, sc, full, limit=3)
            if len(candidates) and candidates[0][0] < best_score:
                best_score = candidates[0][0]
                best_block = b1.copy()
                print(f"m={m}, start={s}, score={best_score:.6f}")
            if best_score < 1.495:
                break

    # Final full-space polishing on the best candidates.
    final_deadline = total_deadline - 8.0

    if candidates and time.monotonic() < final_deadline - 30.0:
        # 1) Direct polish of current best.
        limit = min(360.0, final_deadline - time.monotonic() - 30.0)
        if limit > 40.0:
            best_seq = candidates[0][1].copy()
            out = local_search_sequence(
                best_seq, limit, rng,
                k_search=18, use_refine=False,
                kick_scale=0.16, max_kicks=12,
                anchor=best_seq
            )
            sc = float(evaluate_sequence(out.tolist()))
            insert_candidate(candidates, sc, out, limit=3)
            best_score = candidates[0][0]
            print(f"final polish 1 best={best_score:.6f}")

        # 2) Kicked start anchored to current best.
        if time.monotonic() < final_deadline - 30.0:
            limit = min(240.0, final_deadline - time.monotonic() - 30.0)
            if limit > 25.0 and len(candidates):
                base = candidates[0][1].copy()
                kick = random_kick(base, rng, 0.28)
                out = local_search_sequence(
                    kick, limit, rng,
                    k_search=18, use_refine=False,
                    kick_scale=0.22, max_kicks=8,
                    anchor=base
                )
                sc = float(evaluate_sequence(out.tolist()))
                insert_candidate(candidates, sc, out, limit=3)
                best_score = candidates[0][0]
                print(f"final polish 2 best={best_score:.6f}")

        # 3) If a second distinct candidate exists, try it briefly.
        if time.monotonic() < final_deadline - 25.0 and len(candidates) > 1:
            limit = min(160.0, final_deadline - time.monotonic() - 25.0)
            if limit > 20.0:
                seq2 = candidates[1][1].copy()
                out = local_search_sequence(
                    seq2, limit, rng,
                    k_search=16, use_refine=False,
                    kick_scale=0.18, max_kicks=6,
                    anchor=candidates[0][1].copy() if len(candidates) else seq2
                )
                sc = float(evaluate_sequence(out.tolist()))
                insert_candidate(candidates, sc, out, limit=3)
                best_score = candidates[0][0]
                print(f"final polish 3 best={best_score:.6f}")

    if candidates:
        final_seq = safe_normalize(candidates[0][1])
    else:
        final_seq = safe_normalize(np.ones(n_target, dtype=float))

    return [float(x) for x in final_seq]
```
