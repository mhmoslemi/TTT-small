import time


"""
Minimizes 2*n*max(convolve(f,f))/(sum(f))^2 using momentum subgradient
descent with FFT-accelerated softmax-weighted gradient, adaptive temperature,
and exploratory restarts. Starts from height_sequence_1 and iteratively
flattens the autocorrelation peak.
"""


def _compute_score(f):
    """Compute the evaluation score for a sequence."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    b = np.convolve(f, f)
    return 2.0 * n * float(np.max(b)) / (S * S)


def _subgradient_fft(f, T):
    """
    Compute the softmax-weighted subgradient of max(convolve(f,f)) using FFT.
    
    Returns (gradient, max_b) where gradient has the same length as f.
    """
    n = len(f)
    L = 2 * n - 1
    b = np.convolve(f, f)
    max_b = float(np.max(b))

    # Stable softmax: weight each convolution position by exp((b[m]-max_b)/T)
    exponents = (b - max_b) / T
    w = np.exp(exponents)
    w_sum = float(np.sum(w))
    if w_sum < 1e-300:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= w_sum

    # FFT-based computation of:
    #   second_sum[j] = convolve(f, w)[j]       for j = 0..n-1
    #   first_sum[j]  = convolve(f, w_rev)[L-1-j] for j = 0..n-1
    w_rev = w[::-1]

    # Linear convolution length: n + L - 1 = 3n - 2
    conv_len = n + L - 1
    N_fft = 1
    while N_fft < conv_len:
        N_fft *= 2

    F = np.fft.rfft(f, N_fft)
    W = np.fft.rfft(w, N_fft)
    WR = np.fft.rfft(w_rev, N_fft)

    conv_fw = np.fft.irfft(F * W, N_fft)
    conv_fwr = np.fft.irfft(F * WR, N_fft)

    # L-1 = 2n-2; j ranges 0..n-1, so indices are 2n-2, 2n-3, ..., n-1
    # That's conv_fwr[n-1 : 2*n-1] reversed
    first_sum = conv_fwr[n - 1:2 * n - 1][::-1]

    g = first_sum + second_sum
    return g, max_b

    """
    Create a binary-inspired non-negative sequence to use as a restart point.
    Uses a simple LFSR-like construction for low autocorrelation.
    """
    # Use a maximum-length LFSR pattern
    # Find a primitive polynomial degree
    d = 12  # 2^12 - 1 = 4095
    # Simple LFSR with a known primitive polynomial (x^12 + x^6 + x^4 + x + 1)
    # Taps: 12, 6, 4, 1, 0 (0-indexed)
    lfsr = 1  # initial state (non-zero)
    bits = []
    bits_len = (1 << d) - 1  # 4095
    for _ in range(bits_len):
        bits.append(lfsr & 1)
        feedback = ((lfsr >> 11) ^ (lfsr >> 5) ^ (lfsr >> 3) ^ (lfsr >> 0) ^ (lfsr >> 0)) & 1
        # Use standard Fibonacci LFSR
        feedback = lfsr & 1
        lfsr >>= 1
        if feedback:
            lfsr ^= 0x1001  # polynomial for a max-length sequence of degree 12
            # Actually let me use a simpler approach: just use a known good tap set
            lfsr ^= 0x0F01  # x^12 + x^6 + x^4 + x + 1 -> taps at 12,6,4,1,0

    # Simpler: just use a random binary sequence with smoothing
    x = rng.integers(0, 2, size=n).astype(np.float64)
    # Smooth slightly
    if n > 10:
        kernel = np.ones(3) / 3.0
        x = np.convolve(x, kernel, mode='same')
    # Ensure non-negative and positive sum
    x = np.maximum(x, 0.01)
    x = x / np.sum(x)
    return x


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(convolve(f,f))/(sum(f))^2.
    
    Uses momentum subgradient descent with FFT-accelerated gradient,
    adaptive softmax temperature, and exploratory restarts.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Initialize from given sequence
    f_init = np.asarray(height_sequence_1, dtype=np.float64)
    f_init = np.clip(f_init, 0.0, 1000.0)
    n = len(f_init)
    S_orig = float(np.sum(f_init))
    if S_orig < 0.01:
        f_init = np.ones(1024, dtype=np.float64)
        n = f_init.size
        S_orig = 1.0
    f = f_init / S_orig  # normalize to sum=1

    best_f = f.copy()
    best_score = _compute_score(best_f)
    print(f"Initial: n={n}, score={best_score:.6f}", flush=True)

    # Parameters
    lr = 0.008
    momentum = 0.9
    T = 0.01  # initial softmax temperature (relative to max_b)
    min_T = 1e-8
    v = np.zeros(n)
    no_improve = 0
    patience = 80
    max_patience = 400
    restart_count = 0

    start_time = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Compute gradient
        g, max_b = _subgradient_fft(best_f, T * max_b if max_b > 1e-15 else T)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-18:
            break
        g = g / g_norm

        # Momentum update
        v = momentum * v - lr * g
        f_new = best_f + v
        f_new = np.maximum(f_new, 0.0)
        S_new = float(np.sum(f_new))
        if S_new < 0.01:
            v *= 0.5
            continue
        f_new = f_new / S_new

        s_new = _compute_score(f_new)

        if s_new < best_score:
            best_f = f_new
            best_score = s_new
            no_improve = 0
            lr = min(lr * 1.01, 0.04)
            T = max(T * 0.998, min_T)
            patience = max(80, patience - 1)
        else:
            no_improve += 1
            lr *= 0.85
            v *= 0.5

            if no_improve >= patience:
                no_improve = 0
                restart_count += 1

                if restart_count % 7 == 0:
                    # Binary-inspired restart
                    f_try = _try_binary_restart(n, rng)
                    # Blend with current best
                    t_blend = 0.3
                    f_try = (1.0 - t_blend) * best_f + t_blend * f_try
                    f_try = np.maximum(f_try, 0.0)
                    S_try = float(np.sum(f_try))
                    if S_try > 0.01:
                        f_try /= S_try
                        s_try = _compute_score(f_try)
                        if s_try < best_score:
                            best_f = f_try
                            best_score = s_try
                            print(f"  Iter {iteration}: binary restart -> {best_score:.6f}", flush=True)
                    # Reset temperature to explore more
                    T = min(max(T * 5.0, 0.01), 0.1)
                else:
                    # Random multiplicative perturbation
                    sigma = 0.003 + 0.004 * min(restart_count, 20)
                    noise = rng.normal(0.0, sigma, size=n)
                    f_try = best_f * np.exp(noise)
                    f_try = np.maximum(f_try, 0.0)
                    S_try = float(np.sum(f_try))
                    if S_try > 0.01:
                        f_try /= S_try
                        s_try = _compute_score(f_try)
                        if s_try < best_score:
                            best_f = f_try
                            best_score = s_try
                            print(f"  Iter {iteration}: perturbation -> {best_score:.6f}", flush=True)
                    T = min(T * 2.0, 0.1)

                patience = min(patience + 30, max_patience)

        # Periodic logging
        if iteration % 500 == 0:
            elapsed = time.monotonic() - start_time
            remaining = deadline - time.monotonic()
            print(f"Iter {iteration:5d} | score={best_score:.6f} | lr={lr:.5f} | "
                  f"T={T:.2e} | max_b={max_b:.4e} | {elapsed:.0f}s elapsed | "
                  f"{remaining:.0f}s left", flush=True)

    # Return best sequence in original scale
    result = best_f * S_orig
    result = np.clip(result, 0.0, 1000.0)

    # Final verification
    final_score = _compute_score(result)
    total_time = time.monotonic() - start_time
    print(f"Final: score={final_score:.6f}, n={len(result)}, time={total_time:.0f}s", flush=True)

    return [float(x) for x in result]
