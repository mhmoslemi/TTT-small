"""
Step function optimization using LP-based iterative method.
Minimizes 2n * max(convolve(a,a)) / (sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_convolution_matrix(p, n):
    """Build sparse matrix A where (A @ g)[k] = (p*g)[k]."""
    total_cols = n
    total_rows = 2 * n - 1
    nnnz = n * n  # approximate
    rows = np.empty(nnnz, dtype=np.int32)
    cols = np.empty(nnnz, dtype=np.int32)
    vals = np.empty(nnnz, dtype=np.float64)
    
    idx = 0
    for k in range(total_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows[idx:idx+length] = k
        cols[idx:idx+length] = np.arange(j_start, j_end + 1)
        vals[idx:idx+length] = p[k - j_start:k - j_start + length]
        idx += length
    
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(total_rows, total_cols))


def lp_direction(p, M, n):
    """Solve LP to find direction g that maximizes sum(g) s.t. p*g <= M, g >= 0."""
    A = build_convolution_matrix(p, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'presolve': True, 'time_limit': 60})
    
    if res.success and res.x is not None:
        return res.x
    return None


def gradient_smooth_max(p, n, beta=100.0):
    """Compute gradient of smooth max approximation of 2n * max(p*p)."""
    c = np.convolve(p, p)
    max_c = c.max()
    
    # Softmax approximation
    scaled = beta * c / max_c
    scaled -= scaled.max()  # for numerical stability
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # d(max_c) / d(p[j]) ≈ sum_k weights[k] * d(c[k])/d(p[j])
    # c[k] = sum_i p[i]*p[k-i]
    # d(c[k])/d(p[j]) = p[k-j] (if valid) + p[j-k] (if valid, but since p*p is symmetric, = p[|k-j|] if valid)
    # Actually d(c[k])/d(p[j]) = p[k-j] if 0 <= k-j <= n-1, else 0
    # (since c[k] = sum_i p[i]*p[k-i], the derivative w.r.t. p[j] is p[k-j] if k-j is valid)
    
    # Gradient of the smooth max:
    # grad[j] = 2n * sum_k weights[k] * p[k-j] (for valid k-j)
    
    # This is: 2n * (weights_reversed * p) ... let me think
    # sum_k weights[k] * p[k-j] = (weights * p_reversed)[j]? 
    # Let's just compute it directly:
    # grad[j] = 2n * sum_k weights[k] * p[k-j] where 0 <= k-j <= n-1
    # = 2n * sum_{i=0}^{n-1} weights[j+i] * p[i]
    # = 2n * (weights_etended * p)[j] where weights_etended is weights with appropriate shifted windows
    
    # Easier: grad = 2n * convolve(weights, p_reversed)[:n]
    # Wait: sum_k weights[k] * p[k-j] where k-j in [0,n-1]
    # Let i = k-j: sum_{i=0}^{n-1} weights[j+i] * p[i]
    # This is the cross-correlation of weights and p at lag j.
    # = np.correlate(weights, p, mode='full')[(2n-1)-1-j]... this is getting confusing.
    
    # Let me just use the convolution matrix approach:
    # grad = 2n * A^T @ weights where A is the convolution matrix
    # But A is (2n-1) x n, and weights is (2n-1), so A^T @ weights is n-vector.
    # (A^T @ weights)[j] = sum_k A[k,j] * weights[k] = sum_k p[k-j] * weights[k] (for valid k-j)
    
    # This is what we want!
    A = build_convolution_matrix(p, n)
    grad = 2 * n * (A.T @ weights)
    
    return grad, max_c

```python
def build_convolution_matrix_fast(p, n):
    """Build sparse convolution matrix more efficiently."""
    total_rows = 2 * n - 1
    
    # Pre-allocate
    nnnz = n * n  # exact count
    rows = np.empty(nnnz, dtype=np.int32)
    cols = np.empty(nnnz, dtype=np.int32)
    vals = np.empty(nnnz, dtype=np.float64)
    
    idx = 0
    for k in range(total_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        rows[idx:idx+length] = k
        cols[idx:idx+length] = j_start + np.arange(length)
        vals[idx:idx+length] = p[k - j_start:k - j_start + length]
        idx += length
    
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(total_rows, n))
```

The inner operations (arange, slicing) are O(length) in numpy, so the total is O(n^2) but with low constant. For n=4806, this should take about 1-3 seconds.
