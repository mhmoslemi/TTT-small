def optimize_autocorrelation(a, n_iter=10000, lr=0.01):
    a = a.astype(float).copy()
    a = a / a.sum()  # normalize
    
    for it in range(n_iter):
        c = np.convolve(a, a, mode='full')
        k_star = np.argmax(c)
        
        # Compute gradient
        grad = np.zeros_like(a)
        n = len(a)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]

            best_seq = _optimize(best_seq, t, seed + 999)
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"  After refinement: score={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(base_n) / base_n
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"  Fallback uniform: score={best_score:.6f}")

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
