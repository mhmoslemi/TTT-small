    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            A[k, i] = a[i]
    return A

"""
LP-based iterative optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses linear programming to find directions that flatten the auto-convolution,
with multiple initializations and adaptive step sizes.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng):
    """LP-based optimization loop. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.03)
        else:
            stall += 1
            step = max(0.005, step * 0.75)
            if stall % 20 == 0 and stall > 20:
                # Random perturbation to escape local minimum
                noise = np.abs(rng.normal(0, 0.08, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                        stall = 0
                    else:
                        # Accept with small probability (annealing)
                        if rng.random() < 0.15:
                            a = a_new
                        else:
                            a = best_a
                else:

            if stall > 150:
                break

        if it % 500 == 0:

def optimize_gradient(a, deadline, rng):
    """Subgradient/multiplicative update for larger sequences."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.05
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        threshold = 0.85 * M
        peak_mask = c > threshold

        # Compute contribution of each a[i] to peak convolutions
        # a[i] contributes a[i]*a[k-i] to c[k]
        # contribution[i] = sum_{k: peak} a[k-i] (for valid k-i)
        contrib = np.zeros(n)
        peak_indices = np.where(peak_mask)[0]
        for i in range(n):
            js = peak_indices - i
            valid = (js >= 0) & (js < n)
            if valid.any():
                contrib[i] = a[js[valid]].sum()

        # Multiplicative update: reduce high-contribution positions
        avg_c = contrib.mean() + 1e-15
        update = np.exp(-lr * contrib / avg_c)
        a_new = a * update
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            break
        a_new /= a_new.sum()
        sc = compute_objective(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
        else:
            stall += 1
            lr = max(0.001, lr * 0.85)
            if stall > 40:
                noise = np.abs(rng.normal(0, 0.05, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                    stall = 0
                else:
                    break
        if stall > 200:
            break
        if it % 500 == 0:
            print(f"      grad iter {it}, score={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based and gradient-based
    iterative optimization with multiple initializations and lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    # Define search configurations: (n, init_func, method)
    configs = [
        (256, make_cosine, 'lp'),
        (256, make_gaussian, 'lp'),
        (256, make_parabolic, 'lp'),
        (512, make_cosine, 'lp'),
        (512, make_gaussian, 'lp'),
        (512, make_parabolic, 'lp'),
        (1024, make_cosine, 'lp'),
        (1024, make_gaussian, 'lp'),
        (1024, make_parabolic, 'lp'),
        (2048, make_cosine, 'lp'),
        (2048, make_gaussian, 'lp'),
    ]

    # Add parent-based configs at the front
    if parent is not None:
        for n_p in [min(parent.size, 512), min(parent.size, 1024)]:
            def make_parent(n, p=parent):
                return p[:n].copy()
            configs.insert(0, (n_p, make_parent, 'lp'))

    # Add random initializations
    def make_random(n):
        return np.abs(np.random.default_rng(int(time.monotonic()*1000)).normal(0, 1, n))
    configs.append((512, make_random, 'lp'))
    configs.append((1024, make_random, 'lp'))

    total_configs = len(configs)
    # Allocate time: more time for middle-size configs (likely sweet spot)
    # Weight by how promising each config is
    weights = []
    for n, init_f, method in configs:
        if n == 512 or n == 1024:
            weights.append(1.5)
        elif n == 256:
            weights.append(1.2)
        elif n == 2048:
            weights.append(1.0)
        else:
            weights.append(1.0)

    total_weight = sum(weights)
    available_time = budget_s - 30.0  # buffer for final return
    time_allocs = []
    cum = 0.0
    for w in weights:
        cum += available_time * w / total_weight
        time_allocs.append(cum)

    print(f"Starting search with {total_configs} configs, budget={budget_s}s")

    for idx, (n, init_func, method) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        config_deadline = start_time + time_allocs[idx]
        # Also respect global deadline
        config_deadline = min(config_deadline, deadline)

        if config_deadline <= time.monotonic():
            continue

        try:
            a = init_func(n)
        except Exception:
            a = np.ones(n)

        a = np.clip(a, 1e-15, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)

        elapsed = time.monotonic() - start_time
        remaining = deadline - time.monotonic()
        print(f"[{idx+1}/{total_configs}] n={n}, method={method}, remaining~{remaining:.0f}s")

        if method == 'lp' and n <= 2048:
            a_opt, score = optimize_lp(a, config_deadline, rng)
        else:
            a_opt, score = optimize_gradient(a, config_deadline, rng)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n})")
        else:
            print(f"  => score={score:.6f} (best so far: {best_score:.6f})")

    # Final polish on the best sequence if time permits
    if best_seq is not None and time.monotonic() < deadline - 60:
        print("Final LP polish on best sequence...")
        polish_deadline = deadline
        best_seq, best_score = optimize_lp(best_seq, polish_deadline, rng)
        print(f"  After polish: score={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    # Ensure sum >= 0.01 for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    elapsed_total = time.monotonic() - start_time
    print(f"DONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
