import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def build_convolution_matrix(a):
    """Build the sparse convolution matrix for a of length n.
    The matrix A is (2n-1) x n where A[k,j] = a(k-j) if 0<=k-j<n else 0.
    So (A @ g)[k] = (a * g)(k) = sum_j a(k-j) g(j).
    """
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, i.e., k-n+1 <= j <= k
    # Row k has non-zeros at columns j = max(0, k-n+1) to min(k, n-1)
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
