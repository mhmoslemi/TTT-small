from scipy.fft import fft, ifft

def convolve_fft(a, b):
    """Full convolution using FFT."""
    n = len(a) + len(b) - 1
    fa = fft(a, n)
    fb = fft(b, n)
    return np.real(ifft(fa * fb))

def compute_objective_fast(a):
    """Compute 2*n*max(b)/(sum_a)^2 using FFT convolution."""
    n = len(a)
