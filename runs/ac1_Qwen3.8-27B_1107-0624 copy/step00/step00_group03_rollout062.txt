def compute_conv_and_grad_smooth(a, t):
    """Compute convolution and smooth-max gradient."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    # Softmax weights (temperature t)
    cv_max = conv.max()
    exponents = t * (conv - cv_max)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.full_like(conv, 1.0 / len(conv))
    else:
        w /= w_sum

    prod = np.convolve(a, w_rev, mode='full')

    grad = 2.0 * prod[idx]
    return grad, conv
