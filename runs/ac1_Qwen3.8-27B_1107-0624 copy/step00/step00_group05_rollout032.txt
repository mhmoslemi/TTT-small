```python
def compute_gradient_at_k(a, k):
    n = len(a)
    grad = np.zeros(n)
    # For i where 0 <= k-i < n, i.e., k-n+1 <= i <= k
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        j_indices = k - indices
        # For most, grad = a[j] + a[i], but where k == 2*i (i.e., i = k/2), grad = 2*a[i]
        grad[indices] = a[j_indices] + a[indices]
        # Fix the diagonal case
        if k % 2 == 0:
            mid = k // 2
            if i_min <= mid <= i_max:
                grad[mid] = 2 * a[mid]
    return grad
```

```python
    t = 0
```

```python
"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses subgradient descent on the simplex with momentum, multiple restarts,
and noise injection to escape local minima.
"""

import numpy as np
import time


def compute_max_conv_and_grad(a, k_star):
    """Compute max convolution value and gradient at k_star."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c[k_star]
    
    # Gradient at k_star
    grad = np.zeros(n)
    i_min = max(0, k_star - n + 1)
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        j_indices = k_star - indices
        grad[indices] = a[j_indices] + a[indices]
        if k_star % 2 == 0:
            mid = k_star // 2
            if i_min <= mid <= i_max:
                grad[mid] = 2.0 * a[mid]
    
    return c_max, grad, c


def optimize_sequence(a, rng, deadline, verbose=False):
    """Optimize sequence a using subgradient descent on the simplex."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = np.inf
    
    lr = 0.4
    momentum = np.zeros(n)
    beta = 0.9
    noise_scale = 0.05
    t = 0
    
    while time.monotonic() < deadline and t < 500000:
        t += 1
        
        # Compute convolution and find argmax
        c = np.convolve(a, a)
        # Only consider first n lags (by symmetry, c[k] = c[2n-2-k])
        k_star = np.argmax(c[:n])
        c_max = c[k_star]
        obj_val = 2.0 * n * c_max
        
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a.copy()
            
        if verbose and t % 10000 == 0:
            print(f"  iter {t}: obj={obj_val:.6f}, best={best_obj:.6f}")
        
        # Compute gradient at k_star
        grad = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            j_indices = k_star - indices
            grad[indices] = a[j_indices] + a[indices]
            if k_star % 2 == 0:
                mid = k_star // 2
                if i_min <= mid <= i_max:
                    grad[mid] = 2.0 * a[mid]
        
        # Project gradient onto simplex tangent space
        grad_mean = grad.mean()
        grad_proj = grad - grad_mean
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad_proj
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm > 0:
            momentum = momentum / momentum_norm * min(momentum_norm, 1.0)
        
        # Update
        a = a - lr * momentum
        
        # Add noise (annealed)
        noise_scale_curr = noise_scale / (1.0 + t * 0.00005)
        if noise_scale_curr > 0.001:
            a = a + noise_scale_curr * rng.standard_normal(n)
        
        # Project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Decay learning rate
        lr = 0.4 / (1.0 + t * 0.00002)
    
    return best_a
```
