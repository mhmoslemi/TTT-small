def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)

    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def smooth_J_and_grad(a, tau):

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S ** 3)
    
    return J, grad

"""
Gradient-based optimization of step function coefficients to minimize
J(a) = 2n * max(a*a) / (sum(a))^2 using FFT-accelerated gradients with
log-sum-exp smoothing, Adam optimizer, adaptive tau, and multiple
initialization strategies explored in parallel.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective value and gradient using log-sum-exp.
    
    J = 2n * Tmax / S^2 where Tmax = tau*log(sum_k exp(c[k]/tau))
    Gradient: dJ/da[i] = 4n*cross[i]/S^2 - 4n*Tmax/S^3
    where cross[i] = sum_k p[k]*a[k-i], p[k] = softmax(c[k]/tau)
    """
    n = len(a)
    # Use a single FFT size for both convolution and cross-correlation
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    # Log-sum-exp
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation: cross[i] = sum_k p[k] * a[k-i]
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian bump."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_cosine(n, freq=1.0):
    """Cosine modulation."""
    x = np.arange(n)
    return 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)


def make_random_positive(n, rng, sigma=0.5):
    """Log-normal random positive sequence."""
    return np.exp(rng.normal(0.0, sigma, n))


def make_taper(n):
    """Triangular taper peaking at center."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 1.0 - 0.5 * np.abs(x - mid) / max(mid, 1.0)


def make_edge_boost(n):
    """High at edges, low in middle."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    denom = max(mid, 1.0)
    return 0.5 + 0.5 * np.abs(x - mid) / denom


def make_quadratic(n):
    """Quadratic bump."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 1.0 + 0.3 * (1.0 - (x - mid) / mid) ** 2


def run_gradient_descent(a_init, budget_s, seed, tau_start=5.0, tau_end=0.05, lr_start=0.01):
    """Run Adam gradient descent with adaptive tau. Returns (best_array, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_J = compute_J_fft(a)

    start_time = time.monotonic()
    last_verify = 0.0
    stall_count = 0
    stall_threshold_J = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.5 * t / 1000.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            tau = max(1.0, tau)
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        # Check internal objective periodically
        if t % 10000 == 0:
            local_J = compute_J_fft(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_threshold_J = local_J
            else:
                stall_count += 1
                if stall_count > 5:
                    # Try a perturbation
                    cand = a.copy()
                    w = int(n // 50)
                    s = int(rng.integers(0, max(1, n - w)))
                    cand[s:s+w] *= np.exp(rng.normal(0, 0.2, w))
                    cand = np.maximum(cand, 0)
                    cJ = compute_J_fft(cand)
                    if cJ < stall_threshold_J:
                        a = cand
                        m = np.zeros(n)
                        v = np.zeros(n)
                        stall_count = 0
                    else:
                        a = best_a.copy()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall_count = 0

        # Verify with evaluate_sequence every ~8 seconds
        now = time.monotonic()
        if now - last_verify > 8.0:
            last_verify = now
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    gd t={t} J={val:.6f} tau={tau:.4f}")

    # Final check
    final_local = compute_J_fft(best_a)
    return best_a, min(best_J, final_local)


def run_hill_climbing(a_init, budget_s, seed):
    """Local search with adaptive step sizes. Returns (best_array, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J_fft(a)
    step_scale = 0.05

    while time.monotonic() < deadline:
        # Adaptive perturbation width
        width = int(rng.integers(max(1, n // 200), max(2, n // 15)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)

        cand = a.copy()
        noise = rng.normal(0.0, step_scale, end - start)
        cand[start:end] *= np.exp(noise)

        # Sometimes also perturb a few random individual elements
        n_single = int(rng.integers(1, 5))
        for _ in range(n_single):
            idx = int(rng.integers(0, n))
            cand[idx] *= np.exp(rng.normal(0.0, step_scale * 0.5))

        cand = np.maximum(cand, 0.0)
        cS = np.sum(cand)
        if cS < 0.01:
            continue

        cJ = compute_J_fft(cand)

        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.95)
        else:
            step_scale = min(0.3, step_scale * 1.02)

        # Verify periodically
        if int((time.monotonic() % 10.0) * 100) == 0:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val

    return best_a, best_J
