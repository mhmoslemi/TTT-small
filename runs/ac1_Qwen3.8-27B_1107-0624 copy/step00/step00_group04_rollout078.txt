```python
"""
Heuristic optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2 over
non-negative step-function heights.

Strategy:
- Work with many candidate lengths, starting from structured anti-palindromic
  initializations (Thue-Morse, low-pass Fourier, cosine/ramp) plus the provided
  height_sequence_1 when available.
- Maintain sequences normalized to sum 1 so the objective is 2*n*max_conv.
- Iteratively reduce the largest convolution peaks by computing a weighted
  subgradient from the top k convolution lags and applying multiplicative
  updates, with line search, adaptive step size, and random/mutation restarts.
- Track the best exact sequence over the time budget and return it.
"""

import numpy as np
import time

_THUE_CACHE = {}


def next_pow2(n):
    n = max(1, int(n))
    return 1 << (n - 1).bit_length()


def thue_morse_length(n):
    n = int(max(1, n))
    key = 1
    while key < n:
        key <<= 1
    arr = _THUE_CACHE.get(key)
    if arr is None:
        cur = np.array([1.0], dtype=np.float64)
        while cur.size < key:
            cur = np.concatenate((cur, -cur))
        arr = cur
        _THUE_CACHE[key] = arr
    return arr[:n].copy()


def normalize_nonnegative(f):
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.ones(1, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if not np.isfinite(s) or s <= 1e-12:
        return np.ones(f.size, dtype=np.float64) / f.size
    return f / s


def fft_self_convolution(f):
    n = f.size
    size = 2 * n - 1
    N = next_pow2(size)
    F = np.fft.rfft(f, N)
    out = np.fft.irfft(F * F, N)[:size]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def internal_score_general(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.inf
    s = f.sum()
    if not np.isfinite(s) or s <= 1e-8:
        return np.inf
    c = fft_self_convolution(f)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * max(0.0, m) / (s * s))


def make_fourier_first(m, rng):
    m = int(max(1, m))
    out = np.zeros(m, dtype=np.float64)
    if m <= 2:
        return out
    maxf = max(2, m // 4)
    count = min(10, max(2, maxf // 3))
    x = np.arange(m, dtype=np.float64)
    for _ in range(count):
        freq = int(rng.integers(1, maxf + 1))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        coeff = float(rng.normal(0.0, 1.0) / np.sqrt(count))
        out += coeff * np.cos(2.0 * np.pi * freq * x / m + phase)
    out -= float(out.mean())
    mx = float(np.max(np.abs(out)))
    if not np.isfinite(mx) or mx <= 0.0:
        return np.zeros(m, dtype=np.float64)
    return out / mx


def make_anti_palindromic(n, first, amp):
    n = int(max(2, n))
    if n % 2 == 0:
        m = n // 2
        first = np.asarray(first, dtype=np.float64).ravel()
        if first.size != m:
            if first.size == 0:
                first = np.zeros(m, dtype=np.float64)
            else:
                old = np.arange(first.size, dtype=np.float64)
                new = np.linspace(0.0, first.size - 1.0, m)
                first = np.interp(new, old, first)
    else:
        m = n // 2
        first = np.asarray(first, dtype=np.float64).ravel()
        if first.size != m:
            if first.size == 0:
                first = np.zeros(m, dtype=np.float64)
            else:
                old = np.arange(first.size, dtype=np.float64)
                new = np.linspace(0.0, first.size - 1.0, m)
                first = np.interp(new, old, first)

    first = np.where(np.isfinite(first), first, 0.0)
    first = first - float(first.mean())
    mx = float(np.max(np.abs(first)))
    if not np.isfinite(mx) or mx <= 0.0:
        u = np.zeros(n, dtype=np.float64)
    else:
        first = first / mx
        u = np.empty(n, dtype=np.float64)
        if n % 2 == 0:
            u[:m] = first
            u[m:] = -first[::-1]
        else:
            u[:m] = first
            u[m] = 0.0
            u[m + 1:] = -first[:n - m - 1][::-1]

    f = 1.0 + float(amp) * u
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    mx_f = float(np.max(f)) if f.size else 0.0
    if mx_f > 0.0:
        f = np.maximum(f, 1e-12 * mx_f)
    return normalize_nonnegative(f)


def make_initial(n, rng, kind, amp):
    n = int(max(2, n))
    kind = str(kind)
    if n % 2 == 1:
        n += 1
    m = n // 2

    if kind == "uniform":
        return np.ones(n, dtype=np.float64) / n

    if kind == "thue":
        first = thue_morse_length(m)
        if first.size:
            shift = int(rng.integers(0, m))
            first = np.roll(first, shift)
            if float(rng.random()) < 0.5:
                first = -first
        return make_anti_palindromic(n, first, amp)

    if kind == "fourier":
        first = make_fourier_first(m, rng)
        return make_anti_palindromic(n, first, amp)

    if kind == "cosine":
        first = np.cos(np.pi * (2.0 * np.arange(m) + 1.0) / float(n))
        return make_anti_palindromic(n, first, amp)

    if kind == "ramp":
        first = np.linspace(1.0, -1.0, m)
        return make_anti_palindromic(n, first, amp)

    if kind == "random":
        first = rng.normal(0.0, 1.0, m)
        return make_anti_palindromic(n, first, amp)

    first = thue_morse_length(m)
    return make_anti_palindromic(n, first, amp)


def top_subgradient(f, c, k_lags, power):
    n = f.size
    m = c.size
    if m == 0 or n == 0:
        return np.zeros(n, dtype=np.float64)
    c = np.where(np.isfinite(c), c, 0.0)
    K = int(max(1, min(k_lags, m)))
    if K >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(c, m - K)[-K:]
    vals = np.maximum(c[idx], 0.0)
    vals = vals ** float(power)
    total = float(vals.sum())
    if not np.isfinite(total) or total <= 0.0:
        w = np.ones(K, dtype=np.float64) / K
    else:
        w = vals / total

    grad = np.zeros(n, dtype=np.float64)
    for lag, weight in zip(idx, w):
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            grad[lo:hi + 1] += weight * f[lag - hi:lag - lo + 1][::-1]
    return grad


def multiplicative_update(f, grad, alpha):
    f = np.asarray(f, dtype=np.float64)
    grad = np.asarray(grad, dtype=np.float64)
    if f.size == 0 or grad.size != f.size:
        return None
    if not np.all(np.isfinite(grad)):
        return None
    mean = float(grad.mean())
    if not np.isfinite(mean) or mean <= 0.0:
        return None
    pressure = grad / mean - 1.0
    pressure = np.clip(pressure, -10.0, 10.0)
    alpha = float(np.clip(alpha, 1e-6, 0.9))
    ratio = np.exp(-alpha * pressure)
    new = f * ratio
    new = np.where(np.isfinite(new), new, 0.0)
    new = np.maximum(new, 1e-12)
    s = float(new.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return new / s


def add_noise(f, rng, strength):
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return f.copy()
    strength = float(min(0.75, max(0.0, strength)))
    if strength <= 0.0:
        return f.copy()
    new = f * np.exp(rng.normal(0.0, strength, f.size))
    new = np.where(np.isfinite(new), new, 0.0)
    new = np.maximum(new, 1e-12)
    s = float(new.sum())
    if not np.isfinite(s) or s <= 0.0:
        return f.copy()
    return new / s


def shift_sequence(f, rng):
    f = np.asarray(f, dtype=np.float64)
    if f.size <= 1:
        return f.copy()
    s = int(rng.integers(0, f.size))
    if s == 0:
        return f.copy()
    return normalize_nonnegative(np.roll(f, s))


def blend_sequence(a, b, rng):
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    if a.size != b.size:
        return a.copy()
    t = float(rng.uniform(0.35, 0.95))
    new = t * a + (1.0 - t) * b
    return normalize_nonnegative(new)


def resample_sequence(f, n_new):
    f = np.asarray(f, dtype=np.float64).ravel()
    n_new = int(max(1, n_new))
    if f.size == 0:
        return np.ones(n_new, dtype=np.float64)
    if f.size == n_new:
        return normalize_nonnegative(f)
    old = np.arange(f.size, dtype=np.float64)
    new = np.linspace(0.0, f.size - 1.0, n_new)
    out = np.interp(new, old, f)
    return normalize_nonnegative(out)


def optimize_sequence(f0, budget, rng, tag=""):
    start = time.monotonic()
    budget = max(0.0, float(budget))
    stop = start + budget
    if stop <= start:
        f = normalize_nonnegative(f0)
        return f, internal_score_general(f)

    n = int(max(2, len(f0)))
    f = np.asarray(f0, dtype=np.float64).ravel()
    if f.size != n:
        f = resample_sequence(f, n)
    f = normalize_nonnegative(f)

    best = f.copy()
    best_score = internal_score_general(best)
    curr = f.copy()
    alpha = 0.07
    no_improve = 0
    K_base = max(64, min(512, n // 8))

    while time.monotonic() < stop:
        c = fft_self_convolution(curr)
        maxc = float(np.max(c)) if c.size else 0.0
        if not np.isfinite(maxc):
            curr = best.copy()
            alpha = 0.05
            no_improve += 1
            continue
        score = 2.0 * float(n) * max(0.0, maxc)

        if np.isfinite(score) and score < best_score:
            best = curr.copy()
            best_score = float(score)
            no_improve = 0

        grad = top_subgradient(curr, c, K_base, 2.7)
        if not np.all(np.isfinite(grad)) or float(grad.mean()) <= 0.0:
            curr = add_noise(best, rng, 0.10)
            alpha = 0.07
            no_improve += 1
            continue

        accepted = False
        trial_alphas = (alpha, max(1e-5, alpha * 0.5), max(1e-5, alpha * 0.2))
        best_trial = None
        for a in trial_alphas:
            new = multiplicative_update(curr, grad, a)
            if new is None:
                continue
            cnew = fft_self_convolution(new)
            mnew = float(np.max(cnew)) if cnew.size else 0.0
            sc = 2.0 * float(n) * max(0.0, mnew)
            if np.isfinite(sc) and (best_trial is None or sc < best_trial[0]):
                best_trial = (sc, new, a)

        if best_trial is not None and best_trial[0] < score:
            sc, new, chosen_a = best_trial
            curr = new
            if sc < best_score:
                best = curr.copy()
                best_score = float(sc)
                no_improve = 0
            else:
                no_improve += 1
            if chosen_a >= alpha:
                alpha = min(0.45, alpha * 1.15)
            else:
                alpha = max(1e-4, min(alpha, chosen_a * 1.25))
            accepted = True

        if not accepted:
            no_improve += 1
            alpha = max(1e-5, alpha * 0.72)

        if alpha < 1e-4 or no_improve > 28:
            r = float(rng.random())
            if r < 0.35:
                strength = min(0.35, 0.07 + 0.015 * float(np.clip(no_improve, 0, 30)))
                curr = add_noise(best, rng, strength)
            elif r < 0.60:
                curr = shift_sequence(best, rng)
            else:
                fresh_kind = ("thue" if float(rng.random()) < 0.45
                              else "fourier" if float(rng.random()) < 0.70
                              else "random")
                fresh_amp = float(rng.uniform(0.5, 1.0))
                fresh = make_initial(n, rng, fresh_kind, fresh_amp)
                curr = blend_sequence(best, fresh, rng)
            c = fft_self_convolution(curr)
            sc = 2.0 * float(n) * max(0.0, float(np.max(c)))
            if np.isfinite(sc) and sc < best_score:
                best = curr.copy()
                best_score = float(sc)
            alpha = 0.07
            no_improve = 0

        if no_improve > 70:
            fresh_kind = ("thue" if float(rng.random()) < 0.4
                          else "fourier" if float(rng.random()) < 0.7
                          else "cosine")
            fresh_amp = float(rng.uniform(0.45, 1.0))
            fresh = make_initial(n, rng, fresh_kind, fresh_amp)
            curr = blend_sequence(best, fresh, rng)
            c = fft_self_convolution(curr)
            sc = 2.0 * float(n) * max(0.0, float(np.max(c)))
            if np.isfinite(sc) and sc < best_score:
                best = curr.copy()
                best_score = float(sc)
            alpha = 0.10
            no_improve = 0

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for a non-negative sequence minimizing the given objective."""
    seed = int(seed)
    rng = np.random.default_rng(seed)
    total_budget = max(0.0, float(budget_s))
    start = time.monotonic()

    reserve = max(35.0, 0.045 * total_budget)
    usable = max(1.0, total_budget - reserve)
    hard_stop = start + total_budget - max(5.0, reserve * 0.4)

    if total_budget <= 2.0:
        existing = globals().get("height_sequence_1", None)
        if existing is not None:
            f = normalize_nonnegative(existing)
            if f.size and np.all(np.isfinite(f)):
                return [float(x) for x in f]
        return [1.0] * 256

    stored = []

    existing = globals().get("height_sequence_1", None)
    if existing is not None:
        f = normalize_nonnegative(existing)
        if f.size and np.all(np.isfinite(f)) and f.sum() >= 0.01:
            sc = internal_score_general(f)
            if np.isfinite(sc):
                stored.append((sc, 0, f.copy(), "existing"))

    lengths = [256, 512, 1024, 2048, 4096]
    fractions = [0.04, 0.09, 0.15, 0.25, 0.47]

    global_best = None
    global_best_score = np.inf

    for n_len, frac in zip(lengths, fractions):
        if time.monotonic() >= hard_stop - 30.0:
            break
        budget_len = max(15.0, frac * usable)
        if time.monotonic() + budget_len > hard_stop - 20.0:
            budget_len = max(0.0, hard_stop - 20.0 - time.monotonic())
        if budget_len < 15.0:
            continue

        candidates = []
        idx = len(stored)

        if global_best is not None:
            warm = resample_sequence(global_best, n_len)
            candidates.append(warm)
            idx += 1

        native_specs = [
            ("thue", 0.80),
            ("thue", 0.55),
            ("fourier", 0.70),
            ("cosine", 0.72),
            ("random", 0.55),
        ]
        for kind, amp in native_specs:
            candidates.append(make_initial(n_len, rng, kind, amp))
            idx += 1

        scored = []
        for seq in candidates:
            seq = normalize_nonnegative(seq)
            sc = internal_score_general(seq)
            if np.isfinite(sc):
                scored.append((sc, idx, seq, "cand"))
                idx += 1

        if not scored:
            seq = np.ones(n_len, dtype=np.float64) / n_len
            sc = internal_score_general(seq)
            scored.append((sc, idx, seq, "fallback"))
            idx += 1

        scored.sort()
        top1_score, _, top1_seq, _ = scored[0]
        optim_seq = top1_seq.copy()
        t1 = budget_len
        if len(scored) >= 2 and budget_len > 120.0:
            t1 = 0.62 * budget_len
        if t1 < 10.0:
            t1 = budget_len

        seq1, sc1 = optimize_sequence(optim_seq, t1, rng, tag=f"len{n_len}_1")
        if np.isfinite(sc1) and sc1 < global_best_score:
            global_best_score = float(sc1)
            global_best = seq1.copy()
        if np.isfinite(sc1):
            stored.append((float(sc1), idx + 1000, seq1.copy(), f"len{n_len}_1"))
            idx += 1

        if len(scored) >= 2:
            t2 = max(0.0, min(0.34 * budget_len, hard_stop - 30.0 - time.monotonic()))
            if t2 >= 12.0:
                _, _, top2_seq, _ = scored[1]
                optim2 = top2_seq.copy()
                if global_best is not None:
                    optim2 = blend_sequence(resample_sequence(global_best, n_len), top2_seq, rng)
                seq2, sc2 = optimize_sequence(optim2, t2, rng, tag=f"len{n_len}_2")
                if np.isfinite(sc2) and sc2 < global_best_score:
                    global_best_score = float(sc2)
                    global_best = seq2.copy()
                if np.isfinite(sc2):
                    stored.append((float(sc2), idx + 2000, seq2.copy(), f"len{n_len}_2"))
                    idx += 1

    if global_best is None:
        # Guaranteed valid fallback; if existing exists and is finite, prefer it.
        if existing is not None:
            f = normalize_nonnegative(existing)
            if f.size and np.all(np.isfinite(f)):
                global_best = f
        if global_best is None:
            e = globals().get("height_sequence_1", None)
            if e is not None:
                f = normalize_nonnegative(e)
                if f.size and np.all(np.isfinite(f)):
                    global_best = f
        if global_best is None:
            global_best = np.ones(256, dtype=np.float64) / 256.0
        stored.append((internal_score_general(global_best), idx + 3000, global_best.copy(), "final_fallback"))

    # Final exact evaluation on the best internal candidates.
    finite_stored = [x for x in stored if np.isfinite(x[0])]
    if finite_stored:
        finite_stored.sort()
    else:
        finite_stored = [(np.inf, 0, global_best.copy(), "only")]

    best_exact_score = np.inf
    best_exact_seq = None
    for item in finite_stored[:12]:
        # Evaluate only a few large candidates exactly to stay inside the final reserve.
        try:
            candidate_list = [float(max(0.0, min(1000.0, x))) for x in item[2]]
            e_score = float(evaluate_sequence(candidate_list))
        except Exception:
            e_score = np.inf
        if np.isfinite(e_score) and e_score < best_exact_score:
            best_exact_score = e_score
            best_exact_seq = item[2]
            if best_exact_score <= 1.01:
                break
        if time.monotonic() > start + total_budget - 2.0:
            break

    if best_exact_seq is None:
        best_exact_seq = global_best

    final = np.asarray(best_exact_seq, dtype=np.float64).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.maximum(final, 0.0)
    final = np.minimum(final, 1000.0)
    if final.size == 0:
        final = np.ones(256, dtype=np.float64)
    s = float(final.sum())
    if not np.isfinite(s) or s <= 0.01:
        final = np.ones(256, dtype=np.float64)
    return [float(x) for x in final]
```
