import numpy as np
import time
from scipy.optimize import minimize

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    condition = u > (cssv - 1) / rho
    rho_hat = np.max(np.where(condition)[0]) + 1
    theta = (cssv[rho_hat-1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

def compute_objective(a):
    """Compute 2n*M/S^2 for sequence a."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    M = np.max(c)
    return 2*n*M/S**2

def subgradient_step(a, lr):
    """One subgradient step to minimize max convolution."""
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    # Find all indices where c is close to max
    threshold = M * 0.999 + 1e-10
    max_indices = np.where(c >= threshold)[0]
    
    # Compute subgradient (average of gradients at maximizing indices)
    grad = np.zeros(n)
    for k in max_indices:
        # grad[m] = 2 * a[k-m] for valid k-m
        for m in range(max(0, k-n+1), min(n, k+1)):
            valid = k - m
            if 0 <= valid < n:
                grad[m] += 2 * a[valid]
    grad /= len(max_indices)
    
    # Update
    a_new = a - lr * grad
    # Project to simplex
    a_new = simplex_project(a_new)
    return a_new, M, c

def fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 2*n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)
    return c[:size]

def compute_objective_fast(a):
    """Compute 2n*max(conv)/sum^2 efficiently."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    M = c.max()
    return 2*n*M/(S*S)

def simplex_project(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1.0) / rho
    rho_hat = np.max(np.where(cond))[0] + 1 if np.any(cond) else 0
    if rho_hat == 0:
        return np.ones(n) / n
    theta = (cssv[rho_hat-1] - 1.0) / rho_hat

# Helper functions at top level

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(fa * fa, fft_size)
    return c[:size]

def compute_obj(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)

def simplex_project(v):
    """Project v onto probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        return np.ones(n) / n
    rho_hat = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_hat-1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)

def grad_conv_max(a, topk=5):
    """Compute (sub)gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a)
    # Get top-k indices
    idx = np.argsort(c)[::-1][:topk]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        if k < n:
            grad[:k+1] += 2.0 * a[:k+1][::-1]
        else:
            lo = k - n + 1
            grad[lo:] += 2.0 * a[lo:][::-1]
    grad /= topk
    return grad, c.max()

def subgrad_optimize(a, max_iters, lr0=1.0, verbose=False):
    """Subgradient descent on the simplex to minimize max conv."""
    a = a.copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in range(1, max_iters+1):
        grad, M = grad_conv_max(a)
        lr = lr0 / np.sqrt(t)
        a = a - lr * grad
        a = simplex_project(a)
        obj = 2.0 * a.size * M  # sum=1
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        if verbose and t % 50000 == 0:
            print(f"    subgrad iter {t}: obj={best_obj:.6f}")
    
    return best_a, best_obj
