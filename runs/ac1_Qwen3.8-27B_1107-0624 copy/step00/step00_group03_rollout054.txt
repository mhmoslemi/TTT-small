import numpy as np
import time

def project_to_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        # All values are below the threshold
        return np.ones(n) / n
    rho_max = np.max(rho[cond])
    theta = (cssv[rho_max - 1] - 1) / rho_max
    return np.maximum(v - theta, 0.0)

def convolution_and_peak(f):
    """Compute the full convolution and return (max_value, argmax_index)."""
    g = np.convolve(f, f, mode='full')
    k_star = int(np.argmax(g))
    return g, k_star

def subgradient_at(f, k_star):
    """Compute the gradient of (f*f)[k_star] with respect to f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = f[k_star - indices]
        # Handle the diagonal case
        if k_star % 2 == 0:
            mid = k_star // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * f[mid]
    return grad

def compute_score_fast(f):
    """Fast computation of the score (equivalent to evaluate_sequence for valid inputs)."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    g = np.convolve(f, f, mode='full')
    return 2.0 * n * np.max(g) / (s * s)

    elif init_type == 'linear':

        f /= np.sum(f)
    elif init_type == 'quadratic':

def optimize_sequence(f, rng, deadline, max_iters=500000):
    """Optimize the sequence using subgradient descent."""
    n = len(f)
    f = project_to_simplex(f.copy()).astype(np.float64)
    best_f = f.copy()
    best_score = compute_score_fast(f)

    momentum = 0.9
    velocity = np.zeros(n)

    stuck_count = 0

        if time.monotonic() > deadline:
            break

        peak_indices = np.argsort(g)[::-1][:top_k]

"""
Minimizes 2*n*max(f*f)/(sum(f))^2 using a hybrid of subgradient descent
with top-k peak averaging and LP-based direction finding.
Uses multiple initializations and adaptive learning rates.
"""
import numpy as np
import time
from scipy.optimize import linprog

def project_to_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.max(rho[cond])
    theta = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - theta, 0.0)

def subgradient_at(f, k_star):
    """Compute the gradient of (f*f)[k_star] with respect to f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = f[k_star - indices]
        if k_star % 2 == 0:
            mid = k_star // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * f[mid]
    return grad

def compute_score(f):
    """Compute 2*n*max(f*f)/(sum(f))^2."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    g = np.convolve(f, f, mode='full')
    return 2.0 * n * np.max(g) / (s * s)

def build_conv_matrix(f):
    """Build matrix A such that A @ x = convolve(f, x, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi+1] = f[k - np.arange(lo, hi + 1)]
    return A

def lp_direction(f):
    """Solve the LP to find a good direction g."""
    n = len(f)
    g_conv = np.convolve(f, f, mode='full')
    M = np.max(g_conv)
    if M <= 0:
        return None
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = result.x
            s = np.sum(g)
            if s > 0:
                return g / s
    except Exception:
        pass
    return None

def optimize_subgradient(f, rng, deadline, top_k=5):
    """Subgradient descent with top-k peak averaging and momentum."""
    n = len(f)
    f = project_to_simplex(f.copy()).astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    
    lr = 1.0 / n
    momentum = 0.7
    velocity = np.zeros(n)
    stuck = 0
    
    while time.monotonic() < deadline:
        g_conv = np.convolve(f, f, mode='full')
        current_max = np.max(g_conv)
        current_score = 2.0 * n * current_max  # sum(f)=1
        
        # Top-k peaks
        k = min(top_k, 2 * n - 1)
        peak_indices = np.argpartition(g_conv, -k)[-k:]
        
        grad = np.zeros(n)
        for ki in peak_indices:
            grad += subgradient_at(f, int(ki))
        grad /= k
        
        # Momentum
        velocity = momentum * velocity + grad
        f_new = f - lr * velocity
        f_new = project_to_simplex(f_new)
        
        new_score = compute_score(f_new)
        f = f_new
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Adaptive LR
        if stuck % 500 == 0 and stuck > 0:
            lr *= 0.9
        if stuck > 10000:
            # Restart
            f = best_f + rng.normal(0, 0.05 * np.mean(best_f), n)
            f = project_to_simplex(f)
            velocity = np.zeros(n)
            stuck = 0
            lr = 1.0 / n
    
    return best_f, best_score

def optimize_lp(f, rng, deadline):
    """LP-based optimization."""
    n = len(f)
    f = project_to_simplex(f.copy()).astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    
    t = 0.1  # mixing parameter
    
    while time.monotonic() < deadline:
        direction = lp_direction(f)
        if direction is None:
            break
        
        # Try different mixing parameters
        best_local_f = best_f
        best_local_score = best_score
        
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            f_new = (1 - t) * f + t * direction
            f_new = np.maximum(f_new, 0)
            f_new = f_new / np.sum(f_new)
            score = compute_score(f_new)
            if score < best_local_score:
                best_local_score = score
                best_local_f = f_new
        
        if best_local_score < best_score:
            best_score = best_local_score
            best_f = best_local_f.copy()
            f = best_local_f.copy()
        else:
            # Try subgradient step
            g_conv = np.convolve(f, f, mode='full')
            k_star = int(np.argmax(g_conv))
            grad = subgradient_at(f, k_star)
            f_new = f - (1.0/n) * grad
            f_new = project_to_simplex(f_new)
            f = f_new
    
    return best_f, best_score

"""
Minimizes 2*n*max(f*f)/(sum(f))^2 using a hybrid optimization strategy:
- Subgradient descent with top-k peak averaging and momentum
- LP-based direction finding (maximize sum(g) s.t. f*g <= M, g>=0)
- Multiple initializations (bimodal, raised cosine, Gaussian bumps, etc.)
- Adaptive learning rates and random restarts
"""
import numpy as np
import time
from scipy.optimize import linprog


def project_to_simplex(v):
    """Project v onto the probability simplex {x : x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = int(np.max(rho[cond]))
    theta = (cssv[rho_max - 1] - 1.0) / rho_max
    return np.maximum(v - theta, 0.0)


def subgradient_at(f, k_star):
    """Compute the gradient of (f*f)[k_star] with respect to f."""
    n = len(f)
    grad = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = f[k_star - indices]
        if k_star % 2 == 0:
            mid = k_star // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * f[mid]
    return grad


def compute_score(f):
    """Compute 2*n*max(f*f)/(sum(f))^2 efficiently."""
    n = len(f)
    s = np.sum(f)
    if s < 0.01:
        return float('inf')
    g = np.convolve(f, f, mode='full')
    return 2.0 * n * np.max(g) / (s * s)


def build_conv_matrix(f):
    """Build matrix A such that A @ x = convolve(f, x, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def lp_direction(f, time_limit=20):
    """Solve LP to find direction g maximizing sum(g) s.t. f*g <= max(f*f), g>=0."""
    n = len(f)
    g_conv = np.convolve(f, f, mode='full')
    M = float(np.max(g_conv))
    if M <= 0:
        return None
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success and result.x is not None:
            g = result.x
            s = float(np.sum(g))
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def optimize_subgradient(f, rng, deadline, top_k=5, verbose=False):
    """Subgradient descent with top-k peak averaging, momentum, and restarts."""
    n = len(f)
    f = project_to_simplex(f.astype(np.float64).copy())
    best_f = f.copy()
    best_score = compute_score(f)

    lr = 2.0 / n
    momentum = 0.8
    velocity = np.zeros(n)
    stuck = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1
        g_conv = np.convolve(f, f, mode='full')
        current_max = float(np.max(g_conv))
        current_score = 2.0 * n * current_max

        k = min(top_k, 2 * n - 1)
        peak_indices = np.argpartition(g_conv, -k)[-k:]

        grad = np.zeros(n)
        for ki in peak_indices:
            grad += subgradient_at(f, int(ki))
        grad /= max(1, k)

        velocity = momentum * velocity + grad
        speed = np.linalg.norm(velocity)
        if speed > 0:
            f_new = f - lr * velocity
        else:
            f_new = f.copy()
        f_new = project_to_simplex(f_new)

        new_score = compute_score(f_new)
        f = f_new

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            stuck = 0
            if verbose and iter_count % 1000 == 0:
                print(f"    iter {iter_count}: score={best_score:.6f}")
        else:
            stuck += 1

        if stuck > 0 and stuck % 1000 == 0:
            lr *= 0.85

        if stuck > 15000:
            noise_scale = 0.1 * max(np.mean(best_f), 1.0 / n)
            f = best_f + rng.normal(0.0, noise_scale, n)
            f = project_to_simplex(f)
            velocity = np.zeros(n)
            stuck = 0
            lr = 2.0 / n

    return best_f, best_score


def optimize_lp(f, rng, deadline, verbose=False):
    """LP-based optimization with line search on mixing parameter."""
    n = len(f)
    f = project_to_simplex(f.astype(np.float64).copy())
    best_f = f.copy()
    best_score = compute_score(f)

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        direction = lp_direction(f, time_limit=30)
        if direction is None:
            break

        best_local_f = f.copy()
        best_local_score = best_score

        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            f_new = (1.0 - t) * f + t * direction
            f_new = np.maximum(f_new, 0.0)
            s = np.sum(f_new)
            if s < 0.01:
                continue
            f_new = f_new / s
            sc = compute_score(f_new)
            if sc < best_local_score:
                best_local_score = sc
                best_local_f = f_new

        if best_local_score < best_score:
            best_score = best_local_score
            best_f = best_local_f.copy()
            f = best_local_f.copy()
            if verbose and iter_count % 5 == 0:
                print(f"    LP iter {iter_count}: score={best_score:.6f}")
        else:
            # Fallback: single subgradient step
            g_conv = np.convolve(f, f, mode='full')
            k_star = int(np.argmax(g_conv))
            grad = subgradient_at(f, k_star)
            f_new = f - (1.0 / n) * grad
            f_new = project_to_simplex(f_new)
            sc = compute_score(f_new)
            if sc < best_score:
                best_score = sc
                best_f = f_new.copy()
            f = f_new

    return best_f, best_score


def initialize_sequence(n, init_type, rng):
    """Create an initial non-negative sequence of length n, normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    if init_type == 'uniform':
        f = np.ones(n)
    elif init_type == 'bimodal':
        f = (np.exp(-((i - 0.2 * n) / (0.12 * n)) ** 2) +
             np.exp(-((i - 0.8 * n) / (0.12 * n)) ** 2))
    elif init_type == 'trimodal':
        f = (np.exp(-((i - 0.15 * n) / (0.08 * n)) ** 2) +
             np.exp(-((i - 0.5 * n) / (0.08 * n)) ** 2) +
             np.exp(-((i - 0.85 * n) / (0.08 * n)) ** 2))
    elif init_type == 'raised_cosine':
        f = np.cos(np.pi * i / (2.0 * n)) ** 2
    elif init_type == 'sqrt_decay':
        f = 1.0 / np.sqrt(i + 1.0)
    elif init_type == 'linear':
        f = n - i
    elif init_type == 'edge_concentrated':
        center = (n - 1.0) / 2.0
        f = 1.0 / (1.0 + 10.0 * (np.abs(i - center) / center) ** 2)
        f += 1.0 / (1.0 + ((np.abs(i) + np.abs(i - (n - 1))) / (0.1 * n)) ** 2)
    elif init_type == 'gaussian':
        center = (n - 1.0) / 2.0
        f = np.exp(-0.5 * ((i - center) / (0.3 * n)) ** 2)
    elif init_type == 'noise':
        f = rng.random(n)
    elif init_type == 'sqrt_bimodal':
        base = (np.exp(-((i - 0.2 * n) / (0.15 * n)) ** 2) +
                np.exp(-((i - 0.8 * n) / (0.15 * n)) ** 2))
        f = np.sqrt(np.maximum(base, 0))
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.0)
    s = np.sum(f)
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    return f


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Searches for a non-negative sequence minimizing 2*n*max(f*f)/(sum(f))^2.
    Uses hybrid subgradient + LP optimization with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 10.0

    best_score = float('inf')
    best_f = None

    total_time = float(budget_s) - 10.0

    # Phase 1: Explore diverse initializations with subgradient descent
    explore_deadline = time.monotonic() + 0.45 * total_time
    configs = [
        (2000, 'bimodal'),
        (2000, 'trimodal'),
        (2000, 'raised_cosine'),
        (1500, 'bimodal'),
        (1500, 'trimodal'),
        (2500, 'bimodal'),
        (2000, 'sqrt_bimodal'),
        (2000, 'gaussian'),
        (1800, 'edge_concentrated'),
        (2000, 'noise'),
        (2200, 'bimodal'),
        (1600, 'sqrt_decay'),
    ]

    print(f"Phase 1: Exploration (deadline in {explore_deadline - time.monotonic():.0f}s)")
    for n, init_type in configs:
        remaining = explore_deadline - time.monotonic()
        if remaining < 15:
            break
        config_dl = time.monotonic() + min(100.0, remaining)
        f0 = initialize_sequence(n, init_type, rng)
        f, s = optimize_subgradient(f0, rng, config_dl, top_k=5)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  n={n}, init={init_type}: score={s:.6f} (NEW BEST)")
        else:
            print(f"  n={n}, init={init_type}: score={s:.6f}")

    print(f"\nBest after exploration: {best_score:.6f}")

    # Phase 2: LP-based refinement on best solution
    if best_f is not None:
        lp_deadline = time.monotonic() + 0.35 * total_time
        if lp_deadline > deadline:
            lp_deadline = deadline
        print(f"\nPhase 2: LP refinement (deadline in {lp_deadline - time.monotonic():.0f}s)")
        f, s = optimize_lp(best_f, rng, min(lp_deadline, deadline), verbose=True)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  After LP refinement: score={s:.6f} (NEW BEST)")

    # Phase 3: Final subgradient polish with more top-k
    if best_f is not None and time.monotonic() < deadline:
        print(f"\nPhase 3: Final subgradient polish")
        f, s = optimize_subgradient(best_f, rng, deadline, top_k=10)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  After final polish: score={s:.6f} (NEW BEST)")

    # Verify with the actual evaluator
    if best_f is not None:
        final_score = evaluate_sequence(best_f.tolist())
        print(f"\nFinal verified score: {final_score:.6f}")
        if np.isfinite(final_score):
            return best_f.tolist()

    # Fallback: uniform sequence
    print("WARNING: Using fallback uniform sequence")
    fallback = np.ones(256) / 256.0
    return fallback.tolist()
