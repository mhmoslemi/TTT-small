import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import scipy.sparse as sp

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)[k] = (f * g)[k] for full convolution."""
    n = len(f)
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    
    rows = []
    cols = []
    data = []
    
    for j in range(n):
        # Column j: A[k, j] = f[k-j] for k = j, j+1, ..., j+n-1
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def solve_lp_for_g(f):
    """Solve the LP: min M s.t. (f*g)[k] <= M, sum(g)=1, g>=0"""
    n = len(f)
    
    # Build the convolution matrix A (2n-1 x n)
    A = build_conv_matrix(f)
    
    # For linprog: minimize c @ x
    # x = [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1] (n+1 elements)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ x <= b_ub
    # Constraints: A @ g - M <= 0
    # In terms of x = [g; M]: [A | -1_col] @ x <= 0
    # Where -1_col is a column of -1s with length 2n-1
    
    # A_ub has shape (2n-1, n+1)
    # First n columns: A
    # Last column: -1
    
    # Building this as sparse:
    # A_ub = [A | -e] where e is a column vector of ones
    neg_one_col = sp.csr_matrix(-np.ones((2*n-1, 1)))
    A_ub = sp.hstack([A, neg_one_col], format='csr')
    b_ub = np.zeros(2*n-1)
    
    # A_eq @ x == b_eq
    # sum(g) = 1: [1, 1, ..., 1, 0] @ x = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_j >= 0, M unconstrained (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        g = result.x[:n]
        return np.clip(g, 0, None)
    else:

def build_conv_matrix(f):
    """Build sparse matrix A where (A @ g) = full_convolution(f, g)."""
    n = len(f)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=np.int32) for j in range(n)])
    data = np.tile(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f):
    """Solve LP: min M s.t. (f*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns g or None if LP fails."""
    n = len(f)
    A = build_conv_matrix(f)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub @ x <= b_ub: A@g - M <= 0
    neg_col = sp.csr_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = hstack([A, neg_col], format='csr')
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq @ x == b_eq: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 60})
