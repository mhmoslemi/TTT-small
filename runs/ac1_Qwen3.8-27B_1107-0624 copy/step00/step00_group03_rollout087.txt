```
def params_to_sequence(n, ptype, params):
```

```
def random_perturb_sequence(a, rng, strength):
    a = _normalize_sum(a)
    n=a.size
    x=np.linspace(0,1,n)
    loga=np.log(np.maximum(a,1e-300))
    noise=np.zeros(n)
    max_modes = min(16, max(2, n//32))
    for k in range(1, max_modes+1):
       amp = strength / float(k + 1)
       phase = float(rng.uniform(0.0, 2.0*np.pi))
       coeff = float(rng.normal(0.0, 1.0))
       noise += amp * coeff * np.sin(2.0*np.pi*k*x + phase)
    # small smooth quadratic noise
    noise += strength * 0.1 * float(rng.normal()) * (x*x - 0.5)
    loga = loga + noise
    return _normalize_sum(np.exp(loga - float(np.max(loga))))
```

```python
"""
Search algorithm for minimizing
    2 * n * max(np.convolve(a, a)) / sum(a)^2
over nonnegative sequences a.

The search combines:
1. A randomized scan over low-dimensional asymmetric smooth log-amplitude
   families (exponential, quadratic, rational/logarithmic, and smooth
   Fourier-like shapes).
2. Full-space multiplicative implicit gradient descent on a soft top-lag
   objective.  The gradient uses a smoothed maximum of the convolution lags.
3. Adam-style averaging, adaptive step scaling, periodic perturbations, and
   final authoritative evaluation with evaluate_sequence().

All sequences are kept nonnegative and are normalized to sum 1 during
optimization, since the objective is scale invariant.
"""

import numpy as np
import time


def _safe_array(a):
    """Return a finite, nonnegative, clipped 1D float array."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        arr = np.zeros(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    return np.clip(arr, 0.0, 1000.0)


def _normalize_sum(a):
    """Normalize a nonnegative array to sum 1."""
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if arr.size == 0:
        return np.zeros(1, dtype=np.float64)
    if not np.isfinite(s) or s <= 1e-14:
        return np.full(arr.size, 1.0 / float(arr.size), dtype=np.float64)
    arr /= s
    return arr


def resample_sequence(a, n):
    """Linearly resample a nonnegative sequence to length n and normalize."""
    a = _normalize_sum(a)
    n = int(n)
    if a.size == n:
        return a
    if a.size == 0 or n <= 0:
        return np.zeros(max(1, n), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    return _normalize_sum(np.interp(x_new, x_old, a))


def convolve_fast(x, y):
    """Real linear convolution, using direct for small cases and FFT otherwise."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if x.size == 0 or y.size == 0:
        return np.zeros(0, dtype=np.float64)

    n_out = x.size + y.size - 1
    if x.size * y.size <= 2000000:
        return np.convolve(x, y)

    m = 1 << (n_out - 1).bit_length()
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    Z = np.fft.irfft(X * Y, m)
    return Z[:n_out]


def sequence_score(a):
    """Internal scale-invariant score: 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = convolve_fast(a, a)
    if c.size == 0:
        return np.inf
    mc = float(np.max(c))
    if not np.isfinite(mc) or mc <= 0.0:
        return np.inf
    return float(2.0 * a.size * mc / (s * s))


def p_to_sequence(p):
    """Convert log-amplitude p to a normalized nonnegative sequence."""
    arr = np.asarray(p, dtype=np.float64)
    if arr.size == 0:
        return np.zeros(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = arr - float(np.max(arr))
    arr = np.clip(arr, -50.0, 0.0)
    e = np.exp(arr)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(arr.size, 1.0 / float(arr.size), dtype=np.float64)
    return e / s


def params_to_sequence(n, ptype, params):
    """Build a sequence from a parameterized log-amplitude model."""
    n = int(n)
    if n <= 0:
        return np.zeros(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n)
    p = np.zeros(n, dtype=np.float64)
    ptype = int(ptype)

    if ptype == 0:
        # Pure exponential ramp.
        p += float(params[0]) * x
    elif ptype == 1:
        # Quadratic exponent.
        x2 = x * x
        p += float(params[0]) * x + float(params[1]) * x2
    elif ptype == 2:
        # Cubic exponent.
        x2 = x * x
        p += float(params[0]) * x + float(params[1]) * x2 + float(params[2]) * x2 * x
    elif ptype == 3:
        # Concave logarithmic growth.
        p += float(params[0]) * np.log1p(float(params[1]) * x)
    elif ptype == 4:
        # Polynomial exponent, terms x, x^2, ...
        power = x.copy()
        for idx in range(len(params)):
            p += float(params[idx]) * power
            power *= x
    elif ptype == 5:
        # Linear + low-frequency trigonometric shape.
        p += float(params[0]) * x
        p += float(params[1]) * np.cos(np.pi * x)
        p += float(params[2]) * np.cos(2.0 * np.pi * x)
        p += float(params[3]) * np.sin(np.pi * x)
        p += float(params[4]) * np.sin(2.0 * np.pi * x)
    elif ptype == 6:
        # Rational-concave shape.
        p += float(params[0]) * x
        p += float(params[1]) * (x / (1.0 + x))
    else:
        if len(params) > 0:
            p += float(params[0]) * x

    return p_to_sequence(p)


def random_candidate(rng):
    """Return a random parameterized model candidate."""
    r = float(rng.random())

    if r < 0.15:
        return (0, (float(rng.uniform(0.2, 3.5)),))

    if r < 0.30:
        return (1, (float(rng.uniform(0.2, 3.0)), float(rng.uniform(-1.5, 1.5))))

    if r < 0.42:
        return (2, (float(rng.uniform(0.2, 2.5)),
                    float(rng.uniform(-1.0, 1.0)),
                    float(rng.uniform(-1.0, 1.0))))

    if r < 0.55:
        return (3, (float(rng.uniform(0.5, 3.0)), float(rng.uniform(0.1, 5.0))))

    if r < 0.70:
        vals = [0.0] * 6
        for i in range(6):
            vals[i] = float(rng.normal(0.0, 1.0))
        vals[0] += 0.5
        return (4, tuple(vals))

    if r < 0.85:
        vals = [float(rng.uniform(0.2, 3.0))]
        for _ in range(4):
            vals.append(float(rng.normal(0.0, 0.8)))
        return (5, tuple(vals))

    return (6, (float(rng.uniform(0.2, 3.0)), float(rng.uniform(-1.0, 1.0))))


def add_candidate(top, score, rid, ptype, params, k):
    """Maintain a small top list without using lambdas."""
    if not np.isfinite(score):
        return
    item = (float(score), int(rid), int(ptype), tuple(params))
    if len(top) < k:
        top.append(item)
        top.sort()
    elif score < top[-1][0]:
        top[-1] = item
        top.sort()


def scan_top_candidates(rng, n_scan, deadline, max_evals):
    """Quick random/parametric scan at a short length to find good starts."""
    top = []
    rid = 0
    n_scan = int(max(64, n_scan))
    K = 12

    # Deterministic exponential ramps.
    for alpha in np.linspace(0.1, 3.5, 70):
        alpha = float(alpha)
        a = params_to_sequence(n_scan, 0, (alpha,))
        s = sequence_score(a)
        add_candidate(top, s, rid, 0, (alpha,), K)
        rid += 1
        if time.monotonic() >= deadline:
            break

    # Deterministic quadratic ramps.
    for aa in np.linspace(0.1, 3.0, 12):
        aa = float(aa)
        for bb in np.linspace(-1.5, 1.5, 12):
            bb = float(bb)
            a = params_to_sequence(n_scan, 1, (aa, bb))
            s = sequence_score(a)
            add_candidate(top, s, rid, 1, (aa, bb), K)
            rid += 1
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # Deterministic log-concave ramps.
    for aa in np.linspace(0.5, 3.0, 10):
        aa = float(aa)
        for bb in np.linspace(0.1, 5.0, 10):
            bb = float(bb)
            a = params_to_sequence(n_scan, 3, (aa, bb))
            s = sequence_score(a)
            add_candidate(top, s, rid, 3, (aa, bb), K)
            rid += 1
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    max_evals = int(max(0, max_evals))
    count = 0
    while time.monotonic() < deadline and count < max_evals:
        ptype, params = random_candidate(rng)
        a = params_to_sequence(n_scan, ptype, params)
        s = sequence_score(a)
        add_candidate(top, s, rid, ptype, params, K)
        rid += 1
        count += 1

    return top


def random_perturb_sequence(a, rng, strength):
    """Smoothly perturb a sequence in log-amplitude space."""
    a = _normalize_sum(a)
    n = a.size
    if n <= 0:
        return a

    x = np.linspace(0.0, 1.0, n)
    loga = np.log(np.maximum(a, 1e-300))
    noise = np.zeros(n, dtype=np.float64)

    max_modes = min(16, max(2, n // 32))
    for k in range(1, max_modes + 1):
        amp = strength / float(k + 1)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        coeff = float(rng.normal(0.0, 1.0))
        noise += amp * coeff * np.sin(2.0 * np.pi * k * x + phase)

    noise += strength * 0.1 * float(rng.normal()) * (x * x - 0.5)
    loga += noise
    loga = loga - float(np.max(loga))
    return _normalize_sum(np.exp(loga))


def optimize_sequence(a0, rng, time_limit, verbose_id=0):
    """
    Local optimization from an initial nonnegative sequence.

    Uses a soft top-lag objective and Adam-style multiplicative updates.
    Returns (best_sequence, best_internal_score).
    """
    a = _normalize_sum(a0)
    n = int(a.size)
    if n <= 0:
        return np.zeros(1, dtype=np.float64), np.inf

    time_limit = max(1.0, float(time_limit))
    start_time = time.monotonic()
    end_time = start_time + time_limit
    tiny = 1e-300

    c = convolve_fast(a, a)
    max_c = float(np.max(c))
    score_cur = 2.0 * n * max_c
    best_a = a.copy()
    best_score = score_cur
    prev_score = score_cur

    m_state = np.zeros(n, dtype=np.float64)
    v_state = np.zeros(n, dtype=np.float64)
    t_adam = 0
    scale = 0.10
    stagnation = 0
    last_print = start_time

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - start_time

        # Current objective.
        c = convolve_fast(a, a)
        max_c = float(np.max(c))
        if not np.isfinite(max_c):
            break
        score_cur = 2.0 * n * max_c

        if score_cur < best_score - 1e-13:
            best_score = score_cur
            best_a = a.copy()

        if score_cur < prev_score - 1e-14:
            stagnation = 0
        else:
            stagnation += 1
        prev_score = score_cur

        # If the current point drifted far from the local best, reset.
        if score_cur > best_score * 1.02 + 1e-12 and stagnation > 30:
            a = best_a.copy()
            c = convolve_fast(a, a)
            max_c = float(np.max(c))
            score_cur = best_score
            prev_score = score_cur
            m_state.fill(0.0)
            v_state.fill(0.0)
            t_adam = 0
            stagnation = 0
            continue

        # Periodic escape from local minima.
        if stagnation > 350:
            strength = 0.15 + 0.05 * min(5, stagnation // 350)
            a_try = random_perturb_sequence(best_a, rng, strength)
            c_try = convolve_fast(a_try, a_try)
            s_try = 2.0 * n * float(np.max(c_try))
            if np.isfinite(s_try) and s_try < best_score:
                a = a_try
                c = c_try
                max_c = float(np.max(c_try))
                score_cur = s_try
                best_a = a copy()
                best_score = s_try
            else:
                a = best_a.copy()
                c = convolve_fast(a, a)
                max_c = float(np.max(c))
                score_cur = best_score
            prev_score = score_cur
            m_state.fill(0.0)
            v_state.fill(0.0)
            t_adam = 0
            stagnation = 0
            continue

        # Soft top-lag weights.
        focus = 0.10 - 0.09 * min(1.0, elapsed / 600.0)
        if focus < 0.01:
            focus = 0.01
        threshold = (1.0 - focus) * max_c
        w = np.maximum(c - threshold, 0.0)
        w = w * w
        ws = float(np.sum(w))
        if not np.isfinite(ws) or ws <= tiny:
            w = np.zeros_like(c)
            idx = int(np.argmax(c))
            if 0 <= idx < w.size:
                w[idx] = 1.0
            ws = 1.0
        else:
            w /= ws

        # Gradient of smooth maximum of convolution with respect to log(a).
        full = convolve_fast(a, w[::-1])
        if full.size >= 2 * n - 1:
            h = full[n - 1:2 * n - 1][::-1]
        else:
            h = np.zeros(n, dtype=np.float64)
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)

        g = 2.0 * a * h
        g -= float(np.mean(g))
        grms = float(np.sqrt(np.mean(g * g)))
        if not np.isfinite(grms) or grms < 1e-300:
            a_try = random_perturb_sequence(a, rng, 0.25)
            c_try = convolve_fast(a_try, a_try)
            s_try = 2.0 * n * float(np.max(c_try))
            if np.isfinite(s_try) and s_try < score_cur:
                a = a_try
                score_cur = s_try
                if s_try < best_score:
                    best_score = s_try
                    best_a = a.copy()
            prev_score = score_cur
            m_state.fill(0.0)
            v_state.fill(0.0)
            t_adam = 0
            continue

        g /= grms

        # Adam.
        t_adam += 1
        m_state = 0.9 * m_state + 0.1 * g
        v_state = 0.999 * v_state + 0.001 * (g * g)
        b1 = 0.9
        b2 = 0.999
        m_hat = m_state / (1.0 - (b1 ** t_adam))
        v_hat = v_state / (1.0 - (b2 ** t_adam))
        step = m_hat / (np.sqrt(v_hat) + 1e-8)
        step -= float(np.mean(step))
        step = np.clip(step, -1.5, 1.5)

        # Line search over log-step scale.
        accepted_frac = -1.0
        for frac in (1.0, 0.5, 0.25):
            trial_scale = scale * frac
            if trial_scale <= 0.001:
                break
            a_trial = a * np.exp(trial_scale * step)
            a_trial = np.maximum(a_trial, tiny)
            a_trial = _normalize_sum(a_trial)
            c_trial = convolve_fast(a_trial, a_trial)
            s_trial = 2.0 * n * float(np.max(c_trial))

            if np.isfinite(s_trial) and s_trial < score_cur - 1e-14:
                a = a_trial
                c = c_trial
                max_c = float(np.max(c))
                score_cur = s_trial
                prev_score = s_trial
                if s_trial < best_score:
                    best_score = s_trial
                    best_a = a_trial.copy()
                accepted_frac = frac
                break

        if accepted_frac >= 0.0:
            if abs(accepted_frac - 1.0) <= 1e-12:
                scale = min(0.50, scale * 1.08)
            else:
                scale = max(0.004, scale * accepted_frac)
        else:
            scale = max(0.004, scale * 0.75)

        if time.monotonic() - last_print > 20.0:
            print(
                f"opt {verbose_id}: t={elapsed:.1f}s "
                f"cur={score_cur:.6f} best={best_score:.6f} "
                f"scale={scale:.4f} stag={stagnation}",
                flush=True
            )
            last_print = time.monotonic()

    s_final = sequence_score(best_a)
    if np.isfinite(s_final) and s_final < best_score:
        best_score = s_final
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):

            parent_src = height_sequence_1

            parent_src = None

    parent = None
    if parent_src is not None:
        try:
            parr = _safe_array(parent_src)

                parent = parr

            parent = None

    best_a = None
    best_score = np.inf
    candidates = []

    print(f"initial internal score={best_score:.6f} n={best_a.size}", flush=True)

    # Choose main sequence length.
    if parent is not None and 512 <= parent.size <= 5000:
        n_main = int(parent.size)
    else:
        n_main = 2048

    # Parameter scan on a short length.

        scan_alloc = min(60.0, max(15.0, 0.30 * budget))
    else:
        scan_alloc = min(240.0, max(90.0, 0.12 * budget))
    scan_deadline = start_time + scan_alloc
    n_scan = 256 if budget < 450 else 512
    top_candidates = scan_top_candidates(
        rng,
        n_scan,
        scan_deadline,
        max_evals=20000
    )

    # Build a set of starting sequences at the main length.
    starts = []

    # Current best resampled to main length.
    starts.append(resample_sequence(best_a, n_main))

    # Parent, if useful.
    if parent is not None:
        starts.append(resample_sequence(parent, n_main))

    # Explicit strong defaults.
    starts.append(params_to_sequence(n_main, 0, (1.5,)))
    starts.append(params_to_sequence(n_main, 0, (1.7,)))
    starts.append(params_to_sequence(n_main, 1, (1.8, -0.35)))
    starts.append(params_to_sequence(n_main, 3, (1.8, 1.2)))
    starts.append(params_to_sequence(n_main, 6, (1.8, -0.2)))

    # Top scanned candidates.
    for item in top_candidates[:8]:
        ptype = item[2]
        params = item[3]
        starts.append(params_to_sequence(n_main, ptype, params))

    # Rank starts by their direct internal score.
    scored = []
    for idx, a in enumerate(starts):
        s = sequence_score(a)
        if np.isfinite(s):
            scored.append((s, idx))
    scored.sort()
    starts = [starts[item[1]] for item in scored[:9]]

    # Run local optimizations.
    reserve_for_final = 150.0
    run_deadline = hard_deadline - reserve_for_final

    for idx, a0 in enumerate(starts):
        now = time.monotonic()
        if now > run_deadline:
            break
        remaining = max(0.0, run_deadline - now)
        n_left = max(1, len(starts) - idx)
        alloc = min(240.0, remaining / float(n_left))
        if alloc < 30.0:
            alloc = min(30.0, remaining)
        if alloc <= 0.0:
            break

        a_run, s_run = optimize_sequence(a0, rng, alloc, verbose_id=idx)
        if np.isfinite(s_run) and a_run is not None:
            candidates.append(a_run.copy())
            if s_run < best_score:
                best_score = s_run
                best_a = a_run.copy()
                print(f"new internal best after run {idx}: {best_score:.6f}", flush=True)

    # Final polish of the best sequence.
    now = time.monotonic()
    if now < hard_deadline - 45.0 and best_a is not None:
        alloc = hard_deadline - now - 40.0
        if alloc > 60.0:
            a_pol, s_pol = optimize_sequence(best_a, rng, alloc, verbose_id=99)
            if np.isfinite(s_pol) and a_pol is not None:
                candidates.append(a_pol.copy())
                if s_pol < best_score:
                    best_score = s_pol
                    best_a = a_pol.copy()
                    print(f"polished internal best: {best_score:.6f}", flush=True)

    # Authoritative final selection using evaluate_sequence when available.
    final_a = best_a
    final_score = np.inf

    # Evaluate a few of the best candidates with the official evaluator.
    unique_candidates = []
    seen_lengths = set()
    for cand in reversed(candidates[-20:]):
        cand = np.clip(np.asarray(cand, dtype=np.float64), 0.0, 1000.0)
        # Prefer distinct lengths to cover possible exact-eval differences.
        L = int(cand.size)
        if L in seen_lengths and len(unique_candidates) < 12:
            continue
        seen_lengths.add(L)
        unique_candidates.append(cand)
        if len(unique_candidates) >= 12:
            break
    unique_candidates.reverse()

    try:
        for cand in unique_candidates:
            cand_list = [float(x) for x in cand]
            cs = float(evaluate_sequence(cand_list))
            if np.isfinite(cs) and cs < final_score:
                final_score = cs
                final_a = np.asarray(cand_list, dtype=np.float64)
    except Exception:
        final_score = np.inf

    # Fallback if the evaluator is unavailable or all evaluations are invalid.
    if final_a is None or len(final_a) == 0:
        final_a = np.ones(2048, dtype=np.float64) / 2048.0

    if not np.isfinite(final_score):
        # Use the best internal sequence, clamped and normalized.
        final_a = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
        final_a = _normalize_sum(final_a)
        try:
            final_score = float(evaluate_sequence([float(x) for x in final_a]))
        except Exception:
            final_score = np.inf

    # Last-fallback valid sequence.
    if final_a is None or len(final_a) == 0:
        final_a = np.ones(2048, dtype=np.float64) / 2048.0

    final_a = np.clip(np.asarray(final_a, dtype=np.float64), 0.0, 1000.0)
    final_list = [float(x) for x in final_a]

    print(f"final internal tracking score={best_score:.6f}", flush=True)
    print(f"final evaluate_sequence score={final_score:.6f}", flush=True)
    print(f"final length={len(final_list)}", flush=True)

    return final_list
```
