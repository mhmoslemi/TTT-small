def _normalize_sum1(a):
    """Normalize array to have sum 1, all non-negative."""
    a = np.maximum(a, 0)

    if s < 1e-15:

    Y = np.fft.rfft(y, size)

1. Initialize with several candidates.
2. Main loop:
   a. Compute conv(a,a) via FFT.

   f. Accept if score improves, else reject and reduce step.
   g. Occasionally (every M iterations or when stuck), try a random perturbation.
3. Return best found.

import numpy as np
import time

def _fft_convolve_auto(a):
    """Full auto-convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A*A, size)[:2*n-1]

def _fft_conv_cross(x, y):
    """Full cross-convolution x*y via FFT."""
    n = len(x) + len(y) - 1
    size = 1
    while size < n:
        size <<= 1
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    return np.fft.irfft(X*Y, size)[:n]

def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n)/n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _stable_softmax(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

    """Compute the score (assuming sum a = 1)."""
