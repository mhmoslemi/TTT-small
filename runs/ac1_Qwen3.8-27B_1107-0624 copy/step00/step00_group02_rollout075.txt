LP-guided iterative optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) LP to find flattening direction, (2) adaptive step size,
(3) subgradient polishing phase, (4) multi-length search.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_sparse_A(f):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, m - 1)
        length = k_end - k_start + 1
        rows[idx:idx + length] = np.arange(k_start, k_end + 1)
        cols[idx:idx + length] = j
        vals[idx:idx + length] = f[:length]
        idx += length
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(m, n))


def solve_direction_lp(f, M, time_limit=30.0):
    """Solve max sum(g) s.t. (conv of f,g)_k <= M, g >= 0, g <= g_max."""
    n = len(f)
    try:
        A = build_sparse_A(f)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        g_max = 2.0 / n
        result = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, g_max)] * n,
            method='highs',
            options={"time_limit": time_limit}
        )
        if result.success:
            return np.maximum(result.x, 0.0)
        elif len(result.x) == n and np.isfinite(result.x).all():
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    valid = u * rho_candidates > (cssv - 1.0)
    if not np.any(valid):
        return np.ones(n) / n
    rho = np.where(valid)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_update(f, step_size):
    """One subgradient step on max_k (f*f)_k with simplex projection."""
    n = len(f)
    conv = np.convolve(f, f, mode='full')
    max_val = np.max(conv)
    # Find active set (within tolerance)
    tol = 1e-8 * max_val
    active = np.where(conv >= max_val - tol)[0]
    # Average subgradient over active set
    s = np.zeros(n, dtype=np.float64)
    for k in active:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                s[j] += f[idx]
    s = s / len(active)
    f_new = f - step_size * s
    f_new = project_simplex(f_new)
    return f_new, max_val

def make_initialization(n, init_type, rng):
