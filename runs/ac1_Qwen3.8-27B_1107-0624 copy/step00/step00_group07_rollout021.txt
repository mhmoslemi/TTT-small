def pressure_direction(f, top_k=5):
    """Compute a direction that reduces the max convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    # Find the top-k lags
    top_lags = np.argsort(conv)[-top_k:]

    pressure = np.zeros(n)

```python
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import ...
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Build the constraint matrix A (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # k ranges from 0 to 2n-2, j ranges from 0 to n-1
    
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize c^T g = sum(g) subject to A g <= M, g >= 0
    c = -np.ones(n)  # Minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def build_conv_matrix(f):
    n = len(f)
    # Column j has values f[0], f[1], ..., f[n-1] at rows j, j+1, ..., j+n-1
    # We can use the fact that this is a "hankel-like" sparse matrix
    
    # All columns have the same data (f), just shifted in row
    # Total non-zeros: n * n = n²
    # (But some are outside the [0, 2n-2] range for the first and last columns)
    
    # Actually, column j has non-zeros at rows j to j+n-1, all within [0, 2n-2]
    # (since j >= 0 and j+n-1 <= n-1+n-1 = 2n-2)
    
    # So all n columns have exactly n non-zeros.
    # Total: n² non-zeros.
    
    # For the COO format:
    # cols: [0]*n + [1]*n + ... + [(n-1)]*n
    # rows: [0..n-1] + [1..n] + [2..n+1] + ... + [(n-1)..(2n-2)]
    # data: f + f + f + ... + f (n times)
    
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.concatenate([np.arange(j, j+n) for j in range(n)])
    data = np.tile(f, n)
    
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
```

The `np.concatenate` with n arrays of length n is O(n²) but done in C, so it should be fast. For n=6733, that's 45M elements, which is manageable.
