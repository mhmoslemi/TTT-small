```python
def subgradient_step(a, step_size):
    """One step of subgradient descent on max(a*a) with projection."""
    n = len(a)
    ac = np.convolve(a, a)
    k_star = np.argmax(ac)
    
    # Gradient of (a*a)_{k_star} with respect to a
    grad = np.zeros(n)
    for i in range(n):
        # d(a*a)_{k_star} / da_i = a_{k_star - i} + a_{k_star - i} (if k_star != 2i)
        # = a_{k_star - i} (if k_star == 2i)
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = a[i]
            else:
                grad[i] = 2 * a[j]
    
    # Subgradient of f(a) = 2n * max(a*a) / (sum a)^2
    # = 2n * [(grad * (sum a)^2 - max(a*a) * 2 * (sum a)) / (sum a)^4]
    # = 2n * [grad / (sum a)^2 - 2 * max(a*a) / (sum a)^3]
    s = np.sum(a)
    max_ac = ac[k_star]
    grad_f = 2 * n * (grad / (s * s) - 2 * max_ac / (s ** 3))
    
    # Take step
    a_new = a - step_size * grad_f
    
    # Project to non-negative, sum = 1
    a_new = np.clip(a_new, 0, None)
    s_new = np.sum(a_new)
    if s_new > 0:
        a_new = a_new / s_new
    else:
        a_new = a / n
    
    return a_new, max_ac
```

Uses the linear programming approach from the SOTA description, with

import numpy as np
from scipy.optimize import linprog

def auto_conv(a):
    """Compute full auto-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return np.maximum(result, 0)  # Clip small negative numerical errors


def compute_score(a):
    """Compute the objective value (matches evaluate_sequence logic)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    ac = auto_conv(a)
    return float(2.0 * n * np.max(ac) / (s * s))


def normalize(a):
    """Normalize to sum=1."""
    s = np.sum(a)
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def build_lp_matrix(a):
    """Build the sparse LP constraint matrix.
    
    A[k, j] = a[j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(a)
    # Each column j has non-zeros at rows j, j+1, ..., min(j+n-1, 2n-2)
    col_starts = np.arange(n)
    col_ends = np.minimum(col_starts + n, 2*n - 1)
    
    # Count non-zeros per column
    counts = col_ends - col_starts
    
    # Build COO data
    total_nnz = int(np.sum(counts))
    row_data = np.repeat(col_starts, counts) + np.arange(total_nnz) - np.repeat(col_starts, counts)
    # Actually, let me do this more carefully
    
    rows = np.concatenate([np.arange(s, e) for s, e in zip(col_starts, col_ends)])
    cols = np.concatenate([np.full(e - s, j) for j, s, e in zip(col_starts, col_starts, col_ends)])
    vals = np.concatenate([np.full(e - s, a[j]) for j, s, e in zip(col_starts, col_starts, col_ends)])
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def solve_lp(a, m_factor=1.0, timeout=120):
    """Solve the LP: max sum(b) s.t. b>=0, (a*b)[k] <= M*m_factor.
    
    Returns normalized b or None.
    """
    n = len(a)
    ac = auto_conv(a)
    M = float(np.max(ac)) * m_factor
    
    if M <= 0 or not np.isfinite(M):
        return None
    
    A = build_lp_matrix(a)
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'time_limit': timeout, 'presolve': True})
    
    if res.success or res.status in (2, 9):  # 2=local opt, 9=time limit
        x = res.x
        if x is not None:
            b = np.clip(x, 0, None)
            s = np.sum(b)
            if s > 1e-10:
                return b / s
    return None


def random_init(n, rng):
    """Generate a random non-negative sequence."""
    strategy = rng.integers(0, 5)
    if strategy == 0:
        a = rng.uniform(0.1, 1.0, n)
    elif strategy == 1:
        # Block-structured
        nblocks = rng.integers(5, 30)
        block_len = n // nblocks
        a = np.repeat(rng.uniform(0.5, 2.0, nblocks), block_len)
        if len(a) < n:

        kernel = np.ones(50) / 50
        a = np.convolve(a, kernel, mode='same')
    elif strategy == 3:

6. Repeat

    ac = auto_conv(a)
    max_ac = np.max(ac)
    
    # Find hot spots
    hot_mask = ac > hot_threshold * max_ac
    hot_indices = np.where(hot_mask)[0]
    
    if len(hot_indices) == 0:
        return a.copy()
    
    # For each hot spot, find the most contributing index
    a_new = a.copy()
    for k in hot_indices:
        # ac[k] = sum_j a_j * a_{k-j}
        # The j that maximizes a_j * a_{k-j} is the "most contributing"
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        j_range = range(j_min, j_max + 1)
        contributions = np.array([a[j] * a[k-j] for j in j_range])
        j_best = j_min + np.argmax(contributions)
        # Decrease a[j_best] slightly
        a_new[j_best] *= 0.99
        a_new[k - j_best] *= 0.99
    
    a_new = np.clip(a_new, 0, None)
    return normalize(a_new)

The subgradient of max_k ac[k] at a is the gradient of ac[k*] where k* = argmax.

`grad_ac_kstar[i] = 2 * a[kstar - i]` (if 0 <= kstar-i < n and i != kstar-i)
`grad_ac_kstar[i] = a[i]` (if i == kstar-i, i.e., kstar = 2*i)
`grad_ac_kstar[i] = 0` (otherwise)

Then the gradient of the full objective (with normalization) is:
`grad[i] = grad_ac_kstar[i] / sum(a) - 2 * ac[k*] / (sum(a))^2` (up to a constant factor)

```python
def subgradient_step(a, learning_rate):
    n = len(a)
    ac = auto_conv(a)
    k_star = int(np.argmax(ac))
    s = np.sum(a)
    max_ac = ac[k_star]
    
    # Gradient of max_ac w.r.t. a
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = a[i]
            else:
                grad[i] = 2 * a[j]
    
    # Gradient of score = 2n * max_ac / s^2
    # d(score)/da_i = 2n * (grad_i / s^2 - 2*max_ac / s^3)
    grad_score = 2 * n * (grad / (s * s) - 2 * max_ac / (s ** 3))
    
    # Projected gradient step
    a_new = a - learning_rate * grad_score
    a_new = np.clip(a_new, 0, None)
    a_new = normalize(a_new)
    
    return a_new, max_ac
```
