"""
Minimize  2*n*max(conv(a,a)) / (sum a)^2  for a non-negative step function.

The objective is scale invariant, so internally every candidate is normalized
to sum == 1 and the score being minimized is exactly 2*n*max(conv(a,a)).

Method per starting point:
  * FFT-based full self-convolution for fast evaluation.
  * Subgradient descent on the (top-K, log-sum-exp softened) maximum of the
    autocorrelation, with heavy ball momentum.
  * Projection onto the non-negative probability simplex after each step.
  * Adam-ish adaptive step + annealing on the softening parameter.
  * Small random jitter to escape flat basins.
Multiple lengths / structured initializations are tried; the best exact
(evaluate_sequence) score wins.  A valid fallback is always returned before
the time budget expires.
"""

import time
import numpy as np

# ---------------------------------------------------------------------------
# Top-level helpers (no closures, no lambdas)
# ---------------------------------------------------------------------------

def fast_self_conv(a):
    """Full linear self-convolution of a 1-D float array via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0)
    out_len = 2 * n - 1
    fft_len = 1
    while fft_len < out_len:
        fft_len <<= 1
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:out_len]


def project_simplex(v):
    """Project a 1-D vector onto {x >= 0, sum(x) == 1} (water filling)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if np.sum(v) <= 0.0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0][-1]
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(c, a, top_idx, weights):
    """Weighted subgradient of max_lag c[lag] w.r.t. a.

    For a single active lag k the gradient is g[j] = 2*a[k-j] for valid j.
    """
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for lag, w in zip(top_idx, weights):
        if w == 0.0:
            continue
        j_min = max(0, lag - (n - 1))
        j_max = min(lag, n - 1)
        if j_min > j_max:
            continue
        j = np.arange(j_min, j_max + 1)
        idx = lag - j
        g[j] += (2.0 * w) * a[idx]
    return g


def make_starts(seed):
    """Return a list of (description, array) starting points, big lengths first."""
    rng = np.random.default_rng(seed)
    starts = []

    # Provided known construction if available.

    except Exception:
        pass

    # Structured / uniform initializations, largest length first.
    for n in (8192, 4096, 2048, 1024):
        starts.append(("uniform-%d" % n, np.ones(n, dtype=np.float64)))

    # U-shaped (high at edges, low in middle) heuristic.
    n = 4096

    u = 0.15 + 0.85 * np.clip(2.0 * np.abs(x - 0.5), 0.0, 1.0)

    # A couple of random smooth initializations for diversity.

        base = 1.0 + np.cumsum(np.cumsum(rng.normal(0.0, 0.4, n)))
        base = base - base.min() + 0.05
        starts.append(("rand-%d" % n, base))

    return starts


def optimize_from(a0, tag, rng, deadline):
    """Run momentum subgradient descent from a normalized start; return best."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.ones_like(a)
        s = a.sum()
    a = a / s
    n = a.size

    c = fast_self_conv(a)
    best_obj = 2.0 * n * float(c.max())
    best_a = a.copy()

    mom = np.zeros(n, dtype=np.float64)
    beta = 0.85
    step = 0.05
    tau = 25.0          # softmax temperature on the top peaks
    k_top = 12
    it = 0
    base_step = step

    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        obj = 2.0 * n * float(c.max())
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        it += 1
        # Adaptive step, anneal softening slightly over time.
        step = base_step / (1.0 + it * 1e-4)
        tau = max(12.0, 40.0 - it * 1e-3)

        # Top-K peaks.
        kk = min(k_top + it // 4000, 64, c.size)
        top_idx = np.argpartition(c, -kk)[-kk:]
        cv = c[top_idx]
        z = tau * (cv - cv.max())
        ew = np.exp(z)
        weights = ew / ew.sum()

        g = compute_grad(c, a, top_idx, weights)

        # Heavy-ball momentum + update + projection.
        mom = beta * mom + g
        a = a - step * mom

        # Occasional small jitter to leave flat basins.
        if it % 250 == 0 and rng.random() < 0.3:
            a = a + 0.01 * rng.standard_normal(n)

        a = np.clip(a, 0.0, None)
        a = project_simplex(a)

    # Final measurement.
    c = fast_self_conv(a)
    obj = 2.0 * n * float(c.max())
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the best non-negative coefficient sequence.

    Returns a list of non-negative floats. Always returns a valid sequence
    before the time budget is exhausted.
    """
    budget_s = float(budget_s)
    start_wall = time.monotonic()
    deadline = start_wall + max(1.0, budget_s) - 8.0   # safety margin

    rng = np.random.default_rng(seed)
    starts = make_starts(seed)

    best_seq = None
    best_score = np.inf

    # Fallback valid sequence (normalized so evaluator returns a finite value).
    fallback = np.ones(512, dtype=np.float64)
    fallback_score = float(evaluate_sequence(fallback.tolist()))
    if np.isfinite(fallback_score):
        best_seq = fallback
        best_score = fallback_score

    n_starts = max(1, len(starts))
    for i, (tag, a0) in enumerate(starts):
        remaining = deadline - time.monotonic()
        if remaining <= 5.0:
            break
        # Give each start a share of the remaining time.
        per_start = max(15.0, remaining / float(n_starts - i))
        sub_deadline = time.monotonic() + per_start

        if time.monotonic() >= sub_deadline:
            sub_deadline = time.monotonic() + 1.0
        a_opt, obj = optimize_from(a0, tag, rng, sub_deadline)

        # Confirm with the authoritative evaluator on the top candidates.
        cand = np.clip(a_opt, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        exact = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(exact) and exact < best_score:
            best_score = exact
            best_seq = cand.copy()
            print("[%s] n=%d  obj=%.6f  exact=%.6f  elapsed=%.1fs"
                  % (tag, cand.size, obj, exact, time.monotonic() - start_wall))

    # Always leave a valid answer.
    if best_seq is None:
        best_seq = fallback
    best_seq = np.asarray(best_seq, dtype=np.float64).ravel()
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.ones(512, dtype=np.float64)
    print("FINAL best=%.6f n=%d elapsed=%.1fs"
          % (best_score, best_seq.size, time.monotonic() - start_wall))
    return [float(x) for x in best_seq]
