(the active-set spirit of the state of the art, but self-contained and
robust).  At each step we (i) compute the full convolution, (ii) take the
sub-gradient contributed by every lag that sits within a small tolerance
of the maximum (this flattens the whole top plateau, not just one spike),
(iii) step against it, (iv) project back to the non-negative face and
re-normalize the sum, and (v) line-search / simulated-anneal the step size.
We run several structured seeds (the provided construction, a two-level
block, a ramp, a trapezoid and a random one) in parallel round-robin and
return the overall best sequence verified by evaluate_sequence.
"""

import time
import numpy as np

# ------------------------------------------------------------------
# low level helpers (top level, no closures / lambdas)
# ------------------------------------------------------------------

def _safe_length(base):
    """Return a sane target length from the provided construction."""
    try:
        arr = np.asarray(base, dtype=np.float64)
        if arr.ndim == 1 and arr.size >= 4 and np.isfinite(arr).all():
            L = int(arr.size)
            if 200 <= L <= 9000:
                return L
    except Exception:
        pass
    return 3000


def _normalize(a, S):
    """Clip to >=0 and rescale so the sum is exactly S."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    s = float(a.sum())
    if s <= 1e-12:
        return np.full_like(a, S / a.size, dtype=np.float64)
    return a * (S / s)


def _peak_and_active(a, tol):
    """Return (max_conv, list_of_active_lags, full_conv)."""
    conv = np.convolve(a, a)
    m = float(conv.max())
    thr = m * (1.0 - tol)
    active = np.where(conv >= thr)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(conv))])
    return m, active.tolist(), conv


def _lag_subgradient(a, k):
    """Sub-gradient vector  g[j] = d (a*a)[k] / d a[j]  (zero padded)."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    # v1[j] = a[k-j]   for 0 <= k-j < n
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if hi >= lo:
        x = np.arange(lo, hi + 1)
        g[x] += a[k - x]
    # v2[j] = a[j-k]   for 0 <= j-k < n
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if hi2 >= lo2:
        x2 = np.arange(lo2, hi2 + 1)
        g[x2] += a[x2 - k]
    return g


def _step(a, active, eta, S):
    """One projected sub-gradient step (chase the plateau)."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        g += _lag_subgradient(a, k)
    g /= float(active.size)
    cand = a - eta * g
    return _normalize(cand, S)


def _local_search(a0, S, deadline, rng, tol=0.004, base_eta=0.02,
                  block_every=40, block_prob=0.5):
    """Projected sub-gradient local search with annealed step + random kicks.

    Returns (best_a, best_obj) where best_obj == max_k (a*a)[k] with
    sum(a)=S  (== the score).
    """
    a = _normalize(a0.copy(), S)
    n = a.size
    best_a = a.copy()
    best_obj, active0, _ = _peak_and_active(a, tol)
    best_obj = float(best_a_obj if False else best_obj)  # placeholder removed below
    best_obj, active0, _ = _peak_and_active(a, tol)
    obj = best_obj

    T = 0.5 * obj          # annealing temperature (in obj units)
    eta = base_eta
    it = 0
    impat = 0

    while time.monotonic() < deadline:
        # ---- compute current plateau ----
        obj, active, _ = _peak_and_active(a, tol)

        # ---- direction ----
        direction = None
        if len(active) >= 1:
            g = np.zeros(n, dtype=np.float64)
            for k in active:
                g += _lag_subgradient(a, k)
            g /= float(len(active))
            direction = g

        # ---- occasional random block kick (escape) ----
        if block_every and (it % block_every == 0) and rng.random() < block_prob:
            w = int(np.clip(rng.integers(max(2, n // 40), max(3, n // 6)), 2, n))
            st = int(rng.integers(0, max(1, n - w + 1)))
            kick = a.copy()
            kick[st:st + w] *= float(np.exp(rng.normal(0.0, 0.18)))
            # blend so we still move mostly against the plateau
            if direction is not None:
                kk = _normalize(kick, S)
                kk = kk - 0.5 * eta * direction
                a = _normalize(kk, S)
            else:
                a = _normalize(kick, S)
            it += 1
            continue

        # ---- line search over a few step sizes ----
        improved = False
        for scale in (1.0, 0.5, 0.25, 0.1, 0.05, 1.5, 2.0):
            h = eta * scale
            cand = _step(a, active, h, S)
            c_obj, c_active, _ = _peak_and_active(cand, tol)
            # annealed accept
            if c_obj < obj - 1e-12 or rng.random() < np.exp(-(c_obj - obj) / max(T, 1e-9)):
                a = cand
                obj = c_obj
                active = c_active
                improved = True
                break

        if improved:
            eta = min(0.05, eta * 1.02)
            if obj < best_obj - 1e-9:
                best_obj = obj
                best_a = a.copy()
                impat = 0
                T = max(1e-4 * obj, T * 0.97)
            impat += 1
        else:
            eta = max(0.002, eta * 0.5)
            T = max(1e-4 * obj, T * 0.90)
            if impat > 250:
                # stuck: reheat with a structured re-seed mix
                a = _normalize(0.5 * a + 0.5 * a0, S)
                eta = base_eta
                impat = 0

        it += 1

    # final exact recompute
    fo, _, _ = _peak_and_active(best_a, 0.0)
    return best_a, float(fo)


def _seed_block(n, S):
    a = np.empty(n, dtype=np.float64)
    a[: n // 2] = 1.0
    a[n // 2:] = 2.0
    return _normalize(a, S)


def _seed_ramp(n, S):
    a = np.arange(1, n + 1, dtype=np.float64)
    return _normalize(a, S)


def _seed_trap(n, S):
    r = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(r, n + 1 - r)          # tent (symmetric)
    a = np.maximum(a, 0.3)                # keep a floor to avoid dead zone
    return _normalize(a, S)


def _seed_random(n, S, rng):
    a = rng.random(n) * 0.5 + 0.5
    return _normalize(a, S)

# main entry point
# ------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for the lowest-score non-negative step-function height list."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(10.0, float(budget_s) - 4.0)   # keep a safety margin

    n = _safe_length(height_sequence_1)
    S = float(np.sqrt(2.0 * n))

    # build seeds
    base = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size == n and np.isfinite(arr).all():
            base = _normalize(arr, S)
    except Exception:
        base = None
    if base is None:
        base = _seed_block(n, S)

    seeds = []
    seeds.append(("base", base))
    seeds.append(("block", _seed_block(n, S)))
    seeds.append(("ramp", _seed_ramp(n, S)))
    seeds.append(("trap", _seed_trap(n, S)))
    seeds.append(("rand", _seed_random(n, S, rng)))

    # give each seed a short refinement budget, keep the best per seed,
    # then pour the remaining time into the single best seed (deep polish).
    per_seed = max(8.0, min(60.0, (float(budget_s) - 30.0) / max(1, len(seeds) + 2)))
    results = {}
    for name, s in seeds:
        if time.monotonic() >= deadline - per_seed:
            break
        d = min(time.monotonic() + per_seed, deadline - 2.0)
        a_best, obj_best = _local_search(s, S, d, rng)
        results[name] = (a_best, obj_best)
        print("[seed %-5s] obj=%.6f" % (name, obj_best), flush=True)

    # deep polish the best seed
    if results:
        best_name = min(results, key=lambda k: results[k][1])
        a_best, obj_best = results[best_name]
        print("[best seed] %s  obj=%.6f  (t=%.1fs)" %
              (best_name, obj_best, time.monotonic() - start), flush=True)
        a_pol, obj_pol = _local_search(
            a_best, S, deadline, rng,
            tol=0.002, base_eta=0.01, block_every=60, block_prob=0.3)
        if obj_pol < obj_best:
            a_best, obj_best, best_name = a_pol, obj_pol, best_name + "!"
            print("[polish %.5s] obj=%.6f  (t=%.1fs)" %
                  (best_name, obj_pol, time.monotonic() - start), flush=True)

    # ---- sanity: ensure a valid return; verify with the real evaluator ----
    out = [float(x) for x in np.clip(np.asarray(a_best, dtype=np.float64), 0.0, 1000.0)]
    try:
        sc = float(evaluate_sequence(out))
        print("[final] internal_obj=%.6f  evaluate_sequence=%.6f  n=%d" %
              (obj_best, sc, n), flush=True)
        if not np.isfinite(sc):
            out = [1.0] * n          # guaranteed-valid fallback
    except Exception as e:
        print("[final] evaluator error %r, returning internal best" % (e,), flush=True)

    return out
