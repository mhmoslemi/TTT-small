"""
Hybrid optimization for minimizing 2n*max(f*f)/(sum f)^2 over non-negative sequences.
Uses subgradient descent on the peak autocorrelation, LP-based refinement,
and multiple random restarts across different sequence lengths.
Key insight: asymmetric sequences can break the J>=2 barrier for symmetric ones.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_conv(f):
    """Compute full self-convolution."""
    return np.convolve(f, f)


def compute_obj(f):
    """Compute objective 2n*max(f*f)/(sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    M = np.convolve(f, f).max()
    return 2.0 * n * M / (s * s)


def subgradient_step_vec(f, conv, n_active=30):
    """Compute subgradient vector using top lags of convolution. Returns (v, M)."""
    n = len(f)
    M = conv.max()
    # Find active lags
    if n_active >= len(conv):
        active = np.arange(len(conv))
    else:
        active = np.argsort(conv)[-n_active:]
    # Compute subgradient: v[i] = mean over active k of f[k-i] (if valid)
    v = np.zeros(n)
    for k_idx in active:
        k_idx = int(k_idx)
        i_start = max(0, k_idx - n + 1)
        i_end = min(n, k_idx + 1)
        # v[i] += f[k_idx - i] for i in [i_start, i_end)
        # f[k_idx - i] for i = i_start..i_end-1 gives f[k_idx-i_start]..f[k_idx-(i_end-1)]
        # which is f[k_idx-i_start : k_idx-i_end+1] reversed
        j_start = k_idx - (i_end - 1)
        j_end = k_idx - i_start + 1
        v[i_start:i_end] += f[j_start:j_end][::-1]
    v /= len(active)
    return v, M


def optimize_subgradient(f_init, max_time, rng, verbose=False):
    """Subgradient descent to minimize max(f*f) subject to sum(f)=1, f>=0.
    Returns (best_f, best_M)."""
    f = np.maximum(0, np.asarray(f_init, dtype=np.float64)).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f_init))
        s = f.sum()
    f = f / s
    n = len(f)

    conv = compute_conv(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M

    start = time.monotonic()
    step = 0
    base_alpha = 0.05
    no_improve = 0
    patience = 2000

    while time.monotonic() - start < max_time:
        t0 = time.monotonic()

        conv = compute_conv(f)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Restart with perturbation
            f = best_f.copy()
            pert = rng.random(n) * 0.05
            f = f + pert
            f = np.maximum(0, f)
            f = f / f.sum()
            no_improve = 0

        v, M = subgradient_step_vec(f, conv, n_active=min(50, len(conv)))

        # Project onto sum=0 hyperplane
        v = v - v.mean()

        # Step size: decay with schedule, reset on restart
        alpha = base_alpha / (1.0 + 0.001 * no_improve)

        f_new = f - alpha * v
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f = best_f + rng.random(n) * 0.01
            f = np.maximum(0, f)
            f = f / f.sum()
        else:
            f = f_new / s_new

        step += 1
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad: step={step}, M={M:.8f}, best={best_M:.8f}, "
                  f"time/step={elapsed*1000:.2f}ms, elapsed={time.monotonic()-start:.1f}s")

    return best_f, best_M


def optimize_lp(f_init, max_time, rng, verbose=False):
    """LP-based optimization: alternate between f and LP solution g.
    Returns (best_f, best_M)."""
    f = np.maximum(0, np.asarray(f_init, dtype=np.float64)).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f_init))
        s = f.sum()
    f = f / s
    n = len(f)

    best_f = f.copy()
    best_M = compute_conv(f).max()

    start = time.monotonic()
    t_mix = 0.3
    n_smooth = 0

    while time.monotonic() - start < max_time:
        conv = compute_conv(f)
        M = conv.max()

        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            n_smooth += 1
        else:
            n_smooth = 0
            if n_smooth == 0 and rng.random() < 0.1:
                # Perturb to escape
                f = best_f + rng.standard_normal(n) * 0.01
                f = np.maximum(0, f)
                f = f / f.sum()

        # Select active constraints
        n_act = min(50, len(conv))
        active = np.argsort(conv)[-n_act:]

        # Build LP: max sum(g) s.t. A g <= M, g >= 0
        c = -np.ones(n)
        A_ub = np.zeros((len(active), n))
        for row, k_idx in enumerate(active):
            k_idx = int(k_idx)
            i_start = max(0, k_idx - n + 1)
            i_end = min(n, k_idx + 1)
            j_start = k_idx - (i_end - 1)
            j_end = k_idx - i_start + 1
            A_ub[row, j_start:j_end] = f[i_start:i_end][::-1]

        b_ub = np.full(len(active), M)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except Exception:
            res = None

        if res is None or not res.success:
            t_mix *= 0.9
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Mix
        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = np.maximum(0, f_new)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f = f_new / s_new

        # Anneal mixing
        t_mix = max(0.05, t_mix * 0.995)

        if verbose and time.monotonic() - start > 5:
            print(f"    LP: M={M:.8f}, best={best_M:.8f}, g_sum={g_sum:.4f}, "
                  f"t_mix={t_mix:.4f}, elapsed={time.monotonic()-start:.1f}s")

    return best_f, best_M


def small_n_brute_force(n, rng, max_time):
    """For very small n, try to find good sequences via random search + local opt."""
    start = time.monotonic()
    best_f = None
    best_M = float('inf')

    for _ in range(100):
        if time.monotonic() - start > max_time:
            break
        # Random initialization
        if rng.random() < 0.5:
            f = rng.dirichlet(np.ones(n))
        else:
            f = np.abs(rng.standard_normal(n)) + 0.01
            f = f / f.sum()

        # Local subgradient refinement
        f_ref, M_ref = optimize_subgradient(f, max_time=min(2.0, max_time / 50),
                                            rng=rng, verbose=False)
        if M_ref < best_M:
            best_M = M_ref
            best_f = f_ref.copy()

    if best_f is not None:
        return best_f, best_M
    return None, float('inf')
