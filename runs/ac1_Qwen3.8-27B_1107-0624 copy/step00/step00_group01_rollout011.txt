import numpy as np
import cvxpy as cp
from scipy.sparse import lil_matrix, csr_matrix
import time

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix C where C @ b = convolve(f, b)."""
    n = len(f)
    C = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k-n+1)
        hi = min(k, n-1)
        for j in range(lo, hi+1):
            C[k, j] = f[k-j]
    return C.tocsr()

def solve_lp_direction(f, M, solver=cp.GLOP):
    """Solve LP to find direction b that reduces max convolution."""
    n = len(f)

    try:
