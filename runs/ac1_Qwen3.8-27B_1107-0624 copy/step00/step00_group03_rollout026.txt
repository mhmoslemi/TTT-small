Minimizes 2n*max(conv(a,a))/(sum(a))^2 using momentum gradient descent
on smooth Lp surrogates of the self-convolution, with non-negativity
projection, FFT-accelerated operations, adaptive learning rates, and
multiple restart strategies to escape local minima.

def _fft_conv(a):
    """Compute a*a (full linear convolution) via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _fft_corv(x, y):
    """Compute np.correlate(x, y, mode='valid') via FFT. Requires len(x) >= len(y)."""
    nx, ny = len(x), len(y)
    size = 1 << (nx + ny - 2).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y[::-1], size)
    r = np.fft.irfft(fx * fy, size)
    return r[ny - 1:ny - 1 + nx - ny + 1]

def _fast_score(a):
    """Compute the objective value using FFT (fast, not exact)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return 1e30
    b = _fft_conv(a)
    return float(2.0 * n * b.max() / (s * s))

    else:
