import numpy as np

def compute_smooth_gradient(a, conv_vals, alpha):
    """
    Compute the gradient of the soft-max smoothed objective.
    F = (1/alpha) * log(sum(exp(alpha * conv_k)))
    grad_j = (1/Z) * sum_k exp(alpha*conv_k) * 2*a[k-j] (valid k,j)
    """

    # exp(alpha * conv_vals) might overflow, so subtract max
    shifted = alpha * conv_vals - alpha * np.max(conv_vals)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    w = exp_vals / Z  # w has length 2n-1, sums to 1
    
    # grad[j] = 2 * sum_m w[j+m] * a[m] for j=0..n-1, m=0..n-1
    # Since j+m ranges 0..2n-2 which is valid for w (length 2n-1)
    from numpy.lib.stride_tricks import as_strided
    stride = w.strides[0]
    M = as_strided(w, shape=(n, n), strides=(stride, stride))
    grad = 2.0 * (M @ a)
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed subgradient descent
with soft-max approximation, FFT-based convolution, Hankel-matrix gradient
computation, heavy-ball momentum, adaptive scheduling, and multiple restarts.
"""

import numpy as np
import time


def _fft_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)
    return conv[:2 * n - 1]


def _compute_gradient(a, w):
    """
    Compute gradient of smooth max objective w.r.t. a.
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    Using Hankel matrix formed from w via stride tricks.
    """
    n = len(a)
    from numpy.lib.stride_tricks import as_strided
    stride = w.strides[0]
    M = as_strided(w, shape=(n, n), strides=(stride, stride))
    return 2.0 * (M @ a)


def _score_fast(a):
    """Quick score without calling evaluate_sequence (assumes sum(a) > 0)."""
    n = len(a)
    conv_vals = _fft_convolve(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * np.max(conv_vals) / (s * s)


def _random_init(n, rng, variant):
    """Generate diverse random non-negative initializations."""
    if variant == 0:

        n_freqs = rng.integers(3, 8)

        phases = rng.uniform(0, 2 * np.pi, size=n_freqs)
        amps = rng.uniform(0.3, 1.0, size=n_freqs)
        x = np.arange(n, dtype=np.float64)
        a = np.zeros(n)
        for f, p, am in zip(freqs, phases, amps):
            a += am * np.cos(2 * np.pi * f * x / n + p)
        a = np.abs(a) + 0.3
    elif variant == 1:

        a += 0.05 * np.abs(rng.standard_normal(n))
    elif variant == 2:

        a = np.cumsum(steps)
        a = a - a.min() + 0.5
    elif variant == 3:

        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + np.exp(-0.5 * ((x - c2) / w2) ** 2) + 0.1
    elif variant == 4:
        # Chirp-like

    elif variant == 5:

        a = a + 0.2
    else:

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) * 0.5
        s = n * 0.5
    a = a * n / s
    return a


def _optimize(a, deadline, rng, n, tag="opt", alpha_start=10.0, max_time=None):
    """
    Smooth gradient descent with momentum for minimizing max(conv(a,a))/(sum(a))^2.
    Normalizes to sum(a) = n throughout.
    """
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a * n / s

    best_local = a.copy()
    best_local_score = _score_fast(best_local)
    print(f"  [{tag}] start_score = {best_local_score:.6f}")

    momentum = np.zeros_like(a)
    beta = 0.9
    alpha = alpha_start
    lr = 0.02
    step = 0
    noise_count = 0
    no_improve_count = 0
    last_best = best_local_score

    # Pre-allocate
    L = 1
    while L < 2 * n - 1:
        L *= 2

    while time.monotonic() < deadline:
        # Compute convolution via FFT
        A = np.fft.rfft(a, L)
        conv_vals = np.fft.irfft(A * A, L)[:2 * n - 1]

        max_c = np.max(conv_vals)
        score = 2.0 * max_c / n  # since sum(a) = n

        if score < best_local_score - 1e-8:
            best_local_score = score
            best_local = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Compute soft-max weights (numerically stable)
        shifted = alpha * (conv_vals - max_c)
        exp_vals = np.exp(shifted)
        Z = np.sum(exp_vals)
        w = exp_vals / Z

        # Compute gradient
        grad = _compute_gradient(a, w)

        # Clamp gradient to avoid numerical issues
        grad_max = np.max(np.abs(grad))
        if grad_max > 100:
            grad = grad / grad_max * 100

        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        a = a - lr * momentum
        a = np.maximum(a, 0.0)

        # Renormalize to sum = n
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n)
            s = n
        a = a * n / s

        step += 1

        # Adaptive scheduling

            alpha *= 1.03
            lr *= 0.997

        # If stuck, increase noise or reset momentum

            momentum *= 0.1
            no_improve_count = 0
