def build_conv_matrix(f, n):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[j] * b[k-j] contributes to output position k = j + m for m = 0,...,n-1
        # So A[j + m, j] = f[j] for m = 0,...,n-1
        # i.e., A[j:j+n, j] = f[j]
        A[j:j + n, j] = f[j]
    return A

def objective(s):
    """Compute the evaluation score (without calling evaluate_sequence)."""
    n = len(s)
    conv = np.convolve(s, s)
    max_conv = np.max(conv)
    sum_s = np.sum(s)
    if sum_s < 0.01:
        return np.inf
    return float(2 * n * max_conv / (sum_s ** 2))

def lp_frank_wolfe_step(f, n):
    """Solve the LP to find the Frank-Wolfe direction."""
    M = np.max(np.convolve(f, f))
    if M <= 0:
        return f.copy()
    
    A = build_conv_matrix(f, n)
    
    # LP: max ones^T b s.t. A @ b <= M * ones(2n-1), b >= 0
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = A
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return f.copy()
    
    b = res.x
    sum_b = np.sum(b)
    if sum_b < 1e-10:
        return f.copy()

    g = b / sum_b
    return g

```python
def project_simplex(v, s=1):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def projected_gradient_descent(t, n, steps=10000):
    """Iterative peak shaving using projected subgradient method."""
    t = t.copy()
    t = t / t.sum()
    
    for step in range(steps):
        conv = np.convolve(t, t)
        k_star = np.argmax(conv)
        current_val = conv[k_star]
        
        # Subgradient: sg[i] = t[k_star - i] + t[i + k_star] (with boundaries)
        sg = np.zeros(n)
        for i in range(n):
            val = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                val += t[j1]
            j2 = i + k_star
            if 0 <= j2 < n:
                val += t[j2]
            sg[i] = 2.0 * val
        
        # Step size
        step_size = 1.0 / (step + 1)
        
        # Gradient step
        new_t = t - step_size * sg
        new_t = np.maximum(new_t, 0)
        s = new_t.sum()
        if s > 0:
            new_t = new_t / s
        else:
            new_t = t
        
        # Accept if better
        new_conv = np.convolve(new_t, new_t)
        if np.max(new_conv) < current_val:
            t = new_t
    
    return t
```

def compute_subgradient(t, k_star, n):
    sg = np.zeros(n)
    # Term 1: t[k* - i] for i where 0 <= k*-i < n
    i_lo1 = max(0, k_star - n + 1)
    i_hi1 = min(n - 1, k_star)
    indices1 = np.arange(i_lo1, i_hi1 + 1)
    sg[indices1] += t[k_star - indices1]
    
    # Term 2: t[i + k*] for i where 0 <= i+k* < n
    i_lo2 = 0
    i_hi2 = min(n - 1, n - 1 - k_star)
    if i_hi2 >= i_lo2:
        indices2 = np.arange(i_lo2, i_hi2 + 1)
        sg[indices2] += t[indices2 + k_star]
    
    return 2.0 * sg
```

Now, for the projected gradient method, I need to handle the projection onto the simplex {t ≥ 0, sum(t) = 1}.

The projection is: given v, find p = argmin ||p - v||^2 s.t. p ≥ 0, sum(p) = 1.

This can be done in O(n log n) using the following algorithm:
1. Sort v in descending order: v_(1) ≥ v_(2) ≥ ... ≥ v_(n)
2. Find the largest ρ such that v_(ρ) - (sum_{i=1}^{ρ} v_(i) - 1)/ρ > 0
3. θ = (sum_{i=1}^{ρ} v_(i) - 1)/ρ
4. p_i = max(v_i - θ, 0)

Let me implement this:

```python
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

import numpy as np

from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_topk(t, top_k, n):
    """Compute averaged subgradient over the top-k peak lags of t*t."""
    conv = np.convolve(t, t)
    if top_k < len(conv):
        top_lags = np.argsort(conv)[::-1][:top_k]
    else:
        top_lags = np.arange(len(conv))
    sg = np.zeros(n)
    for k_star in top_lags:
        i_lo1 = max(0, k_star - n + 1)
        i_hi1 = min(n - 1, k_star)
        if i_hi1 >= i_lo1:
            idx1 = np.arange(i_lo1, i_hi1 + 1)
            sg[idx1] += t[k_star - idx1]
        i_hi2 = min(n - 1, n - 1 - k_star)
        if i_hi2 >= 0:
            idx2 = np.arange(0, i_hi2 + 1)
            sg[idx2] += t[idx2 + k_star]
    sg *= 2.0
    sg /= len(top_lags)
    return sg, conv, top_lags

def eval_score(s):
    """Fast evaluation of the objective (no safety checks)."""
    n = len(s)
    conv = np.convolve(s, s)
    max_c = np.max(conv)
    s_sum = np.sum(s)
    if s_sum < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s_sum * s_sum)

def build_conv_matrix(f, n):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f[j]
    return A

def solve_lp_direction(f, n, rng):
    """Solve LP: max sum(b) s.t. A_f @ b <= M, b >= 0. Return normalized b or None."""
    conv_f = np.convolve(f, f)
    M = np.max(conv_f)
    if M <= 0:
        return None
    A = build_conv_matrix(f, n)
    c = -np.ones(n)
    c += rng.random(n) * 1e-6  # tiny perturbation to break ties
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    b = res.x
    sb = np.sum(b)
    if sb < 1e-10:
        return None
    return b / sb
