def optimize_coordinate(a, j, n, S):
    """Optimize a[j] while keeping others fixed. Returns optimal a[j]."""
    # Compute c_k, p_k, q_k
    b = a.copy()
    b[j] = 0
    bb = np.convolve(b, b)  # (b*b)_k
    S = b.sum()  # sum of all except a[j]
    
    # p_k and q_k
    # p_k = 2*a_{k-j} if 0<=k-j<n and k!=2j
    # q_k = 1 if k==2j
    p = np.zeros(2*n-1)
    q = np.zeros(2*n-1)
    # k-j ranges from 0 to n-1 when k ranges from j to j+n-1
    k_lo = j
    k_hi = j + n - 1
    # p[k] = 2*a[k-j] for k in [j, j+n-1], k != 2j
    for k in range(k_lo, k_hi+1):
        idx = k - j
        if k == 2*j:
            q[k] = 1.0
        else:
            p[k] = 2.0 * a[idx]
    
    # Actually, p[k] = 2*a[k-j] for k in [j, j+n-1], k != 2j
    # We can vectorize:
    p[k_lo:k_hi+1] = 2.0 * a[0:n]
    p[2*j] = 0.0
    q[2*j] = 1.0
    
    # Now minimize max_k (bb[k] + p[k]*x + q[k]*x^2) / (S + x)^2 over x >= 0
    # This is a 1D optimization
    
    def objective(x):
        vals = bb + p * x + q * x**2
        return np.max(vals) / (S + x)**2
