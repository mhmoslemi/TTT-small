"""
LP-based iterative optimization for minimum auto-convolution peak step function.
Uses the Picard-type iteration: fix f, solve LP for g that maximizes sum(g)
subject to f*conv(g) <= max(f*conv(f)), then blend f and g.
Combined with gradient polishing and multiple initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_A_sparse(f):
    """Build sparse matrix A where A[m,j] = f[m-j] if 0<=m-j<n else 0.
    Shape: (2n-1, n). Each column has exactly n non-zeros (a copy of f placed at different rows).
    """
    n = len(f)
    total = n * n
    rows = np.empty(total, dtype=np.int64)
    cols = np.empty(total, dtype=np.int64)
    vals = np.empty(total, dtype=np.float64)
    
    for j in range(n):
        s = j * n
        rows[s:s+n] = np.arange(j, j+n)
        cols[s:s+n] = j
        vals[s:s+n] = f
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_solve(f, M):
    """Solve: max sum(g) s.t. A(f) @ g <= M, g >= 0."""
    n = len(f)
    A = build_A_sparse(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=np.zeros((n, 2)) + [0, np.inf], method='highs')
    if res.status != 0:
        return None
    return res.x


def objective(f):
    """Compute the score (lower is better)."""
    n = len(f)
    conv_f = np.convolve(f, f)
    sum_f = f.sum()
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * conv_f.max() / (sum_f ** 2)


def lp_iteration(f, t=1.0):
    """One LP-based iteration. Returns updated f or None if no improvement."""
    n = len(f)
    f = f / f.sum()  # normalize
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    
    g = lp_solve(f, M)
    if g is None:
        return None
    
    sum_g = g.sum()
    if sum_g <= f.sum():
        return None
    
    g_norm = g / g.sum()
    
    # Try full step first, then reduce
    for step in [t, t*0.5, t*0.25, 0.1, 0.01]:

        f_new = np.clip(f_new, 0, None)
        if f_new.sum() < 1e-10:

        f_new = f_new / f_new.sum()
        obj_new = objective(f_new)
        obj_old = objective(f)
        if obj_new < obj_old:
            return f_new
    
    return None

LP-based iterative optimization for minimum auto-convolution peak step function.
Uses Picard-type iteration with LP subproblems, combined with gradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def build_A_sparse(f):
    n = len(f)
    total = n * n
    rows = np.empty(total, dtype=np.int64)
    cols = np.empty(total, dtype=np.int64)
    vals = np.empty(total, dtype=np.float64)
    for j in range(n):
        s = j * n
        rows[s:s+n] = np.arange(j, j+n)
        cols[s:s+n] = j
        vals[s:s+n] = f
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def lp_solve(f, M, timeout_hint=None):
    n = len(f)
    A = build_A_sparse(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def compute_score(f):
    n = len(f)
    conv_f = np.convolve(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(conv_f.max()) / (s * s)


def lp_iterate(f, max_iter=100, verbose=True):
    """Run LP-based iterations. Returns best f found."""
    f = f.copy()
    f = np.clip(f, 0, None)
    if f.sum() < 1e-10:
        return f
    f = f / f.sum()
    
    best_obj = compute_score(f)
    best_f = f.copy()
    n = len(f)
    
    for it in range(max_iter):
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())
        
        g = lp_solve(f, M)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {it}")
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1e-10:
            break
        
        g_norm = g / sum_g
        
        # Try different step sizes
        improved = False
        for t in [1.0, 0.5, 0.3, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g_norm
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            obj = compute_score(f_new)
            if obj < best_obj:
                f = f_new
                best_obj = obj
                best_f = f.copy()
                improved = True
                if verbose and it % 10 == 0:

                break
        
        if not improved:
            if verbose:

            break
    
    return best_f

            k_max = min(n, 2*n-1-j)

T1[j] = sum_k f[k] * r[m-k] = (f * r)[m] = (f * q[::-1])[2n-2-j]

So T1 = conv(f, q[::-1])[2n-2 : 2n-2-(n-1) : -1] = conv(f, q[::-1])[n-1 : 2n-2] (reversed)

Wait, let me be more careful. (f * r)[m] where * is linear convolution, f has length n, r has length 2n-1. The result has length n + (2n-1) - 1 = 3n-2.

T1[j] = (f * r)[m] where m = 2n-2-j.
For j=0: m = 2n-2
For j=n-1: m = n-1

So T1[j] = conv(f, q[::-1])[2n-2-j] for j = 0, ..., n-1.

Second term: T2[j] = sum_k f[k] * p[j-k] for valid k, j-k.
p[j-k] is valid when j-k in [0, 2n-2] and k in [0, n-1].
So k in [max(0, j-(2n-2)), min(n-1, j)].

T2[j] = sum_k f[k] * p[j-k] = (f * p_flip)[j] where p_flip = p (no flip needed, this is standard cross-correlation)

Actually: T2[j] = sum_k f[k] * p[j-k] = (f ⊛ p)[j] where ⊛ is cross-correlation at lag j.
This equals conv(f, p[::-1])[j + (2n-2)] = conv(f, p[::-1])[j + 2n - 2]

Hmm, let me use a different approach. Define h = p (length 2n-1).
T2[j] = sum_k f[k] * h[j-k]

This is the standard convolution (f * h)[j] if we think of it as:
(f * h)[j] = sum_k f[k] * h[j-k]

Yes! So T2 = conv(f, h) = conv(f, p), and we take indices j = 0, ..., n-1.

Wait, but conv(f, p) has length n + (2n-1) - 1 = 3n-2. We want indices 0 to n-1.

So:
T2[j] = np.convolve(f, p)[j] for j = 0, ..., n-1

And for T1:
T1[j] = np.convolve(f, p[::-1])[2n-2-j] for j = 0, ..., n-1

Or equivalently: T1 = np.convolve(f, p[::-1])[n-1:2n-2+1][::-1]

Let me verify with a small example.

f = [a, b, c] (n=3), p = [p0, p1, p2, p3, p4] (length 2n-1=5)

T1[0] = sum_k f[k]*p[k+0] = a*p0 + b*p1 + c*p2
T1[1] = sum_k f[k]*p[k+1] = a*p1 + b*p2 + c*p3
T1[2] = sum_k f[k]*p[k+2] = a*p2 + b*p3 + c*p4

= [a*p4, a*p3+b*p4, a*p2+b*p3+c*p4, a*p1+b*p2+c*p3, a*p0+b*p1+c*p2, b*p0+c*p1, c*p0]

Length = 3+5-1 = 7.

T1[0] = a*p0+b*p1+c*p2 = index 4 in the conv
T1[1] = a*p1+b*p2+c*p3 = index 3
T1[2] = a*p2+b*p3+c*p4 = index 2

So T1[j] = conv(f, p[::-1])[4-j] = conv(f, p[::-1])[2n-2-j] ✓ (2n-2 = 4)

T2[0] = sum_k f[k]*p[0-k]: only k=0 valid → a*p0
T2[1] = sum_k f[k]*p[1-k]: k=0: a*p1, k=1: b*p0 → a*p1+b*p0
T2[2] = sum_k f[k]*p[2-k]: k=0: a*p2, k=1: b*p1, k=2: c*p0 → a*p2+b*p1+c*p0

conv(f, p) = conv([a,b,c], [p0,p1,p2,p3,p4])
= [a*p0, a*p1+b*p0, a*p2+b*p1+c*p0, a*p3+b*p2+c*p1, a*p4+b*p3+c*p2, b*p4+c*p3, c*p4]

T2[0] = a*p0 = index 0 ✓
T2[1] = a*p1+b*p0 = index 1 ✓
T2[2] = a*p2+b*p1+c*p0 = index 2 ✓

So T2[j] = conv(f, p)[j] for j = 0, ..., n-1. ✓

So the gradient is:
grad = T1 + T2
T1 = conv(f, p[::-1])[2n-2 : n-2 : -1]  (i.e., reversed slice from index 2n-2 down to n-1)
T2 = conv(f, p)[:n]

conv1 = np.convolve(f, p[::-1])  # length 3n-2
T1 = conv1[2*n-2 : n-2 : -1]  # indices 2n-2, 2n-3, ..., n-1 → length n

Let me verify the slice: 2n-2 down to n-1 (inclusive) is 2n-2-(n-1)+1 = n elements. ✓

This is O(n log n) using FFT-based convolution in numpy. For n=5000, this is very fast.

Let me update the gradient_polish function:

```python
def gradient_polish(f, max_iter=200, tau=0.01, verbose=True):
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    f = f / f.sum()
    n = len(f)
    best_obj = compute_score(f)
    best_f = f.copy()
    
    lr = 1.0
    for it in range(max_iter):
        conv_f = np.convolve(f, f)
        max_c = conv_f.max()
        exp_c = np.exp((conv_f - max_c) / tau)
        p = exp_c / exp_c.sum()
        
        # Compute gradient using FFT
        conv1 = np.convolve(f, p[::-1], mode='full')  # length 3n-2
        T1 = conv1[2*n-2 : n-2 : -1]  # length n
        conv2 = np.convolve(f, p, mode='full')  # length 3n-2
        T2 = conv2[:n]  # length n
        grad = T1 + T2
        
        # Gradient step with projection
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        f_new = f_new / s
        obj = compute_score(f_new)
        
        if obj < best_obj:
            f = f_new
            best_obj = obj
            best_f = f.copy()
            # Adaptive learning rate: increase if we made good progress
            if lr < 10:
                lr *= 1.05
        else:
            lr *= 0.5
            if lr < 1e-12:
                break
        
        if verbose and it % 20 == 0:
            print(f"  Grad iter {it}: obj={best_obj:.6f}, lr={lr:.6f}")
    
    return best_f
```

This should be much faster. Each gradient step is O(n log n) for the FFT convolutions + O(n) for the log-sum-exp.
