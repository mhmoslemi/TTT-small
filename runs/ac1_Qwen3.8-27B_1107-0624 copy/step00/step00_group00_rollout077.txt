def _score(a):
    """Fast objective computation matching evaluate_sequence."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (s * s))


def _softmax_grad_step(a, beta, step_frac):
    """One gradient descent step using softmax-weighted peak reduction."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = np.max(c)
    weights = np.exp(beta * (c - c_max))
    total_w = np.sum(weights)

        return a
    weights /= total_w

    thresh = 1e-6
    active = np.where(weights > thresh)[0]
    if len(active) == 0:
        return a

    grad = np.zeros(n)
    for k in active:
        wk = weights[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            idx = np.arange(k - i_lo, k - i_hi - 1, -1)
            grad[i_lo:i_hi + 1] += wk * a[idx]

        j_lo = max(k, 0)
        j_hi = n - 1
        if j_lo <= j_hi:
            count = j_hi - j_lo + 1
            grad[j_lo:j_hi + 1] += wk * a[:count]

def _parametric_cosine(n, f, alpha, phase=0.0):
    """Generate a[i] = 1 + alpha * cos(2*pi*f*i/n + phase), clipped to non-negative."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * f * idx / n + phase)
    np.maximum(a, 0.0, out=a)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _parametric_u_shape(n, power):
    """Generate a U-shaped sequence."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** power + (1.0 - t) ** power
    a += 0.01
    return a / np.sum(a)


def _parametric_exp_cos(n, f, alpha):

def _parametric_quad_residue(n):
    """Generate quadratic residue-based sequence."""
    from math import isqrt

        is_prime = True
        for d in range(2, isqrt(p) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p -= 1
    if p < 10:
        return np.ones(n) / n
    residues = set()
    for x in range(p):
        residues.add((x * x) % p)
    a = np.zeros(n)

        positions = np.arange(r, n, p)
        a[positions] = 1.0
    s = np.sum(a)

    return a / s
