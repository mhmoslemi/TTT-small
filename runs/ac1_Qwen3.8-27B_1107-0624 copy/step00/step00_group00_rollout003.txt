    """Try a structured perturbation; return improved sequence or None."""
    f = best.copy()

            if np.isfinite(score_new) and score_new < best_score:
                best = f_new
                best_score = score_new
                no_improve_count = 0
                print(f"Iter {iteration}: gradient improved to {best_score:.10f}")
            else:
                # Try negative gradient (ascent in f where gradient is negative)
                f_neg = best + step_size * np.zeros(n)  # placeholder
                # Just check if uniform is still best
                if best_score > 2.0:
                    # Drift back toward uniform
                    best = 0.9 * best + 0.1 * uniform
                    best = np.clip(best, 0.0, 1000.0)
                    best_score = float(evaluate_sequence(best.tolist()))

        # Strategy 3: Periodically re-evaluate uniform as anchor
        if iteration % 100 == 0:
            u_score = float(evaluate_sequence(uniform.tolist()))
            if u_score < best_score:
                best = uniform.copy()
                best_score = u_score
                print(f"Iter {iteration}: reverted to uniform, score={best_score:.10f}")
            if best_score <= 2.0000001:
                print(f"Iter {iteration}: confirmed at theoretical optimum (~2.0)")

    # Final verification with the actual evaluator
    final_score = float(evaluate_sequence([float(x) for x in best]))
    print(f"Final score (evaluator): {final_score:.10f}")
    print(f"Total iterations: {iteration}")

    return [float(x) for x in best]
