def subgradient_step(p, eta):
    n = len(p)
    c = np.convolve(p, p)
    k_star = np.argmax(c)
    # Subgradient at k*: g[i] = 2 * p[k_star - i] (for valid k_star - i)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = 2.0 * p[j]
    # Update
    p_new = p - eta * g
    # Project onto non-negative simplex (sum = 1)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s > 0:
        p_new = p_new / s
    else:
        p_new = np.ones(n) / n
    return p_new, c[k_star]
```

def optimize_subgradient(p_init, n_iters, eta_init, decay):
    p = p_init.copy()
    best_p = p.copy()
    best_obj = np.inf
    n = len(p)
    
    for t in range(n_iters):
        eta = eta_init / (1 + decay * t)
        c = np.convolve(p, p)
        obj = c.max()
        if obj < best_obj:
            best_obj = obj
            best_p = p.copy()
        
        k_star = np.argmax(c)
        # Subgradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = 2.0 * p[k_star - i]
        
        p = p - eta * g
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s > 0:
            p = p / s
    
    return best_p, best_obj
```
