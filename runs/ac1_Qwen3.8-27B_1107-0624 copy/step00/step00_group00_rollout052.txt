def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _fast_convolve_pair(a, w):
    """Compute conv(a, w) and cross_corr(w, a) using shared FFTs."""
    n = len(a)
    m = len(w)
    N = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    conv_aw = np.fft.irfft(A * W, N)
    xcorr_wa = np.fft.irfft(np.conj(W) * A, N)  # IFFT(conj(W) * A)
    return conv_aw, xcorr_wa

def _objective_grad(a, t):
    """Compute the smooth objective and its gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e10, np.zeros(n)
    
    # Compute c = conv(a, a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2*n-1]

    tc = t * c
    log_max = logsumexp(tc)
    L = log_max / t
    w = np.exp(tc - log_max)  # softmax weights, sum to 1
    
    # Objective
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    # vc[m] = v1[m] + v2[m] where
    # v2[m] = (conv(a, w))[m] = IFFT(FFT(a) * FFT(w))[m]
    # v1[m] = (w ⊛ a)[m] = IFFT(conj(FFT(w)) * FFT(a))[m]
    N2 = _next_pow2(n + len(w) - 1)
    A2 = np.fft.rfft(a, N2)
    W2 = np.fft.rfft(w, N2)
    v2 = np.fft.irfft(A2 * W2, N2)[:n]
    v1 = np.fft.irfft(np.conj(W2) * A2, N2)[:n]
    
    vc = v1 + v2
