"""
Minimize  2*n*max(conv(a,a)) / (sum a)^2  over non-negative step sequences.

Key observations exploited here:
  * The objective is scale invariant and invariant under the clamps
    a <- clip(a, 0, 1000), so every candidate is normalised to mean 1.
  * Constant (symmetric) sequences sit exactly at 2.0 and form a flat
    "col"; small symmetric perturbations leave them stuck there (this is why
    the previous block-perturb + mirror search stalled at 2.0).
  * Asymmetric, moderate dynamic-range profiles (ramps / exponentials /
    skewed bumps) can place the convolution peak off-centre where it is
    smaller relative to (sum a)^2, allowing values below 2.0.

Strategy:
  1. Use an FFT based exact objective (mirrors evaluate_sequence, incl. the
     0.01 sum guard and the [0,1000] clamps) so each trial is O(n log n)
     instead of O(n^2).  This permits ~10^5-10^6 trials inside the budget.
  2. Coarse parameterised scan over a rich family of asymmetric inits and
     several sequence lengths (the given length plus a couple of multiples),
     keeping a single global best.
  3. Local non-commutative search (block multiplicative noise, "push the mass
     to one side", occasional weak symmetrisation, annealed amplitude) rebuilt
     from the best basins found in (2), refining the global best further.
No forced full-symmetrisation (only a small, occasional, partial one) so the
search is not locked into the symmetric 2.0 manifold.
"""

import time
import numpy as np


def _next_pow2(m):
    return int(1) << int((m - 1).bit_length())


def fast_objective(a):
    """Exact mirror of evaluate_sequence (clamps + sum guard) using FFT conv."""
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    m = 2 * n - 1
    size = _next_pow2(m)
    A = np.fft.rfft(a, size)
    C = np.fft.irfft(A * A, size)
    mb = float(C[:m].max())
    return float(2.0 * n * mb / (s * s))


def _normalise(a):
    s = a.sum()
    if s <= 0.0:
        return np.clip(a, 0.0, 1000.0)
    a = a * (float(a.size) / s)          # mean 1 (scale invariant objective)
    return np.clip(a, 0.0, 1000.0)


def init_ramp(n, p, eps, rng):
    x = np.linspace(eps, 1.0, n)
    a = x ** p
    return _normalise(a + rng.normal(0.0, 0.01, n))


def init_exp(n, t, rng):
    a = np.exp(t * np.linspace(0.0, 1.0, n))
    return _normalise(a + rng.normal(0.0, 0.01, n))


def init_skew_bump(n, c, w, d, rng):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - c) ** 2) / (2.0 * w * w)) * (1.0 + d * x)
    a = np.abs(a)
    return _normalise(a + rng.normal(0.0, 0.01, n))


def init_step_slope(n, d, rng):
    a = np.clip(np.linspace(-d, d, n) + 0.5, 1e-4, None)
    return _normalise(a + rng.normal(0.0, 0.01, n))


def init_two_level(n, gap, rng):
    h = np.ones(n)
    j = int(rng.integers(1, max(2, n)))
    h[j:] *= gap
    return _normalise(h + rng.normal(0.0, 0.05, n))


def init_random(n, scale, rng):
    a = np.abs(rng.normal(0.0, scale, n)) + 1e-4
    return _normalise(a)


def init_linramp_perturb(n, rng):
    a = np.linspace(1.0, n, n)
    return _normalise(a + rng.normal(0.0, n * 0.05, n))


def make_candidate(current, rng, amp):
    c = current.copy()
    n = c.size
    if n < 3:
        c *= np.exp(rng.normal(0.0, amp))
        return _normalise(c)
    # 1) smooth block multiplicative noise
    lo = max(2, n // 200)
    hi = max(lo + 1, n // 3)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    c[start:start + width] *= np.exp(rng.normal(0.0, amp, width))
    # 2) asymmetric mass-shift (the move that broke the 2.0 plateau)
    r = rng.random()
    if r < 0.40:
        half = n // 2
        if r < 0.20:
            c[half:] *= np.exp(rng.normal(0.0, amp))
        else:
            c[:half] *= np.exp(rng.normal(0.0, amp))
    elif r < 0.50:
        # grow one end, shrink the other  -> dynamic-range exploration
        c[: n // 4] *= np.exp(rng.normal(0.0, amp))
        c[-(n // 4):] *= np.exp(rng.normal(0.0, -amp + rng.normal(0.0, amp)))
    # 3) occasionally a fresh local bump
    if rng.random() < 0.25:
        bw = int(rng.integers(2, max(3, n // 50)))
        bs = int(rng.integers(0, n - bw + 1))
        c[bs:bs + bw] += amp * np.abs(rng.normal(0.0, 1.0, bw))
    # 4) only occasionally a weak partial symmetrisation (never a full lock)
    if rng.random() < 0.03:
        c = 0.5 * (c + c[::-1])
    return _normalise(c)


def _seed_from_global(rng):
    try:
        g = np.asarray(height_sequence_1, dtype=np.float64).ravel()  # noqa: F821
        if g.size >= 10 and np.isfinite(g).all():
            return _normalise(np.clip(g, 0.0, 1000.0))
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 3.0)

    # ---- decide search lengths ------------------------------------------
    base_n = 8192
    g = _seed_from_global(rng)
    if g is not None:
        base_n = int(g.size)
    if not (100 <= base_n <= 200000):
        base_n = 8192
    lengths = [base_n]
    for mult in (2, 4):
        cand = base_n * mult
        if cand <= 200000 and cand not in lengths:
            lengths.append(cand)

    best = None
    best_score = float("inf")

    def consider(a):
        nonlocal best, best_score
        a = np.asarray(a, dtype=np.float64)
        sc = fast_objective(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()
            return True
        return False

    # ---- Phase 1: coarse parameterised scan (find sub-2 basins) ---------
    scan_budget = 0.35 * max(0.0, float(budget_s) - 3.0)
    scan_end = t0 + scan_budget
    trials = 0
    while time.monotonic() < scan_end:
        n = int(lengths[int(rng.integers(0, len(lengths)))])
        kind = int(rng.integers(0, 8))
        if kind == 0:
            a = init_ramp(n, float(rng.uniform(0.2, 5.0)),
                          float(10.0 ** rng.uniform(-4.0, -1.0)), rng)
        elif kind == 1:
            a = init_exp(n, float(rng.uniform(0.3, 4.5)), rng)
        elif kind == 2:
            a = init_skew_bump(n, float(rng.uniform(0.2, 0.8)),
                               float(rng.uniform(0.02, 0.35)),
                               float(rng.uniform(-1.5, 1.5)), rng)
        elif kind == 3:
            a = init_step_slope(n, float(rng.uniform(0.2, 2.5)), rng)
        elif kind == 4:
            a = init_two_level(n, float(rng.uniform(0.05, 6.0)), rng)
        elif kind == 5:
            a = init_random(n, float(rng.uniform(0.3, 1.5)), rng)
        elif kind == 6:
            a = init_ramp(n, float(rng.uniform(0.2, 5.0)), 1e-4, rng)[::-1].copy()
        else:
            a = init_linramp_perturb(n, rng)
        consider(a)
        trials += 1
        if trials % 2000 == 0:
            print(f"[scan] trials={trials} best={best_score:.6f} "
                  f"len={best.size if best is not None else 0}")

    print(f"[scan done] best={best_score:.6f}")

    # ---- Phase 2: local refinement from the best basin ------------------
    if best is None:
        best = _normalise(np.ones(base_n))
        best_score = fast_objective(best)

    cur = best.copy()
    cur_score = best_score
    amp = 0.5
    last_improve = time.monotonic()
    stagn = 0
    while time.monotonic() < deadline:
        cand = make_candidate(cur, rng, amp)
        sc = fast_objective(cand)
        if np.isfinite(sc) and sc < cur_score - 1e-12:
            cur = cand
            cur_score = sc
            amp *= 0.985
            stagn = 0
            if sc < best_score:
                best_score = sc
                best = cur.copy()
                last_improve = time.monotonic()
        else:
            stagn += 1
            amp = min(0.6, amp * 1.01)
        # reheat / restart if long stagnation
        if (time.monotonic() - last_improve) > 25.0 or stagn > 4000:
            stagn = 0
            amp = 0.5
            r = rng.random()
            if r < 0.5:
                cur = best.copy() + rng.normal(0.0, 0.3, best.size)
            else:
                n2 = int(lengths[int(rng.integers(0, len(lengths)))])
                cur = init_exp(n2, float(rng.uniform(0.5, 3.0)), rng)
            cur = _normalise(cur)
            cur_score = fast_objective(cur)
        if (stagn % 5000) == 0:
            print(f"[refine] cur={cur_score:.6f} best={best_score:.6f} "
                  f"amp={amp:.3f} t={time.monotonic()-t0:.0f}s")

    # ---- final exact validation with the provided evaluator -------------
    out = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    try:
        exact = float(evaluate_sequence(out))  # noqa: F821  (in environment)
    except Exception:
        exact = best_score
    print(f"[final] length={len(out)} fast={best_score:.6f} exact={exact:.6f}")
    return out
