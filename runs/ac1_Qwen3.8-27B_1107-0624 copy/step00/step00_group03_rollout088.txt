```
def _lse_and_weights(c, alpha):
   m=c.max()
   x=alpha*(c-m)
   # stable logsumexp
   ax=x.max()
   expx=np.exp(x-ax)
   lse = m + (ax + np.log(expax.sum()))/alpha

   q=np.fft.ifft(np.conj(SFFT)*PFFT).real[:n]
```

```
import numpy as np
import time

def _fft_next_power_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()

def _project(s):
    s = np.asarray(s, dtype=np.float64).copy()
    s = np.maximum(s, 0.0)
    s = np.where(np.isfinite(s), s, 0.0)
    total = float(s.sum())
    if total <= 0.0:
        s.fill(1.0)
        total = float(s.size)
    s /= total
    return s

def _conv_square(s):
    n = s.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_next_power_two(2*n - 1)
    S = np.fft.rfft(s, L)
    c = np.fft.irfft(S * S, L)
    c = c[:2*n - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _exact_score(s):
    n = s.size
    total = float(s.sum())
    if total < 0.01:
        return float('inf')
    c = _conv_square(s)
    cmax = float(c.max())
    return float(2.0 * n * cmax / (total * total))

        val = float(val)
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float('inf')

    m = float(c.max())
    x = float(alpha) * (c - m)
    ax = float(x.max())
    e = np.exp(x - ax)
    esum = float(e.sum())
    if esum <= 0.0 or not np.isfinite(esum):
        return m
    return m + (ax + float(np.log(esum))) / float(alpha)

    m = float(c.max())
    x = float(alpha) * (c - m)
    ax = float(x.max())
    e = np.exp(x - ax)

    else:

    grad = np.where(np.isfinite(grad), grad, 0.0)
```

```python
"""
Search for non-negative step-function heights minimizing
    2 * n * max(conv(s, s)) / (sum(s))**2
using FFT-based projected gradient descent with adaptive LSE/peak/L2 objectives,
local block perturbations, and multi-start resets.
"""

import numpy as np
import time


def _fft_next_power_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _project(s):
    """Project to the shifted simplex: nonnegative and sum exactly 1."""
    s = np.asarray(s, dtype=np.float64).copy()
    s = np.where(np.isfinite(s), s, 0.0)
    s = np.maximum(s, 0.0)
    total = float(s.sum())
    if total <= 0.0:
        s.fill(1.0)
        total = float(s.size)
    s /= total
    return s


def _conv_square(s):
    """Full linear convolution s * s using FFT, clipped to nonnegative."""
    n = s.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_next_power_two(2 * n - 1)
    S = np.fft.rfft(s, L)
    c = np.fft.irfft(S * S, L)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _exact_score(s):
    """Local exact score using the same formula as evaluate_sequence."""
    n = s.size
    total = float(s.sum())
    if total < 0.01:
        return float('inf')
    c = _conv_square(s)
    cmax = float(c.max())
    return float(2.0 * n * cmax / (total * total))


def _evaluate_safe(seq):
    """Call the external evaluator if present, returning inf on failure."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float('inf')


def _lse_value(c, alpha):
    """Stable log-sum-exp of alpha*c."""
    m = float(c.max())
    a = float(alpha)
    if a <= 0.0:
        return m
    x = a * (c - m)
    ax = float(x.max())
    e = np.exp(x - ax)
    esum = float(e.sum())
    if esum <= 0.0 or not np.isfinite(esum):
        return m
    return m + (ax + float(np.log(esum))) / a


def _weights_lse(c, alpha):
    """Softmax weights proportional to exp(alpha*c), stable."""
    m = float(c.max())
    a = float(alpha)
    if a <= 0.0:
        return np.full_like(c, 1.0 / float(c.size), dtype=np.float64)
    x = a * (c - m)
    ax = float(x.max())
    e = np.exp(x - ax)
    esum = float(e.sum())
    if esum <= 0.0 or not np.isfinite(esum):
        return np.full_like(c, 1.0 / float(c.size), dtype=np.float64)
    return e / esum


def _weights_power(c, power):
    """Weights proportional to c^power, useful for L2/variance polish."""
    c = np.maximum(np.asarray(c, dtype=np.float64), 0.0)
    w = np.power(c, float(power))
    w = np.where(np.isfinite(w), w, 0.0)
    total = float(w.sum())
    if total <= 0.0 or not np.isfinite(total):
        return np.full_like(c, 1.0 / float(c.size), dtype=np.float64)
    return w / total


def _weights_topk(c, k):
    """Uniform weights over the top-k lag coefficients."""
    c = np.asarray(c, dtype=np.float64)
    n = c.size
    k = int(k)
    if n == 0:
        return np.array([], dtype=np.float64)
    if k <= 1:
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p
    k = min(n, k)
    thresh = float(np.partition(c, n - k)[n - k])
    p = (c >= thresh).astype(np.float64)
    total = float(p.sum())
    if total <= 0.0:
        p.fill(1.0 / n)
    else:
        p /= total
    return p


def _gradient_weights(s, p):
    """
    Gradient of sum_k p_k * (s*s)_k with respect to s.
    Since d(s*s)_k / d s_j = 2*s_{k-j} when valid, this is
    2 * cross-correlation(p, s) restricted to the first n lags.
    """
    n = s.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _fft_next_power_two(2 * n - 1)
    S = np.fft.fft(s, L)
    P = np.fft.fft(p, L)
    q = np.fft.ifft(np.conj(S) * P)
    grad = 2.0 * q.real[:n]
    return np.where(np.isfinite(grad), grad, 0.0)


def _random_perturb(s, rng, sigma):
    """Multiplicative lognormal perturbation followed by projection."""
    s = np.asarray(s, dtype=np.float64).copy()
    if s.size == 0:
        return s
    noise = rng.normal(0.0, sigma, s.size)
    noise = np.clip(noise, -10.0, 10.0)
    s = s * np.exp(noise)
    return _project(s)


def _block_perturb(s, rng, sigma=0.1):
    """Perturb a random contiguous block; sometimes symmetrize."""
    s = np.asarray(s, dtype=np.float64).copy()
    n = s.size
    if n == 0:
        return s
    min_w = max(1, n // 1000)
    max_w = max(min_w + 1, n // 25)
    width = int(rng.integers(min_w, max_w))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, sigma)))
    s[start:stop] *= factor
    if rng.random() < 0.30:
        s = 0.5 * (s + s[::-1])
    return _project(s)


def _make_initial_set(base, rng):
    """Create several normalized initial candidates and rank them exactly."""
    n = base.size
    cands = []
    cands.append(_project(base.copy()))
    cands.append(_project(np.exp(rng.normal(0.0, 0.1, n))))

    if n > 1:
        x = np.linspace(0.0, 1.0, n)
    else:
        x = np.array([0.0], dtype=np.float64)

    for r in (0.5, -0.5, 1.0, -1.0):
        cands.append(_project(np.exp(r * x)))

    if n > 2:
        z = (x - 0.5) * 4.0
        cands.append(_project(np.exp(-z * z)))
        cands.append(_project(np.exp(z * z)))

    records = []
    for i, c in enumerate(cands):
        val = _exact_score(c)
        if not np.isfinite(val):
            val = float('inf')
        records.append((val, i, c))
    records.sort()
    return records


def _optimize_s(s, deadline, rng, reset_sources, initial_best_s=None, initial_best_score=None):
    """
    Adaptive projected-descent optimizer on the normalized nonnegative simplex.

    Modes:
      0: LSE peak-smoothing with alpha schedule
      1: L2 variance polish using c^1 weights
      2: top-k peak shaving
    """
    n = s.size
    if n == 0:
        return np.ones(1, dtype=np.float64), float('inf')

    if initial_best_s is not None:
        best_s = np.asarray(initial_best_s, dtype=np.float64).copy()
        best_score = float(initial_best_score) if initial_best_score is not None else _exact_score(best_s)
    else:
        best_s = _project(s.copy())
        best_score = _exact_score(best_s)

    s = _project(s.copy())

    alpha = 6.0
    mode = 0
    topk = max(8, n // 200)
    no_improve = 0
    it = 0
    last_print = time.monotonic()
    last_eval = time.monotonic()
    step_multipliers = (0.50, 0.25, 0.125, 0.0625, 0.03125, 0.015625)
    scale = float(2 * n - 1)

    while time.monotonic() < deadline:
        c = _conv_square(s)
        cmax = float(c.max())
        total = float(s.sum())
        score = float(2.0 * n * cmax / (total * total))
        cs = c * scale

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_s = s.copy()
            if time.monotonic() - last_eval > 3.0:
                ev = _evaluate_safe(best_s.tolist())
                if np.isfinite(ev) and ev < best_score:
                    best_score = ev
                last_eval = time.monotonic()

        cur_lse = _lse_value(cs, alpha)
        if mode == 0:
            p = _weights_lse(cs, alpha)
            cur_l2 = None
        elif mode == 1:
            p = _weights_power(cs, 1.0)
            cur_l2 = float(np.dot(cs, cs))
        else:
            p = _weights_topk(c, topk)
            cur_l2 = None

        grad = _gradient_weights(s, p)
        g = -grad
        g = g - float(np.mean(g))
        g = np.where(np.isfinite(g), g, 0.0)
        gmax = float(np.max(np.abs(g)))

        improved = False

        if not np.isfinite(gmax) or gmax <= 1e-300:
            if alpha > 15.0:
                alpha = max(4.0, alpha * 0.5)
            else:
                alpha = min(alpha * 1.5, 80.0)
            s = _random_perturb(s, rng, 0.05)
            no_improve += 1
        else:
            d = g / gmax
            mean_s = float(s.mean())

            for mult in step_multipliers:
                step = mult * mean_s
                cand = _project(s + step * d)
                c2 = _conv_square(cand)
                c2max = float(c2.max())
                total2 = float(cand.sum())
                score2 = float(2.0 * n * c2max / (total2 * total2))
                c2s = c2 * scale

                better = np.isfinite(score2) and score2 < score - 1e-10

                if mode == 1:
                    l22 = float(np.dot(c2s, c2s))
                    if np.isfinite(l22) and np.isfinite(cur_l2) and l22 < cur_l2 - 1e-7:
                        better = True
                else:
                    lse2 = _lse_value(c2s, alpha)
                    if np.isfinite(lse2) and lse2 < cur_lse - 1e-9:
                        better = True

                if better:
                    s = cand
                    improved = True
                    break

            if not improved:
                # Cheap local block search from current point and best point.
                if no_improve % 8 == 7:
                    src_choices = (s, best_s)
                    for src in src_choices:
                        for _ in range(3):
                            cand = _block_perturb(src, rng, 0.08)
                            val = _exact_score(cand)
                            if np.isfinite(val) and val < best_score - 1e-12:
                                best_score = val
                                best_s = cand
                                s = cand
                                improved = True
                                break
                        if improved:
                            break

                if not improved:
                    no_improve += 1

                    if no_improve == 40:
                        s = _random_perturb(s, rng, 0.07)
                    elif no_improve == 120:
                        if reset_sources and rng.random() < 0.7:
                            src = reset_sources[int(rng.integers(0, len(reset_sources)))]
                        else:
                            src = best_s
                        s = _random_perturb(src, rng, 0.18)
                        no_improve = 0
                    elif no_improve == 250:
                        mode = (mode + 1) % 3
                        if mode == 0:
                            if alpha < 35.0:
                                alpha = min(80.0, alpha * 1.35)
                            else:
                                alpha = max(5.0, alpha * 0.70)
                        elif mode == 2:
                            topk = max(4, topk // 2)
                        no_improve = 0
                    elif no_improve > 1000:
                        if reset_sources:
                            src = reset_sources[int(rng.integers(0, len(reset_sources)))]
                        else:
                            src = best_s
                        s = _random_perturb(src, rng, 0.35)
                        alpha = 6.0
                        mode = 0
                        no_improve = 0

        it += 1
        if it % 1000 == 0 or time.monotonic() - last_print > 5.0:
            print(f"opt it={it} best={best_score:.10f} alpha={alpha:.2f} mode={mode} noImp={no_improve}")
            last_print = time.monotonic()

    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a best nonnegative sequence under the evaluation function.

    The method starts from available known constructions, builds a few diverse
    initial points, then runs adaptive FFT-based projected gradient descent with
    LSE peak smoothing, L2 flattening polish, top-k peak shaving, random resets,
    and final local block perturbations.
    """
    start = time.monotonic()

    try:
        budget = float(budget_s)
    except Exception:
        budget = 1800.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1800.0

    # Leave room for final checks and possible last local search.
    deadline = start + max(0.0, budget - 4.0)
    final_deadline = start + max(0.0, budget - 1.2)

    rng = np.random.default_rng(seed)

    # Load and sanitize the provided global construction if available.
    base = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0:
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.maximum(arr, 0.0)
            total = float(arr.sum())
            if total > 0.0:
                base = arr / total
    except Exception:
        base = None

    if base is None:
        base = np.ones(256, dtype=np.float64)
    base = _project(base)

    records = _make_initial_set(base, rng)

    chosen = None
    for rec in records:
        if np.isfinite(rec[0]):
            chosen = rec
            break

    if chosen is None:
        best_s = base.copy()
        best_score = _exact_score(best_s)
    else:
        best_s = chosen[2].copy()
        best_score = chosen[0]

    ev = _evaluate_safe(best_s.tolist())
    if np.isfinite(ev) and ev < best_score:
        best_score = ev

    print(f"initial n={best_s.size} score={best_score:.10f}")

    # Preserve the strongest initial points for resets.
    reset_sources = [r[2] for r in records[:5]]

    # Start from a slightly perturbed version of the best initial point so that
    # symmetric stationary points are not perfectly stationary.
    s = best_s.copy()
    if s.size > 1:
        sigma = 0.10 if float(np.var(s)) < 1e-20 else 0.02
        s = _random_perturb(s, rng, sigma)

    if deadline > start:
        best_s, best_score = _optimize_s(
            s,
            deadline,
            rng,
            reset_sources,
            initial_best_s=best_s,
            initial_best_score=best_score
        )

    # Use remaining time for a few fast local improvements.
    if time.monotonic() < final_deadline:
        quick_iter = 0
        while time.monotonic() < final_deadline:
            cand = _block_perturb(best_s, rng, 0.04)
            val = _exact_score(cand)
            if np.isfinite(val) and val < best_score - 1e-12:
                best_score = val
                best_s = cand

            cand = _random_perturb(best_s, rng, 0.03)
            val = _exact_score(cand)
            if np.isfinite(val) and val < best_score - 1e-12:
                best_score = val
                best_s = cand

            quick_iter += 1
            if quick_iter % 50 == 0:
                print(f"final local quick it={quick_iter} best={best_score:.10f}")

    final_val = _evaluate_safe(best_s.tolist())
    if np.isfinite(final_val):
        print(f"final evaluated score={final_val:.10f}")
    else:
        print(f"final local score={best_score:.10f}")

    # Ultimate fallback if something pathological happened.
    if best_s.size == 0 or not np.all(np.isfinite(best_s)):
        best_s = base

    return [float(x) for x in best_s]
```
