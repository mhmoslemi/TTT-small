"""
LP-based search for step functions minimizing 2n*max(autocorrelation)/sum².
Uses linear programming to find optimal update directions, with multiple
initializations and varying sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time


def build_conv_mat_sp(f, n):
    """Build sparse (2n-1)xn matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    M = 2 * n - 1
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        L = min(n, M - j)
        row_parts.append(np.arange(j, j + L))
        col_parts.append(np.full(L, j, dtype=np.int32))
        val_parts.append(f[:L])
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    vals = np.concatenate(val_parts)
    return sparse.csr_matrix((vals, (rows, cols)), shape=(M, n))


def solve_lp_for_direction(f, n, time_limit=20):
    """Solve max sum(g) s.t. convolve_matrix(f)@g <= max(f*f), g>=0."""
    conv = np.convolve(f, f)
    M_peak = np.max(conv)
    if M_peak <= 0 or not np.isfinite(M_peak):
        return None
    A = build_conv_mat_sp(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_peak)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None
