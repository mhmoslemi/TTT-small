Minimizes 2*n*max(autocorr)/sum^2 using power-norm gradient descent,
LP-based dual updates, diverse multi-start, and local refinement.
"""

import numpy as np
import time
from scipy.signal import fftconvolve
from scipy.optimize import linprog


def _compute_b(a):
    """Compute full linear convolution a*a via FFT."""
    return fftconvolve(a, a)


def _fast_score(a):
    """Score for sum-normalized sequence: 2*n*max_b."""
    b = _compute_b(a)
    return 2.0 * len(a) * b.max()


def _compute_gradient(a, b, p=4):
    """
    Gradient of (sum b^p)^(1/p) w.r.t. a, where b=convolve(a,a).
    Returns vector of same length as a.
    """
    n = len(a)
    if p <= 1:
        w = np.ones_like(b)
    else:
        w = b ** (p - 1)
    # part1[j] = sum_k w[k]*a[j-k] = (w*a)[j]
    part1 = np.convolve(w, a, 'full')[:n]
    # part2[j] = sum_k w[k]*a[k-j] = correlate(w,a) at lag j
    part2 = np.correlate(w, a, 'full')[n - 1:2 * n - 1]
    raw = part1 + part2
    # Center to preserve sum constraint
    raw = raw - raw.mean()
    return raw


def _build_lp_matrix(a):
    """Build the (2n-1)xn convolution constraint matrix A where A[k,m]=a[k-m]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = a
    return A

    """

    Returns normalized g or None.

        return None

    return None
