def _autocorr(s):
    """Compute s*s using FFT."""
    n = len(s)
    f = np.fft.rfft(s, 2*n)
    return np.fft.irfft(f * f, 2*n)[:2*n-1]

"""
Minimizes the step-function evaluation score using projected gradient descent
with active-set gradients, LP-based direction finding, and random perturbations.
Key insight: the score is 2nM/S^2 where M is the peak of the autocorrelation.
We use FFT-based autocorrelation computation for speed, and combine multiple
optimization strategies to escape local minima near the uniform (score=2) solution.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _autocorr(s):
    """Compute convolution of s with itself using FFT (autocorrelation)."""
    n = len(s)
    size = 2 * n
    f = np.fft.rfft(s, size)
    return np.fft.irfft(f * f, size)[:2 * n - 1]

def _compute_score(s):
    """Compute the error metric 2nM/S^2 efficiently."""
    c = _autocorr(s)
    M = c.max()
    S = s.sum()
    if S < 0.01:
        return np.inf, c
    return 2.0 * len(s) * M / (S * S), c

def _gradient_active_set(s, c, num_peaks=5):
    """
    Compute a subgradient of max(c)/S^2 w.r.t. s.
    Uses top-k peak positions for a smoother gradient signal.
    Returns gradient vector of same shape as s.
    """
    n = len(s)
    S = s.sum()
    M = c.max()
    
    # Find top-k peak positions
    if num_peaks >= len(c):
        peaks = np.arange(len(c))
    else:
        peaks = np.argpartition(c, -num_peaks)[-num_peaks:]
    
    # Weight peaks exponentially for smoothness
    peak_vals = c[peaks]
    weights = np.exp(10.0 * (peak_vals - M))
    weights /= weights.sum()
    
    # Compute gradient: d(c[k])/d(s[j]) = s[k-j] for valid indices
    grad = np.zeros(n)
    for w, k in zip(weights, peaks):
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        if lo < hi:
            idx = np.arange(lo, hi)
            grad[lo:hi] += w * s[k - idx]
    
    # Chain rule: d(M/S^2)/ds = grad_c/S^2 - 2*M/S^3
    grad = grad / (S * S) - 2.0 * M / (S ** 3)
    return grad

def _project_descent(s, lr, c, momentum=0.9):
    """One step of projected gradient descent with momentum."""
    n = len(s)
    S0 = s.sum()
    grad = _gradient_active_set(s, c, num_peaks=5)
    # Momentum would require storing previous grad; simplified here
    s_new = s - lr * grad
    # Project to non-negativity
    s_new = np.maximum(s_new, 0.0)
    # Renormalize to maintain sum
    S_new = s_new.sum()
    if S_new < 0.01:
        return s.copy()
    s_new = s_new * (S0 / S_new)
    return np.clip(s_new, 0.0, 1000.0)

def _lp_direction(s):
    """
    Solve LP: max sum(g) s.t. convolve(s,g)[k] <= M, g >= 0.
    Returns the direction g or None if LP fails.
    """
    n = len(s)
    c = _autocorr(s)
    M = c.max()
    
    # Build sparse constraint matrix A where A[k,j] = s[j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        if s[j] < 1e-15:
            continue
        lo = j
        hi = min(j + n, 2 * n - 1)
        rows.extend(range(lo, hi))
        cols.extend([j] * (hi - lo))
        vals.extend([s[j]] * (hi - lo))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    bounds = [(0.0, 1000.0)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def _block_perturb(s, rng, intensity=0.15):
    """Apply a random block perturbation to escape local minima."""
    n = len(s)
    S0 = s.sum()
    candidate = s.copy()
    num_blocks = rng.integers(1, 5)
    for _ in range(num_blocks):
        width = max(2, n // rng.integers(20, 100))

    candidate = candidate * (S0 / S_new)
    return np.clip(candidate, 0.0, 1000.0)
