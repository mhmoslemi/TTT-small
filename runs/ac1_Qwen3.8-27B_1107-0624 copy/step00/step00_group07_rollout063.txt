```
def _softmax_b(b):
    b=np.asarray(b,dtype=np.float64)
    bmax=b.max()
    if not np.isfinite(bmax): return np.ones(b.size)/b.size
    e=np.exp(b-bmax)
    s=e.sum()
    if not np.isfinite(s) or s<=1e-300:
       a=np.ones_like(b); s=a.size
    else: a=e/s
    return a
```

```
import numpy as np

def _score_key(item):
    return item[0]

def _normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    a = a / s
    return a

def _fft_convolve(a):
    n = int(a.size)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:need]
    c = np.where(c > 0.0, c, 0.0)
    return c

def _fast_score(a, c=None):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    if c is None:
        c = _fft_convolve(a / s)
        denom = s * s
    else:
        # If a not normalized? This c likely conv of normalized? safer recompute if sum not 1?

        if abs(s - 1.0) > 1e-8:

        denom = s * s
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / denom)
```

```python
"""
Search for low-peak self-convolution nonnegative sequences.

The objective is scale invariant, so internal optimization works with sum=1.
Candidates are generated from smooth asymmetric priors and refined with a
multiplicative active-set heuristic: at each step, coefficients of a*a that
are near the maximum are used as weights, and the log-heights are decreased
where they contribute most to those active convolution coefficients.  The
update is accepted by a fast FFT evaluation and the best point is kept.
"""

import numpy as np
import time


def _score_key(item):
    return item[0]


def _normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    a = a / s
    return a


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:need]
    c = np.where(c > 0.0, c, 0.0)
    return c


def _fast_score(a, c=None):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf

    if c is None:
        c = _fft_convolve(a / s)
    else:
        if abs(s - 1.0) > 1e-8:
            c = _fft_convolve(a / s)

    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _log_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.log(np.maximum(a, 1e-300))
    bmean = float(b.mean())
    if np.isfinite(bmean):
        b -= bmean
    return b


def _softmax_b(b):
    b = np.asarray(b, dtype=np.float64).ravel()
    bmax = float(b.max())
    if not np.isfinite(bmax):
        return np.ones(b.size, dtype=np.float64) / float(b.size)
    e = np.exp(b - bmax)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(b.size, dtype=np.float64) / float(b.size)
    return e / s


def _perturb(a, rng, scale):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    noise = rng.normal(0.0, float(scale), size=a.size)
    a *= np.exp(noise)
    return _normalize_positive(a)


def _add_scored(scored, arr):
    a = _normalize_positive(arr)
    if a is None:
        return
    sc = _fast_score(a)
    if np.isfinite(sc):
        scored.append((float(sc), a.copy()))


def _select_candidates(n, rng, top_count=3):
    n = int(n)
    scored = []
    x = np.arange(n, dtype=np.float64) / float(max(1, n))

    # Plain baseline.
    _add_scored(scored, np.ones(n, dtype=np.float64))

    # Simple random positive priors.
    for sig in (0.3, 0.6, 1.0):
        _add_scored(scored, np.exp(rng.normal(0.0, sig, size=n)))

    # Low-frequency asymmetric priors.
    phases = np.linspace(0.0, 2.0 * np.pi, 6, endpoint=False)
    amps = (0.3, 0.6, 0.9)
    for ph in phases:
        for amp in amps:
            _add_scored(scored, 1.0 + amp * np.cos(2.0 * np.pi * x + ph))
            _add_scored(scored, 1.0 + amp * np.sin(2.0 * np.pi * x + ph))

    # Two-mode priors, with phase separation.
    modes = ((1, 2), (1, 3), (2, 3))
    for k1, k2 in modes:
        for amp1 in (0.3, 0.7):
            for amp2 in (-0.7, 0.7):
                for ph in (0.0, 0.5 * np.pi):
                    f1 = 1.0 + amp1 * np.cos(2.0 * np.pi * k1 * x) + amp2 * np.cos(
                        2.0 * np.pi * k2 * x + ph
                    )
                    _add_scored(scored, f1)
                    f2 = 1.0 + amp1 * np.sin(2.0 * np.pi * k1 * x + ph) + amp2 * np.sin(
                        2.0 * np.pi * k2 * x
                    )
                    _add_scored(scored, f2)

    # Random smooth nonnegative shapes.
    for _ in range(12):
        f = np.ones(n, dtype=np.float64)
        for k in range(1, 9):
            f = f + (
                rng.normal(0.0, 0.25)
                * np.cos(2.0 * np.pi * k * x + rng.uniform(0.0, 2.0 * np.pi))
            )
        _add_scored(scored, f)

    # Random multiplicative block perturbations of uniform.
    for _ in range(10):
        a = np.ones(n, dtype=np.float64)
        nb = int(rng.integers(2, 8))
        minlen = max(1, min(n, n // 32))
        maxlen = max(minlen, min(n, n // 2))
        for _b in range(nb):
            start = int(rng.integers(0, n))
            if maxlen > minlen:
                length = int(rng.integers(minlen, maxlen + 1))
            else:
                length = minlen
            stop = min(n, start + length)
            if stop > start:
                factor = float(np.exp(rng.normal(0.0, 0.5)))
                a[start:stop] *= factor
        _add_scored(scored, a)

    scored.sort(key=_score_key)
    if not scored:
        return [(np.inf, np.ones(n, dtype=np.float64) / float(n))]
    return scored[: max(1, int(top_count))]


def _refine_topk(a, seconds, rng, k0=None, step0=0.07):
    a = _normalize_positive(a)
    if a is None:
        return None, np.inf

    seconds = float(seconds)
    n = int(a.size)

    if seconds <= 0.0:
        return a, _fast_score(a)

    if k0 is None:
        k0 = max(8, int(n // 128))

    c = _fft_convolve(a)
    best_score = _fast_score(a, c)
    if not np.isfinite(best_score):
        return a, np.inf

    best = a.copy()
    current_score = best_score
    b = _log_a(a)

    step = float(step0)
    it = 0
    stall = 0
    best_stall = 0
    restarts = 0

    start = time.monotonic()
    end = start + max(0.0, seconds)

    while time.monotonic() < end and restarts < 120:
        # Escape from plateaus / bad drift by perturbing the best point.
        if best_stall > 200 or step < 1e-5 or (
            np.isfinite(current_score) and np.isfinite(best_score) and current_score > best_score + 0.3
        ):
            scale = 0.15 + 0.05 * min(restarts, 8)
            a = _perturb(best, rng, scale)
            c = _fft_convolve(a)
            current_score = _fast_score(a, c)
            if current_score < best_score:
                best = a.copy()
                best_score = current_score
            b = _log_a(a)
            step = float(step0)
            it = 0
            stall = 0
            best_stall = 0
            restarts += 1
            continue

        m = float(np.max(c))
        if not np.isfinite(m) or m <= 0.0:
            break
        current_score = float(2.0 * n * m)

        if current_score < best_score - 1e-12:
            best = a.copy()
            best_score = current_score
            best_stall = 0
        else:
            best_stall += 1

        frac = min(1.0, float(it) / 1500.0)
        Kcur = int(max(1, k0 * (1.0 - 0.85 * frac)))
        Kcur = min(Kcur, int(c.size))

        if Kcur >= int(c.size):
            idx = np.arange(c.size, dtype=np.int64)
        else:
            idx = np.argpartition(c, c.size - Kcur)[-(Kcur):]

        # Include symmetric convolution positions, since c_t = c_{2n-2-t}.
        idx = np.unique(np.concatenate((idx, 2 * (n - 1) - idx)))

        wc = np.asarray(c[idx], dtype=np.float64)
        wsum = float(np.sum(wc))
        if not np.isfinite(wsum) or wsum <= 0.0:
            step *= 0.5
            best_stall += 1
            continue

        # Emphasize the most active coefficients.
        wmax = float(np.max(wc))
        emphasis = 5.0 + 25.0 * frac
        w = np.exp(emphasis * (wc - wmax))
        wsum = float(np.sum(w))
        if not np.isfinite(wsum) or wsum <= 0.0:
            w = np.ones_like(idx, dtype=np.float64)
            wsum = float(w.sum())
        w = w / wsum

        # h_i = sum_t w_t a_{t-i}, the gradient-like contribution to active
        # convolution coefficients.
        h = np.zeros(n, dtype=np.float64)
        for t, wt in zip(idx, w):
            t = int(t)
            if t <= n - 1:
                length = t + 1
                if length > 0:
                    h[:length] += wt * a[t::-1]
            else:
                length = 2 * n - 1 - t
                if length > 0:
                    lo = t - n + 1
                    h[lo:] += wt * a[n - 1::-1][:length]

        hmax = float(np.max(h))
        if not np.isfinite(hmax) or hmax <= 1e-300:
            step *= 0.5
            best_stall += 1
            continue

        b_cand = b - step * (h / hmax)
        b_cand = b_cand - b_cand.mean()
        a_cand = _softmax_b(b_cand)

        c_cand = _fft_convolve(a_cand)
        new_score = _fast_score(a_cand, c_cand)

        it += 1

        accept_tol = 0.02 if frac < 0.3 else 0.005
        if np.isfinite(new_score) and new_score <= current_score + accept_tol:
            a = a_cand
            b = b_cand
            c = c_cand
            current_score = new_score
            stall = 0
            if new_score < best_score - 1e-12:
                best = a.copy()
                best_score = new_score
                best_stall = 0
                step = min(step * 1.04, 0.5)
            else:
                best_stall += 1
                step = min(step * 0.995, 0.5)
        else:
            step *= 0.6
            if step < 1e-5:
                step = float(step0)
            stall += 1
            best_stall += 1

    return best, best_score


def _exact_score(seq):
    try:
        val = evaluate_sequence([float(x) for x in seq])
        return float(val)
    except Exception:
        return np.inf


def _clean_sequence(x):
    try:
        a = np.asarray(x, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a[a < 0.0] = 0.0
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s

    Search for a sequence minimizing 2 * n * max(conv(a,a)) / sum(a)**2.
    """

    rng = np.random.default_rng(int(seed))

    final_reserve = 90.0 if budget_s > 300.0 else max(15.0, 0.15 * budget_s)
    total_end = start + budget_s
    work_end = total_end - final_reserve

    best_seq = None
    best_score = np.inf

    # Use supplied parent if valid.
    try:
        h = _clean_sequence(height_sequence_1)
    except Exception:
        h = None

    if h is not None:
        sc = _fast_score(h)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = h.copy()

    # A few basic lengths.  Keep them reasonably large for asymptotic
    # improvement, but not so large that exact final checks become slow.
    lengths = [1024, 2048, 4096, 8192]
    try:

        if 512 <= hl <= 12000 and hl not in lengths:
            lengths.append(hl)
    except Exception:
        pass

    lengths = sorted(set(lengths))[:5]

    # Allocate more time to smaller lengths (faster evaluations), but give
    # large lengths a larger share because they usually have lower optimum.
    raw_fracs = []
    for i, n in enumerate(lengths):
        raw_fracs.append((1.0 / np.sqrt(float(n))) * (1.0 + 0.1 * i))
    sfr = float(sum(raw_fracs))
    fracs = [v / sfr for v in raw_fracs]

    work_budget = max(1.0, work_end - time.monotonic())

    for n, f in zip(lengths, fracs):
        if time.monotonic() >= work_end:
            break

        if not selected:
            continue

        cand_time = max(1.0, (length_end - time.monotonic()) / float(len(selected)))

        for sc, a in selected:
            if time.monotonic() >= length_end:
                break
            rem = min(cand_time, length_end - time.monotonic())
            if rem < 1.0:
                continue

            seq, score = _refine_topk(a, rem, rng)
            if seq is not None and np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"length={n} score={best_score:.9f}", flush=True)

    # Final polishing of the current champion.
    if best_seq is not None:
        rem = total_end - 60.0 - time.monotonic()
        if rem > 5.0:
            seq, score = _refine_topk(
                best_seq,
                rem,
                rng,
                k0=max(16, int(best_seq.size // 64)),
                step0=0.03,
            )
            if seq is not None and np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"polish score={best_score:.9f}", flush=True)

    # Final exact re-ranking of a few best FFT-scored candidates.
    final_candidates = []
    if best_seq is not None:
        final_candidates.append(best_seq)
    if h is not None:
        final_candidates.append(h)
    if best_seq is not None:
        for scale in (0.05, 0.10, 0.20):
            final_candidates.append(_perturb(best_seq, rng, scale))

    finals = []
    for a in final_candidates:
        a = _normalize_positive(a)
        if a is None:
            continue
        sc = _fast_score(a)
        if np.isfinite(sc):
            finals.append((sc, a))

    finals.sort(key=_score_key)

    chosen = None
    chosen_exact = np.inf
    max_exact = 6 if budget_s > 600.0 else 3

    for _, a in finals[:max_exact]:
        if time.monotonic() > total_end - 2.0:
            break
        val = _exact_score(a)
        if np.isfinite(val) and val < chosen_exact:
            chosen = a
            chosen_exact = val

    if chosen is None and best_seq is not None:
        chosen = best_seq

    if chosen is None:
        chosen = np.ones(1024, dtype=np.float64) / 1024.0

    return [float(x) for x in chosen]
```
