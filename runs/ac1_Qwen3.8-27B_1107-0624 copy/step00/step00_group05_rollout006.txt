def obj_value(a):

import numpy as np
import time

def fast_self_conv(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = int(2 ** np.ceil(np.log2(size)))
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    # Clean up numerical noise
    conv[conv < 0] = 0  # self-convolution of non-negative sequence should be non-negative
    return conv

def compute_objective(a):
    """Compute the objective function value."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fast_self_conv(a)
    max_c = np.max(conv)
    return 2.0 * n * max_c / (S * S)
