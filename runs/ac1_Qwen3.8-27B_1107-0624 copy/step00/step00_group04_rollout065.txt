"""

on the probability simplex with FFT-accelerated convolution, momentum,

def proj_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_conv_sq(a):
    """Compute full linear convolution a*a using FFT (fast for large n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]

    Gradient of conv(a,a)[k] w.r.t. a.

    # Phase 2: Multiple structured initializations (40% budget)
    remaining = deadline - time.monotonic()
    t2 = remaining * 0.40
    inits = generate_structured_inits(n, best_seq, rng)
    per_init = t2 / len(inits)

    for i, a_init in enumerate(inits):
        if time.monotonic() >= deadline - 10:
            break
        a_norm = np.clip(a_init, 0, None)
        s = a_norm.sum()
        if s < 1e-10:
            continue
        a_norm = a_norm / s
        alloc = min(per_init, deadline - time.monotonic() - 5.0)
        if alloc < 2.0:
            break
        a_opt, val = optimize_sequence(a_norm, alloc, rng)
        cand = a_opt * 100.0
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            print(f"Phase 2, init {i}: {best_score:.6f}")

    # Phase 3: Fine-tuning from best (remaining budget)
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        a_norm = np.clip(best_seq, 0, None)
        s = a_norm.sum()
        if s > 0:
            a_norm = a_norm / s
        a_opt, val = optimize_sequence(a_norm, remaining, rng)
        cand = a_opt * 100.0
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc
            print(f"Phase 3 (fine-tune): {best_score:.6f}")

    # Phase 4: Quick multi-restart polish
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        a_norm = np.clip(best_seq, 0, None)
        s = a_norm.sum()
        if s > 0:
            a_norm = a_norm / s
        # Several short independent runs with different seeds
        for trial in range(5):
            if time.monotonic() >= deadline - 3:
                break
            alloc = (deadline - time.monotonic() - 2.0) / (5 - trial)
            a_pert = a_norm + rng.normal(0, 0.05, n)
            a_pert = np.clip(a_pert, 0, None)
            sp = a_pert.sum()
            if sp > 0:
                a_pert = a_pert / sp
            a_opt, val = optimize_sequence(a_pert, alloc, rng)
            cand = a_opt * 100.0
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best_seq = cand.copy()
                best_score = sc
                print(f"Phase 4, trial {trial}: {best_score:.6f}")

    final = float(evaluate_sequence(best_seq.tolist()))
    print(f"FINAL SCORE: {final:.6f}")
    return [float(x) for x in best_seq]
