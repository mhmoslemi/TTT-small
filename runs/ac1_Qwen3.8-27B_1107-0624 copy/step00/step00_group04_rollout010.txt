import time

def compute_score(f):
    """Compute objective and convolution."""
    n = f.size
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf, conv
    score = 2.0 * n * max_b / (sum_a ** 2)
    return score, conv

def subgradient_step(f, conv, active_frac=0.05, eta_scale=0.3):
    """One subgradient step to reduce peak convolution."""
    n = f.size
    max_val = conv.max()
    threshold = max_val * (1 - active_frac)
    active = np.where(conv >= threshold)[0]
    
    # Compute subgradient
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            m = np.arange(j_start, j_end + 1)
            g[m] += f[k - m]
    
    if len(active) > 0:
        g /= len(active)
    
    g -= g.mean()  # Center to preserve sum
    
    # Adaptive step size
    grad_norm_sq = (g ** 2).sum()
    if grad_norm_sq < 1e-20:
        return f.copy()
    eta = eta_scale * max_val / np.sqrt(grad_norm_sq * n)

    candidate = np.clip(candidate, 0.0, 1000.0)
    
    # Renormalize to preserve sum
    s = f.sum()
    if candidate.sum() > 0.01:
        candidate = candidate * s / candidate.sum()
    
    return candidate

    """Compute the objective value and the convolution."""
    n = f.size
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf, conv
    return 2.0 * n * max_b / (sum_a ** 2), conv

def _subgradient(f, conv, top_frac=0.05):
    """Compute centered subgradient of max(conv)."""
    n = f.size
    max_val = conv.max()
    # Use top fraction of bins as "active set"
    n_active = max(3, int(len(conv) * top_frac))
    # Get the top n_active values
    # Use a threshold approach for efficiency
    threshold = max_val - (max_val - np.sort(conv)[-n_active]) 
    # Actually, let's just use a simple threshold
    threshold = max_val * 0.999  # or some fraction
    
    # Better: use the actual top values
    # For efficiency with large conv, use partition
    if len(conv) > 1000:
        # Get approximate top set
        threshold = np.partition(conv, -n_active)[-n_active]
    else:
        threshold = np.sort(conv)[-n_active]
    
    active = np.where(conv >= threshold)[0]
    
    # Compute subgradient: g[j] = mean over active k of f[k-j]
    g = np.zeros(n)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            m = np.arange(j_start, j_end + 1)
            g[m] += f[k - m]
    
    g /= max(1, len(active))
    g -= g.mean()  # Center to preserve sum
    return g

    for _ in range(n_swaps):

        j = int(rng.integers(0, n))

            if rng.random() < 0.5:
                candidate[i] -= amount
                candidate[j] += amount
            else:
                candidate[j] -= amount
                candidate[i] += amount
    return np.clip(candidate, 0.0, 1000.0)

                            amt = min(candidate[i], candidate[j]) * rng.uniform(0.1, 0.5)
                            if rng.random() < 0.5:
                                candidate[i] -= amt
                                candidate[j] += amt
                            else:
                                candidate[j] -= amt
                                candidate[i] += amt
                elif r < 0.70:
                    # Global smooth modulation
                    mod = np.ones(n)
                    for k in range(1, 5):
                        amp = rng.uniform(0.5, 1.5) / k
                        ph = rng.uniform(0, 2 * np.pi)
                        mod += amp * np.cos(2 * np.pi * k * np.arange(n) / n + ph)
                    candidate = current * np.clip(mod, 0.1, 10.0)
                else:
                    # Global multiplicative noise
                    candidate = current * np.exp(rng.normal(0, 0.25, n))
                
                candidate = np.clip(candidate, 0.0, None)
                cs = candidate.sum()
                if cs > 0.01:
                    candidate = candidate * S / cs
                
                cand_score2, _ = _score_and_conv(candidate)
                if cand_score2 < current_score:
                    current = candidate
                    current_score = cand_score2
                    velocity = np.zeros(n)
                    no_improve = 0
                else:
                    # Annealing acceptance
                    delta = cand_score2 - current_score
                    T = max(0.005, 0.05 * remaining / max(1.0, float(budget_remain)))
                    if delta / T < 50.0 and rng.random() < np.exp(-delta / T):
                        current = candidate
                        current_score = cand_score2
                        velocity = np.zeros(n)
                # Reset counter to give perturbation more time
                no_improve = 0
    
    return best, best_score
