import numpy as np
import time

def compute_conv(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]

def fast_score(a):
    """Fast score computation (may differ slightly from evaluate_sequence)."""
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    C = compute_conv(a)
    return 2.0 * len(a) * np.max(C) / (S * S)

def optimize(a, rng, deadline):

    n = len(a)

    a = np.clip(a, 0, 1000)

        return a, np.inf
    a = a / S  # Normalize to sum = 1

    best_score = fast_score(a)
    # Verify with exact evaluation
    exact = evaluate_sequence(a.tolist())
    if np.isfinite(exact) and exact < best_score:
        best_score = exact
    
    step = 1.0
    no_improve = 0
    
    while time.monotonic() < deadline:
        C = compute_conv(a)
        max_C = C.max()
        
        # Active set: peaks within 2% of max
        thresh = max_C * 0.98
        active = np.where(C >= thresh)[0]
        
        # Aggregate gradient
        g = np.zeros(n)
        for m in active:
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            if hi >= lo:

        g_center = (g - g.mean())
        g_max = np.max(np.abs(g_center))
        if g_max < 1e-15:

        a_new = np.clip(a_new, 0, 1000)

C = conv(a, a), where C has length 2n-1.
C[m] = sum_{j: 0≤j<n, 0≤m-j<n} a[j] * a[m-j]

∂C[m]/∂a[i]:
The terms where a[i] appears:
- When j = i: a[i] * a[m-i] (valid if 0 ≤ m-i < n)
- When m-j = i, i.e., j = m-i: a[m-i] * a[i] (valid if 0 ≤ m-i < n)

If i ≠ m-i (i.e., 2i ≠ m): these are two distinct terms, each giving a[i]*a[m-i], so ∂C[m]/∂a[i] = 2*a[m-i]
If i = m-i (i.e., m = 2i): it's a single term a[i]^2, so ∂C[m]/∂a[i] = 2*a[i]

In both cases: ∂C[m]/∂a[i] = 2*a[m-i] (when 0 ≤ m-i < n), which is the same formula!
When m = 2i: 2*a[m-i] = 2*a[i] ✓

So the formula ∂C[m]/∂a[i] = 2*a[m-i] * I(0 ≤ m-i < n) is correct in all cases.

Now, for the gradient computation:
For a given m, the valid range of i is: max(0, m-n+1) ≤ i ≤ min(n-1, m)

When i is in this range, m-i ranges from m-min(n-1,m) to m-max(0,m-n+1) = 0 to n-1 (or less).

So the gradient contribution from peak m to position i is 2*a[m-i], which is 2*a shifted by m.

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses chirp-like and diverse initializations with FFT-based iterative gradient descent,
momentum, adaptive step sizing, and periodic random restarts.
"""

import numpy as np
import time


def _compute_conv(a):
    """Compute full linear convolution using FFT for efficiency."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def _fast_score(a):
    """Compute score quickly using FFT-based convolution."""
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    C = _compute_conv(a)
    return float(2.0 * len(a) * np.max(C) / (S * S))


def _gradient_from_peaks(a, C, active_indices):
    """Compute aggregate gradient from active convolution peaks.
    
    dC[m]/da[i] = 2*a[m-i] if 0 <= m-i < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    for m in active_indices:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if hi >= lo:
            indices = np.arange(lo, hi + 1)
            g[indices] += 2.0 * a[m - indices]
    return g


def _optimize_sequence(a_init, rng, deadline, verbose=False):
    """Iterative gradient optimization to minimize peak autocorrelation.
    
    Works with sum(a)=1 normalization (score is scale-invariant).
    Uses momentum, adaptive step size, and periodic restarts.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S  # Normalize to sum = 1

    best_a = a.copy()
    best_score = _fast_score(a)
    exact = evaluate_sequence(a.tolist())
    if np.isfinite(exact):
        best_score = min(best_score, exact)

    # Optimization state
    step = 1.0
    momentum = np.zeros(n)
    beta = 0.5
    no_improve = 0
    restart_counter = 0

    while time.monotonic() < deadline:
        # Compute convolution (one FFT per iteration)
        C = _compute_conv(a)
        max_C = C.max()

        # Active set: peaks within 3% of max
        thresh = max_C * 0.97
        active = np.where(C >= thresh)[0]

        # Aggregate gradient
        g = _gradient_from_peaks(a, C, active)

        # Center gradient (maintain sum constraint)
        g_center = g - np.mean(g)
        g_norm = np.max(np.abs(g_center))
        if g_norm < 1e-15:
            break

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * g_center / g_norm

        # Update
        a_new = a - step * momentum
        a_new = np.clip(a_new, 0.0, 1000.0)
        new_S = np.sum(a_new)
        if new_S < 1e-10:
            a_new = a
        else:
            a_new = a_new / new_S

        # Fast score check
        C_new = _compute_conv(a_new)
        max_new = C_new.max()
        new_score = 2.0 * n * max_new / new_S / new_S

        if new_score < best_score - 1e-9:
            a = a_new
            no_improve = 0
            step = min(3.0, step * 1.03)
            # Periodic exact verification
            if no_improve % 50 == 0 or restart_counter > 200:

                if np.isfinite(exact) and exact < best_score:
                    best_score = exact
                    best_a = a.copy()
        else:
            no_improve += 1
            step = max(0.005, step * 0.96)

            # Accept if at least not worse (for exploration)
            if new_score < _fast_score(a) + 0.01:
                a = a_new
            else:
                a = a  # Reject

            if no_improve > 80:

                scale = 0.05 * min(1.0, no_improve / 200.0)
                a = a * (1.0 + scale * rng.standard_normal(n))

                S2 = np.sum(a)
                if S2 > 1e-10:

                step = 1.0
                momentum[:] = 0.0
                no_improve = 0

            print(f"    iter: best={best_score:.6f}, step={step:.4f}, no_improve={no_improve}")

    # Final evaluation
    exact = evaluate_sequence(a.tolist())
    if np.isfinite(exact) and exact < best_score:
        best_score = exact
        best_a = a.copy()

    return best_a, best_score


def _hill_climb(a_init, rng, deadline):
    """Simple hill climbing with random perturbations as fallback."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    S = np.sum(a)
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    if not np.isfinite(best_score):
        return best_a, best_score

    for _ in range(50000):
        if time.monotonic() >= deadline:
            break

        # Random block perturbation
        width = int(rng.integers(max(1, n // 500), max(2, n // 20)))
        start = int(rng.integers(0, n - width))
        a_new = a.copy()
        a_new[start:start + width] *= np.exp(rng.normal(0, 0.15, min(width, a_new.size)))

        # Sometimes try swapping blocks
        if rng.random() < 0.3:
            w2 = int(rng.integers(1, n // 10 + 1))
            s2 = int(rng.integers(0, n - w2))
            if s2 != start or w2 != width:
                a_new[s2:s2 + w2] = a[start:start + min(width, w2)]
                if w2 > min(width, w2):
                    a_new[s2 + min(width, w2):s2 + w2] = a[start + min(width, w2):start + width] if width > min(width, w2) else a_new[s2 + min(width, w2):s2 + w2]

        a_new = np.clip(a_new, 0.0, 1000.0)
        S2 = np.sum(a_new)
        if S2 < 1e-10:
            continue
        a_new = a_new / S2

        score = evaluate_sequence(a_new.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score
