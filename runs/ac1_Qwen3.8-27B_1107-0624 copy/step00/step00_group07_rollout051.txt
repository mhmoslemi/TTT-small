import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time

def compute_convolution(f):
    """Compute full convolution using FFT."""
    n = len(f)
    size = 2*n - 1
    # Pad to next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = fft(f, fft_size)
    conv = ifft(F * F, fft_size).real[:size]
    return conv

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    # Build constraint matrix A where A[k, i] = f_{k-i} if valid
    # (f*g)_k = sum_i A[k,i] * g_i <= M
    
    # A is 2n-1 rows, n columns
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)_k = (f*g)_k."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        # Column i: A[k, i] = f[k-i] if 0 <= k-i < n
        # k ranges from i to i+n-1 (so that k-i ranges from 0 to n-1)
        A[i:i+n, i] = f
    return A
