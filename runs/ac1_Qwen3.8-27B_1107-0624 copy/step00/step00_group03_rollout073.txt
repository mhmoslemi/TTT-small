        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        s = s - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project
        s = np.clip(s, 0, None)
        S = s.sum()
        if S < 1e-10:
            s = best_s.copy()
            break
        s /= S
        
        # Periodic evaluation
        if t % 100 == 0:
            score = float(evaluate_sequence(s.tolist()))
            if np.isfinite(score) and score < best_score:
                if best_score < 1.6 and verbose:
                    print(f"    t={t}, score={best_score:.6f}, T={T:.0f}")
                best_score = score
                best_s = s.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        # Adapt
        if t % 500 == 0:
            T *= 1.1  # sharpen
            lr *= 0.95
        
        if stall_count > stall_limit:
            # Restart with perturbation
            s = best_s + 0.01 * rng.normal(0, 1, n)
            s = np.clip(s, 0, None)
            S = s.sum()
            if S > 0:
                s /= S
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            T = float(n)
            lr = 0.8
            stall_count = 0
    
    return best_s, best_score
