import numpy as np
import time

def compute_convolution(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    # Use FFT for efficient convolution
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[:size]
    return conv

def compute_gradient_active(a, active_lags):
    """Compute gradient of sum of b_k for k in active_lags, w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for k in active_lags:
        # Gradient of (a*a)_k = 2 * a[k - i] (with boundary)
        # For each i, the contribution is a[k-i] if valid, with factor 2 (or 1 if k=2i)
        if k == 0:

        else:
