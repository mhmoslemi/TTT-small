```python
import numpy as np
from scipy.optimize import linprog
import time

def build_toeplitz(a, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        idx = k - np.arange(j_min, j_max+1)
        A[k, j_min:j_max+1] = a[idx]
    return A

def lp_solve(a, M, n):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    A = build_toeplitz(a, n)
    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(2*n-1, M),
        bounds=[(0, None)]*n,
        method='highs'
    )
    return res.x if res.success else None

def objective(a):
    """Compute the objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    max_c = np.max(np.convolve(a, a))
    return 2.0 * n * max_c / (s*s)

def lp_iterate(a, n, max_iters=1000, time_limit=None):
    """Run LP-based iterations to improve the sequence."""
    deadline = time.monotonic() + time_limit if time_limit else np.inf
    a = a / np.sum(a)  # normalize to sum = 1
    best_a = a.copy()
    best_obj = objective(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv_a = np.convolve(a, a)
        M = np.max(conv_a)
        
        b = lp_solve(a, M, n)
        if b is None:
            break
        
        S = np.sum(b)
        if S <= 1.0001:
            break  # local optimum
        
        g = b / S
        
        # Line search on t
        best_t = 0.01
        best_conv_max = np.inf
        for t in [0.01, 0.05, 0.1, 0.2, 0.3]:
            a_t = (1-t)*a + t*g
            c_t = np.max(np.convolve(a_t, a_t))
            if c_t < best_conv_max:
                best_conv_max = c_t
                best_t = t
        
        a = (1-best_t)*a + best_t*g
        obj = 2*n*np.max(np.convolve(a, a))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
```
