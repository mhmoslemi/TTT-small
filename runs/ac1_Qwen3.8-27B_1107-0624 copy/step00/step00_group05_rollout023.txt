import numpy as np
import time

def compute_smoothed_grad(a, weights, n_fft):
    """Compute the smoothed gradient using FFT."""
    n = len(a)
    P = np.fft.rfft(weights, n_fft)
    A = np.fft.rfft(a, n_fft)
    corr = np.fft.irfft(P * np.conj(A), n_fft)
    return (2.0) * np.real(corr[:n])

def simplex_project(a):
    """Project onto the simplex {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones_like(a) / len(a)
    return a

def smooth_max_value(a, b, beta):
    """Compute the smoothed max of b."""
    b_max = np.max(b)
    return b_max + np.log(np.sum(np.exp(beta * (b - b_max)))) / beta

"""
Minimizes the peak autocorrelation of a non-negative sequence using
smoothed gradient descent with momentum on the probability simplex.
The objective 2n*max(conv(a,a))/sum(a)^2 is scale-invariant, so we
fix sum(a)=1 and minimize max(conv(a,a)).
"""

import numpy as np
import time


def _fft_convolve(a, n_fft):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    A = np.fft.rfft(a, n_fft)
    b = np.fft.irfft(A * A, n_fft)
    return b[:2 * n - 1]


def _fft_cross_correlate(p, a, n_fft):
    """Cross-correlation: out[m] = sum_k p[k+m]*a[k] for m=0,...,n-1."""
    n = len(a)
    P = np.fft.rfft(p, n_fft)
    A = np.fft.rfft(a, n_fft)
    corr = np.fft.irfft(P * np.conj(A), n_fft)
    return corr[:n]


def _project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s > 1e-15:
        return a / s
    return np.ones_like(a) / len(a)


def _smoothed_grad(a, b, beta, n_fft):
    """Compute gradient of (1/beta)*log(sum exp(beta*b)) w.r.t. a."""
    b_max = np.max(b)
    # Softmax weights (numerically stable)
    log_w = beta * (b - b_max)
    # Clip for numerical stability
    log_w = np.clip(log_w, -700.0, 700.0)
    w = np.exp(log_w)
    w_sum = np.sum(w)
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.ones_like(b) / len(b)
    
    # Gradient = (2/beta) * cross_correlate(w, a)
    corr = _fft_cross_correlate(w, a, n_fft)
    return (2.0 / beta) * corr


def _smoothed_max_value(b, beta):
    """Compute (1/beta)*log(sum exp(beta*b)) as smooth max approximation."""
    b_max = np.max(b)
    log_sum = np.log(np.sum(np.exp(np.clip(beta * (b - b_max), -700, 700))))
    return b_max + log_sum / beta


def _optimize_phase(a_init, n_fft, deadline, beta, lr, momentum=0.9, restart_interval=200, noise_scale=0.05):
    """Run one phase of smoothed gradient descent with momentum."""
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    v = np.zeros(n)
    local_best = a.copy()
    local_best_score = 2 * n * np.max(_fft_convolve(a, n_fft))
    steps = 0
    steps_since_improve = 0
    t_start = time.monotonic()
    
    while time.monotonic() < deadline:
        steps += 1
        
        # Compute convolution
        b = _fft_convolve(a, n_fft)
        b_max = np.max(b)
        
        # Compute gradient
        grad = _smoothed_grad(a, b, beta, n_fft)
        
        # Momentum update
        v = momentum * v - lr * grad
        a = a + v
        
        # Project to simplex
        a = _project_simplex(a)
        
        # Check for restart
        if steps % 20 == 0:
            score = 2 * n * b_max  # using current b (slightly stale but OK)
            # Recompute b for accuracy
            b = _fft_convolve(a, n_fft)
            score = 2 * n * np.max(b)
            if score < local_best_score:
                local_best_score = score
                local_best = a.copy()
                steps_since_improve = 0
            else:
                steps_since_improve += 20
        
        # Random restart / perturbation
        if steps_since_improve >= restart_interval:
            # Add noise
            a = a + noise_scale * np.random.randn(n)
            a = _project_simplex(a)
            v = np.zeros(n)
            steps_since_improve = 0
    
    return local_best, local_best_score, steps
