Optimizes step function coefficients to minimize 2*n*max(conv(f,f))/sum(f)^2.
Uses iterative peak reduction via multiplicative updates (mirror descent)
with adaptive step sizes, combined with block perturbations and restarts.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} via bisection."""
    if v.max() <= 0:
        return np.full(len(v), S / len(v))
    lo = 0.0
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.maximum(v - t, 0.0)


def _compute_pressure(f, active_mask):
    """Compute pressure vector: g[i] = sum_{k: active[k]} f[k-i].
    
    Uses the identity g[i] = (w * fR)[n-1+i] where fR is reversed f padded.
    """

    fR = np.zeros(c_len)
    fR[:n] = f[::-1]
    g_full = _fft_convolve(w, fR)
    return g_full[n - 1: 2 * n - 1]

The pressure vector $p_i = \sum_{k \in A} f_{k-i}$ (summed over active positions). The max of $p$ is at most $|A| \cdot \max f$. The typical value is $(|A| / (2n)) \cdot S$ (roughly).

For the multiplicative update to be effective, $\alpha \cdot p_i / p_{\max}$ should be of order 1 (so that some positions are significantly reduced). With $\alpha = 1$, this happens for positions where $p_i$ is close to $p_{\max}$.

After the update, the sum changes by a factor of $\exp(-\alpha \cdot \bar{p} / p_{\max})$ where $\bar{p}$ is the mean pressure. If $\bar{p} \approx p_{\max} / 2$, the sum is reduced by a factor of $e^{-0.5} \approx 0.6$. After renormalization, the relative changes are moderate.

For large $p$, this gradient is dominated by the largest $c_k$, so it's similar to the subgradient of the max.

"""
Optimizes step function coefficients to minimize 2*n*max(conv(f,f))/sum(f)^2.
Uses p-norm gradient descent with multiplicative updates (mirror descent),
multiple restarts with diverse initializations, and block perturbations.
"""

import numpy as np
import time


def _fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def _project_nonneg_sum(v, S):
    """Project onto {x >= 0, sum(x) = S}."""
    if v.max() <= 0:
        return np.full(len(v), S / len(v))
    lo = 0.0
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.maximum(v - t, 0.0)


def _gradient_pnorm(f, p):
    """Gradient of sum(c_k^p) where c = conv(f,f)."""
    n = len(f)
    c = _fft_convolve(f, f)
    c_pow = np.maximum(c, 0.0) ** (p - 1)
    fR = np.zeros(2 * n - 1)
    fR[:n] = f[::-1]
    g_full = _fft_convolve(c_pow, fR)
    return 2.0 * p * g_full[n - 1: 2 * n - 1]

    best_local = f.copy()
    best_local_score = float(evaluate_sequence(best_local.tolist()))
    
    alpha = alpha_init
    count = 0
    stalled = 0
    
    while time.monotonic() < deadline:
        count += 1
        g = _gradient_pnorm(f, p)
        g_max = g.max()
        if g_max < 1e-15:
            break
        
        # Multiplicative update (mirror descent)
        f_new = f * np.exp(-alpha * g / g_max)
        f_new = np.maximum(f_new, 0.0)

            f_new = f_new * (S / f_new_sum)

        alpha *= decay

                best_local = f.copy()
            else:
                stalled += 1
