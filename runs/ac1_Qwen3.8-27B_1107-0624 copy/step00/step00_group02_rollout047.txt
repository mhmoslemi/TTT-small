def project_simplex(v, target_sum=1.0):
    """Project vector v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v

    if abs(s - target_sum) < 1e-12:

    # Find theta such that sum(max(v_i - theta, 0)) = target_sum
    lo = 0.0
    hi = np.max(v)
    for _ in range(60):

        s_mid = np.sum(np.maximum(v - mid, 0))
        if s_mid > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2
    w = np.maximum(v - theta, 0.0)

    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None

def gradient_step(a, lr, soft_T=0.1):
    """
    Perform a gradient descent step on the soft-max objective.
    a is assumed to have sum = 1 and be non-negative.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    
    if soft_T < 1e-6:
        # Hard max: use only the argmax
        k_star = np.argmax(c)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        # Soft max: weighted average
        c_shifted = c - np.max(c)
        weights = np.exp(c_shifted / soft_T)
        weights = weights / np.sum(weights)
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            if weights[k] < 1e-10:
                continue
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += 2.0 * weights[k] * a[j]
    
    # Take step and project
    a_new = a - lr * grad
    a_new = project_simplex(a_new, 1.0)
    return a_new, np.max(c)

def optimize_with_lp(a_init, time_budget, seed=0):
    """Run LP-based optimization for the given time budget."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    
    # Normalize to sum = 1
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    deadline = time.monotonic() + time_budget
    t = 0.3  # step size
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = np.max(c)
        
        if M < 1e-15:
            break
        
        # Solve LP
        b = lp_direction(a, M)
        
        if b is None:
            # Fallback: gradient step
            a_new, _ = gradient_step(a, lr=0.01, soft_T=0.01)
            a_new = project_simplex(a_new, 1.0)
        else:
            sum_b = np.sum(b)
            if sum_b < 1e-10:
                break
            b_norm = b / sum_b  # normalize to sum = 1
            
            # Update
            a_new = (1 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = project_simplex(a_new, 1.0)
        
        obj = compute_obj(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            # Reset step size occasionally
            if obj < best_obj * 0.999:
                t = min(0.5, t * 1.1)
        else:
            t = max(0.01, t * 0.8)
        
        a = a_new
        
        if it % 50 == 0:
            print(f"  it={it}, obj={best_obj:.6f}, t={t:.4f}")
    
    return best_a, best_obj

def optimize_with_gradient(a_init, time_budget, seed=0):
    """Run gradient-based optimization for the given time budget."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a_init, dtype=np.float64).copy()
    n = len(a)
    
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    deadline = time.monotonic() + time_budget
    lr = 0.1
    soft_T = 1.0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = np.max(c)
        c_max = np.max(c)
        
        # Soft-max gradient
        c_shifted = c - c_max
        weights = np.exp(c_shifted / soft_T)
        weights = weights / np.sum(weights)
        
        # Compute gradient efficiently
        # grad[i] = 2 * sum_k weights[k] * a[k-i] (valid j)
        # This is like a weighted convolution
        grad = np.zeros(n)
        for k in range(2*n-1):
            if weights[k] < 1e-8:
                continue
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += 2.0 * weights[k] * a[k - j_hi : k - j_lo + 1][::-1]
        
        # Take step
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        
        obj = compute_obj(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            lr = min(0.5, lr * 1.05)
        else:
            lr = max(0.001, lr * 0.7)
        
        a = a_new
        
        # Anneal soft_T
        soft_T = max(0.01, soft_T * 0.995)
        
        if it % 100 == 0:
            print(f"  grad_it={it}, obj={best_obj:.6f}, lr={lr:.4f}, T={soft_T:.4f}")
    
    return best_a, best_obj

def make_u_shaped(n, beta=1.0):
    """Create a U-shaped (high at edges) sequence of length n."""
    i = np.arange(n, dtype=np.float64)
    if n > 1:
        x = 2.0 * i / (n - 1) - 1.0  # range [-1, 1]
    else:
        x = np.zeros(1)
    a = np.abs(x) ** beta + 0.1
    return a / np.sum(a)

def make_random_seq(n, seed, distribution='uniform'):
    """Create a random non-negative sequence of length n."""
    rng = np.random.default_rng(seed)
    if distribution == 'uniform':
        a = rng.random(n)
    elif distribution == 'exponential':
        a = rng.exponential(1.0, n)
    elif distribution == 'beta':
        a = rng.beta(0.5, 0.5, n)  # U-shaped
    elif distribution == 'bimodal':
        a = rng.exponential(1.0, n)
        # Emphasize edges
        edge_weight = np.ones(n)
        edge_weight[:n//4] = 3.0
        edge_weight[3*n//4:] = 3.0
        a = a * edge_weight
    else:
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    else:
        a = a / s
    return a

def spectral_shaping(a, n_iter=10, seed=0):
    """Apply spectral domain shaping to flatten the convolution."""
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    L = 2 * n  # FFT length (>= 2n-1)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for it in range(n_iter):
        # Zero-pad and FFT
        A = np.fft.rfft(a, n=L)
        
        # The convolution's spectrum is |A|^2
        # We want |A|^2 to be flat, so we want |A| to be flat
        # Modify A to have more uniform magnitude while preserving phase
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude
        target_mag = np.mean(mag)
        
        # Blend between current and target
        alpha = 0.3
        new_mag = (1 - alpha) * mag + alpha * target_mag
        # Ensure DC component (A[0] = sum(a)) is preserved
        new_mag[0] = mag[0]
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n=L)[:n]
        
        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if np.isfinite(obj) and obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
        else:
            alpha = max(0.01, alpha * 0.5)
    
    return best_a

def refine_with_perturbation(a, time_budget, seed=0):
    """Local search with random perturbations."""
    rng = np.random.default_rng(seed)
    a = np.asarray(a, dtype=np.float64).copy()
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    deadline = time.monotonic() + time_budget
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Generate perturbation
        a_pert = a.copy()
        
        # Random block scaling
        if n > 10:
            width = max(1, n // rng.integers(20, 200))
            start = rng.integers(0, n - width + 1)
            scale = np.exp(rng.normal(0, 0.1))
            a_pert[start:start+width] *= scale
        
        # Random element scaling
        idx = rng.integers(0, n)
        a_pert[idx] *= np.exp(rng.normal(0, 0.05))
        
        # Symmetrize with some probability
        if rng.random() < 0.3:
            a_pert = 0.5 * (a_pert + a_pert[::-1])
        
        a_pert = np.maximum(a_pert, 0.0)
        s_pert = np.sum(a_pert)
        if s_pert < 1e-10:
            continue
        a_pert = a_pert / s_pert
        
        obj = compute_obj(a_pert)
        if np.isfinite(obj) and obj < best_obj:
            best_a = a_pert.copy()
            best_obj = obj
            a = a_pert
    
    return best_a, best_obj
