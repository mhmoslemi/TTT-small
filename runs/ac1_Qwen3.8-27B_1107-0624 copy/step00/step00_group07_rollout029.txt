    def get_score(a):
        s = a.sum()
        if s < 0.01:
            return np.inf
        b = np.convolve(a, a)
        return 2.0 * n * b.max() / (s * s)
    
    def gradient_step(a, eta):
        """One step of gradient descent on sum((b-mu)^2)."""
        a = a / a.sum()  # Normalize
        b = np.convolve(a, a)
        c = b - b.mean()
        # Compute gradient: grad[j] = 4 * dot(c[j:j+n], a)
        window = as_strided(c, shape=(n, n), strides=(c.strides[0], c.strides[0]))
        grad = 4.0 * (window @ a)
        grad = grad - grad.mean()
        # Adaptive step
        norm_g = np.linalg.norm(grad) / np.sqrt(n)
        step = eta / (norm_g + 1e-10)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        return a_new
    
    def lp_step(a, M, active_threshold=0.9):
        """Solve LP to find a direction that maintains ||a*g||_inf <= M."""
        b = np.convolve(a, a)
        # Select active constraints
        active_mask = b >= M * active_threshold
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            return a  # No constraints, shouldn't happen

        constraints = []

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            row[lo:hi+1] = a[k-hi:k-lo:-1]
            constraints.append(row)

            g = g / g.sum()  # Normalize
            return g

For non-negative sequences, the convolution is always non-negative, and the "edge effects" (the triangular shape at the boundaries) make it hard to achieve a perfectly flat convolution. The best we can do is have a convolution that's flat in the middle but tapered at the edges.

If the convolution is `b_k = mu` for `k = K1, ..., K2` (a flat region) and tapers smoothly to 0 at the edges, then `M = mu` and `sum(b) = mu * (K2-K1+1) + (edge contributions) = S^2`.

The fraction of the convolution that's in the flat region determines how close we get to the lower bound. If a fraction `f` of the convolution is at the max value, then `M/S^2 ≈ 1/(f*(2n-1))`, giving score ≈ `2n/(f*(2n-1))`.

So for score = 1.5, we need `r ≈ 1.5`, meaning the max is 1.5 times the average. That's quite flat!

For the uniform sequence, `M = n*a^2` and `S^2/(2n-1) = n^2*a^2/(2n-1)`, so `r = n*(2n-1)/n^2 ≈ 2`. Hence score ≈ 2. ✓
