def build_conv_matrix(s, active_lags):

    A[k, j] = s[k-j] if 0 <= k-j < n else 0

def _build_lp_matrix(s, active_lags):
    """Build the LP constraint matrix for given active lags."""
    n = len(s)
    A = np.zeros((len(active_lags), n), dtype=np.float64)
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[i, j_start:j_end+1] = s[k : k - (j_end - j_start) - 1 : -1]
    return A


def _lp_direction(s, M, active_lags, time_limit=10.0):
    """Solve LP for best direction."""
    n = len(s)
    A = _build_lp_matrix(s, active_lags)
    b = np.full(len(active_lags), M, dtype=np.float64)

                      options={'time_limit': time_limit, 'presolve': True})

        pass
    return None

    """Gradient step on soft-max objective."""
    n = len(s)
    S = np.sum(s)
    c = np.convolve(s, s)
    
    # Use only top n_lags for efficiency
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_lags]
    c_active = c[active]
    
    # Softmax weights
    c_max = np.max(c_active)
    weights = np.exp(T * (c_active - c_max))
    weights /= np.sum(weights)
    
    # Weighted subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += w * 2.0 * s[k : k - (j_end - j_start) - 1 : -1]
    
    # Full gradient of f = 2n*soft_max/S^2
    # Simplified: direction = grad - (weighted_avg_c)/S * ones
    wavg_c = np.sum(weights * c_active)

Combines LP, soft-gradient, multiplicative updates, and exploration.

import numpy as np
import time
from scipy.optimize import linprog


def _conv_matrix_row(s, k):
    """Get the k-th row of the convolution matrix (as a sparse-like vector)."""
    n = len(s)
    row = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        row[j_start:j_end+1] = s[k : k - (j_end - j_start) - 1 : -1]
    return row


def _lp_solve(s, M, active_lags, time_limit=10):
    """Solve the LP direction-finding problem."""
    n = len(s)
    nc = len(active_lags)
    A = np.zeros((nc, n))
    for i, k in enumerate(active_lags):
        A[i] = _conv_matrix_row(s, k)
    b = np.full(nc, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
