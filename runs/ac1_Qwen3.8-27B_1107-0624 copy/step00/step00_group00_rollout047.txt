def compute_conv_and_grad(a, T):
    """Compute convolution and gradient of smoothed max."""
    n = len(a)
    conv = np.convolve(a, a, 'full')
    max_c = np.max(conv)
    T = max(T, 1e-12)
    # Numerically stable softmax
    shifted = conv / T - max_c / T
    exp_shifted = np.exp(shifted)
    w_sum = np.sum(exp_shifted)
    if w_sum > 1e-300:
        w = exp_shifted / w_sum
