"""
Multi-start smoothed gradient descent (log-sum-exp of the self-convolution)
with simplex projection, adaptive step size, simulated-annealing exploration,
and periodic LP-based direction finding.

Goal: minimize  n = len(a)  ->  2*n*max_k (a*a)[k] / (sum a)^2 .
The objective is scale invariant, so we work on the simplex {a>=0, sum a = 1}
where it reduces to 2*n*max_k (a*a)[k]  ~  peak-to-average ratio of the
self-convolution.  Low-PAR (radar/ communications) shaped sequences are the
target; we find them numerically from several analytic/random starts.
"""

import numpy as np
import time
from scipy.fft import fft, ifft

try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:  # pragma: no cover
    _HAS_LINPROG = False


# --------------------------------------------------------------------------- #
# Core numerical helpers (all top-level, no closures / lambdas)
# --------------------------------------------------------------------------- #
def full_convolve_fft(a, b):
    """Full linear convolution of 1-D arrays using an FFT of next power of 2."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = fft(a, size)
    fb = fft(b, size)
    out = np.real(ifft(fa * fb))[:n]
    # FFT round-off can give tiny negative values; they are harmless downstream.
    return out


def project_simplex(v):
    """Euclidean projection of v onto the probability simplex {x>=0, 1.x=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1) > 0
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(0.0, v - theta)


def compute_objective(a):
    """Objective on the simplex (sum a == 1): 2*n*max_k (a*a)[k]."""
    b = full_convolve_fft(a, a)
    return 2.0 * len(a) * float(np.max(b))


def smooth_grad(a, epsilon):
    """
    Smoothed (log-sum-exp) surrogate of max_k (a*a)[k] and its gradient.

    Returns (surrogate_value, grad, weights).  The gradient is
        grad[j] = 2 * sum_k w[k] * a[k-j]
    computed by one more FFT convolution of the flipped weights with a.
    """
    n = len(a)
    b = full_convolve_fft(a, a)
    b_max = np.max(b)
    shifted = (b - b_max) / epsilon
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_b = np.exp(shifted)
    s = np.sum(exp_b)
    w = exp_b / s
    value = b_max + epsilon * np.log(s)
    w_flip = w[::-1]
    conv_wa = full_convolve_fft(w_flip, a)
    grad = 2.0 * conv_wa[n - 1:2 * n - 1][::-1]
    return value, grad, w


def lp_direction(a, n_active=200):
    """
    Solve the SoT linear program:
        maximize  sum b_j
        s.t.      (a*b)[k] <= M  for the top-``n_active`` active lags k
                  b >= 0
    with M = max(a*a).  The solution is a nonnegative sequence that, when
    convolved with a, stays under the current peak; mixing a toward it is a
    globally informed descent direction (helps escape local minima).
    """
    if not _HAS_LINPROG:
        return None
    n = len(a)
    b_aa = full_convolve_fft(a, a)
    M = float(np.max(b_aa))
    # Keep only the tightest / most active constraints to keep the LP tractable.
    order = np.argsort(b_aa)[::-1]
    active = order[:min(n_active, len(order))]
    A_ub = np.zeros((len(active), n))
    for row, k in enumerate(active):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        A_ub[row, i_min:i_max + 1] = a[k - i_max:k - i_min + 1][::-1]
    b_ub = np.full(len(active), M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method="highs",
                      options={"presolve": True})
    except Exception:
        return None
    if res is not None and res.success and res.x is not None:
        xv = np.clip(res.x, 0.0, None)
        if np.sum(xv) > 0:
            return xv / np.sum(xv)  # normalise to the simplex for mixing
    return None


# --------------------------------------------------------------------------- #
# Start generators
# --------------------------------------------------------------------------- #
def start_uniform(n):
    return np.ones(n) / n


def start_perturbed(n, rng, amp=0.05):
    v = np.ones(n) + amp * rng.standard_normal(n)
    v = np.clip(v, 0.0, None)
    v = v / np.sum(v)
    return v


def start_sin(n, rng):
    j = np.arange(n)
    freq = rng.uniform(0.3, 1.5)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    amp = rng.uniform(0.2, 0.5)
    v = 1.0 + amp * np.cos(2.0 * np.pi * freq * j / n + phase)
    v = np.clip(v, 0.0, None)
    v = v / np.sum(v)
    return v


def start_chirp(n, rng):
    j = np.arange(n)
    rate = rng.uniform(0.5, 2.0)
    v = 0.5 + 0.5 * np.cos(2.0 * np.pi * rate * j * j / (2.0 * n))
    v = v / np.sum(v)
    return v


def start_random(n, rng):
    v = np.abs(rng.standard_normal(n)) + 1e-9
    return v / np.sum(v)

def gradient_phase(a, t_end, seed, tag=""):
    """
    Annealed smoothed-gradient descent from a start until t_end.
    Returns the best (on-simplex) sequence and its objective seen in the phase.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    a = a.copy()
    obj = compute_objective(a)
    best_a = a.copy()
    best_obj = obj
    epsilon = 0.08
    step = 0.004
    stuck = 0
    lp_counter = 0
    last_lp = t0
    while time.monotonic() < t_end:
        a_new = project_simplex(a - step * smooth_grad(a, epsilon)[1])
        obj_new = compute_objective(a_new)
        if obj_new < obj:
            a = a_new
            obj = obj_new
            stuck = 0
            step = min(0.04, step * 1.02)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if tag:
                    print("    [%s] obj=%.6f  eps=%.4f t=%.1fs"
                          % (tag, best_obj, epsilon, time.monotonic() - t0))
        else:
            stuck += 1
            step *= 0.6
            # Occasionally blend in an LP-derived direction to escape basins.
            lp_counter += 1
            if stuck > 40 and lp_counter % 7 == 0 and time.monotonic() - last_lp > 6.0:
                last_lp = time.monotonic()
                d = lp_direction(a, n_active=200)
                if d is not None:
                    for mix in (0.3, 0.5, 0.7):
                        cand = project_simplex((1.0 - mix) * a + mix * d)
                        oc = compute_objective(cand)
                        if oc < obj:
                            a = cand
                            obj = oc
                            stuck = 0
                            if oc < best_obj:
                                best_obj = oc
                                best_a = a.copy()
                            break
            if stuck > 120:
                # gentle random kick to leave a flat region
                a = project_simplex(a + 1e-3 * rng.standard_normal(len(a)))
                stuck = 60
        # anneal the surrogate temperature
        el = time.monotonic() - t0
        epsilon = max(0.0015, 0.08 * np.exp(-el / 150.0))
        if step < 1e-5 and best_obj < 1.55:
            break
    return best_a, best_obj


def anneal_phase(a, t_end, seed):
    """
    Simulated-annealing over block multiplicative perturbations (scale invariant
    on the simplex after projection).  Acts as a global explorer between
    gradient phases.
    """
    rng = np.random.default_rng(seed)
    a = a.copy()
    obj = compute_objective(a)
    best_a = a.copy()
    best_obj = obj
    T = 0.02
    t0 = time.monotonic()
    while time.monotonic() < t_end:
        n = len(a)
        width = max(2, int(np.exp(rng.uniform(np.log(3), np.log(max(4, n // 10))))))
        start = int(rng.integers(0, n - width + 1))
        cand = a.copy()
        cand[start:start + width] *= np.exp(rng.uniform(-0.4, 0.4))
        cand = project_simplex(cand)
        obj_c = compute_objective(cand)
        d = obj_c - obj
        accept = d < 0 or rng.random() < np.exp(-d / T)
        if accept:
            a = cand
            obj = obj_c
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print("    [anneal] obj=%.6f  T=%.4f t=%.1fs"
                      % (best_obj, T, time.monotonic() - t0))
        T = max(3e-4, T * 0.985)
    return best_a, best_obj

    best_a = start_uniform(n)
    best_obj = float(evaluate_sequence(best_a.tolist()))

        ("perturb", start_perturbed(n, rng, 0.15)),
        ("sin",     start_sin(n, rng)),
        ("sin",     start_sin(n, rng)),
        ("chirp",   start_chirp(n, rng)),

            pass

    # ---- interleave short gradient + anneal phases over the starts -------------
    n_starts = len(starts)
    # time per slot scales down if we have many starts
