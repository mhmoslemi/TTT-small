import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix

def build_convolve_matrix(n, f):
    """Build the sparse matrix A such that (A @ g)[k] = (convolve(f, g))[k]."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            val = f[k - j]
            if val != 0.0:
                rows.append(k)
                cols.append(j)
                data.append(val)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    if rows:
        A[rows, cols] = data
    return A.tocsr()

def solve_direction_lp(f, M, n):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    A = build_convolve_matrix(n, f)
    c_obj = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None
