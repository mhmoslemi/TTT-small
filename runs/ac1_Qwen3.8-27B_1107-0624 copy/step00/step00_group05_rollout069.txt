"""
Minimize  2*n*max(convolve(a,a)) / (sum(a))^2  over non-negative sequences a.

The objective is scale-invariant, so we optimize on the probability simplex
(sum(a)=1) and minimize  2*n*max(c)  with c=convolve(a,a), then rescale at the end.

Method:
  * Smooth the non-smooth max with a log-sum-exp (soft-max) of the autocorrelation.
  * Compute the exact gradient of that smooth surrogate in O(n log n) using an FFT
    (the gradient is a twice-weighted cross-correlation of a with the soft-max
    weights over the autocorrelation).
  * Perform projected gradient steps on the simplex: subtract the mean of the
    gradient (tangent-space projection onto sum=1), clip to non-negative, renormalize,
    with a backtracking line search.
  * Explore several diverse (deliberately non-palindromic) initializations,
    anneal the soft-max sharpness, and restart with random perturbations on stagnation.
  * Return the best sequence found (scaled and verified with evaluate_sequence).
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m (m >= 1)."""
    p = 1
    while p < m:
        p <<= 1
    return p


def _softmax_weights(c, beta):
    """Soft-max(beta*c) as a probability vector, numerically stable."""
    bc = beta * c
    bc_max = float(np.max(bc))
    e = np.exp(bc - bc_max)
    s = float(np.sum(e))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _smooth_max_grad(a, w, fft_size):
    """
    Gradient of  (1/beta)*log(sum_k exp(beta*c_k))  w.r.t. a, where c=convolve(a,a).
    Equals  2 * sum_m a[m] * w[m+i]  for i = 0..n-1  (a cross-correlation).
    Computed with one FFT-based linear cross-correlation.
    """
    n = len(a)
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, n=fft_size)
    W = np.fft.rfft(w, n=fft_size)
    conv = np.fft.irfft(A * W, n=fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def _normalized_obj(a):
    """2*n*max(convolve(a,a)) (assumes a is arbitrary; we use it on sum-1 vectors)."""
    n = len(a)
    c = np.convolve(a, a)
    return float(np.max(c)) * (2.0 * n)


def _clean_start(s, n):
    """Clip to non-negative and normalize to sum 1; uniform fallback."""
    s = np.asarray(s, dtype=np.float64)
    s = np.clip(s, 0.0, None)
    tot = float(np.sum(s))
    if tot < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return s / tot


def _optimize(a0, time_budget, seed=0, verbose=True):
    """Projected gradient descent (simplex) with backtracking and restarts."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = _clean_start(a0, n)

    fft_size = _next_pow2(3 * n - 2)

    best = a.copy()
    best_raw = float(np.max(np.convolve(a, a)))
    best_obj = 2.0 * n * best_raw

    beta = 25.0
    lr = 0.02
    start_time = time.time()
    deadline = start_time + max(1.0, float(time_budget))
    it = 0
    stall = 0
    max_stall = 300
    local_count = 0

    while time.time() < deadline:
        c = np.convolve(a, a)
        cur_raw = float(np.max(c))
        cur_obj = 2.0 * n * cur_raw

        w = _softmax_weights(c, beta)
        g = _smooth_max_grad(a, w, fft_size)
        g_proj = g - float(np.mean(g))  # project onto sum=0 tangent space

        # Backtracking line search (Armijo-lite: accept any decrease in max(c)).
        accepted = False
        obj_new = cur_obj
        for _ls in range(8):
            trial = a - lr * g_proj
            trial = np.clip(trial, 0.0, None)
            tot = float(np.sum(trial))
            if tot < 1e-15:
                lr *= 0.5
                continue
            trial = trial / tot
            trial_raw = float(np.max(np.convolve(trial, trial)))
            cand_obj = 2.0 * n * trial_raw
            if cand_obj < cur_obj:
                a = trial
                cur_obj = cand_obj
                cur_raw = trial_raw
                obj_new = cand_obj
                accepted = True
                break
            lr *= 0.5

        if accepted:
            local_count += 1
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_obj = cur_obj
                best = a.copy()
                stall = 0
                lr = min(lr * 1.15, 0.6)
            else:
                stall += 1
        else:
            stall += 1
            lr = max(lr * 0.5, 1e-6)

        # Stagnation handling: restart from best with a scaled random perturbation.
        if stall > max_stall:
            amp = 0.04 + 0.10 * float(rng.random())
            a = best + amp * rng.standard_normal(n)
            a = _clean_start(a, n)
            lr = 0.02
            beta = float(rng.choice([10.0, 25.0, 50.0, 100.0, 200.0, 400.0]))
            stall = 0
            max_stall = max(60, max_stall // 2)

        # Gentle annealing of the soft-max sharpness.
        if it and it % 400 == 0:
            beta = min(beta * 1.15, 800.0)

        it += 1
        if verbose and it % 5000 == 0:
            elapsed = time.time() - start_time
            print("    it=%d best_obj=%.6f raw=%.6f lr=%.5f beta=%.0f t=%.0fs"
                  % (it, best_obj, best_raw, lr, beta, elapsed))

    return best, best_obj
