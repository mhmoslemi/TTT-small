import numpy as np
import time

def _fft_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    out_len = 2 * n - 1
    N = 1
    while N < out_len:
        N *= 2
    fa = np.fft.rfft(a, N)
    fc = np.fft.irfft(fa * fa, N)
    return fc[:out_len]

def _cross_correlation(p, a):
    """Compute cross-correlation: r[j] = sum_k p[k] * a[k-j] for j=0..n-1."""
    n = len(a)
    n_fft = 1
    while n_fft < len(p) + n:
        n_fft *= 2
    P = np.fft.rfft(p, n_fft)
    A = np.fft.rfft(a, n_fft)
    r = np.fft.irfft(P * np.conj(A), n_fft)
    return r[:n]

def _softmax_grad_step(a, beta, S):
    """Compute gradient of soft-approximated objective."""
    n = len(a)
    c = _fft_convolve(a)
    # Compute M and p
    max_c = np.max(c)
    exp_bc = np.exp(beta * (c - max_c))
    sum_exp = np.sum(exp_bc)
    M = (max_c + np.log(sum_exp)) / beta
    p = exp_bc / sum_exp
    # Cross-correlation
    gamma = _cross_correlation(p, a)
    # Gradient
    grad = (2.0 * n / (S ** 3)) * (S * gamma - 2.0 * M)
    return grad, M, c

    A = np.fft.rfft(a, n_fft)

    return r[:n]

    n = len(a)
    S = float(np.sum(a))
    c = _fft_convolve_self(a)
    # Log-sum-exp for numerical stability
    max_c = float(np.max(c))

    sum_exp = float(np.sum(exp_shifted))

    p = exp_shifted / sum_exp
    gamma = _fft_cross_corr(p, a)

        m_adam = beta1 * m_adam + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * grad * grad
        mh = m_adam / (1.0 - beta1 ** t_adam)
        vh = v_adam / (1.0 - beta2 ** t_adam)
        lr = lr0 / (1.0 + 0.003 * t_adam)

        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.clip(a, 0.0, 1000.0)
        a = _norm(a, 1.0)

        # Anneal softmax temperature linearly
        beta_s = min(beta_s + 0.02, beta_s_max)

        # Internal fast check
        if it % eval_every == 0:
            sc_fast = _fast_score(a)
            # Full evaluation less frequently
            if it % (eval_every * 5) == 0:
                sc = float(evaluate_sequence(a.tolist()))
            else:
                sc = sc_fast

            if np.isfinite(sc) and sc < lbest_sc - 1e-9:
                lbest = a.copy()
                lbest_sc = float(evaluate_sequence(lbest.tolist()))
                stuck = 0
            else:
                stuck += 1

            # Escape mechanism
            if stuck > 60:
                k = max(1, n // 6)
                ii = rng.choice(n, k, replace=False)
                a[ii] *= np.exp(rng.normal(0, 0.5, k))
                a = np.clip(a, 0.0, 1000.0)
                a = _norm(a, 1.0)
                beta_s = max(1.0, beta_s * 0.3)
                lr0 = lr0 * 1.5
                stuck = 0

    return lbest, lbest_sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-restart gradient descent with annealed softmax to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    Exploits scale invariance by working at sum=1.
    Uses FFT-based gradient computation for O(n log n) per step.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    # --- Load parent sequence ---
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if p_arr.ndim == 1 and p_arr.size >= 50 and np.all(np.isfinite(p_arr)):
            parent = np.clip(p_arr, 0.0, 1000.0)
    except Exception:
        pass
    if parent is None:
        parent = np.ones(512)

    n = len(parent)
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: n={n}, score={best_score:.6f}")
    t0 = time.monotonic()

    # --- Build diverse initializations ---
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("dirichlet", rng.dirichlet(np.ones(n))))
    h = rng.random(n // 2 + 1)
    inits.append(("sym_random", np.concatenate([h, h[::-1][:n - n // 2 - 1]])))
    inits.append(("cosine", 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))))
    inits.append(("gaussian", np.exp(-0.5 * ((np.arange(n) - n / 2.0) / (n / 4.0)) ** 2)))
    inits.append(("asym", np.concatenate([rng.random(n // 2), 0.6 * rng.random(n - n // 2)])))
    # Windowed uniform (tapered edges)
    w = np.linspace(0, 1, n)
    win = 0.3 + 0.7 * np.sin(np.pi * w) ** 0.5
    inits.append(("windowed", win))

    for idx, (name, init) in enumerate(inits):
        if time.monotonic() > deadline - 30:
            break
        rem = deadline - time.monotonic()
        # Allocate time: give more to earlier restarts
        frac = max(0.8, 1.0 - 0.12 * idx)
        t_budget = rem * frac
        t_stop = time.monotonic() + min(t_budget, max(120.0, 600.0 - 60.0 * idx))

        print(f"  Restart {idx} ({name}): budget={min(t_budget, max(120.0, 600.0 - 60.0 * idx)):.0f}s")
        lbest, lbest_sc = _adam_restart(init, n, t_stop, rng)

        if lbest_sc < best_score - 1e-9:
            best = lbest.copy()
            best_score = lbest_sc
            elapsed = time.monotonic() - t0
            print(f"    NEW BEST: {lbest_sc:.6f} (elapsed {elapsed:.0f}s)")
        else:
            print(f"    Final: {lbest_sc:.6f} (no improvement)")

    # --- Final polish: coordinate descent on best ---
    if time.monotonic() < deadline - 30:
        polish_deadline = deadline
        a = _norm(best.copy(), 1.0)
        a_sc = float(evaluate_sequence(a.tolist()))
        improved = True
        while improved and time.monotonic() < polish_deadline:
            improved = False
            for j in range(n):
                if time.monotonic() > polish_deadline:
                    break
                old_val = a[j]
                # Try a few candidate values
                for mult in [0.5, 0.7, 1.3, 1.5, 0.8, 1.2]:
                    a[j] = old_val * mult
                    sc = _fast_score(a)
                    if sc < a_sc - 1e-8:
                        a_sc = sc
                        improved = True
                a[j] = old_val  # revert and recompute properly
                # Actually do proper 1D optimization: try golden section
                # For simplicity, just try the best multiplier
                best_mult = 1.0
                best_sc_j = a_sc
                for mult in np.linspace(0.5, 1.5, 11):
                    a[j] = old_val * mult
                    sc_j = _fast_score(a)
                    if sc_j < best_sc_j:
                        best_sc_j = sc_j
                        best_mult = mult
                a[j] = old_val * best_mult
                a[j] = max(0.0, a[j])
                # Renormalize
                a = _norm(a, 1.0)
                a_sc = _fast_score(a)

        a_sc_exact = float(evaluate_sequence(a.tolist()))
        if a_sc_exact < best_score - 1e-9:
            best = a.copy()
            best_score = a_sc_exact
            print(f"    Polish improved: {best_score:.6f}")

    elapsed = time.monotonic() - t0
    print(f"Done in {elapsed:.0f}s. Final score: {best_score:.6f}, n={len(best)}")
    return [float(x) for x in best]
