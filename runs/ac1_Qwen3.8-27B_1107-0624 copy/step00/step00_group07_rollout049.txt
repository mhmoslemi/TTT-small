def fft_convolve(a, b=None):
    """Compute convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)[:size]

def make_initial_shape(n, shape_type='cosine'):
    """Create an initial shape."""
    if shape_type == 'uniform':
        return np.ones(n)
    elif shape_type == 'cosine':
        k = np.arange(n)
        return 0.5 * (1 + np.cos(2 * np.pi * k / (n + 1)))
    elif shape_type == 'hann':
        k = np.arange(n)
        return 0.5 * (1 - np.cos(2 * np.pi * k / (n - 1)))
    elif shape_type == 'parabolic':
        k = np.arange(n)
        x = 2 * k / (n - 1) - 1
        return 1 - x**2
    elif shape_type == 'quartic':
        k = np.arange(n)
        x = 2 * k / (n - 1) - 1
        return 1 - x**4
    elif shape_type == 'flat_top':

    else:
        return np.ones(n)

    conv_vals = fft_convolve(a, a)
    M = conv_vals.max()
    near_max = np.where(conv_vals >= M * (1 - rel_tol))[0]

    for j in near_max:
        lo = max(0, j - n + 1)
        hi = min(n, j + 1)
        subgrad[lo:hi] += a[j - hi + 1 : j - lo + 1][::-1]

    momentum_vec = momentum_coeff * momentum_vec + (1 - momentum_coeff) * subgrad
    
    # Gradient step
    a_new = a - lr * momentum_vec
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new, M, momentum_vec

def optimize_shape(a_init, n_steps, lr_init, lr_decay, momentum_coeff, rel_tol):
    """Run subgradient descent to minimize max(conv(a,a)) with sum(a)=1."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    momentum_vec = np.zeros(n)

        if M < best_M:
            best_M = M
            best_a = a.copy()

def solve_lp(a, M):

    # Row k (0-indexed), column j: value is a[k-j] if 0 <= k-j < n

Now, can we get below 2? We need 2n * max_b / sum_a^2 < 2, i.e., max_b < sum_a^2 / n.

For the uniform sequence, max_b = n and sum_a^2/n = n^2/n = n. So max_b = sum_a^2/n, giving score = 2.

To get below 2, we need max_b < sum_a^2/n.

But we just showed that conv(a,a)_{n-1} = Σ_k a_k * a_{n-1-k} and by Cauchy-Schwarz, (Σ_k a_k * a_{n-1-k})^2 <= (Σ_k a_k^2)(Σ_k a_{n-1-k}^2) = (Σ_k a_k^2)^2.

So conv(a,a)_{n-1} <= Σ_k a_k^2 (for non-negative a).

And by Cauchy-Schwarz again, Σ_k a_k^2 >= (Σ_k a_k)^2 / n.

So conv(a,a)_{n-1} <= Σ_k a_k^2, and Σ_k a_k^2 >= sum_a^2/n.

This doesn't directly give us max_b < sum_a^2/n. In fact, for the uniform sequence, we achieve equality in both: conv(a,a)_{n-1} = n = Σ a_k^2 and Σ a_k^2 = n = sum_a^2/n.
