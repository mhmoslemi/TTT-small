"""
LP-guided local search minimizing the autocorrelation peak ratio of a step function.
Iteratively solves an LP to find descent directions (maximize sum subject to
bounded cross-correlation), uses line search, multiple initializations, and
noise injection to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score_fast(s):
    """Compute the evaluation score using FFT for speed."""
    n = len(s)
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(s))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_lp_structure(n):
    """Precompute sparse structure for the LP constraint matrix (s*b)[k]."""
    row_list = []
    col_list = []
    s_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row_list.append(k)
            col_list.append(j)
            s_list.append(k - j)
    return (np.asarray(row_list, dtype=np.int32),
            np.asarray(col_list, dtype=np.int32),
            np.asarray(s_list, dtype=np.int32))


def solve_lp_direction(s, row_idx, col_idx, s_idx, time_limit=60):
    """Solve LP: max sum(b) s.t. (s*b)[k] <= M, b >= 0. Returns (direction, M)."""
    n = len(s)
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    M = float(np.max(conv))
    if M < 1e-15:
        return None, M
    vals = s[s_idx]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None, M
    if result.x is not None and float(np.sum(result.x)) > 0:
        return np.maximum(result.x, 0.0), M
    return None, M


def heuristic_direction(s):
    """Fallback: build a direction from the peak-lag gradient."""
    n = len(s)
    size = 2 * n - 1
    f_s = np.fft.rfft(s, n=size)
    conv = np.fft.irfft(f_s * f_s, n=size)
    conv = np.maximum(conv, 0.0)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = s[j]
    direction = s - 0.1 * grad
    return np.maximum(direction, 0.0)


def line_search(s, direction, steps):
    """Find best step size from candidates via evaluation."""
    best_t = steps[0]
    best_score = compute_score_fast(s)
    for t in steps:
        s_trial = (1.0 - t) * s + t * direction
        s_trial = np.clip(s_trial, 0.0, 1000.0)
        if float(np.sum(s_trial)) < 0.01:
            continue
        sc = compute_score_fast(s_trial)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t, best_score


def run_optimization(s_init, deadline, rng, label, verbose=True):
    """Run LP-based local search from initial sequence until deadline."""
    n = len(s_init)
    s = s_init.copy()
    row_idx, col_idx, s_idx = build_lp_structure(n)
    best_s = s.copy()
    best_score = compute_score_fast(s)
    if verbose:
        print(f"  [{label}] n={n}, initial_score={best_score:.6f}")
    normal_steps = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    fine_steps = [0.05, 0.1, 0.15, 0.2, 0.3]
    iteration = 0
    no_improve = 0
    max_no_improve = 20
    while time.monotonic() < deadline:
        iteration += 1
        t0 = time.monotonic()
        direction, M = solve_lp_direction(s, row_idx, col_idx, s_idx, time_limit=60)
        if direction is None or float(np.sum(direction)) <= 0:
            direction = heuristic_direction(s)
            if float(np.sum(direction)) <= 0:
                break
        if iteration < 30:
            steps = normal_steps
        else:
            steps = fine_steps
        best_t, trial_score = line_search(s, direction, steps)
        cur_score = compute_score_fast(s)
        if trial_score >= cur_score + 1e-10:
            no_improve += 1
            if no_improve >= max_no_improve:
                s = s * np.exp(rng.normal(0, 0.04, n))
                s = np.clip(s, 0.0, 1000.0)
                no_improve = 0
                if verbose:
                    print(f"  [{label}] Iter {iteration}: noise injected")
            continue
        s = (1.0 - best_t) * s + best_t * direction
        s = np.clip(s, 0.0, 1000.0)
        score = trial_score
        elapsed = time.monotonic() - t0
        if score < best_score - 1e-9:
            best_score = score
            best_s = s.copy()
            no_improve = 0
            if verbose and (iteration <= 5 or iteration % 15 == 0 or best_score < 1.55):
                print(f"  [{label}] Iter {iteration}: score={best_score:.6f} t={best_t:.2f} {elapsed:.1f}s")
    if verbose:
        print(f"  [{label}] Done: {iteration} iters, best={best_score:.6f}")
    return best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-guided local search for minimizing the step function autocorrelation ratio.
    Uses iterative LP to find directions maximizing sum under cross-correlation
    bounds, with line search, multiple initializations, and noise injection.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Get initial sequence
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        s0 = np.clip(np.array(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    else:
        s0 = np.ones(2535, dtype=np.float64)
    n = len(s0)
    print(f"Starting optimization, n={n}, budget={budget_s}s")

    # Phase 1: Main LP optimization from height_sequence_1
    phase1_end = time.monotonic() + min(1000.0, float(budget_s) * 0.5)
    best_s, best_score = run_optimization(s0.copy(), phase1_end, rng, "main", verbose=True)
    print(f"Phase 1 complete: score = {best_score:.6f}")

    # Phase 2: Alternative initializations
    if time.monotonic() < total_deadline:
        print("\nPhase 2: Alternative initializations")
        alternatives = []
        x = np.linspace(-3, 3, n)
        alternatives.append(("cosine", 1.0 + 0.7 * np.cos(np.pi * x / 3.0))
        alternatives.append(("bimodal", np.exp(-((x - 1.5)**2) / 2.0) + np.exp(-((x + 1.5)**2) / 2.0))
        alternatives.append(("ramp", np.arange(1, n + 1, dtype=np.float64)))
        r = 1.0005
        alternatives.append(("geo", r ** np.arange(n, dtype=np.float64)))
        s_alt2 = np.ones(n)
        s_alt2[::2] = 1.5
        s_alt2[1::2] = 0.5
        alternatives.append(("alternating", s_alt2))
        s_rand = rng.dirichlet(np.full(n, 0.5)) * n
        alternatives.append(("dirichlet", s_rand))

        for name, s_alt in alternatives:
            if time.monotonic() >= total_deadline:
                break
            remaining = total_deadline - time.monotonic()
            if remaining < 100:
                break
            phase_end = time.monotonic() + max(80.0, remaining * 0.2)
            print(f"\n  Trying '{name}'...")
            s_opt, score = run_optimization(s_alt.copy(), phase_end, rng, name, verbose=False)
            if score < best_score:
                best_score = score
                best_s = s_opt.copy()
                print(f"  '{name}' IMPROVED: score = {best_score:.6f}")
            else:
                print(f"  '{name}': score = {score:.6f} (no improvement)")

    # Phase 3: Continue polishing the best solution
    if time.monotonic() < total_deadline and np.isfinite(best_score):
        remaining = total_deadline - time.monotonic()
        if remaining > 30:
            print(f"\nPhase 3: Polishing best solution (score={best_score:.6f})")
            phase3_end = time.monotonic() + remaining - 5.0
            best_s, best_score = run_optimization(best_s.copy(), phase3_end, rng, "polish", verbose=True)

    # Final validation
    print(f"\nBest score (fast): {best_score:.6f}")
    final_score = evaluate_sequence(best_s.tolist())
    print(f"Final validated score: {final_score:.6f}")

    if not np.isfinite(final_score):
        print("ERROR: Non-finite score, returning fallback")
        fb = [1.0] * 100
        fs = evaluate_sequence(fb)
        if np.isfinite(fs):
            return fb
        return [1.0]

    return [float(x) for x in best_s]
