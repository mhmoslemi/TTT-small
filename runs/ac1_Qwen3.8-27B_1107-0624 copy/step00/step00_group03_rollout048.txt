```python
def frank_wolfe(f, max_iters=1000):
    n = len(f)
    f = f.copy().astype(float)
    S = f.sum()
    if S <= 0:
        return f
    f = f / S  # normalize to sum = 1
    
    for it in range(max_iters):
        # Compute convolution
        conv = np.convolve(f, f)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-10)[0]
        
        # Compute subgradient (average over active set)
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += f[i] + f[j]  # d(f*f)[k]/df[i] = f[k-i] + f[i]
        grad /= len(active)
        
        # Linear minimization: minimize grad^T f' s.t. f' >= 0, sum(f') = 1
        # Optimal: f' = e_{argmin(grad)}
        v = np.zeros(n)
        v[np.argmin(grad)] = 1.0
        
        # Step size (line search or fixed)
        # Use backtracking line search
        t = 1.0 / (it + 2)  # standard FW step size
        f_new = (1 - t) * f + t * v
        
        # Check convergence
        if np.max(np.abs(f_new - f)) < 1e-10:
            break
        f = f_new
    
    return f
```

```python
def frank_wolfe_with_line_search(f0, max_iters=10000, tol=1e-8):
    n = len(f0)
    f = np.array(f0, dtype=float).copy()
    S = f.sum()
    if S < 1e-10:
        f = np.ones(n) / n
        return f
    f = f / S
    
    best_f = f.copy()
    best_val = np.inf
    
    for it in range(max_iters):
        conv = np.convolve(f, f)
        max_val = conv.max()
        # Active set: indices where conv is close to max
        active = np.where(conv >= max_val - 1e-8 * max_val)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            # grad[i] += f[k-i] + f[i] (with bounds checking)
            i_min = max(0, k - (n-1))
            i_max = min(n-1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                grad[i] += f[j] + f[i]
        grad /= len(active)
        
        # Linear minimization vertex
        v_idx = np.argmin(grad)
        
        # Line search
        best_t = 0
        best_obj = max_val / (S * S)  # current objective (with sum=1, it's just max_val * 2n)
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            f_new = (1 - t) * f + t * 0  # wait, v is e_{v_idx}
            # Actually: f_new = (1-t)*f + t*v where v = e_{v_idx}
            f_new = f * (1 - t)
            f_new[v_idx] += t
            conv_new = np.convolve(f_new, f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj:
                best_obj = obj_new
                best_t = t
        
        if best_t == 0:
            break
        
        f = f * (1 - best_t)
        f[v_idx] += best_t
        
        if conv.max() < best_val:
            best_val = conv.max()
            best_f = f.copy()
        
        if it % 100 == 0:
            print(f"FW iter {it}: max_conv = {conv.max():.8f}, score = {2*n*conv.max():.6f}")
    
    return best_f
```

```python
def compute_grad_at_k(f, k):
    n = len(f)
    grad = np.zeros(n)
    i_min = max(0, k - (n-1))
    i_max = min(n-1, k)
    indices = np.arange(i_min, i_max + 1)
    j_indices = k - indices
    grad[indices] = f[j_indices] + f[indices]
    return grad
```
