def project_simplex(v, s=1.0):
    """Project v onto {x: sum(x) = s, x >= 0}."""
    n = len(v)
    # Scale
    v = v / s  # so we project onto {x: sum = 1, x >= 0} then rescale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w * s
