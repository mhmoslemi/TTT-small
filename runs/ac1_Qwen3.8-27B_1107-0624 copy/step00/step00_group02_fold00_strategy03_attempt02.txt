    def generate_cellular_line(self, base_coef, position):
        """Generate a new local structure using cellular automata-like rules."""
        n = len(base_coef)

        coef_line = np.zeros(n)

            # Create gradient symmetry
            coef_line[:position] = base_coef[:position]
            coef_line[position:n//2] = base_coef[position:n]
            coef_line[n//2:n] = np.random.randn(n-n//2)

            coef_line[position:] = base_coef[position:n]
        return np.clip(coef_line, 0, 1000) / np.sum(coef_line)
