"""
Hybrid optimizer for step function coefficient sequence.
Strategy: ideal sequence (1-z)^(-1/2) init -> LP direction improvement -> diverse local search.
Minimizes 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def make_ideal_sequence(n):
    """a_k = C(2k,k)/4^k, Taylor coefficients of (1-z)^(-1/2)."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(n - 1):
        a[k + 1] = a[k] * (2 * k + 1) / (2 * k + 2)
    return a


def fft_convolve(a, b=None):
    """Full linear convolution using FFT (returns real part)."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return ifft(fft(a, N) * fft(b, N))[:n].real


def build_conv_matrix(f):
    """Sparse (2n-1) x n matrix A with (A@g)_k = (f*g)_k."""
    n = len(f)
    diags = []
    offsets = []
    for d in range(n):
        diags.append(np.full(n, f[d]))
        offsets.append(d)
    for k in range(1, n):
        diags.append(np.full(n - k, f[k]))
        offsets.append(-k)
    return sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')


def solve_lp_direction(f, M, time_limit=40):
    """LP: maximize sum(g) subject to conv(f,g) <= M, 0 <= g <= 1000."""
    n = len(f)
    if f[0] < 1e-12:
        return None
    try:
        A = build_conv_matrix(f)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, 1000.0)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return np.clip(res.x, 0.0, 1000.0)
    except Exception:
        pass
    return None


def perturb_block(a, rng, sigma=0.12):
    """Multiply a random block by exp(N(0, sigma))."""
    n = len(a)
    out = a.copy()
    w = int(rng.integers(max(2, n // 200), max(3, n // 10)))
    s = int(rng.integers(0, n - w))
    out[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    if rng.random() < 0.25:
        out = 0.5 * (out + out[::-1])
    return np.clip(out, 0.0, 1000.0)


def perturb_subgrad(a, rng):

    conv = fft_convolve(a)

    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    idx = np.arange(lo, hi + 1)
    contrib[idx] = a[k_star - idx]
    top_frac = max(3, n // 30)
    top = np.argpartition(contrib, -top_frac)[-top_frac:]
    out = a.copy()
    out[top] *= (1.0 - rng.uniform(0.02, 0.08))
    rest = np.setdiff1d(np.arange(n), top)
    if len(rest) > 0:
        out[rest] *= (1.0 + rng.uniform(0.005, 0.03))
    return np.clip(out, 0.0, 1000.0)

def perturb_momentum(a, best, rng, alpha=0.3):
    """Interpolate between current and best, add noise."""
    n = len(a)

    out += rng.normal(0.0, 0.005 * max(1e-6, np.max(a)), size=n)
