```
def autocorr_fft(x, m):
    xx = np.asarray(x, dtype=np.float64)
    X = np.fft.rfft(xx, m)
    c = np.fft.irfft(X * np.conj(X), m)
    return c

def crosscorr_fft(w, p, m):
    P = np.fft.rfft(p, m)
    W = np.fft.rfft(w, m)
    g = np.fft.irfft(W * np.conj(P), m)
    return g
```

```
def fft_autocorr(x, m):
```

```
   no_improve=0
   last_progress = t0
```

```python
"""
Soft-max gradient search for a non-negative probability vector p minimizing
2*n*max(convolve(p,p)).  The optimizer runs in a softmax parameterization,
several asymmetric initial profiles are tried, and a homotopy-increasing
soft-max beta drives the objective toward the true maximum auto-correlation.
"""

import time
import operator

import numpy as np


def p2_at_least(x):
    """Smallest power of two >= x (x integer)."""
    val = 1
    while val < int(x):
        val <<= 1
    return val


def softmax(y):
    """Numerically stable softmax."""
    y = np.asarray(y, dtype=np.float64)
    y = y - float(np.max(y))
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        z = np.zeros_like(y)
        z[int(np.argmax(y))] = 1.0
        return z
    return e / s


def normalize_p(x):
    """Non-negative normalization to sum 1."""
    x = np.asarray(x, dtype=np.float64)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-15 or not np.isfinite(s):
        return np.full(x.size, 1.0 / float(x.size) if x.size else 0.0)
    return x / s


def fft_autocorr(x, m):
    """Full linear autocorrelation via zero-padded FFT; also returns FFT of x."""
    P = np.fft.rfft(x, m)
    c = np.fft.irfft(P * np.conj(P), m)
    return c, P


def crosscorr_with_existing_p(w, P, m):
    """
    Compute g[i] = sum_k w[k] * p[k-i] using an already computed FFT of p.
    Only the first len(p) entries are needed by the caller.
    """
    W = np.fft.rfft(w, m)
    g = np.fft.irfft(W * np.conj(P), m)
    return g


def soft_max_metrics(A, beta, n):
    """
    Soft max of A with temperature beta*n.
    Returns (smooth_value, softmax_weights).
    """
    A = np.maximum(np.asarray(A, dtype=np.float64), 0.0)
    L = 2 * n - 1
    c = float(beta) * float(n)
    z = c * A
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, -1e300)
    zm = float(np.max(z))
    ez = np.exp(z - zm)
    s = float(np.sum(ez))
    if s <= 1e-300 or not np.isfinite(s):
        w = np.zeros_like(A, dtype=np.float64)
        w[int(np.argmax(A))] = 1.0
        val = float(np.max(A))
    else:
        w = ez / s
        val = (zm + float(np.log(s)) - float(np.log(L))) / c
    return val, w


def exact_score_fft(p, n, m):
    """Exact score for sum(p)=1 using FFT."""
    A, _ = fft_autocorr(p, m)
    A = A[: 2 * n - 1]
    A = np.maximum(A, 0.0)
    return 2.0 * float(n) * float(np.max(A))


def generate_starts(n, rng):
    """Generate a diverse set of asymmetric non-negative starts."""
    starts = []
    j = np.arange(n)
    L = max(1, n - 1)
    t = j / float(L)
    half = n // 2

    # Two-level monotone profiles (left/right asymmetry).
    ratios = [0.05, 0.1, 0.25, 0.5, 0.75, 1.3, 2.0, 4.0, 10.0]
    for r in ratios:
        p = np.where(j < half, 1.0, float(r))
        starts.append(normalize_p(p))
        p = np.where(j < half, float(r), 1.0)
        starts.append(normalize_p(p))

    # Exponentials.
    alphas = [0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    for a in alphas:
        p = np.exp(-float(a) * t)
        starts.append(normalize_p(p))
        p = np.exp(-float(a) * (1.0 - t))
        starts.append(normalize_p(p))

    # Powers with a small offset so endpoints are nonzero but small.
    pows = [0.1, 0.25, 0.5, 0.8, 1.2, 1.8, 2.5]
    for pw in pows:
        p = (t + 0.02) ** float(pw)
        starts.append(normalize_p(p))
        p = ((1.0 - t) + 0.02) ** float(pw)
        starts.append(normalize_p(p))

    # Beta-law profiles.
    beta_pairs = [
        (0.3, 1.0), (1.0, 0.3), (0.3, 2.5), (2.5, 0.3),
        (0.5, 0.5), (0.5, 3.0), (3.0, 0.5), (1.0, 5.0), (5.0, 1.0),
        (1.5, 1.0), (1.0, 1.5), (2.5, 0.7), (0.7, 2.5)
    ]
    for a, b in beta_pairs:
        p = rng.beta(float(a), float(b), n)
        p = np.maximum(p, 1e-8)
        starts.append(normalize_p(p))

    # Sinusoidal asymmetric waveforms.
    amps = [0.10, 0.25, 0.50, 0.80]
    phases = [0.0, 0.7, 1.3, 2.0, 2.9, 3.7]
    for amp in amps:
        for ph in phases:
            p = 1.0 + float(amp) * np.sin(2.0 * np.pi * t + float(ph))
            p = np.maximum(p, 1e-8)
            starts.append(normalize_p(p))

    # Piecewise polynomial profiles.
    Ks = [4, 8, 16, 32, 64]
    for K in Ks:
        x = np.arange(K) / float(K - 1)
        seg = np.minimum((j * float(K) / float(n)).astype(int), K - 1)

        h = 1.0 + 1.2 * x
        starts.append(normalize_p(h[seg]))

        h = 1.0 + 1.2 * (1.0 - x)
        starts.append(normalize_p(h[seg]))

        h = 1.0 + 0.9 * (x * x)
        starts.append(normalize_p(h[seg]))

        h = 1.0 + 0.9 * ((1.0 - x) * (1.0 - x))
        starts.append(normalize_p(h[seg]))

        h = 1.0 + 0.7 * x + 0.4 * np.sin(2.0 * np.pi * x)
        h = np.maximum(h, 1e-8)
        starts.append(normalize_p(h[seg]))

        h = 1.0 + 0.7 * (1.0 - x) + 0.4 * np.cos(2.0 * np.pi * x)
        h = np.maximum(h, 1e-8)
        starts.append(normalize_p(h[seg]))

    # Random smooth monotone profiles.
    for _ in range(25):
        x = rng.random(n)
        x = x ** float(rng.uniform(0.25, 3.0))
        starts.append(normalize_p(x))

        srt = np.sort(x)
        starts.append(normalize_p(srt[::-1]))
        starts.append(normalize_p(srt))

        p = 1.0 + float(rng.uniform(0.1, 0.8)) * rng.random(n)
        p = np.maximum(p, 1e-8)
        starts.append(normalize_p(p))

    # Use the previous construction if it exists and is meaningful.
    try:
        prev = height_sequence_1
        prev = np.asarray(prev, dtype=np.float64).ravel()
        prev = np.maximum(prev, 0.0)
        if prev.size > 0 and np.sum(prev) > 1e-15:
            if prev.size == n:
                starts.append(normalize_p(prev))
                starts.append(normalize_p(prev[::-1]))
            else:
                xp = np.linspace(0.0, 1.0, prev.size)
                xn = np.linspace(0.0, 1.0, n)
                p = np.interp(xn, xp, prev)
                p = np.maximum(p, 0.0)
                starts.append(normalize_p(p))
                starts.append(normalize_p(p[::-1]))
                starts.append(normalize_p(p * (0.5 + t)))
                starts.append(normalize_p(p * (0.5 + (1.0 - t))))
    except Exception:
        pass

    # Add gentle random perturbations of the first deterministic starts.
    base_count = len(starts)
    for idx in range(min(base_count, 35)):
        base = starts[idx]
        p = base + (0.25 / float(n)) * rng.standard_normal(n)
        p = np.maximum(p, 0.0)
        s = float(np.sum(p))
        if s > 1e-15:
            starts.append(p / s)

    return starts


def smooth_optimize(p0, n, m, time_budget, beta_start, beta_end):
    """
    Softmax-parameterized gradient descent on a soft-max approximation to
    max autocorrelation.  Returns the best exact-score point found.
    """
    if time_budget <= 0.0:
        p0 = normalize_p(p0)
        return p0, exact_score_fft(p0, n, m)

    p0 = normalize_p(p0)
    y = np.log(np.maximum(p0, 1e-300))
    y -= float(np.mean(y))
    p = softmax(y)

    A, _ = fft_autocorr(p, m)
    A = np.maximum(A[: 2 * n - 1], 0.0)
    best_score = 2.0 * float(n) * float(np.max(A))
    best_p = p.copy()

    beta = float(beta_start)
    lr = 0.05
    max_lr = 0.55
    min_lr = 1e-5
    mu = 0.80
    momentum = np.zeros(n, dtype=np.float64)

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        A, P = fft_autocorr(p, m)
        Ax = np.maximum(A[: 2 * n - 1], 0.0)
        if not np.all(np.isfinite(Ax)):
            break

        val, w = soft_max_metrics(Ax, beta, n)

        grad_p = 2.0 * crosscorr_with_existing_p(w, P, m)[:n].real
        if not np.all(np.isfinite(grad_p)):
            break

        # Gradient of objective through softmax:
        # dF/dy_j = p_j * (g_j - sum_i g_i p_i)
        grad_y = p * (grad_p - float(np.dot(grad_p, p)))
        if not np.all(np.isfinite(grad_y)):
            break

        rms = float(np.sqrt(np.mean(grad_y * grad_y)))
        if rms <= 1e-16:
            break

        gy = grad_y / rms
        momentum = mu * momentum + (1.0 - mu) * gy

        y_new = y - lr * momentum
        y_new -= float(np.mean(y_new))
        y_new = np.clip(y_new, -30.0, 30.0)
        p_new = softmax(y_new)
        if not np.all(np.isfinite(p_new)):
            lr *= 0.5
            continue

        A2, _ = fft_autocorr(p_new, m)
        Ax2 = np.maximum(A2[: 2 * n - 1], 0.0)
        if not np.all(np.isfinite(Ax2)):
            lr *= 0.5
            no_improve += 1
            continue

        val2, _ = soft_max_metrics(Ax2, beta, n)
        exact2 = 2.0 * float(n) * float(np.max(Ax2))

        if np.isfinite(val2) and val2 < val + 1e-13:
            y = y_new
            p = p_new
            val = val2
            no_improve = 0
            lr = min(max_lr, lr * 1.02)

            if exact2 < best_score:
                best_score = exact2
                best_p = p_new.copy()
        else:
            no_improve += 1
            lr *= 0.70
            if no_improve > 25:
                beta = min(float(beta_end), beta * 1.18)
                momentum *= 0.0
                no_improve = 0
                if lr < min_lr:
                    break

    # One final exact score for the actual returned point.
    p_best_final = normalize_p(best_p)
    final_score = exact_score_fft(p_best_final, n, m)
    if final_score < best_score:
        best_score = final_score
        best_p = p_best_final

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search entry point.  Splits the wall-clock budget among initial
    profile optimizations, a deep refinement of the current best point, and
    random restarts.  Returns the best sequence found, verified by
    evaluate_sequence when available.
    """
    t_global0 = time.monotonic()
    deadline = t_global0 + max(0.0, float(budget_s) - 20.0)

    n = 1024
    m = p2_at_least(2 * n - 1)
    rng = np.random.default_rng(int(seed) if seed is not None else 42)

    print("n = %d, fft m = %d, budget = %.1fs" % (n, m, float(budget_s)))

    starts = generate_starts(n, rng)

    results = []
    for p in starts:
        p = normalize_p(p)
        sc = exact_score_fft(p, n, m)
        if np.isfinite(sc):
            results.append([sc, p])

    if not results:
        p = normalize_p(np.ones(n))
        results.append([2.0, p])

    results.sort(key=operator.itemgetter(0))

    best_p = normalize_p(results[0][1])
    best_score = float(results[0][0])
    print("Best initial score: %.9f" % best_score)

    # Phase 1: optimize the best diverse starts, leaving time for refinement.
    phase1_end = min(deadline - 900.0, t_global0 + 1050.0)
    top_count = min(6, len(results))
    for i in range(top_count):
        if time.monotonic() > phase1_end:
            break
        allowed = min(230.0, phase1_end - time.monotonic())
        if allowed <= 5.0:
            break
        p_opt, sc_opt = smooth_optimize(
            results[i][1], n, m, allowed, 18.0, 900.0
        )
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = float(sc_opt)
            best_p = normalize_p(p_opt)
            print("Phase1 start %d improved to %.9f" % (i + 1, best_score))

    # Phase 2: deep refinement of the current best.
    phase2_end = min(deadline - 220.0, t_global0 + 1580.0)
    if time.monotonic() < phase2_end:
        allowed = min(520.0, phase2_end - time.monotonic())
        if allowed > 30.0:
            p_opt, sc_opt = smooth_optimize(
                best_p, n, m, allowed, 60.0, 2500.0
            )
            if np.isfinite(sc_opt) and sc_opt < best_score:
                best_score = float(sc_opt)
                best_p = normalize_p(p_opt)
                print("Phase2 refined to %.9f" % best_score)

    # Phase 3: random restarts / local perturbations until near the deadline.
    scale = 0.18 / float(n)
    fail_count = 0

    while time.monotonic() < deadline - 25.0:
        if fail_count % 6 == 0:
            base = rng.beta(1.2, 1.6, n)
        else:
            base = best_p + scale * rng.standard_normal(n)
        p_pert = normalize_p(base)

        allowed = min(75.0, deadline - time.monotonic() - 8.0)
        if allowed <= 20.0:
            break

        p_opt, sc_opt = smooth_optimize(
            p_pert, n, m, allowed, 40.0, 2500.0
        )
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = float(sc_opt)
            best_p = normalize_p(p_opt)
            fail_count = 0
            scale = min(0.45 / float(n), scale * 1.15)
            print("Restart improved to %.9f" % best_score)
        else:
            fail_count += 1
            scale *= 0.78

    # Final candidates: current best plus a few strong initial profiles.
    final_best_p = normalize_p(best_p)
    final_best_val = float(evaluate_sequence(final_best_p.tolist()))
    print("Best internal score: %.9f" % best_score)
    print("Best evaluate_sequence: %.9f" % final_best_val)

    for i in range(min(5, len(results))):
        p = normalize_p(results[i][1])
        val = float(evaluate_sequence(p.tolist()))
        if np.isfinite(val) and val < final_best_val:
            final_best_val = val
            final_best_p = p

    out = []
    for x in final_best_p:
        out.append(float(x))
    print("Returned evaluate_sequence: %.9f" % final_best_val)
    return out
```
