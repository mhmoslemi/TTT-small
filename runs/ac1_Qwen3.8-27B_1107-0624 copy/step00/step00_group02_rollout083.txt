```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(convolve(a,a)) / (sum(a)**2)
using log-space RMS subgradient descent on the normalized sequence, with
periodic symmetry checks, restarts, and multiple stochastic initializations.
"""

import numpy as np
import time

LOG_FLOOR = 1e-15


def safe_uniform(n):
    n = max(1, int(n))
    return np.ones(n, dtype=np.float64) / float(n)


def next_pow2(x):
    x = max(1, int(x))
    return 1 << (x - 1).bit_length()


def normalize_pos(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return safe_uniform(1)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if not np.isfinite(s) or s <= 1e-300:
        return safe_uniform(x.size)
    return x / s


def make_log(a):
    x = normalize_pos(a)
    x = np.maximum(x, LOG_FLOOR)
    s = float(np.sum(x))
    if s <= 0.0 or not np.isfinite(s):
        return safe_uniform(x.size)
    x /= s
    v = np.log(x)
    v -= float(np.mean(v))
    return v


def from_log(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    m = float(np.max(v))
    a = np.exp(v - m)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return safe_uniform(v.size)
    a /= s
    a = np.maximum(a, LOG_FLOOR)
    a /= float(np.sum(a))
    return a


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.zeros(1, dtype=np.float64)
    n = int(a.size)
    L = 2 * n - 1
    N = next_pow2(L)
    fa = np.fft.rfft(a, N)
    b = np.fft.irfft(fa * fa, N)[:L]
    b = np.where(np.isfinite(b) & (b > 0.0), b, 0.0)
    return b


def active_lags(b, max_count, frac=0.75):
    L = int(b.size)
    if L <= 0:
        return np.array([0], dtype=np.int64), np.array([1.0], dtype=np.float64)

    max_count = max(1, int(max_count))
    mx = float(np.max(b))
    if not np.isfinite(mx) or mx <= 0.0:
        return np.array([0], dtype=np.int64), np.array([1.0], dtype=np.float64)

    if max_count >= L:
        top = np.arange(L, dtype=np.int64)
    else:
        top = np.argpartition(b, -max_count)[-max_count:].astype(np.int64)

    thr = float(frac) * mx
    idx = top[b[top] >= thr]

    if idx.size == 0:
        idx = top[np.argmax(b[top])].reshape(1)

    w = np.maximum(b[idx] - thr, 0.0)
    sw = float(np.sum(w))
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(idx.size, dtype=np.float64)
    else:
        w = w / sw

    return idx, w


def gradient_from_active(a, active, weights):
    n = int(a.size)
    g = np.zeros(n, dtype=np.float64)

    for i in range(active.size):
        k = int(active[i])
        w = float(weights[i])
        if w == 0.0:
            continue

        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi >= n:
            hi = n - 1

        if lo <= hi:
            segment = a[k - hi:k - lo + 1][::-1]
            g[lo:hi + 1] += (2.0 * w) * segment

    return g


def raised_cosine_envelope(n):
    x = np.linspace(0.0, 1.0, max(1, int(n)), dtype=np.float64)
    return 0.5 * (1.0 + np.cos(np.pi * (2.0 * x - 1.0)))


def gaussian_envelope(n, sigma=0.22):
    x = np.linspace(0.0, 1.0, max(1, int(n)), dtype=np.float64)
    sigma = max(1e-6, float(sigma))
    return np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)


def modulated_envelope(base, rng, eps=0.8, ncomp=3):
    base = np.asarray(base, dtype=np.float64)
    n = int(base.size)
    if n <= 0:
        return safe_uniform(1)

    x = np.arange(n, dtype=np.float64)
    mod = np.zeros(n, dtype=np.float64)

    for _ in range(max(1, int(ncomp))):
        f = float(rng.uniform(0.05, 0.50))
        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        mod += np.cos(2.0 * np.pi * f * x + phi)

    mod /= max(1, float(ncomp))
    vals = base * (1.0 + float(eps) * mod)
    return normalize_pos(np.maximum(vals, 0.0))


def anti_sign_sequence(n, rng, stick=0.8):
    n = max(1, int(n))
    c = np.ones(n, dtype=np.float64)
    rmat = rng.random(n)
    cmat = rng.random(n)

    for i in range(1, n):
        if rmat[i] < float(stick):
            c[i] = -c[i - 1]
        else:
            c[i] = 1.0 if cmat[i] < 0.5 else -1.0

    return c


def random_positive_init(n, rng):
    n = max(1, int(n))
    p = float(rng.choice([1.0, 1.5, 2.0]))
    x = np.abs(rng.standard_normal(n)) ** p
    return normalize_pos(x)


def resample_nonnegative(x, n):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    n = max(1, int(n))

    if x.size == 0 or not np.isfinite(s) or s <= 1e-300:
        return safe_uniform(n)

    x /= s
    m = int(x.size)
    if m == n:
        return x
    if m == 1:
        return safe_uniform(n)

    old = np.linspace(0.0, 1.0, m, dtype=np.float64)
    new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    y = np.interp(new, old, x)
    return normalize_pos(y)


def run_optimizer(a_init, time_budget, seed=0, active_count=None, base_lr=0.05, best_init=None):
    if a_init is None:
        return safe_uniform(8192)

    a = normalize_pos(a_init)
    n = int(a.size)
    if n <= 0:
        return safe_uniform(8192)

    if active_count is None:
        active_count = int(max(32, min(128, n // 64)))

    rng = np.random.default_rng(int(seed))
    v = make_log(a)
    mom = np.zeros(n, dtype=np.float64)
    var = np.zeros(n, dtype=np.float64)

    b = fast_conv(a)
    cur = float(np.max(b))
    full_cur = 2.0 * float(n) * cur

    best = a.copy()
    best_max = cur
    best_full = full_cur

    if best_init is not None and best_init.size == n:
        bi = normalize_pos(best_init)
        bi_b = fast_conv(bi)
        bi_max = float(np.max(bi_b))
        bi_full = 2.0 * float(n) * bi_max
        if bi_full < best_full:
            best = bi.copy()
            best_max = bi_max
            best_full = bi_full

    budget = max(0.0, float(time_budget))
    if budget <= 0.0:
        return best

    lr = float(base_lr)
    min_lr = 0.002
    max_lr = 0.25
    since_best = 0
    it = 0
    t_start = time.monotonic()
    deadline = t_start + budget
    last_print = t_start

    while time.monotonic() < deadline:
        it += 1

        idx, w = active_lags(b, active_count, 0.75)
        if idx.size < 3:
            idx2, w2 = active_lags(b, max(96, 2 * active_count), 0.50)
            if idx2.size > idx.size:
                idx, w = idx2, w2

        g = gradient_from_active(a, idx, w)
        h = g - float(np.dot(a, g))

        h_abs = np.abs(h)
        h_max = float(np.max(h_abs)) if h_abs.size else 0.0
        if not np.isfinite(h_max) or h_max <= 1e-18:
            break

        mom = 0.85 * mom + 0.15 * h
        var = 0.90 * var + 0.10 * (h * h)

        offset = 0.02 * h_max
        denom = np.sqrt(var) + 1e-12 + offset
        step = (lr * mom) / denom
        step = np.clip(step, -1.0, 1.0)

        v_try = v - step
        v_try -= float(np.mean(v_try))
        span = float(np.max(v_try) - np.min(v_try))
        if np.isfinite(span) and span > 40.0:
            v_try = v_try * (40.0 / span)
            v_try -= float(np.mean(v_try))

        a_try = from_log(v_try)
        b_try = fast_conv(a_try)
        cur_try = float(np.max(b_try))

        if np.isfinite(cur_try):
            accept = False

            if cur_try < cur:
                accept = True
                lr = min(lr * 1.03, max_lr)
            elif cur_try == cur and rng.random() < 0.10:
                accept = True
            elif cur_try < cur * 1.01 and rng.random() < 0.03:
                accept = True

            if accept:
                v = v_try
                a = a_try
                b = b_try
                cur = cur_try

                if cur_try < best_max:
                    best = a.copy()
                    best_max = cur_try
                    best_full = 2.0 * float(n) * best_max
                    since_best = 0
            else:
                lr = max(lr * 0.92, min_lr)
        else:
            v = make_log(best)
            a = best.copy()
            b = fast_conv(a)
            cur = best_max
            lr = min_lr

        since_best += 1

        if since_best > 1500 or cur > best_max * 1.05 + 1e-15:
            sigma = 0.04 + 0.16 * float(rng.random())
            v_p = make_log(best) + sigma * rng.standard_normal(n)
            a_p = from_log(v_p)
            b_p = fast_conv(a_p)
            cur_p = float(np.max(b_p))

            if np.isfinite(cur_p) and cur_p < best_max:
                best = a_p.copy()
                best_max = cur_p
                best_full = 2.0 * float(n) * best_max

            v = make_log(best)
            a = best.copy()
            b = fast_conv(a)
            cur = best_max
            mom.fill(0.0)
            var.fill(0.0)
            lr = max(lr * 0.7, min_lr)
            since_best = 0

        if it % 3000 == 0:
            a_sym = 0.5 * (a + a[::-1])
            b_sym = fast_conv(a_sym)
            cur_sym = float(np.max(b_sym))

            if cur_sym < cur:
                a = a_sym
                b = b_sym
                cur = cur_sym
                v = make_log(a)
                mom.fill(0.0)
                var.fill(0.0)

            if cur_sym < best_max:
                best = a_sym.copy()
                best_max = cur_sym
                best_full = 2.0 * float(n) * best_max
                since_best = 0

        if time.monotonic() - last_print > 30.0:
            print(f"[opt] n={n} iter={it} best={2.0*float(n)*best_max:.6f} "
                  f"cur={2.0*float(n)*cur:.6f} lr={lr:.5f}")
            last_print = time.monotonic()

    return best


def prune_small(a, thr=1e-12):
    a = normalize_pos(a)
    n = int(a.size)
    cur = 2.0 * float(n) * float(np.max(fast_conv(a)))

    x = a.copy()
    x[x < float(thr)] = 0.0
    s = float(np.sum(x))
    if s <= 1e-300 or not np.isfinite(s):
        return a
    x /= s

    new = 2.0 * float(n) * float(np.max(fast_conv(x)))
    if np.isfinite(new) and new < cur:
        return x
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    t0 = time.monotonic()
    total = float(budget_s)
    reserve = min(35.0, max(5.0, 0.02 * total))
    budget = max(10.0, total - reserve)

    n = 8192
    rng = np.random.default_rng(int(seed))

    starts = []

    hseq = globals().get("height_sequence_1", None)
    if hseq is not None:

            harr = np.asarray(hseq, dtype=np.float64)

        except Exception:
            pass

    env = raised_cosine_envelope(n)
    starts.append(normalize_pos(env))

    starts.append(random_positive_init(n, rng))

    if len(starts) >= 2:
        starts.append(normalize_pos(0.5 * starts[0] + 0.5 * starts[1]))

    starts = [normalize_pos(x) for x in starts[:8]]

    global_best = starts[0].copy()
    global_b = fast_conv(global_best)
    global_score = 2.0 * float(global_best.size) * float(np.max(global_b))
    if not np.isfinite(global_score):
        global_best = safe_uniform(n)
        global_score = 2.0

    quick_results = []

    elapsed = time.monotonic() - t0
    remaining = max(0.0, budget - elapsed)
    if remaining > 0.0 and len(starts) > 0:
        per_quick = min(35.0, max(3.0, remaining / (2.0 * len(starts))))
        if per_quick * len(starts) > 0.40 * remaining:
            per_quick = 0.40 * remaining / len(starts)

        for i in range(len(starts)):
            now = time.monotonic()
            avail = max(0.0, t0 + budget - 15.0 - now)
            if avail <= 3.0:
                break

            slice_time = min(per_quick, avail)
            if slice_time <= 0.0:
                break

            a_opt = run_optimizer(starts[i], slice_time, seed=10000 + i,
                                  best_init=global_best)
            b_opt = fast_conv(a_opt)
            score = 2.0 * float(a_opt.size) * float(np.max(b_opt))

            if not np.isfinite(score):
                score = 2.0

            quick_results.append((a_opt, score))

            if score < global_score:
                global_best = a_opt.copy()
                global_score = score

    remaining = max(0.0, t0 + budget - time.monotonic())

    top = []
    for a_opt, score in quick_results:
        pos = 0
        while pos < len(top) and score >= top[pos][1]:
            pos += 1
        top.insert(pos, (a_opt, score))
        if len(top) > 3:
            top.pop()

    long_tasks = []
    if remaining > 10.0:
        long_tasks.append((global_best.copy(), 0.40 * remaining, 20000))

        if len(top) > 1:
            long_tasks.append((top[1][0].copy(), 0.25 * remaining, 20001))
        elif len(starts) > 1:
            long_tasks.append((starts[1].copy(), 0.25 * remaining, 20001))
        else:
            long_tasks.append((global_best.copy(), 0.25 * remaining, 20001))

        v_p = make_log(global_best) + 0.10 * rng.standard_normal(n)
        a_p = from_log(v_p)
        long_tasks.append((a_p, 0.15 * remaining, 20002))

        for a_init, secs, task_seed in long_tasks:
            now = time.monotonic()
            avail = max(0.0, t0 + budget - 5.0 - now)
            if avail <= 0.0:
                break

            actual = min(float(secs), avail)
            if actual <= 0.0:
                break

            a_opt = run_optimizer(a_init, actual, seed=int(task_seed),
                                  best_init=global_best)
            b_opt = fast_conv(a_opt)
            score = 2.0 * float(a_opt.size) * float(np.max(b_opt))

            if not np.isfinite(score):
                score = 2.0

            if score < global_score:
                global_best = a_opt.copy()
                global_score = score

    remaining = max(0.0, t0 + budget - time.monotonic())
    if remaining > 15.0:
        a_opt = run_optimizer(global_best, remaining, seed=30000,
                              best_init=global_best)
        b_opt = fast_conv(a_opt)
        score = 2.0 * float(a_opt.size) * float(np.max(b_opt))
        if np.isfinite(score) and score < global_score:
            global_best = a_opt.copy()
            global_score = score

    a_final = normalize_pos(global_best)
    a_final = prune_small(a_final, 1e-12)
    a_final = np.clip(a_final, 0.0, 1000.0)
    s = float(np.sum(a_final))
    if s > 0.0 and np.isfinite(s):
        a_final /= s
    else:
        a_final = safe_uniform(n)

    try:
        ev = evaluate_sequence(a_final.tolist())
        if (not np.isfinite(ev)) or (ev > 2.0 + 1e-8):
            a_final = safe_uniform(n)
    except Exception:
        pass

    return [float(x) for x in a_final]
```
