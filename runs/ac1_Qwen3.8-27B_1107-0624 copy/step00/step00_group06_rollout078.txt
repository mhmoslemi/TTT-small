    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fconv = fa * fa
    result = np.fft.irfft(fconv, size)[:2*n-1]
    return result

    a = a / s

    best_a = a.copy()
    best_sc = compute_score(a)

    deadline = time.monotonic() + budget_s
    momentum = np.zeros(n)
    mu = 0.85
    base_lr = 0.03
    it = 0
    stuck = 0
    patience = 300

    while time.monotonic() < deadline:
        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Subgradient: grad[i] = a[k_star - i] for valid indices
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        grad[idx] = a[k_star - idx]

        # Center to preserve normalization
        grad -= np.mean(grad)

        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.0005 * it)

        # Momentum
        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a_new = np.clip(a_new, 0.0, None)
        s_new = float(np.sum(a_new))
        if s_new < 0.01:
            a_new = a.copy()
            momentum *= 0.5
        else:
            a_new /= s_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1

        a = a_new
        it += 1

        # Random perturbation to escape local minima
        if stuck > patience:
            frac = min(0.5, 0.1 + 0.05 * (stuck // patience))
            mask = rng.random(n) < frac
            cnt = int(np.sum(mask))
            if cnt > 0:
                a[mask] = rng.random(cnt) * 1.5
            s = float(np.sum(a))
            if s < 0.01:
                a = best_a.copy()
            else:
                a = a / s
            momentum = np.zeros(n)
            stuck = 0
        elif it % 2000 == 0 and rng.random() < 0.2:
            # Small random nudge
            a = a + 0.02 * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            s = float(np.sum(a))
            if s > 0.01:
                a = a / s
            momentum *= 0.5
            stuck = 0

    return best_a, best_sc


def polish(a0, max_iters=200):
    """Coordinate descent polishing step."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0).copy()
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return a
    a = a / s
    best_sc = compute_score(a)
    best_a = a.copy()

    for _ in range(max_iters):
        improved = False
        for i in range(n):
            # Find optimal delta for coordinate i
            # b[k] changes linearly in a[i] with slope 2*a[k-i] for k != 2i
            # and has a[i]^2 term at k=2i
            b = fast_convolve(a)
            peak = float(np.max(b))
            k_star = int(np.argmax(b))

            # Slope of b[k] w.r.t. a[i]
            # b[k] = ... + a[i]*a[k-i] + a[k-i]*a[i] + ... (for k != 2i)
            # slope_k = 2*a[k-i] if 0<=k-i<n and k!=2i
            # For k=2i: slope = 2*a[i] (derivative of a[i]^2)
            # We want to minimize max(b) by adjusting a[i]
            # Try small steps
            best_delta = 0.0
            best_local = peak
            for delta in [-0.5, -0.2, -0.05, 0.05, 0.2, 0.5]:
                trial = a.copy()
                trial[i] = max(0.0, a[i] + delta * max(a[i], 0.1))
                ts = float(np.sum(trial))
                if ts < 0.01:
                    continue
                trial = trial / ts
                tb = fast_convolve(trial)
                tpeak = float(np.max(tb))
                if tpeak < best_local:
                    best_local = tpeak
                    best_delta = delta

            if best_delta != 0.0:
                a[i] = max(0.0, a[i] + best_delta * max(a[i], 0.1))
                s = float(np.sum(a))
                if s > 0.01:
                    a = a / s
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for optimal non-negative step function sequence minimizing
    2*n*max(convolve(a,a))/sum(a)^2.
    Uses diverse initializations, subgradient optimization, and polishing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    best_a = None
    best_sc = np.inf

    def try_update(name, a):
        nonlocal best_a, best_sc
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if len(a) == 0 or float(np.sum(a)) < 0.01:
            return
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"  [{name}] score={sc:.6f}", flush=True)

    # --- Build diverse starting points ---
    starts = []

    # Provided sequence
    try:
        h = np.array(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if len(h) > 0 and float(np.sum(h)) > 0.01:
            starts.append(('provided', h))
            try_update('provided_raw', h)
    except Exception:
        pass

    # Structured starts at various lengths
    lengths = [300, 500, 800, 1200, 2000, 4000, 8000]
    for n in lengths:
        rng_local = np.random.default_rng(seed * 1000 + n)

        # Perturbed uniform
        a = 1.0 + 0.5 * rng_local.random(n)
        starts.append((f'unif{n}', a))
        try_update(f'unif{n}', a)

        # Tapered (higher at center)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.9 * np.sin(np.pi * x)
        starts.append((f'sin{n}', a))
        try_update(f'sin{n}', a)

        # Edge-enhanced
        a = 1.0 + 0.8 * np.abs(2.0 * x - 1.0)
        starts.append((f'edge{n}', a))
        try_update(f'edge{n}', a)

        # Periodic dips
        a = 1.0
        a[::4] = 0.2
        starts.append((f'per4_{n}', a))
        try_update(f'per4_{n}', a)

        # Random
        a = rng_local.random(n) + 0.05
        starts.append((f'rnd{n}', a))
        try_update(f'rnd{n}', a)

    # Sort starts by initial score
    scored_starts = []
    for name, a in starts:
        sc = compute_score(a)
        if np.isfinite(sc):
            scored_starts.append((sc, name, a))
    scored_starts.sort(key=lambda z: z[0])

    print(f"Total starts: {len(scored_starts)}, best init: {scored_starts[0][0]:.6f}", flush=True)

    # --- Optimize from each start ---
    n_starts = len(scored_starts)
    for i, (init_sc, name, a) in enumerate(scored_starts):
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time: better starts get more time
        priority = 1.0 / (1.0 + 0.5 * i)
        base_alloc = 40.0
        alloc = min(base_alloc * priority, remaining * 0.7 / max(1, n_starts - i))
        alloc = max(10.0, alloc)
        alloc = min(alloc, remaining - 10)

        print(f"Optimizing '{name}' (init={init_sc:.4f}, alloc={alloc:.0f}s)", flush=True)
        res_a, res_sc = optimize_subgrad(a, alloc, seed + i)

        # Polish the result
        res_a = polish(res_a, max_iters=50)
        res_sc = compute_score(res_a)

        try_update(name, res_a)

    # --- Final polishing of global best ---
    if best_a is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        alloc_polish = min(120.0, remaining * 0.5)
        if alloc_polish > 10:
            print(f"Final polish, alloc={alloc_polish:.0f}s", flush=True)
            # Run one more optimization from the best
            res_a, res_sc = optimize_subgrad(best_a, alloc_polish, seed + 9999)
            res_a = polish(res_a, max_iters=100)
            res_sc = compute_score(res_a)
            try_update('final_polish', res_a)

    # --- Verify with actual evaluator ---
    if best_a is not None:
        result = [float(x) for x in best_a]
        final_sc = evaluate_sequence(result)
        print(f"FINAL: score={final_sc:.6f}, len={len(result)}", flush=True)
        if not np.isfinite(final_sc):
            # Fallback
            result = [1.0] * 100
            final_sc = evaluate_sequence(result)
            print(f"FALLBACK: score={final_sc:.6f}", flush=True)
        return result
    else:
        result = [1.0] * 100
        print(f"FALLBACK (no best found): score={evaluate_sequence(result):.6f}", flush=True)
        return result
