import time
import numpy as np
import scipy.signal

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cv - 1.0)
    rhat = np.max(r[cond]) if np.any(cond) else 1
    theta = (cv[rhat - 1] - 1.0) / rhat
    return np.maximum(v - theta, 0.0)

def compute_c(a):
    n = len(a)
    a_pad = np.zeros(2 * n)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A.conj())[:2 * n - 1]
    return c

def compute_loss(c, tau):
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(np.clip(c - c_max, -50, 50) * tau)))

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cv - 1.0)
    rhat = np.max(r[cond]) if np.any(cond) else 1
    theta = (cv[rhat - 1] - 1.0) / rhat
    return np.maximum(v - theta, 0.0)

import time
import numpy as np
import scipy.signal

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cv - 1.0)
    rhat = np.max(r[cond]) if np.any(cond) else 1
    theta = (cv[rhat - 1] - 1.0) / rhat
    return np.maximum(v - theta, 0.0)

def compute_c(a):
    n = len(a)
    a_pad = np.zeros(2 * n)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A.conj())[:2 * n - 1]
    return c

def compute_loss(c, tau):
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(np.clip(c - c_max, -50, 50) * tau)))

def optimize_n(n, budget_s, seed):
    rng = np.random.default_rng(seed)
    # initialize with noise to break symmetry
    a = np.ones(n)
    a += rng.normal(0, 0.5, n)
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    
    tau = 5.0
    current_loss = compute_loss(compute_c(a), tau)
    start_time = time.time()
    
    best_a = a.copy()
    best_raw_score = 2.0 * n * np.max(compute_c(a))
    step = 0.05
    
    i = 0
    while time.time() - start_time < budget_s:
        c = compute_c(a)
        c_max = np.max(c)
        p = np.exp(np.clip((c - c_max) * tau, -500, 500))
        p_sum = np.sum(p)
        if p_sum == 0 or not np.isfinite(p_sum):
            return best_a, best_raw_score
        p = p / p_sum
        
        # Gradient of smooth max

        # backtracking line search
        for ls_iter in range(15):
            a_trial = project_simplex(a - step * g)
            c_trial = compute_c(a_trial)
            loss_trial = compute_loss(c_trial, tau)
            if loss_trial <= current_loss + 1e-8:
                current_loss = loss_trial
                a = a_trial
                break
            else:
                step *= 0.5
                if step < 1e-7:
                    break

            a = project_simplex(a + mut)

            best_a = a.copy()

        if i % 500 == 0:
            tau *= 1.5

import time
import numpy as np

def project_simplex(v):
    """
    Project a vector v exactly onto the probability simplex (sum = 1, elements >= 0).
    """
    n = len(v)
    if n <= 1:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cv - 1.0)
    rhat = np.max(r[cond]) if np.any(cond) else 1
    theta = (cv[rhat - 1] - 1.0) / rhat
    return np.maximum(v - theta, 0.0)

def compute_c(a):
    """
    Compute the self-convolution of the sequence a via FFT.
    """
    n = len(a)
    a_pad = np.zeros(2 * n)

    A = np.fft.rfft(a_pad)
    c = np.fft.irfft(A * A.conj())[:2 * n - 1]
    return c
