    W=0.0
    eps=1e-15

        sj=s[j]

    total=y.sum()
    if total<=0: p = np.ones(L)/L

def lp_direction(s,c,n,Klp,rng,floor):
    try: from scipy.optimize import linprog
    except Exception: return None
    L=2*n-1
    if Klp <=0 or Klp > L: Klp=L

       # top Klp

       else: idx=np.argpartition(c, -Klp)[-Klp:]
    else: idx=np.arange(L)
    # extras
    extras = [n-1, 0, L-1, L//2]
    # add some evenly spaced
    for q in np.linspace(0, L-1, 20): extras.append(int(q))
    ks = np.unique(np.concatenate((idx, np.array(extras,dtype=int))))
    R=ks.size
    A=np.zeros((R,n), dtype=np.float64)
    for r,k in enumerate(ks):
       start=max(0, int(k)-n+1); end=min(n-1,int(k))+1
       if end>start: A[r,start:end]=1.0
    # objective perturbed 1/s
    s_obj=np.maximum(s, 1e-6)
    if rng is not None:
       noise = 0.01 * rng.normal(size=n)
       # asymmetry bias
       if n>1:
          tt=np.arange(n,dtype=float)/(n-1)
          sign=1.0 if rng.random() <0.5 else -1.0
          noise += 0.02 * sign * (tt - 0.5)
       else: noise=np.zeros(n)
       s_obj = s_obj * (1.0 + noise)
       s_obj = np.maximum(s_obj, 1e-6)
    inv = 1.0 / s_obj
    c_obj = -inv
    b_ub = np.ones(R, dtype=np.float64)
    try:
       res=linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0,1.0), method='highs')
    except Exception:
       return None
    if res is None or res.x is None: return None
    w=np.asarray(res.x,dtype=float)
    w=np.where(np.isfinite(w),w,0.0)
    w=np.clip(w,0.0,1.0)
    b = w * inv
    b = np.where(np.isfinite(b), b, 0.0)

import numpy as np
import time

"""
Search purpose:
Minimize evaluate_sequence(s) = 2*len(s)*max(s*s)/(sum(s))^2 for nonnegative s.
The program uses:
  1) coarse-to-fine optimization at several sequence lengths,
  2) FFT-based evaluation of the full convolution,
  3) a soft-max / top-lag / dual-covering subgradient family,
  4) multiplicative power updates with backtracking-style line search,
  5) a symmetry-breaking LP-expanded search direction using scipy if possible,
  6) an archive of good sequences and log-space random restarts.

The returned sequence is normalized to sum 1, so the evaluator's scale
invariance and clipping are not affected.
"""

LP_T_VALUES = (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.35, 0.6)


def next_pow2(m):
    n = 1
    m = int(max(1, m))
    while n < m:
        n <<= 1
    return n


def normalize_sequence_with_floor(v, floor):
    x = np.asarray(v, dtype=np.float64)
    if x.ndim != 1 or x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, float(floor))
    s = float(x.sum())
    if not np.isfinite(s) or s <= 0.0:
        x = np.full(x.size, 1.0 / x.size, dtype=np.float64)
        s = float(x.sum())
    x = x / s
    return x


def fast_max_conv(s, N, L):
    if s.size == 0:
        return np.inf
    S = np.fft.rfft(s, N)
    c = np.fft.irfft(S * np.conj(S), N)[:L]
    if c.size == 0:
        return np.inf
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return m


def fast_score(s, n, N, L):
    m = fast_max_conv(s, N, L)
    if not np.isfinite(m):
        return np.inf
    return 2.0 * float(n) * m


def compute_c_s(s, N, L):
    S = np.fft.rfft(s, N)
    c = np.fft.irfft(S * np.conj(S), N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    return c, S


def xcorr_p_s(p, S, N, n):
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(P * np.conj(S), N)[:n]
    corr = np.where(np.isfinite(corr), corr, 0.0)
    return np.maximum(corr, 0.0)


def uniform_weights(length):
    if length <= 0:
        return np.zeros(0)
    return np.full(length, 1.0 / length, dtype=np.float64)


def normalize_weights(p, length=None):
    p = np.asarray(p, dtype=np.float64)
    if length is None:
        length = p.size
    if p.size != length or length <= 0:
        return uniform_weights(length)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        return uniform_weights(length)
    return p / s


def softmax_weights(c, temp_rel):
    L = c.size
    if L <= 0:
        return np.zeros(0)
    cm = float(c.max())
    if not np.isfinite(cm) or cm <= 0.0:
        return uniform_weights(L)
    temp = max(float(temp_rel) * cm, 1e-300)
    z = (c - cm) / temp
    p = np.exp(z)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    return normalize_weights(p, L)


def top_weights(c, K):
    L = c.size
    if L <= 0:
        return np.zeros(0)
    K = int(K)
    if K <= 0 or K >= L:
        return uniform_weights(L)
    if K == 1:
        idx = np.array([int(np.argmax(c))], dtype=int)
    else:
        idx = np.argpartition(c, -K)[-K:]
    p = np.zeros(L, dtype=np.float64)
    p[idx] = 1.0 / float(K)
    return p


def dual_weights(s, n):
    """
    Cheap LP-dual-inspired covering weights.

    For the full relaxed LP
        max sum b_j
        s.t. conv(s,b)_k <= M, b >= 0,
    the dual has sliding-window covering constraints.  The greedy primal-dual
    covering weights here are used only as a nonnegative Lagrange-like p.
    """
    L = 2 * n - 1
    y = np.zeros(L, dtype=np.float64)
    W = 0.0
    eps = 1e-15
    for j in range(n):
        sj = float(s[j])
        if sj < eps:
            r = 1.0 / eps
        else:
            r = 1.0 / sj

        if r > W:
            delta = r - W
            if delta > 1e12:
                delta = 1e12
            pos = j + n - 1
            if 0 <= pos < L:
                y[pos] += delta
                W += delta

        if j < L:
            W -= float(y[j])
            if W < 0.0 and W > -1e-9:
                W = 0.0

    total = float(y.sum())
    if not np.isfinite(total) or total <= 0.0:
        return uniform_weights(L)

    y /= total
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)
    return normalize_weights(y, L)


def compute_p(strategy, c, s, tau_rel, Ktop, n):
    L = c.size
    if strategy == 0:
        return softmax_weights(c, tau_rel)
    if strategy == 1:
        p1 = softmax_weights(c, tau_rel)
        p2 = dual_weights(s, n)
        return normalize_weights(0.65 * p1 + 0.35 * p2, L)
    if strategy == 2:
        p1 = softmax_weights(c, tau_rel)
        p2 = top_weights(c, Ktop)
        return normalize_weights(0.65 * p1 + 0.35 * p2, L)
    if strategy == 3:
        p1 = dual_weights(s, n)
        p2 = top_weights(c, Ktop)
        return normalize_weights(0.65 * p1 + 0.35 * p2, L)
    p1 = softmax_weights(c, tau_rel)
    p2 = dual_weights(s, n)
    p3 = top_weights(c, Ktop)
    return normalize_weights(0.5 * p1 + 0.25 * p2 + 0.25 * p3, L)


def make_candidate_from_u(s, u, eta, floor):
    if eta <= 0.0:
        return normalize_sequence_with_floor(s, floor)
    u = np.asarray(u, dtype=np.float64)
    u = np.where(np.isfinite(u), u, 0.0)
    u = np.maximum(u, 0.0)
    denom = 1.0 + float(eta) * u
    denom = np.maximum(denom, 1e-300)
    x = np.asarray(s, dtype=np.float64) / denom
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, float(floor))
    ss = float(x.sum())
    if not np.isfinite(ss) or ss <= 0.0:
        return None
    x /= ss
    return x


def add_to_archive(archive, seq, score, cap=5):
    if score is None or not np.isfinite(score):
        return
    score = float(score)
    seq = np.asarray(seq, dtype=np.float64).copy()
    if seq.size == 0:
        return

    if not archive:
        archive.append((score, seq))
        return

    if len(archive) >= cap and score >= archive[-1][0] - 1e-12:
        return

    insert = len(archive)
    for i, (old_sc, _) in enumerate(archive):
        if score < old_sc - 1e-12:
            insert = i
            break

    archive.insert(insert, (score, seq))
    if len(archive) > cap:
        archive.pop()


def try_candidate(seq, n, N, L, archive, floor):
    x = normalize_sequence_with_floor(seq, floor)
    sc = fast_score(x, n, N, L)
    add_to_archive(archive, x, sc, cap=5)
    return x, sc


def quick_random_starts(n, N, L, rng, time_limit, start_archive=None):
    archive = [] if start_archive is None else list(start_archive)
    floor = 1e-9
    end = time.monotonic() + max(0.0, float(time_limit))

    try_candidate(np.ones(n, dtype=np.float64), n, N, L, archive, floor)

    ar = np.arange(n, dtype=np.float64)
    if n > 1:
        t = ar / float(n - 1)
    else:
        t = np.zeros(n, dtype=np.float64)
    taper = np.sin(np.pi * t) ** 2
    taper = np.maximum(taper, 1e-6)

    for _ in range(10):
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(0.3 + 0.7 * rng.random())
        base = 1.0 + amp * np.cos(2.0 * np.pi * t + phase)
        base = np.maximum(base, 0.0)
        seq = base * (taper ** 1.5)
        try_candidate(seq, n, N, L, archive, floor)
        if time.monotonic() >= end:
            break

    for center in (0.15, 0.35, 0.65, 0.85):
        sigma = float(0.05 + 0.10 * rng.random())
        seq = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        try_candidate(seq, n, N, L, archive, floor)
        if time.monotonic() >= end:
            break

    count = 0
    while time.monotonic() < end:
        count += 1
        if count % 25 == 0:
            half = N // 2 + 1
            S = np.zeros(half, dtype=np.complex128)
            cut = max(1, N // 8)
            S[:cut] = (rng.normal(size=cut) + 1j * rng.normal(size=cut))
            x = np.fft.irfft(S, N)[:n]
            seq = np.abs(x) + 1e-6 * taper
        else:
            seq = np.exp(0.6 * rng.normal(size=n, loc=0.0, scale=1.0)) * taper

        try_candidate(seq, n, N, L, archive, floor)

        if count > 20000:
            break

    if not archive:
        try_candidate(np.ones(n, dtype=np.float64), n, N, L, archive, floor)

    return archive


def resample_sequence(s, m):
    s = np.asarray(s, dtype=np.float64)
    n0 = s.size
    if m <= 0:
        return np.ones(1, dtype=np.float64)
    if m == n0:
        return s.copy()
    if n0 == 1:
        return np.full(m, float(s[0]), dtype=np.float64)
    x_old = np.arange(n0, dtype=np.float64)
    x_new = np.linspace(0.0, float(n0 - 1), int(m))
    y = np.interp(x_new, x_old, s)
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)
    return normalize_sequence_with_floor(y, 1e-9)


def probe_direction(s, g, n, N, L, floor):
    s = np.asarray(s, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    best_s = normalize_sequence_with_floor(s, floor)
    best_sc = fast_score(best_s, n, N, L)
    if not np.isfinite(best_sc):
        return s, np.inf

    for tv in LP_T_VALUES:
        tv = float(tv)
        if tv <= 0.0:
            continue
        if tv >= 1.0:
            cand = g.copy()
        else:
            cand = (1.0 - tv) * s + tv * g
        cand = normalize_sequence_with_floor(cand, floor)
        sc = fast_score(cand, n, N, L)
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best_s = cand

    return best_s, best_sc


def lp_direction(s, c, n, Klp, rng, floor, max_sec=60.0):
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    L = 2 * n - 1
    Klp = int(Klp)
    if Klp <= 0 or Klp > L:
        Klp = L

    if Klp >= L:
        idx = np.arange(L, dtype=int)
    elif Klp == 1:
        idx = np.array([int(np.argmax(c))], dtype=int)
    else:
        idx = np.argpartition(c, -Klp)[-Klp:]

    extras = [n - 1, 0, L - 1, L // 2, max(0, n // 2), max(0, L - 1 - n // 2)]
    extras.extend(int(x) for x in np.linspace(0, L - 1, 24))
    ks = np.unique(np.concatenate((idx, np.asarray(extras, dtype=int))))
    R = ks.size
    if R == 0:
        return None

    # Build active interval matrix.  Column j is 1 on rows k in [j, j+n-1].
    A = np.zeros((R, n), dtype=np.float64)
    for r, k in enumerate(ks):
        k = int(k)
        start = max(0, k - n + 1)
        end = min(n - 1, k) + 1
        if end > start:
            A[r, start:end] = 1.0

    # Exact change of variables:
    #   z_j = s_j * b_j / M, then w_j = z_j.
    # Objective becomes max sum w_j / s_j, constraints are interval sums <= 1.
    s_obj = np.maximum(np.asarray(s, dtype=np.float64), 1e-6)
    if rng is not None:
        noise = 0.01 * rng.normal(size=n)
        if n > 1:
            tt = np.arange(n, dtype=np.float64) / float(n - 1)
            sign = 1.0 if rng.random() < 0.5 else -1.0
            noise += 0.025 * sign * (tt - 0.5)
        s_obj = s_obj * (1.0 + noise)
        s_obj = np.maximum(s_obj, 1e-6)

    inv_s = 1.0 / s_obj
    obj = -inv_s
    b_ub = np.ones(R, dtype=np.float64)

    res = None
    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0.0, 1.0),
            method="highs",
            options={"time_limit": float(max_sec)},
        )
    except Exception:
        try:
            res = linprog(
                obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0.0, 1.0),
                method="highs",
            )
        except Exception:
            return None

    if res is None or getattr(res, "x", None) is None:
        return None

    w = np.asarray(res.x, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = np.clip(w, 0.0, 1.0)

    b = w * inv_s
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    ss = float(b.sum())
    if not np.isfinite(ss) or ss <= 0.0:
        return None
    b /= ss
    return b


def try_lp_update(s, c, n, N, L, Klp, rng, floor, max_sec=60.0):
    g = lp_direction(s, c, n, Klp, rng, floor, max_sec=max_sec)
    base_sc = fast_score(s, n, N, L)
    if g is None:
        return s, base_sc
    cand, sc = probe_direction(s, g, n, N, L, floor)
    if not np.isfinite(sc):
        return s, base_sc
    if sc >= base_sc:
        return s, base_sc
    return cand, sc

                prev_score = fast_score(s, n, N, L)
                u = np.zeros(n, dtype=np.float64)
                stall = 0
                continue
            corr_norm = corr / mx

        u = 0.6 * u + 0.4 * corr_norm
        mean_u = float(np.dot(s, u))
        if mean_u > 1e-300 and np.isfinite(mean_u):
            u = u / mean_u
        else:
            u = np.zeros(n, dtype=np.float64)

        cand_best = None
        cand_best_sc = cur
        best_e = eta

        else:
            eta = max(eta * 0.65, 1e-4)
            stall += 1

        if it % 2000 == 0:
            tau_rel = max(0.025, tau_rel * 0.93)
            Ktop = max(18, int(Ktop * 0.92))
            if stall > 500:
                strat = (strat + 1) % len(strategies)

        long_stall = 800
        time_limit_for_restart = max(30.0, 0.05 * float(time_limit))
        if stall > long_stall or (
            time.monotonic() - best_time > time_limit_for_restart and stall > 250
        ):
            idx = int(rng.integers(0, len(archive))) if len(archive) > 0 else 0
            base = archive[idx][1].copy()
            sigma = 0.05 + 0.25 * min(1.0, stall / 5000.0)
            local_s = s.copy()
            local_sc = fast_score(local_s, n, N, L)
            for _ in range(25):
                logb = np.log(base + float(floor))
                x = logb + sigma * rng.normal(size=n)
                mx = float(np.max(x))
                cand = np.exp(x - mx)
                cand = normalize_sequence_with_floor(cand, floor)
                sc = fast_score(cand, n, N, L)
                if np.isfinite(sc) and sc < local_sc:
                    local_sc = sc
                    local_s = cand

            if local_sc < cur - 1e-12:
                s = local_s
                prev_score = local_sc
            elif local_sc < best_score - 1e-12:
                s = local_s
                prev_score = local_sc
            else:
                s = best_s.copy()
                prev_score = fast_score(s, n, N, L)

            u = np.zeros(n, dtype=np.float64)
            eta = 0.5
            stall = 0
            strat = (strat + 1) % len(strategies)

        if use_lp and time.monotonic() < end - max(60.0, 2.0 * lp_max_sec):
            interval = 10.0 if n <= 256 else (30.0 if n <= 1024 else (90.0 if n <= 2048 else 200.0))
            if stall > 350 or time.monotonic() - last_lp > interval:
                c_best, _ = compute_c_s(best_s, N, L)
                t0 = time.monotonic()
                s_new, sc_new = try_lp_update(
                    best_s, c_best, n, N, L, Klp, rng, floor, max_sec=lp_max_sec
                )
                dur = time.monotonic() - t0
                last_lp = time.monotonic()
                if dur > max(120.0, 2.5 * lp_max_sec):
                    lp_slow += 1
                if lp_slow > 3:
                    use_lp = False

                if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
                    best_s = s_new
                    best_score = sc_new
                    add_to_archive(archive, best_s, best_score, cap=5)

                if np.isfinite(sc_new) and sc_new < prev_score - 1e-12:
                    s = s_new
                    prev_score = sc_new
                    eta = 0.5
                    stall = 0

        if it % 20000 == 0:
            print(f"  len={n} local_best={best_score:.8f} cur={cur:.8f} stale={stall}")

    final_sc = fast_score(s, n, N, L)
    if np.isfinite(final_sc) and final_sc < best_score:
        best_score = final_sc
        best_s = s.copy()

    return s, best_s, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    total = max(60.0, min(float(budget_s), 1800.0))
    rng = np.random.default_rng(seed)
    floor = 1e-9

    h = None
    n_base = 4096
    try:
        hh = np.asarray(height_sequence_1, dtype=np.float64)
        if hh.ndim == 1 and hh.size > 0 and np.all(np.isfinite(hh)):
            h = np.maximum(hh, 0.0)
            if float(h.sum()) > 0.0:
                n_base = int(h.size)
    except Exception:
        h = None
        n_base = 4096

    n_base = int(max(256, min(n_base, 8192)))
    if h is None:
        h = np.ones(n_base, dtype=np.float64)

    lengths = [256, 512, 1024, 2048]
    if n_base not in lengths:
        lengths.append(n_base)
    lengths = sorted(set(lengths))

    if total < 300.0:
        lengths = [n_base]

    times = []
    for m in lengths:
        if m <= 300:
            t = 8.0
        elif m <= 600:
            t = 12.0
        elif m <= 1100:
            t = 20.0
        elif m <= 2200:
            t = 45.0
        else:
            t = 0.0
        times.append(float(t))

    if len(lengths) == 1:
        times = [total]
    else:
        coarse_sum = sum(times[:-1])
        allow = 0.45 * total
        if coarse_sum > allow and coarse_sum > 0.0:
            scale = allow / coarse_sum
            for i in range(len(lengths) - 1):
                times[i] = max(1.0, times[i] * scale)

        coarse_sum = sum(times[:-1])
        final_time = total - coarse_sum
        if final_time < 30.0:
            deficit = 30.0 - final_time
            for i in range(len(lengths) - 1):
                take = min(times[i] * 0.5, max(0.0, deficit))
                times[i] -= take
                deficit -= take
                if deficit <= 0.0:
                    break
            final_time = total - sum(times[:-1])
        times.append(max(1.0, final_time))

    global_best_s = None
    global_best_score = np.inf
    prev_best_seq = None

    for idx, m in enumerate(lengths):
        if time.monotonic() - start >= total - 3.0:
            break

        remaining = max(0.0, total - (time.monotonic() - start))
        time_budget = max(0.5, min(float(times[idx]), remaining - 1.0))
        if time_budget <= 0.0:
            continue

        N = next_pow2(2 * m - 1)
        L = 2 * m - 1
        archive = []

        if m == n_base:
            try:
                hh = np.asarray(height_sequence_1, dtype=np.float64)
                if hh.size == m:
                    try_candidate(hh, m, N, L, archive, floor)
            except Exception:
                pass

        if prev_best_seq is not None:
            seq = resample_sequence(prev_best_seq, m)
            try_candidate(seq, m, N, L, archive, floor)

        if global_best_s is not None and global_best_s.size != m:
            seq = resample_sequence(global_best_s, m)
            try_candidate(seq, m, N, L, archive, floor)

        start_time = min(5.0, max(0.5, 0.08 * time_budget))
        start_time = min(start_time, 0.2 * time_budget)
        archive = quick_random_starts(m, N, L, rng, start_time, start_archive=archive)
        if not archive:
            x = normalize_sequence_with_floor(np.ones(m, dtype=np.float64), floor)
            archive = [(fast_score(x, m, N, L), x)]

        initial_s = archive[0][1].copy()
        initial_score = archive[0][0]

        remain_after_start = max(0.0, time_budget - start_time - 0.5)
        if remain_after_start > 1.0:
            initial_s, best_s_m, best_score_m = improve_sequence(
                initial_s,
                initial_s.copy(),
                initial_score,
                remain_after_start,
                m,
                N,
                L,
                rng,
                True,
                floor,
                start_archive=archive,
            )
        else:
            best_s_m = initial_s.copy()
            best_score_m = initial_score

        prev_best_seq = best_s_m.copy()
        if np.isfinite(best_score_m) and best_score_m < global_best_score:
            global_best_score = best_score_m
            global_best_s = best_s_m.copy()

        print(
            f"[search] length={m} local_best={best_score_m:.8f} "
            f"global_best={global_best_score:.8f} elapsed={time.monotonic()-start:.1f}s"
        )

    if global_best_s is None:
        global_best_s = normalize_sequence_with_floor(np.ones(n_base, dtype=np.float64), floor)
        global_best_score = fast_sequence_score = 2.0 * float(n_base) * fast_max_conv(
            global_best_s, next_pow2(2 * n_base - 1), 2 * n_base - 1
        )

    result = [float(x) for x in np.asarray(global_best_s, dtype=np.float64)]
    try:
        final_score = float(evaluate_sequence(result))
        if not np.isfinite(final_score):
            raise ValueError
    except Exception:
        fb = normalize_sequence_with_floor(np.ones(n_base, dtype=np.float64), floor)
        result = [float(x) for x in fb]

    return result
