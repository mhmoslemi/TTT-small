import time
from scipy import sparse
from scipy.optimize import linprog

def build_conv_structure(n):
    """Precompute row/col indices for the convolution matrix structure."""
    # Column j has nonzeros at rows j, j+1, ..., j+n-1
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    for j in range(n):
        rows[j*n:(j+1)*n] = np.arange(j, j+n)
        cols[j*n:(j+1)*n] = j
    return rows, cols

def make_window(n, kind='hann'):
    if n <= 1:
        return np.ones(n)
    t = np.arange(n)

    if kind == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0*np.pi*t/m))
    elif kind == 'blackman':

    return w / s

def solve_lp(f, n, rows, cols, time_limit=20.0):
    """Solve: max sum(b) s.t. conv(f,b)_k <= max(conv(f,f)), b >= 0."""
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()
    if M < 1e-15:
        return None
    
    vals = f[cols]  # f[j] for each (row, col) entry
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None

def subgradient_step(f, lr, n):
    """One projected subgradient descent step on max(conv(f,f))."""
    conv_ff = np.convolve(f, f)
    max_val = conv_ff.max()
    # Active set: indices where conv is close to max
    threshold = max_val - 1e-6 * max_val
    active = np.where(conv_ff >= threshold)[0]
    
    if len(active) == 0:
        return f.copy()
    
    # Compute subgradient: grad[j] = (1/|S|) * sum_{k in S} f[k-j] (if valid)
    grad = np.zeros(n)
    for k in active:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        # f[k-j] for j in [j_min, j_max]
        # k-j ranges from k-j_min to k-j_max
        # So we need f[k-j_max : k-j_min+1] reversed
        # Actually: for j = j_min, k-j = k-j_min; for j = j_max, k-j = k-j_max
        # So f[k-j_min], f[k-j_min-1], ..., f[k-j_max]
        # = f[k-j_max : k-j_min+1][::-1]
        seg = f[k-j_max : k-j_min+1][::-1]
        grad[j_min:j_max+1] += seg
    
    grad /= len(active)
    
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s

"""
LP-based direction finding with gradient descent refinement to minimize
2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences of length ~4498.
Uses sparse LP (HiGHS) for powerful direction updates, subgradient descent
for fine-tuning, and periodic perturbations for exploration.
"""

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def build_conv_structure(n):
    """Precompute row/col indices for the convolution matrix A where A@b = conv(f,b).
    Column j has nonzeros at rows j..j+n-1 with values f[0],f[1],...,f[n-1]."""
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).flatten().astype(np.int32)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    return rows, cols


def make_window(n, kind='hann'):
    if n <= 1:
        return np.ones(n)
    t = np.arange(n, dtype=np.float64)
    m = float(n - 1)
    if kind == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / m))
    elif kind == 'blackman':
        w = 0.42 - 0.5 * np.cos(2.0 * np.pi * t / m) + 0.08 * np.cos(4.0 * np.pi * t / m)
    elif kind == 'blackman_harris':
        w = (0.35875 - 0.48829 * np.cos(2.0*np.pi*t/m)
             + 0.14128 * np.cos(4.0*np.pi*t/m) - 0.01168 * np.cos(6.0*np.pi*t/m))
    else:
        w = np.ones(n)
    w = np.clip(w, 0, None)
    s = w.sum()
    if s < 1e-15:
        w = np.ones(n)
        s = n
    return w / s


def fast_score(f, n):
    """Compute score directly with numpy (much faster than evaluate_sequence)."""
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def solve_lp(f, n, rows, cols, time_limit=25.0):
    """Solve: max sum(b) s.t. conv(f,b)_k <= max(conv(f,f)), b >= 0.
    Returns the optimal b or None."""
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()
    if M < 1e-15:
        return None

    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': time_limit, 'presolve': True})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def subgradient_step(f, lr, n):
    """One projected subgradient descent step on max(conv(f,f)).
    grad[j] = (2/|S|) * sum_{k in S} f[k-j] where S = argmax set."""
    conv_ff = np.convolve(f, f)
    max_val = conv_ff.max()
    threshold = max_val - 1e-10 * max_val
    active = np.where(conv_ff >= threshold)[0]

    if len(active) == 0:
        return f.copy()

    # Compute h[j] = sum_{k in active} f[k-j] using convolution
    s_vec = np.zeros(2 * n - 1)
    s_vec[active] = 1.0
    conv_h = np.convolve(f, s_vec[::-1])
    h = conv_h[n - 1:2 * n - 1][::-1]  # h[j] = sum_k s[k]*f[k-j]

    grad = 2.0 * h / len(active)

    f_new = f - lr * grad
    f_new = np.clip(f_new, 0, None)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def propose_candidate():
    """
    Main search function. Uses LP-based direction finding combined with
    subgradient descent and periodic perturbations to minimize the score.
    """
    n = len(height_sequence_1)
    base = np.array(height_sequence_1, dtype=np.float64)
    base = np.clip(base, 0, 1000)

    base = base / base.sum()

    # Power variations of Hann window
    t_arr = np.arange(n, dtype=np.float64)
    m = float(max(n - 1, 1))

        if w.sum() > 1e-10:

    # Gaussian-like window

        w = np.exp(-0.5 * ((t_arr - center) / sigma) ** 2)

    # Evaluate all candidates (use fast_score for speed, validate with evaluate_sequence)
    best_score = np.inf

        if np.isfinite(s) and s < best_score:

    print(f"Initial best score: {best_score:.6f} (n={n})", flush=True)

            best_seq = np.clip(best_seq, 0, None)
            if best_seq.sum() > 1e-10:

            s = fast_score(best_seq, n)
            if np.isfinite(s) and s < best_score:

        # === Occasionally try a global restart with a new window ===
        if no_improve >= 60 and iteration % 30 == 0:
            # Try a random weighted combination of windows
            w1 = make_window(n, 'hann')
            w2 = make_window(n, 'blackman')
            alpha = rng.random()

                s = fast_score(hybrid, n)

    if result.sum() < 0.01:

    else:
        result = result / result.sum()

    final_score = evaluate_sequence(result.tolist())
    print(f"Final score: {final_score:.6f}", flush=True)
    return result.tolist()
