import time

def compute_convolution(a):
    """Compute a * a (full convolution)."""
    return np.convolve(a, a)

def subgradient_step(a, lr):
    """One subgradient descent step on max(a*a)."""
    b = np.convolve(a, a)
    max_b = np.max(b)
    S = np.where(b == max_b)[0]  # positions of max
    
    # Gradient of max(b) w.r.t. a
    # ∂b[k]/∂a[i] = 2*a[k-i] if 0 <= k-i < n else 0
    n = len(a)
    grad = np.zeros(n)
    for k in S:
        # For this k, grad[i] += 2*a[k-i] if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # i ranges from lo to hi, and k-i ranges from k-hi to k-lo
        # a[k-i] for i in [lo, hi] is a[k-hi : k-lo+1] = a[lo..hi] reversed? No.
        # When i = lo, k-i = k-lo. When i = hi, k-i = k-hi.
        # So a[k-i] for i in [lo,hi] is a[k-hi : k-lo+1]
        grad[lo:hi+1] += 2 * a[k-hi:k-lo+1][::-1]

    # Project onto {a >= 0, sum a = 1}
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 0:
        a_new /= s
    else:
        a_new = np.ones(n) / n
    return a_new, max_b

    b = fast_convolve(a)
    max_b = float(np.max(b))
    
    # Softmax weights (numerically stable)
    logits = beta * (b - max_b)  # all <= 0
    logits = np.where(logits < -50, -50, logits)
    exp_b = np.exp(logits)
    total = float(np.sum(exp_b))
    weights = exp_b / total
    
    # Gradient via cross-correlation: grad[i] = (1/beta) * 2 * sum_k b_k * a[k-i]
    # Wait, let me re-derive:
    # f = (1/beta) * log(sum_k exp(beta * b[k]))
    # df/da[i] = (1/beta) * sum_k exp(beta*b[k])/sum_j exp(beta*b[j]) * db[k]/da[i]
    # = sum_k weights[k] * db[k]/da[i]
    # = sum_k weights[k] * 2 * a[k-i] (with bounds)
    # = 2 * (a cross-correlate weights)[i]
    # where (a cross-correlate h)[i] = sum_m a[m] * h[m+i]
    
    corr_size = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (corr_size - 1).bit_length()
    A_f = np.fft.rfft(a, fft_size)
    W_f = np.fft.rfft(weights, fft_size)
    corr = np.fft.irfft(A_f * np.conj(W_f), fft_size)[:n]
    
    grad = 2.0 * corr
    return grad, max_b
