def solve_lp(a, n):
    """Solve the LP: min M s.t. (a*b)_k <= M for all k, b >= 0, sum(b) = 1"""
    import cvxpy as cp
    
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    constraints = [cp.sum(b) == 1]
    # (a * b)_k = sum_i a[i] * b[k-i] <= M
    # For efficiency, use convolution structure
    # We can express a*b as a matrix-vector product
    # Let A be the (2n-1) x n matrix where A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Then (a*b) = A @ b
    
    # Build constraint matrix (sparse)
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or another solver

1. LP-based best-response iterations (structured sparse convolution matrix)

    """FFT-based full convolution of a with itself."""
    f = 2 * n - 1
    # Round up to next power of 2 for FFT efficiency
    f = int(2 ** np.ceil(np.log2(f)))
    fa = np.fft.rfft(a, f)
    return np.fft.irfft(fa * fa, f)[:2*n-1]

    rows = np.empty(total_nnz, dtype=np.int32)

        count = j_end - j_start + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(j_start, j_end + 1, dtype=np.int32)
        idx += count
    
    return rows, cols, m
