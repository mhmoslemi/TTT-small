import numpy as np
import time
from scipy.optimize import linprog


def build_A(f):
    """Build the constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = f[k - j_end:k - j_start + 1]
    return A

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return np.maximum(result.x, 0.0)
    return None

def build_constraint_matrix(f):
    """Build A where A[k,j] = f[k-j] if 0<=k-j<n else 0. Shape (2n-1, n)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = f[k - j_end:k - j_start + 1]
    return A


def solve_direction_lp(f, M):
    """Solve max sum(g) s.t. (f*g)_k <= M, g >= 0, g <= g_max."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    g_max = 2.0 * np.sum(f)
    bounds = [(0.0, g_max)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def make_u_shaped(n, power=2):
    """Create a U-shaped distribution on {0,...,n-1}."""
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    p = 1.0 + 10.0 * np.power(np.abs(x), power)
    p = p / np.sum(p)
    return p


def uniform_dist(n):
    return np.ones(n, dtype=np.float64) / n


def iterate_lp_direction(f, deadline, max_iter=500):
    """Run LP-based iterations starting from f. Returns (best_f, best_score)."""
    n = len(f)

        M = np.max(conv)
        
        g = solve_direction_lp(f, M)
        
        if g is None or np.sum(g) < 1e-15:
            # Fallback: random perturbation
            perturbation = np.exp(np.random.default_rng(it).normal(0, 0.05, n))
            g = f * perturbation
            g = np.maximum(g, 0)
            if np.sum(g) > 0:
                g = g / np.sum(g)
            else:
                break
        else:
            g = g / np.sum(g)
        
        # Try the update with current step
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        s = np.sum(f_new)
        if s < 1e-15:
            break
        f_new = f_new / s
        
        score = evaluate_sequence(f_new.tolist())
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0

                print(f"    it={it}, score={best_score:.6f}, M={M:.6f}")
        else:
            no_improve += 1
            if no_improve > 30:
                break
            # Reduce step size
            step = max(0.05, step * 0.8)
            # Don't update f, try again with smaller step
            continue
        
        it += 1
    
    return best_f, best_score
