def _simplex_project(v):
    """Project vector v onto the non-negative simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_subgradient(a, k_star):
    """Compute subgradient of (a*a)[k_star] w.r.t. a. O(n) time."""
    n = len(a)
    g = a.copy()
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        g[lo:hi + 1] += a[k_star - hi:k_star - lo + 1][::-1]
    return g


def _get_active_set(c, max_c, frac=0.92, max_active=100):

    if len(active) > max_active:
        order = c[active][::-1].argsort()[::-1][:max_active]
        active = active[order]
