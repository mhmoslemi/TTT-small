def project_simplex(v):
    """Project v onto the probability simplex {x : sum(x) = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    rho_star = np.max(rho[cond]) if np.any(cond) else 1
    theta = (cssv[int(rho_star)-1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def compute_conv(a, size):

    n = len(a)

"""

def project_simplex(v):
    """Project onto {x : sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = np.max(rho[cond])
    theta = (cssv[rho_star - 1] - 1) / rho_star
    return np.maximum(v - theta, 0)

def self_conv(a, size):
    """Full self-convolution via FFT."""
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)

def grad_smooth_max(a, c, beta, size):
    """Gradient of logsumexp(beta*c)/beta where c = self_conv(a)."""
    n = len(a)
    c_shifted = beta * c - beta * c.max()
    p = np.exp(c_shifted)

    # grad_j = 2 * sum_m a_m * p_{j+m}

    af = a[::-1]
    fa = np.fft.rfft(af, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa * fp, size)
    return 2.0 * conv[n-1:2*n-1]
