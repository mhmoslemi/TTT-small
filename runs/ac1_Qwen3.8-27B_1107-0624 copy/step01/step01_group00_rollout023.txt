def _proj_simplex(v, target=1.0):
    """Project onto {x>=0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
