"""
Minimize 2*n*max(self_convolution)/(sum)^2 using multi-phase optimization:
1) L-BFGS-B with soft-max surrogate on logit params
2) Active-set subgradient with Armijo line search on simplex
3) Fine polishing with adaptive peak focusing
Key insight: asymmetric distributions are essential (symmetric can't beat uniform).

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

N = 4096
_FFT_N = 1 << (2 * N - 1).bit_length()
_FFT_GRAD = 1 << (3 * N - 1).bit_length()


class SearchTimeout(Exception):
    pass


def _self_conv(p):
    """Full linear convolution p*p via FFT."""
    f = np.fft.rfft(p, _FFT_N)
    c = np.fft.irfft(f * f, _FFT_N).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    """Compute 2*N*max(p*p)."""
    c = _self_conv(p)
    return 2.0 * N * float(np.max(c))


def _grad_of_conv_peak(p, k):
    """Gradient of c_k = sum_m p_m p_{k-m} w.r.t. p.
    Result: g_i = 2 * p_{k-i} (with p_j=0 outside [0,n-1])."""
    g = np.zeros(N, dtype=np.float64)
    # g_i = 2 * p_{k-i} for valid i
    lo = max(0, k - (N - 1))
    hi = min(N - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * p[k - idx]
    return g


def _subgradient_topk(p, c, k):
    """Weighted subgradient of max(c) using top-k peaks."""
    cmax = float(np.max(c))
    idx = np.argpartition(c, -k)[-k:]
    weights = c[idx] + 1e-15
    weights = weights / float(np.sum(weights))
    g = np.zeros(N, dtype=np.float64)
    for j, ki in enumerate(idx):
        g += weights[j] * _grad_of_conv_peak(p, int(ki))
    return g


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u - (cssv - 1.0) / np.arange(1, n + 1)) > 0)
    if rho is None or rho < 0:
        rho = 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _tangent_direction(g, p):
    """Project gradient onto tangent space of simplex: d = g - mean(g) (for free coordinates)."""
    # For simplicity, use d = g - mean(g), then clip p + t*d to [0,inf] and renormalize
    return g - np.mean(g)


def _armijo_line_search(p, g, f0, c1=1e-4, t0=0.5, max_steps=12):
    """Armijo line search for p_new = project_simplex(p - t * d)."""
    d = g - np.mean(g)
    gnorm = float(np.linalg.norm(d))
    if gnorm < 1e-15:
        return p, f0
    d = d / gnorm
    t = t0
    for _ in range(max_steps):
        p_new = p - t * d
        p_new = _project_simplex(p_new)
        f_new = _score(p_new)
        if f_new <= f0 - c1 * t * gnorm:
            return p_new, f_new
        t *= 0.5
    return p, f0


def _make_power_initial(alpha):
    """p_j proportional to (j+1)^alpha / sum."""
    x = np.arange(N, dtype=np.float64)
    p = np.power(x + 1.0, alpha)
    p /= float(np.sum(p))
    return p


def _make_exp_initial(beta):
    """p_j proportional to exp(beta * j/n)."""
    x = np.arange(N, dtype=np.float64) / N
    p = np.exp(beta * x)
    p /= float(np.sum(p))
    return p


def _make_logistic_initial(beta, center=0.5):
    x = np.arange(N, dtype=np.float64) / N
    p = 1.0 / (1.0 + np.exp(-beta * (x - center)))
    p = np.maximum(p, 1e-10)
    p /= float(np.sum(p))
    return p


def _make_ramp_initial(a, b):
    """p_j = a + b*(j/(n-1)), normalized."""
    x = np.arange(N, dtype=np.float64) / (N - 1)
    p = a + b * x
    p = np.maximum(p, 1e-10)
    p /= float(np.sum(p))
    return p


def _logit_from_p(p):
    """Convert probability vector to logit parameters (for L-BFGS-B)."""
    p = np.maximum(p, 1e-12)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -40.0, 40.0)


def _p_from_logit(z):
    """Convert logit parameters to probability vector."""
    z = np.clip(np.asarray(z, dtype=np.float64), -40.0, 40.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _soft_max_objective(z, alpha):
    """Soft-max surrogate objective and gradient for L-BFGS-B."""
    global _deadline
    if time.monotonic() > _deadline:
        raise SearchTimeout
    p = _p_from_logit(z)
    c = _self_conv(p)
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    u = np.clip(u, -50, 50)
    e = np.exp(u)
    s = float(np.sum(e))
    if s < 1e-30:
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s
    # Gradient of w.c w.r.t. p: g_i = 2 * sum_k w_k * p_{k-i}
    # Which is 2 * (w_reflected * p) evaluated at the right slice
    w_rev = w[::-1]
    conv_full = np.fft.irfft(np.fft.rfft(w_rev, _FFT_GRAD) * np.fft.rfft(p, _FFT_GRAD), _FFT_GRAD).real
    gp = 2.0 * conv_full[N - 1:2 * N - 1]
    # Gradient w.r.t. z: gz_i = p_i * (gp_i - sum(gp*p)) for i >= 1
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(J), gz

        cmax = float(np.max(c))
        score = 2.0 * N * cmax

        if score < _best_score_total - 1e-15:
            _best_score_total = score
            _best_p_total = p.copy()

        # Focus on single highest peak
        k_star = int(np.argmax(c))
        g = _grad_of_conv_peak(p, k_star)
        d = g - np.mean(g)
        gnorm = float(np.linalg.norm(d))
        if gnorm < 1e-14:
            break
        d /= gnorm

        # Try small step
        p_trial = p - t * d
        p_trial = _project_simplex(p_trial)
        s_trial = _score(p_trial)

        if s_trial < score - 1e-12:
            p = p_trial
            score = s_trial
            t = min(0.05, t * 1.1)
        else:
            t *= 0.8
            if t < 1e-6:
                # Try random micro-perturbation
                p_trial = p + rng.normal(0, 0.005, N)
                p_trial = _project_simplex(p_trial)
                s_trial = _score(p_trial)
                if s_trial < score - 1e-12:
                    p = p_trial
                    score = s_trial
                    t = 0.01
                else:
                    break

    return _best_p_total.copy() if _best_p_total is not None else p


def _try_topk_sweep(p0, deadline, rng):
    """Try different topk values and focus strategies."""
    global _best_p_total, _best_score_total
    p = p0.copy()
    score = _score(p)

    for topk in [1, 2, 3, 4, 5, 7, 10, 15]:
        if time.monotonic() > deadline:
            break
        # Run a short subgradient with this topk
        p_local = p.copy()
        s_local = score
        t = 0.3
        for step in range(200):
            if time.monotonic() > deadline:
                break
            c = _self_conv(p_local)
            cmax = float(np.max(c))
            s_local = 2.0 * N * cmax
            if s_local < _best_score_total - 1e-15:
                _best_score_total = s_local
                _best_p_total = p_local.copy()

            g = _subgradient_topk(p_local, c, topk)
            d = g - np.mean(g)
            gnorm = float(np.linalg.norm(d))
            if gnorm < 1e-14:
                break
            d /= gnorm

            accepted = False
            for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                tt = t * factor
                p_trial = p_local - tt * d
                p_trial = _project_simplex(p_trial)
                s_trial = _score(p_trial)
                if s_trial < s_local - 1e-11:
                    p_local = p_trial
                    s_local = s_trial
                    accepted = True
                    break
            if accepted:
                t = min(1.0, t * 1.1)
            else:
                t *= 0.7

        p = p_local
        score = s_local

    return _best_p_total.copy() if _best_p_total is not None else p


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """Main search function. Returns best non-negative sequence found."""
    global _best_p_total, _best_score_total, _deadline

    start = time.monotonic()
    budget_s = float(budget_s)
    end = start + max(1.0, budget_s) - 3.0

    _best_p_total = None
    _best_score_total = float('inf')

    rng = np.random.default_rng(seed)

    # --- Generate initial candidates ---
    candidates = []

    # From previous best if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, None)
            if base.size != N:
                old_x = np.linspace(0, 1, base.size)
                new_x = np.linspace(0, 1, N)
                base = np.interp(new_x, old_x, base)
            s_base = float(np.sum(base))
            if s_base > 0.01:
                base = base / s_base
                candidates.append(base)
                candidates.append(base[::-1].copy())  # reversed version
    except Exception:
        pass

    # Structured candidates
    struct_params = [
        ('power_0.3', _make_power_initial(0.3)),
        ('power_0.5', _make_power_initial(0.5)),
        ('power_0.7', _make_power_initial(0.7)),
        ('power_1.0', _make_power_initial(1.0)),
        ('power_1.5', _make_power_initial(1.5)),
        ('power_2.0', _make_power_initial(2.0)),
        ('exp_1', _make_exp_initial(1.0)),
        ('exp_2', _make_exp_initial(2.0)),
        ('exp_3', _make_exp_initial(3.0)),
        ('exp_5', _make_exp_initial(5.0)),
        ('logi_2', _make_logistic_initial(2.0)),
        ('logi_4', _make_logistic_initial(4.0)),
        ('logi_6', _make_logistic_initial(6.0)),
        ('ramp_0.1_0.9', _make_ramp_initial(0.1, 0.9)),
        ('ramp_0.05_1.0', _make_ramp_initial(0.05, 1.0)),
        ('ramp_0.0_1.0', _make_ramp_initial(0.0, 1.0)),
    ]
    for name, p in struct_params:
        candidates.append(p)
        candidates.append(p[::-1].copy())  # reversed

    # Random perturbations of good candidates
    for i in range(min(8, len(candidates))):
        p_pert = candidates[i] + rng.normal(0, 0.05, N)
        p_pert = _project_simplex(p_pert)
        candidates.append(p_pert)

    # Score all candidates
    scored = []
    for p in candidates:
        s = _score(p)
        scored.append((s, p))
    scored.sort(key=lambda x: x[0])

    print(f"Initial candidates: {len(scored)}, best initial score: {scored[0][0]:.6f}")
    for i in range(min(5, len(scored))):
        print(f"  cand {i}: score={scored[i][0]:.6f}")

    # Update global best
    for s, p in scored[:5]:
        if s < _best_score_total:
            _best_score_total = s
            _best_p_total = p.copy()

    # --- Phase 1: L-BFGS-B from top candidates ---
    phase1_deadline = start + 150.0
    top_k_lbfgs = min(5, len(scored))
    alpha = 5e6

    for i in range(top_k_lbfgs):
        if time.monotonic() > phase1_deadline:
            break
        p_init = scored[i][1].copy()
        rem = max(60.0, phase1_deadline - time.monotonic())
        maxfun = int(min(2000, rem / 0.05))
        p_result = _lbfgs_from_p(p_init, alpha, maxfun, phase1_deadline)
        if _best_p_total is not None:
            print(f"  L-BFGS cand {i}: best={_best_score_total:.6f}")

    # --- Phase 2: Subgradient with line search ---
    phase2_deadline = min(start + budget_s - 300.0, start + budget_s * 0.65)
    if _best_p_total is not None:
        p_sg = _subgradient_phase(_best_p_total.copy(), phase2_deadline, rng)
        print(f"  After SG phase: best={_best_score_total:.6f}")

    # --- Phase 3: Top-k sweep ---
    phase3_deadline = min(start + budget_s - 150.0, start + budget_s * 0.82)
    if _best_p_total is not None:
        p_sweep = _try_topk_sweep(_best_p_total.copy(), phase3_deadline, rng)
        print(f"  After topk sweep: best={_best_score_total:.6f}")

    # --- Phase 4: Fine polish ---
    phase4_deadline = end - 5.0
    if _best_p_total is not None:
        p_final = _fine_polish(_best_p_total.copy(), phase4_deadline, rng)
        print(f"  After fine polish: best={_best_score_total:.6f}")

    # --- Phase 5: A few more L-BFGS runs from perturbed best ---
    if _best_p_total is not None and time.monotonic() < end - 20.0:
        for trial in range(3):
            if time.monotonic() > end - 10.0:
                break
            p_pert = _best_p_total + rng.normal(0, 0.02, N)
            p_pert = _project_simplex(p_pert)
            rem = end - 10.0 - time.monotonic()
            if rem > 15.0:
                maxfun = int(min(500, rem / 0.04))
                _lbfgs_from_p(p_pert, 1e7, maxfun, end - 10.0)

    # --- Final: second detailed subgradient from current best ---
    if _best_p_total is not None and time.monotonic() < end - 5.0:
        p_final2 = _subgradient_phase(_best_p_total.copy(), end - 3.0, rng, verbose=False)
        print(f"  Final: best={_best_score_total:.6f}")

    # Construct output sequence
    if _best_p_total is None:
        seq = [1.0 / N] * N
    else:
        p_out = _best_p_total.copy()
        p_out = np.clip(p_out, 0.0, 1000.0)
        s_out = float(np.sum(p_out))
        if s_out < 0.01:
            seq = [1.0 / N] * N
        else:
            seq = [float(v) for v in p_out]

    # Validate
    try:
        val = float(evaluate_sequence(seq))
    except Exception:
        val = float('inf')

    if not np.isfinite(val) or val > 2.0:
        seq = [1.0 / N] * N
        val = 2.0

    # Also check if height_sequence_1 is better
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < val:
                seq = [float(v) for v in base]
                val = base_score
    except Exception:
        pass

    elapsed = time.monotonic() - start
    print(f"FINAL: score={val:.6f}, n={len(seq)}, time={elapsed:.1f}s")
    return seq
