    A = np.zeros((2*n - 1, n))

        # i.e., A[k, j] = a[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = a
    return A

    n = len(a)
    S = a.sum()
    if S < 1e-10:

    c = np.convolve(a, a, mode='full')

      sum(d) = 0
      a[i] + d[i] >= 0 for all i
      |d[i]| <= epsilon (if epsilon is not None)
    
    Returns (d, t) or (None, None) if infeasible.

    c_aa = np.convolve(a, a, mode='full')
    M = c_aa.max()

    # Variables: d (n) + t (1) = n+1 variables
    # Minimize t: c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # Constraint k: conv(a,a)[k] + 2*(conv(a,d))[k] - t <= 0
    # i.e.: 2*A_conv[k,:] @ d - t <= -c_aa[k]

"""
Direct LP-based optimization with asymmetric initializations for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses linearized auto-convolution in LP subproblems,
multiplicative weight updates, and diverse lopsided starting points.

import numpy as np
import time
from scipy.optimize import linprog


def build_convolve_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ d = convolve(a, d)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def lp_find_direction(a, epsilon=None):
    """
    Solve LP: minimize t s.t. conv(a,a)[k] + 2*(conv(a,d))[k] <= t,
    sum(d)=0, a[i]+d[i]>=0, |d[i]|<=epsilon.
    Returns (d, t) or (None, None).
    """
    n = len(a)
    c_aa = np.convolve(a, a, mode='full')
    
    A_conv = build_convolve_matrix(a)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ineq = np.zeros((2 * n - 1, n + 1))
    A_ineq[:, :n] = 2.0 * A_conv
    A_ineq[:, n] = -1.0
    b_ineq = -c_aa
    
    bounds = []
    if epsilon is None:
        epsilon = 0.5 * a.max()
    for i in range(n):
        lo = max(-a[i], -epsilon)
        hi = epsilon
        bounds.append((lo, hi))
    bounds.append((None, None))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n], res.x[n]
    except Exception:
        pass
    return None, None


def lp_classic(a):
    """Classic LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c_aa = np.convolve(a, a, mode='full')
    M = c_aa.max()
    
    A_conv = build_convolve_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def multiplicative_update(a, lr, peak_fraction=0.7):
    """Multiplicative weight update to reduce peak contributions."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    threshold = peak_fraction * M
    peak_mask = c > threshold
    
    peak_indices = np.where(peak_mask)[0]
    contrib = np.zeros(n)
    for i in range(n):
        js = peak_indices - i
        valid = (js >= 0) & (js < n)
        if valid.any():
            contrib[i] = a[js[valid]].sum()
    
    avg_c = contrib.mean() + 1e-15
    update = np.exp(-lr * contrib / avg_c)
    a_new = a * update
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() < 1e-10:
        return a
    return a_new / a_new.sum()

def make_ramp(n, direction=1):
    """Linear ramp: high at one end, low at the other."""
    x = np.arange(n, dtype=np.float64)
    if direction == 1:
        return (n - x)
    else:
        return (x + 1)


def make_exp_decay(n, alpha=2.0):
    """Exponential decay from one end."""
    x = np.arange(n, dtype=np.float64)
    return np.exp(-alpha * x / n)


def make_quadratic_decay(n):
    """Quadratic decay."""
    x = np.arange(n, dtype=np.float64)
    return (1.0 - x / n) ** 2 + 0.1


def make_offset_gaussian(n, center_frac=0.3, sigma_frac=0.2):
    """Gaussian centered off-center."""
    center = center_frac * n
    sigma = sigma_frac * n
    x = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((x - center) / sigma) ** 2)


def make_bimodal(n):
    """Large at both ends, small in middle (U-shape)."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return 1.0 + 0.5 * ((x - center) / center) ** 2


def make_asym_triangular(n, peak_frac=0.3):
    """Asymmetric triangular: peak off-center."""
    x = np.arange(n, dtype=np.float64)
    peak = peak_frac * (n - 1)
    if x <= peak:
        val = (x / peak) if peak > 0 else 0
    else:
        val = (n - 1 - x) / (n - 1 - peak) if (n - 1 - peak) > 0 else 0
    return np.maximum(val, 0.01)


def make_random_asym(n, rng):
    """Random with asymmetric bias."""
    base = np.abs(rng.normal(0, 1, n))
    bias = np.linspace(0, 1, n) ** rng.uniform(0.5, 3.0)
    return base * (0.5 + bias)

        a_opt, score = optimize_hybrid(a, config_deadline, rng, verbose=(n <= 512))
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n}, init={name})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")
    
    # Final polish on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nFinal polish on best (n={len(best_seq)}, score={best_score:.6f})...")
        polish_deadline = deadline
        best_seq, best_score = optimize_hybrid(best_seq, polish_deadline, rng, verbose=True)
        print(f"  After polish: score={best_score:.6f}")
    
    # If we didn't find anything, use a default
    if best_seq is None:
        best_seq = make_ramp(256)
        best_seq = best_seq / best_seq.sum()
        best_score = compute_objective(best_seq)
        print(f"Default: score={best_score:.6f}")
    
    # Ensure valid for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (1.0 / best_seq.sum())
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")
    
    return [float(x) for x in best_seq]
