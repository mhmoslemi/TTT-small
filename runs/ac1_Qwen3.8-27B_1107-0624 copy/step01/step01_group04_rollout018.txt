def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

def build_convolution_matrix(f, n):
    """Build the matrix A such that (A @ g)_k = (f*g)_k for k=0..2n-2."""
    # (f*g)_k = sum_{j} f_j * g_{k-j} where g_{k-j} is valid if 0 <= k-j < n
    # So (f*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j}
    # Let i = k-j, so j = k-i, and i ranges from max(0, k-n+1) to min(n-1, k)
    # But we want A[k, i] * g_i, so A[k, i] = f_{k-i} when 0 <= k-i < n
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A


def lp_solve_full(f, n, M, time_limit=30.0):
    """
    Solve the LP:
        maximize sum(g)
        subject to A @ g <= M (all 2n-1 constraints)
                   g >= 0
    where (A @ g)_k = (f*g)_k
    """
    if _scipy_opt is None:
        return None
    
    A = build_convolution_matrix(f, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    
    try:
        res = _scipy_opt.linprog(
            cobj,
            A_ub=A,
            b_ub=b,
            bounds=(0.0, None),
            method='highs',
            options={'time_limit': float(time_limit)}
        )
    except Exception:
        return None
    
    if res is None or not getattr(res, 'success', False):
        return None
    
    g = np.asarray(res.x, dtype=np.float64).ravel()
    g = np.clip(g, 0.0, None)
    return g

    G = np.fft.irfft(Pf * Pg, L)[:2*n-1].real
    H = np.fft.irfft(Pg * Pg, L)[:2*n-1].real

    G = np.maximum(G, 0)
    H = np.maximum(H, 0)

            best_t = t

    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return np.inf
    if n <= 2048:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(384, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / max(bs, 1e-300))
    return [float(x) for x in b]


def try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    return val if np.isfinite(val) else np.inf


def build_conv_matrix(f, n):
    """Build matrix A such that (A @ g)[k] = (f*g)[k] for k=0..2n-2."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def find_best_step_fft(f, g, n, S0, n_grid=80):
    """
    Find optimal t in [0,1] minimizing score of (1-t)*f + t*g.
    Uses precomputed FFTs of f and g for efficiency.
    Returns (best_t, best_score).
    """
    L = next_power_of_two(2 * n - 1)
    Pf = np.fft.rfft(f, L)
    Pg = np.fft.rfft(g, L)

    F = np.fft.irfft(Pf * Pf, L)[:2 * n - 1].real
    G = np.fft.irfft(Pf * Pg, L)[:2 * n - 1].real
    H = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real

    F = np.maximum(F, 0.0)
    G = np.maximum(G, 0.0)
    H = np.maximum(H, 0.0)

    # h_k(t) = F_k + 2(G_k - F_k)*t + (F_k - 2G_k + H_k)*t^2
    A_coef = F - 2.0 * G + H
    B_coef = 2.0 * (G - F)

    best_t = 0.0
    best_val = np.inf

    ts = np.linspace(0.0, 1.0, n_grid, dtype=np.float64)
    for t in ts:
        h = F + B_coef * t + A_coef * t * t
        val = 2.0 * n * float(np.max(h)) / (S0 * S0)
        if val < best_val:
            best_val = val
            best_t = float(t)

    # Refine around best_t with a finer grid
    if 0.0 < best_t < 1.0:
        delta = 1.0 / n_grid
        ts2 = np.linspace(max(0.0, best_t - 2 * delta), min(1.0, best_t + 2 * delta), 40, dtype=np.float64)
        for t in ts2:
            h = F + B_coef * t + A_coef * t * t
            val = 2.0 * n * float(np.max(h)) / (S0 * S0)
            if val < best_val:
                best_val = val
                best_t = float(t)

    return best_t, best_val


def lp_iteration(p, budget_s, max_iters=300):
    """
    Full-constraint LP-based cutting plane iteration.
    For n <= 1536, uses all 2n-1 constraints.
    For larger n, uses a subset with adaptive constraint addition.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        return p, actual_score_fft(p)

    if _scipy_opt is None:
        return p, actual_score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        f = greedy_normalized(n) * S0
        cur_score = actual_score_fft(f)

    best_f = f.copy()
    best_score = float(cur_score)

    # Determine if we can use full constraints
    use_full = (2 * n - 1) * n <= 4000000  # ~31 MB

    if use_full:
        A_full = build_conv_matrix(f, n)
    else:
        A_full = None

    end = time.monotonic() + max(0.0, budget_s - 5.0)
    stagnant = 0
    min_improvement = 1e-7

    # For non-full case, track active rows
    rows = list(range(2 * n - 1)) if not use_full else None
    if not use_full:
        rows = sorted(np.unique(np.linspace(0, 2 * n - 2, min(1500, 2 * n - 1), dtype=int).astype(int)).tolist())

    it = 0
    while time.monotonic() < end and it < max_iters:
        it += 1
        rem = end - time.monotonic()
        if rem < 10.0:
            break

        # Solve LP
        if use_full:
            # Rebuild A if f changed (it does every iteration)
            A = build_conv_matrix(f, n)
            b = np.full(2 * n - 1, float(np.max(np.convolve(f, f))) + 1e-10, dtype=np.float64)
            n_constraints = 2 * n - 1
        else:
            A = np.zeros((len(rows), n), dtype=np.float64)
            for ii in range(len(rows)):
                k = rows[ii]
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
            M_curr = float(np.max(np.convolve(f, f)))
            b = np.full(len(rows), M_curr + 1e-10, dtype=np.float64)
            n_constraints = len(rows)

        cobj = -np.ones(n, dtype=np.float64)
        tl = max(5.0, min(60.0, rem * 0.3))

        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method='highs',
                options={'time_limit': float(tl)}
            )
        except Exception:
            try:
                res = _scipy_opt.linprog(
                    cobj, A_ub=A, b_ub=b,
                    bounds=(0.0, None), method='highs'
                )
            except Exception:
                break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        g = g * (S0 / sg)

        # Check violations for non-full case
        if not use_full:
            conv_fg = np.convolve(f, g)
            M_curr = float(np.max(np.convolve(f, f)))
            tol = 1e-8 + 1e-8 * M_curr
            violations = conv_fg - M_curr
            if np.max(violations) > tol:
                # Add violated rows
                if len(rows) < min(3000, 2 * n - 1):
                    order = np.argsort(violations)[::-1]
                    added = 0
                    row_set = set(rows)
                    for ix in order:
                        if added >= 200 or added >= n_constraints // 2:
                            break
                        ix = int(ix)
                        if violations[ix] > tol and ix not in row_set:
                            rows.append(ix)
                            row_set.add(ix)
                            added += 1
                    if added == 0 and float(np.max(violations)) > 10 * tol:
                        break
                    continue
                elif float(np.max(violations)) > 10 * tol:
                    break
            else:
                # No violations, proceed
                pass

        # Find best step size
        best_t, best_sc = find_best_step_fft(f, g, n, S0, n_grid=60)

        if not np.isfinite(best_sc) or best_t <= 0.0:
            break

        fn = (1.0 - best_t) * f + best_t * g
        fn = np.clip(fn, 0.0, None)
        sn = float(fn.sum())
        if sn <= 1e-12:
            break
        fn = fn * (S0 / sn)

        # Verify score
        actual_sc = actual_score_fft(fn)
        if not np.isfinite(actual_sc):
            break

        if actual_sc < best_score - min_improvement:
            f = fn
            cur_score = actual_sc
            best_f = f.copy()
            best_score = cur_score
            stagnant = 0
            if it % 10 == 0:
                print("  LP iter %d: score=%.8f" % (it, best_score))
        else:
            stagnant += 1
            if stagnant >= 8:
                break
            # Try smaller step
            best_t2, best_sc2 = find_best_step_fft(f, g, n, S0, n_grid=40)
            if best_t2 > 0:
                # Use half the step
                t_small = best_t2 * 0.5
                fn2 = (1.0 - t_small) * f + t_small * g
                fn2 = np.clip(fn2, 0.0, None)
                sn2 = float(fn2.sum())
                if sn2 > 1e-12:
                    fn2 = fn2 * (S0 / sn2)
                    sc2 = actual_score_fft(fn2)
                    if np.isfinite(sc2) and sc2 < best_score - min_improvement:
                        f = fn2
                        cur_score = sc2
                        best_f = f.copy()
                        best_score = cur_score
                        stagnant = 0
                        continue
            break

    final_score = actual_score_fft(best_f)
    if final_score < best_score:
        best_score = final_score
    return normalize_nonneg(best_f), best_score


def adam_optimize(p, budget_s, seed=0):
    """Adam optimizer with log-sum-exp smoothed max for the convolution objective."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.02
    elif n < 2048:
        lr = 0.012
    else:
        lr = 0.008

    mu, nu, eps = 0.9, 0.999, 1e-8
    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Alpha: start smooth (high) and get sharper
        alpha = 4.0 + 20.0 * np.exp(-3.5 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Smoothed weights
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient of smoothed max wrt p
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_arr = mu * m_arr + (1.0 - mu) * grad_x
        v_arr = nu * v_arr + (1.0 - nu) * (grad_x * grad_x)
        m_hat = m_arr / (1.0 - mu ** t)
        v_hat = v_arr / (1.0 - nu ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Load provided sequence
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                sc = actual_score_fft(arr)
                print("Provided: n=%d, score=%.6f" % (arr.size, sc))
                if np.isfinite(sc):
                    record(arr, sc)
                    best_p = arr.copy()
                    best_score = float(sc)
                    best_by_n[arr.size] = (float(sc), arr.copy())
        except Exception:
            pass

    # Quick baselines
    print("=== Phase 0: Baselines ===")
    base_lengths = [192, 256, 320, 384, 448, 512, 640, 768, 1024]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        p = greedy_normalized(n)
        sc = actual_score_fft(p)
        record(p, sc)
        p2 = power_normalized(n, 0.45)
        sc2 = actual_score_fft(p2)
        record(p2, sc2)
        p3 = power_normalized(n, 0.55)
        sc3 = actual_score_fft(p3)
        record(p3, sc3)

    print("After baselines: best=%.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Phase 1: LP optimization on most promising lengths
    print("=== Phase 1: LP Optimization ===")
    if _scipy_opt is not None:
        # Sort candidates by score, take top few
        sorted_items = sorted(best_by_n.items(), key=lambda x: x[1][0])
        lp_lengths = [item[0] for item in sorted_items[:4]]
        # Ensure we have 384 and nearby
        for tn in [256, 384, 512]:
            if tn not in lp_lengths and len(lp_lengths) < 6:
                if tn in best_by_n:
                    lp_lengths.append(tn)

        for n in lp_lengths:
            rem = deadline - time.monotonic()
            if rem < 80.0:
                break
            rec = best_by_n.get(n)
            if rec is None:
                base = greedy_normalized(n)
            else:
                base = rec[1]
            alloc = min(500.0, max(60.0, 0.35 * rem))
            print("  LP: n=%d, alloc=%.0fs, start_score=%.6f" % (n, alloc, best_by_n.get(n, (99,))[0]))
            p_out, f_out = lp_iteration(base, alloc, max_iters=400)
            if np.isfinite(f_out):
                record(p_out, f_out)
                print("  LP done: n=%d, score=%.6f" % (n, f_out))

        # Second round on the best
        rem = deadline - time.monotonic()
        if rem > 120 and best_p is not None:
            n_best = int(best_p.size)
            alloc = min(400.0, rem - 30.0)
            print("  LP round 2: n=%d, alloc=%.0fs" % (n_best, alloc))
            p_out, f_out = lp_iteration(best_p, alloc, max_iters=300)
            if np.isfinite(f_out):
                record(p_out, f_out)
                print("  LP round 2: score=%.6f" % f_out)
    else:
        print("  scipy not available, skipping LP")

    # Phase 2: Adam polish
    print("=== Phase 2: Adam Polish ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            n_best = int(best_p.size)
            alloc = min(rem - 20.0, 350.0)
            print("  Adam: n=%d, alloc=%.0fs" % (n_best, alloc))
            p_out, f_out = adam_optimize(best_p, alloc, seed=int(seed))
            if np.isfinite(f_out):
                record(p_out, f_out)
                print("  Adam: score=%.6f" % f_out)

        # Perturbed Adam
        rem = deadline - time.monotonic()
        for i in range(3):
            if rem < 45.0:
                break
            sigma = 0.04 + 0.03 * i
            noise = np.exp(rng.normal(0.0, sigma, best_p.size))
            p_pert = normalize_nonneg(best_p * noise)
            alloc = min(rem - 15.0, 180.0)
            p_out, f_out = adam_optimize(p_pert, alloc, seed=int(seed) + 200 + i)
            if np.isfinite(f_out):
                record(p_out, f_out)
            rem = deadline - time.monotonic()

    # Phase 3: Final LP+Adam combo
    print("=== Phase 3: Final Refinement ===")
    if _scipy_opt is not None and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 80.0:
            n_best = int(best_p.size)
            lp_alloc = min(rem * 0.55, 250.0)
            p_out, f_out = lp_iteration(best_p, lp_alloc, max_iters=200)
            if np.isfinite(f_out):
                record(p_out, f_out)
            rem2 = deadline - time.monotonic()
            if rem2 > 30.0:
                p_out2, f_out2 = adam_optimize(p_out, min(rem2 - 10.0, 150.0), seed=int(seed) + 999)
                if np.isfinite(f_out2):
                    record(p_out2, f_out2)

    # Phase 4: Try a few new lengths if time remains
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 100:
            # Try lengths near the best
            n_best = int(best_p.size)
            for n_try in [n_best - 32, n_best + 32, n_best - 64, n_best + 64]:
                if n_try <= 0 or n_try > 4096:
                    continue
                if rem < 40:
                    break
                # Interpolate/resample from best_p to new length
                indices = np.linspace(0.0, best_p.size - 1.0, n_try)
                p_resamp = np.interp(indices, np.arange(best_p.size, dtype=float), best_p)
                p_resamp = normalize_nonneg(p_resamp)
                alloc = min(rem - 20.0, 120.0)
                if _scipy_opt is not None:
                    p_out, f_out = lp_iteration(p_resamp, alloc, max_iters=150)
                else:
                    p_out, f_out = adam_optimize(p_resamp, alloc, seed=int(seed) + n_try)
                if np.isfinite(f_out):
                    record(p_out, f_out)
                rem = deadline - time.monotonic()

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score_fft(best_p)

    print("=== Final: score=%.6f, n=%d ===" % (best_score, best_p.size))

    # Validate with evaluator
    candidates = [best_p]
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in items[:3]:
        if p is not None and p is not best_p:
            candidates.append(p)

    for cand in candidates:
        if deadline - time.monotonic() < 8.0:
            break
        try:
            seq = scale_output(cand, 10.0)
            gscore = try_global_eval(seq)
            if np.isfinite(gscore) and gscore < best_score:
                best_score = gscore
                best_p = np.asarray(cand, dtype=np.float64).ravel()
                print("Evaluator: score=%.6f" % gscore)
        except Exception:
            pass

    try:
        final_seq = scale_output(best_p, 10.0)
    except Exception:
        final_seq = scale_output(greedy_normalized(384), 10.0)

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative step-function sequence minimizing
    2*n*max(convolve(seq, seq)) / sum(seq)^2.

    Uses full-constraint LP iteration as the primary optimizer,
    with FFT-based grid search for step size, and Adam as a polish tool.
    """
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in proposal: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
