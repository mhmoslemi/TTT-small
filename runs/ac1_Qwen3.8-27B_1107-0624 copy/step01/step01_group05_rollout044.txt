"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multi-stage L-BFGS-B with log-sum-exp surrogate, followed by aggressive
subgradient polishing with adaptive active sets and backtracking line search.

import numpy as np

    from scipy.optimize import minimize
except Exception:

def _fft_size(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _self_conv(p):
    n = len(p)
    sz = _fft_size(2 * n - 1)
    f = np.fft.rfft(p, sz)
    c = np.fft.irfft(f * f, sz).real[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    n = len(p)
    c = _self_conv(p)
    return 2.0 * n * float(np.max(c))


def _probs_from_z(z, n):
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(n, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _grad_p_from_weights(p, w, n):
    """Gradient of weighted sum of convolution coefficients w.r.t. p."""
    sz = _fft_size(3 * n - 1)
    fa = np.fft.rfft(p[::-1], sz)
    fb = np.fft.rfft(w, sz)
    conv = np.fft.irfft(fa * fb, sz).real
    h = conv[n - 1:2 * n - 1]
    return 2.0 * h


def _make_initial_points(n, rng):
    """Generate diverse initial points in logit space."""
    x = np.linspace(0.0, 1.0, n)
    points = []

    # Uniform
    points.append(np.zeros(n - 1))

    # Logistic shapes
    for beta in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0]:
        for sign in [1, -1]:
            f = 1.0 / (1.0 + np.exp(-sign * beta * (x - 0.5)))
            f = np.maximum(f, 1e-12)
            z = np.log(f[1:] / f[0])
            points.append(np.clip(z, -40, 40))

    # Power law
    for a in [0.5, 1.0, 2.0, 3.0]:
        for rev in [False, True]:
            if rev:
                f = np.power(1.0 - x + 1e-6, a)
            else:
                f = np.power(x + 1e-6, a)
            f = np.maximum(f, 1e-12)
            z = np.log(f[1:] / f[0])
            points.append(np.clip(z, -40, 40))

    # Parabolic (inverted U)
    for width in [0.3, 0.5, 0.7, 0.9]:
        f = np.exp(-((x - 0.5) / width) ** 2)
        f = np.maximum(f, 1e-12)
        z = np.log(f[1:] / f[0])
        points.append(np.clip(z, -40, 40))

    # U-shaped
    for width in [0.3, 0.5, 0.7]:
        f = np.exp(((x - 0.5) / width) ** 2)
        f = np.maximum(f, 1e-12)
        z = np.log(f[1:] / f[0])
        points.append(np.clip(z, -40, 40))

    # Random
    for _ in range(30):

        z = rng.normal(0, 1.5) * t
        z += rng.normal(0, 0.8) * t * t

            z += rng.normal(0, 1.0 / (fq + 1)) * np.sin(2 * np.pi * fq * x + rng.uniform(0, 6.28))
        points.append(np.clip(z, -40, 40))

    return points


def _objective_and_grad(x, n, alpha, deadline):
    global global_best_seq, global_best_score
    if time.monotonic() >= deadline:
        raise Exception("timeout")

    z = np.clip(x, -50.0, 50.0)
    p = _probs_from_z(z, n)
    c = _self_conv(p)
    hard = 2.0 * n * float(np.max(c))

    if np.isfinite(hard) and hard < global_best_score:
        global_best_score = float(hard)
        global_best_seq = [float(v) for v in p]

    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    gp = _grad_p_from_weights(p, w, n)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e8, 1e8)
    return float(J), gz


def _lbfgs_optimize(z0, n, alpha, max_fun, deadline):
    if minimize is None:
        return
    bounds = [(-50.0, 50.0)] * (n - 1)
    try:
        minimize(
            _objective_and_grad,
            np.copy(z0),
            args=(n, alpha, deadline),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={"maxfun": max_fun, "maxiter": max_fun // 2,
                     "ftol": 1e-15, "gtol": 1e-10, "maxls": 50, "disp": False},
        )
    except Exception:
        pass


def _subgradient_polish(z0, n, deadline, rng, topk=16):
    """Aggressive subgradient polishing with backtracking line search."""
    global global_best_seq, global_best_score

    z = np.clip(z0, -50.0, 50.0).copy()
    p = _probs_from_z(z, n)
    score = _score(p)
    if score < global_best_score:
        global_best_score = score
        global_best_seq = [float(v) for v in p]

    max_steps = 5000
    fails = 0

    for step_idx in range(max_steps):
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p)
        cmax = float(np.max(c))

        # Adaptive active set: include coefficients above 95% of max
        threshold = cmax * 0.95
        active_mask = c >= threshold
        if not np.any(active_mask):
            idx = np.argpartition(c, -topk)[-topk:]
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[idx] = True

        # Also ensure top-k are included
        top_idx = np.argpartition(c, -topk)[-topk:]
        active_mask[top_idx] = True

        # Weights: proportional to (c - cmax + epsilon)^2 on active set
        act_c = c[active_mask]
        w_vals = (act_c - cmax + 1e-10) ** 2 + 1e-20
        w = np.zeros_like(c)
        w[active_mask] = w_vals / float(np.sum(w_vals))

        gp = _grad_p_from_weights(p, w, n)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        if gnorm < 1e-15:
            break

        direction = -gz / gnorm

        # Backtracking line search
        best_step = None
        best_new_score = score
        for step in [0.5, 0.3, 0.2, 0.1, 0.05, 0.025, 0.01, 0.005]:
            zt = z + step * direction
            zt = np.clip(zt, -50.0, 50.0)
            pt = _probs_from_z(zt, n)
            st = _score(pt)
            if st < best_new_score:
                best_new_score = st
                best_step = step
                best_zt = zt
                best_pt = pt

        if best_step is not None:
            z = best_zt
            p = best_pt
            score = best_new_score
            if score < global_best_score:
                global_best_score = score
                global_best_seq = [float(v) for v in p]
            fails = 0
        else:
            # Try random perturbation
            improved = False
            for _ in range(5):
                noise_scale = max(0.005, 0.05 * (1.0 - step_idx / max_steps))
                zt = z + rng.normal(0, noise_scale, n - 1)
                zt = np.clip(zt, -50.0, 50.0)
                pt = _probs_from_z(zt, n)
                st = _score(pt)
                if st < score:
                    z = zt
                    p = pt
                    score = st
                    if score < global_best_score:
                        global_best_score = score
                        global_best_seq = [float(v) for v in p]
                    improved = True
                    break

            if not improved:
                fails += 1
                if fails > 200:
                    break
            else:
                fails = 0
