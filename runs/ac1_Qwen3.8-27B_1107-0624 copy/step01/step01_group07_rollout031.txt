"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based subgradient descent, LP refinement, and coordinate descent
with multiple initializations including asymmetric profiles.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)

def project_simplex(v, s=1.0):
    """Project v onto {x: sum(x)=s, x>=0}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_find_direction(a):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(a,g) <= max(conv(a,a))."""
    n = len(a)
    c = fft_convolve(a)
    M = c.max()
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            g = res.x
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None

def lp_refine(a, deadline, rng):
    """Iterative LP refinement."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    step = 0.1
    stall = 0
    
    while time.monotonic() < deadline:
        c = fft_convolve(a)
        M = c.max()
        g = lp_find_direction(a)
        if g is None:
            break
        
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            s = step * shrink
            a_new = project_simplex((1.0 - s) * best_a + s * g, 1.0)
            sc = 2.0 * n * fft_convolve(a_new).max()
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:

            step = min(0.5, step * 1.05)
        else:
            stall += 1
            step = max(0.002, step * 0.7)

    return best_a, best_score

def subgradient_optimize(a, deadline, rng):
    """Subgradient descent with momentum."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a).max()
    step = 0.01
    momentum = 0.5
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a)
        M = c.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find argmax
        max_indices = np.where(np.isclose(c, M, atol=1e-10))[0]
        k_star = int(max_indices[rng.integers(len(max_indices))])
        
        # Compute subgradient: grad[m] = 2*a[k_star - m] if valid
        grad = np.zeros(n)
        valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
        grad[valid] = 2.0 * a[k_star - np.arange(n)[valid]]
        
        # Update with momentum
        v = momentum * v - step * grad
        a_new = project_simplex(a + v, 1.0)
        
        # Check
        c_new = fft_convolve(a_new)
        if 2.0 * n * c_new.max() <= score + 1e-10:
            a = a_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            step = max(1e-5, step * 0.5)
        
        if it % 500 == 0:
            print(f"    subgrad: iter={it}, best={best_score:.6f}, step={step:.6f}, stall={stall}")
        
        if stall > 500:
            break
    
    return best_a, best_score

def coord_descent(a, deadline, rng):
    """Coordinate descent: optimize each element one at a time."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_score = compute_objective(a)
    
    while time.monotonic() < deadline:
        improved = False
        for i in range(n):
            # Try adjusting a[i]
            a_old = a[i]
            for delta in [0.5 * a_old, 0.1 * a_old, 0.01 * a_old, -0.5 * a_old, -0.1 * a_old]:
                a[i] = max(0.0, a_old + delta)
                # Renormalize
                if a.sum() > 1e-10:
                    a_norm = a / a.sum()
                else:
                    a[i] = a_old
                    continue
                sc = compute_objective(a_norm)
                if sc < best_score - 1e-12:
                    a = a_norm.copy()
                    best_score = sc
                    improved = True
                    break
            a[i] = a_old
            if a.sum() > 1e-10:
                a = a / a.sum()
        
        if not improved:

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based subgradient descent, LP refinement (min-peak cross-convolution),
and multiple diverse initializations including asymmetric profiles.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (S * S)


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fft_convolve(a)
    return 2.0 * n * c.max()


def project_simplex(v, s=1.0):
    """Project v onto the probability simplex scaled to sum s."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho_candidates) == 0:
        return np.full(n, s / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Fix numerical issues
    diff = s - result.sum()
    if abs(diff) > 1e-12:
        idx_max = np.argmax(result)
        result[idx_max] += diff
        result[idx_max] = max(result[idx_max], 0.0)
    return result


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_find_direction(a, time_limit=20):
    """
    LP: find g >= 0, sum(g)=1, minimizing max_k conv(a,g)[k].
    Returns normalized g or None.
    """
    n = len(a)
    if n > 2048:
        return None
    A = build_conv_matrix(a)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            g = res.x[:n]
            g = np.maximum(g, 0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None

def subgradient_optimize(a, deadline, rng):
    """Subgradient descent with momentum for the max-conv objective."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_a = a.copy()
    best_score = compute_obj_normalized(a)
    step = 0.02
    momentum = 0.4
    v = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a)
        M = c.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Find a maximizing index
        max_indices = np.where(c >= M - 1e-12 * max(1.0, M))[0]
        k_star = int(max_indices[rng.integers(len(max_indices))])

        # Subgradient: grad[m] = 2*a[k_star - m] if 0 <= k_star-m < n
        indices = np.arange(n)
        j_vals = k_star - indices
        valid = (j_vals >= 0) & (j_vals < n)
        grad = np.zeros(n)
        grad[valid] = 2.0 * a[j_vals[valid]]

        # Update with momentum
        v = momentum * v - step * grad
        a_new = project_simplex(a + v, 1.0)

        # Accept or reject
        c_new = fft_convolve(a_new)
        if 2.0 * n * c_new.max() <= score + 1e-9:
            a = a_new
        else:
            a = best_a.copy()
            v *= 0.5
            step = max(1e-5, step * 0.7)

        # Adaptive step increase if consistent improvement
        if stall == 0 and step < 0.05:
            step = min(0.05, step * 1.01)

        if it % 2000 == 0:
            print(f"    subgrad: iter={it}, best={best_score:.6f}, step={step:.6f}, stall={stall}")

        if stall > 1000:
            break

    return best_a, best_score

        threshold = 0.7 * M
        peak_mask = c > threshold
        peak_indices = np.where(peak_mask)[0]

        # Compute contribution of each a[i] to peak convolutions
        contrib = np.zeros(n)
        for i in range(n):
            js = peak_indices - i
            valid = (js >= 0) & (js < n)
            if valid.any():
                contrib[i] = a[js[valid]].sum()

        # Multiplicative update
        avg_c = contrib.mean() + 1e-15
        update = np.exp(-lr * contrib / avg_c)
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()

        sc = compute_obj_normalized(a_new)
        if sc < best_score - 1e-12:
            a = a_new
            stall = 0
        else:
            stall += 1
            lr = max(0.01, lr * 0.9)

        if it % 1000 == 0:
            print(f"    mult: iter={it}, best={best_score:.6f}, lr={lr:.4f}")

        if stall > 300:
            # Random restart from best
            a = best_a.copy()
            noise = np.abs(rng.normal(0, 0.03, n))
            a = project_simplex(a * (1 + noise), 1.0)
            stall = 0
            lr = 0.3

    return best_a, best_score


def coord_descent(a, deadline, rng):
    """Coordinate descent: try small changes to individual elements."""
    n = len(a)
    a = project_simplex(a, 1.0)
    best_score = compute_obj_normalized(a)

    while time.monotonic() < deadline:
        improved = False
        # Randomize order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            a_val = a[i]
            if a_val < 1e-15:
                # Try increasing from zero
                for frac in [0.01, 0.005, 0.001]:
                    a[i] = frac
                    a_norm = project_simplex(a, 1.0)
                    sc = compute_obj_normalized(a_norm)
                    if sc < best_score - 1e-12:
                        a = a_norm
                        best_score = sc
                        improved = True
                        break
                a[i] = 0.0
            else:
                # Try scaling
                for factor in [1.5, 1.2, 0.8, 0.5, 0.0]:
                    a[i] = a_val * factor
                    a_norm = project_simplex(a, 1.0)
                    sc = compute_obj_normalized(a_norm)
                    if sc < best_score - 1e-12:
                        a = a_norm
                        best_score = sc
                        improved = True
                        break
                a[i] = a_val

        if not improved:
            break

    return a, best_score


def make_init(n, kind, rng):
    """Create various initialization profiles."""
    if kind == 'cosine':
        return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        center = (n - 1) / 2.0
        sigma = 0.25 * n
        return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
    elif kind == 'exp_left':
        return np.exp(-2.5 * np.linspace(0, 1, n))
    elif kind == 'exp_right':
        return np.exp(2.5 * np.linspace(0, 1, n))
    elif kind == 'beta_l':
        i = np.linspace(0, 1, n)
        return (i + 0.05) * (1.0 - i) ** 2
    elif kind == 'beta_r':
        i = np.linspace(0, 1, n)
        return (1.0 - i + 0.05) * i ** 2
    elif kind == 'beta_sym':
        i = np.linspace(0, 1, n)
        return (i + 0.05) * (1.0 - i + 0.05)
    elif kind == 'random':
        return np.abs(rng.normal(0, 1, n))
    elif kind == 'parabolic':
        c = (n - 1) / 2.0
        return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))
    elif kind == 'sigmoid_left':
        i = np.linspace(0, 1, n)
        return 1.0 / (1.0 + np.exp(10 * (i - 0.3)))
    elif kind == 'sigmoid_right':
        i = np.linspace(0, 1, n)
        return 1.0 / (1.0 + np.exp(10 * (i - 0.7)))
    else:
        return np.ones(n)


def optimize_single(a, deadline, rng, use_lp=True):
    """Full optimization pipeline for a single sequence."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_obj_normalized(a)

    # Phase 1: Multiplicative update (fast, good for initial shaping)
    p1_deadline = min(time.monotonic() + 45, deadline - 90)
    if time.monotonic() < p1_deadline:
        a1, s1 = multiplicative_update(a, p1_deadline, rng)
        if s1 < best_score:
            best_a, best_score = a1, s1
        a = a1 if s1 < compute_obj_normalized(a) else a

    # Phase 2: Subgradient
    p2_deadline = min(time.monotonic() + 60, deadline - 60)
    if time.monotonic() < p2_deadline:
        a2, s2 = subgradient_optimize(a, p2_deadline, rng)
        if s2 < best_score:
            best_a, best_score = a2, s2
        a = a2 if s2 < compute_obj_normalized(a) else a

    # Phase 3: LP refinement (most effective for fine-tuning)
    if use_lp and n <= 2048:
        p3_deadline = min(time.monotonic() + 180, deadline - 30)
        if time.monotonic() < p3_deadline:
            a3, s3 = lp_refine(best_a, p3_deadline, rng)
            if s3 < best_score:
                best_a, best_score = a3, s3

    # Phase 4: Another subgradient pass from LP result
    p4_deadline = min(time.monotonic() + 45, deadline - 20)
    if time.monotonic() < p4_deadline:
        a4, s4 = subgradient_optimize(best_a, p4_deadline, rng)
        if s4 < best_score:
            best_a, best_score = a4, s4

    # Phase 5: Final LP if time remains
    if use_lp and n <= 2048 and time.monotonic() < deadline:
        a5, s5 = lp_refine(best_a, deadline, rng)
        if s5 < best_score:
            best_a, best_score = a5, s5

    # Phase 6: Coordinate descent
    p6_deadline = min(time.monotonic() + 30, deadline)
    if time.monotonic() < p6_deadline:
        a6, s6 = coord_descent(best_a, p6_deadline, rng)
        if s6 < best_score:
            best_a, best_score = a6, s6

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of FFT-based
    subgradient descent, LP refinement, multiplicative updates, and
    coordinate descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    best_seq = None
    best_score = np.inf

    # Collect initializations with (name, array, priority)
    inits = []

    # Parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all() and p.sum() > 0:
            inits.append(('parent', p / p.sum(), 3))
    except Exception:
        pass

    # Diverse initializations
    for n in [384, 512, 768]:
        for kind in ['cosine', 'exp_left', 'exp_right', 'beta_l', 'beta_r',
                     'sigmoid_left', 'sigmoid_right', 'beta_sym']:
            try:
                a = make_init(n, kind, rng)
                a = np.clip(a, 1e-12, None)
                a = a / a.sum()
                inits.append((f'{kind}_n{n}', a, 1))
            except Exception:
                pass

    # A couple of random ones
    for n in [512, 768]:
        a = np.abs(rng.normal(0, 1, n))
        a = a / a.sum()
        inits.append((f'random_n{n}', a, 1))

    # Sort by priority (higher first), then by name for determinism
    inits.sort(key=lambda x: (-x[2], x[0]))

    # Time allocation: parent gets most time
    total_inits = len(inits)
    # Reserve 120s for final polish
    available = deadline - start_time - 120.0
    # Weighted allocation
    weights = [float(x[2]) for x in inits]
    total_w = sum(weights)
    # Give at most 600s to any single init
    max_per_init = min(600.0, available * 0.4)

    print(f"Starting: {total_inits} inits, budget={budget_s}s, available~{available:.0f}s")

    for idx, (name, a, priority) in enumerate(inits):
        if time.monotonic() > deadline - 100:
            break

        # Compute deadline for this init
        remaining_inits = total_inits - idx
        time_share = available * weights[idx] / total_w
        # But don't exceed what's left or the max
        init_deadline = min(
            time.monotonic() + min(time_share, max_per_init),
            deadline - 80 * remaining_inits
        )

        if init_deadline <= time.monotonic() + 10:
            continue

        elapsed = time.monotonic() - start_time
        n_init = len(a)
        print(f"[{idx+1}/{total_inits}] {name}, n={n_init}, elapsed={elapsed:.0f}s, deadline_in {init_deadline - time.monotonic():.0f}s")

        try:
            a_opt, score = optimize_single(a, init_deadline, rng, use_lp=(n_init <= 1024))
        except Exception as e:
            print(f"  Error: {e}")
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n_init})")
        else:
            print(f"  => score={score:.6f} (current best: {best_score:.6f})")

    # Final polish on best sequence
    if best_seq is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish, {remaining:.0f}s remaining, current best={best_score:.6f}")
        if len(best_seq) <= 1024:
            best_seq, best_score = lp_refine(best_seq, deadline, rng)
            print(f"After LP polish: {best_score:.6f}")
        # Coordinate descent
        if time.monotonic() < deadline:
            best_seq, best_score = coord_descent(best_seq, deadline, rng)
            print(f"After coord descent: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512.0
        best_score = 2.0

    # Ensure valid for evaluator
    final_list = [float(x) for x in best_seq]
    s = sum(final_list)
    if s < 0.01:
        scale = 0.01 / s
        final_list = [x * scale for x in final_list]

    # Clip to [0, 1000]
    final_list = [max(0.0, min(1000.0, x)) for x in final_list]

    # Verify
    try:
        eval_score = evaluate_sequence(final_list)
        print(f"\nFINAL: evaluator score={eval_score:.6f}, n={len(final_list)}")
    except Exception as e:
        print(f"Eval error: {e}")

    elapsed_total = time.monotonic() - start_time
    print(f"Total time: {elapsed_total:.1f}s")

    return final_list
