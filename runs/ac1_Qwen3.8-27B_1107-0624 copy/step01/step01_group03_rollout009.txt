```python
def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for tau
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        tau = (lo + hi) / 2
        x = np.maximum(v - tau, 0)
        s = x.sum()
        if s > 1:
            lo = tau
        else:
            hi = tau
    x = np.maximum(v - (lo + hi) / 2, 0)
    # Adjust for numerical precision
    diff = 1 - x.sum()
    if diff != 0:
        # Distribute the difference to the largest elements
        idx = np.argmax(x)
        x[idx] += diff
    return x
```

```python
def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
import time
from scipy.optimize import linprog

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def subgradient_step(s, k_star, alpha):
    """Take one subgradient step to reduce conv[s,s][k_star]."""

    sg[:k_star+1] = 2.0 * s[k_star::-1]
    s_new = s - alpha * sg
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    # Fix numerical issues
    diff = 1.0 - result.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(result)
        result[idx] += diff
    return result

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def compute_subgradient(s, k_star):
    """Compute subgradient of conv(s,s)[k_star] w.r.t. s."""
    n = len(s)
    sg = np.zeros(n)
    m_max = min(k_star, n - 1)
    if m_max >= 0:
        sg[:m_max+1] = 2.0 * s[k_star::-1][:m_max+1]
    return sg

def subgradient_optimize(s, max_iters=100000, deadline=None, verbose=False):
    """Optimize using subgradient descent with momentum."""
    n = len(s)
    s = s.copy()
    s = s / s.sum()
    best_s = s.copy()
    best_score = fast_score(s)
    
    momentum = np.zeros(n)
    beta = 0.85  # momentum parameter
    base_alpha = 0.05
    no_improve = 0

        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = conv.max()
        k_star = conv.argmax()
        
        sg = compute_subgradient(s, k_star)

        alpha = base_alpha / (1 + 0.01 * it)
        
        momentum = beta * momentum - alpha * sg
        s_new = s + momentum
        s_new = simplex_project(s_new)
        
        score = fast_score(s_new)
        if score < best_score:

            no_improve = 0
        else:
            no_improve += 1
        
        s = s_new
        
        if no_improve > 5000:

            no_improve = 0
        
        if verbose and it % 10000 == 0:
```

```python
import time
from scipy.optimize import linprog

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    return result

    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()

    return 2.0 * n * conv.max() / (total * total)

    s = np.maximum(s, 0)
```
