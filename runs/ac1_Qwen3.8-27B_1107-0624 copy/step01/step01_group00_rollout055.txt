import time
import math
import numpy as np
from scipy.optimize import linprog

def _conv_sq(a, m=None):
    """Linear convolution a*a via FFT."""
    n = len(a)
    if m is None:
        m = 1
        while m < 2*n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _obj(a):
    """Compute the objective (assuming sum(a) = 1)."""
    n = len(a)
    c = _conv_sq(a)
    return 2.0 * n * c.max()

def _lp_best_response(f, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    n = len(f)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi+1):
            A[k, j] = f[k-j]
    # Convert to linprog form: A_ub @ b <= M (vector of M's)
    b_vec = np.full(2*n-1, M)
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    # Fallback: return uniform
    return np.full(n, 1.0/n)

    """Multiplicative weights / mirror descent with entropic regularization."""
    n = len(a)
    a = np.abs(a).astype(np.float64)
    a = a / a.sum()
    m = 1
    while m < 2*n - 1:
        m <<= 1
    lr = lr0
    best_a = a.copy()
    best_obj = _obj(a)
    
    while time.monotonic() < t_end:
        c = _conv_sq(a, m)
        mc = c.max()
        cur_obj = 2.0 * n * mc
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Gradient of max(c) w.r.t. a: 2*a[max_lag - i] for valid i
        k_max = c.argmax()
        lo = max(0, k_max - n + 1)
        hi = min(k_max, n - 1)
        grad = np.zeros(n)
        grad[lo:hi+1] = 2.0 * a[k_max - np.arange(lo, hi+1)]
        
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = a / a.sum()
        
        lr *= decay
    
    return best_a, best_obj

def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    rho = cond.nonzero()[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _euclidean_descent(a, t_end, lr0=0.05):

    a = np.abs(a).astype(np.float64)
    a = _proj_simplex(a, 1.0)

    """Precompute row/col indices for the LP constraint matrix."""
    rows = np.repeat(np.arange(n), n)  # column j repeated n times... wait

    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    return rows, cols

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based best response iterations (Nash equilibrium approach)
2. Multiplicative weights (mirror descent) for fast polishing
3. Nesterov-accelerated projected gradient for fine-tuning
4. Symmetrization to exploit problem symmetry
5. Multiple restarts with different n values and initial conditions
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp


def _fft_conv(a):
    """Linear convolution a*a via FFT, returns full result of length 2n-1."""
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _obj_val(a):
    """Objective assuming sum(a)=1: returns 2*n*max(conv)."""
    n = len(a)
    return 2.0 * n * _fft_conv(a).max()


def _build_lp_indices(n):
    """Precompute row/col indices for LP constraint matrix (f*b) <= M."""
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    return rows, cols


def _lp_solve(f, n, rows, cols, M):
    """Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 8, 'presolve': True})
        if res.success and res.x is not None:
            b = np.abs(res.x)
            s = b.sum()
            if s > 1e-12:
                return b / s
    except Exception:
        pass
    return np.full(n, 1.0 / n)


def _symmetrize(a):
    """Enforce a[i] = a[n-1-i]."""
    return 0.5 * (a + a[::-1])


def _lp_iteration(f, t_budget, rows, cols):
    """LP best-response fixed-point iteration."""
    n = len(f)
    f = np.abs(f).astype(np.float64)
    f = _symmetrize(f)
    f = f / f.sum()
    t_start = time.monotonic()
    blend = 0.4
    best_f = f.copy()
    best_obj = _obj_val(f)
    it = 0

    while time.monotonic() < t_start + t_budget:
        c = _fft_conv(f)
        M = c.max()
        b = _lp_solve(f, n, rows, cols, M * (1.0 + 1e-10))
        b = _symmetrize(b)
        b = np.clip(b, 0, None)
        bs = b.sum()
        if bs < 1e-12:
            break
        b = b / bs

        f_new = (1.0 - blend) * f + blend * b
        f_new = np.clip(f_new, 0, None)
        f_new = _symmetrize(f_new)
        f_new = f_new / f_new.sum()

        cur_obj = _obj_val(f_new)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_f = f_new.copy()

        f = f_new
        it += 1

        if it % 15 == 0:
            blend = max(0.08, blend * 0.88)

        if it % 20 == 0:
            print(f"    LP iter {it}: obj={best_obj:.6f}, blend={blend:.3f}, "
                  f"elapsed={time.monotonic()-t_start:.1f}s")

    return best_f


def _mirror_descent(a, t_budget):
    """Multiplicative weights / mirror descent with entropic regularization."""
    n = len(a)
    a = np.abs(a).astype(np.float64)
    a = _symmetrize(a)
    a = a / a.sum()
    t_start = time.monotonic()
    lr = 0.08
    best_a = a.copy()
    best_obj = _obj_val(a)
    it = 0

    while time.monotonic() < t_start + t_budget:
        c = _fft_conv(a)
        mc = c.max()
        cur_obj = 2.0 * n * mc
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        k = int(c.argmax())
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        a = a * np.exp(-lr * grad)
        a = _symmetrize(a)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a = a / s

        it += 1
        if it % 200 == 0:
            lr = max(0.005, lr * 0.93)

    return best_a


def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.full(n, target / n)
    idx = cond.nonzero()[0][-1]
    theta = (cssv[idx] - target) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _projected_gradient(a, t_budget):
    """Nesterov accelerated projected gradient descent."""
    n = len(a)
    a = np.abs(a).astype(np.float64)
    a = _symmetrize(a)
    a = a / a.sum()
    t_start = time.monotonic()
    lr = 0.04
    v = np.zeros(n)
    beta = 0.89
    best_a = a.copy()
    best_obj = _obj_val(a)
    it = 0

    while time.monotonic() < t_start + t_budget:
        c = _fft_conv(a)
        mc = c.max()
        cur_obj = 2.0 * n * mc
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        k = int(c.argmax())
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        v = beta * v - lr * grad
        a_new = a + lr * v
        a_new = _symmetrize(a_new)
        a = _proj_simplex(a_new, 1.0)

        it += 1
        if it % 400 == 0:
            lr = max(0.003, lr * 0.94)
            beta = min(0.96, beta + 0.002)

    return best_a


def _soft_max_descent(a, t_budget):
    """Subgradient with softmax smoothing over top-L lags."""
    n = len(a)
    a = np.abs(a).astype(np.float64)
    a = _symmetrize(a)
    a = a / a.sum()
    t_start = time.monotonic()
    lr = 0.05
    L = min(40, n)
    tau = 0.15
    beta = 0.9
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = _obj_val(a)
    it = 0

    while time.monotonic() < t_start + t_budget:
        c = _fft_conv(a)
        mc = c.max()
        cur_obj = 2.0 * n * mc
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Softmax over top-L lags
        order = np.argpartition(-c, L - 1)[:L]
        wc = c[order]
        w = np.exp(wc / max(tau, 1e-10))
        w = w / w.sum()

        grad = np.zeros(n)
        for k_i, w_i in zip(order, w):
            if w_i < 1e-10:
                continue
            lo = max(0, k_i - n + 1)
            hi = min(k_i, n - 1)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w_i * a[k_i - idx]

        v = beta * v - lr * grad
        a_new = a + lr * v
        a_new = _symmetrize(a_new)
        a = _proj_simplex(a_new, 1.0)

        it += 1
        if it % 300 == 0:
            lr = max(0.003, lr * 0.94)
            tau = max(0.02, tau * 0.85)
            L = max(5, int(L * 0.75))
            beta = min(0.96, beta + 0.002)

    return best_a


def _make_seed(n, kind, base=None):
    """Create initial sequence of length n."""
    if kind == 'interpolated' and base is not None:
        a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(base)), base)
    elif kind == 'uniform':
        a = np.ones(n)
    elif kind == 'bump_center':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    elif kind == 'bump_offset':
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - 0.6 * n) / (0.18 * n)) ** 2)
    elif kind == 'ramp':
        a = np.linspace(0.1, 1.0, n)
    elif kind == 'cosine':
        a = 0.5 + 0.5 * np.cos(np.pi * np.arange(n) / n)
        a = np.clip(a, 0.01, None)
    elif kind == 'sqrt':
        a = np.sqrt(np.linspace(0.01, 1.0, n))
    elif kind == 'random_smooth':
        rng = np.random.default_rng(123)
        a = np.abs(np.cumsum(rng.normal(0, 1, n)))
        a = a + 0.1
    else:
        a = np.ones(n)

    a = np.clip(a, 0, None)
    a = _symmetrize(a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
