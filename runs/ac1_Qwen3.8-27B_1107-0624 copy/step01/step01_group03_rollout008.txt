    val, g_s = smooth_max_and_grad(s, tau)
    # Chain rule: grad_x = s * (g_s - <s, g_s>)
    dot = np.dot(s, g_s)
    g_x = s * (g_s - dot)
    return val, g_x

            smooth_obj_and_grad_x, x, args=(tau,),

            best_s = s.copy()

        print(f"    tau={tau:.1f}: score={cur_score:.6f}, time={stage_time:.1f}s, "
