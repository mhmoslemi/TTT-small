```python
import numpy as np
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. (2011)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u * np.arange(1, n+1) > cssv)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_subgradient(s, conv, frac=1.0):
    """Compute subgradient of max(conv) w.r.t. s.
    
    Uses the top fraction of peak entries for a smoother gradient.
    """
    n = len(s)
    M = conv.max()
    # Find entries that are close to the max
    threshold = M * (1 - 0.01 * (1 - frac))  # or use a fixed fraction
    peaks = np.where(conv >= M * 0.95)[0]
    
    if len(peaks) == 0:
        peaks = np.array([np.argmax(conv)])
    
    g = np.zeros(n)
    for k in peaks:
        # grad_k[i] = 2*s[k-i] if 0 <= k-i < n
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        if np.any(valid):
            g[valid] += 2 * s[idx[valid]]
    g /= len(peaks)
    return g

def smooth_gradient(s, conv, t, weight='hinge'):
    """Compute gradient of smoothed objective.
    
    weight='hinge': F = sum_k max(0, conv[k] - t)^2
    weight='softplus': F = sum_k softplus(conv[k] - t)
    """
    n = len(s)
    if weight == 'hinge':
        w = np.maximum(conv - t, 0) ** 2 * 2  # d/d conv of (conv-t)^2 * I[conv>t]
        # Actually: d/d conv_k [max(0, conv_k - t)^2] = 2*max(0, conv_k - t)
        w = 2 * np.maximum(conv - t, 0)
    elif weight == 'softplus':
        # softplus(x) = log(1+exp(x)), use stable version
        x = conv - t
        w = 1.0 / (1.0 + np.exp(-np.clip(x, -50, 50)))
    
    g = np.zeros(n)
    for k in range(len(conv)):
        if w[k] > 1e-15:
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            if np.any(valid):
                g[valid] += w[k] * 2 * s[idx[valid]]
    return g
```

```python
    n = len(v)
```

    eta = eta0
    no_imp = 0
    stall_count = 0
    
    while time.monotonic() < end_time:
        conv = np.convolve(s, s)
        M = conv.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_s = s.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Escape mechanism
        if no_imp > 1000:
            # Try a bolder restart
            s = best_s + 0.01 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            eta = 0.1
            no_imp = 0
            stall_count += 1
            if stall_count > 10:
                break
            continue
        
        # Compute smoothed gradient
        g = compute_gradient(s, conv, peak_frac=peak_frac)
        g_max = np.max(np.abs(g))
        if g_max < 1e-15:
            break
        
        # Multiplicative (mirror) descent update
        step = eta * g / g_max
        s_new = s * np.exp(-step)
        s_new = np.maximum(s_new, 1e-30)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            break
        s = s_new / s_sum
        
        # Decay step size
        eta *= 0.9995
        if eta < 1e-4:
            eta = 0.01
    
    return best_s, best_score
