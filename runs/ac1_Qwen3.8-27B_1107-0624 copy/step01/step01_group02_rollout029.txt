import numpy as np
import time

    from scipy.optimize import linprog
except Exception:

def normalize_nonneg(a, target=1.0):
    """Normalize a 1-D array to sum=target, keeping it non-negative."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64) * target
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a * (target / s)


def compute_score_normalized(a):
    """Score for a normalized (sum=1) nonnegative sequence: 2n * max(f*f)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    a = a / s
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m)


def soft_max_gradient(f, tau, n_active=None):
    """
    Compute the gradient of the soft-max of (f*f) with temperature tau.
    Returns (grad, soft_max_value, max_convol).
    Uses FFT for efficiency.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    c = np.convolve(f, f)
    mc = float(c.max())
    if mc <= 0:
        return np.zeros(n), 0.0, mc
    scaled = (c - mc) / tau
    scaled = np.clip(scaled, -100.0, 0.0)
    p = np.exp(scaled)
    ps = float(p.sum())
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / ps
    soft_val = mc + tau * float(np.log(ps))
    # Gradient: grad[j] = 2 * sum_m p[m+j] * f[m]
    # This is cross-correlation of p with f at offset j
    size = 1
    need = n + len(p) - 1
    while size < need:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    # p has length 2n-1; we need p[m+j] for m in [0,n-1], j in [0,n-1]
    # So p indices range from 0 to (n-1)+(n-1) = 2n-2, which is fine
    pa = np.zeros(size, dtype=np.float64)
    pa[:len(p)] = p
    # cross_corr[j] = sum_m f[m] * p[m+j] = (f * flip(p))[j + len(p) - 1 - 0]... 
    # Actually: sum_m f[m] p[m+j]. Let's compute via correlation.
    # corr[k] = sum_m f[m] p[m+k] for k = 0, ..., 2n-2
    # This is the cross-correlation. Use FFT:
    # corr = ifft(fft(f) * conj(fft(p)))... no.
    # sum_m f[m] p[m+k]: shift p by -k and dot with f.
    # = sum_m f[m] p[m+k] = (f * p_flipped)(len(p)-1 - k + something)...
    # Let me use: sum_m f[m] p[m+k] = sum_m f[m] p_flipped[len(p)-1-m-k] ... 
    # Easier: use np.correlate(f, p, mode='full') which gives:
    # np.correlate(f, p, 'full')[k] = sum_m f[m] p[m-k+len(f)-1]... 
    # Actually np.correlate(a, v, 'full')[k] = sum_n a[n+k] v[n] for appropriate k.
    # I want sum_m f[m] p[m+j]. Let me just use:
    # result[j] = sum_m f[m] p[m+j]
    # = convolution of f with flipped p, evaluated at appropriate index.
    # conv(f, p_rev)[j + n - 1] where p_rev = p[::-1]
    p_rev = p[::-1]
    conv = np.fft.irfft(np.fft.rfft(f, size) * np.fft.rfft(p_rev, size), size)
    # conv[j + n - 1] = sum_m f[m] p_rev[n-1-j+m]... 
    # Let me just verify: conv[k] = sum_i f[i] p_rev[k-i] = sum_i f[i] p[n-1-(k-i)] = sum_i f[i] p[n-1-k+i]
    # I want: sum_m f[m] p[m+j]. Set n-1-k+i = m+j, so i = m, k = n-1-j.
    # So conv[n-1-j] = sum_m f[m] p[m+j]. 
    # Therefore: grad[j] = 2 * conv[n-1-j]
    grad = 2.0 * conv[n - 1 - np.arange(n)]
    # Clip to avoid numerical issues
    grad = np.clip(grad, -1e10, 1e10)
    return grad, soft_val, mc


def project_simplex(v, target=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0][-1:]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_optimize(f, time_limit, tau_init=0.01, tau_min=0.001, step_init=0.01, rng=None):
    """
    Optimize f using projected gradient on the soft-max objective.
    f is normalized (sum=1, non-negative).
    Returns the optimized f and its true score.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = normalize_nonneg(f)
    n = f.size
    deadline = time.monotonic() + max(0.0, float(time_limit))
    tau = float(tau_init)
    step = float(step_init)
    score = compute_score_normalized(f)
    best = f.copy()
    best_score = score
    momentum = np.zeros(n, dtype=np.float64)
    beta_mom = 0.85
    no_improve = 0
    stall_count = 0

    while time.monotonic() < deadline:
        grad, soft_val, mc = soft_max_gradient(f, tau)
        # Subtract mean to maintain normalization (first-order)
        grad_normalized = grad - np.mean(grad)
        # Adaptive step size
        step_size = step / (1.0 + np.max(np.abs(grad_normalized)) * 100.0)
        step_size = min(step_size, 0.1)
        # Gradient step with momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_normalized
        h = f - step_size * momentum
        h = project_simplex(h, 1.0)
        h = normalize_nonneg(h)
        new_score = compute_score_normalized(h)
        if new_score < best_score - 1e-10:
            best_score = new_score
            best = h.copy()
            f = h
            no_improve = 0
            stall_count = 0
            step = min(step * 1.05, 0.05)
        else:
            no_improve += 1
            stall_count += 1
            # Try smaller step
            h2 = f - 0.3 * step_size * momentum
            h2 = normalize_nonneg(project_simplex(h2, 1.0))
            sc2 = compute_score_normalized(h2)
            if sc2 < score - 1e-10:
                f = h2
                score = sc2
            else:
                step *= 0.7
            if no_improve > 20:
                tau = max(tau_min, tau * 0.9)
                if no_improve > 40:
                    # Random restart perturbation
                    f = normalize_nonneg(f * np.exp(rng.normal(0, 0.05, size=n)))
                    no_improve = 0
                    step = float(step_init)
                if no_improve > 80:
                    stall_count += 1
                    if stall_count > 3:
                        break
                    f = normalize_nonneg(f * np.exp(rng.normal(0, 0.15, size=n)))
                    no_improve = 0
                    step = float(step_init)
                    tau = float(tau_init)
        score = compute_score_normalized(f)
        if score < best_score:
            best_score = score
            best = f.copy()
    return best, best_score


def lp_direction(f, M, max_rows):
    """Solve LP to find improvement direction g: max sum(g) s.t. cross-corr(f,g) <= M, g>=0."""
    n = f.size
    if linprog is None or n > 4096:
        return None
    c = np.convolve(f, f)
    idx = np.argsort(c)[::-1][:max_rows]
    idx = np.sort(idx)
    A = np.zeros((len(idx), n), dtype=np.float64)
    for r, k in enumerate(idx):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[r, jlo:jhi] = f[k - (jlo + np.arange(jhi - jlo))]
    if not np.any(A):
        return None
    A = A / M
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.ones(len(idx)), bounds=(0, None), method='highs')
    if res is not None and res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        if g.sum() > 1.0 + 1e-5:
            return g
    return None


def lp_optimize(f, time_limit, rng=None):
    """Optimize using LP direction finding with line search."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = normalize_nonneg(f)
    n = f.size
    deadline = time.monotonic() + max(0.0, float(time_limit))
    score = compute_score_normalized(f)
    best = f.copy()
    best_score = score
    no_improve = 0

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = float(c.max())
        cur = float(2.0 * n * M)
        if cur < best_score - 1e-10:
            best_score = cur
            best = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
        if n <= 512:
            mr = 2 * n - 1
        elif n <= 1024:
            mr = min(2 * n - 1, 800)
        else:
            mr = min(2 * n - 1, 400)
        g = lp_direction(f, M, mr)
        if g is not None:
            Sg = float(g.sum())
            g = g / Sg
            best_cur = cur
            best_h = f
            if Sg > 10:
                ts = (0.01, 0.03, 0.05, 0.1, 0.2, 0.4)
            else:
                ts = (0.05, 0.1, 0.2, 0.4, 0.7, 1.0)
            for t in ts:
                h = (1.0 - t) * f + t * g
                h = normalize_nonneg(h)
                sc = compute_score_normalized(h)
                if sc < best_cur:
                    best_cur = sc
                    best_h = h
            if best_cur < cur - 1e-10:
                f = best_h
            else:
                no_improve += 2
        else:
            no_improve += 1
            f = normalize_nonneg(f * np.exp(rng.normal(0, 0.05, size=n)))
    return best, best_score


def hybrid_optimize(f, time_limit, rng):
    """Combine LP and smooth-max methods."""
    f = normalize_nonneg(f)
    n = f.size
    deadline = time.monotonic() + max(0.0, float(time_limit))
    score = compute_score_normalized(f)
    best = f.copy()
    best_score = score

    # Phase A: LP if small enough
    if n <= 1024 and time.monotonic() < deadline - 30:
        ta = min(60.0, (deadline - time.monotonic()) * 0.4)
        f, sc_a = lp_optimize(f, ta, rng)
        if sc_a < best_score:
            best_score = sc_a
            best = f.copy()

    # Phase B: Smooth-max refinement
    if time.monotonic() < deadline - 10:
        tb = deadline - time.monotonic() - 5.0
        tau_init = max(0.005, 0.5 * float(np.std(np.convolve(f, f))))
        f, sc_b = smooth_max_optimize(f, tb, tau_init=tau_init, step_init=0.01, rng=rng)
        if sc_b < best_score:
            best_score = sc_b
            best = f.copy()

    return best, best_score


def make_initial(N, kind, rng, best_seq=None, provided_seq=None):
    """Create initial normalized nonnegative sequence of length N."""
    N = int(max(4, N))
    x = np.linspace(0.0, 1.0, N)

    if kind == 99 and best_seq is not None:
        return resize_to_length(best_seq, N)
    if kind == 98 and provided_seq is not None:
        return resize_to_length(provided_seq, N)

    if kind == 0:
        # Two-block: L = ratio*N, mass ratio optimized
        ratio = float(rng.uniform(0.2, 0.45))
        L = max(1, int(N * ratio))
        a = np.zeros(N, dtype=np.float64)
        m1 = float(rng.uniform(0.5, 0.85))
        m2 = 1.0 - m1
        a[:L] = m1 / L
        a[L:] = m2 / (N - L)
    elif kind == 1:
        # Linear ramp
        direction = float(rng.choice([1.0, -1.0]))
        a = np.linspace(0.1, 1.0, N)
        if direction < 0:
            a = a[::-1]
        # Add some curvature
        alpha = float(rng.uniform(0.5, 3.0))
        a = x ** alpha
        if direction < 0:
            a = a[::-1]
    elif kind == 2:
        # Exponential decay
        rate = float(rng.uniform(1.0, 5.0))
        a = np.exp(-rate * x)
        if rng.random() < 0.5:
            a = a[::-1]
    elif kind == 3:
        # Beta-like profile
        alpha = float(rng.uniform(0.3, 4.0))
        beta = float(rng.uniform(0.3, 4.0))
        xs = np.linspace(1e-6, 1 - 1e-6, N)
        a = xs ** alpha * (1 - xs) ** beta
    elif kind == 4:
        # Smoothed binary (Thue-Morse inspired)
        tm = np.zeros(N)
        tm[0] = 1.0
        for i in range(1, N):
            tm[i] = 1.0 - tm[i // 2]
        # Smooth it
        wlen = max(3, N // 32)
        kernel = np.exp(-0.5 * (np.arange(wlen) - wlen // 2) ** 2 / (wlen / 6.0) ** 2)
        kernel /= kernel.sum()
        a = np.convolve(tm, kernel, mode='same')
        a = np.maximum(a, 0.01 * a.max())
    elif kind == 5:
        # Random smooth
        base = rng.normal(0, 1, size=N)
        wlen = max(3, N // 50)
        kernel = np.ones(wlen) / wlen
        base = np.convolve(base, kernel, mode='same')
        a = np.exp(0.8 * base)
    elif kind == 6:
        # Asymmetric two-hump
        x1 = float(rng.uniform(0.1, 0.4))
        x2 = float(rng.uniform(0.6, 0.9))
        s1 = float(rng.uniform(0.05, 0.2))
        s2 = float(rng.uniform(0.05, 0.2))
        a1 = float(rng.uniform(0.5, 1.5))
        a2 = float(rng.uniform(0.3, 1.0))
        a = a1 * np.exp(-0.5 * ((x - x1) / s1) ** 2) + a2 * np.exp(-0.5 * ((x - x2) / s2) ** 2)
        a = np.maximum(a, 1e-6)
    elif kind == 7:
        # Power-law asymmetric
        power = float(rng.uniform(0.5, 3.0))
        a = x ** power
        if rng.random() < 0.5:
            a = (1 - x) ** power
        a = np.maximum(a, 1e-6)
    else:
        a = rng.random(N) + 0.1

    return normalize_nonneg(np.maximum(a, 0.0))


def resize_to_length(a, N):
    """Resize sequence to length N."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == N:
        return normalize_nonneg(a)
    if n < N:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, N)
        out = np.interp(x_new, x_old, a)
    else:
        # Downsample by averaging
        out = np.zeros(N, dtype=np.float64)
        for i in range(N):
            lo = int(i * n / N)
            hi = max(lo + 1, int((i + 1) * n / N))
            out[i] = a[lo:hi].mean()
    return normalize_nonneg(np.maximum(out, 0.0))

    """
    rng = np.random.default_rng(seed)
    hard_deadline = time.monotonic() + max(0.0, float(budget_s) - 80.0)

    state = {"score": float("inf"), "seq": None}

    def update_state(seq, score):
        if np.isfinite(score) and score < state["score"] - 1e-12:
            state["score"] = float(score)
            state["seq"] = normalize_nonneg(seq).copy()
            print(f"  new best: score={score:.10f}, n={seq.size}", flush=True)

    # Load provided sequence
    provided = load_provided_sequence()
    if provided is not None:
        ps = compute_score_normalized(provided)
        update_state(provided, ps)

    # Guaranteed fallback
    fb = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
    update_state(fb, compute_score_normalized(fb))

    # Phase 1: Quick exploration at small sizes (100s)
    print("Phase 1: Quick exploration (n=32..256)", flush=True)
    t_phase1_end = time.monotonic() + 80.0
    for N in (32, 64, 128, 256):
        for kind in (0, 1, 3, 7):
            if time.monotonic() > t_phase1_end:
                break
            seq = make_initial(N, kind, rng, state["seq"], provided)
            sc = compute_score_normalized(seq)
            if sc < state["score"] + 0.5:
                # Worth optimizing
                tl = min(8.0, t_phase1_end - time.monotonic())
                if tl > 2:
                    seq2, sc2 = hybrid_optimize(seq, tl, rng)
                    update_state(seq2, sc2)

    # Phase 2: Medium optimization (n=512..1024) (600s)
    print("Phase 2: Medium optimization (n=512..1024)", flush=True)
    t_phase2_end = time.monotonic() + 550.0
    kinds_medium = [99, 98 if provided is not None else 0, 0, 1, 3, 7, 5]
    for N in (512, 1024):
        for kind in kinds_medium:
            if time.monotonic() > t_phase2_end:
                break
            seq = make_initial(N, kind, rng, state["seq"], provided)
            sc = compute_score_normalized(seq)
            # Skip if clearly bad
            if sc > state["score"] + 0.3 and kind not in (99, 98):
                continue
            tl = min(40.0, t_phase2_end - time.monotonic() - 5.0)
            if tl < 5:
                break
            print(f"  optimizing n={N}, kind={kind}, init_score={sc:.6f}, time={tl:.0f}s", flush=True)
            seq2, sc2 = hybrid_optimize(seq, tl, rng)
            update_state(seq2, sc2)

    # Phase 3: Large refinement (n=2048..4096) (800s)
    print("Phase 3: Large refinement (n=2048..4096)", flush=True)
    t_phase3_end = time.monotonic() + 750.0
    for N in (2048, 4096):
        if time.monotonic() > t_phase3_end - 60:
            break
        if state["seq"] is None:
            continue
        for attempt in range(2):
            if time.monotonic() > t_phase3_end - 10:
                break
            seq = resize_to_length(state["seq"], N)
            tl = min(100.0, t_phase3_end - time.monotonic() - 10.0)
            if tl < 20:
                break
            print(f"  refining n={N}, attempt={attempt}, time={tl:.0f}s", flush=True)
            # For large n, use smooth-max primarily
            c = np.convolve(seq, seq)
            tau_init = max(0.003, 0.3 * float(np.std(c)))
            seq2, sc2 = smooth_max_optimize(seq, tl, tau_init=tau_init, step_init=0.005, rng=rng)
            update_state(seq2, sc2)

    # Phase 4: Final polish on best (200s)
    print("Phase 4: Final polish", flush=True)
    if state["seq"] is not None and time.monotonic() < hard_deadline - 10:
        # Try multiple polish strategies
        cur_n = state["seq"].size
        # Try same size with different tau
        for tau_mult in (1.0, 0.5, 2.0):
            if time.monotonic() > hard_deadline - 10:
                break
            tl = min(50.0, hard_deadline - time.monotonic() - 10.0)
            if tl < 10:
                break
            c = np.convolve(state["seq"], state["seq"])
            tau = max(0.001, tau_mult * 0.3 * float(np.std(c)))
            seq2, sc2 = smooth_max_optimize(state["seq"], tl, tau_init=tau,
                                             tau_min=0.0005, step_init=0.003, rng=rng)
            update_state(seq2, sc2)

        # Try upscaling one more step if beneficial
        if cur_n < 4096:
            next_n = min(4096, cur_n * 2)
            if time.monotonic() < hard_deadline - 60:
                seq = resize_to_length(state["seq"], next_n)
                tl = min(80.0, hard_deadline - time.monotonic() - 15.0)
                if tl > 30:
                    c = np.convolve(seq, seq)
                    tau = max(0.002, 0.3 * float(np.std(c)))
                    seq2, sc2 = smooth_max_optimize(seq, tl, tau_init=tau,
                                                     tau_min=0.0005, step_init=0.003, rng=rng)
                    update_state(seq2, sc2)

    # Also try the provided sequence at its native length if it's in a good range
    if provided is not None and 256 <= provided.size <= 4096 and time.monotonic() < hard_deadline - 30:
        tl = min(40.0, hard_deadline - time.monotonic() - 10.0)
        if tl > 15:
            c = np.convolve(provided, provided)
            tau = max(0.002, 0.3 * float(np.std(c)))
            seq2, sc2 = smooth_max_optimize(provided, tl, tau_init=tau,
                                             tau_min=0.0005, step_init=0.005, rng=rng)
            update_state(seq2, sc2)

    if state["seq"] is None:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]

    cand = [float(x) for x in np.asarray(state["seq"], dtype=np.float64)]
    print(f"Final: score={state['score']:.10f}, n={len(cand)}", flush=True)
    try:
        official = evaluate_sequence(cand)
        print(f"Official score: {official:.10f}", flush=True)
    except Exception:
        pass
    return cand
