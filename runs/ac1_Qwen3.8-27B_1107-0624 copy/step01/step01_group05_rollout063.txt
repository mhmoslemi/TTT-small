"""
Minimizes 2*n*max_k((a*a)[k])/(sum a)^2 via logit-parameterized optimization.
Uses L-BFGS-B with tight log-sum-exp surrogate, followed by accelerated
subgradient polishing with top-k peak weighting.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

N = 4096
X_GRID = (np.arange(N, dtype=np.float64) + 0.5) / float(N)
_FFT_SELF = 1 << (2 * N - 2).bit_length()  # 8192
_FFT_FULL = 1 << (3 * N - 2).bit_length()  # 16384
_BOUNDS = [(-50.0, 50.0)] * (N - 1)

_best_score = float("inf")
_best_z = None
_search_deadline = 0.0


class SearchTimeout(Exception):
    pass


def _sanitized_z(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -50.0, 50.0)
    return x


def _probs_from_z(z):
    z = _sanitized_z(z)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _self_conv(p):
    f = np.fft.rfft(p, _FFT_SELF)
    c = np.fft.irfft(f * f, _FFT_SELF).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_p(p):
    c = _self_conv(p)
    return 2.0 * float(N) * float(np.max(c))


def _score_z(z):
    global _best_score, _best_z
    x = _sanitized_z(z)
    p = _probs_from_z(x)
    s = _score_p(p)
    if np.isfinite(s) and s < _best_score:
        _best_score = float(s)
        _best_z = np.array(x, copy=True)
    return s


def _grad_p_from_weights(p, w):
    """Gradient of sum_k w_k (p*p)[k] w.r.t. p."""
    # (p*p)[k] = sum_i p[i] p[k-i]
    # d/dp[j] sum_k w_k (p*p)[k] = sum_k w_k (p[k-j] + p[k-j]) = 2 * sum_k w_k p[k-j]
    # = 2 * (w_rev * p)[j] where w_rev[k] = w[2N-2-k]
    # Actually: sum_k w_k p[k-j] for k-j in [0,N-1]
    # = sum_i w_{j+i} p[i] for i in [0,N-1] (substituting i=k-j)
    # So grad[j] = 2 * sum_i p[i] w[j+i]
    
    # We can compute this as a cross-correlation
    # Let's use FFT: convolve p reversed with w
    conv = np.fft.irfft(
        np.fft.rfft(p[::-1], _FFT_FULL) * np.fft.rfft(w, _FFT_FULL),
        _FFT_FULL
    ).real
    h = conv[N-1:2*N-1]
    return 2.0 * h


def _objective_and_grad(x, alpha):
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout
    
    x = _sanitized_z(x)
    p = _probs_from_z(x)
    c = _self_conv(p)
    
    cmax = float(np.max(c))
    hard = 2.0 * float(N) * cmax
    if np.isfinite(hard) and hard < _best_score:
        _best_score = float(hard)
        _best_z = np.array(x, copy=True)
    
    # Log-sum-exp surrogate: cmax + log(sum exp(alpha*(c - cmax))) / alpha
    u = alpha * (c - cmax)
    # Clip to avoid overflow (u should be <= 0)
    e = np.exp(np.minimum(u, 0.0))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        idx = int(np.argmax(c))
        w[idx] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s
    
    # Gradient w.r.t. p
    gp = _grad_p_from_weights(p, w)
    # Gradient w.r.t. z: gz[j] = p[j+1] * (gp[j+1] - sum(p*gp))
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e8, 1e8)
    return float(J), gz

def _eval_candidates(cands):

def _polish_subgradient(z0, deadline, rng):
    """Subgradient polishing with top-k weighting and backtracking."""
    global _best_score, _best_z
    
    z = _sanitized_z(z0)
    p = _probs_from_z(z)
    score = _score_p(p)
    _score_z(z)
    
    # Polyak lower bound: average of convolution = 1/(2N-1)
    lb = 1.0 / (2.0 * N - 1)
    obj_lower = 2.0 * N * lb
    
    topk = 16
    noise_level = 0.1
    fails = 0
    max_fails = 200
    step_base = 0.3
    
    while time.monotonic() < deadline and fails < max_fails:
        c = _self_conv(p)
        cmax = float(np.max(c))
        obj_val = 2.0 * N * cmax
        
        # Top-k subgradient
        idx = np.argpartition(c, -topk)[-topk:]
        # Weight proportional to exp(alpha*(c - cmax)) for tightness
        alpha_w = 50000.0
        wc = c[idx] - cmax
        ew = np.exp(np.minimum(alpha_w * wc, 0.0))
        w = np.zeros(2 * N - 1, dtype=np.float64)
        w[idx] = ew / float(np.sum(ew))
        
        gp = _grad_p_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))
        
        if gnorm < 1e-15:
            break
        
        direction = gz / gnorm
        accepted = False
        
        # Try several step sizes (backtracking-style)
        # Polyak step: (f - f_lb) / ||g||^2 in p-space
        # In z-space, approximately: step_z ≈ (f - f_lb) / (||g||^2 * mean(p))
        polyak_obj = (obj_val - obj_lower) / max(gnorm * gnorm, 1e-20)
        polyak_step = min(1.0, polyak_obj * 10.0)  # Scale for z-space
        
        steps = [polyak_step, 0.35, 0.18, 0.09, 0.045, 0.022, 0.01]
        steps = [max(0.001, s) for s in steps]
        # Remove duplicates and sort descending
        steps = sorted(set(steps), reverse=True)
        
        for step in steps:
            zt = z - step * direction
            zt = _sanitized_z(zt)
            pt = _probs_from_z(zt)
            st = _score_p(pt)
            if st < score - 1e-13:
                z = zt
                p = pt
                score = st
                _score_z(z)
                accepted = True
                break
        
        if not accepted:
            # Try noisy perturbations
            found = False
            for _ in range(5):
                zt = z + rng.normal(0, noise_level, N - 1)
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _score_z(z)
                    accepted = True
                    found = True
                    break
            if not found:
                # Try smaller step in direction
                for step in [0.005, 0.002, 0.001]:
                    zt = z - step * direction
                    zt = _sanitized_z(zt)
                    pt = _probs_from_z(zt)
                    st = _score_p(pt)
                    if st < score - 1e-13:
                        z = zt
                        p = pt
                        score = st
                        _score_z(z)
                        accepted = True
                        break
        
        if accepted:
            fails = 0
        else:
            fails += 1
            noise_level = max(0.01, noise_level * 0.85)
    
    return z, score

    f = np.exp(beta * (x - 0.5))
    return _density_to_z(f)


def _init_random_poly(rng, deg=5):
    x = X_GRID
    t = x - 0.5
    coef = rng.normal(scale=1.0, size=deg + 1)
    f = np.ones(N, dtype=np.float64)
    val = np.ones(N, dtype=np.float64)
    for k in range(deg, 0, -1):
        val = val * t + coef[k]
    f += val
    f = np.maximum(f, 1e-10)
    return _density_to_z(f)


def _init_random_sinusoidal(rng):
    x = X_GRID[1:]
    z = np.zeros_like(x)
    n_freq = int(rng.integers(2, 6))
    for _ in range(n_freq):
        fq = int(rng.integers(1, 10))
        amp = rng.normal(0, 2.0 / (fq + 0.5))
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.sin(2 * np.pi * fq * x + ph)
    return np.clip(z, -50, 50)


def _init_random_smooth(rng):
    x = X_GRID[1:]
    n_comp = 5
    z = np.zeros_like(x)
    for _ in range(n_comp):
        freq = rng.uniform(0.5, 8.0)
        amp = rng.normal(0, 1.5)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.sin(2 * np.pi * freq * (x - 0.5) + ph)
    return np.clip(z, -50, 50)


def _make_candidates(rng, n_random=80):
    cands = []
    cands.append(_init_uniform())
    for beta in [1, 2, 3, 4, 5, 7, 10]:
        cands.append(_init_logistic(beta))
        cands.append(_init_logistic(-beta))
    for rho in [0.3, 0.5, 0.7, 0.9]:
        cands.append(_init_linear(rho))
        cands.append(_init_linear(-rho))
    for a in [0.5, 1.0, 1.5, 2.0, 3.0]:
        cands.append(_init_power(a))
        cands.append(_init_power(a, reverse=True))
    for ampl in [0.3, 0.5, 0.8]:
        for freq in [1, 2, 3, 4, 6]:
            cands.append(_init_cosine(ampl, freq))
            cands.append(_init_cosine(-ampl, freq))
    for depth in [0.5, 1.0, 2.0, 4.0]:
        cands.append(_init_bathtub(depth))
    for width in [1, 2, 4, 8]:
        cands.append(_init_bell(width))
    for beta in [1, 2, 3, 5]:
        cands.append(_init_exp(beta))
        cands.append(_init_exp(-beta))
    for _ in range(n_random // 3):
        cands.append(_init_random_poly(rng, deg=int(rng.integers(3, 8))))
    for _ in range(n_random // 3):
        cands.append(_init_random_sinusoidal(rng))
    for _ in range(n_random // 3):
        cands.append(_init_random_smooth(rng))
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0, None)
            if float(np.sum(base)) > 0:
                if base.size != N:
                    old = np.linspace(0, 1, base.size)
                    new = np.linspace(0, 1, N)
                    base = np.interp(new, old, base)
                base = base / float(np.sum(base))
                cands.append(_density_to_z(base))
    except Exception:
        pass
    return cands


def _eval_candidates(cands):
    scored = []
    for z in cands:
        z = _sanitized_z(z)
        s = _score_z(z)
        if np.isfinite(s):
            scored.append((s, z))
    scored.sort(key=lambda t: t[0])
    return scored


def _choose_alpha(score):
    if not np.isfinite(score):
        return 1e6
    if score > 1.90:
        return 5e5
    if score > 1.75:
        return 2e6
    if score > 1.60:
        return 5e6
    if score > 1.52:
        return 20e6
    if score > 1.51:
        return 50e6
    return 100e6


def _polish_subgradient(z0, deadline, rng, topk=16, alpha_w=1e5):
    global _best_score, _best_z
    z = _sanitized_z(z0)
    p = _probs_from_z(z)
    score = _score_p(p)
    _update_best(z, score)
    obj_lower = 2.0 * N / (2.0 * N - 1)
    noise_level = 0.1
    fails = 0
    max_fails = 300
    conv_len = 2 * N - 1
    while time.monotonic() < deadline and fails < max_fails:
        c = _self_conv(p)
        cmax = float(np.max(c))
        obj_val = 2.0 * N * cmax
        idx = np.argpartition(c, -topk)[-topk:]
        wc = c[idx] - cmax
        ew = np.exp(np.minimum(alpha_w * wc, 0.0))
        w = np.zeros(conv_len, dtype=np.float64)
        w[idx] = ew / float(np.sum(ew))
        gp = _grad_p_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))
        if gnorm < 1e-15:
            break
        direction = gz / gnorm
        accepted = False
        polyak_step = min(1.0, max(0.01, 20.0 * (obj_val - obj_lower) / max(gnorm * gnorm, 1e-20)))
        steps = sorted(set([polyak_step, 0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006]), reverse=True)
        for step in steps:
            zt = z - step * direction
            zt = _sanitized_z(zt)
            pt = _probs_from_z(zt)
            st = _score_p(pt)
            if st < score - 1e-13:
                z = zt
                p = pt
                score = st
                _update_best(z, score)
                accepted = True
                break
        if not accepted:
            for _ in range(5):
                zt = z + rng.normal(0, noise_level, N - 1)
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(z, score)
                    accepted = True
                    break
        if not accepted:
            for step in [0.003, 0.001]:
                zt = z - step * direction
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(z, score)
                    accepted = True
                    break
        if accepted:
            fails = 0
        else:
            fails += 1
            noise_level = max(0.005, noise_level * 0.87)
    return z, score
