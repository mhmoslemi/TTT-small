    result = np.fft.irfft(fa * fa, size)
    return result[:2*n-1]

    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def compute_grad(a, top_k=3):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n), 1e15
    
    # Find top-k lags
    top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]  # sort descending
    
    # Softmax weights over top-k
    local_c = c[top_idx]
    local_max = local_c[0]
    spread = local_max - local_c[-1]
    tau = max(1e-10, spread * 0.05) if spread > 0 else 1e-10
    w = np.exp((local_c - local_max) / tau)
    w /= w.sum()
    
    weighted_max = np.dot(w, local_c)
    
    # Repulsion vector
    r = np.zeros(n)
    for k, wi in zip(top_idx, w):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        r[i_lo:i_hi+1] += wi * a[i_lo:i_hi+1][::-1]
    
    # Gradient
    grad = (4.0 * n / (S * S)) * (r - weighted_max / S)
    
    return grad, weighted_max

"""

Uses exact subgradient with heavy-ball momentum, multiplicative updates,

Key improvements over prior: exact subgradient (no smoothing), conflict detection,

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/S^2 using FFT."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def compute_grad_exact(a, k_star):
    """Compute exact subgradient at a single active lag k_star.
    Returns (gradient array, max_c value)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n), 1e15
    c = convolve_fft(a)
    max_c = c[k_star]
    # Repulsion: r[i] = a[k_star - i] for valid i
    r = np.zeros(n)
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(n - 1, k_star)
    r[i_lo:i_hi + 1] = a[i_lo:i_hi + 1][::-1]
    grad = (4.0 * n / (S * S)) * (r - max_c / S)
    return grad, max_c


def compute_grad_weighted(a, top_k=3, tau_frac=0.05):
    """Compute weighted gradient over top-k peaks with softmax weights."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n), 1e15
    # Find top-k
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    local_c = c[top_idx]
    local_max = local_c[0]
    spread = local_max - local_c[-1]
    tau = max(1e-10, spread * tau_frac) if spread > 1e-15 else 1e-10
    w = np.exp((local_c - local_max) / tau)
    w /= w.sum()
    weighted_max = np.dot(w, local_c)
    # Repulsion vector
    r = np.zeros(n)
    for k, wi in zip(top_idx, w):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        r[i_lo:i_hi + 1] += wi * a[i_lo:i_hi + 1][::-1]
    grad = (4.0 * n / (S * S)) * (r - weighted_max / S)
    return grad, weighted_max


def find_best_lag(a):
    """Find the lag where convolution is maximized."""
    c = convolve_fft(a)
    return int(np.argmax(c))

def run_multiplicative(a_init, budget_s, seed):
    """Multiplicative (log-space) updates: a[i] *= exp(-eta * r[i] / max_r)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    eta = 0.05
    t = 0
    stall_count = 0

    while time.monotonic() < deadline:
        # Compute repulsion from the active peak
        k_star = find_best_lag(a)
        c = convolve_fft(a)
        max_c = c[k_star]
        S_cur = np.sum(a)

        r = np.zeros(n)
        i_lo = max(0, k_star - (n - 1))
        i_hi = min(n - 1, k_star)
        r[i_lo:i_hi + 1] = a[i_lo:i_hi + 1][::-1]

        # Normalize r
        max_r = np.max(r)
        if max_r < 1e-15:
            break
        r_norm = r / max_r

        # Multiplicative update: reduce hot spots, boost cold spots
        # a_new[i] = a[i] * exp(-eta * (r_norm[i] - mean(r_norm)))
        mean_r = np.mean(r_norm)
        update = np.exp(-eta * (r_norm - mean_r))
        a = a * update
        a = np.maximum(a, 0.0)
        S_new = np.sum(a)
        if S_new < 0.01:
            a = best_a.copy()
            stall_count += 1
            eta *= 0.5
            continue
        a = a * (n / S_new)
        t += 1

        # Decay eta
        eta *= 0.9995

        # Check improvement
        if t % 500 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                stall_count = 0
                if J < 1.51:
                    print(f"    multiplicative t={t} J={J:.6f}")
            else:
                stall_count += 1
                if stall_count > 30:
                    # Restart
                    a = best_a.copy()
                    eta = 0.05
                    # Small perturbation
                    idx = rng.choice(n, size=int(n // 30), replace=False)
                    a[idx] *= np.exp(rng.normal(0, 0.05, len(idx)))
                    a = np.maximum(a, 0)
                    a = a * (n / np.sum(a))
                    stall_count = 0

        # Verify periodically
        if t % 50000 == 0:
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    multiplicative verified t={t} J={val:.6f}")

    J_final = compute_J(best_a)
    best_J = min(best_J, J_final)
    return best_a, best_J


def run_scipy_lbfgs(a_init, budget_s, seed, tau=0.5):
    """Use scipy L-BFGS-B with log-sum-exp smoothing."""
    from scipy.optimize import minimize
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)
    deadline = time.monotonic() + budget_s

    def objective(x):
        x = np.maximum(x, 0)
        c = convolve_fft(x)
        S_x = np.sum(x)
        if S_x < 0.01:
            return 1e15
        c_max = np.max(c)
        shifted = (c - c_max) / tau
        exp_shifted = np.exp(shifted)
        if np.max(exp_shifted) > 1e300:
            return 1e15
        T = c_max + tau * np.log(np.sum(exp_shifted))
        val = 2.0 * n * T / (S_x * S_x)

        def grad_inner():
            p = exp_shifted / np.sum(exp_shifted)
            size = 1
            while size < 2 * n - 1:
                size <<= 1
            fa = np.fft.rfft(x, size)
            fp = np.fft.rfft(p, size)
            cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
            g = (4.0 * n * cross) / (S_x * S_x) - (4.0 * n * T) / (S_x * S_x * S_x)
            return g

        return val, grad_inner()

    # Run minimize with time check
    best_x = a.copy()
    current_tau = tau
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        options = {
            'maxiter': int(remaining * 50),
            'maxfun': int(remaining * 50),
            'ftol': 1e-10,
            'gtol': 1e-8,
            'maxls': 20,
        }
        try:
            res = minimize(objective, best_x, method='L-BFGS-B',
                           options=options, jac=True)
            if res.success or res.fun < best_J:
                best_x = np.maximum(res.x, 0)
                S_b = np.sum(best_x)
                if S_b > 0:
                    best_x = best_x * (n / S_b)
                J_check = compute_J(best_x)
                if J_check < best_J:
                    best_J = J_check
                    best_a = best_x.copy()
                    print(f"    lbfgs tau={current_tau:.3f} J={J_check:.6f}")
        except Exception:
            pass

        # Decrease tau for sharper approximation
        current_tau *= 0.5
        if current_tau < 0.01:
            break

        # Perturb to escape
        if time.monotonic() < deadline:
            rng = np.random.default_rng(seed + int(tau * 1000))
            best_x *= (1 + 0.02 * rng.normal(0, 1, n))
            best_x = np.maximum(best_x, 0)
            S_b = np.sum(best_x)
            if S_b > 0:
                best_x = best_x * (n / S_b)

    J_final = compute_J(best_a)
    best_J = min(best_J, J_final)
    return best_a, best_J


def run_local_search(a_init, budget_s, seed):
    """Local search with structured block perturbations."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    step_scale = 0.03
    t = 0
    stall = 0

    while time.monotonic() < deadline:
        # Choose perturbation type
        mt = t % 4
        cand = a.copy()

        if mt == 0:
            # Block modulation
            w = int(rng.integers(max(1, n // 500), max(2, n // 20)))
            s = int(rng.integers(0, max(1, n - w)))
            cand[s:s + w] *= np.exp(rng.normal(0, step_scale, w))
        elif mt == 1:
            # Pair swap (move mass from hot to cold)
            k_star = find_best_lag(cand)
            c = convolve_fft(cand)
            # Find positions that contribute most to the peak
            i_lo = max(0, k_star - (n - 1))
            i_hi = min(n - 1, k_star)
            contrib = cand[i_lo:i_hi + 1] * cand[i_lo:i_hi + 1][::-1]
            # Move mass from high-contrib to low-contrib
            n_move = int(rng.integers(1, 5))
            for _ in range(n_move):
                hi_idx = i_lo + int(np.argmax(contrib))
                lo_idx = int(rng.integers(0, n))
                if hi_idx == lo_idx:
                    continue
                delta = 0.01 * cand[hi_idx]
                cand[hi_idx] -= delta
                cand[lo_idx] += delta
        elif mt == 2:
            # Sinusoidal phase shift
            freq = rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2 * np.pi)
            mod = 1 + 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            cand *= mod
        else:
            # Random individual boosts
            n_single = int(rng.integers(3, 15))
            idx = rng.choice(n, size=n_single, replace=False)
            cand[idx] *= np.exp(rng.normal(0, step_scale, n_single))

        cand = np.maximum(cand, 0)
        S_c = np.sum(cand)
        if S_c < 0.01:
            continue
        cand = cand * (n / S_c)

        cJ = compute_J(cand)
        t += 1

        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.97)
            stall = 0
            if best_J < 1.51:
                print(f"    local_search t={t} J={best_J:.6f}")
        else:
            stall += 1
            step_scale = min(0.1, step_scale * 1.01)
            if stall > 500:
                a = best_a.copy()
                step_scale = 0.03
                stall = 0

        # Verify periodically
        if t % 50000 == 0:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()

    J_final = compute_J(best_a)
    best_J = min(best_J, J_final)
    return best_a, best_J


def make_chirp_init(n):
    """Create a chirp-based initialization (real non-negative)."""
    x = np.arange(n)
    # Quadratic chirp, take absolute value and normalize
    a = np.abs(np.cos(np.pi * x * x / (2.0 * n)))
    a = a + 0.1  # Ensure positivity
    S = np.sum(a)
    return a * (n / S)


def make_multi_bump_init(n, rng):
    """Create a multi-bump initialization."""
    a = np.zeros(n)
    n_bumps = int(rng.integers(3, 8))
    for _ in range(n_bumps):
        center = int(rng.integers(0, n))
        width = int(rng.integers(n // 50, n // 10))
        height = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a += height * np.exp(-(x - center) ** 2 / (2.0 * width ** 2))
    a = a + 0.01
    S = np.sum(a)
    return a * (n / S)


def make_fourier_init(n, rng, n_coeffs=15):
    """Create a Fourier-based initialization with random coefficients."""
    a = np.ones(n) * 0.5
    for k in range(1, n_coeffs + 1):
        amp_c = rng.uniform(-0.3, 0.3) / k
        amp_s = rng.uniform(-0.3, 0.3) / k
        a += amp_c * np.cos(2 * np.pi * k * np.arange(n) / n)
        a += amp_s * np.sin(2 * np.pi * k * np.arange(n) / n)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    return a


def run_adaptive_gd(a_init, budget_s, seed):
    """Adaptive gradient descent with phase-based strategy:
    Phase 1: Smooth (large tau) for broad optimization
    Phase 2: Sharper (medium tau)
    Phase 3: Exact subgradient (tau->0)
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    start_time = time.monotonic()
    total_budget = budget_s

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))

        # Adaptive tau: start smooth, end sharp
        if frac < 0.3:
            tau = 2.0 * (1.0 - frac / 0.3) + 0.5
            top_k = 5
            lr = 0.005
        elif frac < 0.6:
            tau = 0.5 * (1.0 - (frac - 0.3) / 0.3) + 0.1
            top_k = 3
            lr = 0.003
        elif frac < 0.85:
            tau = 0.1 * (1.0 - (frac - 0.6) / 0.25) + 0.02
            top_k = 2
            lr = 0.001
        else:
            tau = 0.02
            top_k = 1
            lr = 0.0005

        # Compute gradient
        if top_k == 1:
            k_star = find_best_lag(a)
            grad, mc = compute_grad_exact(a, k_star)
        else:
            grad, mc = compute_grad_weighted(a, top_k=top_k, tau_frac=tau / mc if mc > 0 else 0.05)

        if not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S_new = np.sum(a)
        if S_new < 0.01:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        a = a * (n / S_new)

        # Check periodically
        if t % 3000 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                if best_J < 1.51:
                    print(f"    adaptive_gd t={t} frac={frac:.2f} J={J:.6f}")
            elif t % 15000 == 0:
                # Perturb
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                w = int(n // 100)
                s = int(rng.integers(0, max(1, n - w)))
                a[s:s + w] *= np.exp(rng.normal(0, 0.05, w))
                a = np.maximum(a, 0)
                a = a * (n / np.sum(a))

        # Verify
        if t % 300000 == 0:
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    adaptive_gd verified t={t} J={val:.6f}")

    J_final = compute_J(best_a)
    best_J = min(best_J, J_final)
    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer combining exact subgradient descent,
    multiplicative updates, scipy L-BFGS-B, and local search.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    parent = parent * (n / np.sum(parent))
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Phase 1: Multi-start (300s)
    p1_budget = min(300.0, (total_deadline - time.monotonic()) - 1300.0)
    p1_budget = max(p1_budget, 60.0)
    print(f"\n=== Phase 1: Multi-start ({p1_budget:.0f}s) ===")

    init_parent = parent.copy()
    init_chirp = make_chirp_init(n)
    init_fourier = make_fourier_init(n, rng, 15)
    init_bump = make_multi_bump_init(n, rng)
    init_perturbed = parent * (1 + 0.05 * rng.normal(0, 1, n))
    init_perturbed = np.maximum(init_perturbed, 0)
    init_perturbed = init_perturbed * (n / np.sum(init_perturbed))

    p1_results = []
    p1_strategies = [
        ("exact_gd_p", init_parent, "exact"),
        ("adaptive_gd_p", init_parent, "adaptive"),
        ("mult_p", init_parent, "mult"),
        ("exact_gd_chi", init_chirp, "exact"),
        ("adaptive_gd_f", init_fourier, "adaptive"),
        ("local_p", init_perturbed, "local"),
    ]

    per_strat = p1_budget / len(p1_strategies)
    for i, (name, init_arr, method) in enumerate(p1_strategies):
        remaining = total_deadline - time.monotonic() - 1000.0
        if remaining < 10:
            break
        b = min(per_strat, remaining)
        if b < 10:
            break
        print(f"  P1: {name} (budget={b:.0f}s)")
        try:
            if method == "exact":
                res_a, res_J = run_exact_gd(init_arr, b, seed + i * 100, lr_start=0.004)
            elif method == "adaptive":
                res_a, res_J = run_adaptive_gd(init_arr, b, seed + i * 100)
            elif method == "mult":
                res_a, res_J = run_multiplicative(init_arr, b, seed + i * 100)
            else:
                res_a, res_J = run_local_search(init_arr, b, seed + i * 100)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"  *** P1 NEW BEST: {val:.6f} ***")
            else:
                print(f"  P1: {name} -> J={res_J:.6f}")
            p1_results.append((res_a, res_J))
        except Exception as e:
            print(f"  P1: {name} failed: {e}")

    # Phase 2: Intensive optimization (900s)
    p2_budget = min(900.0, (total_deadline - time.monotonic()) - 400.0)
    p2_budget = max(p2_budget, 200.0)
    print(f"\n=== Phase 2: Intensive optimization ({p2_budget:.0f}s) ===")

    # Use the best from Phase 1 for intensive work
    best_p2_init = best_global.copy()
    if p1_results:
        best_p1 = min(p1_results, key=lambda x: x[1])
        best_p2_init = best_p1[0].copy()

    # Sub-phase 2a: Exact GD (40% of Phase 2)
    p2a_budget = p2_budget * 0.4
    print(f"  P2a: exact_gd (budget={p2a_budget:.0f}s)")
    res_a, res_J = run_exact_gd(best_p2_init, p2a_budget, seed + 5000, lr_start=0.003)
    val = float(evaluate_sequence(res_a.tolist()))
    if np.isfinite(val) and val < best_global_score:
        best_global = res_a.copy()
        best_global_score = val
        print(f"  *** P2a NEW BEST: {val:.6f} ***")
    else:
        print(f"  P2a: J={res_J:.6f}, verified={val:.6f}")

    # Sub-phase 2b: Multiplicative (30% of Phase 2)
    p2b_budget = p2_budget * 0.3
    remaining = total_deadline - time.monotonic() - 400.0
    p2b_budget = min(p2b_budget, remaining)
    if p2b_budget > 50:
        print(f"  P2b: multiplicative (budget={p2b_budget:.0f}s)")
        res_a, res_J = run_multiplicative(best_global, p2b_budget, seed + 6000)
