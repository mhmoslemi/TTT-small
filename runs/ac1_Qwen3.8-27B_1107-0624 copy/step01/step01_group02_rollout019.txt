```python
def compute_smooth_grad(a, temp):
    """
    Compute the gradient of the smooth max of auto-convolution.
    Returns (grad, c, M) where grad[j] = 2 * sum_i a[i] * p[i+j]
    """
    n = a.size
    c = np.convolve(a, a)  # length 2n-1
    M = c.max()
    e = (c - M) / max(temp, 1e-30)
    e = np.clip(e, -50, 0)
    p = np.exp(e)
    s = p.sum()
    if s < 1e-30:
        p = np.zeros_like(c)
        p[c.argmax()] = 1.0
    else:
        p = p / s
    
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * p[i+j]
```

Looking at this problem, I need to minimize `2*n*max(auto_convolution)/(sum)^2` over non-negative sequences. Since the score is scale-invariant, I normalize to sum=1 and minimize `2*n*max(a*a)`.

Key insights:
1. For non-negative sequences, the peak of the auto-convolution isn't always at the center (unlike auto-correlation)
2. Spectral shaping (making |FFT(a)|² flat) directly flattens the auto-convolution since FT(a*a) = |FT(a)|²
3. A combined approach of smooth-max gradient descent + spectral shaping + progressive upsampling should work well

```python
"""
Step function optimizer using smooth-max gradient descent, spectral shaping,
and progressive upsampling to minimize 2*n*max(auto_conv)/(sum^2).
"""
import numpy as np
import time


def normalize_nonneg(a):
    """Normalize a 1-D array to sum 1, keeping non-negative."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1)
    a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def auto_conv(a):
    """Compute auto-convolution using FFT."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def score_normalized(a):
    """Score for normalized (sum=1) non-negative sequence."""
    if a.size == 0:
        return np.inf
    c = auto_conv(a)
    return float(2.0 * a.size * c.max())


def compute_smooth_grad(a, temp):
    """
    Compute gradient of smooth max of auto-convolution.
    grad[j] = 2 * sum_i a[i] * p[i+j] where p is softmax over convolution values.
    Returns (grad, c, M, p).
    """
    n = a.size
    c = auto_conv(a)
    M = float(c.max())
    e = (c - M) / max(temp, 1e-30)
    e = np.clip(e, -60.0, 0.0)
    p = np.exp(e)
    s = p.sum()
    if s < 1e-30:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / s

    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * p[i+j]
    # This is 2 * cross-correlation of a and p at lag j
    N = 1
    while N < n + (2 * n - 1) - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(fa * np.conj(fp), N)
    grad = 2.0 * corr[:n]

    return grad, c, M, p


def spectral_shape_step(a, alpha, t):
    """
    Apply spectral shaping: compress |FFT(a)| to make auto-convolution flatter.
    alpha < 1 compresses the spectrum. t blends with original.
    """
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    mag = np.abs(fa)
    phase = fa / np.maximum(mag, 1e-30)

    # Compress magnitude: new_mag = mag^alpha (alpha < 1 makes flatter)
    new_mag = np.power(np.maximum(mag, 1e-30), alpha)
    new_fa = new_mag * phase

    a_new = np.fft.irfft(new_fa, size)[:n]
    a_new = np.clip(a_new, 0.0, None)
    a_new = normalize_nonneg(a_new)

    result = (1.0 - t) * a + t * a_new
    return normalize_nonneg(np.maximum(result, 0.0))


def gradient_step(a, grad, c, M, p, temp, max_steps=30):
    """
    Take a gradient descent step with backtracking.
    Uses both additive and multiplicative updates.
    Returns improved a or None.
    """
    n = a.size
    cur_score = float(2.0 * n * M)

    # Try multiplicative update first
    for eta in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
        # Normalize gradient for stability
        grad_norm = float(np.max(np.abs(grad)))
        if grad_norm < 1e-15:
            break
        grad_n = grad / grad_norm

        # Multiplicative: a * exp(-eta * grad_n)
        log_shift = -eta * grad_n
        log_shift -= log_shift.max()  # numerical stability
        a_new = a * np.exp(log_shift)
        a_new = normalize_nonneg(a_new)
        sc = score_normalized(a_new)
        if sc < cur_score - 1e-12:
            return a_new

    # Try additive update
    for eta in (0.01, 0.03, 0.1, 0.3, 0.5):
        grad_norm = float(np.max(np.abs(grad)))
        if grad_norm < 1e-15:
            break
        a_new = a - eta * grad / grad_norm
        a_new = normalize_nonneg(a_new)
        sc = score_normalized(a_new)
        if sc < cur_score - 1e-12:
            return a_new

    return None


def random_perturb(a, rng, scale=0.1):
    """Random multiplicative perturbation."""
    n = a.size
    if n <= 1:
        return a.copy()
    noise = np.exp(rng.normal(0.0, scale, size=n))
    a_new = a * noise
    return normalize_nonneg(a_new)


def block_perturb(a, rng, scale=0.2):
    """Perturb a random block of the sequence."""
    n = a.size
    if n < 4:
        return a.copy()
    width = max(2, int(rng.integers(n // 20, n // 3 + 1)))
    start = int(rng.integers(0, n - width + 1))
    a_new = a.copy()
    a_new[start:start + width] *= np.exp(rng.normal(0.0, scale, size=width))
    return normalize_nonneg(a_new)


def upsample(a, N):
    """Upsample to length N via interpolation."""
    m = a.size
    if m == N:
        return a.copy()
    if m == 1:
        return np.full(N, 1.0 / N)
    x = np.linspace(0.0, float(m - 1), N)
    i0 = np.floor(x).astype(np.int64)
    i1 = np.minimum(i0 + 1, m - 1)
    frac = x - i0
    out = a[i0] * (1.0 - frac) + a[i1] * frac
    return normalize_nonneg(np.maximum(out, 0.0))


def downsample(a, N):
    """Downsample to length N by averaging bins."""
    m = a.size
    if m == N:
        return a.copy()
    if N == 1:
        return np.array([1.0])
    out = np.zeros(N, dtype=np.float64)
    for i in range(N):
        lo = (i * m) // N
        hi = ((i + 1) * m) // N
        if hi > lo:
            out[i] = a[lo:hi].sum()
    return normalize_nonneg(out)


def resize(a, N):
    """Resize sequence to length N."""
    if a.size == N:
        return a.copy()
    if a.size < N:
        return upsample(a, N)
    return downsample(a, N)


def make_initial(N, kind, rng, best_seq=None, provided_seq=None):
    """Create initial normalized non-negative sequence of length N."""
    N = int(max(2, N))

    if kind == -1 and best_seq is not None:
        return resize(best_seq, N)
    if kind == -2 and provided_seq is not None:
        return resize(provided_seq, N)
    if kind in (-1, -2):
        kind = 3

    x = np.linspace(0.0, 1.0, N)
    xs = np.linspace(1.0 / (N + 1), float(N) / (N + 1), N)

    if kind == 0:
        mu = float(rng.uniform(0.2, 0.8))
        sigma = float(rng.uniform(0.15, 0.35))
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    elif kind == 1:
        alpha = float(rng.choice([-2.0, -1.0, -0.5, 0.5, 1.0, 2.0]))
        a = np.exp(alpha * x)
    elif kind == 2:
        alpha = float(rng.uniform(-0.5, 1.5))
        beta = float(rng.uniform(-0.5, 1.5))
        a = xs ** alpha * (1.0 - xs) ** beta
    elif kind == 3:
        # Two adjacent blocks (2/3 mass, 1/3 mass)
        L = N * 2 // 3
        a = np.zeros(N, dtype=np.float64)
        if L > 0:
            a[:L] = (2.0 / 3.0) / L
        if N - L > 0:
            a[L:] = (1.0 / 3.0) / (N - L)
    elif kind == 4:
        # Multi-block
        K = int(rng.integers(2, min(7, N // 4 + 1)))
        K = max(2, K)
        lengths = rng.dirichlet(np.ones(K)) * N
        lengths = np.maximum(1, lengths.astype(np.int64))
        while lengths.sum() < N:
            lengths[int(rng.integers(K))] += 1
        while lengths.sum() > N:
            i = int(np.argmax(lengths))
            if lengths[i] > 1:
                lengths[i] -= 1
        masses = rng.dirichlet(0.5 * np.ones(K))
        a = np.zeros(N, dtype=np.float64)
        pos = 0
        for L, pm in zip(lengths, masses):
            L = int(L)
            if L > 0 and pos + L <= N:
                a[pos:pos + L] = float(pm) / L
                pos += L
    elif kind == 5:
        # Smoothed random
        base = rng.normal(0.0, 1.0, size=N)
        wlen = max(1, min(N, 1 + N // 50))
        if wlen > 1:
            w = np.ones(wlen)
            w /= w.sum()
            base = np.convolve(base, w, mode="same")
        a = np.exp(0.5 * base)
    elif kind == 6:
        # Sinusoidal
        freq = int(rng.integers(2, 8))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.5, 0.99))
        a = 1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    elif kind == 7:
        # "Ramp" shape (asymmetric)
        a = np.linspace(0.1, 2.0, N)
    elif kind == 8:
        # Two separated blocks
        gap = N // 5
        L1 = (N - gap) // 2
        a = np.zeros(N, dtype=np.float64)
        if L1 > 0:
            a[:L1] = (0.6 / L1)
        L2 = N - gap - L1
        if L2 > 0:
            a[gap + L1:] = (0.4 / L2)
    else:
        a = rng.random(N)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        a = np.ones(N)
        s = N
    return a / s


def optimize_trajectory(f, time_limit, rng, state):
    """
    Run a local search trajectory combining:
    - Smooth-max gradient descent (additive + multiplicative)
    - Spectral shaping for exploration
    - Random perturbations for escape
    """
    f = normalize_nonneg(f)
    n = f.size
    if n < 2:
        f = np.array([1.0 / 3.0, 2.0 / 3.0])
        n = f.size

    c = auto_conv(f)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best = f.copy()

    if best_score < state["score"]:
        state["score"] = best_score
        state["seq"] = best.copy()

    deadline = time.monotonic() + max(0.0, float(time_limit))
    it = 0
    no_improve = 0
    temp = 0.15 * M  # temperature for smooth max

    while time.monotonic() < deadline:
        it += 1

        c = auto_conv(f)
        M = float(c.max())
        cur = float(2.0 * n * M)

        if cur < best_score - 1e-12:
            best_score = cur
            best = f.copy()
            state["score"] = best_score
            state["seq"] = best.copy()
            no_improve = 0
            temp = max(0.01 * M, temp * 0.92)
        else:
            no_improve += 1

        if no_improve > 50:
            temp = min(0.5 * M, temp * 1.3)
            # Try spectral shaping as a bigger jump
            alpha = float(rng.uniform(0.3, 0.8))
            t = float(rng.uniform(0.3, 1.0))
            f = spectral_shape_step(f, alpha, t)
            continue

        moved = False

        # Gradient step
        if M > 1e-30:
            grad, c2, M2, p = compute_smooth_grad(f, temp)
            h = gradient_step(f, grad, c2, M2, p, temp)
            if h is not None:
                f = h
                moved = True

        if not moved:
            # Try spectral shaping
            if rng.random() < 0.4:
                alpha = float(rng.uniform(0.4, 0.9))
                t = float(rng.uniform(0.2, 0.8))
                f_new = spectral_shape_step(f, alpha, t)
                if score_normalized(f_new) < cur:
                    f = f_new
                    moved = True

        if not moved:
            # Random perturbations
            r = rng.random()
            if r < 0.5:
                f = random_perturb(f, rng, 0.08)
            else:
                f = block_perturb(f, rng, 0.15)

    # Final check
    c = auto_conv(f)
    final_sc = float(2.0 * f.size * float(c.max()))
    if final_sc < best_score:
        best_score = final_sc
        best = f.copy()
        state["score"] = best_score
        state["seq"] = best.copy()

    return best, best_score


def optimize_with_refinement(f, time_limit, rng, state):
    """
    Enhanced optimization with periodic spectral shaping restarts.
    """
    f = normalize_nonneg(f)
    n = f.size
    deadline = time.monotonic() + max(0.0, float(time_limit))

    c = auto_conv(f)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best = f.copy()

    if best_score < state["score"]:
        state["score"] = best_score
        state["seq"] = best.copy()

    temp = 0.1 * M
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = auto_conv(f)
        M = float(c.max())
        cur = float(2.0 * n * M)

        if cur < best_score - 1e-12:
            best_score = cur
            best = f.copy()
            state["score"] = best_score
            state["seq"] = best.copy()
            no_improve = 0
            temp = max(0.005 * M, temp * 0.95)
        else:
            no_improve += 1

        if no_improve > 40:
            # Escape strategy
            if rng.random() < 0.5:
                alpha = float(rng.uniform(0.2, 0.7))
                t = 1.0
                f = spectral_shape_step(f, alpha, t)
            elif rng.random() < 0.5:
                f = random_perturb(f, rng, 0.15)
            else:
                f = block_perturb(f, rng, 0.25)
            temp = min(0.5 * M, temp * 1.4)
            continue

        if M > 1e-30:
            grad, c2, M2, p = compute_smooth_grad(f, temp)

            # Multiplicative update with various step sizes
            grad_max = float(np.max(np.abs(grad)))
            if grad_max > 1e-15:
                gn = grad / grad_max
                for eta in (0.01, 0.03, 0.1, 0.3, 1.0):
                    log_shift = -eta * gn
                    log_shift -= log_shift.max()
                    a_new = f * np.exp(log_shift)
                    a_new = normalize_nonneg(a_new)
                    sc = score_normalized(a_new)
                    if sc < cur - 1e-12:
                        f = a_new
                        break
                else:
                    # Additive fallback
                    for eta in (0.005, 0.02, 0.1, 0.5):
                        a_new = f - eta * gn
                        a_new = normalize_nonneg(a_new)
                        sc = score_normalized(a_new)
                        if sc < cur - 1e-12:
                            f = a_new
                            break

    c = auto_conv(f)
    final_sc = float(2.0 * f.size * float(c.max()))
    if final_sc < best_score:
        best_score = final_sc
        best = f.copy()
        state["score"] = best_score
        state["seq"] = best.copy()

    return best, best_score

def propose_candidate(seed=42, budget_s=1800):
    """
    Search for low-score step-function heights using a combination of:
    - Smooth-max gradient descent (multiplicative + additive)
    - Spectral shaping (compresses |FFT|^2 to flatten auto-convolution)
    - Progressive upsampling
    - Multiple diverse initializations
    """
    rng = np.random.default_rng(seed)
    hard_deadline = time.monotonic() + max(1.0, float(budget_s) - 80.0)

    state = {"score": float("inf"), "seq": None}

    provided = load_provided_sequence()

    # Initial guaranteed fallbacks
    state["score"] = 16.0 / 9.0
    state["seq"] = np.array([1.0 / 3.0, 2.0 / 3.0])

    if provided is not None:
        ps = score_normalized(provided)
        if ps < state["score"]:
            state["score"] = ps
            state["seq"] = provided.copy()
            print(f"Provided sequence: score={ps:.10f}, n={provided.size}", flush=True)

    # Phase 1: Quick exploration at moderate length
    # Try multiple initializations at n=256 and n=512
    for N in (128, 256, 512):
        now = time.monotonic()
        if now > hard_deadline:
            break

        kinds = [3, 0, 1, 5, 7, 8, -1, -2]
        if provided is not None:
            kinds = [-2, -1, 3, 0, 5, 7, 8]

        for kind in kinds:
            now = time.monotonic()
            if now + 5.0 > hard_deadline:
                break
            limit = min(15.0, hard_deadline - now - 5.0)
            if limit < 3.0:
                continue

            seq = make_initial(N, kind, rng, state["seq"], provided)
            seq_sc = score_normalized(seq)
            print(f"  N={N} kind={kind}: init_score={seq_sc:.6f}", flush=True)

            if seq_sc < state["score"] * 1.5:
                optimize_trajectory(seq, limit, rng, state)

    print(f"After Phase 1: best={state['score']:.10f}, n={state['seq'].size if state['seq'] is not None else 0}", flush=True)

    # Phase 2: Focused refinement at best length
    now = time.monotonic()
    if now < hard_deadline and state["seq"] is not None:
        cur_n = state["seq"].size
        limit = min(120.0, hard_deadline - now - 30.0)
        if limit > 30.0:
            f = state["seq"].copy()
            optimize_with_refinement(f, limit, rng, state)
            print(f"After Phase 2: best={state['score']:.10f}, n={state['seq'].size if state['seq'] is not None else 0}", flush=True)

    # Phase 3: Progressive upsampling
    now = time.monotonic()
    target_sizes = [256, 512, 1024, 2048, 4096, 8192]
    current_best_n = state["seq"].size if state["seq"] is not None else 256

    for target_n in target_sizes:
        now = time.monotonic()
        if now + 60.0 > hard_deadline:
            break
        if target_n < current_best_n:
            continue

        limit = min(
            60.0 + 20.0 * (target_n // 1024),
            hard_deadline - now - 40.0
        )
        if limit < 30.0:
            break

        # Start from current best, upsampled
        f = resize(state["seq"], target_n)

        # Also try spectral-shaped version
        f_alt = spectral_shape_step(f, 0.6, 0.5)
        if score_normalized(f_alt) < score_normalized(f):
            f = f_alt

        sc_before = state["score"]
        optimize_with_refinement(f, limit, rng, state)

        if state["score"] < sc_before - 1e-8:
            print(f"  Upsampled to n={target_n}: score improved to {state['score']:.10f}", flush=True)
        else:
            print(f"  n={target_n}: no improvement (still {state['score']:.10f})", flush=True)
            if target_n >= 2048:
                break

    # Phase 4: Final refinement with multiple restarts at best length
    now = time.monotonic()
    if now < hard_deadline and state["seq"] is not None:
        best_n = state["seq"].size
        # Try a few more restarts at this length
        for restart in range(3):
            now = time.monotonic()
            if now + 40.0 > hard_deadline:
                break
            limit = min(40.0, hard_deadline - now - 20.0)
            if limit < 15.0:
                break

            # Perturb the best and re-optimize
            f = random_perturb(state["seq"], rng, 0.1)
            optimize_with_refinement(f, limit, rng, state)

        # Also try spectral shaping on the best
        now = time.monotonic()
        if now + 30.0 < hard_deadline:
            for alpha in (0.5, 0.6, 0.7, 0.8):
                f = spectral_shape_step(state["seq"], alpha, 0.3)
                sc = score_normalized(f)
                if sc < state["score"]:

                    if limit > 10.0:
                        optimize_with_refinement(f, limit, rng, state)

    print(f"Final best: score={state['score']:.10f}, n={state['seq'].size if state['seq'] is not None else 0}", flush=True)

    if state["seq"] is None:
```
