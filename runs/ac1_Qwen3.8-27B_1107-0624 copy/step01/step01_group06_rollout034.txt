    """Compute gradient using top-K subgradient averaging."""
    c = convolve_fft(a)  # length 2n-1
    S = np.sum(a)
    max_c = np.max(c)
    
    # Get top-K indices
    topk_idx = np.argsort(c)[-K:]
    p = np.zeros(2*n-1)
    p[topk_idx] = 1.0 / K
    
    # Gradient of T = (1/K) * sum_{k in topK} c[k]
    # dT/da[i] = (1/K) * sum_{k in topK} a[k-i] (where k-i in [0,n-1])
    # This is the cross-correlation of p with a, evaluated at [0, n-1]
    
    # Use FFT for cross-correlation
    size = 1
    while size < 3*n-2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)

"""
Multi-stage gradient optimization of step function coefficients to minimize
J(a) = 2*n*max(conv(a,a)) / (sum(a))^2 using FFT-accelerated subgradients
with adaptive top-K active set and staged tau cooling.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def compute_grad_topk(a, K, n):
    """Compute smoothed J and gradient using top-K subgradient.
    
    Returns (J_approx, grad) where J_approx = 2n*T/S^2, T = mean of top-K conv values.
    """
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Get top-K convolution values
    if K >= len(c):
        # Use all (equivalent to uniform weight)
        p = np.ones(len(c)) / len(c)
        T = np.mean(c)
    else:
        # Top-K
        topk = np.argpartition(c, -K)[-K:]
        p = np.zeros(len(c))
        p[topk] = 1.0 / K
        T = np.sum(c[topk]) / K
    
    # Cross-correlation: cross[i] = sum_j a[j] * p[j+i]
    # Using FFT: cross = ifft(fft(p, size) * conj(fft(a, size)))[:n]
    size = 1
    while size < len(c) + n - 1:
        size <<= 1
    
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    # J_approx = 2n*T/S^2
    J = 2.0 * n * T / (S * S)
    
    # dJ/da[i] = 2*n*cross[i]/S^2 - 4*n*T/S^3
    grad = 2.0 * n * cross / (S * S) - 4.0 * n * T / (S * S * S)
    
    return J, grad


def compute_grad_softmax(a, tau, n):
    """Compute smoothed J and gradient using softmax weighting."""
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Softmax weights (numerically stable)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    T = c_max + tau * np.log(sum_exp)
    
    # Cross-correlation
    size = 1
    while size < len(c) + n - 1:
        size <<= 1
    
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    J = 2.0 * n * T / (S * S)
    grad = 2.0 * n * cross / (S * S) - 4.0 * n * T / (S * S * S)
    
    return J, grad

    n = len(a_init)

    best_J = compute_J(a)

    stages = [

"""
Multi-stage gradient optimization of step function coefficients to minimize
J(a) = 2*n*max(conv(a,a)) / (sum(a))^2 using FFT-accelerated subgradients
with corrected gradient formula, adaptive top-K active set, staged tau cooling,
frequency-domain perturbation, and local refinement.

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute exact J = 2n*max(conv(a,a))/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def cross_corr_fft(a, p, n):
    """Compute (a cross p)[i] = sum_j a[j]*p[j+i] for i=0..n-1.
    a has length n, p has length 2n-1.
    Uses the identity: (a cross p)[i] = conv(a, p_rev)[2n-2-i].
    """
    p_rev = p[::-1].copy()
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    c = np.fft.irfft(fa * fp, size)
    # (a cross p)[i] = c[2n-2-i] for i=0..n-1
    # So result[i] = c[2n-2-i], i.e., c[2n-2], c[2n-3], ..., c[n-1]
    result = c[n - 1: 2 * n - 1][::-1]
    return result


def compute_grad_topk(a, K, n):
    """Compute smoothed J and gradient using top-K subgradient averaging.
    
    T = mean of top-K convolution values.
    grad[m] = (4n/S^2) * (cross[m] - T/S)
    """
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    if K >= len(c):
        p = np.ones(len(c)) / len(c)
        T = np.mean(c)
    else:
        topk = np.argpartition(c, -K)[-K:]
        p = np.zeros(len(c))
        p[topk] = 1.0 / K
        T = np.sum(c[topk]) / K

    cross = cross_corr_fft(a, p, n)

    J = 2.0 * n * T / (S * S)
    grad = (4.0 * n / (S * S)) * (cross - T / S)

    return J, grad


def compute_grad_softmax(a, tau, n):
    """Compute smoothed J and gradient using softmax weighting (log-sum-exp).
    
    T = tau * log(sum exp(c/tau)) (numerically stable log-sum-exp)
    grad[m] = (4n/S^2) * (cross[m] - T/S)
    """
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    T = c_max + tau * np.log(sum_exp)

    cross = cross_corr_fft(a, p, n)

    J = 2.0 * n * T / (S * S)
    grad = (4.0 * n / (S * S)) * (cross - T / S)

    return J, grad

        if best_in_stage < compute_J(a):
            a = best_a_in_stage.copy()

    # Final check
    final_J = compute_J(best_a)
    best_J = min(best_J, final_J)
    return best_a, best_J


def freq_domain_perturb(a, rng, strength=0.3):
    """Perturb in frequency domain for global changes."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    A = np.fft.rfft(a, size)
    # Random phase perturbation
    phase_noise = rng.normal(0, strength, len(A))
    A_pert = A * np.exp(1j * phase_noise)
    a_pert = np.fft.irfft(A_pert, size)[:n]
    a_pert = np.maximum(a_pert, 0.0)
    S = np.sum(a_pert)
    if S < 0.01:
        return a
    return a_pert * (n / S)


def run_local_refinement(a_init, budget, seed):
    """Local refinement with adaptive block and coordinate updates."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget
    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    step_scale = 0.03
    no_improve = 0

    while time.monotonic() < deadline - 2:
        # Adaptive block size
        block_size = int(rng.integers(max(1, n // 200), n // 25))
        start = int(rng.integers(0, max(1, n - block_size)))

        cand = a.copy()
        if rng.random() < 0.7:
            # Multiplicative noise
            cand[start:start + block_size] *= np.exp(rng.normal(0, step_scale, block_size))
        else:
            # Additive noise
            cand[start:start + block_size] += rng.normal(0, step_scale * np.mean(a), block_size)
        cand = np.maximum(cand, 0.0)

        # Sometimes try frequency-domain perturbation
        if rng.random() < 0.15:
            cand = freq_domain_perturb(a, rng, strength=step_scale)

        cJ = compute_J(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.95)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                step_scale = min(0.2, step_scale * 1.1)
                if no_improve > 200:
                    # Frequency domain jump
                    cand = freq_domain_perturb(a, rng, strength=0.5)
                    cJ = compute_J(cand)
                    if cJ < best_J:
                        best_J = cJ
                        best_a = cand.copy()
                        a = cand.copy()
                        no_improve = 0
                        step_scale = 0.03
                    else:
                        a = best_a.copy()
                        no_improve = 0

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage gradient optimization with corrected gradient (factor 4n),
    adaptive top-K and softmax subgradients, staged tau cooling,
    frequency-domain perturbation for global exploration, and local refinement.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    total_budget = float(budget_s) - 10.0

    # Get base sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    rng = np.random.default_rng(seed)

    # === Phase 1: Main staged optimization (55% of time) ===
    main_budget = 0.52 * total_budget
    print(f"\n=== Phase 1: Main staged optimization ({main_budget:.0f}s) ===")
    res_a, res_J = run_staged_optimization(best_global, main_budget, seed=seed)
    val = float(evaluate_sequence(res_a.tolist()))
    if np.isfinite(val) and val < best_global_score:
        best_global = res_a.copy()
        best_global_score = val
        print(f"  Phase 1 result: {val:.6f}")
    else:
        print(f"  Phase 1: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # === Phase 2: Secondary optimizations from perturbations (25%) ===
    sec_budget = 0.23 * total_budget
    if time.monotonic() < total_deadline - sec_budget * 0.5:
        print(f"\n=== Phase 2: Secondary optimizations ({sec_budget:.0f}s) ===")
        per_attempt = sec_budget / 3.0

        for attempt in range(3):
            if time.monotonic() >= total_deadline - sec_budget * 0.3:
                break

            # Create perturbed version
            pert = best_global.copy()
            strategy = attempt % 3
            if strategy == 0:
                # Block multiplicative
                w = n // 6
                s = int(rng.integers(0, max(1, n - w)))
                pert[s:s + w] *= np.exp(rng.normal(0, 0.4, w))
            elif strategy == 1:
                # Frequency domain
                pert = freq_domain_perturb(best_global, rng, strength=0.4)
            else:
                # Multi-block
                for _ in range(3):
                    w = int(rng.integers(n // 30, n // 10))
                    s = int(rng.integers(0, max(1, n - w)))
                    pert[s:s + w] *= np.exp(rng.normal(0, 0.5, w))
            pert = np.maximum(pert, 0.0)
            bJ = compute_J(pert)
            print(f"  Attempt {attempt}: perturbed J = {bJ:.6f}")

            remaining = min(per_attempt, total_deadline - time.monotonic() - sec_budget * 0.3)
            if remaining < 15:
                break

            res_a2, res_J2 = run_staged_optimization(pert, remaining, seed=seed + 500 * (attempt + 1))
            val2 = float(evaluate_sequence(res_a2.tolist()))
            if np.isfinite(val2) and val2 < best_global_score:
                best_global = res_a2.copy()
                best_global_score = val2
                print(f"  *** NEW BEST (phase2, attempt {attempt}): {val2:.6f} ***")
            else:
                print(f"  Attempt {attempt}: verified={val2:.6f}, best={best_global_score:.6f}")

    # === Phase 3: Fine staged optimization on current best (12%) ===
    fine_budget = 0.12 * total_budget
    if time.monotonic() < total_deadline - 10:
        print(f"\n=== Phase 3: Fine-tuning ({fine_budget:.0f}s) ===")
        remaining = min(fine_budget, total_deadline - time.monotonic() - 8)
        if remaining > 10:
            # Use a variant with smaller tau range, focused on polishing
            n = len(best_global)
            a_start = best_global.copy()
            deadline3 = time.monotonic() + remaining

            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            best3_J = compute_J(a_start)
            best3_a = a_start.copy()
            K_fine = max(15, n // 60)

            while time.monotonic() < deadline3 - 2:
                frac3 = (time.monotonic() - (deadline3 - remaining)) / max(remaining, 1)
                tau = 0.02 * (0.003 / 0.02) ** frac3
                K_use = max(8, int(K_fine * (1 - 0.5 * frac3)))

                if tau > 0.05:
                    J_sm, grad = compute_grad_softmax(a_start, tau, n)
                else:
                    J_sm, grad = compute_grad_topk(a_start, K_use, n)

                if not np.isfinite(J_sm) or not np.all(np.isfinite(grad)):
                    a_start = best3_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t_adam = 0
                    continue

                t_adam += 1
                b1, b2, eps = 0.9, 0.999, 1e-8
                m = b1 * m + (1 - b1) * grad
                v = b2 * v + (1 - b2) * grad * grad
                m_hat = m / (1 - b1 ** t_adam)
                v_hat = v / (1 - b2 ** t_adam)
                lr = 0.002 / (1 + 0.1 * t_adam / 5000)

                a_start = a_start - lr * m_hat / (np.sqrt(v_hat) + eps)
                a_start = np.maximum(a_start, 0.0)

                if t_adam % 2000 == 0:
                    local_J = compute_J(a_start)
                    if local_J < best3_J:
                        best3_J = local_J
                        best3_a = a_start.copy()

            val3 = float(evaluate_sequence(best3_a.tolist()))
            if np.isfinite(val3) and val3 < best_global_score:
                best_global = best3_a.copy()
                best_global_score = val3
                print(f"  Phase 3: {val3:.6f}")
            else:
                print(f"  Phase 3: verified={val3:.6f}, best={best_global_score:.6f}")

    # === Phase 4: Local refinement (8%) ===
    refine_budget = 0.08 * total_budget
    if time.monotonic() < total_deadline - 5:
        print(f"\n=== Phase 4: Local refinement ({refine_budget:.0f}s) ===")
        remaining = min(refine_budget, total_deadline - time.monotonic() - 5)
        if remaining > 5:
            res_a4, res_J4 = run_local_refinement(best_global, remaining, seed=seed + 7777)
            val4 = float(evaluate_sequence(res_a4.tolist()))
            if np.isfinite(val4) and val4 < best_global_score:
                best_global = res_a4.copy()
                best_global_score = val4
                print(f"  Phase 4: {val4:.6f}")
            else:
                print(f"  Phase 4: verified={val4:.6f}, best={best_global_score:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
