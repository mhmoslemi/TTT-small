Actually, I just realized I can use FFT much more efficiently. Let me think again.

r[m] = sum_{j=0}^{n-1} w[m+j] * s[j] for m = 0, ..., n-1.

Define:
- W: length 2n-1 (the weights)
- S: length n (the sequence)

We want the "valid" cross-correlation where we slide S over W starting at offset 0.

This is equivalent to: r = conv(w, s_flipped) at specific positions.

Let s_flipped = s[::-1] (length n).
conv(w, s_flipped) has length (2n-1) + n - 1 = 3n-2.
conv(w, s_flipped)[k] = sum_j w[j] * s_flipped[k-j] = sum_j w[j] * s[n-1-(k-j)]

We want r[m] = sum_{j=0}^{n-1} w[m+j] * s[j].
Let i = m+j: r[m] = sum_{i=m}^{m+n-1} w[i] * s[i-m].

Now, conv(w, s_flipped)[m + n - 1] = sum_j w[j] * s_flipped[m+n-1-j] = sum_j w[j] * s[n-1-m-n+1+j] = sum_j w[j] * s[j-m].

For j >= m: = sum_{j=m}^{2n-2} w[j] * s[j-m] (with s[j-m] = 0 if j-m >= n)
= sum_{j=m}^{m+n-1} w[j] * s[j-m] (since s[j-m] = 0 for j-m >= n)
= sum_{i=m}^{m+n-1} w[i] * s[i-m] (substituting i=j)
= r[m] ✓

So r[m] = conv(w, s_flipped)[m+n-1] for m = 0, ..., n-1.

Therefore: r = conv(w, s[::-1])[n-1 : 2n-1]

And the gradient is 2*r.

This can be computed with a single FFT-based convolution! Great.

import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
import time

"""
Optimized search for non-negative sequences minimizing 2n*max(conv(s,s))/(sum(s))^2.
Uses gradient-based (log-sum-exp + mirror descent) and LP-guided methods.
"""

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def compute_gradient(s, beta=50.0):
    """Compute gradient of log-sum-exp approximation of max(conv(s,s))."""
    n = len(s)
    c = np.convolve(s, s)
    # Normalize weights
    c_shifted = c - c.max()
    exp_c = np.exp(beta * c_shifted)
    weights = exp_c / exp_c.sum()
    # Gradient: 2 * conv(weights, s[::-1])[n-1 : 2n-1]
    s_flip = s[::-1]
    conv_ws = fftconvolve(weights, s_flip)
    grad = 2.0 * conv_ws[n-1:2*n-1]
    return grad, c.max()

def mirror_descent(s, grad, eta):
    """Mirror descent step on the simplex."""
    # s_new = s * exp(-eta * grad) / sum(s * exp(-eta * grad))
    log_factor = -eta * grad
    # For numerical stability
    log_factor -= log_factor.max()
    factor = np.exp(log_factor)
    s_new = s * factor
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:
        return s
    s_new = s_new / s_sum
    return s_new

def gradient_descent(s, max_iters=1000, deadline=None, verbose=False):
    """Run gradient descent with log-sum-exp smoothing."""
    best_s = s.copy()
    best_sc = fast_score(s)
    s = s.copy()
    s = s / s.sum()
    
    beta = 30.0  # Start with moderate smoothing
    eta = 0.1    # Initial step size
    
    no_improve = 0
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        grad, cmax = compute_gradient(s, beta)
        s = mirror_descent(s, grad, eta)
        
        sc = fast_score(s)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_s = s.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                # Increase beta (sharper approximation of max) and reduce eta
                beta *= 1.5
                eta *= 0.5
                no_improve = 0
                if beta > 1000:
                    break
        
        if verbose and it % 100 == 0:
            print(f"    GD iter {it}: score={sc:.6f}, beta={beta:.0f}, eta={eta:.4f}")
    
    return best_s, best_sc

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if 0<=k-j<n else 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def solve_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) subject to (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def lp_iterate(s, max_iters=200, deadline=None):
    """Perform LP-guided iterations starting from s."""
    best = s.copy()
    best_sc = fast_score(s)
    no_imp = 0
    rng = np.random.default_rng()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = np.max(conv)
        
        b = solve_direction(s, M, time_limit=3.0)
        if b is None or np.sum(b) < 1e-14:
            break
        
        b = b / b.sum()
        
        # Line search over t
        best_t = 0.0
        best_ts = best_sc
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            ts = fast_score(cand)
            if ts < best_ts - 1e-13:
                best_ts = ts
                best_t = t
        
        if best_t == 0.0:
            no_imp += 1
            if no_imp > 10:
                break
            # Perturb
            s = s + 0.001 * rng.standard_normal(len(s))
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue
        
        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        cur_sc = fast_score(s)
        no_imp = 0
        
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = s.copy()
    
    return best, best_sc
