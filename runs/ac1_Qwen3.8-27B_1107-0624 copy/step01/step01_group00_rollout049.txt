def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.maximum(v - (v.sum() - target) / n, 0.0)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
