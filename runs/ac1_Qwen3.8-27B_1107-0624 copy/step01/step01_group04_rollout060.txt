"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences a.
Uses multi-start Adam optimization on the probability simplex with
adaptive softmax smoothing, active-set peak refinement, and warm-starting
from the provided height_sequence_1.

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(16, dtype=np.float64) / 16.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def greedy_normalized(n):
    n = int(max(n, 2))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def uniform_normalized(n):
    return np.ones(int(n), dtype=np.float64) / int(n)


def mix_greedy_uniform(n, lam):
    g = greedy_normalized(n)
    u = uniform_normalized(n)
    return normalize_nonneg(lam * g + (1.0 - lam) * u)


def score_normalized(p):
    """Score for a probability vector (sum=1): 2n * max(convolve(p,p))"""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 0:
        return np.inf
    a = a / s
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2*n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max())


def scale_output(p, target_sum=10.0):

def adam_optimize_v2(p, budget_s, alpha_init=20.0, alpha_end=80.0,
                     lr_init=0.04, lr_end=0.005, seed=0):
    """
    Adam optimizer on the log-probability simplex.
    p is a probability vector (sum=1, all >= 0).
    Returns (best_p, best_score).
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        p = greedy_normalized(16)
        n = 16
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 5.0:
        return p, score_normalized(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3*n - 2)

    best_p = p.copy()
    best_f = score_normalized(p)
    no_improve = 0
    best_streak_f = best_f

    start = time.monotonic()
    end = start + max(0.0, budget_s - 3.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Anneal temperature: start smooth, end sharp
        alpha = alpha_init + (alpha_end - alpha_init) * frac
        # Anneal learning rate: cosine schedule
        lr = lr_end + 0.5 * (lr_init - lr_end) * (1.0 + np.cos(np.pi * frac))

        # Compute convolution
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # If stuck, perturb and continue
        if no_improve > 500:
            x += 0.02 * np.random.default_rng(t).normal(0, 1, n)
            x -= x.mean()
            no_improve = 0

        # Compute softmax weights on convolution values
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2*n-1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Compute gradient
        E = float(np.dot(w, c))  # weighted average of convolution values
        W_rev = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W_rev, L)
        # Extract the relevant part: D_i = sum_k w_k * p_{k-i}
        # The full conv of w_rev (len 2n-1) and p (len n) has length 3n-2
        # We need D_i for i = 0,...,n-1
        # conv(w_rev, p)[m] = sum_j w_rev[j] * p[m-j]
        # w_rev[j] = w[2n-2-j]
        # D_i = sum_k w_k * p_{k-i}
        # Let me think... conv(w_rev, p) has length (2n-1)+n-1 = 3n-2
        # Element m = sum_j w_rev[j] * p[m-j] where 0<=j<2n-1, 0<=m-j<n
        # = sum_j w[2n-2-j] * p[m-j]
        # Let l = 2n-2-j: = sum_l w[l] * p[m - 2n + 2 + l]
        # We want D_i = sum_k w_k * p_{k-i}
        # Setting m - 2n + 2 = -i gives m = 2n - 2 - i
        # So D_i = conv(w_rev, p)[2n-2-i]
        
        D = Cfull[2*n-2 : 2*n-2+n-1][::-1].real  # indices 2n-2, 2n-3, ..., n-1 → reversed
        # Wait, let me re-derive.
        # conv(w_rev, p)[m] for m = n-1, ..., 2n-2
        # Actually, let me just use the same indexing as the original code:
        # Cfull[n-1 : 2n-1][::-1]
        
        # Let me just use the original formula:
        D = Cfull[n-1:2*n-1][::-1].real
        D = np.maximum(D, 0.0)  # p values are non-negative
        
        grad_p = (4.0 * n) * (D - E)
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu**t)
        vhat = v / (1.0 - nu**t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -100.0, 100.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = score_normalized(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def active_set_refine(p, budget_s, top_k=10):

    Active-set refinement: identify the top-k peaks in the convolution,
    then use subgradient steps to reduce them.
    """
    p = np.asarray(p, dtype=np.float64).ravel()

        return p, score_normalized(p)

def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None or p.size == 0:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s <= 1e-15:
            return
        a = a / s
        if score is None:
            score = score_normalized(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("[%.1fs] New best: %.8f (n=%d)" % (time.monotonic()-t0, best_score, a.size))
        n = int(a.size)
        if n not in best_by_n or score < best_by_n[n][0]:
            best_by_n[n] = (float(score), a.copy())

    # Phase 1: Quick baselines (50s)
    print("Phase 1: Baselines")
    base_lengths = [128, 192, 256, 320, 384, 448, 512]
    for n in base_lengths:
        if time.monotonic() >= deadline - 1700:
            break
        for p in [greedy_normalized(n), power_normalized(n, 0.45),
                  power_normalized(n, 0.55), mix_greedy_uniform(n, 0.5),
                  uniform_normalized(n)]:
            record(p)

    # Warm start from height_sequence_1
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)

        except Exception as e:
            print("  Could not load height_sequence_1: %s" % e)

    print("  Current best: %.8f" % best_score)

    # Phase 2: Adam on warm start (300s)
    print("\nPhase 2: Adam on warm start (n=%d)" % (best_p.size if best_p is not None else 0))
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(300.0, max(20.0, rem * 0.35))
        p_out, f_out = adam_optimize_v2(best_p.copy(), alloc, seed=int(seed))
        record(p_out, f_out)

    # Phase 3: Adam on best baseline (200s)
    print("\nPhase 3: Adam on best baseline")
    if best_p is not None:
        rem = deadline - time.monotonic()
        n_best = best_p.size
        # Use a slightly different starting point
        p_start = best_p.copy()
        noise = np.exp(rng.normal(0.0, 0.05, n_best))
        p_start = normalize_nonneg(p_start * noise)
        alloc = min(200.0, max(20.0, rem * 0.2))
        p_out, f_out = adam_optimize_v2(p_start, alloc,
                                         alpha_init=15.0, alpha_end=100.0,
                                         lr_init=0.03, lr_end=0.003,
                                         seed=int(seed)+1)
        record(p_out, f_out)

    # Phase 4: Active-set refinement (200s)
    print("\nPhase 4: Active-set refinement")
    if best_p is not None:
        rem = deadline - time.monotonic()

def active_set_refine(p, budget_s, top_k=8):
    """
    Active-set subgradient refinement: identify the top peaks in the
    convolution and use projected subgradient steps to reduce them.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        return p, score_normalized(p)
    budget_s = float(budget_s)
    if budget_s < 10.0:
        return p, score_normalized(p)

    p = normalize_nonneg(p)
    best_p = p.copy()
    best_f = score_normalized(p)

    if n <= 512:
        L = None
    else:
        L = next_power_of_two(2 * n - 1)

    step_size = 0.005
    no_improve = 0
    it = 0

    start = time.monotonic()
    end = start + max(0.0, budget_s - 5.0)

    while time.monotonic() < end:
        it += 1
        # Compute convolution
        if L is None:
            c = np.convolve(p, p)
        else:
            Pg = np.fft.rfft(p, L)
            c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        f = float(2.0 * n * maxc)

        if f < best_f - 1e-12:
            best_f = f
            best_p = p.copy()
            no_improve = 0
            step_size = min(0.02, step_size * 1.02)
        else:
            no_improve += 1
            step_size *= 0.95
            if step_size < 1e-6 or no_improve > 500:
                break

        # Find active peaks
        if c.size <= top_k:
            top_indices = np.arange(c.size)
        else:
            top_indices = np.argsort(c)[::-1][:top_k]

        # Compute subgradient: average of gradients of active peaks
        grad = np.zeros(n, dtype=np.float64)
        for k in top_indices:
            k = int(k)
            # dC_k/dp_i = 2 * p_{k-i} if 0 <= k-i < n, else 0
            lo_i = max(0, k - n + 1)
            hi_i = min(k, n - 1)
            if lo_i <= hi_i:
                # For i in [lo_i, hi_i]: grad_k[i] = 2 * p[k-i]
                # k-i ranges from k-hi_i to k-lo_i
                grad[lo_i:hi_i + 1] += 2.0 * p[k - hi_i:k - lo_i + 1][::-1]
        grad /= float(top_k)

        if not np.all(np.isfinite(grad)):
            break

        # Projected step: move opposite to gradient, maintain sum=1 and p>=0
        grad_centered = grad - np.mean(grad)
        p_new = p - step_size * grad_centered
        p_new = np.maximum(np.asarray(p_new, dtype=np.float64), 0.0)
        sn = float(p_new.sum())
        if sn <= 1e-15:
            break
        p_new = p_new / sn
        p = p_new

    return best_p, best_f

    L2 = next_power_of_two(2 * n - 1)

    # Start with a set of constraint rows
    max_rows = min(200, 2 * n - 1)
    rows = list(np.linspace(0, 2 * n - 2, max_rows, dtype=int))

    for _ in range(3):
        if budget_s <= 0:
            return None
        # Build constraint matrix
        A = np.zeros((len(rows), n), dtype=np.float64)
        for ii, k in enumerate(rows):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(len(rows), M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        tl = max(5.0, min(30.0, budget_s * 0.3))
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl), "presolve": True}
            )
        except Exception:
            return None

        if res is None or not getattr(res, "success", False) or res.x is None:
            return None

        g = np.clip(res.x, 0.0, None)
        if g.sum() <= 1e-15:
            return None

        # Check for violations
        Pg = np.fft.rfft(g, L2)
        Pf = np.fft.rfft(f, L2)
        conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        violations = conv - M
        if np.max(violations) < 1e-6 * M:
            return g

        # Add violated rows
        if len(rows) < 500:
            viol_idx = np.where(violations > 1e-6 * M)[0]
            row_set = set(rows)
            for vi in viol_idx[:100]:
                if vi not in row_set:
                    rows.append(int(vi))
                    row_set.add(vi)
            budget_s -= tl
    return None


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s <= 1e-15:
            return
        a = a / s
        if score is None:
            score = score_normalized(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("[%.1fs] New best: score=%.8f (n=%d)" %
                  (time.monotonic() - t0, best_score, a.size))
        n_val = int(a.size)
        if n_val not in best_by_n or score < best_by_n[n_val][0]:
            best_by_n[n_val] = (float(score), a.copy())

    # Phase 1: Quick baselines
    print("=== Phase 1: Baselines ===")
    base_lengths = [64, 128, 192, 256, 320, 384, 448, 512, 768]
    for n in base_lengths:
        if time.monotonic() >= deadline - 1700:
            break
        for p in [greedy_normalized(n), power_normalized(n, 0.4),
                  power_normalized(n, 0.5), power_normalized(n, 0.6),
                  mix_greedy_uniform(n, 0.3), mix_greedy_uniform(n, 0.6),
                  uniform_normalized(n)]:
            record(p)

    # Warm start from height_sequence_1
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 1 and arr.sum() > 0:
                record(arr)
                print("  height_sequence_1: n=%d, score=%.8f" %
                      (arr.size, score_normalized(normalize_nonneg(arr))))
        except Exception as e:
            print("  Could not load height_sequence_1: %s" % e)

    print("  Best after phase 1: %.8f" % best_score)

    # Phase 2: Adam on warm start (most important - give it the most time)
    print("\n=== Phase 2: Adam on warm start ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(400.0, max(30.0, rem * 0.35))
        print("  Allocating %.0fs for n=%d" % (alloc, best_p.size))
        p_out, f_out = adam_optimize(best_p.copy(), alloc,
                                     alpha_init=15.0, alpha_end=120.0,
                                     lr_init=0.05, lr_end=0.002,
                                     seed=int(seed))
        record(p_out, f_out)

    # Phase 3: Adam with perturbed restart
    print("\n=== Phase 3: Perturbed restart ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(200.0, max(20.0, rem * 0.2))
        noise = np.exp(rng.normal(0.0, 0.08, best_p.size))
        p_start = normalize_nonneg(best_p * noise)
        p_out, f_out = adam_optimize(p_start, alloc,
                                     alpha_init=10.0, alpha_end=150.0,
                                     lr_init=0.06, lr_end=0.003,
                                     seed=int(seed) + 1)
        record(p_out, f_out)

    # Phase 4: Active-set refinement
    print("\n=== Phase 4: Active-set refinement ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(180.0, max(20.0, rem * 0.15))
        p_out, f_out = active_set_refine(best_p.copy(), alloc, top_k=6)
        record(p_out, f_out)

    # Phase 5: Adam with sharp temperature (focus on true max)
    print("\n=== Phase 5: Sharp Adam ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(200.0, max(20.0, rem * 0.18))
        p_out, f_out = adam_optimize(best_p.copy(), alloc,
                                     alpha_init=50.0, alpha_end=300.0,
                                     lr_init=0.03, lr_end=0.001,
                                     seed=int(seed) + 2)
        record(p_out, f_out)

    # Phase 6: Multi-length exploration
    print("\n=== Phase 6: Multi-length exploration ===")
    for n_try in [320, 352, 384, 416, 448, 480, 512]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        # Find best nearby length to start from
        start_p = None
        for dn in [0, -32, 32, -64, 64, -96, 96]:
            nn = n_try + dn
            if nn in best_by_n and nn > 0:
                sp = best_by_n[nn][1].copy()
                if sp.size != n_try:
                    # Resample
                    old_x = np.linspace(0, 1, sp.size)
                    new_x = np.linspace(0, 1, n_try)
                    sp = np.interp(new_x, old_x, sp)
                start_p = normalize_nonneg(sp)
                break
        if start_p is None:
            start_p = greedy_normalized(n_try)
        alloc = min(60.0, max(15.0, rem * 0.08))
        p_out, f_out = adam_optimize(start_p, alloc,
                                     alpha_init=10.0, alpha_end=80.0,
                                     lr_init=0.04, lr_end=0.004,
                                     seed=int(seed) + n_try)
        record(p_out, f_out)

    # Phase 7: Long Adam with very low LR (polish)
    print("\n=== Phase 7: Long polish Adam ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        alloc = min(250.0, max(30.0, rem * 0.25))
        p_out, f_out = adam_optimize(best_p.copy(), alloc,
                                     alpha_init=80.0, alpha_end=400.0,
                                     lr_init=0.015, lr_end=0.0005,
                                     seed=int(seed) + 100)
        record(p_out, f_out)

    # Phase 8: Another active-set pass
    print("\n=== Phase 8: Final active-set ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            alloc = min(120.0, max(15.0, rem * 0.15))
            p_out, f_out = active_set_refine(best_p.copy(), alloc, top_k=4)
            record(p_out, f_out)

    # Phase 9: One more Adam restart from slightly perturbed best
    print("\n=== Phase 9: Final Adam restart ===")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            alloc = min(150.0, max(20.0, rem * 0.2))
            noise = np.exp(rng.normal(0.0, 0.03, best_p.size))
            p_start = normalize_nonneg(best_p * noise)
            p_out, f_out = adam_optimize(p_start, alloc,
                                         alpha_init=30.0, alpha_end=200.0,
                                         lr_init=0.02, lr_end=0.001,
                                         seed=int(seed) + 200)
            record(p_out, f_out)

    # Final validation
    print("\n=== Final validation ===")
    if best_p is not None:
        print("  Internal best score: %.8f (n=%d)" % (best_score, best_p.size))
        candidates_to_check = [best_p]
        # Also check a few other top candidates
        items = sorted(best_by_n.items(), key=lambda x: x[1][0])
        for n_val, (sc, arr) in items[:3]:
            if arr is not None and not np.allclose(arr, best_p):
                candidates_to_check.append(arr)

        for cand in candidates_to_check:
            if deadline - time.monotonic() < 10:
                break
            seq = scale_output(cand, 10.0)
            gscore = try_global_eval(seq)
            if np.isfinite(gscore):
                print("  Candidate (n=%d): global score = %.8f" %
                      (cand.size, gscore))

    if best_p is None:
        best_p = greedy_normalized(384)

    # Final slight refinement: try a few target sums for the output
    best_output_score = np.inf
    best_output_seq = scale_output(best_p, 10.0)
    for ts in [5.0, 10.0, 20.0, 50.0]:
        seq = scale_output(best_p, ts)
        sc = try_global_eval(seq)
        if np.isfinite(sc) and sc < best_output_score:
            best_output_score = sc
            best_output_seq = seq

    print("\nFinal answer: score=%.8f, length=%d" %
          (best_output_score, len(best_output_seq)))
    return best_output_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the score."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % e)
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
