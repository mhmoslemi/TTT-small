```python
        grad = 2 * shift_correlation(p, a)  # factor of 2 because of symmetry

        a_new = a - lr * grad
        a_new = project(a_new)
        a = a_new
    return a
```

```python
import numpy as np
```

```python
def _normalize(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    if a.size == 0:
        return a
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def _conv(a):
    return np.convolve(a, a, mode='full')


def _score_normalized(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size < 2:
        return np.inf
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    c = _conv(a)
    return float(2.0 * a.size * c.max())


def _shift_corr_fft(p, a):
    """Compute grad_i = sum_k p_k * (a_{i+k} + a_{i-k}) using FFT."""
    n = a.size
    m = p.size
    size = 1
    while size < 2 * n:
        size <<= 1

    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)

    s1 = np.fft.irfft(fa * np.conj(fp), size)[:n]
    # S2 = (p * a)_i = sum_k p_k * a_{i-k} (linear, zero-padded)
    s2 = np.fft.irfft(fp * fa, size)[:n]

    return s1 + s2


def _smooth_gradient(a, beta):
    """Compute the gradient of log(sum exp(beta * conv(a,a))) w.r.t. a."""
    n = a.size
    c = _conv(a)
    cm = c - c.max()
    e = np.exp(beta * cm)
    p = e / e.sum()
    grad = _shift_corr_fft(p, a)
    return grad, c, p


def _line_search(a, grad, c_max_current, max_iter=30):
    """Backtracking line search to find a step that reduces the score."""
    n = a.size
    best_a = a.copy()
    best_score = 2.0 * n * c_max_current
    lr = 1.0 / (n * 0.01)
    cur_score = best_score
    for _ in range(max_iter):
        a_new = a - lr * grad
        a_new = _normalize(a_new)
        c_new = _conv(a_new)
        m_new = float(c_new.max())
        sc_new = 2.0 * n * m_new
        if sc_new < best_score:
            best_a = a_new.copy()
            best_score = sc_new
            return best_a, best_score, lr
        lr *= 0.7
    return best_a, best_score, lr


def _optimize_smooth(a, time_limit, beta_start=30.0, beta_end=300.0,
                     momentum_start=0.7, momentum_end=0.9, seed=42):

    Main smooth gradient optimization with Nesterov momentum and adaptive beta.

    rng = np.random.default_rng(seed)
    a = _normalize(a.copy())
    n = a.size
    c = _conv(a)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c
    best_a = a.copy()

    v = np.zeros_like(a)  # momentum buffer

    stuck = 0

        progress = min(1.0, it / max(1, int(time_limit * 3)))
        beta = beta_start + (beta_end - beta_start) * progress

        max_c = float(c.max())

def _upsample(a, target_n):
    """Upsample a to target_n using cubic-like interpolation."""
    n = a.size
    if n == target_n:
        return a.copy()
    if n < 2:
        return np.full(target_n, 1.0 / target_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(x_new, x_old, a)
    return _normalize(a_interp)


def _downsample(a, target_n):
    """Downsample a to target_n by binning."""
    n = a.size
    if n == target_n:
        return a.copy()
    out = np.zeros(target_n)
    for i in range(target_n):
        lo = int(i * n / target_n)
        hi = int((i + 1) * n / target_n)
        out[i] = a[lo:hi].sum()
    return _normalize(out)


def _two_block_init(n, rng):
    """Two-block initialization with random block boundary."""
    a = np.zeros(n)
    split = int(n * rng.uniform(0.35, 0.65))
    p1 = rng.uniform(0.4, 0.7)
    a[:split] = p1 / split
    a[split:] = (1 - p1) / (n - split)
    return _normalize(a)


def _smooth_random_init(n, rng):
    """Smooth random initialization using low-frequency DFT components."""
    x = np.linspace(0, 1, n)
    a = 1.0

        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    return _normalize(a)

        chunk_time = remaining / (n_iter + 1)

        for phase in range(n_iter):
            if time.monotonic() > hard_deadline - 30:
                break
            beta_lo = 50.0 * (phase + 1)
            beta_hi = 200.0 * (phase + 1) + 50
            t_chunk = min(chunk_time, hard_deadline - time.monotonic() - 20)
            if t_chunk < 10:
                break

            print(f"  Phase 2.{phase}: beta=[{beta_lo},{beta_hi}], t={t_chunk:.0f}s, "
                  f"current best={best_score:.10f}", flush=True)

            a_opt, sc = _optimize_smooth(best_seq, t_chunk,
                                         beta_start=beta_lo, beta_end=beta_hi,
                                         momentum_start=0.7, momentum_end=0.92,
                                         seed=seed + phase)
            update_best(a_opt, sc)

            # Try a slightly perturbed version
            if time.monotonic() < hard_deadline - 60:
                a_pert = _normalize(best_seq * np.exp(
                    rng.normal(0, 0.05, size=best_seq.size)))
                t_pert = min(30.0, (hard_deadline - time.monotonic()) * 0.1)
                if t_pert > 5:
                    a_opt2, sc2 = _optimize_smooth(a_pert, t_pert,
                                                   beta_start=beta_lo, beta_end=beta_hi,
                                                   momentum_start=0.7, momentum_end=0.9,
                                                   seed=seed + 100 + phase)
                    update_best(a_opt2, sc2)

    # Phase 3: Try upscaling to longer sequences
    remaining = hard_deadline - time.monotonic()
    if remaining > 120 and best_seq is not None:
        print(f"Phase 3: upscaling, remaining={remaining:.0f}s", flush=True)

        current_n = best_seq.size
        target_lengths = []
        if current_n <= 256:
            target_lengths = [512, 1024]
        elif current_n <= 512:
            target_lengths = [1024, 2048]
        elif current_n <= 1024:
            target_lengths = [2048, 4096]
        else:
            target_lengths = []

        for target_n in target_lengths:
            if time.monotonic() > hard_deadline - 80:
                break
            a_up = _upsample(best_seq, target_n)
            t_up = min(150.0, (hard_deadline - time.monotonic()) * 0.3)
            if t_up < 20:
                break
            print(f"  Upscale to n={target_n}, t={t_up:.0f}s", flush=True)
            a_opt, sc = _optimize_smooth(a_up, t_up,
                                         beta_start=30, beta_end=200,
                                         momentum_start=0.7, momentum_end=0.9,
                                         seed=seed + 200)
            update_best(a_opt, sc)

    # Phase 4: Final polish - concentrate on the very best
    remaining = hard_deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"Phase 4: final polish, t={remaining:.0f}s", flush=True)
        a_opt, sc = _optimize_smooth(best_seq, remaining - 10,
                                     beta_start=100, beta_end=500,
                                     momentum_start=0.8, momentum_end=0.95,
                                     seed=seed + 500)
        update_best(a_opt, sc)

    # Final: try a few more perturbations on the best
    if best_seq is not None and time.monotonic() < hard_deadline - 10:
        for pert_trial in range(5):
            if time.monotonic() > hard_deadline - 10:
                break
            scale = 0.03 * (5 - pert_trial) / 5
            a_p = _normalize(best_seq * np.exp(
                rng.normal(0, scale, size=best_seq.size)))
            t_small = min(20.0, hard_deadline - time.monotonic() - 5)
            if t_small < 3:
                break
            a_opt, sc = _optimize_smooth(a_p, t_small,
                                         beta_start=100, beta_end=400,
                                         momentum_start=0.8, momentum_end=0.95,
                                         seed=seed + 600 + pert_trial)
            update_best(a_opt, sc)

    if best_seq is None:
        # Fallback: two-element optimal
        return [1.0 / 3.0, 2.0 / 3.0]

    cand = [float(x) for x in best_seq]

    try:
        official = evaluate_sequence(cand)
        if np.isfinite(official):
            print(f"FINAL: official_score={official:.10f}, n={len(cand)}", flush=True)
        else:
            print(f"WARNING: official_score={official}", flush=True)
    except Exception as ex:
        print(f"WARNING: could not evaluate: {ex}", flush=True)
        print(f"Internal score: {best_score:.10f}, n={len(cand)}", flush=True)

    return cand
```
