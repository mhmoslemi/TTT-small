```python
import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

N = 4096
# Precompute FFT sizes
FFT_SELF = 1 << (2*N - 2).bit_length()  # for self-convolution
FFT_CROSS = 1 << (2*N - 2).bit_length()  # for cross-convolution (same size since both have length N)

def self_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    fa = np.fft.rfft(a, FFT_SELF)
    c = np.fft.irfft(fa * fa, FFT_SELF).real
    return c[:2*N-1]

def cross_conv(a, b):
    """Compute full linear convolution of a with b using FFT."""
    fa = np.fft.rfft(a, FFT_CROSS)
    fb = np.fft.rfft(b, FFT_CROSS)
    c = np.fft.irfft(fa * fb, FFT_CROSS).real
    return c[:2*N-1]

def build_convolution_matrix(a):
    """Build sparse matrix A where A[k,i] = a[i] if 0 <= k-i <= N-1."""
    # A is (2N-1) x N
    rows = []
    cols = []
    data = []
    for i in range(N):
        if a[i] == 0:
            continue
        # a[i] contributes to A[k,i] for k = i, i+1, ..., i+N-1
        # which is k from i to i+N-1
        k_start = i
        k_end = i + N - 1
        rows.extend(range(k_start, k_end + 1))
        cols.extend([i] * N)
        data.extend([a[i]] * N)
    A = csr_matrix((data, (rows, cols)), shape=(2*N-1, N))
    return A

def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    A = build_convolution_matrix(a)
    c = -np.ones(N)  # minimize -sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(2*N-1, M), bounds=[(0, None)]*N, method='highs')
    if res.status == 0:
        return res.x
    return None

def evaluate_obj(a):
    """Compute 2*N*max(conv) for sequence a (sum=1 normalization)."""
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = self_conv(a)
    return 2.0 * N * np.max(c) / (s*s)
```

"""
Minimize 2*n*max_k (a*a)[k] / (sum a)^2 using LP-based direction finding
and smooth refinement. The LP finds directions that keep the cross-convolution
bounded, enabling monotone decrease of the objective.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

N = 4096
FFT_SIZE = 1 << (2 * N - 2).bit_length()

# Precompute sparsity pattern for convolution matrix
def _precompute_pattern():
    # Column i has non-zeros in rows i to i+N-1
    cols = np.repeat(np.arange(N), N)
    rows_parts = [np.arange(i, i + N) for i in range(N)]
    rows = np.concatenate(rows_parts)
    return rows, cols

_ROWS, _COLS = _precompute_pattern()

def self_conv(a):
    fa = np.fft.rfft(a, FFT_SIZE)
    c = np.fft.irfft(fa * fa, FFT_SIZE).real
    return c[:2*N - 1]

def cross_conv(a, b):
    fa = np.fft.rfft(a, FFT_SIZE)
    fb = np.fft.rfft(b, FFT_SIZE)
    c = np.fft.irfft(fa * fb, FFT_SIZE).real
    return c[:2*N - 1]

def compute_obj(a):
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = self_conv(a)
    return float(2.0 * N * np.max(c) / (s * s))

def lp_direction(a, M, timeout=30.0):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    data = np.repeat(a, N)  # each column i has value a[i] repeated N times
    A = csr_matrix((data, (_ROWS, _COLS)), shape=(2*N - 1, N))
    c = -np.ones(N)
    b_ub = np.full(2*N - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * N, 
                      method='highs', options={'time_limit': timeout, 'presolve': True})
        if res.status == 0 and np.isfinite(res.fun):
            b = res.x
            if np.sum(b) > 0:
                return b
    except Exception as e:
        print(f"LP error: {e}")
    return None

def line_search(a, b, max_evals=20):
    """Find best t in [0,1] for a_new = (1-t)*a + t*b."""
    def obj_at_t(t):
        a_new = (1 - t) * a + t * b
        s = np.sum(a_new)
        if s < 0.01:
            return np.inf
        c = self_conv(a_new)
        return float(np.max(c))
    
    # Try a grid
    best_t = 0.0
    best_obj = obj_at_t(0.0)
    for t in np.linspace(0.1, 1.0, max_evals):
        o = obj_at_t(t)
        if o < best_obj:
            best_obj = o
            best_t = t
    return best_t, best_obj

def lp_iterate(a, num_iters, deadline):
    """Run LP-based iterations to minimize the objective."""
    best_a = a.copy()
    best_score = compute_obj(a)
    
    for it in range(num_iters):
        if time.monotonic() > deadline:
            break
        
        s = np.sum(a)
        a_norm = a / s  # normalize to sum=1
        c = self_conv(a_norm)
        M = float(np.max(c))
        
        # Solve LP
        b = lp_direction(a_norm, M, timeout=20.0)
        if b is None:
            # Fallback: use subgradient direction
            break
        
        b_norm = b / np.sum(b)
        
        # Line search
        t, obj_val = line_search(a_norm, b_norm, max_evals=15)
        
        if t > 0.01:
            a_new = (1 - t) * a_norm + t * b_norm
            score_new = 2.0 * N * float(np.max(self_conv(a_new)))
            if score_new < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = score_new
                a = a_new
        
        if it % 10 == 0:
            print(f"  LP iter {it}: best={best_score:.6f}, M={M:.6f}, t={t:.3f}")
    
    return best_a, best_score
