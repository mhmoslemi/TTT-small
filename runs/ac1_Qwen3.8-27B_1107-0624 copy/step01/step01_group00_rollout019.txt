def _fft_conv_sq(a, m):
    """Compute linear convolution a*a via FFT."""

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)

def _subgradient(a, c, top_k, tau):
    """Compute smoothed subgradient of max(c) w.r.t. a."""
    n = len(a)
    k_eff = min(top_k, n)
    # Get top-k lags
    if k_eff == n:
        order = np.arange(n)  # all of them... wait, c has 2n-1 elements
    # c has length 2n-1
    c_len = len(c)
    k_eff = min(top_k, c_len)
    if k_eff == c_len:
        idx = np.arange(c_len)
        weights = np.ones(c_len) / c_len
    else:
        # Use softmax over top-k
        order = np.argpartition(-c, k_eff)[:k_eff]
        peak_val = c[order].max()
        # Softmax with temperature
        logits = (c[order] - peak_val) / max(tau, 1e-10)
        weights = np.exp(logits)
        weights /= weights.sum()
    
    # Compute gradient: grad[i] = 2 * sum_k weights[k_idx] * a[k_idx - i] (with bounds)
    grad = np.zeros(n)
    for k_idx, w in zip(order, weights):
        if w < 1e-10:
            continue
        # k_idx ranges from 0 to 2n-2
        # a[k_idx - i] is valid when 0 <= k_idx - i < n, i.e., k_idx-n+1 <= i <= k_idx
        lo = max(0, int(k_idx) - n + 1)
        hi = min(int(k_idx), n - 1)
        if lo > hi:
            continue
        i_range = np.arange(lo, hi + 1)
        src = int(k_idx) - i_range
        grad[lo:hi+1] += 2.0 * w * a[src]
    return grad

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequence a.
Uses three complementary methods: (1) LogSumExp-smoothed subgradient with
multiplicative updates, (2) Gerchberg-Saxton style peak shaving, and
(3) exact subgradient with Euclidean projection. Results are time-shared
and the best candidate is returned.
"""

import time
import math
import numpy as np


def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_sq(a, m):
    """Linear convolution a*a via FFT, length 2n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * len(a) - 1]


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} via Euclidean projection."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _objective_fast(a, m):
    """Compute max(c) where c = convolve(a,a), assuming sum(a)=1."""
    c = _conv_sq(a, m)
    return float(c.max())


def _compute_grad_single_peak(a, c, k_peak):
    """Exact subgradient of c_k_peak w.r.t. a. Returns vector of length n."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, int(k_peak) - n + 1)
    hi = min(int(k_peak), n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        src = int(k_peak) - idx
        g[lo:hi + 1] = 2.0 * a[src]
    return g


def _compute_grad_lse(a, c, tau):
    """Gradient of tau*log(sum(exp(c/tau))) w.r.t. a, via softmax weights."""
    n = len(a)
    c_len = len(c)
    # Softmax weights
    mc = c.max()
    logits = (c - mc) / max(tau, 1e-12)
    exp_logits = np.exp(logits)
    w = exp_logits / exp_logits.sum()
    # Prune small weights
    threshold = 1e-8
    mask = w > threshold
    if not np.any(mask):
        return np.zeros(n)
    active_idx = np.where(mask)[0]
    active_w = w[active_idx]
    g = np.zeros(n)
    for k_idx, wk in zip(active_idx, active_w):
        ki = int(k_idx)
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        src = ki - idx
        g[lo:hi + 1] += 2.0 * wk * a[src]
    return g


def _optimize_lse(a, time_budget, tau0=0.005, lr0=0.03, top_ignore=None):
    """LogSumExp-smoothed subgradient with multiplicative updates."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    a = _normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = _objective_fast(a, m)
    lr = lr0
    tau = tau0
    last_best = time.monotonic()
    stall = 0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 0.5:
        c = _conv_sq(a, m)
        obj = float(c.max())
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_best = time.monotonic()
            stall = 0
        else:
            stall += 1
        
        # Anneal temperature
        if stall > 500:
            tau = max(0.001, tau * 0.7)
            lr = max(0.001, lr * 0.85)
            stall = 0
        
        # Compute gradient
        g = _compute_grad_lse(a, c, tau)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Multiplicative update (mirror descent)
        scale = lr * g / max(g_norm, 1e-10)
        scale = np.clip(scale, -1.0, 1.0)
        a_new = a * np.exp(-scale)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a, best_obj


def _optimize_subgrad_eucl(a, time_budget, lr0=0.01):
    """Exact subgradient (top peak) with Euclidean projection."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    a = _normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = _objective_fast(a, m)
    lr = lr0
    start = time.monotonic()
    stall = 0

        c = _conv_sq(a, m)
        obj = float(c.max())
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr = max(0.0005, lr * 0.9)
                stall = 0
        
        k_peak = int(np.argmax(c))
        g = _compute_grad_single_peak(a, c, k_peak)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        a_new = a - lr * g / g_norm
        a_new = _proj_simplex(a_new, 1.0)
        a = a_new
    
    return best_a, best_obj


def _optimize_gs(a, time_budget, clip_frac=0.995, blend=0.3):
    """Gerchberg-Saxton style: clip autocorrelation peaks, re-factor, project."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    a = _normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = _objective_fast(a, m)
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget - 0.5:
        # Current autocorrelation
        A = np.fft.fft(a, m)
        C = A * A
        c = np.fft.ifft(C).real[:2 * n - 1]
        c = np.maximum(c, 0)  # numerical fix
        obj = float(c.max())
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Clip peaks
        thresh = obj * clip_frac
        c_clipped = np.minimum(c, thresh)
        
        # Pad and FFT
        c_padded = np.zeros(m)
        c_padded[:len(c_clipped)] = c_clipped
        R_new = np.fft.fft(c_padded)
        # Take magnitude for new amplitude (handle potential negative values)
        R_new = np.maximum(np.real(R_new), 0.0)
        new_amp = np.sqrt(R_new)
        
        # Blend amplitudes
        A = np.fft.fft(a, m)
        old_amp = np.abs(A)
        phase = np.angle(A)
        blended_amp = (1 - blend) * old_amp + blend * new_amp
        A_new = blended_amp * np.exp(1j * phase)
        
        # Inverse FFT and project
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        a = a_new
        
        # Gradually increase clip aggression
        clip_frac = min(0.9999, clip_frac * 1.0005)
    
    return best_a, best_obj


def _optimize_topk(a, time_budget, k=5, tau=0.01, lr0=0.05):
    """Top-k peaks subgradient with multiplicative updates."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    a = _normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = _objective_fast(a, m)
    lr = lr0
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < time_budget - 0.5:
        c = _conv_sq(a, m)
        obj = float(c.max())
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                lr = max(0.002, lr * 0.85)
                tau = max(0.002, tau * 0.7)
                stall = 0
        
        # Top-k peaks with softmax
        k_eff = min(k, len(c))
        order = np.argpartition(-c, k_eff - 1)[:k_eff]
        peak_vals = c[order]
        logits = (peak_vals - peak_vals.max()) / max(tau, 1e-10)
        w = np.exp(logits)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        for ki, wi in zip(order, w):
            ki = int(ki)
            lo = max(0, ki - n + 1)
            hi = min(ki, n - 1)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            src = ki - idx
            g[lo:hi + 1] += 2.0 * wi * a[src]
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Multiplicative step
        scale = lr * g / g_norm
        scale = np.clip(scale, -0.8, 0.8)
        a_new = a * np.exp(-scale)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        a = a_new
    
    return best_a, best_obj


def _make_seeds(base, n):
    """Generate diverse starting seeds."""
    seeds = [base.copy()]
    # Reverse

    # Sine-modulated

    # Triangular

    seeds.append(tri.astype(np.float64))
    # Gaussian

    sigma = n * 0.25
    seeds.append(np.exp(-0.5 * (np.arange(n) - mu) ** 2 / sigma ** 2) + 0.01)
    # Power-law
    seeds.append(1.0 / (np.arange(n) + 1) ** 0.5)
    # Two bumps

    b2 = np.exp(-0.5 * ((np.arange(n) - n * 0.7) / (n * 0.15)) ** 2)
    seeds.append(b1 + b2 + 0.01)
    # Perlin-like: sum of sines

    sig = np.sin(2 * np.pi * 1 * t) + 0.5 * np.sin(2 * np.pi * 2 * t + 0.5) + 0.3 * np.sin(2 * np.pi * 3 * t + 1.0)
