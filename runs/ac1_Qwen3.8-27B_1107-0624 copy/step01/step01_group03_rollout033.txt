import numpy as np
from scipy.optimize import linprog
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2"""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def softmax_alpha(c, alpha):
    """Compute softmax with temperature alpha, numerically stable."""
    c_shifted = alpha * c - alpha * np.max(c)
    e = np.exp(c_shifted)
    return e / np.sum(e)

def compute_grad_s(s, alpha):
    """Compute gradient of softmax approximation of max(conv(s,s)) w.r.t. s."""
    n = len(s)
    c = np.convolve(s, s)
    w = softmax_alpha(c, alpha)
    # dG/ds[i] = (2/alpha) * sum_m w[m+i] * s[m]
    # = (2/alpha) * conv(w, s[::-1])[n-1 : 2n-1]
    r = s[::-1]
    conv_wr = np.convolve(w, r)
    grad_s = (2.0 / alpha) * conv_wr[n-1 : 2*n-1]
    return grad_s

def compute_grad_t(t, alpha):
    """Compute gradient w.r.t. log-parameter t where s = softmax(t)."""
    # s = exp(t) / sum(exp(t))
    t_shifted = t - np.max(t)
    e = np.exp(t_shifted)
    s = e / np.sum(e)
    grad_s = compute_grad_s(s, alpha)
    # grad_t[j] = s[j] * (grad_s[j] - dot(grad_s, s))
    dot_gs_s = np.dot(grad_s, s)
    grad_t = s * (grad_s - dot_gs_s)
    return grad_t, s

class AdamOptimizer:
    def __init__(self, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
        self.m = None
        self.v = None
        self.t = 0
    
    def step(self, params, grad):
        self.t += 1
        if self.m is None:
            self.m = np.zeros_like(params)
            self.v = np.zeros_like(params)
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad * grad
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return params - self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def gradient_descent(t_init, alpha_init, alpha_final, n_steps, deadline, lr=0.001):
    """Run gradient descent with annealing alpha."""
    n = len(t_init)
    opt = AdamOptimizer(lr=lr)
    t = t_init.copy()
    best_score = float('inf')
    best_t = t.copy()
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Anneal alpha
        frac = step / max(n_steps - 1, 1)
        alpha = alpha_init + (alpha_final - alpha_init) * frac
        
        t, s = compute_grad_t(t, alpha)
        
        # Evaluate actual score
        score = fast_score(s)
        if score < best_score:
            best_score = score
            best_t = t.copy()
        
        grad_t, _ = compute_grad_t(t, alpha)
        t = opt.step(t, grad_t)
        
    return best_t, best_score

import numpy as np
from scipy.optimize import linprog
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def get_s_from_t(t):
    """Convert log-parameter to probability vector."""
    t_shifted = t - np.max(t)
    e = np.exp(t_shifted)
    return e / np.sum(e)

def compute_grad_s(s, alpha):
    """Compute gradient of softmax-approx max(conv(s,s)) w.r.t. s."""
    n = len(s)
    c = np.convolve(s, s)
    c_offset = np.max(c)
    w = np.exp(alpha * (c - c_offset))
    w /= np.sum(w)
    r = s[::-1]
    conv_wr = np.convolve(w, r)
    grad_s = 2.0 * conv_wr[n-1:2*n-1]
    return grad_s

def adam_step(params, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Perform one Adam step. Returns updated params, m, v."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params_new = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params_new, m, v

def gradient_search(t_init, deadline, alpha_start=1.0, alpha_end=100.0, 
                    lr=0.05, eval_interval=50):
    """Run Adam-based gradient descent with alpha annealing."""
    n = len(t_init)
    t = t_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    step_count = 0
    best_score = float('inf')
    best_t = t.copy()
    
    while time.monotonic() < deadline:
        step_count += 1
        frac = min(step_count / 10000.0, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        s = get_s_from_t(t)
        grad_s = compute_grad_s(s, alpha)
        
        # Gradient w.r.t. t
        dot_gs_s = np.dot(grad_s, s)
        grad_t = s * (grad_s - dot_gs_s)
        
        t, m, v = adam_step(t, grad_t, m, v, step_count, lr)
        t -= np.mean(t)
        
        # Periodic evaluation
        if step_count % eval_interval == 0:
            s_check = get_s_from_t(t)
            score = fast_score(s_check)
            if score < best_score:
                best_score = score
                best_t = t.copy()
    
    # Final evaluation
    s_final = get_s_from_t(t)
    score_final = fast_score(s_final)
    if score_final < best_score:
        best_score = score_final
        best_t = t.copy()
    
    return best_t, best_score

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

    best = s.copy()
    best_sc = fast_score(s)
    rng = np.random.default_rng(123)
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        M = np.max(conv)
        
        b = solve_lp_direction(s, M, time_limit=3.0)
        if b is None or np.sum(b) < 1e-14:
            break
        
        b = b / b.sum()
        
        best_t = 0.0
        best_ts = best_sc
        for t_val in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            cand = np.maximum(cand, 0.0)
            ts = fast_score(cand)
            if ts < best_ts - 1e-12:

                best_t = t_val

            continue

        s = s / s.sum()
        no_imp = 0

            best = s.copy()
    
    return best, best_sc

Gradient-based search using softmax-approximation of max-autocorrelation,
with log-parameterization for non-negativity and normalization.
Uses Adam optimizer with alpha annealing. Multiple sequence lengths and

import numpy as np
from scipy.optimize import linprog
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 without external calls."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def get_s_from_t(t):
    """Convert log-parameter to probability vector (softmax)."""
    t_shifted = t - np.max(t)
    e = np.exp(t_shifted)
    return e / np.sum(e)


def compute_grad_s(s, alpha):
    """Compute gradient of (1/alpha)*log(sum_k exp(alpha*(s*s)[k])) w.r.t. s."""
    n = len(s)
    c = np.convolve(s, s)
    c_max = np.max(c)
    w = np.exp(alpha * (c - c_max))
    w_sum = np.sum(w)
    w = w / w_sum
    r = s[::-1]
    conv_wr = np.convolve(w, r)
    grad_s = 2.0 * conv_wr[n - 1:2 * n - 1]
    return grad_s


def run_gradient_descent(t_init, deadline, alpha_start, alpha_end, lr, eval_interval=200):
    """Adam-based gradient descent with alpha annealing."""
    n = len(t_init)
    t = t_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    best_score = float('inf')
    best_t = t.copy()
    
    while time.monotonic() < deadline:
        step_count += 1
        frac = min(step_count / 5000.0, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        # Decay learning rate
        cur_lr = lr * (0.5 ** (step_count / 10000.0))
        
        s = get_s_from_t(t)
        grad_s = compute_grad_s(s, alpha)
        
        # Gradient w.r.t. t
        dot_gs_s = np.dot(grad_s, s)
        grad_t = s * (grad_s - dot_gs_s)

        v = beta2 * v + (1 - beta2) * grad_t * grad_t

        t = t - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        t -= np.mean(t)

                best_score = score

    # Final evaluation
    s_final = get_s_from_t(t)
    score_final = fast_score(s_final)
    if score_final < best_score:
        best_score = score_final
        best_t = t.copy()
    
    return best_t, best_score


def build_conv_matrix(s):
    """Build (2n-1) x n matrix for LP direction finding."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_polish(s, deadline, max_iters=80):
    """LP-guided local search to polish a sequence."""
    n = len(s)
    best = s.copy()
    best_sc = fast_score(s)
    rng = np.random.default_rng(42)
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = np.max(conv)
        
        b = solve_lp_direction(s, M, time_limit=2.0)
        if b is None or np.sum(b) < 1e-14:
            break
        
        b = b / b.sum()
        
        best_t_val = 0.0
        best_ts = best_sc
        for t_val in [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - t_val) * s + t_val * b
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                continue
            ts = fast_score(cand)
            if ts < best_ts - 1e-12:
                best_ts = ts
                best_t_val = t_val

            no_imp += 1
            if no_imp > 12:

            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue
        
        s = (1.0 - best_t_val) * s + best_t_val * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        no_imp = 0
        
        if best_ts < best_sc:
            best_sc = best_ts
            best = s.copy()
    
    return best, best_sc

        step_count += 1
        frac = min(step_count / 8000.0, 1.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        cur_lr = lr * (1.0 / (1.0 + 0.01 * step_count / 1000.0))

        s = get_s_from_t(t)
        grad_s = compute_grad_s(s, alpha)

        # Gradient w.r.t. t (softmax Jacobian)
        dot_gs_s = np.dot(grad_s, s)
        grad_t = s * (grad_s - dot_gs_s)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t * grad_t
        m_hat = m / (1 - beta1 ** step_count)
        v_hat = v / (1 - beta2 ** step_count)
        t = t - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        t -= np.mean(t)

        # Periodic evaluation
        if step_count % eval_interval == 0:
            s_check = get_s_from_t(t)
            score = fast_score(s_check)
            if score < best_score:
                best_score = score
                best_t = t.copy()

    # Final evaluation
    s_final = get_s_from_t(t)
    score_final = fast_score(s_final)
    if score_final < best_score:
        best_score = score_final
        best_t = t.copy()

    return best_t, best_score


def build_conv_matrix(s):
    """Build (2n-1) x n matrix for LP direction finding."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_polish(s, deadline, max_iters=60):
    """LP-guided local search to polish a sequence."""
    n = len(s)
    best = s.copy()
    best_sc = fast_score(s)
    rng = np.random.default_rng(42)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        M = np.max(conv)

        b = solve_lp_direction(s, M, time_limit=2.0)
        if b is None or np.sum(b) < 1e-14:
            break

        b = b / b.sum()

        best_t_val = 0.0
        best_ts = best_sc
        for t_val in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - t_val) * s + t_val * b
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                continue
            ts = fast_score(cand)
            if ts < best_ts - 1e-12:
                best_ts = ts
                best_t_val = t_val

        if best_t_val == 0.0:
            no_imp += 1
            if no_imp > 12:
                break
            s = s + 0.002 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue

        s = (1.0 - best_t_val) * s + best_t_val * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        no_imp = 0

        if best_ts < best_sc:
            best_sc = best_ts
            best = s.copy()

    return best, best_sc


def make_initializations(n, rng, hseq=None):
    """Generate diverse log-parameter initializations."""
    inits = []
    x = np.arange(n) / max(n - 1, 1)

    # Uniform
    inits.append(('uniform', np.zeros(n)))

    # Random
    for i in range(5):
        t = rng.standard_normal(n) * 0.3
        inits.append((f'rand_{i}', t))

    # Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3]:
            t = -((x - center) ** 2) / (2 * sigma ** 2)
            inits.append((f'g_{center}_{sigma}', t))

    # U-shape (mass at edges)
    for beta in [0.5, 1.0, 2.0, 4.0]:
        t = beta * (2 * x - 1) ** 2
        inits.append((f'u_{beta}', t))

    # Cosine modulations
    for freq in [1, 2, 3, 4]:
        for amp in [1.0, 2.0, 3.0]:
            t = amp * np.cos(2 * np.pi * freq * x)
            inits.append((f'cos{freq}_{amp}', t))

    # Sine modulations
    for freq in [1, 2, 3]:
        for amp in [0.5, 1.0, 2.0]:
            t = amp * np.sin(2 * np.pi * freq * x)
            inits.append((f'sin{freq}_{amp}', t))

    # Linear ramps
    for beta in [-2.0, -1.0, 1.0, 2.0]:
        t = beta * (2 * x - 1)
        inits.append((f'ramp_{beta}', t))

    # From height_sequence_1
    if hseq is not None and len(hseq) > 10:
        x_old = np.linspace(0, 1, len(hseq))
        x_new = np.linspace(0, 1, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 1e-10)
        t = np.log(h_r)
        t -= np.mean(t)
        inits.append(('hseq', t))

    # Two-level
    for frac in [0.3, 0.4, 0.5]:
        for ratio in [0.5, 2.0, 3.0]:
            s_raw = np.ones(n)
            k = int(frac * n)
            s_raw[:k] = ratio
            s_raw = s_raw / s_raw.sum()
            t = np.log(s_raw)
            t -= np.mean(t)
            inits.append((f'two_{frac}_{ratio}', t))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimizes 2n*max(conv(s,s))/(sum(s))^2 using gradient-based search with
    softmax approximation, log-parameterization, Adam, and alpha annealing.
    Multiple sequence lengths (512, 1024, 2048) and diverse initializations
    are explored. LP polishing refines the best result.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 20.0

    # Load height_sequence_1 if available
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h / h.sum()
            print(f"Loaded height_sequence_1, n={h.size}, score={fast_score(hseq):.6f}")
    except Exception:
        pass

    global_best_score = float('inf')
    global_best_seq = None

    # Time allocation per sequence length
    n_configs = [
        (512, 0.15),
        (1024, 0.35),
        (2048, 0.50),
    ]

    available_time = total_deadline - time.monotonic()

    for n, time_frac in n_configs:
        if time.monotonic() > total_deadline - 40:
            break

        n_time = available_time * time_frac
        n_deadline = time.monotonic() + n_time
        remaining = n_deadline - time.monotonic()
        print(f"\n=== n={n}, budget={remaining:.0f}s ===")

        inits = make_initializations(n, rng, hseq)

        # Score all initializations
        init_scores = []
        for name, t_init in inits:
            s_init = get_s_from_t(t_init)
            sc = fast_score(s_init)
            init_scores.append((sc, name, t_init))

        init_scores.sort(key=lambda x: x[0])

        print("  Top inits:")
        for sc, name, _ in init_scores[:5]:
            print(f"    {name}: {sc:.6f}")

        # Run gradient search on top initializations
        n_trials = min(8, len(init_scores))
        time_per_trial = n_time * 0.72 / n_trials

        for i in range(n_trials):
            if time.monotonic() > n_deadline - 15:
                break

            sc0, name, t_init = init_scores[i]
            trial_deadline = time.monotonic() + time_per_trial
            print(f"  Trial [{name}] init={sc0:.6f}, time={time_per_trial:.1f}s")

            alpha_start = n
            alpha_end = 15 * n
            lr = 0.03

            best_t, best_sc = run_gradient_descent(t_init, trial_deadline,
                                                    alpha_start, alpha_end, lr)

            s_best = get_s_from_t(best_t)
            final_sc = fast_score(s_best)

            if final_sc < global_best_score:
                global_best_score = final_sc
                global_best_seq = s_best.copy()
                print(f"  ** New global best: {final_sc:.6f} (n={n}, {name})")
            else:
                print(f"  Result: {final_sc:.6f}")

        # LP polishing if time permits and n <= 1024
        if (time.monotonic() < n_deadline - 5 and n <= 1024
                and global_best_seq is not None and len(global_best_seq) == n):
            print(f"  LP polishing (n={n})...")
            pol_deadline = min(n_deadline, total_deadline - 15)
            polished, pol_sc = lp_polish(global_best_seq, pol_deadline, max_iters=40)
            if pol_sc < global_best_score:
                global_best_score = pol_sc
                global_best_seq = polished.copy()
                print(f"  ** Polished: {pol_sc:.6f}")
            else:
                print(f"  Polish result: {pol_sc:.6f}")

    # Final polish on best if time permits
    if global_best_seq is not None and time.monotonic() < total_deadline - 10:
        n_best = len(global_best_seq)
        if n_best <= 1024:
            print(f"\nFinal polish (n={n_best})...")
            polished, pol_sc = lp_polish(global_best_seq, total_deadline, max_iters=30)
            if pol_sc < global_best_score:
                global_best_score = pol_sc
                global_best_seq = polished.copy()
                print(f"  ** Final polished: {pol_sc:.6f}")
        else:
            # For large n, do a quick gradient polish
            t_init = np.log(global_best_seq + 1e-30)
            t_init -= np.mean(t_init)
            quick_deadline = total_deadline
            alpha_s = 5 * n_best
            alpha_e = 20 * n_best
            best_t, best_sc = run_gradient_descent(t_init, quick_deadline,
                                                    alpha_s, alpha_e, lr=0.01,
                                                    max_steps=50000)
            s_best = get_s_from_t(best_t)
            final_sc = fast_score(s_best)
            if final_sc < global_best_score:
                global_best_score = final_sc
                global_best_seq = s_best.copy()
                print(f"  ** Gradient polish: {final_sc:.6f}")

    # Ensure valid result
    if global_best_seq is None:
        global_best_seq = np.ones(256) / 256.0

    # Final verification
    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(global_best_seq)}")

    # Safety fallback
    if np.isinf(final_score) or final_score < 0:
        global_best_seq = np.ones(256) / 256.0
        final_score = float(evaluate_sequence(global_best_seq.tolist()))
        print(f"  Fallback score: {final_score:.6f}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
