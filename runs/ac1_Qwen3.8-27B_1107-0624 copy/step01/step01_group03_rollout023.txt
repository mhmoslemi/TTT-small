def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the dual algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u > (cssv - 1) / rho
    rho_hat = rho[cond][-1]

from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """FFT-based convolution for efficiency."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = fft_convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1) / rho
    if not cond.any():
        return np.ones(n) / n
    rho_hat = np.where(cond)[0][-1] + 1
    theta = (cssv[rho_hat - 1] - 1) / rho_hat
    return np.maximum(v - theta, 0)

    """Gradient of soft-max of conv(s,s)."""
    c = fft_convolve(s, s)
    max_c = c.max()
    w = np.exp(alpha * (c - max_c))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    grad = np.correlate(w, s, mode='valid')
    return grad

    """Build (2n-1) x n matrix for LP."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

import numpy as np

import time

"""
Multi-strategy search for low-peak autocorrelation sequences.
Combines smooth gradient optimization (soft-max annealing) with LP-guided
local search for polishing. Tries multiple sequence lengths and diverse
initializations to minimize 2n*max(conv(s,s))/(sum(s))^2.

def fft_convolve(a, b):
    """FFT-based linear convolution."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = fft_convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / rho
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho_hat = idx[-1] + 1
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def soft_max_grad(s, alpha):
    """Compute gradient of soft-max-alpha of conv(s,s)."""
    c = fft_convolve(s, s)
    max_c = np.max(c)
    shifted = c - max_c
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    grad = np.correlate(w, s, mode='valid')
    return grad

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):

            if time.monotonic() > polish_deadline - 5:
                break
            s_new, sc_new = lp_polish(s_curr, polish_deadline, max_iters=10)
            if sc_new < sc_curr:
                s_curr = s_new
                sc_curr = sc_new
                print(f"  Polish round {round_idx}: {sc_new:.6f}")
            else:
                # Try a small perturbation and re-optimize
                s_pert = s_curr + 0.001 * rng.standard_normal(n_polish)
                s_pert = simplex_project(s_pert)
                s_new, sc_new = lp_polish(s_pert, polish_deadline, max_iters=5)
                if sc_new < sc_curr:
                    s_curr = s_new
                    sc_curr = sc_new
                    print(f"  Perturbed polish round {round_idx}: {sc_new:.6f}")
                else:
                    break

        if sc_curr < global_best_score:
            global_best_score = sc_curr
            global_best_seq = s_curr.copy()

        print(f"  After final polishing: {global_best_score:.6f}")

    # Fallback if nothing was found
    if global_best_seq is None:
        print("WARNING: No valid sequence found, using uniform fallback")
        global_best_seq = np.ones(256) / 256.0

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(result)}")

    return result
