```python
"""
LP-based iterative optimization for minimizing J(a) = 2n*max(a*a)/(sum(a))^2.
Uses the key insight that for a fixed f, the problem of finding g that
maximizes sum(g) subject to ||f*g||_inf <= M and g >= 0 is a linear program.
Alternates between LP solves and convex combination updates.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def convolve_fft(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    result = irfft(fa * fb, size)
    return result[:n+m-1]

def compute_J(a):
    """Compute the objective value."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def build_conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[m] = (f*g)[m]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        j_start = max(0, m - n + 1)
        j_end = min(n, m + 1)
        for j in range(j_start, j_end):
            A[m, j] = f[m - j]
    return A

def solve_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[m] <= M for all m, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    
    c_obj = -np.ones(n)
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None

def project_simplex(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection on threshold."""
    # Find tau such that sum(max(a - tau, 0)) = S
    # Externally, we can use the water-filling approach
    sorted_a = np.sort(a)[::-1]
    cumulative = np.cumsum(sorted_a)
    # Find the right number of active variables
    rho = np.where(sorted_a > (cumulative - S) / np.arange(1, len(a)+1))[0]
    if len(rho) == 0:
        return np.zeros(len(a))
    rho = rho[-1]
    tau = (cumulative[rho] - S) / (rho + 1)
    return np.maximum(a - tau, 0)
```

```python
import numpy as np
```

def convolve_fft(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def project_onto_positive_sum(a, S):
    """Project a onto {x >= 0, sum(x) = S} using bisection."""
    n = len(a)
    if a.max() < S / n:
        # All can be increased
        result = a.copy()
        result += (S - a.sum()) / n
        return result
    # Bisection on threshold tau: sum(max(a - tau, 0)) = S
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(a - mid, 0.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(a - tau, 0.0)
    # Fix sum exactly
    diff = S - result.sum()
    if abs(diff) > 1e-12:
        pos = result > 0
        if pos.sum() > 0:
            result[pos] += diff / pos.sum()
            result = np.maximum(result, 0.0)
    return result

    S = a.sum()
    if S < 0.01:

def smooth_objective_and_grad(a, tau, n_peaks=20):
    """Compute smoothed objective and gradient using log-sum-exp on top peaks."""
    n = len(a)
    c = convolve_fft(a)
    S = a.sum()
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Use log-sum-exp for smooth max
    c_max = c.max()
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp  # softmax weights
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Gradient via cross-correlation
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = rfft(a, size)
    fp = rfft(p, size)
    cross = irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad
