"""
LP-based optimization of step function coefficients to minimize
J(a) = 2n * max(a*a) / (sum(a))^2.

Uses the state-of-the-art LP approach: at each step, solve an LP to find
a direction that reduces the peak auto-correlation, then take a line-search
step. Combined with gradient descent refinement and multiple initializations.
"""

import numpy as np
import time
import scipy.optimize as sco
from scipy.sparse import lil_matrix, csr_matrix


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def cross_correlate_fft(f, y):
    """Compute cross-correlation of f (length n) with y (length 2n-1).
    
    Returns array of length n: result[j] = sum_m f[m] * y[j+m]
    (with y[k]=0 for k outside [0, 2n-2]).
    """
    n = len(f)
    size = 1
    while size < 3 * n:
        size <<= 1
    # We want: result[j] = sum_m f[m] * y[j+m]
    # This is like a "sliding dot product" or cross-correlation.
    # Using FFT: result = ifft(fft(f_padded) * conj(fft(y_padded)))... 
    # Actually, let me think more carefully.
    # 
    # result[j] = sum_{m=0}^{n-1} f[m] * y[j+m]
    # Let's define f_rev[m'] = f[n-1-m']
    # Then result[j] = sum_{m'=0}^{n-1} f_rev[m'] * y[j + n - 1 - m']
    # = (f_rev * y)[j + n - 1]  (linear convolution)
    # 
    # So: result = convolve(f_rev, y)[n-1 : n-1+n]
    
    f_rev = f[::-1]
    # Linear convolution of f_rev (length n) and y (length 2n-1) has length n + 2n-1 - 1 = 3n-2
    conv_size = 1
    while conv_size < 3 * n - 2:
        conv_size <<= 1
    F = np.fft.rfft(f_rev, conv_size)
    Y = np.fft.rfft(y, conv_size)
    conv = np.fft.irfft(F * Y, conv_size)
    return conv[n-1 : n-1+n]


def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix T(f).
    T(f)[k,j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        # Column j: A[k, j] = f[k-j] for k = j, j+1, ..., min(2n-2, j+n-1)
        k_start = j
        k_end = min(2 * n - 2, j + n - 1)
        length = k_end - k_start + 1
        A[k_start:k_end+1, j] = f[:length]
    return A.tocsr()


def solve_lp_full(f, M):
    """Solve full LP: max sum(g) s.t. T(f)@g <= M, g >= 0.
    Returns (optimal_value, g) or (None, None).
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    t0 = time.monotonic()
    res = sco.linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    t_solve = time.monotonic() - t0
    
    if res.status == 0:
        optimal_value = -res.fun
        return optimal_value, res.x, t_solve
    return None, None, t_solve


def solve_lp_reduced(f, M, K=200):
    """Solve reduced LP using only the K largest auto-correlation lags as constraints.
    Returns (optimal_value, g) or (None, None).
    """
    n = len(f)
    c = convolve_fft(f)
    # Get the K largest lags
    top_k = np.argsort(c)[-K:][::-1]
    
    # Build reduced constraint matrix
    A_red = lil_matrix((K, n))
    for idx, k in enumerate(top_k):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A_red[idx, j] = f[k - j]
    A_red = A_red.tocsr()
    
    c_obj = -np.ones(n)
    b = np.full(K, M)
    
    res = sco.linprog(c_obj, A_ub=A_red, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.status == 0:
        optimal_value = -res.fun
        return optimal_value, res.x
    return None, None


def lp_iteration(f, M, use_full=True, K=200):
    """One LP-based iteration. Returns (new_f, improved, solve_time)."""
    n = len(f)
    
    if use_full:
        g_opt, g, t_solve = solve_lp_full(f, M)
    else:
        g_opt, g = solve_lp_reduced(f, M, K)
        t_solve = 0
    
    if g_opt is None or g is None:
        return f, False, t_solve
    
    if g_opt <= 1.0 + 1e-4:
        # At (or near) local optimum
        return f, False, t_solve
    
    # h = g / g_opt, so sum(h) = 1
    g_sum = np.sum(g)
    if g_sum < 0.01:
        return f, False, t_solve
    h = g / g_sum
    
    # Line search
    best_t = 0.0
    best_J = compute_J(f)
    for t in [1.0, 0.8, 0.5, 0.3, 0.1]:
        cand = (1 - t) * f + t * h
        cand = np.maximum(cand, 0)
        S = np.sum(cand)
        if S < 0.01:
            continue
        cand = cand / S  # renormalize (though sum should already be 1)
        J_cand = compute_J(cand)
        if J_cand < best_J:
            best_J = J_cand
            best_t = t
    
    if best_t == 0.0:
        return f, False, t_solve
    
    new_f = (1 - best_t) * f + best_t * h
    new_f = np.maximum(new_f, 0)
    S = np.sum(new_f)
    if S > 0:
        new_f = new_f / S
    return new_f, True, t_solve


def run_lp_loop(f_init, budget_s, use_full=True, K=200, seed=42):
    """Run the LP-based iteration loop. Returns (best_f, best_J, iterations)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    f = f_init.copy()
    S = np.sum(f)
    if S > 0:
        f = f / S
    
    c = convolve_fft(f)
    M = np.max(c)
    best_f = f.copy()
    best_J = 2.0 * len(f) * M
    iterations = 0
    stall_count = 0
    last_verify = time.monotonic()
    
    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2:
            break
        
        f_new, improved, t_solve = lp_iteration(f, M, use_full, K)
        iterations += 1
        
        if not improved:
            stall_count += 1
            if stall_count > 5:
                # Perturb and restart
                f = f + 0.01 * rng.normal(0, 1, len(f))
                f = np.maximum(f, 0)
                S = np.sum(f)
                if S > 0.01:
                    f = f / S
                c = convolve_fft(f)
                M = np.max(c)
                stall_count = 0
                use_full = not use_full  # switch between full and reduced
                continue
        else:
            f = f_new
            c = convolve_fft(f)
            M = np.max(c)
            stall_count = 0
            J = 2.0 * len(f) * M
            if J < best_J:
                best_J = J
                best_f = f.copy()
        
        # Periodic verification
        if time.monotonic() - last_verify > 15:
            last_verify = time.monotonic()
            val = float(evaluate_sequence(best_f.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                print(f"    LP iter {iterations}: verified J={val:.6f}, M={M:.6f}")
    
    # Final check
    final_J = compute_J(best_f)
    best_J = min(best_J, final_J)
    return best_f, best_J, iterations


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def run_gradient_descent(a_init, budget_s, seed, tau_start=2.0, tau_end=0.01, lr_start=0.01):
    """Adam gradient descent with adaptive tau."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a / S  # normalize to sum 1
    
    m = np.zeros(n)

"""
LP-based optimization of step function coefficients to minimize
J(a) = 2n * max(a*a) / (sum(a))^2.

Uses reduced-constraint LP iterations (solving max sum(g) s.t. top-K auto-correlation
constraints), combined with gradient descent refinement, multiple initializations,
and periodic verification. The LP approach provides principled descent directions
by finding the optimal non-negative sequence g with bounded f*g peak.
"""

import numpy as np
import time
import scipy.optimize as sco
from scipy.sparse import lil_matrix, csr_matrix


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective value and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad

                # Try a larger perturbation
                scale = 0.005 + 0.01 * min(stall_count / 20.0, 1.0)
                f = f + scale * rng.normal(0, 1, n)
                f = np.maximum(f, 0)
                S = np.sum(f)
                if S > 0.01:
                    f = f / S
                    c = convolve_fft(f)
                    M = np.max(c)
                stall_count = 0
            continue

        h = g / g_sum

        # Adaptive line search
        best_t = 0.0
        best_J_local = compute_J(f)
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            cand = (1.0 - t) * f + t * h
            cand = np.maximum(cand, 0)
            S_c = np.sum(cand)
            if S_c < 0.01:
                continue
            cand = cand / S_c
            J_c = compute_J(cand)
            if J_c < best_J_local:
                best_J_local = J_c
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * h
            f = np.maximum(f, 0)
            S = np.sum(f)
            if S > 0:
                f = f / S
            c = convolve_fft(f)
            M = np.max(c)
            J = 2.0 * n * M
            if J < best_J:
                best_J = J
                best_f = f.copy()
            iterations += 1
            stall_count = 0
        else:
            stall_count += 1

        # Periodic verification
        if time.monotonic() - last_verify > 10.0:
            last_verify = time.monotonic()
            val = float(evaluate_sequence(best_f.tolist()))
            if np.isfinite(val):
                best_J = min(best_J, val)
                print(f"    LP iter {iterations}: J={val:.6f}, M={M:.6f}, g_opt={g_opt:.4f}, K={K_use}")

    return best_f, best_J, iterations


def run_lp_full_loop(f_init, budget_s, seed=42):
    """Run LP-based iterations using the full constraint set."""
    n = len(f_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    f = f_init.copy().astype(np.float64)
    S = np.sum(f)
    if S > 0:
        f = f / S

    c = convolve_fft(f)
    M = np.max(c)
    best_f = f.copy()
    best_J = 2.0 * n * M
    iterations = 0
    stall_count = 0
    last_verify = time.monotonic()

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2.0:
            break

        t0 = time.monotonic()

        # Build full constraint matrix
        A = lil_matrix((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)
            length = k_end - k_start + 1
            A[k_start:k_end + 1, j] = f[:length]
        A = A.tocsr()

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        res = sco.linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': deadline - time.monotonic() - 3.0})

        t_solve = time.monotonic() - t0

        if res.status != 0:
            stall_count += 1
            if stall_count > 5:
                f = f + 0.003 * rng.normal(0, 1, n)
                f = np.maximum(f, 0)
                S = np.sum(f)
                if S > 0.01:
                    f = f / S
                    c = convolve_fft(f)
                    M = np.max(c)
                stall_count = 0
            continue

        g_opt = -res.fun
        g = res.x
        g_sum = np.sum(g)

        if g_opt <= 1.0 + 1e-4 or g_sum < 0.01:
            stall_count += 1
            if stall_count > 5:
                f = f + 0.005 * rng.normal(0, 1, n)
                f = np.maximum(f, 0)
                S = np.sum(f)
                if S > 0.01:
                    f = f / S
                    c = convolve_fft(f)
                    M = np.max(c)
                stall_count = 0
            continue

        h = g / g_sum

        best_t = 0.0
        best_J_local = compute_J(f)
        for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
            cand = (1.0 - t) * f + t * h
            cand = np.maximum(cand, 0)
            S_c = np.sum(cand)
            if S_c < 0.01:
                continue
            cand = cand / S_c
            J_c = compute_J(cand)
            if J_c < best_J_local:
                best_J_local = J_c
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * h
            f = np.maximum(f, 0)
            S = np.sum(f)
            if S > 0:
                f = f / S
            c = convolve_fft(f)
            M = np.max(c)
            J = 2.0 * n * M
            if J < best_J:
                best_J = J
                best_f = f.copy()
            iterations += 1
            stall_count = 0
            if iterations % 10 == 0:
                print(f"    Full LP iter {iterations}: J={J:.6f}, solve={t_solve:.2f}s, g_opt={g_opt:.4f}")
        else:
            stall_count += 1

        if time.monotonic() - last_verify > 15.0:
            last_verify = time.monotonic()
            val = float(evaluate_sequence(best_f.tolist()))
            if np.isfinite(val):
                best_J = min(best_J, val)

    return best_f, best_J, iterations


def run_gradient_descent(a_init, budget_s, seed, tau_start=2.0, tau_end=0.01, lr_start=0.008):
    """Adam gradient descent with log-sum-exp smoothing."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    start_time = time.monotonic()

    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S > 0:
        a = a / S

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    last_verify = time.monotonic()
    stall_count = 0
    stall_threshold = best_J

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 1.0:
            break

        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.05 * t / 1000.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0)
        S = np.sum(a)
        if S < 0.01:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        a = a / S

        if t % 3000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_threshold = local_J
            else:
                stall_count += 1
                if stall_count > 10:
                    # Perturbation
                    w = max(1, n // 100)
                    s = int(rng.integers(0, n - w))
                    a[s:s + w] *= np.exp(rng.normal(0, 0.3, w))
                    a = np.maximum(a, 0)
                    S = np.sum(a)
                    if S > 0:
                        a = a / S
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0

        if time.monotonic() - last_verify > 10.0:
            last_verify = time.monotonic()
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    GD t={t}: J={val:.6f}, tau={tau:.4f}")

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase LP + GD optimizer for step function coefficient sequences.
    
    Phase 1: Reduced-constraint LP iterations (fast, many iterations).
    Phase 2: Full LP iterations on the best result (fewer, more precise iterations).
    Phase 3: Gradient descent refinement with log-sum-exp smoothing.
    
    Multiple diverse initializations are explored.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}, total budget = {budget_s}s")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Build diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ones", np.ones(n)))
    inits.append(("gauss_25", make_gaussian(n, 0.25)))
    inits.append(("gauss_40", make_gaussian(n, 0.40)))
    inits.append(("cos1", make_cosine(n, 1.0, 0.3)))
    inits.append(("cos2", make_cosine(n, 2.0, 0.2)))
    inits.append(("taper", make_tapered(n, 1.5)))
    inits.append(("rand", np.exp(rng.normal(0, 0.4, n))))
    inits.append(("edge_boost", 0.5 + 0.5 * np.abs(2.0 * np.arange(n) / (n - 1) - 1.0)))

    # Score and sort initializations
    scored = []
    for name, arr in inits:
        arr = np.maximum(arr, 0).astype(np.float64)
        S = np.sum(arr)
        if S > 0:
            arr = arr / S
        J = compute_J(arr)
        scored.append((name, J, arr))

    scored.sort(key=lambda x: x[1])
    for name, J, _ in scored:
        print(f"  init '{name}': J = {J:.6f}")

    # === Phase 1: Reduced LP iterations on top 3 initializations ===
    remaining = total_deadline - time.monotonic()
    # Allocate time: 40% to best, 30% to second, 20% to third, 10% reserve
    p1_allocs = [remaining * 0.40, remaining * 0.30, remaining * 0.20]

    for i in range(min(3, len(scored))):
        if time.monotonic() >= total_deadline - 30:
            break

        name, init_J, init_arr = scored[i]
        budget = min(p1_allocs[i], total_deadline - time.monotonic() - 25)
        if budget < 10:
            break

        print(f"\nPhase 1 [{name}]: reduced LP, budget={budget:.1f}s, init_J={init_J:.4f}")

        f_out, J_out, iters = run_lp_reduced_loop(
            init_arr, budget, K=200, seed=seed + i * 137
        )
        val = float(evaluate_sequence(f_out.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = f_out.copy()
            best_global_score = val
            print(f"  *** NEW BEST (reduced LP on {name}): {val:.6f} after {iters} iters ***")
        else:
            print(f"  ({name}: local_J={J_out:.4f}, verified={val:.4f}, best={best_global_score:.6f})")

    # === Phase 2: Full LP iterations on the best ===
    remaining = total_deadline - time.monotonic() - 30
    if remaining > 20:
        p2_budget = min(remaining * 0.5, 300)
        print(f"\nPhase 2: full LP on best, budget={p2_budget:.1f}s")

        f_out, J_out, iters = run_lp_full_loop(
            best_global, p2_budget, seed=seed + 555
        )
        val = float(evaluate_sequence(f_out.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = f_out.copy()
            best_global_score = val
            print(f"  *** NEW BEST (full LP): {val:.6f} after {iters} iters ***")
        else:
            print(f"  (full LP: local_J={J_out:.4f}, verified={val:.4f})")

    # === Phase 3: Reduced LP again on best (different K, different seed) ===
    remaining = total_deadline - time.monotonic() - 25
    if remaining > 15:
        p3_budget = min(remaining * 0.5, 200)
        print(f"\nPhase 3: reduced LP (K=400) on best, budget={p3_budget:.1f}s")

        f_out, J_out, iters = run_lp_reduced_loop(
            best_global, p3_budget, K=400, seed=seed + 999
        )
        val = float(evaluate_sequence(f_out.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = f_out.copy()
            best_global_score = val
            print(f"  *** NEW BEST (reduced LP K=400): {val:.6f} ***")
        else:
            print(f"  (reduced LP K=400: local_J={J_out:.4f}, verified={val:.4f})")

    # === Phase 4: Gradient descent refinement ===
    remaining = total_deadline - time.monotonic() - 10
    if remaining > 10:
        print(f"\nPhase 4: GD refinement, budget={remaining:.1f}s")
        f_ref, J_ref = run_gradient_descent(
            best_global, remaining, seed=seed + 7777,
            tau_start=1.0, tau_end=0.01, lr_start=0.005
        )
        val = float(evaluate_sequence(f_ref.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = f_ref.copy()
            best_global_score = val
            print(f"  *** NEW BEST (GD): {val:.6f} ***")
        else:
            print(f"  (GD: local_J={J_ref:.4f}, verified={val:.4f})")

    # === Final verification ===
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
