    return 2.0 * n * c.max() / (S * S)

def subgradient_direction(a, threshold_frac=0.95):
    """Compute subgradient of max(conv(a,a)) using peaks above threshold."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    # Get peaks above threshold
    threshold = threshold_frac * M
    peaks = np.where(c >= threshold)[0]
    
    # Weight peaks by how close they are to M
    weights = np.maximum(0, c[peaks] - threshold) / (M - threshold + 1e-15)
    weights = weights / (weights.sum() + 1e-15)
    
    subgrad = np.zeros(n)
    for idx, k in enumerate(peaks):
        w = weights[idx]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                subgrad[i] += w * a[j]
    
    return subgrad, M

"""
Hybrid subgradient + LP optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated multiplicative weight updates with multi-peak subgradients,
followed by LP-based polishing for final refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def get_subgradient(a, threshold_frac=0.95):
    """Compute weighted subgradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    threshold = threshold_frac * M
    
    peak_mask = c >= threshold
    peak_idx = np.where(peak_mask)[0]
    
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(c)])
    
    # Weight by how close each peak is to max
    peak_vals = c[peak_idx]
    w = (peak_vals - threshold + 1e-15)
    w = w / w.sum()
    
    subgrad = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        subgrad[lo:hi + 1] += w[idx] * 2.0 * a[k - hi:k - lo + 1]
    
    return subgrad, M, len(peak_idx)


def optimize_subgradient(a, deadline, rng, threshold_frac=0.95):
    """Accelerated multiplicative weight update for minimizing max(conv(a,a))."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.05
    stall = 0
    it = 0
    f_lower = 2.0 * n / (2.0 * n - 1.0)
    
    while time.monotonic() < deadline:
        it += 1
        
        subgrad, M, npeaks = get_subgradient(a, threshold_frac)
        
        if np.max(subgrad) < 1e-15:
            break
        
        # Normalize subgradient for stable updates
        sg_max = np.max(subgrad)
        subgrad_norm = subgrad / sg_max  # Scale to [0, 1]
        
        # Adaptive step size
        sc = 2.0 * n * M / (a.sum() ** 2)
        gap = max(0.0, sc - f_lower)
        polyak_lr = gap / (np.sum(subgrad_norm * (a > 0)) + 1e-10)
        lr = min(0.3, max(0.001, polyak_lr))
        
        # Multiplicative update
        a_new = a * np.exp(-lr * subgrad_norm)
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        
        sc_new = compute_score(a_new)
        if sc_new < best_score - 1e-11:
            best_a = a_new.copy()
            best_score = sc_new
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall == 10:
                # Try additive update
                a_new = a - lr * 0.1 * subgrad
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    sc_try = compute_score(a_new)
                    if sc_try < best_score:
                        best_a = a_new.copy()
                        best_score = sc_try
                        a = a_new
                        stall = 0
            elif stall == 25:
                # Directional perturbation
                a_new = best_a * (1.0 + 0.03 * rng.standard_normal(n))
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                stall = 0
            if stall > 100:
                # Restart from best with larger perturbation
                a_new = best_a * (1.0 + 0.1 * np.abs(rng.standard_normal(n)))
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                    stall = 50  # Give it more chances
            if stall > 300:
                break
        
        # Occasionally try different threshold
        if it % 500 == 0:
            threshold_frac = [0.9, 0.95, 0.98, 0.85][it // 500 % 4]
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, max_iters=200):
    """LP-based optimization loop."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.4, step * 1.05)
        else:
            stall += 1
            step = max(0.005, step * 0.7)
            if stall % 30 == 0:
                noise = np.abs(rng.normal(0, 0.05, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                    stall = 0
            if stall > 100:
                break
    
    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_random(n, rng):

"""
Hybrid subgradient + LP optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated multiplicative weight updates with multi-peak subgradients for fast
broad exploration, followed by LP-based polishing and local coordinate search for
final refinement. Key improvements over previous approach:
- Vectorized multi-peak subgradient with adaptive weighting
- Polyak-style adaptive step size
- Two-scale optimization (coarse-to-fine)
- Periodic threshold cycling to explore different peak structures
- More aggressive LP polishing with warm-start line search
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def get_subgradient(a, threshold_frac=0.95):
    """Compute weighted subgradient of max(conv(a,a)) using peaks above threshold.
    
    Returns (subgrad, max_val, num_peaks) where subgrad[i] = weighted sum of 2*a[k-i]
    over all peaks k above threshold, weighted by proximity to max.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    if M < 1e-15:
        return np.zeros(n), M, 0
    
    threshold = threshold_frac * M
    peak_mask = c >= threshold
    peak_idx = np.where(peak_mask)[0]
    
    if len(peak_idx) == 0:
        peak_idx = np.array([np.argmax(c)])
    
    peak_vals = c[peak_idx]
    spread = M - threshold
    if spread < 1e-15:
        w = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        w = (peak_vals - threshold) / spread
        w = w / w.sum()
    
    subgrad = np.zeros(n)
    for idx in range(len(peak_idx)):
        k = peak_idx[idx]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # subgrad[i] += 2*w*a[k-i] for i in [lo, hi]
        # a[k-i] for i=lo..hi gives a[k-lo], a[k-lo-1], ..., a[k-hi]
        # which is a[k-hi:k-lo+1] reversed
        subgrad[lo:hi + 1] += 2.0 * w[idx] * a[k - hi:k - lo + 1][::-1]
    
    return subgrad, M, len(peak_idx)


def optimize_subgradient(a, deadline, rng, threshold_start=0.95):
    """Accelerated multiplicative weight update for minimizing max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a.copy(), np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.05
    stall = 0
    it = 0
    f_lower = 2.0 * n / (2.0 * n - 1.0)
    threshold_frac = threshold_start
    threshold_cycle = 0
    momentum = 0.3
    v = np.zeros(n)
    
    while time.monotonic() < deadline:
        it += 1
        
        subgrad, M, npeaks = get_subgradient(a, threshold_frac)
        
        if np.max(subgrad) < 1e-15:
            break
        
        # Momentum on subgradient
        v = momentum * v + (1.0 - momentum) * subgrad
        
        # Normalize for stable multiplicative updates
        sg_max = np.max(v)
        if sg_max < 1e-15:
            break
        v_norm = v / sg_max
        
        # Adaptive step size (Polyak-inspired)
        sc = 2.0 * n * M / (a.sum() ** 2)
        gap = max(0.0, sc - f_lower)
        total_mass = np.sum(v_norm * (a > 1e-15))
        if total_mass < 1e-10:
            break
        polyak_lr = gap / (total_mass + 1e-10)
        lr = min(0.4, max(0.002, polyak_lr))
        
        # Multiplicative update
        a_new = a * np.exp(-lr * v_norm)
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        
        sc_new = compute_score(a_new)
        if sc_new < best_score - 1e-11:

                    sc_try = compute_score(a_new)

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, max_iters=300):
    """LP-based optimization with line search and adaptive step."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    step = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        if time.monotonic() - start_lp > 60 and it < 10:
            break  # Skip if not enough time
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = np.clip(g, 0, None)
        if g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.35, step * 1.04)
        else:
            stall += 1
            step = max(0.003, step * 0.65)
            if stall % 25 == 0:

                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    a = a_new
                    stall = 0
            if stall > 120:
                break
    
    return best_a, best_score


def coordinate_descent(a, deadline, rng, max_iters=50000):
    """Local coordinate descent for final polishing."""
    n = len(a)
    a = np.clip(a, 0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # Pick 2 random coordinates to adjust
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            j = (j + 1) % n
        
        # Try several step sizes
        best_local = best_score
        best_local_a = None
        for delta_frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.005]:
            for sign_i in [1, -1]:
                for sign_j in [1, -1]:
                    a_new = best_a.copy()
                    a_new[i] += sign_i * delta_frac * best_a.max()
                    a_new[j] += sign_j * delta_frac * best_a.max() * 0.5
                    a_new = np.clip(a_new, 0, None)
                    if a_new.sum() < 1e-10:
                        continue
                    a_new /= a_new.sum()
                    sc = compute_score(a_new)
                    if sc < best_local - 1e-12:
                        best_local = sc
                        best_local_a = a_new
        
        if best_local_a is not None:
            best_a = best_local_a.copy()
            best_score = best_local
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
    
    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_bell(n, width=0.4):
    """Bell-shaped with sharper peak."""
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    return np.exp(-(x / (width * n)) ** 2)


def make_two_hump(n):
    """Two-hump initialized sequence."""
    c1 = n * 0.3
    c2 = n * 0.7
    sig = n * 0.12
    x = np.arange(n)
    return np.exp(-0.5 * ((x - c1) / sig) ** 2) + np.exp(-0.5 * ((x - c2) / sig) ** 2)


def make_flat_top(n, frac=0.6):
    """Flat-top window."""
    x = np.arange(n)
    c = (n - 1) / 2.0
    hw = frac * n / 2.0
    result = np.ones(n)
    mask = np.abs(x - c) > hw
    # Smooth transition
    transition = 0.1 * n
    edge = np.abs(x - c)
    result[mask] = 0.5 * (1 + np.cos(np.pi * (edge[mask] - hw) / transition))
    result[edge > hw + transition] = 0
    return np.clip(result, 0, None)


def upsample_coarse_to_fine(coarse, fine_n):
    """Upsample a coarse sequence to fine_n by linear interpolation."""
    coarse_n = len(coarse)
    if coarse_n >= fine_n:
        return coarse[:fine_n].copy()
    x_old = np.linspace(0, 1, coarse_n)
    x_new = np.linspace(0, 1, fine_n)
    return np.interp(x_new, x_old, coarse)
