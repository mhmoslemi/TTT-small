def solve_lp_selective(a, M, frac=0.95):
    """Solve LP with only near-active constraints."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Select constraints where c_k > frac * M
    active = np.where(c >= frac * M)[0]
    
    # Build sparse matrix for only active constraints
    # For each active constraint k, the row is:
    # A[k, :] = [a_{k-0}, a_{k-1}, ..., a_{k-(n-1)}] (with 0 for out of range)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                rows.append(idx)
                cols.append(j)
                vals.append(a[i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

"""
Hybrid LP + subgradient optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses selective-constraint LPs, adaptive backtracking, multi-phase search
with diverse initializations, and a final polish phase.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_sparse(a, active_indices=None):
    n = len(a)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            i = k - j
            rows.append(idx)
            cols.append(j)
            vals.append(a[i])
    if not rows:
        return sparse.csr_matrix((len(active_indices), n))
    return sparse.csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_lp_for_direction(a, M, frac=0.90, max_constraints=3000):
    n = len(a)
    c = np.convolve(a, a, mode='full')
    threshold = frac * M
    active = np.where(c >= threshold)[0]
    if len(active) > max_constraints:
        order = np.argsort(c[active])[::-1]
        active = active[order[:max_constraints]]
    A = build_conv_sparse(a, active)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, verbose=False):
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.12
    stall = 0
    it = 0
    frac_schedule = [0.90, 0.92, 0.95, 0.97]

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        frac = frac_schedule[min(stall // 40, len(frac_schedule) - 1)]
        g = solve_lp_for_direction(a, M, frac=frac)
        if g is None or g.sum() < 1e-15:
            if stall > 20:
                break
            continue
        g = g / g.sum()
        improved = False
        s = step
        for _ in range(35):
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                s *= 0.5
                continue
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-14:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
            s *= 0.5
        if improved:
            stall = 0
            step = min(0.45, step * 1.04)
        else:
            stall += 1
            step = max(0.003, step * 0.78)
            if stall % 25 == 0 and stall > 25:
                scale = min(0.15, 0.04 + 0.005 * (stall // 50))
                noise = rng.normal(0.0, scale, n)
                a_pert = np.clip(best_a * (1.0 + noise), 0.0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc_pert = compute_objective(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        a = a_pert
                        best_score = sc_pert
                        stall = 0
                    elif rng.random() < 0.25:
                        a = a_pert
                        stall = 0
            if stall > 250:
                break
        if verbose and it % 100 == 0:
            print(f"    LP iter={it}, score={best_score:.8f}, step={step:.4f}, stall={stall}")

    return best_a, best_score


def optimize_subgrad(a, deadline, rng):
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.08
    stall = 0

    while time.monotonic() < deadline:
        c = np.convolve(a, a, mode='full')
        M = c.max()
        threshold = M * 0.95
        P = np.where(c >= threshold)[0]
        g = np.zeros(n)
        p_vec = np.zeros(2 * n - 1)
        p_vec[P] = 1.0
        # g_j = (2/|P|) * sum_{k in P} a_{k-j}
        # This is cross-correlation: for each j, sum over k of p_k * a_{k-j}
        # = (p * flip(a)) evaluated appropriately
        # Let's just do it directly with numpy which is fast enough
        for k in P:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += a[k - np.arange(lo, hi + 1)]
        g *= 2.0 / len(P)
        a_new = project_simplex(a - lr * g)
        sc = compute_objective(a_new)
        if sc < best_score:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.4, lr * 1.02)
        else:
            stall += 1
            lr = max(0.002, lr * 0.96)
            if stall > 80:
                break
    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n) / n))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_hann(n):
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / n))


def make_flat_taper(n, taper_frac=0.3):
    a = np.ones(n)
    t = int(n * taper_frac)
    if t > 0:
        taper = 0.5 * (1.0 - np.cos(np.pi * np.arange(t) / t))
        a[:t] = taper
        a[-t:] = taper[::-1]
    return a


def make_bspline(n, levels=3):
    a = np.ones(n)
    for _ in range(levels):
        a = np.convolve(a, np.ones(2) / 2, mode='same')
        a = a / a.sum() if a.sum() > 0 else a
    return a
