    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:

    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)

"""
Refined gradient-based optimization with multi-phase adaptive smoothing,
peak-shaving heuristics, and focused local search to minimize
J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute exact J = 2n*max(conv)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (2.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def multi_arg_subgrad(a):
    """Subgradient using uniform average over near-argmax entries."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    max_c = np.max(c)
    threshold = max_c - 0.01 * max_c
    active = c >= threshold
    if not np.any(active):
        active = c >= max_c - 1e-6

    num_active = np.sum(active)
    cross = np.zeros(n)
    for k in np.where(active)[0]:
        k_int = int(k)
        for i in range(n):
            j = k_int - i
            if 0 <= j < n:
                cross[i] += a[j]
    cross /= num_active

    J = 2.0 * n * max_c / (S * S)
    grad = (2.0 * n * cross) / (S * S) - (4.0 * n * max_c) / (S * S * S)
    return J, grad


def peak_shave(a, num_rounds=200, eta=0.005):
    """Reduce peak by shaving contributing elements."""
    n = len(a)
    for _ in range(num_rounds):
        c = convolve_fft(a)
        k_star = int(np.argmax(c))
        max_c = c[k_star]

        # Sensitivity: which a[i] contribute to c[k_star]?
        sens = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                sens[i] = a[j]

        # Normalize sensitivity
        s_max = np.max(sens)
        if s_max < 1e-10:
            break
        sens_norm = sens / s_max

        # Decrease a where sensitive, scaled by eta
        a = a * (1.0 - eta * sens_norm)
        a = np.maximum(a, 1e-10)

    return a


def coordinate_polish(a, budget_s, seed):
    """Coordinate descent polish on the best candidate."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a.copy()
    best_J = compute_J(a)
    best_a = a.copy()

    while time.monotonic() < deadline:
        # Pick a random subset of coordinates to try
        n_tries = int(rng.integers(5, 30))
        improved = False

        for _ in range(n_tries):
            idx = int(rng.integers(0, n))
            # Try multiplicative perturbation
            for factor in [0.9, 0.95, 1.05, 1.1, 0.8, 1.2]:
                cand = a.copy()
                cand[idx] *= factor
                cand = np.maximum(cand, 0.0)
                S = np.sum(cand)
                if S < 0.01:
                    continue
                cand = cand * (n / S)
                cJ = compute_J(cand)
                if cJ < best_J - 1e-10:
                    best_J = cJ
                    best_a = cand.copy()
                    a = cand.copy()
                    improved = True
                    break

        if not improved:
            # Try window perturbations
            w = int(rng.integers(2, 20))
            s = int(rng.integers(0, n - w))
            cand = a.copy()
            cand[s:s+w] *= np.exp(rng.normal(0, 0.1, w))

def make_diverse_init(n, rng, strategy):
    """Generate diverse initializations."""
    if strategy == 0:
        return np.ones(n)
    elif strategy == 1:
        x = np.arange(n) - n / 2.0
        return np.exp(-x * x / (2 * (0.2 * n) ** 2))
    elif strategy == 2:
        x = np.arange(n)
        return 1.0 + 0.3 * np.cos(2 * np.pi * x / n)
    elif strategy == 3:
        return np.exp(rng.normal(0, 0.5, n))
    elif strategy == 4:
        x = np.arange(n)
        mid = (n - 1) / 2.0
        return 1.0 - 0.4 * np.abs(x - mid) / mid
    elif strategy == 5:
        x = np.arange(n)
        mid = (n - 1) / 2.0
        return 0.5 + 0.5 * np.abs(x - mid) / mid
    elif strategy == 6:
        # Two bumps
        x = np.arange(n)
        return np.exp(-((x - n*0.3)/(n*0.1))**2) + np.exp(-((x - n*0.7)/(n*0.1))**2) + 0.1
    else:
        return 1.0 + 0.2 * rng.normal(0, 1, n)


def run_gradient_phase(a_init, budget_s, seed, tau_start, tau_end, lr_start, lr_end):
    """Run gradient descent with given tau and lr schedule."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s

    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_at_stall = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget_s, 1.0))

        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        J_s, grad = smooth_obj_grad(a, tau)
        if not np.isfinite(J_s) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            continue

        t += 1
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = np.maximum(a, 1e-10)
        S = np.sum(a)
        if S > 0.01:
            a = a * (n / S)

        # Check progress periodically
        if t % 20 == 0:
            J_check = compute_J(a)
            if J_check < best_J - 1e-8:
                best_J = J_check
                best_a = a.copy()
                stall = 0
                best_at_stall = best_J
            else:
                stall += 1
                if stall > 10:
                    # Kick perturbation
                    w = int(n // 30)
                    s = int(rng.integers(0, max(1, n - w)))
                    cand = a.copy()
                    cand[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 1e-10)
                    Sc = np.sum(cand)
                    if Sc > 0.01:
                        cand = cand * (n / Sc)
                    cJ = compute_J(cand)
                    if cJ < best_at_stall:
                        a = cand
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall = 0
                    else:
                        a = best_a.copy()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall = 0

    return best_a, best_J
