import numpy as np
import time

def full_convolve_fft(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    elif mn < 0:
        c = c - mn
    return c

def score_of(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = full_convolve_fft(a)
    return 2.0 * n * c.max() / (s * s)

def make_init_profile(n, etype='linspace', **kwargs):
    """Create various initial profiles."""
    if etype == 'linear':
        eta = kwargs.get('eta', 1.5)
        t = np.linspace(0, 1, n)

        t = np.linspace(0, np.pi, n)

"""
Multi-scale gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses log-space Adam with Nesterov momentum at
multiple sequence lengths, followed by L-BFGS polishing. Key improvements:
direct optimization at large n, better initialization from parent, adaptive
beta schedule, and hybrid first-order/quasi-Newton approach.
"""
import numpy as np
import time
from scipy.optimize import minimize


def full_convolve_fft(a):
    """Full linear convolution of a with itself via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    elif mn < 0.0:
        c = c - mn
    return c


def cross_corr_w_a(w, a):
    """Compute h[k] = sum_j w[k+j] * a[j] for k=0..n-1 via FFT."""
    n = a.size
    N = 1 << (2 * n - 2).bit_length()
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)[n - 1:2 * n - 1]
    return h


def soft_max_and_weights(c, beta):
    """Compute soft max value and interpolation weights."""
    m = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    x = (c - m) / beta
    max_x = float(x.max())
    ex = np.exp(x - max_x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return m, w
    w = ex / s
    val = m + beta * float(np.log(s) + max_x)
    return val, w


def z_to_seq(z):
    """Convert log-space parameters to normalized non-negative sequence."""
    z = np.clip(z, -80.0, 80.0)
    y = z - float(z.max())
    a = np.exp(y)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-30:
        return np.ones(z.size) / z.size
    return a / s


def seq_to_z(a):
    """Convert positive sequence to centered log-space parameters."""
    a = np.maximum(a, 1e-30)
    z = np.log(a)
    z = z - float(z.mean())
    return z


def score_from_seq(a):
    """Compute score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    c = full_convolve_fft(a)
    return 2.0 * n * float(c.max())


def compute_grad_z(z, n, beta):
    """Compute gradient of score w.r.t. z (log-space params)."""
    a = z_to_seq(z)
    c = full_convolve_fft(a)
    L, w = soft_max_and_weights(c, beta)
    h = cross_corr_w_a(w, a)
    dot_ha = float(np.dot(h, a))
    grad = 4.0 * n * a * (h - dot_ha)
    score = 2.0 * n * L
    return grad, score


def make_linear_profile(n, eta=1.5):
    """Monotone linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    return np.maximum(a, 1e-12) / a.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Anti-symmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    return np.maximum(a, 1e-12) / a.sum()


def make_power_profile(n, p=2.0):
    """Power-law profile."""
    t = np.linspace(0.0, 1.0, n)
    a = t ** p + 1e-8
    return a / a.sum()


def make_cosine_ramp(n, alpha=1.0):
    """Cosine-modulated ramp."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * t + 0.3 * np.sin(np.pi * t)
    return np.maximum(a, 1e-12) / a.sum()


def interpolate_seq(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = a.size
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    return a_new / a_new.sum()


def adam_logspace(z0, n, beta_init, deadline, lr=0.5, beta_decay=0.999,
                  restart_patience=3000, restart_amp=0.2, momentum_b1=0.9,
                  momentum_b2=0.999, eps=1e-8):
    """Adam optimizer in log-space with adaptive beta."""
    z = np.asarray(z0, dtype=np.float64).copy()
    dim = z.size
    m = np.zeros(dim)
    v = np.zeros(dim)
    t_adam = 0
    beta = beta_init

    a = z_to_seq(z)
    c = full_convolve_fft(a)
    L, w = soft_max_and_weights(c, beta)
    score = 2.0 * n * L

    best_z = z.copy()
    best_score = score
    best_a = a.copy()
    no_imp = 0
    base_lr = lr

    it = 0
    while time.monotonic() < deadline:
        if it % 200 == 0:
            beta = max(1e-8 / n, beta * beta_decay)

        grad, _ = compute_grad_z(z, n, beta)

        if not np.all(np.isfinite(grad)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            it += 1
            continue

        t_adam += 1
        m = momentum_b1 * m + (1.0 - momentum_b1) * grad
        v = momentum_b2 * v + (1.0 - momentum_b2) * (grad * grad)
        m_hat = m / (1.0 - momentum_b1 ** t_adam)
        v_hat = v / (1.0 - momentum_b2 ** t_adam)
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if not np.all(np.isfinite(z_new)):
            z = best_z.copy()
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            it += 1
            continue

        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - float(z_new.mean())

        a_new = z_to_seq(z_new)
        c_new = full_convolve_fft(a_new)
        L_new, w_new = soft_max_and_weights(c_new, beta)
        score_new = 2.0 * n * L_new

        if np.isfinite(score_new) and score_new < best_score - 1e-12:
            best_score = score_new
            best_z = z_new.copy()
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.05, lr * 0.95)

        z = z_new
        a = a_new
        c = c_new

        if no_imp > restart_patience:
            z = best_z + restart_amp * np.random.randn(dim)
            z = z - float(z.mean())
            z = np.clip(z, -80.0, 80.0)
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            lr = base_lr
            no_imp = 0
            restart_amp *= 0.6
            beta = max(1e-8 / n, beta_init * 0.5)

        it += 1

    return best_z, best_a, best_score


def lbfgs_polish(z0, n, beta, max_iter=2000):
    """L-BFGS-B polish in log-space."""
    def obj(z):
        a = z_to_seq(z)
        c = full_convolve_fft(a)
        L, _ = soft_max_and_weights(c, beta)
        return 2.0 * n * L

    def grad(z):
        g, _ = compute_grad_z(z, n, beta)
        return g

    try:
        res = minimize(obj, z0.copy(), method='L-BFGS-B', jac=grad,
                       options={'maxiter': max_iter, 'gtol': 1e-12,
                                'maxfun': 5000, 'iprint': 0})
        if res.fun < np.inf:
            a = z_to_seq(res.x)
            return a, float(res.fun)
    except Exception:
        pass
    return None, np.inf
