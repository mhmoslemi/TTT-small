import math
import numpy as np

def _conv_full(a):
    """Full linear convolution using direct method (exact)."""
    return np.convolve(a, a, mode='full')

def _conv_fft(a, m=None):
    """Full linear convolution using FFT (fast for large n)."""
    n = len(a)
    if m is None:
        m = 1
        while m < 2*n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _score_fast(a):
    """Fast approximate score using FFT convolution."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = _conv_fft(a)
    return 2.0 * n * c.max() / (S * S)

def _smoothed_max_grad(c, L, tau):
    """Compute smooth approximation of max(c) and its gradient w.r.t. c."""
    L = min(L, len(c))
    order = np.argpartition(-c, L-1)[:L]
    vals = c[order]
    w = np.exp(vals / tau)
    w = w / w.sum()
    smooth_max = np.dot(w, vals)
    return smooth_max, w, order
