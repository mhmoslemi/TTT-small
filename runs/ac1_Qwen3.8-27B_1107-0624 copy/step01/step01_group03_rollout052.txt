```python
def smooth_max_grad(s, alpha):
    """Compute smooth max of conv(s,s) and its gradient."""
    n = len(s)
    conv = np.convolve(s, s)
    # Smooth max
    shifted = alpha * conv - alpha * conv.max()
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    smooth_val = conv.max() + np.log(exp_shifted.sum()) / alpha
    
    # Gradient
    w_nv = weights[:n]
    part1_full = np.correlate(w_nv, s, mode='full')
    part1 = part1_full[:n][::-1]
    part2 = np.convolve(s, w_nv, mode='full')[:n]
    grad = part1 + part2
    
    return smooth_val, grad, conv

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)

        return np.ones(n) / n
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
```

```python
"""
Hybrid smooth-max gradient descent + LP-guided refinement to minimize
2n*max(conv(s,s))/(sum(s))^2 for non-negative sequences.
Uses FFT-based convolution, adaptive smooth-max parameter, momentum,
and periodic LP direction-finding for local refinement.
"""
import numpy as np
from scipy.optimize import linprog
import time


def fast_score(s):
    """Compute objective without external calls."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def smooth_max_and_grad(s, alpha):
    """Compute smooth max of conv(s,s) and gradient wrt s."""
    n = len(s)
    conv = np.convolve(s, s)
    shifted = alpha * (conv - conv.max())
    max_exp = np.max(shifted)
    exp_shifted = np.exp(shifted - max_exp)
    total_exp = exp_shifted.sum()
    weights = exp_shifted / total_exp
    smooth_val = conv.max() + (max_exp + np.log(total_exp)) / alpha

    w_nv = weights[:n]
    part1_full = np.correlate(w_nv, s, mode='full')
    part1 = part1_full[:n][::-1]
    part2 = np.convolve(s, w_nv, mode='full')[:n]
    grad = part1 + part2
    return smooth_val, grad, conv, weights


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def gradient_descent_smooth_max(s0, deadline, alpha_start=3.0, alpha_end=80.0,
                                lr=0.005, beta_mom=0.85, max_iters=100000,
                                perturb_every=500, perturb_scale=0.01,
                                rng=None):
    """Smooth-max projected gradient descent with momentum."""
    n = len(s0)
    s = s0.astype(np.float64).copy()
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    momentum = np.zeros(n)
    best_s = s.copy()
    best_sc = fast_score(s)
    no_improve = 0
    lr_current = lr

    if rng is None:
        rng = np.random.default_rng(0)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(it / max_iters, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        smooth_val, grad_conv, conv, weights = smooth_max_and_grad(s, alpha)
        S = s.sum()

        grad_obj = (2.0 * n / (S * S)) * grad_conv - (4.0 * n * smooth_val / (S * S * S)) * np.ones(n)

        momentum = beta_mom * momentum - lr_current * grad_obj
        s_new = s + momentum
        s_new = project_simplex(s_new)

        new_sc = fast_score(s_new)
        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_s = s_new.copy()
            no_improve = 0
            lr_current = min(lr_current * 1.05, lr * 3.0)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr_current = max(lr_current * 0.7, lr * 0.1)

        s = s_new

        if it > 0 and it % perturb_every == 0 and no_improve > 100:
            idx = rng.integers(0, n, size=n // 4)
            s[idx] += perturb_scale * rng.standard_normal(n // 4)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            no_improve = 0
            lr_current = lr

    return best_s, best_sc


def build_c

onv_matrix_sparse(s, top_k=200):
    """Build constraint matrix for LP using only top-k active constraints."""
    n = len(s)
    conv = np.convolve(s, s)
    M = conv.max()
    # Get indices of constraints closest to M
    sorted_idx = np.argsort(conv)[::-1]
    active_idx = sorted_idx[:top_k]

    A_rows = []
    for k in active_idx:
        row = np.zeros(n)
        # conv[k] = sum_i s[i]*s[k-i], gradient wrt b is:
        # d(conv[s,b][k])/db[j] = s[k-j] if valid + s[j-k] if valid
        for j in range(min(n, k + 1)):
            if k - j >= 0 and k - j < n:
                row[j] += s[k - j]
            if j - k >= 0 and j - k < n:
                row[j] += s[j - k]
        A_rows.append(row)
    return np.array(A_rows), M, active_idx


def lp_direction(s, deadline, top_k=300, time_per_lp=3.0):
    """Find LP direction: maximize sum(b) s.t. conv(s,b)[k] <= M, b >= 0."""
    if time.monotonic() > deadline:
        return None
    n = len(s)
    conv = np.convolve(s, s)
    M = conv.max()

    # Build constraint matrix (only top-k constraints)
    sorted_idx = np.argsort(conv)[::-1]
    active_idx = sorted_idx[:top_k]

    A = np.zeros((len(active_idx), n))
    for ii, k in enumerate(active_idx):
        for j in range(n):
            idx1 = k - j
            idx2 = j - k
            if 0 <= idx1 < n:
                A[ii, j] += s[idx1]
            if 0 <= idx2 < n:
                A[ii, j] += s[idx2]

    c = -np.ones(n)
    b_ub = np.full(len(active_idx), M)
    bounds = [(0.0, None)] * n

    lp_deadline = min(deadline, time.monotonic() + time_per_lp)
    if time.monotonic() > lp_deadline:
        return None

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_per_lp, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-14:
                return b / b.sum()
    except Exception:
        pass
    return None


def lp_iterate(s0, deadline, max_iters=200, top_k=300):
    """LP-guided local search iterations."""
    n = len(s0)
    s = s0.astype(np.float64).copy()
    s = s / s.sum()
    best_s = s.copy()
    best_sc = fast_score(s)
    no_imp = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break

        b = lp_direction(s, deadline, top_k=top_k)
        if b is None:
            break

        # Line search
        best_t = 0.0
        best_ts = best_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-10:
                continue
            cand = cand / cand.sum()
            ts = fast_score(cand)
            if ts < best_ts - 1e-12:
                best_ts = ts
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 15:
                break
            # Small random perturbation
            s = s + 0.002 * np.random.default_rng(it).standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue

        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        cur_sc = fast_score(s)
        no_imp = 0

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_s = s.copy()

    return best_s, best_sc


def try_parameterized(n, rng):
    """Generate diverse parameterized starting points."""
    results = []

    # Anti-symmetric ramp: high at start, low at end
    for t in np.linspace(0.1, 0.9, 9):
        s = np.zeros(n)
        # Two-level: high for first t*n, low for rest
        k = int(t * n)
        for ratio in [2.0, 3.0, 4.0, 5.0, 8.0]:
            s2 = np.zeros(n)
            s2[:k] = ratio
            s2[k:] = 1.0
            s2 = s2 / s2.sum()
            results.append((fast_score(s2), s2.copy()))

    # Exponential decay from one end
    for alpha in [2.0, 4.0, 6.0, 8.0, 12.0]:
        s = np.exp(-alpha * np.arange(n) / n)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))
        s2 = np.exp(-alpha * (n - 1 - np.arange(n)) / n)
        s2 = s2 / s2.sum()
        results.append((fast_score(s2), s2.copy()))

    # Biased distribution
    for mu in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            x = np.arange(n) / n
            s = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            s = s / s.sum()
            results.append((fast_score(s), s.copy()))

    # Power law
    for p in [2.0, 3.0, 4.0, 5.0]:
        s = (np.arange(n) + 1) ** (-p)
        s = s / s.sum()
        results.append((fast_score(s), s.copy()))
        s2 = (n - np.arange(n)) ** (-p)
        s2 = s2 / s2.sum()
        results.append((fast_score(s2), s2.copy()))

    # Random starts
    for _ in range(20):
        r = rng.random(n)
        r = r ** rng.uniform(0.3, 3.0)
        r = np.maximum(r, 0.01)
        r = r / r.sum()
        results.append((fast_score(r), r.copy()))

    results.sort(key=lambda x: x[0])
    return results


def coordinate_descent(s, deadline, max_rounds=50):
    """Basic coordinate descent: optimize each element in turn."""
    n = len(s)
    s = s.copy()
    best_sc = fast_score(s)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 2:
                return s, best_sc
            old_val = s[j]
            # Try a few values around old_val
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5, 0.0]
            best_lv = old_val
            best_lscore = fast_score(s)
            for lv in candidates:
                lv = max(0.0, min(1000.0, lv))
                s[j] = lv
                sc = fast_score(s)
                if sc < best_lscore:
                    best_lscore = sc
                    best_lv = lv
            s[j] = best_lv
        # Renormalize
        s = s / s.sum()
        sc = fast_score(s)
        if sc < best_sc:
            best_sc = sc
            improved = True
        if not improved and rnd > 10:
            break

    return s, best_sc
```
