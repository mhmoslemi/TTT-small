"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 using LP-based iteration with full
convolution constraints, combined with Adam gradient polish.

Key improvements over previous approach:
- Full constraint LP (no cutting planes) for n <= 2048
- Adaptive line search
- Focused time allocation on best lengths
- Multiple diverse starting points

import time
import numpy as np
import scipy.optimize

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(max(a.size, 16))
        s = float(a.sum())
    return a / s

def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-6:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

def direct_score(p):
    """Use direct convolve for small n (more accurate)."""
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-6:
        return np.inf
    c = np.convolve(a, a)
    maxc = float(np.max(c))
    return float(2.0 * n * maxc / (s * s))

def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return [1.0 / 16] * 16
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]

def greedy_normalized(n):
    n = max(int(n), 2)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return normalize_nonneg(a)

def power_normalized(n, s):
    n = max(int(n), 2)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

def build_conv_mat(f):
    """
    Build the convolution matrix A such that (f*g) = A @ g.
    A has shape (2n-1, n).
    A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0.
    """
    n = f.size
    A = np.zeros((2*n - 1, n), dtype=np.float64)
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # A[k, lo:hi+1] = f[k-hi : k-lo+1][::-1]
            A[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return A

def lp_solve(f, M, time_limit):
    """
    Solve: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns g (optimal) or None.
    """
    n = f.size
    A = build_conv_mat(f)
    b = np.full(2*n - 1, M, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    
    try:
        res = scipy.optimize.linprog(
            cobj, A_ub=A, b_ub=b,
            bounds=(0.0, None),
            method='highs',
            options={'time_limit': float(time_limit), 'presolve': True}
        )
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0.0, None)

def lp_iterate(p0, budget_s, verbose=True):
    """
    Main LP-based iteration.
    p0: initial sequence (will be normalized internally)
    """
    p = normalize_nonneg(p0)
    n = p.size
    S0 = np.sqrt(2.0 * n)
    f = p * S0
    
    # Compute initial score
    def score_fn(x):
        s = float(x.sum())
        if s < 1e-12:
            return np.inf
        xn = x / s
        # Score is scale invariant, so compute with sum=1
        L = next_power_of_two(2*n - 1)
        P = np.fft.rfft(xn, L)
        c = np.fft.irfft(P*P, L)[:2*n-1].real
        maxc = float(np.max(np.maximum(c, 0.0)))
        return 2.0 * n * maxc
    
    cur_score = score_fn(f)
    best_f = f.copy()
    best_score = float(cur_score)
    
    end = time.monotonic() + budget_s
    it = 0
    stall = 0
    
    while time.monotonic() < end:
        it += 1
        if time.monotonic() > end - 2.0:
            break
        
        rem = end - time.monotonic()
        if rem < 3.0:
            break
        
        # Compute M
        c = np.convolve(f, f)
        M = float(np.max(np.maximum(c, 0.0)))
        
        if M <= 0 or not np.isfinite(M):
            break
        
        # Solve LP
        tl = min(30.0, max(2.0, rem * 0.25))
        g = lp_solve(f, M, tl)
        
        if g is None:
            break
        
        sg = float(g.sum())
        if sg <= 1e-12:
            break
        
        # Line search: try various step sizes
        accepted = None
        accepted_score = np.inf
        for t in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
            fn = (1.0 - t) * f + t * g
            fn = np.maximum(fn, 0.0)
            sc = score_fn(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()
        
        if accepted is None:
            break
        
        # Check if improvement
        if accepted_score < best_score:
            f = accepted
            best_f = f.copy()
            best_score = accepted_score
            stall = 0
            if verbose and (it <= 10 or it % 10 == 0):
                print(f"  LP n={n} it={it}: score={best_score:.6f}")
        else:
            stall += 1
            f = accepted  # still move, but track stall
            if stall >= 8:
                break
    
    # Return normalized
    best_norm = normalize_nonneg(best_f)
    return best_norm, best_score

def adam_polish(p0, budget_s, sigma=5.0):
    """Adam gradient descent on smoothed max conv."""
    p = normalize_nonneg(p0)
    n = p.size
    x = np.log(p + 1e-300)
    x -= x.mean()
    
    lr = 0.02 if n > 512 else 0.03
    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    L = next_power_of_two(3*n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)
    
    start = time.monotonic()
    end = start + budget_s
    
    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = sigma + (50.0 - sigma) * np.exp(-3.0 * frac)
        
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            break
        
        f_now = 2.0 * n * maxc  # sum(p)=1
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()
        
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2*n-1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz
        
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n-1:2*n-1][::-1].real
        C = np.maximum(C, 0.0)
        
        grad_p = (4.0 * n) * C
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)
        
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu**t)
        vhat = v / (1.0 - nu**t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()
        
        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se
    
    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f

    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def lp_solve_constraints(f, M, time_limit):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0. Uses all constraints."""
    n = f.size
    A = build_conv_mat(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    cobj = -np.ones(n, dtype=np.float64)
    try:
        res = scipy.optimize.linprog(
            cobj, A_ub=A, b_ub=b,
            bounds=(0.0, None),
            method='highs',
            options={'time_limit': float(time_limit), 'presolve': True}
        )
    except Exception:
        return None
    if not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
        return None
    return np.clip(res.x, 0.0, None)


def lp_iterate(p0, budget_s, verbose=True):
    """LP-based iteration with full constraints and line search."""
    n = p0.size
    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p0) * S0

    cur_score = score_sum_one(f / S0)
    best_f = f.copy()
    best_score = float(cur_score)

    end = time.monotonic() + budget_s
    it = 0
    stall = 0

    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 3.0:
            break

        # Compute M using direct convolve for accuracy
        c = np.convolve(f, f)
        M = float(np.max(np.maximum(c, 0.0)))
        if M <= 0 or not np.isfinite(M):
            break

        # Solve LP
        tl = min(45.0, max(3.0, rem * 0.3))
        g = lp_solve_constraints(f, M, tl)
        if g is None:
            break
        sg = float(g.sum())
        if sg <= 1e-12:
            break

        # Line search
        accepted = None
        accepted_score = np.inf
        for t in [1.0, 0.7, 0.5, 0.35, 0.2, 0.12, 0.06, 0.03]:
            fn = (1.0 - t) * f + t * g
            fn = np.maximum(fn, 0.0)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            sc = score_sum_one(fn / sn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None:
            break

        if accepted_score < best_score:
            f = accepted
            best_f = f.copy()
            best_score = accepted_score
            stall = 0
            if verbose and (it <= 15 or it % 10 == 0):
                print(f"  LP n={n} it={it}: score={best_score:.6f}")
        else:
            stall += 1
            f = accepted
            if stall >= 12:
                break

    best_norm = normalize_nonneg(best_f)
    return best_norm, best_score


def adam_polish(p0, budget_s, sigma_init=5.0, sigma_end=60.0):
    """Adam gradient descent on smoothed max convolution."""
    p = normalize_nonneg(p0)
    n = p.size
    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.025
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.1, budget_s)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = sigma_init + (sigma_end - sigma_init) * np.exp(-3.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = 2.0 * n * maxc
        if f_now < best_f:
            best_f = float(f_now)
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -40.0, 40.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = float(final_f)
        best_p = p.copy()
    return best_p, best_f


def is_better_candidate(p, score, best_ref):
    """Check if p is better than best_ref (a list [vec, score])."""
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0 or not np.isfinite(score):
        return best_ref
    if score < best_ref[1]:
        best_ref[0] = a.copy()
        best_ref[1] = float(score)
    return best_ref


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.5, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed))

    best_ref = [None, np.inf]
    best_by_n = {}

    # Phase 0: Load previous best
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                s = float(arr.sum())
                if s > 0.01:
                    arr_norm = arr / s
                    sc = actual_score_fft(arr_norm)
                    print(f"Loaded height_sequence_1: n={arr.size}, score={sc:.6f}")
                    best_ref = is_better_candidate(arr_norm, sc, best_ref)
                    best_by_n[arr.size] = (sc, arr_norm.copy())
        except Exception as ex:
            print(f"Failed to load height_sequence_1: {ex}")

    # Phase 1: Quick baselines
    base_lengths = [128, 192, 256, 384, 512, 768, 1024]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        inits = [
            greedy_normalized(n),
            power_normalized(n, 0.40),
            power_normalized(n, 0.50),
            power_normalized(n, 0.60),
            power_normalized(n, 0.70),
            raised_cosine_normalized(n, 1.0),
            raised_cosine_normalized(n, 1.5),
            raised_cosine_normalized(n, 2.0),
        ]
        for p in inits:
            sc = actual_score_fft(p)
            best_ref = is_better_candidate(p, sc, best_ref)
            old = best_by_n.get(n)
            if old is None or sc < old[0]:
                best_by_n[n] = (sc, p.copy())

    print(f"Phase 1 done. Best: {best_ref[1]:.6f} at n={best_ref[0].size if best_ref[0] is not None else 0}")

    # Phase 2: LP iteration on promising lengths
    lp_lengths = []
    if best_ref[0] is not None:
        bn = best_ref[0].size
        if bn not in lp_lengths:
            lp_lengths.append(bn)
    for n in [256, 384, 512, 768, 1024]:
        if n not in lp_lengths:
            lp_lengths.append(n)

    # Sort: prioritize the length with best current score
    def length_priority(n):
        rec = best_by_n.get(n)
        if rec is not None:
            return (rec[0], n)
        return (99.0, n)

    lp_lengths.sort(key=length_priority)

    # Allocate LP time proportionally (shorter = more iterations possible)
    total_lp_time = min(1000.0, max(150.0, 0.55 * (deadline - time.monotonic())))
    n_lp = len(lp_lengths)
    # Weight: give more time to better-performing lengths
    weights = []
    for n in lp_lengths:
        rec = best_by_n.get(n)
        sc = rec[0] if rec is not None else 2.0
        w = 1.0 / max(sc, 1.01)
        weights.append(w)
    total_w = sum(weights)
    allocs = [total_lp_time * w / total_w for w in weights]

    for i, n in enumerate(lp_lengths):
        rem = deadline - time.monotonic()
        if rem < 40.0:
            break

        alloc = min(allocs[i], rem - 10.0)
        if alloc < 15.0:
            continue

        # Get starting point
        rec = best_by_n.get(n)
        if rec is not None:
            start_p = rec[1].copy()
        else:
            start_p = greedy_normalized(n)

        # Run LP
        lp_alloc = alloc * 0.65
        if lp_alloc > 10.0:
            p_out, f_out = lp_iterate(start_p, lp_alloc, verbose=True)
            sc = actual_score_fft(p_out)
            best_ref = is_better_candidate(p_out, sc, best_ref)
            old = best_by_n.get(n)
            if old is None or sc < old[0]:
                best_by_n[n] = (sc, p_out.copy())

        # Adam polish
        rem2 = deadline - time.monotonic()
        if rem2 > 20.0:
            adam_alloc = min(alloc * 0.35, rem2 - 5.0)
            if adam_alloc > 8.0:
                start_adam = best_ref[0].copy() if best_ref[0] is not None and best_ref[0].size == n else start_p
                p_adam, f_adam = adam_polish(start_adam, adam_alloc, sigma_init=5.0, sigma_end=50.0)
                sc = actual_score_fft(p_adam)
                best_ref = is_better_candidate(p_adam, sc, best_ref)
                old = best_by_n.get(n)
                if old is None or sc < old[0]:
                    best_by_n[n] = (sc, p_adam.copy())

    print(f"Phase 2 done. Best: {best_ref[1]:.6f} at n={best_ref[0].size if best_ref[0] is not None else 0}")

    # Phase 3: Diverse perturbation + re-optimization
    rem = deadline - time.monotonic()
    if rem > 60.0 and best_ref[0] is not None:
        sigmas = [0.03, 0.06, 0.10, 0.15]
        for sigma in sigmas:
            rem = deadline - time.monotonic()
            if rem < 25.0:
                break
            noise = np.exp(rng.normal(0.0, sigma, best_ref[0].size))
            p0 = normalize_nonneg(best_ref[0] * noise)
            alloc = min(150.0, rem * 0.25)
            if alloc < 10.0:
                break
            # Try LP first, then Adam
            lp_alloc = alloc * 0.5
            adam_alloc = alloc * 0.5
            if lp_alloc > 8.0:
                p_lp, f_lp = lp_iterate(p0, lp_alloc, verbose=False)
                sc = actual_score_fft(p_lp)
                best_ref = is_better_candidate(p_lp, sc, best_ref)
                if best_ref[0].size != best_ref[1] and False:
                    pass
            if rem - time.monotonic() > 8.0:
                p_adam, f_adam = adam_polish(best_ref[0], adam_alloc, sigma_init=8.0, sigma_end=40.0)
                sc = actual_score_fft(p_adam)
                best_ref = is_better_candidate(p_adam, sc, best_ref)

    # Phase 4: Final LP push on the best
    rem = deadline - time.monotonic()
    if rem > 60.0 and best_ref[0] is not None:
        p_out, f_out = lp_iterate(best_ref[0], min(300.0, rem - 15.0), verbose=True)
        sc = actual_score_fft(p_out)
        best_ref = is_better_candidate(p_out, sc, best_ref)

    # Phase 5: Final Adam polish
    rem = deadline - time.monotonic()
    if rem > 15.0 and best_ref[0] is not None:
        p_adam, f_adam = adam_polish(best_ref[0], min(120.0, rem - 5.0), sigma_init=20.0, sigma_end=80.0)
        sc = actual_score_fft(p_adam)
        best_ref = is_better_candidate(p_adam, sc, best_ref)

    if best_ref[0] is None:
        best_ref[0] = greedy_normalized(256)
        best_ref[1] = actual_score_fft(best_ref[0])

    print(f"FINAL: score={best_ref[1]:.6f}, n={best_ref[0].size}")

    # Validate
    rem = deadline - time.monotonic()
    if rem > 3.0:
        try:
            seq = scale_output(best_ref[0], 10.0)
            gscore = evaluate_sequence(seq)
            if np.isfinite(gscore):
                print(f"Validator confirms: {gscore:.6f}")
        except Exception:
            pass

    return scale_output(best_ref[0], 10.0)
