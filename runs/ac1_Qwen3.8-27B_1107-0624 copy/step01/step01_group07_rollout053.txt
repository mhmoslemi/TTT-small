import numpy as np

from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def make_barker_like(n):

    # Add a secondary modulation
    mod = 0.5 + 0.5 * np.cos(4 * np.pi * t + np.pi / 3)
    return base * mod + 0.1


def make_random_smooth(n, rng):
    """Generate a random smooth sequence."""
    # Random frequencies and amplitudes
    n_freqs = 3
    freqs = rng.uniform(0.5, 3.0, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.5, 1.0, n_freqs)
    t = np.arange(n) / n
    seq = np.ones(n)
    for i in range(n_freqs):
        seq += amps[i] * np.cos(2 * np.pi * freqs[i] * t + phases[i])
    return np.clip(seq, 0.01, None)


def upsample(a, factor=2):
    """Upsample a sequence by inserting interpolated values."""
    n = len(a)
    new_n = n * factor
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)


def downsample(a, factor=2):
    """Downsample by averaging blocks."""
    n = len(a)
    new_n = n // factor
    a = a[:new_n * factor]
    return a.reshape(new_n, factor).mean(axis=1)

"""
Hybrid LP + smoothed gradient optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smooth gradient for fast exploration, LP-based refinement for
precision, adaptive M targets, multi-n search, and periodic restarts to escape
local minima. Builds upon parent sequence with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def smooth_grad_step(a, T, lr):
    """One FFT-based gradient step on softmax-smoothed max of auto-convolution."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    weights = np.exp((c - c_max) / T)
    weights /= weights.sum()

    # grad[i] = 2 * sum_m a[m] * weights[m+i]  (cross-correlation)
    N = 1
    while N < 2 * n:
        N *= 2
    A_fft = np.fft.rfft(a, N)
    W_fft = np.fft.rfft(weights, N)
    cc = np.fft.irfft(np.conj(A_fft) * W_fft, N)
    grad = 2.0 * cc[:n]

    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new / a_new.sum()
    return a_new


def optimize_smooth(a, deadline, rng):
    """Smooth gradient descent with adaptive temperature and learning rate."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_score = compute_objective(a)
    T = max(0.05, c_peak_scale(a))
    lr = 0.3
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        a = smooth_grad_step(a, T, lr)
        sc = compute_objective(a)
        if sc < best_score - 1e-13:
            best_a = a.copy()
            best_score = sc
            stall = 0
            lr = min(0.5, lr * 1.01)
        else:
            stall += 1
            lr = max(0.01, lr * 0.95)
            T = min(0.5, T * 1.02)

                    a_new /= a_new.sum()
                    a = a_new
                    stall = 0
                if stall > 400:

    return best_a, best_score


def c_peak_scale(a):
    """Estimate a scale for temperature based on peak-to-mean ratio."""
    c = np.convolve(a, a, mode='full')
    return max(0.02, (c.max() - c.mean()) / max(c.mean(), 1e-15) * 0.01)


def optimize_lp(a, deadline, rng, verbose=False):

    n = len(a)

    best_score = compute_objective(a)
    step = 0.2
    stall = 0
    it = 0
    m_shrink = 1.0

        c = np.convolve(a, a, mode='full')

        g = g / g.sum()

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def make_flattop(n):
    """Flattop window - has very flat top, good for reducing peaks."""
    t = np.arange(n) / (n - 1.0)
    return (0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5) / 1.0)) ** 2


def make_barker_smooth(n, rng):
    """Smooth sequence designed to have low auto-correlation side lobes."""
    t = np.arange(n) / n
    # Multiple frequency components to flatten auto-correlation
    seq = np.ones(n) * 0.3
    for k in range(1, 5):
        seq += 0.15 * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    return np.clip(seq, 0.01, None)


def make_random_smooth(n, rng):
    """Random smooth non-negative sequence."""
    n_freqs = 4
    freqs = rng.uniform(0.3, 4.0, n_freqs)
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    amps = rng.uniform(0.3, 1.0, n_freqs)
    t = np.arange(n) / n
    seq = np.ones(n) * 0.5
    for i in range(n_freqs):
        seq += amps[i] * np.cos(2 * np.pi * freqs[i] * t + phases[i])
    return np.clip(seq, 0.001, None)


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, n)
    t_new = np.linspace(0, 1, new_n)
    return np.interp(t_new, t_old, a)
