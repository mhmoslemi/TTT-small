def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    c = convolve_fft(a)
    return float(2.0 * n * np.max(c) / (S * S))

def smooth_obj_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exps / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S ** 3)
    return J, grad

    """LP-based reset: find g maximizing sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    # Build constraint matrix A (sparse)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is the matrix for linear convolution
    
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    # For each column j (variable g[j]), the non-zero entries are at rows j, j+1, ..., j+n-1
    # with values a[0], a[1], ..., a[n-1]
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        rz = a > 0  # Only add non-zero entries
        if not rz.any():
            continue
        idx = np.where(rz)[0]
        rows.extend((j + idx).tolist())
        cols.extend([j] * len(idx))
        vals.extend(a[idx].tolist())

        g = np.maximum(g, 0)
        S = np.sum(g)
        if S > 0:

        return g
    return None
