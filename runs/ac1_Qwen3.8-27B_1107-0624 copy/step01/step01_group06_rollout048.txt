def fft_convolve(a):

    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)

def compute_J(a):

    n = len(a)
    S = np.sum(a)
    if S < 0.01:

    c = fft_convolve(a)
    return 2.0 * n * np.max(c) / (S * S)


def circular_autocorr(a):
    """Compute circular autocorrelation via FFT. Returns real array of length n."""
    n = len(a)
    A = np.fft.fft(a)
    P = A * np.conj(A)
    r = np.fft.ifft(P).real
    return r


def spectral_flatten(a, budget_s, seed, alpha=1.0):
    """
    Spectral flattening: iteratively modify the power spectrum to reduce
    the peak circular autocorrelation, while maintaining a >= 0.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    S0 = np.sum(a)
    a = a.copy()
    if S0 > 0:
        a = a * (S0 / S0)  # keep as is, we'll normalize at end

    best_a = a.copy()
    best_J = compute_J(a)
    it = 0
    stall = 0
    stall_thresh = best_J

    while time.monotonic() < deadline:
        S = np.sum(a)
        if S < 0.01:
            break

        # Compute circular autocorrelation
        A = np.fft.fft(a)
        P = A * np.conj(A)
        r = np.fft.ifft(P).real
        r_max = np.max(r)
        tau_max = np.argmax(r)

        # Gradient of r[tau_max] w.r.t. P[k]: (1/n)*exp(2*pi*i*k*tau_max/n)
        # To decrease r[tau_max], decrease P in direction Re(exp(2*pi*i*k*tau_max/n))
        # = cos(2*pi*k*tau_max/n)

        grad_dir = np.cos(2.0 * np.pi * k_freq * tau_max / n)

        # Weight lags above threshold

        active_lags = np.where(r > r_threshold)[0]

        P_new = np.maximum(P_new, 0.0)  # power spectrum must be non-negative

        phase = A / (np.abs(A) + 1e-30)

        a_new = np.fft.ifft(A_new).real

def multi_lag_gradient(a, K=30, tau_smoothing=0.0):
    """
    Compute gradient using top-K lags of the convolution.
    Returns (J, grad) where grad is the subgradient of J.
    """
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c = fft_convolve(a)
    c_max = np.max(c)
    J = 2.0 * n * c_max / (S * S)

    # Get top K lags
    if K >= len(c):
        active = np.arange(len(c))
        weights = np.ones(len(c))
    else:
        top_idx = np.argsort(c)[::-1][:K]
        # Weight by how much above the K-th value
        threshold = c[top_idx[-1]]
        active = top_idx
        weights = np.maximum(0, c[active] - threshold) + 1e-10
        weights = weights / np.sum(weights)

    # Compute subgradient: for each lag k, dc[k]/da[j] = a[j] + a[k-j]
    # Weighted combination
    grad_c = np.zeros(n)
    for i, k in enumerate(active):
        w = weights[i]
        # a[j] + a[k-j] for j=0..n-1, with a[m]=0 if m out of [0,n-1]
        # a[j] contributes: just a[j]
        grad_c += w * a
        # a[k-j] contributes: a[k-j] where 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # a[k-j] for j in [lo, hi] = a[hi] down to a[lo] (reversed)
            grad_c[lo:hi + 1] += w * a[hi:lo - 1:-1] if lo > 0 else w * a[hi::-1]

    # Gradient of J = 2n*c_max/S^2 w.r.t. a[j]
    # = (2n/S^2)*dc_max/da[j] - (4n*c_max/S^3)*S
    # = (2n/S^2)*grad_c[j] - (4n*c_max/S^2)
    grad = (2.0 * n / (S * S)) * grad_c - (4.0 * n * c_max / (S * S))
    return J, grad

        m_hat = m / (1.0 - beta1 ** t)

def coordinate_descent(a_init, budget_s, seed):
    """
    Coordinate descent: optimize individual elements (or small groups)
    by trying to reduce the max convolution.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    S0 = np.sum(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    c = fft_convolve(a)
    it = 0
    no_improve = 0

    # Precompute which lags are "active" (near max)
    c_max = np.max(c)

    while time.monotonic() < deadline:
        # Recompute convolution periodically
        if it % 10 == 0:
            c = fft_convolve(a)
            c_max = np.max(c)
            S = np.sum(a)
            best_J = 2.0 * n * c_max / (S * S)

        # Pick a random coordinate (or small group)
        group_size = int(rng.integers(1, 4))
        start = int(rng.integers(0, n - group_size + 1))
        indices = np.arange(start, start + group_size)

        # For each coordinate in the group, find the optimal value
        # that minimizes the max convolution
        for idx in indices:
            # Current contribution to convolution
            # c[k] involves a[idx]*a[k-idx] and a[k-idx]*a[idx]
            # So changing a[idx] changes c[k] for all k where a[k-idx] is nonzero
            # i.e., k in [idx, idx+n-1]

            lo_k = idx
            hi_k = min(idx + n - 1, 2 * n - 2)

            if lo_k > hi_k:
                continue

            # Current c values in affected range
            c_affected = c[lo_k:hi_k + 1].copy()
            # The "partner" values: a[k-idx] for k in [lo_k, hi_k]
            partner = a[0:n]  # a[k-idx] for k=idx..idx+n-1 is a[0]..a[n-1]

            # Actually c[k] = sum_j a[j]*a[k-j], so the terms involving a[idx] are:
            # a[idx]*a[k-idx] (when j=idx) and a[k-idx]*a[idx] (when j=k-idx)
            # If idx == k-idx (k=2*idx), it's a[idx]^2, derivative is 2*a[idx]

            # (approximately, for the linear part)

            # We want max_k c[k]_new to be minimized.

            # and slope_k = 2*partner[k-idx]

            base = c_affected - 2.0 * a[idx] * partner
            slope = 2.0 * partner

            # Find x that minimizes max_k (base_k + slope_k * x)
            # The optimal x is where the highest and lowest "lines" cross
            # Simple approach: binary search on x
            # Or: the optimal x makes the max of (base_k + slope_k * x) as small as possible

            # For simplicity, use a few candidate values:

            # x = a[idx] (no change)
            # x that equalizes the top 2 affected lags

            if current_val < 1e-10:

            # Try: reduce a[idx] to balance the top affected lag with others
            # The most affected lag is where partner is largest

            # We want to reduce the max c to match the second-max
            target_reduction = c_max - np.sort(c)[::-1][min(5, len(c) - 1)]
            if target_reduction <= 0:

            max_partner = np.max(partner)
            if max_partner < 1e-10:
                continue

            delta = target_reduction / (2.0 * max_partner + 1e-10)
            new_val = max(0.0, current_val - delta)

            # Also try increasing slightly for "cold" coordinates
            if current_val < S0 / n * 0.5:

            # Evaluate

            S = np.sum(a)

            S = np.sum(a)

def make_hann(n):
    """Hann window."""
    x = np.arange(n)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))


def make_blackman(n):
    """Blackman window."""
    x = np.arange(n)
    N = n - 1
    return 0.42 - 0.5 * np.cos(2.0 * np.pi * x / N) + 0.08 * np.cos(4.0 * np.pi * x / N)


def make_optimal_taper(n, gamma=2.0):
    """Parabolic taper: (1 - (2x/n - 1)^gamma) clipped."""
    x = np.arange(n)
    t = 2.0 * x / (n - 1) - 1.0
    return np.maximum(0.0, 1.0 - np.abs(t) ** gamma)

        J = compute_J(arr)
        init_scores.append((J, name, arr))
        print(f"  init '{name}': J = {J:.6f}")

    init_scores.sort(key=lambda x: x[0])

    # Phase allocation
    remaining = total_deadline - time.monotonic()
    # Reserve 20s for final verification
    work_time = remaining - 20.0

    # Phase 1: Spectral flattening on top 3 inits (25% time)
    # Phase 2: Gradient descent on top 3 (45% time)
    # Phase 3: Coordinate descent on best (20% time)
    # Phase 4: Final GD on global best (10% time)

    phase1_time = work_time * 0.25
    phase2_time = work_time * 0.45
    phase3_time = work_time * 0.20
    phase4_time = work_time * 0.10

    # Phase 1: Spectral flattening
    print(f"\n=== Phase 1: Spectral Flattening ({phase1_time:.0f}s) ===")
    phase1_deadline = time.monotonic() + phase1_time
    top_inits = init_scores[:4]
    spectral_time_each = phase1_time / len(top_inits)

    for J_init, name, arr in top_inits:
        if time.monotonic() >= phase1_deadline:
            break
        budget = min(spectral_time_each, phase1_deadline - time.monotonic())
        if budget < 5:
            break
        print(f"  Spectral on '{name}' ({budget:.0f}s)")
        res_a, res_J = spectral_flatten(arr, budget, seed=seed + hash(name) % 100000, alpha=0.5)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST (spectral): {val:.6f} ***")
        else:
            print(f"    ({name} spectral: local={res_J:.4f}, verified={val:.4f})")

    # Phase 2: Gradient descent
    print(f"\n=== Phase 2: Gradient Descent ({phase2_time:.0f}s) ===")
    phase2_deadline = time.monotonic() + phase2_time

    # Run GD on: global best + top 2 inits
    gd_targets = [(best_global, 'global_best')]
    for J_init, name, arr in top_inits[:2]:
        gd_targets.append((arr, name))

    # Also add a version of the parent with slight modification
    mod_parent = parent * (1.0 + 0.1 * np.cos(2.0 * np.pi * np.arange(n) / n))
    mod_parent = np.maximum(mod_parent, 0)
    gd_targets.append((mod_parent, 'mod_parent'))

    gd_time_each = phase2_time / len(gd_targets)
    for arr, name in gd_targets:
        if time.monotonic() >= phase2_deadline:
            break
        budget = min(gd_time_each, phase2_deadline - time.monotonic())
        if budget < 5:
            break
        print(f"  GD on '{name}' ({budget:.0f}s)")
        res_a, res_J = run_gradient_descent(arr, budget, seed=seed + 999, K=50, lr_start=0.3)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST (GD): {val:.6f} ***")
        else:
            print(f"    ({name} GD: local={res_J:.4f}, verified={val:.4f})")

    # Phase 3: Coordinate descent
    print(f"\n=== Phase 3: Coordinate Descent ({phase3_time:.0f}s) ===")
    if time.monotonic() < total_deadline - 25:
        budget = min(phase3_time, total_deadline - time.monotonic() - 25)
        if budget > 10:
            print(f"  CD on best ({budget:.0f}s)")
            res_a, res_J = coordinate_descent(best_global, budget, seed=seed + 555)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"    *** NEW BEST (CD): {val:.6f} ***")
            else:
                print(f"    (CD: local={res_J:.4f}, verified={val:.4f})")

    # Phase 4: Final refinement
    print(f"\n=== Phase 4: Final Refinement ===")
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 10:
        # Try spectral again on the best found
        budget_spec = remaining * 0.4
        budget_gd = remaining * 0.4
        budget_cd = remaining * 0.2

        if budget_spec > 5:
            print(f"  Final spectral ({budget_spec:.0f}s)")
            res_a, res_J = spectral_flatten(best_global, budget_spec, seed=seed + 7777, alpha=0.3)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"    Refined (spectral): {val:.6f}")

        remaining = total_deadline - time.monotonic() - 15.0
        if remaining > 5:
            budget_gd = remaining * 0.5
            if budget_gd > 5:
                print(f"  Final GD ({budget_gd:.0f}s)")
                res_a, res_J = run_gradient_descent(best_global, budget_gd, seed=seed + 8888, K=80, lr_start=0.1)
                val = float(evaluate_sequence(res_a.tolist()))
                if np.isfinite(val) and val < best_global_score:
                    best_global = res_a.copy()
                    best_global_score = val
                    print(f"    Refined (GD): {val:.6f}")

        remaining = total_deadline - time.monotonic() - 15.0
        if remaining > 5:
            budget_cd = remaining * 0.6
            if budget_cd > 5:
                print(f"  Final CD ({budget_cd:.0f}s)")
                res_a, res_J = coordinate_descent(best_global, budget_cd, seed=seed + 9999)
                val = float(evaluate_sequence(res_a.tolist()))
                if np.isfinite(val) and val < best_global_score:
                    best_global = res_a.copy()
                    best_global_score = val
                    print(f"    Refined (CD): {val:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n=== FINAL SCORE: {best_global_score:.6f} ===")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
