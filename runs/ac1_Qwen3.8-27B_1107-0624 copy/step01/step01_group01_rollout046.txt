"""
LP-based peak reduction optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact LP subproblems to find descent directions, combined with multiple
diverse starting points, line search, and gradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    """Fast linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 96:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0:
        c = c - mn
    return c


def _score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _convolve_fft(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def _build_A(a):
    """Build the constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = a.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # A[k, j] = a[k-j] for j in [j_lo, j_hi]
            # k-j ranges from k-j_hi to k-j_lo
            A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def _solve_lp(a, M):
    """Solve: max 1^T b  s.t.  A(a) @ b <= M, b >= 0.
    Returns b* or None if infeasible.
    """
    n = a.size
    A = _build_A(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def _lp_step(a, rng):
    """One LP-based descent step. Returns (new_a, improved)."""
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return a, False
    a_norm = a / s

    c = _convolve_fft(a_norm)
    M = float(c.max())

    b = _solve_lp(a_norm, M)
    if b is None:
        return a, False

    b_sum = float(b.sum())
    if b_sum < 1e-10:
        return a, False

    b_dir = b / b_sum  # normalize to sum 1

    # Line search
    base_sc = _score(a_norm)
    best_t = 0.0
    best_sc = base_sc
    for t in (0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 0.1, 0.2, 0.4, 0.6):
        a_try = (1.0 - t) * a_norm + t * b_dir
        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc = _score(a_try)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    if best_t == 0.0:
        return a, False

    new_a = (1.0 - best_t) * a_norm + best_t * b_dir
    new_a = np.maximum(new_a, 0.0)
    s_new = float(new_a.sum())
    if s_new > 1e-10:
        new_a = new_a / s_new
    return new_a, True

            else:

def _soft_max_gradient_step(a, beta, lr):
    """One gradient step using soft-max approximation of the peak."""
    n = a.size
    c = _convolve_fft(a)
    mval = float(c.max())
    if mval < 1e-15:
        return a
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ex / s

    # Gradient of soft_max w.r.t. a: 2 * corr(w, a)
    N = _next_pow2(2 * n - 1)
    fw = rfft(w, N)
    fa = rfft(a[::-1], N)
    grad = irfft(fw * fa, N)[n - 1:2 * n - 1]
    grad *= 2.0

    # Logit-space update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= log_a.mean()
    log_a -= lr * grad / (np.abs(grad).max() + 1e-30)
    log_a = np.clip(log_a, -50, 50)
    log_a -= log_a.mean()

    new_a = np.exp(log_a)
    new_a = np.maximum(new_a, 0.0)
    s = float(new_a.sum())
    if s > 1e-10:
        new_a = new_a / s
    return new_a


def _gradient_polish(a, deadline, rng, lr=0.1):
    """Polish using soft-max gradient descent in logit space."""
    n = a.size
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _convolve_fft(a)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)

    best_a = a.copy()
    best_sc = _score(a)
    no_imp = 0
    step = lr

    while time.monotonic() < deadline:
        new_a = _soft_max_gradient_step(a, beta, step)
        sc = _score(new_a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = new_a.copy()
            a = new_a
            no_imp = 0
        else:
            a = best_a.copy()
            no_imp += 1
            if no_imp % 200 == 0:
                step = max(0.01, step * 0.95)

        beta = max(1e-12, beta * 0.999)

        if no_imp > 500:
            log_a = np.log(np.maximum(best_a, 1e-15))
            log_a += 0.3 * rng.normal(size=n)
            log_a -= log_a.mean()
            a = np.exp(np.clip(log_a, -50, 50))
            a = a / a.sum()
            beta = max(1e-10, 0.3 * _convolve_fft(a).max())
            step = lr
            no_imp = 0

    return best_a, best_sc


def _make_starting_profiles(n, rng):

    profiles = []
    t = np.linspace(0.0, 1.0, n)

    # Linear ramps
    for eta in (0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0):

        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Quadratic

        a = 1.0 + a1 * t + a2 * t * t
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Exponential
    for alpha in (1.0, 2.0, 3.0, -1.0, -2.0, 0.5, -0.5):
        a = np.exp(alpha * t)
        profiles.append(a / a.sum())

    # Bump (Gaussian-like)
    for center, width in ((0.3, 0.15), (0.7, 0.15), (0.5, 0.1), (0.5, 0.2), (0.2, 0.1), (0.8, 0.1)):
        a = np.exp(-((t - center) / width) ** 2)
        profiles.append(a / a.sum())

    # Product of exponentials (asymmetric bump)
    for a_left, a_right in ((2.0, 1.0), (1.0, 2.0), (3.0, 1.0), (1.0, 3.0), (2.0, 0.5), (0.5, 2.0)):
        a = np.exp(-a_left * t) * np.exp(-a_right * (1.0 - t))
        profiles.append(a / a.sum())

    # Random smooth
    for _ in range(8):
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(freq * 2 * np.pi * t + phase) + 0.3 * rng.uniform(-1, 1) * t
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())

    # Power-modulated exponential
    for p in (2.0, 3.0, 0.5):
        a = np.exp(2.0 * t ** p)
        profiles.append(a / a.sum())

def _interpolate_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)
