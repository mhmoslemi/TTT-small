def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def convolve_fft(a):
    """Full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    c = np.maximum(c, 0.0)  # Remove numerical negatives
    return c

def logsumexp(x, beta):
    """Compute beta * log(sum(exp(x/beta))) - min(x), stable."""
    x = np.asarray(x, dtype=np.float64)
    xmax = x.max()
    if beta < 1e-15:
        return xmax, np.array([x == xmax])
    e = np.exp((x - xmax) / beta)
    s = e.sum()
    val = xmax + beta * np.log(s)
    w = e / s
    return val, w

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 32:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1]
    return np.maximum(c, 0.0)


def _corr_sliding(p, q, n):
    """Compute h[i] = sum_j p[j]*q[i+j] for i=0..n-1, where q has length 2n-1."""
    N = _next_pow2(3 * n)
    fp = np.fft.rfft(p[::-1], N)
    fq = np.fft.rfft(q, N)
    h = np.fft.irfft(fp * fq, N)[n-1:2*n-1]
    return h


def _logit_grad(p, gp, n):
    """Convert gradient in p-space to z-space (logit) for simplex."""
    return p * (gp - p.dot(gp))


def _adam_step(z, m, v, t, gz, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1-b1) * gz
    v = b2 * v + (1-b2) * gz**2
    mh = m / (1 - b1**t)
    vh = v / (1 - b2**t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m, v
