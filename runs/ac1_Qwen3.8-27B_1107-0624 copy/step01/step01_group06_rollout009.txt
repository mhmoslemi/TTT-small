    deadline = time.monotonic() + budget_s
    
    a = normalize_sum(a_init.copy(), target=float(n))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    start = time.monotonic()
    stall = 0
    stall_best = best_J
    verify_interval = 5.0
    last_verify = 0.0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.6)
        lr_t = lr / (1.0 + 0.2 * t / 3000.0)
        
        J, grad = smooth_obj_grad(a, tau)
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        a = a - lr_t * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0.0)
        
        # Periodic local check
        if t % 3000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall = 0
                stall_best = local_J
            else:
                stall += 1
                if stall > 8:
                    # Escaped perturbation
                    cand = best_a.copy()
                    w = max(1, n // 50)
                    s = int(rng.integers(0, max(1, n - w)))
                    cand[s:s+w] = np.maximum(cand[s:s+w] * np.exp(rng.normal(0, 0.3, w)), 0)
                    cJ = compute_J(cand)
                    if cJ < stall_best:
                        a = cand.copy()
                    else:
                        # Try a different perturbation
                        cand = best_a.copy()
                        w2 = max(1, n // 20)
                        s2 = int(rng.integers(0, max(1, n - w2)))
                        cand[s2:s2+w2] = np.maximum(cand[s2:s2+w2] * np.exp(rng.normal(0, 0.5, w2)), 0)
                        cJ2 = compute_J(cand)
                        if cJ2 < stall_best:
                            a = cand.copy()
                        else:
                            a = best_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall = 0
                    stall_best = best_J
        
        # Verify with evaluate_sequence
        now = time.monotonic()
        if now - last_verify > verify_interval:
            last_verify = now
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
    
    return best_a, best_J


def subgradient_polish(a_init, budget_s, seed):
    """Polish using exact subgradients with averaged step."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()
    S = np.sum(a)
    if S < 0.01:
        return a
    a = normalize_sum(a, target=float(n))
    best_a = a.copy()
    best_J = compute_J(a)
    step_size = 0.1
    momentum = np.zeros(n)
    mu = 0.9
    
    while time.monotonic() < deadline:
        J, grad = subgradient_step(a)
        if not np.isfinite(J):
            break
        momentum = mu * momentum + (1 - mu) * grad
        a = a - step_size * momentum
        a = np.maximum(a, 0.0)
        
        local_J = compute_J(a)
        if local_J < best_J:
            best_J = local_J
            best_a = a.copy()
            step_size = max(0.01, step_size * 1.02)
        else:
            a = best_a.copy()
            step_size *= 0.95
            if step_size < 0.001:
                break
    
    return best_a


def make_sinusoidal(n, freq, amp, phase=0.0):
    """Sinusoidal modulation."""
    x = np.arange(n)
    return 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n + phase)


def make_multi_sin(n, freqs, amps):
    """Sum of multiple sinusoids."""
    x = np.arange(n)
    result = np.ones(n)
    for f, a in zip(freqs, amps):
        result += a * np.cos(2.0 * np.pi * f * x / n)
    return np.maximum(result, 0.1)


def make_gaussian_mod(n, sigma, modulation_freq, modulation_amp):
    """Gaussian envelope with sinusoidal modulation."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    envelope = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    mod = 1.0 + modulation_amp * np.cos(2.0 * np.pi * modulation_freq * x / n)
    return np.maximum(envelope * mod, 0.01)


def make_walsh(n, order=2):
    """Walsh-Hadamard based pattern (predictor: good for flat convolution)."""
    x = np.arange(n)
    result = np.ones(n)
    for k in range(1, order + 1):
        # Walsh function at frequency k
        phase = np.floor(x * k / n * 2.0) % 2.0
        result *= 1.0 + 0.15 * (1.0 - 2.0 * phase)
    return np.maximum(result, 0.01)

            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)
    
    n = parent.size
    print(f"n = {n}")
    
    # Initial score
    best_global = parent.copy()

    print(f"Initial score: {best_global_score:.6f}")
    
    # Build diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("ones", np.ones(n)))
    
    # Various sinusoidal patterns
    sin_configs = [
        (1, 0.3), (1, 0.5), (1, 0.7),
        (2, 0.3), (2, 0.5),

        (5, 0.3), (5, 0.5),
        (7, 0.4),

        (15, 0.3),
        (20, 0.2),
    ]
    for f, a in sin_configs:
        inits.append((f"sin_{f}_{a}", make_sinusoidal(n, f, a)))
    
    # Multi-sinusoidal
    inits.append(("msin1", make_multi_sin(n, [1, 3, 7], [0.2, 0.15, 0.1])))
    inits.append(("msin2", make_multi_sin(n, [2, 5, 11], [0.2, 0.1, 0.08])))
    inits.append(("msin3", make_multi_sin(n, [1, 2, 4, 8], [0.15, 0.12, 0.1, 0.08])))

    inits.append(("gauss_mod1", make_gaussian_mod(n, n*0.3, 5, 0.3)))
    inits.append(("gauss_mod2", make_gaussian_mod(n, n*0.4, 10, 0.5)))
    
    # Walsh-like
    inits.append(("walsh2", make_walsh(n, 2)))
    inits.append(("walsh3", make_walsh(n, 3)))

    for s in [0.2, 0.4, 0.7]:

    # Normalize and score
    scored_inits = []
    for name, arr in inits:
        arr = np.asarray(arr, dtype=np.float64)
        arr = normalize_sum(arr, target=float(n))
        J = compute_J(arr)

    scored_inits.sort(key=lambda x: x[1])
    print("\nTop initializations:")

    # Phase 1: Gradient descent on top initializations (50% of time)
