def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

    """Smallest power of two >= n, at least 2."""

def full_convolve_fft(a):
    """Full linear convolution of a real 1D array with itself."""
    a = np.asarray(a, dtype=np.float64).ravel()

def soft_mx(c, beta):
    """Smooth max of c with width beta, returning value and interpolating weights."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])

    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))

        ex[int(np.argmax(c))] = 1.0

def corr_weights(w, p):

def make_basis_mixed(n, K):
    """Mix linear drift and trigonometric harmonics; no constant column."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def make_basis_poly(n, K):
    """Centered polynomial basis, no constant column."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = []
    for k in range(1, K + 1):
        cols.append(u ** k)
    return np.column_stack(cols)

def interpolate_profile(p0, n_new):

    return normalize_nonneg(np.interp(x1, x0, p0))


def param_adam(theta0, basis, deadline, rng, lr=0.25, reg=1e-6):

    Adam optimization in the low-dimensional parameter space theta -> softmax(basis@theta).
    Returns (best_theta, best_p, best_score).
    """

    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(dim, dtype=np.float64)
    v = np.zeros(dim, dtype=np.float64)
    adam_t = 0

    p = profile_from_basis(theta, basis)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.25 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(theta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        theta = np.clip(theta, -6.0, 6.0)
        p = profile_from_basis(theta, basis)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.04, lr * 0.97)
        else:
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

            beta = max(1e-12, beta * 0.75)

            no_imp = 0

def logit_anneal(p0, deadline, rng, lr=0.08, reg=1e-6):
    """
    Full logit-space Adam polish of a nonnegative simplex sequence.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.25 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 500 == 0:
                    lr = max(0.02, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 100 == 0:
            beta = max(1e-12, beta * 0.75)

        if no_imp > max(500, 6 * n):
            z = 0.82 * best_z + 0.32 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.25 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score
