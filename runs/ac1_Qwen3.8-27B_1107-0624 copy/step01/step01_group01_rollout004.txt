def tuple_first(item):

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def make_basis_mixed(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m += 1
        if len(cols) > K:
            break
    return np.column_stack(cols[:K])


def make_basis_poly(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = []
    for k in range(1, K + 1):
        cols.append(u ** k)
    return np.column_stack(cols)

def make_piecewise_profile(n, segments):

    s = float(a.sum())
    return a / s

        for width in (0.15, 0.25, 0.35):
            profs.append(make_gaussian_profile(n, center, width))
    # Bimodal profiles
    profs.append(make_bimodal_profile(n, 0.2, 0.8, 0.15, 0.15, 1.0))
    profs.append(make_bimodal_profile(n, 0.1, 0.9, 0.1, 0.1, 1.5))
    profs.append(make_bimodal_profile(n, 0.3, 0.7, 0.2, 0.2, 0.5))
    # Piecewise profiles
    profs.append(make_piecewise_profile(n, [(0, 0.3, 2.0), (0.3, 0.7, 1.0), (0.7, 1.0, 0.5)]))
    profs.append(make_piecewise_profile(n, [(0, 0.2, 3.0), (0.2, 0.5, 1.0), (0.5, 1.0, 0.3)]))
    profs.append(make_piecewise_profile(n, [(0, 0.5, 1.5), (0.5, 1.0, 0.5)]))
    # Anti-symmetric cubic
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    for a1, a3 in ((1.5, -1.0), (2.0, -2.0), (1.0, 0.5), (2.5, -1.5)):
        a = 1.0 + a1 * u + a3 * u ** 3
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())
    return profs


def make_random_profiles(n, count, rng):
    profs = list(make_deterministic_profiles(n))
    bm = make_basis_mixed(n, 16)
    bp = make_basis_poly(n, 14)
    for _ in range(max(0, int(count))):
        th = rng.uniform(-4.0, 4.0, bm.shape[1])
        profs.append(profile_from_basis(th, bm))
        th = rng.uniform(-4.0, 4.0, bp.shape[1])
        profs.append(profile_from_basis(th, bp))
    for _ in range(max(1, count // 3)):
        th = rng.uniform(-3.0, 3.0, bm.shape[1])
        p = profile_from_basis(th, bm)
        for pw in (2.0, 3.0, 0.5):
            q = np.power(p, pw)
            q /= q.sum()
            profs.append(q)
    return profs


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    if n0 == 1:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def param_adam(theta0, basis, deadline, rng, lr=0.3, reg=1e-7, max_clip=7.0):
    basis = np.asarray(basis, dtype=np.float64)
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(dim, dtype=np.float64)
    v = np.zeros(dim, dtype=np.float64)
    adam_t = 0

    p = profile_from_basis(theta, basis)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.2 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(theta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        theta = np.clip(theta, -max_clip, max_clip)
        p = profile_from_basis(theta, basis)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 600 == 0:
                    lr = max(0.03, lr * 0.96)
        else:
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.72)

        if no_imp > max(300, 8 * dim):
            kick_scale = 0.7 if no_imp > 800 else 0.5
            theta = best_theta + kick_scale * rng.normal(size=dim)
            theta = np.clip(theta, -max_clip, max_clip)
            p = profile_from_basis(theta, basis)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_theta, best_p, best_score


def logit_anneal(p0, deadline, rng, lr=0.08, reg=1e-7, max_clip=50.0):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.2 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -max_clip, max_clip)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.015, lr * 0.97)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-12, beta * 0.70)

        if no_imp > max(400, 4 * n):
            kick = 0.6 if no_imp > 1000 else 0.4
            z = (1.0 - kick) * best_z + kick * rng.normal(size=n) * 0.5
            z = np.clip(z, -max_clip, max_clip)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.2 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_descent(p0, deadline, rng, steps_per_entry=3):
    """Direct coordinate descent on p with projection to simplex."""
    p = np.asarray(p0, dtype=np.float64).ravel().copy()
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = p.size

    c = full_convolve_fft(p)
    maxc = float(c.max())
    best_score = float(2.0 * n * maxc)
    best_p = p.copy()
    no_imp = 0

    while time.monotonic() < deadline and no_imp < 5000:
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            if time.monotonic() > deadline:
                break
            # Try modifying p[i] while keeping sum = 1
            # Move mass from p[i] to a random other position
            # Subgradient of max c at p[i]: sum_{k: c[k]=max} (p[k-i+1] + p[i-k+1])
            # We use the soft-max weights as a proxy
            L, w = soft_mx(c, max(1e-12, 0.1 * maxc))
            gp = 2.0 * corr_weights(w, p)
            grad_i = gp[i]

            if grad_i > 0:
                # Try decreasing p[i], increase a position with low gradient
                delta = p[i] * 0.1 * rng.uniform(0.1, 0.5)
                # Find j with lowest gradient (excluding i)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                j = int(np.argmin(gp[mask] * np.where(mask, 1, 1000)))
                if not mask[j]:
                    j = (i + n // 2) % n
                p_new = p.copy()
                p_new[i] -= delta
                p_new[j] += delta
                if p_new[i] < 0:
                    p_new[i] = 0
                    p_new[j] += p[i]
                p_new = p_new / p_new.sum()
                c_new = full_convolve_fft(p_new)
                max_new = float(c_new.max())
                score_new = float(2.0 * n * max_new)
                if score_new < best_score - 1e-14:
                    p = p_new
                    c = c_new
                    maxc = max_new
                    best_score = score_new
                    best_p = p.copy()
                    no_imp = 0
                    improved = True
                else:
                    no_imp += 1
            else:
                no_imp += 1

        if not improved:
            # Try a random perturbation
            if no_imp > 500:
                idx = rng.integers(0, n, size=min(10, n))
                for i in idx:
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    delta = p[i] * 0.05
                    p[i] -= delta
                    p[j] += delta
                p = np.maximum(p, 1e-15)
                p = p / p.sum()
                c = full_convolve_fft(p)
                maxc = float(c.max())
                no_imp = 0

    return best_p, best_score


def multi_start_logit_anneal(p0, deadline, rng, n_starts=3, lr_base=0.08):
    """Try multiple perturbations of p0 and anneal each."""
    best_p = p0.copy()
    best_score = internal_score(p0)
    n = p0.size

    for s in range(n_starts):
        if time.monotonic() > deadline:
            break
        frac = (deadline - time.monotonic()) / (n_starts - s)
        pert_scale = 0.15 * (s + 1)
        z0 = np.log(np.maximum(p0, 1e-12))
        z0 -= np.mean(z0)
        z_pert = z0 + pert_scale * rng.normal(size=n)
        z_pert = np.clip(z_pert, -40.0, 40.0)
        z_pert -= np.mean(z_pert)
        p_pert = softmax_from_z(z_pert)
        lr = lr_base * (0.7 + 0.3 * s / max(1, n_starts - 1))
        p_res, sc = logit_anneal(p_pert, time.monotonic() + frac, rng, lr=lr)
        if sc < best_score:
            best_score = sc
            best_p = p_res

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search: diverse profiles -> param Adam -> logit anneal ->
    coordinate descent -> multi-resolution refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []

    def local_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
        add_candidate(candidates, p, sc, 24)

    # --- Get parent sequence ---
    parent_valid = np.array([])
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = normalize_nonneg(parent)
            if pn is not None:
                parent_valid = pn
    except Exception:
        parent_valid = np.array([])

    if parent_valid.size > 0:
        sc_par = internal_score(parent_valid)
        local_update(parent_valid, sc_par)
        print(f"parent: n={parent_valid.size}, score={sc_par:.10f}")

    # --- Safe fallbacks ---
    for nf, eta in ((128, 2.0), (256, 2.0), (512, 2.5), (512, 3.0)):
        p = make_linear_profile(nf, eta)
        sc = internal_score(p)
        local_update(p, sc)

    print(f"initial best={best_score:.10f}")

    # --- Phase A: Low-dim parametric search at n=128 ---
    n0 = 128
    phase_a_end = min(deadline, t_start + 240.0)
    bm = make_basis_mixed(n0, 16)
    bp = make_basis_poly(n0, 14)

    for basis, dims in ((bm, bm.shape[1]), (bp, bp.shape[1])):
        starts = []
        for eta in (1.5, 2.0, 2.5, 3.0):
            th = np.zeros(dims)
            th[0] = eta
            starts.append(th)
        for _ in range(20):
            starts.append(rng.uniform(-4.0, 4.0, dims))

        for th in starts:
            if time.monotonic() > phase_a_end:
                break
            per = min(phase_a_end, time.monotonic() + 4.0)
            _, p_best, sc = param_adam(th, basis, per, rng, lr=0.3)
            local_update(p_best, sc)

    print(f"after phase A best={best_score:.10f}")

    # --- Phase B: Logit anneal at n0 and n=256 ---
    phase_b_end = min(deadline, t_start + 540.0)

    # Interpolate top candidates to n=256
    n1 = 256
    for i in range(min(8, len(candidates))):
        if time.monotonic() > phase_b_end:
            break
        p0 = candidates[i][1]
        if p0.size != n1:
            p0 = interpolate_profile(p0, n1)
        if p0 is None:
            continue
        # Also try at original small n
        if p0.size == n0:
            per = min(phase_b_end, time.monotonic() + 18.0)
            p_best, sc = logit_anneal(p0, per, rng, lr=0.08)
            local_update(p_best, sc)

        per = min(phase_b_end, time.monotonic() + 22.0)
        p_best, sc = logit_anneal(p0, per, rng, lr=0.07)
        local_update(p_best, sc)

    # Try deterministic profiles at n=256
    for p0 in make_deterministic_profiles(n1)[:8]:
        if time.monotonic() > phase_b_end:
            break
        per = min(phase_b_end, time.monotonic() + 20.0)
        p_best, sc = logit_anneal(p0, per, rng, lr=0.07)
        local_update(p_best, sc)

    print(f"after phase B best={best_score:.10f}")

    # --- Phase C: Refine at n=512 (and 1024 if time permits) ---
    phase_c_end = deadline - 120.0

    if parent_valid.size >= 512:
        p_work = parent_valid[:512] if parent_valid.size > 512 else parent_valid
        p_work = normalize_nonneg(p_work)
    else:
        p_work = interpolate_profile(best_seq, 512) if best_seq is not None else make_linear_profile(512, 2.5)

    if p_work is not None:
        p_work = p_work[:512] if len(p_work) > 512 else p_work
        if len(p_work) < 512:
            p_work = interpolate_profile(p_work, 512)

        # Multi-start anneal at n=512
        per_total = min(phase_c_end - time.monotonic(), 300.0)
        if per_total > 30:
            p_best, sc = multi_start_logit_anneal(p_work, time.monotonic() + per_total, rng, n_starts=3, lr_base=0.06)
            local_update(p_best, sc)
            print(f"phase C n=512 multi-start best={best_score:.10f}")

    # Additional candidates at n=512
    n2 = 512
    for i in range(min(6, len(candidates))):
        if time.monotonic() > phase_c_end:
            break
        p0 = interpolate_profile(candidates[i][1], n2)
        if p0 is None:
            continue
        per = min(phase_c_end, time.monotonic() + 25.0)
        p_best, sc = logit_anneal(p0, per, rng, lr=0.06)
        local_update(p_best, sc)

    # Try n=1024 if time permits
    if time.monotonic() < phase_c_end - 200.0 and best_score < 1.55:
        n3 = 1024
        p0 = interpolate_profile(best_seq, n3)
        if p0 is not None:
            per = min(phase_c_end - 100.0, time.monotonic() + 180.0)
            p_best, sc = logit_anneal(p0, per, rng, lr=0.05)
            local_update(p_best, sc)
            print(f"phase C n=1024 best={sc:.10f}")

    print(f"after phase C best={best_score:.10f}")

    # --- Phase D: Coordinate descent polish ---
    if time.monotonic() < phase_c_end - 30.0 and best_seq is not None:
        n_cd = min(512, best_seq.size)
        p_cd = best_seq[:n_cd] if best_seq.size > n_cd else best_seq
        if len(p_cd) < n_cd:
            p_cd = interpolate_profile(p_cd, n_cd)
        per_cd = min(phase_c_end - 30.0, time.monotonic() + 120.0)
        p_best, sc = coordinate_descent(p_cd, time.monotonic() + per_cd, rng)
        local_update(p_best, sc)
        print(f"after CD polish best={best_score:.10f}")

    # --- Phase E: Final aggressive polish with multiple short restarts ---
    if time.monotonic() < deadline - 30.0 and best_seq is not None:
        n_final = min(512, best_seq.size)
        p_final = best_seq[:n_final] if best_seq.size > n_final else best_seq
        if len(p_final) < n_final:
            p_final = interpolate_profile(p_final, n_final)

        remaining = deadline - 30.0 - time.monotonic()
        n_restarts = max(1, int(remaining // 25.0))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30.0:
                break
            per = min(deadline - 30.0, time.monotonic() + 20.0)
            # Small perturbation
            z0 = np.log(np.maximum(p_final, 1e-12))
            z0 -= np.mean(z0)
            scale = 0.08 * (r + 1)
            z_pert = z0 + scale * rng.normal(size=n_final)
            z_pert = np.clip(z_pert, -40.0, 40.0)
            z_pert -= np.mean(z_pert)
            p_pert = softmax_from_z(z_pert)
            p_best, sc = logit_anneal(p_pert, per, rng, lr=0.05)
            local_update(p_best, sc)
            p_final = best_seq[:n_final] if best_seq.size >= n_final else interpolate_profile(best_seq, n_final)

    print(f"after phase E best={best_score:.10f}")

    # --- Final: exact scoring and return ---
    if best_seq is not None:
        final_seq = best_seq
        final_score = best_score

        # Exact re-score
        if final_seq.size <= 2048:
            sc_exact = true_score(final_seq)
            if np.isfinite(sc_exact) and sc_exact < final_score:
                final_score = sc_exact

        # Check all candidates exactly
        for sc_c, p_c in candidates:
            if p_c.size <= 2048:
                sc_c2 = true_score(p_c)
                if np.isfinite(sc_c2) and sc_c2 < final_score:
                    final_score = sc_c2
                    final_seq = p_c

        # Validate with evaluate_sequence
        if final_seq.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass

        final_seq = normalize_nonneg(final_seq)
        if final_seq is None:
            final_seq = make_linear_profile(512, 2.0)
            final_score = internal_score(final_seq)

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"FINAL: score={final_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Ultimate fallback
    fallback = make_linear_profile(512, 2.0)
    return [float(x) for x in fallback]
