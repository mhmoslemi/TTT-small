"""
FFT-accelerated mirror descent with top-K peak tracking for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses multiplicative weight updates on the
simplex with adaptive step sizes, multiple initializations, and varied
sequence lengths. Includes LP polishing for the best candidate.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def compute_gradient_topk(a, top_k):
    """Compute subgradient of max(conv(a,a)) using top-k peaks.
    Returns (gradient, max_value)."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    if top_k < 2 * n - 1:
        peaks = np.argpartition(c, -top_k)[-top_k:]
    else:
        peaks = np.arange(2 * n - 1)
    grad = np.zeros(n)
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            grad[i_start:i_end + 1] += a[k - np.arange(i_start, i_end + 1)]
    return grad, M


def mirror_descent(a0, deadline, rng, top_k=10, initial_lr=0.05, ema_alpha=0.3):
    """Mirror descent (multiplicative weight) on the simplex.
    Minimizes max(conv(a,a)) with sum(a)=1, a>=0."""
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return a0, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)

    lr = initial_lr
    stall = 0
    it = 0
    grad_ema = np.zeros(n)

    while time.monotonic() < deadline:
        it += 1
        grad, M = compute_gradient_topk(a, top_k)
        grad_ema = (1.0 - ema_alpha) * grad_ema + ema_alpha * grad

        # Multiplicative update
        a_new = a * np.exp(-lr * grad_ema)
        a_new = np.clip(a_new, 1e-300, None)
        Sn = a_new.sum()
        if Sn < 1e-10:
            a_new = a.copy()
            Sn = S
        a_new = a_new / Sn

        sc = compute_objective(a_new)

        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.3, lr * 1.003)
        else:
            stall += 1
            lr = max(0.001, lr * 0.98)
            if stall % 200 == 0:
                # Structured perturbation
                noise_scale = 0.1 + 0.1 * (stall // 500)
                noise = 1.0 + rng.normal(0, noise_scale, n)
                a_pert = best_a * noise
                a_pert = np.clip(a_pert, 0, None)
                sp = a_pert.sum()
                if sp > 1e-10:
                    a_pert = a_pert / sp
                    sc_pert = compute_objective(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                        a = a_pert
                        grad_ema = np.zeros(n)
                        stall = 0
                        lr = initial_lr
                    elif rng.random() < 0.15:
                        a = a_pert
                        grad_ema = np.zeros(n)
                    else:
                        a = best_a.copy()
                        grad_ema = np.zeros(n)
                if stall > 2000:
                    break

        if it % 50000 == 0:
            print(f"      MD iter {it}, score={best_score:.6f}, lr={lr:.5f}, stall={stall}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_polish(a, deadline, rng):
    """LP-based polishing for the final sequence (works best for n <= 4096)."""
    n = len(a)
    if n > 8192:
        return a, compute_objective(a)

    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        A = build_conv_matrix(a)

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        if g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            sc = compute_objective(a_new)
            if sc < best_score - 1e-14:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.02)
        else:
            stall += 1
            step = max(0.005, step * 0.8)
            if stall > 100:
                break

    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_with_edges(n):
    """Flat with slightly boosted edges to reduce central peak."""
    a = np.ones(n)
    edge_width = max(1, n // 10)
    a[:edge_width] = 1.2
    a[-edge_width:] = 1.2
    return a


def make_modular(n, period):
    """Periodic pattern with slight randomness."""
    base = np.ones(period)
    base[period // 2] = 0.5
    a = np.tile(base, n // period + 1)[:n]
    return a
