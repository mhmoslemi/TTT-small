**Alternative: Coordinate Descent**

Pick one coordinate at a time and optimize it (or a small block) while keeping the rest fixed. For a single coordinate a[i], the objective as a function of a[i] is:
`f(a[i]) = 2n * max_k (conv[k](a[i])) / (sum - a[i] + a[i])^2`

Specifically, if we increase a[i] by δ (and decrease some other coordinate to maintain normalization), the convolution changes by δ * (a[k-i] + a[i-k]) at each lag k. We want to choose δ to minimize the new maximum.

This is a 1D minimax problem that can be solved by bisection!

**Coordinate descent with bisection:**
For each coordinate i:
1. Compute the "sensitivity vector" s[k] = a[k-i] + a[i-k] (zero-padded)
2. The new convolution is c + δ * s (approximately, for small δ)
3. Find δ that minimizes max(c + δ * s) using bisection
4. Update a[i] += δ, and renormalize

- Each step is exact (in 1D)
- It's simple to implement
- It naturally respects non-negativity (δ >= -a[i])

"""
Step function height optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses Adam optimizer on smooth-max (log-sum-exp) approximation with adaptive temperature,
multiple length scales, and population-based diversification.

import numpy as np
import time


def normalize_nonneg(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    np.clip(a, 0.0, None, out=a)
    s = a.sum()
    if s < 1e-300:
        return np.full_like(a, 1.0 / max(1, a.size))
    return a / s


def score_normalized(a):
    """Compute 2n * max(convolve(a,a)) for sum-1 non-negative a."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())


def smooth_max_logsumexp(c, beta):
    """Return (smooth_max_value, softmax_weights) for the vector c."""
    m = c.max()
    shifted = (c - m) * beta
    shifted = np.clip(shifted, -60.0, 0.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return float(m), w
    w = e / s
    smooth_val = m + np.log(s) / beta
    return float(smooth_val), w


def gradient_smooth_max(a, weights):
    """
    Gradient of smooth_max(convolve(a,a)) w.r.t. a.
    grad[i] = 2 * sum_j weights[i+j] * a[j]  (cross-correlation)
    """
    n = len(a)
    # np.correlate(weights, a, 'valid') gives:
    # out[i] = sum_j weights[i+j] * a[j] for i=0..n-1
    g = np.correlate(weights, a, mode='valid')
    return 2.0 * g


def make_initial_sequence(n, kind, rng, provided_seq=None):

    n = max(2, int(n))
    x = np.linspace(0.0, 1.0, n)

    if kind == -1 and provided_seq is not None:
        # Resize provided sequence
        m = len(provided_seq)
        if m == n:
            return normalize_nonneg(provided_seq)
        x_old = np.linspace(0.0, 1.0, m)
        a = np.interp(x, x_old, provided_seq)

    if kind == -2 and provided_seq is not None:

        m = len(provided_seq)
        x_old = np.linspace(0.0, 1.0, m)

    elif kind == 2:

    elif kind == 3:

def optimize_with_adam(a, time_budget, rng, state, beta_start=30.0, beta_end=3000.0):
    """
    Optimize a normalized non-negative sequence using Adam on smooth-max.
    state: dict with 'score' and 'seq' keys (updated in place).
    Returns (best_seq, best_score) as arrays.
    """
    n = len(a)
    a = normalize_nonneg(a.copy())

    best_score = score_normalized(a)
    best_a = a.copy()

    if best_score < state['score'] - 1e-12:
        state['score'] = best_score
        state['seq'] = best_a.copy()
        print(f"  [adam] init score={best_score:.10f} n={n}", flush=True)

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    # Schedule params
    total_steps_est = 80000  # rough estimate
    base_lr = 0.05 / n ** 0.3
    no_improve = 0
    patience = 2000

    deadline = time.monotonic() + max(0.1, float(time_budget))

    while time.monotonic() < deadline:
        t += 1

        # Temperature schedule: cosine annealing from beta_start to beta_end
        progress = min(1.0, t / max(1, total_steps_est))
        beta = beta_start + (beta_end - beta_start) * (1.0 - np.cos(np.pi * progress)) / 2.0
        # Add slight randomization
        beta *= rng.uniform(0.9, 1.1)

        # Compute convolution and smooth max
        c = np.convolve(a, a)
        smooth_val, weights = smooth_max_logsumexp(c, beta)

        # Compute gradient
        grad = gradient_smooth_max(a, weights)

        # Learning rate schedule
        lr = base_lr * (1.0 - 0.5 * progress)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - lr * step

        # Project: clip to non-negative, normalize
        np.clip(a_new, 0.0, None, out=a_new)
        s = a_new.sum()
        if s < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s

        a = a_new

        # Evaluate
        if t % 50 == 0 or t == 1:
            cur_score = score_normalized(a)
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
                if best_score < state['score'] - 1e-10:
                    state['score'] = best_score
                    state['seq'] = best_a.copy()
                    print(f"  [adam] t={t} score={best_score:.10f} n={n} beta={beta:.1f}", flush=True)
            else:
                no_improve += 1

        # Restart logic
        if no_improve > patience:
            # Try a perturbation
            scale = rng.uniform(0.05, 0.3)
            noise = rng.normal(0, scale, size=n)
            a_new = a * np.exp(noise)
            a = normalize_nonneg(a_new)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t = 0
            no_improve = 0
            base_lr = base_lr * rng.uniform(0.5, 1.5)
            print(f"  [adam] restart at t={t}, base_lr={base_lr:.6f}", flush=True)

    # Final evaluation
    final_score = score_normalized(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if best_score < state['score'] - 1e-12:
        state['score'] = best_score
        state['seq'] = best_a.copy()

    return best_a, best_score


def optimize_with_gradient_descent(a, time_budget, rng, state, beta_start=20.0, beta_end=5000.0):
    """
    Pure gradient descent with momentum and backtracking line search.
    More aggressive than Adam for this problem.
    """
    n = len(a)
    a = normalize_nonneg(a.copy())

    best_score = score_normalized(a)
    best_a = a.copy()

    if best_score < state['score'] - 1e-12:
        state['score'] = best_score
        state['seq'] = best_a.copy()

    # Momentum
    velocity = np.zeros(n, dtype=np.float64)
    momentum = 0.9
    beta = beta_start
    base_lr = 0.1 / n ** 0.2
    no_improve = 0
    patience = 3000

    deadline = time.monotonic() + max(0.1, float(time_budget))
    t = 0

    while time.monotonic() < deadline:
        t += 1

        # Temperature increase
        progress = min(1.0, t / 100000.0)
        beta = beta_start + (beta_end - beta_start) * progress

        c = np.convolve(a, a)
        smooth_val, weights = smooth_max_logsumexp(c, beta)
        grad = gradient_smooth_max(a, weights)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-300:
            grad = grad / gnorm

        # Momentum update
        velocity = momentum * velocity + (1.0 - momentum) * grad

        # Line search (try a few step sizes)
        lr = base_lr * max(0.1, 1.0 - progress)
        tried = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            a_try = a - lr * factor * velocity
            np.clip(a_try, 0.0, None, out=a_try)
            s = a_try.sum()
            if s < 1e-300:
                continue
            a_try = a_try / s
            sc = score_normalized(a_try)
            if sc < best_score - 1e-12 or sc < score_normalized(a) - 1e-8:
                a = a_try
                tried = True
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                    if best_score < state['score'] - 1e-10:
                        state['score'] = best_score
                        state['seq'] = best_a.copy()
                        print(f"  [gd] t={t} score={best_score:.10f} n={n} beta={beta:.0f}", flush=True)
                break
        if not tried:
            no_improve += 1

        if no_improve > patience:
            # Perturb and reset
            scale = rng.uniform(0.1, 0.4)
            a = normalize_nonneg(a * np.exp(rng.normal(0, scale, size=n)))
            velocity = np.zeros(n, dtype=np.float64)
            no_improve = 0
            base_lr *= rng.uniform(0.3, 2.0)

    final_score = score_normalized(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if best_score < state['score'] - 1e-12:
        state['score'] = best_score
        state['seq'] = best_a.copy()

    return best_a, best_score


def coordinate_descent_pass(a, state, rng, max_iters=5000):
    """
    Perform coordinate descent: for each position, find optimal value.
    Uses bisection on the 1D subproblem.
    """
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_score = score_normalized(a)

    improved = False
    for it in range(max_iters):
        if time.monotonic() > deadline_global:
            break
        # Pick a random coordinate (or sweep)
        if rng.random() < 0.5:
            i = int(rng.integers(n))
        else:
            i = it % n

        old_val = a[i]
        if old_val < 1e-15:
            # Try increasing from 0
            for delta in (0.001, 0.005, 0.01, 0.05) / n:
                a[i] = delta
                a = a / a.sum()
                sc = score_normalized(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
                    break
            continue

        # Try decreasing
        for frac in (0.7, 0.5, 0.3, 0.1, 0.0):
            a[i] = old_val * frac
            a = a / a.sum()
            sc = score_normalized(a)
            if sc < best_score - 1e-12:
                best_score = sc
                improved = True
            else:
                a[i] = old_val
                a = a / a.sum()
                break

    if best_score < state['score'] - 1e-12:
        state['score'] = best_score
        state['seq'] = a.copy()

    return a, best_score


def refine_short(a, time_budget, rng, state):
    """
    Refine a short sequence (n <= 12) by exhaustive-ish search.
    """
    n = len(a)
    if n > 12:
        return a, score_normalized(a)

    best_score = score_normalized(a)
    best_a = a.copy()
    deadline = time.monotonic() + max(0.1, float(time_budget))

    # Try various structured sequences
    candidates = []

    # Symmetric patterns
    for _ in range(200):
        if time.monotonic() > deadline:
            break
        # Random non-negative, normalize
        c = rng.random(n)
        if rng.random() < 0.3:
            c[1::2] *= 0.1  # sparse
        c = normalize_nonneg(c)
        sc = score_normalized(c)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()
            if best_score < state['score'] - 1e-12:
                state['score'] = best_score
                state['seq'] = best_a.copy()

    # Local perturbation of current best
    for _ in range(2000):
        if time.monotonic() > deadline:
            break
        c = best_a.copy()
        i = int(rng.integers(n))
        c[i] *= rng.uniform(0.3, 3.0)
        c = normalize_nonneg(c)
        sc = score_normalized(c)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = c.copy()
            if best_score < state['score'] - 1e-12:
                state['score'] = best_score
                state['seq'] = best_a.copy()

    return best_a, best_score


# Global deadline for time management
deadline_global = 0.0


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for low-score step-function heights using:
    1. Adam optimizer on smooth-max with adaptive temperature
    2. Gradient descent with momentum as alternative
    3. Multiple length scales (from short to long)
    4. Population-based diversification
    5. Final refinement on the best found sequence

    Returns a list of non-negative floats.
    """
    global deadline_global
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline_global = start_time + max(1.0, float(budget_s) - 30.0)

    state = {'score': float('inf'), 'seq': None}

    # Load provided sequence
    provided = load_provided()
    if provided is not None:
        p_score = score_normalized(provided)
        print(f"Provided sequence: n={len(provided)}, score={p_score:.10f}", flush=True)
        state['score'] = min(state['score'], p_score)
        state['seq'] = provided.copy()

    # Guaranteed fallback
    fallback = np.array([1.0/3, 2.0/3], dtype=np.float64)
    fb_score = score_normalized(fallback)
    print(f"2-element fallback: score={fb_score:.10f}", flush=True)

    # Phase 1: Quick exploration of short sequences (0-60s)
    print("\n=== Phase 1: Short sequence exploration ===", flush=True)
    phase1_end = start_time + 60.0

    for n_short in (2, 3, 4, 5, 6, 8, 10, 12):
        if time.monotonic() > phase1_end:
            break
        for kind in range(5):
            if time.monotonic() > phase1_end:
                break
            a = make_initial_sequence(n_short, kind, rng, provided)
            refine_short(a, 2.0, rng, state)

    # Phase 2: Medium length optimization (60s - 1200s)
    print("\n=== Phase 2: Medium length optimization ===", flush=True)

    lengths_to_try = []
    if provided is not None:
        lengths_to_try.append(len(provided))

    # Standard lengths
    for n in (64, 128, 256, 512, 1024, 2048):
        if n not in lengths_to_try:
            lengths_to_try.append(n)
    lengths_to_try.sort()

    # Allocate time proportional to length (longer = more time)
    # But prioritize lengths near the current best
    for n in lengths_to_try:
        if time.monotonic() > deadline_global - 200:
            break

        # Determine time budget for this length
        if n <= 128:
            per_kind_time = 10.0
        elif n <= 512:
            per_kind_time = 25.0
        elif n <= 1024:
            per_kind_time = 40.0
        else:
            per_kind_time = 50.0

        kinds_to_try = []
        if provided is not None:
            kinds_to_try += [-1, -2]  # Resize provided
        kinds_to_try += [0, 1, 2, 4, 5, 6]

        for kind in kinds_to_try:
            if time.monotonic() > deadline_global - 150:
                break

            a = make_initial_sequence(n, kind, rng, provided)
            init_score = score_normalized(a)
            print(f"  n={n}, kind={kind}: init_score={init_score:.6f}", flush=True)

            # Skip if obviously bad and we already have a good solution
            if init_score > max(3.0, 2.5 * state['score']) and kind not in (-1, -2) and n > 128:
                continue

            remaining = deadline_global - time.monotonic() - 150.0
            budget = min(per_kind_time, remaining)
            if budget < 3.0:
                continue

            # Choose optimizer based on length
            if n <= 512:
                optimize_with_adam(a, budget, rng, state,
                                   beta_start=20.0, beta_end=5000.0)
            else:
                # For longer sequences, use lower temperature to avoid numerical issues
                optimize_with_adam(a, budget, rng, state,
                                   beta_start=15.0, beta_end=2000.0)

    # Phase 3: Focused refinement of the best (1200s - 1750s)
    print(f"\n=== Phase 3: Focused refinement ===\nCurrent best: {state['score']:.10f}", flush=True)

    if state['seq'] is not None:
        best_n = len(state['seq'])
        refine_budget = min(300.0, deadline_global - time.monotonic() - 50.0)

        if refine_budget > 30:
            # Refine at current length with aggressive parameters
            a = state['seq'].copy()
            optimize_with_adam(a, refine_budget * 0.5, rng, state,
                               beta_start=100.0, beta_end=10000.0)

            # Try gradient descent variant
            if time.monotonic() < deadline_global - 50:
                a2 = state['seq'].copy()
                gd_budget = min(100.0, deadline_global - time.monotonic() - 50.0)
                if gd_budget > 10:
                    optimize_with_gradient_descent(a2, gd_budget, rng, state,
                                                   beta_start=50.0, beta_end=8000.0)

            # Try upscaling to next power of 2
            if best_n < 4096 and time.monotonic() < deadline_global - 60:
                next_n = best_n * 2
                if next_n <= 4096:
                    a3 = make_initial_sequence(next_n, -1, rng, state['seq'])
                    up_budget = min(80.0, deadline_global - time.monotonic() - 50.0)
                    if up_budget > 10:
                        print(f"  Trying upscale to n={next_n}", flush=True)
                        optimize_with_adam(a3, up_budget, rng, state,
                                           beta_start=20.0, beta_end=3000.0)

    # Phase 4: Final polish (last 50s)
    print(f"\n=== Phase 4: Final polish ===", flush=True)
    if state['seq'] is not None:
        polish_budget = min(40.0, deadline_global - time.monotonic() - 5.0)
        if polish_budget > 5:
            a = state['seq'].copy()
            # Very high temperature = essentially exact max
            optimize_with_adam(a, polish_budget, rng, state,
                               beta_start=2000.0, beta_end=50000.0)

    # Verify with official evaluator
    if state['seq'] is not None:
        cand = [float(x) for x in state['seq']]
        try:
            official = evaluate_sequence(cand)
            print(f"\nOfficial score: {official:.10f} (n={len(cand)})", flush=True)
            if np.isfinite(official) and official < state['score']:
                state['score'] = official
        except Exception as e:
            print(f"Evaluation error: {e}", flush=True)
    else:
        print("No valid sequence found, using fallback.", flush=True)

    # Return
    if state['seq'] is not None:
        result = [float(max(0.0, x)) for x in state['seq']]
        print(f"Returning sequence of length {len(result)}, internal score={state['score']:.10f}", flush=True)
        return result
    else:
        return [1.0 / 3.0, 2.0 / 3.0]
