def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)

    if n <= 1024:

    else:

        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2*n-1].real

    maxc = float(np.max(c))
    if not np.isfinite(maxc):

    return float(2.0 * n * maxc / (s * s))

Actually, for the convolution to be flat, we want the "energy" to be spread out in the time domain. This is related to the uncertainty principle: a sequence that's spread out in time has a concentrated spectrum, and vice versa.

The convolution in the time domain corresponds to multiplication in the frequency domain: C(ω) = A(ω)^2.

If A(ω) is flat (constant), then C(ω) is also flat, and c (the convolution) is a delta function (concentrated at one point). That's the opposite of what we want.

If A(ω) is concentrated (like a sinc), then C(ω) is also concentrated, and c is more spread out. But c will have a large peak where it's concentrated.

The ideal would be for c to be as flat as possible, which means C(ω) should be as concentrated as possible (a delta at ω=0). But C(ω) = A(ω)^2, so A(ω) should be a delta at ω=0, meaning a is a constant sequence. But a constant sequence gives a triangular c, not flat.

So there's a fundamental tension. The best we can do is find a compromise.

For non-negative sequences, the constraint a >= 0 limits the possible A(ω). In particular, A(0) = sum(a) > 0, and A(ω) for other ω is constrained by the non-negativity.

I think the greedy sequence (1/sqrt(k) decay) is close to optimal because it balances the peak at k=0 (a[0]^2) with the spread of the rest.

"""
Search for non-negative sequences minimizing 2*n*max(conv(a))/sum(a)^2.
Uses adaptive smoothed gradient descent + LP cutting plane refinement.
Focus on length ~384 with warm start from previous best.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def greedy_init(n):
    n = int(n)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    return normalize(a)


def score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    L = next_pow2(2*n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max() / (s * s))


def score_direct(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max() / (s * s))


def smooth_grad_descent(p, budget, alpha0=30.0, alpha1=5.0):
    """Main optimization loop."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    p = normalize(p)
    L = next_pow2(2*n + n - 1)  # for conv(p, w_rev)

    x -= x.mean()
    
    mu, nu, eps = 0.9, 0.999, 1e-8
    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    t = 0
    
    best_p = p.copy()
    best_f = score_fft(p)
    
    start = time.monotonic()
    end = start + budget
    no_improve_count = 0
    patience = 500
    
    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget, 1.0))
        
        # Alpha schedule
        alpha = alpha0 + (alpha1 - alpha0) * (1.0 - np.exp(-2.5 * frac))
        
        # Compute convolution
        Pg = np.fft.rfft(p, Lc)
        c = np.fft.irfft(Pg * Pg, Lc)[:2*n-1].real
        c = np.maximum(c, 0.0)

def lp_direction(f, M, max_iter=50, budget=60.0):
    """
    LP cutting plane to find a direction g that reduces max(f*f).
    f is normalized to sum = sqrt(2n).
    Returns g (same sum as f) or None.
    """
    if _scipy_opt is None:
        return None
    
    n = f.size
    S0 = np.sqrt(2.0 * n)
    
    # Initial constraints: evenly spaced + the argmax
    n_init = min(200, 2*n - 1)
    init_rows = np.unique(np.concatenate([
        np.linspace(0, 2*n-2, n_init, dtype=int),
        [int(np.argmax(np.convolve(f, f)))]
    ]))
    
    rows = list(init_rows)
    row_set = set(rows)
    max_rows = min(800, 2*n - 1)
    
    start = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start > budget:
            break
        
        # Build constraint matrix
        rows_arr = np.array(rows, dtype=int)
        nr = rows_arr.size
        A = np.zeros((nr, n), dtype=np.float64)
        for i in range(nr):
            k = int(rows_arr[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[i, lo:hi+1] = f[k-hi:k-lo+1][::-1]
        
        b = np.full(nr, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)
        
        # Solve LP
        try:
            res = _scipy_opt.linprog(
                c_obj, A_ub=A, b_ub=b,
                bounds=[(0, None)] * n,
                method='highs',
                options={'time_limit': max(1.0, (budget - (time.monotonic()-start)) * 0.3)}
            )
        except Exception:
            return None
        
        if not res.success or res.x is None:
            return None
        
        g = np.maximum(res.x, 0.0)
        sg = g.sum()
        if sg < 1e-12:
            return None
        
        # Check violations
        conv_fg = np.convolve(f, g) if n <= 512 else (
            np.fft.irfft(np.fft.rfft(f, next_pow2(2*n-1)) * np.fft.rfft(g, next_pow2(2*n-1)), next_pow2(2*n-1))[:2*n-1].real
        )
        viol = conv_fg - M
        if viol.max() > 1e-6 * M:
            # Add violated rows
            if len(rows) < max_rows:
                order = np.argsort(viol)[::-1]
                added = 0
                for idx in order:
                    if added >= 100:
                        break
                    idx = int(idx)
                    if viol[idx] > 1e-6 * M and idx not in row_set:
                        rows.append(idx)
                        row_set.add(idx)
                        added += 1
                continue
        
        # Rescale g to match f's sum
        g = g * (S0 / sg)
        return g
    return None

    )))

    best_score = cur_score
    M = cur_score
    
    start = time.monotonic()

            break
