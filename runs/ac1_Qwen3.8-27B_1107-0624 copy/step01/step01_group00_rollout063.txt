def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target/n)
    lam = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - lam, 0.0)
