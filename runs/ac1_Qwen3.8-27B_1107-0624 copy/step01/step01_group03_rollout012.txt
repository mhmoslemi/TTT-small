import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def water_fill(v, target=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    css = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n+1) > (css - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (css[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(s, t):
    """Compute gradient of (2n/t) * log(sum exp(t * conv(s,s)))."""
    n = len(s)
    conv = np.convolve(s, s)
    # Stable log-sum-exp

    corr = correlate(w, s, mode='valid')

def lp_direction(s, M, time_limit=3.0):

            return np.maximum(res.x, 0.0)
    except:
        pass
    return None

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def water_fill(v, target=1.0):
    """Euclidean projection onto simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    cond = sv * (np.arange(1, n+1, dtype=float)) + target > css
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.max(np.where(cond)[0])
    theta = (css[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_gradient(s, t):
    """Gradient of (2n/t)*log(sum(exp(t*conv(s,s))))."""
    n = len(s)

    mc = np.max(conv)

"""
Hybrid LP-guided + log-sum-exp gradient optimization for minimizing
2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Combines LP direction-finding, smooth gradient descent with annealing,
and multiplicative updates across multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def water_fill(v, target=1.0):
    """Euclidean projection onto simplex {x >= 0, sum(x) = target}."""
    n = len(v)
    sv = np.sort(v)[::-1]
    css = np.cumsum(sv)
    idx = np.arange(1, n + 1, dtype=float)
    cond = sv * idx + target > css
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.max(np.where(cond)[0])
    theta = (css[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(s, t):
    """Gradient of (2n/t)*log(sum(exp(t*conv(s,s))))."""
    n = len(s)
    conv = np.convolve(s, s)
    mc = np.max(conv)
    w = np.exp(t * (conv - mc))
    w /= np.sum(w)
    corr = correlate(w, s, mode='valid')
    return (4.0 * n / t) * corr


def subgradient_mean(s):
    """Mean subgradient of 2n*max(conv(s,s)) over active set."""
    n = len(s)
    conv = np.convolve(s, s)
    M = np.max(conv)
    # Active set: lags where conv is within 1% of max
    active = conv >= M * 0.99
    if not np.any(active):
        active = conv >= M - 1e-10
    wa = np.zeros(2 * n - 1)
    wa[active] = 1.0 / np.sum(active)
    corr = correlate(wa, s, mode='valid')
    return 4.0 * n * corr


def lp_direction(s, M, tlimit=2.0):
    """LP: max sum(b) s.t. (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': tlimit, 'presolve': True})
        if res.success:
            x = np.maximum(res.x, 0)
            if x.sum() > 1e-14:
                return x / x.sum()
    except Exception:
        pass
    return None


def lp_refine(s, rng, deadline, max_iters=10000):
    """LP-guided local search with line search and perturbation."""
    n = len(s)
    best = s.copy()
    best_sc = fast_score(s)
    stall = 0
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(s, s)
        M = np.max(conv)
        tl = 1.5 if n <= 512 else 2.5
        b = lp_direction(s, M, tlimit=tl)
        if b is None:
            break
        # Line search with geometric candidates
        bt = 0.0
        bst = best_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            c = np.maximum((1 - t) * s + t * b, 0)
            cs = c.sum()
            if cs < 1e-15:
                continue
            c /= cs
            sc = fast_score(c)
            if sc < bst - 1e-13:
                bst = sc
                bt = t
        if bt == 0.0:
            stall += 1
            if stall > 10:
                # Geometric perturbation
                s = s * (1 + 0.01 * rng.standard_normal(n))
                s = np.maximum(s, 0)
                s /= s.sum()
            if stall > 40:
                break
            continue
        stall = 0
        s = np.maximum((1 - bt) * s + bt * b, 0)
        s /= s.sum()
        sc = fast_score(s)
        if sc < best_sc:
            best_sc = sc
            best = s.copy()
    return best, best_sc


def grad_refine(s, deadline, t_start=15.0, t_end=800.0, n_anneal=25, lr_init=0.008):
    """Log-sum-exp gradient descent with geometric annealing."""
    n = len(s)
    best = s.copy()
    best_sc = fast_score(s)
    ts = np.geomspace(t_start, t_end, n_anneal)
    per_phase = max(5, int(300 / n_anneal))
    for t in ts:
        if time.monotonic() > deadline:
            break
        lr = lr_init * np.sqrt(t_start / t)
        for _ in range(per_phase):
            if time.monotonic() > deadline:
                break
            g = smooth_gradient(s, t)
            gc = g - g.mean()
            gnorm = np.linalg.norm(gc)
            if gnorm < 1e-15:
                break
            # Backtracking line search
            accepted = False
            for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * shrink
                v = s - lr_try * gc
                s_new = water_fill(v, 1.0)
                sc = fast_score(s_new)
                if sc <= best_sc + 1e-14:
                    s = s_new
                    accepted = True
                    break
            if not accepted:
                # Tiny step to avoid stagnation
                s = water_fill(s - 0.001 * lr * gc, 1.0)
            sc = fast_score(s)
            if sc < best_sc:
                best_sc = sc
                best = s.copy()
    return best, best_sc


def grad_refine_subgrad(s, deadline, n_iters=2000, lr_init=0.004):
    """Mean-subgradient descent on the true max (complementary to smooth grad)."""
    n = len(s)
    best = s.copy()
    best_sc = fast_score(s)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g = subgradient_mean(s)
        gc = g - g.mean()
        gnorm = np.linalg.norm(gc)
        if gnorm < 1e-15:
            break
        lr = lr_init / (1.0 + 0.01 * it)
        accepted = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            v = s - lr * shrink * gc / gnorm
            s_new = water_fill(v, 1.0)
            sc = fast_score(s_new)
            if sc <= best_sc + 1e-14:
                s = s_new
                accepted = True
                break
        if not accepted:
            break
        sc = fast_score(s)
        if sc < best_sc:
            best_sc = sc
            best = s.copy()
    return best, best_sc


def multiplicative_update(s, b, t=0.3):
    """Multiplicative (mirror) update toward b."""
    eps = 1e-12
    ratio = b / (s + eps)
    s_new = s * np.power(ratio, t)
    s_new = np.maximum(s_new, 0)
    if s_new.sum() < 1e-15:
        return s
    return s_new / s_new.sum()
