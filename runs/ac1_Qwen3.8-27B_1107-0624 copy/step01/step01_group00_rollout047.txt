def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    r = np.max(np.where(cond)[0])
    theta = (cssv[r] - target) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_sq(a, m):
    """Full linear convolution a*a via FFT."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*len(a)-1]

def _compute_grad(c, a, m, L, tau):
    """Compute gradient of smoothed max of c (convolution) w.r.t. a."""
    n = len(a)
    L = min(L, 2*n-1)
    # Get top-L indices
    if L < 2*n-1:
        top_idx = np.argpartition(-c, L-1)[:L]
    else:
        top_idx = np.arange(2*n-1)
    wc = c[top_idx]
    # Softmax weights
    wc_shift = wc - wc.max()
    w_all = np.zeros(2*n-1)
    w_top = np.exp(wc_shift / tau)
    w_top = w_top / w_top.sum()
    w_all[top_idx] = w_top

    fw = np.fft.rfft(w_all, m)
    h = np.fft.irfft(fw * np.conj(fa), m)
    return 2.0 * h[:n]

    a_new = np.fft.irfft(fa_new, m)
    a_new = a_new[:n]

    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = np.full(n, 1.0/n)
    return a_new

        # a[k-p] is valid when 0 <= k-p < n, i.e., k-n+1 <= p <= k
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        # For p in [lo, hi]: a[k-p] where p goes from lo to hi, so k-p goes from k-lo to k-hi
        p_range = np.arange(lo, hi+1)
        a_vals = a[k - p_range]  # a[k-lo], a[k-lo-1], ..., a[k-hi]
        grad[lo:hi+1] += 2.0 * wk * a_vals
    
    return grad

"""
Minimize 2*n*max(b)/(sum(a))^2 where b = full_convolve(a, a), a >= 0.
Strategy: population-based search with FFT-based gradient descent,
spectral refinement (Gerchberg-Saxton), and multiple sequence lengths.
"""

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n, dtype=np.float64)
    r = np.max(np.where(cond)[0])
    theta = (cssv[r] - target) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_sq(a, m):
    """Full linear convolution a*a via zero-padded FFT."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * len(a) - 1]


def _compute_grad(c, a, L, tau):
    """Compute gradient of smoothed-max objective w.r.t. a."""
    n = len(a)
    num_c = len(c)
    L = min(L, num_c)
    top_idx = np.argpartition(-c, L - 1)[:L]
    wc = c[top_idx]
    wc_max = wc.max()
    exp_w = np.exp((wc - wc_max) / max(tau, 1e-12))
    w = exp_w / exp_w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for k, wk in zip(top_idx, w):
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        p_range = np.arange(lo, hi + 1)
        a_vals = a[k - p_range]
        grad[lo:hi + 1] += 2.0 * wk * a_vals
    return grad


def _spectral_refine(a, m, alpha=1.0):
    """One Gerchberg-Saxton style spectral refinement step."""
    n = len(a)
    fa = np.fft.rfft(a, m)
    freqs = np.fft.rfftfreq(m, d=1.0 / m)
    # Gaussian low-pass target (reduces time-domain ringing)
    sigma = 0.15
    target_mag = np.exp(-0.5 * (freqs / sigma) ** 2)
    target_mag[0] = 1.0
    phase = np.angle(fa)
    fa_new = target_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, m)
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = np.full(n, 1.0 / n, dtype=np.float64)
    return a_new


def _spectral_flat(a, m, alpha=1.0):
    """Spectral refinement with flat target magnitude."""
    n = len(a)
    fa = np.fft.rfft(a, m)
    mag = np.abs(fa)
    # Replace magnitude with its geometric mean (flat target)
    log_mag = np.log(np.maximum(mag, 1e-15))
    flat_mag = np.exp(np.median(log_mag))
    phase = np.angle(fa)
    fa_new = flat_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(fa_new, m)
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = np.full(n, 1.0 / n, dtype=np.float64)
    return a_new


def _make_seeds(n, rng, provided_seq=None):
    """Generate diverse initial seeds of length n."""
    seeds = []
    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    # Raised cosine
    t = np.linspace(0, np.pi, n)
    rc = np.sin(t / 2) ** 2
    seeds.append(rc / rc.sum())
    # Bell
    x = np.linspace(-1, 1, n)
    bell = np.exp(-x ** 2 / (2 * 0.35 ** 2))
    seeds.append(bell / bell.sum())
    # Flat-top with smooth edges
    ft = np.ones(n, dtype=np.float64)
    w = int(0.7 * n)
    start = (n - w) // 2
    ft[:start] = 0
    ft[start + w:] = 0
    edge = max(2, w // 8)
    for i in range(edge):
        frac = i / edge
        ft[start + i] *= 0.5 * (1 - np.cos(np.pi * frac))
        ft[start + w - 1 - i] *= 0.5 * (1 - np.cos(np.pi * frac))
    ft = np.maximum(ft, 0)
    if ft.sum() > 0:
        seeds.append(ft / ft.sum())
    # Power-law shape
    x = np.linspace(0.01, 1, n)
    pw = x ** 2
    seeds.append(pw / pw.sum())
    # Symmetric double-bump
    x = np.linspace(-1, 1, n)
    db = np.exp(-0.5 * ((np.abs(x) - 0.4) / 0.25) ** 2)
    seeds.append(db / db.sum())
    # Random smooth
    x = np.arange(n)
    center = rng.uniform(0.1 * n, 0.9 * n)
    width = rng.uniform(0.15 * n, 0.4 * n)
    rs = np.exp(-0.5 * ((x - center) / width) ** 2)
    seeds.append(rs / rs.sum())
    # From provided sequence
    if provided_seq is not None:
        try:
            b = np.asarray(provided_seq, dtype=np.float64)
            b = np.clip(b, 0, 1000)
            if b.ndim == 1 and b.size == n and b.sum() > 0:
                seeds.append(b / b.sum())
            elif b.ndim == 1 and b.size >= 4:
                new_x = np.linspace(0, 1, n)
                old_x = np.linspace(0, 1, b.size)
                resampled = np.interp(new_x, old_x, b)
                resampled = np.maximum(resampled, 0)
                if resampled.sum() > 0:
                    seeds.append(resampled / resampled.sum())
        except Exception:
            pass
    # Add noisy variants of best-potential seeds
    for s in [seeds[1], seeds[3]]:
        noisy = s * np.exp(rng.normal(0, 0.15, n))
        noisy = np.maximum(noisy, 0)
        if noisy.sum() > 0:
            seeds.append(noisy / noisy.sum())
    return seeds


def _optimize_single(a_init, m, t_limit, rng, epoch_idx):
    """Run gradient descent from a_init until t_limit. Return (best_a, best_local_obj)."""
    n = len(a_init)
    a = _proj_simplex(a_init.copy(), 1.0)
    best_a = a.copy()
    c = _conv_sq(a, m)
    best_mc = c.max()
    step = 0
    L = 80
    tau = 0.15
    lr = 0.3 / math.sqrt(n)
    momentum = 0.75
    buf = np.zeros(n, dtype=np.float64)
    stall = 0
    stall_thresh = 0.0005

    while time.monotonic() < t_limit:
        c = _conv_sq(a, m)
        mc = c.max()
        if mc < best_mc - 1e-10:
            best_mc = mc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive parameters
        if step % 1000 == 0 and step > 0:
            L = max(4, int(L * 0.85))
            tau = max(0.005, tau * 0.9)
            momentum = min(0.93, momentum + 0.005)
        if step % 200 == 0:
            lr = max(1e-6, lr * 0.97)

        # Gradient step
        g = _compute_grad(c, a, L, tau)
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g = g / gn
        buf = momentum * buf - lr * g
        a_new = a + lr * buf
        a = _proj_simplex(a_new, 1.0)

        # Periodic spectral refinement
        if step % 300 == 150:
            a_spec = _spectral_flat(a, m)
            blend = 0.3 + 0.2 * math.sin(step * 0.01)
            a = (1 - blend) * a + blend * a_spec
            a = _proj_simplex(a, 1.0)
        if step % 500 == 250:
            a_spec = _spectral_refine(a, m)
            a = 0.6 * a + 0.4 * a_spec
            a = _proj_simplex(a, 1.0)

        # Restart if stalled
        if stall > 5000:
            a = best_a * np.exp(rng.normal(0, 0.3, n))
            a = np.maximum(a, 0)
            if a.sum() > 0:
                a = a / a.sum()
            else:
                a = best_a.copy()
            buf = np.zeros(n, dtype=np.float64)
            lr = 0.3 / math.sqrt(n)
            momentum = 0.7
            stall = 0

        step += 1

    return best_a, 2.0 * n * best_mc
