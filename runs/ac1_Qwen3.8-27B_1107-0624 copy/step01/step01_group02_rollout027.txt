"""
Step function optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-guided direction search with FFT-accelerated convolution,
adaptive active-set selection, and multi-length portfolio search.
"""
import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _normalize(a):
    """Normalize non-negative array to sum 1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(a.size) / a.size
    return a / s


def _score(a):
    """Score = 2*n*max(conv) for a normalized (sum-1) non-negative sequence."""
    n = a.size
    if n < 1:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return 2.0 * n * m


def _resize(a, N):
    """Resize a sequence to length N via interpolation."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m <= 1 or m == N:
        if m == N:
            return _normalize(a)
        return np.full(N, 1.0 / N)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, N)
    return _normalize(np.interp(x_new, x_old, a))


def _load_provided():
    """Load the provided height_sequence_1 if available."""
    try:
        raw = height_sequence_1
    except Exception:
        return None
    try:
        a = np.asarray(raw, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.size == 0 or a.sum() < 1e-300:
            return None
        return _normalize(a)
    except Exception:
        return None


def _make_initial(N, kind, rng, best_seq=None, provided=None):
    """Generate an initial normalized non-negative sequence of length N."""
    N = int(N)
    x = np.linspace(0.0, 1.0, N)

    if kind == -1 and best_seq is not None:
        return _resize(best_seq, N)
    if kind == -2 and provided is not None:
        return _resize(provided, N)
    if kind < 0:
        kind = 6

    if kind == 0:
        # Uniform + linear asymmetry
        a = np.ones(N) + 0.7 * (x - 0.5)
    elif kind == 1:
        # Two blocks (asymmetric)
        L = N // 2
        a = np.ones(N)
        a[:L] = 1.6
        a[L:] = 0.4
    elif kind == 2:
        # Quadratic shape
        a = np.ones(N) + 1.0 * (2.0 * x - 1.0) ** 2
    elif kind == 3:
        # Sine perturbation
        a = np.ones(N) + 0.8 * np.sin(2.0 * np.pi * x + 0.3)
    elif kind == 4:
        # Cosine bump
        a = 0.5 + 0.5 * np.cos(np.pi * (x - 0.3))
    elif kind == 5:
        # Random smooth
        base = rng.normal(0, 1, N)
        w = np.ones(max(1, N // 20))
        w /= w.sum()
        base = np.convolve(base, w, mode='same')
        a = np.exp(0.5 * base)
    elif kind == 6:
        # Smooth random
        a = np.ones(N) + 0.4 * rng.normal(0, 1, N)
        w = np.ones(max(1, N // 30))
        w /= w.sum()
        a = np.convolve(a, w, mode='same')
    else:
        a = np.ones(N)

    return _normalize(a)


def _build_matrix(f, rows):
    """Build LP constraint matrix: A[r, j] = f[k-r - j] for valid indices."""
    n = f.size
    rows = np.asarray(rows, dtype=np.int64)
    A = np.zeros((rows.size, n), dtype=np.float64)
    for r in range(rows.size):
        k = int(rows[r])
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            A[r, lo:hi] = f[k - np.arange(lo, hi)]
    return A


def _lp_direction(f, M, active_rows):
    """
    Solve LP: max sum(g) s.t. A*g <= M, g >= 0.
    Returns normalized g (sum 1) if sum(g) > 1, else None.
    """
    if linprog is None:
        return None
    n = f.size
    if n > 8192 or M < 1e-300:
        return None

    try:
        A = _build_matrix(f, active_rows)
    except Exception:
        return None

    if A.size == 0 or not np.any(A > 0):
        return None

    A_scaled = A / M
    c_vec = -np.ones(n)
    b_vec = np.ones(A_scaled.shape[0])

    try:
        opts = {'maxiter': 5000, 'presolve': True}
        res = linprog(c_vec, A_ub=A_scaled, b_ub=b_vec,
                      bounds=(0.0, None), method='highs', options=opts)
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1.0 + 1e-8:
            return g / s
    return None


def _subgradient_step(f, c, M, alpha):
    """Take a subgradient descent step on max(conv)."""
    n = f.size
    # Find active positions
    threshold = M * 0.98
    active = np.where(c >= threshold)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))])
    if active.size > 50:
        active = active[np.argsort(c[active])[-50:]]

    # Compute subgradient (average of gradients at active positions)
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            grad[lo:hi] += 2.0 * f[k - np.arange(lo, hi)]
    grad /= active.size

    # Remove mean to stay on sum=1 hyperspace
    grad -= grad.mean()

    h = f - alpha * grad
    return _normalize(h)


def _optimize_trajectory(f, time_limit, rng, state, verbose=False):
    """
    Run local optimization on a single trajectory.
    Combines LP directions, subgradient steps, and random escapes.
    """
    f = _normalize(f)
    n = f.size
    if n < 2:
        f = np.array([1.0/3, 2.0/3])
        n = 2

    deadline = time.monotonic() + max(0.1, time_limit)
    best_score = _score(f)
    best_f = f.copy()
    no_improve = 0
    it = 0

    # Update state
    if best_score < state['score']:
        state['score'] = best_score
        state['seq'] = best_f.copy()
        if verbose:
            print(f"    init: score={best_score:.10f} n={n}", flush=True)

    while time.monotonic() < deadline:
        it += 1

        c = np.convolve(f, f)
        M = float(c.max())
        cur = 2.0 * n * M

        if cur < best_score - 1e-12:
            best_score = cur
            best_f = f.copy()
            no_improve = 0
            if best_score < state['scale']:
                state['score'] = best_score
                state['seq'] = best_f.copy()
                if verbose or best_score < 1.52:
                    print(f"    it={it} score={best_score:.10f} n={n}", flush=True)
        else:
            no_improve += 1

        # Escape mechanism
        if no_improve > 30:
            scale = 0.05 + 0.1 * min(1.0, no_improve / 60.0)
            f = _normalize(f * np.exp(rng.normal(0, scale, n)))
            no_improve = 15
            continue

        moved = False

        # LP direction (primary method)
        if linprog is not None and n <= 8192:
            # Select active rows: positions near the peak
            frac = 0.96 if no_improve < 10 else 0.93
            threshold = M * frac
            active = np.where(c >= threshold)[0]

            # Limit number of rows for LP efficiency
            max_rows = min(len(active), max(20, n // 3))
            if len(active) > max_rows:
                idx = np.argsort(c[active])[::-1][:max_rows]
                active = active[idx]

            if active.size >= 3:
                g = _lp_direction(f, M, active)
                if g is not None:
                    # Line search
                    best_t = 0.0
                    best_sc = cur
                    ts = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
                    for t in ts:
                        h = _normalize((1.0 - t) * f + t * g)
                        sc = _score(h)
                        if sc < best_sc:
                            best_sc = sc
                            best_t = t
                    if best_t > 0:
                        f = _normalize((1.0 - best_t) * f + best_t * g)
                        moved = True

        # Subgradient fallback
        if not moved:
            for alpha in [0.02, 0.05, 0.1, 0.2, 0.4]:
                h = _subgradient_step(f, c, M, alpha)
                sc = _score(h)
                if sc < cur - 1e-12:
                    f = h
                    moved = True
                    break

        # Last resort: small random perturbation
        if not moved:
            f = _normalize(f * np.exp(rng.normal(0, 0.03, n)))

    # Final check
    final_sc = _score(f)
    if final_sc < best_score:
        best_score = final_sc
        best_f = f.copy()
    if best_score < state['score']:
        state['score'] = best_score
        state['seq'] = best_f.copy()

    return best_f, best_score

    f = _normalize(f)
    n = f.size
    deadline = time.monotonic() + time_limit

                state['score'] = best_score

                print(f"  polish: score={best_score:.12f} n={n}", flush=True)

        # Use very tight active set and small steps
        threshold = M * 0.995
        active = np.where(c >= threshold)[0]
        max_rows = min(len(active), max(10, n // 5))
        if len(active) > max_rows:

    fb = np.array([1.0/3, 2.0/3])
    fb_score = _score(fb)
    if fb_score < state['score']:
        state['score'] = fb_score
        state['seq'] = fb.copy()

    print(f"Initial best: {state['score']:.10f}", flush=True)

    # Define tasks: (length, kind, time_budget)
    tasks = []

    # Start from provided
    if provided is not None:
        pN = provided.size
        if 32 <= pN <= 8192:
            tasks.append((pN, -2, 60))
        # Also try resized versions

                tasks.append((target_n, -2, 40))

    # Start from current best
    for target_n in [256, 512, 1024]:
        tasks.append((target_n, -1, 50))

    # Parametric starts at key lengths
    for N in [128, 256, 512]:
        for k in range(7):
            tasks.append((N, k, 12))

    for N in [1024, 2048]:
        for k in [0, 1, 3, 5, 6]:
            tasks.append((N, k, 20))

    # Sort by expected value (prioritize provided/best-resized)
    tasks.sort(key=lambda x: -x[2])

    # Run search
    for N, kind, base_time in tasks:
        now = time.monotonic()
        remaining = hard_deadline - now
        if remaining < 5.0:
            break
        limit = min(float(base_time), remaining)

    # Polish phase: refine the best found
    if state['seq'] is not None:
        remaining = hard_deadline - time.monotonic()
        if remaining > 20:
            polish_budget = min(remaining - 5, 120)
            print(f"\nPolish phase: budget={polish_budget:.0f}s, "
                  f"current best={state['score']:.10f}", flush=True)
            _polish(state['seq'], polish_budget, rng, state)

        # Try upscaling the best and polishing again
        remaining = hard_deadline - time.monotonic()
        if remaining > 30 and state['seq'] is not None:
            cur_n = state['seq'].size
            if cur_n < 1024:
                for next_n in [1024, 2048]:
                    remaining = hard_deadline - time.monotonic()
                    if remaining < 20:
                        break
                    big = _resize(state['seq'], next_n)
                    _optimize_trajectory(big, min(remaining - 10, 60), rng, state)
                    # Polish the result
                    remaining = hard_deadline - time.monotonic()
                    if remaining > 15 and state['seq'] is not None:
                        _polish(state['seq'], min(remaining - 5, 40), rng, state)

    # Final verification
    if state['seq'] is None:
        return [float(1.0/3), float(2.0/3)]

    cand = [float(x) for x in np.asarray(state['seq'], dtype=np.float64)]
    print(f"\nFinal: n={len(cand)}, internal_score={state['score']:.12f}", flush=True)

    try:
        official = evaluate_sequence(cand)
        print(f"Official score: {official:.12f}", flush=True)
    except Exception as e:
        print(f"Error calling evaluate_sequence: {e}", flush=True)

    elapsed = time.monotonic() - t0
    print(f"Total time: {elapsed:.1f}s", flush=True)

    return cand
