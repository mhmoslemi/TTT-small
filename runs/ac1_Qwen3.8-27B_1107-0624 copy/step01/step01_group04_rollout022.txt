def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _norm(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def _score(a):
    """Compute 2n*max(conv)/sum^2 using FFT for large n."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-15:
        return np.inf
    if n <= 1024:
        c = np.convolve(a, a)
    else:
        L = _next_pow2(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def _conv_full(a):
    """Full convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 1024:

    L = _next_pow2(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _conv_ab(a, b):

    P = np.fft.rfft(a, L)
    Q = np.fft.rfft(b, L)
    c = np.fft.irfft(P * Q, L)[:n + m - 1].real
    return c

        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = 2.0 * n * maxc  # sum(p)=1 so score = 2n*maxc
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights on convolution
        z = alpha * (c / maxc - 1.0)
        ez = np.exp(z)
        sz = float(ez.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ez / sz

        # Gradient of smoothed max w.r.t. p
        # d/dp_k [sum_j w_j * c_j] where c_j = sum_i p_i * p_{j-i}
        # = 2 * sum_j w_j * (p * p)_j weighted by p_k
        # Actually: dc_j/dp_k = p_{j-k} + p_{k-j} (if valid)
        # So d(smoothed_max)/dp_k = 2 * sum_j w_j * (p_{j-k} + p_{k-j})/2
        # = 2 * (w * p)_k (by convolution symmetry and commutativity)
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = 2.0 * n * 2.0 * C  # d(score)/dp
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_buf = mu * m_buf + (1.0 - mu) * grad_x
        v_buf = nu * v_buf + (1.0 - nu) * (grad_x * grad_x)
        mhat = m_buf / (1.0 - mu ** t)
        vhat = v_buf / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = _score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _lbfgs_smooth(p, budget_s, alpha=10.0):
    """L-BFGS on softmax-smoothed log-sum-exp of convolution."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, _score(p)
    p = _norm(p)
    budget_s = max(2.0, float(budget_s))

    x0 = np.log(p + 1e-300)
    x0 -= x0.mean()
    L = _next_pow2(2 * n - 1)

    def objective(xv):
        xv = np.asarray(xv, dtype=np.float64).ravel()
        xv = np.clip(xv, -50.0, 50.0)
        xv -= xv.max()
        e = np.exp(xv)
        se = float(e.sum())
        if se <= 0 or not np.isfinite(se):
            return 1e10, np.zeros(n)
        p_local = e / se
        Pg = np.fft.rfft(p_local, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 1e-300)
        # Smoothed max: (1/alpha)*log(sum(exp(alpha*c/maxc)))
        maxc = float(c.max())
        if maxc <= 0:
            return 1e10, np.zeros(n)
        z = alpha * (c / maxc - 1.0)
        ez = np.exp(z)
        sz = float(ez.sum())
        if not np.isfinite(sz) or sz <= 0:
            return 1e10, np.zeros(n)
        w = ez / sz
        val = 2.0 * n * maxc * (1.0 + np.log(sz) / alpha)  # approx
        # Actually the smoothed max is (1/alpha)*log(sum exp(alpha*c_k))
        # = maxc + (1/alpha)*log(sum exp(alpha*(c_k - maxc)))
        smax = maxc + np.log(sz) / alpha
        val = 2.0 * n * smax

        # Gradient
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        grad_p = 2.0 * n * 2.0 * C
        gp = float(np.dot(p_local, grad_p))
        grad_x = p_local * (grad_p - gp)
        return val, grad_x

    t0 = time.monotonic()
    max_iter = 200

    def cbk(xk):
        return (time.monotonic() - t0) >= budget_s * 0.85

    try:
        res = sp_opt.minimize(objective, x0, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'maxfun': 5000,
                                       'callback': cbk, 'gtol': 1e-8})
    except Exception:
        return p, _score(p)

    if res is not None and res.x is not None:
        xv = np.clip(res.x, -50.0, 50.0)
        xv -= xv.max()
        e = np.exp(xv)
        se = float(e.sum())
        if se > 0 and np.isfinite(se):
            p_new = e / se
            f_new = _score(p_new)
            f_old = _score(p)
            if f_new < f_old:
                return p_new, f_new
    return p, _score(p)


def _lp_cutting(p, budget_s, max_rows=2000):
    """LP cutting-plane: iteratively find direction g maximizing sum(g) s.t. (f*g)<=M, g>=0."""
    if not HAS_SCIPY:
        return p, _score(p)
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, _score(p)
    budget_s = max(10.0, float(budget_s))
    if budget_s < 20.0:
        return p, _score(p)

    S0 = np.sqrt(2.0 * n)
    f = _norm(p) * S0
    cur_M = float(_conv_full(f).max())
    cur_score = float(2.0 * n * cur_M / (S0 * S0))
    best_f = f.copy()
    best_score = cur_score

    # Initial row set
    r0 = min(max_rows, max(100, 2 * n))
    rows = list(np.unique(np.linspace(0, 2 * n - 2, r0, dtype=float)).astype(int))
    row_set = set(rows)

    end = time.monotonic() + budget_s
    it = 0

    while time.monotonic() < end and len(rows) > 0:
        it += 1
        rows_arr = np.asarray(rows, dtype=np.int64)
        nr = int(rows_arr.size)

        # Build constraint matrix A (nr x n)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                # A[ii, i] = f[k-i] for i in [lo, hi]
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, cur_M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 3.0)
        tl = max(3.0, min(60.0, rem * 0.3))

        try:
            res = sp_opt.linprog(cobj, A_ub=A, b_ub=b,
                                  bounds=(0.0, None), method='highs',
                                  options={'time_limit': tl, 'presolve': True})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or res.x is None:
            break

        g = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
        sg = float(g.sum())
        if sg <= 1e-12 or not np.isfinite(sg):
            break

        # Check full convolution constraint violation
        fg = _conv_ab(f, g)
        tol = 1e-6 + 1e-6 * cur_M
        viol = fg - cur_M
        max_viol = float(viol.max())

        if max_viol > tol:
            # Add violated rows
            if len(rows) < max_rows:
                order = np.argsort(viol)[::-1]
                added = 0
                for ix in order:
                    if added >= 200 or added >= 0.3 * (max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if viol[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added == 0:
                    break
                continue
            else:
                if max_viol > 10.0 * tol:
                    break
                continue

        # Scale g to match normalization
        g = g * (S0 / sg)

        # Line search
        best_step = None
        best_step_score = np.inf
        for step in (1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02):
            fn = (1.0 - step) * f + step * g
            fn = np.maximum(fn, 0.0)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = _score(fn)
            if np.isfinite(sc) and sc < best_step_score:
                best_step_score = sc
                best_step = fn.copy()

        if best_step is None or best_step_score >= best_score:
            break

        f = best_step
        cur_M = float(_conv_full(f).max())
        cur_score = _score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            if it % 3 == 0:
                print("  LP n=%d it=%d score=%.6f rows=%d" % (n, it, best_score, len(rows)))

    final_score = _score(best_f)
    if final_score < best_score:
        best_score = final
    return _norm(best_f), best_score


def _peak_reduction(p, budget_s):
    """Iteratively reduce the peak of the convolution by adjusting contributing entries."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, _score(p)
    p = _norm(p)
    budget_s = max(2.0, float(budget_s))
    t0 = time.monotonic()
    best_p = p.copy()
    best_score = _score(p)

    while time.monotonic() - t0 < budget_s:
        c = _conv_full(p)
        mc = float(c.max())
        if mc <= 0:
            break
        # Find top peaks
        k_peak = int(np.argmax(c))
        # Entry a[k_peak] and a contributions to c[k_peak]:
        # c[k_peak] = sum_i p[i]*p[k_peak-i]
        # Reduce p at indices that contribute most to the peak
        lo = max(0, k_peak - n + 1)
        hi = min(k_peak, n - 1)
        contrib = np.zeros(n, dtype=np.float64)
        for i in range(lo, hi + 1):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] += p[j]
                contrib[j] += p[i] if j != i else 0.0

        # Slightly reduce high contributors, slightly increase low ones
        reduce_mask = contrib > np.median(contrib[contrib > 0]) if np.any(contrib > 0) else np.ones(n, dtype=bool)
        p[reduce_mask] *= 0.97
        p[~reduce_mask] *= 1.01
        p = np.maximum(p, 0.0)
        p = _norm(p)

        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    return best_p, best_score


def _scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / max(bs, 1e-300))
    return [float(x) for x in b]


def _record(cand, best_p, best_score, best_by_n):
    if cand is None:
        return best_p, best_score
    a = np.asarray(cand, dtype=np.float64).ravel()
    if a.size == 0:
        return best_p, best_score
    sc = _score(a)
    if not np.isfinite(sc):
        return best_p, best_score
    if sc < best_score:
        best_p = a.copy()
        best_score = sc
    n = int(a.size)
    if n not in best_by_n or sc < best_by_n[n][0]:
        best_by_n[n] = (sc, a.copy())
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search entry point."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    print("Starting search, budget=%.0fs" % budget_s)

    # Load previous best if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                best_p, best_score = _record(arr, best_p, best_score, best_by_n)
                print("Loaded previous best: n=%d score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Generate diverse starts at multiple lengths
    lengths = [128, 192, 256, 320, 384, 448, 512, 640, 768, 1024, 1280, 1536, 2048]
    start_kinds = ['greedy', 'cosine', 'gaussian', 'power45', 'flat_top']

    # Quick evaluation of all starts
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for kind in start_kinds:
            if kind == 'greedy':
                p = _start_greedy(n)
            elif kind == 'cosine':
                p = _start_cosine(n)
            elif kind == 'gaussian':
                p = _start_gaussian(n, 0.35)
            elif kind == 'power45':
                p = _start_power(n, 0.45)
            elif kind == 'flat_top':
                p = _start_flat_top(n, 0.5)
            else:
                p = _start_greedy(n)
            best_p, best_score = _record(p, best_p, best_score, best_by_n)

    print("After initialization: best_score=%.6f best_n=%d" % (best_score, int(np.asarray(best_p).size) if best_p is not None else 0))

    # Phase 1: Adam gradient on top candidates
    # Sort by_n and pick top 8
    items = sorted(best_by_n.items(), key=lambda kv: kv[1][0])
    top_candidates = [(n, p) for n, (sc, p) in items[:8]]

    # Allocate gradient time
    grad_budget_total = min(600.0, (deadline - time.monotonic()) * 0.35)
    per_grad = max(10.0, grad_budget_total / max(1, len(top_candidates)))

    for n, p in top_candidates:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        alloc = min(per_grad, rem - 10.0)
        if alloc < 5.0:
            break
        p_out, f_out = _adam_smooth(p, alloc, seed=int(seed) + n)
        best_p, best_score = _record(p_out, best_p, best_score, best_by_n)
        # Also try L-BFGS on the result
        rem2 = deadline - time.monotonic()
        if rem2 > 30.0 and HAS_SCIPY:
            p_lb, f_lb = _lbfgs_smooth(p_out, min(40.0, rem2 - 10.0), alpha=12.0)
            best_p, best_score = _record(p_lb, best_p, best_score, best_by_n)

    print("After gradient phase: best_score=%.6f best_n=%d" % (best_score, int(np.asarray(best_p).size) if best_p is not None else 0))

    # Phase 2: LP cutting-plane on the best candidates
    if HAS_SCIPY:
        # Re-sort
        items = sorted(best_by_n.items(), key=lambda kv: kv[1][0])
        lp_candidates = [(n, p) for n, (sc, p) in items[:6]]

        lp_budget_total = min(800.0, (deadline - time.monotonic()) * 0.55)
        per_lp = max(30.0, lp_budget_total / max(1, len(lp_candidates)))

        for n, p in lp_candidates:
            rem = deadline - time.monotonic()
            if rem < 50.0:
                break
            alloc = min(per_lp, rem - 30.0)
            if alloc < 25.0:
                break
            print("  LP starting n=%d budget=%.0fs current_score=%.6f" % (n, alloc, _score(p)))
            p_out, f_out = _lp_cutting(p, alloc, max_rows=min(3000, 4 * n))
            best_p, best_score = _record(p_out, best_p, best_score, best_by_n)
            print("  LP done n=%d score=%.6f" % (n, _score(p_out)))

            # Follow up with a short Adam run
            rem2 = deadline - time.monotonic()
            if rem2 > 25.0:
                p_ad, f_ad = _adam_smooth(p_out, min(30.0, rem2 - 10.0), seed=int(seed) + 777 + n)
                best_p, best_score = _record(p_ad, best_p, best_score, best_by_n)

    print("After LP phase: best_score=%.6f best_n=%d" % (best_score, int(np.asarray(best_p).size) if best_p is not None else 0))

    # Phase 3: Peak reduction + fine Adam on the absolute best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            p_pr, f_pr = _peak_reduction(best_p, min(30.0, rem * 0.3))
            best_p, best_score = _record(p_pr, best_p, best_score, best_by_n)

        rem = deadline - time.monotonic()
        if rem > 40.0:
            p_fin, f_fin = _adam_smooth(best_p, min(60.0, rem - 10.0), seed=int(seed) + 4242)
            best_p, best_score = _record(p_fin, best_p, best_score, best_by_n)

        if HAS_SCIPY:
            # One more LP on the final best
            rem = deadline - time.monotonic()
            if rem > 60.0:
                n_best = int(np.asarray(best_p).size)
                p_lp2, f_lp2 = _lp_cutting(best_p, min(120.0, rem - 20.0), max_rows=min(3000, 4 * n_best))
                best_p, best_score = _record(p_lp2, best_p, best_score, best_by_n)

                # Final Adam polish
                rem = deadline - time.monotonic()
                if rem > 20.0:
                    p_pol, f_pol = _adam_smooth(best_p, min(30.0, rem - 5.0), seed=int(seed) + 9999)
                    best_p, best_score = _record(p_pol, best_p, best_score, best_by_n)

    print("Final best_score=%.6f best_n=%d time_used=%.1fs" % (
        best_score, int(np.asarray(best_p).size) if best_p is not None else 0, time.monotonic() - t0))

    # Validate with evaluate_sequence
    if best_p is not None:
        try:
            seq = _scale_output(best_p, 10.0)
            gscore = evaluate_sequence(seq)
            if np.isfinite(gscore) and gscore < best_score:
                best_score = float(gscore)
                print("Evaluator confirms: %.6f" % best_score)
            elif np.isfinite(gscore):
                print("Evaluator score: %.6f (vs internal %.6f)" % (gscore, best_score))
        except Exception as ex:
            print("Evaluator check failed: %s" % str(ex))

    if best_p is None:
        best_p = _start_greedy(256)

    return _scale_output(best_p, 10.0)
