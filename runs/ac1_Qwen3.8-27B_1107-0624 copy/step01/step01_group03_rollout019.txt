import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def solve_hankel(w, n, eps=1e-8):
    """Solve (H + eps*I) s = 1 where H[i,j] = w[i+j]."""
    # Build Hankel matrix
    H = np.fromfunction(lambda i, j: w[i+j], (n, n))
    H[np.diag_indices(n)] += eps
    s = np.linalg.solve(H, np.ones(n))
    s = np.maximum(s, 0)
    s_sum = s.sum()
    if s_sum < 1e-15:
        return np.ones(n) / n
    return s / s_sum

def weighted_qp_optimize(s_init, n, rng, max_iters=200, beta_start=1.0, beta_end=100.0, deadline=None):
    """
    Optimize using weighted anti-diagonal QP with softmax weight updates.
    """
    s = s_init.copy()
    s = np.maximum(s, 0)
    s /= s.sum()
    best_s = s.copy()
    best_sc = fast_score(s)
    
    conv = np.convolve(s, s)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
            
        # Compute weights (softmax)
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Center the convolution values for numerical stability
        conv_max = np.max(conv)
        conv_centered = conv - conv_max
        w = np.exp(beta * conv_centered)
        w /= w.sum()
        
        # Solve the QP
        eps = 1e-8
        s_new = solve_hankel(w, n, eps)
        
        # Blend
        t = 0.5  # could be adapted
        s = (1 - t) * s + t * s_new
        s = np.maximum(s, 0)
        s /= s.sum()
        
        conv = np.convolve(s, s)
        sc = fast_score(s)
        
        if sc < best_sc:
            best_sc = sc
            best_s = s.copy()
    
    return best_s, best_sc

```python
import numpy as np
import time
from scipy.optimize import linprog

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u, _ = np.sort(v)[::-1], None
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (sv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (sv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def hankel_matvec(w, v, eps=1e-10):
    """Compute (H + eps*I) @ v where H[i,j] = w[i+j]."""
    n = len(v)
    N = 1
    while N < 3 * n:
        N *= 2
    W = np.fft.rfft(w, N)
    V = np.fft.rfft(v, N)
    result = np.fft.irfft(np.conj(V) * W, N)[:n]
    return result + eps * v

def cg_solve_hankel(w, n, eps=1e-8, tol=1e-10, max_iter=1000):
    """Solve (H + eps*I) x = 1 using conjugate gradient."""
    rhs = np.ones(n)
    x = np.ones(n) / n  # initial guess
    r = rhs - hankel_matvec(w, x, eps)
    p = r.copy()
    rs_old = r @ r
    for _ in range(max_iter):
        Ap = hankel_matvec(w, p, eps)
        pAp = p @ Ap
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = r @ r
        if np.sqrt(rs_new) < tol * np.linalg.norm(rhs):
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    return x

def weighted_qp_optimize(s_init, max_iters=200, beta_start=0.5, beta_end=50.0, 
                         deadline=None, verbose=False):
    """
    Weighted anti-diagonal QP with softmax weight updates.
    At each step: compute weights from current conv, solve QP, update s.
    """
    s = s_init.copy().astype(np.float64)
    s = np.maximum(s, 0)
    s_sum = s.sum()
    if s_sum < 1e-15:
        s = np.ones_like(s)
        s_sum = len(s)
    s /= s_sum
    n = len(s)
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Progress-based beta
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / max(beta_start, 1e-10)) ** progress
        
        # Compute convolution
        conv = np.convolve(s, s)
        
        # Compute softmax weights
        conv_max = conv.max()
        exponents = beta * (conv - conv_max)
        # Clip for numerical stability
        exponents = np.clip(exponents, -500, 0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.ones_like(conv) / len(conv)
        else:
            w /= w_sum
        
        # Solve (H + eps*I) x = 1
        eps = 1e-10
        x = cg_solve_hankel(w, n, eps=eps, tol=1e-8, max_iter=500)
        x = np.maximum(x, 0)
        x_sum = x.sum()
        if x_sum < 1e-15:
            x = np.ones(n) / n
        else:
            x /= x_sum
        
        # Line search: blend s toward x
        best_t = 0.0
        best_sc_t = best_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1 - t) * s + t * x
            cand = project_simplex(cand)
            sc = fast_score(cand)
            if sc < best_sc_t - 1e-15:
                best_sc_t = sc
                best_t = t
        
        if best_t == 0.0:
            # No improvement; try smaller steps or break
            if it > 50:
                break
            s = s + 0.001 * np.random.default_rng(it).standard_normal(n)
            s = project_simplex(s)
            continue
        
        s = (1 - best_t) * s + best_t * x
        s = project_simplex(s)
        sc = fast_score(s)
        
        if sc < best_sc:
            best_sc = sc
            best_s = s.copy()
            if verbose:
                print(f"    WQP iter {it}: score={sc:.6f}, beta={beta:.2f}")
    
    return best_s, best_sc

def gradient_optimize(s_init, max_iters=500, beta_start=1.0, beta_end=200.0,
                      deadline=None, verbose=False):
    """
    Gradient-based optimization using log-sum-exp approximation of max.
    """
    s = s_init.copy().astype(np.float64)
    s = np.maximum(s, 0)
    s /= s.sum()
    n = len(s)
    conv_len = 2 * n - 1
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    # FFT size for cross-correlation
    N_fft = 1
    while N_fft < 3 * n:
        N_fft *= 2
    
    lr = 0.1  # learning rate (will be adapted)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        progress = it / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / max(beta_start, 1e-10)) ** progress
        
        # Compute convolution
        conv = np.convolve(s, s)
        
        # Compute gradient using log-sum-exp
        conv_max = conv.max()
        exponents = beta * (conv - conv_max)
        exponents = np.clip(exponents, -500, 0)
        p = np.exp(exponents)
        p /= p.sum()
        
        # Gradient: grad[j] = 4n * sum_m p[m+j] * s[m]
        # = 4n * cross_correlation(p, s)[j]
        # cross[p,s][j] = sum_m p[m+j] * s[m] = ifft(conj(FFT(s)) * FFT(p))[j]
        P = np.fft.rfft(p, N_fft)
        S = np.fft.rfft(s, N_fft)
        cross = np.fft.irfft(np.conj(S) * P, N_fft)[:n]
        grad = 4.0 * n * cross
        
        # Normalize gradient (the objective also has a (sum s)^2 denominator, 
        # but since we project to simplex, we handle it implicitly)
        
        # Gradient step with backtracking
        alpha = lr
        for _ in range(20):
            s_new = s - alpha * grad
            s_new = project_simplex(s_new)
            sc_new = fast_score(s_new)
            sc_old = fast_score(s)
            if sc_new < sc_old or alpha < 1e-10:
                break
            alpha *= 0.5
        
        s = s_new
        sc = sc_new
        
        if sc < best_sc:
            best_sc = sc
            best_s = s.copy()
            if verbose:
                print(f"    Grad iter {it}: score={sc:.6f}, beta={beta:.1f}, alpha={alpha:.6f}")
        
        # Adapt learning rate
        if it % 50 == 49:
            lr = max(lr * 0.7, 1e-8)
    
    return best_s, best_sc

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if 0<=k-j<n else 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) subject to (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def spectral_shaping(s_init, max_iters=100, deadline=None, verbose=False):
    """
    Spectral shaping: modify the power spectrum to flatten the autocorrelation.
    """
    s = s_init.copy().astype(np.float64)
    s = np.maximum(s, 0)
    s /= s.sum()
    n = len(s)
    
    best_s = s.copy()
    best_sc = fast_score(s)
    
    N_fft = 1
    while N_fft < 4 * n:
        N_fft *= 2
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        M = conv.max()
        
        # Target: reduce the peak
        # In frequency domain: P = |FFT(s)|^2, ifft(P) = conv (up to scaling)
        S = np.fft.rfft(s, N_fft)
        P = np.abs(S) ** 2
        
        # Create a mask that reduces components contributing to the peak
        # The peak is at lag n-1 (center). The frequency components that 
        # contribute to the peak are those where the phase alignment creates 
        # constructive interference at that lag.
        # Simple approach: reduce the overall amplitude where P is large.
        P_max = P.max()
        if P_max < 1e-30:
            break
        
        # Soft threshold: reduce large spectral components
        alpha = 0.1 + 0.1 * it / max(max_iters, 1)
        P_new = P * (1.0 - alpha * (P / P_max))
        P_new = np.maximum(P_new, 1e-20)
        
        # New spectrum: preserve phase, modify amplitude
        S_new = S * np.sqrt(P_new / P)
        
        # Inverse FFT, take real part, clip
        s_new = np.fft.irfft(S_new, N_fft)[:n]
        s_new = np.maximum(s_new, 0)
        s_new_sum = s_new.sum()
        if s_new_sum < 1e-15:
            break
        s_new /= s_new_sum
        
        sc = fast_score(s_new)
        if sc < best_sc:
            best_sc = sc
            best_s = s_new.copy()
            s = s_new.copy()
            if verbose:
                print(f"    Spec iter {it}: score={sc:.6f}")
    
    return best_s, best_sc
