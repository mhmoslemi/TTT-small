# Precompute FFT sizes
FFT_SELF = 1 << (2*N-1).bit_length()  # For self-convolution
FFT_CROSS = 1 << (2*N-1).bit_length() # For cross-convolution

# Global state
_best_score = float('inf')
_best_p = None

def probs_from_z(z):
    """Convert logits to probabilities."""
    z = np.clip(z, -50, 50)
    z = np.where(np.isfinite(z), z, 0)
    m = np.max(z) if z.size else 0
    exp_z = np.exp(z - m)
    exp_0 = np.exp(-m)
    Z = exp_0 + np.sum(exp_z)
    p = np.empty(N)
    p[0] = exp_0 / Z
    p[1:] = exp_z / Z
    return p

def self_conv(p):
    """Compute self-convolution using FFT."""
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f*f, FFT_SELF).real
    c = c[:2*N-1]
    np.maximum(c, 0, out=c)
    return c

def score(p):
    c = self_conv(p)
    return 2*N*np.max(c)

def cross_conv(p, w):
    """Compute p convolved with w (padded to same length)."""
    f = np.fft.rfft(p, FFT_CROSS)
    g = np.fft.rfft(w, FFT_CROSS)
    return np.fft.irfft(f*g, FFT_CROSS).real

def grad_from_weights(p, w):
    """Gradient of sum_k w[k]*(p*p)[k] w.r.t. p."""
    # d/dp[i] (p*p)[k] = p[k-i] + p[i-k] (symmetric)
    # So grad[i] = sum_k w[k] * (p[k-i] + p[i-k])
    # = sum_k w[k] * p[k-i] + sum_k w[k] * p[i-k]
    # = (w * p)[i] + (w_rev * p)[i] where w_rev[k] = w[2N-2-k]
    # Actually by symmetry of the convolution, both terms are the same:
    # sum_k w[k]*p[k-i] = sum_j w[i+j]*p[j] (substituting j=k-i)
    # And by the symmetry of the problem, the gradient is:
    # grad[i] = 2 * sum_j p[j] * w[i+j]
    # This can be computed as a cross-correlation.
    
    # Using FFT: grad = 2 * (p[::-1] * w) convolved and extracted
    # (p[::-1] * w)[k] = sum_j p[n-1-j] * w[k-j]
    # We want sum_j p[j] * w[i+j] for each i
    # = sum_j p[j] * w[i+j]
    # Let's define q[j] = p[n-1-j], then sum_j p[j]*w[i+j] = sum_j q[n-1-j]*w[i+j]
    # = sum_j q[m]*w[i+n-1-m] where m = n-1-j
    # = (q * w_rev)[i] where w_rev[k] = w[2N-2-k]... 
    
    # Actually, the simplest way: 
    # grad_p[i] = 2 * sum_j p[j] * w[i+j]
    # This is 2 * (p * w_flipped_reversed)[i] or something like that.
    
    # Let me just use the direct FFT approach:
    # Define the "full" correlation: (p ⊛ w)[k] = sum_j p[j] * w[k-j]
    # We want sum_j p[j] * w[i+j] = sum_j p[j] * w[(i+j)]
    # Let k = i+j, j = k-i: sum_k p[k-i] * w[k] = (p_shifted * w)[i]
    # where p_shifted[k] = p[k-i]... this is getting circular.
    
    # Simplest: use the fact that derivative of (p*p)[k] w.r.t. p[i] is p[k-i] (if in range)
    # So grad[i] = sum_k w[k] * p[k-i] (summing over valid k)
    # = sum_k w[k] * p[k-i]
    # Let j = k-i: sum_j w[j+i] * p[j]
    # This is the i-th element of the "cross-correlation" of w and p.
    # cross_corr[i] = sum_j w[i+j] * p[j]
    
    # Using FFT: cross_corr = ifft(fft(p, 2N) * conj(fft(w, 2N)))... no that's not right either.
    
    # Let me just do it directly with the approach from the previous code:
    # _grad_p_from_weights computes: conv = full_convolve(p[::-1], w), then h = conv[N-1:2N-1]
    # full_convolve(p[::-1], w)[k] = sum_j p[::-1][j] * w[k-j] = sum_j p[N-1-j] * w[k-j]
    # For h[i] = conv[N-1+i]: sum_j p[N-1-j] * w[N-1+i-j]
    # Let m = N-1-j: sum_m p[m] * w[N-1+i-(N-1-m)] = sum_m p[m] * w[i+m]
    # So h[i] = sum_m p[m] * w[i+m] ✓
    # And grad = 2*h ✓
    
    conv = cross_conv(p[::-1], w)
    h = conv[N-1:2*N-1]
    return 2.0 * h

def objective_and_grad(x, alpha):
    """Smooth objective and gradient in logit space."""
    global _deadline, _best_score, _best_p
    if time.monotonic() >= _deadline:
        raise Timeout
    
    z = np.clip(x, -40, 40)
    z = np.where(np.isfinite(z), z, 0)
    p = probs_from_z(z)
    c = self_conv(p)
    
    # Update best
    s = 2*N*np.max(c)
    if s < _best_score:
        _best_score = s
        _best_p = p.copy()
    
    # Smooth surrogate
    cmax = np.max(c)
    u = alpha * (c - cmax)
    e = np.exp(np.clip(u, -50, 50))
    s_exp = np.sum(e)
    if s_exp <= 0 or not np.isfinite(s_exp):
        J = cmax
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        J = cmax + np.log(s_exp) / alpha
        w = e / s_exp
    
    gp = grad_from_weights(p, w)
    sdot = np.dot(gp, p)
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz

def lbfgs_optimize(z0, alpha, max_fun, deadline):

                'gtol': 1e-10,
                'maxls': 40,
                'disp': False,
            }

    except Exception:
        pass

"""
Minimize 2n * max_k (p*p)[k] over non-negative probability vectors p of length n=4096.

Algorithm:
1. Generate diverse initial candidates (structured families + previous best)
2. Coarse-to-fine L-BFGS-B optimization with log-sum-exp surrogate
3. Momentum-accelerated subgradient polishing with adaptive peak count
4. Periodic L-BFGS-B restarts during polishing
5. Final sharp-phase optimization

import numpy as np
import time
from scipy.optimize import minimize

N = 4096
FFT_SELF = 1 << (2 * N - 1).bit_length()
FFT_CROSS = 1 << (2 * N - 1).bit_length()

_best_score = float('inf')
_best_p = None
_deadline = 0.0


class _Timeout(Exception):
    pass


def _probs_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    exp_z = np.exp(z - m)
    exp_0 = np.exp(-m)
    Z = exp_0 + float(np.sum(exp_z))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp_0 / Z
    p[1:] = exp_z / Z
    return p


def _self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    c = _self_conv(p)
    return 2.0 * N * float(np.max(c))


def _cross_conv(a, b):
    fa = np.fft.rfft(a, FFT_CROSS)
    fb = np.fft.rfft(b, FFT_CROSS)
    return np.fft.irfft(fa * fb, FFT_CROSS).real


def _grad_from_weights(p, w):
    conv = _cross_conv(p[::-1], w)
    h = conv[N - 1:2 * N - 1]
    return 2.0 * h


def _update_best(p, s):
    global _best_score, _best_p
    if np.isfinite(s) and s < _best_score:
        _best_score = float(s)
        _best_p = p.copy()


def _score_z(z):
    p = _probs_from_z(z)
    s = _score(p)
    _update_best(p, s)
    return s


def _obj_grad(x, alpha):
    global _deadline, _best_score, _best_p
    if time.monotonic() >= _deadline:
        raise _Timeout
    z = np.asarray(x, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -45.0, 45.0)
    p = _probs_from_z(z)
    c = _self_conv(p)
    s = 2.0 * N * float(np.max(c))
    _update_best(p, s)
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    u = np.clip(u, -60.0, 60.0)
    e = np.exp(u)
    s_exp = float(np.sum(e))
    if s_exp <= 0.0 or not np.isfinite(s_exp):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s_exp) / alpha
        w = e / s_exp
    gp = _grad_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e5, 1e5)
    return float(J), gz


def _lbfgs(z0, alpha, max_fun, deadline):
    global _deadline
    _deadline = float(deadline)
    bounds = [(-45.0, 45.0)] * (N - 1)
    try:
        minimize(
            _obj_grad, z0, args=(float(alpha),),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'maxfun': int(max_fun), 'maxiter': max(int(max_fun) // 2, 50),
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40, 'disp': False}
        )
    except _Timeout:
        pass
    except Exception:
        pass


def _polish(z0, deadline, rng, use_restarts=True):
    global _deadline, _best_score, _best_p
    _deadline = float(deadline)
    z = np.asarray(z0, dtype=np.float64).copy()
    p = _probs_from_z(z)
    sc = _score(p)
    _update_best(p, sc)
    v = np.zeros_like(z)
    beta_mom = 0.85
    step_base = 0.4
    noise = 0.15
    fails = 0
    consecutive_fails = 0
    last_restart = time.monotonic()

    while time.monotonic() < deadline:
        c = _self_conv(p)
        cmax = float(np.max(c))
        # Adaptive top-K: more peaks when far from optimum
        ratio = sc / (2.0 * N * (1.0 / N))  # compared to uniform
        if ratio > 1.7:
            topk = 32
        elif ratio > 1.55:
            topk = 16
        elif ratio > 1.52:
            topk = 8
        else:
            topk = 4
        topk = min(topk, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros_like(c)
        wv = c[idx] + 1e-18
        w[idx] = wv / float(np.sum(wv))
        gp = _grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))
        accepted = False
        if gnorm > 1e-15:

            v = beta_mom * v - step_base * direction

                if np.isfinite(s_try) and s_try < sc - 1e-14:
                    z = z_try
                    p = p_try
                    sc = s_try
                    _update_best(p, sc)
                    accepted = True
                    consecutive_fails = 0
                    break
        if not accepted:

            best_local = sc
            best_z_local = z
            best_p_local = p
            n_tries = 7
            for _ in range(n_tries):
                z_try = z + rng.normal(0.0, noise, N - 1)
                z_try = np.clip(z_try, -45.0, 45.0)
                p_try = _probs_from_z(z_try)
                s_try = _score(p_try)

                    best_z_local = z_try
                    best_p_local = p_try
            if best_local < sc - 1e-12:

                p = best_p_local
                sc = best_local
                v = np.zeros_like(z)

                consecutive_fails = 0
                accepted = True
            else:

                consecutive_fails += 1

                v *= 0.5

                if consecutive_fails > 250:
                    break
        else:
            noise = min(0.4, noise * 1.08)
            step_base = min(0.5, step_base * 1.02)

        # Periodic L-BFGS-B restart
        if use_restarts and accepted and (fails % 40 == 0 or consecutive_fails == 0) and fails > 20:
            now = time.monotonic()
            if now - last_restart > 30.0 and now < deadline - 60.0:
                last_restart = now
                z_restart = z + rng.normal(0.0, 0.02, N - 1)
                alpha_r = 5e6 if sc > 1.53 else 2e7
                _lbfgs(z_restart, alpha_r, 300, min(now + 45.0, deadline - 30.0))
                # Update z from best_p

                    f = np.maximum(_best_p, 1e-15)
                    z = np.clip(np.log(f[1:] / f[0]), -45.0, 45.0)
                    p = _best_p.copy()
                    sc = _best_score
                    v = np.zeros_like(z)

    return z, p, sc
