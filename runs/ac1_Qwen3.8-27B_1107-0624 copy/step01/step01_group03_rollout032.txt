def balance_step(s, alpha=0.1):
    """One balancing step: reduce peak by moving mass to quiet positions."""
    n = len(s)
    conv = np.convolve(s, s)
    M = conv.max()
    k_star = np.argmax(conv)
    
    # Find elements that contribute to the peak
    # s[j] contributes s[j]*s[k_star-j] to conv[k_star]
    contributions = np.zeros(n)
    for j in range(n):
        kj = k_star - j
        if 0 <= kj < n:
            contributions[j] = s[j] * s[kj]
    
    # Total contribution = M (should be, up to floating point)
    # We want to reduce the top contributors
    top_frac = 0.5  # Reduce the top 50% of contributors
    
    # Find the "quietest" positions (smallest convolution values)
    # Moving mass to position j' increases conv[k] for k near j' by about 2*mass*s[k-j']
    # To minimize the increase, move to positions where the local convolution is lowest
    
    # Simple heuristic: move mass from high contributors to low contributors
    sorted_by_contrib = np.argsort(contributions)[::-1]
    sorted_by_conv = np.argsort(conv)  # Quietest lags first
    
    # Positions to reduce
    reduce_indices = sorted_by_contrib[:int(n * top_frac)]
    # Positions to increase (choose from quiet areas)
    increase_indices = sorted_by_conv[:int(n * top_frac)]
    # Make sure they don't overlap too much
    increase_indices = [i for i in increase_indices if i not in set(reduce_indices)][:len(reduce_indices)]
    
    if len(increase_indices) == 0:
        return s
    
    # Move mass
    total_reduce = sum(alpha * contributions[j] / M * s[j] for j in reduce_indices)
    per_reduce = total_reduce / len(reduce_indices) if len(reduce_indices) > 0 else 0
    per_increase = total_reduce / len(increase_indices)
    
    s_new = s.copy()
    for j in reduce_indices:
        s_new[j] = max(0, s_new[j] - per_reduce)
    for i in increase_indices:
        s_new[i] += per_increase
    
    # Renormalize
    s_new = s_new / s_new.sum() * s.sum()
    return s_new

At each step, we want to find a direction d such that moving from s to s + t*d (for small t > 0) reduces the peak of conv(s+td, s+td).
