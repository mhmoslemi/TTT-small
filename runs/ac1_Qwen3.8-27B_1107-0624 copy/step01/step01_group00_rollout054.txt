"""
Minimize 2*n*max(autocorr)/(sum a)^2 for non-negative step functions.
Strategy: FISTA with log-sum-exp smoothed max-autocorrelation objective,
temperature annealing, Adam optimizer, multiple n values and restarts.
"""

import time
import math
import numpy as np

def _proj_simplex(v, target=1.0):
    """Project v onto the standard simplex scaled by target: {x>=0, sum(x)=target}."""
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_sq(a):
    """Linear autocorrelation a*a via FFT."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]

def _compute_grad(a, p):
    """Compute gradient of smoothed objective: grad[i] = 2*sum_j p[i+j]*a[j]."""
    n = a.size
    r = p[::-1]
    m2 = 1
    while m2 < 3 * n - 1:
        m2 <<= 1
    A = np.fft.rfft(a, m2)
    R = np.fft.rfft(r, m2)
    conv = np.fft.irfft(A * R, m2)[:3 * n - 2]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

def _adam_fista_lse(a_init, n, t_end, rng, verbose=False):
    """Optimize using Adam with LSE-smoothed max-autocorrelation."""
    a = _proj_simplex(np.abs(a_init).astype(np.float64), 1.0)
    
    best_a = a.copy()
    best_mc = _conv_sq(a).max()
    
    # Adam state
    m_state = np.zeros_like(a)
    v_state = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1e-3  # Adam learning rate (adjusted for scale)
    
    # Temperature
    tau = 0.005  # initial temperature (relative to c scale)
    
    step = 0
    t_start = time.monotonic()
    last_improve = 0
    no_improve_counter = 0
    
    while time.monotonic() < t_end - 1.0:
        c = _conv_sq(a)
        mc = c.max()
        
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        
        if no_improve_counter > 5000:
            # Restart with perturbation
            a = _proj_simplex(best_a * np.exp(rng.normal(0, 0.1, n)), 1.0)
            m_state = np.zeros_like(a)
            v_state = np.zeros_like(a)
            no_improve_counter = 0
            tau = 0.01
            continue
        
        # Scale tau relative to current max
        tau_eff = max(mc * 0.05, 1e-10)
        
        # LSE weights
        shifted = (c - mc) / tau_eff
        shifted = np.clip(shifted, -100, 0)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
        
        # Gradient
        grad = _compute_grad(a, p)
        
        # Adam update
        step += 1
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad ** 2
        m_hat = m_state / (1 - beta1 ** step)
        v_hat = v_state / (1 - beta2 ** step)

    # Random smooth
    x = np.sort(rng.random(n))
    r = np.exp(-((np.arange(n) - 0.5*n) / (0.1*n + 0.01))**2 * x)
    seeds.append(r / r.sum())
    
    # If base sequence provided, include it (resized to n)
    if base_seq is not None and len(base_seq) > 0:
        b = np.asarray(base_seq, dtype=np.float64)
        b = np.clip(b, 0, None)
        if b.sum() > 0:
            # Resample to length n
            idx = np.linspace(0, len(b)-1, n).astype(int)
            b_r = b[idx]
            if b_r.sum() > 0:
                seeds.append(b_r / b_r.sum())
    
    # Normalize all seeds
    out = []
    for s in seeds:
        s = np.clip(np.asarray(s, dtype=np.float64), 0, None)
        if s.sum() > 0:
            out.append(s / s.sum())
        else:
            out.append(np.ones(n) / n)
    return out

def propose_candidate(seed=42, budget_s=1800):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_total = time.monotonic() + max(0.0, float(budget_s)) - 5.0
    
    # Get base sequence if available
    base = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        if b.ndim == 1 and b.size >= 4 and np.isfinite(b).all():
            base = b
    except:
        pass
    
    # Try multiple n values
    n_values = [256, 512, 1024]
    
    best_overall_a = None
    best_overall_score = float('inf')
    
    # Allocate time: give more time to n=512 (previous best was at this n)
    time_weights = {256: 0.25, 512: 0.45, 1024: 0.30}
    
    for n in n_values:
        t_limit = t_total - (time.monotonic() - (t_total - (budget_s - 5))) * time_weights[n]
        # Simpler: split remaining time
        remaining = t_total - time.monotonic()
        if remaining < 10:
            break
        t_for_n = remaining * time_weights[n] / sum(time_weights.values())
        t_limit = time.monotonic() + t_for_n - 1.0
        
        print(f"\n=== n={n}, time budget={t_for_n:.1f}s ===")
        
        seeds = _get_seeds(n, base, rng)
        
        for si, s in enumerate(seeds):
            if time.monotonic() > t_limit:
                break
            t_for_seed = t_limit - time.monotonic()
            if t_for_seed < 5:
                break
            
            print(f"  Seed {si}: ", end="", flush=True)
            a_best, mc = _adam_fista_lse(s, n, t_for_seed, rng)
            
            # Evaluate
            try:
                score = float(evaluate_sequence([float(x) for x in a_best]))
            except:
                score = float('inf')
            
            obj = 2 * n * mc  # our internal objective (with sum=1)
            print(f"mc={mc:.8f} obj={obj:.6f} official={score:.6f}")
            
            if score < best_overall_score:
                best_overall_score = score
                best_overall_a = a_best.copy()
                print(f"  *** NEW BEST: {score:.6f} ***")
        
        # After trying all seeds for this n, do a refinement on the best
        if best_overall_a is not None and len(best_overall_a) == n:
            t_rem = t_limit - time.monotonic()
            if t_rem > 10:
                print(f"  Refinement pass for n={n}...")
                a_ref, mc_ref = _adam_fista_lse(best_overall_a, n, time.monotonic() + t_rem - 1, rng)
                try:
                    score = float(evaluate_sequence([float(x) for x in a_ref]))
                except:
                    score = float('inf')
                if score < best_overall_score:
                    best_overall_score = score
                    best_overall_a = a_ref.copy()
                    print(f"  *** REFINED BEST: {score:.6f} ***")
    
    # Final
    if best_overall_a is None:
        # Fallback: uniform
        n = 512
        best_overall_a = np.ones(n) / n
        best_overall_score = float(evaluate_sequence([float(x) for x in best_overall_a]))
    
    print(f"\nFINAL: score={best_overall_score:.6f}, n={len(best_overall_a)}")
    return [float(x) for x in best_overall_a]
