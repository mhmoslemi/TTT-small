"""
Optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Iteratively reweighted projected gradient (QP-based)
2. Smooth gradient descent with log-sum-exp
3. Peak subgradient refinement
All with simplex projection for normalization.
"""

import numpy as np
import time


def convolve_fft(a):
    n = len(a)
    size = 2 * n - 1
    # Round up to power of 2 for FFT efficiency
    ps = 1
    while ps < size:
        ps <<= 1
    fa = np.fft.rfft(a, ps)
    result = np.fft.irfft(fa * fa, ps)
    return result[:2 * n - 1]


def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_grad(a, tau):
    n = len(a)
    ps = 2 * n - 1
    p2 = 1
    while p2 < ps:
        p2 <<= 1
    
    fa = np.fft.rfft(a, p2)
    c = np.fft.irfft(fa * fa, p2)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    cmax = np.max(c)
    sh = np.clip((c - cmax) / tau, -50, 50)
    es = np.exp(sh)
    se = np.sum(es)
    Tmax = cmax + tau * np.log(se)
    p = es / se
    
    J = 2.0 * n * Tmax / (S * S)
    
    fp = np.fft.rfft(p, p2)
    cross = np.fft.irfft(np.conj(fa) * fp, p2)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S ** 3)
    return J, grad


def peak_subgrad(a, frac=0.01):
    """Subgradient using peaks within `frac` of the max."""
    n = len(a)
    ps = 2 * n - 1
    p2 = 1
    while p2 < ps:
        p2 <<= 1
    
    fa = np.fft.rfft(a, p2)
    c = np.fft.irfft(fa * fa, p2)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    cmax = np.max(c)
    J = 2.0 * n * cmax / (S * S)
    
    active = c >= cmax * (1.0 - frac)
    h = active.astype(np.float64)
    cnt = np.sum(h)
    if cnt == 0:
        h = c == cmax
        cnt = np.sum(h)
    
    fp = np.fft.rfft(h / cnt, p2)
    cross = np.fft.irfft(np.conj(fa) * fp, p2)[:n]
    
    grad = (2.0 * n * cross) / (S * S) - (4.0 * n * cmax) / (S ** 3)
    return J, grad


def project_simplex(v, target):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.maximum(v - tau, 0.0)


def qp_gradient(a, w, p2, r_cache):
    """Compute 2*(W*a) where (W*a)[i] = sum_j w[i+j]*a[j]."""
    n = len(a)
    fa = np.fft.rfft(a, p2)
    conv = np.fft.irfft(fa * r_cache, p2)[:2 * n - 1]
    Wa = np.flip(conv[n - 1:])
    return 2.0 * Wa
