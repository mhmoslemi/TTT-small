def _next_pow2(x):
    """Next power of 2 >= x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_sq(a, m):
    """Linear convolution a*a via FFT, length 2*len(a)-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * len(a) - 1]


def _lse_grad(a, t, m):
    """Compute gradient of LSE_t(conv(a,a)) and the conv values."""
    n = len(a)
    c = _conv_sq(a, m)
    # Stable log-sum-exp
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    w = exp_c / exp_c.sum()
    v = w * c / t  # length 2n-1
    # grad[i] = 2 * sum_{j=0}^{n-1} v[i+j] * a[j]
    # = 2 * cross_correlation(v, a)[i]
    # Using FFT: need m >= (2n-1) + n - 1 = 3n-2
    m_corr = _next_pow2(3 * n)
    fv = np.fft.rfft(v, m_corr)
    fa = np.fft.rfft(a, m_corr)
    grad = 2.0 * np.fft.irfft(fv * np.conj(fa), m_corr)[:n]
    return grad, c


def _mirror_step(a, grad, eta):
    """Multiplicative weights update + simplex normalization."""
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s <= 0 or not np.isfinite(s):
        return a
    return a_new / s

def _make_seeds(n, base_seq, rng):

    # Gaussian at center

            # Resize to n if needed
            if b.size == n:
                seeds.append(b.copy())

                # Linear interpolation to resize
                old_x = np.linspace(0, 1, b.size)
                new_x = np.linspace(0, 1, n)

                seeds.append(np.clip(resized, 0, None))

        freq1 = rng.uniform(1, 5)
        freq2 = rng.uniform(2, 8)

        r = np.clip(r, 0.01, None)
        seeds.append(r)
    
    # Normalize all
    seeds = [s / s.sum() if s.sum() > 0 else np.ones(n)/n for s in seeds]
    return seeds

def _next_pow2(x):
    p = 1
    while p < x:
        p <<= 1
    return p

def _conv_sq_fft(a, m):
    n = len(a)
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2*n-1]

def _lse_and_grad(a, t, m, m_corr):
    """Compute LSE value and gradient."""
    n = len(a)
    c = _conv_sq_fft(a, m)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    w = exp_c / exp_c.sum()
    lse_val = c_max + t * math.log(exp_c.sum())  # stable LSE
    
    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Cross-correlation of w (len 2n-1) and a (len n)
    fw = np.fft.rfft(w, m_corr)
    fa = np.fft.rfft(a, m_corr)
    grad = 2.0 * np.fft.irfft(fw * np.conj(fa), m_corr)[:n]
    
    return grad, c, lse_val

def _mirror_descent_step(a, grad, eta):

    log_a = np.log(np.maximum(a, 1e-300)) - eta * grad

    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s

def _make_seed(n, kind, rng, base=None):
    """Create a seed sequence of length n."""
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'cosine':
        return 0.5 * (1 + np.cos(np.linspace(0, np.pi, n)))
    elif kind == 'gaussian':
        mid = (n - 1) / 2
        return np.exp(-0.5 * ((np.arange(n) - mid) / (n / 4)) ** 2)
    elif kind == 'symmetric_exp':
        mid = (n - 1) / 2
        return np.exp(-2.0 * np.abs(np.arange(n) - mid) / mid)
    elif kind == 'sine':
        return np.sin(np.pi * np.arange(n) / n) ** 2
    elif kind == 'parabolic':
        x = np.arange(n) / (n - 1)
        return 1.0 - 2.0 * (x - 0.5) ** 2
    elif kind == 'sigmoid_blend':
        x = np.arange(n) / (n - 1)
        return 1.0 / (1.0 + 5.0 * np.exp(-8.0 * (x - 0.3))) + 1.0 / (1.0 + 5.0 * np.exp(8.0 * (x - 0.7)))
    elif kind == 'base':
        if base is not None and base.size >= 4:
            if base.size == n:
                return base.astype(np.float64).copy()
            else:
                old_x = np.linspace(0, 1, base.size)
                new_x = np.linspace(0, 1, n)
                return np.interp(new_x, old_x, base)
        return np.ones(n)
    elif kind == 'base_rev':
        if base is not None and base.size >= 4:
            if base.size == n:
                return base.astype(np.float64)[::-1].copy()
            else:
                old_x = np.linspace(0, 1, base.size)
                new_x = np.linspace(0, 1, n)
                r = np.interp(new_x, old_x, base)
                return r[::-1].copy()
        return np.ones(n)
    elif kind == 'random_smooth':
        freq = rng.uniform(1, 4)
        amp = rng.uniform(0.3, 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        r = 1.0 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
