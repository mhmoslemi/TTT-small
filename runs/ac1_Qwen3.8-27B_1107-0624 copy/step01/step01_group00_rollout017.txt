2. Project to the simplex.
3. This gives the new point.

This is a nice parameterization! The gradient w.r.t. `w` is:

`da[i]/dw[j] = a[i] * (δ_{ij} - a[j])` (standard softmax Jacobian)

This is the gradient in the "logit" space. Then I can use vanilla gradient descent or Adam on `w` without any projection!

This is a cleaner approach. Let me use this.

    a0 = np.clip(a0, 1e-10, None)
    a0 = a0 / a0.sum()
    w = np.log(a0)  # logit

    v = np.zeros(n)
    t = 0
    
    best_w = w.copy()
    best_mc = float('inf')

        a = softmax(w)

        wc = c[order]
        w_soft = np.exp(wc / tau)
        w_soft /= w_soft.sum()

        grad_a = compute_grad(a, w_full, n)

        grad_w = a * (grad_a - mean_grad)
        
        # Adam update

        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * grad_w**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr_init / (1 + step * lr_decay)
        w = w - lr * (m_hat / (np.sqrt(v_hat) + 1e-8))
        
        # Evaluate
        mc = c.max()
        if mc < best_mc:

            best_w = w.copy()
    
    a_best = softmax(best_w)
    score = 2 * n * best_mc  # (sum a = 1, so objective = 2n * max_conv / 1)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for a >= 0 using:
- Softmax parameterization (avoids explicit projection)
- Adam optimizer with curriculum on smoothing parameters
- Multiple sequence lengths and diverse initializations
- Hybrid objective (smoothed max + variance penalty)
"""

import time
import math
import numpy as np

def _softmax(w):
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()

def _conv_sq(a, m=None):
    n = len(a)
    if m is None:
        m = 1
        while m < 2 * n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _compute_grad_topL(a, c, L, tau):
    """Compute gradient of smoothed max (top-L softmax) w.r.t. a."""
    n = len(a)
    L = min(L, 2*n - 1)
    # Get top-L indices
    if L == 2*n - 1:
        order = np.arange(2*n - 1)
    else:
        order = np.argpartition(-c, L - 1)[:L]
    wc = c[order]
    # Softmax weights
    wc_shifted = wc - wc.max()
    w_soft = np.exp(wc_shifted / tau)
    w_soft /= w_soft.sum()
    # Gradient: grad[j] = 2 * sum_{k in topL} w_soft[k'] * a[k - j] for valid k-j
    grad = np.zeros(n)
    for idx, (k, wk) in enumerate(zip(order, w_soft)):
        if wk < 1e-10:
            continue
        # a[k-j] for j = 0..n-1: valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # grad[j] += 2 * wk * a[k-j] for j in [j_lo, j_hi]
            # a[k-j] for j in [j_lo, j_hi] = a[j_hi..j_lo] (reversed range)
            # Specifically: j=j_lo -> a[k-j_lo], j=j_hi -> a[k-j_hi]
            # k - j_lo = k - max(0, k-n+1), k - j_hi = k - min(k, n-1)
            if k < n:
                # j_lo=0, j_hi=k: a[k], a[k-1], ..., a[0]
                src = a[:k+1][::-1]  # a[k], a[k-1], ..., a[0]
                grad[:k+1] += 2.0 * wk * src
            else:
                # j_lo = k-n+1, j_hi = n-1: a[k-(k-n+1)], ..., a[k-(n-1)] = a[n-1], ..., a[1]
                # Wait: k-j for j=k-n+1 is n-1, for j=n-1 is k-(n-1)

                grad[k-n+1:] += 2.0 * wk * src
    return grad

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for a >= 0.
Uses softmax parameterization + Adam optimizer with curriculum on
smoothing parameters, multiple sequence lengths, and diverse initializations.
"""

def _softmax(w):
    """Stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(w)) / len(w)
    return e / s


def _conv_sq(a):
    """Linear convolution a*a via FFT."""
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]


def _grad_smoothed_max(a, c, L, tau):
    """Gradient of smoothed max (top-L softmax) of c w.r.t. a."""
    n = len(a)
    L = min(L, 2 * n - 1)
    if L >= 2 * n - 1:
        idx = np.arange(2 * n - 1)
    else:
        idx = np.argpartition(-c, L - 1)[:L]
    vc = c[idx]
    vc = vc - vc.max()
    w = np.exp(vc / tau)
    w /= w.sum()
    grad = np.zeros(n)
    for k, wk in zip(idx, w):
        if wk < 1e-12:
            continue
        # grad[j] += 2*wk * a[k-j] for valid j
        # valid j: max(0, k-n+1) <= j <= min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo <= j_hi:
            # For j in [j_lo, j_hi], a[k-j] ranges from a[k-j_lo] down to a[k-j_hi]
            # k - j_lo to k - j_hi
            src_lo = k - j_hi  # smallest index in a
            src_hi = k - j_lo  # largest index in a
            # grad[j_lo..j_hi] += 2*wk * a[src_hi, src_hi-1, ..., src_lo]
            grad[j_lo:j_hi+1] += 2.0 * wk * a[src_lo:src_hi+1][::-1]
    return grad

    """Optimize from initial point a0 (on simplex, length n) using Adam in logit space."""
    a0 = np.clip(a0, 1e-15, None)
    a0 = a0 / a0.sum()
    w = np.log(a0)
    
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    
    best_mc = _conv_sq(a0).max()
    best_a = a0.copy()
    
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_base = 0.05
    
    step = 0
    t_start = time.monotonic()
    
    while time.monotonic() - t_start < time_limit:
        step += 1
        
        # Curriculum: schedule L and tau
        progress = min(1.0, step / 20000.0)  # full curriculum over 20K steps
        L = max(5, int(L_init * (1 - progress) + 5 * progress)) if L_init > 0 else max(5, int(n * (1 - progress) + 5 * progress))
        # tau: start large, end small
        c = _conv_sq(a := _softmax(w))
        c_scale = c.max()
        tau = c_scale * tau_frac * (0.01 + 0.99 * (1 - progress))
        
        # Gradient
        g_a = _grad_smoothed_max(a, c, L, tau)
        # Convert to logit gradient
        g_w = a * (g_a - np.dot(a, g_a))
        
        # Normalize gradient to prevent instability
        gn = np.linalg.norm(g_w)
        if gn > 1e-10:
            g_w = g_w / gn
        
        # Adam step
        t_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * g_w
        v_adam = beta2 * v_adam + (1 - beta2) * g_w * g_w
        m_hat = m_adam / (1 - beta1 ** t_adam)
        v_hat = v_adam / (1 - beta2 ** t_adam)
        
        # Learning rate with decay
        lr = lr_base * (0.5 + 0.5 * (1 - progress))
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Track best
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
        
        # Periodic output
        if step % 10000 == 0:
            obj = 2 * n * mc
            best_obj = 2 * n * best_mc
            print(f"  step={step} mc={mc:.6e} best_mc={best_mc:.6e} obj={obj:.4f} best_obj={best_obj:.4f} L={L}")
    
    # Final: return best
    a_best = _softmax(np.log(np.clip(best_a, 1e-15, None)))
    return a_best, 2.0 * n * best_mc


def _make_cosine_window(n, c=0.5, width=1.0, power=1.0):
    """Create a cosine-based window centered at c*n with given width."""
    x = np.arange(n) / n
    w = np.cos(np.pi * (x - c) / width * 0.5) ** power
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        w = np.ones(n)
        s = n
    return w / s


def _make_ramp(n, direction='up', power=1.0):
    x = np.linspace(0, 1, n)
    if direction == 'up':
        w = (x + 0.1) ** power
    else:
        w = (1 - x + 0.1) ** power
    return w / w.sum()

    x = np.arange(n) / n
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)

    w = np.exp(-((x - c) / s) ** 2 * (1 + 0.3 * (x - c)))
    w = np.maximum(w, 1e-10)
