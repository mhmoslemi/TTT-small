def project_simplex(v, total=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0: return v
    v = np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
    # Quick if all positive and sum close? no need
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    sv = cssum[-1]
    # If sv <= total, then projection is v? Actually if sum v <= total and all positive? For nonnegative total larger than sum, projection onto simplex with larger total: x = v + (total-sv)/n.     else:
        rho = 0
    theta = (cssum[rho] - total) / (rho + 1.0)

def stationary_weights(B, alpha=1.0):
    m,n=B.shape
    if m==0: return np.array([])
    if m==1: return np.ones(1)
    S = B @ B.T
    bsum = B.sum(axis=1)
    M = S - np.outer(bsum,bsum) / n
    M += alpha*np.eye(m)
    u=np.ones(m)/m
    try: w=np.linalg.solve(M, alpha*u)
    except: w=u
    w=np.nan_to_num(w, nan=0, posinf=0, neginf=0)
    w=np.maximum(w,0)
    sw=float(w.sum())
    if sw <= 1e-15: w=u
    else: w=w/sw
    return w

We treat the evaluator as a mostly black-box fitness, but use a strong surrogate

  1. Active-lag directional corrections from the largest off-peak convolutions.

  7. Occasionally, an LP direction that maximizes L1 mass subject to current
     active convolution constraints.

All sequences are maintained on the simplex (non-negative, sum = 1). The final
result is selected using the real evaluate_sequence() when available.
"""


def project_simplex(v, total=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = total}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    sv = float(cssum[-1])
    idx = np.nonzero(u * (np.arange(1, n + 1, dtype=np.float64)) > (sv - total))[0]
    if idx.size:
        rho = int(idx[-1])
    else:
        rho = 0
    theta = (float(cssum[rho]) - total) / float(rho + 1)
    w = v - theta
    w[w < 0.0] = 0.0
    sw = float(w.sum())
    if sw > 1e-15:
        w = w * (total / sw)
    else:
        w = np.full(n, total / n, dtype=np.float64)
    return w


def normalize_positive(s):
    """Clip to non-negative and normalize to sum 1."""
    s = np.asarray(s, dtype=np.float64)
    s = np.nan_to_num(s, nan=0.0, posinf=0.0, neginf=0.0)
    s = np.maximum(s, 0.0)
    tot = float(s.sum())
    if tot < 1e-15:
        return np.full(s.size, 1.0 / max(1, s.size), dtype=np.float64)
    return s / tot


def resample_height(hseq, n):
    """Resample a nonnegative height sequence to length n and normalize."""
    h = np.asarray(hseq, dtype=np.float64)
    if h.size == 0:
        return np.ones(n) / float(n)
    if h.size == n:
        return normalize_positive(h)
    x_old = np.linspace(0.0, 1.0, h.size)
    x_new = np.linspace(0.0, 1.0, n)
    h2 = np.interp(x_new, x_old, np.maximum(h, 0.0))
    return normalize_positive(h2)


def auto_conv(s):
    """Linear convolution s * s, using FFT for longer sequences."""
    n = s.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 2048:
        return np.convolve(s, s)
    target = 2 * n - 1
    m = 1
    while m < target:
        m <<= 1
    F = np.fft.rfft(s, m)
    c = np.fft.irfft(F * F, m)[:target]
    return np.nan_to_num(c, nan=0.0, posinf=0.0, neginf=0.0)


def proxy_offpeak(s, c=None):
    """
    Proxy objective:
        2 n max_{t != n-1} (s*s)_t / (sum s)^2
    This is scale-invariant and matches the likely hidden off-peak objective.
    """
    s = np.asarray(s, dtype=np.float64)
    n = s.size
    if n <= 1:
        return 2.0
    if c is None:
        c = auto_conv(s)
    S = float(np.sum(s))
    if S < 1e-12:
        return np.inf
    central = n - 1
    if central == 0:
        M = 0.0
    else:
        M_left = float(np.max(c[:central])) if central > 0 else 0.0
        M_right = float(np.max(c[central + 1:])) if central + 1 < c.size else 0.0
        M = max(M_left, M_right)
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (S * S))


def proxy_full(s, c=None):
    """Full peak proxy, mostly for diagnostics/fallbacks."""
    s = np.asarray(s, dtype=np.float64)
    n = s.size
    if n <= 0:
        return np.inf
    if c is None:
        c = auto_conv(s)
    S = float(np.sum(s))
    if S < 1e-12:
        return np.inf
    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (S * S))


def score_actual(s):
    """Call the real evaluator, with a safe fallback to the proxy."""
    s = np.asarray(s, dtype=np.float64)
    seq = [float(x) for x in s]
    try:
        sc = evaluate_sequence(seq)
    except Exception:
        sc = proxy_offpeak(s)
    try:
        sc = float(sc)
    except Exception:
        sc = np.inf
    if not np.isfinite(sc):
        return np.inf
    return sc


def active_indices(c, n, top_k=48, rel=0.90):
    """Return high-activity off-peak convolution indices."""
    if n <= 1:
        return [], []
    central = n - 1
    off = c.copy()
    off[central] = -np.inf
    off = np.nan_to_num(off, nan=-1e300, posinf=-1e300, neginf=-1e300)
    M = float(np.max(off))
    if not np.isfinite(M) or M <= 1e-15:
        return [], []
    order = np.argsort(off)[::-1]
    active = []
    thresh = max(M * rel, M - 1e-12)
    min_count = max(4, max(1, int(top_k) // 4))
    for idx in order:
        if len(active) >= top_k:
            break
        idx = int(idx)
        if off[idx] >= thresh or len(active) < min_count:
            active.append(idx)
    if len(active) < min_count:
        present = set(active)
        for idx in order:
            if len(active) >= min_count:
                break
            idx = int(idx)
            if idx not in present:
                active.append(idx)
                present.add(idx)
    return active, off[active]


def build_active_rows(s, active):
    """
    Build rows A where (s*b)_t = A_t dot b for given full convolution indices t.
    """
    s = np.asarray(s, dtype=np.float64)
    n = s.size
    m = len(active)
    A = np.zeros((m, n), dtype=np.float64)
    for r, t in enumerate(active):
        t = int(t)
        if t < 0 or t >= 2 * n - 1:
            continue
        lo = max(0, t - (n - 1))
        hi = min(n - 1, t)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            A[r, js] = s[t - js]
    return A


def stationary_weights(B, alpha=1.0):
    """
    Find nonnegative weights on active rows that make the combined gradient as
    constant as possible, with regularization toward uniform weights.
    """
    B = np.asarray(B, dtype=np.float64)
    m, n = B.shape
    if m == 0:
        return np.array([], dtype=np.float64)
    if m == 1:
        return np.ones(1, dtype=np.float64)
    S = B @ B.T
    bsum = B.sum(axis=1)
    M = S - np.outer(bsum, bsum) / float(n)
    M = M + alpha * np.eye(m, dtype=np.float64)
    u = np.ones(m, dtype=np.float64) / float(m)
    try:
        w = np.linalg.solve(M, alpha * u)
    except Exception:
        w = u
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    w = np.maximum(w, 0.0)
    sw = float(w.sum())
    if sw <= 1e-15:
        w = u
    else:
        w = w / sw
    return w


def rank_weights(vals, p=8):
    """Strongly emphasize the largest active lag values."""
    vals = np.asarray(vals, dtype=np.float64)
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    vals = np.maximum(vals, 0.0)
    w = vals ** p
    sw = float(w.sum())
    if sw <= 1e-30:
        return np.ones(vals.size, dtype=np.float64) / max(1, vals.size)
    return w / sw


def make_dir_candidates(s, D, alphas=(0.1, 1.0, 10.0, 100.0)):
    """Generate simplex-projected candidates moving opposite to D."""
    s = np.asarray(s, dtype=np.float64)
    D = np.asarray(D, dtype=np.float64)
    out = []
    maxD = float(np.max(np.abs(D)))
    if maxD <= 1e-30:
        return out
    Dn = D / maxD
    base = float(np.mean(s))
    for a in alphas:
        out.append(project_simplex(s - a * base * Dn))
    return out


def fourier_noise_candidate(s, rng, amp=0.2, limit=12, beta=0.7):
    """Add smooth zero-mean Fourier noise in log-density space."""
    s = np.asarray(s, dtype=np.float64)
    n = s.size
    if n <= 1:
        return normalize_positive(s)
    if amp <= 0.0:
        return normalize_positive(s)
    x = np.log(np.maximum(s, 1e-12))
    x = x - float(np.mean(x))
    m = 1
    while m < n:
        m <<= 1
    if m < 4:
        m = 4
    lim = min(int(limit), max(1, int(n // 16)), m // 2 - 1)
    delta = np.zeros(n, dtype=np.float64)
    if lim > 0:
        ks = np.arange(1, lim + 1, dtype=np.float64)
        coeff = amp / (ks ** beta)
        F = np.zeros(m, dtype=np.complex128)
        F[1:lim + 1] = coeff * (rng.standard_normal(lim) + 1j * rng.standard_normal(lim))
        F[-lim:] = np.conj(F[1:lim + 1])
        delta = np.fft.irfft(F, m)[:n]
        delta = delta - float(np.mean(delta))
    cand = np.exp(np.clip(x + delta, -30.0, 30.0))
    return project_simplex(cand)


def bump_candidate(s, rng, amp=0.3, width_frac=0.1):
    """Add a smooth periodic Gaussian-like bump."""
    s = np.asarray(s, dtype=np.float64)
    n = s.size
    if n <= 1:
        return normalize_positive(s)
    center = float(rng.uniform(0.0, n))
    w = max(1.0, float(n) * float(width_frac))
    x = np.arange(n, dtype=np.float64)
    d = np.abs(x - center)
    d = np.minimum(d, n - d)
    bump = np.exp(-(d * d) / (2.0 * w * w))
    sign = 1.0 if rng.random() < 0.5 else -1.0
    cand = s + sign * float(amp) * bump * float(np.mean(s))
    return project_simplex(cand)


def lp_polish(s0, rng, deadline, use_actual=True):
    """
    LP-guided improvement.

    For current s and current off-peak peak M, solve approximately:
        maximize sum(b)
        subject to (s*b)_t <= M on chosen off-peak lags, b >= 0, b <= U.
    Then line-search blend towards b.
    """
    s = normalize_positive(s0)
    n = s.size
    if not HAVE_LINPROG or n <= 1:
        return s, (score_actual(s) if use_actual else proxy_offpeak(s))

    cur = score_actual(s) if use_actual else proxy_offpeak(s)
    best_seq = s.copy()
    best_score = cur
    no_imp = 0
    t_options = (0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0)

    for it in range(2000):
        if time.monotonic() > deadline - 3.0:
            break
        c = auto_conv(s)
        central = n - 1
        off = c.copy()
        off[central] = -np.inf
        off = np.nan_to_num(off, nan=-1e300, posinf=-1e300, neginf=-1e300)
        M = float(np.max(off))
        if not np.isfinite(M) or M <= 1e-15:
            break

        if n <= 300:
            row_idx = [t for t in range(2 * n - 1) if t != central]
        else:
            active, _ = active_indices(c, n, top_k=80, rel=0.95)
            row_idx = active[:80]
        if not row_idx:
            break

        A = build_active_rows(s, row_idx)
        U = max(float(np.max(s)) * 1.5, 20.0 / float(n), 1e-4)
        cobj = -np.ones(n, dtype=np.float64) + 1e-8 * rng.standard_normal(n)
        b_ub = np.full(len(row_idx), M * (1.0 + 1e-12) + 1e-15, dtype=np.float64)

        b = None
        try:
            bounds = [(0.0, float(U))] * n
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"time_limit": 0.6, "presolve": True},
            )
            if res.success and res.x is not None:
                b = np.maximum(np.asarray(res.x, dtype=np.float64), 0.0)
                tb = float(b.sum())
                if tb > 1e-12:
                    b = b / tb
        except Exception:
            b = None

        if b is None:
            no_imp += 1
            if no_imp > 8:
                break
            continue

        best_local = cur
        best_cand = s

        for t in t_options:
            cand = project_simplex((1.0 - t) * s + t * b)
            sc = score_actual(cand) if use_actual else proxy_offpeak(cand)
            if sc < best_local - 1e-12:
                best_local = sc
                best_cand = cand

        if best_local >= cur - 1e-12:
            noisy_b = normalize_positive(b + 0.01 * rng.standard_normal(n))
            for t in (0.2, 0.5, 0.8):
                cand = project_simplex((1.0 - t) * s + t * noisy_b)
                sc = score_actual(cand) if use_actual else proxy_offpeak(cand)
                if sc < best_local - 1e-12:
                    best_local = sc
                    best_cand = cand

        if best_local < cur - 1e-12:
            s = best_cand
            cur = best_local
            no_imp = 0
            if cur < best_score:
                best_score = cur
                best_seq = s.copy()
        else:
            no_imp += 1
            if no_imp > 10:
                break

    return best_seq, best_score


def local_search(s0, n, rng, deadline, use_actual=True):
    """
    Local search around a sequence.

    Returns:
        best_seq, best_score,
        actual_best_seq, actual_best_score

    If use_actual is True, candidates are scored with the real evaluator.
    Otherwise candidates are scored by the off-peak proxy, with periodic
    real-evaluator feedback.
    """
    s0 = np.asarray(s0, dtype=np.float64)
    n = s0.size
    s = normalize_positive(s0)

    if use_actual:
        cur_score = score_actual(s)
    else:
        cur_score = proxy_offpeak(s)

    best_seq = s.copy()
    best_score = cur_score
    proxy_best_seq = s.copy()
    proxy_best_score = proxy_offpeak(s)
    actual_best_seq = s.copy()
    actual_best_score = score_actual(s)

    if np.isfinite(cur_score) and cur_score < best_score:
        best_score = cur_score
        best_seq = s.copy()

    no_imp = 0
    it = 0
    max_cands = 24 if n > 1024 else (28 if n > 512 else 32)
    check_interval = 10 if n > 4096 else 5

    while time.monotonic() < deadline - 3.0:
        it += 1
        c = auto_conv(s)
        top_k = min(64, max(24, int(n // 6)))
        active, act_vals = active_indices(c, n, top_k=top_k, rel=0.90)

        cands = []

        if active:
            B = build_active_rows(s, active)
            if B.shape[0] > 0:
                w_rank = rank_weights(act_vals, p=8)
                D = B.T @ w_rank
                cands.extend(make_dir_candidates(s, D, (0.1, 1.0, 10.0, 100.0)))

                w_stat = stationary_weights(B, alpha=1.0)
                D = B.T @ w_stat
                cands.extend(make_dir_candidates(s, D, (0.3, 3.0, 30.0)))

                for r in range(min(3, int(B.shape[0]))):
                    D = B[r].reshape(n)
                    cands.extend(make_dir_candidates(s, D, (0.5, 5.0, 50.0)))

        for amp in (0.02, 0.10, 0.40, 1.50, 4.00):
            cands.append(
                fourier_noise_candidate(
                    s,
                    rng,
                    amp=amp,
                    limit=min(24, max(4, int(n // 16))),
                    beta=0.7,
                )
            )

        cands.append(bump_candidate(s, rng, amp=0.2, width_frac=0.05))
        cands.append(bump_candidate(s, rng, amp=0.5, width_frac=0.10))

        if len(cands) > max_cands:
            cands = cands[:max_cands]

        if use_actual:
            best_cand = s
            best_sc = cur_score
            for cand in cands:
                cand = project_simplex(cand)
                sc = score_actual(cand)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_cand = cand

            if best_sc < cur_score - 1e-12:
                s = best_cand
                cur_score = best_sc
                no_imp = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = s.copy()
                if cur_score < actual_best_score:
                    actual_best_score = cur_score
                    actual_best_seq = s.copy()
                pr = proxy_offpeak(s)
                if pr < proxy_best_score:
                    proxy_best_score = pr
                    proxy_best_seq = s.copy()
            else:
                no_imp += 1
                if no_imp > 30:
                    noise = rng.standard_normal(n)
                    s = project_simplex(best_seq + 0.3 * float(np.mean(best_seq)) * noise)
                    cur_score = score_actual(s)
                    no_imp = 0
        else:
            scored = []
            ctr = 0
            for cand in cands:
                cand = project_simplex(cand)
                pr = proxy_offpeak(cand)
                scored.append((pr, ctr, cand))
                ctr += 1
            scored.sort()

            if scored and scored[0][0] < cur_score - 1e-12:
                s = scored[0][2]
                cur_score = scored[0][0]
                no_imp = 0
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = s.copy()
                    proxy_best_score = cur_score
                    proxy_best_seq = s.copy()
            else:
                no_imp += 1
                if no_imp > 30:
                    noise = rng.standard_normal(n)
                    s = project_simplex(best_seq + 0.3 * float(np.mean(best_seq)) * noise)
                    cur_score = proxy_offpeak(s)
                    no_imp = 0

            if scored and (it % check_interval == 0):
                top_to_check = 1 if n > 2048 else 2
                for _, _, cand in scored[:top_to_check]:
                    ac = score_actual(cand)
                    if ac < actual_best_score:
                        actual_best_score = ac
                        actual_best_seq = cand.copy()

                ac_cur = score_actual(s)
                if ac_cur < actual_best_score:
                    actual_best_score = ac_cur
                    actual_best_seq = s.copy()

                # If the real evaluator says the proxy best is different, shift
                # the local search toward the real best.
                if np.isfinite(actual_best_score) and actual_best_score < best_score - 0.01:
                    s = actual_best_seq.copy()
                    cur_score = proxy_offpeak(s)
                    no_imp = 0
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = s.copy()
                        proxy_best_score = cur_score
                        proxy_best_seq = s.copy()

        # Keep the returned proxy-best in sync with s if s improves proxy.
        pr_s = proxy_offpeak(s)
        if pr_s < proxy_best_score:
            proxy_best_score = pr_s
            proxy_best_seq = s.copy()

        if best_score < 1.0 + 1e-12:
            break

    if not np.isfinite(actual_best_score):
        actual_best_score = score_actual(best_seq)
    if not np.isfinite(actual_best_score):
        actual_best_seq = best_seq

    return best_seq, best_score, actual_best_seq, actual_best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: search for a low-scoring nonnegative sequence."""
    if budget_s is None:
        budget_s = 1800
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + (budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # Load previously found heights if available.
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.nan_to_num(h, nan=0.0, posinf=0.0, neginf=0.0)
        h = np.maximum(h, 0.0)
        if h.ndim == 1 and h.size >= 8 and float(h.sum()) > 0.01:
            hseq = h / float(h.sum())
            print(f"Loaded height_sequence_1 length={h.size} sum={float(h.sum()):.3f}")
    except Exception:
        hseq = None

    # Initial global best.
    if hseq is not None:
        best_seq = normalize_positive(hseq)
    else:
        best_seq = np.ones(256, dtype=np.float64) / 256.0

    best_actual = score_actual(best_seq)
    print(f"Initial best actual score: {best_actual:.8f}")

    candidates = []
    counter = 0
    p0 = proxy_offpeak(best_seq)
    candidates.append((p0, counter, best_seq.copy()))
    counter += 1

    # Build a small set of lengths to explore.
    length_set = [256, 512, 1024, 2048, 4096, 8192]
    if hseq is not None:
        Lh = int(np.clip(hseq.size, 64, 8192))
        if Lh not in length_set:
            length_set.append(Lh)

    seen = set()
    lengths = []
    for L in length_set:
        if 64 <= L <= 8192 and L not in seen:
            seen.add(L)
            lengths.append(L)

    prio = []
    for L in lengths:
        p = abs(float(np.log(max(1.0, float(L)) / 256.0)))
        if L < 256:
            p += 1.0
        elif L > 256:
            p += 0.05
        prio.append((p, L))
    prio.sort()
    lengths = [L for _, L in prio][:6]

    weights = [40, 25, 15, 10, 6, 4]
    while len(weights) < len(lengths):
        weights.append(4)

    for i, L in enumerate(lengths):
        if time.monotonic() > deadline - 60.0:
            break
        rem_weights = 0.0
        for w in weights[i:]:
            rem_weights += float(w)
        if rem_weights <= 0.0:
            break
        phase_deadline = time.monotonic() + (deadline - time.monotonic()) * (float(weights[i]) / rem_weights)
        phase_deadline = min(phase_deadline, deadline - 20.0)
        if phase_deadline <= time.monotonic() + 20.0:
            continue

        print(f"\n--- Starting length {L}, phase leaves {(phase_deadline - time.monotonic()):.1f}s")

        starts = []
        start_weights = []

        if hseq is not None:
            starts.append(resample_height(hseq, L))
            start_weights.append(0.35)

        starts.append(resample_height(best_seq, L))
        start_weights.append(0.30 if hseq is not None else 0.40)

        uniform = np.ones(L, dtype=np.float64) / float(L)
        starts.append(normalize_positive(uniform))
        start_weights.append(0.10)

        for _ in range(2):
            sm = fourier_noise_candidate(uniform, rng, amp=0.8, limit=min(24, max(4, L // 16)), beta=0.7)
            starts.append(normalize_positive(sm))
            start_weights.append(0.075)

        two_level = np.ones(L, dtype=np.float64)
        two_level[: L // 2] *= 2.0
        starts.append(normalize_positive(two_level))
        start_weights.append(0.075)

        wsum = float(sum(start_weights))
        if wsum <= 0.0:
            start_weights = [1.0 / len(start_weights)] * len(start_weights)
        else:
            start_weights = [w / wsum for w in start_weights]

        phase_left = phase_deadline - time.monotonic()
        for j, s0 in enumerate(starts):
            sb_deadline = time.monotonic() + phase_left * start_weights[j]
            if sb_deadline <= time.monotonic() + 15.0:
                continue
            s0 = normalize_positive(s0)
            print(f"  start {j} (weight {start_weights[j]:.2f}) base proxy={proxy_offpeak(s0):.8f}")

            if HAVE_LINPROG and L <= 512 and j == 0:
                lp_deadline = time.monotonic() + 0.30 * max(1.0, sb_deadline - time.monotonic())
                lb, lsc = lp_polish(s0, rng, lp_deadline, use_actual=True)
                lac = score_actual(lb)
                if np.isfinite(lac) and lac < best_actual:
                    best_actual = lac
                    best_seq = lb.copy()
                    print(f"    LP improved actual to {best_actual:.8f}")
                pbb = proxy_offpeak(lb)
                candidates.append((pbb, counter, lb.copy()))
                counter += 1
                s0 = lb

            use_actual = L <= 1024
            bp, psc, ba, asc = local_search(s0, L, rng, sb_deadline, use_actual=use_actual)

            for seq, sc in [(bp, psc), (ba, asc)]:
                seq = np.asarray(seq, dtype=np.float64)
                seq = normalize_positive(seq)
                a = score_actual(seq)
                if np.isfinite(a) and a < best_actual:
                    best_actual = a
                    best_seq = seq.copy()
                    print(f"    new global actual {best_actual:.8f} length={seq.size}")
                pp = proxy_offpeak(seq)
                candidates.append((pp, counter, seq.copy()))
                counter += 1

            if len(candidates) > 220:
                candidates.sort()
                candidates = candidates[:160]

        # Downsample/interpolate the current best into this length for later?
        # Not needed explicitly because next length will resample best_seq.

    # Final polishing from the global best, if time remains.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 180.0:
        Lbest = len(best_seq)
        print(f"\nFinal polish remaining={remaining:.1f}s length={Lbest}")
        for amp in (0.10, 0.30, 1.00):
            if time.monotonic() > deadline - 60.0:
                break
            s0 = project_simplex(best_seq + amp * float(np.mean(best_seq)) * rng.standard_normal(Lbest))
            deadline_dp = time.monotonic() + 0.25 * remaining
            deadline_dp = min(deadline_dp, deadline - 40.0)
            bp, psc, ba, asc = local_search(s0, Lbest, rng, deadline_dp, use_actual=(Lbest <= 1024))
            for seq, sc in [(bp, psc), (ba, asc)]:
                seq = normalize_positive(seq)
                a = score_actual(seq)
                if np.isfinite(a) and a < best_actual:
                    best_actual = a
                    best_seq = seq.copy()
                    print(f"  final polish actual {best_actual:.8f}")
                pp = proxy_offpeak(seq)
                candidates.append((pp, counter, seq.copy()))
                counter += 1

    # Final verification over a top set of proxy-scored candidates.
    if candidates:
        candidates.sort()
        checked = 0
        print("\nFinal candidate verification:")
        for pp, _, seq in candidates:
            if checked >= 30:
                break
            if time.monotonic() > deadline - 10.0:
                break
            seq = np.asarray(seq, dtype=np.float64)
            seq = normalize_positive(seq)
            a = score_actual(seq)
            checked += 1
            if np.isfinite(a) and a < best_actual:
                best_actual = a
                best_seq = seq.copy()
                print(f"  improved final actual to {a:.8f} at proxy {pp:.8f}")

    # Safety fallback.
    if best_seq is None or best_seq.size == 0 or not np.all(np.isfinite(best_seq)):
        best_seq = np.ones(256, dtype=np.float64) / 256.0

    best_seq = normalize_positive(best_seq)
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    final_score = score_actual(np.asarray(result, dtype=np.float64))
    print(f"Final returned score: {final_score:.8f} length={len(result)}")
    return result
