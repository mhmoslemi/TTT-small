    phase = alpha * t * t
    # Convert to positive: use |cos| or 1 + cos
    modulated = 1.0 + 0.3 * np.cos(2 * np.pi * (n * phase + 0.5 * t))
    return np.maximum(0.01, envelope * modulated)

import numpy as np

def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return 1e15
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_and_grad(x, tau):
    """Returns (J, grad) for smoothed objective."""
    n = len(x)
    c = convolve_fft(x)
    S = np.sum(x)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Cross-correlation for gradient
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(x, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad


def run_lbfgsb(a_init, tau, max_iter=2000, ftol=1e-14):
    """Run L-BFGS-B optimization with given tau."""
    n = len(a_init)

        J, g = smooth_obj_and_grad(x, tau)

        return J, g
    
    bounds = [(0.0, 1000.0)] * n

            obj_grad, a, method='L-BFGS-B',
            jac=True, bounds=bounds,

        )
        if result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    
    return a

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return 1e15
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_and_grad(x, tau):
    """Returns (J, grad) for smoothed objective with log-sum-exp."""
    n = len(x)
    c = convolve_fft(x)
    S = np.sum(x)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation for gradient: cross[i] = sum_k p[k]*a[k-i]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(x, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def run_lbfgsb_timed(a_init, tau, budget_s, max_iter=5000):
    """Run L-BFGS-B with time limit. Returns best array found."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    deadline = time.monotonic() + budget_s
    best_a = a.copy()
    best_J = compute_J(a)
    state = {'done': False}

    def obj_grad(x):
        if state['done'] or time.monotonic() > deadline:
            state['done'] = True
            return 1e15, np.zeros(n)
        J, g = smooth_obj_and_grad(x, tau)
        if not np.isfinite(J):
            return 1e15, np.zeros(n)
        if not np.isfinite(g).all():
            g = np.nan_to_num(g, nan=0.0, posinf=1e6, neginf=-1e6)
        return J, g

    bounds = [(0.0, 1000.0)] * n

    try:
        result = minimize(
            obj_grad, a, method='L-BFGS-B',
            jac=True, bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-14,
                     'gtol': 1e-8, 'maxls': 30}
        )
        if result.x is not None:
            cand = np.maximum(result.x, 0.0)
            cand_J = compute_J(cand)
            if cand_J < best_J:
                best_a = cand
                best_J = cand_J
    except Exception:
        pass

    return best_a


def run_annealed_lbfgsb(a_init, budget_s):
    """Sequential L-BFGS-B with decreasing tau. Returns best array."""
    best_a = a_init.copy()
    best_J = compute_J(a_init)

    # Tau schedule: start large (global), end small (local)
    tau_schedule = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    # Allocate time: give more to smaller tau
    n_phases = len(tau_schedule)
    # Weights increase for later phases
    weights = np.array([1.0 + 0.5 * i for i in range(n_phases)])
    weights /= weights.sum()

    for i, tau in enumerate(tau_schedule):
        remaining = budget_s - (time.monotonic() - (budget_s - (time.monotonic() - time.monotonic())))
        # Just use a fraction of remaining
        frac_rem = (time.monotonic_budget_end - time.monotonic()) if hasattr(time, 'monotonic_budget_end') else budget_s
        # Simpler: allocate proportionally
        alloc = budget_s * weights[i]
        # But don't exceed remaining time
        elapsed_in_this = sum(budget_s * weights[j] for j in range(i))
        remaining = budget_s - elapsed_in_this - 2.0
        alloc = min(alloc, max(0, remaining))

        if alloc < 3.0:
            break

        res = run_lbfgsb_timed(best_a, tau, alloc)
        res_J = compute_J(res)
        if res_J < best_J:
            best_J = res_J
            best_a = res

    return best_a, best_J


def run_coordinate_descent(a, budget_s, seed):
    """Coordinate descent targeting indices contributing to max convolution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    J = best_J

    step_size = 0.12
    no_improve = 0

    while time.monotonic() < deadline:
        c = convolve_fft(a)
        S = np.sum(a)
        J = 2.0 * n * np.max(c) / (S * S)

        if J < best_J - 1e-10:
            best_J = J
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            step_size = max(0.003, step_size * 0.85)
            no_improve = 0
            if step_size < 0.005 and no_improve > 200:
                break
            if rng.random() < 0.4:
                a = best_a.copy()
                J = best_J

        # Target the peak convolution index
        max_lag = int(np.argmax(c))

        # Find top contributors to c[max_lag]
        lo = max(0, max_lag - (n - 1))
        hi = min(n - 1, max_lag)
        contrib = np.zeros(n)
        for i in range(lo, hi + 1):
            j = max_lag - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Try adjusting top contributors
        active = np.where(contrib > 1e-10)[0]
        if len(active) > 0:
            top_k = min(8, len(active))
            top_idx = active[np.argsort(contrib[active])[-top_k:]]
            for idx in top_idx:
                old = a[idx]
                if old < 1e-10:
                    continue
                improved = False
                for scale in [1.0 - step_size, 1.0 + step_size,
                              1.0 - 2.0 * step_size, 1.0 - 0.5 * step_size]:
                    a[idx] = max(0.0, old * scale)
                    new_J = compute_J(a)
                    if new_J < J:
                        J = new_J
                        improved = True
                        break
                    a[idx] = old
                if improved:
                    break

        # Random perturbations
        for _ in range(4):
            idx = int(rng.integers(0, n))
            old = a[idx]
            new_val = old * np.exp(rng.normal(0, step_size))
            a[idx] = max(0.0, min(1000.0, new_val))
            new_J = compute_J(a)
            if new_J < J:
                J = new_J
            else:
                a[idx] = old

    return best_a, best_J


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for optimization."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    inits = {}

    if parent is not None and len(parent) == n:
        inits['parent'] = parent.copy()

    inits['ones'] = np.ones(n)

    # Gaussian shapes
    inits['gauss_nar'] = np.exp(-((x - mid) / (0.15 * n)) ** 2)
    inits['gauss_mid'] = np.exp(-((x - mid) / (0.25 * n)) ** 2)
    inits['gauss_wide'] = np.exp(-((x - mid) / (0.40 * n)) ** 2)

    # Chirp modulations
    t = x / n
    inits['chirp1'] = 1.0 + 0.3 * np.cos(2 * np.pi * (3.0 * t * t + 0.3))
    inits['chirp2'] = 1.0 + 0.4 * np.cos(2 * np.pi * (7.0 * t * t))
    inits['chirp3'] = 1.0 + 0.25 * np.cos(2 * np.pi * (3.0 * t * t)) + \
                     0.25 * np.cos(2 * np.pi * (7.0 * t * t + 0.5))

    # Multi-frequency
    inits['mf1'] = 1.0 + 0.2 * np.cos(2 * np.pi * 1 * x / n) + \
                  0.2 * np.cos(2 * np.pi * 5 * x / n + 0.7) + \
                  0.15 * np.cos(2 * np.pi * 13 * x / n + 1.3)
    inits['mf2'] = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x / n + 0.2) + \
                  0.3 * np.cos(2 * np.pi * 11 * x / n + 1.1)

    # Taper
    inits['taper'] = np.maximum(0.05, 1.0 - 0.6 * np.abs(x - mid) / max(mid, 1))

    # Edge boost
    inits['edge'] = 0.3 + 0.7 * (np.abs(x - mid) / max(mid, 1)) ** 2

    # Smoothed random
    for k in range(3):
        raw = np.exp(rng.normal(0, 0.4 + 0.2 * k, n))
        ks = max(3, n // 300)
        kernel = np.ones(ks) / ks
        inits[f'rand_{k}'] = np.convolve(raw, kernel, mode='same')

    # Make all positive and normalize
    for name in inits:
        inits[name] = np.maximum(inits[name], 0.01)
        S = np.sum(inits[name])
        if S > 0:
            inits[name] = inits[name] * (n / S)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer: L-BFGS-B with annealed log-sum-exp tau,
    coordinate descent on active indices, and iterated local search.
    Explores diverse initializations including chirp and multi-frequency patterns.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial parent score: {best_global_score:.6f}")

    # Generate and score initializations
    inits = make_initializations(n, rng, parent)
    init_list = []
    for name, arr in inits.items():
        J = compute_J(arr)
        if np.isfinite(J):
            init_list.append((J, arr.copy(), name))

    init_list.sort(key=lambda x: x[0])
    print("\nTop initializations:")
    for J, _, name in init_list[:6]:
        print(f"  {name}: J={J:.6f}")

    # ============ PHASE 1: L-BFGS-B annealing on top initializations ============
    phase1_total = (total_deadline - time.monotonic()) * 0.55
    top_k = min(4, len(init_list))

    # Weight allocation: best gets most time
    weights = [1.0, 0.7, 0.5, 0.4][:top_k]
    w_sum = sum(weights)

    for i in range(top_k):
        remaining = total_deadline - time.monotonic()
        if remaining < 30.0:
            break
        alloc = phase1_total * weights[i] / w_sum
        alloc = min(alloc, remaining - 25.0)
        if alloc < 15.0:
            break

        J_init, arr_init, name = init_list[i]
        print(f"\n[P1] L-BFGS-B on '{name}' (J0={J_init:.4f}, t={alloc:.0f}s)")
        t0 = time.monotonic()

        res_a = run_annealed_lbfgsb_inner(arr_init, alloc)
        res_J = compute_J(res_a)

        elapsed = time.monotonic() - t0
        print(f"    Result: J={res_J:.6f} (time={elapsed:.1f}s)")

        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST: {val:.6f} ***")

    # ============ PHASE 2: Coordinate descent polishing ============
    remaining = total_deadline - time.monotonic()
    if remaining > 30.0:
        cd_budget = min(remaining * 0.2, 200.0)
        print(f"\n[P2] Coordinate descent (t={cd_budget:.0f}s)")
        t0 = time.monotonic()
        res_a, res_J = run_coordinate_descent(best_global, cd_budget, seed + 55555)
        elapsed = time.monotonic() - t0

        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST: {val:.6f} ***")
        else:
            print(f"    (verified={val:.6f}, time={elapsed:.1f}s)")

    # ============ PHASE 3: Iterated local search ============
    remaining = total_deadline - time.monotonic()
    if remaining > 60.0:
        print(f"\n[P3] Iterated local search (t={remaining:.0f}s)")
        ilt_deadline = time.monotonic() + remaining * 0.85
        perturb_scale = 0.25
        ilt_count = 0

        while time.monotonic() < ilt_deadline:
            ilt_count += 1
            inner_budget = min(80.0, ilt_deadline - time.monotonic() - 5.0)
            if inner_budget < 15.0:
                break

            # Create perturbation
            perturbed = best_global * np.exp(rng.normal(0, perturb_scale, n))
            perturbed = np.clip(perturbed, 0.01, 1000.0)

            # Quick annealed optimization
            res_a = run_annealed_lbfgsb_inner(perturbed, inner_budget)
            res_J = compute_J(res_a)

            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"  [ILS #{ilt_count}] *** NEW BEST: {val:.6f} *** (scale={perturb_scale:.3f})")
                perturb_scale = max(0.08, perturb_scale * 0.85)
            else:
                if ilt_count % 3 == 0:
                    print(f"  [ILS #{ilt_count}] no improvement (verified={val:.6f}, scale={perturb_scale:.3f})")
                perturb_scale = min(1.5, perturb_scale * 1.15)

    # ============ FINAL: Verify and return ============
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_global_score:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]


def run_annealed_lbfgsb_inner(a_init, budget_s):
    """Internal annealed L-BFGS-B with proper time tracking."""
    best_a = a_init.copy()
    start = time.monotonic()

    # Adaptive tau schedule based on available time
    if budget_s > 200:
        tau_schedule = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        time_weights = [0.06, 0.08, 0.10, 0.10, 0.10, 0.10, 0.10, 0.10, 0.10, 0.08, 0.06, 0.02]
    elif budget_s > 100:
        tau_schedule = [20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
        time_weights = [0.08, 0.12, 0.14, 0.14, 0.14, 0.12, 0.10, 0.08, 0.06, 0.02]
    else:
        tau_schedule = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
        time_weights = [0.12, 0.16, 0.18, 0.16, 0.14, 0.10, 0.08, 0.06]

    for i, tau in enumerate(tau_schedule):
        elapsed = time.monotonic() - start
        remaining = budget_s - elapsed
        if remaining < 3.0:
            break

        alloc = budget_s * time_weights[i]
        alloc = min(alloc, remaining)
        if alloc < 2.0:
            break

        res = run_lbfgsb_timed(best_a, tau, alloc)
        res_J = compute_J(res)
        if res_J < compute_J(best_a):
            best_a = res

    return best_a
