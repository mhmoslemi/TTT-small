import numpy as np
import time
from scipy.optimize import linprog

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.clip(a, 0.0, None, out=a)
    s = a.sum()
    if s <= 1e-300:
        a[:] = 1.0 / a.size
        return a
    a /= s
    return a

def score_norm(a):
    n = a.size
    c = np.convolve(a, a)
    m = c.max()
    return 2.0 * n * float(m)

def build_full_conv_matrix_rows(f, n):
    """
    Build the constraint matrix A for the LP:
    Row k has A[k,j] = f[k-j] if 0 <= k-j < n, else 0.
    This gives A @ g = conv(f, g) (element-wise).
    """
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            idx = np.arange(jlo, jhi)
            A[k, jlo:jhi] = f[k - idx]
    return A

def solve_lp(f, M, max_time=5.0):
    """
    Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns g or None.
    """
    n = f.size

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based direction finding, smooth-max subgradient, block-level global search,
and simulated annealing with multiple length trials.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

try:
    from scipy.optimize import differential_evolution
except Exception:
    differential_evolution = None


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.clip(a, 0.0, None, out=a)
    s = a.sum()
    if s <= 1e-300:
        a[:] = 1.0 / max(1, a.size)
        return a
    a /= s
    return a


def score_norm(a):
    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * float(c.max())


def build_A_matrix(f):
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[k, jlo:jhi] = f[k - jhi + 1: k - jlo + 1][::-1]
    return A


def solve_lp_direction(f, M, timeout=3.0):
    if linprog is None:
        return None
    n = f.size
    if n > 2048:
        return None
    try:
        A = build_A_matrix(f)
        A_norm = A / M
        c_obj = -np.ones(n, dtype=np.float64)
        b_vec = np.ones(2 * n - 1, dtype=np.float64)
        res = linprog(c_obj, A_ub=A_norm, b_ub=b_vec, bounds=(0.0, None),
                      method='highs', options={'time_limit': timeout})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1.0 + 1e-5:
                return g
    except Exception:
        pass
    return None


def line_search(f, g, f_conv):
    n = f.size
    best_t = 0.0
    best_score = 2.0 * n * float(f_conv.max())
    ts_coarse = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0]
    best_t_coarse = 0.0
    best_s_coarse = best_score
    for t in ts_coarse:
        h = (1.0 - t) * f + t * g
        h = normalize_nonneg(h)
        sc = score_norm(h)
        if sc < best_s_coarse:
            best_s_coarse = sc
            best_t_coarse = t
    if best_t_coarse > 0:
        t_lo = max(0.0, best_t_coarse - 0.08)
        t_hi = min(1.0, best_t_coarse + 0.08)
        ts_fine = np.linspace(t_lo, t_hi, 9)
        for t in ts_fine:
            h = (1.0 - t) * f + t * g
            h = normalize_nonneg(h)
            sc = score_norm(h)
            if sc < best_s_coarse:
                best_s_coarse = sc
                best_t_coarse = t
    if best_s_coarse < best_score:
        return best_t_coarse, best_s_coarse
    return 0.0, best_score


def smooth_max_step(f, conv, M, temp_frac):
    n = f.size
    temp = max(1e-12, temp_frac * M)
    excess = np.maximum(conv - M, 0.0)
    weights = excess / temp
    weights = np.minimum(weights, 50.0)
    total_w = weights.sum()
    if total_w < 1e-15:
        peak_idx = np.argmax(conv)
        weights = np.zeros(2 * n - 1)
        weights[peak_idx] = 1.0
    else:
        weights /= total_w

    grad = np.zeros(n, dtype=np.float64)
    for k in np.where(weights > 1e-10)[0]:
        k = int(k)
        w = weights[k]
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        for j in range(jlo, jhi):
            grad[j] += 2.0 * w * f[k - j]
    mean_grad = grad.mean()
    grad -= mean_grad
    grad_norm = np.abs(grad).max()
    if grad_norm < 1e-15:
        return None
    grad /= grad_norm
    best_h = None
    best_sc = None
    alphas = [0.03, 0.08, 0.15, 0.3, 0.5, 0.8]
    for alpha in alphas:
        h = f - alpha * grad
        h = normalize_nonneg(h)
        sc = score_norm(h)
        if best_sc is None or sc < best_sc:
            best_sc = sc
            best_h = h
    return best_h, best_sc


def resize_sequence(a, new_n):
    a = np.asarray(a, dtype=np.float64)
    old_n = a.size
    if old_n == new_n:
        return normalize_nonneg(a)
    if old_n < new_n:
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, new_n)
        interp = np.interp(x_new, x_old, a)
        return normalize_nonneg(np.maximum(interp, 0))
    else:
        bins = np.zeros(new_n, dtype=np.float64)
        for i in range(new_n):
            lo = int(i * old_n / new_n)
            hi = int((i + 1) * old_n / new_n)
            bins[i] = a[lo:hi].sum()
        return normalize_nonneg(bins)


def block_objective(params, n, n_blocks):
    heights = np.abs(params[:n_blocks])
    s = heights.sum()
    if s < 1e-10:
        return 100.0
    heights = heights / s
    a = np.zeros(n, dtype=np.float64)
    for i in range(n_blocks):
        lo = int(i * n / n_blocks)
        hi = int((i + 1) * n / n_blocks)
        length = hi - lo
        if length > 0:
            a[lo:hi] = heights[i] / length
    return score_norm(a)


def block_search(n, n_blocks, rng, time_budget=30.0):
    if differential_evolution is None:
        return None
    bounds = [(-1.0, 1.0)] * n_blocks
    deadline = time.monotonic() + time_budget
    try:
        res = differential_evolution(
            block_objective, bounds, args=(n, n_blocks),
            seed=int(rng.integers(0, 2**31)), maxiter=200,
            tol=1e-8, popsize=15, init='sobol',
            options={'maxiter': 200},
            callback=None
        )
        if res.success or res.fun < 2.5:
            heights = np.abs(res.x[:n_blocks])
            s = heights.sum()
            if s > 1e-10:
                heights /= s
                a = np.zeros(n, dtype=np.float64)
                for i in range(n_blocks):
                    lo = int(i * n / n_blocks)
                    hi = int((i + 1) * n / n_blocks)
                    length = hi - lo
                    if length > 0:
                        a[lo:hi] = heights[i] / length
                return a
    except Exception:
        pass
    return None


def local_refine(f, time_budget, rng, best_state):
    n = f.size
    f = normalize_nonneg(f.copy())
    conv = np.convolve(f, f)
    M = float(conv.max())
    cur_score = 2.0 * n * M
    best_local = f.copy()
    best_local_score = cur_score

    if cur_score < best_state['score']:
        best_state['score'] = cur_score
        best_state['seq'] = f.copy()
        print(f"  local_refine init: score={cur_score:.8f}", flush=True)

    deadline = time.monotonic() + time_budget
    no_improve = 0
    temp = 0.15
    lp_count = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(conv.max())
        cur_score = 2.0 * n * M

        if cur_score < best_local_score - 1e-10:
            best_local_score = cur_score
            best_local = f.copy()
            if cur_score < best_state['score']:
                best_state['score'] = cur_score
                best_state['seq'] = f.copy()
                print(f"  new best: {cur_score:.8f} (lp={lp_count})", flush=True)
            no_improve = 0
            temp = max(0.02, temp * 0.92)
        else:
            no_improve += 1

        if no_improve > 50:
            f = normalize_nonneg(f * np.exp(rng.normal(0, 0.3, n)))
            temp = min(0.5, temp * 1.5)
            continue

        improved = False

        # LP direction
        if linprog is not None and n <= 1500 and (no_improve % 2 == 0):
            g = solve_lp_direction(f, M, timeout=2.0)
            lp_count += 1
            if g is not None:
                t, sc = line_search(f, g, conv)
                if sc < cur_score - 1e-10:
                    f = normalize_nonneg((1.0 - t) * f + t * g)
                    improved = True

        # Smooth-max subgradient
        if not improved:
            h, sc = smooth_max_step(f, conv, M, temp)
            if h is not None and sc < cur_score - 1e-10:
                f = h
                improved = True

        # Small random step
        if not improved and time.monotonic() < deadline - 0.5:
            scale = 0.05 + 0.1 * (no_improve / 50.0)
            f = normalize_nonneg(f * np.exp(rng.normal(0, scale, n)))

    return best_local, best_local_score
