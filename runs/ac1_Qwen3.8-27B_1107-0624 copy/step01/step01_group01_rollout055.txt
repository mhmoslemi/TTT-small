Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterations (novel) combined with gradient-based polishing.

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn if mn > -1e-9 else np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w

def lp_solve(f, M, timeout=10.0):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f, g)[k] <= M for all k.
    Returns g or None.
    """
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # Use sparse representation
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # maximize sum = minimize -sum
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'time_limit': timeout})
    
    if result.success and result.x is not None:
        g = np.maximum(result.x, 0)
        return g
    return None

        M_new = float(c_new.max())
        if M_new < best_M:

            best_f = f_new.copy()
    
    return best_f, best_M, best_M < M - 1e-12

    z -= float(np.mean(z))

    v = np.zeros(n)

    best_z = z.copy()

        gz -= float(np.mean(gz))

            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); t_adam = 0
            continue

        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)
        
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:

                if no_imp % 300 == 0:
                    lr = max(0.01, lr * 0.96)

            m.fill(0); v.fill(0); t_adam = 0
