    """Solve the LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0"""
    # Build the constraint matrix A where A[k, :] is the row of the convolution
    # (f*g)_k = sum_i f[i] * g[k-i] (with appropriate bounds)
    # For row k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0

"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Full-constraint LP direction finding (no cutting plane for n <= ~2000)
2. Nesterov-accelerated gradient descent with adaptive smoothing
3. Multi-start with diverse initializations
4. Final polishing and validation
"""

import time
import numpy as np
try:
    from scipy.optimize import linprog
    HAS_SCIPY = True
except:
    HAS_SCIPY = False

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L

def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.size)
    return a / s

def greedy_normalized(n):
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    return normalize_nonneg(a)

def power_normalized(n, s):
    idx = np.arange(1, n+1, dtype=float)
    a = idx ** (-s)
    return normalize_nonneg(a)

def exponential_normalized(n, rate):
    idx = np.arange(n, dtype=float)
    a = np.exp(-rate * idx / n)
    return normalize_nonneg(a)

def score_fft(p):
    """Compute 2n*max(conv)/sum^2 using FFT"""
    a = np.asarray(p, dtype=float).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    L = next_power_of_two(2*n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    return float(2.0 * n * maxc / (s*s))

def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=float).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return [1.0]*16
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]

def lp_direction(f, n, M):
    """Solve max sum(g) s.t. (f*g)_k <= M for all k, g >= 0. Returns g."""
    if not HAS_SCIPY:
        return None
    
    # Build A matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    b = np.full(2*n-1, M)
    c_obj = -np.ones(n)
    
    result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if result is None or not result.success:
        return None
    g = result.x
    g = np.clip(g, 0.0, None)
    if g.sum() <= 1e-12:
        return None
    return g

def lp_iteration(f, budget_s, tol=1e-7):
    """
    Iterative LP method:
    - f is normalized to sum = sqrt(2n)
    - At each step, solve LP to find direction g
    - Take a step toward g
    - Update M = max(f*f)
    """
    n = f.size
    S0 = np.sqrt(2.0 * n)
    f = f * (S0 / float(f.sum()))
    M = float(np.max(np.convolve(f, f)))
    best_score = 2.0 * n * M / (S0 * S0)  # = M since sum = S0
    best_f = f.copy()
    
    end = time.monotonic() + budget_s
    it = 0
    
    while time.monotonic() < end - 2.0:
        it += 1
        g = lp_direction(f, n, M)
        if g is None:
            break
        
        g = g * (S0 / float(g.sum()))
        
        # Line search
        best_t = None
        best_sc = best_score
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625]:
            fn = (1.0 - t) * f + t * g
            fn = np.clip(fn, 0.0, None)
            sf = float(fn.sum())
            if sf <= 1e-12:
                continue
            fn = fn * (S0 / sf)
            c = np.convolve(fn, fn)
            maxc = float(np.max(c))
            sc = 2.0 * n * maxc / (S0 * S0)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_t = t
        
        if best_t is None:
            # Try smaller steps
            for t in [0.0078125, 0.00390625, 0.001953125]:
                fn = (1.0 - t) * f + t * g
                fn = np.clip(fn, 0.0, None)
                sf = float(fn.sum())
                if sf <= 1e-12:
                    continue
                fn = fn * (S0 / sf)
                c = np.convolve(fn, fn)
                maxc = float(np.max(c))
                sc = 2.0 * n * maxc / (S0 * S0)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_t = t
        
        if best_t is None:
            break

def nesterov_gradient(p, budget_s):
    """Nesterov-accelerated gradient descent in log-space"""
    n = p.size
    p = normalize_nonneg(p)
    S0 = np.sqrt(2.0 * n)
    f = p * S0
    
    L = next_power_of_two(2*n - 1)
    x = np.log(f + 1e-300)
    x_out = x.copy()  # momentum position
    v = np.zeros(n)
    
    lr = 0.02
    mu = 0.9
    best_f = f.copy()
    best_sc = score_fft(f)
    
    end = time.monotonic() + budget_s
    t = 0
    
    while time.monotonic() < end - 1.0:
        t += 1
        frac = min(1.0, t / max(1.0, budget_s))
        alpha = 3.0 + 47.0 * np.exp(-5.0 * frac)
        
        # Nesterov lookahead
        x_new = x + mu * v
        f_trial = np.exp(x_new - np.max(x_new))
        f_trial = f_trial / f_trial.sum() * S0
        
        # Compute gradient at trial point
        Pf = np.fft.rfft(f_trial, L)
        c = np.fft.irfft(Pf * Pf, L)[:2*n-1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0:
            break
        
        # Smoothed max weights
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if sz <= 0:
            w = np.zeros(2*n-1)
            w[np.argmax(c)] = 1.0
        else:
            w = expz / sz
        
        # Gradient: d/d(f_i) [2n * sum_k w_k * (f*f)_k] = 4n * (w_rev * f)_i
        W = np.fft.rfft(w[::-1], L)
        C = np.fft.irfft(Pf * W, L)[n-1:2*n-1][::-1].real
        C = np.maximum(C, 0.0)
        grad = 4.0 * n * C
        
        # Project gradient to simplex (in log-space)
        gp = float(np.dot(f_trial, grad))
        grad_x = f_trial * (grad - gp)
        
        # Nesterov update
        v_new = -lr * grad_x
        v = mu * v + (1 - mu) * v_new  # Actually this isn't quite right for Nesterov
        x = x + v_new
        x -= x.mean()
        
        f_trial = np.exp(x - np.max(x))
        f_trial = f_trial / f_trial.sum() * S0
        
        sc = score_fft(f_trial)
        if sc < best_sc:
            best_sc = sc
            best_f = f_trial.copy()
    
    return normalize_nonneg(best_f), best_sc

def build_A_matrix(f, n):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n"""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        for ii in range(k_start - j + 1):
            k = k_start + ii
            i = k - j
            A[k, j] = f[i]
    return A

def lp_solve_full(A, b, n):
    """Solve max sum(g) s.t. A@g <= b, g >= 0. Returns g or None."""
    if not HAS_SCIPY:
        return None

    try:

    except Exception:
        try:

            return None

    g = np.asarray(result.x, dtype=np.float64).ravel()
    g = np.clip(g, 0.0, None)
    if float(g.sum()) <= 1e-15:

    hi = 1.0
    best_t = 0.0
    best_s = s0

    for _ in range(50):
        mid = 0.5 * (lo + hi)
        smid = score_at_t(mid)
        if smid < best_s:
            best_s = smid
            best_t = mid

        # Estimate subgradient: derivative of max component at mid
        h = (1.0 - mid) * f + mid * g
        h = np.maximum(h, 0.0)
        sh = float(h.sum())
        if sh <= 1e-15:
            break
        h = h * (S0 / sh)
        c = np.convolve(h, h)
        max_idx = int(np.argmax(c))

        # (h*h)'(mid) at max_idx: 2*(1-mid)*(f*g)[max_idx] + 2*mid*(g*g)[max_idx] - 2*(1-mid)*mid*(f*f)[max_idx]... 
        # Actually: h(t) = (1-t)f + t*g, (h*h)_k = (1-t)^2 (f*f)_k + 2t(1-t)(f*g)_k + t^2 (g*g)_k
        # d/dt = -2(1-t)(f*f)_k + 2(1-2t)(f*g)_k + 2t(g*g)_k
        ff = np.convolve(f, f)
        fg = np.convolve(f, g)
        gg = np.convolve(g, g)
        deriv = -2.0 * (1.0 - mid) * ff[max_idx] + 2.0 * (1.0 - 2.0 * mid) * fg[max_idx] + 2.0 * mid * gg[max_idx]

        if deriv < 0:
            lo = mid
        else:
            hi = mid

        if hi - lo < 1e-8:
            break

    return best_t, best_s


def lp_iteration(p_init, budget_s, verbose=True):
    """
    Iterative LP method with full constraints.
    p_init: initial sequence (will be normalized to sum = sqrt(2n))
    Returns (best_p, best_score) where best_p sums to 1.
    """
    n = int(p_init.size)
    if n < 2:
        return normalize_nonneg(p_init), np.inf

    S0 = np.sqrt(2.0 * n)
    p = normalize_nonneg(p_init) * S0

    # Precompute A matrix
    A = build_A_matrix_fast(p, n)

    M = float(np.max(np.convolve(p, p)))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    best_p = p.copy()
    best_score = float(2.0 * n * M / (S0 * S0))  # = M since sum=p is S0

    if verbose:
        print(f"    LP start: n={n}, score={best_score:.8f}")

    end = time.monotonic() + budget_s
    it = 0
    no_improve = 0

    while time.monotonic() < end - 2.0:
        it += 1
        g = lp_solve_full(A, b, n)
        if g is None:
            break

        g = g * (S0 / float(g.sum()))

        # Line search
        best_t, best_t_score = line_search_convex(p, g, S0)

        if best_t <= 1e-8:
            no_improve += 1
            if no_improve >= 3:
                break
            # Try a random perturbation and restart
            perturb = np.ones(n) * (1.0 + 0.1 * np.random.randn(n))
            p_new = p * perturb
            p_new = np.clip(p_new, 0.0, None)
            p_new = p_new * (S0 / float(p_new.sum()))
            M_new = float(np.max(np.convolve(p_new, p_new)))
            if M_new < M * 1.01:
                p = p_new
                M = M_new
                b[:] = M
                A = build_A_matrix_fast(p, n)
            else:
                break
            continue

        # Update
        p = (1.0 - best_t) * p + best_t * g
        p = np.maximum(p, 0.0)
        p = p * (S0 / float(p.sum()))
        M = float(np.max(np.convolve(p, p)))
        cur_score = float(M)  # since 2n/S0^2 = 1

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                break

        b[:] = M
        A = build_A_matrix_fast(p, n)

        if verbose and it % 5 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={best_t:.4f}")

        if best_score < 1.5027:
            if verbose:
                print(f"    Target reached at iter {it}!")
            break

    # Final: normalize to sum=1
    best_p_norm = normalize_nonneg(best_p)
    final_score = score_fft(best_p_norm)
    if final_score < best_score:
        best_score = final_score
    return best_p_norm, best_score


def nesterov_gradient(p_init, budget_s, verbose=False):
    """Nesterov-accelerated gradient descent in log-space with adaptive temperature."""
    n = int(p_init.size)
    if n < 2:
        return normalize_nonneg(p_init), np.inf

    p = normalize_nonneg(p_init) * np.sqrt(2.0 * n)
    x = np.log(p + 1e-300)
    x_mom = x.copy()
    v = np.zeros(n, dtype=np.float64)

    if n < 128:
        base_lr = 0.04
    elif n < 512:
        base_lr = 0.025
    else:
        base_lr = 0.015

    mu = 0.85
    L = next_power_of_two(2 * n - 1)

    best_p = p.copy()
    best_score = float(np.max(np.convolve(p / p.sum() * np.ones(1), p / p.sum())))
    best_score = 2.0 * n * best_score

    end = time.monotonic() + budget_s
    t = 0

    while time.monotonic() < end - 1.0:
        t += 1
        elapsed = t
        frac = min(1.0, t / max(1.0, budget_s))
        alpha = 4.0 + 50.0 * np.exp(-5.0 * frac)
        lr = base_lr * (1.0 + 0.5 * np.exp(-3.0 * frac))

        # Nesterov lookahead
        x_look = x_mom + mu * v
        p_trial = np.exp(x_look - np.max(x_look))
        p_trial = p_trial / float(p_trial.sum()) * np.sqrt(2.0 * n)

        # Compute convolution and gradient
        Pt = np.fft.rfft(p_trial, L)
        c = np.fft.irfft(Pt * Pt, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            break

        # Softmax weights for smoothed max
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if sz <= 1e-30:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient via FFT
        W = np.fft.rfft(w[::-1], L)
        C = np.fft.irfft(Pt * W, L)[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        grad_p = 4.0 * n * C

        # Convert to log-space gradient
        gp = float(np.dot(p_trial, grad_p))
        grad_x = p_trial * (grad_p - gp)

        # Nesterov update
        v = -lr * grad_x
        x_mom = x_mom + v
        x = x_mom
        x -= x.mean()

        p_trial = np.exp(x - np.max(x))
        p_trial = p_trial / float(p_trial.sum()) * np.sqrt(2.0 * n)

        sc = float(np.max(np.convolve(p_trial, p_trial)))
        sc = 2.0 * n * sc
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_trial.copy()

        if verbose and t % 100 == 0:
            print(f"    Grad iter {t}: score={best_score:.8f}")

    best_p_norm = normalize_nonneg(best_p)
    final_score = score_fft(best_p_norm)
    if final_score < best_score:
        best_score = final_score
    return best_p_norm, best_score


def additive_gradient(p_init, budget_s, verbose=False):
    """Additive gradient descent: directly optimize in a-space with projection."""
    n = int(p_init.size)
    if n < 2:
        return normalize_nonneg(p_init), np.inf

    p = normalize_nonneg(p_init)
    S0 = np.sqrt(2.0 * n)
    f = p * S0
    L = next_power_of_two(2 * n - 1)

    best_f = f.copy()
    best_score = float(np.max(np.convolve(f, f)))

    end = time.monotonic() + budget_s
    t = 0
    step_size = 1.0 / (n * 0.5)

    while time.monotonic() < end - 1.0:
        t += 1
        frac = min(1.0, t / max(1.0, budget_s))
        alpha = 5.0 + 40.0 * np.exp(-4.0 * frac)
        ss = step_size * (1.0 + 2.0 * np.exp(-3.0 * frac))

        Pf = np.fft.rfft(f, L)
        c = np.fft.irfft(Pf * Pf, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0 or not np.isfinite(maxc):
            break

        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if sz <= 1e-30:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        C = np.fft.irfft(Pf * W, L)[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        grad = 4.0 * n * C

        # Gradient step with projection onto {x >= 0, sum(x) = S0}
        f_new = f - ss * grad
        f_new = np.maximum(f_new, 0.0)
        sn = float(f_new.sum())
        if sn <= 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new * (S0 / sn)

        sc = float(np.max(np.convolve(f_new, f_new)))
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()

        # Also try the negative gradient (in case we're at a boundary)
        f_new2 = f + 0.5 * ss * grad
        f_new2 = np.maximum(f_new2, 0.0)
        sn2 = float(f_new2.sum())
        if sn2 > 1e-15:
            f_new2 = f_new2 * (S0 / sn2)
            sc2 = float(np.max(np.convolve(f_new2, f_new2)))
            if sc2 < best_score:
                best_score = sc2
                best_f = f_new2.copy()
                f = best_f.copy()
                continue

        f = f_new

        if verbose and t % 200 == 0:
            print(f"    AddGrad iter {t}: score={best_score:.8f}")

    best_p_norm = normalize_nonneg(best_f)
    final_score = score_fft(best_p_norm)
    if final_score < best_score:
        best_score = final_score
    return best_p_norm, best_score

def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, sc=None):
        nonlocal best_p, best_score
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        a = np.clip(a, 0.0, None)
        sa = float(a.sum())
        if sa <= 1e-15:
            return
        if sc is None:
            sc = score_fft(a)
        if not np.isfinite(sc):
            return
        if sc < best_score:
            best_p = a.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: n={a.size}, score={sc:.8f} (time={time.monotonic()-t0:.1f}s)")
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or sc < old:
            best_by_n[n] = float(sc)

    # Phase 0: Warm start from global variable
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and float(arr.sum()) > 0.01:
                record(arr)
        except Exception:
            pass

    # Phase 1: Quick diversity sweep
    print(f"Phase 1: Initialization sweep")
    lengths = [64, 96, 128, 160, 192, 224, 256, 320, 384, 448, 512, 640, 768, 1024, 1280, 1536, 2048]
    init_fns = [
        make_start_greedy,
        make_start_power40,
        make_start_power45,
        make_start_power50,
        make_start_power55,
        make_start_power60,
        make_start_exp05,
        make_start_exp10,
        make_start_exp20,
        make_start_exp30,
        make_start_log2,
        make_start_hybrid1,
        make_start_hybrid2,
        make_start_hybrid3,
    ]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        for fn in init_fns:
            try:
                p = fn(n)
                record(p)
            except Exception:
                pass

    print(f"  After sweep: best_score={best_score:.8f}, best_n={best_p.size if best_p is not None else 'N/A'}")

    # Phase 2: LP-based optimization (main workhorse)
    if HAS_SCIPY:
        print(f"Phase 2: LP optimization")
        # Pick top lengths to focus on
        sorted_n = sorted(best_by_n.keys(), key=lambda x: best_by_n[x])
        lp_lengths = sorted_n[:8]
        # Ensure 384 is included
        if 384 not in lp_lengths:
            lp_lengths.append(384)
        # Ensure we have some diversity
        for extra in [256, 512, 384]:
            if extra not in lp_lengths and len(lp_lengths) < 10:
                lp_lengths.append(extra)

        for n in lp_lengths:
            rem = deadline - time.monotonic()
            if rem < 40:
                break

            # Multiple starts per length
            starts = [
                greedy_normalized(n),
                power_normalized(n, 0.45),
                exponential_normalized(n, 1.0),
            ]
            # Add warm start if matching length
            if h is not None:
                try:
                    arr = np.asarray(h, dtype=np.float64).ravel()
                    if arr.size == n and float(arr.sum()) > 0.01:
                        starts.append(normalize_nonneg(arr))
                except Exception:
                    pass

            # Also add perturbed best if available
            if n in best_by_n and best_p is not None and best_p.size == n:
                starts.append(normalize_nonneg(best_p * np.exp(rng.normal(0, 0.1, n))))

            for si, start in enumerate(starts[:4]):
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                if si == 0:
                    alloc = min(rem * 0.35, 400.0)
                else:
                    alloc = min(rem * 0.2, 180.0)
                if alloc < 15:
                    break

                print(f"    LP n={n}, start={si}, alloc={alloc:.0f}s")
                p_out, sc_out = lp_iteration(start, alloc, verbose=(si == 0))
                record(p_out, sc_out)

    # Phase 3: Gradient polishing
    print(f"Phase 3: Gradient polishing")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(rem * 0.3, 200.0)
            print(f"    Nesterov grad, alloc={alloc:.0f}s")
            p_out, sc_out = nesterov_gradient(best_p, alloc)
            record(p_out, sc_out)

    # Phase 4: More LP from best
    if HAS_SCIPY and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            alloc = min(rem * 0.4, 350.0)
            print(f"    LP from best (n={best_p.size}), alloc={alloc:.0f}s")
            p_out, sc_out = lp_iteration(best_p, alloc, verbose=True)
            record(p_out, sc_out)

    # Phase 5: Additive gradient
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(rem * 0.3, 150.0)
            print(f"    Additive grad, alloc={alloc:.0f}s")
            p_out, sc_out = additive_gradient(best_p, alloc)
            record(p_out, sc_out)

    # Phase 6: Final LP polish
    if HAS_SCIPY and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(rem - 10, 200.0)
            print(f"    Final LP polish, alloc={alloc:.0f}s")
            p_out, sc_out = lp_iteration(best_p, alloc, verbose=True)
            record(p_out, sc_out)

    # Phase 7: Final gradient
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            alloc = min(rem - 5, 100.0)
            p_out, sc_out = nesterov_gradient(best_p, alloc)
            record(p_out, sc_out)

    # Phase 8: Validate with global evaluator
    print(f"Phase 8: Validation")
    if best_p is not None:
        seq = scale_output(best_p, 10.0)
        try:
            gv = evaluate_sequence(seq)
            gv = float(gv)
            print(f"  Global eval result: {gv:.8f}")
            if np.isfinite(gv) and gv < best_score:
                best_score = gv
        except Exception as e:
            print(f"  Eval error: {e}")

    # Also validate top few by length
    items = sorted(best_by_n.items(), key=lambda x: x[1])[:5]
    for n, _ in items:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        # We don't store the full arrays by n separately, so skip unless best_p has this n
        pass

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = score_fft(best_p)

    print(f"Final: n={best_p.size}, score={best_score:.8f}, total_time={time.monotonic()-t0:.1f}s")
    return scale_output(best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    try:

    except Exception as e:
        import traceback
        traceback.print_exc()
