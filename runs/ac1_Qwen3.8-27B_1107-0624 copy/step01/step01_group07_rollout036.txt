                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        stall = 0
                    elif rng.random() < 0.25:
                        best_a = a_new  # Accept worse (simulated annealing)
                        stall = 0

            if stall > 80:
                break

        if verbose and it % 50 == 0:
            print(f"      LP iter {it}, score={best_score:.8f}, step={step:.4f}, stall={stall}")

    return best_a, best_score


def peak_equalization(a, deadline, rng, verbose=False):
    """
    Iteratively equalize peak values of the auto-convolution.
    At each step, identify the highest peak and reduce contributions to it.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(a, a, mode='full')
        M = conv.max()
        peak_idx = np.where(conv >= 0.95 * M)[0]

        if len(peak_idx) == 0:
            break

        # For each peak position k, the variables involved are a[j] where
        # conv[k] depends on a[j]*a[k-j]. We want to reduce a[j] values
        # that appear in many peak constraints.
        exposure = np.zeros(n)
        for k in peak_idx:
            for j in range(max(0, k - n + 1), min(k + 1, n)):
                other = k - j
                if 0 <= other < n:
                    exposure[j] += a[other]
                    if other != j:
                        exposure[other] += a[j]

        # Reduce highly exposed variables
        lr = 0.08
        exp_norm = exposure / (exposure.mean() + 1e-15)
        a_new = a * np.exp(-lr * exp_norm)
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break
        a_new /= S_new

        score_new = compute_objective(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a * np.exp(-lr * 0.3 * exp_norm)
            a_new2 = np.clip(a_new2, 0, None)
            S2 = a_new2.sum()
            if S2 > 1e-10:
                a_new2 /= S2
                sc2 = compute_objective(a_new2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_new2.copy()
                    a = a_new2
                else:
                    lr = max(0.005, lr * 0.7)
                    if lr < 0.01:
                        # Perturb
                        noise = rng.random(n) * 0.1
                        a = a * (1.0 + noise)
                        a = np.clip(a, 0, None) / max(a.sum(), 1e-10)
                        lr = 0.05

    return best_a, best_score


def run_optimization(a_init, deadline, rng, methods=('cd', 'pe', 'lp')):
    """Run a sequence of optimization methods, passing results forward."""
    a = np.array(a_init, dtype=np.float64)
    a = np.clip(a, 0, None)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_score = compute_objective(a)
    best_a = a.copy()
    n_methods = len(methods)

    for m_i, method in enumerate(methods):
        m_deadline = deadline - (n_methods - 1 - m_i) * 10
        m_deadline = min(m_deadline, deadline)
        if time.monotonic() > m_deadline - 5:
            break

        if method == 'cd':
            time_frac = 0.7 if n_methods > 1 else 1.0
            m_dl = time.monotonic() + (m_deadline - time.monotonic()) * time_frac
            a, score = coordinate_descent_fast(a, m_dl, rng)
        elif method == 'pe':
            a, score = peak_equalization(a, m_deadline, rng)
        elif method == 'lp':
            a, score = lp_optimize(a, m_deadline, rng)
        else:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    - Start from parent sequence and diverse initializations
    - Use hybrid CD + peak-equalization + LP optimization
    - Multiple sequence lengths
    - Final polish on best result
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
            parent_score = compute_objective(parent)
            print(f"Parent: n={p.size}, score={parent_score:.8f}")
    except Exception:
        pass

    # Build configurations: (n, init_name, methods)
    configs = []

    # Parent-based (highest priority)
    if parent is not None:
        np_s = parent.size
        configs.append((np_s, 'parent', ('cd', 'pe', 'lp')))
        configs.append((np_s, 'parent_perturb', ('cd', 'lp')))

    # Key sizes with diverse inits
    for n in [384, 512, 640, 768]:
        for init_name in ['cosine', 'gaussian', 'asym_bump', 'waterfall', 'two_bump']:
            configs.append((n, init_name, ('cd', 'pe', 'lp')))

    # A couple of longer sequences
    for n in [1024, 1280]:
        for init_name in ['cosine', 'gaussian']:
            configs.append((n, init_name, ('cd', 'lp')))

    # Assign time weights
    weights = []
    for n, init_name, methods in configs:
        if 'parent' in init_name:
            weights.append(3.0)
        elif n in [384, 512, 640] and init_name in ['asym_bump', 'waterfall']:
            weights.append(2.0)
        elif n in [512, 640, 768]:
            weights.append(1.5)
        elif n in [384]:
            weights.append(1.2)
        else:
            weights.append(0.8)

    total_w = sum(weights)
    avail = budget_s - 60.0  # reserve for final polish

    # Time allocation
    cum = 0
    time_marks = []
    for w in weights:
        cum += avail * w / total_w
        time_marks.append(start_time + cum)

    print(f"{len(configs)} configs, total time {avail:.0f}s, deadline at t+{budget_s-20:.0f}s")

    for idx in range(len(configs)):
        if time.monotonic() > deadline:
            break
        if time.monotonic() > time_marks[idx] - 20:
            continue

        n, init_name, methods = configs[idx]
        t_mark = time_marks[idx]
        t_deadline = min(t_mark, deadline)

        # Generate initial
        try:
            if init_name == 'parent' and parent is not None:
                a = parent[:n].copy()
            elif init_name == 'parent_perturb' and parent is not None:
                a = parent[:n].copy()
                a = a * (1.0 + rng.random(n) * 0.2 - 0.1)
            elif init_name == 'cosine':
                a = make_cosine(n)
            elif init_name == 'gaussian':
                a = make_gaussian(n)
            elif init_name == 'asym_bump':
                a = make_asymmetric_bump(n)
            elif init_name == 'waterfall':
                a = make_waterfall(n)
            elif init_name == 'two_bump':
                a = make_two_bump(n)
            else:
                a = np.ones(n)
        except Exception:
            a = np.ones(n)

        a = np.clip(a, 1e-10, None)
        init_score = compute_objective(a)

        # Quick skip if clearly bad and we have a good best
        if best_score < 1.52 and init_score > 2.0 and 'parent' not in init_name:
            continue

        a_out, score = run_optimization(a, t_deadline, rng, methods)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_out.copy()
            print(f"  [{idx+1}/{len(configs)}] n={n} init={init_name}: NEW BEST {best_score:.8f}")
        else:
            print(f"  [{idx+1}/{len(configs)}] n={n} init={init_name}: score={score:.6f} (best={best_score:.6f})")

    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 90:
        print(f"\n=== Final polish (score={best_score:.8f}, n={len(best_seq)}) ===")
        a_pol, sc_pol = coordinate_descent_fast(best_seq, deadline - 30, rng, verbose=True)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol
            print(f"  After CD: {best_score:.8f}")

        if time.monotonic() < deadline - 40:
            a_pol, sc_pol = lp_optimize(best_seq, deadline, rng, verbose=True)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol
                print(f"  After LP: {best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    best_seq = np.clip(best_seq, 0, None)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / max(best_seq.sum(), 1e-30))

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: {elapsed:.1f}s, score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
