    cmax = c.max()

    if score < _best_score:
        _best_score = score
        _best_z = x.copy()

    e = np.exp(u)
    s = e.sum()
    if s < 1e-300:
        J = cmax
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s
    
    gp = _grad_p(p, w)
    sdot = gp @ p
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.clip(np.where(np.isfinite(gz), gz, 0.0), -1e6, 1e6)
    return J, gz

Search for non-negative step function heights minimizing

where a*a is the full linear self-convolution.

Method: Softmax parameterization + multi-stage L-BFGS-B (soft-max surrogate)
        + active-set subgradient polishing + population-based restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize

N = 4096
FFT_SELF = 8192
FFT_GRAD = 16384

    pass

def _sz(x):
    """Sanitize and clip z-vector."""
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    return np.clip(x, -ZMAX, ZMAX)


def _zp(z):
    """z-vector → probability vector (softmax)."""
    z = _sz(z)
    m = float(z.max())
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    D = e0 + float(ez.sum())
    p = np.empty(N, dtype=np.float64)
    p[0] = e0 / D
    p[1:] = ez / D
    return p


def _ac(p):
    """Self-convolution (autocorrelation) of p, length 2N-1."""
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real[:2*N-1]
    np.maximum(c, 0.0, out=c)
    return c

    """Objective value for probability vector."""
    return 2.0 * N * float(np.max(_ac(p)))

def _d2z(f):
    """Density (length N, positive) → z-vector (length N-1)."""
    f = np.asarray(f, dtype=np.float64)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 1e-12)
    lf = np.log(f)
    return np.clip(lf[1:] - lf[0], -ZMAX, ZMAX)


def _i_log(beta):
    return _d2z(2.0 / (1.0 + np.exp(-beta * (_X - 0.5))))


def _i_pwr(a, rev=False):
    x = (1.0 - _X) if rev else _X
    return _d2z(np.maximum(x + 1e-6, 1e-8) ** a)


def _i_exp(beta):
    return _d2z(np.exp(beta * _X))


def _i_gauss(mu, sig):
    f = np.exp(-0.5 * ((_X - mu) / sig) ** 2)
    return _d2z(np.maximum(f, 1e-12))


def _i_rand(rng):
    z = np.zeros(N-1)
    X1 = _X[1:]
    for fq in range(1, 9):
        z += rng.normal(0, 2.0 / fq) * np.sin(2*np.pi*fq*X1 + rng.uniform(0, 2*np.pi))
    return np.clip(z, -ZMAX, ZMAX)


def _i_poly(rng):
    t = _X - 0.5
    deg = int(rng.integers(3, 9))
    cf = rng.normal(0, 1.5, deg+1)
    f = np.ones(N)
    tn = np.ones(N)
    for k in range(1, deg+1):
        tn = tn * t
        f += cf[k] * tn
    return _d2z(np.maximum(f, 1e-8))


def _i_bimodal(rng):
    """Bimodal distribution: two Gaussian bumps."""
    mu1 = rng.uniform(0.1, 0.4)
    mu2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.5, 1.0)
    f = w1 * np.exp(-0.5*((_X-mu1)/0.08)**2) + np.exp(-0.5*((_X-mu2)/0.08)**2)
    return _d2z(np.maximum(f, 1e-12))


def _i_uniform():
    return np.zeros(N-1)
