2. Repeat:
   a. Compute c = a*g (FFT)
   b. For positions k where c[k] > M, compute the "excess" e[k] = max(0, c[k] - M)
   c. Compute responsibility: r[i] = sum_k e[k] * 2*a[k-i] (using FFT cross-correlation)
   d. Update: g[i] = g[i] * exp(-eta * r[i] / sum(g)) (or some normalization)
   e. Clip g to [0, inf]
3. Return g

This is essentially a cutting-plane / multiplicative weights approach to the LP. It should converge to the LP solution (or close to it) with the right step size.

    """Solve the soft LP: find g approximating max sum(g) s.t. (a*g) <= M, g>=0."""
    g = a.copy()
    size = 1
    while size < 3*n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    
    for it in range(n_inner):
        # Compute c = a * g (convolution)
        fg = np.fft.rfft(g, size)
        c = np.fft.irfft(fa * fg, size)[:2*n-1]
        
        # Excess
        e = np.maximum(0, c - M)
        
        # Cross-correlation: r[i] = sum_j a[j] * e[j+i]

        # Gradient

        # Update
        g = np.maximum(0, g - lr * grad)

        if S_g > 0:

Iterated Soft LP with multiplicative weights + gradient refinement
for minimizing J(a) = 2n * max(a*a) / (sum(a))^2.
"""

import numpy as np
import time

def fft_size(n):
    """Smallest power of 2 >= n."""
    s = 1
    while s < n:
        s <<= 1
    return s

def convolve_fft(a, b=None):
    """Linear convolution using FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n+m-1]

def cross_correlate_fft(a, b):
    """Compute r[i] = sum_j a[j] * b[j+i] for i = 0, ..., len(a)-1.
    Using FFT: r = ifft(conj(fft(a)) * fft(b))[:len(a)]
    Need size >= len(a) + len(b) - 1."""
    na = len(a)
    nb = len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    r = np.fft.irfft(np.conj(fa) * fb, size)
    return r[:na]

def compute_J(a):
    """Compute J = 2n * max(a*a) / (sum a)^2."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    c = convolve_fft(a)
    return 2.0 * n * np.max(c) / (S * S)

def soft_lp_inner(a, M, n_inner, eta, nfft):
    """Solve soft LP: find g approximating max sum(g) s.t. (a*g) <= M, g >= 0.
    Uses multiplicative weights (Hedge) algorithm.
    
    Args:
        a: current sequence (length n)
        M: target max convolution value
        n_inner: number of inner iterations
        eta: step size (will be normalized)
        nfft: FFT size (precomputed)
    
    Returns:
        g: approximate LP solution
    """
    n = len(a)
    g = a.copy()
    fa = np.fft.rfft(a, nfft)
    
    for it in range(n_inner):
        # Compute c = a * g
        fg = np.fft.rfft(g, nfft)
        c = np.fft.irfft(fa * fg, nfft)[:2*n-1]
        
        # Excess (soft threshold)
        e = np.maximum(0.0, c - M)
        
        # If no excess, we're done
        if e.max() < 1e-10:
            break
        
        # Responsibility: r[i] = sum_k e[k] * a[k-i]
        fe = np.fft.rfft(e, nfft)
        r = np.fft.irfft(np.conj(fa) * fe, nfft)[:n]
        
        # Normalize eta by the scale of r
        r_max = r.max()
        if r_max < 1e-15:
            break
        step = eta / r_max
        
        # Multiplicative update
        g = g * np.exp(-step * r)
        g = np.maximum(g, 0.0)
        
        # Rescale to maintain sum(g) ≈ sum(a)
        S_g = np.sum(g)
        S_a = np.sum(a)
        if S_g > 1e-10:
            g = g * (S_a / S_g)
    
    return g

def _convolve_fft(a, b, fft_size=None):
    """Linear convolution of a and b using FFT."""
    n = len(a)
    m = len(b)
    needed = n + m - 1

    fa = np.fft.rfft(a, fft_size)

    return result[:needed]
