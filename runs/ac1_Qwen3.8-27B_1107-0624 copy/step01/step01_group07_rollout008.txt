**Alternating Minization with Peak-Targeting:**

1. Compute `c = conv(a, a)`
2. Find the top-k peaks in `c`
3. For each peak at position `p`, the value is `sum_i a[i] * a[p-i]`
4. We want to reduce these peaks. The most effective way is to reduce the `a[i]` values that contribute most to the peaks.
5. But we also want to maintain `sum(a)` (or equivalently, work with normalized `a`).
6. So: reduce `a[i]` proportional to its contribution to the peaks, and redistribute the mass to positions that don't contribute to peaks.

This is essentially a "Lovasz theta"-like approach or a "water-filling" approach.

```python
def peak_targeting_update(a, top_k=5, strength=0.1):
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Find top peaks
    peak_indices = np.argsort(c)[-top_k:]
    
    # Compute contribution of each a[i] to the peaks
    contrib = np.zeros(n)
    for p in peak_indices:
        for i in range(n):
            j = p - i
            if 0 <= j < n:
                contrib[i] += a[j]
    
    # Reduce high-contribution, increase low-contribution
    a_new = a.copy()
    avg = contrib.mean()
    a_new *= (1 - strength * (contrib - avg) / (avg + 1e-15))
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() > 0:
        a_new /= a_new.sum()
    return a_new
```

**Weinberger algorithm for flat autocorrelation:**
For each element a[i]:
- Compute the "residual" if we change a[i]
- The auto-convolution at lag m is `c[m] = sum_j a[j] * a[m-j]`

- We want to choose δ to minimize the max of c

When a[i] changes from `old` to `v` (change δ = v - old):
- For m where 0 <= m-i < n and m != 2i: c[m] changes by 2*a[m-i]*δ
- For m = 2i: c[m] changes by 2*a[i]*δ + δ^2 = 2*old*δ + δ^2
- For all other m: no change

So:
- c_new[m] = c_old[m] + 2*a[m-i]*δ (for valid m-i, m ≠ 2i)
- c_new[2i] = c_old[2i] + 2*old*δ + δ^2

The new max is:
max( max over m≠2i of (c_old[m] + 2*a[m-i]*δ), c_old[2i] + 2*old*δ + δ^2, max over invalid m of c_old[m] )

This can be computed in O(n) per candidate δ, and we need to try maybe 10-20 values of δ per coordinate. Total: O(n * n * 20) = O(20n^2) per pass. For n=512, that's about 5M operations per pass, which is very fast.

"""
Hybrid coordinate descent + LP optimization for minimizing
2*n*max(auto-convolution)/(sum)^2 of non-negative sequences.
Uses fast analytic coordinate updates, periodic LP refinement,
multiple initializations, and adaptive time management.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n))) + 1e-10


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 1e-10


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)) + 1e-10


def make_paley(q):
    """Paley difference set based sequence (q must be prime power ≡ 3 mod 4)."""
    residues = set()
    for x in range(1, q):
        residues.add((x * x) % q)
    seq = np.zeros(q)
    for r in residues:
        seq[r] = 1.0
    # Smooth it a bit
    kernel = np.ones(3) / 3.0
    seq = np.convolve(seq, kernel, mode='same')
    return seq + 1e-10


def make_quadratic_residue_convolution(n, prime=None):
    """Create a sequence inspired by quadratic residue structures."""
    if prime is None:
        # Find a prime near n
        for p in range(max(n, 7), max(n, 7) + 200):
            if p % 4 == 3 and all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
                prime = p
                break
    if prime is None:
        prime = 7
    residues = set()
    for x in range(1, prime):
        residues.add((x * x) % prime)
    # Create periodic extension
    seq = np.zeros(n)
    for i in range(n):
        if i % prime in residues:
            seq[i] = 1.0
    # Smooth
    k = max(2, n // 50)
    kernel = np.ones(k) / k
    seq = np.convolve(seq, kernel, mode='same')
    return seq + 1e-10

            g = lp_direction(a, M, time_limit=8.0)
            if g is not None and g.sum() > 1e-10:
                g = g / g.sum()
                improved_lp = False
                for t in [step, step * 0.5, step * 0.25, 0.05]:
                    a_try = (1.0 - t) * a + t * g
                    a_try = np.clip(a_try, 0.0, None)
                    if a_try.sum() < 1e-10:
                        continue
                    a_try = a_try / a_try.sum()
                    sc = compute_score(a_try)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try
                        improved_lp = True
                        stall = 0
                        step = min(0.35, step * 1.05)
                        break

                if not improved_lp:
                    stall += 1
                    step = max(0.02, step * 0.8)

        # Phase 3: Perturbation when stuck
        if stall > 8:
            noise_scale = min(0.3, 0.05 * stall)
            noise = np.abs(rng.normal(0, noise_scale, n))
            a_try = a * (1.0 + noise)
            a_try = np.clip(a_try, 0.0, None)
            if a_try.sum() > 1e-10:
                a_try = a_try / a_try.sum()
                # Quick CD to see if perturbation helps
                a_try2, sc_try = coordinate_descent_fast(a_try, n_passes=10,
                                                          deadline=min(deadline, time.monotonic() + 5), rng=rng)
                if sc_try < best_score - 1e-12:
                    best_score = sc_try
                    best_a = a_try2.copy()
                    a = a_try2
                    stall = 0
                elif rng.random() < 0.1:
                    a = a_try  # Accept with small probability (annealing)
            stall = max(0, stall - 3)

        if stall > 60:
            break

        if verbose and stall % 20 == 0 and stall > 0:
            print(f"    LP-iter stall={stall}, score={best_score:.6f}")

    return best_a, best_score


def pure_cd_optimize(a, deadline, rng, verbose=False):
    """Pure coordinate descent optimization (for larger n where LP is too slow)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64).copy()
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    stall = 0

    while time.monotonic() < deadline:
        a_cd, sc_cd = coordinate_descent_fast(a, n_passes=30, deadline=deadline, rng=rng)
        if sc_cd < best_score - 1e-12:
            best_score = sc_cd
            best_a = a_cd.copy()
            a = a_cd
            stall = 0
        else:
            stall += 1
            if stall > 3:
                # Perturb
                noise_scale = min(0.2, 0.03 * stall)
                noise = np.abs(rng.normal(0, noise_scale, n))
                a_try = a * (1.0 + noise)
                a_try = np.clip(a_try, 0.0, None)
                if a_try.sum() > 1e-10:
                    a_try = a_try / a_try.sum()
                    a = a_try
                    stall = 1
            if stall > 30:
                break

    return best_a, best_score


def multi_start_search(budget_s, seed=42):
    """Main search function with multiple initializations and lengths."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p.copy()
            parent = parent / parent.sum()
    except Exception:
        pass

    # Define search configurations
    configs = []

    # Parent-based (highest priority)
    if parent is not None:
        pn = len(parent)
        configs.append((pn, lambda n, p=parent: p.copy(), 'lp'))
        if pn > 512:
            configs.append((512, lambda n, p=parent: p[:n].copy(), 'lp'))
        if pn > 256:
            configs.append((256, lambda n, p=parent: p[:n].copy(), 'lp'))

    # Standard initializations for various n
    n_values = [256, 384, 512, 640, 768, 1024, 1280, 1536, 2048]
    inits = [make_cosine, make_gaussian, make_parabolic]

    for n in n_values:
        for init_f in inits:
            configs.append((n, init_f, 'cd' if n > 1500 else 'lp'))

    # Paley-based
    for q in [19, 43, 67, 83, 103, 127, 163, 179, 211, 283, 367, 419, 443, 467, 523, 547]:
        if q % 4 == 3:
            configs.append((q, make_paley, 'cd'))

    # Quadratic residue convolution
    configs.append((512, lambda n: make_quadratic_residue_convolution(n, 503), 'lp'))
    configs.append((768, lambda n: make_quadratic_residue_convolution(n, 751), 'lp'))
    configs.append((1024, lambda n: make_quadratic_residue_convolution(n, 1021), 'lp'))

    # Random
    configs.append((512, lambda n, r=rng: np.abs(r.normal(0, 1, n)), 'lp'))
    configs.append((768, lambda n, r=rng: np.abs(r.normal(0, 1, n)), 'lp'))
    configs.append((1024, lambda n, r=rng: np.abs(r.normal(0, 1, n)), 'lp'))

    # Deduplicate by (n, init type) - keep first occurrence
    seen = set()
    unique_configs = []
    for cfg in configs:
        key = (cfg[0], cfg[2])
        if key not in seen:
            seen.add(key)
            unique_configs.append(cfg)
    configs = unique_configs

    # Allocate time: weight by promise
    total_configs = len(configs)
    weights = []
    for n, init_f, method in configs:
        w = 1.0
        if 384 <= n <= 1024:
            w = 1.5
        if 512 <= n <= 768:
            w *= 1.3
        if method == 'lp':
            w *= 1.2
        if parent is not None and n == len(parent):
            w *= 2.0
        weights.append(w)

    total_weight = sum(weights)
    available_time = budget_s - 60.0  # buffer for final polish
    time_allocs = []
    cum = 0.0
    for w in weights:
        cum += available_time * w / total_weight
        time_allocs.append(cum)

    print(f"Starting search: {total_configs} configs, budget={budget_s}s")
    print(f"Time per config (avg): {available_time/total_configs:.1f}s")

    results = []
    for idx, (n, init_func, method) in enumerate(configs):
        elapsed = time.monotonic() - start_time
        config_deadline = min(start_time + time_allocs[idx], deadline)

        if time.monotonic() > config_deadline:
            continue

        config_time = config_deadline - time.monotonic()
        if config_time < 10:
            continue

        try:
            a = init_func(n)
        except Exception:
            a = make_cosine(n)

        a = np.clip(a, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = make_cosine(n)
        a = a / a.sum()

        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        print(f"[{idx+1}/{total_configs}] n={n}, method={method}, "
              f"time_alloc={config_time:.0f}s, remaining={remaining:.0f}s")

        if method == 'lp' and n <= 2048:
            a_opt, score = lp_based_optimize(a, config_deadline, rng, verbose=False)
        else:
            a_opt, score = pure_cd_optimize(a, config_deadline, rng)

        results.append((score, n, a_opt.copy()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")

    # Sort results by score
    results.sort(key=lambda x: x[0])
    print(f"\nTop 5 results:")
    for i, (sc, nn, _) in enumerate(results[:5]):
        print(f"  {i+1}. score={sc:.6f}, n={nn}")

    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal polish on best (n={len(best_seq)}, score={best_score:.6f})...")
        polish_deadline = deadline
        best_seq, best_score = lp_based_optimize(best_seq, polish_deadline, rng, verbose=True)
        print(f"  After polish: score={best_score:.6f}")

    # Also try top-2 and top-3 with remaining time
    if len(results) > 1:
        for ri in [1, 2]:
            if ri >= len(results):
                break
            if time.monotonic() > deadline - 30:
                break
            sc_r, n_r, a_r = results[ri]
            if sc_r > best_score * 1.001:
                continue
            print(f"\nPolishing result #{ri+1} (n={n_r}, score={sc_r:.6f})...")
            a_pol, sc_pol = lp_based_optimize(a_r, min(deadline, time.monotonic() + 120), rng, verbose=False)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Improved! score={best_score:.6f}")

    if best_seq is None:
        best_seq = make_cosine(512)
        best_score = compute_score(best_seq)

    # Ensure validity for evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid coordinate descent
    and LP-based optimization with multiple initializations and sequence lengths.

    Algorithm:
    1. Multiple diverse initializations (cosine, Gaussian, parabolic, Paley, random)
    2. Fast analytic coordinate descent as the primary local optimizer
    3. Periodic LP steps to escape plateaus (finds optimal direction given current peak)
    4. Structured perturbations for global exploration
    5. Adaptive time allocation across (n, initialization) combinations
    6. Final polish on the best candidate found
    """
    return multi_start_search(budget_s, seed=seed)
