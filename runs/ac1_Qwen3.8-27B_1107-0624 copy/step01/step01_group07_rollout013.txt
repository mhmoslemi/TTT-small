"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) soft-max continuation with L-BFGS for fast smooth optimization,
(2) FFT-based exponentiated gradient (exactly matching subgradient) for polishing,
(3) LP direction finding for small-n final polish.
Multiple diverse initializations and adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import minimize, linprog


def fft_convolve_full(a, b):
    """Full linear convolution via FFT."""
    n_out = len(a) + len(b) - 1
    size = next_fast_len(n_out)
    return np.real(ifft(fft(a, size) * fft(b, size)))[:n_out]


def fft_auto_conv(a):
    """Auto-convolution a*a via FFT."""
    n_out = 2 * len(a) - 1
    size = next_fast_len(n_out)
    fa = fft(a, size)
    return np.real(ifft(fa * fa))[:n_out]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_auto_conv(a)
    return 2.0 * n * c.max() / (S * S)


def soft_max_and_weights(c, temperature):
    """Compute soft max and softmax weights of c at given temperature."""
    c_max = c.max()
    c_shifted = (c - c_max) / temperature
    c_shifted = np.minimum(c_shifted, 500.0)  # prevent overflow
    exp_c = np.exp(c_shifted)
    total = exp_c.sum()
    weights = exp_c / total
    soft_max_val = c_max + temperature * np.log(total)
    return soft_max_val, weights, c_max


def soft_grad(a, temperature):
    """
    Compute objective and gradient of 2n * soft_max(conv(a,a), tau) / (sum(a))^2.
    Returns (obj_value, gradient).
    """
    n = len(a)
    c = fft_auto_conv(a)
    soft_max_val, weights, c_max = soft_max_and_weights(c, temperature)
    S = a.sum()

    # dM/da_i = 2 * sum_k weights[k] * a[k-i]
    # = 2 * xcorr(weights, a)[i]
    size = next_fast_len(2 * (2 * n - 1))
    w_freq = fft(weights, size)
    a_freq = fft(a, size)
    xcorr = np.real(ifft(np.conj(w_freq) * a_freq))[:n]
    dM = 2.0 * xcorr

    obj = 2.0 * n * soft_max_val / (S * S)
    # grad_i = (2n/S^3) * (S * dM_i - 2*M)
    grad = (2.0 * n / (S ** 3)) * (S * dM - 2.0 * soft_max_val)

    return obj, grad


def run_lbfgs_soft(a, tau_list, max_iter_per_tau=200):
    """
    Continuation method: minimize soft objective with decreasing temperature.
    Returns optimized a.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    if a.sum() < 1e-10:
        return a
    a = a / a.sum()

    for tau in tau_list:
        def fun(x):
            obj, grad = soft_grad(x, tau)
            return obj, grad

        res = minimize(fun, a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': max_iter_per_tau, 'ftol': 1e-12,
                                'gtol': 1e-8, 'maxfun': max_iter_per_tau * 3})
        a_new = np.maximum(res.x, 0.0)
        if a_new.sum() > 1e-10:
            a = a_new / a_new.sum()

    return a


def exact_subgradient(a, peak_frac=0.0):
    """
    Compute exact subgradient of max(conv(a,a)) using FFT.
    peak_frac: use all peaks within (1-peak_frac) of max.
    Returns (subgradient_vector, c_max).
    """
    n = len(a)
    c = fft_auto_conv(a)
    c_max = c.max()

    if peak_frac > 0:
        threshold = c_max * (1.0 - peak_frac)
        K = np.where(c >= threshold)[0]
    else:
        K = np.where(c == c_max)[0]

    h = np.zeros(2 * n - 1)
    h[K] = 1.0

    # sg[i] = sum_k h[k] * a[k-i] = xcorr(h, a)[i]
    size = next_fast_len(2 * (2 * n - 1))
    h_freq = fft(h, size)
    a_freq = fft(a, size)
    sg = np.real(ifft(np.conj(h_freq) * a_freq))[:n]

    return sg, c_max


def exponentiated_gradient_step(a, sg, lr):
    """One step of exponentiated gradient: a_i *= exp(-lr * sg_i / max(sg))."""
    sg_max = sg.max()
    if sg_max < 1e-15:
        return a.copy()
    update = np.exp(-lr * sg / sg_max)
    a_new = a * update
    a_new = np.maximum(a_new, 1e-30)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def run_subgradient_polish(a, deadline, rng, peak_fracs=None):

    FFT-based exponentiated gradient method for polishing.
    Uses adaptive learning rate, momentum, and periodic restarts.
    """

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * sg
        sg_eff = momentum

        a_new = exponentiated_gradient_step(a, sg_eff, lr)
        score = compute_score(a_new)

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_direction(a, M):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 2048:
        return None
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_plateau(n, frac=0.6):
    """Plateau with smooth edges."""
    x = np.linspace(0, 1, n)
    edge = (1.0 - frac) / 2.0
    a = np.ones(n)
    mask_left = x < edge
    mask_right = x > 1.0 - edge
    a[mask_left] = 0.5 * (1 + np.cos(np.pi * x[mask_left] / edge))
    a[mask_right] = 0.5 * (1 + np.cos(np.pi * (x[mask_right] - (1 - edge)) / edge))
    return a


def make_flat(n):
    return np.ones(n)
