import numpy as np
from scipy.optimize import linprog
import time

def make_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that A @ g = f * g (linear conv)."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    return A

def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)  # minimize -sum(g)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_step(f, t=0.3):

    M = c.max()
    g = solve_lp_for_g(f, M)

        return f, M, False

    f_new = f_new / s

import numpy as np
from scipy.optimize import linprog
import time
import math

"""
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

Key idea: At each step, fix current f, solve LP to find g maximizing sum(g)
subject to g >= 0 and (f*g)[k] <= max(f*f) for all k. Blend f with normalized g
using a line-searched step size. This produces directions that push mass away
from convolution peaks. Multiple starting points and adaptive restarts avoid
local minima.
"""


def make_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = linear convolution of f and g."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_lp_for_g(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A_ub = make_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for non-negative f."""
    n = len(f)
    s = float(f.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(f, f)
    return float(2.0 * n * c.max() / (s * s))


def compute_M(f):
    """Compute max of linear convolution of f with itself."""
    c = np.convolve(f, f)
    return float(c.max())


def normalize_nonneg(f):
    """Clip to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    if s < 1e-30:
        return None
    return f / s


def make_linear_profile(n, slope=2.0):
    """Linear ramp profile, normalized to sum 1."""
    t = np.arange(n, dtype=np.float64)
    f = 1.0 + slope * t / max(n - 1, 1)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_cubic_profile(n, a1=1.5, a3=-1.0):
    """Anti-symmetric cubic profile."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    f = 1.0 + a1 * u + a3 * u ** 3
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_gaussian_bump(n, mu=0.5, sigma=0.2):
    """Gaussian bump profile."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def make_dual_bump(n):
    """Two bumps at edges."""
    t = np.linspace(0.0, 1.0, n)
    f = np.exp(-0.5 * (t / 0.15) ** 2) + np.exp(-0.5 * ((t - 1) / 0.15) ** 2) + 0.1
    f = np.maximum(f, 1e-12)
    return f / f.sum()


def get_starting_points(n, rng, count=20):

def interpolate_profile(f, n_new):
    """Linearly interpolate profile to new length."""
    n0 = len(f)
    if n0 == n_new:
        return f.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    return np.interp(x1, x0, f)

def soft_max_gradient(c, beta):
    """Compute gradient of soft-max of c with respect to input."""
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    return mval + beta * np.log(s), w


def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1 (using FFT)."""
    n = len(p)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]

    f = np.maximum(f_init, 1e-12)

    n = len(f)

    z -= z.mean()

    beta = max(1e-12, 0.2 * M)

    best_z = z.copy()
    best_M = M
    no_imp = 0
    base_lr = lr

        if time.monotonic() > deadline:
            break
        
        L, w = soft_max_gradient(c, beta)
        gp = 2.0 * corr_weights(w, f)
        vp = gp + 1e-7 * f
        dot = float(np.dot(f, vp))
        gz = f * (vp - dot)
        gz -= gz.mean()
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            adam_t = 0
            continue
        
        z = np.clip(z, -50, 50)
        z -= z.mean()
        e = np.exp(z)
        f = e / e.sum()
        c = np.convolve(f, f)
        M = float(c.max())
        
        if M < best_M - 1e-12:
            best_M = M
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)
        
        if adam_t % 150 == 0:
            beta = max(1e-12, beta * 0.8)
        
        if no_imp > max(300, 3 * n):
            z = 0.8 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z -= z.mean()
            e = np.exp(z)
            f = e / e.sum()
            c = np.convolve(f, f)
            M = float(c.max())
            beta = max(1e-10, 0.2 * M)
            m.fill(0)
            v.fill(0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_f, best_M
