def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def bump_normalized(n, width):

    n = int(n)

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def bump_normalized(n, width_frac):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = width_frac * n
    a = np.exp(-0.5 * ((idx - center) / max(1.0, width)) ** 2)
    return normalize_nonneg(a)


def flat_top_normalized(n, edge_frac):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.ones(n, dtype=np.float64)
    edge = max(1, int(edge_frac * n))
    for i in range(min(edge, n // 2)):
        t = i / max(1, edge)
        val = 0.5 * (1.0 - np.cos(np.pi * t))
        a[i] = val
        if n - 1 - i != i:
            a[n - 1 - i] = val
    return normalize_nonneg(a)


def mix_greedy_uniform(n, lam):

def actual_score_fft(p, direct_limit=512):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01 * 1e-6:
        return np.inf
    if n <= direct_limit:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

def adam_prefixed(p, budget_s, seed=0, t_init=80.0, t_end=8.0, lr_scale=1.0):
    """
    Adam gradient descent on log-sum-exp smoothed max-convolution objective.
    p is normalized to sum 1. We minimize 2*n*max(p*p).
    """
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 0.5:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.06 * lr_scale
    elif n < 256:
        lr = 0.04 * lr_scale
    elif n < 1024:
        lr = 0.025 * lr_scale
    else:
        lr = 0.018 * lr_scale

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    step_count = 0

    L_fft = next_power_of_two(2 * n - 1)

    best_p = p.copy()
    best_f = actual_score_fft(p)
    global_best_f = best_f
    global_best_p = best_p.copy()

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    stagnation = 0
    patience = 150

    while time.monotonic() < end:
        step_count += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Temperature: convex combination from t_init to t_end
        temp = t_init + (t_end - t_init) * frac

        # Compute convolution
        Pg = np.fft.rfft(p, L_fft)
        c = np.fft.irfft(Pg * Pg, L_fft)[:2 * n - 1].real
        c = np.maximum(c, 0.0)

        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()
            stagnation = 0
        else:
            stagnation += 1

        if np.isfinite(f_now) and f_now < global_best_f:
            global_best_f = f_now
            global_best_p = p.copy()

        if stagnation > patience:
            lr *= 0.6
            t_end = max(3.0, t_end * 0.7)
            stagnation = 0
            if lr < 0.002:
                break

        # Softmax weights: w_k = exp(temp * c_k) / sum exp(temp * c_j)
        c_shifted = temp * (c - maxc)
        w = np.exp(c_shifted)
        sz = float(w.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / sz

        # Gradient via cross-correlation of w and p
        W = np.fft.rfft(w[::-1], L_fft)
        Cfull = np.fft.irfft(Pg * W, L_fft).real
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        # The gradient of the log-sum-exp w.r.t. p is proportional to C
        # Objective = 2n * L, so grad_obj = 2n * (1/temp) * gradient of log-sum-exp
        # But since we're doing gradient descent on p (not x = log p), and
        # using softmax parametrization, we just need the direction.
        grad_p = C

        # Project to tangent space of simplex (sum p = 1)
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Adam
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** step_count)
        vhat = v / (1.0 - nu ** step_count)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -70.0, 70.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    if best_f < global_best_f:
        global_best_f = best_f
        global_best_p = best_p.copy()
    return global_best_p, global_best_f


def lp_improve(p, budget_s, max_rows=None):
    """LP cutting-plane: maximize sum(g) s.t. f*g <= M, g >= 0."""
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2:
        return p, np.inf
    budget_s = float(budget_s)
    if budget_s < 20.0:
        return p, actual_score_fft(p)

    if max_rows is None:
        if n > 4000:
            max_rows = 600
        elif n > 1500:
            max_rows = 800
        else:
            max_rows = 900

    max_elements = 5000000
    while max_rows * n > max_elements:
        max_rows = max(80, int(max_rows * 0.8))
    max_rows = max(80, min(int(max_rows), max_elements // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)
    end = time.monotonic() + max(0.0, budget_s - 5.0)

    r0 = min(max_rows, max(40, max_rows // 2))
    rows0 = np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int).astype(int))
    rows = [int(k) for k in rows0.tolist()]
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)

        A = np.zeros((int(rows_arr.size), n), dtype=np.float64)
        for ii in range(int(rows_arr.size)):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(int(rows_arr.size), cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": float(tl)},
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full constraint violations via FFT
        L2 = next_power_of_two(2 * n - 1)
        Pf = np.fft.rfft(f, L2)
        Pg = np.fft.rfft(g, L2)
        conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        tol = 1e-7 * max(1.0, cur_score)
        violations = conv - cur_score
        add_needed = np.any(violations > tol)

        if add_needed:
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                for ix in order:
                    if added >= min(128, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0:
                if float(np.max(violations)) > 10.0 * tol:
                    break
            else:
                continue

        g = g * (S0 / sg)

        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.25, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score):
            break
        if accepted_score >= cur_score * (1.0 - 1e-9):
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

    return normalize_nonneg(best_f), best_score

        candidate = random_perturb(p, rng, sigma)

        num_evals += 1

        if cand_f < cur_f or rng.random() < np.exp(-(cand_f - cur_f) / max(1e-10, temp)):
            p = candidate
            cur_f = cand_f
            if cur_f < best_f:
                best_f = cur_f
                best_p = p.copy()

    return best_p, best_f
