def normalize_nn(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s

def smooth_max_grad(a, c, M, temp):
    """
    Compute the subgradient of the smooth-max approximation of max(c).
    Returns grad of shape (n,) where grad[j] = sum_k w[k]*a[k-j] (valid indices).
    This is the cross-correlation of w with a.
    """
    n = len(a)
    m = len(c)
    # Softmax weights (emphasizing the max)
    e = (c - M) / temp
    e = np.clip(e, -80.0, 0.0)
    w = np.exp(e)
    sw = float(w.sum())
    if sw < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / sw

    # Cross-correlation: grad[j] = sum_{i=0}^{n-1} w[j+i] * a[i]
    # = ifft(FFT(w_padded) * conj(FFT(a_padded)))[:n]
    size = 1
    while size < m + n - 1:
        size <<= 1
    w_pad = np.zeros(size, dtype=np.float64)
    w_pad[:m] = w
    a_pad = np.zeros(size, dtype=np.float64)
    a_pad[:n] = a

    w_fft = np.fft.rfft(w_pad, size)
    a_fft = np.fft.rfft(a_pad, size)
    cc = np.fft.irfft(w_fft * np.conj(a_fft), size)
    grad = cc[:n].copy()
    return grad

def lp_direction(f, M, max_rows, n):
    """Solve LP to find improvement direction g: max sum(g) s.t. A@b <= M, g >= 0."""
    if linprog is None or n > 3000:
        return None
    c = fft_convolve_self(f)
    m = len(c)
    k = min(max_rows, m)
    if k < 2:
        return None
    # Select top-k lags
    rows = np.argpartition(c, -k)[-k:]
    rows = rows.astype(np.int64)

    # Build constraint matrix
    A = np.zeros((k, n), dtype=np.float64)
    for idx, k_val in enumerate(rows):
        k_val = int(k_val)
        # c[k_val] = sum_i f[i]*f[k_val-i] for valid i
        # d c[k_val] / d f[j] = f[k_val-j] (if 0 <= k_val-j < n)
        j_lo = max(0, k_val - n + 1)
        j_hi = min(n, k_val + 1)
        if j_hi > j_lo:
            A[idx, j_lo:j_hi] = f[k_val - (j_lo + np.arange(j_hi - j_lo))]

    # Normalize constraints
    if M <= 1e-30:
        return None
    A = A / M
    b = np.ones(k)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0.0, None), method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, 30.0 * n / 512.0)})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-12:
                return g
    except Exception:
        pass
    return None

def block_coordinate_descent(a, c, M, block_size, n_steps, rng):
    """Optimize blocks of coordinates while keeping others fixed."""
    n = len(a)
    a = a.copy()
    for _ in range(n_steps):
        idx = rng.integers(0, n - block_size + 1)
        end = idx + block_size
        # Fix all except a[idx:end]
        fixed = a.copy()
        fixed[idx:end] = 0.0
        fixed_sum = fixed.sum()
        # We want to optimize a[idx:end] to minimize max of convolution
        # The convolution with the fixed part is known
        # c_fixed[k] = conv(fixed, fixed)[k] + 2*conv(fixed, block)[k] + conv(block, block)[k]
        # This is complex; use a simple approach: try random perturbations of the block
        best_block = a[idx:end].copy()
        best_sc = score_norm(a)
        for _ in range(12):
            candidate = a.copy()
            # Random perturbation of the block
            candidate[idx:end] = a[idx:end] * np.exp(rng.normal(0, 0.3, size=block_size))
            candidate = normalize_nn(candidate)
            sc = score_norm(candidate)
            if sc < best_sc:
                best_sc = sc
                best_block = candidate[idx:end].copy()
                a = candidate
    return a

def tapered_block(n, ratio, taper_frac):
    """Two-block sequence with tapered edges."""
    a = np.zeros(n)
    split = int(n * ratio)
    # Block 1: from 0 to split
    if split > 0:
        t = np.linspace(0, 1, split)
        # Taper: ramp up then plateau
        tw = int(taper_frac * split)
        for i in range(split):
            if i < tw:
                a[i] = (i + 1) / tw
            else:
                a[i] = 1.0
    # Block 2: from split to n
    if n - split > 0:
        t = np.linspace(0, 1, n - split)
        tw2 = int(taper_frac * (n - split))
        for i in range(n - split):
            if i < tw2:
                a[split + i] = (i + 1) / tw2
            elif i >= (n - split) - tw2:
                a[split + i] = ((n - split - 1 - i) + 1) / tw2
            else:
                a[split + i] = 1.0
    a = np.clip(a, 0, None)
    return normalize_nn(a)

def exp_decay(n, alpha):
    """Exponentially decaying sequence."""
    x = np.arange(n, dtype=np.float64) / n
    a = np.exp(-alpha * x)
    return normalize_nn(a)

def gaussian_mixture(n, n_comp, rng):
    """Gaussian mixture sequence."""
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for _ in range(n_comp):
        mu = rng.uniform(0.1, 0.9)
        sig = rng.uniform(0.05, 0.3)
        w = rng.uniform(0.5, 2.0)
        a += w * np.exp(-0.5 * ((x - mu) / sig) ** 2)
    return normalize_nn(np.maximum(a, 0))

def block_sequence(n, ratio):
    """Simple two-block sequence."""
    a = np.zeros(n)
    split = max(1, int(n * ratio))
    a[:split] = 1.0 / split
    a[split:] = 1.0 / (n - split)
    return normalize_nn(a)

def sine_modulated(n, freq, amp, rng):
    """Sine-modulated sequence."""
    x = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + amp * np.sin(2 * np.pi * freq * x + phase)
    return normalize_nn(np.maximum(a, 0))

def optimize_sequence(a, time_limit, rng, state, use_lp=True, lp_interval=5):
    """Main optimization loop for a single sequence."""
    a = normalize_nn(a.copy())
    n = len(a)
    c = fft_convolve_self(a)
    M = float(c.max())
    cur_score = 2.0 * n * M

    best_a = a.copy()
    best_score = cur_score
    if cur_score < state['score']:
        state['score'] = cur_score
        state['seq'] = a.copy()
        print(f"  init: score={cur_score:.10f}, n={n}", flush=True)

    deadline = time.monotonic() + time_limit
    temp = max(0.02 * M, 1e-8)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve_self(a)
        M = float(c.max())
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            if best_score < state['score'] - 1e-14:
                state['score'] = best_score
                state['seq'] = best_a.copy()
                if it % 50 == 1 or best_score < state.get('last_print', 10) - 0.001:
                    print(f"  it={it}: score={best_score:.10f}, n={n}, M={M:.6e}", flush=True)
                    state['last_print'] = best_score
            no_improve = 0
            temp = max(0.01 * M, 1e-10) * (0.97 ** min(it, 100))
        else:
            no_improve += 1
            if no_improve > 20:
                temp = min(0.5 * M, temp * 1.3)

        # Compute gradient
        grad = smooth_max_grad(a, c, M, temp)
        # Center the gradient (remove mean to preserve sum approximately)
        grad = grad - grad.mean()
        grad_norm = float(np.sqrt(np.sum(grad**2)))
        if grad_norm < 1e-15:
            break

        # Adaptive step size with backtracking
        max_step = 0.4
        improved = False
        for atrial in range(8):
            alpha = max_step * (0.5 ** atrial)
            step = alpha * grad / grad_norm
            h = a - step
            h = normalize_nn(h)
            sc = score_norm(h)
            if sc < cur_score - 1e-12:
                a = h
                improved = True
                break

        if not improved:
            # Try LP direction
            if use_lp and (it % lp_interval == 0 or no_improve > 8) and n <= 1024:
                max_rows = min(2 * n - 1, max(100, n // 2))
                g = lp_direction(a, M, max_rows, n)
                if g is not None:
                    g = normalize_nn(g)
                    best_t = None
                    for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0):
                        h = normalize_nn((1 - t) * a + t * g)
                        sc = score_norm(h)
                        if sc < cur_score - 1e-12:
                            if best_t is None or sc < best_t[1]:
                                best_t = (h, sc)
                    if best_t is not None:
                        a = best_t[0]
                        improved = True

        if not improved:
            # Random perturbation (escape)
            scale = 0.08 + 0.04 * np.sqrt(no_improve)
            h = a * np.exp(rng.normal(0, scale, size=n))
            a = normalize_nn(h)
            if no_improve > 40:
                # More aggressive: block swap
                if n >= 64:
                    bs = int(rng.integers(n // 8, n // 3))
                    pos = int(rng.integers(0, n - bs + 1))
                    h = a.copy()
                    h[pos:pos+bs] = a[pos:pos+bs] * rng.uniform(0.3, 2.0, size=bs)
                    a = normalize_nn(h)

    # Final check
    sc = score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    if best_score < state['score']:
        state['score'] = best_score
        state['seq'] = best_a.copy()
    return best_a, best_score

def resize_seq(a, target_n):
    """Resize sequence to target length via interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.full(target_n, 1.0 / target_n)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    new_a = np.interp(x_new, x_old, a)
    return normalize_nn(np.maximum(new_a, 0))
