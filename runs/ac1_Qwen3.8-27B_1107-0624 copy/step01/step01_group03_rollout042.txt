import numpy as np
from scipy.optimize import linprog
import time

def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2 without external calls."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if 0<=k-j<n else 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) subject to (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            if b.sum() > 1e-14:
                return b
    except Exception:
        pass
    return None

def subgradient_step(s, step_size):
    """Compute subgradient of max(conv(s,s)) and take a step."""
    n = len(s)
    conv = np.convolve(s, s)
    M = np.max(conv)
    # Find active set (within tolerance of max)
    tol = 0.01 * M + 1e-15
    active = np.where(conv >= M - tol)[0]
    
    # Compute subgradient: average of gradients of active terms
    # grad_k[i] = 2 * s[k-i] for valid k-i
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += 2.0 * s[ki]
    grad /= len(active)
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return s.copy(), 0.0
    
    # Step in negative gradient direction
    new_s = s - step_size * grad
    new_s = np.maximum(new_s, 0.0)
    if new_s.sum() < 1e-15:
        return s.copy(), 0.0
    new_s = new_s / new_s.sum()
    return new_s, M

def line_search(s, direction, current_score, max_steps=20):
    """Find best step size in given direction."""
    best_t = 0.0
    best_sc = current_score
    t = 0.1
    for _ in range(max_steps):
        cand = s + t * direction

            t *= 0.5
            continue

        if sc < best_sc:
            best_sc = sc
            best_t = t
        t *= 0.7  # or 0.5
        if t < 1e-6:
            break
    return best_t, best_sc

    n = len(s)
