```python
import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_solve(a, M):
    n = len(a)
    if n > 4096:
        return None
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_step(a, lr):
    """One subgradient step for minimizing max(conv(a,a)) subject to sum(a)=1, a>=0."""
    c = np.convolve(a, a, mode='full')
    M = c.max()
    peak_mask = c >= M - 1e-12 * M
    
    n = len(a)
    s = np.zeros(n)
    peak_indices = np.where(peak_mask)[0]
    for k in peak_indices:
        for j in range(n):
            d = k - j
            if 0 <= d < n:
                if 2 * j == k:
                    s[j] += a[d]
                else:
                    s[j] += 2.0 * a[d]
    s /= len(peak_indices)
    
    a_new = a - lr * s
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() < 1e-10:
        return a
    a_new /= a_new.sum()
    return a_new
```

```python
def compute_subgradient(a, peak_indices):
    n = len(a)
    s = np.zeros(n)
    for k in peak_indices:
        # For each j, contribution is 2*a[k-j] if 0 <= k-j < n and k != 2j
        # or a[j] if k = 2j
        d = k - np.arange(n)  # d[j] = k - j
        valid = (d >= 0) & (d < n)
        contrib = np.zeros(n)
        contrib[valid] = 2.0 * a[d[valid]]
        # Fix the center: if 2j == k, contribution should be a[j] not 2*a[j]
        if k % 2 == 0:
            center = k // 2
            if 0 <= center < n:
                contrib[center] = a[center]  # Not 2*a[center]
        s += contrib
    s /= len(peak_indices)
    return s
```

```python
"""
Hybrid subgradient + adaptive LP optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradients for fast initial convergence, adaptive LP targets for refinement,
and coordinate descent for final polishing. Multiple diverse initializations at different
sequence lengths are explored with time-weighted allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_convolve_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = np.convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M):
    """Solve max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 4096:
        return None
    A = build_convolve_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None


def exact_subgradient(a, c):
    """Compute exact subgradient of max_k (a*a)[k]."""
    n = len(a)
    M = c.max()
    tol = 1e-10 * M
    peak_idx = np.where(c >= M - tol)[0]
    s = np.zeros(n)
    for k in peak_idx:
        d = k - np.arange(n)
        valid = (d >= 0) & (d < n)
        contrib = np.zeros(n)
        contrib[valid] = 2.0 * a[d[valid]]
        if k % 2 == 0:
            ctr = k // 2
            if 0 <= ctr < n:
                contrib[ctr] = a[ctr]
        s += contrib
    s /= len(peak_idx)
    return s


def subgradient_optimize(a, max_time, rng):
    """Fast subgradient optimization. Returns (a, score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.15
    stall = 0
    it = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < max_time:
        it += 1
        c = np.convolve(a, a, mode='full')
        s = exact_subgradient(a, c)
        
        a_new = a - lr * s
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            lr *= 0.5
            continue
        a_new /= a_new.sum()
        sc = compute_score(a_new)
        
        if sc < best_score - 1e-13:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.4, lr * 1.005)
        else:
            stall += 1
            lr = max(0.002, lr * 0.97)
            if stall > 80:
                noise = np.abs(rng.normal(0, 0.04, n))
                a_pert = best_a * (1.0 + noise)
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc_pert = compute_score(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                    a = a_pert
                    stall = 0
                else:
                    break
            if stall > 300:
                break
    
    return best_a, best_score


def lp_optimize(a, max_time, rng):
    """LP-based refinement with adaptive M target."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    it = 0
    M_target = None  # Will be set to current max initially
    
    while time.monotonic() - t0 < max_time:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        if M_target is None:
            M_target = M
        else:
            # Gradually decrease M_target towards M
            M_target = 0.9 * M_target + 0.1 * M
            # But never go below M (infeasible)
            M_target = max(M_target, M * 0.999)
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None or g.sum() < 1e-14:
            # Try with slightly larger M
            g = solve_lp(a, M * 1.1)
            if g is None or g.sum() < 1e-14:
                M_target = M  # Reset
                continue
        
        g = g / g.sum()
        
        # Backtracking line search
        improved = False
        for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            a_new = (1.0 - t) * best_a + t * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if not improved:
            # Perturbation
            noise = np.abs(rng.normal(0, 0.02, n))
            a_pert = best_a * (1.0 + noise)
            a_pert = np.clip(a_pert, 0.0, None)
            if a_pert.sum() > 1e-10:
                a_pert /= a_pert.sum()
                a = a_pert
        
        if it % 50 == 0:
            print(f"    LP iter {it}, score={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent(a, max_time, rng):
    """Coordinate descent polish."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < max_time:
        improved = False
        c = np.convolve(a, a, mode='full')
        M = c.max()
        S = a.sum()
        
        for j in rng.permutation(n):
            if time.monotonic() - t0 > max_time:
                break
            
            # Try different values for a[j]
            old_val = a[j]
            # Compute the effect of changing a[j]
            # c[k] changes by: a[j] * a[k-j] + a[k-j] * a[j] - old_val * a[k-j] - a[k-j] * old_val
            # = (new_val - old_val) * (a[k-j] + a[k-j]) = (new_val - old_val) * 2 * a[k-j]
            # (except when k-j == j, where it's (new_val - old_val) * a[j])
            
            # Try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5,
                         old_val + 0.01, old_val - 0.01, 0.0]
            
            for cv in candidates:
                cv = max(0.0, cv)
                delta = cv - old_val
                if abs(delta) < 1e-12:
                    continue
                
                # Update convolution efficiently
                # c_new[k] = c[k] + delta * (a[k-j] + a[k-j]) for valid k
                # But this is O(n) per candidate, and we have O(n) candidates per j, so O(n^2) total
                # For n=512, this is ~262K operations, which is fine
                
                new_c = c.copy()
                for k in range(2 * n - 1):
                    d = k - j
                    if 0 <= d < n:
                        if d == j:
                            # a[j]*a[j] term: (old+delta)^2 - old^2 = 2*old*delta + delta^2
                            new_c[k] += 2.0 * old_val * delta + delta * delta
                        else:
                            new_c[k] += delta * a[d] * 2.0
                    # Also need to handle the case where j is the other index
                    # k - j gives d, but we also need j - (k-j) = 2j - k
                    # Actually, the convolution c[k] = sum_i a[i]*a[k-i]
                    # The terms involving a[j] are: a[j]*a[k-j] (when i=j) and a[k-j]*a[j] (when k-i=j, i.e., i=k-j)
                    # These are the same term when j = k-j, i.e., k = 2j
                    # So the total contribution of a[j] to c[k] is:
                    #   a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] (if k != 2j)
                    #   a[j]*a[j] = a[j]^2 (if k = 2j)
                    # When we change a[j] to a[j]+delta:
                    #   New contribution: (a[j]+delta)*a[k-j] + a[k-j]*(a[j]+delta) = 2*(a[j]+delta)*a[k-j] (if k != 2j)
                    #   or (a[j]+delta)^2 (if k = 2j)
                    #   Change: 2*delta*a[k-j] (if k != 2j) or 2*delta*a[j] + delta^2 (if k = 2j)
                
                new_S = S + delta
                if new_S < 1e-10:
                    continue
                new_M = new_c.max()
                new_score = 2.0 * n * new_M / (new_S * new_S)
                
                if new_score < best_score - 1e-13:
                    a[j] = cv
                    c = new_c
                    S = new_S
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    break
            
            a = np.clip(a, 0.0, None)
            if a.sum() < 1e-10:
                a = best_a.copy()
                c = np.convolve(a, a, mode='full')
                S = a.sum()
        
        if not improved:
            break
    
    return best_a, best_score
```

```python
Hybrid subgradient + adaptive LP + coordinate descent optimization
to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradients for fast initial convergence, LP with adaptive

import numpy as np

from scipy.optimize import linprog


def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_convolve_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M):
    n = len(a)
    if n > 4096:
        return None
    A = build_convolve_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            if g.sum() > 1e-14:
                return g
    except Exception:
        pass
    return None


def exact_subgradient(a, c):
    n = len(a)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n)
    tol = 1e-9 * M
    peak_idx = np.where(c >= M - tol)[0]
    s = np.zeros(n)
    for k in peak_idx:
        d = k - np.arange(n)
        valid = (d >= 0) & (d < n)
        contrib = np.zeros(n)
        contrib[valid] = 2.0 * a[d[valid]]
        if k % 2 == 0:
            ctr = k // 2
            if 0 <= ctr < n:
                contrib[ctr] = a[ctr]
        s += contrib
    s /= len(peak_idx)
    return s

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n))) + 1e-12


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 1e-12


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.clip(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 1e-12, None)


def make_biphasic(n):

    a[half:] = 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n - half)))
    return a + 1e-12
```
