```python
def solve_lp(f, M, n):
    """Solve: max sum(g) s.t. f*g <= M (component-wise), g >= 0"""
    # Build the constraint matrix A where A[k,i] = f[k-i] if 0 <= k-i < n
    # A is (2n-1) x n, sparse
    # Use scipy.optimize.linprog
    from scipy import sparse
    from scipy.optimize import linprog
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(max(0, k-n+1), min(k+1, n)):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
    
    A = sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
    
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return f  # fallback

    return best_f, best_J
```

```python
def build_conv_matrix(f, n):
    """Build sparse matrix A where A[k,i] = f[k-i] if valid."""
    from scipy import sparse
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # For row k, cols are i_start to i_end, values are f[k-i]
        # f[k-i_start] to f[k-i_end] = f[k-i_start] to f[k-i_end]
        # which is f[k-i_start], f[k-i_start-1], ..., f[k-i_end]
        # i.e., f[rev] where rev goes from k-i_start down to k-i_end
        indices = np.arange(i_start, i_end + 1)
        vals_k = f[k - indices]
        rows.extend([k] * len(indices))
        cols.extend(indices)
        vals.extend(vals_k)
    return sparse.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n)).tocsr()
```

```python
"""
LP-based iterative optimization with active set reduction, combined with
Nesterov subgradient descent and local search, to minimize the peak
auto-convolution ratio J(a) = 2n*max(a*a)/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def solve_reduced_lp(f, active_indices, M, n):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0.
    Returns optimal g.
    """
    K = len(active_indices)
    # Build constraint matrix A (K x n), A[k_idx, i] = f[k - i] if valid
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            A[idx, i_start:i_end+1] = f[k - i_start:k - i_end + 1][::-1]
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(K, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

def lp_iteration(f, n, budget_s, seed, top_k=100):
    """LP-based iterative method with active set."""
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(seed)
    
    f = f.copy()
    S = np.sum(f)
    if S > 0:
        f = f * (np.sqrt(2 * n) / S)  # normalize
    
    best_f = f.copy()
    best_J = compute_J_fft(f)
    t_blend = 0.5
    stall_count = 0
    
    iteration = 0
    while time.monotonic() < deadline:
        c = convolve_fft(f)
        M = np.max(c)
        
        # Active set: top top_k bins
        top_k_actual = min(top_k, 2 * n - 1)
        active_idx = np.argsort(c)[-top_k_actual:]
        
        # Solve LP
        g = solve_reduced_lp(f, active_idx, M, n)
        if g is None or not np.isfinite(g).all() or np.sum(g) < 1e-10:
            t_blend *= 0.9
            if t_blend < 0.01:
                break
            continue
        
        # Normalize g to match sum of f
        S_f = np.sum(f)
        S_g = np.sum(g)
        g = g * (S_f / S_g)
        
        # Blend
        f_new = (1 - t_blend) * f + t_blend * g
        f_new = np.maximum(f_new, 0.0)
        
        J_new = compute_J_fft(f_new)
        if J_new < best_J:
            best_J = J_new
            best_f = f_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                # Reduce blend and try again
                t_blend = max(0.05, t_blend * 0.8)
                if t_blend < 0.05:
                    break
        f = f_new
        iteration += 1
        
        # Adaptively adjust top_k and t_blend
        if iteration % 20 == 0:
            t_blend = max(0.1, t_blend * 0.95)
        
        if iteration % 50 == 0:
            print(f"    LP iter {iteration}: J={best_J:.6f}, M={M:.2f}, t={t_blend:.3f}")
    
    return best_f, best_J


def nesterov_subgrad(f, n, budget_s, seed):
    """Nesterov accelerated subgradient descent."""
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(seed)
    
    a = f.copy()
    S = np.sum(a)
    target_S = np.sqrt(2 * n)
    a = a * (target_S / S)
    
    y = a.copy()
    best_a = a.copy()
    best_J = compute_J_fft(a)
    
    L = 1.0  # Lipschitz constant estimate
    t_k = 1.0
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        c = convolve_fft(y)
        M = np.max(c)
        # Subgradient: average of a[k*-i] over all argmax k*
        max_indices = np.where(np.abs(c - M) < 1e-6 * M)[0]
        
        S = np.sum(y)
        # Gradient of J = 2n*M/S^2 with respect to a[i]
        # dM/da[i] = mean over argmax k of: y[k-i] (with bounds)
        # dJ/da[i] = 2n/S^2 * dM/da[i] - 4n*M/S^3
        
        grad_M = np.zeros(n)
        for k in max_indices:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                grad_M[i_start:i_end+1] += y[k - i_start:k - i_end + 1][::-1]
        grad_M /= len(max_indices)
        
        grad_J = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M) / (S * S * S)
        
        # Step size
        eta = target_S / (np.linalg.norm(grad_J) + 1e-10)
        
        # Nesterov update
        a_new = y - eta * grad_J
        # Project onto {a >= 0, sum(a) = target_S}
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new > 0:
            a_new = a_new * (target_S / S_new)
        
        # Update y
        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        beta = (t_k - 1) / t_new
        y = a_new + beta * (a_new - a)
        a = a_new
        t_k = t_new
        
        J = compute_J_fft(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    return best_a, best_J


def local_search(f, n, budget_s, seed):
    """Local search with block perturbations."""
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(seed)
    
    a = f.copy()
    best_a = a.copy()
    best_J = compute_J_fft(a)
    step_scale = 0.03
    
    while time.monotonic() < deadline:
        width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)
        
        cand = a.copy()
        noise = rng.normal(0.0, step_scale, end - start)
        cand[start:end] *= np.exp(noise)
        cand = np.maximum(cand, 0.0)
        
        cJ = compute_J_fft(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.97)
        else:
            step_scale = min(0.2, step_scale * 1.03)
    
    return best_a, best_J
```

```python
LP-based active set iteration combined with Nesterov subgradient descent
and local search to minimize J(a) = 2n*max(a*a)/(sum(a))^2.
Key innovation: reduced-constraint LP using only top-K active bins,

import numpy as np

from scipy.optimize import linprog


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def build_lp_matrix(f, active_indices, n):
    """Build dense constraint matrix for reduced LP."""
    K = len(active_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            A[idx, i_start:i_end + 1] = f[k - i_start:k - i_end + 1][::-1]
    return A


def solve_reduced_lp(f, active_indices, M, n):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    K = len(active_indices)
    A = build_lp_matrix(f, active_indices, n)
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None

        if iteration % 10 == 0:
            progress = min(1.0, iteration / max(1, budget_s / 2.0))
            t_blend = t_blend_start * (t_blend_end / t_blend_start) ** progress
            t_blend = max(t_blend, t_blend_end)

        if t_blend < t_blend_end * 1.05 and stall_count > 30:
            break

        if iteration % 30 == 0:
            print(f"    LP iter={iteration} J={best_J:.6f} M={M:.2f} "
                  f"top_k={top_k} t_blend={t_blend:.3f} stall={stall_count}")

    # Final: return best found
    final_J = compute_J_fft(best_f)
    best_J = min(best_J, final_J)
    return best_f, best_J


def nesterov_subgrad(f, n, budget_s, seed):
    """Nesterov accelerated subgradient descent with projection."""
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(seed)

    a = np.maximum(f.copy(), 0.0)
    S = np.sum(a)
    target_S = np.sqrt(2.0 * n)
    a = a * (target_S / S)

    y = a.copy()
    best_a = a.copy()
    best_J = compute_J_fft(a)
    t_k = 1.0
    mu = 1.0  # step size scale

    it = 0
    while time.monotonic() < deadline:
        c = convolve_fft(y)
        M = np.max(c)
        S = np.sum(y)

        # Find argmax bins (within tolerance)
        tol = 1e-8 * M
        max_mask = c >= (M - tol)
        max_indices = np.where(max_mask)[0]
        if len(max_indices) == 0:
            max_indices = np.array([np.argmax(c)])

        # Subgradient of M = max(c) w.r.t. y[i]
        # d(c[k])/dy[i] = y[k-i] if 0 <= k-i < n
        # Subgradient of max = average over argmax
        grad_M = np.zeros(n)
        for k in max_indices:
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            if i_start <= i_end:
                grad_M[i_start:i_end + 1] += y[k - i_start:k - i_end + 1][::-1]
        grad_M /= len(max_indices)

        # Gradient of J = 2n*M/S^2
        grad_J = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M) / (S ** 3) * y
        # Actually the second term should be uniform: -4n*M/S^3 for each component
        # Let me redo: J = 2n*M/S^2, dJ/dy[i] = 2n/S^2 * dM/dy[i] - 4n*M/S^3 * dy[i]/dS
        # S = sum(y), so dS/dy[i] = 1
        # dJ/dy[i] = 2n/S^2 * dM/dy[i] - 4n*M/S^3
        # The second term is the same for all i
        grad_J = (2.0 * n / (S * S)) * grad_M - np.full(n, 4.0 * n * M / (S ** 3))

        # Adaptive step size
        grad_norm = np.linalg.norm(grad_J)
        if grad_norm < 1e-15:
            break
        eta = mu / grad_norm
        mu *= 0.999  # slow decay

        # Nesterov update
        a_new = y - eta * grad_J
        # Project: a >= 0, sum(a) = target_S
        a_new = np.maximum(a_new, 0.0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = np.full(n, target_S / n)
        else:
            a_new = a_new * (target_S / S_new)

        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new
        y = a_new + beta * (a_new - a)
        a = a_new
        t_k = t_new

        it += 1
        if it % 500 == 0:
            J = compute_J_fft(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
            mu = max(0.01, mu * 0.95)

    final_J = compute_J_fft(best_a)
    best_J = min(best_J, final_J)
    return best_a, best_J


def local_search(f, n, budget_s, seed):
    """Local search with adaptive block perturbations."""
    deadline = time.monotonic() + budget_s
    rng = np.random.default_rng(seed)

    a = np.maximum(f.copy(), 0.0)
    best_a = a.copy()
    best_J = compute_J_fft(a)
    step_scale = 0.02
    no_improve = 0

    while time.monotonic() < deadline:
        # Multi-scale perturbation
        scale = rng.choice([n // 200, n // 50, n // 20, n // 8],
                           p=[0.3, 0.3, 0.25, 0.15])
        width = max(2, int(scale))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)

        # Choose perturbation type
        ptype = rng.random()
        cand = a.copy()
        if ptype < 0.4:
            # Multiplicative noise
            noise = rng.normal(0.0, step_scale, end - start)
            cand[start:end] *= np.exp(noise)
        elif ptype < 0.7:
            # Additive noise
            noise = rng.normal(0.0, step_scale * np.mean(a[start:end]), end - start)
            cand[start:end] += noise
        else:
            # Swap-based: redistribute mass within block
            block_sum = cand[start:end].copy()
            perm = rng.permutation(end - start)
            cand[start:end] = block_sum[perm] * np.exp(rng.normal(0, step_scale * 0.5, end - start))

        # Also perturb a few random individual elements
        for _ in range(rng.integers(1, 6)):
            idx = int(rng.integers(0, n))
            cand[idx] *= np.exp(rng.normal(0.0, step_scale * 0.3))

        cand = np.maximum(cand, 0.0)
        cS = np.sum(cand)
        if cS < 0.01:
            continue

        cJ = compute_J_fft(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.97)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                step_scale = min(0.15, step_scale * 1.04)
            if no_improve > 500:
                # Reset to best with small perturbation
                a = best_a * np.exp(rng.normal(0, 0.01, n))
                a = np.maximum(a, 0)
                no_improve = 0

    return best_a, best_J


def make_u_shaped(n, alpha=2.0):
    """U-shaped: high at edges, low in middle."""
    x = np.arange(n) / max(n - 1, 1)
    return 0.5 + alpha * (x - 0.5) ** 2


def make_centered(n, width=0.3):
    """Centered bump with specified width."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = width * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_periodic(n, freq=3.0):
    """Periodic modulation."""
    x = np.arange(n)
    return 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)


def make_optimal_spectrum(n):
    """
    Design a sequence whose DFT has flat magnitude (ideal for flat convolution).
    Use a chirp-like phase: a[i] = exp(-alpha * (i-mid)^2 / n) * (1 + beta*cos(2*pi*gamma*i/n))
    """
    x = np.arange(n) - (n - 1) / 2.0
    base = np.exp(-0.3 * (x / (n * 0.4)) ** 2)
    mod = 1.0 + 0.2 * np.cos(2.0 * np.pi * 2.0 * x / n)
    return np.maximum(base * mod, 0.01)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: LP-based active set iteration (primary),
    Nesterov subgradient (secondary), local search (polishing).
    Uses diverse initializations and adaptive time allocation.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    # Get base sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Build diverse initializations
    inits = []
    inits.append(("given", parent.copy()))
    inits.append(("ones", np.ones(n)))
    inits.append(("centered_0.2", make_centered(n, 0.2)))
    inits.append(("centered_0.35", make_centered(n, 0.35)))
    inits.append(("u_shaped", make_u_shaped(n, 1.5)))
    inits.append(("u_shaped_3", make_u_shaped(n, 3.0)))
    inits.append(("periodic_3", make_periodic(n, 3.0)))
    inits.append(("periodic_7", make_periodic(n, 7.0)))
    inits.append(("opt_spec", make_optimal_spectrum(n)))

    # Random inits
    x = np.arange(n) - (n - 1) / 2.0
    for sf in [0.4, 0.6]:
        sigma = sf * n
        inits.append((f"rand_gauss_{sf}", np.exp(rng.normal(0, 0.3, n)) *
                      np.exp(-x * x / (2.0 * sigma * sigma))))

    # Score and sort
    scored = []
    for name, arr in inits:
        arr = np.maximum(arr, 0.0)
        S = np.sum(arr)
        if S > 0:
            arr = arr * (np.sqrt(2.0 * n) / S)
        J = compute_J_fft(arr)
        scored.append((J, name, arr))

    scored.sort(key=lambda x: x[0])
    for J, name, _ in scored:
        print(f"  init '{name}': J = {J:.6f}")

    # === Phase 1: LP iteration on top-2 initializations ===
    remaining = total_deadline - time.monotonic() - 30.0
    lp1_budget = remaining * 0.40
    lp2_budget = remaining * 0.25
    lp3_budget = remaining * 0.15
    nest_budget = remaining * 0.10
    local_budget = remaining * 0.10

    # LP on best init
    if lp1_budget > 15:
        print(f"\n=== Phase 1a: LP on '{scored[0][1]}' ({lp1_budget:.0f}s) ===")
        res_a, res_J = lp_iteration(scored[0][2], n, lp1_budget, seed=seed,
                                     start_top_k=50, max_top_k=250)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # LP on second best
    if lp2_budget > 10 and len(scored) > 1:
        print(f"\n=== Phase 1b: LP on '{scored[1][1]}' ({lp2_budget:.0f}s) ===")
        res_a, res_J = lp_iteration(scored[1][2], n, lp2_budget, seed=seed + 100,
                                     start_top_k=75, max_top_k=300)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # LP on current best (refinement)
    if lp3_budget > 10:
        print(f"\n=== Phase 1c: LP refinement on best ({lp3_budget:.0f}s) ===")
        res_a, res_J = lp_iteration(best_global, n, lp3_budget, seed=seed + 200,
                                     start_top_k=100, max_top_k=400,
                                     t_blend_start=0.3, t_blend_end=0.05)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # === Phase 2: Nesterov subgradient ===
    if nest_budget > 5:
        print(f"\n=== Phase 2: Nesterov ({nest_budget:.0f}s) ===")
        res_a, res_J = nesterov_subgrad(best_global, n, nest_budget, seed=seed + 300)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # === Phase 3: Local search polishing ===
    if local_budget > 5:
        print(f"\n=== Phase 3: Local search ({local_budget:.0f}s) ===")
        res_a, res_J = local_search(best_global, n, local_budget, seed=seed + 400)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # === Phase 4: Final LP refinement if time remains ===
    remaining = total_deadline - time.monotonic() - 10.0
    if remaining > 20:
        print(f"\n=== Phase 4: Final LP refinement ({remaining:.0f}s) ===")
        res_a, res_J = lp_iteration(best_global, n, remaining, seed=seed + 500,
                                     start_top_k=150, max_top_k=500,
                                     t_blend_start=0.2, t_blend_end=0.03)
        val = float(evaluate_sequence(np.clip(res_a, 0, 1000).tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Result: local={res_J:.6f}, verified={val:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(np.clip(best_global, 0.0, 1000.0).tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_global_score:.6f}")
    print(f"{'='*50}")

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
```
