    Result: (2/K) * sum_{k in topK} a_flipped_shifted_by_k
    """
    n = a.size
    n_c = len(c)
    K = min(K, n_c)
    top_idx = np.argpartition(-c, K - 1)[:K]
    
    grad = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        # Add (2/K) * a[k-p] for p in valid range
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        # For p = lo,...,hi: a[k-p] = a[k-hi], a[k-hi+1], ..., a[k-lo]
        # Wait: p=lo gives a[k-lo], p=hi gives a[k-hi]
        # So the vector is [a[k-lo], a[k-lo-1], ..., a[k-hi]] = a[k-hi:k-lo+1][::-1]
        grad[lo:hi+1] += (2.0 / K) * a[k-hi:k-lo+1][::-1]
    return grad


def _topK_avg_value(c, K):
    """Value of (1/K) * sum of top-K of c."""
    n_c = len(c)
    K = min(K, n_c)
    top = np.argpartition(-c, K - 1)[:K]

"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences a.
Uses FISTA (Nesterov accelerated projected gradient) with top-K average
as a convex surrogate for the peak, adaptive K scheduling, and
multi-phase optimization with restarts.

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} via Euclidean projection."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(sv * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_sq(a, m):
    """Linear convolve(a,a) via FFT of size m (m >= 2*len(a)-1)."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _grad_topK_avg(c, a, K):
    """Subgradient of (1/K)*sum of top-K c_k w.r.t. a.
    Each c_k contributes (2/K) * a[k-p] for valid p."""
    n = a.size
    n_c = len(c)
    K = min(K, n_c)
    if K == n_c:
        top_idx = np.arange(n_c)
    else:
        top_idx = np.argpartition(-c, K - 1)[:K]
    
    grad = np.zeros(n, dtype=np.float64)
    coef = 2.0 / K
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        grad[lo:hi + 1] += coef * a[k - hi:k - lo + 1][::-1]
    return grad


def _grad_softmax(c, a, K, tau):
    """Gradient of softmax-smoothed top-K max: tau * log(sum exp(c_k/tau) over top K)."""
    n = a.size
    n_c = len(c)
    K = min(K, n_c)
    if K == n_c:
        top_idx = np.arange(n_c)
    else:
        top_idx = np.argpartition(-c, K - 1)[:K]
    
    wc = c[top_idx]
    w = np.exp(wc / max(tau, 1e-15))
    w = w / w.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi <= 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        grad[lo:hi + 1] += 2.0 * wi * a[k - hi:k - lo + 1][::-1]
    return grad

            order = np.argsort(a_kp)  # sorted by a[k-p], smallest first

            a_new[p_inc] += amount

            n = b.size
    except Exception:
        pass
    if a0 is None:
        a0 = np.ones(n, dtype=np.float64) / n
    
    print(f"Starting optimization with n={n}")
    
    # Initial evaluation
    best_a = a0.copy()
    try:
        best_score = float(evaluate_sequence([float(x) for x in a0]))
    except Exception:
        best_score = float('inf')
    print(f"Initial score: {best_score:.6f}")
    
    # Phase 1: FISTA with large K (broad optimization) - 45% of time
    t_phase1 = t_start + 0.45 * (budget_s - 5)
    print(f"Phase 1: FISTA K={max(50, n//10)} -> 5, time until {t_phase1 - t_start:.0f}s")
    a1, mc1 = _fista_optimize(a0.copy(), t_phase1, K_start=max(50, n // 10), K_end=5, step_size=0.35)
    try:
        s1 = float(evaluate_sequence([float(x) for x in a1]))
    except Exception:
        s1 = float('inf')
    print(f"Phase 1 done: max_c={mc1:.8f}, score={s1:.6f}")
    if s1 < best_score:
        best_score = s1
        best_a = a1.copy()
    
    # Phase 2: FISTA from best with smaller K (focused) - 25% of time
    t_phase2 = t_start + 0.72 * (budget_s - 5)
    print(f"Phase 2: FISTA K=15 -> 2, from best")
    a2, mc2 = _fista_optimize(best_a.copy(), t_phase2, K_start=15, K_end=2, step_size=0.3)
    try:
        s2 = float(evaluate_sequence([float(x) for x in a2]))
    except Exception:
        s2 = float('inf')
    print(f"Phase 2 done: max_c={mc2:.8f}, score={s2:.6f}")
    if s2 < best_score:
        best_score = s2
        best_a = a2.copy()
    
    # Phase 3: FISTA from perturbed best - 15% of time
    t_phase3 = t_start + 0.85 * (budget_s - 5)
    a_pert = _proj_simplex(best_a * np.exp(0.05 * rng.standard_normal(n)), 1.0)
    a3, mc3 = _fista_optimize(a_pert, t_phase3, K_start=20, K_end=3, step_size=0.35)
    try:
        s3 = float(evaluate_sequence([float(x) for x in a3]))
    except Exception:
        s3 = float('inf')
    print(f"Phase 3 done: max_c={mc3:.8f}, score={s3:.6f}")
    if s3 < best_score:
        best_score = s3
        best_a = a3.copy()
    
    # Phase 4: Coordinate polish on the best - 10% of time
    t_phase4 = t_end
    print(f"Phase 4: Coordinate polish")
    a4 = _coordinate_polish(best_a.copy(), t_phase4, n_top=3)
    try:
        s4 = float(evaluate_sequence([float(x) for x in a4]))
    except Exception:
        s4 = float('inf')
    print(f"Phase 4 done: score={s4:.6f}")
    if s4 < best_score:
        best_score = s4
        best_a = a4.copy()
    
    # Final short FISTA polish
    if time.monotonic() < t_end - 10:
        a5, mc5 = _fista_optimize(best_a.copy(), t_end, K_start=5, K_end=1, step_size=0.25)
        try:
            s5 = float(evaluate_sequence([float(x) for x in a5]))
        except Exception:
            s5 = float('inf')
        print(f"Final polish: score={s5:.6f}")
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()
    
    # Ensure valid output
    out = np.clip(best_a, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 0.01 / n)
    
    try:
        final_score = float(evaluate_sequence([float(x) for x in out]))
    except Exception:
        final_score = float('inf')
    print(f"FINAL score: {final_score:.6f} (target: 1.5027)")
    
    return [float(x) for x in out]
