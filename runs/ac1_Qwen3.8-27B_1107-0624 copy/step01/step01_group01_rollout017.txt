def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n <= 128:
        return np.convolve(a, a).astype(np.float64)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def soft_max_full(c, beta):
    """Log-sum-exp soft max. Returns (value, weights)."""
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    mval = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-10
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    # Clip to avoid overflow
    x = np.maximum(x, -50.0)
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute grad component: h[i] = sum_j w[i+j] * p[j]."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]


def softmax_stable(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -40.0, 40.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / e.size
    return e / s

    """Clip to [0, 1000] and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:

def internal_score(a):
    """Fast score: 2*n*max(conv)/sum^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:

    s = float(a.sum())
    if s <= 1e-30:
        return np.inf

    mval = float(c.max())

def logit_and_score(z, beta):
    """Compute smooth-max score and gradient w.r.t. z from logit vector z."""
    p = softmax_stable(z)
    n = p.size
    c = full_convolve_fft(p)
    L, w = soft_max_full(c, beta)
    # Gradient w.r.t. p: dL/dp[i] = 2 * sum_k w[k] * (p[i-k] + p[i+k])
    # = 2 * corr_weights(w, p) [approximately, for the middle part]
    grad_p = 2.0 * corr_weights(w, p)
    # Gradient w.r.t. z: dL/dz[j] = sum_i (dL/dp[i]) * p[i] * (delta_ij - p[j])
    # = p[j] * (sum_i grad_p[i] * p[i] - grad_p[j] * ... )
    # Actually: dL/dz[j] = p[j] * (grad_p[j] - sum_i p[i] * grad_p[i])
    dot = float(np.dot(p, grad_p))

def cosine_lr(step, total_steps, lr_max, lr_min):
    """Cosine learning rate schedule."""
    if step >= total_steps:
        return lr_min
    return lr_min + 0.5 * (lr_max - lr_min) * (1.0 + np.cos(np.pi * step / total_steps))

        w[int(np.argmax(c))] = 1.0
        return mval, w
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Cross-correlation: h[i] = sum_j w[i+j]*p[j] for i=0..n-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1: 2 * n - 1]


def softmax_stable(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -45.0, 45.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(e) / e.size
    return e / s


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def power_profile(n, p_val, alpha):
    """Profile: a_j = (j/n + alpha)^(p_val), normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = (t + alpha) ** p_val
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def cosine_lr(step, total_steps, lr_max, lr_min):
    if step >= total_steps:
        return lr_min
    frac = float(step) / float(total_steps)
    return lr_min + 0.5 * (lr_max - lr_min) * (1.0 + np.cos(np.pi * frac))


def adam_run(p0, deadline, rng, lr_init, lr_min, beta_init_frac, beta_min_frac,
             reg, patience, noise_init, noise_decay):
    """Adam in logit space with cosine LR. Returns (best_p, best_score)."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    est_steps = 15000
    z = np.log(p0)
    z -= float(np.mean(z))
    b1, b2, eps_a = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t_adam = 0
    p = p0.copy()
    c = full_convolve_fft(p)
    max_c = float(c.max())
    beta = beta_init_frac * max_c
    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * max_c)
    no_imp = 0
    step = 0
    noise_scale = noise_init
    while time.monotonic() < deadline:
        L, w = soft_max_vals(c, beta)
        grad_p = 2.0 * corr_weights(w, p)
        grad_p = grad_p + reg * p
        dot = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - dot)
        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        t_adam += 1
        lr = cosine_lr(step, est_steps, lr_init, lr_min)
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps_a)
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            continue
        z = np.clip(z, -45.0, 45.0)
        z -= float(np.mean(z))
        p = softmax_stable(z)
        c = full_convolve_fft(p)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)
        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
        step += 1
        beta = max(beta_min_frac * max_c, beta * 0.9965)
        if no_imp > patience:
            z = best_z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -45.0, 45.0)
            z -= float(np.mean(z))
            p = softmax_stable(z)
            c = full_convolve_fft(p)
            max_c = float(c.max())
            beta = beta_init_frac * max_c
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            no_imp = 0
            noise_scale *= noise_decay
            step = 0
    return best_p, best_score


def scipy_obj(z, n, beta):
    p = softmax_stable(z)
    c = full_convolve_fft(p)
    L, _ = soft_max_vals(c, beta)
    return float(2.0 * n * L)


def scipy_grad(z, n, beta):
    p = softmax_stable(z)
    c = full_convolve_fft(p)
    L, w = soft_max_vals(c, beta)
    grad_p = 2.0 * corr_weights(w, p)
    dot = float(np.dot(p, grad_p))
    return p * (grad_p - dot)


def scipy_polish_run(p0, beta_frac, max_iter):
    """Polish using L-BFGS-B in logit space. Returns (best_p, best_score)."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size
    z0 = np.log(p0)
    z0 -= float(np.mean(z0))
    beta = beta_frac
    best_p = p0.copy()
    best_score = float(2.0 * n * float(full_convolve_fft(p0).max()))
    for _ in range(3):
        res = minimize(scipy_obj, z0, jac=lambda zz: scipy_grad(zz, n, beta),
                       args=(), method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                                'iprint': -1})
        p_new = softmax_stable(res.x)
        c = full_convolve_fft(p_new)
        sc = float(2.0 * n * float(c.max()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            z0 = res.x - float(np.mean(res.x))
        else:
            break
        beta *= 0.3
    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: Logit Adam with cosine LR on multiple initializations.
    Phase 2: scipy L-BFGS-B polish.
    Phase 3: Fine Adam with tight beta.
    Phase 4: Exact evaluation and return.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    best_seq = None
    best_score = np.inf
    targets_n = 512

    # Gather initial candidates
    candidates = []

    # Parent
    try:
        par_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if par_arr.ndim == 1 and par_arr.size > 0:
            par = normalize_nonneg(par_arr)
            if par is not None:
                candidates.append(par)
                sc = internal_score(par)
                if sc < best_score:
                    best_score = sc
                    best_seq = par.copy()
                print(f"Parent: score={sc:.10f}, n={par.size}")
    except Exception:
        pass

    # Analytic profiles at target length
    for eta in (1.5, 1.75, 2.0, 2.25, 2.5, 3.0):
        p = linear_profile(targets_n, eta)
        candidates.append(p)
        sc = internal_score(p)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()

    for pval in (2.0, 2.5, 3.0, 4.0):
        for alpha in (0.01, 0.05, 0.1, 0.2):
            p = power_profile(targets_n, pval, alpha)
            candidates.append(p)
            sc = internal_score(p)
            if sc < best_score:
                best_score = sc
                best_seq = p.copy()

    # Random smooth profiles
    t_arr = np.linspace(0.0, 1.0, targets_n)
    for _ in range(8):
        a0 = rng.uniform(0.5, 2.0)
        a1 = rng.uniform(-1.0, 3.0)
        a2 = rng.uniform(-0.5, 1.5)
        a3 = rng.uniform(-1.0, 1.0)
        prof = a0 + a1 * t_arr + a2 * t_arr**2 + a3 * t_arr**3
        prof = np.maximum(prof, 1e-12)
        prof = prof / prof.sum()
        candidates.append(prof)
        sc = internal_score(prof)
        if sc < best_score:
            best_score = sc
            best_seq = prof.copy()

    print(f"After init: best={best_score:.10f}")

    # Sort candidates by score, keep top 10
    scored = []
    for p in candidates:
        sc = internal_score(p)
        scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    top_candidates = [p for _, p in scored[:10]]

    # Phase 1: Adam optimization on top candidates (use ~55% of budget)
    phase1_end = t_start + 0.55 * (deadline - t_start)
    num_phase1 = min(6, len(top_candidates))
    per_time = max(30.0, (phase1_end - t_start) / max(1, num_phase1))

    for idx in range(num_phase1):
        if time.monotonic() > phase1_end:
            break
        p_init = top_candidates[idx]
        if p_init.size != targets_n:
            x_old = np.linspace(0, 1, p_init.size)
            x_new = np.linspace(0, 1, targets_n)
            p_init = normalize_nonneg(np.interp(x_new, x_old, p_init))
        if p_init is None:
            continue
        end_this = min(phase1_end, time.monotonic() + per_time)
        if end_this - time.monotonic() < 10.0:
            continue

        # Vary hyperparameters for diversity
        lr_i = 0.12 + 0.02 * (idx % 3)
        beta_i = 0.12 + 0.03 * (idx % 4)
        p_opt, sc_opt = adam_run(
            p_init, end_this, rng,
            lr_init=lr_i, lr_min=0.004,
            beta_init_frac=beta_i, beta_min_frac=0.002,
            reg=1e-7, patience=500, noise_init=0.4, noise_decay=0.85
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = p_opt.copy()
            print(f"Phase1 cand{idx}: {sc_opt:.10f} (improved)")
        else:
            print(f"Phase1 cand{idx}: {sc_opt:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: scipy L-BFGS-B polish on the best (~20% of remaining)
    if best_seq is not None and time.monotonic() < deadline - 180.0:
        phase2_end = min(deadline - 180.0, time.monotonic() + 0.20 * (deadline - time.monotonic()))
        p_cur = best_seq
        if p_cur.size != targets_n:
            x_old = np.linspace(0, 1, p_cur.size)
            x_new = np.linspace(0, 1, targets_n)
            p_cur = normalize_nonneg(np.interp(x_new, x_old, p_cur))
        if p_cur is not None:
            # Multiple scipy polishes with decreasing beta
            for beta_f in (0.02, 0.008, 0.003, 0.001):
                if time.monotonic() > phase2_end:
                    break
                p_pol, sc_pol = scipy_polish_run(p_cur, beta_f, max_iter=300)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = p_pol.copy()
                    p_cur = p_pol.copy()
                    print(f"Scipy beta={beta_f}: {sc_pol:.10f} (improved)")
                else:
                    print(f"Scipy beta={beta_f}: {sc_pol:.10f}")
                    break

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Fine Adam with very small beta on the best (~15% of remaining)
    if best_seq is not None and time.monotonic() < deadline - 80.0:
        phase3_end = min(deadline - 80.0, time.monotonic() + 0.15 * (deadline - time.monotonic()))
        p_fine = best_seq
        if p_fine.size != targets_n:
            x_old = np.linspace(0, 1, p_fine.size)
            x_new = np.linspace(0, 1, targets_n)
            p_fine = normalize_nonneg(np.interp(x_new, x_old, p_fine))
        if p_fine is not None:
            p_fo, sc_fo = adam_run(
                p_fine, phase3_end, rng,
                lr_init=0.04, lr_min=0.001,
                beta_init_frac=0.03, beta_min_frac=0.0005,
                reg=1e-8, patience=300, noise_init=0.15, noise_decay=0.8
            )
            if sc_fo < best_score:
                best_score = sc_fo
                best_seq = p_fo.copy()
                print(f"Phase3 fine Adam: {sc_fo:.10f} (improved)")
            else:
                print(f"Phase3 fine Adam: {sc_fo:.10f}")

    # Phase 4: One more scipy polish if time allows
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        p_final = best_seq
        if p_final.size != targets_n:
            x_old = np.linspace(0, 1, p_final.size)
            x_new = np.linspace(0, 1, targets_n)
            p_final = normalize_nonneg(np.interp(x_new, x_old, p_final))
        if p_final is not None:
            p_sp, sc_sp = scipy_polish_run(p_final, 0.001, max_iter=500)
            if sc_sp < best_score:
                best_score = sc_sp
                best_seq = p_sp.copy()
                print(f"Final scipy: {sc_sp:.10f} (improved)")

    # Final exact evaluation
    if best_seq is not None:
        if best_seq.size <= 4096:
            sc_exact = true_score(best_seq)
            if np.isfinite(sc_exact):
                print(f"Exact score: {sc_exact:.10f}")
                if sc_exact < best_score:
                    best_score = sc_exact
        try:
            sc_eval = float(evaluate_sequence(best_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"evaluate_sequence: {sc_eval:.10f}")
        except Exception:
            pass

    # Cleanup
    if best_seq is not None:
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(best_seq.sum())
        if s > 0:
            best_seq = best_seq / s
        print(f"FINAL: score={best_score:.10f}, n={best_seq.size}")
        return [float(x) for x in best_seq]

    # Fallback
    fb = linear_profile(256, 2.0)
    return [float(x) for x in fb]
