        conv = np.fft.ifft(a_fft * p_rev_fft).real
        # conv[l] = sum_m a[m] p_rev[l-m] = sum_m a[m] p[2n-2-(l-m)] 
        # We want corr[j] = conv[2n-2-j]
        idx = 2*n-2 - np.arange(n)
        g = 2.0 * conv[idx]
    
    return g, c


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def smooth_gradient_step(a, beta, alpha_candidates=None):
    """
    One smoothed gradient descent step with simplex projection.
    Returns the new sequence and the score improvement.
    """
    n = a.size
    c = np.convolve(a, a)
    cur_max = c.max()
    cur_score = 2.0 * n * cur_max
    
    g, c = compute_gradient(a, beta)
    if not np.all(np.isfinite(g)):
        return a, cur_score
    
    # Line search
    if alpha_candidates is None:
        alpha_candidates = [0.1, 0.05, 0.02, 0.01, 0.005]
    
    best_a = a
    best_score = cur_score
    for alpha in alpha_candidates:
        new_a = project_simplex(a - alpha * g)
        new_score = score_normalized(new_a)
        if new_score < best_score:
            best_score = new_score
            best_a = new_a
    
    return best_a, best_score


def solve_lp_escape(a, max_rows=None):
    """
    Solve the LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    where M = max(a*a).
    """
    if linprog is None:
        return None
    n = a.size
    c = np.convolve(a, a)
    M = c.max()
    if M < 1e-15:
        return None
    
    if max_rows is None:
        max_rows = 2*n - 1
    max_rows = min(max_rows, 2*n-1)
    
    # Select active rows (top max_rows by c value)
    if max_rows < 2*n-1:
        rows = np.argpartition(c, -max_rows)[-max_rows:]
    else:
        rows = np.arange(2*n-1)
    
    # Build constraint matrix: (a*g)_k = sum_j a[k-j] * g[j] (with a[m]=0 for m out of range)
    # For row k: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((max_rows, n))
    for idx, k in enumerate(rows):
        k = int(k)
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_lo = max(0, k - n + 1)
        j_hi = min(k + 1, n)
        if j_hi > j_lo:
            j_idx = np.arange(j_lo, j_hi)
            A[idx, j_lo:j_hi] = a[k - j_idx]
    
    # Normalize constraints: A g <= M  =>  (A/M) g <= 1
    A = A / M
    c_obj = -np.ones(n)
    b = np.ones(max_rows)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1.0 + 1e-6:
                return g / g.sum()  # normalize
    except:
        pass
    return None


def make_initial(N, kind, rng, best_seq=None, provided=None):
    """Create diverse initial sequences."""
    N = int(N)
    x = np.linspace(0, 1, N)
    
    if kind == -1 and best_seq is not None:
        # Resize best
        if best_seq.size != N:
            from scipy.interpolate import interp1d
            f = interp1d(np.linspace(0, 1, best_seq.size), best_seq, kind='cubic', fill_value='extrapolate')
            a = np.maximum(f(x), 0)
        else:
            a = best_seq.copy()
        return normalize_nonneg(a)
    
    if kind == -2 and provided is not None:
        if provided.size != N:
            from scipy.interpolate import interp1d
            f = interp1d(np.linspace(0, 1, provided.size), provided, kind='cubic', fill_value='extrapolate')
            a = np.maximum(f(x), 0)
        else:
            a = provided.copy()
        return normalize_nonneg(a)
    
    if kind == 0:  # Uniform
        a = np.ones(N)
    elif kind == 1:  # Two-block
        L = N // 2
        a = np.zeros(N)
        a[:L] = (2/3) / L
        a[L:] = (1/3) / (N - L)
    elif kind == 2:  # Raised cosine
        a = 0.5 * (1 + np.cos(2*np.pi*x))
        a = np.maximum(a, 1e-10)
    elif kind == 3:  # Asymmetric ramp
        a = np.linspace(0.1, 1.0, N)
    elif kind == 4:  # Gaussian
        mu = rng.uniform(0.2, 0.8)
        sigma = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5*((x-mu)/sigma)**2)
    elif kind == 5:  # Sine-modulated
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        a = 0.5 + 0.4*np.sin(2*np.pi*freq*x + phase)
        a = np.maximum(a, 0.01)
    elif kind == 6:  # Random smooth
        base = rng.normal(size=N)
        # Smooth
        w = np.ones(max(1, N//32))
        w /= w.sum()
        a = np.convolve(base, w, mode='same')
        a = np.exp(0.5 * a)
    elif kind == 7:  # Three-block asymmetric
        a = np.zeros(N)
        a[:N//4] = 0.5 / (N//4)
        a[N//4:3*N//4] = 0.3 / (3*N//4 - N//4)
        a[3*N//4:] = 0.2 / (N - 3*N//4)
    else:
        a = rng.random(N)
    
    return normalize_nonneg(np.maximum(a, 0))


def optimize_sequence(a, time_limit, rng, state):
    """Optimize a single sequence using smoothed gradient + LP escape."""
    a = normalize_nonneg(a).copy()
    n = a.size
    cur_score = score_normalized(a)
    best_a = a.copy()
    best_score = cur_score
    update_global(state, best_a, best_score)
    
    deadline = time.monotonic() + time_limit
    no_improve = 0
    beta = 50.0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = c.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            update_global(state, best_a, best_score)
            no_improve = 0
            beta = max(10.0, beta * 0.95)  # gradually increase sharpness
        else:
            no_improve += 1
            beta = min(200.0, beta * 1.05)  # increase smoothing to escape
        
        if no_improve > 100:
            # Try LP escape
            g = solve_lp_escape(a)
            if g is not None:
                # Line search on the LP direction
                best_t = 0
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    h = normalize_nonneg((1-t)*a + t*g)
                    s = score_normalized(h)
                    if s < cur_score:
                        best_t = t
                        cur_score = s
                if best_t > 0:
                    a = normalize_nonneg((1-best_t)*a + best_t*g)
                    no_improve = 0
                    continue
            else:
                # Random perturbation
                a = normalize_nonneg(a * np.exp(rng.normal(0, 0.1, n)))
                no_improve = 0
                continue
        
        # Smoothed gradient step
        g, _ = compute_gradient(a, beta)
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            a = normalize_nonneg(a * np.exp(rng.normal(0, 0.05, n)))
            continue
        
        # Try several step sizes
        base_alpha = 0.5 / max(g_norm, 0.1)
        alphas = [base_alpha * f for f in [1.0, 0.5, 0.25, 0.1, 0.05]]
        
        moved = False
        for alpha in alphas:
            new_a = project_simplex(a - alpha * g)
            new_score = score_normalized(new_a)
            if new_score < cur_score:
                a = new_a
                moved = True
                break
        
        if not moved:
            # Smaller perturbation
            a = normalize_nonneg(a * np.exp(rng.normal(0, 0.03, n)))
    
    final_score = score_normalized(best_a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    update_global(state, best_a, best_score)
    return best_a, best_score


def update_global(state, seq, score):
    if np.isfinite(score) and score < state["score"] - 1e-12:
        state["score"] = score
        state["seq"] = normalize_nonneg(seq).copy()
        print(f"  new best: score={score:.10f} n={seq.size}", flush=True)
