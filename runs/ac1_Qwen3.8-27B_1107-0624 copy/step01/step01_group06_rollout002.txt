    n = len(a)

    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def convolve_fft(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def subgradient_max(a):
    """Exact subgradient of J using the true max (average over tied peaks)."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    max_c = np.max(c)
    # Find active set (positions within numerical tolerance of max)
    tol = 1e-10 * max_c + 1e-15
    active = np.where(c >= max_c - tol)[0]

    # Compute subgradient: average of d(c_k)/d(a_j) over active k
    grad_c = np.zeros(n)
    for k in active:
        for j in range(n):
            kij = k - j
            if 0 <= kij < n:
                if k == 2 * j:
                    grad_c[j] += a[j]
                else:
                    grad_c[j] += 2.0 * a[kij]

    if len(active) > 0:
        grad_c /= len(active)

    J = 2.0 * n * max_c / (S * S)
    grad = (2.0 * n * grad_c) / (S * S) - (4.0 * n * max_c) / (S * S * S)
    return J, grad


def peak_peel(a, n_peaks=5, peel_frac=0.02):
    """Targeted reduction of top convolution peaks.
    For each peak position k*, find the dominant pair (i, k*-i) and
    reduce both, redistributing mass to low-contribution positions."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)

    # Get top peak positions (excluding near-boundary which are small)
    order = np.argsort(c)[::-1]
    top_k = order[:n_peaks]

    a_new = a.copy()
    for k in top_k:
        # Find dominant contributing pair
        best_prod = 0
        best_i = -1
        for i in range(n):
            kij = k - i
            if 0 <= kij < n and i != kij:
                prod = a_new[i] * a_new[kij]
                if prod > best_prod:
                    best_prod = prod
                    best_i = i
            elif 0 <= kij < n and i == kij:
                prod = a_new[i] ** 2
                if prod > best_prod:
                    best_prod = prod
                    best_i = i

        if best_i < 0:
            continue

        kij = k - best_i
        if best_i == kij:
            # a[j]^2 term
            reduction = a_new[best_i] * peel_frac
            a_new[best_i] -= reduction
            removed = reduction
        else:
            # 2*a[i]*a[kij] term
            reduction_i = a_new[best_i] * peel_frac * 0.5
            reduction_j = a_new[kij] * peel_frac * 0.5
            a_new[best_i] -= reduction_i
            a_new[kij] -= reduction_j
            removed = reduction_i + reduction_j

        # Redistribute removed mass to positions with low convolution contribution
        if removed > 0:
            # Add to random positions that are "far" from the peak
            n_add = min(10, n // 10)
            # Prefer positions near the edges (less contribution to center peaks)
            edge_positions = list(range(0, max(1, n // 4))) + list(range(max(1, 3 * n // 4), n))
            chosen = np.random.choice(edge_positions, size=n_add, replace=False)
            for pos in chosen:
                a_new[pos] += removed / n_add

    a_new = np.maximum(a_new, 0.0)
    return a_new


def spectral_init(n, rng):
    """Create sequence with approximately flat power spectrum."""
    size = 1
    while size < 2 * n:
        size <<= 1

    A = np.zeros(size // 2 + 1, dtype=complex)
    freqs = np.arange(size // 2 + 1) / size
    # Band-limit
    mask = (freqs > 0.02) & (freqs < 0.48)
    A[mask] = np.exp(1j * rng.uniform(0, 2 * np.pi, mask.sum()))
    A[0] = 0  # Remove DC to make mean zero

    a = np.fft.irfft(A, size)[:n]
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)
    else:
        a = np.ones(n)
    return a


def chirp_init(n):
    """Quadratic phase (chirp) - has flat autocorrelation property."""
    x = np.arange(n)
    phase = np.pi * x * x / n
    a = np.cos(phase)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)
    else:
        a = np.ones(n)
    return a


def structured_init(n, freq):
    """Multi-frequency cosine structure."""
    x = np.arange(n)
    a = np.ones(n)
    a += 0.3 * np.cos(2 * np.pi * freq * x / n)
    a += 0.2 * np.cos(2 * np.pi * 2 * freq * x / n + 0.5)
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S > 0.01:
        a = a * (n / S)
    return a

        b2 = min(phase2_budget, total_deadline - time.monotonic() - 25.0)
        if b2 > 10:
            res_a, res_J = run_gradient_descent(
                scored[1][2], b2, seed=seed + 1000,
                tau_start=5.0, tau_end=0.01
            )
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global_score = val
                best_global = res_a.copy()
                print(f"  Phase 2 result: {val:.8f}")

    # Phase 3: Coordinate polish
    remaining = total_deadline - time.monotonic() - 10.0
    if remaining > 15:
        print(f"\n=== Phase 3: Coordinate polish ({remaining:.0f}s) ===")
        res_a, res_J = coordinate_polish(best_global, remaining, seed=seed + 2000)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global_score = val
            best_global = res_a.copy()
            print(f"  Phase 3 result: {val:.8f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val) and final_val < best_global_score:
        best_global_score = final_val

    print(f"\nFinal score: {best_global_score:.8f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
