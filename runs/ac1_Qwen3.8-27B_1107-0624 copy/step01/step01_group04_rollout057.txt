def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def greedy_normalized(n):
    """Coefficients of (1-z)^{-1/2}, normalized to sum 1."""
    n = int(n)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    s = a.sum()
    return a / s


def power_normalized(n, s):
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + 1.0, -s)
    return a / a.sum()


def actual_score(a):
    """Compute f(a) = 2*n*max(conv(a,a)) / sum(a)^2"""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = a.sum()
    if n == 0 or s < 1e-10:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    mc = c.max()
    return float(2.0 * n * mc / (s * s))


def convolve_fft(a):

    n = a.size
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    return np.fft.irfft(P * P, L)[:2*n-1]

def coordinate_descent_polish(a, budget_s):

    given all others fixed (by solving a 1D problem).

    s = a.sum()
    if s <= 1e-10:

    a /= s
    
    n = a.size

    best_score = actual_score(a)

    # c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[m] = a[k-m] (if valid) + a[m] * (if m appears in a[k-m]... wait)

    # If we change a[m], it affects c[k] for all k where j=m or k-j=m
    # i.e., k = m + j for any j (first term) or k = m + j where k-j=m, i.e., j = k-m
    # So c[k] = a[m]*a[k-m] + a[k-m]*a[m] + rest = 2*a[m]*a[k-m] + rest (if m != k-m)
    #          = a[m]^2 + rest (if m = k-m, i.e., k = 2m)
    
    # For a fixed m, c[k] as a function of a[m]:
    # c[k] = a[m]*a[k-m] + a[k-m]*a[m] + (terms not involving a[m])
    #      = 2*a[m]*a[k-m] + c0[k]  (if m != k-m)
    #      = a[m]^2 + c0[k]  (if k = 2m)
    
    # The max is the max over k of these quadratic/linear functions.
    # To minimize the max, we binary search on the target value T and check
    # if there exists a[m] in [0, (1-a[m])/sum(others)] such that all c[k] <= T.
    
    # This is complex. Let me use a simpler approach: 
    # For each coordinate m, try a few values and keep the best.
    
    n_iter = 0
    convergence = 0
    step = 0.01

        improved = False

                # Also try the value that would make the peak at the "worst" position equal to the next peak
            ]

            best_sc = np.inf
            
            # Compute current max
            c = convolve_fft(a)
            c = np.maximum(c, 0.0)
            current_max = c.max()
            
            for cv in candidates:
                if cv < 0 or not np.isfinite(cv):
                    continue
                a[m] = cv
                s = a.sum()
                if s <= 1e-10:
                    a[m] = old_val
                    continue
                # Quick score check
                c_new = convolve_fft(a)
                c_new = np.maximum(c_new, 0.0)
                sc = 2.0 * n * c_new.max()
                if sc < best_sc:
                    best_sc = sc

            a[m] = best_val

def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()

        s = float(a.sum())

"""
Search for non-negative step-function sequences minimizing
f(seq) = 2*n*max(convolve(seq,seq)) / sum(seq)^2
using fast projected subgradient descent in log-space (mirror descent)
with exact max, multi-length exploration, and adaptive step scheduling.
Starts from the previous best (height_sequence_1) and refines aggressively.
"""

import time
import numpy as np


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def greedy_normalized(n):
    """Coefficients of (1-z)^{-1/2}, normalized to sum 1."""
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    s = a.sum()
    if s <= 0:
        return np.ones(n) / n
    return a / s


def power_normalized(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return a / a.sum()


def mix_normalized(n, lam, rng=None):
    g = greedy_normalized(n)
    if rng is not None:
        noise = np.exp(rng.normal(0, 0.1, n))
        g = g * noise
    u = np.ones(n, dtype=np.float64) / n
    a = lam * g + (1.0 - lam) * u
    return a / a.sum()


def actual_score(a):
    """Compute f(a) = 2*n*max(conv(a,a)) / sum(a)^2"""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(a.sum())
    if n == 0 or s < 1e-10:
        return np.inf
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def convolve_fft(a):
    """Full self-convolution via FFT."""
    n = a.size
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    return np.fft.irfft(P * P, L)[:2 * n - 1]


def mirror_descent(a_init, budget_s, alpha_mode='adaptive', lr0=0.08, lr_decay=0.99997):
    """
    Mirror descent (exponential weights / log-space gradient) on the simplex.
    Uses the exact max of the convolution (subgradient).
    Optionally uses log-sum-exp smoothing with increasing alpha.
    """
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-10:
        return a, np.inf
    a /= s

    n = a.size
    L = next_power_of_two(2 * n - 1)

    # Work in log-space
    log_a = np.log(a + 1e-300)
    log_a -= log_a.max()

    best_a = a.copy()
    best_score = 2.0 * n * convolve_fft(a).max()  # since sum=1

    start = time.monotonic()
    end = start + budget_s
    lr = lr0
    it = 0
    stall = 0
    stall_threshold = 500 if n < 500 else 1000

    while time.monotonic() < end:
        elapsed = (time.monotonic() - start) / max(1e-9, budget_s)

        # Reconstruct a from log_a
        la = log_a - log_a.max()
        e = np.exp(la)
        se = float(e.sum())
        if se <= 0 or not np.isfinite(se):
            break
        a = e / se

        # Compute convolution
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)

        mc = float(c.max())
        if mc <= 0 or not np.isfinite(mc):
            break

        cur_score = 2.0 * n * mc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_threshold:
            # Restart with perturbation
            log_a = np.log(np.clip(a, 1e-15, None))
            log_a += 0.1 * np.random.randn(n)
            log_a -= log_a.max()
            stall = 0
            lr = max(lr, lr0 * 0.5)
            continue

        # Compute subgradient / smoothed gradient
        if alpha_mode == 'adaptive':
            alpha = 5.0 + 50.0 * elapsed * elapsed
        else:
            alpha = 1.0  # exact max

        if alpha > 1.0:
            # Softmax smoothing: weights on near-max positions
            z = alpha * (c / mc - 1.0)
            z = np.clip(z, -80.0, 0.0)
            ez = np.exp(z)
            sz = float(ez.sum())
            if sz <= 0 or not np.isfinite(sz):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ez / sz
        else:
            # Exact: pick argmax (or a few near-max)
            near_max = c >= 0.9999 * mc
            idx = np.where(near_max)[0]
            if len(idx) > 20:
                idx = np.argsort(c)[::-1][:20]
            w = np.zeros(2 * n - 1)
            w[idx] = 1.0 / len(idx)

        # Gradient: grad[m] = sum_k w[k] * a[k-m] for valid indices
        # = cross-correlation of w and a
        L2 = next_power_of_two(3 * n - 2)
        FFT_w = np.fft.rfft(w, L2)
        FFT_a2 = np.fft.rfft(a, L2)
        cross = np.fft.irfft(FFT_w * np.conj(FFT_a2), L2)
        grad = cross[:n]
        grad = np.maximum(grad, 0.0)

        # Update in log-space
        log_a -= lr * grad
        log_a -= log_a.max()

        # Clip for numerical stability
        log_a = np.clip(log_a, -50.0, 5.0)

        lr *= lr_decay

        it += 1
        if it % 2000 == 0:
            sc = actual_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if it % 10000 == 0:
                print(f"    MD iter {it}, score={best_score:.6f}, lr={lr:.6f}")

    # Final
    la = log_a - log_a.max()
    e = np.exp(la)
    se = float(e.sum())
    if se > 0 and np.isfinite(se):
        a_final = e / se
        sc = actual_score(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final

    return best_a, best_score


def subgradient_projected(a_init, budget_s, lr0=0.05, lr_decay=0.99999):
    """
    Projected subgradient descent: a <- clip(a - lr*g, 0), then renormalize.
    Uses exact max (subgradient at argmax).
    """
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-10:
        return a, np.inf
    a /= s

    n = a.size
    L = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_score = 2.0 * n * float(convolve_fft(a).max())

    start = time.monotonic()
    end = start + budget_s
    lr = lr0
    it = 0
    stall = 0

    while time.monotonic() < end:
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        mc = float(c.max())
        if mc <= 0:
            break

        cur_score = 2.0 * n * mc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            break

        # Subgradient at argmax
        k_star = int(np.argmax(c))
        g = np.zeros(n, dtype=np.float64)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        if lo <= hi:
            g[lo:hi + 1] = a[k_star - hi: k_star - lo + 1][::-1]

        # Step
        a_new = a - lr * g
        a_new = np.clip(a_new, 0.0, None)
        sn = float(a_new.sum())
        if sn > 1e-10:
            a = a_new / sn
        else:
            break

        lr *= lr_decay
        it += 1

    sc = actual_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def multi_peak_subgradient(a_init, budget_s, lr0=0.04, n_peaks=5):
    """
    Subgradient descent that averages over the top-n_peaks convolution positions.
    This helps when multiple peaks are close to the maximum.
    """
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-10:
        return a, np.inf
    a /= s

    n = a.size
    L = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_score = 2.0 * n * float(convolve_fft(a).max())

    start = time.monotonic()
    end = start + budget_s
    lr = lr0
    it = 0
    stall = 0

    while time.monotonic() < end:
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        mc = float(c.max())
        if mc <= 0:
            break

        cur_score = 2.0 * n * mc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            break

        # Top-k peaks
        k_peaks = min(n_peaks, 2 * n - 1)
        peak_indices = np.argsort(c)[::-1][:k_peaks]
        # Select peaks that are within 99.5% of max
        peak_indices = peak_indices[c[peak_indices] > 0.995 * mc]
        if len(peak_indices) == 0:
            peak_indices = np.array([int(np.argmax(c))])

        # Average subgradient over selected peaks
        g = np.zeros(n, dtype=np.float64)
        for k in peak_indices:
            k = int(k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += a[k - hi: k - lo + 1][::-1]
        g /= len(peak_indices)

        # Step
        a_new = a - lr * g
        a_new = np.clip(a_new, 0.0, None)
        sn = float(a_new.sum())
        if sn > 1e-10:
            a = a_new / sn
        else:
            break

        lr *= 0.999995
        it += 1

    sc = actual_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def coordinate_polish(a_init, budget_s, n_trials_per_coord=4):
    """
    Coordinate descent: for each coordinate, try several values and keep the best.
    Uses FFT for fast evaluation.
    """
    a = np.asarray(a_init, dtype=np.float64).ravel().copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-10:
        return a
    a /= s

    n = a.size
    L = next_power_of_two(2 * n - 1)
    best_a = a.copy()
    best_score = 2.0 * n * float(convolve_fft(a).max())

    start = time.monotonic()
    stall_rounds = 0

    while time.monotonic() < start + budget_s:
        improved = False
        for m in range(n):
            if time.monotonic() > start + budget_s:
                break

            old_val = a[m]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [1e-5, 1e-4, old_val + 1e-5]
            else:
                candidates = [
                    old_val * 0.3,
                    old_val * 0.6,
                    old_val * 1.4,
                    old_val * 0.1,
                ]

            best_val = old_val
            best_c = np.inf

            for cv in candidates:
                cv = max(0.0, cv)
                a[m] = cv
                sn = a.sum()
                if sn <= 1e-10:
                    a[m] = old_val
                    continue
                # Quick score (sum changes, so use unscaled)
                P = np.fft.rfft(a, L)
                c = np.fft.irfft(P * P, L)[:2 * n - 1]
                c = np.maximum(c, 0.0)
                sc = 2.0 * n * c.max() / (sn * sn)
                if sc < best_c:
                    best_c = sc
                    best_val = cv

            a[m] = best_val
            a = np.clip(a, 0.0, None)
            sn = a.sum()
            if sn > 1e-10:
                a /= sn
            else:
                a[m] = old_val
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            if best_val != old_val:
                improved = True

        # Check overall improvement
        sc = actual_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall_rounds = 0
        else:
            stall_rounds += 1

        if stall_rounds > 3:
            break

    return best_a


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(16, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]

    # Phase 4: More mirror descent with different params
    print("Phase 4: Extended mirror descent")
    sorted_items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in sorted_items[:8]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        alloc = min(80.0, max(15.0, rem * 0.1))
        # Try with higher initial lr
        p_out, f_out = mirror_descent(p, alloc, alpha_mode='adaptive', lr0=0.12, lr_decay=0.99995)
        record(p_out, f_out)
        # Try with lower lr for finer convergence
        p_out2, f_out2 = mirror_descent(p_out, min(30.0, alloc * 0.5), alpha_mode='adaptive', lr0=0.02, lr_decay=0.999999)
        record(p_out2, f_out2)

    print(f"  After extended MD: best={best_score:.6f}")

    # Phase 5: Random restarts
    print("Phase 5: Random restarts")
    restart_count = 0
    while time.monotonic() < deadline - 120:
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        # Pick length near best
        best_n = min(best_by_n.keys(), key=lambda k: best_by_n[k][0]) if best_by_n else 384
        n_try = int(best_n + rng.integers(-32, 33))
        n_try = max(192, min(800, n_try))

        # Various initializations
        choice = rng.integers(0, 5)
        if choice == 0:
            p_init = greedy_normalized(n_try)
        elif choice == 1:
            p_init = power_normalized(n_try, float(rng.uniform(0.35, 0.65)))
        elif choice == 2:
            base = best_by_n.get(best_n, (np.inf, greedy_normalized(best_n)))[1]
            if n_try == best_n:
                perturbed = base + rng.normal(0, 0.08, n_try)
                p_init = np.clip(perturbed, 0, None)
                p_init = p_init / (p_init.sum() + 1e-30)
            else:
                # Interpolate to new length
                p_int = np.interp(
                    np.linspace(0, 1, n_try),
                    np.linspace(0, 1, best_n),
                    base
                )
                p_int += rng.normal(0, 0.03, n_try)
                p_init = np.clip(p_int, 0, None)
                p_init = p_init / (p_init.sum() + 1e-30)
        elif choice == 3:
            p_init = mix_normalized(n_try, float(rng.uniform(0.3, 0.8)), rng)
        else:
            # Sparse initialization: few large values
            p_init = np.zeros(n_try)
            n_ones = int(n_try * rng.uniform(0.3, 0.7))
            positions = rng.choice(n_try, n_ones, replace=False)
            p_init[positions] = 1.0
            p_init *= np.exp(rng.normal(0, 0.5, n_try))
            p_init = np.clip(p_init, 0, None)
            p_init = p_init / (p_init.sum() + 1e-30)

        # Optimize
        alloc = min(35.0, max(10.0, rem * 0.06))
        p_out, f_out = mirror_descent(p_init, alloc, alpha_mode='adaptive', lr0=0.08, lr_decay=0.99997)
        record(p_out, f_out)

        alloc2 = min(20.0, max(5.0, rem * 0.03))
        p_out2, f_out2 = mirror_descent(p_out, alloc2, alpha_mode='adaptive', lr0=0.03, lr_decay=0.999999)
        record(p_out2, f_out2)

        restart_count += 1
        if restart_count % 5 == 0:
            print(f"    Restart {restart_count}, best={best_score:.6f}, time_left={rem:.0f}s")

    print(f"  After restarts: best={best_score:.6f}")

    # Phase 6: Coordinate polish on the best
    print("Phase 6: Coordinate polish")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            alloc = min(rem - 20, 60.0)
            p_out = coordinate_polish(best_p, alloc)
            record(p_out)

    # Phase 7: Final fine-grained mirror descent
    print("Phase 7: Final optimization")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(rem - 15, 50.0)
            p_out, f_out = mirror_descent(best_p, alloc, alpha_mode='adaptive', lr0=0.015, lr_decay=0.9999999)
            record(p_out, f_out)

        rem = deadline - time.monotonic()
        if rem > 20:
            # Try a slightly perturbed version
            p_pert = best_p + rng.normal(0, 0.02, best_p.size)
            p_pert = np.clip(p_pert, 0, None)
            p_pert = p_pert / p_pert.sum()
            alloc = min(rem - 10, 30.0)
            p_out, f_out = mirror_descent(p_pert, alloc, alpha_mode='adaptive', lr0=0.01, lr_decay=0.9999999)
            record(p_out, f_out)

    # Final validation
    print(f"=== Final best: {best_score:.6f} (n={best_p.size if best_p is not None else 0}) ===")

    if best_p is None:
        best_p = greedy_normalized(384)

    # Validate with the global evaluator
    final_seq = scale_output(best_p, 10.0)
    val = try_eval(final_seq)
    if np.isfinite(val):
        print(f"Validated score: {val:.6f}")
    else:
        print("Primary validation failed, trying sum=1")
        final_seq = scale_output(best_p, 1.0)
        val2 = try_eval(final_seq)
        if np.isfinite(val2):
            print(f"Validated score (sum=1): {val2:.6f}")
        else:
            print("WARNING: all validations failed")

    # Also validate top few candidates and pick the best
    if len(best_by_n) > 1 and time.monotonic() < deadline:
        items = sorted(best_by_n.items(), key=lambda x: x[1][0])
        for n, (sc, p) in items[:3]:
            if time.monotonic() > deadline - 5:
                break
            seq = scale_output(p, 10.0)
            v = try_eval(seq)
            if np.isfinite(v) and v < best_score:
                best_score = v
                best_p = p
                final_seq = seq
                print(f"  Updated from validation: n={n}, score={v:.6f}")

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation function."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print(f"Critical error: {e}")
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
