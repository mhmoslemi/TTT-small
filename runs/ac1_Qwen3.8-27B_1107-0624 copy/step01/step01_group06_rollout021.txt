import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_fft(a, b=None):
    """Compute convolution of a with b (or a with a if b is None) using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]

def compute_J(a):
    """Compute J = 2n * max(conv) / sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Cross-correlation for gradient
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad

def solve_lp_best_response(a, M):
    """Solve the LP: maximize sum(b) subject to b >= 0 and (a*b)[k] <= M for all k.
    
    Returns the optimal b.
    """
    n = len(a)
    
    # Build the constraint matrix A (shape: (2n-1) x n)
    # A[k, i] = a[i] if 0 <= k-i < n, else 0
    # Equivalently: A[k, i] = a[k - (n-1) + j] ... let me think more carefully.
    # 
    # (a*b)[k] = sum_i a[i] * b[k-i] where 0<=i<n, 0<=k-i<n
    # = sum_i a[i] * b[k-i] where max(0, k-n+1) <= i <= min(k, n-1)
    #
    # So A[k, i] = a[i] if max(0, k-n+1) <= i <= min(k, n-1), else 0
    #
    # Let me build this as a sparse matrix.
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M (all constraints are upper bounds)
    # b >= 0 (handled by bounds)
    
    result = linprog(
        c=c_obj,
        A_ub=A,
        b_ub=np.full(2 * n - 1, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'dual_feasibility_tolerance': 1e-8, 'primal_feasibility_tolerance': 1e-8}
    )
    
    if result.success:
        return result.x
    else:
        return None

def frank_wolfe_iteration(a, a_current, M, t):
    """One Frank-Wolfe iteration."""
    b = solve_lp_best_response(a, M)
    if b is None:
        return a_current
    # Rescale b to have same sum as a_current
    S_b = np.sum(b)
    S_a = np.sum(a_current)
    if S_b < 1e-10:
        return a_current
    b = b * (S_a / S_b)
    return (1 - t) * a_current + t * b

This is a 1D piecewise-linear (or rational) optimization problem that can be solved by checking the breakpoints.

"""
Hybrid optimizer for step function coefficients minimizing
J(a) = 2n * max(convolve(a,a)) / (sum(a))^2.

Uses:
1. FFT-based gradient descent with log-sum-exp smoothing (Adam)
2. LP-based Frank-Wolfe directions with reduced constraint sets
3. Multiple diverse initializations and adaptive time allocation
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def convolve_fft(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_J(a):
    """Compute J = 2n * max(conv) / sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_J_and_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = np.clip((c - c_max) / tau, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def solve_reduced_lp(a, M, K=500):
    """Solve reduced LP: max sum(b) s.t. (a*b)[k] <= M for top-K positions, b >= 0."""
    n = len(a)
    c = convolve_fft(a)
    
    # Find top K positions
    if K >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:K]
    
    # Build constraint matrix for these K positions
    rows = []
    cols = []
    vals = []
    
    for k in idx:
        i_start = max(0, int(k) - n + 1)
        i_end = min(int(k), n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(int(k))
            cols.append(i)
            vals.append(a[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    try:
        result = linprog(c=c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_cosine(n, freq=1.0):
    x = np.arange(n)
    return 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * x / n)


def make_edge_boost(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    denom = max(mid, 1.0)
    return 0.3 + 0.7 * (np.abs(x - mid) / denom) ** 2


def make_taper(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 1.0 - 0.5 * np.abs(x - mid) / max(mid, 1.0)


def make_quadratic(n):
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return 0.5 + 0.5 * (1.0 - (x - mid) / max(mid, 1.0)) ** 2


def run_gd_with_lp(a_init, budget_s, seed, tau_start=5.0, tau_end=0.05,
                   lr_start=0.01, lp_every=200, lp_K=500, lp_weight=0.1):
    """Gradient descent with periodic LP-based Frank-Wolfe corrections."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy().astype(np.float64)
    # Normalize to sum = sqrt(2n)
    S = np.sum(a)
    if S > 0:
        a = a * (np.sqrt(2.0 * n) / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    stall_count = 0
    stall_best = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_start / (1.0 + 0.3 * t / 500.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        # Normalize to maintain sum = sqrt(2n)
        S = np.sum(a)
        if S > 0.01:
            a = a * (np.sqrt(2.0 * n) / S)

        # Periodic LP correction
        if t % lp_every == 0 and budget_s > 30:
            current_J = compute_J(a)
            M = np.max(convolve_fft(a))
            b = solve_reduced_lp(a, M, K=lp_K)
            if b is not None:
                S_b = np.sum(b)
                S_a = np.sum(a)
                if S_b > 1e-10:
                    b_norm = b * (S_a / S_b)
                    a = (1 - lp_weight) * a + lp_weight * b_norm
                    a = np.maximum(a, 0.0)
                    # Reset optimizer state
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0

        # Check improvement periodically
        if t % 500 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_best = local_J
            else:
                stall_count += 1
                if stall_count > 8:
                    # Perturbation
                    cand = a.copy()
                    w = max(1, n // 100)
                    s = int(rng.integers(0, n - w))
                    cand[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 0.0)
                    cJ = compute_J(cand)
                    if cJ < stall_best:
                        a = cand
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall_count = 0
                    else:
                        a = best_a.copy()
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall_count = 0

    return best_a, best_J

def convolve_fft(a, b=None):
    """Fast convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:n + m - 1]


def compute_J(a):
    """Compute J = 2n * max(conv) / sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)
