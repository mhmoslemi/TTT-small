        _update_best(p, hard)
        u = alpha * (b - cmax)
        e = np.exp(u - np.max(u))
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            return hard, np.zeros(n)
        w = e / s
        L = _fft_len(2 * n)
        wf = np.zeros(L)
        wf[:lb] = w
        pf = np.zeros(L)
        pf[:n] = p
        W = np.fft.rfft(wf)
        P = np.fft.rfft(pf)
        cross = np.fft.irfft(np.conj(W) * P, L)[:n]
        g = 2.0 * n * 2.0 * cross
        g = g - np.mean(g)
        g = np.where(np.isfinite(g), g, 0.0)
        return hard + float(np.log(s)) / alpha, g

    try:
        res = minimize(
            _obj, a0.copy(), method="L-BFGS-B", jac=True,
            bounds=[(0.0, None)] * n,
            options={"maxfun": max_fun, "maxiter": max_fun // 3,
                     "ftol": 1e-15, "gtol": 1e-10, "maxls": 20}
        )
        p = _project_simplex(res.x)
        sc = _score(p)
        _update_best(p, sc)
    except _Timeout:
        pass
    except Exception:
        pass


def _init_uniform(n):
    return np.ones(n) / n


def _init_tukey(n, r):
    x = np.linspace(0, 1, n)
    w = np.ones(n)
    if r <= 0:
        return w / n
    edge = r / 2.0
    tl = (x < edge)
    tr = (x > 1.0 - edge)
    xm = (x - edge) / r
    w[tl] = 0.5 + 0.5 * np.cos(np.pi * xm[tl])
    xr = (1.0 - x - edge) / r
    w[tr] = 0.5 + 0.5 * np.cos(np.pi * xr[tr])
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def _init_hann(n):
    x = np.linspace(0, 1, n)
    w = 0.5 + 0.5 * np.cos(2.0 * np.pi * x)
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s if s > 0 else np.ones(n) / n


def _init_raised_cos(n):
    x = np.linspace(0, 1, n)
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    w = np.maximum(w, 1e-15)
    return w / w.sum()


def _init_gaussian(n, sigma):
    x = (np.arange(n) - (n - 1) / 2.0) / float(n)
    w = np.exp(-x ** 2 / (2 * sigma * sigma))
    return w / w.sum()


def _init_bump(n, center, width):
    x = np.arange(n) / float(n)
    w = np.exp(-((x - center) / width) ** 2)
    return w / w.sum()


def _init_two_bumps(n, c1, c2, w1, w2):
    x = np.arange(n) / float(n)
    w = w1 * np.exp(-((x - c1) / 0.05) ** 2) + w2 * np.exp(-((x - c2) / 0.05) ** 2)
    w = np.maximum(w, 0)
    s = w.sum()
    return w / s if s > 0 else np.ones(n) / n


def _init_asymmetric(n, beta):
    x = np.linspace(0, 1, n)
    w = np.exp(beta * x)
    return w / w.sum()


def _init_poly(n, rng):
    x = np.linspace(0, 1, n)
    deg = int(rng.integers(2, 7))
    c = rng.normal(0, 1, deg + 1)
    w = np.ones(n)
    for k in range(deg, -1, -1):
        w = w * (x - 0.5) + c[k]
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _init_random_smooth(n, rng):
    freqs = rng.choice([1, 2, 3, 4, 5, 6, 8], size=int(rng.integers(2, 5)))
    amps = rng.normal(0, 1, len(freqs))
    phs = rng.uniform(0, 2 * np.pi, len(freqs))
    x = np.linspace(0, 1, n)
    w = np.ones(n)
    for f, a, p in zip(freqs, amps, phs):
        w += a * np.sin(2 * np.pi * f * x + p)
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _resample(seq, n):
    old_n = len(seq)
    if old_n == n:
        return np.array(seq, dtype=np.float64)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    return np.interp(new_x, old_x, np.asarray(seq, dtype=np.float64))


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_seq, _deadline

    start = time.monotonic()
    _deadline = start + budget_s - 8.0
    _best_score = float("inf")
    _best_seq = None
    rng = np.random.default_rng(seed)

    # Try to load previous best
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)
        prev_seq = np.where(np.isfinite(prev_seq), prev_seq, 0.0)
        prev_seq = np.maximum(prev_seq, 0)
        if prev_seq.sum() < 0.01:
            prev_seq = None
    except Exception:
        prev_seq = None

    if prev_seq is not None and len(prev_seq) > 0:
        pn = len(prev_seq)
        pnorm = prev_seq / prev_seq.sum()
        _update_best(pnorm, _score(pnorm))
        print(f"Initial best from height_sequence_1: {_best_score:.6f} (n={pn})")

    # Define n values to try (sorted by priority)
    n_values = [4096, 2048, 8192, 1024, 512]
    if budget_s > 600:
        n_values.append(16384)

    # Time allocation: give more time to n=4096 and n=2048
    # Total available time
    total_time = _deadline - start
    # Allocate proportionally
    weights = {4096: 3.0, 2048: 2.0, 8192: 1.5, 1024: 1.0, 512: 1.0, 16384: 1.0}
    wsum = sum(weights.get(nv, 1.0) for nv in n_values)

    elapsed = 0
    for n in n_values:
        time_left = _deadline - time.monotonic()
        if time_left < 20:
            break
        alloc = total_time * weights.get(n, 1.0) / wsum
        alloc = min(alloc, time_left - 10)
        if alloc < 15:
            alloc = min(15, time_left - 5)
        if alloc < 5:
            continue

        phase_deadline = time.monotonic() + alloc
        print(f"\n=== n={n}, time_alloc={alloc:.0f}s ===")

        # Build initializations
        inits = []
        labels = []

        # Uniform
        inits.append(_init_uniform(n))
        labels.append("uniform")

        # Tukey windows
        for r in [0.3, 0.5, 0.7, 0.9, 1.0]:
            inits.append(_init_tukey(n, r))
            labels.append(f"tukey_r{r}")

        # Gaussian
        for sig in [0.15, 0.2, 0.3, 0.4]:
            inits.append(_init_gaussian(n, sig))
            labels.append(f"gauss_{sig}")

        # Asymmetric
        for beta in [1.0, 2.0, 3.0, 5.0]:
            inits.append(_init_asymmetric(n, beta))
            inits.append(_init_asymmetric(n, -beta))
            labels.append(f"asym_+{beta}")
            labels.append(f"asym_-{beta}")

        # Two bumps
        inits.append(_init_two_bumps(n, 0.3, 0.7, 1.0, 1.0))
        labels.append("two_bumps_30_70")

        # If we have previous best, resample it
        if prev_seq is not None and len(prev_seq) > 10:
            resampled = _resample(prev_seq, n)
            resampled = np.maximum(resampled, 0)
            s = resampled.sum()
            if s > 0:
                inits.append(resampled / s)
                labels.append("prev_resampled")
            # Also try perturbed versions
            for _ in range(3):
                pert = resampled + rng.normal(0, 0.02, n)
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 0:
                    inits.append(pert / ps)
                    labels.append("prev_pert")

        # Random inits (if enough time)
        if alloc > 60:
            for _ in range(8):
                inits.append(_init_poly(n, rng))
                labels.append("poly_rand")
            for _ in range(4):
                inits.append(_init_random_smooth(n, rng))
                labels.append("smooth_rand")

        # Score all inits and sort
        scores = np.array([_score(a) for a in inits])
        for i, sc in enumerate(scores):
            _update_best(inits[i], sc)
        order = np.argsort(scores)

        # Adaptive topk based on n
        if n <= 1024:
            topk_base = 30
        elif n <= 4096:
            topk_base = 20
        else:
            topk_base = 15

        # Optimize top candidates
        n_opt = min(len(inits), max(3, int(alloc / 15)))
        opt_deadline = phase_deadline

        for rank in range(n_opt):
            if rank >= len(order):
                break
            if time.monotonic() > opt_deadline - 5:
                break
            idx = int(order[rank])
            a_init = inits[idx]
            label = labels[idx] if idx < len(labels) else f"init_{idx}"
            rem = opt_deadline - time.monotonic()
            if rem < 5:
                break

            # Determine max_iter based on remaining time
            # Each iteration costs ~2 FFTs of size ~2n, roughly 0.1ms for n=4096
            # So for 30s, we can do ~300000 iterations (very optimistic)
            # More realistically ~5000-20000 iterations per opt
            if rem > 30:
                max_iter = 8000
            elif rem > 10:
                max_iter = 4000
            else:
                max_iter = 1500

            print(f"  Opt {rank+1}/{n_opt}: {label} (init_score={scores[idx]:.6f}, max_iter={max_iter})")
            _opt_grad(a_init, topk_base, max_iter, opt_deadline, verbose=False, label=label)

            # If we've beaten the initial score significantly, also try L-BFGS
            if time.monotonic() < opt_deadline - 30 and _best_score < 1.55:
                a_best_local = a_init.copy()
                # Use current best as starting point for L-BFGS
                if _best_seq is not None and len(_best_seq) == n:
                    a_best_local = _best_seq.copy()
                max_fun = max(100, int((opt_deadline - time.monotonic()) * 50))
                alpha = 5e6 if _best_score < 1.52 else 1e6
                print(f"  L-BFGS polish: alpha={alpha:.2e}, max_fun={max_fun}")
                _opt_smooth(_best_local if 'best_local' in dir() else a_best_local, alpha, max_fun, opt_deadline)

        # Additional random restarts from best if time remains
        rem = phase_deadline - time.monotonic()
        if rem > 20 and _best_seq is not None and len(_best_seq) == n:
            n_restarts = min(5, int(rem / 20))
            for ri in range(n_restarts):
                if time.monotonic() > phase_deadline - 5:
                    break
                noise_scale = 0.05 + 0.1 * ri
                a_restart = _best_seq + rng.normal(0, noise_scale, n)
                a_restart = _project_simplex(a_restart)
                rem2 = phase_deadline - time.monotonic()
                mi = min(3000, max(500, int(rem2 * 50)))
                _opt_grad(a_restart, topk_base, mi, phase_deadline, verbose=False,
                          label=f"restart_{ri}")

        print(f"  After n={n}: best={_best_score:.6f}")

    # Final: if we have a best, try to polish with the evaluate_sequence
    if _best_seq is not None:
        seq_out = [float(v) for v in _best_seq]
        try:
            actual_score = float(evaluate_sequence(seq_out))
            print(f"Actual score from evaluate_sequence: {actual_score:.6f}")
            if not np.isfinite(actual_score) or actual_score > 2.0:
                # Fallback to uniform
                seq_out = [1.0 / len(seq_out)] * len(seq_out)
                print("Fallback to uniform")
        except Exception:
            seq_out = [1.0 / len(_best_seq)] * len(_best_seq)

        # Compare with previous best if available
        if prev_seq is not None and len(prev_seq) > 0:
            try:
                prev_out = [float(max(0, min(1000, v))) for v in prev_seq]
                prev_sc = float(evaluate_sequence(prev_out))
                if np.isfinite(prev_sc) and prev_sc < actual_score:
                    seq_out = prev_out
                    actual_score = prev_sc
                    print(f"Keeping previous best: {actual_score:.6f}")
            except Exception:
                pass
    else:
        # Fallback: uniform of length 4096
        seq_out = [1.0 / 4096] * 4096
        actual_score = 2.0

    try:
        final_sc = float(evaluate_sequence(seq_out))
        if not np.isfinite(final_sc) or final_sc > 2.0:
            seq_out = [1.0 / 4096] * 4096
            final_sc = 2.0
    except Exception:
        seq_out = [1.0 / 4096] * 4096
        final_sc = 2.0

    print(f"\nFinal: score={final_sc:.6f}, len={len(seq_out)}")
    return seq_out
