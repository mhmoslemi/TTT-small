- n linear inequalities: b[j] >= 0

    from scipy.optimize import linprog
    c_obj = -np.ones(N)
    b_ub = M * np.ones(2*N-1)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    
    if result.success:
        b = result.x
        return b
