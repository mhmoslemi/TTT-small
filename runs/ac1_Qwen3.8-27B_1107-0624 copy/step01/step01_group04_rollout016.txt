```python
"""
Minimizes f(seq) = 2*n*max(convolve(seq,seq))/sum(seq)^2 using:
1. Adam gradient descent with smoothly-weighted max-convolution objective
2. LP cutting-plane for principled direction finding (improved constraint management)
3. Local perturbation-based search to escape local minima
4. Multi-length exploration with warm-starting from previous best
"""

import time
import numpy as np
import scipy.optimize as scipy_opt


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def actual_score_fft(p, direct_limit=1024):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    if n <= direct_limit:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    if c.size == 0:
        return np.inf
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def compute_conv(a, b):
    la, lb = len(a), len(b)
    L = next_power_of_two(la + lb - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:la + lb - 1].real


def adam_optimize(p, budget_s, seed=0):
    """Softmax-projected Adam for smoothed max-convolution objective."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.04
    elif n < 256:
        lr = 0.025
    elif n < 1024:
        lr = 0.015
    else:
        lr = 0.01

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 0.5)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 3.0 + 60.0 * np.exp(-4.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c - maxc)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W_rev = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W_rev, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)
        grad_p = (4.0 * n) * C

        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -40.0, 40.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_improve(p, budget_s):
    """
    LP cutting-plane: maximize sum(g) s.t. (f*g)(k) <= M, g >= 0.
    f is normalized to sum = sqrt(2n). M = max(f*f).
    Uses peak-first constraint initialization and aggressive violation handling.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 2:
        return p, actual_score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 15.0:
        return p, actual_score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)
    end = time.monotonic() + max(0.0, budget_s - 3.0)

    # Initial constraint set: peak positions + uniform spread
    fg = compute_conv(f, f)
    fg = np.maximum(fg, 0.0)
    n_init = min(2 * n - 1, max(100, 2 * n // 3))
    peak_idx = np.argsort(fg)[::-1][:n_init]
    rows = [int(k) for k in peak_idx.tolist()]
    row_set = set(rows)
    # Add uniform spread
    uniform = np.linspace(0, 2 * n - 2, max(32, n // 4), dtype=int)
    for k in uniform:
        if int(k) not in row_set:
            rows.append(int(k))
            row_set.add(int(k))

    max_rows = min(2 * n - 1, 6 * n)

    it = 0
    while time.monotonic() < end:
        it += 1
        rem = end - time.monotonic()
        if rem < 5.0:
            break

        rows_arr = np.asarray(rows, dtype=int)
        nr = rows_arr.size
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        tl = max(2.0, min(12.0, rem * 0.3))
        res = None
        try:
            res = scipy_opt.linprog(cobj, A_ub=A, b_ub=b,
                                    bounds=(0.0, None), method='highs',
                                    options={'time_limit': float(tl), 'presolve': True})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check all constraints
        conv = compute_conv(f, g)
        conv = np.maximum(conv, 0.0)
        M = cur_score
        violations = conv - M
        tol = 1e-7 * max(1.0, M)
        max_viol = float(violations.max())

        if max_viol > tol:
            # Add violated constraints
            n_added = 0
            viols = np.where(violations > tol * 0.1)[0]
            # Sort by violation magnitude
            viols = viols[np.argsort(violations[viols])[::-1]]
            for ix in viols:
                if n_added >= 512:
                    break
                if len(rows) >= max_rows:
                    break
                ix = int(ix)
                if ix not in row_set:
                    rows.append(ix)
                    row_set.add(ix)
                    n_added += 1
            if n_added == 0 and max_viol > 10.0 * tol:
                # Cannot add more, try to proceed anyway with current solution
                pass
            continue

        # LP converged for current constraint set - update f
        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score):
            break
        if accepted_score >= best_score - 1e-10:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

        # Reset constraint set to peak positions of new f*f
        fg_new = compute_conv(f, f)
        fg_new = np.maximum(fg_new, 0.0)
        new_peaks = np.argsort(fg_new)[::-1][:n_init]
        rows = [int(k) for k in new_peaks.tolist()]
        row_set = set(rows)
        uniform2 = np.linspace(0, 2 * n - 2, max(32, n // 4), dtype=int)
        for k in uniform2:
            if int(k) not in row_set:
                rows.append(int(k))
                row_set.add(int(k))

        if it % 5 == 0:
            print("  LP iter=%d score=%.8f rows=%d" % (it, best_score, len(rows)))

    final_score = actual_score_fft(best_f)
    if final_score < best_score:
        best_score = final_score
    return normalize_nonneg(best_f), best_score


def local_perturbation_search(p, budget_s, rng):
    """Try random perturbations of p and keep improvements."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    best = normalize_nonneg(p)
    best_score = actual_score_fft(best)
    end = time.monotonic() + budget_s

    while time.monotonic() < end:
        # Try different perturbation scales
        sigma = rng.uniform(0.02, 0.15)
        noise = np.exp(rng.normal(0.0, sigma, n))
        cand = normalize_nonneg(best * noise)
        sc = actual_score_fft(cand)
        if sc < best_score - 1e-8:
            best = cand
            best_score = sc

    return best, best_score

        p_out, f_out = lp_improve(best_p, min(rem - 10.0, 200.0))
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("After final LP: best_score=%.8f" % best_score)

    # Validate with global evaluator
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 10.0:
            seq = scale_output(best_p, 10.0)
            gscore = try_global_eval(seq)
            print("Global eval of best: %.8f" % gscore)
            if np.isfinite(gscore) and gscore < best_score:
                best_score = gscore

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score_fft(best_p)

    print("FINAL: best_score=%.8f, n=%d" % (best_score, len(best_p)))
    try:
        return scale_output(best_p, 10.0)
    except Exception:
        return scale_output(greedy_normalized(384), 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
```
