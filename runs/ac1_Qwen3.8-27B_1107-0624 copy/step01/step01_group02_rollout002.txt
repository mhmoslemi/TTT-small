    except Exception:
        return None
    try:
        a = np.asarray(raw, dtype=np.float64)
    except Exception:
        return None
    if a is None or a.size == 0:
        return None
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.clip(a, 0.0, None)
    if float(a.sum()) <= 1e-300:
        return None
    return normalize_nonneg(a)


def build_full_matrix(f, n_lags=None):
    """
    Build the full convolution matrix A where row k has A[k,j] = f[k-j]
    whenever 0 <= k-j < len(f), else 0.
    Returns A with shape (2n-1, n) or subset of rows.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    total_lags = 2 * n - 1
    if n_lags is None or n_lags >= total_lags:
        rows = np.arange(total_lags, dtype=np.int64)
    else:
        # Use all rows (we want full constraints)
        rows = np.arange(total_lags, dtype=np.int64)

    A = np.zeros((rows.size, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            idx = jlo + np.arange(jhi - jlo, dtype=np.int64)
            A[r, jlo:jhi] = f[k - idx]
    return A, rows


def solve_lp_direction_full(f, M, max_time=2.0):
    """
    Solve the full LP: maximize sum(g) subject to A g <= M, g >= 0.
    Uses all 2n-1 constraints.
    """
    if linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n > 2048:
        return None
    if M <= 1e-300:
        return None

    t0 = time.monotonic()
    try:
        A, rows = build_full_matrix(f)
    except Exception:
        return None

    if A.size == 0 or not np.any(A):
        return None

    A_norm = A / M
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(A_norm.shape[0], dtype=np.float64)

    try:
        res = linprog(
            cvec,
            A_ub=A_norm,
            b_ub=bvec,
            bounds=(0.0, None),
            method="highs",
            options={"time_limit": max_time, "presolve": True},
        )
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) > 1e-12:
            return g
    return None


def solve_lp_direction_subset(f, M, top_k, max_time=1.0):
    """Solve LP with only the top_k most active constraints."""
    if linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if M <= 1e-300:
        return None

    c = np.convolve(f, f)
    total_lags = c.size
    k = min(top_k, total_lags)
    if k == total_lags:
        rows = np.arange(total_lags, dtype=np.int64)
    else:
        rows = np.argpartition(c, -k)[-k:].astype(np.int64)

    try:
        A = np.zeros((k, n), dtype=np.float64)
        for r, ki in enumerate(rows):
            ki = int(ki)
            jlo = max(0, ki - n + 1)
            jhi = min(n, ki + 1)
            if jhi > jlo:
                idx = jlo + np.arange(jhi - jlo, dtype=np.int64)
                A[r, jlo:jhi] = f[ki - idx]
    except Exception:
        return None

    if not np.any(A):
        return None

    A_norm = A / M
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(k, dtype=np.float64)

    try:
        res = linprog(
            cvec, A_ub=A_norm, b_ub=bvec,
            bounds=(0.0, None), method="highs",
            options={"time_limit": max_time, "presolve": True},
        )
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) > 1e-12:
            return g
    return None


def subgradient_step(f, c, M, temp_frac, cur_score):
    """
    Smooth-max subgradient step for the normalized sequence f.
    Returns an improved sequence or None.
    """
    f = np.asarray(f, dtype=np.float64)
    c = np.asarray(c, dtype=np.float64)
    n = f.size
    if M <= 1e-300:
        return None

    temp = max(1e-15, float(temp_frac) * M)
    e = (c - M) / temp
    e = np.clip(e, -60.0, 0.0)
    p = np.exp(e)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p = p / s

    # Compute r[j] = sum_i f[i] * p[i+j] using FFT
    need = n + c.size - 1
    size = 1
    while size < need:
        size <<= 1

    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    pb = np.zeros(size, dtype=np.float64)
    pb[:c.size] = p[::-1]

    out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(pb, size), size)
    idx = (c.size - 1) - np.arange(n, dtype=np.int64)
    r = out[idx]

    d = r - np.mean(r)
    scale = float(np.max(np.abs(d)))
    if not np.isfinite(scale) or scale <= 1e-300:
        return None

    alphas = (0.03, 0.08, 0.15, 0.3, 0.5, 0.8)
    best_h = None
    best_sc = cur_score
    for alpha in alphas:
        h = f - alpha * scale * d
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h
    return best_h


def compute_subgradient_direct(f, active_weights):
    """
    Compute the exact subgradient of max_k (f*f)_k at f,
    given weights on the active lags.
    The subgradient w.r.t. f is: 2 * sum_k w_k * (shifted f)_k
    where (shifted f)_k[i] = f[k-i] if valid.
    """
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    c = np.asarray(active_weights, dtype=np.float64)
    total = c.size

    # r[j] = sum_k c[k] * f[k-j] for valid k-j
    # = sum_k c[k] * f[k-j]
    # This is a cross-correlation of c (reversed) with f
    need = n + total - 1
    size = 1
    while size < need:
        size <<= 1

    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = f
    ca = np.zeros(size, dtype=np.float64)
    ca[:total] = c[::-1]

    out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(ca, size), size)
    idx = (total - 1) - np.arange(n, dtype=np.int64)
    return 2.0 * out[idx]


def random_perturb(f, rng, scale=0.1):
    """Random multiplicative perturbation."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    h = np.maximum(f, 0.0).copy()
    if n > 1:
        h = h * np.exp(rng.normal(0.0, float(scale), size=n))
        # Block perturbation
        w1 = max(1, n // 200)
        w2 = max(2, n // 8)
        width = int(rng.integers(w1, w2 + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        h[start:start + width] *= np.exp(float(scale) * 2.0 * rng.normal(0.0, 1.0))
    return normalize_nonneg(h)


def block_swap(f, rng):
    """Swap two random blocks of the sequence."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n < 8:
        return f.copy()
    h = f.copy()
    block_len = int(rng.integers(max(2, n // 30), n // 4 + 1))
    i = int(rng.integers(0, max(1, n - block_len)))
    j = int(rng.integers(0, max(1, n - block_len)))
    if abs(i - j) < block_len:
        return f.copy()
    h[i:i + block_len], h[j:j + block_len] = h[j:j + block_len].copy(), h[i:i + block_len].copy()
    return normalize_nonneg(h)


def reverse_seq(f):
    """Reverse the sequence."""
    return normalize_nonneg(np.asarray(f, dtype=np.float64)[::-1].copy())


def circular_shift(f, rng):
    """Apply a random circular shift."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    shift = int(rng.integers(1, n))
    return normalize_nonneg(np.roll(f, shift))


def update_global(state, seq, score):
    """Update the search state if score is finite and better."""
    if not np.isfinite(score):
        return False
    seq = np.asarray(seq, dtype=np.float64)
    if seq.size == 0:
        return False
    old = state["score"]
    if score < old - 1e-12:
        state["score"] = float(score)
        state["seq"] = normalize_nonneg(seq).copy()
        if not np.isfinite(old) or score < old - 1e-4:
            print(f"  NEW BEST: score={state['score']:.10f} n={state['seq'].size}", flush=True)
        return True
    return False


def make_two_block(N, rng):
    """Create a two-block initial sequence."""
    ratio = float(rng.uniform(0.6, 1.4))
    L = int(N / (1.0 + ratio))
    L = max(1, min(N - 1, L))
    a = np.zeros(N, dtype=np.float64)
    a[:L] = 1.0 / L
    a[L:] = ratio / (N - L)
    return normalize_nonneg(a)


def make_smooth_bump(N, rng):
    """Create a smooth bump initial sequence."""
    x = np.linspace(0.0, 1.0, N)
    mu = float(rng.uniform(0.3, 0.7))
    sigma = float(rng.uniform(0.15, 0.35))
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    return normalize_nonneg(a)


def make_ramp(N, rng):
    """Create a ramp initial sequence."""
    x = np.linspace(0.0, 1.0, N)
    alpha = float(rng.uniform(0.5, 3.0))
    direction = rng.integers(0, 2)
    if direction == 0:
        a = x ** alpha
    else:
        a = (1.0 - x) ** alpha
    return normalize_nonneg(np.maximum(a, 1e-10))


def make_random_smooth(N, rng):
    """Create a random smooth sequence."""
    base = rng.normal(0.0, 1.0, size=N)
    wlen = max(1, min(N, 1 + N // 50))
    if wlen > 1:
        w = np.ones(wlen, dtype=np.float64)
        w /= w.sum()
        base = np.convolve(base, w, mode="same")
    a = np.exp(0.8 * base)
    return normalize_nonneg(a)


def local_search(f, time_limit, rng, state, use_full_lp=True):
    """
    Run a local search trajectory combining LP, subgradient, and escape.
    """
    f = normalize_nonneg(f)
    n = f.size
    if n < 2:
        return f, score_normalized(f)

    c = compute_conv(f)
    M = float(c.max())
    cur_score = float(2.0 * n * M)
    best_score = cur_score
    best = f.copy()
    update_global(state, best, best_score)

    deadline = time.monotonic() + max(0.5, float(time_limit))
    it = 0
    no_improve = 0
    temp_frac = 0.20
    lp_count = 0
    subgrad_count = 0

    while time.monotonic() < deadline:
        it += 1
        c = compute_conv(f)
        M = float(c.max())
        cur_score = float(2.0 * n * M)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = f.copy()
            update_global(state, best, best_score)
            no_improve = 0
            temp_frac = max(0.02, temp_frac * 0.96)
        else:
            no_improve += 1

        # Escape if stuck
        if no_improve > 25:
            escape_choice = rng.integers(0, 5)
            if escape_choice == 0:
                f = random_perturb(f, rng, 0.15)
            elif escape_choice == 1:
                f = block_swap(f, rng)
            elif escape_choice == 2:
                f = reverse_seq(f)
            elif escape_choice == 3:
                f = circular_shift(f, rng)
            else:
                f = random_perturb(f, rng, 0.30)
            temp_frac = min(0.6, temp_frac * 1.3)
            no_improve = 0
            continue

        moved = False

        # LP step (most powerful)
        lp_interval = 1 if n <= 512 else 3
        if (it % lp_interval == 0) and linprog is not None and n <= 2048:
            t0 = time.monotonic()
            g = None

            if use_full_lp and n <= 1024:
                g = solve_lp_direction_full(f, M, max_time=3.0)
            if g is None:
                top_k = min(2 * n - 1, max(100, n))
                g = solve_lp_direction_subset(f, M, top_k, max_time=2.0)

            if g is not None:
                lp_count += 1
                Sg = float(g.sum())
                if Sg > 1.0 + 1e-6:
                    g = g / Sg
                    best_h = f
                    best_cur = cur_score

                    # Adaptive step sizes
                    if Sg > 5.0:
                        ts = (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4)
                    elif Sg > 2.0:
                        ts = (0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.7)
                    else:
                        ts = (0.05, 0.1, 0.15, 0.25, 0.4, 0.6, 0.8, 1.0)

                    for t in ts:
                        h = (1.0 - float(t)) * f + float(t) * g
                        h = normalize_nonneg(h)
                        sc = score_normalized(h)
                        if sc < best_cur:
                            best_cur = sc
                            best_h = h

                    if best_cur < cur_score - 1e-14:
                        f = best_h
                        moved = True

        # Subgradient step
        if not moved and time.monotonic() < deadline:
            h = subgradient_step(f, c, M, temp_frac, cur_score)
            if h is not None:
                subgrad_count += 1
                f = h
                moved = True

        # Fallback: small random step
        if not moved:
            f = random_perturb(f, rng, 0.04)

        # Safety: check time
        if time.monotonic() > deadline - 0.5:
            break

    # Final evaluation
    c = compute_conv(f)
    final_sc = float(2.0 * f.size * float(c.max()))
    if final_sc < best_score:
        best_score = final_sc
        best = f.copy()
    update_global(state, best, best_score)

    if it > 10:
        print(f"  Local search: n={n}, iters={it}, lp={lp_count}, subgrad={subgrad_count}, "
              f"best={best_score:.10f}", flush=True)

    return best, best_score


def coordinate_polish(f, time_limit, rng):
    """
    Fine-grained coordinate descent polish.
    For each coordinate (or small group), try to reduce the score.
    """
    f = normalize_nonneg(f)
    n = f.size
    best = f.copy()
    best_score = score_normalized(best)
    deadline = time.monotonic() + max(1.0, float(time_limit))

    # Group coordinates into small blocks for efficiency
    group_size = max(1, min(4, n // 64))
    n_groups = (n + group_size - 1) // group_size

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        improved_this_pass = False

        for g_idx in range(n_groups):
            if time.monotonic() >= deadline:
                break
            lo = g_idx * group_size
            hi = min(n, lo + group_size)
            gsize = hi - lo

            # Try small perturbations of this group
            for attempt in range(3):
                pert = rng.normal(0.0, 0.02, size=gsize)
                h = f.copy()
                h[lo:hi] = np.maximum(h[lo:hi] * np.exp(pert), 0.0)
                h = normalize_nonneg(h)
                sc = score_normalized(h)
                if sc < best_score - 1e-14:
                    best = h.copy()
                    best_score = sc
                    f = h
                    improved_this_pass = True
                    break

        if not improved_this_pass and iteration > 5:
            # Try larger step
            for lo in range(0, n, max(1, n // 20)):
                hi = min(n, lo + max(1, n // 20))
                h = f.copy()
                h[lo:hi] *= np.exp(rng.normal(0.0, 0.1, size=hi - lo))
                h = normalize_nonneg(h)
                sc = score_normalized(h)
                if sc < best_score - 1e-14:
                    best = h.copy()
                    best_score = sc
                    f = h
                    improved_this_pass = True
                    break

        if not improved_this_pass:
            break

    update_global_state = {"score": float("inf"), "seq": None}
    # Return best
    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for low-score step-function heights using a multi-strategy approach:
    1. Load best known sequence and resize to candidate lengths.
    2. Run local search with full LP direction finding (all 2n-1 constraints).
    3. Use subgradient method as fallback with adaptive temperature.
    4. Employ diverse escape mechanisms (block swap, reversal, shift, random).
    5. Apply coordinate polish for final refinement.
    6. Try multiple sequence lengths and return the best.
    """
    rng = np.random.default_rng(seed)
    hard_deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    state = {"score": float("inf"), "seq": None}

    # Load provided sequence
    provided = load_provided_sequence()
    provided_score = float("inf")
    if provided is not None:
        provided_score = score_normalized(provided)
        print(f"Provided sequence: n={provided.size}, score={provided_score:.10f}", flush=True)
        update_global(state, provided, provided_score)

    # Guaranteed fallback
    fallback = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
    update_global(state, fallback, 16.0 / 9.0)

    # Determine which lengths to try
    lengths_to_try = [512, 256, 1024, 768, 384]

    # Allocate time: prioritise the best-known length and nearby sizes
    base_time = 120.0
    # First, run on the provided length (if reasonable)
    if provided is not None and 128 <= provided.size <= 2048:
        lengths_to_try.insert(0, provided.size)

    # Deduplicate
    seen = set()
    deduped = []
    for L in lengths_to_try:
        if L not in seen and L >= 64:
            seen.add(L)
            deduped.append(L)
    lengths_to_try = deduped[:5]

    print(f"Lengths to try: {lengths_to_try}", flush=True)

    # Time budget allocation
    total_budget = hard_deadline - time.monotonic()
    # Give more time to shorter sequences (faster iterations) and the provided length
    time_alloc = {}
    for i, L in enumerate(lengths_to_try):
        if L == (provided.size if provided is not None else -1):
            time_alloc[L] = 0.35 * total_budget
        elif i == 0:
            time_alloc[L] = 0.25 * total_budget
        elif i == 1:
            time_alloc[L] = 0.20 * total_budget
        else:
            time_alloc[L] = max(20.0, 0.08 * total_budget)

    # Normalize times to fit budget
    total_alloc = sum(time_alloc.values())
    if total_alloc > total_budget:
        scale = total_budget / total_alloc
        time_alloc = {k: v * scale for k, v in time_alloc.items()}

    # Run searches for each length
    for L in lengths_to_try:
        if time.monotonic() >= hard_deadline - 60.0:
            break

        t_alloc = min(time_alloc.get(L, 60.0), hard_deadline - time.monotonic() - 40.0)
        if t_alloc < 5.0:
            continue

        print(f"\n=== Searching length n={L} (budget {t_alloc:.0f}s) ===", flush=True)
        t0 = time.monotonic()

        # Create initial candidates for this length
        candidates = []

        # Primary: resize provided sequence
        if provided is not None:
            candidates.append(("provided", resize_to_length(provided, L)))

        # Secondary: two-block
        candidates.append(("twoblock", make_two_block(L, rng)))
        # Tertiary: smooth bump
        candidates.append(("bump", make_smooth_bump(L, rng)))
        # Quaternary: ramp
        candidates.append(("ramp", make_ramp(L, rng)))
        # Quinary: random smooth
        candidates.append(("randsmooth", make_random_smooth(L, rng)))

        # Sort by initial score
        scored = []
        for name, cand in candidates:
            sc = score_normalized(cand)
            scored.append((sc, name, cand))
        scored.sort(key=lambda x: x[0])

        # Run local search on top candidates
        per_cand_time = t_alloc / min(3, len(scored))
        for i, (sc, name, cand) in enumerate(scored[:min(3, len(scored))]):
            if time.monotonic() >= hard_deadline - 40.0:
                break
            remaining = hard_deadline - time.monotonic() - 40.0
            cand_time = min(per_cand_time, remaining)
            if cand_time < 3.0:
                break

            print(f"  Candidate '{name}' (initial score={sc:.6f}), time={cand_time:.0f}s", flush=True)
            local_search(cand, cand_time, rng, state, use_full_lp=(L <= 1024))

        # Coordinate polish on the best found so far
        if state["seq"] is not None and time.monotonic() < hard_deadline - 20.0:
            # Resize best to this length if different
            polish_seq = resize_to_length(state["seq"], L) if state["seq"].size != L else state["seq"].copy()
            polish_time = min(30.0, hard_deadline - time.monotonic() - 15.0)
            if polish_time > 5.0:
                print(f"  Polish at n={L} for {polish_time:.0f}s", flush=True)
                local_search(polish_seq, polish_time, rng, state, use_full_lp=(L <= 1024))

        elapsed = time.monotonic() - t0
        print(f"  Done n={L} in {elapsed:.0f}s, current best: {state['score']:.10f}", flush=True)

    # Final polish phase on the overall best
    if state["seq"] is not None and time.monotonic() < hard_deadline - 10.0:
        best_n = state["seq"].size
        polish_time = min(60.0, hard_deadline - time.monotonic() - 10.0)
        if polish_time > 5.0:
            print(f"\n=== Final polish at n={best_n} for {polish_time:.0f}s ===", flush=True)
            local_search(state["seq"].copy(), polish_time, rng, state, use_full_lp=True)

    # Verify with official evaluator
    if state["seq"] is None:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]

    cand = [float(x) for x in np.asarray(state["seq"], dtype=np.float64)]

    try:
        official = evaluate_sequence(cand)
        if np.isfinite(official):
            print(f"\nFinal official score: {official:.10f}, n={len(cand)}", flush=True)
        else:
            print(f"\nWARNING: Official evaluation returned {official}", flush=True)
    except Exception as e:
        print(f"\nEvaluation error: {e}", flush=True)

    return cand
