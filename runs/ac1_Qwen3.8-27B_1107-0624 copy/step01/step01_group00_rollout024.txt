def _next_pow2(m):

        p <<= 1

def _conv_sq(a, m):
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[: (2 * a.size - 1)]


def _proj_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    cond = sv * np.arange(1, n + 1) > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.zeros(n)
    rho = idx[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)

def _smooth_objective_and_grad(a, m1, m2, tau):
    """
    Compute the smoothed objective f(a) = tau * log(sum exp(c_k/tau))
    and its gradient, where c = a*a (linear convolution).
    m1: FFT length for convolution (>= 2n-1)
    m2: FFT length for gradient (>= 3n-2)
    """
    n = a.size
    c = _conv_sq(a, m1)
    # Log-sum-exp for numerical stability
    c_max = c.max()
    exp_c = np.exp((c - c_max) / tau)
    log_z = c_max / tau + np.log(exp_c.sum())

    w = exp_c / exp_c.sum()

    A = np.fft.rfft(a, m2)
    xcorr = np.fft.irfft(W * np.conj(A), m2)
    grad = 2.0 * xcorr[:n]
    return f, grad, c

    """Run FISTA with temperature annealing on seed a0 until t_end."""
    n = a0.size
    m1 = _next_pow2(2 * n - 1)
    m2 = _next_pow2(3 * n - 2)
    
    a = a0.copy()
    a_prev = a.copy()
    buf = np.zeros(n)  # momentum buffer
    
    best_a = a.copy()
    best_f = _smooth_objective_and_grad(a, m1, m2, tau_start)[0]
    best_mc = _conv_sq(a, m1).max()
    
    tau = tau_start
    lr = lr0
    step = 0
    total_steps = 50000  # approximate
    momentum = momentum_sched[0]
    
    while time.monotonic() < t_end - 0.5:
        progress = min(1.0, step / total_steps)
        # Temperature annealing: exponential
        tau = tau_start * (tau_end / tau_start) ** progress
        # Learning rate decay
        lr = lr0 * (1.0 / (1.0 + 0.01 * step))
        # Momentum schedule
        momentum = momentum_sched[0] + (momentum_sched[1] - momentum_sched[0]) * progress
        
        # FISTA extrapolation
        if step > 0:
            t_k = (1 + math.sqrt(1 + 4 * step * step)) / 2
            y = a + ((t_k - 1) / t_k) * (a - a_prev)
        else:
            y = a.copy()
        
        # Compute gradient at y (projected to feasible set)
        y = _proj_simplex(y, 1.0)
        f_y, grad_y, c_y = _smooth_objective_and_grad(y, m1, m2, tau)
        
        # Backtracking line search
        a_new = y - lr * grad_y
        a_new = _proj_simplex(a_new, 1.0)
        
        # Check if we made progress (on the smoothed objective)
        f_new, _, _ = _smooth_objective_and_grad(a_new, m1, m2, tau)
        backtrack = 0
        while f_new > f_y + 1e-10 * np.abs(f_y) and backtrack < 10:
            lr *= 0.5
            a_new = y - lr * grad_y
            a_new = _proj_simplex(a_new, 1.0)
            f_new, _, _ = _smooth_objective_and_grad(a_new, m1, m2, tau)
            backtrack += 1
        
        a_prev = a
        a = a_new
        
        # Track best by max convolution coefficient (the actual objective)
        mc = c_y.max()  # This is approximate; we use c at y, not at a
        # Actually let's compute max at a
        c_a = _conv_sq(a, m1)
        mc_a = c_a.max()
        if mc_a < best_mc:
            best_mc = mc_a
            best_a = a.copy()
        
        step += 1
        
        if step % 5000 == 0 and step < 20000:
            print(f"  step={step}, tau={tau:.4f}, mc={mc_a:.8f}, lr={lr:.6f}")
    
    return best_a, best_mc


def _make_seeds(n, base_seq, rng):

    seeds = []

    # 10. 1/x type
    inv = 1.0 / (x + 1)
    seeds.append(_to_sum1(inv))
    
    return seeds

Minimize 2*n*max(conv(a,a))/(sum(a))^2 via temperature-annealed FISTA with
full log-sum-exp smoothing, backtracking line search, diverse seed strategies,
and multi-phase diversification. Improves on top-L softmax by using a globally
smooth surrogate, and adds structured perturbations + coarse-to-fine
temperature schedules to escape local minima.
"""
import time
import math
import numpy as np


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _conv_sq(a, m):
    """Linear self-convolution via rfft."""
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[: 2 * a.size - 1]


def _proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _to_sum1(a):
    s = a.sum()
    if s <= 1e-15:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _smooth_obj_grad(a, m1, m2, tau):
    """Return (sm_obj, gradient, convolution) for f = tau*log(sum exp(c/tau))."""
    n = a.size
    c = _conv_sq(a, m1)
    cmax = c.max()
    ec = np.exp((c - cmax) / tau)
    logz = cmax / tau + math.log(ec.sum())
    f = tau * logz
    w = ec / ec.sum()
    Wf = np.fft.rfft(w, m2)
    Af = np.fft.rfft(a, m2)
    xcorr = np.fft.irfft(Wf * np.conj(Af), m2)
    grad = 2.0 * xcorr[:n]
    return f, grad, c


def _fista_run(a0, t_end, tau0, tau1, lr0, mom0, mom1, total_est, verbose_tag):
    """FISTA with exponential temperature annealing; returns best (a, mc)."""
    n = a0.size
    m1 = _next_pow2(2 * n - 1)
    m2 = _next_pow2(3 * n - 2)
    a = a0.copy()
    a_prev = a.copy()
    best_a = a.copy()
    c = _conv_sq(a, m1)
    best_mc = c.max()
    step = 0
    while time.monotonic() < t_end - 0.3:
        prog = min(1.0, step / total_est)
        tau = tau0 * (tau1 / tau0) ** prog
        lr = lr0 / (1.0 + 0.005 * step)
        mom = mom0 + (mom1 - mom0) * prog
        if step >= 1:
            tk = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * step * step))
            y = a + ((tk - 1.0) / tk) * (a - a_prev)
        else:
            y = a.copy()
        y = _proj_simplex(y, 1.0)
        _, gy, _ = _smooth_obj_grad(y, m1, m2, tau)
        gn = np.linalg.norm(gy)
        if gn > 1e-12:
            gy = gy / gn
        a_new = _proj_simplex(y - lr, 1.0)
        # momentum update
        a_new = _proj_simplex(a + mom * (a_new - a), 1.0)
        a_prev = a.copy()
        a = a_new
