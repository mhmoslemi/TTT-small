def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def gaussian_decay(n, center, sigma):
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    return normalize_nonneg(a)

    a = np.asarray(a, dtype=np.float64).ravel()
    noise = np.exp(rng.normal(0.0, sigma, a.size))
    return normalize_nonneg(a * noise)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.size)
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    if float(b.sum()) < 0.01:
        b = b * (0.01 / max(float(b.sum()), 1e-30))
    return [float(x) for x in b]


def try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def compute_convolution_fft(p, L):
    Pg = np.fft.rfft(p, L)
    c = np.fft.irfft(Pg * Pg, L)[:2 * p.size - 1].real
    return np.maximum(c, 0.0)


def nesterov_smoothed_max(p_init, budget_s, alpha_min=3.0, alpha_max=200.0):
    """
    Nesterov-accelerated gradient descent on smoothed max in log-space.
    p_init: starting point (sums to 1, non-negative)
    Returns: (best_p, best_score)
    """
    p = np.asarray(p_init, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s < 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 0.5)

    # Nesterov momentum
    z = x.copy()  # lookahead variable
    mu = 1.0  # Nesterov parameter
    lr = 0.1
    prev_x = x.copy()

    it = 0
    while time.monotonic() < end:
        it += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Adaptive temperature: start high (sharp) if we're near optimum, start low if far
        # Use a schedule that adapts based on score
        if best_f > 1.6:
            alpha = 10.0 + 80.0 * np.sin(np.pi * frac)
        elif best_f > 1.52:
            alpha = 30.0 + 120.0 * np.sin(np.pi * frac)
        else:
            alpha = 80.0 + 200.0 * np.sin(np.pi * 0.8 * frac)
        alpha = max(alpha_min, min(alpha_max, alpha))

        # Compute p from x (softmax)
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        p = e / e.sum()

        # Compute convolution
        if n <= 512:
            c = np.convolve(p, p)
            c = np.maximum(c, 0.0)
        else:
            c = compute_convolution_fft(p, L)

        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)
        cur_score = f_now  # since sum(p)=1, score = 2n*maxc

        if cur_score < best_f:
            best_f = cur_score
            best_p = p.copy()

        # Compute smoothed weights
        # w_k = exp(alpha * (c[k]/maxc - 1)) normalized
        zvals = alpha * (c / maxc - 1.0)
        zvals -= zvals.max()
        expz = np.exp(zvals)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Compute gradient in log-space
        # grad_x[i] = 2 * p[i] * (h[i] - f_smooth)
        # where h[i] = sum_k w[k] * p[k-i] (cross-correlation of w with p)
        # f_smooth = sum_k w[k] * c[k]

        f_smooth = float(np.dot(w, c))

        # h[i] = sum_k w[k] * p[k-i] for valid k-i in [0, n-1]
        # This is the cross-correlation: h = correlate(w, p) at lags 0..n-1
        # h[i] = sum_{k=i}^{i+n-1} w[k] * p[k-i]  ... actually let me be more careful
        # h[i] = sum_k w[k] * p[k-i], where p[m]=0 for m outside [0,n-1]
        # So for a given i, k must satisfy 0 <= k < 2n-1 and 0 <= k-i < n
        # i.e., i <= k <= i+n-1, and 0 <= k <= 2n-2
        # h[i] = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} w[k] * p[k-i]

        if n <= 512:
            h = np.zeros(n)
            for i in range(n):
                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n)
                if k_start < k_end:
                    h[i] = np.dot(w[k_start:k_end], p[0:k_end - k_start])
        else:
            # FFT-based cross-correlation
            # h[i] = sum_k w[k] * p[k-i]
            # = (w * p_rev)[i + n - 1] where p_rev = p reversed
            # Actually: let's use the same trick as before
            W = np.fft.rfft(w[::-1], L)
            Pg = np.fft.rfft(p, L)
            Cfull = np.fft.irfft(Pg * W, L)
            # Cfull[m] = sum_k p[k] * w_rev[m-k mod L]
            # w_rev[j] = w[2n-2-j]
            # We want h[i] = sum_k w[k]*p[k-i]
            # Let's verify: Cfull[i + n - 1] = sum_k p[k] * w_rev[i+n-1-k]
            # = sum_k p[k] * w[2n-2-i-n+1+k] = sum_k p[k] * w[n-1-i+k]
            # Let j = n-1-i+k, so k = j-n+1+i:
            # = sum_j p[j-n+1+i] * w[j]
            # For valid j in [0, 2n-2] and valid p index: 0 <= j-n+1+i <= n-1
            # h[i] = sum_j w[j] * p[j-(n-1-i)] = sum_j w[j] * p[j - (n-1-i)]
            # This is the same as sum_k w[k] * p[k - (n-1-i)]
            # But we want sum_k w[k] * p[k-i]
            # So h[i] = Cfull[i + n - 1] gives sum_k w[k]*p[k-(n-1-i)]
            # Hmm, the indexing is off by (n-1). Let me just use a different approach.

            # Direct: h[i] = sum_k w[k] * p[k-i]
            # Define q = w (length 2n-1), r = p (length n)
            # h[i] = sum_k q[k] * r[k-i] = (q * r_rev)[i + n - 1] shifted
            # Actually the cross-correlation (q ⋆ r)[i] = sum_k q[k] * r[k+i]
            # So h[i] = (q ⋆ r)[−i] ... this is getting confusing.
            # Let me just use a valid slice approach with FFT.
            # h[i] = sum_{k=0}^{2n-2} w[k] * p[k-i]
            # Define extended p: pe = [p, zeros(n-1)] (length 2n-1)
            # Then h[i] = sum_k w[k] * pe[k-i] which is circular correlation
            # Use FFT: h = IFFT(FFT(w) * conj(FFT(pe))) ... no that's not right either.
            # Let me just do it directly for n <= 2048 and use the trick for larger.
            # For n up to 2048, O(n^2) is at most 4M operations per step, which is fine.
            h = np.zeros(n)
            for i in range(n):
                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n)
                if k_start < k_end:
                    h[i] = np.dot(w[k_start:k_end], p[0:k_end - k_start])

        grad_x = 2.0 * p * (h - f_smooth)

        # Adaptive learning rate
        lr = 0.08 * (1.0 + 0.5 * (1.0 - frac))

        # Nesterov update
        mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        beta = (mu - 1.0) / mu_new
        z = x + beta * (x - prev_x)
        x = z - lr * grad_x
        x -= x.mean()
        prev_x = x
        mu = mu_new

        # Clip for numerical stability
        x = np.clip(x, -100.0, 100.0)
        x -= x.mean()

        # Periodically re-normalize to prevent drift
        if it % 100 == 0:
            p_check = np.exp(x - x.max())
            p_check = p_check / p_check.sum()
            x = np.log(p_check + 1e-300)
            x -= x.mean()

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_polish(p_init, budget_s):
    """
    LP-based polishing: find direction g that maximizes sum(g) subject to f*g <= M.
    """
    if _scipy_opt is None:
        return p_init, actual_score_fft(p_init)

    p = np.asarray(p_init, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, actual_score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 15.0:
        return p, actual_score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)

    # Start with fewer rows, add more as we go
    max_rows = min(400, 2 * n - 1)
    end = time.monotonic() + max(0.0, budget_s - 3.0)

    # Initial rows: evenly spaced
    r0 = min(max_rows, 64)
    rows = list(np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int)).astype(int))
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        nr = len(rows_arr)

        # Build constraint matrix A (nr x n)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, cur_score, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic())
        tl = max(3.0, min(30.0, rem * 0.3))

        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False) or res.x is None:
            break

        g = np.clip(np.asarray(res.x, dtype=np.float64).ravel(), 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check all constraints
        if n <= 512:
            conv = np.convolve(f, g)
        else:
            L2 = next_power_of_two(2 * n - 1)
            Pf = np.fft.rfft(f, L2)
            Pg = np.fft.rfft(g, L2)
            conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        violations = conv - cur_score
        tol = 1e-7 + 1e-7 * cur_score

        if np.any(violations > tol):
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    ix = int(ix)
                    if violations[ix] <= tol:
                        break
                    if ix not in row_set and added < 64:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added > 0:
                    continue
            else:
                break

        g = g * (S0 / sg)

        accepted = None
        accepted_score = np.inf
        for step in (0.3, 0.2, 0.15, 0.1, 0.05, 0.15, 0.25):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or accepted_score >= best_score:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score
        print("  LP iter %d: score=%.6f rows=%d" % (it, best_score, len(rows)))

        if best_score < 1.505:
            break

    return normalize_nonneg(best_f), best_score


def coordinate_descent(p_init, budget_s):
    """
    Simple coordinate descent: at each step, find the max lag and adjust
    the two coordinates that most reduce it.
    """
    a = np.asarray(p_init, dtype=np.float64).ravel().copy()
    n = a.size
    if n < 2:
        return a, actual_score_fft(a)
    a = normalize_nonneg(a)
    best_a = a.copy()
    best_score = actual_score_fft(a)
    start = time.monotonic()
    end = start + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < end:
        if n <= 512:
            c = np.maximum(np.convolve(a, a), 0.0)
        else:
            L = next_power_of_two(2 * n - 1)
            P = np.fft.rfft(a, L)
            c = np.maximum(np.fft.irfft(P * P, L)[:2 * n - 1].real, 0.0)

        k_star = int(np.argmax(c))
        maxc = c[k_star]

        # Gradient of c[k_star] w.r.t. a[j] is 2*a[k_star - j] (if valid)
        # We want to decrease c[k_star], so move in negative gradient direction
        # But we need to keep sum(a) = 1 and a >= 0
        # Simple approach: find j that maximizes a[k_star-j] (most contributing)
        # and redistribute from that coordinate to others

        # Compute effective gradient
        grad = np.zeros(n)
        for j in range(n):
            k2 = k_star - j
            if 0 <= k2 < n:
                grad[j] = 2.0 * a[k2]
            else:
                grad[j] = 0.0

        # Projected gradient step in a-space (with simplex constraint)
        # Use a small step
        eta = 0.01
        a_new = a - eta * grad
        # Project to simplex: a >= 0, sum = 1
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s
        else:
            a_new = a.copy()

        sc = actual_score_fft(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("  NEW BEST: score=%.6f n=%d" % (best_score, a.size))
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Load previous best if available
    h_prev = globals().get("height_sequence_1", None)
    if h_prev is not None:
        try:
            arr = np.asarray(h_prev, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                arr_norm = normalize_nonneg(arr)
                sc = actual_score_fft(arr_norm)
                record(arr_norm, sc)
                print("Previous best loaded: n=%d score=%.6f" % (arr.size, sc))
        except Exception:
            pass

    # Quick baselines for various n
    base_lengths = [64, 128, 192, 256, 320, 384, 448, 512, 640, 768, 1024]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for mode in ["greedy", "pow05", "gauss_c", "mix50"]:
            p = mixed_init(n, rng, mode)
            record(p)

    # Main optimization: Nesterov smoothed max
    # Schedule jobs by (n, mode, time_allocation)
    # Prioritize lengths around the current best
    if best_p is not None:
        best_n = best_p.size
    else:
        best_n = 384

    # Define which lengths to try and how much time to allocate
    n_candidates = sorted(set([
        best_n,
        best_n - 32 if best_n > 64 else 64,
        best_n + 32,
        best_n // 2 if best_n > 64 else 64,
        best_n * 2 if best_n < 2048 else 2048,
        192, 256, 320, 384, 448, 512, 640,
    ]))

    jobs = []
    for n in n_candidates:
        if n < 16 or n > 2048:
            continue
        for mode in ["greedy", "pow05", "gauss_c"]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            # Allocate time based on proximity to best_n
            prox = 1.0 / (1.0 + abs(n - best_n) / 64.0)
            alloc = min(rem * 0.15, max(8.0, 30.0 * prox))
            jobs.append((n, mode, alloc))

    # Sort by expected quality (proximity to best_n, then by initial score)
    def job_key(j):
        n, mode, alloc = j
        p = mixed_init(n, rng, mode)
        sc = actual_score_fft(p)
        return (sc, abs(n - best_n))

    jobs.sort(key=job_key)

    for n, mode, alloc in jobs[:15]:
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        base = mixed_init(n, rng, mode)
        # If we have a best at this n, use it
        rec = best_by_n.get(n)
        if rec is not None:
            base = rec[1].copy()

        alloc_use = min(alloc, max(5.0, rem - 5.0))
        p_out, f_out = nesterov_smoothed_max(base, alloc_use)
        record(p_out, f_out)

    # LP polishing phase
    if _scipy_opt is not None:
        lp_ns = sorted(set([
            best_n,
            best_n - 32 if best_n > 64 else 64,
            best_n + 32,
            256, 384, 512,
        ]))

        lp_candidates = []
        for n in lp_ns:
            if n < 16 or n > 2048:
                continue
            rec = best_by_n.get(n)
            sc = rec[0] if rec is not None else 2.0
            lp_candidates.append((sc, n))
        lp_candidates.sort()

        for _, n in lp_candidates[:5]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            rec = best_by_n.get(n)
            base = rec[1] if rec is not None else greedy_normalized(n)
            alloc = min(60.0, max(10.0, rem * 0.3))
            p_out, f_out = lp_polish(base, alloc)
            record(p_out, f_out)

            # Follow up with Nesterov from LP result
            if np.isfinite(f_out) and f_out < best_score:
                rem2 = deadline - time.monotonic()
                if rem2 > 15:
                    p_out2, f_out2 = nesterov_smoothed_max(p_out, min(rem2 * 0.5, 40.0))
                    record(p_out2, f_out2)

    # Coordinate descent polish
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            p_out, f_out = coordinate_descent(best_p, min(rem * 0.3, 30.0))
            record(p_out, f_out)

    # Additional Nesterov runs from perturbed best
    if best_p is not None:
        for sigma in [0.05, 0.1, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            p0 = perturb(best_p, rng, sigma)
            alloc = min(rem * 0.25, 50.0)
            p_out, f_out = nesterov_smoothed_max(p0, alloc)
            record(p_out, f_out)

    # Final: run Nesterov directly from best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 15:
            p_out, f_out = nesterov_smoothed_max(best_p, min(rem * 0.5, 60.0))
            record(p_out, f_out)

    if best_p is None:
        best_p = greedy_normalized(256)
        best_score = actual_score_fft(best_p)

    # Validate best few candidates
    candidates = [(best_score, best_p)]
    items = sorted(best_by_n.items(), key=lambda x: x[1][0])
    for n, (sc, p) in items[:3]:
        candidates.append((sc, p))

    for sc, p in candidates:
        if deadline - time.monotonic() < 5:
            break
        seq = scale_output(p, 10.0)
        gscore = try_global_eval(seq)
        if np.isfinite(gscore) and gscore < best_score:
            best_score = gscore
            best_p = np.asarray(p, dtype=np.float64).ravel()
            print("  Global eval improved: %.6f" % best_score)

    final_seq = scale_output(best_p, 10.0)
    print("Final: n=%d, score=%.6f, time=%.1fs" % (
        best_p.size, best_score, time.monotonic() - t0))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Exception in main_search: %s" % str(e))
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
