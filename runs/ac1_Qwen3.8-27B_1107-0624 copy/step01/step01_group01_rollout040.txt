```python

LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses iterative LP direction finding with blending, combined with
gradient-based polishing and multi-start exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c

def compute_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def compute_max_conv(a):
    """Max of conv(a,a) assuming sum(a)=1."""
    c = fft_convolve(a)
    return float(c.max())

def make_lp_matrix(f, n):
    """Build the LP constraint matrix A where A[k,j] = f[k-j] if valid."""
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp(f, M, n):
    """Solve the LP to find direction g."""
    A = make_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

    """Run LP-based optimization on f0 (normalized to sum 1)."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size
    
    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        g = solve_lp(f, M, n)
        if g is None or g.sum() < 1e-15:
            # LP infeasible or trivial, try perturbation
            break
        
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-9:

            # Try perturbing
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = compute_max_conv(f)
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
```

```python
LP-based iterative optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses iterative linear programming for direction finding, combined with
soft-max gradient polishing and multi-start exploration at varying sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        c = np.convolve(a, a)
        return np.maximum(c, 0.0)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def compute_score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-30 or not np.isfinite(s):
        return np.inf
    c = fft_convolve(a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def compute_max_conv(a):
    c = fft_convolve(a)
    return float(c.max())


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cubic_profile(n, a1, a3):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_smooth_random(n, rng, n_harm=8):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for k in range(1, n_harm + 1):
        amp = rng.uniform(0.05, 0.8) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * t + phase)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_deterministic_profiles(n):
    profs = []
    for eta in (0.8, 1.0, 1.2, 1.5, 1.8, 2.0, 2.5):
        profs.append(make_linear_profile(n, eta))
    for a1, a3 in ((1.2, -0.8), (1.5, -1.2), (1.8, -1.5), (2.0, -2.0),
                   (1.0, -0.5), (1.5, -2.5), (2.0, -3.0), (0.8, 0.5)):
        profs.append(make_cubic_profile(n, a1, a3))
    t = np.linspace(0, 1, n)
    a = 1.0 + 1.3 * t - 0.5 * t * t
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    a = 1.0 + 0.8 * t + 1.2 * t ** 3
    a = np.maximum(a, 1e-12)
    profs.append(a / a.sum())
    return profs


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<=n-1
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def lp_optimize(f0, deadline, rng, max_iter=20000, verbose=False):
    """Run LP-based iterative optimization. f0 must be normalized to sum=1."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return f0, np.inf
    f = f / s
    n = f.size

    M = compute_max_conv(f)
    best_f = f.copy()
    best_M = M
    no_imp = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = solve_lp_direction(f, M, n)
        if g is None:
            no_imp += 1
            if no_imp > 50:
                break
            f = best_f + 0.01 * rng.normal(size=n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-30:
                f = best_f.copy()
                M = best_M
                continue
            f = f / fs
            M = compute_max_conv(f)
            if M < best_M:
                best_M = M
                best_f = f.copy()
                no_imp = 0
            continue

        g_sum = float(g.sum())
        if g_sum <= 1.0 + 1e-7:
            no_imp += 1
            if no_imp > 200:
                f = best_f + 0.02 * rng.normal(size=n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-30:
                    f = best_f.copy()
                    M = best_M
                    no_imp = 0
                    continue
                f = f / fs
                M = compute_max_conv(f)
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
            continue

        g = g / g_sum

        improved = False
        for t in (0.4, 0.2, 0.6, 0.1, 0.8, 0.3, 0.05, 0.5):
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-30:
                continue
            f_new = f_new / fs
            M_new = compute_max_conv(f_new)
            if M_new < M - 1e-14:
                f = f_new
                M = M_new
                improved = True
                if M < best_M:
                    best_M = M
                    best_f = f.copy()
                    no_imp = 0
                break

        if not improved:
            no_imp += 1
            if no_imp % 100 == 0:
                f = 0.85 * best_f + 0.15 * np.abs(rng.normal(size=n)) + 1e-8
                f = f / f.sum()
                M = compute_max_conv(f)
                no_imp = 0

        if verbose and it % 200 == 0:
            print(f"    LP iter {it}: M={M:.12f}, best={best_M:.12f}")

    return best_f, best_M


def softmax_gradient_polish(f0, deadline, rng, lr=0.06, max_iter=50000):
    """Polish using soft-max gradient in logit space with the refined gradient formula."""
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    n = f.size

    z = np.log(f)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    c = fft_convolve(f)
    M_val = float(c.max())
    beta = max(1e-12, 0.15 * M_val)

    best_f = f.copy()
    best_M = M_val
    best_z = z.copy()
    no_imp = 0
    N_corr = next_pow2(3 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Soft-max weights
        x = (c - M_val) / beta
        ex = np.exp(np.clip(x, -100.0, 5.0))
        s_ex = float(ex.sum())
        if s_ex < 1e-30:
            ex = np.zeros_like(c)
            ex[int(np.argmax(c))] = 1.0
            s_ex = 1.0
        w = ex / s_ex
        L = M_val + beta * float(np.log(s_ex))

        # Gradient: grad_z = 2 * a * (h - L)
        # where h[j] = sum_p w[p] * a[p-j] = cross-correlation (w cross a)[j]
        fw = np.fft.rfft(w, N_corr)
        fa = np.fft.rfft(f, N_corr)
        h = np.fft.irfft(fw * np.conj(fa), N_corr)[:n]

        grad_z = 2.0 * f * (h - L)

        if not np.all(np.isfinite(grad_z)):
            z = best_z.copy()
            m.fill(0)
            v.fill(0)
            t_adam = 0
            continue

        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        f = np.exp(z - z.max())
        f = f / f.sum()
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        c = fft_convolve(f)
        M_new = float(c.max())

        if M_new < best_M - 1e-14:
            best_M = M_new
            best_f = f.copy()
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t_adam % 300 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(300, 3 * n):
            z = 0.85 * best_z + 0.4 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()
            f = np.exp(z - z.max())
            f = f / f.sum()
            c = fft_convolve(f)
            M_val = float(c.max())
            beta = max(1e-12, 0.15 * M_val)
            m.fill(0)
            v.fill(0)
            t_adam = 0
            no_imp = 0

    return best_f, best_M


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        return p0 / s if s > 1e-30 else None
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p1 = np.interp(x1, x0, p0)
    p1 = np.maximum(p1, 0.0)
    s = p1.sum()
    return p1 / s if s > 1e-30 else None


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
    1. LP-based direction finding at n=512 from diverse starts.
    2. Extended LP optimization on best candidates.
    3. Soft-max gradient polish.
    4. Interpolation to n=1024 with further refinement.
    5. Final exact evaluation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def try_update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # Get parent if available
    parent = None
    try:
        pv = np.asarray(height_sequence_1, dtype=np.float64)
        if pv.ndim == 1 and pv.size > 0:
            pv = np.maximum(pv, 0.0)
            s = pv.sum()
            if s > 1e-30:
                parent = pv / s
    except Exception:
        parent = None

    if parent is not None:
        sc = compute_score(parent)
        try_update(parent, sc)
        print(f"Parent score: {sc:.10f}, n={parent.size}")

    # Phase 1: LP optimization at n=512 from multiple starts
    n1 = 512
    phase1_end = min(deadline, t_start + 600.0)

    starts = []
    if parent is not None:
        p_int = interpolate_profile(parent, n1)
        if p_int is not None:
            starts.append(p_int)

    starts.extend(make_deterministic_profiles(n1)[:6])
    for _ in range(8):
        starts.append(make_smooth_random(n1, rng))

    # Also try linear profiles with various slopes
    for eta in (1.0, 1.3, 1.6, 1.9, 2.2, 2.6):
        starts.append(make_linear_profile(n1, eta))

    candidates_512 = []
    per_start = max(5.0, (phase1_end - t_start) / max(1, len(starts)))

    for i, p0 in enumerate(starts):
        if time.monotonic() > phase1_end:
            break
        p0 = np.maximum(p0, 0.0)
        s = p0.sum()
        if s < 1e-30:
            continue
        p0 = p0 / s

        per = min(phase1_end, time.monotonic() + per_start)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=5000)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        candidates_512.append((sc, p_best))
        if i % 5 == 0:
            print(f"  Phase 1 start {i}/{len(starts)}: best={best_score:.10f}")

    candidates_512.sort(key=lambda x: x[0])
    print(f"After Phase 1: best={best_score:.10f}")

    # Phase 2: Extended LP on top candidates
    phase2_end = min(deadline, t_start + 1080.0)
    top_k = min(5, len(candidates_512))
    per_cand = max(10.0, (phase2_end - t_start) / max(1, top_k))

    for i in range(top_k):
        if time.monotonic() > phase2_end:
            break
        _, p0 = candidates_512[i]
        per = min(phase2_end, time.monotonic() + per_cand)
        p_best, M_best = lp_optimize(p0, per, rng, max_iter=30000, verbose=True)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"  Phase 2 cand {i}: score={sc:.10f}")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Soft-max gradient polish on best candidates
    phase3_end = min(deadline, t_start + 1440.0)
    if best_seq is not None and best_seq.size == n1:
        per = min(phase3_end, time.monotonic() + 120.0)
        p_best, M_best = softmax_gradient_polish(best_seq, per, rng, lr=0.06)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)
        print(f"After Phase 3 polish (n={n1}): best={best_score:.10f}")

    # Also polish a few other candidates
    for i in range(min(3, len(candidates_512))):
        if time.monotonic() > phase3_end:
            break
        _, p0 = candidates_512[i]
        per = min(phase3_end, time.monotonic() + 60.0)
        p_best, M_best = softmax_gradient_polish(p0, per, rng, lr=0.05)
        sc = 2.0 * n1 * M_best
        try_update(p_best, sc)

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Try n=1024
    n2 = 1024
    phase4_end = min(deadline, t_start + 1680.0)
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        p_ext = interpolate_profile(best_seq, n2)
        if p_ext is not None:
            per = min(phase4_end, time.monotonic() + 80.0)
            p_best, M_best = lp_optimize(p_ext, per, rng, max_iter=15000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)
            print(f"Phase 4 LP (n={n2}): score={sc:.10f}")

            if time.monotonic() < phase4_end - 30:
                per2 = min(phase4_end, time.monotonic() + 60.0)
                p_best2, M_best2 = softmax_gradient_polish(p_best, per2, rng, lr=0.04)
                sc2 = 2.0 * n2 * M_best2
                try_update(p_best2, sc2)
                print(f"Phase 4 polish (n={n2}): score={sc2:.10f}")

    # Also try direct deterministic profiles at n=1024
    if time.monotonic() < deadline - 60:
        for p0 in make_deterministic_profiles(n2)[:4]:
            if time.monotonic() > deadline - 60:
                break
            per = min(deadline - 60, time.monotonic() + 25.0)
            p_best, M_best = lp_optimize(p0, per, rng, max_iter=8000)
            sc = 2.0 * n2 * M_best
            try_update(p_best, sc)

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Final LP refinement on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = best_seq.size
        per = min(deadline - 5, time.monotonic() + 60.0)
        p_best, M_best = lp_optimize(best_seq, per, rng, max_iter=20000)
        sc = compute_score(p_best)
        try_update(p_best, sc)
        print(f"After Phase 5: best={best_score:.10f}, n={best_seq.size if best_seq is not None else 0}")

    # Final evaluation
    if best_seq is not None:
        final = np.maximum(best_seq, 0.0)
        s = final.sum()
        if s > 1e-30:
            final = final / s
        # Exact score check
        sc_final = compute_score(final)
        print(f"Final internal score: {sc_final:.10f}, n={final.size}")

        # Try the evaluator for exact check on small sequences
        if final.size <= 2048:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(sc_eval):
                    print(f"Final evaluated score: {sc_eval:.10f}")
            except Exception as e:
                print(f"Evaluation error: {e}")

        return [float(x) for x in final]

    # Fallback
    fb = make_linear_profile(512, 1.5)
    return [float(x) for x in fb]
```
