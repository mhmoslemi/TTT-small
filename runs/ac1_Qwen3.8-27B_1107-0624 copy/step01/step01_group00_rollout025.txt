def _proj_simplex(v, target=1.0):

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using FISTA-style momentum descent with
FFT-accelerated gradient of the softmax-smoothed max. The key speedup over the
previous approach is computing the gradient via FFT cross-correlation in O(n log n)
instead of O(L*n), enabling 10x+ more iterations in the time budget. Multiple
diverse seeds and adaptive hyperparameters ensure robustness against local minima.

import time

import numpy as np


def _next_pow2(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using the standard sorted algorithm."""
    n = np.size(v)
    if target <= 1e-15:
        return np.zeros(n, dtype=np.float64)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_sq_fft(a, nfft):
    """Linear convolution a*a via real FFT. Returns array of length 2*len(a)-1."""
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa * fa, nfft)
    return c[:2 * len(a) - 1]


def _grad_smoothed_max(a, c, tau, nfft_corr):
    """Gradient of tau*log(sum_k exp(c_k/tau)) w.r.t. a, computed via FFT.
    
    grad[p] = 2 * sum_j a[j] * w[j+p] where w = softmax(c/tau)
    Computed as 2 * ifft(conj(fft(a)) * fft(w)).real[:n]
    """
    n = len(a)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    w = exp_c / exp_c.sum()
    A = np.fft.fft(a, nfft_corr)
    W = np.fft.fft(w, nfft_corr)
    r = np.fft.ifft(np.conj(A) * W)
    return 2.0 * r.real[:n]


def _grad_topk(a, c, topk_list):
    """Subgradient: average gradient over top-k peak lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for k in topk_list:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= len(topk_list)
    return grad

        c = _conv_sq_fft(a, nfft)
        mc = c.max()
        
        if mc < best_mc:

    # --- Phase 3: Fine-tuning the best solution ---
    remaining = t_end - time.monotonic() - 10.0
    if remaining > 10.0:
        print(f"\nPhase 3: fine-tuning, {remaining:.1f}s available")
        
        # Multiple short runs with tiny perturbations around the best
        n_fine = 4
        per_fine = remaining / n_fine
        for fi in range(n_fine):
            if time.monotonic() > t_end - 12.0:
                break
            a_pert = best_a * np.exp(rng.normal(0, 0.01, n))
            a_pert = _proj_simplex(np.abs(a_pert), 1.0)
            
            a_best, mc = _optimize_seed(a_pert, per_fine, nfft, nfft_corr,
                                         tau_init=0.002, step_init=0.01)
            try:
                sc = float(evaluate_sequence([float(x) for x in a_best]))
            except Exception:
                sc = float('inf')
            print(f"  Fine {fi}: max_c={mc:.10f}, score={sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
                best_mc = mc
    
    # Final safety checks and return
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0)
    
    try:
        final_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"\nFINAL score = {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score
    except Exception:
        print("Final evaluation failed, using last tracked score")
    
    elapsed = time.monotonic() - t_global_start
    print(f"Total time: {elapsed:.1f}s")
    return [float(x) for x in out]
