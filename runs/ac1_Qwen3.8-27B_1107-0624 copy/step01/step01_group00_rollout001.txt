Minimize 2*n*max(convolve(a,a))/(sum(a))^2 over non-negative sequences.
Uses multi-phase smoothed subgradient descent with Nesterov acceleration,
adaptive parameters, stall detection with random kicks, and a final
exact-subgradient polish phase. Scale-invariant: fix sum(a)=1, minimize max(c).

import time
import math
import numpy as np


def _proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.full(n, target / n, dtype=np.float64)
    rho_val = rho[cond][-1]
    theta = (cssv[rho_val - 1] - target) / rho_val
    return np.maximum(v - theta, 0.0)


def _fft_len(n):
    """Next power of 2 >= n."""
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_sq(a, m):
    """Linear auto-convolution via FFT. m must be >= 2*len(a)-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[: 2 * len(a) - 1]


def _grad_smoothed(a, c, L, tau):
    """Smoothed subgradient of sum_k w_k * c_k w.r.t. a, where w from softmax over top-L."""
    n = len(a)
    nc = len(c)
    L = min(L, nc)
    if L < nc:
        order = np.argpartition(-c, L - 1)[:L]
    else:
        order = np.arange(nc)
    vals = c[order]
    vmax = vals.max()
    w = np.exp((vals - vmax) / max(tau, 1e-12))
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros_like(w)
        w[vals.argmax()] = 1.0
    else:
        w /= wsum

    grad = np.zeros(n, dtype=np.float64)
    for idx in range(len(order)):

        if lo > hi:
            continue

        src = k - j
        grad[j] += 2.0 * wk * a[src]
    return grad

    n = len(a)

    grad = np.zeros(n, dtype=np.float64)
    cnt = len(ties)
    for k in ties:

        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:

        j = np.arange(lo, hi + 1)
        src = k - j
