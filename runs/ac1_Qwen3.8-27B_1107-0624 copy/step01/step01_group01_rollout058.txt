```python
def solve_lp(a, M):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0."""
    n = len(a)
    # Build the constraint matrix C where C[k,j] = a[k-j] if 0<=k-j<n else 0
    # Actually, (a*b)_k = sum_j a_{k-j} * b_j for the full convolution
    # C[k,j] = a[k-j] if 0 <= k-j < n, else 0
    from scipy.optimize import linprog
    
    # Objective: minimize -sum(b)
    c = -np.ones(n)
    
    # Constraint: C @ b <= M (vector of M's, length 2n-1)
    # C[k,j] = a[k-j] if 0 <= k-j < n else 0
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                C[k,j] = a[idx]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
```

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating minimization
and gradient polish. Key insight: asymmetric ramps give score ~4/3, and
LP alternation further reduces the peak autocorrelation.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_self(a):
    """Full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = np.maximum(c, 0.0)
    return c


def _score_internal(a):
    """Fast score computation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = _convolve_self(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _score_exact(a):
    """Exact score using direct convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def _normalize(a):
    """Normalize to sum 1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def _linear_ramp(n, power=1.0, flip=False):
    """Create a power-law ramp profile."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    if flip:
        a = a[::-1]
    return a / a.sum()


def _build_conv_matrix(a):
    """Build the convolution constraint matrix C where (C@b)_k = (a*b)_k."""
    n = len(a)
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            C[k, j] = a[k - j]
    return C


def _lp_solve(a, M):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns b (optimal solution) or None."""
    n = len(a)
    c_obj = -np.ones(n, dtype=np.float64)
    C = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_alternate(a_start, max_iters=2000, tol=1e-7, deadline=None):
    """LP-based alternating minimization. Returns (best_a, best_score)."""
    a = _normalize(a_start)
    if a is None:
        return a_start, _score_internal(a_start)
    n = a.size
    best_a = a.copy()
    best_sc = _score_internal(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _convolve_self(a)
        M = float(c.max())
        if M < 1e-15:
            break
        b = _lp_solve(a, M)
        if b is None:
            break
        s_b = float(b.sum())
        if s_b <= 1.0 + tol:
            break
        b_norm = b / s_b
        # Try several step sizes and pick the best
        best_local = None
        best_local_sc = best_sc
        for t in (1.0, 0.75, 0.5, 0.3, 0.15):
            a_try = (1.0 - t) * a + t * b_norm
            a_try = np.maximum(a_try, 0.0)
            sc_try = _score_internal(a_try)
            if sc_try < best_local_sc - 1e-12:
                best_local_sc = sc_try
                best_local = a_try
        if best_local is None:
            break
        a = best_local
        if best_local_sc < best_sc - 1e-13:
            best_sc = best_local_sc
            best_a = a.copy()
        if it % 200 == 0 and it > 0:
            print(f"    LP iter {it}: score={best_sc:.8f}, sum_b={s_b:.6f}")

    return best_a, best_sc


def _interpolate(a, n_new):
    """Linearly interpolate a profile to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    return _normalize(a_new)


def _adam_polish(a_start, deadline, lr=0.05, reg=1e-7):
    """Adam optimization in logit space to polish a sequence."""
    n = a_start.size
    p = np.maximum(a_start, 1e-15)
    p = p / p.sum()
    z = np.log(p)
    z -= z.mean()

    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    c = _convolve_self(p)
    maxc = float(c.max())
    beta = max(1e-10, 0.2 * maxc)

    best_p = p.copy()
    best_sc = float(2.0 * n * maxc)
    no_imp = 0

    N = _next_pow2(2 * n - 1)

    while time.monotonic() < deadline:
        # Soft-max weights
        x = (c - maxc) / beta
        ex = np.exp(np.clip(x, -50, 50))
        w = ex / ex.sum()

        # Gradient: gp_i = 2 * sum_j w_{i+j} * p_j (circular correlation)
        fw = np.fft.rfft(w, N)
        fp = np.fft.rfft(p[::-1], N)
        h = np.fft.irfft(fw * fp, N)
        gp = 2.0 * h[n - 1:2 * n - 1]

        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= gz.mean()

        if not np.all(np.isfinite(gz)):
            break

        t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -40, 40)
        z -= z.mean()

        e = np.exp(z)
        p = e / e.sum()

        c = _convolve_self(p)
        maxc = float(c.max())
        sc = float(2.0 * n * maxc)

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp % 500 == 0:
                lr = max(0.01, lr * 0.95)

        if t % 100 == 0:
            beta = max(1e-10, beta * 0.7)

        if no_imp > 2000:
            break

    return best_p, best_sc

        best_score = _score_internal(best_seq)

    # Exact score check
    if best_seq.size <= 4096:
        sc_exact = _score_exact(best_seq)
        if np.isfinite(sc_exact):
            best_score = min(best_score, sc_exact)
            print(f"Exact score: {best_score:.10f}")

    # Try to verify with evaluate_sequence
    if best_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in best_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
                best_score = min(best_score, sc_eval)
        except Exception:
            pass

    # Ensure valid output
    best_seq = np.asarray(best_seq, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(best_seq.sum())
    if s > 0:
        best_seq = best_seq / s

    print(f"FINAL: score={best_score:.10f}, n={best_seq.size}")
    return [float(x) for x in best_seq]
```
