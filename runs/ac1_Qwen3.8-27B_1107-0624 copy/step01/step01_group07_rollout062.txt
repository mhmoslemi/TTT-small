"""
LP-based iterative optimization with multi-M strategy and subgradient
refinement to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng):
    """Main LP-based optimization loop."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S
    
    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0
    stall = 0

        c = np.convolve(a, a, mode='full')
        M = c.max()

            g = solve_lp(a, M_try)

            g = g / g.sum()

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. A@G <= M, g >= 0 where A@G = conv(a,g)."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

"""
Multi-start LP optimization with asymmetric initializations and multi-M
strategy to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: optimal sequences must be asymmetric to break the Cauchy-Schwarz
barrier that limits symmetric sequences to objective >= 2.
Uses LP direction finding with multiple constraint tightening levels,
adaptive step sizes, and subgradient refinement.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0 where A@g = conv(a,g)."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, deadline, rng, m_factors=None):
    """LP-based optimization with multi-M strategy and adaptive step."""
    if m_factors is None:
        m_factors = (1.0, 0.97, 0.94, 1.02)
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_objective(a)
    it = 0
    stall = 0
    step = 0.3

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        improved = False
        for M_f in m_factors:
            M_try = M * M_f
            g = solve_lp(a, M_try)
            if g is None or g.sum() < 1e-15:
                continue
            g = g / g.sum()

                a_new = (1.0 - s) * best_a + s * g
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() < 1e-10:
                    continue
                a_new = a_new / a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score - 1e-13:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    improved = True
                    break
            if improved:
                break

        if improved:
            stall = 0
            step = min(0.5, step * 1.015)
        else:
            stall += 1
            step = max(0.02, step * 0.82)

def subgradient_refine(a, deadline, rng):

        direction = np.zeros(n)
        for k in peaks:
            js = k - np.arange(n)
            valid = (js >= 0) & (js < n)
            if valid.any():
                direction += a[js[valid]]

        if max_d > 1e-15:
            direction /= max_d
            a_new = a * np.exp(-lr * direction)
            a_new = np.clip(a_new, 0.0, None)

                    if stall > 400:

    return best_a, best_score


def make_hann(n):
    """Symmetric Hann window."""
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n) / max(n - 1, 1)))


def make_cosine_asym(n):
    """Asymmetric cosine: goes from 1 to 0."""
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n) / n))


def make_gaussian_offcenter(n, center_frac=0.35, sigma_frac=0.22):
    """Gaussian window with off-center peak for asymmetry."""
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_gaussian_wide(n, center_frac=0.4, sigma_frac=0.28):
    """Wider Gaussian with off-center peak."""
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_linear_up(n):
    """Linearly increasing sequence."""
    return np.linspace(0.15, 1.0, n)


def make_exponential_decay(n, rate=1.2):
    """Exponentially decreasing."""
    return np.exp(-rate * np.arange(n) / n) + 0.05


def make_parabola_asym(n):
    """Asymmetric parabolic window."""
    x = np.arange(n) / max(n - 1, 1)
    return 0.6 * (1.0 - x) + 0.4 * 4.0 * x * (1.0 - x)


def make_sin_shift(n):
    """Sine-based asymmetric window."""
    return np.abs(np.sin(np.pi * (np.arange(n) + 0.3 * n) / (1.6 * n))) + 0.1


def make_quadratic_asym(n):
    """Quadratic asymmetric window."""
    x = np.arange(n) / max(n - 1, 1)
    return (0.5 + 0.5 * x) ** 2


def make_bump_combination(n):
    """Combination of two Gaussian bumps."""
    x = np.arange(n)
    bump1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    bump2 = 0.6 * np.exp(-0.5 * ((x - 0.65 * n) / (0.12 * n)) ** 2)
    return bump1 + bump2
