    """Compute full convolution using FFT."""
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()  # Next power of 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]

Hybrid coordinate descent + subgradient optimization for minimizing

from scipy.optimize import minimize_scalar


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fft_convolve(a):
    """Full convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]


def coordinate_descent_step(a, conv, i, n):
    """
    Optimize coordinate i of a, given current convolution.
    Returns new a[i] value.
    """
    S = a.sum() - a[i]  # Sum without a[i]
    
    # Base convolution (with a[i] = 0)
    # We need to subtract the contribution of a[i] from conv
    base = conv.copy()
    old_x = a[i]
    for k in range(max(0, i), min(2*n-1, i+n)):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                base[k] -= old_x * old_x
            else:
                base[k] -= 2 * old_x * a[j]
    
    # Ensure base is non-negative (numerical errors)
    base = np.maximum(base, 0)
    
    # 1D optimization
    def obj(x):
        if x < 0:
            return np.inf
        new_conv = base.copy()
        for k in range(max(0, i), min(2*n-1, i+n)):
            j = k - i
            if 0 <= j < n:
                if k == 2 * i:
                    new_conv[k] += x * x
                else:
                    new_conv[k] += 2 * x * a[j]
        total = S + x
        if total < 1e-15:
            return np.inf
        return 2.0 * n * new_conv.max() / (total * total)
    
    # Upper bound for x
    x_max = max(3 * a.max(), 1.0)
    result = minimize_scalar(obj, bounds=(0, x_max), method='bounded',
                             options={'xatol': 1e-10, 'maxiter': 100})
    return result.x

"""
Coordinate descent with FFT-accelerated convolution for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Multiple initializations, adaptive
global moves, and length exploration.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fft_convolve_full(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat(n):
    return np.ones(n)


def make_spike_pair(n):
    a = np.zeros(n)
    a[0] = 1.0
    a[n-1] = 1.0
    return a


def make_tapered(n, alpha=2.0):
    """Power-law taper: a[i] = (i+1)^alpha * (n-i)^alpha, normalized."""
    i = np.arange(n) + 1
    return i**alpha * (n - i + 1)**alpha


def coordinate_optimize(a, conv, i, n):
    """
    Optimize coordinate i. Returns new value for a[i].
    a: current sequence
    conv: current full convolution
    i: index to optimize
    n: length
    """
    S = a.sum() - a[i]
    old_x = a[i]
    
    # Indices in convolution affected by a[i]
    k_start = max(0, i)
    k_end = min(2 * n - 1, i + n - 1)
    ks = np.arange(k_start, k_end + 1)
    js = ks - i  # corresponding j indices, all in [0, n-1]
    
    # Contribution of old a[i] to conv at these positions
    if old_x > 0:
        old_contrib = np.where(ks == 2 * i, old_x * old_x, 2.0 * old_x * a[js])
    else:
        old_contrib = np.zeros(len(ks))
    
    # Base (convolution without a[i]'s contribution)
    base_ks = conv[ks].copy()
    base_ks -= old_contrib
    
    # Max of conv at positions NOT affected by a[i]
    mask = np.ones(2*n-1, dtype=bool)
    mask[ks] = False
    max_outside = conv[mask].max() if mask.any() else 0.0
    
    def obj(x):
        if x < -1e-15:
            return np.inf
        x = max(0.0, x)
        new_contrib = np.where(ks == 2 * i, x * x, 2.0 * x * a[js])
        max_ks = (base_ks + new_contrib).max()
        max_conv = max(max_ks, max_outside)
        total = S + x
        if total < 1e-15:
            return np.inf
        return 2.0 * n * max_conv / (total * total)
    
    x_upper = max(2.0 * a.max() + 0.1, 1.0)
    try:
        res = minimize_scalar(obj, bounds=(0.0, x_upper), method='bounded',
                              options={'xatol': 1e-12, 'maxiter': 80})
        return max(0.0, res.x)
    except:
        return old_x

"""
Coordinate descent with FFT-accelerated convolution for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Combines exact 1D coordinate optimization,
spectral gradient moves for global exploration, multiple initializations,
and adaptive time allocation across different sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fft_convolve_full(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.01, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_tapered(n, alpha=2.0):
    i = np.arange(n).astype(float) + 1.0
    return i ** alpha * (n - i + 1.0) ** alpha


def make_sinc_like(n):
    center = (n - 1) / 2.0
    x = np.arange(n) - center
    return np.sinc(x / (n * 0.3)) ** 2 * (n * 0.3)


def coordinate_optimize(a, conv, i, n):
    """Optimize coordinate i given current sequence and convolution."""
    S = a.sum() - a[i]
    old_x = a[i]

    k_start = max(0, i)
    k_end = min(2 * n - 1, i + n - 1)
    ks = np.arange(k_start, k_end + 1)
    js = ks - i

    if old_x > 1e-15:
        old_contrib = np.where(ks == 2 * i, old_x * old_x, 2.0 * old_x * a[js])
    else:
        old_contrib = np.zeros(ks.shape)

    base_ks = conv[ks] - old_contrib

    mask = np.ones(2 * n - 1, dtype=bool)
    mask[ks] = False
    max_outside = conv[mask].max() if mask.any() else 0.0

    a_js = a[js].copy()
    is_center = (ks == 2 * i)

    def obj(x):
        if x < -1e-12:
            return 1e10
        x = max(0.0, x)
        new_contrib = np.where(is_center, x * x, 2.0 * x * a_js)
        max_ks = np.max(base_ks + new_contrib)
        max_conv = max(max_ks, max_outside)
        total = S + x
        if total < 1e-15:
            return 1e10
        return 2.0 * n * max_conv / (total * total)

    x_upper = max(3.0 * a.max() + 0.5, 2.0 * S / n + 1.0, 1.0)
    try:
        res = minimize_scalar(obj, bounds=(0.0, x_upper), method='bounded',
                              options={'xatol': 1e-13, 'maxiter': 100})
        return max(0.0, res.x)
    except:
        return old_x


def spectral_gradient_step(a, conv, lr, rng):
    """Spectral subgradient step to reduce peak convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a
    M = conv.max()
    peak_threshold = 0.7 * M
    peak_mask = conv > peak_threshold

    # Compute gradient: for each i, how much does a[i] contribute to peak conv positions?
    grad = np.zeros(n)
    # conv[k] = sum_j a[j]*a[k-j], gradient wrt a[i] is sum_{k in peaks, valid j} a[k-i] (with factor 2 except k=2i)
    # Efficient: grad[i] = 2 * sum_{j: peak_mask[i+j]} a[j] (approx, ignoring k=2i correction)
    # Use cross-correlation: grad[i] ~ 2 * (a cross-correlate peak_indicator)[i]
    # But peak_mask is on conv indices [0, 2n-2], and we need for each i:
    # grad[i] = sum_j a[j] * peak_mask[i+j] (for valid i+j in [0, 2n-2])
    # This is a correlation!
    # corr[i] = sum_j a[j] * peak_mask[i+j]
    # We can compute this via FFT:
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    P_fft = np.fft.rfft(peak_mask.astype(float), fft_size)
    corr = np.fft.irfft(A_fft * P_fft, fft_size)
    grad = 2.0 * corr[:n]

    # Subgradient step
    max_grad = np.max(grad) + 1e-15
    a_new = a - lr * grad / max_grad
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() < 1e-10:
        return a
    return a_new


def run_coordinate_descent(a, deadline, rng, verbose=False):
    """Run coordinate descent with periodic global moves."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    conv = fft_convolve_full(a)
    best_a = a.copy()
    best_score = compute_objective(a)

    sweep = 0
    no_improve = 0
    lr = 0.1

    while time.monotonic() < deadline:
        sweep += 1
        old_score = best_score

        # Coordinate descent sweep (random order)
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            new_x = coordinate_optimize(a, conv, i, n)
            if abs(new_x - a[i]) > 1e-14:
                k_start = max(0, i)
                k_end = min(2 * n - 1, i + n - 1)
                ks = np.arange(k_start, k_end + 1)
                js = ks - i
                old_x = a[i]
                old_c = np.where(ks == 2 * i, old_x * old_x, 2.0 * old_x * a[js])
                new_c = np.where(ks == 2 * i, new_x * new_x, 2.0 * new_x * a[js])
                conv[ks] += new_c - old_c
                a[i] = new_x

        # Periodic full conv recomputation for numerical stability
        if sweep % 5 == 0:
            conv = fft_convolve_full(a)

        # Evaluate
        S = a.sum()
        if S < 1e-15:
            a = best_a
            conv = fft_convolve_full(a)
            break
        score = 2.0 * n * conv.max() / (S * S)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            # Revert to best if current is worse
            if score > best_score * 1.01:
                a = best_a.copy()
                conv = fft_convolve_full(a)
            no_improve += 1

        # Global moves
        if no_improve >= 5:
            move = rng.integers(4)
            if move == 0:
                # Multiplicative noise
                noise = np.exp(rng.normal(0, 0.15, n))
                a_new = a * noise
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a = a_new / a_new.sum()
                    conv = fft_convolve_full(a)
            elif move == 1:
                # Spectral gradient step
                a_new = spectral_gradient_step(a, conv, lr, rng)
                if a_new.sum() > 1e-10:
                    a = a_new / a_new.sum()
                    conv = fft_convolve_full(a)
                lr = max(0.01, lr * 0.8)
            elif move == 2:
                # Reverse
                a = a[::-1].copy()
                conv = fft_convolve_full(a)
            else:
                # Add small random vector
                a_new = a + rng.normal(0, 0.05 / n, n)
                a_new = np.clip(a_new, 0, None)
                if a_new.sum() > 1e-10:
                    a = a_new / a_new.sum()
                    conv = fft_convolve_full(a)
            no_improve = 0

        if no_improve > 40:
            break

        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}, score={best_score:.7f}, no_improve={no_improve}")

    return best_a, best_score

        else:
            print(f"  {name}: score={score:.7f} (best so far={best_score:.7f})", flush=True)

    # Final polish on best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\nPolishing best sequence for {remaining:.0f}s...")
            best_seq, best_score = run_coordinate_descent(best_seq, deadline, rng, verbose=True)
            print(f"After polish: score={best_score:.7f}", flush=True)

        # Extra short sweeps at the end
        remaining = deadline - time.monotonic()
        if remaining > 10:
            n = len(best_seq)
            a = best_seq / best_seq.sum()
            # Try a spectral step on the best
            conv = fft_convolve_full(a)
            for lr_try in [0.05, 0.02, 0.01]:
                a_trial = spectral_gradient_step(a, conv, lr_try, rng)
                a_trial = np.clip(a_trial, 0, None)
                if a_trial.sum() > 1e-10:
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_objective(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a_trial
                        print(f"  Spectral improvement: {best_score:.7f}")

    if best_seq is None:
        best_seq = make_cosine(512)
        best_score = compute_objective(best_seq)

    # Ensure compatibility with evaluator (sum >= 0.01)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.02 / best_seq.sum())

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.7f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
