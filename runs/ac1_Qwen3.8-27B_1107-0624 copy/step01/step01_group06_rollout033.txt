```python

Multi-scale L-BFGS optimization with log-sum-exp smoothing, log-parameterization,
and peak-directed coordinate descent for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import minimize


def convolve_fft(a):
    """Fast self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(conv(a,a)) / (sum(a)^2)."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_objective_and_grad(x, tau):
    """
    Compute smooth objective J and gradient w.r.t. x in log-parameterization.
    a = exp(x), J = 2n * Tmax / S^2 where Tmax = tau * log(sum(exp(c/tau)))
    dJ/dx[i] = dJ/da[i] * a[i]
    """
    n = len(x)
    a = np.exp(x)
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # dJ/da[i] = 4n * cross[i] / S^2 - 4n * Tmax / S^3
    # where cross[i] = sum_k p[k] * a[(k-i) mod ...] 
    # Actually: cross[i] = sum_k p[k] * a[k-i] for valid k-i
    # This is the cross-correlation of p and a, computed via FFT
    
    fp = np.fft.rfft(p, size)
    # cross[i] = sum_k p[k] * a[k-i]
    # This is (p * a)[i] in the convolution sense? No...
    # Actually: sum_k p[k] * a[k-i] is the cross-correlation.
    # In FFT: cross = IFFT(conj(FFT(a)) * FFT(p)) gives sum_j a[j] * p[j+i]?
    # Let me think carefully.
    
    # We want: cross[i] = sum_{k} p[k] * a[k-i]
    # This is (p cross-correlate a)[i] = sum_k p[k] a[k-i]
    # In terms of convolution: if we define q[k] = p[-k] (reversed), then
    # cross[i] = sum_k p[k] a[k-i] = sum_j q[j] a[j+i] where j = -k
    # Hmm, this is getting confusing. Let me use a different approach.
    
    # cross[i] = sum_k p[k] * a[k-i]
    # = sum_k p[k] * a[k-i]
    # Let's compute this via FFT.
    # Define b[k] = a[k] for 0 <= k < n, 0 otherwise.
    # Then cross[i] = sum_k p[k] * b[k-i]
    # This is the convolution (p * b_reversed)[2n-2-i]? No...
    
    # Actually, the simplest way: cross-correlation of p and a.
    # (p ⋆ a)[i] = sum_k p[k+i] * a[k] = sum_j p[j] * a[j-i] (with j = k+i)
    # Hmm, conventions vary. Let me just compute it directly via FFT.
    
    # cross[i] = sum_{k=i}^{n-1} p[k] * a[k-i]  (only valid terms where both indices in range)
    # Wait, p has length 2n-1 and a has length n.
    # cross[i] = sum_{k} p[k] * a[k-i] where 0 <= k-i < n, i.e., i <= k < i+n
    # = sum_{k=i}^{min(2n-2, i+n-1)} p[k] * a[k-i]
    
    # This is the cross-correlation of p (length 2n-1) and a (length n).
    # Using FFT: IFFT(FFT(p) * conj(FFT(a))) gives the cross-correlation.
    # Specifically, if P = FFT(p) and A = FFT(a), then IFFT(P * conj(A))[i] = sum_k p[k] * conj(a[k-i])
    # Since everything is real: IFFT(P * conj(A))[i] = sum_k p[k] * a[k-i] = cross[i] ✓
    
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]
    
    dJ_da = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    dJ_dx = dJ_da * a  # chain rule: dJ/dx[i] = dJ/da[i] * da[i]/dx[i] = dJ/da[i] * a[i]
    
    return J, dJ_dx


def run_lbfgs(x_init, tau, max_iter, ftol, gtol, max_fun=None):
    """Run L-BFGS-B on the smoothed objective."""
    def func_and_grad(x):
        J, grad = smooth_objective_and_grad(x, tau)
        if not np.isfinite(J):
            return 1e15, np.zeros_like(x)
        if not np.isfinite(grad).all():
            return J, np.zeros_like(x)
        return J, grad
    
    result = minimize(
        func_and_grad, x_init, method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol, 'maxfun': max_fun or 100000}
    )
    return result.x, result.fun
```

```python
"""
Multi-scale L-BFGS optimization with log-sum-exp smoothing, log-parameterization,
and peak-directed coordinate descent to minimize J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses structured initializations, adaptive tau annealing, and iterative refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize


def convolve_fft(a):
    """Fast self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(conv(a,a)) / (sum(a)^2) using FFT."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_grad(x, tau):
    """
    Compute smooth objective J and gradient w.r.t. x (log-parameterization).
    a = exp(x). Returns (J, dJ_dx).
    """
    n = len(x)
    a = np.exp(x)

    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    if sum_exp < 1e-300:
        return 1e15, np.zeros(n)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # cross[i] = sum_k p[k] * a[k-i] (cross-correlation)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(fp * np.conj(fa), size)[:n]

    # dJ/da[i] = 4n*cross[i]/S^2 - 4n*Tmax/S^3
    dJ_da = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    # dJ/dx[i] = dJ/da[i] * a[i]
    dJ_dx = dJ_da * a

    if not np.isfinite(J):
        return 1e15, np.zeros(n)
    if not np.isfinite(dJ_dx).all():
        dJ_dx = np.zeros(n)

    return J, dJ_dx


def run_lbfgs(x_init, tau, max_iter, time_budget):
    """Run L-BFGS-B on the smoothed objective with time limit."""
    deadline = time.monotonic() + time_budget
    n = len(x_init)
    call_count = [0]

    def func_and_grad(x):
        call_count[0] += 1
        return smooth_obj_grad(x, tau)

    try:
        result = minimize(
            func_and_grad, x_init, method='L-BFGS-B', jac=True,
            options={'maxiter': max_iter, 'ftol': 1e-11, 'gtol': 1e-8,
                     'maxfun': max(1000, max_iter * 20)}
        )
        return result.x, result.fun
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return x_init, smooth_obj_grad(x_init, tau)[0]


def make_initializations(parent, n, seed):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Parent
    inits.append(('parent', parent.copy()))

    inits.append(('uniform', np.ones(n)))

    # Cosine modulations

    inits.append(('cos2', np.maximum(0.1, 1.0 + 0.4 * np.cos(4.0 * np.pi * np.arange(n) / n))))

def peak_coordinate_descent(a, budget_s, seed):
    """Peak-directed coordinate descent to refine the solution."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    best_a = a.copy()
    best_J = compute_J(a)
    no_improve = 0
    factors = [0.97, 0.95, 0.9, 0.85, 0.75]

    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = convolve_fft(best_a)
        k_star = int(np.argmax(c))

        # Find valid indices: i such that 0<=i<n and 0<=k_star-i<n
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)

        if i_min > i_max:
            break

        indices = np.arange(i_min, i_max + 1)
        contributions = best_a[indices] * best_a[k_star - indices]

        # Top contributors
        top_k = min(40, len(indices))
        top_local = np.argsort(contributions)[::-1][:top_k]
        top_indices = indices[top_local]

        improved = False
        for idx in top_indices:
            j = k_star - idx

            # Try decreasing a[idx]
            for factor in factors:
                cand = best_a.copy()
                cand[idx] *= factor
                cJ = compute_J(cand)
                if cJ < best_J - 1e-9:
                    best_J = cJ
                    best_a = cand
                    improved = True
                    break

            # Try decreasing a[j] (if different)
            if j != idx and j >= 0 and j < n:
                for factor in factors:
                    cand = best_a.copy()
                    cand[j] *= factor
                    cJ = compute_J(cand)
                    if cJ < best_J - 1e-9:
                        best_J = cJ
                        best_a = cand
                        improved = True
                        break

        if not improved:
            no_improve += 1
            # Periodic random perturbation to escape local minima
            if no_improve > 20:
                w = int(rng.integers(max(1, n // 200), n // 20 + 1))
                s = int(rng.integers(0, max(1, n - w)))
                cand = best_a.copy()
                cand[s:s+w] *= np.exp(rng.normal(0, 0.03, w))
                cJ = compute_J(cand)
                if cJ < best_J:
                    best_J = cJ
                    best_a = cand
                    no_improve = 0
                elif no_improve > 200:
                    # Larger perturbation
                    w = int(rng.integers(n // 50, n // 5 + 1))
                    s = int(rng.integers(0, max(1, n - w)))
                    cand = best_a.copy()
                    cand[s:s+w] *= np.exp(rng.normal(0, 0.1, w))
                    cJ = compute_J(cand)
                    if cJ < best_J:
                        best_J = cJ
                        best_a = cand
                        no_improve = 0
                    else:
                        no_improve = 0  # reset to avoid infinite loop
        else:
            no_improve = 0

        # Periodic verification
        if it % 200 == 0:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            print(f"    CD iter {it}: J={best_J:.6f}")

    return best_a, best_J


def gradient_fallback(a, budget_s, seed, tau=0.1):

    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    x = np.log(np.maximum(a, 1e-10))
    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.05

    while time.monotonic() < deadline:
        J, grad = smooth_obj_grad(x, tau)
        if not np.isfinite(J) or not np.isfinite(grad).all():
            break
        grad_norm = np.sqrt(np.sum(grad**2))
        if grad_norm < 1e-12:
            break
        step = lr * grad / grad_norm
        x = x - step
        a_new = np.exp(x)
        a_new = np.clip(a_new, 1e-10, 1000.0)
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        else:
            lr *= 0.8

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-scale L-BFGS with log-sum-exp smoothing, log-parameterization,
    peak-directed coordinate descent, and structured initializations.
    """
    total_start = time.monotonic()
    total_deadline = total_start + budget_s - 15.0  # 15s safety buffer

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121)

    n = parent.size
    print(f"n = {n}")

    # Score parent
    parent_J = float(evaluate_sequence(parent.tolist()))
    print(f"Parent J = {parent_J:.6f}")

    # Generate and score initializations
    inits = make_initializations(parent, n, seed)
    print("\nInitializations:")
    for i in range(len(inits)):
        name, arr = inits[i]
        J = compute_J(arr)
        inits[i] = (name, J, arr)
        print(f"  {name}: J = {J:.6f}")

    # Sort by quality
    inits.sort(key=lambda x: x[1])
    print(f"\nBest init: '{inits[0][0]}' with J={inits[0][1]:.6f}")

    # Start from the best initialization
    best_name = inits[0][0]
    best_a = inits[0][2].copy()
    best_J = inits[0][1]

    # If parent is better than the best init, use parent
    if parent_J < best_J:
        best_a = parent.copy()
        best_J = parent_J
        best_name = 'parent'
        print(f"Using parent as start (J={best_J:.6f})")

    # === Phase 1: Coarse L-BFGS (large tau, quick global structure) ===
    remaining = total_deadline - time.monotonic()
    phase1_time = min(80.0, remaining * 0.08)
    print(f"\n=== Phase 1: Coarse L-BFGS ({phase1_time:.0f}s) ===")

    x = np.log(np.maximum(best_a, 1e-10))
    for tau in [50.0, 10.0, 5.0]:
        if time.monotonic() > total_deadline - 600:
            break
        t0 = time.monotonic()
        tau_time = min(phase1_time / 3.0, total_deadline - time.monotonic() - 500)
        if tau_time < 5:
            break
        x, _ = run_lbfgs(x, tau, max_iter=200, time_budget=tau_time)
        a = np.exp(x)
        a = np.clip(a, 1e-10, 1000.0)
        a_J = compute_J(a)
        if a_J < best_J:
            best_J = a_J
            best_a = a.copy()
            print(f"  tau={tau}: J={a_J:.6f} *** NEW BEST ***")
        else:
            print(f"  tau={tau}: J={a_J:.6f} (best={best_J:.6f})")
        # Always continue from best
        x = np.log(np.maximum(best_a, 1e-10))

    # Verify
    val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(val) and val < best_J:
        best_J = val
    print(f"  After Phase 1: verified J={best_J:.6f}")

    # === Phase 2: Fine L-BFGS (decreasing tau, multi-scale) ===
    remaining = total_deadline - time.monotonic()
    phase2_time = remaining * 0.55
    print(f"\n=== Phase 2: Fine L-BFGS ({phase2_time:.0f}s) ===")

    x = np.log(np.maximum(best_a, 1e-10))

    # Tau schedule: from medium to very fine
    tau_schedule = [2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
    # Time weights (more time for finer taus)
    weights = [0.04, 0.06, 0.08, 0.10, 0.12, 0.14, 0.16, 0.14, 0.12, 0.14]
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    for i, tau in enumerate(tau_schedule):
        if time.monotonic() > total_deadline - 300:
            break
        tau_time = phase2_time * weights[i]
        tau_time = min(tau_time, total_deadline - time.monotonic() - 200)
        if tau_time < 5:
            break

        t0 = time.monotonic()
        max_iter = max(50, int(tau_time * 3))
        x, _ = run_lbfgs(x, tau, max_iter=max_iter, time_budget=tau_time)
        a = np.exp(x)
        a = np.clip(a, 1e-10, 1000.0)
        a_J = compute_J(a)

        elapsed = time.monotonic() - t0
        if a_J < best_J:
            best_J = a_J
            best_a = a.copy()
            print(f"  tau={tau:.4f}: J={a_J:.6f} *** NEW BEST *** ({elapsed:.1f}s)")
        else:
            print(f"  tau={tau:.4f}: J={a_J:.6f} (best={best_J:.6f}) ({elapsed:.1f}s)")

        # Always restart from best
        x = np.log(np.maximum(best_a, 1e-10))

    # Verify
    val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(val) and val < best_J:
        best_J = val
    print(f"  After Phase 2: verified J={best_J:.6f}")

    # === Phase 3: Peak-directed coordinate descent ===
    remaining = total_deadline - time.monotonic() - 60
    phase3_time = min(remaining * 0.5, 500.0)
    if phase3_time > 20:
        print(f"\n=== Phase 3: Coordinate Descent ({phase3_time:.0f}s) ===")
        best_a, best_J = peak_coordinate_descent(best_a, phase3_time, seed + 100)
        val = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(val) and val < best_J:
            best_J = val
        print(f"  After Phase 3: verified J={best_J:.6f}")

    # === Phase 4: Final L-BFGS polish ===
    remaining = total_deadline - time.monotonic() - 20
    if remaining > 30:
        print(f"\n=== Phase 4: Final Polish ({remaining:.0f}s) ===")
        x = np.log(np.maximum(best_a, 1e-10))
        for tau in [0.005, 0.002]:
            if time.monotonic() > total_deadline - 25:
                break
            tau_time = min(remaining / 2.0, total_deadline - time.monotonic() - 25)
            if tau_time < 5:
                break
            x, _ = run_lbfgs(x, tau, max_iter=200, time_budget=tau_time)
            a = np.exp(x)
            a = np.clip(a, 1e-10, 1000.0)
            a_J = compute_J(a)
            if a_J < best_J:
                best_J = a_J
                best_a = a.copy()
                print(f"  Polish tau={tau}: J={a_J:.6f} *** NEW BEST ***")
            else:
                print(f"  Polish tau={tau}: J={a_J:.6f} (best={best_J:.6f})")
            x = np.log(np.maximum(best_a, 1e-10))

    # === Phase 5: Try second-best initialization if time remains ===
    remaining = total_deadline - time.monotonic() - 20
    if remaining > 60 and len(inits) > 1:
        second_init = inits[1]
        if second_init[1] < best_J + 0.5:  # Only if reasonably close
            print(f"\n=== Phase 5: Try '{second_init[0]}' ({remaining:.0f}s) ===")
            x = np.log(np.maximum(second_init[2], 1e-10))
            for tau in [5.0, 1.0, 0.1, 0.02]:
                if time.monotonic() > total_deadline - 25:
                    break
                tau_time = min(remaining / 4.0, total_deadline - time.monotonic() - 25)
                if tau_time < 5:
                    break
                x, _ = run_lbfgs(x, tau, max_iter=100, time_budget=tau_time)
                a = np.exp(x)
                a = np.clip(a, 1e-10, 1000.0)
                a_J = compute_J(a)
                if a_J < best_J:
                    best_J = a_J
                    best_a = a.copy()
                    print(f"  Phase5 tau={tau}: J={a_J:.6f} *** NEW BEST ***")
                x = np.log(np.maximum(best_a, 1e-10))

    # Final verification
    final_val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(final_val):
        best_J = min(best_J, final_val)

    total_time = time.monotonic() - total_start
    print(f"\n{'='*50}")
    print(f"FINAL: J = {best_J:.6f}")
    print(f"Total time: {total_time:.1f}s")
    print(f"{'='*50}")

    result = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in result]
```
