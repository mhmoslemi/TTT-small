def _lp_direction(f, M, max_rows):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    if linprog is None:
        return None
    n = f.size
    if n > 2048:
        return None
    if M <= 1e-30:
        return None

    c = np.convolve(f, f)
    if max_rows >= 2 * n - 1:
        rows = np.arange(2 * n - 1)
    else:
        rows = np.argpartition(c, -max_rows)[-max_rows:]

    A_ub = np.zeros((len(rows), n))
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            idx = np.arange(jlo, jhi)
            A_ub[r, jlo:jhi] = f[k - idx]

    if not np.any(A_ub):
        return None
    A_ub = A_ub / M
    try:
        res = linprog(-np.ones(n), A_ub=A_ub, b_ub=np.ones(len(rows)),
                      bounds=(0, None), method='highs')
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0, None)
        if g.sum() > 1e-10:
            return g
    return None

    elif kind == 2:
        L = N // 2
        a = np.zeros(N)
        a[:L] = (2.0 / 3.0) / L
        a[L:] = (1.0 / 3.0) / (N - L)
    elif kind == 3:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif kind == 4:
        # Sine bump
        a = np.maximum(0.1, np.sin(np.pi * x) ** 2)
    elif kind == 5:
        # Multi-block
        K = int(rng.integers(2, 6))
        a = np.zeros(N)
        blocks = np.sort(rng.choice(N, K, replace=False))
        for k in range(K):
            start = blocks[k] if k == 0 else blocks[k - 1] + 1
            end = blocks[k] + 1 if k < K - 1 else N
            a[start:end] = rng.uniform(0.3, 1.0) / max(1, end - start)
    elif kind == 6:
        # Smooth random
        base = rng.normal(0, 1, N)
        w = np.ones(max(1, N // 20))
        w /= w.sum()
        base = np.convolve(base, w, mode='same')
        a = np.exp(0.5 * base)
    else:
        a = rng.random(N)
    return _project_simplex(np.maximum(a, 0.0))


def _resize(a, N):
    """Resize sequence to length N."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == N:
        return _project_simplex(a)
    if m < N:
        x_new = np.linspace(0, m - 1, N)
        i0 = np.floor(x_new).astype(int)
        i1 = np.minimum(i0 + 1, m - 1)
        frac = x_new - i0
        out = a[i0] * (1 - frac) + a[i1] * frac
    else:
        out = np.zeros(N)
        for i in range(N):
            lo = (i * m) // N
            hi = ((i + 1) * m) // N
            out[i] = a[lo:hi].sum()
    return _project_simplex(np.maximum(out, 0.0))


def _load_provided():
    """Load height_sequence_1 if available."""
    try:
        raw = height_sequence_1
    except Exception:
        return None
    try:
        a = np.asarray(raw, dtype=np.float64)
    except Exception:
        return None
    if a is None or a.size == 0 or a.ndim != 1:
        return None
    a = np.clip(a, 0, None)
    if a.sum() < 1e-30:
        return None
    return _project_simplex(a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for low-score non-negative step function heights.
    Combines coordinate descent, smooth-max subgradient (FFT-accelerated),
    LP direction finding, and diverse initialization on a time budget.
    """
    rng = np.random.default_rng(seed)
    hard_deadline = time.monotonic() + max(60.0, float(budget_s) - 45.0)

    best_a = None
    best_sc = np.inf

    def update(a, sc):
        nonlocal best_a, best_sc
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best_a = np.asarray(a, dtype=np.float64).copy()
            print(f"  NEW BEST: score={best_sc:.12f}, n={best_a.size}", flush=True)

    # Load provided sequence
    provided = _load_provided()
    if provided is not None:
        sc_p = _score(provided)
        update(provided, sc_p)
        print(f"Provided sequence: score={sc_p:.12f}, n={provided.size}", flush=True)

    # Fallback
    fallback = np.array([1.0 / 3.0, 2.0 / 3.0])
    update(fallback, _score(fallback))

    # Phase 1: Quick exploration at multiple lengths
    print("Phase 1: Exploration", flush=True)
    lengths = [128, 256, 512, 1024]
    kinds = [0, 1, 2, 3, 4, 5, 6]

    for N in lengths:
        if time.monotonic() > hard_deadline - 300:
            break
        for k in kinds:
            if time.monotonic() > hard_deadline - 60:
                break
            a = _make_initial(N, k, rng)
            # Quick coordinate descent (2 passes)
            a, sc = _coordinate_descent(a, max_passes=3, time_limit=5.0, rng=rng)
            update(a, sc)

    # Phase 2: Deep optimization of best candidates
    print("Phase 2: Deep optimization", flush=True)

    # Build a pool of starting points for the dominant length
    target_N = 512
    if provided is not None:
        pool = [_resize(provided, target_N)]
    else:
        pool = []

    # Add the current best resized
    if best_a is not None:
        pool.append(_resize(best_a, target_N))

    # Add diverse starts
    for k in range(7):
        pool.append(_make_initial(target_N, k, rng))

    # Also try the provided at its native length if reasonable
    if provided is not None and 128 <= provided.size <= 2048:
        pool.append(provided.copy())

    # Optimize each pool member
    time_per = max(5.0, (hard_deadline - time.monotonic() - 120.0) / max(1, len(pool)))
    for idx, a in enumerate(pool):
        if time.monotonic() > hard_deadline - 60:
            break
        a = _project_simplex(a)
        n_trial = a.size

        # Coordinate descent (primary, fast)
        a_cd, sc_cd = _coordinate_descent(a, max_passes=500, time_limit=time_per * 0.5, rng=rng)
        update(a_cd, sc_cd)

        # Smooth-max subgradient (secondary)
        if time.monotonic() < hard_deadline - 60:
            a_sm, sc_sm = _smooth_max_optimize(a_cd, time_limit=min(30.0, time_per * 0.4), rng=rng)
            update(a_sm, sc_sm)

            # If smooth-max found something better, do more CD on it
            if sc_sm < sc_cd:
                a_cd2, sc_cd2 = _coordinate_descent(a_sm, max_passes=200, time_limit=15.0, rng=rng)
                update(a_cd2, sc_cd2)

        # LP direction if n is small enough
        if n_trial <= 1024 and time.monotonic() < hard_deadline - 60:
            a_now = best_a if best_a.size == n_trial else a_cd
            a_now = _project_simplex(a_now)
            b_now = np.convolve(a_now, a_now)
            M = float(b_now.max())
            rows = min(2 * n_trial - 1, 500)
            g = _lp_direction(a_now, M, rows)
            if g is not None:
                g = g / g.sum()
                for t in (0.1, 0.3, 0.5, 0.7, 0.9):
                    h = _project_simplex((1 - t) * a_now + t * g)
                    sc_h = _score_norm(h)
                    update(h, sc_h)
                # CD on the best blend
                h_best = a_now
                sc_hb = _score_norm(a_now)
                for t in (0.1, 0.3, 0.5, 0.7, 0.9):
                    h = _project_simplex((1 - t) * a_now + t * g)
                    sc_h = _score_norm(h)
                    if sc_h < sc_hb:
                        sc_hb = sc_h
                        h_best = h
                a_lp, sc_lp = _coordinate_descent(h_best, max_passes=100, time_limit=10.0, rng=rng)
                update(a_lp, sc_lp)

    # Phase 3: Progressive upsampling
    print("Phase 3: Upsampling", flush=True)
    if best_a is not None:
        cur = best_a.copy()
        for target in [512, 1024, 2048]:
            if time.monotonic() > hard_deadline - 30:
                break
            if cur.size >= target:
                continue
            up = _resize(cur, target)
            up, sc = _coordinate_descent(up, max_passes=300, time_limit=60.0, rng=rng)
            update(up, sc)
            if sc < _score(_project_simplex(cur)):
                cur = up
            # Also try smooth-max on the upscaled
            if time.monotonic() < hard_deadline - 30:
                up2, sc2 = _smooth_max_optimize(up, time_limit=30.0, rng=rng)
                update(up2, sc2)

    # Phase 4: Final polish on the best
    print("Phase 4: Final polish", flush=True)
    if best_a is not None and time.monotonic() < hard_deadline - 10:
        a_final = _project_simplex(best_a)
        # A few more CD passes
        a_final, sc_final = _coordinate_descent(a_final, max_passes=500,
                                                 time_limit=min(30.0, hard_deadline - time.monotonic() - 10), rng=rng)
        update(a_final, sc_final)
        # Smooth max polish
        if time.monotonic() < hard_deadline - 10:
            a_final2, sc_final2 = _smooth_max_optimize(a_final, time_limit=15.0, rng=rng)
            update(a_final2, sc_final2)
            if sc_final2 < sc_final:
                a_final2, sc_final3 = _coordinate_descent(a_final2, max_passes=200,
                                                          time_limit=10.0, rng=rng)
                update(a_final2, sc_final3)

    # Final evaluation
    if best_a is not None:
        best_a = np.clip(best_a, 0, None)
        final_score = _score(best_a)
        print(f"Final best: score={final_score:.12f}, n={best_a.size}", flush=True)
        cand = [float(x) for x in best_a]
        try:
            official = evaluate_sequence(cand)
            print(f"Official score: {official:.12f}", flush=True)
        except Exception:
            pass
        return cand

    # Absolute fallback
    return [1.0 / 3.0, 2.0 / 3.0]
