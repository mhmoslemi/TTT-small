```python
"""
Minimize 2*n*max(conv(a,a)) / (sum(a))^2 over non-negative sequences a.

Uses a primal-dual method: the primal iterates optimize a on the non-negative
simplex using a weighted-average gradient (weights given by dual variable p over
lags), and the dual iterates update p to focus on the dominant lags of the
autocorrelation. This adapts the "effective subgradient" over time, giving
better descent directions than fixed top-K methods.
"""

import time
import math
import numpy as np


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _conv_sq(a):
    """Linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.zeros(n)
    rho_idx = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho_idx - 1] - target) / rho_idx
    return np.maximum(v - theta, 0.0)


def _grad_w(a, p, m):
    """Gradient of a^T M(p) a where M(p)_{i,j} = p_{i+j}.
    p has length 2n-1. Uses FFT-based convolution."""
    n = a.size
    a_rev = a[::-1]
    # Compute a_rev * p (full convolution) via FFT
    # Length of result: n + (2n-1) - 1 = 3n-2
    m2 = _next_pow2(n + 2 * n - 1)
    fa = np.fft.rfft(a_rev, m2)
    fp = np.fft.rfft(p, m2)
    conv = np.fft.irfft(fa * fp, m2)
    # g_i = 2 * conv[i + n - 1] for i = 0, ..., n-1
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def _softmax_p(lags, c, temp):
    """Compute softmax distribution over lags with given temperature."""
    x = c / max(temp, 1e-12)
    x = x - x.max()
    w = np.exp(x)
    s = w.sum()
    if s <= 0:
        return np.ones(len(lags)) / len(lags)
    return w / s


def _run_primal_dual(a, t_end, eta_init, eta_final, lr_init, lr_final,
                     temp_init, temp_final, seed):
    """Run the primal-dual optimization from initial point a until t_end.
    Returns (best_a, best_score_proxy) where score_proxy = max(conv(a,a))."""
    n = a.size
    a = _proj_simplex(a.copy(), 1.0)
    c = _conv_sq(a)
    best_a = a.copy()
    best_mc = c.max()
    
    # Initialize p proportional to c^alpha (alpha=1 initially)
    p = c.copy()
    p = p / p.sum()
    
    m = _next_pow2(2 * n - 1)
    m2 = _next_pow2(3 * n - 2)
    
    momentum_buf = np.zeros(n)
    momentum = 0.85
    step = 0
    total_iters = 0
    
    while time.monotonic() < t_end - 1.0:
        # Adaptive parameters based on progress
        frac = min(1.0, step / 50000.0)  # ramp over 50K steps
        eta = eta_init + (eta_final - eta_init) * frac
        lr = lr_init + (lr_final - lr_init) * frac
        temp = temp_init + (temp_final - temp_init) * frac if temp_final < temp_init \
               else temp_init + (temp_final - temp_init) * frac
        
        # Compute current autocorrelation
        c = _conv_sq(a)
        mc = c.max()
        
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
        
        # Update p (dual variable) using softmax of c
        if step % 10 == 0:  # update p every 10 steps
            p = _softmax_p(np.arange(2 * n - 1), c, temp)
        
        # Compute weighted gradient
        g = _grad_w(a, p, m2)
        
        # Normalized gradient step with momentum
        gnorm = np.linalg.norm(g)
        if gnorm > 1e-15:
            g = g / gnorm
        momentum_buf = momentum * momentum_buf - lr * g
        a_new = a + lr * momentum_buf
        a_new = _proj_simplex(a_new, 1.0)
        
        a = a_new
        step += 1
        total_iters += 1
        
        # Periodic momentum increase
        if step % 10000 == 0:
            momentum = min(0.95, momentum + 0.01)
    
    return best_a, best_mc


def _run_topk_polish(a, t_end, topk, lr, seed):
    """Polish using exact top-K subgradient. Returns best_a, best_mc."""
    n = a.size
    a = _proj_simplex(a.copy(), 1.0)
    m = _next_pow2(2 * n - 1)
    c = _conv_sq(a)
    best_a = a.copy()
    best_mc = c.max()
    momentum_buf = np.zeros(n)
    momentum = 0.9
    step = 0
    
    while time.monotonic() < t_end - 1.0:
        c = _conv_sq(a)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
        
        # Top-K subgradient
        K = min(topk, 2 * n - 1)
        # Get the values at the top-K lags
        # grad_i = 2 * sum_{k in topK} a[k-i] (for valid indices)
        # Use the formula: for lag k, contribution to grad_i is 2*a[max(0,k-n+1)...min(k,n-1)]
        # Specifically: grad_i += 2 * a[k-i] if 0 <= k-i < n, and 2 * a[i-k] if 0 <= i-k < n
        # Combined: for each k in topK, add 2*a[k-i] to grad_i (if valid) and 2*a[i-k] (if valid)
        # But since k >= 0 and i >= 0: k-i can be negative (then a[i-k] contributes)
        # Actually: (a*a)_k = sum_j a_j * a[k-j], d/d a_i = a[k-i] + a[i-k] (with out-of-range = 0)
        
        order = np.argpartition(-c, K - 1)[:K]
        g = np.zeros(n)
        for k in order:
            # d(a*a)_k / da_i = a[k-i] + a[i-k] for valid i
            # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            # a[i-k] is valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
            # Combined: i ranges from max(0, k-n+1) to min(2n-2, k+n-1) intersect [0, n-1]
            # = i from max(0, k-n+1) to min(n-1, k+n-1)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k + n - 1)
            # For i in [lo, hi]:

            # a[i-k] valid when i-k in [0, n-1], i.e., i in [k, k+n-1]
```
