"""
Multi-scale gradient optimization with adaptive soft-max and subgradient polishing
to minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FFT-based convolution, logits parameterization, and a two-phase strategy:
(1) L-BFGS-B on smooth-max surrogate at reduced resolution, (2) full-resolution
refinement with sharp surrogate and subgradient polishing.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

# --- Global state ---
_best_score = float("inf")
_best_p = None  # probability vector (sum=1)
_search_deadline = 0.0


class SearchTimeout(Exception):
    pass


# --- Core utilities ---

def _pow2_at_least(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _self_conv_fft(p, fft_size=None):
    """Compute full self-convolution of p using FFT."""
    n = len(p)
    if fft_size is None:
        fft_size = _pow2_at_least(2 * n - 1)
    f = np.fft.rfft(p, fft_size)
    c = np.fft.irfft(f * f, fft_size).real
    return c[:2 * n - 1]


def _score_from_p(p):
    """Compute score = 2n * max(conv(p,p)) for probability vector p."""
    c = _self_conv_fft(p)
    np.maximum(c, 0.0, out=c)  # numerical safety
    n = len(p)
    return 2.0 * n * float(np.max(c))


def _convolve_padded(a, b, fft_size):
    """Full linear convolution via FFT."""
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size).real


def _grad_from_weights(p, w):
    """Compute grad_p of max-conv with weights w on peaks.
    Returns g where g[m] = 2 * sum_k w[k] * p[k-m] (valid terms only).
    """
    n = len(p)
    # w has length 2n-1 (full convolution)
    # g[m] = 2 * sum_k w[k] * p[k-m] = 2 * (convolve(p_flipped, w))[n-1+m]
    p_flipped = p[::-1]
    fft_size = _pow2_at_least(n + len(w) - 1)
    conv = _convolve_padded(p_flipped, w, fft_size)
    h = conv[n - 1:2 * n - 1]  # length n
    return 2.0 * h


def _softmax_probs(z):
    """Convert logits z (length n-1) to probability vector of length n.
    p[0] = 1/Z, p[j] = exp(z[j-1])/Z."""
    n = len(z) + 1
    m = float(np.max(z)) if len(z) > 0 else 0.0
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    Z = e0 + float(np.sum(ez))
    p = np.empty(n, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = ez / Z
    return p


def _p_to_z(p):
    """Convert probability vector to logits."""
    p = np.maximum(p, 1e-30)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -50.0, 50.0)


def _z_to_p(z):
    return _softmax_probs(z)


def _update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = np.array(p, copy=True)


def _objective_and_grad(z, alpha, n):
    """Smooth-max objective and gradient in z-space."""
    if time.monotonic() >= _search_deadline:
        raise SearchTimeout

    p = _z_to_p(z)
    c = _self_conv_fft(p)
    cmax = float(np.max(c))
    hard = 2.0 * n * cmax
    _update_best(p, hard)

    # Soft-max smoothing
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s < 1e-300 or not np.isfinite(s):
        J = cmax
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s

    # Gradient in p-space
    gp = _grad_from_weights(p, w)

    # Chain rule: dJ/dz_j = p[j+1] * (gp[j+1] - sum(p * gp))
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)

    return float(J), gz


# --- Initialization functions ---

def _uniform_z(n):
    return np.zeros(n - 1, dtype=np.float64)


def _logistic_z(n, beta):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    u = beta * (x - 0.5)
    u = np.clip(u, -50, 50)
    f = 2.0 / (1.0 + np.exp(-u))
    p = f / np.sum(f)
    return _p_to_z(p)


def _raised_cosine_z(n):
    x = np.arange(n, dtype=np.float64) / n
    f = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    f = np.maximum(f, 1e-10)
    p = f / np.sum(f)
    return _p_to_z(p)


def _sine_z(n):
    x = np.arange(n, dtype=np.float64) / n
    f = np.sin(np.pi * x)
    f = np.maximum(f, 1e-10)
    p = f / np.sum(f)
    return _p_to_z(p)


def _triangle_z(n):
    x = np.arange(n, dtype=np.float64) / n
    f = 1.0 - np.abs(2.0 * x - 1.0)
    f = np.maximum(f, 1e-10)
    p = f / np.sum(f)
    return _p_to_z(p)


def _power_z(n, a):
    x = np.arange(n, dtype=np.float64) / n
    f = np.power(x + 1e-6, a) + np.power(1 - x + 1e-6, a)
    f = np.maximum(f, 1e-10)
    p = f / np.sum(f)
    return _p_to_z(p)


def _bimodal_z(n, center_spread):
    x = np.arange(n, dtype=np.float64) / n
    f = np.exp(-((x - 0.3) ** 2) / (2 * center_spread ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * center_spread ** 2))
    p = f / np.sum(f)
    return _p_to_z(p)


def _random_z(n, rng, complexity=4):
    x = np.arange(n, dtype=np.float64) / n
    z = np.zeros(n)
    for freq in range(1, complexity + 1):
        amp = rng.normal(0, 2.0 / freq)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.sin(2 * np.pi * freq * x + ph)
    f = np.exp(np.clip(z, -3, 3))
    p = f / np.sum(f)
    return _p_to_z(p)

def _lbfags_optimize(z0, n, alpha, maxfun, deadline):
    """Run L-BFGS-B optimization from z0."""
    global _search_deadline
    if minimize is None:
        return
    _search_deadline = deadline
    bounds = [(-50.0, 50.0)] * (n - 1)
    try:
        minimize(
            _objective_and_grad,
            z0,
            args=(alpha, n),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxfun": maxfun,
                "maxiter": maxfun // 2,
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 40,
                "disp": False,
            },
        )
    except SearchTimeout:
        pass
    except Exception:
        pass


def _subgradient_polish(z0, n, deadline, rng, top_k=16):
    """Subgradient method with adaptive step size."""
    z = np.array(z0, copy=True)
    p = _z_to_p(z)
    score = _score_from_p(p)
    _update_best(p, score)

    best_score = score
    step = 0.5
    fails = 0
    noise = 0.05

    while time.monotonic() < deadline:
        c = _self_conv_fft(p)
        np.maximum(c, 0.0, out=c)
        cmax = float(np.max(c))

        # Select top-k peaks with geometric weighting
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(2 * n - 1)
        c_top = c[idx]
        # Exponential weighting (sharper = more focus on the max)
        wv = np.exp(10.0 * (c_top - cmax))
        w[idx] = wv / float(np.sum(wv))

        gp = _grad_from_weights(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        accepted = False
        if gnorm > 1e-15:
            direction = -gz / gnorm
            # Backtracking
            for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
                zt = z + step * factor * direction
                zt = np.clip(zt, -50, 50)
                pt = _z_to_p(zt)
                st = _score_from_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(p, score)
                    accepted = True
                    break

        if not accepted:
            # Try random perturbations
            for _ in range(5):
                zt = z + rng.normal(0, noise, n - 1)
                zt = np.clip(zt, -50, 50)
                pt = _z_to_p(zt)
                st = _score_from_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(p, score)
                    accepted = True
                    break

        if accepted:
            fails = 0
            step = min(2.0, step * 1.1)
            noise = min(0.3, noise * 1.15)
        else:
            fails += 1
            step = max(0.01, step * 0.8)
            noise = max(0.01, noise * 0.85)
            if fails > 200:
                break

    return z, score


def _upscale_z(z_small, n_small, n_large):
    """Upscale a logit vector from n_small to n_large via interpolation."""
    # Convert to probability, interpolate, convert back
    p_small = _z_to_p(z_small)
    # Interpolate to n_large
    x_old = np.linspace(0, 1, n_small)
    x_new = np.linspace(0, 1, n_large)
    p_large = np.interp(x_new, x_old, p_small)
    p_large = p_large / np.sum(p_large)  # renormalize
    return _p_to_z(p_large)

                    z, p, sc = zt, pt, st
                    acc = True
                    break
        if not acc:
            for _ in range(5):
                zt = np.clip(z + rng.normal(0, noise, N - 1), -50, 50)
                pt = _z2p(zt)
                st = _sc(pt, N, FFT_S)
                if st < sc - 1e-13:
                    z, p, sc = zt, pt, st
                    acc = True
                    break
        if acc:
            fails = 0
            stp = min(2.5, stp * 1.06)
            noise = min(0.4, noise * 1.08)
        else:
            fails += 1
            stp = max(0.005, stp * 0.82)
            noise = max(0.005, noise * 0.88)
            if fails > 600:
                break
        _upd(p, sc)
    return z, sc


def _pe(z0, dl, rng):
    """Peak equalization: try to flatten the top of the convolution."""
    z = z0.copy()
    p = _z2p(z)
    sc = _sc(p, N, FFT_S)
    _upd(p, sc)
    stp = 0.3
    for _ in range(200):
        if time.monotonic() >= dl:
            break
        c = _cv(p, FFT_S)
        cm = float(c.max())
        # Pick the single max peak and try to reduce it
        kmax = int(c.argmax())
        w = np.zeros(2 * N - 1)
        w[kmax] = 1.0
        gp = _gw(p, w, FFT_G)
        sd = float(gp @ p)
        gz = p[1:] * (gp[1:] - sd)
        gn = float(np.linalg.norm(gz))
        if gn < 1e-15:
            break
        d = -gz / gn
        acc = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            zt = np.clip(z + stp * f * d, -50, 50)
            pt = _z2p(zt)
            st = _sc(pt, N, FFT_S)
            if st < sc - 1e-13:
                z, p, sc = zt, pt, st
                acc = True
                break
        if acc:
            stp = min(1.0, stp * 1.05)
        else:
            stp *= 0.7
            # Random kick
            zt = np.clip(z + rng.normal(0, 0.05, N - 1), -50, 50)
            pt = _z2p(zt)
            st = _sc(pt, N, FFT_S)
            if st < sc:
                z, p, sc = zt, pt, st
        _upd(p, sc)
    return z, sc


def _iu():
    return np.zeros(N - 1)


def _il(beta):
    x = (np.arange(N) + 0.5) / N
    f = 2.0 / (1.0 + np.exp(-beta * (x - 0.5)))
    return _p2z(f / f.sum())


def _isin():
    x = np.arange(N) / N
    f = np.maximum(np.sin(np.pi * x), 1e-10)
    return _p2z(f / f.sum())


def _ipow(a):
    x = np.arange(N) / N
    f = np.maximum(x, 1e-8)**a + np.maximum(1 - x, 1e-8)**a
    return _p2z(f / f.sum())


def _ibi(s):
    x = np.arange(N) / N
    f = np.exp(-0.5 * ((x - 0.3) / s)**2) + np.exp(-0.5 * ((x - 0.7) / s)**2)
    return _p2z(f / f.sum())


def _i3m(s):
    x = np.arange(N) / N
    f = np.exp(-0.5 * ((x - 0.2) / s)**2) + np.exp(-0.5 * ((x - 0.5) / s)**2) + np.exp(-0.5 * ((x - 0.8) / s)**2)
    return _p2z(f / f.sum())


def _irc(rng):
    x = np.arange(N) / N
    z = np.zeros(N)
    for fq in range(1, 7):
        z += rng.normal(0, 1.8 / fq) * np.sin(2 * np.pi * fq * x + rng.uniform(0, 6.2832))
    f = np.exp(np.clip(z, -3, 3))
    return _p2z(f / f.sum())


def _irc_p(rng):
    """Random polynomial shape."""
    x = np.arange(N) / N
    t = x - 0.5
    deg = rng.integers(2, 6)
    c = rng.normal(0, 1, deg)
    f = np.ones(N)
    for i in range(deg):
        f += c[i] * t**(i + 1)
    f = np.maximum(f, 1e-8)
    return _p2z(f / f.sum())


def _inits(rng):
    out = [_iu()]
    for b in [1, 2, 3, 5, 8, 12, 16, 24]:
        out.append(_il(b))
    out.append(_isin())
    for a in [0.5, 1, 1.5, 2, 3, 5, 8]:
        out.append(_ipow(a))
    for s in [0.04, 0.07, 0.1, 0.15, 0.25]:
        out.append(_ibi(s))
    for s in [0.05, 0.1, 0.15]:
        out.append(_i3m(s))
    for _ in range(6):
        out.append(_irc(rng))
    for _ in range(6):
        out.append(_irc_p(rng))
    return out


def _inits_r(rng):
    """Initializations at reduced size N_R."""
    x = (np.arange(N_R) + 0.5) / N_R
    out = [np.zeros(N_R - 1)]
    for b in [1, 2, 3, 5, 8, 12]:
        f = 2.0 / (1.0 + np.exp(-b * (x - 0.5)))
        out.append(_p2z(f / f.sum()))
    x2 = np.arange(N_R) / N_R
    f = np.maximum(np.sin(np.pi * x2), 1e-10)
    out.append(_p2z(f / f.sum()))
    for a in [0.5, 1, 2, 3, 5]:
        f = np.maximum(x2, 1e-8)**a + np.maximum(1 - x2, 1e-8)**a
        out.append(_p2z(f / f.sum()))
    for s in [0.05, 0.1, 0.2]:
        f = np.exp(-0.5 * ((x2 - 0.3) / s)**2) + np.exp(-0.5 * ((x2 - 0.7) / s)**2)
        out.append(_p2z(f / f.sum()))
    for _ in range(4):
        z = np.zeros(N_R)
        for fq in range(1, 6):
            z += rng.normal(0, 1.8 / fq) * np.sin(2 * np.pi * fq * x2 + rng.uniform(0, 6.2832))
        f = np.exp(np.clip(z, -3, 3))
        out.append(_p2z(f / f.sum()))
    return out


def propose_candidate(seed=42, budget_s=1800.0, **kw):
    global _best_score, _best_p, _deadline
    t0 = time.monotonic()
    B = float(budget_s)
    end = t0 + B - 3.0
    _best_score = float('inf')
    _best_p = None
    rng = np.random.default_rng(seed)

    # Try inherited base
    base_seq = None
    try:
        base = height_sequence_1
        base = np.asarray(base, float)
        base = np.where(np.isfinite(base), base, 0.0)
        base = np.clip(base, 0, 1000)
        if base.size >= 2 and float(base.sum()) > 0.01:
            base_seq = base
            p = base / base.sum()
            if p.size != N:
                p = np.interp(np.linspace(0, 1, N), np.linspace(0, 1, p.size), p)
                p /= p.sum()
            _upd(p, _sc(p, N, FFT_S))
    except Exception:
        pass

    # Multi-scale: optimize at N_R to get good inits
    # Phase A: reduced-size optimization (first ~15% of budget)
    phaseA_dl = t0 + min(270.0, 0.15 * B)
    inits_r = _inits_r(rng)
    scored_r = []
    for zr in inits_r:
        pr = _z2p(zr)
        sr = _sc(pr, N_R, FFT_SR)
        scored_r.append((sr, zr))
    scored_r.sort(key=lambda x: x[0])

    # Optimize top reduced-size candidates
    for i, (sr, zr) in enumerate(scored_r[:8]):
        rem = phaseA_dl - time.monotonic()
        if rem < 5:
            break
        alpha = 2e5
        mf = max(80, min(400, int(rem / 0.008)))
        _lb(zr, alpha, mf, phaseA_dl, red=True)

    # Upscale best reduced solutions to full size
    # Re-score at full size
    full_inits = _inits(rng)
    # Add upscaled solutions
    for i, (sr, zr) in enumerate(scored_r[:6]):
        pr = _z2p(zr)
        pf = _up(pr, N)
        zf = _p2z(pf)
        full_inits.append(zf)

    # Score all full-size inits
    scored_f = []
    for zf in full_inits:
        pf = _z2p(zf)
        sf = _sc(pf, N, FFT_S)
        _upd(pf, sf)
        scored_f.append((sf, zf))
    scored_f.sort(key=lambda x: x[0])

    print(f"After init: best={_best_score:.6f} t={time.monotonic()-t0:.1f}s")

    # Phase B: main L-BFGS optimization loop
    phaseB_dl = end - max(90, 0.1 * B)
    run = 0
    last_best = _best_score
    stagnant = 0

    while time.monotonic() < phaseB_dl:
        rem = phaseB_dl - time.monotonic()
        if rem < 15:
            break

        if run < len(scored_f):
            z0 = scored_f[run][1]
        else:
            # Perturb best
            zb = _p2z(_best_p) if _best_p is not None else np.zeros(N - 1)
            frac = max(0.0, rem / max(1.0, phaseB_dl - t0))
            if stagnant > 20:
                noise = 0.5 + 0.5 * frac
            else:
                noise = 0.03 + 0.25 * frac
            z0 = np.clip(zb + rng.normal(0, noise, N - 1), -50, 50)

        run += 1
        # Adaptive alpha: sharper as we improve
        if _best_score > 1.55:
            alpha = 3e5
        elif _best_score > 1.52:
            alpha = 1e6
        elif _best_score > 1.51:
            alpha = 3e6
        else:
            alpha = 8e6
        mf = max(100, min(900, int(rem / 0.025)))
        _lb(z0, alpha, mf, phaseB_dl)

        if _best_score < last_best - 1e-8:
            stagnant = 0
            last_best = _best_score
            print(f"  run {run}: best={_best_score:.6f} t={time.monotonic()-t0:.0f}s")
        else:
            stagnant += 1

    # Phase C: intensive polishing
    if _best_p is not None:
        zb = _p2z(_best_p)
        pol_dl = end - max(20, 0.02 * B)

        # Subgradient polish
        zb, _ = _pol(zb, pol_dl, rng, tk=16)
        print(f"After polish1: best={_best_score:.6f} t={time.monotonic()-t0:.0f}s")

        # Peak equalization
        rem = end - 5 - time.monotonic()
        if rem > 30:
            zb, _ = _pe(zb, end - 5.0, rng)
            print(f"After peak-eq: best={_best_score:.6f} t={time.monotonic()-t0:.0f}s")

        # Final L-BFGS with very sharp alpha
        rem = end - 3 - time.monotonic()
        if rem > 25:
            z0 = np.clip(_p2z(_best_p) + rng.normal(0, 0.015, N - 1), -50, 50)
            mf = max(100, int(min(600, rem / 0.03)))
            _lb(z0, 15e6, mf, end - 3.0)

        # Final short polish
        rem = end - 2 - time.monotonic()
        if rem > 10:
            _pol(_p2z(_best_p), end - 2.0, rng, tk=8)

    # Build final sequence
    if _best_p is not None:
        p = _best_p
    else:
        p = np.ones(N) / N

    seq = [float(v) for v in p]
    try:
        sc = float(evaluate_sequence(seq))
    except Exception:
        sc = float('inf')

    if not np.isfinite(sc) or sc > 2.001:
        seq = [1.0 / N] * N
        sc = 2.0

    # Compare with base
    if base_seq is not None:
        try:
            bs = float(evaluate_sequence([float(v) for v in base_seq]))
            if np.isfinite(bs) and bs < sc:
                seq = [float(v) for v in base_seq]
                sc = bs
        except Exception:
            pass

    print(f"FINAL: score={sc:.6f} n={len(seq)} t={time.monotonic()-t0:.0f}s")
    return seq
