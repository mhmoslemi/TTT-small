using a combination of:
  1. Parametric family exploration (power laws, windowed profiles)

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def score_direct(a):
    """Compute the objective directly for small sequences."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = np.convolve(a, a)
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def score_fft(a, L_cache=None):
    """Compute the objective using FFT for larger sequences."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    if n <= 768:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def conv_direct(a):
    """Direct convolution, returning the full result."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    c = np.convolve(a, a)
    return np.maximum(c, 0.0)


def params_power(n, alpha, beta=0.0, gamma=0.0, c1=1.0, c2=1.0):
    """a_i = (i+c1)^(-alpha) * (n-1-i+c2)^(-beta) * exp(-gamma*(i/n-0.5)^2)"""
    idx = np.arange(n, dtype=np.float64)
    a = np.power(idx + c1, -alpha)
    if beta > 1e-12:
        a = a * np.power(n - 1.0 - idx + c2, -beta)
    if gamma > 1e-12:
        a = a * np.exp(-gamma * (idx / n - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def params_sinc_window(n, alpha, win_beta):
    """a_i = (i+1)^(-alpha) * sin(pi*(i+1)/(2*(n+1)))^(2*win_beta)"""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -alpha)
    w = np.sin(np.pi * idx / (2.0 * (n + 1)))
    a = a * np.power(w, 2.0 * win_beta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def params_greedy(n):
    """Coefficients of 1/sqrt(1-x): C(2k,k)/4^k, decays as 1/sqrt(pi*k)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def coordinate_descent(a, budget_s, n_sweeps=None, rng=None):
    """
    Coordinate descent with incremental convolution updates.
    Optimizes each coordinate by trying candidate values.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n < 2:
        return a, score_fft(a)
    if rng is None:
        rng = np.random.default_rng(0)

    # Use direct convolution for efficiency (n is small enough)
    if n > 2048:
        return a, score_fft(a)

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones(n) / n
        s = 1.0

    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 2.0)

    conv = conv_direct(a)
    maxc = float(conv.max())
    cur_score = float(2.0 * n * maxc / (s * s))
    best_score_local = cur_score
    best_a = a.copy()

    # Number of candidate values per coordinate
    n_cand = 9
    # Geometric ratios for candidates
    ratios = np.array([0.5, 0.65, 0.78, 0.9, 1.0, 1.12, 1.3, 1.55, 2.2])

    sweep = 0
    while time.monotonic() < deadline:
        sweep += 1
        # Random coordinate order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() >= deadline:
                break
            
            old_val = a[i]
            if old_val <= 1e-30:
                # Try activating this coordinate
                candidates = [a.sum() / n * ratios[idx] for idx in range(n_cand)]
            else:
                candidates = [old_val * r for r in ratios]
            candidates = [max(0.0, c) for c in candidates]

            best_v = old_val
            best_sc = cur_score
            old_sum = s
            old_conv = conv.copy()

            for v in candidates:
                if abs(v - old_val) < 1e-15 * max(1.0, old_val):
                    continue
                # Update sum
                new_sum = old_sum - old_val + v
                if new_sum <= 1e-12:
                    continue
                # Update convolution incrementally
                delta = v - old_val
                new_conv = old_conv.copy()
                # Update all lags that involve a[i]
                # For lag k, the contribution of a[i] is:
                #   a[i]*a[k-i] if 0 <= k-i < n
                #   plus a[k-i]*a[i] if k-i != i (i.e., k != 2i)
                # So total: 2*a[i]*a[k-i] for k != 2i, a[i]^2 for k = 2i
                # Delta: for k != 2i: 2*delta*a[k-i]
                #         for k = 2i: v^2 - old_val^2 + delta*a[i] (but a[i] is being changed)
                # Actually: new contribution = v * a[k-i] (for k-i valid)
                #            old contribution = old_val * a[k-i]
                #            delta_conv[k] = delta * a[k-i] (one side)
                # But we also have the mirror: a[k-i]*v vs a[k-i]*old_val
                # For k where both i and k-i are in range and i != k-i: two terms
                # For k = 2i: one term v^2 vs old_val^2
                
                for k in range(2 * n - 1):
                    ki = k - i  # other index
                    if 0 <= ki < n:
                        if ki == i:  # k = 2i
                            new_conv[k] = old_conv[k] - old_val * old_val + v * v
                        else:
                            new_conv[k] = old_conv[k] + 2.0 * delta * a[ki]
                    # Also check the other direction: k-i could give i on the other side
                    # Actually in the convolution sum_p a[p]*a[k-p], a[i] appears when:
                    #   p = i: contributes a[i]*a[k-i] (if k-i valid)
                    #   p = k-i: contributes a[k-i]*a[i] (if k-i valid and k-i != i)
                    # So for k != 2i, we get 2*delta*a[k-i]
                    # For k = 2i, we get delta^2 + 2*delta*old_val = v^2 - old_val^2
                    # This is already handled above.
                
                new_conv = np.maximum(new_conv, 0.0)
                new_maxc = float(new_conv.max())
                new_score = 2.0 * n * new_maxc / (new_sum * new_sum)
                if new_score < best_sc:
                    best_sc = new_score
                    best_v = v

            # Apply best
            if best_v != old_val:
                delta = best_v - old_val
                s = s - old_val + best_v
                a[i] = best_v
                # Update convolution
                for k in range(2 * n - 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if ki == i:
                            conv[k] += best_v * best_v - old_val * old_val
                        else:
                            conv[k] += 2.0 * delta * a[ki]
                conv = np.maximum(conv, 0.0)
                maxc = float(conv.max())
                cur_score = 2.0 * n * maxc / (s * s)
                if cur_score < best_score_local:
                    best_score_local = cur_score
                    best_a = a.copy()

        # Optional: damped full-sequence update
        if sweep % 3 == 0 and time.monotonic() < deadline:
            # Slightly perturb and see if it helps
            pass

    final_score = score_fft(best_a)
    if final_score < best_score_local:
        best_score_local = final_score
        best_a = a.copy()
    else:
        best_score_local = score_fft(best_a)
    return best_a, best_score_local

def adam_optimize(a, budget_s, seed=0, temp_start=8.0, temp_end=2.0):
    """
    Adam optimizer on the log-parameterized non-negative sequence.
    Uses softened max (Laplace smoothing) for the convolution peak.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n < 2:
        return a, score_fft(a)
    
    a = np.clip(a, 1e-300, 1000.0)
    a = normalize_nonneg(a)
    
    x = np.log(a + 1e-300)
    x = x - x.mean()

    # Learning rate schedule
    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.025
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m_buf = np.zeros(n, dtype=np.float64)
    v_buf = np.zeros(n, dtype=np.float64)
    t_step = 0

    L = next_power_of_two(3 * n - 2)
    best_p = a.copy()
    best_f = score_fft(a)

    t0 = time.monotonic()
    deadline = t0 + max(0.5, float(budget_s) - 2.0)

    while time.monotonic() < deadline:
        t_step += 1
        elapsed = time.monotonic() - t0
        total = float(budget_s) - 2.0
        frac = min(1.0, elapsed / max(1e-9, total))
        
        # Temperature schedule: start high (smooth), end low (sharp)
        alpha = temp_end + (temp_start - temp_end) * np.exp(-3.0 * frac)

        # Compute convolution
        Pg = np.fft.rfft(a, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)  # sum(a) = 1
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = a.copy()

        # Smoothed max weights
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: d/d a_i of [sum_k w_k * conv_k]
        # conv_k = sum_p a[p] * a[k-p]
        # d(conv_k)/d(a_i) = a[k-i] + a[i-k] (symmetric terms, careful with double counting)
        # Actually: conv_k = sum_p a[p]*a[k-p]
        # d/d a_i = a[k-i] (if 0<=k-i<n) + a[k-i] (if i!=k-i and 0<=k-i<n) 
        # = a[k-i] + (a[k-i] if k!=2i else 0) ... no
        # Let me be precise: conv_k = sum_{p=0}^{n-1} a[p] * a[k-p] (where a[j]=0 if j not in [0,n-1])
        # d(conv_k)/d(a_i) = a[k-i] (from p=i term: derivative of a[i]*a[k-i] w.r.t. a[i] is a[k-i])
        #                    + a[k-i] (from p=k-i term: derivative of a[k-i]*a[i] w.r.t. a[i] is a[k-i])
        # But if k-i == i (i.e., k=2i), these are the same term, so we only get a[i] once.
        # So: d(conv_k)/d(a_i) = a[k-i] + [a[k-i] if k != 2i else 0] (with a[j]=0 for j out of range)
        #                         = a[k-i] * (2 if k != 2i else 1) 

        # Gradient of the weighted sum:
        # grad_i = sum_k w_k * d(conv_k)/d(a_i)
        #        = sum_k w_k * a[k-i] * (1 + [k != 2i])
        # By symmetry and the weight structure, we can compute this via convolution:
        # grad_i = sum_k w_k * a[k-i] + sum_k w_k * a[k-i] * [k != 2i]
        #        = (w * a)_reversed[i] + (w * a)_reversed[i] - w[2i] * a[i]
        #        = 2 * (w_reversed * a)[i] - w[2i]*a[i] (if 2i < 2n-1)
        
        # Simpler: compute via FFT
        # grad_i = sum_k w[k] * (a[k-i] + a[i-k]) where we define a[j]=0 for j out of [0,n-1]
        # = sum_k w[k]*a[k-i] + sum_k w[k]*a[i-k]
        # The first sum is a cross-correlation, computable via FFT.
        # Let's just do it directly for clarity (n is small enough):
        # grad_i = sum_{k: 0<=k-i<n} w[k]*a[k-i] + sum_{k: 0<=i-k<n} w[k]*a[i-k]
        # = sum_{j: 0<=j<n} w[i+j]*a[j] + sum_{j: 0<=j<n} w[i-j]*a[j]
        
        # For efficiency, use the fact that this is a convolution:
        # grad = ifft(fft(a) * fft(w_padded)) appropriately indexed
        # Let's use the direct approach for n <= 2048
        
        if n <= 2048:
            # grad[i] = sum_j a[j] * (w[i+j] + w[i-j]) where w is indexed 0..2n-2
            # w[i+j] is valid when 0 <= i+j <= 2n-2
            # w[i-j] is valid when 0 <= i-j <= 2n-2
            grad = np.zeros(n, dtype=np.float64)
            for i in range(n):
                # sum_j a[j] * w[i+j] for j where 0<=i+j<=2n-2, i.e., -i <= j <= 2n-2-i
                # j in [max(0, -i), min(n-1, 2n-2-i)] = [0, min(n-1, 2n-2-i)]
                # sum_j a[j] * w[i-j] for j where 0<=i-j<=2n-2, i.e., i-(2n-2) <= j <= i
                # j in [max(0, i-2n+2), min(n-1, i)] = [0, i] (since i < n < 2n-2)
                # So:
                # First part: sum_{j=0}^{min(n-1, 2n-2-i)} a[j]*w[i+j]
                # Second part: sum_{j=0}^{i} a[j]*w[i-j]
                hi1 = min(n - 1, 2 * n - 2 - i)
                if hi1 >= 0:
                    grad[i] += np.dot(a[:hi1 + 1], w[i:i + hi1 + 1])
                if i >= 0:
                    grad[i] += np.dot(a[:i + 1], w[i::-1]) if i > 0 else a[0] * w[0]
        else:
            # FFT-based gradient
            # grad[i] = sum_j a[j]*(w[i+j] + w[i-j])
            # = (a * w_flipped)[2n-2-i] ... this gets complicated, let's do direct for now
            # For large n, use the FFT trick:
            # sum_k w[k]*a[k-i] = (w_reversed * a)[i] where * is cross-correlation
            # We can compute this as: ifft(fft(w_padded) * conj(fft(a_padded)))
            from numpy import fft as np_fft
            Lg = next_power_of_two(4 * n)
            A_f = np_fft.rfft(a, Lg)
            W_f = np_fft.rfft(w, Lg)
            # cross_corr[i] = sum_k w[k+i] * a[k] = ifft(conj(W_f) * A_f)[i]
            # Actually let's just do: grad = 2 * cross_corr - diagonal_correction
            # For now, fallback to the same loop (slower but correct)
            grad = np.zeros(n, dtype=np.float64)
            for i in range(n):
                hi1 = min(n - 1, 2 * n - 2 - i)
                if hi1 >= 0:
                    grad[i] += np.dot(a[:hi1 + 1], w[i:i + hi1 + 1])
                if i >= 0:
                    if i > 0:
                        grad[i] += np.dot(a[:i + 1], w[i::-1])
                    else:
                        grad[i] += a[0] * w[0]

        # Project to simplex: gradient descent in log-space
        # minimize sum_k w_k * conv_k = a^T S a (quadratic form)
        # In log-space: f(x) = exp(x)^T S exp(x), grad_x = exp(x) * (2*S*exp(x) - mean(2*S*exp(x)))
        # Actually for the softmax parameterization p_i = exp(x_i) / sum(exp(x)):
        # The gradient of the objective w.r.t. x is: p * (grad_obj - p @ grad_obj)
        # where grad_obj is the gradient w.r.t. p
        
        # Since we're using p = softmax(x), and the objective is 2n*max(conv):
        # d(dobj)/dp_i = 2n/dp_i of max_k conv_k ≈ 2n * grad[i] (from smoothed version, ignoring the 2n/sum^2 factor since sum=1)
        # In log-space: dx_i = p_i * (grad[i] - sum_j p_j * grad[j])
        
        obj_grad = 2.0 * n * grad  # gradient of the objective w.r.t. a (with sum=1)
        # Actually, the objective is 2n * max(conv) / (sum a)^2. If sum a = 1, it's 2n * max(conv).
        # The gradient w.r.t. a_i of 2n * sum_k w_k * conv_k is 2n * grad[i] (as computed).
        # But we need to account for the constraint sum(a) = 1.
        # In the log-parametrization, this is handled automatically.
        
        gp = float(np.dot(a, obj_grad))
        grad_x = a * (obj_grad - gp)

        # Adam update
        m_buf = mu * m_buf + (1.0 - mu) * grad_x
        v_buf = nu * v_buf + (1.0 - nu) * (grad_x * grad_x)
        m_hat = m_buf / (1.0 - mu ** t_step)
        v_hat = v_buf / (1.0 - nu ** t_step)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x = x - x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        a = e / se

    final_f = score_fft(a)
    if final_f < best_f:
        best_f = final_f
        best_p = a.copy()
    return best_p, best_f


def subgradient_step(a, n_steps=100, lr=0.01):
    """
    Take steps in the direction of the exact subgradient (gradient at argmax lag).
    This handles the non-smoothness of the max directly.
    """
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    a = normalize_nonneg(a)
    
    for _ in range(n_steps):
        conv = conv_direct(a) if n <= 2048 else None
        if conv is None:
            L = next_power_of_two(2 * n - 1)
            P = np.fft.rfft(a, L)
            conv = np.fft.irfft(P * P, L)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        
        # Find all lags that achieve (or nearly achieve) the max
        maxc = float(conv.max())
        # Subgradient: gradient of conv_k* w.r.t. a
        k_star = int(np.argmax(conv))
        # d(conv_k*)/d(a_i) = a[k*-i] + a[i-k*] (with range checks)
        grad = np.zeros(n, dtype=np.float64)
        for i in range(n):
            ki = k_star - i
            val = 0.0
            if 0 <= ki < n:
                val += a[ki]
                if ki != i:
                    val += a[ki]  # the mirror term
            # Wait, let me redo: conv_k = sum_p a[p]*a[k-p]
            # d/d a_i = a[k-i] (from p=i) + a[k-i] (from p=k-i) if k-i != i
            #         = a[k-i] + (a[k-i] if k!=2i else 0)
            #         = a[k-i] * (2 if k != 2i else 1)
            ki = k_star - i
            if 0 <= ki < n:
                if k_star == 2 * i:
                    grad[i] = a[ki]
                else:
                    grad[i] = 2.0 * a[ki]
        
        # Also add a small regularization to keep things positive
        grad = grad + 1e-6 * a
        
        # Normalize the gradient
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn
        
        # Projected gradient step (stay in simplex)
        a_new = a - lr * grad
        # Project onto non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn > 1e-12:
            a_new = a_new / sn
        else:
            a_new = a.copy()
        
        # Accept if it improves
        sc_new = score_fft(a_new)
        sc_old = score_fft(a)
        if sc_new < sc_old:
            a = a_new
        else:
            # Dampen
            a_new = a - lr * 0.1 * grad
            a_new = np.maximum(a_new, 0.0)
            sn = a_new.sum()
            if sn > 1e-12:
                a_new = a_new / sn
            else:
                break
            a = a_new

    return a, score_fft(a)


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    return val if np.isfinite(val) else np.inf


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    print("t=0.0: Starting search, budget=%.0fs" % budget_s)

    # === Phase 0: Parametric scan (~60s) ===
    scan_deadline = t0 + min(60.0, 0.08 * budget_s)
    lengths = [96, 128, 160, 192, 256, 320, 384, 448, 512, 640, 768, 1024, 1280, 1536, 2048]
    
    for n in lengths:
        if time.monotonic() >= scan_deadline:
            break
        # Power law
        for alpha in [0.4, 0.45, 0.5, 0.52, 0.55, 0.6]:
            a = params_power(n, alpha)
            best_p, best_score = record_candidate(a, best_p, best_score, best_by_n)
        # Two-sided power
        for alpha in [0.45, 0.5, 0.55]:
            for beta in [0.0, 0.1, 0.2, 0.5]:
                a = params_power(n, alpha, beta)
                best_p, best_score = record_candidate(a, best_p, best_score, best_by_n)
            for gamma in [0.0, 0.5, 1.0, 2.0, 5.0]:
                a = params_power(n, alpha, gamma=gamma)
                best_p, best_score = record_candidate(a, best_p, best_score, best_by_n)
        # Sinc window
        for win_beta in [0.0, 0.25, 0.5, 1.0, 2.0]:
            a = params_sinc_window(n, 0.5, win_beta)
            best_p, best_score = record_candidate(a, best_p, best_score, best_by_n)
        # Greedy
        a = params_greedy(n)
        best_p, best_score = record_candidate(a, best_p, best_score, best_by_n)

    # Try the global initial sequence if available
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size and arr.sum() > 0.01:
                best_p, best_score = record_candidate(arr, best_p, best_score, best_by_n)
                print("  Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    print("t=%.1f: Phase 0 done, best=%.6f (n=%d)" % (
        time.monotonic() - t0, best_score, 
        int(np.asarray(best_p).size) if best_p is not None else 0))

    # === Phase 1: Coordinate descent from best candidates (~500s) ===
    cd_deadline = t0 + min(560.0, 0.32 * budget_s)
    
    # Get top candidates
    items = []
    for n, rec in best_by_n.items():
        items.append((rec[0], int(n), rec[1]))
    items.sort(key=lambda x: x[0])
    
    # Also add the global best
    top_candidates = []
    seen_ns = set()
    for sc, n, p in items:
        if n not in seen_ns:
            top_candidates.append(p)
            seen_ns.add(n)
        if len(top_candidates) >= 5:
            break
    if best_p is not None and best_p.size not in seen_ns:
        top_candidates.append(best_p)

    for idx, cand in enumerate(top_candidates):
        if time.monotonic() >= cd_deadline:
            break
        rem = cd_deadline - time.monotonic()
        if rem < 30:
            break
        alloc = min(120.0, rem * 0.6)
        a_out, sc_out = coordinate_descent(cand, alloc, rng=rng)
        best_p, best_score = record_candidate(a_out, best_p, best_score, best_by_n, sc_out)
        print("  CD cand %d (n=%d): score=%.6f" % (idx, a_out.size, sc_out))

    print("t=%.1f: Phase 1 done, best=%.6f" % (time.monotonic() - t0, best_score))

    # === Phase 2: Adam optimization (~600s) ===
    adam_deadline = t0 + min(1160.0, 0.64 * budget_s)
    
    # Prepare Adam starting points
    adam_starts = []
    if best_p is not None:
        adam_starts.append(best_p)
    for sc, n, p in items[:3]:
        if p.size not in [x.size for x in adam_starts]:
            adam_starts.append(p)
    
    # Add some diverse starts
    for n in [256, 512, 1024]:
        a = params_power(n, 0.5, gamma=1.0)
        # Add random perturbation
        noise = np.exp(rng.normal(0, 0.1, n))
        a = normalize_nonneg(a * noise)
        adam_starts.append(a)

    for idx, start in enumerate(adam_starts):
        if time.monotonic() >= adam_deadline:
            break
        rem = adam_deadline - time.monotonic()
        if rem < 40:
            break
        alloc = min(180.0, rem * 0.5)
        # Vary temperature
        ts = 6.0 + 4.0 * (idx % 3)
        te = 1.5 + 0.5 * (idx % 2)
        a_out, sc_out = adam_optimize(start, alloc, seed=int(seed) + idx * 100,
                                       temp_start=ts, temp_end=te)
        best_p, best_score = record_candidate(a_out, best_p, best_score, best_by_n, sc_out)
        print("  Adam cand %d (n=%d): score=%.6f" % (idx, a_out.size, sc_out))

    print("t=%.1f: Phase 2 done, best=%.6f" % (time.monotonic() - t0, best_score))

    # === Phase 3: Subgradient + CD polish (~400s) ===
    polish_deadline = t0 + min(1600.0, 0.89 * budget_s)
    
    # Apply subgradient steps to the best
    if best_p is not None:
        rem = polish_deadline - time.monotonic()
        if rem > 60:
            a_sub, sc_sub = subgradient_step(best_p, n_steps=200, lr=0.005)
            best_p, best_score = record_candidate(a_sub, best_p, best_score, best_by_n, sc_sub)
            print("  Subgradient: score=%.6f" % sc_sub)
        
        # Coordinate descent polish
        rem = polish_deadline - time.monotonic()
        if rem > 60 and best_p.size <= 2048:
            alloc = min(rem - 20, 300.0)
            a_cd, sc_cd = coordinate_descent(best_p, alloc, rng=rng)
            best_p, best_score = record_candidate(a_cd, best_p, best_score, best_by_n, sc_cd)
            print("  CD polish: score=%.6f" % sc_cd)
        
        # More Adam with low temperature
        rem = polish_deadline - time.monotonic()
        if rem > 40:
            alloc = min(rem - 10, 150.0)
            a_ad, sc_ad = adam_optimize(best_p, alloc, seed=int(seed) + 9999,
                                         temp_start=3.0, temp_end=1.0)
            best_p, best_score = record_candidate(a_ad, best_p, best_score, best_by_n, sc_ad)
            print("  Adam polish: score=%.6f" % sc_ad)
        
        # Final CD
        rem = polish_deadline - time.monotonic()
        if rem > 30 and best_p.size <= 2048:
            alloc = min(rem - 5, 100.0)
            a_cd2, sc_cd2 = coordinate_descent(best_p, alloc, rng=rng)
            best_p, best_score = record_candidate(a_cd2, best_p, best_score, best_by_n, sc_cd2)
            print("  Final CD: score=%.6f" % sc_cd2)

    print("t=%.1f: Phase 3 done, best=%.6f (n=%d)" % (
        time.monotonic() - t0, best_score,
        int(np.asarray(best_p).size) if best_p is not None else 0))

    # === Phase 4: Diversity / restart if stuck ===
    if best_score > 1.505 and time.monotonic() < deadline - 100:
        # Try a few more diverse approaches
        for n_try in [384, 512, 768]:
            if time.monotonic() >= deadline - 80:
                break
            # Randomized restart
            start = params_power(n_try, 0.5)
            noise = np.exp(rng.normal(0, 0.3, n_try))
            start = normalize_nonneg(start * noise)
            
            a_cd3, sc3 = coordinate_descent(start, min(80.0, deadline - time.monotonic() - 30), rng=rng)
            best_p, best_score = record_candidate(a_cd3, best_p, best_score, best_by_n, sc3)
            if best_score < 1.505:
                break

    # === Final validation ===
    if best_p is None:
        best_p = params_greedy(384)
    
    final_seq = scale_output(best_p, 10.0)
    
    # Validate with the global evaluator
    gscore = try_global_eval(final_seq)
    if np.isfinite(gscore):
        print("  Global eval: %.6f" % gscore)
    
    # Also try a few other scalings
    for ts in [1.0, 5.0, 50.0]:
        alt_seq = scale_output(best_p, ts)
        alt_score = try_global_eval(alt_seq)
        if np.isfinite(alt_score):
            gscore2 = gscore if np.isfinite(gscore) else np.inf
            if alt_score < gscore2:
                final_seq = alt_seq
                gscore = alt_score
    
    print("t=%.1f: Final score=%.6f, n=%d" % (
        time.monotonic() - t0, 
        gscore if np.isfinite(gscore) else best_score,
        len(final_seq)))
    
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        # Fallback: greedy sequence
        a = params_greedy(384)
        return scale_output(a, 10.0)
