def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)

    if not np.isfinite(s) or s <= 1e-12:

def greedy_normalized(n):

def compute_conv_weights(p, alpha):
    """Compute softmax weights on the autocorrelation with temperature alpha."""
    n = p.size
    L = next_power_of_two(2 * n - 1)
    Pg = np.fft.rfft(p, L)
    c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(c.max())
    if maxc <= 0 or not np.isfinite(maxc):
        return c, np.zeros(2 * n - 1), 0.0
    z = alpha * (c / maxc - 1.0)
    expz = np.exp(z)
    sz = float(expz.sum())
    if not np.isfinite(sz) or sz <= 0:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sz
    return c, w, maxc


def adam_optimize(p, budget_s, seed=0, initial_alpha=8.0, final_alpha=80.0):
    """
    Adaptive-temperature softmax-projected gradient descent.
    Uses log-sum-exp reparameterization for stability.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.06
    elif n < 256:
        lr = 0.035
    elif n < 1024:
        lr = 0.02
    else:
        lr = 0.012

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    stall_count = 0
    stall_threshold = 200

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Adaptive alpha: start moderate, tighten over time
        alpha = initial_alpha + (final_alpha - initial_alpha) * (1.0 - np.exp(-3.0 * frac))

        # Compute autocorrelation and weights
        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)  # p sums to 1
        if np.isfinite(f_now) and f_now < best_f:
            if best_f - f_now > 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            best_f = f_now
            best_p = p.copy()
        else:
            stall_count += 1

        # If stalled for too long, increase alpha or break
        if stall_count > stall_threshold:
            if alpha < 150.0:
                initial_alpha = alpha
                final_alpha = min(200.0, alpha * 2.0)
                alpha *= 1.5
                stall_count = 0
            else:
                break

        # Compute softmax weights
        z = alpha * (c / maxc - 1.0)
        z -= z.max()  # numerical stability
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient: grad_x[i] = p[i] * (grad_p[i] - mean(p*grad_p))
        # grad_p = 2*n * convolve(p, w_reversed)[n-1:2n-1][::-1]
        Wr = w[::-1]
        W = np.fft.rfft(Wr, L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Skip if gradient is essentially zero
        gnorm = float(np.sqrt(np.dot(grad_x, grad_x)))
        if gnorm < 1e-12:
            break

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f

    Cutting-plane LP: at each iteration, solve
      maximize sum(g)
      subject to (f*g)[k] <= M, g >= 0

    """

        return p, actual_score_fft(p)

    n = p.size
    if n < 2:
        n = 16
        p = normalize_nonneg(np.ones(n, dtype=np.float64))
    budget_s = float(budget_s)
    if budget_s < 20.0:
        return p, actual_score_fft(p)

    if max_rows is None:
        if n > 4000:
            max_rows = 1200
        elif n > 1500:
            max_rows = 1000
        else:
            max_rows = 800

    # Limit dense matrix size
    max_elements = 8000000
    while max_rows * n > max_elements:
        max_rows = max(128, int(max_rows * 0.8))
    max_rows = max(128, min(int(max_rows), max_elements // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_F = actual_score_fft(f)
    if not np.isfinite(cur_F):
        return p, cur_F

    best_f = f.copy()
    best_score = float(cur_F)
    M = float(cur_F)  # This is the max autocorrelation value (since sum=f.sum()=S0)
    # Actually M = max(f*f) = cur_F * sum(f)^2 / (2*n) = cur_F * 2n / (2n) = cur_F
    # Wait: cur_F = 2*n*max(f*f)/(sum(f))^2 = 2*n*max(f*f)/(2n) = max(f*f)
    # So M = max(f*f) = cur_F when sum(f) = sqrt(2n)
    M = float(cur_F)

    # Initialize constraint set with top autocorrelation positions
    c_full = np.convolve(f, f) if n <= 1500 else None
    if c_full is None:
        L2 = next_power_of_two(2 * n - 1)
        Pf = np.fft.rfft(f, L2)
        c_full = np.fft.irfft(Pf * Pf, L2)[:2 * n - 1].real
    c_full = np.maximum(c_full, 0.0)

    # Start with top 40% of autocorrelation positions
    n_initial = max(100, min(max_rows // 2, int(0.4 * (2 * n - 1))))
    top_idx = np.argsort(c_full)[::-1][:n_initial]
    rows = [int(k) for k in top_idx.tolist()]
    row_set = set(rows)

    end = time.monotonic() + max(0.0, budget_s - 3.0)
    it = 0
    lp_call_count = 0

    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        nr = int(rows_arr.size)

        # Build constraint matrix A (nr x n)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            # A[ii, j] = f[k-j] for valid j (0 <= k-j < n)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                A[ii, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]

        b = np.full(nr, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(5.0, min(120.0, rem * 0.3))
        lp_call_count += 1

        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl), "dual_feasibility_tolerance": 1e-8,
                         "primal_feasibility_tolerance": 1e-8}
            )
        except Exception:
            try:
                res = _scipy_opt.linprog(
                    cobj, A_ub=A, b_ub=b,
                    bounds=(0.0, None), method="highs"
                )
            except Exception:
                break

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)

            L2 = next_power_of_two(2 * n - 1)

        else:

        conv = np.maximum(conv, 0.0)
        tol = 1e-6 + 1e-6 * M

            # Add violated constraints
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= (max_rows - len(rows)) or added >= 200:
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added == 0 and float(np.max(violations)) > 10.0 * tol:
                    break
                continue
            else:
                # At max rows, check if violations are severe
                if float(np.max(violations)) > 5.0 * tol:
                    break
                # Otherwise proceed with current solution (slightly infeasible)
                g = np.clip(g, 0.0, None)
                sg = float(g.sum())
                if sg <= 1e-12:
                    break

        # Rescale g to match normalization
        g = g * (S0 / sg)

        # Line search: try different step sizes
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.10, 0.05, 0.02):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score):
            break
        if accepted_score >= best_score * (1.0 - 1e-10):
            break

        f = accepted
        M = float(accepted_score)
        best_score = accepted_score
        best_f = f.copy()

        if it % 3 == 0:
            print("  LP iter=%d score=%.6f rows=%d" % (it, best_score, len(rows)))

        # If improvement is very small, stop
        if best_score < 1.48:
            break

    final_score = actual_score_fft(best_f)
    if final_score < best_score:
        best_score = float(final_score)
    return normalize_nonneg(best_f), best_score


def random_perturb(p, rng, sigma):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return greedy_normalized(16)
    noise = np.exp(rng.normal(0.0, float(sigma), p.size))
    return normalize_nonneg(p * noise)


def prepare_start(n, kind, rng):
    n = int(n)
    if kind == "greedy":
        return greedy_normalized(n)
    if kind == "power45":
        return power_normalized(n, 0.45)
    if kind == "power55":
        return power_normalized(n, 0.55)
    if kind == "power30":
        return power_normalized(n, 0.30)
    if kind == "mix25":
        p = 0.25 * greedy_normalized(n) + 0.75 * np.ones(n, dtype=np.float64)
        return normalize_nonneg(p)
    if kind == "mix50":
        p = 0.5 * greedy_normalized(n) + 0.5 * np.ones(n, dtype=np.float64)
        return normalize_nonneg(p)
    if kind == "mix75":
        p = 0.75 * greedy_normalized(n) + 0.25 * np.ones(n, dtype=np.float64)
        return normalize_nonneg(p)
    if kind == "pert05":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.05, n))
        return normalize_nonneg(p * noise)
    if kind == "pert15":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.15, n))
        return normalize_nonneg(p * noise)
    if kind == "pert30":
        p = greedy_normalized(n)
        noise = np.exp(rng.normal(0.0, 0.30, n))
        return normalize_nonneg(p * noise)
    if kind == "dirichlet":
        try:
            return normalize_nonneg(rng.dirichlet(np.ones(n, dtype=np.float64)))
        except Exception:
            return greedy_normalized(n)
    return greedy_normalized(n)


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_state = {'p': None, 'score': np.inf}
    best_by_n = {}

    # Load prior best if available
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                best_state = record_candidate(arr, best_state, best_by_n)
                print("Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, best_state['score']))
        except Exception:
            pass

    # Phase 1: Quick baseline evaluation
    print("Phase 1: Baselines...")
    base_lengths = [64, 96, 128, 192, 256, 384, 512, 768, 1024, 1536, 2048, 3072, 4096]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for kind in ["greedy", "power45", "power55", "power30"]:
            p = prepare_start(n, kind, rng)
            best_state = record_candidate(p, best_state, best_by_n)

    print("  Best after baselines: %.6f (n=%d)" % (best_state['score'],
          best_state['p'].size if best_state['p'] is not None else 0))

    # Phase 2: Adam optimization on diverse starts
    print("Phase 2: Adam optimization...")
    jobs = []
    # Prioritize lengths near the current best
    if best_state['p'] is not None:
        bn = best_state['p'].size
        # Add the best length and neighbors
        for dn in [0, -64, 64, -128, 128, -256, 256]:
            candidate_n = bn + dn
            if 32 <= candidate_n <= 6000:
                jobs.append((candidate_n, "best", 0))

    # Standard schedule
    standard_jobs = [
        (128, "greedy", 40),
        (192, "greedy", 50),
        (256, "greedy", 60),
        (384, "greedy", 70),
        (512, "greedy", 70),
        (768, "greedy", 80),
        (1024, "greedy", 80),
        (1536, "greedy", 70),
        (2048, "greedy", 80),

        (256, "pert15", 50),
        (256, "mix50", 50),
        (384, "pert30", 60),
        (384, "mix75", 60),
        (512, "pert15", 70),
        (512, "mix50", 70),
        (512, "power45", 70),
        (768, "pert30", 60),
        (768, "mix50", 60),
        (1024, "pert15", 70),
        (1024, "mix75", 70),
        (1536, "pert30", 60),
        (2048, "pert15", 70),
    ]
    jobs.extend(standard_jobs)

    # Deduplicate (keep first occurrence)
    seen = set()
    deduped = []
    for item in jobs:
        key = (item[0], item[1])
        if key not in seen:
            seen.add(key)
            deduped.append(item)
    jobs = deduped

    for n, kind, alloc in jobs:
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        if kind == "best":
            rec = best_by_n.get(int(n))
            if rec is not None:
                base = rec[1]
            else:
                base = greedy_normalized(n)
        else:
            base = prepare_start(n, kind, rng)
        alloc_use = min(float(alloc), max(10.0, rem * 0.15))
        p_out, f_out = adam_optimize(base, alloc_use, seed=int(seed) + int(n) * 7)
        best_state = record_candidate(p_out, best_state, best_by_n, f_out)
        if int(n) % 512 == 0:
            print("  Adam n=%d: %.6f (best: %.6f)" % (n, f_out, best_state['score']))

    # Phase 3: LP refinement on top candidates
    print("Phase 3: LP refinement...")
    if _scipy_opt is not None:
        # Sort candidates by score, try top ones
        items = []
        for n, rec in best_by_n.items():
            items.append((rec[0], int(n), rec[1]))
        items.sort(key=lambda x: x[0])

        lp_done = set()
        for sc, n, p in items:
            rem = deadline - time.monotonic()
            if rem < 60.0:
                break
            if n in lp_done:
                continue
            if n > 5000:
                continue
            lp_done.add(n)
            alloc = min(300.0, max(60.0, rem * 0.35))
            print("  LP n=%d (current score %.6f), alloc=%.0fs" % (n, sc, alloc))
            p_out, f_out = lp_improve(p, alloc)
            best_state = record_candidate(p_out, best_state, best_by_n, f_out)
            print("  LP n=%d done: %.6f (best: %.6f)" % (n, f_out, best_state['score']))
            if best_state['score'] < 1.49:
                print("  Breaking LP phase: score < 1.49")
                break
    else:
        # Fallback: more Adam iterations
        while deadline - time.monotonic() > 100.0:
            if best_state['p'] is not None:
                base = random_perturb(best_state['p'], rng, 0.1)
            else:
                base = greedy_normalized(256)
            alloc = min(200.0, max(30.0, (deadline - time.monotonic()) * 0.4))
            p_out, f_out = adam_optimize(base, alloc, seed=int(seed) + int(time.monotonic() * 100) % 10000)
            best_state = record_candidate(p_out, best_state, best_by_n, f_out)

    # Phase 4: Final polishing - try LP on the absolute best, then Adam
    print("Phase 4: Final polishing...")
    if best_state['p'] is not None:
        rem = deadline - time.monotonic()
        if rem > 80.0 and _scipy_opt is not None:
            n_best = best_state['p'].size
            alloc = min(200.0, rem * 0.6)
            p_out, f_out = lp_improve(best_state['p'], alloc)
            best_state = record_candidate(p_out, best_state, best_by_n, f_out)
            print("  Final LP: %.6f" % best_state['score'])

        rem = deadline - time.monotonic()
        if rem > 30.0:
            p0 = random_perturb(best_state['p'], rng, 0.05)
            alloc = min(120.0, rem - 5.0)
            p_out, f_out = adam_optimize(p0, alloc, seed=int(seed) + 77777,
                                         initial_alpha=15.0, final_alpha=120.0)
            best_state = record_candidate(p_out, best_state, best_by_n, f_out)
            print("  Final Adam (perturbed): %.6f" % best_state['score'])

        rem = deadline - time.monotonic()
        if rem > 20.0:
            p0 = normalize_nonneg(best_state['p'])
            alloc = min(80.0, rem - 5.0)
            p_out, f_out = adam_optimize(p0, alloc, seed=int(seed) + 99999,
                                         initial_alpha=20.0, final_alpha=150.0)
            best_state = record_candidate(p_out, best_state, best_by_n, f_out)
            print("  Final Adam (direct): %.6f" % best_state['score'])

    if best_state['p'] is None:
        best_state['p'] = greedy_normalized(256)
        best_state['score'] = actual_score_fft(best_state['p'])

    print("Final best: %.6f (n=%d)" % (best_state['score'], best_state['p'].size))

    # Validate with global evaluator
    candidates_to_check = [best_state['p']]
    items = []
    for n, rec in best_by_n.items():
        items.append((rec[0], int(n), rec[1]))
    items.sort(key=lambda x: x[0])
    for sc, n, p in items[:3]:
        if p is not None and p is not best_state['p']:
            candidates_to_check.append(p)

    for cand in candidates_to_check:
        if deadline - time.monotonic() < 10.0:
            break
        seq = scale_output(cand, 10.0)
        gscore = try_global_eval(seq)
        if np.isfinite(gscore) and gscore < best_state['score']:
            best_state['score'] = float(gscore)
            best_state['p'] = np.asarray(cand, dtype=np.float64).ravel()
            print("  Global eval improved to: %.6f" % best_state['score'])

    # Also check a scaled-up version (sometimes helps with discrete effects)
    for target in [5.0, 10.0, 20.0, 50.0]:
        if deadline - time.monotonic() < 10.0:
            break
        seq = scale_output(best_state['p'], target)
        gscore = try_global_eval(seq)
        if np.isfinite(gscore) and gscore < best_state['score']:
            best_state['score'] = float(gscore)
            print("  scale=%.1f improved to: %.6f" % (target, best_state['score']))

    final_seq = scale_output(best_state['p'], 10.0)
    print("Returning sequence of length %d" % len(final_seq))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Exception in main_search: %s" % str(e))
        # Fallback
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
