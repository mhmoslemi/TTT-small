import numpy as np
import time

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 without external calls."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * np.max(conv) / (total * total))

def simplex_project(v):
    """Project v onto {s >= 0, sum(s) = 1} using the Euclidean projection."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Use the algorithm from the "Euclidean projection onto the simplex" paper
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    k = np.where(cond)[0]
    if len(k) == 0:
        return np.ones(n) / n
    k = k[-1] + 1  # 1-indexed
    theta = (cssv[k-1]) / k
    return np.maximum(v - theta, 0)

def compute_gradient_fft(s, p):
    """Compute grad[i] = 2 * sum_k p[k] * s[k-i] using FFT.
    s has length n, p has length 2n-1."""
    n = len(s)
    N = 1
    while N < (2*n - 1) + n - 1:
        N *= 2
    
    S = np.fft.rfft(s, N)
    P = np.fft.rfft(p, N)
    
    # cross-correlation: ifft(P * conj(S))
    corr = np.fft.irfft(P * np.conj(S), N)
    
    return 2.0 * corr[:n]

                             max_step=0.1, rng=None):

    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)
    
    total_iters = num_phases * iters_per_phase
    tau_schedule = np.geomspace(tau_start, tau_end, total_iters)
    
    step = max_step
    no_improve = 0
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        tau = tau_schedule[it]
        conv = np.convolve(s, s)
        
        # Softmax weights
        conv_shifted = conv / tau
        conv_shifted -= conv_shifted.max()  # for numerical stability
        p = np.exp(conv_shifted)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.ones_like(conv) / len(conv)
        else:
            p /= p_sum
        
        # Gradient
        grad = compute_gradient_fft(s, p)
        
        # Gradient descent step
        s_new = s - step * grad
        s_new = simplex_project(s_new)
        
        # Backtracking line search
        score_new = fast_score(s_new)
        score_cur = fast_score(s)
        
        tries = 0
        while score_new >= score_cur - 1e-12 and tries < 10:
            step *= 0.5
            s_new = s - step * grad
            s_new = simplex_project(s_new)
            score_new = fast_score(s_new)
            tries += 1
        
        if score_new < score_cur - 1e-12:
            s = s_new
            step = min(step * 1.5, max_step)
            no_improve = 0
        else:
            step *= 0.5
            no_improve += 1
            if no_improve > 30:
                break

def peak_shave(s, deadline, num_iters=100, rng=None):
    """Peak shaving: reduce coordinates contributing most to the peak."""
    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)
    
    for it in range(num_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(s, s)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        
        # Responsibility: r[i] = 2 * s[max_idx - i] (with zero-padding)
        r = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                r[i] = 2 * s[j]
        
        # Reduce top contributors
        top_k = max(1, n // 10)
        top_idx = np.argsort(-r)[:top_k]
        
        # Reduce them
        reduction = 0.01 * s[top_idx].max()
        s[top_idx] -= reduction
        s = simplex_project(s)
        
        score = fast_score(s)
        if score < best_score:
            best_score = score
            best_s = s.copy()
    
    return best_s, best_score

"""
Hybrid optimizer for minimizing 2n*max(conv(s,s))/(sum(s))^2 over non-negative sequences.
Combines: (1) projected gradient descent with log-sum-exp smoothing, (2) peak shaving
via mass transfer, (3) multi-scale tau annealing, (4) multiple initializations and lengths.

import numpy as np
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 without external calls."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return float(2.0 * n * np.max(conv) / (total * total))


def simplex_project(v):
    """Project v onto {s >= 0, sum(s) = 1} via Euclidean projection."""
    n = len(v)
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    k = idx[-1] + 1
    theta = cssv[k - 1] / k
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(s, p):
    """Compute grad[i] = 2 * sum_k p[k] * s[k-i] using FFT (linear cross-correlation)."""
    n = len(s)
    m = len(p)
    N = 1
    while N < m + n - 1:
        N *= 2
    S = np.fft.fft(s, N)
    P = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(S) * P)
    return 2.0 * np.real(corr[:n])


def smooth_gradient_descent(s, deadline, tau_start=0.5, tau_end=0.005,
                            total_iters=100, max_step=0.5, verbose=False):
    """Projected gradient descent with log-sum-exp smoothing of the peak convolution."""
    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)

    tau_schedule = np.geomspace(tau_start, tau_end, total_iters)
    step = max_step
    no_improve = 0

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break

        tau = tau_schedule[it]
        conv = np.convolve(s, s)

        # Softmax weights (numerically stable)
        conv_s = conv / tau
        conv_s -= conv_s.max()
        exp_vals = np.exp(conv_s)
        p = exp_vals / exp_vals.sum()

        # Gradient
        grad = compute_gradient_fft(s, p)

        # Gradient descent step with backtracking
        s_new = simplex_project(s - step * grad)
        score_new = fast_score(s_new)
        score_cur = fast_score(s)

        backtracks = 0
        while score_new >= score_cur - 1e-13 and backtracks < 12:
            step *= 0.5
            s_new = simplex_project(s - step * grad)
            score_new = fast_score(s_new)
            backtracks += 1

        if score_new < score_cur - 1e-13:
            s = s_new
            step = min(step * 1.3, max_step * 2.0)
            no_improve = 0
        else:
            step *= 0.5
            no_improve += 1
            if no_improve > 40:
                break

        if score_new < best_score:
            best_score = score_new
            best_s = s.copy()
            if verbose and it % 20 == 0:
                print(f"    it={it}, tau={tau:.4f}, score={best_score:.6f}")

    return best_s, best_score


def peak_shave(s, deadline, num_iters=500, verbose=False):
    """Peak shaving: transfer mass from peak contributors to low-impact coordinates."""
    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)
    no_improve = 0

    for it in range(num_iters):

        conv = np.convolve(s, s)

        i_lo = max(0, k_star - n + 1)
        i_hi = min(n, k_star + 1)
        contribs = np.array([s[i] * s[k_star - i] for i in range(i_lo, i_hi)])
        i_max = i_lo + int(np.argmax(contribs))
        j_max = k_star - i_max

        if eps > 1e-12:

                s[j_max] -= eps
                s[i_inc] += 2.0 * eps
            else:
                s[i_inc] += eps
            s = simplex_project(s)

        score = fast_score(s)
        if score < best_score - 1e-12:
            best_score = score
            best_s = s.copy()

            no_improve += 1
            if no_improve > 50:

                no_improve = 0
                if no_improve > 100:

def active_set_gradient(s, deadline, total_iters=200, num_active=5, max_step=0.3):
    """Gradient descent using only the top convolution peaks (active set)."""
    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)
    step = max_step

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        # Use top num_active peaks
        top_idx = np.argsort(conv)[::-1][:num_active]
        p = np.zeros_like(conv)
        for k in top_idx:
            p[k] = 1.0 / num_active

        grad = compute_gradient_fft(s, p)
        s_new = simplex_project(s - step * grad)
        score_new = fast_score(s_new)
        score_cur = fast_score(s)

        backtracks = 0
        while score_new >= score_cur - 1e-13 and backtracks < 10:
            step *= 0.5

            backtracks += 1

            s = s_new
            step = min(step * 1.2, max_step * 1.5)
        else:
            step *= 0.5

        if score_new < best_score:
            best_score = score_new

def spectral_flatten(s, deadline, num_iters=200, alpha=0.3):
    """Spectral flattening: iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(s)
    s = simplex_project(s.copy())
    best_s = s.copy()
    best_score = fast_score(s)

    N = 1
    while N < 2 * n - 1:
        N *= 2

    for it in range(num_iters):
        if time.monotonic() > deadline:
            break

        S = np.fft.fft(s, N)
        mag_sq = np.abs(S) ** 2
        target = mag_sq.mean()

        # Move magnitude toward flat
        mag = np.sqrt(mag_sq)
        mag_new = mag * (1.0 - alpha) + np.sqrt(target) * alpha

        # Reconstruct with same phase
        phase = np.angle(S)
        S_new = mag_new * np.exp(1j * phase)
        s_new = np.real(np.fft.ifft(S_new))
        s_new = np.maximum(s_new, 0.0)
        if s_new.sum() < 1e-10:
            break
        s_new = s_new / s_new.sum()

        score = fast_score(s_new)
        if score < best_score:
            best_score = score
            best_s = s_new.copy()
            s = s_new
        else:
            # Try smaller alpha
            alpha *= 0.9
            if alpha < 0.01:
                break

    return best_s, best_score

                total_iters=int(50 * time_per_cand / 5.0) + 30,
                num_active=3, max_step=0.3
            )

            # Phase 4: Peak shaving
            s4, sc4 = peak_shave(
                s3, deadline - 20,
                num_iters=int(200 * time_per_cand / 5.0) + 100
            )

            # Phase 5: Final gradient polish
            s5, sc5 = smooth_gradient_descent(
                s4, deadline - 20,
                tau_start=0.01, tau_end=0.002,
                total_iters=50, max_step=0.2
            )

            # Choose best from all phases
            best_local = min(
                [(sc1, s1), (sc2, s2), (sc3, s3), (sc4, s4), (sc5, s5)],
                key=lambda x: x[0]
            )

            if best_local[0] < global_best_score:
                global_best_score = best_local[0]
                global_best_seq = best_local[1].copy()
                print(f"  ** New global best: {global_best_score:.6f} (n={n}, phase took {time.monotonic()-t0:.1f}s)")

            # Also try spectral flattening from the best so far
            if time.monotonic() < deadline - 40:
                s_spec, sc_spec = spectral_flatten(s5, deadline - 20, num_iters=100, alpha=0.2)
                if sc_spec < global_best_score:
                    global_best_score = sc_spec
                    global_best_seq = s_spec.copy()
                    print(f"  ** Spectral best: {global_best_score:.6f}")

    # Additional random restarts with the best algorithm if time permits
    if time.monotonic() < deadline - 30:
        n_best = len(global_best_seq)
        print(f"\n=== Random restarts with n={n_best} ===")
        for restart in range(10):
            if time.monotonic() > deadline - 15:
                break
            s_rand = simplex_project(rng.random(n_best) + 0.01)
            s_opt, sc_opt = smooth_gradient_descent(
                s_rand, deadline - 15,
                tau_start=0.5, tau_end=0.01,
                total_iters=80, max_step=0.5
            )
            if sc_opt < global_best_score:
                global_best_score = sc_opt
                global_best_seq = s_opt.copy()
                print(f"  ** Restart {restart+1}: {sc_opt:.6f}")

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_score = float(evaluate_sequence(result))
    print(f"\nFinal verified score: {final_score:.6f}, length={len(result)}")

    return result
