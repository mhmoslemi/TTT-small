def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    sv = np.sort(v)[::-1]
    cssv = np.cumsum(sv)
    cond = sv * np.arange(1, n + 1) > (cssv - target)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_idx[-1]
    lam = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - lam, 0.0)


def _fft_len(n):
    """Next power of 2 >= n."""
    m = 1
    while m < n:
        m <<= 1
    return m


def _conv_sq(a, m):
    """Linear convolution a*a via FFT. m must be >= 2*len(a)-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[: 2 * a.size - 1]


def _grad_topk(a, c, top_idx, weights):
    """Gradient of sum_k weights[k]*c[k] where c = a*a (convolution)."""
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for k, w in zip(top_idx, weights):
        if w < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        src = k - idx
        grad[idx] += w * a[src]
    return 2.0 * grad


def _fista_optimize(a_init, n_steps, m, tau0, lr0, L0, momentum0, t_deadline):
    """
    FISTA-style accelerated descent on smoothed max convolution.
    Returns (best_a, best_max_c, best_a_fista, buf, t_nest).
    """
    n = a_init.size
    a = _proj_simplex(a_init.copy(), 1.0)
    a_prev = a.copy()
    buf = np.zeros(n, dtype=np.float64)
    y = a.copy()  # FISTA extrapolation point
    t_nest = 1.0

    best_a = a.copy()
    best_mc = float('inf')

    tau = tau0
    lr = lr0
    L = L0
    momentum = momentum0

    for step in range(n_steps):
        if time.monotonic() > t_deadline:
            break

        # Evaluate at extrapolated point y
        c_y = _conv_sq(y, m)
        mc_y = c_y.max()

        # Get top-L indices from c_y
        conv_len = 2 * n - 1
        if L >= conv_len:
            top_idx = np.arange(conv_len)
        else:
            top_idx = np.argpartition(-c_y, L - 1)[:L]

        # Softmax weights over top-L
        top_c = c_y[top_idx]
        max_top = top_c.max()
        exp_vals = np.exp((top_c - max_top) / tau)
        weights = exp_vals / exp_vals.sum()

        # Gradient at y
        g = _grad_topk(y, c_y, top_idx, weights)
        gn = np.linalg.norm(g)
        if gn > 1e-15:
            g = g / gn

        # FISTA step
        a_new = _proj_simplex(y - lr * g, 1.0)

        # Update FISTA parameter
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = a_new + ((t_nest - 1.0) / t_new) * (a_new - a_prev)
        # Ensure y stays in a reasonable range
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0:
            y = y / y_sum  # keep y on simplex for stability

        a_prev = a
        a = a_new
        t_nest = t_new

        # Track best (using actual a, not y)
        c_a = _conv_sq(a, m)
        mc_a = c_a.max()
        if mc_a < best_mc:
            best_mc = mc_a
            best_a = a.copy()

        # Adaptive schedule
        frac = step / max(1, n_steps)
        if step % 200 == 0 and step > 0:
            tau = max(0.005 / n, tau * 0.65)
            L = max(5, int(L * 0.75))
            lr = max(1e-7, lr * 0.88)
            momentum = min(0.99, momentum + 0.003)

    return best_a, best_mc, a, buf, t_nest


def _exact_subgrad_polish(a_init, n_steps, m, t_deadline):
    """Polish using exact subgradient (pick argmax lag, step, project)."""
    n = a_init.size
    a = _proj_simplex(a_init.copy(), 1.0)
    buf = np.zeros(n, dtype=np.float64)
    best_a = a.copy()
    best_mc = _conv_sq(a, m).max()
    lr = 0.05
    momentum = 0.9

    for step in range(n_steps):
        if time.monotonic() > t_deadline:
            break

        c = _conv_sq(a, m)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()

        # Use top-5 lags with uniform weight for a better subgradient
        k = np.argmax(c)
        # Also include a few neighbors of the argmax for stability
        top5 = np.argpartition(-c, 4)[:5]

        grad = np.zeros(n, dtype=np.float64)
        for kk in top5:
            lo = max(0, kk - n + 1)
            hi = min(kk, n - 1)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                src = kk - idx
                grad[idx] += a[src]
        grad *= 2.0 / 5.0

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn

        buf = momentum * buf - lr * grad
        a_new = a + lr * buf
        a = _proj_simplex(a_new, 1.0)

        if step % 100 == 0 and step > 0:
            lr = max(1e-7, lr * 0.9)

    return best_a, best_mc


def _make_profiles(n, rng):
    """Generate diverse initial profiles, all normalized to sum=1, non-negative."""
    x = np.arange(n, dtype=np.float64) / n
    profiles = []

    # Uniform
    profiles.append(np.ones(n))

    # Ramps
    profiles.append(np.linspace(0.05, 1.0, n))
    profiles.append(np.linspace(1.0, 0.05, n))

    # Sine bump (symmetric)
    profiles.append(np.sin(np.pi * (x + 0.5 / n)) ** 2)
    profiles.append(np.sin(np.pi * (x + 0.5 / n)) ** 0.5)

    # Gaussian bumps at different positions and widths
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            profiles.append(np.exp(-((x - center) ** 2) / (2 * width * width)))

    # Exponential decay (left-heavy)
    for alpha in [1.0, 2.0, 4.0, 8.0]:
        profiles.append(np.exp(-alpha * x))

    # Exponential decay (right-heavy)
    for alpha in [2.0, 4.0]:
        profiles.append(np.exp(alpha * (x - 1.0)))

    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(1.0 / (np.arange(n) + 1) ** p)

    # Two bumps
    profiles.append(np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2))
    profiles.append(np.exp(-30 * (x - 0.2) ** 2) + np.exp(-30 * (x - 0.8) ** 2))

    # Triangle
    profiles.append(1.0 - 2.0 * np.abs(x - 0.5))

    # Raised cosine
    profiles.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))

    # Quartic bump
    profiles.append((1.0 - (2.0 * x - 1.0) ** 2) ** 2)

    # Random
    profiles.append(rng.random(n))
    profiles.append(np.sort(rng.random(n)))
    profiles.append(np.sort(rng.random(n))[::-1])

    # Structured: alternating
    alt = np.zeros(n)
    alt[0::2] = 1.0
    profiles.append(alt)

    # Block: first half on
    block = np.zeros(n)
    block[: n // 2] = 1.0
    profiles.append(block)

    return [_proj_simplex(np.maximum(p, 0), 1.0) for p in profiles]

            if sc < best_score:
                best_score = sc
                best_a = ps.copy()
                print(f"Provided: n={provided_n}, score={sc:.6f}")
    except Exception:
        pass

    # Sequence lengths to try (larger n = more DOF = potentially lower objective)
    n_values = [1024, 2048, 4096, 8192]

    # If provided sequence exists, also try its length
    if provided_n in n_values:
        n_values.remove(provided_n)
    # Put provided_n first if it exists
    if provided is not None:
        n_values.insert(0, provided_n)

    total_rem = t_end - time.monotonic()
    per_n = total_rem / len(n_values)

    for n_idx, n in enumerate(n_values):
        if time.monotonic() > t_end - 30:
            break

        m = _fft_len(2 * n - 1)
        t_n_end = min(t_end - 30, time.monotonic() + per_n * 1.2)

        print(f"\n=== n={n} (FFT len={m}), time budget ~{per_n:.0f}s ===")

        # Generate profiles
        profiles = _make_profiles(n, rng)

        # Add resampled provided/best sequence
        ref = provided if provided is not None else best_a
        if ref is not None and ref.size != n:
            resampled = np.interp(
                np.linspace(0, 1, n),
                np.linspace(0, 1, ref.size),
                ref
            )
            profiles.insert(0, _proj_simplex(np.maximum(resampled, 0), 1.0))
        elif ref is not None and ref.size == n:
            profiles.insert(0, _proj_simplex(np.maximum(ref, 0), 1.0))

        # Add noisy variants of the best
        if best_a is not None:
            for noise_scale in [0.2, 0.5]:
                ref2 = best_a
                if ref2.size != n:
                    ref2 = np.interp(
                        np.linspace(0, 1, n),
                        np.linspace(0, 1, ref2.size),
                        ref2
                    )
                noisy = _proj_simplex(ref2, 1.0) * np.exp(
                    rng.normal(0, noise_scale, n)
                )
                profiles.append(_proj_simplex(noisy, 1.0))

        # Time per profile
        rem = t_n_end - time.monotonic()
        n_profiles = len(profiles)
        per_profile = rem / n_profiles

        print(f"  {n_profiles} profiles, ~{per_profile:.1f}s each")

        for pi, prof in enumerate(profiles):
            if time.monotonic() > t_n_end:
                break

            # Scale hyperparameters with n
            tau0 = 1.0 / n
            lr0 = min(0.15, 50.0 / n)
            L0 = min(500, max(20, n // 4))
            n_steps = max(300, int(per_profile * 80))  # ~80 steps/sec target

            # Phase 1: FISTA with smoothed max
            a1, mc1, _, _, _ = _fista_optimize(
                prof, n_steps, m, tau0, lr0, L0, 0.9,
                min(t_n_end, time.monotonic() + per_profile * 0.7)
            )
            score1 = 2.0 * n * mc1

            # Phase 2: Polish with exact subgradient
            a2, mc2 = _exact_subgrad_polish(
                a1, max(100, n_steps // 3), m,
                min(t_n_end, time.monotonic() + per_profile * 0.3)
            )
            score2 = 2.0 * n * mc2

            # Phase 3: More FISTA from polished point with lower tau
            a3, mc3, _, _, _ = _fista_optimize(
                a2, max(100, n_steps // 4), m,
                tau0 * 0.1, lr0 * 0.5, max(5, L0 // 4), 0.95,
                t_n_end
            )
            score3 = 2.0 * n * mc3

            # Pick the best from all phases
            best_local_score = min(score1, score2, score3)
            if score3 <= score2 and score3 <= score1:
                best_local_a = a3
            elif score2 <= score1:
                best_local_a = a2
            else:
                best_local_a = a1

            if best_local_score < best_score:
                best_score = best_local_score
                best_a = best_local_a.copy()
                print(f"  [{pi:2d}] NEW BEST: score={best_local_score:.6f} "
                      f"(tau phases: {score1:.6f}/{score2:.6f}/{score3:.6f})")

    # Final authoritative evaluation
    if best_a is not None:
        try:
            final_score = float(evaluate_sequence([float(x) for x in best_a]))
            print(f"\nFINAL authoritative: n={best_a.size}, score={final_score:.6f}")
        except Exception as e:
            print(f"Eval error: {e}")
            final_score = best_score

    # Validate and return
    out = np.clip(np.asarray(best_a, dtype=np.float64), 0.0, 1000.0) if best_a is not None else np.ones(512)
    if out.sum() < 0.01:
        out = np.ones(512)

    # Final sanity check
    try:
        s = float(evaluate_sequence([float(x) for x in out]))
        print(f"Returned sequence: n={out.size}, score={s:.6f}")
    except Exception:
        pass

    return [float(x) for x in out]
