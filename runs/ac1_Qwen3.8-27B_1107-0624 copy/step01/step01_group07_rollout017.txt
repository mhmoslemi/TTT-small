"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast multiplicative subgradient, periodic LP refinement,
and coordinate descent polishing with multiple initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar


def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def subgradient_step(a, lr, top_k=3):
    """One multiplicative subgradient step targeting top peaks."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    max_c = c.max()
    
    # Get top_k peaks
    if top_k < len(c):
        top_idx = np.argsort(c)[-top_k:]
    else:
        top_idx = np.where(c > 0.99 * max_c)[0]
    
    # Compute gradient contribution
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    
    max_g = g.max()
    if max_g < 1e-15:
        return a.copy()
    
    a_new = a * np.exp(-lr * g / max_g)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_refine(a, iterations=5):
    """Apply LP-based refinement for a few iterations."""
    n = len(a)
    if n > 2048:
        return a, compute_score(a)
    
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    for it in range(iterations):
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        if g.sum() < 1e-15:
            break
        
        # Try different step sizes
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.1]:
            a_new = (1.0 - t) * a + t * (g / g.sum())
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def coordinate_descent(a, max_rounds=3, rng=None):
    """Coordinate descent: optimize each element in 1D."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    
    for round_num in range(max_rounds):
        improved = False
        indices = list(range(n))
        if rng is not None:
            rng.shuffle(indices)
        
        for i in indices:
            if a[i] < 1e-15:
                # Try setting it to a small value
                lo, hi = 0.0, a.max() * 2
            else:
                lo = 0.0
                hi = a[i] * 3 + a.max()
            
            def obj(x):
                a_temp = a.copy()
                a_temp[i] = x
                if a_temp.sum() < 1e-10:
                    return np.inf
                return compute_score(a_temp / a_temp.sum())
            
            try:
                res = minimize_scalar(obj, bounds=(lo, hi), method='bounded',
                                       options={'maxiter': 50})
                if res.success and res.fun < best_score - 1e-10:
                    a[i] = res.x
                    best_score = res.fun
                    improved = True
            except Exception:
                pass
        
        if not improved:
            break
    
    return a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n))) + 1e-10


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 1e-10


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.01, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_blackman(n):
    """Blackman window."""
    N = n - 1
    if N == 0:
        return np.array([1.0])
    m = np.arange(n) / N
    return np.maximum(1e-10, 0.42 - 0.5 * np.cos(2 * np.pi * m) + 0.08 * np.cos(4 * np.pi * m))


def run_subgradient(a, deadline, rng, lr_init=0.3, top_k=3):
    """Run multiplicative subgradient optimization until deadline."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    stall = 0
    it = 0
    lp_interval = max(50, n // 10)  # Do LP every this many iterations
    
    while time.monotonic() < deadline:
        it += 1
        a_new = subgradient_step(a, lr, top_k=top_k)
        sc = compute_score(a_new)
        
        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
        else:
            stall += 1
            lr = max(0.01, lr * 0.9)
            a = best_a.copy()
            # Try with slightly different lr
            if stall > 20:
                lr = lr_init * np.exp(-stall * 0.02)
                a_new = subgradient_step(a, lr * 2, top_k=top_k)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_a = a_new.copy()
                    best_score = sc
                    a = a_new
                    stall = 0
            
            if stall > 200:
                # Random restart perturbation
                noise = rng.random(n) * 0.1
                a = best_a * (1.0 + noise)
                a = np.clip(a, 1e-10, None)
                a = a / a.sum()
                stall = 0
        
        # Periodic LP refinement

            if sc_lp < best_score:

                best_score = sc_lp
                a = a_lp
        
        if it % 500 == 0:

    return best_a, best_score

The key algorithm is:
1. Compute the time-domain signal: `x = ifft(fft(a))` (just `a` itself)

3. The "peak" in time domain corresponds to the auto-convolution peak
4. Apply a "peak reduction" in time domain: `x_reduced = peak_reduction(x)`
5. Transform back to frequency: `X_reduced = fft(x_reduced)`
6. Update: `a_new = x_reduced` (the peak-reduced signal)

Hybrid optimization for min 2*n*max(conv(a,a))/(sum(a))^2.
Uses fast multiplicative subgradient + periodic LP + coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar


def compute_score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fast_convolve_auto(a):
    """Compute auto-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    return c


def subgrad_step_fast(a, lr, top_k=3):
    """Fast multiplicative subgradient step."""
    n = len(a)
    c = fast_convolve_auto(a)
    max_c = c.max()
    
    # Top k peaks
    if top_k < 2 * n - 1:
        top_idx = np.argsort(c)[-top_k:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            g[i] += a[j]
    
    max_g = g.max()
    if max_g < 1e-15:
        return a.copy()
    
    a_new = a * np.exp(-lr * g / max_g)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def lp_refine(a, max_iters=5):
    """LP-based refinement."""
    n = len(a)
    if n > 2048:
        return a, compute_score(a)
    
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Build convolution matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    for it in range(max_iters):
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        if g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def coord_descent(a, max_rounds=2, rng=None):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = compute_score(a)
    
    for rnd in range(max_rounds):
        improved_any = False
        indices = list(range(n))
        if rng is not None:
            rng.shuffle(indices)
        
        for i in indices:
            old_val = a[i]
            
            if old_val < 1e-12:
                lo, hi = 0.0, np.max(a) * 1.5
            else:
                lo, hi = 0.0, old_val * 2.5
            
            # Quick 1D optimization
            # Try a few candidate values first (fast)
            candidates = np.linspace(lo, hi, 20)
            best_x = old_val
            best_f = best_score
            
            for x in candidates:
                a_temp = a.copy()
                a_temp[i] = x
                if a_temp.sum() < 1e-10:
                    continue
                a_temp = a_temp / a_temp.sum()
                f = compute_score(a_temp)
                if f < best_f:
                    best_f = f
                    best_x = x
            
            # Refine with golden section
            if best_x != old_val or best_f < best_score:
                # Narrow the search
                mask = np.abs(candidates - best_x) < (hi - lo) / 15
                if mask.sum() >= 2:
                    lo2, hi2 = candidates[mask].min(), candidates[mask].max()
                else:
                    lo2 = max(lo, best_x * 0.5)
                    hi2 = min(hi, best_x * 1.5 + 0.01 * np.max(a))
                
                def obj(x):
                    a_temp = a.copy()
                    a_temp[i] = x
                    if a_temp.sum() < 1e-10:
                        return np.inf
                    return compute_score(a_temp / a_temp.sum())
                
                try:
                    res = minimize_scalar(obj, bounds=(lo2, hi2), method='bounded',
                                           options={'maxiter': 30})
                    if res.fun < best_f:
                        best_f = res.fun
                        best_x = res.x
                except Exception:
                    pass
                
                if best_f < best_score - 1e-12:
                    a[i] = best_x
                    a = a / a.sum()
                    best_score = best_f
                    improved_any = True
        
        if not improved_any:
            break
    
    return a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(2 * np.pi * np.arange(n) / (n + 1))) + 1e-10


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 1e-10


def make_blackman(n):
    N = n - 1
    if N == 0:
        return np.array([1.0])
    m = np.arange(n) / N
    return np.maximum(1e-10, 0.42 - 0.5 * np.cos(2 * np.pi * m) + 0.08 * np.cos(4 * np.pi * m))


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(1e-10, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast multiplicative subgradient, LP refinement with line search,
and coordinate descent polishing across multiple diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize_scalar


def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fast_auto_conv(a):
    """Compute auto-convolution via FFT."""
    n = len(a)
    total_len = 2 * n - 1
    size = 1
    while size < total_len:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:total_len]


def subgrad_step(a, lr, top_k=3):
    """One multiplicative subgradient step targeting top peaks."""
    n = len(a)
    c = fast_auto_conv(a)
    max_c = c.max()
    
    if top_k < 2 * n - 1:
        top_idx = np.argsort(c)[-top_k:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            j = k - i
            if 0 <= j < n:
                g[i] += a[j]
    
    max_g = g.max()
    if max_g < 1e-15:
        return a.copy()
    
    a_new = a * np.exp(-lr * g / max_g)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ x = convolve(a, x)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_one_step(a, A):
    """One LP step: find direction g maximizing sum(g) s.t. conv(a,g)<=M, g>=0."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    M = c.max()
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def line_search_direction(a, g):
    """Find best t in [0,1] for (1-t)*a + t*g."""
    a_n = a / a.sum()
    g_n = g / g.sum()
    
    best_t = 1.0
    best_f = np.inf
    
    # Coarse search
    for t in np.linspace(0.02, 1.0, 25):
        a_new = (1.0 - t) * a_n + t * g_n
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        f = compute_score(a_new)
        if f < best_f:
            best_f = f
            best_t = t
    
    # Fine search around best
    if 0 < best_t < 1:
        lo = max(0.01, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        for t in np.linspace(lo, hi, 20):
            a_new = (1.0 - t) * a_n + t * g_n
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            f = compute_score(a_new)
            if f < best_f:
                best_f = f
                best_t = t
    
    return best_t, best_f


def lp_refine(a, max_iters=10, time_limit=None):
    """LP-based refinement with line search."""
    n = len(a)
    if n > 2500:
        return a.copy(), compute_score(a)
    
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    start_t = time.monotonic()
    
    A = build_conv_matrix(a)
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        g = lp_one_step(a, A)
        if g is None or g.sum() < 1e-12:
            break
        
        t, sc = line_search_direction(a, g)
        if sc < best_score - 1e-13:
            a_new = (1.0 - t) * a + t * (g / g.sum())
            a_new = np.clip(a_new, 0, None)
            a_new = a_new / a_new.sum()
            best_a = a_new
            best_score = sc
            a = a_new
            A = build_conv_matrix(a)
        else:
            # Try smaller steps
            found = False
            for t_try in [0.1, 0.05, 0.02]:
                a_new = (1.0 - t_try) * a + t_try * (g / g.sum())
                a_new = np.clip(a_new, 0, None)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                f = compute_score(a_new)
                if f < best_score - 1e-13:
                    best_a = a_new
                    best_score = f
                    a = a_new
                    A = build_conv_matrix(a)
                    found = True
                    break
            if not found:
                break
    
    return best_a, best_score


def coordinate_descent(a, max_rounds=2, rng=None, time_limit=None):
    """Coordinate descent: optimize each element in 1D."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = compute_score(a)
    
    for rnd in range(max_rounds):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        improved_any = False
        indices = list(range(n))
        if rng is not None:
            rng.shuffle(indices)
        
        for i in indices:
            if time_limit is not None and time.monotonic() > time_limit:
                break
            
            old_val = a[i]
            max_a = np.max(a)
            
            if old_val < 1e-12:
                lo, hi = 0.0, max_a * 1.5
            else:
                lo, hi = 0.0, old_val * 3.0 + 0.01 * max_a
            
            # Quick scan
            candidates = np.linspace(lo, hi, 15)
            best_x = old_val
            best_f = best_score
            
            for x in candidates:
                a_temp = a.copy()
                a_temp[i] = x
                s = a_temp.sum()
                if s < 1e-10:
                    continue
                f = compute_score(a_temp / s)
                if f < best_f:
                    best_f = f
                    best_x = x
            
            if best_f < best_score - 1e-12:
                # Refine with golden section
                span = (hi - lo) / 15
                lo2 = max(0, best_x - 2 * span)
                hi2 = best_x + 2 * span
                
                def obj_1d(x):
                    a_temp = a.copy()
                    a_temp[i] = x
                    s = a_temp.sum()
                    if s < 1e-10:
                        return np.inf
                    return compute_score(a_temp / s)
                
                try:
                    res = minimize_scalar(obj_1d, bounds=(lo2, hi2), method='bounded',
                                           options={'maxiter': 20})
                    if res.fun < best_f:
                        best_f = res.fun
                        best_x = res.x
                except Exception:
                    pass
                
                if best_f < best_score - 1e-12:
                    a[i] = best_x
                    a = a / a.sum()
                    best_score = best_f
                    improved_any = True
        
        if not improved_any:
            break
    
    return a, best_score


def make_cosine(n):
    """Hann-like window."""
    return 0.5 * (1.0 + np.cos(2 * np.pi * np.arange(n) / (n + 1))) + 1e-10


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2) + 1e-10


def make_blackman(n):
    """Blackman window."""
    N = n - 1
    if N == 0:
        return np.array([1.0])
    m = np.arange(n) / N
    return np.maximum(1e-10, 0.42 - 0.5 * np.cos(2 * np.pi * m) + 0.08 * np.cos(4 * np.pi * m))


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(1e-10, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_asymmetric(n, rng):
    """Asymmetric window."""
    a = np.ones(n)
    # Taper left and right differently
    left_taper = int(n * rng.uniform(0.2, 0.5))
    right_taper = int(n * rng.uniform(0.2, 0.5))
    for i in range(left_taper):
        a[i] *= i / left_taper
    for i in range(right_taper):
        a[n - 1 - i] *= i / right_taper
    return a + 1e-10


def run_subgradient(a, deadline, rng, lr_init=0.3, top_k=3):
    """Main subgradient optimization loop."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    stall = 0
    it = 0
    lp_interval = max(100, n // 4)
    
    while time.monotonic() < deadline:
        it += 1
        
        a_new = subgrad_step(a, lr, top_k=top_k)
        sc = compute_score(a_new)
        
        if sc < best_score - 1e-13:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
        else:
            stall += 1
            lr = max(0.01, lr * 0.95)
            a = best_a.copy()
            
            if stall > 50:
                noise = rng.random(n)
                a = best_a * (0.5 + noise)
                a = np.clip(a, 1e-10, None)
                a = a / a.sum()
                stall = 0
                lr = lr_init
            elif stall > 500:
                break
        
        # Periodic LP
        if it % lp_interval == 0 and time.monotonic() < deadline - 60 and n <= 1500:
            a_lp, sc_lp = lp_refine(best_a, max_iters=3,
                                     time_limit=time.monotonic() + 30)
            if sc_lp < best_score:
                best_a = a_lp
                best_score = sc_lp
                a = a_lp
                stall = 0
        
        if it % 2000 == 0:
            print(f"      subgrad iter {it}, score={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score


def get_config_priority(cfg):
