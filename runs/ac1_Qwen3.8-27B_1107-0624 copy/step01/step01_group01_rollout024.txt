```python
def lp_step(f, M):
    # Solve: max sum(b) s.t. f*b <= M (element-wise), b >= 0
    n = len(f)
    # convolution constraint: (f*b)[k] <= M for k=0,...,2n-2
    # (f*b)[k] = sum_j f[k-j]*b[j] (with f out of range = 0)
    # This is a matrix A * b <= M where A[k,j] = f[k-j] if 0<=k-j<n else 0
    
    # Build A (2n-1 by n matrix)
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Solve LP: max c^T b s.t. A b <= M, b >= 0
    c = np.ones(n)
    res = scipy.optimize.linprog(-c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
Optimization of non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterative refinement (best-response dynamics) combined with

import numpy as np
import time

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size

        return np.array([])

        return np.convolve(a, a)

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

    f = f / s  # normalize to sum 1

        c = full_convolve_fft(f)

        # This is a (2n-1) x n matrix
        # Use the fact that it's a "convolution matrix"
        # A[k,j] = f[k-j] for max(0, k-n+1) <= j <= min(n-1, k)

        A = np.zeros((2*n-1, n))
        for j in range(n):

            # i.e., rows j to j+n-1, column j = f[0], f[1], ..., f[n-1]
            A[j:j+n, j] = f

        if sb < 1e-15:
            break

        b = b / sb

            f = f_new

            f = f_new  # still update f for next LP

            # f stays the same
        
        if it % 100 == 0:
            print(f"  LP iter {it}: score={best_score:.10f}")
    
    return best_f, best_score
```

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based best-response iteration combined with soft-max gradient polish.
Key innovation: LP iteration provides a principled descent direction
that avoids the local minima issues of pure gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-10:
        c = np.maximum(c, 0.0)
    elif mn < 0.0:
        c = c - mn
    return c


def score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return None
    return a / s


def soft_max_weights(c, beta):
    mval = float(c.max())
    if beta < 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / cnt
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval + beta * float(np.log(s)), w


def gradient_soft_conv(a, w):
    """
    Compute gradient of soft_max(conv(a,a)) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i] (with a[j]=0 for j outside [0,n-1])
             = 2 * sum_j a[j] * w[j+i]
    Using FFT: h = ifft(conj(fft(a)) * fft(w))[:n]
    """
    n = len(a)
    N = next_pow2(3 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def make_gaussian_profile(n, sigma_frac):
    """Gaussian profile centered in the sequence."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return normalize_nonneg(a)


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, power):
    """Raised cosine: cos^power on [0, pi]."""
    t = np.linspace(0.0, np.pi, n)
    a = np.cos(t / 2.0) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_parabolic_profile(n, c):
    """Parabolic: 1 - c*(2t-1)^2."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 - c * u * u
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_lorentz_profile(n, gamma):
    """Lorentzian: 1/(1+gamma*(2t-1)^2)."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 / (1.0 + gamma * u * u)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = f[k-j]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_best_response(f, M):
    """
    Solve the LP: max sum(b) s.t. A@b <= M, b >= 0.
    Returns the optimal b (normalized to sum 1) or None on failure.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            return None
        b = np.maximum(res.x, 0.0)
        sb = float(b.sum())
        if sb < 1e-15:
            return None
        return b / sb
    except Exception:
        return None


def lp_iteration(f0, max_iters=300, deadline=None, verbose=False):
    """
    LP-based best-response iteration.
    Returns (best_f, best_score).
    """
    f = normalize_nonneg(f0)
    if f is None:
        return f0, np.inf
    n = f.size
    best_f = f.copy()
    best_score = score_fast(f)
    t_step = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = full_convolve_fft(f)
        M = float(c.max())
        
        b = lp_best_response(f, M)
        if b is None:
            break
        
        # Try step sizes from large to small
        accepted = False
        for t in [t_step, 0.5 * t_step, 0.25 * t_step, 0.1 * t_step]:
            f_new = (1.0 - t) * f + t * b
            f_new = normalize_nonneg(f_new)
            if f_new is None:
                continue
            sc = score_fast(f_new)
            if sc < best_score - 1e-14:
                f = f_new
                best_f = f.copy()
                best_score = sc
                no_improve = 0
                accepted = True
                break
        
        if not accepted:
            no_improve += 1
            if no_improve > 50:
                t_step = max(0.05, t_step * 0.8)
                if t_step < 0.05:
                    break
        
        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}")
    
    return best_f, best_score


def gradient_polish(f0, deadline, rng, lr=0.1, max_iters=20000):
    """
    Soft-max gradient descent in log-space with Adam.
    Returns (best_f, best_score).
    """
    f = normalize_nonneg(f0)
    if f is None:
        return f0, np.inf
    n = f.size
    
    # Initialize in log-space
    f_safe = np.maximum(f, 1e-15)
    z = np.log(f_safe)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    c = full_convolve_fft(f)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)
    
    best_f = f.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        L, w = soft_max_weights(c, beta)
        g_a = gradient_soft_conv(f, w)
        
        # Project to simplex: g_z = f * (g_a - dot(f, g_a))
        dot_fa = float(np.dot(f, g_a))
        g_z = f * (g_a - dot_fa)
        
        if not np.all(np.isfinite(g_z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * g_z
        v = b2 * v + (1 - b2) * (g_z * g_z)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0); v.fill(0); adam_t = 0
            continue
        
        z = np.clip(z, -50
