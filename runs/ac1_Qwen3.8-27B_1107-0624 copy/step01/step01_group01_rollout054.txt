"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions for the peak autocorrelation,
combined with soft-max Adam polishing for final refinements.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_step(f, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, deadline, rng, verbose=True):
    """Main LP-based optimization loop."""
    f = np.maximum(np.asarray(f_init, dtype=np.float64).ravel(), 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        return f_init.copy()
    f = f / s
    n = len(f)

    conv_f = np.convolve(f, f)
    best_M = float(conv_f.max())
    best_f = f.copy()
    t = 0.3
    stagnant = 0
    iteration = 0

    while time.monotonic() < deadline:
        M = best_M if stagnant > 0 else float(np.convolve(f, f).max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Try the step
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new / s_new

        M_new = float(np.convolve(f_new, f_new).max())

        if M_new <= best_M + 1e-12:
            f = f_new
            if M_new < best_M - 1e-12:
                best_M = M_new
                best_f = f.copy()
                stagnant = 0
            else:
                stagnant += 1
            iteration += 1
            if verbose and iteration % 100 == 0:
                print(f"  LP iter {iteration}: M={best_M:.10f}, t={t:.4f}")
        else:
            t *= 0.5
            if t < 0.002:
                t = 0.002
            stagnant += 1

        # Decay t slowly
        t = max(0.005, t * 0.998)

        # Random restart when stuck
        if stagnant > 150:
            noise_scale = 0.1 * max(1.0 / n, np.std(best_f))
            f = best_f + rng.normal(0, noise_scale, n)
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs > 1e-10:
                f = f / fs
            else:
                f = best_f.copy()
            t = 0.3
            stagnant = 0
            if verbose:
                print(f"  LP restart at iter {iteration}, M={best_M:.10f}")

    return best_f


def soft_mx(c, beta):
    """Smooth max with log-sum-exp."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] * p[j] for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal(p0, deadline, rng, lr=0.08, reg=1e-7):
    """Full logit-space Adam polish."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.2 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 1000 == 0:
                    lr = max(0.01, lr * 0.98)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 80 == 0:
            beta = max(1e-14, beta * 0.8)

        if no_imp > max(800, 4 * n):
            z = 0.8 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-12, 0.25 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def make_linear_profile(n, eta=1.5):
    """Monotone linear profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_profile(n, k=1):
    """Cosine-based profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + 0.5 * np.cos(2.0 * np.pi * k * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_sine_sq_profile(n, k=1):
    """Sine-squared profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 + 0.5 * np.sin(np.pi * k * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_quadratic_profile(n, a=0.0, b=1.0):
    """Quadratic profile 1 + a*t + b*t^2."""
    t = np.linspace(0.0, 1.0, n)
    a_arr = 1.0 + a * t + b * t ** 2
    a_arr = np.maximum(a_arr, 1e-12)
    return a_arr / a_arr.sum()


def make_mixed_osc_profile(n, k1=2, k2=3, amp=0.3):
    """Mixed oscillatory profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + amp * np.cos(2 * np.pi * k1 * t) + 0.5 * amp * np.cos(2 * np.pi * k2 * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 1e-10 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return np.interp(x1, x0, p0)


def get_starting_profiles(n, rng):

    # Quadratic
    profiles.append(("quad", make_quadratic_profile(n, 1.0, -0.5)))
    profiles.append(("quad2", make_quadratic_profile(n, -0.5, 1.0)))
    # Mixed oscillatory
    profiles.append(("mix1", make_mixed_osc_profile(n, 2, 3, 0.3)))
    profiles.append(("mix2", make_mixed_osc_profile(n, 1, 4, 0.2)))
    # Random smooth
    for _ in range(4):
        coeffs = rng.uniform(-0.5, 0.5, 5)
        t = np.linspace(0.0, 1.0, n)
        a = 1.0
        for i, c in enumerate(coeffs[1:], 1):

        a = np.maximum(a, 1e-12)
        profiles.append((f"rand_{len(profiles)}", a / a.sum()))
    return profiles

        if sc < best_score + 0.01:
            print(f"  Profile {name}: score={sc:.10f}")

    print(f"After Phase 1: best={best_score:.10f}")

    # ============ Phase 2: LP at n=1024 ============
    n_large = 1024
    phase2_end = min(deadline, t_start + 1000.0)

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        p_large = interpolate_profile(best_seq, n_large)
        sc = compute_score(p_large)
        update(p_large, sc)
        print(f"Phase 2 start: n={n_large}, score={sc:.10f}")

        lp_deadline2 = min(phase2_end, time.monotonic() + 400.0)
        result = lp_optimize(p_large, lp_deadline2, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After large LP: score={sc:.10f}")

        # Also try different profiles at large n
        for name, p_small in profiles[:4]:
            if time.monotonic() > phase2_end - 5:
                break
            p_lp = interpolate_profile(p_small, n_large)
            sc_p = compute_score(p_lp)
            if sc_p >= best_score + 0.05:
                continue
            per_deadline = min(phase2_end, time.monotonic() + 60.0)
            result = lp_optimize(p_lp, per_deadline, rng, verbose=False)
            sc = compute_score(result)
            update(result, sc)

    print(f"After Phase 2: best={best_score:.10f}")

    # ============ Phase 3: Soft-max Adam polish ============
    phase3_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        n_polish = best_seq.size
        print(f"Phase 3: polishing at n={n_polish}, current best={best_score:.10f}")

        # Multiple polish rounds with different parameters
        for lr in [0.08, 0.04, 0.06]:
            if time.monotonic() > phase3_end - 5:
                break
            per_deadline = min(phase3_end, time.monotonic() + 120.0)
            p0 = best_seq.copy()
            # Add tiny noise to avoid getting stuck
            p0 = p0 + 0.001 * rng.normal(size=n_polish)
            p0 = np.maximum(p0, 1e-12)
            p0 = p0 / p0.sum()
            result, sc = logit_anneal(p0, per_deadline, rng, lr=lr)
            update(result, sc)
            print(f"  Adam polish lr={lr}: score={sc:.10f}")

    print(f"After Phase 3: best={best_score:.10f}")

    # ============ Phase 4: Try n=2048 if time remains ============
    n_xlarge = 2048
    phase4_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None and best_score > 1.505:
        p_xl = interpolate_profile(best_seq, n_xlarge)
        sc = compute_score(p_xl)
        update(p_xl, sc)
        print(f"Phase 4: n={n_xlarge}, score={sc:.10f}")

        lp_deadline3 = min(phase4_end, time.monotonic() + 100.0)
        result = lp_optimize(p_xl, lp_deadline3, rng, verbose=True)
        sc = compute_score(result)
        update(result, sc)
        print(f"After xlarge LP: score={sc:.10f}")

    print(f"After Phase 4: best={best_score:.10f}")

    # ============ Phase 5: Final short polish ============
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per = min(deadline - 5, time.monotonic() + 30.0)
        n_final = best_seq.size
        p0 = best_seq + 0.0005 * rng.normal(size=n_final)
        p0 = np.maximum(p0, 1e-12)
        p0 = p0 / p0.sum()
        result, sc = logit_anneal(p0, per, rng, lr=0.03)
        update(result, sc)
        print(f"Final polish: score={sc:.10f}")

    # ============ Final validation ============
    if best_seq is not None:
        final_seq = best_seq
    else:
        final_seq = make_linear_profile(512, 1.5)

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = make_linear_profile(256, 1.5)

    final_score = compute_score(final_seq)
    print(f"FINAL: n={final_seq.size}, score={final_score:.10f}")

    # Validate with evaluate_sequence if possible
    try:
        eval_score = float(evaluate_sequence([float(x) for x in final_seq]))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    return [float(x) for x in final_seq]
