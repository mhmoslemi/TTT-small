"""
Hybrid optimization: log-sum-exp gradient descent + LP refinement
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_solve(a, M, top_k=None):
    n = len(a)
    A = build_conv_matrix(a)
    c = np.convolve(a, a, mode='full')
    
    if top_k is not None and top_k < 2*n-1:
        # Only use the top_k constraints
        idx = np.argsort(c)[-top_k]
        A = A[idx]
        b_ub = np.full(len(idx), M)
    else:
        b_ub = np.full(2*n-1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def optimize_gradient_lse(a, deadline, rng, beta_start=50, beta_end=500):
    """Log-sum-exp gradient descent with momentum."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    
    momentum = np.zeros(n)
    lr = 0.1
    beta = beta_start
    stall = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Log-sum-exp approximation to max
        # For numerical stability, subtract max
        lc = c - M
        exp_lc = np.exp(beta * lc)
        weights = exp_lc / exp_lc.sum()
        
        # Gradient: grad_i = 2 * sum_k weights[k] * a[k-i] (for valid k-i)
        grad = np.zeros(n)
        for i in range(n):
            # k ranges from i to i+n-1 (so that k-i is in [0, n-1])
            k_start = i
            k_end = min(i + n, 2*n - 1)
            if k_start < 2*n - 1:
                grad[i] = 2.0 * np.sum(weights[k_start:k_end] * a[k_start-i:k_end-i])
        
        # Normalize gradient
        grad_mean = grad.mean() + 1e-15
        grad = (grad - grad_mean) / (grad.std() + 1e-15)
        
        # Momentum update
        momentum = 0.9 * momentum + 0.1 * grad
        a_new = a * np.exp(-lr * momentum)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / a_new.sum()
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-10:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.3, lr * 1.01)
        else:
            stall += 1
            lr = max(0.001, lr * 0.9)
            if stall > 30:
                # Perturbation
                noise = np.abs(rng.normal(0, 0.1, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                a_new = a_new / a_new.sum()
                a = a_new
                stall = 0
                momentum = np.zeros(n)
            if stall > 200:
                break
        
        # Increase beta over time (sharpen the max approximation)
        beta = min(beta_end, beta * 1.005)

"""
Hybrid optimization combining log-sum-exp gradient descent, LP refinement,
and diverse initializations (including flat-top filter windows) to minimize
2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M, top_k=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Optionally use only top_k constraints."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.convolve(a, a, mode='full')
    
    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(c)[-top_k]
        A_sub = A[idx, :]
        b_ub = np.full(top_k, M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_lse(a, deadline, rng, max_iter=5000):
    """Log-sum-exp gradient descent with momentum."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    
    momentum = np.zeros(n)
    lr = 0.05
    beta = 30.0
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Softmax weights (numerically stable)
        shifted = c - M
        exp_shifted = np.exp(beta * shifted)
        total = exp_shifted.sum()
        if total < 1e-300:
            break
        weights = exp_shifted / total
        
        # Compute gradient: grad[i] = 2 * sum_{k=i}^{min(i+n-1, 2n-2)} weights[k] * a[k-i]
        grad = np.zeros(n)
        for i in range(n):
            end = min(i + n, 2 * n - 1)
            grad[i] = 2.0 * np.dot(weights[i:end], a[:end - i])
        
        # Normalize for stable updates
        g_mean = grad.mean()
        g_std = grad.std() + 1e-15
        grad_normed = (grad - g_mean) / g_std
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * grad_normed
        
        # Multiplicative update
        a_new = a * np.exp(-lr * momentum)
        a_new = np.clip(a_new, 1e-15, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.2, lr * 1.005)
        else:
            stall += 1
            lr = max(0.001, lr * 0.92)
            if stall > 40:
                # Perturbation to escape local min
                scale = rng.choice([0.05, 0.1, 0.15, 0.2])
                noise = rng.normal(0, scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                s = a_new.sum()
                if s > 1e-10:
                    a = a_new / s
                    stall = 0
                    momentum = np.zeros(n)
                else:
                    break
            if stall > 300:
                break
        
        # Gradually increase beta
        beta = min(200.0, beta * 1.002)
        
        if it % 500 == 0:
            print(f"        LSE iter {it}, score={best_score:.6f}, lr={lr:.4f}, beta={beta:.1f}")
    
    return best_a, best_score


def lp_refine(a, deadline, rng, max_iter=200):
    """LP-based refinement."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        # Use top_k constraints for speed
        top_k = min(2 * n - 1, max(50, n))
        g = lp_solve(a, M, top_k=top_k)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 1e-15, None)
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.3, step * 1.05)
        else:
            stall += 1
            step = max(0.01, step * 0.7)
            if stall > 50:
                noise = np.abs(rng.normal(0, 0.05, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                a_new = a_new / a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                else:
                    a = best_a
                step = 0.1
            if stall > 150:
                break
        
        if it % 20 == 0:
            print(f"        LP iter {it}, score={best_score:.6f}, step={step:.4f}")
    
    return best_a, best_score


def make_flat_top(n):

def make_cosine(n):
    """Raised cosine (Hann) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return 1.0 - np.abs(np.arange(n) - c) / (c + 1.0)


def make_optimized_init(n, rng):

"""
Hybrid optimization combining log-sum-exp gradient descent with variance regularization,
LP-based directional refinement, and flat-top filter window initializations to minimize
2*n*max(conv(a,a))/(sum(a))^2. Uses multi-phase time allocation with deep optimization
on the most promising candidates.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M, top_k=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = np.convolve(a, a, mode='full')

    if top_k is not None and top_k < 2 * n - 1:
        idx = np.argsort(c)[-top_k]
        A_sub = A[idx, :]
        b_ub = np.full(top_k, M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_descent_lse(a, deadline, rng, use_variance=True, var_weight=0.3):
    """Log-sum-exp gradient descent with momentum and optional variance regularization."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)

    momentum = np.zeros(n)
    lr = 0.04
    beta = 20.0
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        c_mean = c.mean()

        # Softmax weights for max approximation
        shifted = c - M
        exp_shifted = np.exp(beta * shifted)
        total = exp_shifted.sum()
        if total < 1e-300:
            break
        weights = exp_shifted / total

        # Primary gradient: weighted sum over peaks
        grad = np.zeros(n)
        for i in range(n):
            end = min(i + n, 2 * n - 1)
            grad[i] = 2.0 * np.dot(weights[i:end], a[:end - i])

        # Variance regularization: push convolution toward its mean
        if use_variance:
            c_centered = c - c_mean
            for i in range(n):
                end = min(i + n, 2 * n - 1)
                grad[i] += var_weight * 2.0 * np.dot(c_centered[i:end] / (2 * n - 1), a[:end - i])

        # Normalize gradient
        g_mean = grad.mean()
        g_std = grad.std() + 1e-15
        grad_normed = (grad - g_mean) / g_std

        # Momentum
        momentum = 0.85 * momentum + 0.15 * grad_normed

        # Multiplicative update
        a_new = a * np.exp(-lr * momentum)
        a_new = np.clip(a_new, 1e-15, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = compute_objective(a_new)
        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.15, lr * 1.003)
        else:
            stall += 1
            lr = max(0.002, lr * 0.93)
            if stall > 50:
                scale = rng.choice([0.05, 0.1, 0.2])
                noise = rng.normal(0, scale, n)
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                s2 = a_new.sum()
                if s2 > 1e-10:
                    a = a_new / s2
                    stall = 0
                    momentum = np.zeros(n)
                else:
                    break
            if stall > 400:
                break

        beta = min(150.0, beta * 1.003)

        if it % 1000 == 0:
            print(f"        LSE iter {it}, score={best_score:.6f}, lr={lr:.4f}, beta={beta:.1f}")

    return best_a, best_score


def lp_refine(a, deadline, rng, max_iter=500):
    """LP-based refinement with adaptive step and top-k constraints."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        # Use top_k constraints for speed (scale with n)
        top_k = min(2 * n - 1, max(80, n))
        g = lp_solve(a, M, top_k=top_k)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 1e-15, None)
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.35, step * 1.04)
        else:
            stall += 1
            step = max(0.01, step * 0.7)
            if stall > 40:
                # Try perturbation
                noise = np.abs(rng.normal(0, 0.08, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                a_new = a_new / a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                else:
                    # Accept with small probability (annealing)
                    if rng.random() < 0.2:
                        a = a_new
                    else:
                        a = best_a
                    stall = 0
                step = 0.12
            if stall > 200:
                break

        if it % 50 == 0:
            print(f"        LP iter {it}, score={best_score:.6f}, step={step:.4f}")

    return best_a, best_score


def make_flat_top(n):
    """Flat-top window (designed for maximally flat frequency magnitude response)."""
    m = np.arange(n, dtype=np.float64)
    M = n - 1
    if M == 0:
        return np.ones(1)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006540287
    w = (a0 - a1 * np.cos(2 * np.pi * m / M)
              + a2 * np.cos(4 * np.pi * m / M)
              - a3 * np.cos(6 * np.pi * m / M)
              + a4 * np.cos(8 * np.pi * m / M))
    return np.maximum(w, 1e-10)


def make_blackman_harris(n):
    """Blackman-Harris 4-term window."""
    m = np.arange(n, dtype=np.float64)
    M = n - 1
    if M == 0:
        return np.ones(1)
    a0 = 0.35875
    a1 = 0.48829
    a2 = 0.14128
    a3 = 0.01168
    w = (a0 - a1 * np.cos(2 * np.pi * m / M)
              + a2 * np.cos(4 * np.pi * m / M)
              - a3 * np.cos(6 * np.pi * m / M))
    return np.maximum(w, 1e-10)


def make_nuttall(n):
    """Nuttall window."""
    m = np.arange(n, dtype=np.float64)
    M = n - 1
    if M == 0:
        return np.ones(1)
    a0 = 0.3635819
    a1 = 0.4891775
    a2 = 0.1365995
    a3 = 0.0106411
    w = (a0 - a1 * np.cos(2 * np.pi * m / M)
              + a2 * np.cos(4 * np.pi * m / M)
              - a3 * np.cos(6 * np.pi * m / M))
    return np.maximum(w, 1e-10)


def make_cosine(n):
    """Raised cosine (Hann) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.22):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 1e-10)


def make_super_gaussian(n, p=2.5, sigma=0.25):
    """Super-Gaussian window: exp(-(x/sigma)^p)."""
    center = (n - 1) / 2.0
    x = (np.arange(n) - center) / (sigma * n)
    return np.exp(-np.abs(x) ** p)

        (make_cosine, "hann"),
        (make_gaussian, "gauss"),

    ]

    # Quick gradient runs
    candidates = []  # (score, a, n, label)

    n_explore = len(exploration)

        if time.monotonic() > phase1_end:

        if callable(init_src) or isinstance(init_src, str) == False and not isinstance(init_src, np.ndarray):

            print(f"  P1 [{idx+1}/{n_explore}] n={n} {label}: {score:.6f} "

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Top 5: " + ", ".join(

    # ==============================================================

    phase2_end = start_time + 0.80 * (budget_s - 10)

    return [float(x) for x in best_seq]
