    import scipy.optimize as _scipy_opt

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)


def mix_greedy_uniform(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)

def structured_perturb(p, rng, mode):
    """Apply structured perturbation to a normalized sequence."""
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    idx = np.arange(n, dtype=np.float64)
    if mode == 0:
        # Sinusoidal modulation
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.05, 0.3)
        p = p * (1.0 + amp * np.sin(2.0 * np.pi * freq * idx / n))
    elif mode == 1:
        # Half-scale
        half = n // 2
        ratio = rng.uniform(0.7, 1.3)
        p[:half] *= ratio
        p[half:] *= (2.0 - ratio) / 1.0
    elif mode == 2:
        # Decomposition: boost center
        center = n // 2
        width = max(1, n // 4)
        mask = np.zeros(n, dtype=np.float64)
        lo = max(0, center - width)
        hi = min(n, center + width + 1)
        mask[lo:hi] = 1.0
        amp = rng.uniform(0.1, 0.4)
        p = p + amp * mask * p.mean()
    else:
        # Random smooth perturbation
        smooth = np.exp(rng.normal(0.0, 0.2, n)).cumsum()
        smooth = smooth / smooth[-1] * n
        p = p * (1.0 + 0.2 * (smooth - 1.0))
    return normalize_nonneg(np.maximum(p, 0.0))


def actual_score(p):

    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 768:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))

    n = p.size
    budget_s = float(budget_s)

        return p, actual_score(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    mu, nu, eps = 0.9, 0.999, 1e-8

    v_arr = np.zeros(n, dtype=np.float64)

    best_p = p.copy()
    best_f = actual_score(p)

    # Learning rate based on size

    elif n < 512:

    start = time.monotonic()

        elapsed = time.monotonic() - start

def coordinate_polish(p, budget_s, n_passes_max=500):
    """
    Coordinate-wise line search: for each coordinate, try several multipliers,
    rescale to maintain sum=1, keep the best. Costs O(n * n_mult) evaluations per pass.
    """
    p = normalize_nonneg(p).copy()
    n = p.size
    best_f = actual_score(p)

    # Multipliers to try (skip 1.0)
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0]

    start = time.monotonic()
    end = start + budget_s
    passes = 0

    while time.monotonic() < end and passes < n_passes_max:
        passes += 1
        improved = False

        # Get argmax positions to guide ordering
        a = np.maximum(p, 0.0)
        c = np.convolve(a, a)
        c = np.maximum(c, 0.0)
        argmax_pos = int(np.argmax(c))

        # Prioritize coordinates near the argmax diagonal
        order = sorted(range(n), key=lambda i: abs(i - argmax_pos))

        for i in order:
            if time.monotonic() > end:
                break
            orig_val = p[i]
            if orig_val < 1e-15:
                continue
            best_local_val = orig_val
            best_local_f = best_f
            best_local_p = None

            for mult in multipliers:
                new_val = orig_val * mult
                others_sum = 1.0 - new_val
                orig_others = 1.0 - orig_val
                if others_sum <= 1e-12 or orig_others <= 1e-12:
                    continue
                scale = others_sum / orig_others
                new_p = p * scale
                new_p[i] = new_val
                new_p = np.maximum(new_p, 0.0)
                f_new = actual_score(new_p)
                if f_new < best_local_f:
                    best_local_f = f_new
                    best_local_val = new_val
                    best_local_p = new_p.copy()

            if best_local_p is not None and best_local_f < best_f - 1e-12:
                p = best_local_p
                best_f = best_local_f
                improved = True

        if not improved:
            break

    return p, best_f


def lp_improve(p, budget_s, max_rows=800):
    """
    Cutting-plane LP: maximize sum(g) s.t. f*g <= M, g >= 0.
    Then blend f with g.
    """
    if _scipy_opt is None:
        return p, actual_score(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4 or budget_s < 20.0:
        return p, actual_score(p)

    budget_s = float(budget_s)
    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)
    f_sum = float(f.sum())

    end = time.monotonic() + max(0.0, budget_s - 3.0)

    # Initial rows: spread across the convolution
    r0 = min(max_rows, 200)
    rows = set(int(k) for k in np.linspace(0, 2 * n - 2, r0, dtype=int).astype(int))

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(sorted(rows), dtype=int)
        nr = int(rows_arr.size)

        # Build constraint matrix A (nr x n)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b_ub = np.full(nr, float(np.max(np.convolve(f, f))), dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.3))

        try:
            res = _scipy_opt.linprog(
                c_obj, A_ub=A, b_ub=b_ub,
                bounds=(0.0, None),
                method="highs",
                options={"time_limit": float(tl)},
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full constraint violations
        conv_check = np.convolve(f, g)
        M_val = float(np.max(np.convolve(f, f)))
        violations = conv_check - M_val
        if np.any(violations > 1e-5 * M_val):
            if len(rows) < max_rows:
                # Add most violated constraints
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= 100 or len(rows) >= max_rows:
                        break
                    ix = int(ix)
                    if violations[ix] > 1e-5 * M_val and ix not in rows:
                        rows.add(ix)
                        added += 1
                if added > 0:
                    continue
            else:
                break

        # Rescale g to match f's sum
        g = g * (f_sum / sg)

        # Line search: try different blend ratios
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.maximum(fn, 0.0)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (f_sum / sn)
            sc = actual_score(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_score >= best_score - 1e-10:
            break

        f = accepted
        best_score = accepted_score
        best_f = f.copy()
        cur_score = accepted_score

        if it % 3 == 0:
            print("  LP iter=%d score=%.6f rows=%d" % (it, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score


def main_search(seed, budget_s):
    global _rng_comm
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))
    _rng_comm = np.random.default_rng(int(seed) + 777)

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
        n = int(a.size)
        old = best_by_n.get(n)
        if old is None or score < old:
            best_by_n[n] = float(score)

    def get_best_for_n(n):
        # Return best sequence of length n from archive, or generate
        if n in best_by_n:
            pass
        return None

    # Load height_sequence_1 if available
    h1 = None
    try:
        h1 = globals().get("height_sequence_1", None)
    except Exception:
        pass
    if h1 is not None:
        try:
            arr = np.asarray(h1, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                rec = normalize_nonneg(arr)
                sc = actual_score(rec)
                record(rec, sc)
                print("Loaded height_sequence_1: n=%d score=%.6f" % (arr.size, sc))
        except Exception:
            pass

    # Phase 1: Multi-start Adam (majority of budget)
    print("Phase 1: Multi-start Adam optimization")
    p1_end = t0 + (deadline - t0) * 0.60

    # Define the search jobs: (n, init_type, time_alloc)
    lengths_focus = [256, 320, 360, 384, 408, 432, 480, 512, 640, 768, 1024]
    init_types = ["greedy", "power45", "perturb", "dirichlet", "structured"]

    jobs = []
    # Primary focus on 384
    jobs.append((384, "greedy", 80))
    jobs.append((384, "perturb", 60))
    jobs.append((384, "dirichlet", 50))
    jobs.append((384, "structured", 50))
    jobs.append((384, "power45", 40))

    # Nearby lengths
    jobs.append((360, "greedy", 50))
    jobs.append((360, "perturb", 40))
    jobs.append((408, "greedy", 50))
    jobs.append((408, "perturb", 40))
    jobs.append((432, "greedy", 45))
    jobs.append((320, "greedy", 40))
    jobs.append((480, "greedy", 45))
    jobs.append((512, "greedy", 50))
    jobs.append((512, "perturb", 40))
    jobs.append((640, "greedy", 40))
    jobs.append((768, "greedy", 45))
    jobs.append((1024, "greedy", 40))
    jobs.append((256, "greedy", 30))

    total_alloc = sum(j[2] for j in jobs)
    available = p1_end - t0
    scale = available / total_alloc if total_alloc > 0 else 1.0

    for n, init_type, alloc in jobs:
        rem = p1_end - time.monotonic()
        if rem < 15.0:
            break
        alloc_use = min(float(alloc) * scale, max(5.0, rem - 3.0))

        if init_type == "greedy":
            base = greedy_normalized(n)
        elif init_type == "power45":
            base = power_normalized(n, 0.45)
        elif init_type == "perturb":
            base = perturb_greedy(n, rng, 0.12)
        elif init_type == "dirichlet":
            base = dirichlet_start(n, 0.7, rng)
        elif init_type == "structured":
            base = structured_perturb(greedy_normalized(n), rng, int(rng.integers(0, 4)))
        else:
            base = greedy_normalized(n)

        p_out, f_out = adam_optimize(base, alloc_use, seed=int(seed) + n)
        record(p_out, f_out)
        print("  Adam n=%d type=%s alloc=%.0fs -> score=%.6f (best=%.6f)" %
              (n, init_type, alloc_use, f_out, best_score))

    # Phase 2: Coordinate-wise polishing
    print("Phase 2: Coordinate-wise polishing")
    p2_end = t0 + (deadline - t0) * 0.80
    if best_p is not None and time.monotonic() < p2_end:
        alloc = min(300.0, max(10.0, p2_end - time.monotonic()))
        p_polished, f_polished = coordinate_polish(best_p, alloc)
        record(p_polished, f_polished)
        print("  Coordinate polish: %.6f -> %.6f" % (best_score, f_polished))

        # Also polish the second-best
        items = sorted(best_by_n.items(), key=lambda x: x[1])
        for n_key, sc in items[1:3]:
            if time.monotonic() >= p2_end:
                break
            # Reconstruct from archive (we don't store full arrays, so skip)
            pass

    # Phase 3: LP refinement
    print("Phase 3: LP refinement")
    p3_end = t0 + (deadline - t0) * 0.92
    if _scipy_opt is not None and best_p is not None and time.monotonic() < p3_end:
        lp_n = int(best_p.size)
        alloc = min(200.0, max(30.0, p3_end - time.monotonic()))
        p_lp, f_lp = lp_improve(best_p, alloc)
        record(p_lp, f_lp)
        print("  LP: %.6f -> %.6f" % (best_score, f_lp))

        # Second LP pass with fresh start from best
        if time.monotonic() < p3_end and best_p is not None:
            rem = p3_end - time.monotonic()
            if rem > 30.0:
                p_lp2, f_lp2 = lp_improve(best_p, rem)
                record(p_lp2, f_lp2)
                print("  LP2: -> %.6f" % f_lp2)

    # Phase 4: Final polishing and validation
    print("Phase 4: Final polishing")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30.0:
            # One more coordinate polish with tighter budget
            alloc = min(rem - 10.0, 120.0)
            p_final, f_final = coordinate_polish(best_p, max(5.0, alloc))
            record(p_final, f_final)
            print("  Final polish: -> %.6f" % f_final)

        # Short Adam on the polished result
        rem = deadline - time.monotonic()
        if rem > 20.0:
            p_final2, f_final2 = adam_optimize(best_p, max(5.0, rem - 10.0),
                                                seed=int(seed) + 99999,
                                                alpha_start=15.0, alpha_end=2.0)
            record(p_final2, f_final2)
            print("  Final Adam: -> %.6f" % f_final2)

        # One last coordinate polish
        rem = deadline - time.monotonic()
        if rem > 15.0 and best_p is not None:
            p_last, f_last = coordinate_polish(best_p, max(5.0, rem - 8.0))
            record(p_last, f_last)
            print("  Last polish: -> %.6f" % f_last)

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score(best_p)

    print("Final best score: %.6f at n=%d" % (best_score, best_p.size))

    # Validate against the official evaluator
    final_seq = scale_output(best_p, 10.0)
    try:
        val = evaluate_sequence(final_seq)
        print("Official evaluation: %.6f" % val)
    except Exception as ex:
        print("Warning: official eval failed: %s" % str(ex))

    return final_seq
