    """Compute the grad of (1/eps)*log(sum exp(eps*(s*s)[k]))"""
    n = len(s)
    conv = np.convolve(s, s)
    weights = np.exp(epsilon * conv - epsilon * conv.max())  # numerical stability

    return grad / epsilon

"""
Multi-strategy optimization for minimizing 2*n*max(conv(s,s))/(sum(s))^2.
Uses smoothed gradient descent, LP-guided search, coordinate descent,
and population-based search with diverse initializations.
"""
import numpy as np
from scipy.optimize import linprog
import time
from numpy.lib.stride_tricks import sliding_window_view

def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where (A @ b)[k] = (s*b)[k]."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def smoothed_gradient(s, epsilon):
    """Compute gradient of (1/eps)*log(sum_k exp(eps*(s*s)[k]))."""
    n = len(s)
    conv = np.convolve(s, s)
    # Softmax weights (numerically stable)
    logits = epsilon * conv
    logits -= logits.max()
    weights = np.exp(logits)
    weights /= weights.sum()
    # grad[j] = (1/eps) * sum_k weights[k] * s[k-j] for valid k-j
    # = (1/eps) * sum_{m=0}^{n-1} weights[j+m] * s[m]
    windows = sliding_window_view(weights, n)  # shape (n, n)
    grad = (windows @ s) / epsilon
    return grad

                s = s / s.sum()
            
            cur_score = fast_score(s)
            if cur_score < best_score:

Multi-strategy optimization for minimizing 2*n*max(conv(s,s))/(sum(s))^2.
Uses smoothed power-sum gradient descent, LP-guided search, excess-based
projections, and population-based exploration with diverse initializations.

import numpy as np
from scipy.optimize import linprog
import time
from numpy.lib.stride_tricks import sliding_window_view


def fast_score(s):
    """Compute 2*n*max(conv(s,s))/(sum(s))^2 efficiently."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where (A @ b)[k] = (s*b)[k]."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_direction(s, M, time_limit=3.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M for all k, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def power_sum_gradient(s, alpha):
    """Compute gradient of (1/(2n-1))*sum_k conv_k^alpha."""
    n = len(s)
    conv = np.convolve(s, s)
    # Weights: conv_k^(alpha-1)
    if alpha == 1.0:
        weights = np.ones_like(conv) / (2 * n - 1)
    else:
        weights = np.power(np.maximum(conv, 0), alpha - 1)
        weights /= (2 * n - 1)
    # grad[j] = sum_k weights[k] * s[k-j] for valid k-j
    windows = sliding_window_view(weights, n)
    grad = windows @ s
    return grad, conv


def excess_gradient(s, threshold_frac=1.0):
    """Gradient focusing only on lags exceeding threshold."""
    n = len(s)
    conv = np.convolve(s, s)
    mean_conv = conv.sum() / (2 * n - 1)
    threshold = threshold_frac * mean_conv
    # Soft indicator: how much each lag exceeds threshold
    excess = np.maximum(conv - threshold, 0)
    if excess.sum() < 1e-15:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = excess / excess.sum()
    windows = sliding_window_view(weights, n)
    grad = windows @ s
    return grad, conv, np.max(conv)

        print(f"n={n}, remaining time: {remaining:.0f}s")
        print(f"Current global best: {global_best_score:.6f}")
        
        # Generate initializations
        inits = generate_initializations(n, rng)
        
        # Add resized height_sequence_1
        if hseq is not None:
            if hseq.size == n:
                inits.append(hseq.copy())
            else:
                x_old = np.linspace(0, 1, hseq.size)
                x_new = np.linspace(0, 1, n)
                h_r = np.interp(x_new, x_old, hseq)
                h_r = np.maximum(h_r, 0.001)
                inits.append(h_r)
        
        # Score and sort
        scored = []
        for s in inits:
            s = np.maximum(s, 0).astype(float)
            if s.sum() < 1e-10:
                continue
            s = s / s.sum()
            sc = fast_score(s)
            scored.append((sc, s))
        
        scored.sort(key=lambda x: x[0])
        
        print(f"  Generated {len(scored)} initializations")
        print(f"  Top 5 init scores: {[f'{s:.4f}' for s, _ in scored[:5]]}")
        
        # If we already beat the target, focus more on refinement
        target = 1.5027
        
        # Allocate time: top candidates get more time
        top_k = min(8, len(scored))
        time_per_main = max(15, (remaining - 60) / top_k)
        
        for rank in range(top_k):
            if time.monotonic() > deadline - 40:
                break
            
            sc_init, s_init = scored[rank]
            if sc_init > global_best_score * 1.5 and global_best_score < float('inf'):
                continue  # Skip if much worse than current best
            
            print(f"  [{rank+1}/{top_k}] Init score: {sc_init:.6f}", end="")
            
            # Strategy 1: Power-sum gradient descent
            sub_dl = min(deadline - 40, time.monotonic() + time_per_main * 0.4)
            best_s, best_sc = gradient_descent_v2(s_init.copy(), sub_dl, n_iters=15000, method='power')
            print(f" -> GD: {best_sc:.6f}", end="")
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f" *** NEW GLOBAL BEST ***", end="")
            
            # Strategy 2: Excess-based gradient
            if time.monotonic() < deadline - 40:
                sub_dl = min(deadline - 40, time.monotonic() + time_per_main * 0.3)
                best_s2, best_sc2 = gradient_descent_v2(best_s.copy(), sub_dl, n_iters=10000, method='excess')
                print(f" -> ExcessGD: {best_sc2:.6f}", end="")
                if best_sc2 < global_best_score:
                    global_best_score = best_sc2
                    global_best_seq = best_s2.copy()
                    print(f" *** NEW GLOBAL BEST ***", end="")
                best_s = best_s2 if best_sc2 < best_sc else best_s
            
            # Strategy 3: LP search (expensive but powerful)
            if time.monotonic() < deadline - 60 and n <= 384:
                sub_dl = min(deadline - 60, time.monotonic() + time_per_main * 0.4)
                best_s3, best_sc3 = lp_search(best_s.copy(), sub_dl, max_iters=200, rng=rng)
                print(f" -> LP: {best_sc3:.6f}", end="")
                if best_sc3 < global_best_score:
                    global_best_score = best_sc3
                    global_best_seq = best_s3.copy()
                    print(f" *** NEW GLOBAL BEST ***", end="")
                best_s = best_s3 if best_sc3 < best_sc else best_s
            
            # Strategy 4: Perturbation refinement
            if time.monotonic() < deadline - 40:
                sub_dl = min(deadline - 40, time.monotonic() + min(30, time_per_main * 0.3))
                best_s4, best_sc4 = refine_with_perturbation(best_s.copy(), sub_dl, rng, n_refinements=20)
                print(f" -> Refine: {best_sc4:.6f}")
                if best_sc4 < global_best_score:
                    global_best_score = best_sc4
                    global_best_seq = best_s4.copy()
                    print(f" *** NEW GLOBAL BEST ***")
            else:
                print()
        
        # Random restarts
        n_restarts = 0
        while time.monotonic() < deadline - 40 and n_restarts < 8:
            r = rng.random(n) + 0.01
            r = r / r.sum()
            sub_dl = min(deadline - 40, time.monotonic() + 20)
            best_s, best_sc = gradient_descent_v2(r, sub_dl, n_iters=3000, method='power')
            n_restarts += 1
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  Random restart {n_restarts}: {best_sc:.6f} *** NEW BEST ***")
    
    # Final: if we have a good solution, do one more refinement pass
    if global_best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal refinement pass...")
        sub_dl = deadline - 5
        best_final, best_sc_final = gradient_descent_v2(
            global_best_seq.copy(), sub_dl, n_iters=20000, method='power')
        if best_sc_final < global_best_score:
            global_best_score = best_sc_final
            global_best_seq = best_final.copy()
            print(f"  After final refinement: {global_best_score:.6f}")
        
        # Also try LP on the final best if n is small enough
        if len(global_best_seq) <= 384 and time.monotonic() < deadline - 10:
            best_lp, best_sc_lp = lp_search(global_best_seq.copy(), deadline - 5,
                                             max_iters=100, rng=rng)
            if best_sc_lp < global_best_score:
                global_best_score = best_sc_lp
                global_best_seq = best_lp.copy()
                print(f"  After final LP: {global_best_score:.6f}")
    
    # Verification
    if global_best_seq is not None:
        verified_score = float(evaluate_sequence(global_best_seq.tolist()))
        print(f"\n{'='*50}")
        print(f"Final result: score={verified_score:.6f}, length={len(global_best_seq)}")
        print(f"Target: {1.5027}")
        
        result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
        return result
    else:
        # Fallback
        s = np.ones(256) / 256
        print("Falling back to uniform sequence!")
        return s.tolist()
