        r = rng.dirichlet(np.ones(n))
        seeds.append(r)
    # 8. Two-peak
    for c1, c2 in [(0.3, 0.7), (0.25, 0.75), (0.2, 0.8)]:
        x = np.arange(n)
        g = np.exp(-0.5 * ((x - c1 * n) / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - c2 * n) / (0.15 * n)) ** 2)
        if g.sum() > 0:
            seeds.append(g / g.sum())
    # Deduplicate (keep unique seeds)
    unique_seeds = []
    for s in seeds:
        s = proj_simplex(s, 1.0)
        if s.sum() <= 0:
            continue
        is_dup = False
        for us in unique_seeds:
            if np.linalg.norm(s - us) < 1e-6:
                is_dup = True
                break
        if not is_dup:
            unique_seeds.append(s)
    return unique_seeds


def run_optimization(a0, m, n, t_end, rng, label=""):
    """Run gradient descent from seed a0 until time budget. Returns (best_a, best_score)."""
    a = a0.copy()
    a = proj_simplex(a, 1.0)
    best_a = a.copy()
    best_mc, _ = true_max_conv(a, m)
    best_score = 2.0 * n * best_mc
    buf = np.zeros(n)
    momentum = 0.85
    tau = 0.15
    top_k = 30
    lr = 0.5
    step = 0
    stall_count = 0
    stall_threshold_mc = best_mc
    
    while time.monotonic() < t_end:
        c = conv_sq(a, m)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()
            best_score = 2.0 * n * mc
            stall_count = 0
            stall_threshold_mc = mc * (1 + 1e-6)
        else:
            stall_count += 1
        
        step += 1
        # Compute gradient
        _, grad = smoothed_grad(a, c, top_k, tau, m)
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad = grad / gn
        
        # Momentum step
        buf = momentum * buf - lr * grad
        a_new = a + lr * buf
        a = proj_simplex(a_new, 1.0)
        
        # Adaptive scheduling
        if step % 500 == 0:
            tau = max(0.02, tau * 0.85)
            top_k = max(5, int(top_k * 0.75))
            momentum = min(0.95, momentum + 0.005)
        
        if step % 200 == 0:
            lr = lr * 0.97
        
        # Stall restart: jump to a perturbed version of best
        if stall_count > 3000:
            a = best_a * np.exp(rng.normal(0, 0.1, n))
            a = proj_simplex(a, 1.0)
            buf = np.zeros(n)
            tau = 0.3
            top_k = 50
            lr = 0.5
            stall_count = 0
            stall_threshold_mc = best_mc
        
        # Periodic bigger jump
        if step % 10000 == 5000:
            jump = 0.05 + 0.05 * rng.random()
            a = best_a * np.exp(rng.normal(0, jump, n))
            a = proj_simplex(a, 1.0)
            buf = np.zeros(n)
            tau = 0.25
            top_k = 40
            lr = 0.4
            stall_count = 0
    
    # Final check
    mc, _ = true_max_conv(a, m)
    if mc < best_mc:
        best_mc = mc
        best_a = a.copy()
        best_score = 2.0 * n * mc
    
    return best_a, best_score


def coordinate_polish(a, m, n, t_end):
    """Coordinate-wise optimization: for each coordinate, find best value."""
    a = a.copy()
    a = proj_simplex(a, 1.0)
    best_mc, _ = true_max_conv(a, m)
    improved = True
    while improved and time.monotonic() < t_end:
        improved = False
        n_iter = 0
        while n_iter < n and time.monotonic() < t_end:
            # Pick a random coordinate (or cycle)
            j = n_iter % n
            old_val = a[j]
            # Try a few values around current
            best_local = old_val
            best_local_mc = best_mc
            # Binary search or golden section for best a[j]
            # Simple: try a grid
            for frac in [0.5, 0.8, 1.2, 1.5, 2.0, 0.0, 3.0, 0.3]:
                trial = a.copy()
                old_sum = trial.sum()
                trial[j] = old_val * frac
                # Renormalize to sum 1
                trial_sum = trial.sum()
                if trial_sum <= 0:
                    continue
                trial = trial / trial_sum
                mc, _ = true_max_conv(trial, m)
                if mc < best_local_mc:
                    best_local_mc = mc
                    best_local = trial[j]
            if best_local_mc < best_mc - 1e-12:
                trial = a.copy()
                trial[j] = best_local
                trial_sum = trial.sum()
                if trial_sum > 0:
                    trial = trial / trial_sum
                    a = trial
                    best_mc = best_local_mc
                    improved = True
            n_iter += 1
    return a, 2.0 * n * best_mc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    Uses multi-start smoothed-subgradient descent with adaptive scheduling,
    diverse seeds, peak-shaving restarts, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_end = t_start + max(0.0, float(budget_s)) - 5.0

    # Get base sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
        if base.ndim != 1 or base.size < 4 or not np.all(np.isfinite(base)):
            raise ValueError("bad base")
    except Exception:
        base = np.ones(512, dtype=np.float64)
        print("WARNING: using default base of length 512")

    n = base.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    print(f"n={n}, FFT size m={m}")

    # Create seeds
    seeds = make_seeds(base, n, rng)
    print(f"Created {len(seeds)} seeds")

    # Evaluate initial scores
    best_a = seeds[0].copy()
    best_score = float('inf')
    for s in seeds[:5]:  # Evaluate first few
        mc, _ = true_max_conv(s, m)
        sc = 2.0 * n * mc
        if sc < best_score:
            best_score = sc
            best_a = s.copy()
    print(f"Initial best score: {best_score:.6f}")

    # Phase 1: Run optimization on each seed with staggered times
    # Allocate time: give more time to better seeds
    n_seeds = min(len(seeds), 15)  # Limit number of seeds
    active_seeds = seeds[:n_seeds]
    
    # Run in rounds, allocating time dynamically
    round_time = 60.0  # seconds per round per seed
    while time.monotonic() < t_end - 30.0:
        rem = t_end - time.monotonic() - 10.0
        if rem <= 0:
            break
        per_seed = max(5.0, min(round_time, rem / max(1, len(active_seeds))))
        
        still_active = []
        for i, s in enumerate(active_seeds):
            tower = time.monotonic() + per_seed
            a_best, sc = run_optimization(s, m, n, tower, rng, label=f"seed{i}")
            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
                t_left = t_end - time.monotonic()
                print(f"NEW BEST: {sc:.6f} (seed {i}, t_left={t_left:.1f}s)")
            # Carry forward
            if t_end - time.monotonic() > 20.0:
                still_active.append(a_best)
        
        active_seeds = still_active
        # Add some new seeds based on best
        if len(active_seeds) < 10 and time.monotonic() < t_end - 60.0:
            for _ in range(3):
                jumpscale = 0.05 + 0.1 * rng.random()
                new_s = best_a * np.exp(rng.normal(0, jumpscale, n))
                new_s = proj_simplex(new_s, 1.0)
                active_seeds.append(new_s)
        
        # Reduce round time as we go
        round_time *= 0.7

    # Phase 2: Coordinate polish on the best
    if time.monotonic() < t_end - 10.0:
        print(f"Starting coordinate polish, score={best_score:.6f}")
        a_polished, sc_polished = coordinate_polish(best_a, m, n, t_end - 10.0)
        if sc_polished < best_score:
            best_score = sc_polished
            best_a = a_polished
            print(f"After polish: {best_score:.6f}")

    # Phase 3: Final gradient fine-tuning
    if time.monotonic() < t_end - 5.0:
        a_final, sc_final = run_optimization(best_a, m, n, t_end - 3.0, rng, label="final")
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final

    # Validate with official evaluator
    out = np.clip(best_a.astype(np.float64), 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0)
    
    try:
        official_score = float(evaluate_sequence([float(x) for x in out]))
        print(f"Official score: {official_score:.6f}")
        if official_score < best_score:
            best_score = official_score
    except Exception as e:
        print(f"Evaluator error: {e}")

    # One more try: evaluate a slightly modified version
    for scale in [0.9, 1.0, 1.1]:
        try:
            trial = out * scale
            trial = np.clip(trial, 0.0, 1000.0)
            s = float(evaluate_sequence([float(x) for x in trial]))
            if s < best_score:
                best_score = s
                out = trial
                print(f"Scale {scale}: {s:.6f}")
        except Exception:
            pass

    print(f"FINAL: score={best_score:.6f}, n={n}")
    return [float(x) for x in out]
