Log-space gradient descent with FFT-based exact gradients to minimize
2*n*max(convolve(a,a)) / (sum(a))^2 for non-negative sequences.
Uses softmax parameterization, Adam optimizer, and adaptive smoothed-max.
"""

import numpy as np
import time
import math


def _conv_sq(a, m):
    """Linear convolution a*a via FFT, returns array of length 2*n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _softmax(theta):
    """Stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.full(theta.shape, 1.0 / theta.size)
    return e / s


def _grad_logspace(a, c, w, m2, n):
    """
    Compute gradient of smoothed objective w.r.t. log-space params theta.
    grad[j] = 2*a[j] * (D[j] - S_w)
    where D[j] = sum_i a[i]*w[i+j], S_w = sum_k w[k]*c[k].
    D computed via FFT: D = ifft(fft(a_rev, m2) * fft(w_padded, m2))[n-1:2n-1]
    """
    a_rev = a[::-1]
    w_pad = np.zeros(m2)
    w_pad[:2 * n - 1] = w
    fa = np.fft.rfft(a_rev, m2)
    fw = np.fft.rfft(w_pad, m2)
    conv = np.fft.irfft(fa * fw, m2)
    D = conv[n - 1:2 * n - 1]
    S_w = np.dot(w, c)
    grad = 2.0 * a * (D - S_w)
    return grad
