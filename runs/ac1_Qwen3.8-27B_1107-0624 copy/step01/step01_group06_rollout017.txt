"""
Optimizes step function coefficients to minimize J(a) = 2n*max(a*a)/(sum(a))^2.
Uses hybrid gradient (log-sum-exp + true subgradient), Adam optimizer,
spectral equalization, local search, and coordinate descent.
Tries multiple sequence lengths and initializations.

import numpy as np
import time

def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def lse_gradient(a, tau):
    """Log-sum-exp smoothed gradient."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad

def subgradient_step(a):
    """True subgradient of J at a."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    max_c = np.max(c)
    # Find argmax positions (with tolerance for numerical issues)
    threshold = max_c - 1e-10 * max(max_c, 1e-10)
    max_indices = np.where(c >= threshold)[0]
    # Subgradient of max(c)
    sub_c = np.zeros(n)
    for m in max_indices:
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        idx = np.arange(lo, hi + 1)
        sub_c[idx] += 2.0 * a[m - idx]
    sub_c /= len(max_indices)
    # Subgradient of J
    grad = (2.0 * n / (S * S)) * sub_c - (4.0 * n * max_c / (S * S * S)) * a
    J = 2.0 * n * max_c / (S * S)
    return J, grad

def make_gaussian(n, sigma_frac):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))

def make_hann(n):
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))

def make_blackman(n):
    n1 = n - 1
    x = np.arange(n)
    return 0.42 - 0.5 * np.cos(2 * np.pi * x / n1) + 0.08 * np.cos(4 * np.pi * x / n1)

def make_flat_top(n, frac=0.5):
    """Flat top with raised cosine edges."""
    a = np.zeros(n)
    edge = int(frac * n / 2)
    mid_start = edge
    mid_end = n - edge - 1
    a[mid_start:mid_end+1] = 1.0
    # Raised cosine taper
    for i in range(edge):
        t = (i + 1) / (edge + 1)
        a[i] = 0.5 * (1 - np.cos(np.pi * t))
        a[n-1-i] = a[i]
    return a

def adam_update(a, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    a = np.maximum(a, 0.0)
    return a, m, v, t

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    start_time = time.monotonic()
    use_subgrad = rng.random() < use_subgrad_frac
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (1.0 + t / 50000.0) ** (-0.5)
        
        # Get gradient
        if use_subgrad and t > 100:
            J_true, grad_sg = subgradient_step(a)
            J_lse, grad_lse = lse_gradient(a, tau)
            # Blend
            grad = (1 - use_subgrad_frac) * grad_lse + use_subgrad_frac * grad_sg
            J = J_lse  # Use smoothed J for tracking
        else:
            J, grad = lse_gradient(a, tau)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        a, m, v, t = adam_update(a, grad, m, v, t, lr)
        
        # Periodic check
        if t % 5000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
    
    return best_a, best_J

def run_local_search(a_init, budget_s, seed):
    """Local search with structured perturbations."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    step_scale = 0.03
    
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            # Block perturbation
            width = int(rng.integers(max(1, n//100), max(2, n//10)))
            start = int(rng.integers(0, max(1, n-width)))
            cand = a.copy()
            cand[start:start+width] *= np.exp(rng.normal(0, step_scale, width))
        elif r < 0.6:
            # Sinusoidal perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2*np.pi)
            amp = step_scale * np.sqrt(n)
            cand = a * (1 + amp * np.sin(2*np.pi*freq*np.arange(n)/n + phase))
            cand = np.maximum(cand, 0)
        elif r < 0.8:
            # Random individual elements
            cand = a.copy()
            n_mod = int(rng.integers(1, 10))
            idx = rng.integers(0, n, n_mod)
            cand[idx] *= np.exp(rng.normal(0, step_scale, n_mod))
            cand = np.maximum(cand, 0)
        else:
            # Two-block swap (move some mass from one region to another)
            cand = a.copy()
            w1 = int(rng.integers(n//50, n//10))
            w2 = w1
            s1 = int(rng.integers(0, n-w1))
            s2 = int(rng.integers(0, n-w2))
            if s1 == s2:
                continue
            ratio = np.sum(cand[s1:s1+w1]) / (np.sum(cand[s2:s2+w2]) + 1e-10)
            epsilon = 0.05
            cand[s1:s1+w1] *= (1 - epsilon)
            cand[s2:s2+w2] *= (1 + epsilon * ratio)
            cand = np.maximum(cand, 0)
        
        cS = np.sum(cand)
        if cS < 0.01:
            continue
        cJ = compute_J(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.97)
        else:
            step_scale = min(0.2, step_scale * 1.01)
    
    return best_a, best_J

def run_coordinate_descent(a_init, budget_s, seed):
    """Coordinate descent: optimize one element at a time."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    best_J = compute_J(a)
    best_a = a.copy()
    order = rng.permutation(n)
    
    while time.monotonic() < deadline:
        for i in order:
            # Try several values of a[i]
            base = a[i]
            if base < 1e-10:
                continue
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                cand = a.copy()
                cand[i] = base * factor
                cJ = compute_J(cand)
                if cJ < best_J:
                    best_J = cJ
                    best_a = cand.copy()
                    a = cand.copy()
        
        if time.monotonic() >= deadline:
            break
    
    return best_a, best_J

Hmm, coordinate descent with full recomputation of J each time is O(n^2) per full pass (n coordinates × O(n) to compute J). For n=6121, that's about 37 million operations per pass. Each pass might take a second or two. In 1800 seconds, we could do maybe 500-1000 passes. That should be enough for convergence.

But actually, computing J requires the full convolution, which is O(n log n) with FFT. So each coordinate step is O(n log n), and a full pass is O(n^2 log n). For n=6121: 6121^2 * 13 ≈ 487 million operations. That might be too slow per pass.

Let me optimize: instead of recomputing the full convolution for each coordinate change, I can update it incrementally. If I change a[i] to a[i]+δ, then:
c_new[k] = c_old[k] + δ*(a[k-i] + a[k-i]) + δ^2 (if k=2i)
         = c_old[k] + 2δ*a[k-i] + δ^2*[k==2i]

This is O(n) per coordinate step (updating all c[k]), much faster than O(n log n).

Actually, for the purpose of finding the optimal a[i], I need to minimize max(c_new[k]) over i, which requires checking all k. But I can do this in O(n) per trial value of a[i].

Let me implement this more carefully:

For a fixed a[i] = v, the new convolution is:
c_new[k] = c_old[k] - old_ai * (a[k-i] + a[i-(k-i)]) + v * (a[k-i] + a[i-(k-i)])
Wait, this isn't quite right because c[k] = sum_j a[j]*a[k-j], and changing a[i] affects only the terms where j=i or k-j=i.

c_old[k] = a[i]*a[k-i] + a[k-i]*a[i] + sum_{j≠i, k-j≠i} a[j]*a[k-j]
         = 2*a[i]*a[k-i] + rest (if i ≠ k-i)
         = a[i]^2 + rest (if i = k-i, i.e., k=2i)

c_new[k] = 2*v*a[k-i] + rest (if i ≠ k-i)
          = v^2 + rest (if i = k-i)

So c_new[k] = c_old[k] + (v - old_ai) * 2 * a[k-i] (for k ≠ 2i where k-i is valid)
             = c_old[k] + (v^2 - old_ai^2) (for k = 2i)
             = c_old[k] (for k where k-i is out of range)

This is O(n) to compute c_new from c_old. Then finding max(c_new) is O(n). So each coordinate trial is O(n).

For the binary search / golden section to find optimal v, I'd need O(log(1/ε)) trials. Total per coordinate: O(n * log(1/ε)). For n=6121 and 10 iterations: ~61K operations per coordinate. For all n coordinates: ~373 million per pass. That's still a lot but more manageable.

Actually, let me simplify: instead of finding the exact optimal v, just try a few multiplicative factors (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5) and pick the best. That's 7 evaluations × O(n) per coordinate = 7n per coordinate, or 7n^2 per pass. For n=6121: ~263 million. Still a lot.

import numpy as np
import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S*S)

def lse_grad(a, tau):
    n = len(a)
    size = 1
    while size < 3*n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa*fa, size)[:2*n-1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    cmax = np.max(c)
    sh = (c - cmax) / tau
    es = np.exp(sh)
    se = np.sum(es)
    Tmax = cmax + tau * np.log(se)
    p = es / se
    J = 2.0*n*Tmax/(S*S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa)*fp, size)[:n]
    grad = 4.0*n*cross/(S*S) - 4.0*n*Tmax/(S*S*S)
    return J, grad

def true_subgrad(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    maxc = np.max(c)
    tol = maxc * 1e-8
    mpos = np.where(c >= maxc - tol)[0]
    sub_c = np.zeros(n)
    for m in mpos:
        lo = max(0, m-n+1)
        hi = min(m, n-1)
        idx = np.arange(lo, hi+1)
        sub_c[idx] += 2.0 * a[m-idx]
    sub_c /= len(mpos)
    J = 2.0*n*maxc/(S*S)
    grad = 2.0*n*sub_c/(S*S) - 4.0*n*maxc*a/(S*S*S)
    return J, grad
