```python
"""
Search for low-score non-negative step-function heights.

The program optimizes normalized sequences a (sum(a)=1) for the score
    2 * n * max_k conv(a, a)[k].

The main local search uses an L-BFGS iteration on log-softmax parameters with a
smoothed-maximum objective, accepting only moves that lower the exact discrete
score.  Subgradient, random-mass-transfer, and a small linear-programming
refiner are used as auxiliary escapes.  The best sequence is continuously kept
and is returned before the hard time limit.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


class TimeLimit(Exception):
    pass


def normalize_nonneg(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    return a * (float(target) / s)


def conv_self(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    c = c[:need].copy()
    c[c < 0.0] = 0.0
    return c


def exact_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return float("inf")
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    c = conv_self(a)
    m = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m / (s * s))


def shift_correlation(p, a):
    """
    Return r[i] = sum_t a[t] p[t+i], for i=0..len(a)-1.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    p = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    L = p.size
    if n == 0 or L == 0:
        return np.zeros(max(n, L), dtype=np.float64)
    p_rev = p[::-1]
    need = n + L - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    out = np.fft.irfft(fa * fp, size)
    idx = (2 * n - 2) - np.arange(n)
    r = out[idx]
    return r


def resize_to_length(a, N):
    a = normalize_nonneg(a)
    N = int(max(1, N))
    m = a.size
    if m == N:
        return a
    if m > N:
        out = np.zeros(N, dtype=np.float64)
        for i in range(N):
            lo = (i * m) // N
            hi = ((i + 1) * m) // N
            if hi > lo:
                out[i] = float(np.sum(a[lo:hi]))
        return normalize_nonneg(out)
    if m <= 1:
        return np.full(N, 1.0 / float(N), dtype=np.float64)
    x = np.linspace(0.0, float(m - 1), N)
    i0 = np.floor(x).astype(np.int64)
    i1 = np.minimum(i0 + 1, m - 1)
    frac = x - i0
    out = a[i0] * (1.0 - frac) + a[i1] * frac
    return normalize_nonneg(np.maximum(out, 0.0))

    return a / s

        state["seq"] = seq
        if not np.isfinite(old) or score < old - 1e-5:
            print(f"update_best score={score:.12f} n={seq.size}", flush=True)
        return True
    return False


def make_one_step_seed(N, alpha=0.20, ratio=2.56, reverse=False):
    N = int(max(2, N))
    a = np.zeros(N, dtype=np.float64)
    k = int(round(float(alpha) * float(N)))
    k = int(max(1, min(N - 1, k)))
    if not reverse:
        a[:k] = float(ratio)
        a[k:] = 1.0
    else:
        a[N - k:] = float(ratio)
        a[:N - k] = 1.0
    return normalize_nonneg(a)


def make_stair_seed(N, spec, reverse=False):
    N = int(max(4, N))
    a = np.zeros(N, dtype=np.float64)
    items = list(spec)
    if reverse:
        items = list(reversed(items))
    pos = 0
    total_frac = 0.0
    for frac, h in items:
        total_frac += float(frac)
    if not reverse:
        for idx, (frac, h) in enumerate(items):
            if idx == len(items) - 1:
                L = N - pos
            else:
                L = int(round(float(frac) * float(N)))
            L = int(max(1, min(L, N - pos)))
            a[pos:pos + L] = float(h)
            pos += L
            if pos >= N:
                break
    else:
        # reverse handled by items; fill from left
        for idx, (frac, h) in enumerate(items):
            if idx == len(items) - 1:
                L = N - pos
            else:
                L = int(round(float(frac) * float(N)))
            L = int(max(1, min(L, N - pos)))
            a[pos:pos + L] = float(h)
            pos += L
            if pos >= N:
                break
    if pos < N:
        a[pos:] = float(items[-1][1])
    return normalize_nonneg(a)


def make_block_seed(N, rng, K=16):
    N = int(max(4, N))
    K = int(max(2, min(K, N)))
    lengths = (rng.dirichlet(np.ones(K)) * float(N)).astype(np.int64)
    lengths = np.maximum(1, lengths)
    while int(np.sum(lengths)) < N:
        lengths[int(rng.integers(K))] += 1
    while int(np.sum(lengths)) > N:
        idx = int(np.argmax(lengths))
        if lengths[idx] > 1:
            lengths[idx] -= 1
        else:
            break
    w = np.linspace(1.0, 2.4, K)
    heights = np.exp(rng.normal(0.0, 0.55, size=K)) * w
    heights = np.maximum(heights, 0.25)
    a = np.zeros(N, dtype=np.float64)
    pos = 0
    for L, h in zip(lengths, heights):
        L = int(L)
        if L <= 0:
            continue
        a[pos:pos + L] = float(h)
        pos += L
        if pos >= N:
            break
    if pos < N:
        a[pos:] = float(heights[-1])
    return normalize_nonneg(a)


def make_smooth_seed(N, rng):
    N = int(max(8, N))
    base = rng.normal(0.0, 1.0, size=N)
    wlen = int(max(2, min(N, 1 + N // 30)))
    kernel = np.ones(wlen, dtype=np.float64)
    kernel /= kernel.sum()
    base = np.convolve(base, kernel, mode="same")
    base = np.convolve(base, kernel, mode="same")
    a = np.exp(0.95 * base)
    return normalize_nonneg(a)


def make_gaussian_seed(N, rng):
    N = int(max(8, N))
    x = np.linspace(0.0, 1.0, N)
    mu = float(rng.uniform(0.2, 0.8))
    sigma = float(rng.uniform(0.11, 0.35))
    tilt = float(rng.normal(0.0, 0.7))
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2 + tilt * x)
    return normalize_nonneg(a)


def make_exponential_seed(N, rng):
    N = int(max(8, N))
    x = np.linspace(0.0, 1.0, N)
    alphas = (-2.2, -1.2, -0.4, 0.4, 1.2, 2.4)
    alpha = float(rng.choice(alphas))
    a = np.exp(alpha * x) + 0.05
    return normalize_nonneg(a)


def random_perturb(a, rng, scale=0.1):
    h = normalize_nonneg(a).copy()
    n = h.size
    if n == 0:
        return h
    scale = float(max(1e-4, scale))
    h = h * np.exp(rng.normal(0.0, scale, size=n))
    lo = int(max(1, min(n, n // 25)))
    hi = int(max(lo + 1, min(n, n // 5)))
    hi = int(min(hi, n))
    if hi > lo:
        width = int(rng.integers(lo, hi + 1))
    else:
        width = n
    start = int(rng.integers(0, max(1, n - width + 1)))
    if start + width > n:
        start = 0
        width = n
    h[start:start + width] *= np.exp(rng.normal(0.0, scale * 1.8, size=width))
    return normalize_nonneg(h)


def get_parameterization(a):
    a = normalize_nonneg(a)
    n = a.size
    fixed = int(np.argmax(a)) if n else 0
    if fixed < 0 or fixed >= n:
        fixed = 0
    if fixed == 0:
        left = np.empty(0, dtype=np.int64)
    else:
        left = np.arange(0, fixed, dtype=np.int64)
    if fixed + 1 >= n:
        right = np.empty(0, dtype=np.int64)
    else:
        right = np.arange(fixed + 1, n, dtype=np.int64)
    free = np.concatenate((left, right)).astype(np.int64)
    return fixed, free


def make_w(a, fixed_index, free_indices):
    a = normalize_nonneg(a)
    mx = float(np.max(a))
    eps = max(1e-300, 1e-12 * mx)
    a_safe = np.maximum(a, eps)
    x = np.log(a_safe)
    x -= x[fixed_index]
    x = np.clip(x, -700.0, 700.0)
    return np.asarray(x[free_indices], dtype=np.float64)


def a_from_w(w, n, fixed_index, free_indices):
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64).ravel()
    if w.size != free_indices.size:
        w = np.zeros(free_indices.size, dtype=np.float64)
    if not np.all(np.isfinite(w)):
        w = np.where(np.isfinite(w), w, 0.0)
    x = np.empty(n, dtype=np.float64)
    if free_indices.size:
        x[free_indices] = w
    x[fixed_index] = 0.0
    x -= float(np.max(x))
    x = np.clip(x, -700.0, 700.0)
    a = np.exp(x)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def choose_maxiter(n, time_left):
    if n <= 128:
        base = 120
    elif n <= 256:
        base = 100
    elif n <= 512:
        base = 80
    elif n <= 1024:
        base = 60
    elif n <= 2048:
        base = 40
    else:
        base = 25
    factor = 1.0 if time_left > 120.0 else max(0.25, float(time_left) / 120.0)
    return int(max(5, base * factor))


class LbfgsContext:
    def __init__(self, reference_a, reference_score, temp, deadline, state,
                 n, fixed_index, free_indices):
        self.n = int(n)
        self.fixed_index = int(fixed_index)
        self.free_indices = np.asarray(free_indices, dtype=np.int64)
        self.temp = float(max(1e-12, temp))
        self.deadline = float(deadline)
        self.state = state
        self.obj_scale = 2.0 * float(self.n)
        self.best_a = np.asarray(reference_a, dtype=np.float64).copy()
        self.best_sc = float(reference_score)
        self.cached_w = None
        self.cached_g = None
        self.last_a = None
        self.last_m = 0.0
        self.last_s = 1.0

    def compute(self, w):
        w = np.asarray(w, dtype=np.float64).ravel()
        if w.size != self.free_indices.size:
            w = np.zeros(self.free_indices.size, dtype=np.float64)
        if not np.all(np.isfinite(w)):
            w = np.where(np.isfinite(w), w, 0.0)

        a = a_from_w(w, self.n, self.fixed_index, self.free_indices)
        s = float(np.sum(a))
        c = conv_self(a)
        if c.size == 0:
            m = 1e-300
        else:
            m = float(np.max(c))
        if not np.isfinite(m) or m <= 1e-300:
            c = np.maximum(c, 1e-300)
            m = float(np.max(c))

        tau = self.temp
        z = (c - m) / tau
        zmax = float(np.max(z))
        ez = np.exp(z - zmax)
        ss = float(np.sum(ez))
        if not np.isfinite(ss) or ss <= 0.0:
            p = np.zeros_like(c, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
            lse = m
        else:
            p = ez / ss
            lse = m + tau * (zmax + float(np.log(ss)))

        g_a = 2.0 * shift_correlation(p, a)
        g_a = np.asarray(g_a, dtype=np.float64)
        dot = float(np.dot(g_a, a))
        g_full = a * (g_a - dot)
        g_free = self.obj_scale * g_full[self.free_indices]
        if not np.all(np.isfinite(g_free)):
            g_free = np.zeros_like(g_free, dtype=np.float64)

        if not np.isfinite(lse):
            obj = 1e10
        else:
            obj = float(self.obj_scale * lse)

        self.cached_w = w
        self.cached_g = g_free
        self.last_a = a
        self.last_m = m
        self.last_s = s
        return obj

    def fun(self, w):
        return self.compute(w)

    def jac(self, w):
        w = np.asarray(w, dtype=np.float64).ravel()
        if (self.cached_w is not None and w.size == self.cached_w.size and
                np.array_equal(w, self.cached_w)):
            return self.cached_g
        self.compute(w)
        return self.cached_g

    def __call__(self, pack):
        try:
            x = getattr(pack, "x", None)
            if x is not None and self.last_a is None:
                self.compute(np.asarray(x, dtype=np.float64).ravel())
            if self.last_a is not None and self.last_s > 0.0:
                exact = float(2.0 * self.n * self.last_m / (self.last_s * self.last_s))
                if np.isfinite(exact) and exact < self.best_sc - 1e-12:
                    self.best_sc = exact
                    self.best_a = np.asarray(self.last_a, dtype=np.float64).copy()
                    update_global(self.state, self.best_a, self.best_sc)
        except Exception:
            pass
        if time.monotonic() > self.deadline:
            raise TimeLimit


def lbfgs_try(start_a, ref_a, ref_sc, temp, maxiter, deadline, state, rng):
    if minimize is None:
        return None, ref_sc
    start_a = normalize_nonneg(start_a)
    n = start_a.size
    if n < 2:
        return None, ref_sc
    fixed, free = get_parameterization(start_a)
    if free.size == 0:
        sc0 = exact_score(start_a)
        if sc0 < ref_sc - 1e-12:
            return start_a.copy(), sc0
        return None, ref_sc
    w0 = make_w(start_a, fixed, free)
    ctx = LbfgsContext(ref_a, ref_sc, temp, deadline, state, n, fixed, free)
    options = {
        "maxiter": int(maxiter),
        "ftol": 1e-14,
        "gtol": 1e-7,
        "maxls": 5,
        "maxcor": 10,
        "disp": False,
    }
    res = None
    try:
        res = minimize(ctx.fun, w0, jac=ctx.jac, method="L-BFGS-B",
                       options=options, callback=ctx)
    except TimeLimit:
        res = None
    except Exception:
        res = None

    if res is not None:
        try:
            x = np.asarray(res.x, dtype=np.float64).ravel()
            a_end = a_from_w(x, n, fixed, free)
            sc_end = exact_score(a_end)
            if np.isfinite(sc_end) and sc_end < ctx.best_sc - 1e-12:
                ctx.best_sc = float(sc_end)
                ctx.best_a = a_end.copy()
                update_global(state, ctx.best_a, ctx.best_sc)
        except Exception:
            pass

    if ctx.best_sc < ref_sc - 1e-12:
        return ctx.best_a.copy(), ctx.best_sc
    return None, ref_sc


def lbfgs_refine(a, sc, time_limit, state, rng):
    best = normalize_nonneg(a)
    best_sc = exact_score(best)
    update_global(state, best, best_sc)
    start_sc = best_sc
    n = best.size
    if n < 2 or minimize is None or time_limit <= 5.0:
        return best, best_sc, False

    deadline = time.monotonic() + max(0.0, float(time_limit))
    temp_fracs = (0.50, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001)
    ti = 0
    fails = 0
    ops = 0

    while time.monotonic() + 8.0 < deadline:
        if best_sc <= 0.0:
            break
        M = best_sc / (2.0 * float(n))
        if not np.isfinite(M) or M <= 0.0:
            M = 1.0 / (2.0 * float(n))
        frac = temp_fracs[ti % len(temp_fracs)]
        tau = frac * M
        if tau <= 1e-300:
            tau = 1e-3 * M

        maxiter = choose_maxiter(n, deadline - time.monotonic())
        start_a = best
        if ops > 0 and fails >= 2:
            scale = 0.08 + 0.04 * float(min(fails, 10))
            start_a = random_perturb(best, rng, scale=scale)

        cand, cand_sc = lbfgs_try(start_a, best, best_sc, tau, maxiter,
                                  max(0.0, deadline - 4.0), state, rng)
        if cand is not None:
            best = normalize_nonneg(cand)
            best_sc = exact_score(best)
            update_global(state, best, best_sc)
            fails = 0
            ti = 2
        else:
            fails += 1
            ti += 1
            if fails > 2 * len(temp_fracs):
                ti = 0
                fails = 1

        ops += 1
        if ops % 20 == 0:
            print(f"lbfgs_refine n={n} score={best_sc:.12f}", flush=True)

    return best, best_sc, (best_sc < start_sc - 1e-12)


def subgrad_refine(a, sc, time_limit, state, rng):
    best = normalize_nonneg(a)
    best_sc = exact_score(best)
    update_global(state, best, best_sc)
    start_sc = best_sc
    n = best.size
    if n < 2 or time_limit <= 3.0:
        return best, best_sc, False

    deadline = time.monotonic() + max(0.0, float(time_limit))
    base_frac = 0.50
    v = None
    fails = 0
    ops = 0

    while time.monotonic() + 3.0 < deadline:
        start = best
        if fails >= 3:
            start = random_perturb(best, rng, scale=0.10 + 0.05 * float(min(fails, 5)))
        cur = normalize_nonneg(start)
        cur_sc = exact_score(cur)

        if n <= 256:
            steps = 80
        elif n <= 512:
            steps = 50
        elif n <= 1024:
            steps = 30
        else:
            steps = 18

        alphas = (0.02, 0.05, 0.12, 0.30, 0.70, 1.50, 3.00)
        improved = False

        for _ in range(steps):
            if time.monotonic() + 1.0 > deadline:
                break
            c = conv_self(cur)
            Mcur = float(np.max(c)) if c.size else 0.0
            s = float(np.sum(cur))
            if Mcur <= 0.0 or not np.isfinite(Mcur) or s <= 0.0:
                break
            cur_sc = float(2.0 * n * Mcur / (s * s))

            if cur_sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = cur_sc
                update_global(state, best, best_sc)
                improved = True

            Mref = best_sc / (2.0 * float(n))
            tau = base_frac * max(Mcur, Mref, 1e-300)
            z = (c - Mcur) / tau
            zmax = float(np.max(z))
            ez = np.exp(z - zmax)
            ss = float(np.sum(ez))
            if not np.isfinite(ss) or ss <= 0.0:
                p = np.zeros_like(c, dtype=np.float64)
                p[int(np.argmax(c))] = 1.0
            else:
                p = ez / ss

            g = 2.0 * shift_correlation(p, cur)
            g = np.asarray(g, dtype=np.float64)
            g = g - float(np.mean(g))

            if v is None:
                v = g.copy()
            else:
                v = 0.8 * v + 0.2 * g

            scale = float(np.max(np.abs(v)))
            if not np.isfinite(scale) or scale <= 1e-300:
                break
            vdir = v / scale

            moved = False
            for alpha in alphas:
                h = normalize_nonneg(cur - float(alpha) * vdir)
                sc_h = exact_score(h)
                if sc_h < cur_sc - 1e-12:
                    cur = h
                    cur_sc = sc_h
                    moved = True
                    if sc_h < best_sc - 1e-12:
                        best = h.copy()
                        best_sc = sc_h
                        update_global(state, best, best_sc)
                        improved = True
                    break
            if not moved:
                # Try a steeper move without clipping the direction sign.
                pass

            if improved and cur_sc <= best_sc + 1e-12:
                v = None
                break

        if improved:
            fails = 0
            base_frac = max(0.05, base_frac * 0.75)
        else:
            fails += 1
            base_frac = min(0.90, base_frac * 1.30)

        ops += 1
        if ops % 10 == 0:
            print(f"subgrad_refine n={n} score={best_sc:.12f}", flush=True)
        if fails > 12:
            break

    return best, best_sc, (best_sc < start_sc - 1e-12)


def random_move_refine(a, sc, time_limit, state, rng):
    best = normalize_nonneg(a)
    best_sc = exact_score(best)
    update_global(state, best, best_sc)
    start_sc = best_sc
    n = best.size
    if n < 2 or time_limit <= 2.0:
        return best, best_sc, False

    deadline = time.monotonic() + max(0.0, float(time_limit))
    fracs = (0.01, 0.03, 0.08, 0.20, 0.50, 0.80)
    fails = 0

    while time.monotonic() + 2.0 < deadline:
        improved = False
        attempts = 30 if n <= 512 else 12
        for _ in range(attempts):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            for f in fracs:
                h = best.copy()
                m = float(f) * float(h[j])
                if m <= 0.0:
                    continue
                h[j] -= m
                h[i] += m
                h = normalize_nonneg(h)
                sc_h = exact_score(h)
                if sc_h < best_sc - 1e-12:
                    best = h
                    best_sc = sc_h
                    update_global(state, best, best_sc)
                    improved = True
                    break
            if improved:
                break

        if improved:
            fails = 0
        else:
            fails += 1
            if fails > 6:
                break

    return best, best_sc, (best_sc < start_sc - 1e-12)


def solve_lp_direction(a, M, rows, upper):
    if linprog is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    rows = np.asarray(rows, dtype=np.int64).ravel()
    if n < 2 or rows.size == 0 or M <= 0.0:
        return None

    A = np.zeros((rows.size, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            idx = np.arange(lo, hi, dtype=np.int64)
            A[r, lo:hi] = a[k - idx]

    cvec = np.concatenate((np.zeros(n, dtype=np.float64), np.ones(1, dtype=np.float64)))
    A_ub = np.zeros((rows.size, n + 1), dtype=np.float64)
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(rows.size, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1, dtype=np.float64)

    upper = float(max(upper, 1e-300))
    bounds = [(0.0, upper)] * n + [(0.0, 10.0 * float(M))]

    try:
        res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method="highs")
    except Exception:
        return None
    if res is not None and getattr(res, "success", False):
        h = np.asarray(res.x[:n], dtype=np.float64)
        h = np.clip(h, 0.0, None)
        s = float(np.sum(h))
        if np.isfinite(s) and s > 0.0:
            return h / s
    return None


def lp_refine(a, sc, time_limit, state, rng):
    best = normalize_nonneg(a)
    best_sc = exact_score(best)
    update_global(state, best, best_sc)
    start_sc = best_sc
    n = best.size
    if n < 2 or linprog is None or n > 1024 or time_limit <= 5.0:
        return best, best_sc, False

    deadline = time.monotonic() + max(0.0, float(time_limit))
    rows_count = min(2 * n - 1, 128 if n > 256 else 2 * n - 1)
    fails = 0

    while time.monotonic() + 5.0 < deadline:
        c = conv_self(best)
        M = best_sc / (2.0 * float(n))
        if not np.isfinite(M) or M <= 0.0:
            break
        if rows_count >= c.size:
            rows = np.arange(c.size, dtype=np.int64)
        else:
            rows = np.argpartition(c, -rows_count)[-rows_count:].astype(np.int64)
        upper = max(6.0 * float(np.max(best)), 20.0 / float(n), 1e-6)
        h = solve_lp_direction(best, M, rows, upper)
        improved = False
        if h is not None:
            cand = best.copy()
            cand_sc = best_sc
            for t in (0.10, 0.20, 0.40, 0.60, 0.80, 1.00):
                ht = normalize_nonneg((1.0 - float(t)) * best + float(t) * h)
                sc_t = exact_score(ht)
                if sc_t < cand_sc - 1e-12:
                    cand = ht
                    cand_sc = sc_t
            if cand_sc < best_sc - 1e-12:
                best = cand
                best_sc = cand_sc
                update_global(state, best, best_sc)
                improved = True
        if improved:
            fails = 0
        else:
            fails += 1
            if fails > 4:
                break

    return best, best_sc, (best_sc < start_sc - 1e-12)


def run_local(a, time_limit, state, rng):
    a = normalize_nonneg(a)
    best = a.copy()
    best_sc = exact_score(best)
    update_global(state, best, best_sc)
    if time_limit <= 4.0:
        return

    n = best.size
    deadline = time.monotonic() + max(0.0, float(time_limit))
    phase = 0
    no_improve_cycles = 0

    while time.monotonic() + 8.0 < deadline:
        old_sc = best_sc
        time_left = deadline - time.monotonic()
        if time_left <= 0.0:
            break

        if phase == 0:
            if minimize is not None:
                b, sc, _ = lbfgs_refine(best, best_sc, min(240.0, max(10.0, time_left * 0.55)), state, rng)
            else:
                b, sc, _ = subgrad_refine(best, best_sc, min(120.0, max(5.0, time_left * 0.5)), state, rng)
        elif phase == 1:
            b, sc, _ = subgrad_refine(best, best_sc, min(120.0, max(5.0, time_left * 0.33)), state, rng)
        elif phase == 2:
            b, sc, _ = random_move_refine(best, best_sc, min(80.0, max(3.0, time_left * 0.22)), state, rng)
        elif phase == 3:
            if linprog is not None and n <= 1024:
                b, sc, _ = lp_refine(best, best_sc, min(80.0, max(3.0, time_left * 0.25)), state, rng)
            else:
                b, sc = best, best_sc
            b, sc, _ = random_move_refine(b, sc, min(30.0, max(2.0, time_left * 0.10)), state, rng)
        else:
            break

        if sc < best_sc - 1e-12:
            best = normalize_nonneg(b)
            best_sc = exact_score(best)
            update_global(state, best, best_sc)
            no_improve_cycles = 0
        else:
            no_improve_cycles += 1

        phase += 1
        if phase > 3:
            phase = 0

        if no_improve_cycles >= 6:
            break

    return best, best_sc


def remaining_time(deadline):
    return max(0.0, float(deadline) - time.monotonic())


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    reserve = 90.0
    hard_deadline = time.monotonic() + max(1.0, float(budget_s) - reserve)

    state = {"score": float("inf"), "seq": None}

    fb = np.array([1.0 / 3.0, 2.0 / 3.0], dtype=np.float64)
    update_global(state, fb, exact_score(fb))
    fb256 = np.ones(256, dtype=np.float64)
    update_global(state, fb256, exact_score(fb256))

    provided = load_provided_sequence()
    if provided is not None:
        update_global(state, provided, exact_score(provided))

    quick = []
    if provided is not None:
        quick.append(resize_to_length(provided, 512))
        quick.append(resize_to_length(provided[::-1], 512))

    quick.append(make_one_step_seed(512, 0.20, 2.56, False))
    quick.append(make_one_step_seed(512, 0.24, 2.20, False))
    quick.append(make_stair_seed(
        512,
        ((0.22, 2.30), (0.24, 1.10), (0.54, 0.55)),
        False
    ))
    quick.append(make_block_seed(512, rng, 16))
    quick.append(make_block_seed(512, rng, 32))
    quick.append(make_smooth_seed(512, rng))
    quick.append(make_gaussian_seed(512, rng))
    quick.append(make_exponential_seed(512, rng))

    for seq in quick:
        if remaining_time(hard_deadline) < 12.0:
            break
        run_local(seq, min(30.0, remaining_time(hard_deadline)), state, rng)

    if state["seq"] is None:
        state_seq = make_one_step_seed(512, 0.20, 2.56, False)
        state["seq"] = state_seq
        state["score"] = exact_score(state_seq)

    # Long refinement at the best current resolution.
    if remaining_time(hard_deadline) > 40.0:
        if state["seq"].size <= 1024:
            seq = resize_to_length(state["seq"], 512)
        else:
            seq = state["seq"].copy()
        run_local(seq, min(330.0, remaining_time(hard_deadline)), state, rng)

    # Upsample and refine.
    if remaining_time(hard_deadline) > 60.0:
        seq = resize_to_length(state["seq"], 1024)
        run_local(seq, min(450.0, remaining_time(hard_deadline)), state, rng)

    if remaining_time(hard_deadline) > 60.0:
        seq = resize_to_length(state["seq"], 2048)
        run_local(seq, min(360.0, remaining_time(hard_deadline)), state, rng)

    # If still above target and there is enough remaining budget, try a short 4096 attempt.
    if (remaining_time(hard_deadline) > 120.0 and
            state["score"] > 1.5027 and
            state["seq"].size < 4096):
        seq = resize_to_length(state["seq"], 4096)
        run_local(seq, min(160.0, remaining_time(hard_deadline)), state, rng)

    # Final polish at the currently best length.
    if remaining_time(hard_deadline) > 10.0:
        run_local(state["seq"], remaining_time(hard_deadline), state, rng)

    if state["seq"] is None:
        return [float(1.0 / 3.0), float(2.0 / 3.0)]

    cand_np = np.asarray(state["seq"], dtype=np.float64).ravel()
    cand_np = np.clip(cand_np, 0.0, 1000.0)
    cand = [float(x) for x in cand_np]

    try:
        official = evaluate_sequence(cand)
        print(f"official_best score={official:.12f} n={len(cand)}", flush=True)
    except Exception:
        pass

    return cand
```
