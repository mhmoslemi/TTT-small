def _fft_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if target <= 0:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:

    lam = (cssv[rho[-1]] - target) / (rho[-1] + 1.0)
    return np.maximum(v - lam, 0.0)

def _coord_descent(a, t_end, rng, n_passes_target=500):

    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-12:
        return a, float('inf')

    max_c = c.max()

    best_a = a.copy()

    conv_len = 2 * n - 1
    
    t0 = time.monotonic()
    pass_num = 0
    coords = np.arange(n)

        rng.shuffle(coords)
        pass_num += 1
        
        for i in coords:
            v = a[i]
            if v < 1e-15:
                continue

            # k ranges from i to i+n-1 (clamped to [0, conv_len-1])
            k_lo = max(0, i)
            k_hi = min(conv_len - 1, i + n - 1)
            
            # The contribution of a_i to c_k:
            # For k in [k_lo, k_hi]:
            #   if k == 2*i: contribution = v^2
            #   else: contribution = 2*v*a[k-i]
            
            # Base c (with a_i = 0):
            # c_base[k] = c[k] - contribution for k in [k_lo, k_hi]
            
            # Trial values
            candidates = [0.0, v, v*0.5, v*1.5, v*0.8, v*1.2]
            # Add a "balanced" candidate: value that would make c[2i] equal to current max
            if 2*i < conv_len:
                c_2i_base = c[2*i] - v*v
                # c_2i = c_2i_base + t^2, want c_2i = max_c
                # t = sqrt(max_c - c_2i_base) if max_c > c_2i_base
                if max_c > c_2i_base:
                    candidates.append(math.sqrt(max_c - c_2i_base))
            
            best_t = v
            best_obj = best_score
            
            # Compute the "unaffected max" (max of c outside [k_lo, k_hi])
            # This is a bit expensive to compute each time; let's approximate
            # Actually, for efficiency, let's just compute the full max each time
            # since n=4498 and we have vectorized numpy
            
            for t in candidates:
                if t < -1e-12:
                    continue
                t = max(0.0, t)
                
                # Update c in-place (affected part only)
                # Remove old contribution
                # For k in [k_lo, k_hi]:
                #   if k == 2*i: c[k] -= v*v
                #   else: c[k] -= 2*v*a[k-i]
                
                # This is O(n) per candidate. Let's vectorize.
                idx = np.arange(k_lo, k_hi + 1)
                j_idx = idx - i  # = a[j_idx]

                old contrib = 2 * v * a[j_idx]
                # But c[2*i] gets v^2 instead of 2*v*a[i]
                # 2*v*a[i] vs v^2: they're equal (2*v*a[i] = 2*v*v = 2v^2 ≠ v^2)
                # Wait: 2*v*a[k-i] when k=2*i gives 2*v*a[i] = 2*v*v = 2*v^2
                # But the actual contribution at k=2*i is a_i^2 = v^2
                # So the formula "2*v*a[k-i]" overcounts by v^2 at k=2*i.
                # We need: contribution[k] = 2*v*a[k-i] for k≠2i, and v^2 for k=2i.
                # equivalently: 2*v*a[k-i] - v^2*[k=2i]
                
                contrib = 2.0 * v * a[j_idx]
                if 2*i >= k_lo and 2*i <= k_hi:
                    pos_2i = 2*i - k_lo
                    contrib[pos_2i] = v * v  # override

                c[idx] -= contrib

                new_contrib = 2.0 * t * a[j_idx]
                if 2*i >= k_lo and 2*i <= k_hi:
                    pos_2i = 2*i - k_lo
                    new_contrib[pos_2i] = t * t
                
                c[idx] += new_contrib

                if new_S < 1e-12:
                    new_obj = float('inf')
                else:
                    new_max_c = c.max()
                    new_obj = 2.0 * n * new_max_c / (new_S * new_S)
                
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_t = t
                
                # Restore c (remove new, add back old)
                c[idx] -= new_contrib
                c[idx] += contrib
            
            # Commit the best
            if best_t != v:
                idx = np.arange(k_lo, k_hi + 1)
                j_idx = idx - i
                
                contrib_old = 2.0 * v * a[j_idx]
                if 2*i >= k_lo and 2*i <= k_hi:

                c[idx] += contrib_new
                
                a[i] = best_t
                S += (best_t - v)
                v = best_t
            
            # Update best
            if best_obj < best_score:
                best_score = best_obj
                best_a = a.copy()
    
    return best_a, best_score
