def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:n + m - 1]

def autocorr_fft(a):
    """Full autocorrelation (a * a) using FFT."""
    return fft_convolve(a, a)

def score_of(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = autocorr_fft(a)
    return 2.0 * n * c.max() / (s * s)

def topk_score(a, k=10, beta=0.1):
    """Differentiable approximation using top-k soft-max."""
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf, None
    c = autocorr_fft(a)
    # Sort and get top k
    m = min(k, len(c))

    # Actually this is just the soft-max of the top-k
    score = 2.0 * n * soft_topk / (s * s)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def fft_convolve(a, b):
    n, m = len(a), len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

import numpy as np
import time

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation with adaptive soft-max, multi-resolution
progressive refinement, and multi-start search across different lengths.
"""


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _autocorr(a):
    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    s = a.sum()
    if s <= 1e-30:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_score(a, beta):
    n = len(a)
    s = a.sum()
    c = _autocorr(a)
    cmax = c.max()
    x = (c - cmax) / beta
    e = np.exp(x)
    e_sum = e.sum()
    w = e / e_sum
    soft_max = cmax + beta * np.log(e_sum)

    w_rev = w[::-1]
    H1_full = _fft_convolve(a, w_rev)
    H1 = H1_full[n - 1:2 * n - 1][::-1]
    H2_full = _fft_convolve(w, a)
    H2 = H2_full[:n]

    grad_soft = H1 + H2
    grad = 2.0 * n * grad_soft / s - 4.0 * n * soft_max * a / (s ** 3)
    return grad, soft_max


def _softmax(z):
    z = z - z.max()
    z = np.clip(z, -50, 50)
    e = np.exp(z)
    p = e + 1e-14
    return p / p.sum()


def _normalize(a):
    a = np.clip(a, 0, 1000.0)
    s = a.sum()
    if s <= 1e-30:
        return None
    return a / s


def _interp_to_length(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = p.size
    if n0 == n_new:
        return _normalize(p)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p))


def _adam_logit_optimize(p0, deadline, rng, lr_init=0.12, beta_init=None,

def _block_optimize(n_blocks, n_total, deadline, rng, lr=0.3):
    """Optimize at block level: n_blocks blocks covering n_total positions."""
    # Parametrize block heights in logit space
    z = np.zeros(n_blocks)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n_blocks)
    v = np.zeros(n_blocks)
    t = 0
    best_score = np.inf
    best_z = z.copy()
    no_imp = 0
    beta = 1.0

    while time.monotonic() < deadline:
        t += 1
        pb = _softmax(z + 2.0)  # shift to get non-trivial variation
        # Expand to full length
        indices = np.repeat(np.arange(n_blocks), n_total // n_blocks + 1)[:n_total]
        a = pb[indices]
        a = a / a.sum()

        c = _autocorr(a)
        cmax = c.max()
        sc = 2.0 * n * cmax  # n = n_total, sum = 1

        # Gradient w.r.t. block parameters
        grad_a, _ = _grad_score(a, beta)
        # Aggregate gradient to block level
        grad_b = np.zeros(n_blocks)
        counts = np.zeros(n_blocks)
        for j in range(n_blocks):
            mask = indices == j
            grad_b[j] = grad_a[mask].sum()
            counts[j] = mask.sum()
        counts = np.maximum(counts, 1)

        # Chain rule through softmax
        gp = grad_b / counts
        gp = gp - pb * gp.mean()
        gz = pb * gp

        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * gz * gz
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -10, 10)

            best_score = sc
            best_z = z.copy()
            no_imp = 0
        else:
            no_imp += 1

            m[:] = 0
            v[:] = 0
            t = 0
            no_imp = 0

    # Return the best block-level sequence expanded to full length
    pb = _softmax(best_z + 2.0)
    indices = np.repeat(np.arange(n_blocks), n_total // n_blocks + 1)[:n_total]
    a = pb[indices]
    return _normalize(a)


def _make_linear(n, eta):
    t = np.linspace(0, 1, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cubic(n, a1, a3):
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u ** 3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_basis(n, n_basis):
    t = np.linspace(0, 1, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m = 1
    while len(cols) < n_basis:
        cols.append(np.cos(2 * np.pi * m * t))
        cols.append(np.sin(2 * np.pi * m * t))
        m += 1
    return np.column_stack(cols[:n_basis])

    z = basis @ theta
    return _softmax(z)

            p, sc = _adam_logit_optimize(p_init, per, rng, lr_init=0.04,
                                          beta_init=0.05 * _autocorr(p_init).max(),
                                          patience=2000)
            sc = _score(p)
            _update(p, sc)
            print(f"  Phase4 n={tn}: score={sc:.8f}, best={best_score:.8f}")

    # Phase 5: Final fine polish on the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = best_seq.size
        per = deadline - 30
        p, sc = _adam_logit_optimize(best_seq, per, rng, lr_init=0.02,
                                      beta_init=0.03 * _autocorr(best_seq).max(),
                                      patience=3000)
        sc = _score(p)
        _update(p, sc)
        print(f"  Phase5 final polish: score={sc:.8f}")

    # Final verification
    if best_seq is not None:
        final = best_seq.copy()
        final = np.clip(final, 0, 1000)
        s = final.sum()
        if s > 0:
            final = final / s
        final_score = _score(final)
        print(f"Final: score={final_score:.10f}, n={final.size}")
        return [float(x) for x in final]

    # Fallback
    fb = _make_linear(256, 1.5)
    return [float(x) for x in fb]
