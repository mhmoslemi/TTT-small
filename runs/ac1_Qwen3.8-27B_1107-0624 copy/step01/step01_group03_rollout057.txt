        A[j:j + n, j] = s

        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

    n = len(s)
    A = build_conv_matrix(s)  # shape (2n-1, n)
    
    # Variables: b[0..n-1], M (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

import numpy as np
from scipy.optimize import linprog
import time


"""
LP-based search to minimize 2n*max(conv(s,s))/(sum(s))^2 for non-negative sequences.
Uses direct LP minimization of the cross-correlation peak, combined with mirror descent
polishing and coordinate descent. Tries multiple sequence lengths with adaptive time allocation.
"""


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2 efficiently."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where (s * b)[k] = A[k,:] @ b."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_lp_min_peak(s, time_limit=10.0):
    """
    Solve LP: minimize M subject to (s*b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt) or (None, None) on failure.
    """
    n = len(s)
    A = build_conv_matrix(s)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b_opt = np.maximum(res.x[:n], 0.0)
            M_opt = max(res.x[n], 0.0)
            return b_opt, M_opt
    except Exception:
        pass
    return None, None


def solve_lp_restricted(s, top_k=20, time_limit=5.0):
    """
    Solve restricted LP: minimize M subject to (s*b)[k] <= M for top peak positions k,
    sum(b)=1, b>=0. Faster than full LP.
    """
    n = len(s)
    conv = np.convolve(s, s)
    # Get top peak positions
    peak_idx = np.argsort(conv)[-top_k:][::-1]
    # Also include position 0 and n-1 for safety
    peak_idx = list(set(list(peak_idx) + [0, n - 1]))
    peak_idx = sorted(peak_idx)
    K = len(peak_idx)

    A = build_conv_matrix(s)
    A_sub = A[peak_idx, :]  # shape (K, n)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = A_sub
    A_ub[:, n] = -1.0
    b_ub = np.zeros(K)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            b_opt = np.maximum(res.x[:n], 0.0)
            return b_opt
    except Exception:
        pass
    return None


def lp_search(s, rng, max_iters=200, deadline=None, use_full_lp=True, verbose=True):
    """
    Iteratively solve LP to minimize the objective.
    Uses full LP or restricted LP depending on time budget.
    """
    n = len(s)
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_s = s.copy()
    best_sc = fast_score(s)
    no_imp_count = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Choose LP variant based on n and remaining time
        if use_full_lp and n <= 2048:
            b, M_opt = solve_lp_min_peak(s, time_limit=5.0)
        else:
            b = solve_lp_restricted(s, top_k=30, time_limit=3.0)
            M_opt = None

        if b is None or np.sum(b) < 1e-14:
            break

        b = b / b.sum()
        b_sc = fast_score(b)

        # Try blending to find best step
        best_t = 0.0
        best_blend_sc = best_sc
        t_candidates = [0.0, 0.3, 0.5, 0.7, 0.8, 0.9, 1.0]
        for t in t_candidates:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = fast_score(cand)
            if sc < best_blend_sc - 1e-13:
                best_blend_sc = sc
                best_t = t

        if best_t == 0.0:
            no_imp_count += 1
            if no_imp_count > 5:
                break
            # Perturbation to escape
            s = s + 0.02 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue

        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        cur_sc = fast_score(s)
        no_imp_count = 0

        if cur_sc < best_sc - 1e-13:
            best_sc = cur_sc
            best_s = s.copy()
            if verbose and (it % 20 == 0 or cur_sc < 1.52):
                print(f"    LP iter {it}: score={cur_sc:.8f} (t={best_t:.1f}, M_opt={M_opt:.6f})")

    return best_s, best_sc


def mirror_descent_polish(s, rng, max_iters=500, deadline=None, verbose=False):
    """
    Mirror descent with entropy regularizer on the simplex.
    Uses subgradient of max(conv(s,s)) to guide updates.
    """
    n = len(s)
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_s = s.copy()
    best_sc = fast_score(s)
    no_imp = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        max_val = np.max(conv)
        # Get argmax positions (top few for stability)
        threshold = max_val - max_val * 0.001
        K = np.where(conv >= threshold)[0].tolist()
        if not K:
            K = [int(np.argmax(conv))]

        # Compute subgradient
        g = np.zeros(n)
        for k in K:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                g[i] += 2.0 * s[k - i]
        g /= len(K)

        # Backtracking line search over eta
        best_eta = 0.0
        best_md_sc = best_sc
        for eta in [0.1, 0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
            log_s = np.log(s + 1e-30) - eta * g
            log_s -= np.max(log_s)  # numerical stability
            s_new = np.exp(log_s)
            s_new = s_new / s_new.sum()
            sc = fast_score(s_new)
            if sc < best_md_sc - 1e-13:
                best_md_sc = sc
                best_eta = eta

        if best_eta == 0.0:
            no_imp += 1
            if no_imp > 30:
                break
            # Small random perturbation
            s = s + 0.005 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s = s / s.sum()
            continue

        log_s = np.log(s + 1e-30) - best_eta * g
        log_s -= np.max(log_s)
        s = np.exp(log_s)
        s = s / s.sum()
        cur_sc = fast_score(s)
        no_imp = 0

        if cur_sc < best_sc - 1e-13:
            best_sc = cur_sc
            best_s = s.copy()
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: score={cur_sc:.8f}")

    return best_s, best_sc


def coordinate_polish(s, max_passes=30, deadline=None):
    """
    Coordinate descent polish. For each coordinate, try several multipliers
    and keep the best.
    """
    n = len(s)
    s = np.maximum(s, 0.0)
    s = s / s.sum()
    best_sc = fast_score(s)

    factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1]

    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = s[i]
            best_local = best_sc
            best_local_val = old_val
            for f in factors:
                new_val = old_val * f if old_val > 1e-10 else 1.0 / n
                s[i] = new_val
                total = s.sum()
                if total < 1e-15:
                    s[i] = old_val
                    continue
                conv = np.convolve(s, s)
                sc = 2.0 * n * np.max(conv) / (total * total)
                if sc < best_local - 1e-13:
                    best_local = sc
                    best_local_val = new_val
            s[i] = best_local_val
            if best_local < best_sc:
                best_sc = best_local
                improved = True
        if not improved:
            break

    s = s / s.sum()
    return s, best_sc


def make_starts(n, rng, hseq=None):
    """Generate diverse starting points of length n."""
    candidates = []

    # From height_sequence_1 if available
    if hseq is not None:
        x_old = np.linspace(0, 1, hseq.size)
        x_new = np.linspace(0, 1, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            candidates.append(h_r)

    # Random
    for _ in range(4):
        r = rng.random(n) + 0.01
        r = r / r.sum()
        candidates.append(r)

    # Bimodal (concentrated at two ends)
    k = max(1, n // 4)
    s_bi = np.zeros(n)
    s_bi[:k] = 1.0
    s_bi[-k:] = 1.0
    s_bi = s_bi / s_bi.sum()
    candidates.append(s_bi)

    # Triangular
    x = np.arange(n) / (n - 1)
    s_tri = 1.0 - np.abs(2.0 * x - 1.0)
    s_tri = np.maximum(s_tri, 0.01)
    s_tri = s_tri / s_tri.sum()
    candidates.append(s_tri)

    # Bimodal smooth (two Gaussians)
    x = np.arange(n) / (n - 1)
    s_2g = np.exp(-((x - 0.25) ** 2) / 0.02) + np.exp(-((x - 0.75) ** 2) / 0.02)
    s_2g = np.maximum(s_2g, 0.001)
    s_2g = s_2g / s_2g.sum()
    candidates.append(s_2g)

    # Ramp
    s_ramp = np.arange(1, n + 1, dtype=float)
    s_ramp = s_ramp / s_ramp.sum()
    candidates.append(s_ramp)

    # "U-shaped" (high at edges, low in middle)
    x = np.arange(n) / (n - 1)
    s_u = (2.0 * x - 1.0) ** 2 + 0.01
    s_u = s_u / s_u.sum()
    candidates.append(s_u)

    # Cosine
    s_cos = np.cos(np.pi * x) ** 2 + 0.01
    s_cos = s_cos / s_cos.sum()
    candidates.append(s_cos)

    return candidates

                                        verbose=True)

            # Phase 2: Mirror descent polish
            if time.monotonic() < deadline - 30 and best_sc < global_best_score + 0.01:
                md_deadline = min(cand_deadline, time.monotonic() + time_per_cand * 0.4)
                best_s, md_sc = mirror_descent_polish(best_s, rng, max_iters=300,
                                                      deadline=md_deadline, verbose=False)
                if md_sc < best_sc:
                    best_sc = md_sc

            # Phase 3: Coordinate polish
            if time.monotonic() < deadline - 30:
                best_s, coord_sc = coordinate_polish(best_s, max_passes=20,
                                                     deadline=deadline - 30)
                if coord_sc < best_sc:
                    best_sc = coord_sc

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** NEW GLOBAL BEST: {global_best_score:.8f} (n={n}, cand={ci})")

        # Random restart
        if time.monotonic() < deadline - 30:
            r = rng.random(n) + 0.01
            r = r / r.sum()
            rest_deadline = min(deadline - 30, time.monotonic() + time_per_cand)
            best_s, best_sc = lp_search(r, rng, max_iters=200,
                                        deadline=rest_deadline,
                                        use_full_lp=(n <= 2048), verbose=False)
            if time.monotonic() < deadline - 30:
                best_s, best_sc = coordinate_polish(best_s, max_passes=10,
                                                     deadline=deadline - 30)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart best: {global_best_score:.8f} (n={n})")

    # Final polish on the global best
    if global_best_seq is not None and time.monotonic() < deadline:
        print(f"\nFinal polishing...")
        global_best_seq, final_sc = mirror_descent_polish(
            global_best_seq, rng, max_iters=500, deadline=deadline, verbose=False)
        if final_sc < global_best_score:
            global_best_score = final_sc
        global_best_seq, final_sc = coordinate_polish(
            global_best_seq, max_passes=50, deadline=deadline)
        if final_sc < global_best_score:
            global_best_score = final_sc

    # Verify with actual evaluator
    if global_best_seq is not None:
        final_score = float(evaluate_sequence(global_best_seq.tolist()))
        print(f"\nFinal verified score: {final_score:.8f}, length={len(global_best_seq)}")
    else:
        print("\nWARNING: No valid sequence found!")
        global_best_seq = np.ones(256) / 256.0
        final_score = float(evaluate_sequence(global_best_seq.tolist()))
        print(f"Fallback score: {final_score:.8f}")

    # Ensure all values are valid for the evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
