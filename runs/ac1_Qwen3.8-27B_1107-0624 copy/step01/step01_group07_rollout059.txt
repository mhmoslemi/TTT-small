```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth subgradient with adaptive step size, LP polishing,
multiple diverse initializations (especially asymmetric ones),
and spectral initializations for good starting points.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def get_peaks(c, frac=0.95):
    """Get positions in c that are close to the maximum."""
    M = c.max()
    threshold = frac * M
    return np.where(c >= threshold)[0]


def compute_subgradient(a, c, peak_frac=0.95):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    peaks = get_peaks(c, peak_frac)
    g = np.zeros(n)
    for k in peaks:
        # g[j] += a[k-j] for j = 0..min(n-1, k)
        end = min(n, k + 1)
        g[:end] += a[k - np.arange(end)]
    norm = np.max(g)
    if norm > 1e-15:
        g /= norm
    return g


def compute_smooth_gradient(a, c, tau=None):
    """Compute smooth gradient using softmax weighting."""
    n = len(a)
    if tau is None:
        tau = 0.05 * c.max()
    # Softmax weights
    c_shifted = (c - c.max()) / tau
    weights = np.exp(c_shifted)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for k in range(len(c)):
        if weights[k] < 1e-10:
            continue
        end = min(n, k + 1)
        g[:end] += weights[k] * a[k - np.arange(end)]
    norm = np.max(g)
    if norm > 1e-15:
        g /= norm
    return g


def optimize_smooth(a, deadline, rng, verbose=False):
    """Optimize using smooth gradient descent with momentum."""
    n = len(a)

def init_spectral(n, rng):
    """Generate a signal with approximately flat spectrum."""
    N = 2 * n  # Use a larger FFT size for better frequency resolution
    # Random phases with conjugate symmetry
    phases = rng.uniform(0, 2 * np.pi, N)
    # Ensure conjugate symmetry: phases[k] = -phases[N-k]
    for k in range(1, N // 2):
        phases[N - k] = -phases[k]
    spec = np.exp(1j * phases)
    spec[0] = 1.0  # DC component is positive
    if N % 2 == 0:
        spec[N // 2] = 1.0
    
    b = np.real(np.fft.ifft(spec)) * N
    # Make non-negative
    a = np.abs(b[:n])
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def init_spectral_squared(n, rng):
    """Generate signal from squared magnitude of random spectrum."""
    N = 2 * n
    magnitudes = rng.uniform(0.5, 1.5, N)
    # Conjugate symmetry for real output
    for k in range(1, N // 2):
        magnitudes[N - k] = magnitudes[k]
    phases = rng.uniform(0, 2 * np.pi, N)
    for k in range(1, N // 2):
        phases[N - k] = -phases[k]
    
    spec = magnitudes * np.exp(1j * phases)
    b = np.real(np.fft.ifft(spec)) * N
    a = b ** 2  # Square to ensure positivity
    a = np.clip(a[:n], 1e-10, None)
    return a / a.sum()
```

```python
def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)

    """Compute smooth gradient using softmax weighting of convolution values."""
    n = len(a)
    M = c.max()

    weights = np.exp(c_shifted)
    weights /= weights.sum()

        end = min(n, k + 1)
        g[:end] += weights[k] * a[k - np.arange(end)]
    norm = np.max(g)
    if norm > 1e-15:

    return g

            step = max(0.003, step * 0.78)
            
            if stall % 12 == 0 and stall > 12:
                scale = min(0.15, 0.04 + 0.005 * stall)
                noise = 1.0 + scale * rng.standard_normal(n)
                a_new = best_a * noise
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 1e-10:
                    a_new /= a_new.sum()
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new.copy()
                        best_score = sc
                    a = a_new
                    stall = 0
                    step = 0.05
            
            if stall > 80:
                break
        
        if verbose and it % 50 == 0:
            print(f"          LP iter {it}, score={best_score:.6f}, step={step:.4f}")
    
    return best_a, best_score


def init_cosine(n):
    """Raised cosine window (symmetric)."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def init_gaussian(n, center_frac=0.5, sigma_frac=0.25):
    """Gaussian window."""
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def init_ramp(n, base=0.5, slope=1.0):
    """Asymmetric ramp."""
    return base + slope * np.arange(n) / (n - 1)


def init_exponential(n, rate=1.5):
    """Exponential decay."""
    return np.exp(-rate * np.arange(n) / n)


def init_spectral(n, rng):
    """Signal with approximately flat spectrum (random phase)."""
    N = 2 * n
    phases = rng.uniform(0, 2 * np.pi, N)
    for k in range(1, N // 2):
        phases[N - k] = -phases[k]
    spec = np.exp(1j * phases)
    spec[0] = 1.0
    if N % 2 == 0:
        spec[N // 2] = 1.0
    b = np.real(np.fft.ifft(spec)) * N
    a = np.abs(b[:n])
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def init_spectral_tapered(n, rng):
    """Spectral init with tapering to reduce edge effects."""
    N = 4 * n
    phases = rng.uniform(0, 2 * np.pi, N)
    for k in range(1, N // 2):
        phases[N - k] = -phases[k]
    spec = np.exp(1j * phases)
    spec[0] = 1.0
    if N % 2 == 0:
        spec[N // 2] = 1.0
    b = np.real(np.fft.ifft(spec)) * N
    # Take a windowed portion
    center = N // 4
    half = n // 2
    a = np.abs(b[center - half:center + half])
    # Apply gentle taper
    taper = np.ones(n)
    edge = n // 10
    taper[:edge] = np.linspace(0.3, 1.0, edge)
    taper[-edge:] = np.linspace(1.0, 0.3, edge)
    a = a * taper
    a = np.clip(a, 1e-10, None)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization:
    smooth subgradient search with diverse initializations, followed by
    LP polishing and refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 10.0
    
    best_seq = None
    best_score = np.inf
    
    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass
    
    print(f"Starting search. Budget={budget_s}s, parent: {parent is not None}")
    if parent is not None:
        try:
            ps = compute_objective(parent)
            print(f"  Parent: n={parent.size}, score={ps:.6f}")
        except:
            pass
    
    # Phase 1: Diverse exploration with smooth method
    # Define configs: (name, init_array, time_alloc)
    configs = []
    
    if parent is not None:
        configs.append(('parent_lp', parent.copy(), 'lp', 180))
        # Truncations
        for n_tr in [384, 512, 768, 1024]:
            if parent.size >= n_tr:
                configs.append((f'parent_trunc_{n_tr}', parent[:n_tr].copy(), 'smooth', 60))
            elif parent.size > 32:
                new_x = np.linspace(0, 1, n_tr)
                old_x = np.linspace(0, 1, parent.size)
                from scipy.interpolate import interp1d
                func = interp1d(old_x, parent, kind='cubic', fill_value='extrapolate')
                a_interp = np.clip(func(new_x), 0, None)
                if a_interp.sum() > 1e-10:
                    configs.append((f'parent_interp_{n_tr}', a_interp, 'smooth', 60))
    
    # Standard initializations
    for n in [384, 512, 768, 1024]:
        configs.append((f'cos_{n}', init_cosine(n), 'smooth', 50))
        configs.append((f'ramp_{n}', init_ramp(n, 0.3, 1.2), 'smooth', 50))
        configs.append((f'exp_{n}', init_exponential(n, 2.0), 'smooth', 50))
        configs.append((f'spec_{n}', init_spectral(n, rng), 'smooth', 50))
        configs.append((f'spec_t_{n}', init_spectral_tapered(n, rng), 'smooth', 50))
    
    # Off-center Gaussians
    for n in [512, 1024]:
        for cf in [0.35, 0.65]:
            configs.append((f'gauss_{cf}_{n}', init_gaussian(n, cf, 0.3), 'smooth', 40))
    
    # Random
    for n in [512, 1024]:
        a_rand = np.abs(rng.normal(1, 0.5, n))
        configs.append((f'rand_{n}', a_rand, 'smooth', 40))
    
    print(f"Phase 1: {len(configs)} configs")
    
    total_alloc = sum(c[3] for c in configs)
    if total_alloc > (deadline - start_time - 300):
        # Scale down allocations
        scale = (deadline - start_time - 300) / total_alloc
        configs = [(name, arr, method, max(15, int(alloc * scale))) for name, arr, method, alloc in configs]
        print(f"  Scaled allocations by {scale:.2f}")
    
    for idx, (name, a_init, method, alloc) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 100:
            print(f"  Stopping Phase 1 ({remaining:.0f}s remaining)")
            break
        
        a = np.clip(a_init, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(len(a))
        a = a / a.sum()
        
        n = len(a)
        config_deadline = time.monotonic() + min(alloc, remaining - 100)
        
        print(f"  [{idx+1}/{len(configs)}] {name} (n={n}, {method}, {alloc}s)", end=" ")
        
        if method == 'lp' and n <= 1024:
            a_opt, score = optimize_lp(a, config_deadline, rng)
        else:
            a_opt, score = optimize_smooth(a, config_deadline, rng)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"=> NEW BEST {best_score:.6f}")
        else:
            print(f"=> {score:.6f} (best: {best_score:.6f})")
    
    print(f"\nPhase 1 done. Best: {best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")
    
    # Phase 2: LP polish on best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 2: LP polish")
        polish_deadline = deadline
        # Determine if LP is feasible for this size
        n_best = len(best_seq)
        if n_best <= 2048:
            best_seq, best_score = optimize_lp(best_seq, polish_deadline, rng, verbose=True)
            print(f"  After LP polish: {best_score:.6f}")
        else:
            best_seq, best_score = optimize_smooth(best_seq, polish_deadline, rng)
            print(f"  After smooth polish: {best_score:.6f}")
    
    # Phase 3: Extra refinement attempts
    if best_seq is not None and time.monotonic() < deadline - 15:
        print("Phase 3: Refinement attempts")
        n_best = len(best_seq)
        
        for attempt in range(8):
            if time.monotonic() > deadline - 10:
                break
            
            # Different perturbation strategies
            if attempt % 3 == 0:
                # Small uniform noise
                a_try = best_seq * (1 + 0.03 * rng.standard_normal(n_best))
            elif attempt % 3 == 1:
                # Structural perturbation: swap halves
                a_try = best_seq.copy()
                split = n_best // 2
                a_try[:split] = best_seq[split:split + split]
                if n_best % 2 == 0:
                    a_try[split:] = best_seq[:split]
            else:
                # Blend with a new random direction
                a_rand = np.abs(rng.normal(1, 0.5, n_best))
                a_rand /= a_rand.sum()
                a_try = 0.8 * best_seq + 0.2 * a_rand
            
            a_try = np.clip(a_try, 0.0, None)
            if a_try.sum() < 1e-10:
                continue
            a_try /= a_try.sum()
            
            short_deadline = time.monotonic() + min(60, deadline - time.monotonic() - 10)
            
            if n_best <= 1024 and attempt < 4:
                a_opt, score = optimize_lp(a_try, short_deadline, rng)
            else:
                a_opt, score = optimize_smooth(a_try, short_deadline, rng)
            
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Attempt {attempt}: NEW BEST {best_score:.6f}")
            else:
                print(f"  Attempt {attempt}: {score:.6f}")
    
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0
    
    # Ensure validity for evaluator
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())
    
    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final: score={best_score:.6f}, n={len(best_seq)}")
    
    return [float(x) for x in best_seq]
```
