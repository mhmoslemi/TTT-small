Algorithm:
1. Compute c = a * a.
2. Find k* = argmax(c).
3. Optimize c[k*] directly (it's a quadratic function of a, so we can compute its gradient exactly).
4. When the max position changes (another k becomes the new max), switch to optimizing the new peak.
5. Use a "safety margin" to ensure we're reducing the current max while not increasing others too much.

This is like a **pursuit method** or **best coordinate descent** on the peaks.

Let me develop this:

At any point, let k* be the current argmax of c. The gradient of c[k*] w.r.t. a[i] is:
- a[k*-i] if 0 <= k*-i < n
- 0 otherwise

So the gradient vector is just a shifted version of a itself!

grad_c[i] = a[k*-i] (if valid) else 0.

To minimize c[k*] while keeping S = sum(a) constant, we want to:
- Decrease a[i] where grad_c[i] is large (i.e., where a[k*-i] is large).
- Increase a[j] where grad_c[j] is small.

The optimal direction (projected onto the hyperplane sum = 0) is:
d[i] = grad_c[i] - mean(grad_c)

Then update: a <- a - lr * d (projected to a >= 0, sum(a) = S).

But we also need to consider the other peaks. If reducing c[k*] causes another c[k'] to increase, we might make things worse.

**Hybrid approach: Track top-K peaks simultaneously.**

Use a weighted sum of the top K peaks as the objective:
F(a) = sum_{k in topK} w_k * c[k]

where w_k are learned weights (e.g., based on how close c[k] is to the max).

The gradient of F is:
dF/da[i] = sum_{k in topK} w_k * a[k-i]

This is a weighted cross-correlation of a with the weights.

import numpy as np

def convolve_fft(a):
    """Fast self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n * max(convolve(a,a)) / sum(a)^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_obj_grad(a, tau, top_boost=1.0):
    """
    Compute smoothed objective and gradient using log-sum-exp with optional
    top-peak boosting.
    
    Returns (J_smoothed, gradient)
    """
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    if top_boost > 1.5:
        k = int(top_boost)
        if k < len(c):
            idx = np.argpartition(c, -k)[-k:]
            p[idx] *= 1.5
            p /= np.sum(p)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(fa * np.conj(fp), size)[:n]

    J = 2.0 * n * Tmax / (S * S)

def peak_adaptive_tau(a, base_tau, floor=0.02):
    """
    Adapt tau based on the peakiness of the convolution.
    If the top peak dominates, use smaller tau (sharper).
    If peaks are close, use larger tau (smoother).
    """
    c = convolve_fft(a)
    c_sorted = np.sort(c)[::-1]
    if c_sorted[0] < 1e-15:
        return base_tau
    ratio = c_sorted[1] / c_sorted[0] if len(c_sorted) > 1 else 1.0
    # If second peak is close to first (ratio > 0.9), use larger tau
    # If first peak is isolated (ratio < 0.7), use smaller tau
    adjustment = 1.0 + (0.9 - ratio) * 2.0
    adjustment = np.clip(adjustment, 0.5, 2.0)
    return max(floor, base_tau * adjustment)

            # a[k_star - i] for i in range where 0 <= k_star - i < n
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            for i in range(lo, hi + 1):
                grad_peak[i] += a[k_star - i]
                if k_star - i != i:
                    grad_peak[i] += a[i]
                # Actually this double-counts. Let me just use the FFT approach.
        
        # Use FFT cross-correlation for efficiency
        p = np.zeros(2 * n - 1)
        p[k_star] = 1.0
        size = 1
        while size < 3 * n - 2:
            size <<= 1
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p, size)
        grad_peak = np.fft.irfft(fa * np.conj(fp), size)[:n]
        # Double it because c[k] = sum a[i]*a[k-i] and both factors depend on a
        grad_peak *= 2.0

        # Project gradient to maintain sum constraint
        grad_peak -= np.mean(grad_peak)

        # Momentum update
        momentum = 0.85 * momentum + 0.15 * grad_peak
        a = a - lr * momentum
        a = np.maximum(a, 0.0)

        # Adaptive lr
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        lr = 0.005 * (1.0 + 2.0 * (1.0 - frac))

        if t % 2000 == 0 if False else False:
            pass

        # Check periodically
        if int(t) % 2000 == 0 or True:  # every iteration is fine since it's fast
            pass

        t_global = int((time.monotonic() - start_time) * 1000)
        if t_global % 500 < 4:  # roughly every 500ms
            lj = compute_J(a)
            if lj < best_J:
                best_J = lj
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 10:
                    a = best_a.copy()
                    momentum[:] = 0.0
                    # Small perturbation
                    w = rng.integers(n // 30, n // 10)
                    s = rng.integers(0, max(1, n - w))
                    a[s:s + w] *= np.exp(rng.normal(0, 0.2, w))
                    a = np.maximum(a, 0.0)
                    stall = 0

        now = time.monotonic()
        if now - last_verify > 5.0:
            last_verify = now
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()

    return best_a, best_J


def run_multiscale(a_init, budget_s, seed):
    """
    Multi-scale optimization: first optimize a downsampled version,
    then upsample and refine at full resolution.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s

    # Scale 1: downsample by factor 4
    factor = 4
    n_small = n // factor
    a_small = np.zeros(n_small)
    for i in range(n_small):
        a_small[i] = np.mean(a_init[i * factor:(i + 1) * factor])
    a_small = normalize_to_mean_one(a_small)

    # Optimize at small scale (use 30% of budget)
    small_budget = budget_s * 0.3
    a_small_opt, _ = run_gradient_descent(
        a_small, small_budget, seed=seed,
        tau_start=3.0, tau_end=0.05, lr_start=0.01
    )

    # Upsample
    a_up = np.zeros(n)
    for i in range(n_small):
        a_up[i * factor:(i + 1) * factor] = a_small_opt[i]
    # Blend with original in the "gaps"
    a_full = a_init.copy()
    for i in range(n_small):
        a_full[i * factor:(i + 1) * factor] = a_small_opt[i]
    a_full = np.maximum(a_full, 0.0)

    # Refine at full scale (use remaining 70%)
    full_budget = budget_s * 0.7 - 5.0
    a_opt, J = run_gradient_descent(
        a_full, full_budget, seed=seed + 1,
        tau_start=2.0, tau_end=0.02, lr_start=0.006
    )
    return a_opt, J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function: multi-strategy optimization combining gradient descent,
    peak-directed methods, multi-scale refinement, and structured perturbations.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"Starting optimization with n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Generate diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("chirp", make_chirp_modulated(n, parent, 0.1)))
    inits.append(("chirp_neg", make_chirp_modulated(n, parent, -0.1)))
    inits.append(("edge", make_edge_boosted(n, parent, 0.1)))
    inits.append(("center", make_center_boosted(n, parent, 0.1)))
    inits.append(("harmonic1", make_harmonic(n, parent, 1.0, 0.05)))
    inits.append(("harmonic2", make_harmonic(n, parent, 2.0, 0.03)))

    # Also try random initialization
    rand_init = np.exp(rng.normal(0, 0.3, n))
    inits.append(("random", rand_init))

    # Evaluate all initializations
    scored = []
    for name, arr in inits:
        arr = np.maximum(arr, 0.0)
        J = compute_J(arr)
        scored.append((J, name, arr))
        print(f"  init '{name}': J_local = {J:.6f}")

    scored.sort(key=lambda x: x[0])

    # Time allocation
    # Top 2 inits get 35% each, 3rd gets 20%, remaining 10% for final refinement
    # Plus 5% for multi-scale
    remaining = total_deadline - time.monotonic()
    reserve_final = max(30.0, remaining * 0.08)
    alloc = remaining - reserve_final

    # Strategy: run top 3 with gradient descent, 1 with peak-directed, 1 with multiscale
    budgets = [alloc * 0.35, alloc * 0.30, alloc * 0.20, alloc * 0.10, alloc * 0.05]

    strategies = []
    for i in range(min(3, len(scored))):
        strategies.append(("gd", scored[i], budgets[i] if i < len(budgets) else alloc * 0.05))
    if len(scored) > 3:
        strategies.append(("peak", scored[3], budgets[3] if len(budgets) > 3 else alloc * 0.05))
    if len(scored) > 4:
        strategies.append(("multiscale", scored[4], budgets[4] if len(budgets) > 4 else alloc * 0.03))

    # Execute strategies
    for method, (init_J, name, init_arr), budget in strategies:
        if time.monotonic() >= total_deadline - reserve_final:
            break
        budget = min(budget, total_deadline - time.monotonic() - 10.0)
        if budget < 10.0:
            break

        print(f"\nRunning '{method}' on init '{name}' (budget={budget:.1f}s, init_J={init_J:.4f})")
        t0 = time.monotonic()

        if method == "gd":
            res_a, res_J = run_gradient_descent(
                init_arr, budget, seed=seed + 1000 * (hash(name) % 100),
                tau_start=4.0, tau_end=0.03, lr_start=0.008, use_adaptive_tau=True
            )
        elif method == "peak":
            res_a, res_J = run_peak_directed(init_arr, budget, seed=seed + 2000)
        else:  # multiscale
            res_a, res_J = run_multiscale(init_arr, budget, seed=seed + 3000)

        elapsed = time.monotonic() - t0
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST: {val:.6f} (after {elapsed:.1f}s) ***")
        else:
            verified = val if np.isfinite(val) else res_J
            print(f"  '{name}/{method}': local={res_J:.6f}, verified={verified:.6f}, best={best_global_score:.6f}")

    # Final refinement on global best
    remaining = total_deadline - time.monotonic() - 5.0
    if remaining > 20.0:
        print(f"\nFinal refinement ({remaining:.1f}s)")
        # Run with tighter parameters
        res_a, res_J = run_gradient_descent(
            best_global, remaining * 0.6, seed=seed + 99999,
            tau_start=1.5, tau_end=0.02, lr_start=0.004, use_adaptive_tau=True
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Refined: {val:.6f}")

        # Peak-directed fine-tuning
        remaining2 = total_deadline - time.monotonic() - 3.0
        if remaining2 > 15.0:
            res_a2, res_J2 = run_peak_directed(best_global, remaining2, seed=seed + 88888)
            val2 = float(evaluate_sequence(res_a2.tolist()))
            if np.isfinite(val2) and val2 < best_global_score:
                best_global = res_a2.copy()
                best_global_score = val2
                print(f"  Peak-directed refined: {val2:.6f}")

    # Final verification
    best_global = np.clip(best_global, 0.0, 1000.0)
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)
    else:
        # Fallback: return parent
        best_global = parent.copy()
        best_global_score = float(evaluate_sequence(best_global.tolist()))

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in best_global]
