"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP warm-start, soft-max gradient descent, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M, method='highs'):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, max_iters=100):
    """LP-based optimization. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.02)
        else:
            stall += 1
            step = max(0.005, step * 0.7)
            if stall > 30:
                break

    return best_a, best_score


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_objective_and_grad(a, t):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Soft-max
    c_max = c.max()
    exp_c = np.exp(t * (c - c_max))
    softmax_w = exp_c / exp_c.sum()
    sm_val = c_max + np.log(exp_c.sum()) / t
    
    # Gradient: d(sm)/da[i] = sum_k softmax_w[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] if 0 <= k-i < n
    grad = np.zeros(n)
    for i in range(n):
        # k ranges over 0..2n-2, k-i must be in 0..n-1
        # So k ranges over i..i+n-1
        k_min = i
        k_max = min(i + n - 1, 2 * n - 2)
        if k_min <= k_max:
            grad[i] = np.dot(softmax_w[k_min:k_max+1], a[k_min-i:k_max+1-i])
    
    return sm_val, grad


def softmax_grad_descent(a, deadline, rng, t_start=50.0, t_end=500.0, 
                         lr_start=0.1, lr_end=0.01, max_iters=50000):
    """Soft-max gradient descent on the simplex. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    it = 0
    total_iters = 0

    while time.monotonic() < deadline and total_iters < max_iters:
        total_iters += 1
        # Anneal t and lr
        progress = min(1.0, total_iters / max_iters)
        t = t_start + (t_end - t_start) * progress
        lr = lr_start + (lr_end - lr_start) * progress
        
        # Compute gradient
        sm_val, grad = softmax_objective_and_grad(a, t)
        
        # Gradient step
        a_new = a - lr * grad
        # Project onto simplex
        a_new = simplex_project(a_new, 1.0)
        
        if a_new.sum() < 1e-10:
            break
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
        else:
            a = a_new  # still move (to escape flat regions)
            stall += 1
            if stall > 100:
                # Perturbation
                noise = np.abs(rng.normal(0, 0.03, n))
                a_pert = best_a * (1.0 + noise)
                a_pert = simplex_project(a_pert, 1.0)
                sc_pert = compute_objective(a_pert)
                if sc_pert < best_score:
                    best_a = a_pert
                    best_score = sc_pert
                    a = a_pert
                    stall = 0
                else:
                    a = a_pert  # accept perturbation to explore
                    if stall > 300:
                        break
            if stall > 500:
                break

        if total_iters % 5000 == 0:
            print(f"        sm-grad iter {total_iters}, score={best_score:.8f}, t={t:.0f}, lr={lr:.4f}")

    return best_a, best_score


def coord_descent(a, deadline, rng, max_passes=200):
    """Coordinate descent polish. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    passes = 0

    while time.monotonic() < deadline and passes < max_passes:
        passes += 1
        improved_this_pass = False
        
        # Random order for diversity
        if passes % 5 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            if time.monotonic() > deadline:
                return best_a, best_score
            
            # Try a few values for a[i]
            # Current value, 0, and some perturbations
            current = a[i]
            candidates = [0.0]
            if current > 0:
                candidates.extend([current * 0.5, current * 1.5, current * 0.1])
            
            best_val = current
            best_sc = best_score  # we'll recompute
            
            for val in candidates:
                a[i] = val
                # Renormalize
                s = a.sum()
                if s < 1e-10:
                    a[i] = current

def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_top(n, taper=0.1):

        taper_vals = 0.5 * (1 + np.cos(np.linspace(0, np.pi, m)))
        a[:m] = taper_vals
        a[-m:] = taper_vals[::-1]
    return a

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP warm-start, soft-max gradient descent with annealing,
and coordinate descent polish with golden-section 1D subproblem solving.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def optimize_lp(a, deadline, rng, max_iters=50):
    """LP-based optimization. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:

            a_new = (1.0 - s) * best_a + s * g

                continue
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-14:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.35, step * 1.02)
        else:
            stall += 1
            step = max(0.003, step * 0.7)
            if stall > 25:
                break

    return best_a, best_score


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_objective_and_grad(a, t):
    """Compute soft-max of conv(a,a) and its gradient w.r.t. a (a summing to 1)."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    c_max = c.max()
    exp_c = np.exp(t * (c - c_max))
    exp_sum = exp_c.sum()
    softmax_w = exp_c / exp_sum
    sm_val = c_max + np.log(exp_sum) / t

    # Gradient: grad[i] = sum_{j: 0<=j<n, 0<=i+j<2n-1} softmax_w[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        j_max = min(n, 2 * n - 1 - i)
        if j_max <= 0:
            continue
        grad[i] = np.dot(softmax_w[i:i + j_max], a[:j_max])

    return sm_val, grad


def softmax_grad_descent(a, deadline, rng, t_start=30.0, t_end=800.0,
                         lr_start=0.05, lr_end=0.005, max_iters=100000):
    """Soft-max gradient descent on the probability simplex with annealing."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    total_iters = 0

    while time.monotonic() < deadline and total_iters < max_iters:
        total_iters += 1
        progress = min(1.0, total_iters / max_iters)
        # Use a plateau-then-spike schedule for t
        if progress < 0.5:
            t = t_start + (t_end * 0.3 - t_start) * (progress / 0.5)
        elif progress < 0.8:
            t = t_end * 0.3 + (t_end * 0.7 - t_end * 0.3) * ((progress - 0.5) / 0.3)
        else:
            t = t_end
        lr = lr_start * (lr_end / lr_start) ** progress

        sm_val, grad = softmax_objective_and_grad(a, t)

        # Multiple step sizes: try the main step and a smaller one
        improved = False
        for lr_try in [lr, lr * 0.5, lr * 0.25]:
            a_new = a - lr_try * grad
            a_new = simplex_project(a_new, 1.0)
            if a_new.sum() < 1e-10:
                continue
            sc = compute_objective(a_new)
            if sc < best_score - 1e-14:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                improved = True
                break
            else:
                a = a_new  # accept anyway to explore
                if not improved:
                    a = a_new
                    break

        if not improved:
            stall += 1
            if stall % 50 == 0 and stall > 20:
                # Perturbation
                strength = min(0.15, 0.02 + 0.01 * (stall // 100))
                noise = rng.normal(0, strength, n)
                a_pert = best_a + noise
                a_pert = simplex_project(a_pert, 1.0)
                sc_pert = compute_objective(a_pert)
                if sc_pert < best_score:
                    best_a = a_pert.copy()
                    best_score = sc_pert
                    a = a_pert
                    stall = 0
                else:
                    # Annealing acceptance
                    if rng.random() < 0.3:
                        a = a_pert
                    else:
                        a = best_a.copy()
            if stall > 400:
                break

        if total_iters % 10000 == 0:
            print(f"        sm-grad iter {total_iters}, score={best_score:.8f}, t={t:.0f}, lr={lr:.5f}")

def coord_descent_golden(a, deadline, rng, max_passes=500):
    """Coordinate descent with golden-section 1D subproblem solving."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    passes = 0
    golden_ratio = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() < deadline and passes < max_passes:
        passes += 1
        improved_this_pass = False

        if passes % 3 == 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            if time.monotonic() > deadline:
                return best_a, best_score

            current = a[i]
            S_rest = 1.0 - current  # since sum = 1

            # Precompute rest of convolution (excluding a[i]'s contribution)
            # c[k] = a[i]*a[k-i] + rest[k]
            # rest[k] = sum_{j != i, j+k-j valid} a[j]*a[k-j]
            # Efficient: c_full = conv(a, a), then subtract a[i]'s contribution
            c_full = np.convolve(a, a, mode='full')
            # a[i] contributes to c_full at positions i..i+n-1
            contrib = np.zeros(2 * n - 1)
            j_max = min(n, 2 * n - 1 - i)
            contrib[i:i + j_max] = current * a[:j_max]
            rest = c_full - contrib
            rest = np.maximum(rest, 0.0)  # numerical safety

            # Objective as function of x = a[i] (with S_rest fixed, sum = x + S_rest = 1)
            # f(x) = 2n * max_k(x * a[k-i] + rest[k]) / 1^2  (since sum=1)
            # = 2n * max_k(x * a[k-i] + rest[k])

            # Find optimal x by golden section on [0, upper_bound]
            upper_bound = max(current * 3.0, 2.0 / n)
            if upper_bound > 1.0:
                upper_bound = 1.0

            def obj_1d(x):
                """Objective value when a[i] = x (rest unchanged, sum constraint met by scaling others)."""
                # If we set a[i] = x, others scale by S_rest/(S_rest) = 1 (already sum to S_rest)
                # Actually others sum to S_rest, so total sum = x + S_rest
                # We need to renormalize: new_a[i] = x/(x+S_rest), new_a[j] = a[j]/(x+S_rest)
                # But for the convolution, this is equivalent to scaling
                # Let's compute directly
                c_new = rest + x * np.zeros(2 * n - 1)
                j_max2 = min(n, 2 * n - 1 - i)
                c_new[i:i + j_max2] += x * a[:j_max2]
                total_sum = x + S_rest
                if total_sum < 1e-15:
                    return np.inf
                return 2.0 * n * c_new.max() / (total_sum * total_sum)

            # Golden section search on [0, upper_bound]
            if current < 1e-10:
                x_lo, x_hi = 0.0, upper_bound
            else:
                x_lo = max(0.0, current * 0.3)
                x_hi = upper_bound

            # Initial bracket check: make sure current is in bracket
            if x_lo > current or x_hi < current:
                x_lo = max(0.0, min(current, x_hi) * 0.5)
                x_hi = max(current, x_lo * 2.0, upper_bound)
                x_hi = min(x_hi, 1.0)

            if x_hi <= x_lo:
                x_lo = 0.0
                x_hi = upper_bound

            # Golden section
            c1 = x_hi - golden_ratio * (x_hi - x_lo)
            c2 = x_lo + golden_ratio * (x_hi - x_lo)
            f1 = obj_1d(c1)
            f2 = obj_1d(c2)
            f0 = obj_1d(current)

            for _ in range(40):
                if time.monotonic() > deadline:
                    return best_a, best_score
                if f1 < f2:
                    x_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = x_hi - golden_ratio * (x_hi - x_lo)
                    f1 = obj_1d(c1)
                else:
                    x_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = x_lo + golden_ratio * (x_hi - x_lo)
                    f2 = obj_1d(c2)

            x_opt = (x_lo + x_hi) / 2.0
            f_opt = obj_1d(x_opt)

            # Compare with current best
            if f_opt < best_score - 1e-12:
                a[i] = x_opt
                # Renormalize
                a = a / a.sum()
                sc = compute_objective(a)
                if sc < best_score - 1e-14:
                    best_a = a.copy()
                    best_score = sc
                    improved_this_pass = True

        if improved_this_pass:
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_flat_top(n, taper=0.15):
    """Flat top with cosined tapered edges."""
    a = np.ones(n)
    m = max(1, int(taper * n))
    if m < n // 2:
        taper_vals = 0.5 * (1 + np.cos(np.linspace(0, np.pi, m)))
        a[:m] = taper_vals
        a[-m:] = taper_vals[::-1]
    return a


def make_sinc_like(n, width=5):
    """Sinc-like window with bandwidth control."""
    center = (n - 1) / 2.0
    x = (np.arange(n) - center) / width
    # Use a generalized sinc: (sin(x)/x)^2 type
    mask = np.abs(x) > 1e-10
    result = np.ones(n)
    result[mask] = np.sin(x[mask]) / x[mask]
    result = np.abs(result)
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization:
    LP warm-start + soft-max gradient descent with t-annealing +
    coordinate descent with golden-section 1D subproblem solving.
    Multiple initializations and lengths explored with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0  # safety buffer

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
    except Exception:
        pass

    print(f"Starting hybrid search. Budget: {budget_s}s")
    if parent is not None:
        print(f"Parent sequence: n={parent.size}, score={compute_objective(parent):.8f}")
        if compute_objective(parent) < best_score:
            best_score = compute_objective(parent)
            best_seq = parent.copy()

    # Time allocation
    total_time = deadline - start_time
    t_phase1 = 0.25 * total_time  # LP warm-start
    t_phase2 = 0.50 * total_time  # Soft-max gradient
    t_phase3 = 0.20 * total_time  # Coordinate descent
    # 5% buffer

    phase1_deadline = start_time + t_phase1
    phase2_deadline = start_time + t_phase1 + t_phase2
    phase3_deadline = start_time + t_phase1 + t_phase2 + t_phase3

    # Define initializations to try
    init_list = [

        (768, make_gaussian),
        (1024, make_cosine),
        (1024, make_gaussian),

        (1024, make_flat_top),
    ]

    # Phase 1: LP warm-start on diverse initializations
    print(f"\n{'='*60}")
    print(f"Phase 1: LP warm-start ({t_phase1:.0f}s)")
    print(f"{'='*60}")

    # Prepare configs (include parent if available)
    lp_configs = []
    if parent is not None:
        for n_p in [min(parent.size, 512), min(parent.size, 1024)]:
            lp_configs.append((n_p, parent[:n_p].copy()))

    # Add a subset of standard inits for phase 1 (focus on promising sizes)
    phase1_inits = [
        (256, make_cosine),

    for n, init_f in phase1_inits:

            a = np.clip(a, 1e-15, None)
            a = a / a.sum()
            lp_configs.append((n, a))
        except Exception:
            pass

    # Run LP on each with time-limited budget
    phase1_per_config = t_phase1 / max(1, len(lp_configs))
    lp_results = []  # (score, a) pairs

    for idx, (n, a) in enumerate(lp_configs):
        if time.monotonic() > phase1_deadline:

        if config_deadline <= time.monotonic():

        # Quick initial evaluation

        else:

    # Phase 2: Soft-max gradient descent

    print(f"Phase 3: Coordinate descent polish ({t_phase3:.0f}s)")

    if best_seq is not None:

    if best_seq is None:

    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / max(best_seq.sum(), 1e-30))

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*60}")
    print(f"DONE in {elapsed:.1f}s")
    print(f"Final score: {best_score:.8f}, n={len(best_seq)}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
