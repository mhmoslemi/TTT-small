def score_normalized(f):
    """Score for normalized (sum=1) non-negative sequence."""
    n = len(f)
    c = np.convolve(f, f)
    M = c.max()
    return 2.0 * n * M

def normalize(f):
    """Normalize to sum 1, clip to non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-300:
        return np.ones_like(f) / len(f)
    return f / s

def optimize_smooth(f, time_limit, seed=0):
    """Optimize using smooth-max gradient with adaptive step size."""
    rng = np.random.default_rng(seed)
    f = normalize(f.copy())
    n = len(f)
    deadline = time.monotonic() + time_limit
    
    best_score = score_normalized(f)
    best = f.copy()
    
    tau = 50.0  # temperature (1/temperature in the exponent)
    alpha = 0.05  # base step size
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    
    no_improve = 0
    patience = 50
    
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive tau: increase when making progress, decrease when stuck
        if no_improve < 5:
            tau = min(tau * 1.02, 200.0)
        elif no_improve > patience:
            tau = max(tau * 0.8, 5.0)
            # Random restart in the neighborhood
            f = normalize(f + rng.normal(0, 0.02, n))
            no_improve = 0
            continue
        
        # Compute gradient
        p = smooth_max_gradient(c, tau)
        grad = compute_gradient(f, p)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-300:
            break
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        step = -alpha * momentum / grad_norm  # normalized step direction
        
        # Try a few step sizes (backtracking)
        improved = False
        for trial_alpha in [alpha, alpha * 0.5, alpha * 0.25, alpha * 2.0]:
            f_new = f + trial_alpha * (-momentum / max(grad_norm, 1e-300))
            f_new = normalize(f_new)
            new_score = score_normalized(f_new)
            if new_score < cur_score:
                f = f_new
                improved = True
                break
        
        if not improved:
            # Small random perturbation
            f = normalize(f + rng.normal(0, 0.005, n))
    
    return best, best_score

def make_gaussian_init(n, center, width, rng):
    """Create a Gaussian bump."""
    x = np.arange(n, dtype=np.float64)
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    return normalize(f)

def make_two_bump_init(n, rng):
    """Two asymmetric bumps."""
    c1 = rng.uniform(0.2, 0.5) * n
    c2 = rng.uniform(0.55, 0.8) * n
    w1 = rng.uniform(0.08, 0.25) * n
    w2 = rng.uniform(0.08, 0.25) * n
    a1 = rng.uniform(0.3, 1.0)
    a2 = rng.uniform(0.3, 1.0)
    x = np.arange(n, dtype=np.float64)
    f = a1 * np.exp(-0.5 * ((x - c1) / w1) ** 2) + a2 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    return normalize(f)

def make_fourier_init(n, rng):
    """Fourier-based initialization."""
    K = rng.integers(3, 10)
    coeffs = [1.0] + list(rng.normal(0, 0.5, 2*K))
    x = np.arange(n, dtype=np.float64) / n
    f = np.ones(n)
    for k in range(1, K+1):
        f += coeffs[2*(k-1)] * np.cos(2*np.pi*k*x)
        f += coeffs[2*(k-1)+1] * np.sin(2*np.pi*k*x)
    return normalize(f)
