LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Generate diverse initial profiles (linear, cubic, power-law, sigmoid, etc.)
2. Apply LP-based annealing: solve LP to find best direction under current peak constraint,
   then line-search for optimal blend parameter.
3. Polish with logit-space Adam optimization using soft-max smoothing for the peak.
4. Try multiple sequence lengths and return the best candidate.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def next_pow2(n):
    """Smallest power of two >= n, at least 2."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    """Full linear convolution of a real 1D array with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0 and mn > -1e-9:
        c = c - mn
    elif mn < 0:
        c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    """Fast score using FFT convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    """Exact score using numpy direct convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    """Clip to nonnegative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def build_conv_mat(f):
    """Build (2n-1) x n matrix A where A[k,l] = f[k-l] if 0 <= k-l < n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))

        end = min(n, 2 * n - 1 - l)
        A[l:l + end, l] = f[:end]
    return A


def lp_step(f, M):

    n = len(f)
    if n <= 0 or M <= 0:

    A = build_conv_mat(f)

            if g.sum() > 1e-15:

        pass
    return None


def soft_mx(c, beta):
    """Smooth max of c with width beta, returning value and interpolating weights."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    """Compute h[i] = sum_j w[i+j] p[j], for i=0..len(p)-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def logit_anneal(p0, deadline, rng, lr=0.06, max_iter=5000):
    """
    Full logit-space Adam polish of a nonnegative simplex sequence.
    Returns (best_p, best_score).
    """
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-12)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break

        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        dot = float(np.dot(p, gp))
        gz = p * (gp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 800 == 0:
                    lr = max(0.015, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 150 == 0:
            beta = max(1e-14, beta * 0.7)

        if no_imp > max(1500, 8 * n):
            z = 0.8 * best_z + 0.25 * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.3 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def lp_anneal(f0, deadline, max_iter=150, verbose=True, seed=0):
    """
    LP-based annealing: iteratively solve LP for best direction, line search for blend.
    Returns (best_f, best_score).
    """
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0.0)
    s = f.sum()
    f = f / s

    best_f = f.copy()
    best_score = internal_score(f)
    rng = np.random.default_rng(seed)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = full_convolve_fft(f)
        M = float(c.max())

        g = lp_step(f, M)
        if g is None or g.sum() < 1e-12:
            if verbose:
                print(f"    LP step failed at iter {it}")
            break

        g = g / g.sum()

        # Line search: try multiple t values
        best_t = 0.01
        best_sc = best_score
        t_candidates = [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in t_candidates:
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            sc = internal_score(f_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        sc = internal_score(f)

        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f.copy()

        if verbose and it % 15 == 0:
            print(f"    LP iter {it}: score={best_score:.8f}, t={best_t:.3f}, M={M:.6f}")

        if best_score < 1.49:
            break

        # Check for stagnation
        if it > 20 and best_score > best_f_score_check(f, it, best_score, lp_anneal._hist if hasattr(lp_anneal, '_hist') else None):
            break

    return best_f, best_score


def best_f_score_check(f, it, best_score, hist):
    """Helper for stagnation detection (placeholder)."""
    return best_score - 1e-10


def make_profile_linear(n, eta):
    """Linear profile: 1 + eta*(t - 0.5), normalized."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_cubic(n, a1, a3):
    """Cubic profile: 1 + a1*u + a3*u^3."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a3 * u**3
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_power(n, p):
    """Power-law profile: (j+1)^p normalized."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_sigmoid(n, eta, mid=0.5):
    """Sigmoid profile."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 / (1.0 + np.exp(-eta * (t - mid)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profile_exp(n, eta):
    """Exponential profile: exp(eta * t)."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(eta * t)
    return a / a.sum()


def make_profile_gaussian(n, sigma, mu=0.5):
    """Gaussian profile centered at mu with width sigma."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - mu) / sigma) ** 2)
    return a / a.sum()


def make_diverse_profiles(n, count=30, seed=42):
    """Generate a diverse set of initial profiles."""
    rng = np.random.default_rng(seed)
    profs = []

    # Linear profiles
    for eta in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        profs.append(make_profile_linear(n, eta))

    # Cubic profiles
    for a1, a3 in [(1.0, -0.5), (1.5, -1.0), (2.0, -1.5), (2.0, -2.0),
                   (1.5, -1.5), (2.5, -2.0), (1.0, 0.5), (1.5, 1.0)]:
        profs.append(make_profile_cubic(n, a1, a3))

    # Power-law profiles
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profs.append(make_profile_power(n, p))

    # Sigmoid profiles
    for eta in [3.0, 5.0, 8.0, 12.0]:
        for mid in [0.3, 0.5, 0.7]:
            profs.append(make_profile_sigmoid(n, eta, mid))

    # Exponential profiles
    for eta in [1.0, 2.0, 3.0, 4.0]:
        profs.append(make_profile_exp(n, eta))

    # Gaussian profiles
    for sigma in [0.15, 0.2, 0.25, 0.3]:
        for mu in [0.2, 0.35, 0.5, 0.65, 0.8]:
            profs.append(make_profile_gaussian(n, sigma, mu))

    # Random smooth profiles via random Fourier features
    for _ in range(max(5, count // 3)):
        z = np.zeros(n)
        for m in range(1, 8):
            z += rng.normal(0, 1) * np.cos(2 * np.pi * m * np.linspace(0, 1, n) + rng.normal(0, 1))
        a = np.exp(2.0 * z - 2.0 * np.mean(z))
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    # Mix of existing profiles
    for _ in range(max(3, count // 4)):
        i1 = rng.integers(0, len(profs))
        i2 = rng.integers(0, len(profs))
        w = rng.uniform(0.2, 0.8)
        a = w * profs[i1] + (1 - w) * profs[i2]
        a = np.maximum(a, 1e-12)
        profs.append(a / a.sum())

    return profs


def interpolate_profile(p0, n_new):
    """Linear interpolation of a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search for nonnegative sequences minimizing
    2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
      1. Generate diverse initial profiles at multiple lengths.
      2. Apply LP-based annealing to find good local optima.
      3. Polish best candidates with logit-space Adam.
      4. Try extension to longer sequences.
      5. Return the best sequence found.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    candidates = []  # list of (score, sequence)

    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is None or sc is None or not np.isfinite(sc):
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        if sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: score={best_score:.10f}, n={p.size}")
        # Add to candidate list (keep top 20)
        if len(candidates) < 20:
            candidates.append((float(sc), p.copy()))
            candidates.sort(key=lambda x: x[0])
        else:
            worst_idx = max(range(len(candidates)), key=lambda i: candidates[i][0])
            if float(sc) < candidates[worst_idx][0]:
                candidates[worst_idx] = (float(sc), p.copy())
                candidates.sort(key=lambda x: x[0])

    # Check for parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = normalize_nonneg(parent)
            if pn is not None:
                sc = internal_score(pn)
                update_best(pn, sc)
                print(f"Parent sequence: n={pn.size}, score={sc:.10f}")
    except Exception:
        pass

    print(f"Initial best: {best_score:.10f}")

    # === Phase 1: LP-based optimization at n=512 ===
    n_main = 512
    phase1_end = min(deadline, t_start + 600.0)

    print(f"\nPhase 1: LP annealing at n={n_main}")
    profiles = make_diverse_profiles(n_main, count=40, seed=int(seed))

    # Score all profiles and sort
    scored_profiles = []
    for p in profiles:
        sc = internal_score(p)
        scored_profiles.append((sc, p))
    scored_profiles.sort(key=lambda x: x[0])

    print(f"  Generated {len(profiles)} profiles, best initial score={scored_profiles[0][0]:.6f}")

    # Run LP anneal on top profiles
    n_lp_tries = 8
    for i in range(min(n_lp_tries, len(scored_profiles))):
        if time.monotonic() > phase1_end:
            break
        sc_init, p_init = scored_profiles[i]
        per = min(phase1_end, time.monotonic() + 45.0)
        p_best, sc_best = lp_anneal(p_init, per, max_iter=80, verbose=(i < 3), seed=int(seed) + i)
        update_best(p_best, sc_best)

    print(f"After Phase 1: best={best_score:.10f}")

    # === Phase 2: LP-based optimization at n=256 (easier problem, find good shapes) ===
    n_short = 256
    phase2_end = min(deadline, t_start + 850.0)

    if best_score > 1.505 and time.monotonic() < phase2_end:
        print(f"\nPhase 2: LP annealing at n={n_short}")
        profiles_short = make_diverse_profiles(n_short, count=30, seed=int(seed) + 100)
        scored_short = []
        for p in profiles_short:
            sc = internal_score(p)
            scored_short.append((sc, p))
        scored_short.sort(key=lambda x: x[0])

        for i in range(min(6, len(scored_short))):
            if time.monotonic() > phase2_end:
                break
            sc_init, p_init = scored_short[i]
            per = min(phase2_end, time.monotonic() + 30.0)
            p_best, sc_best = lp_anneal(p_init, per, max_iter=60, verbose=False, seed=int(seed) + 100 + i)
            update_best(p_best, sc_best)

        # Also try extending best n=256 results to n=512
        for i in range(min(4, len(candidates))):
            if time.monotonic() > phase2_end:
                break
            sc_c, p_c = candidates[i]
            if p_c.size == n_short:
                p_ext = interpolate_profile(p_c, n_main)
                sc_ext = internal_score(p_ext)
                per = min(phase2_end, time.monotonic() + 30.0)
                p_best, sc_best = lp_anneal(p_ext, per, max_iter=50, verbose=False, seed=int(seed) + 200 + i)
                update_best(p_best, sc_best)

        print(f"After Phase 2: best={best_score:.10f}")

    # === Phase 3: Logit Adam polish on best candidates ===
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 3: Logit Adam polish")

    # Polish top candidates at their native length
    polished = set()
    for i in range(min(12, len(candidates))):
        if time.monotonic() > phase3_end:
            break
        sc_c, p_c = candidates[i]
        key = (p_c.size, i)
        if key in polished:
            continue
        polished.add(key)

        n_p = p_c.size
        if n_p > 1024:
            p_work = interpolate_profile(p_c, 512)
        else:
            p_work = p_c.copy()

        per = min(phase3_end, time.monotonic() + (40.0 if n_p <= 512 else 60.0))
        p_best, sc_best = logit_anneal(p_work, per, rng, lr=0.05)
        sc_final = internal_score(p_best)
        update_best(p_best, sc_final)

        # Also try a second polish with different lr
        if time.monotonic() < phase3_end:
            per2 = min(phase3_end, time.monotonic() + 30.0)
            p_best2, sc_best2 = logit_anneal(p_best, per2, rng, lr=0.03)
            update_best(p_best2, internal_score(p_best2))

    print(f"After Phase 3: best={best_score:.10f}")

    # === Phase 4: Try n=1024 if time allows ===
    if time.monotonic() < deadline - 200.0 and best_score > 1.50:
        print(f"\nPhase 4: Try n=1024")
        n_long = 1024
        phase4_end = min(deadline, t_start + 1650.0)

        # Interpolate best profile to n=1024
        if best_seq is not None:
            p_long = interpolate_profile(best_seq, n_long)
            sc_long = internal_score(p_long)
            print(f"  Interpolated to n=1024: score={sc_long:.8f}")

            # Short LP anneal at n=1024
            per = min(phase4_end, time.monotonic() + 40.0)
            p_best, sc_best = lp_anneal(p_long, per, max_iter=30, verbose=False, seed=int(seed) + 300)
            update_best(p_best, sc_best)

            # Adam polish
            if time.monotonic() < phase4_end:
                per2 = min(phase4_end, time.monotonic() + 80.0)
                p_best2, sc_best2 = logit_anneal(p_best, per2, rng, lr=0.04)
                update_best(p_best2, internal_score(p_best2))

        # Also try fresh profiles at n=1024
        if time.monotonic() < phase4_end - 30.0:
            for eta in [1.5, 2.0, 2.5]:
                if time.monotonic() > phase4_end - 10.0:
                    break
                p_l = make_profile_linear(n_long, eta)
                per = min(phase4_end - 10.0, time.monotonic() + 30.0)
                p_best, sc_best = lp_anneal(p_l, per, max_iter=20, verbose=False, seed=int(seed) + 400 + int(eta * 10))
                update_best(p_best, sc_best)

        print(f"After Phase 4: best={best_score:.10f}")

    # === Phase 5: Final exact scoring and validation ===
    print(f"\nPhase 5: Final scoring")
    final_seq = best_seq
    final_score = best_score

    if final_seq is not None:
        # Re-score with exact convolution for sequences up to length 2048
        if final_seq.size <= 2048:
            sc_exact = true_score(final_seq)
            if np.isfinite(sc_exact) and sc_exact < final_score:
                final_score = sc_exact
                print(f"  Exact score improved: {final_score:.10f}")

        # Check all candidates exactly
        for sc_c, p_c in candidates[:10]:
            if p_c.size <= 2048:
                sc_e = true_score(p_c)
                if np.isfinite(sc_e) and sc_e < final_score:
                    final_score = sc_e
                    final_seq = p_c
                    print(f"  Candidate exact score: {final_score:.10f}, n={p_c.size}")

        # Call the authoritative evaluator
        if final_seq is not None and final_seq.size <= 5000:
            try:
                sc_eval = float(evaluate_sequence(final_seq.tolist()))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
                    print(f"  Evaluator score: {final_score:.10f}")
            except Exception as e:
                print(f"  Evaluator error: {e}")

        # Ensure valid output
        final_seq = normalize_nonneg(final_seq)
        if final_seq is None:
            final_seq = make_profile_linear(512, 1.5)
            final_score = internal_score(final_seq)

        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0:
            final_seq = final_seq / s

        print(f"\nFinal: score={final_score:.10f}, n={final_seq.size}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = make_profile_linear(512, 1.5)
    print(f"Fallback: n=512, score={internal_score(fallback):.10f}")
    return [float(x) for x in fallback]
