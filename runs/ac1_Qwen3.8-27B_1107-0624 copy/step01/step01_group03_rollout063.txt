def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2"""
    n = len(s)

    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def subgradient_step(s, lr):
    """One subgradient step to reduce max(conv(s,s))."""
    n = len(s)
    conv = np.convolve(s, s)
    M = np.max(conv)
    # Find all lags where conv achieves max (within tolerance)
    active = np.where(np.abs(conv - M) < 1e-10 * M)[0]
    
    # Compute subgradient: sum over active lags of grad(conv[k])
    # grad(conv[k])[i] = 2*s[k-i] if 0 <= k-i < n
    g = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] += 2 * s[j]
    
    # Project gradient onto sum=0 hyperplane (to maintain sum(s)=1)
    g = g - np.mean(g)
    
    # Update
    s_new = s - lr * g
    s_new = np.maximum(s_new, 0)
    s_new = s_new / s_new.sum()
    return s_new, M

    return 2.0 * n * np.max(conv) / (total * total)

def compute_subgradient(s, conv):
    """Compute subgradient of max(conv) with respect to s."""
    n = len(s)
    M = np.max(conv)
    # Active set: lags where conv is close to max
    tol = 1e-8 * M
    active = np.where(conv >= M - tol)[0]

        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        indices = np.arange(i_start, i_end + 1)
        g[indices] += 2.0 * s[k - indices]
    
    return g, M, len(active)

def subgradient_descent(s, max_iters=50000, deadline=None, verbose=False):
    """Run subgradient descent to minimize max(conv(s,s))."""

            no_imp = 0

Optimization of non-negative sequences to minimize 2n*max(conv(s,s))/(sum(s))^2.
Uses subgradient descent followed by coordinate descent with convex 1D minimization.

import numpy as np

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def make_binomial_seq(n):
    """Create sequence proportional to binom(2k,k)/4^k."""
    k = np.arange(n)
    # binom(2k,k) / 4^k can be computed iteratively to avoid overflow
    s = np.zeros(n)
    s[0] = 1.0
    for i in range(1, n):
        s[i] = s[i-1] * (2*i) * (2*i-1) / (4*i*i)
    s = s / s.sum()
    return s

    s = s / s.sum()
    return s
