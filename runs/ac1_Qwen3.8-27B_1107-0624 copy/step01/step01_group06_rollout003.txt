"""
Optimized step function coefficient search using gradient descent with
adaptive log-sum-exp smoothing, peak-directed local search, and
population-based exploration to minimize J(a) = 2n*max(a*a)/(sum(a))^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_max_and_grad(a, tau):
    """Compute smoothed max of convolution and its gradient w.r.t. a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    # Gradient computation via cross-correlation
    fp = np.fft.rfft(p, size)
    # cross[i] = sum_k p[k] * a[k-i] (circular, but we handle bounds)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    dTmax_da = cross  # dTmax/da[j] = sum_k p[k] * a[k-j] (approximately)
    grad = (2.0 * n / (S * S)) * dTmax_da - (4.0 * n * Tmax / (S * S * S))

    J = 2.0 * n * Tmax / (S * S)
    return J, grad

    Tau is held constant for a fixed number of iterations, then dropped.

    rng = np.random.default_rng(seed)

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    best_exact_J = float(evaluate_sequence(a.tolist()))

    # Staircase tau levels
    tau_levels = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
    iters_per_level = 500  # iterations at each tau level

    level = 0
    iter_in_level = 0
    stall_count = 0
    stall_threshold = best_J

    while time.monotonic() < deadline:
        if level >= len(tau_levels):
            # Stay at the last level
            tau = tau_levels[-1]
        else:
            tau = tau_levels[level]
            if iter_in_level >= iters_per_level:
                level += 1
                iter_in_level = 0
                stall_count = 0
                stall_threshold = best_J

            t = 0

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            if np.isfinite(val) and val < best_exact_J:

                best_a = a.copy()
                best_J = val

        peak_val = c[peak_idx]

        # c[peak_idx] = sum_j a[j] * a[peak_idx - j] (for valid j)

        for j in range(n):

        # Top fraction to reduce, bottom fraction to increase

        for j in reduce_indices:
            cand[j] = max(0, cand[j] - reduce_amount * influence[j] / max(influence.max(), 1e-10))
        for j in increase_indices:
            cand[j] += increase_amount / max(1.0, len(increase_indices)) * n_increase

        cand = np.maximum(cand, 0)
        cS = np.sum(cand)
        if cS < 0.01:
            continue

        cJ = compute_J(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_size = max(0.001, step_size * 0.98)
