import time

import numpy as np
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution a*a via FFT."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]

def compute_grad(a, c, tau):
    """Compute gradient of LSE(c, tau) / tau wrt a. (i.e., gradient of the smoothed max."""
    n = a.size
    # softmax weights
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    s = w.sum()
    w = w / s
    # gradient: g[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    # Use cross-correlation via FFT
    m = 1
    while m < 3 * n:  # need m >= 3n-2 for cross-corr of len(2n-1) and len(n)
        m <<= 1
    fw = np.fft.rfft(w, m)
    fa = np.fft.rfft(a, m)
    # cross-corr: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # = ifft(conj(FFT(w)) * FFT(a))[i] ... 
    # Wait, let me double-check.
    # (x ⋆ y)[n] = sum_k x[k+n] y[k]
    # FFT{x ⋆ y} = conj(FFT{x}) * FFT{y}
    # So x ⋆ y = ifft(conj(FFT{x}) * FFT{y})
    # Here x=w, y=a: (w ⋆ a)[i] = sum_k w[k+i] * a[k]
    # But I want sum_k w[k] * a[k-i] = sum_k w[k] * a[k-i]
    # Let j = k-i: = sum_j w[j+i] * a[j] = (w ⋆ a)[i]
    # Yes! So what I want is exactly (w ⋆ a)[i].
    g = np.fft.irfft(np.conj(fw) * fa, m)[:n]
    g *= 2.0
    return g

def fista_step(a, tau, lr, z, t):
    """One FISTA step."""
    c = compute_conv(a)
    g = compute_grad(a, c, tau)
    t_new = 0.5 * (1 + math.sqrt(1 + 4*t*t))
    z = a + ((t - 1) / t_new) * (a - z)
    a_new = z - lr * g
    # Project to simplex {a >= 0, sum a = 1}
    a_new = project_simplex(a_new, 1.0)
    return a_new, z, t_new

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_refine_step(a, c):
    """LP-based refinement step."""
    n = a.size
    # Variables: d_0,...,d_{n-1}, t
    # Minimize t
    # s.t. c[k] + 2*(d*a)_k <= t for all k
    #      sum(d) = 0
    #      a[i] + d[i] >= 0
    
    # Build constraint matrix
    # For each k: sum_i d[i] * 2*a[k-i] - t <= -c[k]
    # Coefficient of d[i] in row k: 2*a[k-i] if 0 <= k-i < n, else 0
    # Coefficient of t in row k: -1
    
    # For speed, build the matrix efficiently
    # A has (2n-1) + n rows and (n+1) columns
    
    # Actually for large n, this matrix is very sparse. Let me use scipy sparse.
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(i)
            vals.append(2.0 * a[j])
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    b_ineq = -c  # length 2n-1
    
    # Non-negativity: a[i] + d[i] >= 0 => -d[i] <= a[i]
    for i in range(n):
        rows.append(2*n-1 + i)
        cols.append(i)
        vals.append(-1.0)
    b_ineq = np.concatenate([b_ineq, a])  # a[i] for the -d[i] <= a[i] constraints
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1+n, n+1))
    
    # Equality: sum(d) = 0
    A_eq_row = np.zeros(n+1)
    A_eq_row[:n] = 1.0
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ineq, A_eq=[A_eq_row], b_eq=[0.0], method='highs')
    
    if res.success:
        d = res.x[:n]
        t_val = res.x[n]
        return d, t_val
    return None, None

"""
Minimize 2*n*max(conv(a,a))/(sum a)^2 using annealed top-L log-sum-exp
objective with FISTA optimization, LP-based polishing, and multi-n multi-start search.

Key ideas:
- Scale invariance allows working on the simplex (sum a = 1)
- Top-L log-sum-exp provides a smooth surrogate for max with controllable bias
- FISTA (Nesterov) gives O(1/k^2) convergence for smooth objectives
- LP successive linear programming for final polishing
- Multiple sequence lengths n to exploit the n-dependent optimization landscape
"""

import time
import math
import numpy as np


def _fft_size(n):
    """Next power of 2 >= n."""
    m = 1
    while m < n:
        m <<= 1
    return m


def compute_conv(a):
    """Linear convolution a*a via FFT. Returns array of length 2*n-1."""
    n = a.size
    m = _fft_size(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def compute_grad_topL(a, c, tau, L_top):
    """Gradient of top-L log-sum-exp of c w.r.t. a, where c = conv(a,a).
    
    Returns g where g[i] = 2 * sum_{k in topL} w_k * a[k-i] (valid idx).
    """
    n = a.size
    K = c.size  # 2n-1
    L = min(L_top, K)
    
    # Get top-L indices
    if L < K:
        top_idx = np.argpartition(-c, L - 1)[:L]
    else:
        top_idx = np.arange(K)
    
    # Softmax weights over top-L
    c_top = c[top_idx]
    c_max = c_top.max()
    w = np.exp((c_top - c_max) / tau)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(L) / L
    else:
        w = w / s
    
    # Build full-length w vector (zero outside top-L)
    w_full = np.zeros(K)
    w_full[top_idx] = w
    
    # Cross-correlation: g[i] = 2 * sum_k w_full[k] * a[k-i]
    # = 2 * (w_full ⋆ a)[i]
    # FFT: (w ⋆ a)[i] = ifft(conj(FFT(w)) * FFT(a))[i]
    m = _fft_size(3 * n)  # need >= len(w_full) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
    fw = np.fft.rfft(w_full, m)
    fa = np.fft.rfft(a, m)
    g = np.fft.irfft(np.conj(fw) * fa, m)[:n]
    g *= 2.0
    return g


def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, target / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    """Ensure a >= 0 and sum(a) = 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _make_seeds(n, rng, base_seq=None):
    """Create diverse initial seeds for optimization."""
    seeds = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    seeds.append(np.ones(n, dtype=np.float64) / n)
    
    # Sine (raised cosine)
    s = np.sin(np.pi * x) ** 2
    seeds.append(s / s.sum())
    
    # Sine to power 0.5 (wider)
    s = np.sin(np.pi * x) ** 0.5
    seeds.append(s / s.sum())
    
    # Gaussian
    g = np.exp(-8.0 * (x - 0.5) ** 2)
    seeds.append(g / g.sum())
    
    # Wide Gaussian
    g = np.exp(-2.0 * (x - 0.5) ** 2)
    seeds.append(g / g.sum())
    
    # Linear ramp
    r = np.linspace(0.01, 1.0, n)
    seeds.append(r / r.sum())
    
    # Cosine window
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    seeds.append(np.maximum(c, 1e-6) / np.maximum(c, 1e-6).sum())
    
    # Two-bump
    b1 = np.exp(-50 * (x - 0.3) ** 2)
    b2 = np.exp(-50 * (x - 0.7) ** 2)
    t = b1 + b2
    seeds.append(t / t.sum())
    
    # Random smooth (low-pass filtered random)
    r = rng.random(n)
    # Simple low-pass: moving average
    k = max(3, n // 64)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    seeds.append(r_smooth / r_smooth.sum())
    
    # Interpolated base sequence
    if base_seq is not None and base_seq.size >= 4:
        bp = np.maximum(base_seq, 0).astype(np.float64)
        if bp.sum() > 0:
            bp = bp / bp.sum()
            if bp.size == n:
                seeds.append(bp.copy())
            else:
                interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, bp.size),
                    bp
                )
                seeds.append(interp / interp.sum())
            # Reversed
            if bp.size == n:
                seeds.append(bp[::-1].copy())
    
    # All seeds normalized
    seeds = [_normalize(s) for s in seeds]
    return seeds


def _fista_optimize(a, t_end, rng):
    """Run FISTA optimization on sequence a (sum=1) until t_end.
    Returns best sequence found (sum=1)."""
    n = a.size
    a = _normalize(a)
    z = a.copy()
    t_counter = 1.0
    
    c = compute_conv(a)
    mc_start = c.max()
    lr = 1.0 / (4.0 * mc_start)  # initial learning rate
    lr = max(lr, 1e-5)
    lr = min(lr, 0.1)
    
    best_a = a.copy()
    best_mc = mc_start
    
    total_steps = 0
    tau_ratio = 0.15  # initial tau as fraction of max
    L_ratio = 1.0  # initial L as fraction of K
    
    consecutive_no_improve = 0
    
    while time.monotonic() < t_end - 2.0:
        total_steps += 1
        progress = min(1.0, total_steps / 3000.0)
        
        # Anneal: tau decreases, L decreases
        c = compute_conv(a)
        mc = c.max()
        K = c.size
        tau = max(1e-5, tau_ratio * mc * (1.0 - 0.8 * progress))
        L_top = max(1, int(L_ratio * K * (1.0 - 0.6 * progress)))
        
        g = compute_grad_topL(a, c, tau, L_top)
        
        # FISTA update
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_counter * t_counter))
        z_old = z.copy()
        z = a + ((t_counter - 1.0) / t_new) * (a - z)
        a_candidate = z - lr * g
        a_candidate = project_simplex(a_candidate, 1.0)
        
        # Evaluate candidate
        c_cand = compute_conv(a_candidate)
        mc_cand = c_cand.max()
        
        if mc_cand < best_mc:
            best_mc = mc_cand
            best_a = a_candidate.copy()
            a = a_candidate
            consecutive_no_improve = 0
            # Adjust learning rate: increase slightly if improving
            lr = min(lr * 1.01, 0.1)
        else:
            a = a_candidate
            consecutive_no_improve += 1
            if consecutive_no_improve > 100:
                lr *= 0.8
                consecutive_no_improve = 0
                # Small random perturbation to escape
                if rng.random() < 0.3:
                    pert = rng.normal(0, 0.01, n)
                    a = project_simplex(a + pert, 1.0)
        
        t_counter = t_new
        
        # Periodic learning rate decay
        if total_steps % 1000 == 0:
            lr *= 0.95
            tau_ratio *= 0.7
            L_ratio = max(0.05, L_ratio * 0.7)
        
        # Adaptive: if stuck for a long time, do a bigger restart
        if consecutive_no_improve > 500 and total_steps > 500:
            pert = rng.normal(0, 0.05, n)
            a = project_simplex(a + pert, 1.0)
            z = a.copy()
            t_counter = 1.0
            lr = 1.0 / (4.0 * mc)
            lr = max(lr, 1e-5)
            consecutive_no_improve = 0
    
    return best_a


def _lp_polish(a, t_end, max_iters=200):
    """LP-based polishing: successive linear programming steps."""
    n = a.size
    if n > 2048:
        return a  # LP too expensive for very large n
    
    a = _normalize(a)
    best_a = a.copy()
    best_mc = compute_conv(a).max()
    
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    for it in range(max_iters):
        if time.monotonic() > t_end - 3.0:
            break
        
        c = compute_conv(a)
        c_max = c.max()
        
        # Build LP: minimize t
        # s.t. c[k] + 2*(d*a)_k <= t  for k=0,...,2n-2
        #      sum(d) = 0
        #      d[i] >= -a[i]  (i.e., a[i]+d[i] >= 0)
        #      d[i] <= 2*a[i] (upper bound to prevent too-large steps)
        
        nvars = n + 1  # d_0,...,d_{n-1}, t
        
        # Objective: minimize t
        c_obj = np.zeros(nvars)
        c_obj[n] = 1.0
        
        # Build inequality constraint matrix (sparse)
        rows = []
        cols = []
        vals = []
        b_ub = np.zeros(2 * n - 1 + 2 * n)
        
        # Convolution constraints: 2*(d*a)_k - t <= -c[k]
        for k in range(2 * n - 1):
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            for i in range(i_lo, i_hi + 1):
                j = k - i
                rows.append(k)
                cols.append(i)
                vals.append(2.0 * a[j])
            rows.append(k)
            cols.append(n)
            vals.append(-1.0)
            b_ub[k] = -c[k]
        
        # Lower bound: -d[i] <= a[i]
        for i in range(n):
            rows.append(2 * n - 1 + i)
            cols.append(i)
            vals.append(-1.0)
            b_ub[2 * n - 1 + i] = a[i]
        
        # Upper bound: d[i] <= 2 * a[i]
        for i in range(n):
            rows.append(3 * n - 1 + i)
            cols.append(i)
            vals.append(1.0)
            b_ub[3 * n - 1 + i] = 2.0 * a[i]
        
        # Equality: sum(d) = 0
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        A_ub_sparse = csr_matrix((vals, (rows, cols)), shape=(len(b_ub), nvars))
        
        res = linprog(c_obj, A_ub=A_ub_sparse, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, method='highs',
                      options={'time_limit': 5.0})
        
        if not res.success:
            break
        
        d = res.x[:n]
        
        # Line search
        alpha = 1.0
        improved = False
        for _ in range(15):
            a_trial = a + alpha * d
            a_trial = _normalize(a_trial)
            mc_trial = compute_conv(a_trial).max()
            if mc_trial < c_max * (1.0 - 1e-10):
                a = a_trial
                if mc_trial < best_mc:
                    best_mc = mc_trial
                    best_a = a_trial.copy()
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            break
    
    return best_a


def _try_n(n, t_end, rng, base_seq=None):
    """Try to find the best sequence of length n within time t_end."""
    best_a = None
    best_mc = float('inf')
    
    seeds = _make_seeds(n, rng, base_seq)
    
    # Allocate time among seeds

        seed_t_end = time.monotonic() + time_per_seed

                best_a = a_polished.copy()
    
    return best_a
