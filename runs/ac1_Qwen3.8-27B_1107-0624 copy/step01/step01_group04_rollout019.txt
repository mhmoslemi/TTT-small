Key improvements over previous approach:

import time

from scipy.optimize import linprog, minimize_scalar
from scipy import sparse


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _fft_conv(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n + m - 1].real


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_conv(a)
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def _build_A_rows(f, rows):
    """Build constraint matrix rows for given row indices."""
    n = len(f)
    R = len(rows)
    if R == 0:
        return np.zeros((0, n))
    A = np.zeros((R, n))
    for ii in range(R):
        k = int(rows[ii])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def _build_A_sparse(f, rows):
    """Build sparse constraint matrix for given row indices."""
    n = len(f)
    R = len(rows)
    if R == 0:
        return sparse.csc_matrix((0, n))
    rr, cc, vv = [], [], []
    for ii in range(R):
        k = int(rows[ii])
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rr.append(k)
            cc.append(j)
            vv.append(f[k - j])
    return sparse.csc_matrix((vv, (rr, cc)), shape=(R, n))


def _line_search(f, g, M, use_fft=True):
    """Find optimal t minimizing max_k q_k(t) where q_k is the k-th conv value of ((1-t)f+tg)."""
    n = len(f)
    if use_fft:
        L = _next_pow2(3 * n - 2)
        F = np.fft.rfft(f, L)
        G = np.fft.rfft(g, L)
        # Compute c_f = f*f, c_fg = f*g, c_g = g*g
        c_f = np.fft.irfft(F * F, L)[:2 * n - 1].real
        c_fg = np.fft.irfft(F * G, L)[:2 * n - 1].real
        c_g = np.fft.irfft(G * G, L)[:2 * n - 1].real
    else:
        c_f = _fft_conv(f)
        c_fg = _fft_conv(f, g)
        c_g = _fft_conv(g)

    c_f = np.maximum(c_f, 0.0)
    c_fg = np.maximum(c_fg, 0.0)
    c_g = np.maximum(c_g, 0.0)

    def phi(t):
        t = float(t)
        val = (1 - t) ** 2 * c_f + 2 * t * (1 - t) * c_fg + t ** 2 * c_g
        return float(val.max())

    res = minimize_scalar(phi, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    t_star = float(res.x)
    return t_star, phi(t_star)


def _lp_optimize(f_init, time_budget, verbose=False):
    """
    LP-based cutting-plane optimization.
    f_init: initial sequence (will be normalized to sum=sqrt(2n))
    Returns: best sequence found and its score.
    """
    f_init = np.asarray(f_init, dtype=np.float64).ravel()
    n = len(f_init)
    if n < 2:
        return f_init, _score(f_init)

    S0 = np.sqrt(2.0 * n)
    f = _normalize(f_init) * S0
    c_f = _fft_conv(f)
    M = float(c_f.max())
    if M <= 0:
        return f, _score(f)

    best_f = f.copy()
    best_score = _score(f)

    # Start with a subset of constraints: those near the peak
    all_rows = np.arange(2 * n - 1)
    # Sort by c_f value descending, take top ones
    top_frac = min(1.0, 800.0 / (2 * n - 1))
    n_top = max(50, int(top_frac * (2 * n - 1)))
    top_idx = np.argsort(c_f)[::-1][:n_top]
    # Also add some evenly spaced ones
    spaced = np.linspace(0, 2 * n - 2, min(100, 2 * n - 1), dtype=int).astype(int)
    rows = np.unique(np.concatenate([top_idx, spaced]))
    rows = rows[:min(len(rows), max(200, min(1200, 2 * n - 1)))]
    row_set = set(int(r) for r in rows)

    max_rows = min(2 * n - 1, max(500, min(2000, 4 * n)))
    end_time = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < end_time:
        it += 1
        rem = end_time - time.monotonic()
        if rem < 3.0:
            break

        rows = list(rows)
        A = _build_A_rows(f, rows)
        b = np.full(len(rows), M)
        c_obj = -np.ones(n)

        # Time limit for LP call
        lp_time = max(2.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0.0, None),
                          method='highs', options={'time_limit': lp_time})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full constraint violations
        c_fg = _fft_conv(f, g)
        c_fg = np.maximum(c_fg, 0.0)
        tol = 1e-8 * max(M, 1.0)
        violations = c_fg - M
        max_viol = float(violations.max())

        if max_viol > tol:
            # Add violated constraints
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order:
                    if added >= min(500, max_rows - len(rows)):
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                if added == 0 and max_viol > 10 * tol:
                    break
                continue
            else:
                # At max rows, if heavily violated, stop
                if max_viol > 100 * tol:
                    break

        # Rescale g to match sum of f
        g = g * (S0 / sg)

        # Line search
        t_star, phi_val = _line_search(f, g, M)
        if t_star <= 1e-6:
            break

        f_new = (1.0 - t_star) * f + t_star * g
        f_new = np.clip(f_new, 0.0, None)
        s_new = float(f_new.sum())
        if s_new <= 1e-12:
            break
        f_new = f_new * (S0 / s_new)

        c_new = _fft_conv(f_new)
        M_new = float(c_new.max())

        if M_new >= M * (1.0 - 1e-10):
            # Try smaller step
            for t_try in [0.5 * t_star, 0.25 * t_star, 0.1 * t_star, 0.05 * t_star]:
                f_try = (1.0 - t_try) * f + t_try * g

                s_try = float(f_try.sum())
                if s_try <= 1e-12:
                    continue
                f_try = f_try * (S0 / s_try)
                c_try = _fft_conv(f_try)
                M_try = float(c_try.max())
                if M_try < M * (1.0 - 1e-10):
                    f_new = f_try
                    M_new = M_try
                    break
            else:
                break

        f = f_new
        M = M_new
        c_f = c_new
        sc = _score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()

        if verbose and it % 10 == 0:
            print("  LP iter=%d M=%.6f score=%.6f" % (it, M, sc))

        # Converged?
        if M < 1e-10:
            break

    return best_f, best_score


def _adam_optimize(p_init, time_budget, seed=0):
    """Adam-based optimization with smoothed max objective."""
    p_init = np.asarray(p_init, dtype=np.float64).ravel()
    n = len(p_init)
    p = _normalize(p_init)
    S0 = np.sqrt(2.0 * n)
    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.025
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    L = _next_pow2(3 * n - 2)

    best_p = p.copy()
    best_f = _score(p)
    start = time.monotonic()
    end = start + max(0.0, time_budget - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, time_budget))
        alpha = 8.0 + 50.0 * np.exp(-5.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -50.0, 50.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = _score(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def _greedy_init(n):
    """Central binomial coefficients (sqrt(1/(1-x)) coefficients)."""
    a = np.empty(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return _normalize(a)


def _power_init(n, s):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return _normalize(a)


def _uniform_init(n):

def _perturb_init(base, rng, sigma):

            if kind == 'greedy':
                base = _greedy_init(nl)
            elif kind == 'perturb':
                base = _perturb_init(_greedy_init(nl), rng, 0.15)
            else:
                base = _power_init(nl, 0.5)
            p_out, sc = _adam_optimize(base, min(alloc, rem - 5.0), seed=seed + nl)
            record(p_out, sc)

    # Phase 3: Hybrid - Adam to get good starting point, then LP to polish
    print("Phase 3: Hybrid Adam+LP")
    if time.monotonic() < deadline:
        hybrid_lengths = [384, 512, 256, 640]
        for nl in hybrid_lengths:
            if time.monotonic() >= deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            # Adam for shorter time
            base = _greedy_init(nl)
            adam_alloc = min(40.0, rem * 0.3)
            p_adam, sc_adam = _adam_optimize(base, adam_alloc, seed=seed + nl + 100)
            record(p_adam, sc_adam)
            # Then LP on the result
            if time.monotonic() < deadline:
                rem2 = deadline - time.monotonic()
                if rem2 > 20:
                    p_out, sc = _lp_optimize(p_adam, min(rem2 - 5.0, 120.0))
                    record(p_out, sc)

    # Phase 4: If we have time, do more LP iteration on the best found
    print("Phase 4: Final polishing")
    if best_p is not None and time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem > 30:
            p_best = best_p.copy()
            nn = len(p_best)
            # Try several perturbations of the best and LP polish
            for trial in range(3):
                if time.monotonic() >= deadline:
                    break
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                if trial == 0:
                    base = p_best.copy()
                else:
                    base = _perturb_init(p_best, rng, 0.03 * trial)
                alloc = min(rem - 5.0, 90.0)
                if alloc < 10:
                    break
                p_out, sc = _lp_optimize(base, alloc)
                record(p_out, sc)

        # Also try Adam on the best
        if time.monotonic() < deadline:
            rem = deadline - time.monotonic()
            if rem > 20:
                p_out, sc = _adam_optimize(best_p, min(rem - 5.0, 60.0), seed=seed + 9999)
                record(p_out, sc)

    # Phase 5: Try some unusual initializations
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        weird_inits = [
            (384, _power_init(384, 0.2)),
            (384, _power_init(384, 0.8)),
            (384, _power_init(384, 1.0)),
            (512, _power_init(512, 0.3)),
            (256, _power_init(256, 0.7)),
            (384, _perturb_init(_uniform_init(384), rng, 0.3)),
            (512, _perturb_init(_uniform_init(512), rng, 0.3)),
        ]
        for nl, init in weird_inits:
            if time.monotonic() >= deadline:
                break
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            p_out, sc = _lp_optimize(init, min(rem - 5.0, 40.0))
            record(p_out, sc)

    # Final validation
    if best_p is None:
        best_p = _greedy_init(384)
        best_score = _score(best_p)

    print("\nFinal best: score=%.8f n=%d time=%.1fs" % (best_score, len(best_p), time.monotonic() - t0))

    # Scale output to reasonable values
    final_seq = _scale_output(best_p, 10.0)

    # Validate with actual evaluator if time permits
    if time.monotonic() < deadline:
        gscore = _try_eval(final_seq)
        if np.isfinite(gscore):
            print("Global eval: %.8f" % gscore)
        else:
            # Try different scaling
            for ts in [1.0, 5.0, 50.0, 100.0]:
                fs = _scale_output(best_p, ts)
                gs = _try_eval(fs)
                if np.isfinite(gs):
                    print("Global eval (sum=%.1f): %.8f" % (ts, gs))
                    if gs < best_score or not np.isfinite(best_score):
                        final_seq = fs
                        best_score = gs
                    break

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: search for optimal sequence and return it."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("ERROR in main_search: %s" % e)
        import traceback
        traceback.print_exc()
        # Fallback
        p = _greedy_init(384)
        return _scale_output(p, 10.0)
