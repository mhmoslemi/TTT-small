def peak_shave(a, c, fraction=0.1, strength=0.05):
    """Directly reduce the peak by adjusting coefficients that contribute to it."""
    n = len(a)
    c_max = np.max(c)
    # Find hot indices
    threshold = c_max * (1 - fraction)
    hot_mask = c >= threshold
    
    # For each hot index k, the contribution of a[i] is a[k-i] (for valid i)
    # Compute total "pressure" on each a[i]
    pressure = np.zeros(n)
    for k in np.where(hot_mask)[0]:
        # a contributes to c[k] through pairs (i, k-i)
        # da[k]/da[i] = a[k-i] + a[i-k] (with zero outside range)
        # We add a[k-i] to pressure[i] for each hot k
        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
        partners = k - valid_i
        valid_partners = (partners >= 0) & (partners < n)
        valid_i = valid_i[valid_partners]
        if len(valid_i) > 0:
            pressure[valid_i] += a[partners[valid_partners]]
    
    # Normalize pressure
    pressure /= (np.max(pressure) + 1e-10)
    
    # Adjust: reduce where pressure is high, increase where low
    # Maintain sum
    S = np.sum(a)
    mean_pressure = np.mean(pressure)
    delta = strength * (mean_pressure - pressure)
    a_new = a * (1 + delta)
    a_new = np.maximum(a_new, 0)
    # Renormalize
    S_new = np.sum(a_new)
    if S_new > 0.01:
        a_new = a_new * (S / S_new)
    return a_new

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad

def peak_shave_step(a, c, strength=0.03, top_frac=0.1):
    """Reduce the peak by decreasing coefficients that contribute to hot convolution indices."""
    n = len(a)
    c_sorted = np.sort(c)[::-1]
    n_top = max(1, int(len(c) * top_frac))
    threshold = c_sorted[n_top - 1]
    hot = c >= threshold
    
    # Compute pressure: for each i, how much does a[i] contribute to hot indices?
    # c[k] = sum_j a[j] a[k-j]
    # d c[k] / d a[i] = a[k-i] + a[i-k]
    # pressure[i] = sum_{k in hot} a[k-i] (approximate, using one term)
    
    # Use FFT for efficiency
    p_hot = np.zeros(2 * n - 1)
    p_hot[hot] = 1.0
    
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_hot, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    # cross[i] ≈ sum_k p_hot[k] * a[k-i] (the "partner" contribution)
    pressure = cross
    
    # Normalize
    p_max = np.max(pressure)
    if p_max > 0:
        pressure /= p_max
    
    # Reduce where pressure is high
    S = np.sum(a)
    a_new = a * (1.0 - strength * pressure + strength * np.mean(pressure))
    a_new = np.maximum(a_new, 0.0)
    S_new = np.sum(a_new)
    if S_new > 0.01:
        a_new *= S / S_new
    return a_new
