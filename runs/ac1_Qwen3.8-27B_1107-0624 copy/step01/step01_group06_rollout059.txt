The key issue: for any sequence with a[0] > 0, we have b[0] = a[0]^2. And the sum is at least a[0]. So J >= 2*n*a[0]^2/(a[0]+...)^2. If a[0] is a significant fraction of the sum, this is large.

For the uniform sequence, a[0] = 1 and sum = n, so the contribution from b[0] to J is 2*n*1/n^2 = 2/n, which is negligible for large n. The maximum for the uniform sequence is b[n-1] = n (the center), giving J = 2.

So for large n, the endpoint values (b[0], b[2n-2]) are negligible, and the maximum is determined by the "bulk" of the convolution. The question is: can we make the bulk of the convolution flatter than the triangular shape of the uniform sequence?

For the uniform sequence, b[k] = k+1 for k < n, and b[k] = 2n-1-k for k >= n. The max is n at k = n-1.

For a "flattened" version, we want b[k] to be closer to its average (sum(a)^2/(2n-1)) for all k. The average for the uniform sequence is n^2/(2n-1) ≈ n/2. The max is n, which is 2x the average. We want to reduce this ratio.

This is exactly the "flat autocorrelation" problem. And the answer is: for non-negative sequences, it's hard to achieve a flat autocorrelation because all cross-terms are positive.

However, we can do slightly better than the uniform sequence by using a non-uniform shape that "spreads out" the energy. The optimal shape is likely a smooth curve that's slightly different from uniform.

"""
Multi-stage optimization of step function coefficients to minimize
J(a) = 2n * max(convolve(a,a)) / (sum(a))^2 using:
1. FFT-accelerated log-sum-exp smoothed objective
2. L-BFGS-B quasi-Newton optimization with adaptive tau annealing
3. Peak-shaving post-processing
4. Multiple diverse initializations
5. Coordinate descent finishing
"""

import numpy as np
import time
from scipy.optimize import minimize


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_objective_and_grad(a, tau):
    """Compute log-sum-exp smoothed J and its gradient.
    J_smooth = 2n * Tmax / S^2 where Tmax = tau*log(sum exp(c_k/tau))
    """
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def make_init_ramp(n):
    """Linearly decreasing from 1.0 to 0.3."""
    return 1.0 - 0.7 * np.arange(n) / max(n - 1, 1)


def make_init_ramp2(n):
    """Linearly decreasing from 1.0 to 0.1."""
    return 1.0 - 0.9 * np.arange(n) / max(n - 1, 1)


def make_init_cosine_decay(n):
    """Cosine decay from 1 to 0.2."""
    x = np.linspace(0, 1, n)
    return 0.2 + 0.8 * (1.0 + np.cos(np.pi * x)) / 2.0


def make_init_gaussian(n, sigma_frac=0.3, center_frac=0.5):
    """Gaussian centered at center_frac."""
    x = np.arange(n)
    center = center_frac * (n - 1)
    sigma = sigma_frac * n
    return np.exp(-(x - center) ** 2 / (2.0 * sigma * sigma))


def make_init_quadratic(n):
    """Quadratic bump offset from center."""
    x = np.arange(n)
    mid = 0.4 * (n - 1)
    return 1.0 - 0.5 * ((x - mid) / (n - 1)) ** 2


def make_init_sinusoidal(n, freq=1.0, phase=0.5):
    """Sinusoidal modulation."""
    x = np.arange(n)
    return 0.6 + 0.4 * np.sin(2 * np.pi * freq * x / n + phase)


def normalize(a, target_sum=None):
    """Normalize to target sum (default: len(a))."""
    n = len(a)
    if target_sum is None:
        target_sum = n
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n)
    return a * (target_sum / S)


def stage_l_bfgs(a_init, tau_start, tau_end, time_budget, seed=42):
    """Run L-BFGS-B with adaptive tau annealing. Returns (best_a, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    def objective(x):
        x = np.maximum(x, 0.0)
        x = x * (n / max(np.sum(x), 1e-10))
        J, _ = smooth_objective_and_grad(x, current_tau[0])
        return J

    def gradient(x):
        x = np.maximum(x, 0.0)
        x = x * (n / max(np.sum(x), 1e-10))
        _, g = smooth_objective_and_grad(x, current_tau[0])
        return g

    best_a = a.copy()
    best_J = compute_J(a)
    current_tau = [tau_start]

    # Anneal tau over the time budget
    num_stages = 8
    tau_schedule = tau_start * (tau_end / tau_start) ** (np.arange(num_stages) / max(num_stages - 1, 1))

def peak_shaving(a, n_iterations=50, step=0.01):
    """Direct peak shaving: reduce contributions to the max convolution entry."""
    n = len(a)
    a = a.copy()
    best_J = compute_J(a)
    best_a = a.copy()

    for it in range(n_iterations):
        c = convolve_fft(a)
        k_star = np.argmax(c)
        max_val = c[k_star]
        S = np.sum(a)
        J = 2.0 * n * max_val / (S * S)

        if J < best_J:
            best_J = J
            best_a = a.copy()

        # Find which a[i] contribute to b[k_star]
        # b[k_star] = sum_i a[i] * a[k_star - i] for valid i
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        indices = np.arange(i_min, i_max + 1)
        partners = k_star - indices
        # Contribution of a[i] to b[k_star] is a[i] * a[partner]
        contribs = np.zeros(n)
        contribs[indices] = a[indices] * a[partners]

        # Reduce elements with high contribution
        total_contrib = np.sum(contribs)
        if total_contrib < 1e-15:
            break

        # Reduce proportionally to contribution, but not below 0
        reductions = step * contribs / total_contrib
        new_a = a - reductions
        new_a = np.maximum(new_a, 0.0)

        # Check if this helps
        new_J = compute_J(new_a)
        if new_J < J:
            a = new_a
        else:
            # Try a smaller step
            reductions = step * 0.1 * contribs / total_contrib
            new_a = np.maximum(a - reductions, 0.0)
            new_J = compute_J(new_a)
            if new_J < J:
                a = new_a
            else:
                break

    return best_a, best_J


def coordinate_descent(a, time_budget, seed=42):
    """Coordinate descent: try small multiplicative changes to random blocks."""
    n = len(a)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + time_budget

    a = a.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    block_size = max(1, n // 100)

    while time.monotonic() < deadline:
        # Try random block perturbation
        w = int(rng.integers(1, block_size + 1))
        s = int(rng.integers(0, max(1, n - w)))
        factor = np.exp(rng.normal(0, 0.05, w))

        cand = a.copy()
        cand[s:s+w] *= factor
        cand = np.maximum(cand, 0.0)
        S = np.sum(cand)
        if S < 0.01:
            continue
        cand = cand * (n / S)

        J = compute_J(cand)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
            a = cand.copy()

    return best_a, best_J
