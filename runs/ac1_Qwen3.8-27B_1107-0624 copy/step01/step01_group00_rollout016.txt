"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative a of length n.
Uses FFT-based full gradient of the smoothed max (log-sum-exp), projected gradient
descent with heavy-ball momentum, adaptive temperature cooling, multiple diverse
seeds, stuck-detection restarts, and a coarse-to-fine multi-scale strategy.
"""

def proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def conv_sq(a, m):
    """Linear self-convolution a*a via FFT, returns length 2n-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * len(a) - 1]


def full_gradient(c, a, tau):
    """
    Compute full gradient of smooth_max(c, tau) w.r.t. a.
    grad_a = 2 * (w * a[::-1])[n-1:2n-1] where w = softmax(c/tau).
    Uses FFT for the convolution.
    """
    n = len(a)
    c_max = c.max()
    w = np.exp((c - c_max) / tau)
    w /= w.sum()
    # Compute (w * a_rev) via FFT
    a_rev = a[::-1]
    M = 1
    while M < len(w) + len(a_rev) - 1:
        M <<= 1
    W = np.fft.rfft(w, M)
    A = np.fft.rfft(a_rev, M)
    conv_full = np.fft.irfft(W * A, M)
    grad_a = 2.0 * conv_full[n - 1:2 * n - 1]
    return grad_a


def smooth_max_val(c, tau):
    """Value of the smoothed max."""
    c_max = c.max()
    return c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))


def make_seeds(n, rng):
    """Generate diverse initial seeds of length n."""
    seeds = []
    t = np.linspace(0, 1, n)
    # Uniform
    seeds.append(np.ones(n) / n)

    seeds.append(project_and_norm(0.5 + 0.5 * np.cos(2 * np.pi * t)))

    seeds.append(project_and_norm(np.exp(-50 * (t - 0.5) ** 2)))

    # Bell (sinc-like, positive part)

    sinc = np.sinc(x / np.pi)
    seeds.append(project_and_norm(np.abs(sinc)))

def project_and_norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, 1.0 / len(a))
    return a / s


def run_optimization(a0, t_end, rng, verbose=False, tau_start=0.05, alpha_mom=0.9):
    """
    Run projected gradient descent with momentum on the simplex.
    Minimizes max(convolve(a,a)) subject to a>=0, sum(a)=1.
    Returns best (a, value) found.
    """

    a = project_and_norm(a0.copy())
    m = 1

    best_a = a.copy()
    best_val = conv_sq(a, m).max()

    tau = tau_start
    lr = 0.1 / n ** 0.4
    momentum = alpha_mom
    buf = np.zeros(n)
    step = 0
    no_improve = 0
    last_time = time.monotonic()

    while time.monotonic() < t_end - 0.5:
        c = conv_sq(a, m)
        cur_val = c.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        step += 1

        # Adaptive schedule
        if step % 300 == 0:
            tau = max(0.005, tau * 0.93)
            momentum = min(0.97, momentum + 0.001)
        if step % 50 == 0:
            lr = max(lr * 0.995, 1e-6 / n)

        # Full FFT-based gradient

        # Normalize gradient for stable steps
        gn = np.linalg.norm(g)
        if gn > 1e-14:
            g = g / gn

        # Heavy-ball momentum (Nesterov-like)
        buf = momentum * buf - lr * g
        a_new = project_and_norm(a + lr * buf)

        # Simple acceptance: accept if not much worse (to allow exploration)
        if no_improve > 3000 and rng.random() < 0.1:
            # Stuck: try a larger perturbation
            # Find peak lag and push mass away
            peak_k = int(np.argmax(c))
            # Gradient at peak only
            lo = max(0, peak_k - n + 1)
            hi = min(peak_k, n - 1)
            contrib = np.zeros(n)
            idx = np.arange(lo, hi + 1)
            contrib[idx] = a[idx] * a[peak_k - idx]
            # Perturbation: move mass from high contribution to low
            pert = -5.0 * contrib / max(contrib.max(), 1e-15) + 1.0 / n
            a_pert = project_and_norm(a + 0.03 * pert)

            if c_pert.max() < cur_val * 1.05:
                a = a_pert
                buf = np.zeros(n)
                no_improve = 0
            elif rng.random() < 0.2:
                a = a_pert
                buf = np.zeros(n)
        else:
            a = a_new

    return best_a, best_val


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)
