from scipy.optimize import linprog

def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_solve(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

1. **Adaptive M**: Instead of using the current max as M, use M = (1+ε) * current_max where ε is small. This allows a small increase in the peak in exchange for a better overall shape.

Wait, that doesn't make sense. The LP maximizes sum(g) subject to conv(a,g) ≤ M. If M = current_max, the current a is feasible (conv(a,a) ≤ M element-wise). The LP finds the direction g that maximizes sum(g) while keeping conv(a,g) ≤ M.

This suggests using small step sizes (t ≈ 0.01-0.1) and many iterations.

Hybrid multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Phase 1: L2-norm gradient descent (smooth warmup) on diverse initializations.
Phase 2: LP-based block coordinate descent with line search (direct peak minimization).
Phase 3: Intensive final polish combining LP and L2 methods.
Uses momentum, adaptive learning rates, and perturbation-based restarts.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def normalize(a):
    a = np.clip(a, 0.0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    return a / s


def make_gaussian(n, sigma_frac):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n) / (n - 1)))


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 0.5))


def make_power(n, p):
    c = (n - 1) / 2.0
    x = np.abs(np.arange(n) - c) / (c + 0.5)
    return np.power(np.maximum(0.0, 1.0 - x), p)


def optimize_l2(a, deadline, rng, verbose=True):

    b = normalize(a.copy())
    best_b = b.copy()
    best_score = compute_objective(b)
    lr = 0.02
    stall = 0
    it = 0
    momentum = np.zeros(n)
    beta = 0.85

        it += 1
        c = np.convolve(b, b, mode='full')
        grad = 4.0 * np.correlate(c, b, mode='valid')

        b_new = b - lr * momentum
        b_new = np.clip(b_new, 0.0, None)
        if b_new.sum() < 1e-10:

        b_new = b_new / b_new.sum()

                b_new = normalize(b_new)

def optimize_lp(a, deadline, rng, verbose=True):

    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.06
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        step_candidates = [step, step * 0.5, step * 0.25, step * 1.5, 0.02, 0.005]
        step_candidates = sorted(set(max(0.001, min(0.4, s)) for s in step_candidates), reverse=True)

                improved = True
                step = min(0.25, s * 1.05)
                break

        if not improved:
            stall += 1
            step = max(0.003, step * 0.88)
            if stall % 12 == 0 and stall > 12:
                scale = min(0.2, 0.008 * stall)
                a_new = a * (1.0 + rng.normal(0, scale, n))
                a_new = normalize(a_new)
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new.copy()
                    best_score = sc
                    stall = 0
                a = a_new
            if stall > 120:
                break
        else:
            stall = 0

        if verbose and it % 200 == 0:
            print(f"      LP: iter={it}, score={best_score:.6f}, step={step:.4f}, stall={stall}")

    return best_a, best_score


def optimize_pressure(a, deadline, rng, verbose=True):
    n = len(a)
    b = normalize(a.copy())
    best_b = b.copy()
    best_score = compute_objective(b)
    lr = 0.04
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(b, b, mode='full')
        max_c = c.max()
        w = (c / max_c) ** 3
        pressure = np.correlate(w, b, mode='valid')
        p_mean = pressure.mean()
        p_std = pressure.std() + 1e-15
        z = (pressure - p_mean) / p_std
        b_new = b * np.exp(-lr * z)
        b_new = np.clip(b_new, 0.0, None)
        if b_new.sum() < 1e-10:
            break
        b_new = b_new / b_new.sum()
        sc = compute_objective(b_new)
        if sc < best_score - 1e-13:
            best_b = b_new.copy()
            best_score = sc
            b = b_new
            stall = 0
            lr = min(0.25, lr * 1.01)
        else:
            stall += 1
            lr = max(0.003, lr * 0.93)
            if stall > 40:
                noise = rng.normal(0, 0.06, n)
                b_new = b * (1.0 + noise)
                b_new = normalize(b_new)
                b = b_new
                stall = 0
            if stall > 200:
                break
        if verbose and it % 2000 == 0:
            print(f"      Press: iter={it}, score={best_score:.6f}, lr={lr:.5f}")

    return best_b, best_score


def optimize_combined(a, deadline, rng, verbose=True):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = compute_objective(a)

    while time.monotonic() < deadline - 5:
        elapsed = time.monotonic()
        remaining = deadline - elapsed
        if remaining < 20:
            break

        if remaining > 60:
            sub_dl = min(elapsed + 40, deadline - 10)
            a, sc = optimize_lp(a, sub_dl, rng, verbose=False)
        else:
            sub_dl = deadline - 5
            a, sc = optimize_l2(a, sub_dl, rng, verbose=False)

        if sc < best_score:
            best_a = a.copy()
            best_score = sc
            if verbose:
                print(f"      Combined: improved to {best_score:.6f}, remaining={remaining:.0f}s")

    return best_a, best_score


def try_configuration(a_init, deadline, rng, n):
    if n <= 1024:
        a_l2, sc_l2 = optimize_l2(a_init, min(deadline, time.monotonic() + 60), rng, verbose=False)
        a_opt, sc_opt = optimize_lp(a_l2, deadline, rng, verbose=False)
        if sc_l2 < sc_opt:
            a_opt, sc_opt = a_l2, sc_l2
    else:
        a_l2, sc_l2 = optimize_l2(a_init, min(deadline, time.monotonic() + 120), rng, verbose=False)
        a_opt, sc_opt = optimize_pressure(a_l2, deadline, rng, verbose=False)
        if sc_l2 < sc_opt:
            a_opt, sc_opt = a_l2, sc_l2
    return a_opt, sc_opt


def propose_candidate(seed=42, budget_s=1800):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p.copy()
    except Exception:
        pass

    print(f"=== Starting search, budget={budget_s}s ===")
    if parent is not None:
        ps = compute_objective(parent)
        print(f"Parent: n={len(parent)}, score={ps:.6f}")

    # Build list of (n, init_array, label, weight)
    candidates = []

    if parent is not None:
        pn = min(len(parent), 512)
        candidates.append((pn, parent[:pn].copy(), "parent_512", 4.0))
        pn2 = min(len(parent), 768)
        candidates.append((pn2, parent[:pn2].copy(), "parent_768", 2.5))

    for sigma in [0.33, 0.36, 0.38, 0.40, 0.42, 0.45]:
        candidates.append((512, make_gaussian(512, sigma), f"gauss512_{sigma}", 2.0))
    candidates.append((512, make_cosine(512), "cos512", 1.5))
    candidates.append((512, make_parabolic(512), "par512", 1.0))
    candidates.append((512, make_power(512, 1.5), "pow512_1.5", 1.0))
    candidates.append((512, make_power(512, 2.5), "pow512_2.5", 1.0))

    for n in [384, 448, 640, 768]:
        candidates.append((n, make_gaussian(n, 0.38), f"gauss{n}", 1.5))
    for n in [256, 1024, 1536, 2048]:
        candidates.append((n, make_gaussian(n, 0.38), f"gauss{n}", 1.0))

    r = np.abs(rng.normal(0, 1, 512))
    candidates.append((512, r, "rand512", 0.8))

    total_weight = sum(w for _, _, _, w in candidates)
    available_time = budget_s - 120.0
    time_per_weight = available_time / total_weight

    # Sort by weight descending
    candidates.sort(key=lambda x: -x[3])

    for idx, (n, a_init, label, w) in enumerate(candidates):
        if time.monotonic() > deadline - 30:
            break
        alloc = w * time_per_weight
        config_deadline = min(time.monotonic() + alloc, deadline)

        if config_deadline - time.monotonic() < 15:
            continue

        print(f"\n[{idx+1}/{len(candidates)}] n={n}, {label}, alloc={alloc:.0f}s, time_left={config_deadline-time.monotonic():.0f}s")
        a0_score = compute_objective(a_init)
        print(f"  Init score: {a0_score:.6f}")

        t0 = time.monotonic()
        a_opt, score = try_configuration(a_init, config_deadline, rng, n)
        elapsed_cfg = time.monotonic() - t0
        print(f"  Result: {score:.6f} (took {elapsed_cfg:.1f}s)")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.6f} (n={n}, {label}) ***")

    # Final intensive polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polish, {remaining:.0f}s remaining ===")
        print(f"  Before polish: {best_score:.6f}")

        # LP polish
        dl1 = time.monotonic() + remaining * 0.5
        best_seq, sc1 = optimize_lp(best_seq, dl1, rng, verbose=True)
        if sc1 < best_score:
            best_score = sc1
            print(f"  After LP: {best_score:.6f}")

        # L2 polish
        if time.monotonic() < deadline - 10:
            dl2 = deadline - 5
            best_seq, sc2 = optimize_l2(best_seq, dl2, rng, verbose=True)
            if sc2 < best_score:
                best_score = sc2
                print(f"  After L2: {best_score:.6f}")

        # One more LP if time
        if time.monotonic() < deadline - 15:
            dl3 = deadline - 5
            best_seq, sc3 = optimize_lp(best_seq, dl3, rng, verbose=True)
            if sc3 < best_score:
                best_score = sc3
                print(f"  After final LP: {best_score:.6f}")

    if best_seq is None:
        best_seq = make_gaussian(512, 0.38)
        best_score = compute_objective(best_seq)

    best_seq = np.clip(best_seq, 0.0, None)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    final_time = time.monotonic() - start_time
    print(f"\n=== DONE in {final_time:.1f}s. Final: score={best_score:.6f}, n={len(best_seq)} ===")

    return [float(x) for x in best_seq]
