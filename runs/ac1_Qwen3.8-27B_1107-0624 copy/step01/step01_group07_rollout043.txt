Let me think about a better approach. Instead of the subgradient of `max(c)`, let me use a **smooth approximation**:

`f_ε(a) = ε * log(sum_k exp(c_k / ε))` (log-sum-exp)

This is a smooth upper bound on `max(c_k)` and its gradient is:
`∇f_ε(a)_j = sum_k (exp(c_k/ε) / sum_m exp(c_m/ε)) * a_{k-j}`

This is a **weighted** sum where the weights are the softmax of `c`. For small ε, this concentrates on the peaks. For larger ε, it's smoother.

This might give better convergence properties than the hard subgradient.

```python
    """Compute gradient of log-sum-exp approximation of max(conv(a,a))."""
```

Wait, I need to double-check the formula. The smooth objective is:
`f(a) = ε * log(sum_k exp(c_k / ε))`

`df/da_j = ε * (1/sum_m exp(c_m/ε)) * sum_k exp(c_k/ε) * dc_k/da_j`
`= ε * sum_k (exp(c_k/ε) / sum_m exp(c_m/ε)) * a_{k-j}`
`= ε * sum_k w_k * a_{k-j}`

where `w_k = exp(c_k/ε) / sum_m exp(c_m/ε)`.

So the gradient is `ε * sum_k w_k * a_{k-j}` for each `j`.

The factor of ε just scales the gradient, which we can absorb into the learning rate. So effectively:
`g_j = sum_k w_k * a_{k-j}`

This is the cross-correlation of `w` and `a`, which I can compute with FFT.

This is much better than the hard subgradient because:
1. It's smooth (differentiable)
2. It uses information from all peaks, not just the argmax

```python
def build_conv_matrix_sparse(a, peak_indices):
    """Build sparse constraint rows for LP: only for peak indices."""
    n = len(a)
    # For each peak index k, the constraint is sum_j a_j * g_{k-j} <= M
    # which is: for each j, the coefficient of g_j is a_{k-j} (if 0 <= k-j < n)
    # Return as dense matrix for simplicity (or use scipy sparse)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in peak_indices:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)  # constraint index (will remap)
                cols.append(j)
                vals.append(a[idx])
    # Remap row indices to 0..len(peak_indices)-1
    k_to_row = {k: i for i, k in enumerate(peak_indices)}
    rows = [k_to_row[r] for r in rows]
    return csr_matrix((vals, (rows, cols)), shape=(len(peak_indices), n))

def lp_step(a, M, top_frac=0.3):
    """
    Solve: max sum(g) s.t. conv(a,g)_k <= M for top (top_frac) peaks, g >= 0.
    Returns g or None.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Get top peaks
    num_peaks = max(1, int(top_frac * (2 * n - 1)))
    top_vals = np.sort(c)[::-1][:num_peaks]
    threshold = top_vals[-1]
    peak_mask = c >= threshold
    peak_indices = np.where(peak_mask)[0]
    
    # For the LP, use only a subset of constraints (the top ones)
    # to keep the LP small
    if len(peak_indices) > 64:
        # Use only the top 64
        top_idx = np.argsort(c)[::-1][:64]
        peak_indices = top_idx
    
    # Build constraint matrix
    A = np.zeros((len(peak_indices), n))
    for i, k in enumerate(peak_indices):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))

def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)

def make_flat_top(n, flat_frac=0.5):
    """Flat top with cosine roll-off."""
    x = np.arange(n) / n
    center = 0.5
    half_flat = flat_frac / 2.0
    a = np.ones(n)
    # Roll off at edges
    edge = 1.0 - flat_frac
    for i in range(n):
        if x[i] < edge / 2:
            a[i] = 0.5 * (1 + np.cos(np.pi * x[i] / (edge / 2)))
        elif x[i] > 1 - edge / 2:
            a[i] = 0.5 * (1 + np.cos(np.pi * (x[i] - (1 - edge / 2)) / (edge / 2)))
    return np.maximum(a, 0)

def make_kaiser(n, beta=5.0):
    """Kaiser window."""
    n = n - 1
    alpha = np.arange(-n/2, n/2 + 1) / (n/2)
    from scipy.special import i0
    return i0(beta * np.sqrt(np.maximum(1 - alpha**2, 0))) / i0(beta)
```

```python
def project_simplex(v, cap=1.0):
    """Project v onto {x >= 0, sum(x) = cap}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]  # decreasing order
    cssv = np.cumsum(u)  # cssv[j] = sum of top (j+1) elements (0-indexed)
    # Find largest j (0-indexed) such that u[j] - (cssv[j] - cap) / (j+1) > 0
    # i.e., (j+1) * u[j] - cssv[j] + cap > 0
    # i.e., (j+1) * u[j] - cssv[j] > -cap
    js = np.arange(n)
    cond = u * (js + 1) - cssv + cap > 0
    if not cond.any():
        return np.full(n, cap / n)
    rho = np.where(cond)[0][-1]  # 0-indexed
    theta = (cssv[rho] - cap) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based smooth gradient
optimization with adaptive epsilon annealing, momentum, simplex projection,
and periodic LP refinement. Multiple initializations across different
sequence lengths and window types.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import i0


def project_simplex(v, cap=1.0):
    """Project v onto {x >= 0, sum(x) = cap} using Duchi et al. (2011)."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    js = np.arange(n)
    cond = u * (js + 1) - cssv + cap > 0
    if not cond.any():
        return np.full(n, cap / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - cap) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max()


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_flat_top(n, flat_frac=0.5):
    """Flat top with cosine roll-off at edges."""
    x = np.arange(n) / n
    a = np.ones(n)
    edge = (1.0 - flat_frac) / 2.0
    if edge > 0:
        left_mask = x < edge
        right_mask = x > 1.0 - edge
        a[left_mask] = 0.5 * (1.0 + np.cos(np.pi * x[left_mask] / edge))
        a[right_mask] = 0.5 * (1.0 + np.cos(np.pi * (x[right_mask] - (1.0 - edge)) / edge))
    return np.maximum(a, 0.0)


def make_kaiser(n, beta=5.0):
    """Kaiser window."""
    if n <= 1:
        return np.ones(n)
    m = n // 2
    alpha = np.arange(-m, m + 1) / m
    return i0(beta * np.sqrt(np.maximum(1.0 - alpha ** 2, 0.0))) / i0(beta)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(1.0 - np.abs(np.arange(n) - c) / (c + 1.0), 0.0)


def optimize_smooth(a, deadline, rng, verbose=True):
    """
    FFT-based smooth gradient optimization with momentum, epsilon annealing,
    and adaptive step size. Assumes sum(a)=1.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_obj_normalized(a)

    N = 1
    while N < 2 * n:
        N *= 2

    lr = 0.08
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    it = 0

    eps_coarse = 0.04
    eps_fine = 0.001
    anneal_iters = 5000

    while time.monotonic() < deadline:
        it += 1

        # Anneal epsilon
        if it <= anneal_iters:
            t_frac = it / anneal_iters
            eps = eps_coarse * (eps_fine / eps_coarse) ** t_frac
        else:
            eps = eps_fine

        # Compute convolution via FFT
        a_fft = np.fft.rfft(a, N)
        c = np.fft.irfft(a_fft * a_fft, N)[:2 * n - 1]
        c_max = c.max()
        cur_score = 2.0 * n * c_max

        # Update best and stall counter
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
            lr = min(0.15, lr * 1.01)
        else:
            stall += 1
            lr = max(0.002, lr * 0.995)

        # Compute softmax weights for smooth gradient
        c_shifted = c - c_max
        with np.errstate(over='ignore', under='ignore'):
            weights = np.exp(c_shifted / eps)
        w_sum = weights.sum()
        if not np.isfinite(w_sum) or w_sum < 1e-300:
            weights = (c >= c_max - 1e-10).astype(np.float64)
            w_sum = weights.sum()
            if w_sum == 0:
                weights = np.ones(2 * n - 1)
                w_sum = weights.sum()
        weights /= w_sum

        # Gradient via FFT: g_j = sum_k w_k * a_{k-j}
        W = np.fft.rfft(weights, N)
        g = np.fft.irfft(np.conj(a_fft) * W, N)[:n]

        # Normalize gradient for stable step size
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        else:
            g = np.zeros(n)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * g
        a_new = a - lr * v
        a_new = project_simplex(a_new, 1.0)
        a = a_new

        # Stalling handling
        if stall > 200 and it % 40 == 0:
            # Try wider epsilon for exploration
            eps_alt = eps * 5.0
            c_shifted_alt = c - c_max
            with np.errstate(over='ignore'):
                weights_alt = np.exp(c_shifted_alt / eps_alt)
            w_sum_alt = weights_alt.sum()
            if np.isfinite(w_sum_alt) and w_sum_alt > 1e-300:
                weights_alt /= w_sum_alt
                W_alt = np.fft.rfft(weights_alt, N)
                g_alt = np.fft.irfft(np.conj(a_fft) * W_alt, N)[:n]
                g_norm_alt = np.linalg.norm(g_alt)
                if g_norm_alt > 1e-15:
                    g_alt /= g_norm_alt
                v = momentum * v + (1.0 - momentum) * g_alt
                a = project_simplex(a - lr * v, 1.0)
            stall = 100

        if stall > 600:
            # Random perturbation from best
            noise_scale = 0.1 + 0.05 * rng.random()
            noise = np.abs(rng.normal(0, noise_scale, n))
            a = project_simplex(best_a * (1.0 + noise), 1.0)
            v = np.zeros(n)
            lr = 0.06
            stall = 0

        if stall > 2000:
            # More aggressive restart
            a = project_simplex(best_a * (1.0 + np.abs(rng.normal(0, 0.3, n))), 1.0)
            v = np.zeros(n)
            lr = 0.04
            momentum = 0.7
            stall = 0

        if verbose and it % 5000 == 0:
            print(f"        [smooth] iter={it}, best={best_score:.6f}, eps={eps:.5f}, lr={lr:.4f}, stall={stall}")

    return best_a, best_score


def lp_step(a, M, max_constraints=128):
    """
    Solve LP: max sum(g) s.t. conv(a,g)_k <= M for top peaks, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Use top max_constraints peaks
    num_c = min(max_constraints, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:num_c]
    threshold = c[top_idx[-1]]
    peak_indices = top_idx[c[top_idx] >= threshold]
    peak_indices = np.sort(peak_indices)

    if len(peak_indices) == 0:
        return None

    # Build constraint matrix: A[i, j] = a[peak_indices[i] - j] if valid
    A = np.zeros((len(peak_indices), n))
    for i, k in enumerate(peak_indices):
        j_start = max(0, k - (n - 1))
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]

    c_obj = -np.ones(n)
    b_ub = np.full(len(peak_indices), M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-10:
                return g / g.sum()
    except Exception:
        pass
    return None


def optimize_with_lp(a, deadline, rng, lp_interval=2000, verbose=True):
    """
    Smooth gradient optimization with periodic LP refinement.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    a = a / S

    best_a = a.copy()
    best_score = compute_obj_normalized(a)

    N = 1
    while N < 2 * n:
        N *= 2

    lr = 0.08
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    it = 0
    last_lp = 0

    eps_coarse = 0.04
    eps_fine = 0.001
    anneal_iters = 5000

    while time.monotonic() < deadline:
        it += 1

        # Anneal epsilon
        if it <= anneal_iters:
            t_frac = it / anneal_iters
            eps = eps_coarse * (eps_fine / eps_coarse) ** t_frac
        else:
            eps = eps_fine

        # Compute convolution via FFT
        a_fft = np.fft.rfft(a, N)
        c = np.fft.irfft(a_fft * a_fft, N)[:2 * n - 1]
        c_max = c.max()
        cur_score = 2.0 * n * c_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
            lr = min(0.15, lr * 1.01)
        else:
            stall += 1
            lr = max(0.002, lr * 0.995)

        # Periodic LP step for escape
        if stall > 100 and it - last_lp > lp_interval:
            last_lp = it
            g_lp = lp_step(a, c_max, max_constraints=min(96, n))
            if g_lp is not None:
                for t_mix in [0.5, 0.3, 0.2, 0.1]:
                    a_try = project_simplex((1.0 - t_mix) * a + t_mix * g_lp, 1.0)
                    sc_try = compute_obj_normalized(a_try)
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a_try.copy()
                        a = a_try
                        stall = 0
                        if verbose:
                            print(f"        [LP] iter={it}, improved to {best_score:.6f}")
                        break
                if verbose:
                    print(f"        [LP] iter={it}, stall reset")
                continue

        # Smooth gradient
        c_shifted = c - c_max
        with np.errstate(over='ignore', under='ignore'):
            weights = np.exp(c_shifted / eps)
        w_sum = weights.sum()
        if not np.isfinite(w_sum) or w_sum < 1e-300:
            weights = (c >= c_max - 1e-10).astype(np.float64)
            w_sum = weights.sum()
            if w_sum == 0:
                weights = np.ones(2 * n - 1)
                w_sum = weights.sum()
        weights /= w_sum

        W = np.fft.rfft(weights, N)
        g = np.fft.irfft(np.conj(a_fft) * W, N)[:n]

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm

        v = momentum * v + (1.0 - momentum) * g
        a_new = a - lr * v
        a_new = project_simplex(a_new, 1.0)
        a = a_new

        # Stalling
        if stall > 500:
            noise = np.abs(rng.normal(0, 0.1, n))
            a = project_simplex(best_a * (1.0 + noise), 1.0)
            v = np.zeros(n)
            lr = 0.06
            stall = 0

        if stall > 2000:
            break

        if verbose and it % 5000 == 0:
            print(f"        [lp-smooth] iter={it}, best={best_score:.6f}, lr={lr:.4f}, stall={stall}")

    return best_a, best_score
```
