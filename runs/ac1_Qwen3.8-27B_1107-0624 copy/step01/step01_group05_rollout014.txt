            break

def subgradient_polish(z0, deadline, rng):
    """Aggressive subgradient polishing with adaptive step sizes."""
    global _best_score, _best_z
    z = np.clip(z0.astype(np.float64), -50, 50)
    p = probs(z)
    s = score(p)
    update_best(z, s)
    
    step = 0.5
    patience = 0
    max_patience = 50
    noise_scale = 0.1
    
    while time.monotonic() < deadline:
        c = self_conv(p)
        cmax = np.max(c)
        
        # Use top peaks for subgradient
        topk = min(16, 2*N-1)
        idx = np.argpartition(c, -topk)[-topk:]
        # Weight by peak height (softer than uniform)
        w = np.zeros(2*N-1)
        peaks = c[idx]
        w[idx] = peaks ** 4  # Emphasize larger peaks
        w /= w.sum()
        
        # Compute subgradient
        q = np.flip(w)
        M = 1 << (3*N - 2).bit_length()
        fp = np.fft.rfft(p, M)
        fq = np.fft.rfft(q, M)
        conv = np.fft.irfft(fp * fq, M)
        gp = 2.0 * conv[N-1:2*N-1][::-1]
        
        # Projected subgradient on simplex (via logits)
        sdot = np.dot(gp, p)
        gz = p[1:] * (gp[1:] - sdot)
        
        gnorm = np.linalg.norm(gz)
        if gnorm < 1e-15:
            break
        
        direction = -gz / gnorm
        
        # Backtracking line search
        accepted = False
        for trial in range(15):
            step_t = step * (0.5 ** trial)
            if step_t < 1e-8:
                break
            zt = z + step_t * direction
            zt = np.clip(zt, -50, 50)
            pt = probs(zt)
            st = score(pt)
            if st < s - 1e-14:
                z = zt
                p = pt
                s = st
                update_best(z, s)
                accepted = True
                step = min(2.0, step * 1.1)
                break
        
        if not accepted:
            patience += 1
            step *= 0.7
            # Add noise to escape local minima
            if patience > 10:
                z_noise = z + rng.normal(0, noise_scale, N-1)
                z_noise = np.clip(z_noise, -50, 50)
                p_noise = probs(z_noise)
                s_noise = score(p_noise)
                if s_noise < s:
                    z = z_noise
                    p = p_noise
                    s = s_noise
                    update_best(z, s)
                    patience = 0
                    step = 0.5
                else:
                    noise_scale *= 1.3
                    if patience > max_patience:
                        break
        else:
            patience = 0
    
    return z

    global _obj_alpha, _obj_deadline
    _obj_alpha = float(alpha)
    _obj_deadline = float(deadline)

    z0 = np.clip(np.asarray(z0, dtype=np.float64), -50.0, 50.0)
    try:
        res = minimize(
            _obj_and_grad, z0, method='L-BFGS-B', jac=True,
            bounds=_BOUNDS,
            options={'maxiter': 400, 'maxfun': maxfun, 'ftol': 1e-15, 'gtol': 1e-11}
        )
        return np.clip(res.x, -50.0, 50.0)
    except Timeout:
        return z0
    except Exception:
        return z0


def _subgradient_polish(z0, deadline, rng):
    """Aggressive subgradient polishing with backtracking and noise."""
    global _best_score, _best_z
    z = np.clip(np.asarray(z0, dtype=np.float64), -50.0, 50.0)
    p = _probs(z)
    s = _score(p)
    _update_best(z, s)

    step = 0.4
    patience = 0
    noise_scale = 0.08

    while time.monotonic() < deadline:
        c = _self_conv(p)
        cmax = float(np.max(c))

        # Weight top peaks with power emphasis
        topk = min(32, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * N - 1, dtype=np.float64)
        peaks = c[idx]
        w[idx] = peaks ** 6
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w[int(np.argmax(c))] = 1.0
        else:
            w /= wsum

        # Compute subgradient
        q = np.flip(w)
        fp = np.fft.rfft(p, FFT_GRAD)
        fq = np.fft.rfft(q, FFT_GRAD)
        conv = np.fft.irfft(fp * fq, FFT_GRAD)
        gp = 2.0 * conv[N - 1:2 * N - 1][::-1]

        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))

        if gnorm < 1e-15:
            break

        direction = -gz / gnorm

        # Backtracking line search
        accepted = False
        for trial in range(18):
            step_t = step * (0.55 ** trial)
            if step_t < 1e-9:
                break
            zt = z + step_t * direction
            zt = np.clip(zt, -50.0, 50.0)
            pt = _probs(zt)
            st = _score(pt)
            if st < s - 1e-15:
                z = zt
                p = pt
                s = st
                _update_best(z, s)
                accepted = True
                step = min(1.5, step * 1.15)
                break

        if not accepted:
            patience += 1
            step *= 0.65
            if patience > 15:
                # Try noisy restart
                best_noise_s = s
                best_noise_z = z
                best_noise_p = p
                n_tries = min(8, max(3, patience // 3))
                for _ in range(n_tries):
                    zt = z + rng.normal(0.0, noise_scale, N - 1)
                    zt = np.clip(zt, -50.0, 50.0)
                    pt = _probs(zt)
                    st = _score(pt)
                    if st < best_noise_s:
                        best_noise_s = st
                        best_noise_z = zt
                        best_noise_p = pt
                if best_noise_s < s:
                    z = best_noise_z
                    p = best_noise_p
                    s = best_noise_s
                    _update_best(z, s)
                    patience = 0
                    step = 0.4
                else:
                    noise_scale *= 1.4
                    if patience > 100:
                        break
        else:
            patience = max(0, patience - 1)

    return z


def _density_to_z(f):
    """Convert a density array to logits."""
    f = np.asarray(f, dtype=np.float64)
    f = np.maximum(f, 1e-15)
    z = np.log(f[1:]) - np.log(f[0])
    return np.clip(z, -50.0, 50.0)


def _make_candidates(rng):
    """Generate diverse structured initial candidates."""
    cands = []
    x = _X

    # Uniform
    cands.append(np.zeros(N - 1, dtype=np.float64))

    # Logistic (various steepness)
    for beta in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0, 15.0, 25.0]:
        f = 2.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        cands.append(_density_to_z(f))

    # Power law
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
        f = np.power(x + 1e-8, a)
        cands.append(_density_to_z(f))
        f2 = np.power(1.0 - x + 1e-8, a)
        cands.append(_density_to_z(f2))

    # U-shape (bathtub)
    for a in [0.3, 0.5, 1.0, 2.0, 4.0, 8.0]:
        t = 2.0 * x - 1.0
        f = 1.0 + a * t ** 2
        cands.append(_density_to_z(f))

    # Deep U-shape
    for a in [2.0, 5.0, 10.0, 20.0]:
        t = 2.0 * x - 1.0
        f = 1.0 + a * t ** 4
        cands.append(_density_to_z(f))

    # Cosine variations
    for fq in [1, 2, 3, 4, 5]:
        for amp in [0.2, 0.4, 0.6, 0.8, 0.95]:
            f = 1.0 + amp * np.cos(2.0 * np.pi * fq * x)
            f = np.maximum(f, 1e-10)
            cands.append(_density_to_z(f))

    # Sine
    for fq in [1, 2, 3]:
        for phase in [0.0, np.pi / 3, np.pi / 2, np.pi]:
            f = 1.0 + 0.5 * np.sin(2.0 * np.pi * fq * x + phase)
            f = np.maximum(f, 1e-10)
            cands.append(_density_to_z(f))

    # Two-bump
    for sigma in [0.05, 0.08, 0.1, 0.15, 0.2]:
        for center in [0.15, 0.2, 0.3]:
            f = (np.exp(-((x - center) ** 2) / (2 * sigma ** 2)) +
                 np.exp(-((x - (1.0 - center)) ** 2) / (2 * sigma ** 2)) + 0.05)
            cands.append(_density_to_z(f))

    # Gaussian
    for sigma in [0.08, 0.12, 0.15, 0.2, 0.25, 0.35]:
        f = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
        cands.append(_density_to_z(f))

    # Exponential
    for beta in [1.0, 2.0, 3.0, 5.0, 8.0]:
        f = np.exp(beta * x)
        cands.append(_density_to_z(f))
        f2 = np.exp(-beta * x)
        cands.append(_density_to_z(f2))

    # Triple cosine (richer shapes)
    for _ in range(20):
        f = np.ones(N, dtype=np.float64)
        for k in range(1, 5):
            amp = rng.uniform(0.1, 0.8) / k
            ph = rng.uniform(0, 2 * np.pi)
            f += amp * np.cos(2 * np.pi * k * x + ph)
        f = np.maximum(f, 1e-10)
        cands.append(_density_to_z(f))

    # Random smooth (low-freq Fourier)
    for _ in range(25):
        z_rand = np.zeros(N - 1, dtype=np.float64)
        xz = _X[1:]
        for k in range(1, 10):
            amp = rng.normal(0, 2.0 / (k + 0.5))
            ph = rng.uniform(0, 2 * np.pi)
            z_rand += amp * np.sin(2 * np.pi * k * xz + ph)
        cands.append(np.clip(z_rand, -50.0, 50.0))

    # From height_sequence_1 if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 1e-10, None)
        if base.size != N:
            old_x = np.linspace(0, 1, base.size)
            new_x = np.linspace(0, 1, N)
            base = np.interp(new_x, old_x, base)
        base = base / base.sum()
        z_base = _density_to_z(base)
        cands.append(z_base)
        for scale in [0.3, 0.5, 0.8, 1.2, 2.0]:
            z_pert = z_base + rng.normal(0, 0.3 * scale, N - 1)
            cands.append(np.clip(z_pert, -50.0, 50.0))
    except Exception:
        pass

    return cands


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    """Main search function. Returns best sequence found."""
    global _best_score, _best_z, _obj_alpha, _obj_deadline

    start_time = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    hard_deadline = start_time + budget_s - 3.0

    _best_score = float('inf')
    _best_z = None

    rng = np.random.default_rng(seed)

    print(f"Starting search with budget={budget_s:.0f}s, N={N}")

    # Phase 1: Evaluate diverse candidates
    print("Phase 1: Evaluating diverse candidates...")
    phase1_deadline = start_time + min(150.0, 0.12 * budget_s)
    cands = _make_candidates(rng)

    scored = []
    for i, z in enumerate(cands):
        if time.monotonic() > phase1_deadline:
            break
        p = _probs(z)
        s = _score(p)
        _update_best(z, s)
        scored.append((s, z))

    scored.sort(key=lambda x: x[0])
    print(f"Phase 1 done: {len(scored)} evaluated, best={_best_score:.6f}")

    # Phase 2: L-BFGS-B with alpha annealing from top candidates
    print("Phase 2: L-BFGS-B with alpha annealing...")
    phase2_deadline = start_time + 0.55 * budget_s

    top_k = min(10, len(scored))
    for i in range(top_k):
        if time.monotonic() > phase2_deadline:
            break
        s0, z0 = scored[i]
        print(f"  Optimizing candidate {i} (score={s0:.6f})...")

        # Alpha annealing: geometric progression
        alphas = [5e3, 1e4, 3e4, 1e5, 3e5, 1e6, 3e6, 8e6]
        z_current = z0.copy()

        for j, alpha in enumerate(alphas):
            if time.monotonic() > phase2_deadline:
                break
            remaining_frac = (phase2_deadline - time.monotonic()) / max(1.0, phase2_deadline - start_time)
            maxfun = int(max(200, min(2000, 800 * remaining_frac * (1.0 + 0.5 / (j + 1)))))
            z_current = _lbfgs_run(z_current, alpha, maxfun, phase2_deadline)

        p = _probs(z_current)
        s = _score(p)
        _update_best(z_current, s)
        print(f"  Candidate {i} -> {s:.6f}")

    # Phase 3: Subgradient polishing
    print(f"Phase 3: Subgradient polishing. Best so far: {_best_score:.6f}")
    phase3_deadline = start_time + 0.82 * budget_s

    if _best_z is not None and time.monotonic() < phase3_deadline:
        _subgradient_polish(_best_z, phase3_deadline, rng)

    # Phase 4: Final L-BFGS-B + short polish
    print(f"Phase 4: Final optimization. Best so far: {_best_score:.6f}")

    if _best_z is not None and time.monotonic() < hard_deadline - 10:
        # Multiple short L-BFGS-B from perturbed best
        n_tries = 5
        for attempt in range(n_tries):
            if time.monotonic() > hard_deadline - 15:
                break
            noise = 0.05 + 0.1 * attempt
            z0 = _best_z + rng.normal(0.0, noise, N - 1)
            z0 = np.clip(z0, -50.0, 50.0)

            alpha = 5e6 if attempt < 2 else 1e7
            maxfun = int(max(300, min(1500, (hard_deadline - time.monotonic() - 10) / 0.008)))
            z_new = _lbfgs_run(z0, alpha, maxfun, hard_deadline - 10)

            p = _probs(z_new)
            s = _score(p)
            _update_best(z_new, s)

        # Final short subgradient polish
        if time.monotonic() < hard_deadline - 5:
            _subgradient_polish(_best_z, hard_deadline - 1, rng)

    # Convert best to sequence
    if _best_z is not None:
        p = _probs(_best_z)
    else:
        p = np.ones(N, dtype=np.float64) / N

    seq = [float(v) for v in p]

    # Verify with evaluate_sequence
    try:
        verified = evaluate_sequence(seq)
        print(f"Verified score: {verified:.8f}")
    except Exception as e:
        print(f"Verification error: {e}")
        verified = _best_score

    # Fallback if something went wrong
    if not np.isfinite(verified) or verified > 2.0:
        seq = [1.0 / N] * N
        verified = 2.0
        print("Using uniform fallback, score=2.0")

    # Compare with inherited sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            base_seq = [float(v) for v in base]
            base_sc = evaluate_sequence(base_seq)
            if np.isfinite(base_sc) and base_sc < verified:
                seq = base_seq
                verified = base_sc
                print(f"Using inherited sequence, score={verified:.8f}")
    except Exception:
        pass

    print(f"FINAL SCORE: {verified:.8f}, length={len(seq)}")
    return seq
