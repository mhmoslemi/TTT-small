def evaluate_sequence(sequence: list[float]) -> float:
    """Evaluates a sequence of coefficients. Lower is better."""
    # Reject if not a list

def full_convolve_fft(arr):
    """FFT convolution for full mode"""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = arr.size
    if n == 0:
        return np.array([])

    fc = np.fft.irfft(fa * fa, next_size)[:2*n-1]
