```python
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iterations
combined with L-BFGS-B gradient polishing.

The key idea: fix current probability vector p, solve the LP
    max sum(q) s.t. (p*q)_k <= M, q >= 0
where M = max(p*p). The optimal q gives a direction that reduces the peak
convolution coefficient. Blend p toward q and iterate.

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

N = 4096
FFT_SIZE = 8192  # next power of 2 >= 2*N - 1

_best_score = float('inf')
_best_p = None
_deadline = 0.0

def _self_conv(p):
    """Compute linear self-convolution p*p."""
    f = np.fft.rfft(p, FFT_SIZE)
    c = np.fft.irfft(f * f, FFT_SIZE).real[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c

def _score(p):
    """Compute the objective value."""
    c = _self_conv(p)
    return 2.0 * float(N) * float(np.max(c))

def _update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = np.array(p, copy=True)

    """Build row/col index arrays for the convolution operator matrix."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        if k < n:
            c = np.arange(k + 1, dtype=np.int32)
        else:
            c = np.arange(k - n + 1, n, dtype=np.int32)
        rows_parts.append(np.full(len(c), k, dtype=np.int32))
        cols_parts.append(c)
    return np.concatenate(rows_parts), np.concatenate(cols_parts)

def _make_conv_matrix(p, rows, cols, n):
    """Create sparse convolution matrix A where A[k,j] = p[k-j]."""
    idx = (rows - cols).astype(np.intp)
    vals = p[idx]
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _solve_lp(p, M, rows, cols, n):
    """Solve max sum(q) s.t. A@q <= M, q >= 0."""
    A = _make_conv_matrix(p, rows, cols, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None

def _blend_step(p, q, M):
    """Try different blend factors and return the best."""
    q_sum = float(np.sum(q))
    if q_sum <= 0:

    q_norm = q / q_sum
    best_p = p
    best_s = _score(p)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

        p_new = np.maximum(p_new, 0.0)

        s = _score(p_new)
        if s < best_s:

            best_p = p_new
    return best_p, best_s
```

  (p_pad * w_pad_rev)[i] = sum_l p_pad[l] * w_pad_rev[i-l] = sum_l p_pad[l] * w_pad[(i-l) mod S]

For i in [0, N-1] and l in [0, N-1]: (i-l) mod S = i-l if i>=l, or S+i-l if i<l.
- If i >= l: w_pad[i-l] = w[i-l] (since 0 <= i-l <= N-1 < 2N-1)
- If i < l: w_pad[S+i-l] = 0 (since S >= 2N-1 > i+l >= S+i-l... wait, S+i-l >= S-(N-1) which might still be < 2N-1)

Hmm, this is getting complicated. Let me just use a simpler approach.

Actually, the cleanest way is:

g[i] = sum_{l=0}^{N-1} p[l] * w[i+l]

This is the same as: reverse p, cross-correlate with w, and extract the right part.

Let p_rev[l] = p[N-1-l]
Then sum_l p[l] * w[i+l] = sum_l p_rev[N-1-l] * w[i+l]

Let m = N-1-l, so l = N-1-m:
= sum_m p_rev[m] * w[i+N-1-m]
= (p_rev * w)[i+N-1]  (linear convolution)

So I need the linear convolution of p_rev and w, evaluated at positions N-1 to 2N-2.

Linear convolution of two sequences of lengths N and 2N-1 has length N + (2N-1) - 1 = 3N - 2.
FFT size needed: at least 3N-2, so next power of 2 after 3*4096-2 = 12286, which is 16384.

```python
def _compute_gp(p, w):
    """Compute gp[i] = 2 * sum_l p[l] * w[i+l] for i=0..N-1."""
    size = 1 << (3 * N - 2).bit_length()  # 16384 for N=4096
    p_rev = p[::-1]
    P = np.fft.rfft(p_rev, size)
    W = np.fft.rfft(w, size)
    conv = np.fft.irfft(P * W, size).real
    gp = 2.0 * conv[N-1:2*N-1]
    return gp
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based best-response iterations
combined with L-BFGS-B gradient polishing and subgradient descent.

Phase 1: LP-based iterations - solve max sum(q) s.t. (p*q)_k <= M, q>=0
Phase 2: L-BFGS-B with adaptive softmax-smoothed max surrogate
Phase 3: Subgradient polishing with top-K averaging
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
import time

N = 4096
FFT_SELF = 8192
FFT_GRAD = 16384

_best_score = float('inf')
_best_p = None
_deadline = 0.0
_rows = None
_cols = None


def _init_structure():
    global _rows, _cols
    if _rows is None:
        parts_r = []
        parts_c = []
        for k in range(2 * N - 1):
            if k < N:
                c = np.arange(k + 1, dtype=np.int32)
            else:
                c = np.arange(k - N + 1, N, dtype=np.int32)
            parts_r.append(np.full(len(c), k, dtype=np.int32))
            parts_c.append(c)
        _rows = np.concatenate(parts_r)
        _cols = np.concatenate(parts_c)


def _self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_p(p):
    c = _self_conv(p)
    return 2.0 * float(N) * float(np.max(c))


def _update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = np.array(p, copy=True)


def _make_conv_matrix(p):
    idx = (_rows - _cols).astype(np.intp)
    vals = p[idx]
    return sp.csr_matrix((vals, (_rows, _cols)), shape=(2 * N - 1, N))


def _lp_solve(p, M):
    A = _make_conv_matrix(p)
    c_obj = -np.ones(N)
    b_ub = np.full(2 * N - 1, M)
    bounds = [(0.0, None)] * N
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _lp_iteration(p):
    c = _self_conv(p)
    M = float(np.max(c))
    q = _lp_solve(p, M)
    if q is None or float(np.sum(q)) <= 0:
        return p, 2.0 * N * M
    q_norm = q / float(np.sum(q))
    best_p = p
    best_s = 2.0 * N * M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0]:
        p_new = (1.0 - t) * p + t * q_norm
        p_new = np.maximum(p_new, 0.0)
        s = float(np.sum(p_new))
        if s <= 1e-10:
            continue
        p_new /= s
        sc = _score_p(p_new)
        if sc < best_s:
            best_s = sc
            best_p = p_new
    return best_p, best_s


def _compute_gp(p, w):
    p_rev = p[::-1]
    P = np.fft.rfft(p_rev, FFT_GRAD)
    W = np.fft.rfft(w, FFT_GRAD)
    conv = np.fft.irfft(P * W, FFT_GRAD).real
    return 2.0 * conv[N - 1:2 * N - 1]


def _probs_from_z(z):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -50.0, 50.0)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _p_to_z(p):
    p = np.maximum(p, 1e-300)
    z = np.log(p[1:] / p[0])
    return np.clip(z, -50.0, 50.0)


def _obj_grad(z, alpha):
    if time.monotonic() >= _deadline:
        raise ValueError("timeout")
    p = _probs_from_z(z)
    c = _self_conv(p)
    cmax = float(np.max(c))
    _update_best(p, 2.0 * N * cmax)
    u = alpha * (c - cmax)
    e = np.exp(u)
    s_e = float(np.sum(e))
    if s_e <= 0 or not np.isfinite(s_e):
        J = cmax
        w = np.zeros(2 * N - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s_e) / alpha
        w = e / s_e
    gp = _compute_gp(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz


def _lbfgs(z0, alpha, max_fun):
    global _deadline
    x0 = np.clip(np.asarray(z0, dtype=np.float64), -50.0, 50.0)
    bounds = [(-50.0, 50.0)] * (N - 1)
    max_iter = max(50, max_fun // 3)
    try:
        minimize(_obj_grad, x0, args=(alpha,), method='L-BFGS-B', jac=True,
                 bounds=bounds,
                 options={'maxfun': max_fun, 'maxiter': max_iter,
                          'ftol': 1e-14, 'gtol': 1e-11, 'maxls': 50})
    except Exception:
        pass


def _subgrad_polish(p, end_time, rng):
    score = _score_p(p)
    _update_best(p, score)
    fails = 0
    while time.monotonic() < end_time:
        c = _self_conv(p)
        cmax = float(np.max(c))
        topk = min(16, 2 * N - 1)
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * N - 1)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))
        gp = _compute_gp(p, w)
        gp_proj = gp - float(np.mean(gp))
        gnorm = float(np.linalg.norm(gp_proj))
        if gnorm < 1e-15:
            break
        direction = -gp_proj / gnorm
        accepted = False
        for step in [0.3, 0.15, 0.07, 0.03, 0.01, 0.005]:
            p_new = p + step * direction
            p_new = np.maximum(p_new, 0.0)
            s = float(np.sum(p_new))
            if s <= 0:
                continue
            p_new /= s
            sc = _score_p(p_new)
            if sc < score - 1e-13:
                p = p_new
                score = sc
                _update_best(p, score)
                accepted = True
                break
        if not accepted:
            for _ in range(4):
                p_new = p + rng.normal(0, 0.015, N)
                p_new = np.maximum(p_new, 0.0)
                s = float(np.sum(p_new))
                if s <= 0:
                    continue
                p_new /= s
                sc = _score_p(p_new)
                if sc < score - 1e-13:
                    p = p_new
                    score = sc
                    _update_best(p, score)
                    accepted = True
                    break
        if accepted:
            fails = 0
        else:
            fails += 1
            if fails > 60:
                break
    return p, score


def _make_candidates(seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(N, dtype=np.float64) + 0.5) / N
    cands = []
    cands.append(np.ones(N) / N)
    for beta in [1, 2, 3, 5, 8, 12]:
        f = np.exp(beta * (x - 0.5))
        cands.append(f / f.sum())
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        f = np.power(x + 1e-8, a)
        cands.append(f / f.sum())
        f2 = np.power(1.0 - x + 1e-8, a)
        cands.append(f2 / f2.sum())
    for mu in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sig in [0.05, 0.1, 0.15, 0.25, 0.4]:
            f = np.exp(-0.5 * ((x - mu) / sig) ** 2)
            cands.append(f / f.sum())
    for _ in range(30):
        freqs = rng.choice([1, 2, 3, 4, 6, 8, 12, 16], size=rng.integers(3, 7), replace=False)
        amps = rng.normal(0, 0.8, size=len(freqs))
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        f = np.ones(N)
        for fq, amp, ph in zip(freqs, amps, phases):
            f += amp * np.sin(2.0 * np.pi * fq * x + ph)
        f = np.maximum(f, 0.01)
        cands.append(f / f.sum())
    return cands
```
