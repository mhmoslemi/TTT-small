        z = (c - M) / eps

def _normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s


def _score(f):
    """Compute score for a sum-1 non-negative sequence."""
    n = len(f)
    c = np.convolve(f, f)
    return 2.0 * n * float(c.max())


def _grad_smooth_max(f, temp):
    """
    Compute gradient of soft-max of conv(f,f) with temperature temp.
    Returns (gradient, M, c).
    If temp <= 0, uses hard max (equal weights on all max lags).
    """
    n = len(f)
    c = np.convolve(f, f)
    M = float(c.max())
    
    if temp > 1e-14:
        z = (c - M) / temp
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        sw = w.sum()
        if sw < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= sw
    else:
        w = np.zeros(2 * n - 1)
        active = c >= M - 1e-12
        w[active] = 1.0 / active.sum()
    
    # grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * f[m]  (where j+m < 2n-1)

    f_rev = f[::-1]
    cc = np.convolve(w, f_rev, mode='full')
    g = cc[n - 1: 2 * n - 1]
    grad = 2.0 * g
    return grad, M, c


def _proj_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        # All below zero after projection
        out = np.maximum(v, 0.0)
        s = out.sum()
        if s < 1e-300:
            return np.full(n, target / n)
        return out * (target / s)
    rho = idx[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _resize(f, N):
    """Resize sequence to length N via interpolation/averaging."""
    f = _normalize(f)
    m = f.size

        return f.copy()

def _lp_direction(f, c, M, max_rows, rng):
    """
    Solve LP: max sum(g) s.t. A g <= M, g >= 0.
    A has rows from active convolution lags.
    Returns normalized g or None.
    """
    try:
        from scipy.optimize import linprog
    except ImportError:
        return None
    
    n = len(f)
    if n > 4096:
        return None
    if M < 1e-300:
        return None
    
    # Choose active rows
    if max_rows >= 2 * n - 1:
        rows = np.arange(2 * n - 1)
    else:
        rows = np.argpartition(c, -max_rows)[-max_rows:]
    
    # Build A
    A = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[i, jlo:jhi] = f[k - (jlo + np.arange(jhi - jlo))]
    
    if not np.any(A):
        return None
    
    A = A / M
    c_obj = -np.ones(n)
    b_ub = np.ones(len(rows))
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception:
        return None
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1.0 + 1e-5:
            g = g / s
            return _normalize(g)
    return None
