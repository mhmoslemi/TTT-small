    s_new = s - lr * grad
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:
        s_new = np.ones(n) / n
    else:
        s_new /= s_sum

    return s_new, peak


def spectral_shaping(s, alpha=0.3, num_iters=3, fft_size=None):
    """Partially flatten the magnitude spectrum (Gerchberg-Saxton style)."""
    n = len(s)
    if fft_size is None:
        fft_size = _next_pow2(2 * n - 1)

    for _ in range(num_iters):
        S = np.fft.rfft(s, fft_size)
        mag = np.abs(S)
        target = mag.mean()
        if target < 1e-15:
            break
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.where(mag > 1e-15, target / mag, 1.0)
            new_mag = mag * (ratio ** alpha)
        phase = np.angle(S)
        S_new = new_mag * np.exp(1j * phase)
        s_new = np.real(np.fft.irfft(S_new, fft_size)[:n])
        s_new = np.maximum(s_new, 0.0)
        s_sum = s_new.sum()
        if s_sum < 1e-15:
            s_new = np.ones(n) / n
        else:
            s_new /= s_sum
        s = s_new
    return s


def run_gradient_optimize(s, num_iters, beta_start, beta_end, lr_start, lr_end,
                          spectral_every, perturb_every, rng, deadline):
    """Main smoothed gradient descent loop with annealing."""
    n = len(s)
    fft_size = _next_pow2(3 * n - 1)
    s = s.copy()
    best_s = s.copy()
    best_score = fast_score(s)

    no_improve = 0
    max_no_improve = 300

    for it in range(num_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(num_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        s_new, peak = gradient_step(s, beta, lr, fft_size)

        sc = fast_score(s_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_s = s_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        s = s_new

        # Periodic spectral shaping
        if spectral_every > 0 and it % spectral_every == 0 and it > 0:
            alpha = 0.15 + 0.25 * progress
            s = spectral_shaping(s, alpha=alpha, num_iters=3, fft_size=fft_size)
            sc = fast_score(s)
            if sc < best_score - 1e-13:
                best_score = sc
                best_s = s.copy()
                no_improve = 0

        # Periodic random perturbation
        if perturb_every > 0 and it % perturb_every == 0 and it > 0:
            if no_improve > 50:
                noise_scale = 0.03 + 0.02 * min(no_improve / 200.0, 1.0)
            else:
                noise_scale = 0.005
            s = s + noise_scale * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s_sum = s.sum()
            if s_sum < 1e-15:
                s = np.ones(n) / n
            else:
                s /= s_sum
            no_improve = 0

        # Major restart if stuck too long
        if no_improve > max_no_improve:
            blend = 0.3 + 0.3 * rng.random()
            s = blend * best_s + (1 - blend) * (np.abs(rng.standard_normal(n)) + 0.1)
            s = np.maximum(s, 0.0)
            s /= s.sum()
            no_improve = 0

    return best_s, best_score


def build_conv_matrix_A(s, top_k=None):
    """Build constraint matrix A where A[k,j] = s[k-j] if valid, else 0."""
    n = len(s)
    if top_k is None:
        c = compute_conv(s, _next_pow2(2 * n - 1))
        top_k = min(2 * n - 1, max(50, n // 2))
        idx = np.argsort(c)[::-1][:top_k]
    else:
        idx = np.arange(2 * n - 1)[:top_k]

    A = np.zeros((len(idx), n))
    for row, k in enumerate(idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[row, j] = s[k - j]
    return A, idx


def lp_refine(s, rng, max_iters=50, deadline=None):
    """LP-guided refinement using top-K constraint relaxation."""
    n = len(s)
    best_s = s.copy()
    best_sc = fast_score(s)
    no_imp = 0

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = compute_conv(s, _next_pow2(2 * n - 1))
        M = np.max(c)

        # Use top K constraints (K = min(n, 2n-1, 200))
        K = min(n, 2 * n - 1, 200)
        top_idx = np.argsort(c)[::-1][:K]

        A = np.zeros((K, n))
        for row, k in enumerate(top_idx):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            A[row, j_min:j_max + 1] = s[k - j_min:k - j_max]

        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 5.0, 'presolve': True})
            if not res.success or res.x is None:
                break
            b = np.maximum(res.x, 0.0)
        except Exception:
            break

        b_sum = b.sum()
        if b_sum < 1e-14:
            break

        b = b / b_sum

        # Line search
        cur_sc = fast_score(s)
        best_t = 0.0
        best_ts = cur_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * s + t * b
            cand = np.maximum(cand, 0.0)
            cand_s = cand.sum()
            if cand_s < 1e-15:
                continue
            cand /= cand_s
            ts = fast_score(cand)
            if ts < best_ts - 1e-13:
                best_ts = ts
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 10:
                break
            s = s + 0.01 * rng.standard_normal(n)
            s = np.maximum(s, 0.0)
            s /= s.sum()
            continue

        s = (1.0 - best_t) * s + best_t * b
        s = np.maximum(s, 0.0)
        s /= s.sum()
        no_imp = 0

        sc = fast_score(s)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_s = s.copy()

    return best_s, best_sc


def generate_starters(n, rng, hseq=None):
    """Generate diverse starting sequences of length n."""
    candidates = []

    # Previous best solution resampled
    if hseq is not None:
        x_old = np.linspace(0.0, 1.0, hseq.size)
        x_new = np.linspace(0.0, 1.0, n)
        h_r = np.interp(x_new, x_old, hseq)
        h_r = np.maximum(h_r, 0.0)
        if h_r.sum() > 0.01:
            h_r /= h_r.sum()
            candidates.append(h_r)

    # Cosine bump (centered)
    x = np.arange(n) / n
    for center in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            s = np.exp(-((x - center) ** 2) / (2 * width ** 2))
            s = np.maximum(s, 0.01)
            s /= s.sum()
            candidates.append(s)

    # Two-hump
    for c1, c2, w in [(0.25, 0.75, 0.1), (0.3, 0.7, 0.15), (0.2, 0.8, 0.2)]:
        s = np.exp(-((x - c1) ** 2) / (2 * w ** 2)) + np.exp(-((x - c2) ** 2) / (2 * w ** 2))
        s = np.maximum(s, 0.01)
        s /= s.sum()
        candidates.append(s)

    # Alternating high-low
    for ratio in [2.0, 3.0, 5.0, 10.0]:
        s = np.ones(n)
        s[::2] = ratio
        s /= s.sum()
        candidates.append(s)

    # Exponential decay from center
    for alpha in [2.0, 5.0, 10.0]:
        s = np.exp(-alpha * np.abs(x - 0.5))
        s = np.maximum(s, 0.01)
        s /= s.sum()
        candidates.append(s)

    # Random
    for _ in range(6):
        r = np.abs(rng.standard_normal(n)) + 0.1
        r /= r.sum()
        candidates.append(r)

    # Smoothed random (low-pass)
    for _ in range(4):
        r = np.abs(rng.standard_normal(n)) + 0.1
        # Simple moving average smoothing
        kernel = np.ones(7) / 7
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0.01)
        r /= r.sum()
        candidates.append(r)

    # Scale to unit sum
    out = []
    for s in candidates:
        s = np.maximum(s, 0.0)
        s_sum = s.sum()
        if s_sum > 0.01:
            out.append(s / s_sum)

    return out


def resample_sequence(s, new_n):
    """Resample a sequence to a new length using cubic interpolation."""
    old_n = len(s)
    if old_n == new_n:
        return s.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    # Use np.interp for linear (cubic needs scipy, keep it simple)
    s_new = np.interp(x_new, x_old, s)
    s_new = np.maximum(s_new, 0.0)
    s_sum = s_new.sum()
    if s_sum < 1e-15:
        return np.ones(new_n) / new_n
    return s_new / s_sum


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimizer: smoothed gradient descent + spectral shaping
    + LP refinement + random restarts. Tries multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    global_best_score = float('inf')
    global_best_seq = np.ones(256) / 256.0

    # Load previous best if available
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h / h.sum()
            print(f"Loaded height_sequence_1: length={h.size}, score={fast_score(hseq):.6f}")
    except Exception as e:
        print(f"No height_sequence_1 available: {e}")

    # Try multiple sequence lengths
    lengths = [256, 512, 1024, 2048, 4096]
    # Time allocation: more time for larger sequences
    time_allocation = [0.10, 0.15, 0.20, 0.25, 0.30]

    for idx, n in enumerate(lengths):
        if time.monotonic() > deadline - 30:
            break
        alloc_time = time_allocation[idx] * (budget_s - 30)
        phase_deadline = time.monotonic() + alloc_time

        print(f"\n{'='*50}")
        print(f"n={n}, time_alloc={alloc_time:.0f}s")
        print(f"Current global best: {global_best_score:.6f}")
        print(f"{'='*50}")

        # Generate starters
        starters = generate_starters(n, rng, hseq)

        # Score all starters
        scored = [(fast_score(s), i) for i, s in enumerate(starters)]
        scored.sort(key=lambda x: x[0])
        print(f"Generated {len(starters)} starters, best={scored[0][0]:.6f}")

        # Pick top candidates (up to 4)
        top_k = min(4, len(starters))
        selected = [starters[scored[i][1]] for i in range(top_k)]

        # Phase A: Gradient optimization on each candidate
        iters_per_cand = max(200, int(alloc_time * 0.6 / max(top_k, 1) / 0.001))
        # Cap iterations to avoid excessive time
        iters_per_cand = min(iters_per_cand, 20000)

        for ci, s0 in enumerate(selected):
            if time.monotonic() > phase_deadline - 20:
                break

            print(f"  Candidate {ci+1}/{top_k}, init score={fast_score(s0):.6f}")

            best_s, best_sc = run_gradient_optimize(
                s0, num_iters=iters_per_cand,
                beta_start=5.0, beta_end=1000.0,
                lr_start=0.08, lr_end=0.002,
                spectral_every=200, perturb_every=100,
                rng=rng, deadline=phase_deadline - 10
            )

            print(f"    After gradient: {best_sc:.6f}")

            # Phase B: LP refinement (only for n <= 1024 where LP is feasible)
            if n <= 1024 and time.monotonic() < phase_deadline - 15:
                best_s2, best_sc2 = lp_refine(
                    best_s, rng, max_iters=30, deadline=phase_deadline - 5
                )
                if best_sc2 < best_sc:
                    best_s = best_s2
                    best_sc = best_sc2
                    print(f"    After LP: {best_sc:.6f}")

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** NEW GLOBAL BEST: {global_best_score:.6f} (n={n})")

        # Phase C: Random restarts if time remains
        restart_deadline = phase_deadline - 5
        restart_count = 0
        while time.monotonic() < restart_deadline and restart_count < 3:
            # Blend best with random
            r = np.abs(rng.standard_normal(n)) + 0.1
            r /= r.sum()
            blend = 0.4 + 0.3 * rng.random()
            s0 = blend * global_best_seq[:n] if len(global_best_seq) == n else blend * resample_sequence(global_best_seq, n) + (1 - blend) * r
            s0 = np.maximum(s0, 0.0)
            s0 /= s0.sum()

            best_s, best_sc = run_gradient_optimize(
                s0, num_iters=max(500, iters_per_cand // 2),
                beta_start=10.0, beta_end=2000.0,
                lr_start=0.05, lr_end=0.001,
                spectral_every=150, perturb_every=80,
                rng=rng, deadline=restart_deadline
            )

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restart_count+1}: {global_best_score:.6f}")

            restart_count += 1

    # Final: one more round of gradient polishing on the global best
    if time.monotonic() < deadline - 10:
        print(f"\nFinal polishing phase...")
        final_n = len(global_best_seq)
        best_s, best_sc = run_gradient_optimize(
            global_best_seq, num_iters=10000,
            beta_start=100.0, beta_end=5000.0,
            lr_start=0.02, lr_end=0.001,
            spectral_every=500, perturb_every=300,
            rng=rng, deadline=deadline
        )
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s
            print(f"  Polishing improved to: {global_best_score:.6f}")

        # Final LP polish if feasible
        if final_n <= 1024 and time.monotonic() < deadline - 5:
            best_s2, best_sc2 = lp_refine(global_best_seq, rng, max_iters=20, deadline=deadline)
            if best_sc2 < global_best_score:
                global_best_score = best_sc2
                global_best_seq = best_s2
                print(f"  Final LP: {global_best_score:.6f}")

    # Verify with actual evaluator
    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    final_verified = evaluate_sequence(result)
    print(f"\n{'='*50}")
    print(f"FINAL: internal={global_best_score:.6f}, verified={final_verified:.6f}, length={len(result)}")
    print(f"{'='*50}")

    return result
