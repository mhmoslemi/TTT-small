Wait, I realize there's a potential issue with the symmetric LP. When I symmetrize `a` as `(a + a[::-1])/2`, this changes the auto-convolution. The symmetrization is a valid operation (it doesn't increase the objective due to convexity), but I need to be careful.

Actually, let me verify: if `a*` is optimal, then `(a* + a*[::-1])/2` is also optimal. This is because:
- `conv((a+b)/2, (a+b)/2) = (conv(a,a) + conv(a,b) + conv(b,a) + conv(b,b))/4`
- If a and b are both optimal with the same max auto-conv, then the max of the average is at most the original max (by convexity of max).
- Actually, this isn't quite right because the max of a sum is not necessarily ≤ the max of each term.

Let me think again. Let `a1` and `a2 = a1[::-1]` be two sequences. We want to show that `max_k conv((a1+a2)/2, (a1+a2)/2)[k] ≤ max(max conv(a1,a1), max conv(a2,a2))`.

`conv((a1+a2)/2, (a1+a2)/2)[k] = (1/4)(conv(a1,a1)[k] + conv(a1,a2)[k] + conv(a2,a1)[k] + conv(a2,a2)[k])`

Note that `conv(a2,a1) = conv(a1,a2)` and `conv(a2,a2)[k] = conv(a1,a1)[2(n-1)-k]`.

So `= (1/4)(conv(a1,a1)[k] + 2*conv(a1,a2)[k] + conv(a1,a1)[2(n-1)-k])`

And `conv(a1,a2)[k] = sum_j a1[j] * a2[k-j] = sum_j a1[j] * a1[2(n-1)-(k-j)] = sum_j a1[j] * a1[2(n-1)-k+j]`

This is the cross-correlation, which in general is different from the auto-convolution. So the max of the symmetrized version could be larger than the max of the original.

However, by the triangle inequality and properties of convolution, we have:
`|conv(a1,a2)[k]| ≤ sqrt(max conv(a1,a1) * max conv(a2,a2))` (Cauchy-Schwarz for convolutions)

Since `max conv(a2,a2) = max conv(a1,a1) = M`, we get `|conv(a1,a2)[k]| ≤ M`.

So `conv((a1+a2)/2, (a1+a2)/2)[k] ≤ (1/4)(M + 2M + M) = M`.

So the symmetrization doesn't increase the max auto-convolution! Great, so we can safely restrict to symmetric sequences.

"""
Hybrid symmetric LP + coordinate descent optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses the symmetry of auto-convolution
to halve the LP variables, with multiple initializations and adaptive
step sizes.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_symmetric(half):
    n_half = len(half)
    if n_half % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])


def build_A_sym(a, n_half):
    n = len(a)
    A = np.zeros((2 * n - 1, n_half))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                if idx < n_half:
                    A[k, idx] += a[j]
                else:
                    A[k, n - 1 - idx] += a[j]
    return A


def lp_solve_sym(a, M):
    n = len(a)
    n_half = (n + 1) // 2
    A = build_A_sym(a, n_half)
    c_obj = -np.ones(n_half)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n_half
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return make_symmetric(res.x)
    except Exception:
        pass
    return None


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def optimize_lps(a, deadline, rng):
    n = len(a)
    n_half = (n + 1) // 2
    a = symmetrize(a)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0
    best_step = 0.15

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve_sym(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        g = symmetrize(g)

        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            a_new = symmetrize(a_new)
            a_new = np.clip(a_new, 0, None)
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new
                a = a_new
                improved = True
                break

            stall = 0
            step = min(0.35, step * 1.03)
            best_step = step

def coordinate_descent(a, deadline, rng):
    n = len(a)
    n_half = (n + 1) // 2
    a = symmetrize(a)
    a = np.clip(a, 0, None)
    a = a / a.sum()
    best_score = compute_objective(a)
    best_a = a.copy()

    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        it += 1
        improved = False
        for i in rng.permutation(n_half):
            a_save = a[i]
            # Also the mirror position
            mirror = n - 1 - i
            best_val = a_save
            best_sc = np.inf
            # Try different values
            candidates = [0, a_save * 0.3, a_save * 0.6, a_save * 1.4, a_save * 2.0,
                          a_save + 0.005, a_save - 0.005]
            for cv in candidates:
                cv = max(0.0, cv)
                a[i] = cv
                a[mirror] = cv
                if a.sum() < 1e-10:
                    a[i] = a_save
                    a[mirror] = a_save
                    continue
                sc = compute_objective(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
            a[i] = best_val
            a[mirror] = best_val

        a = np.clip(a, 0, None)
        a = a / a.sum()
        sc = compute_objective(a)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1
            if no_improve > 5:
                break

    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def interpolate_to_length(a, new_n):
    if len(a) == new_n:

    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)

Symmetric LP-based optimization with coordinate descent refinement.
Exploits auto-convolution symmetry to halve LP variables, uses multiple
initializations and adaptive step sizes to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def make_symmetric(half):
    nh = len(half)
    if nh % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def build_A_sym(a):
    n = len(a)
    nh = (n + 1) // 2
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    if n % 2 == 0:
        A_sym = A[:, :n//2].copy()
        mirror_cols = A[:, n-1:n//2-1:-1]
        A_sym += mirror_cols
    else:
        A_sym = A[:, :nh].copy()
        mirror_cols = A[:, n-1:n//2:-1]
        A_sym[:, :n//2] += mirror_cols
    return A_sym


def lp_solve_sym(a, M):
    n = len(a)
    nh = (n + 1) // 2
    A = build_A_sym(a)
    c_obj = -np.ones(nh)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * nh
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs')
        if res.status == 0:
            return make_symmetric(res.x)
    except Exception:
        pass
    return None


def optimize_lps(a, deadline, rng):
    n = len(a)
    nh = (n + 1) // 2
    a = symmetrize(a)
    a = np.clip(a, 0, None)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve_sym(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            a_new = symmetrize(a_new)
            a_new = np.clip(a_new, 0, None)
            a_new = a_new / a_new.sum()
            sc = compute_objective(a_new)

                best_a = a_new

def coordinate_descent(a, deadline, rng):
    n = len(a)
    nh = (n + 1) // 2
    a = symmetrize(a).copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()
    best_score = compute_objective(a)
    best_a = a.copy()
    no_improve = 0

    while time.monotonic() < deadline:
        improved = False
        for i in rng.permutation(nh):
            a_save = a[i]
            mirror = n - 1 - i
            best_val = a_save
            best_sc = np.inf
            step_base = max(a_save * 0.1, 1e-4)
            for frac in [-0.3, -0.15, 0.15, 0.3, 0.5, -0.5, 1.0, -1.0]:

                cv = max(0.0, cv)

                    best_val = cv

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n) / (n - 1)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0

    c = (n - 1) / 2.0
