"""
Search for non-negative step-function sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact sub-gradient descent with momentum, LP-based refinement, and coordinate descent polishing.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(max(n, 2))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(max(n, 2))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    # Work with normalized sequence
    pn = a / s
    if n <= 512:
        c = np.convolve(pn, pn)
    else:
        L = next_power_of_two(2*n - 1)
        P = np.fft.rfft(pn, L)
        c = np.fft.irfft(P * P, L)[:2*n-1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    return float(2.0 * n * maxc)  # since sum(pn) = 1


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(max(n, 2))
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(max(n, 2))
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def exp_decay_normalized(n, gamma):
    n = int(max(n, 2))
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-float(gamma) * idx / n)
    return normalize_nonneg(a)


def cosine_taper(n, beta):
    n = int(max(n, 2))
    idx = np.arange(n, dtype=np.float64)
    t = idx / max(1, n - 1)
    a = np.cos(np.pi * t / 2.0) ** float(beta)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.ones(n)
    return normalize_nonneg(a)

    a = np.asarray(p, dtype=np.float64).ravel()

    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc)


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / max(bs, 1e-300))
    return [float(x) for x in b]


def try_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def compute_convolution(p, n=None):
    """Compute full convolution p*p."""
    a = np.asarray(p, dtype=np.float64).ravel()
    if n is None:
        n = a.size
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def topk_gradient(p, c, k_top=10):
    """
    Compute the smoothed gradient of max(c) using top-k averaging.
    Returns grad_p of shape (n,).
    """
    n = p.size
    maxc = float(c.max())
    if maxc <= 0:
        return np.zeros(n)
    # Find top-k indices
    if k_top >= c.size:
        k_top = c.size
    if k_top == 1:
        top_idx = [int(np.argmax(c))]
    else:
        top_idx = np.argsort(c)[::-1][:k_top].astype(int).tolist()
    # Average gradient over top-k
    # grad_k[j] = 2 * p[k-j] if 0 <= k-j < n
    grad = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * p[k - hi:k - lo + 1][::-1]
    grad /= float(len(top_idx))
    return grad


def subgrad_optimize(p, budget_s, k_top=5, lr_init=0.1, exponent=0.7):
    """
    Sub-gradient descent on log-parameterization with momentum.
    minimizes 2n * max(p*p) over p >= 0, sum(p) = 1.
    """
    p = normalize_nonneg(p)
    n = p.size
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    # Adaptive learning rate
    mu = 0.9  # momentum
    m = np.zeros(n, dtype=np.float64)
    t = 0

    best_p = p.copy()
    best_f = actual_score_fft(p)

    L_fft = next_power_of_two(2 * n - 1)

    start = time.monotonic()
    end = start + max(0.5, budget_s - 1.0)

    # Adaptive k_top: start broad, narrow down
    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Current k_top: decrease over time for sharper focus
        kt = max(1, int(k_top * (1.0 - 0.5 * frac) + 1))

        # Compute convolution
        Pg = np.fft.rfft(p, L_fft)
        c = np.fft.irfft(Pg * Pg, L_fft)[:2 * n - 1].real
        c = np.maximum(c, 0.0)

        f_now = float(2.0 * n * c.max())
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Compute smoothed gradient
        grad_p = topk_gradient(p, c, kt)
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        if not np.all(np.isfinite(grad_x)):
            break

        # Scale gradient
        gnorm = float(np.sqrt(np.dot(grad_x, grad_x)))
        if gnorm < 1e-15:
            break

        # Learning rate schedule: decay over time
        lr = lr_init * (1.0 + t) ** (-exponent)
        lr = min(lr, 0.5)

        # Momentum
        m = mu * m + (1.0 - mu) * grad_x / gnorm
        step = lr * m

        # Update x
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break

        # Stabilize
        if t % 100 == 0:
            x = np.clip(x, -50.0, 50.0)
            x -= x.mean()

        # Convert back to p
        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

        # Check for stagnation
        if t % 200 == 0 and t > 400:
            # Try a small perturbation to escape plateau
            rng = np.random.default_rng(t)
            noise = 0.01 * rng.normal(0.0, 1.0, n)
            x += noise
            x -= x.mean()

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def softmax_optimize(p, budget_s, alpha_init=20.0):
    """
    Smoothed-max optimization using adaptive alpha softmax.
    """
    p = normalize_nonneg(p)
    n = p.size
    budget_s = float(budget_s)
    if budget_s < 2.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    best_p = p.copy()
    best_f = actual_score_fft(p)

    L_fft = next_power_of_two(2 * n - 1)

    # Schedule alpha: start moderate, increase to very sharp
    alpha_end = 200.0
    lr_init = 0.05 if n < 512 else 0.03

    start = time.monotonic()
    end = start + max(0.5, budget_s - 2.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Alpha schedule: ramp up
        alpha = alpha_init + (alpha_end - alpha_init) * (frac ** 0.5)

        Pg = np.fft.rfft(p, L_fft)
        c = np.fft.irfft(Pg * Pg, L_fft)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0:
            break

        f_now = float(2.0 * n * maxc)
        if f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        # Softmax weights
        z = alpha * (c / maxc - 1.0)
        z = z - z.max()
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # Gradient via correlation
        W = np.fft.rfft(w[::-1], L_fft)
        Cfull = np.fft.irfft(Pg * W, L_fft)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break

        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)

        lr = lr_init * (1.0 + t * 0.01) ** (-0.5)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)

        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break

        if t % 50 == 0:
            x = np.clip(x, -60.0, 60.0)
            x -= x.mean()

        x_shifted = x - x.max()
        e = np.exp(x_shifted)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

    final_f = actual_score_fft(best_p)
    if final_f < best_f:
        best_f = final_f
    return best_p, best_f


def lp_refine(p, budget_s):
    """
    LP-based cutting plane refinement.
    Maximizes sum(g) s.t. f*g <= M, g >= 0, where f = p*sqrt(2n), M = max(f*f).
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = normalize_nonneg(p)
    n = p.size
    budget_s = float(budget_s)
    if budget_s < 15.0 or n < 4:
        return p, actual_score_fft(p)

    S0 = np.sqrt(2.0 * n)
    f = p * S0
    c = compute_convolution(f, n)
    M = float(c.max())
    if M < 1e-12:
        return p, actual_score_fft(p)

    best_f = f.copy()
    best_score = float(2.0 * n * M / (float(f.sum()) ** 2))

    # Manage constraint rows
    max_rows = min(800, max(100, 2000000 // n))
    rows = list(range(0, 2 * n - 1, max(1, (2 * n - 1) // max_rows)))
    row_set = set(rows)

    end = time.monotonic() + max(1.0, budget_s - 3.0)
    it = 0

    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        n_rows = rows_arr.size

        # Build constraint matrix
        A = np.zeros((n_rows, n), dtype=np.float64)
        for ii in range(n_rows):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(n_rows, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 0.5)
        tl = max(3.0, min(60.0, rem * 0.4))

        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None),
                method='highs',
                options={'time_limit': float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full violations
        conv = compute_convolution(f, n)
        if n > 512:
            L2 = next_power_of_two(2 * n - 1)
            Pf = np.fft.rfft(f, L2)
            Pg = np.fft.rfft(g, L2)
            fgc = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        else:
            fgc = np.convolve(f, g)
        fgc = np.maximum(fgc, 0.0)
        violations = fgc - M
        tol = 1e-8 * M + 1e-12

        if np.any(violations > tol):
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                added = 0
                for ix in order[:min(200, order.size)]:
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                        if added >= 100:
                            break
                if added == 0:
                    break
            else:
                break
            continue

        # Scale g to match f's sum
        g = g * (S0 / sg)

        # Try mixing
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.3, 0.2, 0.1, 0.05):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            cn = compute_convolution(fn, n)
            mn = float(cn.max())
            sc = float(2.0 * n * mn / (sn / S0 * S0) ** 2)
            # Actually just use the raw formula
            sc = float(2.0 * n * mn)  # since sum(fn) = S0
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None:
            break
        if accepted_score >= float(2.0 * n * M):
            break

        f = accepted
        c = compute_convolution(f, n)
        M = float(c.max())
        cur_score = float(2.0 * n * M / (S0 ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    return normalize_nonneg(best_f), best_score


def coordinate_polish(p, budget_s):
    """
    Coordinate descent: for each coordinate, optimize its value given others fixed.
    """
    p = normalize_nonneg(p).copy()
    n = p.size
    budget_s = float(budget_s)
    if budget_s < 5.0:
        return p, actual_score_fft(p)

    best_f = actual_score_fft(p)
    best_p = p.copy()

    end = time.monotonic() + max(1.0, budget_s - 2.0)
    rng = np.random.default_rng(42)

    while time.monotonic() < end:
        # Process coordinates in random order
        order = rng.permutation(n)
        improved = 0
        for j in order:
            j = int(j)
            if not np.isfinite(p[j]) or p[j] < 0:
                p[j] = 0.0
                continue

            # Compute objective as function of p[j] with others fixed
            # sum = S + p_j where S = sum of others
            # conv_k includes p_j * p_{k-j} terms
            S_others = float(p.sum() - p[j])
            if S_others < 1e-15:
                continue

            # The convolution contribution from coordinate j:
            # For each k, conv_k += p_j * p_{k-j}
            # We need to find the optimal p_j that minimizes 2n * max(conv) / (S_others + p_j)^2

            # Get current convolution without j's contribution
            # This is expensive to do exactly, so use approximation:
            # Just try a few candidate values

            pj = p[j]
            candidates = [0.0, pj * 0.5, pj * 0.8, pj, pj * 1.2, pj * 1.5, pj * 2.0]
            candidates = [max(0.0, min(10.0, c)) for c in candidates]

            best_val = None
            best_score = np.inf
            for cv in candidates:
                p[j] = cv
                p = np.clip(p, 0.0, None)
                s = float(p.sum())
                if s < 1e-15:
                    p[j] = pj
                    break
                pn = p / s
                c = compute_convolution(pn, n)
                sc = float(2.0 * n * c.max())
                if sc < best_score:
                    best_score = sc
                    best_val = cv

            if best_val is not None and best_score < best_f:
                p[j] = best_val
                best_f = best_score
                best_p = p.copy()
                improved += 1
            else:
                p[j] = pj

        if improved == 0:
            break

    return best_p, best_f


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = actual_score_fft(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print("  New best: score=%.6f, n=%d, t=%.0fs" % (score, a.size, time.monotonic() - t0))
        n = a.size
        old = best_by_n.get(n)
        if old is None or score < old:
            best_by_n[n] = float(score)
            if n not in [k for k in best_by_n]:
                pass
            # Store the sequence
            if 'best_seqs' not in globals():
                pass
            best_seqs[n] = a.copy()

    best_seqs = {}

    # Phase 1: Diverse initialization (first ~120s)
    print("Phase 1: Diverse initialization")
    lengths = [64, 96, 128, 160, 192, 256, 320, 384, 512, 640, 768, 1024]

    inits = []
    for n in lengths:
        inits.append(('greedy', n, greedy_normalized(n)))
        inits.append(('p045', n, power_normalized(n, 0.45)))
        inits.append(('p055', n, power_normalized(n, 0.55)))
        inits.append(('p065', n, power_normalized(n, 0.65)))
        inits.append(('exp05', n, exp_decay_normalized(n, 0.5)))
        inits.append(('exp10', n, exp_decay_normalized(n, 1.0)))
        inits.append(('cos2', n, cosine_taper(n, 2.0)))
        inits.append(('mix_g1', n, mix_seqs(greedy_normalized(n), np.ones(n), 0.3)))
        inits.append(('mix_g2', n, mix_seqs(greedy_normalized(n), np.ones(n), 0.5)))
        # Perturbed greedy
        noise = np.exp(rng.normal(0, 0.1, n))
        inits.append(('pert', n, normalize_nonneg(greedy_normalized(n) * noise)))

    for name, n, p in inits:
        sc = actual_score_fft(p)
        record(p, sc)

    # Check height_sequence_1
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0:
                sc = actual_score_fft(arr)
                record(arr, sc)
                print("  Loaded height_sequence_1: n=%d, score=%.6f" % (arr.size, sc))
        except Exception:
            pass

    print("  Best after init: score=%.6f" % best_score)

    # Phase 2: Focused optimization (~1200s)
    print("Phase 2: Focused optimization")

    # Get top candidates
    items = [(sc, n, best_seqs[n]) for n, sc in best_by_n.items() if n in best_seqs]
    items.sort(key=lambda x: x[0])

    #_OPTIMIZE top 5
    top_candidates = items[:5]
    if best_p is not None and best_p.size not in [c[1] for c in top_candidates]:
        top_candidates.append((best_score, best_p.size, best_p))

    for rank, (sc, n, p) in enumerate(top_candidates[:4]):
        rem = deadline - time.monotonic()
        if rem < 60.0:
            break

        # Allocate time: more for better candidates
        if rank == 0:
            alloc = min(400.0, rem * 0.35)
        elif rank == 1:
            alloc = min(300.0, rem * 0.25)
        elif rank == 2:
            alloc = min(250.0, rem * 0.2)
        else:
            alloc = min(200.0, rem * 0.15)

        alloc = max(30.0, alloc)
        print("  Optimizing candidate %d (n=%d, score=%.6f) for %.0fs" % (rank, n, sc, alloc))

        # Try multiple optimization methods
        if rank == 0:
            # Best candidate: try both methods
            half = alloc * 0.45
            p1, f1 = softmax_optimize(p.copy(), half, alpha_init=15.0)
            record(p1, f1)

            rem2 = deadline - time.monotonic()
            if rem2 > 30:
                p2, f2 = subgrad_optimize(p.copy(), min(alloc * 0.35, rem2 - 5), k_top=5)
                record(p2, f2)

            rem2 = deadline - time.monotonic()
            if rem2 > 30:
                # Try from a perturbed version
                p3 = normalize_nonneg(p * np.exp(rng.normal(0, 0.05, n)))
                p4, f4 = softmax_optimize(p3, min(alloc * 0.2, rem2 - 5), alpha_init=10.0)
                record(p4, f4)
        else:
            p_out, f_out = softmax_optimize(p.copy(), alloc, alpha_init=20.0)
            record(p_out, f_out)

            rem2 = deadline - time.monotonic()
            if rem2 > alloc * 0.3:
                p_out2, f_out2 = subgrad_optimize(p_out.copy(), min(alloc * 0.4, rem2 - 5), k_top=3)
                record(p_out2, f_out2)

    # Phase 2b: LP refinement
    print("Phase 2b: LP refinement")
    if _scipy_opt is not None and best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 80:
            lp_n = best_p.size
            alloc_lp = min(200.0, rem * 0.4)
            print("  LP refining n=%d for %.0fs" % (lp_n, alloc_lp))
            p_lp, f_lp = lp_refine(best_p.copy(), alloc_lp)
            record(p_lp, f_lp)

            # Also try LP on a different length
            rem2 = deadline - time.monotonic()
            if rem2 > 80:
                for n_try in [256, 384, 512]:
                    if n_try == lp_n:
                        continue
                    if n_try in best_seqs:
                        p_try = best_seqs[n_try].copy()
                    else:
                        p_try = greedy_normalized(n_try)
                    alloc_try = min(100.0, rem2 * 0.3)
                    if alloc_try < 30:
                        break
                    p_lp2, f_lp2 = lp_refine(p_try, alloc_try)
                    record(p_lp2, f_lp2)
                    rem2 = deadline - time.monotonic()
                    if rem2 < 60:
                        break

    # Phase 3: Additional optimization rounds
    print("Phase 3: Additional optimization")
    rem = deadline - time.monotonic()
    if best_p is not None and rem > 60:
        # Re-optimize the best with fresh start
        for attempt in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            sigma = 0.03 + 0.02 * attempt
            p0 = normalize_nonneg(best_p * np.exp(rng.normal(0, sigma, best_p.size)))
            alloc = min(150.0, rem * 0.3)
            p_out, f_out = softmax_optimize(p0, alloc, alpha_init=30.0)
            record(p_out, f_out)

            # Also try sub-gradient
            rem = deadline - time.monotonic()
            if rem > 40:
                p_out2, f_out2 = subgrad_optimize(p_out.copy(), min(80.0, rem * 0.3), k_top=3)
                record(p_out2, f_out2)

    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate polish")
    rem = deadline - time.monotonic()
    if best_p is not None and rem > 30:
        n_bd = best_p.size
        if n_bd <= 512:
            alloc_cd = min(rem * 0.4, 120.0)
            p_cd, f_cd = coordinate_polish(best_p.copy(), alloc_cd)
            record(p_cd, f_cd)

    # Phase 5: Final multi-restart
    print("Phase 5: Final restarts")
    rem = deadline - time.monotonic()
    if best_p is not None and rem > 50:
        for i in range(5):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            n_best = best_p.size
            p0 = normalize_nonneg(best_p * np.exp(rng.normal(0, 0.08, n_best)))
            alloc = min(40.0, rem * 0.35)
            p_out, f_out = softmax_optimize(p0, alloc, alpha_init=50.0)
            record(p_out, f_out)

    # Final validation
    print("Final: best_score=%.6f, n=%d" % (best_score, best_p.size if best_p is not None else 0))

    if best_p is None:
        best_p = greedy_normalized(256)
        best_score = actual_score_fft(best_p)

    # Validate with the actual evaluator if possible
    candidates_to_check = []
    if best_p is not None:
        candidates_to_check.append(best_p)
    for n, sc in sorted(best_by_n.items(), key=lambda x: x[1])[:3]:
        if n in best_seqs:
            candidates_to_check.append(best_seqs[n])

    for cand in candidates_to_check[:5]:
        if deadline - time.monotonic() < 10:
            break
        try:
            seq = scale_output(cand, 10.0)
            gs = try_eval(seq)
            if np.isfinite(gs) and gs < best_score:
                print("  Global eval improved: %.6f" % gs)
                best_score = gs
                best_p = np.asarray(cand, dtype=np.float64).ravel()
        except Exception:
            pass

    # Also check the raw best_p scaled differently
    for ts in [5.0, 10.0, 20.0]:
        if deadline - time.monotonic() < 5:
            break
        try:
            seq = scale_output(best_p, ts)
            gs = try_eval(seq)
            if np.isfinite(gs) and gs < best_score:
                best_score = gs
        except Exception:
            pass

    final_seq = scale_output(best_p, 10.0)
    print("Done. Final score=%.6f, length=%d, time=%.0fs" % (
        best_score, best_p.size, time.monotonic() - t0))
    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""
    try:
        result = main_search(int(seed), float(budget_s))
        if result is not None and len(result) > 0:
            return result
        else:
            p = greedy_normalized(256)
            return scale_output(p, 10.0)
    except Exception as e:
        print("Error in propose_candidate: %s" % str(e))
        p = greedy_normalized(256)
        return scale_output(p, 10.0)
