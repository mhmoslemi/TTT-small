    cols = j

        pass
    return None

        improved = False

        # Choose LP variant
        if stall % 40 == 30:
            use_min_peak = not use_min_peak

        if not use_min_peak:
            g, obj_val = solve_lp_max_volume(a, M)
            if g is not None:
                g = g / (g.sum() + 1e-15)
                # Check if g is meaningfully different from a
                diff = np.abs(g - a).max()
                if diff > 1e-6:
                    for shrink in [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]:
                        s = step * shrink
                        a_new = (1.0 - s) * a + s * g
                        a_new = np.clip(a_new, 0.0, None)
                        if a_new.sum() < 1e-10:
                            continue
                        a_new = a_new / a_new.sum()
                        sc = compute_score(a_new)
                        if sc < best_score - 1e-13:
                            best_a = a_new.copy()
                            best_score = sc
                            a = a_new
                            improved = True
                            break
        else:
            g, t_val = solve_lp_min_peak(a)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g / g_sum
                    diff = np.abs(g - a).max()
                    if diff > 1e-6:
                        for s in [0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
                            a_new = (1.0 - s) * a + s * g
                            a_new = np.clip(a_new, 0.0, None)
                            if a_new.sum() < 1e-10:
                                continue
                            a_new = a_new / a_new.sum()
                            sc = compute_score(a_new)
                            if sc < best_score - 1e-13:
                                best_a = a_new.copy()
                                best_score = sc
                                a = a_new
                                improved = True
                                break
                            elif sc < best_score:
                                # Accept smaller improvement
                                if s <= 0.1:
                                    best_a = a_new.copy()
                                    best_score = sc
                                    a = a_new
                                    improved = True
                                    break

        if improved:
            stall = 0
            stall_since_improve = 0
            step = min(0.35, step * 1.04)
        else:
            stall += 1
            stall_since_improve += 1
            step = max(0.003, step * 0.85)

            # Subgradient step when moderately stuck
            if stall == 15 or (stall >= 20 and stall % 10 == 0):
                lr = 0.02 + 0.01 * min(stall // 10, 5)
                a_new = subgradient_step(a, lr, rng)
                sc = compute_score(a_new)
                if sc < best_score:
                    best_a = a_new.copy()
                    best_score = sc
                    a = a_new
                    stall = 0
                    step = 0.1
                elif rng.random() < 0.2:
                    a = a_new

            # Random perturbation when very stuck
            if stall >= 25:
                noise_scale = 0.03 + 0.01 * min((stall - 25) // 10, 10)
                noise = rng.normal(0, noise_scale, n)
                a_new = a * (1.0 + noise)
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    sc = compute_score(a_new)
                    if sc < best_score:
                        best_a = a_new.copy()
                        best_score = sc
                        a = a_new
                        stall = 0
                        step = 0.12
                    elif rng.random() < 0.25 and stall < 60:
                        a = a_new
                        stall = max(0, stall - 10)
                    else:
                        a = best_a.copy()

            # Major reset when extremely stuck
            if stall > 100:
                # Blend with a smoother version
                from scipy.ndimage import uniform_filter1d
                a_smooth = uniform_filter1d(a, size=max(3, n // 32))
                a_new = 0.5 * best_a + 0.5 * a_smooth
                a_new = np.clip(a_new, 0.0, None)
                if a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    a = a_new
                    stall = 0
                    step = 0.1

            if stall > 200:
                break

        if verbose and it % 200 == 0:
            print(f"    iter {it}, score={best_score:.8f}, step={step:.4f}, stall={stall}, use_min_peak={use_min_peak}")

    return best_a, best_score


def make_hann(n):
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))


def make_gaussian(n, sigma=0.3):
    center = (n - 1) / 2.0
    return np.exp(-0.5 * ((np.arange(n) - center) / (sigma * n)) ** 2)


def make_symmetric_bump(n, width=0.6):
    """Symmetric bump peaked at center."""
    center = (n - 1) / 2.0
    x = np.abs(np.arange(n) - center) / (width * center)
    return np.exp(-x * x)


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative optimization
    with sparse matrices, dual LP formulations, subgradient refinement,
    and adaptive step control.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p
            parent_score = compute_score(p)
            print(f"Parent: n={p.size}, score={parent_score:.8f}")
    except Exception:
        pass

    # Build search plan
    # Primary: parent (most time)
    # Secondary: other inits at similar and different sizes
    plan = []  # (init_array, label, time_fraction)

    if parent is not None:
        plan.append((parent, "parent", 0.50))
        n_p = len(parent)
        if n_p > 512:
            plan.append((parent[:512], "parent_512", 0.10))
        if n_p < 1024:
            plan.append((make_hann(1024), "hann_1024", 0.10))
        else:
            plan.append((parent[:1024], "parent_1024", 0.10))
    else:
        plan.append((make_hann(512), "hann_512", 0.30))
        plan.append((make_gaussian(512, 0.3), "gauss_512", 0.20))
        plan.append((make_hann(1024), "hann_1024", 0.20))
        plan.append((make_gaussian(1024, 0.25), "gauss_1024", 0.15))

    # Add a few more diverse inits
    plan.append((make_symmetric_bump(512, 0.5), "bump_512", 0.05))
    plan.append((make_hann(2048), "hann_2048", 0.05))

    # Normalize time fractions
    total_frac = sum(f for _, _, f in plan)
    for i in range(len(plan)):
        plan[i] = (plan[i][0], plan[i][1], plan[i][2] / total_frac)

    available_time = budget_s - 50.0
    print(f"Plan: {len(plan)} configs, available_time={available_time:.0f}s")

    elapsed = 0.0
    for idx, (init_a, label, frac) in enumerate(plan):
        if time.monotonic() >= deadline:
            break

        n = len(init_a)
        alloc_time = available_time * frac
        config_deadline = min(time.monotonic() + alloc_time, deadline)

        remaining = config_deadline - time.monotonic()
        if remaining < 20:
            continue

        print(f"[{idx+1}/{len(plan)}] {label} (n={n}), alloc~{alloc_time:.0f}s")

        a_opt, score = optimize_lp(init_a, config_deadline, rng, verbose=(frac > 0.1))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.8f} ({label}, n={n})")
        else:
            print(f"  => score={score:.8f} (best: {best_score:.8f})")

    # Final polish on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"Final polish (remaining {deadline - time.monotonic():.0f}s)...")
        best_seq, best_score = optimize_lp(best_seq, deadline, rng, verbose=True)
        print(f"  After polish: score={best_score:.8f}")

    if best_seq is None:
        best_seq = np.ones(256)
        best_score = 2.0

    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    elapsed_total = time.monotonic() - start_time
    print(f"DONE in {elapsed_total:.1f}s. Final score={best_score:.8f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
