"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines fast multiplicative updates with LP-based peak minimization,
using FFT acceleration, multiple diverse initializations, and adaptive
step sizes to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def compute_score(a):
    """Compute score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1
    while size < n_out:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def multiplicative_optimize(a, deadline, rng, verbose=False):
    """
    Fast multiplicative (exponential) update to reduce convolution peaks.
    Uses pressure-based deflation with adaptive step size.
    """
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64).copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    eta = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a, a)
        M = c.max()
        threshold = M * (1.0 - 0.15)
        peak_indices = np.where(c >= threshold)[0]

        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c)])

        # Compute pressure using correlation
        # p[j] = sum_{k in peaks} a[k-j] for valid k-j
        # This is cross-correlation of a with peak indicator
        b = np.zeros(2 * n - 1)
        b[peak_indices] = 1.0
        # p[j] = sum_k b[k] * a[k-j] = correlate(b, a)
        # np.correlate(b, a)[j] = sum_k b[k] * a[k-j] (with appropriate indexing)
        p = np.correlate(b, a, mode='full')
        # The correlation output has length (2n-1) + n - 1 = 3n-2
        # We want p[j] for j in 0..n-1
        # correlate(b, a, 'full')[k] = sum_i b[i] * a[i + k - (n-1)]
        # We need: p[j] = sum_k b[k] * a[k-j]
        # Let m = k-j, so k = m+j: p[j] = sum_m b[m+j] * a[m]
        # = (b shifted by j) correlated with a
        # Alternatively: p[j] = np.correlate(b, a, 'full')[n-1-j] ... let me just compute directly for safety
        if n <= 4096:
            p = np.zeros(n)
            for k in peak_indices:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)
                if j_max >= j_min:
                    p[j_min:j_max + 1] += a[k - j_max:k - j_min + 1][::-1]
            p /= len(peak_indices)
        else:
            p = np.zeros(n)
            for k in peak_indices:
                j_min = max(0, k - n + 1)
                j_max = min(n - 1, k)

            p /= len(peak_indices)

        p_normalized = p / (p.max() + 1e-15)
        a_new = a * np.exp(-eta * p_normalized)
        a_new = np.clip(a_new, 1e-15, None)

            break
        a_new /= a_new.sum()

        sc = compute_score(a_new)

            best_score = sc

            stall = 0

def lp_peak_minimize(a, deadline, rng, verbose=False):
    """
    LP-based optimization: minimize the linearized new peak.
    LP: min t s.t. c + 2*A@g <= t, sum(g)=1, g>=0
    Then move towards g with line search.
    """
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64).copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    stall = 0

    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a, a)
        M = c.max()

        # Build LP: min t_new
        # s.t. c[k] + 2 * (A @ g)[k] <= t_new for k=0..2n-2
        #      sum(g) = 1
        #      g >= 0
        # Variables: [g_0, ..., g_{n-1}, t_new]
        A = build_conv_matrix(a)
        A_lp = np.hstack([2.0 * A, np.zeros((2 * n - 1, 1))])
        A_lp = A_lp - np.eye(2 * n - 1, 2 * n)[:, :1]  # This doesn't work simply

        # Let me reformulate. Standard LP form: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
        # Variables: x = [g_0,...,g_{n-1}, t]
        # Objective: minimize t -> c_obj = [0,...,0, 1]
        # Constraints:
        #   (A_conv @ g) + c_vec - t <= 0  =>  [2*A_conv, -I] @ [g; t] <= -c_vec
        #   sum(g) = 1  =>  [1,...,1, 0] @ [g; t] = 1
        #   g >= 0, t free (or t >= 0)

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t

        # Inequality: 2*A_conv @ g - t <= -c_vec
        # i.e., [2*A_conv | -1] @ [g; t] <= -c_vec
        A_ub = np.zeros((2 * n - 1, n + 1))
        A_ub[:, :n] = 2.0 * A
        A_ub[:, n] = -1.0
        b_ub = -c  # c[k] + 2*(A@g)[k] - t <= 0 => 2*(A@g)[k] - t <= -c[k]

        # Equality: sum(g) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                # Try the alternative formulation
                g = _lp_alternative(a, c, M)
            else:
                g = res.x[:n]
                g = np.clip(g, 0, None)
                if g.sum() < 1e-10:
                    g = a.copy()
                else:
                    g /= g.sum()
        except Exception:
            g = _lp_alternative(a, c, M)

        if g is None:
            break

        # Line search: try different step sizes
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.15, 0.1, 0.05, 0.02]:
            a_new = (1.0 - alpha) * best_a + alpha * g
            a_new = np.clip(a_new, 1e-15, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Perturb to escape
                a = best_a * (1.0 + 0.05 * rng.normal(0, 1, n))
                a = np.clip(a, 1e-15, None)
                a /= a.sum()
                stall = 0
            if stall > 50:
                break

        if verbose and it % 50 == 0:
            print(f"      lp iter {it}, score={best_score:.6f}")

    return best_a, best_score


def lp_alternative(a, c, M):
    """Alternative LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x.sum() > 1e-10:
            return res.x / res.x.sum()
    except Exception:
        pass
    return None


def _lp_alternative(a, c, M):
    """Wrapper to avoid closure issues."""
    return lp_alternative(a, c, M)


def subgradient_optimize(a, deadline, rng, verbose=False):
    """
    Subgradient method with momentum for the peak minimization.
    Uses the analytical subgradient of max conv(a,a).
    """
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64).copy()

    best_a = a.copy()
    best_score = compute_score(a)

    # Momentum
    y = a.copy()
    t = 1.0
    beta_init = 1.0
    stall = 0
    it = 0
    lr = 0.05

    while time.monotonic() < deadline:
        it += 1
        t += 1.0
        beta = (1.0 + beta_init) / (2.0 + beta_init + t)

        # Compute subgradient at y
        c_y = fft_convolve(y, y)
        M_y = c_y.max()
        # Use top positions within 90% of max
        threshold = 0.85 * M_y
        peak_mask = c_y >= threshold
        peak_indices = np.where(peak_mask)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(c_y)])

        # Compute subgradient: sg[j] = (1/|S|) * sum_{k in S} 2*y[k-j]
        sg = np.zeros(n)
        for k in peak_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_max >= j_min:
                sg[j_min:j_max + 1] += 2.0 * y[k - j_max:k - j_min + 1][::-1]
        sg /= len(peak_indices)

        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg = sg / sg_norm

        # Nesterov update
        y_new = (1.0 + beta) * a - beta * y - lr / np.sqrt(t) * sg
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = _project_simplex(y_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_a = a_new.copy()

                a /= a.sum()
                y = a.copy()
                stall = 0
            if stall > 200:
                break

        y = a_new
        a = a_new

        if verbose and it % 500 == 0:
            print(f"      subgrad iter {it}, score={best_score:.6f}")

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _project_simplex(v):
    """Top-level wrapper for simplex projection."""
    return project_simplex(v)


def make_raised_cosine(n, width_frac=1.0):
    """Raised cosine window with adjustable width."""
    x = np.linspace(0, np.pi / width_frac, n)
    return 0.5 * (1.0 + np.cos(2.0 * x - np.pi))


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_linear_taper(n):
    """Linear taper (inspired by n=2 optimal)."""
    return np.linspace(1.0, 0.1, n)


def make_symmetric_taper(n):
    """Symmetric bell-shaped taper."""
    x = np.linspace(-1, 1, n)
    return np.exp(-x ** 2 * 2.0)


def make_block_pattern(n, n_blocks=4):
    """Alternating block pattern."""
    block_size = max(1, n // n_blocks)
    a = np.zeros(n)
    for i in range(n_blocks):

        end = min(n, start + block_size)
        if i % 2 == 0:
            a[start:end] = 1.0
        else:
            a[start:end] = 0.3
    return a
