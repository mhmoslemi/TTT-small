```python
def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_sq(a, m=None):

    n = a.size

        m = 1
        while m < 2*n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2*n-1]
```

```python
"""
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for a >= 0 using:
- Log-parameterization (softmax) with L-BFGS-B
- Adaptive temperature annealing on log-sum-exp smoothed max
- Peak-shaving polish
- Multiple sequence lengths and diverse initializations
- Alternating peak-shave as a secondary optimizer

import time
import math
import numpy as np
from scipy.optimize import minimize


def _proj_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_conv_sq(a, m=None):
    """Linear convolution a*a via FFT."""
    n = a.size
    if m is None:
        m = 1
        while m < 2 * n - 1:
            m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def _smoothed_max_value(c, t):
    c_max = c.max()
    shifted = (c - c_max) / t
    return c_max + t * np.log(np.sum(np.exp(shifted)))


def _smoothed_max_weights(c, t):
    c_max = c.max()
    shifted = (c - c_max) / t
    w = np.exp(shifted)
    return w / w.sum()


def _lbbf_obj_and_grad(x, n, t, m):
    """Objective and gradient in log-parametrization."""
    a = _softmax(x)
    c = _fft_conv_sq(a, m)
    val = _smoothed_max_value(c, t)
    obj = 2.0 * n * val

    w = _smoothed_max_weights(c, t)
    # Gradient w.r.t. a: grad_a[i] = 2n * sum_k w[k] * a[k-i]
    grad_a = np.zeros(n)
    # Only iterate over significant weights
    threshold = w.max() * 1e-10
    sig = w > threshold
    ks = np.where(sig)[0]
    for k in ks:
        k = int(k)
        wk = w[k]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        src = k - idx
        grad_a[idx] += wk * a[src]
    grad_a *= 2.0 * n

    # Chain rule through softmax: grad_x = grad_a - a*(a^T grad_a)
    grad_x = grad_a - a * np.dot(a, grad_a)
    return obj, grad_x


def _optimize_lbbf(a_init, t_end, n_init_t=0.3):
    """Optimize using L-BFGS-B with log-parametrization and temperature annealing."""
    n = a_init.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1

    a = np.clip(a_init, 1e-15, None)
    a = a / a.sum()
    x0 = np.log(a)

    # Temperature schedule: start smooth, get sharp
    stages = [
        (0.3, 0.15),   # stage 1: t from 0.3*max_c to 0.15*max_c
        (0.15, 0.05),  # stage 2

    best_a = a.copy()
    best_obj = 2.0 * n * _fft_conv_sq(a).max()

def _peak_shave(a, t_end, n_steps=200):
    """Polish by directly reducing the max (and near-max) lags."""
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1

    best_a = a.copy()
    best_obj = 2.0 * n * _fft_conv_sq(a).max()
    rng = np.random.default_rng(12345)

    for it in range(n_steps):
        if time.monotonic() > t_end - 0.5:
            break

        c = _fft_conv_sq(a, m)
        obj = 2.0 * n * c.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Find top K lags
        K = min(5, 2 * n - 1)
        top_ks = np.argsort(-c)[:K]

        # Build gradient from top lags (equal weight)
        g = np.zeros(n)
        for k in top_ks:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo > hi:
                continue
            idx = np.arange(lo, hi + 1)
            src = k - idx
            g[idx] += a[src]

        gm = g.max()
        if gm < 1e-15:
            break

        # Step: reduce a where g is large
        alpha = 0.05 / (1.0 + it * 0.1)
        a_new = a - alpha * g / gm
        a = _proj_simplex(a_new, 1.0)

        # Occasionally add small noise to escape local minima
        if it % 50 == 49:
            a = a * (1.0 + 0.01 * rng.standard_normal(n))
            a = _proj_simplex(a, 1.0)

    return best_a, best_obj

            best_obj = obj
            best_a = a.copy()

        # Pick the top lag cyclically to avoid over-shaving one peak
        K = min(7, 2 * n - 1)
        top_ks = np.argsort(-c)[:K]
        k_target = top_ks[it % K]
        k_target = int(k_target)

        lo = max(0, k_target - n + 1)
        hi = min(k_target, n - 1)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1)
        src = k_target - idx

        g = np.zeros(n)
        g[idx] = a[src]

        gm = g.max()
        if gm < 1e-15:
            break
        alpha = 0.03
        a_new = a - alpha * g / gm
        a = _proj_simplex(a_new, 1.0)

    return best_a, best_obj


def _make_inits(n, base_seq=None):
    """Create diverse initial sequences of length n, all normalized to sum=1."""
    inits = []

    # From base sequence (interpolated)
    if base_seq is not None and len(base_seq) > 1:
        b = np.clip(base_seq, 0, None)
        x_old = np.linspace(0, 1, len(b))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, b)
        a = np.clip(a, 0, None)
        if a.sum() > 0:
            inits.append(a / a.sum())

    if not inits:
        inits.append(np.ones(n) / n)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian bump (centered)
    x = np.linspace(-1, 1, n)
    for sigma in [0.5, 0.7, 1.0]:
        g = np.exp(-x ** 2 / (2 * sigma ** 2))
        inits.append(g / g.sum())

    # Power-law window
    x = np.linspace(-1, 1, n)
    for p in [1, 2, 3]:
        w = (1 - x ** 2) ** p
        w = np.clip(w, 1e-10, None)
        inits.append(w / w.sum())

    # Ramp (asymmetric)
    for power in [1, 2, 3]:
        r = np.linspace(0.01, 1.0, n) ** power
        inits.append(r / r.sum())

    # Two bumps
    x = np.arange(n) / n
    for sep in [0.25, 0.3]:
        tb = np.exp(-((x - sep) / 0.08) ** 2) + np.exp(-((x - 1 + sep) / 0.08) ** 2)
        inits.append(tb / tb.sum())

    # U-shaped
    x = np.linspace(-1, 1, n)
    u = 0.5 + x ** 2
    inits.append(u / u.sum())

    # Cosine window
    x = np.linspace(0, 1, n)
    cw = 0.5 * (1 + np.cos(2 * math.pi * x))
    cw = np.clip(cw, 1e-10, None)
    inits.append(cw / cw.sum())

    # Random smooth
    rng = np.random.default_rng(42)
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) ** 0.5
        # Smooth it
        kernel = np.ones(5) / 5
        r = np.convolve(r, kernel, mode='same')
        r = np.clip(r, 1e-10, None)
        inits.append(r / r.sum())

    return inits


def _try_length(n, t_end, base_seq=None):
    """Try to optimize for a specific sequence length n."""
    inits = _make_inits(n, base_seq)
    n_inits = len(inits)

    best_a = None
    best_score = float('inf')

    # Allocate time: give more time to base-seq init, split rest among others
    if base_seq is not None:
        # First init is from base - give it more time
        per_init = [(t_end - time.monotonic()) * 0.4 / max(1, n_inits)]
        remaining = (t_end - time.monotonic()) * 0.6 / max(1, n_inits - 1)
        per_init.extend([remaining] * (n_inits - 1))
    else:
        per_init = [(t_end - time.monotonic()) / n_inits] * n_inits

    for i, init in enumerate(inits):
        t_this = time.monotonic() + per_init[i] - 0.5
        if t_this <= time.monotonic() + 2:
            continue

        # Choose optimizer based on init quality and remaining time
        window = t_this - time.monotonic()

        if window > 30:
            # Use L-BFGS first, then projected gradient, then polish
            t_lbbf = time.monotonic() + window * 0.4
            a1, o1 = _optimize_lbbf(init, t_lbbf)

            t_proj = time.monotonic() + window * 0.3
            a2, o2 = _optimize_projected(a1, t_proj)

            t_polish = t_this
            a3, o3 = _peak_shave(a2, t_polish, n_steps=150)
            a4, o4 = _alternating_shave(a3, t_polish, n_steps=80)

            candidates = [a3, a4]
        elif window > 10:
            t_proj = t_this
            a2, o2 = _optimize_projected(init, t_proj)
            t_polish = t_this
            a3, o3 = _peak_shave(a2, t_polish, n_steps=50)
            candidates = [a3]
        else:
            a3, o3 = _peak_shave(init, t_this, n_steps=20)
            candidates = [a3]

        for a_cand in candidates:
            try:
                s = float(evaluate_sequence([float(x) for x in a_cand]))
            except Exception:
                s = float('inf')
            if s < best_score:
                best_score = s
                best_a = a_cand.copy()
                if i == 0:
                    print(f"    n={n}, init {i}: {s:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequence a.
    
    Uses log-parameterized L-BFGS with adaptive temperature annealing,
    projected subgradient descent, peak-shaving polish, and alternating
    peak balancing. Tries multiple sequence lengths and diverse initializations.
    """
    t_end = time.monotonic() + max(0.0, float(budget_s)) - 5.0
    rng = np.random.default_rng(seed)

    # Try to use provided sequence
    base_seq = None
    try:
        b = np.asarray(height_sequence_1, dtype=np.float64)
        b = np.clip(b, 0, 1000.0)
        if b.ndim == 1 and b.size >= 4 and np.isfinite(b).all():
            base_seq = b
            print(f"Using provided sequence, length={b.size}")
    except Exception:
        pass

    # Determine n values to try based on available time
    total_time = t_end - time.monotonic()

    # n values: focus on 384-1024 range
    n_options = [384, 512, 640, 768, 1024]

    # Weight allocation: favor n=512 (proven) and n=1024 (more DOF)
    if total_time > 1200:
        n_weights = {384: 0.10, 512: 0.30, 640: 0.10, 768: 0.15, 1024: 0.35}
    elif total_time > 600:
        n_weights = {384: 0.0, 512: 0.40, 640: 0.10, 768: 0.10, 1024: 0.40}
    else:
        n_weights = {384: 0.0, 512: 0.50, 640: 0.0, 768: 0.0, 1024: 0.50}

    n_options = [n for n in n_options if n_weights.get(n, 0) > 0]
    total_w = sum(n_weights[n] for n in n_options)

    best_a = None
    best_score = float('inf')

    for n in n_options:
        w = n_weights[n] / total_w
        t_this_end = time.monotonic() + (t_end - time.monotonic()) * w - 1.0
        if t_this_end <= time.monotonic() + 5:
            continue

        print(f"Trying n={n}, budget={t_this_end - time.monotonic():.1f}s")
        a_opt, score = _try_length(n, t_this_end, base_seq)

        if a_opt is not None and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"  => New best: n={n}, score={score:.6f}")

    # Final polish on the best found
    if best_a is not None and time.monotonic() < t_end - 30:
        n_best = len(best_a)
        t_polish = t_end - 5
        # Quick L-BFGS refinement
        a_ref, _ = _optimize_lbbf(best_a.copy(), time.monotonic() + 60)
        a_ps, _ = _peak_shave(a_ref, time.monotonic() + 40, n_steps=100)
        a_alt, _ = _alternating_shave(a_ps, t_polish, n_steps=50)

        for a_cand in [a_ps, a_alt, best_a]:
            try:
                s = float(evaluate_sequence([float(x) for x in a_cand]))
            except Exception:
                s = float('inf')
            if s < best_score:
                best_score = s
                best_a = a_cand.copy()
                print(f"  After final polish: {s:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512
        try:
            best_score = float(evaluate_sequence([float(x) for x in best_a]))
        except Exception:
            best_score = 2.0

    # Ensure valid output
    out = np.clip(best_a, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(512)

    try:
        final = float(evaluate_sequence([float(x) for x in out]))
        print(f"FINAL: n={len(out)}, score={final:.6f}")
    except Exception:
        print("Warning: final evaluation failed")

    return [float(x) for x in out]
```
