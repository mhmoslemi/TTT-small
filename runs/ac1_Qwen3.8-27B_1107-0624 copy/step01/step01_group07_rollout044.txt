    c_norm = c / c_max
    if power != 1.0:
        weights = c_norm ** power
    else:
        weights = c_norm
    blame = np.correlate(weights, a, mode='valid')
    return blame


def fast_iterate(a, deadline, rng, power=8.0, lr_init=0.15, verbose=True):
    """
    Fast iterative optimization using multiplicative peak-pressure updates.
    """
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = lr_init
    stall = 0
    it = 0
    power_adaptive = True
    current_power = power

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        M = c.max()

        if power_adaptive:
            # Adapt power: higher when peaked, lower when flat
            c_min_positive = np.percentile(c[c > 0], 10) if np.any(c > 0) else 0
            peak_ratio = c_min_positive / (M + 1e-15)
            # If already fairly flat, reduce power for smoother updates
            if peak_ratio > 0.7:
                current_power = max(2.0, current_power * 0.98)
            else:
                current_power = min(30.0, current_power * 1.02)

        blame = compute_blame(a, c, power=current_power)
        max_blame = blame.max()
        if max_blame < 1e-15:
            break

        # Multiplicative update
        normed_blame = blame / max_blame
        update = np.exp(-lr * normed_blame)
        a_new = a * update
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            lr *= 0.5
            continue
        a_new /= a_new.sum()

        sc = compute_objective(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.5, lr * 1.02)
        else:
            stall += 1
            lr = max(0.001, lr * 0.9)

            # Annealing: occasionally accept worse moves
            if stall > 5 and rng.random() < 0.1 * np.exp(-stall / 50.0):
                a = a_new
                stall = max(0, stall - 2)

            # Perturbation to escape local minima
            if stall % 30 == 0 and stall > 30:
                noise_scale = 0.05 * np.sqrt(stall / 30.0)
                noise = np.abs(rng.normal(0, noise_scale, n))
                a_pert = best_a * (1.0 + noise - noise.mean())
                a_pert = np.clip(a_pert, 0.0, None)
                if a_pert.sum() > 1e-10:
                    a_pert /= a_pert.sum()
                    sc_pert = compute_objective(a_pert)
                    if sc_pert < best_score:
                        best_a = a_pert
                        best_score = sc_pert
                        a = a_pert
                        stall = 0
                        lr = lr_init
                    elif rng.random() < 0.2:
                        a = a_pert
                        stall = 0

            if stall > 200:
                break

        if verbose and it % 200 == 0:
            print(f"      fast_iter: it={it}, score={best_score:.6f}, lr={lr:.4f}, power={current_power:.1f}")

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_polish(a, deadline, rng):
    """LP-based polishing step."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        M = c.max()

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        if g is None or g.sum() < 1e-15:
            break

        g = g / g.sum()
        improved = False

        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.05)
        else:
            stall += 1
            step = max(0.005, step * 0.7)
            if stall > 100:
                break

        if it % 20 == 0:
            print(f"      lp_polish: it={it}, score={best_score:.6f}, step={step:.4f}")

    return best_a, best_score


def make_cosine(n):
    """Raised cosine window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_anti_cosine(n):
    """U-shaped: high at edges, low in center."""
    return 0.5 * (1.0 - np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_bimodal(n, separation=0.3):
    """Bimodal: two Gaussian bumps."""
    x = np.arange(n) / (n - 1.0)
    c1 = 0.5 - separation / 2
    c2 = 0.5 + separation / 2
    sigma = 0.08
    return np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)


def make_sinc_abs(n, freq=1.0):
    """|sinc|-like shape."""
    x = np.arange(n) - (n - 1) / 2.0
    vals = np.sinc(x / n * freq * 2)
    return np.abs(vals) + 1e-8


def make_truncated_sine(n, num_peaks=2, phase=0.0):
    """Truncated positive sine with multiple peaks."""
    x = np.arange(n) / (n - 1.0) * num_peaks * np.pi
    return np.clip(np.sin(x + phase), 0, None) + 1e-8


def propose_candidate(seed=42, budget_s=1800):
    """
    Minimize 2*n*max(conv(a,a))/(sum(a))^2 using hybrid peak-pressure + LP
    optimization with multi-start and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p.copy()
    except Exception:
        pass

    print(f"Parent available: {parent is not None}, budget={budget_s}s")

    # Define search configurations
    # Each config: (n, init_func, init_kwargs, description)
    all_configs = []

    # Parent-based
    if parent is not None:
        for n_p in [min(parent.size, 384), min(parent.size, 512), min(parent.size, 768)]:
            all_configs.append((n_p, lambda n, p=parent: p[:n].copy(), {}, "parent"))

    # Diverse initializations at various lengths
    inits = [
        (make_cosine, {}),
        (make_anti_cosine, {}),
        (make_gaussian, {'sigma_frac': 0.2}),
        (make_gaussian, {'sigma_frac': 0.35}),
        (make_bimodal, {'separation': 0.25}),
        (make_bimodal, {'separation': 0.4}),
        (make_truncated_sine, {'num_peaks': 2, 'phase': 0.0}),
        (make_truncated_sine, {'num_peaks': 3, 'phase': 0.5}),
        (make_sinc_abs, {'freq': 1.0}),
        (make_sinc_abs, {'freq': 1.5}),
    ]

    lengths = [384, 512, 640, 768, 1024]

    for n in lengths:
        for init_f, kwargs in inits:
            all_configs.append((n, init_f, kwargs, init_f.__name__))

    # Add some random initializations
    def make_random_smoothed(n, rng_local):
        raw = np.abs(rng_local.normal(0, 1, n))
        # Smooth with a small kernel
        kernel = np.ones(5) / 5.0
        return np.convolve(raw, kernel, mode='same') + 1e-8

    for n in [512, 768, 1024]:
        all_configs.append((n, lambda n, r=rng: make_random_smoothed(n, r), {}, "random"))

    # Weight configs by promise
    weights = []
    for n, init_f, kwargs, desc in all_configs:
        w = 1.0
        if n in [512, 640, 768]:
            w *= 1.5
        elif n == 384:
            w *= 1.2
        elif n == 1024:
            w *= 1.0
        else:
            w *= 0.8
        if desc == "parent":
            w *= 2.0
        if "bimodal" in desc or "truncated" in desc or "anti" in desc:
            w *= 1.3
        weights.append(w)

    total_weight = sum(weights)
    available_time = deadline - start_time - 30.0
    if available_time < 60:
        available_time = 60

    # Sort configs by weight (try promising ones first)
    sorted_idx = np.argsort(weights)[::-1]
    time_allocs = np.cumsum(weights[sorted_idx]) * available_time / total_weight

    print(f"Total configs: {len(all_configs)}")
    print(f"Available time: {available_time:.0f}s")

    config_count = 0
    for pos, idx in enumerate(sorted_idx):
        if time.monotonic() > deadline - 30:
            break

        n, init_f, kwargs, desc = all_configs[idx]
        config_count += 1

        # Time allocation for this config
        if pos == len(sorted_idx) - 1:
            time_left = deadline - time.monotonic() - 15
        else:
            time_left = time_allocs[pos] - time_allocs[pos - 1]
        time_left = min(time_left, deadline - time.monotonic())

        if time_left < 10:
            continue

        try:
            a = init_f(n, **kwargs)
        except Exception:
            a = make_cosine(n)

        a = np.clip(a, 1e-10, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()

        # Initial score
        init_score = compute_objective(a)
        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()

        remaining = deadline - time.monotonic()
        # Split time: 70% fast iteration, 30% LP polish (if time allows)
        fast_time = remaining * 0.65 if n <= 1024 else remaining * 0.85
        fast_deadline = time.monotonic() + fast_time

        # Run fast iterative optimization
        a_opt, score = fast_iterate(a.copy(), fast_deadline, rng,
                                     power=10.0, lr_init=0.12, verbose=(config_count <= 10))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  [{config_count}] n={n}, {desc}: FAST score={score:.6f} => NEW BEST")
        else:
            print(f"  [{config_count}] n={n}, {desc}: FAST score={score:.6f} (best: {best_score:.6f})")

        # LP polish if time remains and n is manageable
        if time.monotonic() < deadline - 60 and n <= 768:
            lp_deadline = min(deadline - 15, time.monotonic() + (deadline - time.monotonic()) * 0.3)
            a_opt2, score2 = lp_polish(a_opt.copy(), lp_deadline, rng)
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_seq = a_opt2.copy()
                print(f"  [{config_count}] n={n}, {desc}: LP score={score2:.6f} => NEW BEST")

    # Final polish on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"Final polish on best (n={n_best}, score={best_score:.6f})...")

        # First, fast iteration
        polish_deadline1 = min(deadline - 15, time.monotonic() + (deadline - time.monotonic()) * 0.5)
        best_seq, best_score = fast_iterate(best_seq.copy(), polish_deadline1, rng,
                                             power=6.0, lr_init=0.05, verbose=True)
        print(f"  After fast polish: {best_score:.6f}")

        # Then LP if n is small enough
        if n_best <= 768 and time.monotonic() < deadline - 15:
            best_seq, best_score = lp_polish(best_seq.copy(), deadline - 5, rng)
            print(f"  After LP polish: {best_score:.6f}")

    if best_seq is None:
        best_seq = make_cosine(512)
        best_score = compute_objective(best_seq)

    # Ensure validity
    best_seq = np.clip(best_seq, 0.0, None)
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())

    # Cap values at 1000 (evaluator does this, but let's be safe)
    best_seq = np.clip(best_seq, 0, 1000.0)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE in {elapsed:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
