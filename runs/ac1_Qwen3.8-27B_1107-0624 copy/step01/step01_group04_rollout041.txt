def next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)

    win = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t - 1.0) ** gamma))
    return normalize_nonneg(p * win)

    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def try_eval(seq):
    try:
        val = evaluate_sequence(list(seq))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def record_candidate(p, best_p, best_score, best_by_n, score=None):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        return best_p, best_score
    if score is None:
        score = actual_score(a)
    if not np.isfinite(score):
        return best_p, best_score
    if score < best_score:
        best_p = a.copy()
        best_score = float(score)
    n = int(a.size)
    old = best_by_n.get(n)
    if old is None or score < old[0]:
        best_by_n[n] = (float(score), a.copy())
    return best_p, best_score


def power_sum_adam(p, budget_s, p_start=4, p_end=64, seed=0):
    p = normalize_nonneg(p)
    n = p.size
    if n == 0:
        return p, np.inf
    if budget_s < 3.0:
        return p, actual_score(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 128:
        lr = 0.04
    elif n < 512:
        lr = 0.025
    else:
        lr = 0.015

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    L = next_pow2(3 * n - 2)
    best_p = p.copy()
    best_f = actual_score(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    while time.monotonic() < end:
        t += 1
        frac = min(1.0, (time.monotonic() - start) / max(1e-9, budget_s))
        pp = int(p_start + (p_end - p_start) * min(1.0, frac * 1.5))
        pp = max(2, pp)

        Pa = np.fft.rfft(p, L)
        c = np.fft.irfft(Pa * Pa, L)[:2 * n - 1].real
        c = np.maximum(c, 1e-30)

        cp = np.power(c, pp - 1)
        norm = float(np.sum(np.power(c, pp))) ** (1.0 / pp)
        if not np.isfinite(norm) or norm <= 0:
            break

        W = np.fft.rfft(cp[::-1], L)
        Cfull = np.fft.irfft(Pa * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_a = C / norm
        gp = float(np.dot(p, grad_a))
        grad_x = p * (grad_a - gp)

        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -60.0, 60.0)
        x -= x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0:
            break
        p = e / se

        if t % 200 == 0:
            f_now = actual_score(p)
            if f_now < best_f:
                best_f = f_now
                best_p = p.copy()

    f_final = actual_score(p)
    if f_final < best_f:
        best_f = f_final
        best_p = p.copy()
    return best_p, best_f


def peak_deflation(p, budget_s, n_iters=200):
    p = normalize_nonneg(p)
    n = p.size
    if n < 2 or budget_s < 5.0:
        return p, actual_score(p)

    best_p = p.copy()
    best_f = actual_score(p)
    L = next_pow2(2 * n - 1)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    eta = 0.05

    while time.monotonic() < end:
        Pa = np.fft.rfft(p, L)
        c = np.fft.irfft(Pa * Pa, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if maxc <= 0:
            break

        kstar = int(np.argmax(c))

        if kstar < n:
            pressure = p[kstar::-1] if kstar > 0 else np.array([p[0]])
            pressure = np.zeros(n)
            for i in range(min(n, kstar + 1)):
                pressure[i] = p[kstar - i]
        else:
            lo = kstar - n + 1
            hi = n - 1
            pressure = np.zeros(n)
            for i in range(lo, hi + 1):
                pressure[i] = p[kstar - i]

        mean_p = float(pressure.mean())
        mask = p > 1e-15
        update = np.zeros(n)
        update[mask] = eta * (mean_p - pressure[mask])
        update = np.clip(update, -0.5 * p, 0.5 * p)

        p_new = p + update
        p_new = np.clip(p_new, 0.0, None)
        sn = float(p_new.sum())
        if sn <= 1e-12:
            break
        p_new = p_new / sn

        f_new = actual_score(p_new)
        if np.isfinite(f_new) and f_new < best_f:
            best_f = f_new
            best_p = p_new.copy()

        eta *= 0.98

    return best_p, best_f


def lp_polish(p, budget_s, max_rows=800):
    if _linprog is None:
        return p, actual_score(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4 or budget_s < 30.0:
        return p, actual_score(p)

    max_elements = 4000000
    while max_rows * n > max_elements:
        max_rows = max(64, int(max_rows * 0.8))
    max_rows = max(64, min(max_rows, max_elements // max(1, n)))

    S0 = np.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score(f)
    best_f = f.copy()
    best_score = float(cur_score)

    end = time.monotonic() + max(0.0, budget_s - 5.0)

    r0 = min(max_rows, max(40, max_rows // 3))
    rows0 = np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int)).astype(int)
    rows = [int(k) for k in rows0.tolist()]
    row_set = set(rows)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=np.int32)
        A = np.zeros((rows_arr.size, n), dtype=np.float64)
        for ii in range(rows_arr.size):
            k = int(rows_arr[ii])
            lo = k - n + 1
            if lo < 0:
                lo = 0
            hi = k if k < n else n - 1
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(rows_arr.size, cur_score, dtype=np.float64)
        cobj = -np.ones(n)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.2))
        try:
            res = _linprog(cobj, A_ub=A, b_ub=b, bounds=(0.0, None),
                           method='highs', options={'time_limit': float(tl), 'presolve': True})
        except Exception:
            break

        if res is None or not getattr(res, 'success', False) or getattr(res, 'x', None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        L2 = next_pow2(2 * n - 1)
        Pf = np.fft.rfft(f, L2)
        Pg = np.fft.rfft(g, L2)
        conv = np.fft.irfft(Pf * Pg, L2)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)

        tol = 1e-6 + 1e-6 * cur_score
        violations = conv - cur_score
        if np.any(violations > tol):
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                for ix in order:
                    if added >= 200:
                        break
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0 and float(np.max(violations)) > 5.0 * tol:
                break
            continue

        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in (1.0, 0.5, 0.3, 0.15, 0.08, 0.04):
            fn = (1.0 - step) * f + step * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score) or accepted_score >= best_score:
            break

        f = accepted
        cur_score = accepted_score
        best_f = f.copy()
        best_score = cur_score

        if it % 3 == 0:
            print("  LP it=%d score=%.6f rows=%d" % (it, best_score, len(rows)))

    return normalize_nonneg(best_f), best_score


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    # Phase 0: Quick baselines
    print("Phase 0: Baselines")
    base_lengths = [128, 192, 256, 320, 384, 448, 512, 640, 768]
    for n in base_lengths:
        if time.monotonic() >= deadline:
            break
        for kind in ['greedy', 'power45', 'power55', 'win0.8', 'win1.2']:
            if kind == 'greedy':
                p = greedy_normalized(n)
            elif kind == 'power45':
                p = power_normalized(n, 0.45)
            elif kind == 'power55':
                p = power_normalized(n, 0.55)
            elif kind == 'win0.8':
                p = windowed_greedy(n, 0.8)
            else:
                p = windowed_greedy(n, 1.2)
            best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)

    # Load previous best if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size > 0 and arr.sum() > 0.01:
                best_p, best_score = record_candidate(arr, best_p, best_score, best_by_n)
                print("  Loaded height_sequence_1: n=%d score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    print("  Current best: %.6f" % best_score)

    # Phase 1: Power-sum Adam optimization
    print("Phase 1: Power-sum Adam")
    jobs = [
        (384, 'greedy', 45),
        (384, 'perturb', 30),
        (448, 'greedy', 40),
        (320, 'greedy', 35),
        (512, 'greedy', 45),
        (256, 'greedy', 25),
        (640, 'greedy', 40),
        (384, 'win', 30),
        (448, 'perturb', 30),
        (320, 'perturb', 25),
        (512, 'win', 35),
        (768, 'greedy', 45),
    ]

    for n, kind, alloc in jobs:
        rem = deadline - time.monotonic()
        if rem < 20.0:
            break
        if kind == 'greedy':
            base = greedy_normalized(n)
        elif kind == 'perturb':
            base = perturb_greedy(n, rng, 0.12)
        elif kind == 'win':
            base = windowed_greedy(n, 1.0 + rng.random() * 0.4)
        else:
            rec = best_by_n.get(n)
            base = rec[1] if rec else greedy_normalized(n)

        alloc_use = min(float(alloc), max(5.0, rem - 10.0))
        p_out, f_out = power_sum_adam(base, alloc_use, p_start=4, p_end=48, seed=int(seed) + n)
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("  n=%d kind=%s: %.6f (best=%.6f)" % (n, kind, f_out, best_score))

    # Phase 2: Peak deflation
    print("Phase 2: Peak deflation")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            for trial in range(3):
                base = perturb_greedy(int(best_p.size), rng, 0.03) if trial > 0 else best_p.copy()
                p_out, f_out = peak_deflation(base, min(60.0, rem - 10.0) / 3.0)
                best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
                print("  trial %d: %.6f (best=%.6f)" % (trial, f_out, best_score))
                rem = deadline - time.monotonic()
                if rem < 15:
                    break

    # Phase 3: LP polishing
    print("Phase 3: LP polishing")
    if _linprog is not None:
        lp_n = int(best_p.size) if best_p is not None else 384
        lp_candidates = [lp_n]
        for n in [384, 448, 512, 320]:
            if n != lp_n:
                lp_candidates.append(n)

        lp_scored = []
        for n in lp_candidates:
            rec = best_by_n.get(n)
            sc = rec[0] if rec else 99.0
            lp_scored.append((sc, n))
        lp_scored.sort(key=lambda x: x[0])

        for sc, n in lp_scored:
            rem = deadline - time.monotonic()
            if rem < 60.0:
                break
            rec = best_by_n.get(n)
            base = rec[1] if rec else greedy_normalized(n)
            alloc = min(180.0, max(40.0, rem * 0.4))
            p_out, f_out = lp_polish(base, alloc, max_rows=600)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  LP n=%d: %.6f (best=%.6f)" % (n, f_out, best_score))

    # Phase 4: Final Adam polish on best
    print("Phase 4: Final polish")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            p0 = normalize_nonneg(best_p)
            p_out, f_out = power_sum_adam(p0, min(rem - 5.0, 120.0), p_start=16, p_end=64, seed=int(seed) + 777)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  Polish: %.6f (best=%.6f)" % (f_out, best_score))

        rem = deadline - time.monotonic()
        if rem > 25 and _linprog is not None:
            p_out, f_out = lp_polish(best_p, min(rem - 5.0, 100.0), max_rows=500)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  Final LP: %.6f (best=%.6f)" % (f_out, best_score))

    # Phase 5: Validation
    print("Phase 5: Validation")
    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score(best_p)

    final_seq = scale_output(best_p, 10.0)
    gscore = try_eval(final_seq)
    if np.isfinite(gscore):
        print("  Global eval: %.6f" % gscore)
    else:
        print("  Global eval failed, using local score: %.6f" % best_score)

    # Also try nearby lengths for the final answer
    if time.monotonic() < deadline - 10:
        candidates = [(0, best_p)]
        for n in sorted(best_by_n.keys()):
            if best_by_n[n][0] < 1.55:
                candidates.append((n, best_by_n[n][1]))
        candidates.sort(key=lambda x: (best_by_n.get(x[0], (99,))[0] if x[0] else best_score))
        for n, cand in candidates[:3]:
            seq = scale_output(cand, 10.0)
            gs = try_eval(seq)
            if np.isfinite(gs) and gs < gscore:
                gscore = gs
                best_p = cand.copy()
                final_seq = seq
                print("  Better from n=%d: %.6f" % (n, gs))

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Error in main_search: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
