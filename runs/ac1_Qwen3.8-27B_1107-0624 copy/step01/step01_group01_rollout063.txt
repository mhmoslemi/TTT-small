"""
Multi-restart gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, soft-max approximation, Adam optimizer in log-space,
with diverse monotone initializations and coordinate descent refinement.

import numpy as np

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        mask = c >= mval - 1e-12
        if not np.any(mask):
            idx = int(np.argmax(c))
            mask[idx] = True
        w[mask] = 1.0 / float(np.count_nonzero(mask))
        return mval, w
    x = (c - mval) / beta
    x = x - x.max()
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, a):
    n = len(a)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a[::-1], N)
    h = np.fft.irfft(fw * fa, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-14
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def make_linear_profile(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * (t - 0.5)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_power_profile(n, alpha):
    t = np.arange(1, n + 1)
    a = t ** alpha
    s = float(a.sum())
    return a / s


def make_exp_profile(n, beta):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(beta * t)
    s = float(a.sum())
    return a / s


def make_cosine_profile(n, gamma):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + gamma * np.sin(np.pi * t / 2.0)
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    return a / s


def make_random_monotone(n, rng):
    increments = rng.exponential(1.0, n)
    a = np.cumsum(increments)
    a = a / a.sum()
    return a

def logit_optimize(p0, deadline, rng, lr=0.06, reg=1e-7, beta_scale=0.2):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    p0 = np.maximum(p0, 1e-14)
    p0 = p0 / p0.sum()
    n = p0.size

    z = np.log(p0)
    z -= float(np.mean(z))

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_t = 0

    p = p0.copy()
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, beta_scale * maxc)

    best_p = p0.copy()
    best_z = z.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr
    fail_count = 0

    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break

        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + reg * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gz -= float(np.mean(gz))

        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            fail_count += 1
            if fail_count > 5:
                break
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        step = lr * mh / (np.sqrt(vh) + eps)
        z = z - step

        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            fail_count += 1
            if fail_count > 5:
                break
            continue

        z = np.clip(z, -50.0, 50.0)
        z -= float(np.mean(z))
        p = softmax_from_z(z)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.015, lr * 0.96)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            fail_count += 1
            if fail_count > 5:
                break
            continue

        if adam_t % 80 == 0:
            beta = max(1e-12, beta * 0.72)

        if no_imp > max(400, 4 * n):
            strength = 0.3 + 0.3 * min(1.0, (no_imp - 400) / (4.0 * n))
            z = best_z * (1.0 - strength) + strength * rng.normal(size=n)
            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.25 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score


def coordinate_descent_refine(p, deadline_min, rng, iters_per_elem=8):
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    p = np.maximum(p, 1e-14)
    p = p / p.sum()
    best_score = internal_score(p)
    t_start = time.monotonic()
    deadline = t_start + deadline_min

    elem_cycle = 0
    while time.monotonic() < deadline:
        i = elem_cycle % n
        elem_cycle += 1

        a_i = p[i]
        rest_sum = 1.0 - a_i

        lo = 0.0
        hi = min(0.9999, 0.5)

        def obj(x):
            q = p.copy()
            q[i] = x
            q = np.maximum(q, 1e-14)
            q = q / q.sum()
            return internal_score(q)

        best_x = a_i
        best_f = obj(a_i)

        for _ in range(iters_per_elem):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            f1 = obj(mid1)
            f2 = obj(mid2)
            if f1 < f2:
                hi = mid2
                if f1 < best_f:
                    best_f = f1
                    best_x = mid1
            else:
                lo = mid1
                if f2 < best_f:
                    best_f = f2
                    best_x = mid2

        p[i] = best_x
        p = p / p.sum()

        sc = internal_score(p)
        if sc < best_score:
            best_score = sc

    return p, best_score


def param_adam(theta0, basis, deadline, rng, lr=0.2):
    basis = np.asarray(basis, dtype=np.float64)
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(dim, dtype=np.float64)
    v = np.zeros(dim, dtype=np.float64)
    adam_t = 0

    p = softmax_from_z(basis @ theta)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.2 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + 1e-6 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        adam_t += 1
        m = b1 * m + (1.0 - b1) * gtheta
        v = b2 * v + (1.0 - b2) * (gtheta * gtheta)
        mh = m / (1.0 - b1 ** adam_t)
        vh = v / (1.0 - b2 ** adam_t)
        theta = theta - lr * mh / (np.sqrt(vh) + eps)
        theta = np.clip(theta, -8.0, 8.0)

        p = softmax_from_z(basis @ theta)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 200 == 0:
                    lr = max(0.03, lr * 0.95)
        else:
            theta = best_theta.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue

        if adam_t % 60 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(200, 8 * dim):
            theta = best_theta + 0.5 * rng.normal(size=dim)
            theta = np.clip(theta, -8.0, 8.0)
            p = softmax_from_z(basis @ theta)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.25 * maxc)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0

    return best_theta, best_p, best_score


def make_basis_trig(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = [u]
    m_k = 1
    while len(cols) < K:
        ang = 2.0 * np.pi * m_k * t
        cols.append(np.cos(ang))
        cols.append(np.sin(ang))
        m_k += 1
    return np.column_stack(cols[:K])


def make_basis_poly(n, K):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    cols = []
    for k in range(1, K + 1):
        cols.append(u ** k)
    return np.column_stack(cols)


def add_candidate(candidates, p, score, limit=20):
    if p is None or score is None or not np.isfinite(score):
        return
    p = np.asarray(p, dtype=np.float64).copy()
    if len(candidates) < limit:
        candidates.append((float(score), p))
        candidates.sort(key=lambda x: x[0])
    else:
        worst_idx = 0
        worst_val = candidates[0][0]
        for i in range(1, len(candidates)):
            if candidates[i][0] > worst_val:
                worst_val = candidates[i][0]
                worst_idx = i
        if float(score) < worst_val:
            candidates[worst_idx] = (float(score), p)
            candidates.sort(key=lambda x: x[0])
