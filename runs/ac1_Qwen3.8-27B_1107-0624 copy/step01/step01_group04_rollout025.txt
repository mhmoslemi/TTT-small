"""
Search for non-negative step-function sequence minimizing
    2*n*max(convolve(seq,seq)) / sum(seq)^2
using L-BFGS optimization with log-sum-exp smoothing,
multiple diverse initializations, and coordinate-descent polishing.

import time
import numpy as np

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mix_greedy_uniform(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def compute_conv(a):
    """Compute full convolution of a with itself via FFT."""
    n = a.size
    if n <= 256:
        return np.convolve(a, a)
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def objective_and_grad(x, alpha):
    """
    Objective and gradient for the log-parameterized softmax approach.
    x: log-parameters (length n)
    alpha: smoothing parameter
    Returns: (objective_value, gradient_w.r.t. x)
    """
    n = x.size
    # Compute softmax
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    se = float(e.sum())
    a = e / se  # normalized, sum = 1
    
    # Compute convolution
    c = compute_conv(a)
    m = c.size  # 2n-1
    
    # Log-sum-exp
    maxc = float(c.max())
    if maxc <= 0:
        return 2.0 * n, np.zeros(n)
    z = alpha * (c - maxc)
    expz = np.exp(z)
    lsm = maxc + np.log(float(expz.sum()) / m)  # Actually LSM = maxc + log(sum(exp(alpha*(c-maxc))))
    # LSM(alpha * c) = alpha * maxc + log(sum(exp(alpha*(c - maxc))))
    # LSM/alpha = maxc + (1/alpha) * log(sum(exp(alpha*(c - maxc))))
    obj_val = 2.0 * n * (maxc + (1.0 / alpha) * np.log(float(expz.sum())))
    
    # Weights
    w = expz / float(expz.sum())
    
    # Gradient of w.r.t. a: grad_a_j = 2 * sum_k w_k * a_{k-j} (for valid k-j)
    # This is 2 * cross_correlation(w, a) evaluated at positions 0..n-1
    L = next_power_of_two(3 * n - 2)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    # cross_correlation[w, a][j] = sum_k w[k] * a[k-j] = ifft(conj(fft(a)) * fft(w)).real
    cc = np.fft.irfft(np.conj(A) * W, L).real[:n]
    grad_a = 2.0 * cc
    
    # Gradient w.r.t. x (chain rule through softmax)
    # df/dx_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    dot = float(np.dot(a, grad_a))
    grad_x = a * (grad_a - dot)
    
    return float(obj_val), grad_x


def lbfags_optimize(p, budget_s, seed=0):
    """
    Optimize using scipy L-BFGS-B with log-sum-exp smoothing.
    p: initial point (will be normalized)
    budget_s: time budget in seconds
    """
    if scipy_minimize is None:
        return p, actual_score_fft(p)
    
    p = normalize_nonneg(p)
    n = p.size
    budget_s = float(budget_s)
    
    if budget_s < 5.0:
        return p, actual_score_fft(p)
    
    x = np.log(p + 1e-300)
    x -= x.mean()
    
    best_p = p.copy()
    best_f = actual_score_fft(p)
    
    start = time.monotonic()
    end = start + max(0.0, budget_s - 3.0)
    
    # Progressive smoothing: start with moderate alpha, increase over time
    # Use multiple phases with increasing alpha
    phases = [
        (20.0, 0.3),   # (alpha, fraction of budget)
        (50.0, 0.3),
        (100.0, 0.2),
        (200.0, 0.15),
        (500.0, 0.1),   # very sharp, close to true max
    ]
    
    for alpha, frac in phases:
        phase_time = frac * (end - start)
        phase_end = time.monotonic() + phase_time
        if time.monotonic() >= end - 5.0:
            break
        if phase_time < 3.0:
            continue
        
        def objfun(xv):
            val, _ = objective_and_grad(xv, alpha)
            return val
        
        def objgrad(xv):
            _, g = objective_and_grad(xv, alpha)
            return g
        
        try:
            maxiter = max(10, int(phase_time * 10))
            res = scipy_minimize(
                objfun, x, method='L-BFGS-B', jac=objgrad,
                options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                         'ftol': 1e-12, 'gtol': 1e-10}
            )
            if res.x is not None:
                xv = res.x
                xv_shifted = xv - xv.max()
                e = np.exp(xv_shifted)
                se = float(e.sum())
                if np.isfinite(se) and se > 1e-12:
                    a_new = e / se
                    f_new = actual_score_fft(a_new)
                    if np.isfinite(f_new) and f_new < best_f:
                        best_f = float(f_new)
                        best_p = a_new.copy()
                    x = res.x.copy()
        except Exception:
            pass
    
    # Final check
    f_final = actual_score_fft(best_p)
    if f_final < best_f:
        best_f = float(f_final)
    
    return best_p, best_f


def coordinate_descent(p, budget_s):
    """
    Coordinate descent: for each coordinate, try small multiplicative changes.
    """
    p = normalize_nonneg(p).copy()
    n = p.size
    best_f = actual_score_fft(p)
    
    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    
    rng = np.random.default_rng(12345)
    
    # Precompute FFT length
    L = next_power_of_two(2 * n - 1)
    
    while time.monotonic() < end:
        improved = False
        # Sweep through coordinates (possibly in random order)
        order = np.arange(n)
        if n > 200:
            rng.shuffle(order)
        
        for j in order:
            if time.monotonic() >= end:
                break
            orig = p[j]
            if orig < 1e-10:
                continue
            
            best_j_val = orig
            best_j_f = best_f
            
            # Try a few multiplicative factors
            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                new_val = orig * factor
                if new_val < 0 or new_val > 1000.0:
                    continue
                p[j] = new_val
                # Quick score computation
                s = float(p.sum())
                if s <= 1e-12:
                    p[j] = orig
                    continue
                a = p  # already has sum that we'll normalize
                if n <= 256:
                    c = np.convolve(a, a)
                else:
                    P = np.fft.rfft(a, L)
                    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
                c = np.maximum(c, 0.0)
                maxc = float(c.max())
                f_new = 2.0 * n * maxc / (s * s)
                
                if f_new < best_j_f - 1e-12:
                    best_j_f = f_new
                    best_j_val = new_val
            
            p[j] = best_j_val
        
        if time.monotonic() < end:
            # Recompute score
            s = float(p.sum())
            if s <= 1e-12:
                break
            a = p / s  # normalize
            f_new = actual_score_fft(a)
            p = a
            if f_new < best_f - 1e-12:
                best_f = float(f_new)
                improved = True
    
    return p, best_f

"""
Search for non-negative step-function sequence minimizing
    2*n*max(convolve(seq,seq)) / sum(seq)^2
using L-BFGS with log-sum-exp smoothing, progressive temperature,
diverse initializations, and coordinate-descent polishing.

import time
import numpy as np

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mix_greedy_uniform(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def perturb_greedy(n, rng, sigma):
    p = greedy_normalized(n)
    noise = np.exp(rng.normal(0.0, float(sigma), int(n)))
    return normalize_nonneg(p * noise)


def dirichlet_start(n, alpha, rng):
    try:
        u = rng.dirichlet(float(alpha) * np.ones(n, dtype=np.float64))
    except Exception:
        u = np.ones(n, dtype=np.float64)
    return normalize_nonneg(u)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def compute_conv(a):
    n = a.size
    if n <= 256:
        return np.convolve(a, a)
    L = next_power_of_two(2 * n - 1)
    P = np.fft.rfft(a, L)
    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def softmax_obj_grad(x, alpha):
    """
    Objective and gradient for log-parameterized softmax variables.
    x: log-parameters (length n)
    alpha: smoothing parameter for log-sum-exp
    Returns: (objective, gradient w.r.t. x)
    """
    n = x.size
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    se = float(e.sum())
    if not np.isfinite(se) or se <= 0:
        return 2.0 * n, np.zeros(n)
    a = e / se

    c = compute_conv(a)
    m = c.size
    maxc = float(c.max())
    if maxc <= 1e-300:
        return 2.0 * n, np.zeros(n)

    z = alpha * (c - maxc)
    expz = np.exp(z)
    sz = float(expz.sum())
    if not np.isfinite(sz) or sz <= 0:
        w = np.zeros(m, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w = expz / sz

    obj_val = 2.0 * n * (maxc + (1.0 / alpha) * np.log(sz))

    # grad_a_j = 2 * sum_k w[k] * a[k-j] = 2 * (w * a_flip)[n-1 + j]
    L = next_power_of_two(3 * n - 2)
    W = np.fft.rfft(w, L)
    Af = np.fft.rfft(a[::-1], L)
    conv = np.fft.irfft(W * Af, L).real
    g = 2.0 * conv[n - 1:2 * n - 1]

    if not np.all(np.isfinite(g)):
        return float(obj_val), np.zeros(n)

    dot = float(np.dot(a, g))
    grad_x = a * (g - dot)

    return float(obj_val), grad_x


def lbfags_optimize(p, budget_s, seed=0):
    """
    Multi-phase L-BFGS optimization with progressive smoothing.
    """
    if scipy_minimize is None:
        return p, actual_score_fft(p)

    p = normalize_nonneg(p)
    n = p.size
    budget_s = float(budget_s)
    if budget_s < 3.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)

    # Progressive alpha schedule
    alphas = [15.0, 40.0, 100.0, 250.0, 600.0]
    fracs = [0.25, 0.25, 0.22, 0.18, 0.10]

    for i, (alpha, frac) in enumerate(zip(alphas, fracs)):
        if time.monotonic() >= end - 3.0:
            break
        phase_time = frac * (end - start)
        if phase_time < 2.0:
            continue

        def objfun(xv):
            val, _ = softmax_obj_grad(xv, alpha)
            return val

        def objgrad(xv):
            _, g = softmax_obj_grad(xv, alpha)
            return g

        maxiter = max(20, int(phase_time * 15))
        try:
            res = scipy_minimize(
                objfun, x.copy(), method='L-BFGS-B', jac=objgrad,
                options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                         'ftol': 1e-14, 'gtol': 1e-12, 'maxls': 50}
            )
            if res.x is not None:
                xv = np.asarray(res.x, dtype=np.float64).ravel()
                xv_shifted = xv - xv.max()
                e = np.exp(xv_shifted)
                se = float(e.sum())
                if np.isfinite(se) and se > 1e-12:
                    a_new = e / se
                    f_new = actual_score_fft(a_new)
                    if np.isfinite(f_new) and f_new < best_f:
                        best_f = float(f_new)
                        best_p = a_new.copy()
                    x = xv.copy()
        except Exception:
            pass

    f_final = actual_score_fft(best_p)
    if f_final < best_f:
        best_f = float(f_final)

    return best_p, best_f


def coordinate_polish(p, budget_s, seed=0):
    """
    Coordinate descent: sweep through coordinates trying multiplicative changes.
    """
    p = normalize_nonneg(p).copy()
    n = p.size
    best_f = actual_score_fft(p)

    if budget_s < 2.0 or n > 1024:
        return p, best_f

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)
    rng = np.random.default_rng(seed)

    L = next_power_of_two(2 * n - 1)
    factors = [0.6, 0.75, 0.85, 0.92, 1.08, 1.15, 1.25, 1.4]

    sweep = 0
    while time.monotonic() < end:
        sweep += 1
        order = np.arange(n)
        if sweep % 2 == 0:
            rng.shuffle(order)

        for j in order:
            if time.monotonic() >= end:
                break
            orig = p[j]
            if orig < 1e-12:
                continue

            best_val = orig
            best_fj = best_f

            for fc in factors:
                new_val = orig * fc
                if new_val < 0 or new_val > 50.0:
                    continue
                p[j] = new_val
                s = float(p.sum())
                if s <= 1e-12:
                    p[j] = orig
                    continue
                if n <= 256:
                    c = np.convolve(p, p)
                else:
                    P = np.fft.rfft(p, L)
                    c = np.fft.irfft(P * P, L)[:2 * n - 1].real
                c = np.maximum(c, 0.0)
                maxc = float(c.max())
                fj = 2.0 * n * maxc / (s * s)
                if fj < best_fj - 1e-10:
                    best_fj = fj
                    best_val = new_val

            p[j] = best_val

        s = float(p.sum())
        if s > 1e-12:
            p = p / s
            f_new = actual_score_fft(p)
            if f_new < best_f:
                best_f = float(f_new)

    return p, best_f

    for n in opt_lengths:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        # For each length, try several initializations
        inits = [
            ("greedy", greedy_normalized(n)),
            ("p04", power_normalized(n, 0.4)),
            ("p06", power_normalized(n, 0.6)),
            ("mix30", mix_greedy_uniform(n, 0.3)),
            ("mix60", mix_greedy_uniform(n, 0.6)),
            ("pert10", perturb_greedy(n, rng, 0.10)),
            ("pert30", perturb_greedy(n, rng, 0.30)),
        ]

        # If we have a best at or near this length, use it as init
        if best_p is not None:
            bp_n = best_p.size
            if abs(bp_n - n) <= n * 0.3:
                # Truncate or pad
                if bp_n >= n:
                    inits.append(("best_trunc", normalize_nonneg(best_p[:n])))
                else:
                    padded = np.zeros(n, dtype=np.float64)
                    padded[:bp_n] = best_p
                    inits.append(("best_pad", normalize_nonneg(padded)))

        # Allocate time: more time for lengths near the current best
        if best_p is not None and abs(n - best_p.size) <= 100:
            time_per = max(15.0, rem * 0.8 / len(opt_lengths) / max(1, len(inits)) * 4)
        else:
            time_per = max(8.0, rem * 0.8 / len(opt_lengths) / max(1, len(inits)) * 2)

        for name, init_p in inits:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(time_per, max(5.0, rem - 5.0))
            n_out, f_out = lbfags_optimize(init_p, alloc, seed=int(seed) + n)
            if n_out is not None:
                record(n_out, f_out)

    print("After L-BFGS: best=%.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Coordinate descent polishing on top candidates
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            alloc = min(120.0, max(15.0, rem * 0.4))
            p_out, f_out = coordinate_polish(best_p, alloc, seed=int(seed) + 77)
            record(p_out, f_out)
            print("After coord polish (best): %.6f" % best_score)

        # Polish the best at a few other lengths too
        items = []
        for n, sc in best_by_n.items():
            if n != int(n):
                continue
            items.append((float(sc), int(n)))
        items.sort()
        for sc, n in items[:4]:
            if time.monotonic() >= deadline - 30:
                break
            rec = best_by_n.get(float(n) + 0.5)
            if rec is None:
                continue
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            alloc = min(60.0, max(10.0, rem * 0.3))
            p_out, f_out = coordinate_polish(rec, alloc, seed=n)
            record(p_out, f_out)

    # Additional L-BFGS from perturbed best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            n_best = best_p.size
            for sigma in [0.05, 0.10, 0.15, 0.20, 0.30, 0.50]:
                if time.monotonic() >= deadline - 15:
                    break
                p0 = perturb_greedy(n_best, rng, sigma) * 0.5 + 0.5 * best_p
                p0 = normalize_nonneg(p0)
                alloc = min(80.0, max(10.0, rem - 20.0))
                n_out, f_out = lbfags_optimize(p0, alloc, seed=int(seed) + 9999)
                record(n_out, f_out)
                rem = deadline - time.monotonic()
                if best_score < 1.505:
                    break

    # Try a few more lengths near the best
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            n_best = best_p.size
            for dn in [-32, -16, -8, 8, 16, 32, 48, 64]:
                n_try = n_best + dn
                if n_try < 64 or n_try > 4096:
                    continue
                if time.monotonic() >= deadline - 15:
                    break
                # Interpolate from best
                p_interp = np.zeros(n_try, dtype=np.float64)
                for i in range(n_try):
                    src = int(i * n_best / n_try)
                    if src < n_best:
                        p_interp[i] = best_p[src]
                p_interp = normalize_nonneg(p_interp)
                alloc = min(50.0, max(8.0, rem - 20.0))
                n_out, f_out = lbfags_optimize(p_interp, alloc, seed=int(seed) + n_try)
                record(n_out, f_out)

    print("Before validation: best=%.6f (n=%d)" % (best_score, best_p.size if best_p is not None else 0))

    # Validate top candidates with the authoritative evaluator
    if best_p is not None:
        candidates = [best_p]
        items = []
        for n, sc in best_by_n.items():
            if int(n) == n:
                rec = best_by_n.get(float(int(n)) + 0.5)
                if rec is not None:
                    items.append((float(sc), int(n), rec))
        items.sort()
        for sc, n, p in items[:5]:
            if p is not None and p is not best_p:
                candidates.append(p)

        for cand in candidates:
            if time.monotonic() >= deadline - 10:
                break
            try:
                seq = scale_output(cand, 10.0)
                gscore = try_global_eval(seq)
                if np.isfinite(gscore):
                    if gscore < best_score:
                        best_score = float(gscore)
                        best_p = np.asarray(cand, dtype=np.float64).ravel()
                    print("  Candidate n=%d: local=%.6f, global=%.6f" % (
                        cand.size, actual_score_fft(cand), gscore))
            except Exception:
                pass

    if best_p is None:
        best_p = greedy_normalized(384)
        best_score = actual_score_fft(best_p)

    final_seq = scale_output(best_p, 10.0)
    final_score = try_global_eval(final_seq)
    if np.isfinite(final_score):
        print("FINAL: n=%d, score=%.6f" % (best_p.size, final_score))
    else:
        print("FINAL: n=%d, local=%.6f (global eval failed)" % (best_p.size, actual_score_fft(best_p)))

    return final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the score."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("Exception: %s" % str(e))
        import traceback
        traceback.print_exc()
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
