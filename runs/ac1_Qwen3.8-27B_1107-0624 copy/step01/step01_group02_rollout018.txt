def smooth_step_nesterov(a, c, M, temp, step_size, momentum):
    """Nesterov smooth max gradient step."""
    n = a.size
    grad = compute_gradient_smooth_max(a, a, c, M, temp)
    # Project gradient: remove mean component (since we're on the simplex)
    grad = grad - np.mean(grad)
    # Step
    a_new = a - step_size * grad
    a_new = np.clip(a_new, 0.0, None)
    a_new = normalize_nonneg(a_new)
    return a_new

def make_initial(N, kind, rng, best_seq=None, provided=None):
    N = max(2, int(N))
    if kind == -1 and best_seq is not None:
        return resize_seq(best_seq, N)
    if kind == -2 and provided is not None:
        return resize_seq(provided, N)
    if kind in (-1, -2):
        kind = 3
    
    x = np.linspace(0, 1, N)
    if kind == 0:  # Gaussian
        mu = rng.uniform(0.2, 0.8)
        sig = rng.uniform(0.1, 0.35)
        a = np.exp(-0.5*((x-mu)/sig)**2)
    elif kind == 1:  # Exponential
        alpha = rng.choice([-3, -1.5, 0.5, 1.5, 3])
        a = np.exp(alpha * x)
    elif kind == 2:  # Beta-like
        alpha = rng.uniform(-0.5, 1.5)
        beta = rng.uniform(-0.5, 1.5)
        xs = np.linspace(1/(N+1), N/(N+1), N)
        a = xs**alpha * (1-xs)**beta
    elif kind == 3:  # Two-block
        L = N // 2
        a = np.zeros(N)
        if L > 0:
            a[:L] = (2.0/3.0)/L
        if N-L > 0:
            a[L:] = (1.0/3.0)/(N-L)
    elif kind == 4:  # U-shaped (arcsine-like)
        xs = np.linspace(0.01, 0.99, N)
        a = 1.0/(np.sqrt(xs*(1-xs)) + 0.1)
    elif kind == 5:  # Multi-block
        K = int(rng.integers(2, min(8, N//32+2)))
        lengths = (N/K * rng.dirichlet(np.ones(K))).astype(int)
        lengths = np.maximum(1, lengths)
        while lengths.sum() < N:
            lengths[rng.integers(K)] += 1
        while lengths.sum() > N:
            i = np.argmax(lengths)
            if lengths[i] > 1:
                lengths[i] -= 1
            else:
                break
        masses = rng.dirichlet(0.3*np.ones(lengths.size))
        a = np.zeros(N)
        pos = 0
        for L, p in zip(lengths, masses):
            L = int(L)
            if L > 0 and pos+L <= N:
                a[pos:pos+L] = float(p)/L
                pos += L
            elif L > 0:
                a[pos:N] = float(p)/max(1, N-pos)
                break
    elif kind == 6:  # Smooth U-shape
        a = 1.0 + 0.8*np.cos(2*np.pi*x)
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(N)
    return normalize_nonneg(np.maximum(a, 0.0))

def resize_seq(a, N):
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == N:
        return normalize_nonneg(a)
    if m < N:
        x_old = np.linspace(0, m-1, m)
        x_new = np.linspace(0, N-1, N)
        a_new = np.interp(x_new, x_old, a)
    else:
        x_old = np.linspace(0, m-1, m)
        x_new = np.linspace(0, N-1, N)
        a_interp = np.interp(x_new, x_old, a)
        # Average bins for downsampling
        a_new = np.zeros(N)
        for i in range(N):
            lo = int(i * m / N)
            hi = int((i+1) * m / N)
            a_new[i] = a[lo:hi].sum()
    return normalize_nonneg(np.maximum(a_new, 0.0))

def optimize_local(a, time_limit, rng, state):
    """Local optimization using smooth max gradient + LP + perturbations."""
    a = normalize_nonneg(a)
    n = a.size
    deadline = time.monotonic() + time_limit
    
    c = np.convolve(a, a)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best_a = a.copy()
    if best_score < state['score']:
        state['score'] = best_score
        state['seq'] = best_a.copy()
        print(f"  init score={best_score:.10f} n={n}", flush=True)
    
    # Nesterov state
    v = np.zeros(n)
    temp = 0.15 * M
    momentum = 0.8
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        M = float(c.max())
        cur = float(2.0 * n * M)
        
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            state['score'] = best_score
            state['seq'] = best_a.copy()
            no_improve = 0
            temp = max(0.01 * M, temp * 0.93)
        else:
            no_improve += 1
        
        if no_improve > 50:
            # Restart with perturbation
            a = a * np.exp(rng.normal(0, 0.15, n))
            a = normalize_nonneg(a)
            v = np.zeros(n)
            temp = 0.2 * M
            no
