"""
Step function coefficient optimization using multiplicative mirror descent
with FFT-accelerated log-sum-exp smoothing, peak-focused reweighting,
adaptive learning rates, and diverse asymmetric initialization strategies.
Key innovation: clean gradient derivation using reverse-convolution identity,
combined with geometric-decay mirror updates for robust non-negativity handling.
"""

import numpy as np
import time


def convolve_fft(a, b=None):
    """Fast linear convolution a*b (or a*a if b is None) using FFT."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * fb, size)
    return result[:na + nb - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_gradient(a, tau, top_k=None):
    """
    Compute smoothed objective gradient using the identity:
    dTmax/da[i] = 2*tau * conv(a_rev, p)[n-1+i]
    where p = softmax(c/tau) (or restricted to top-k peaks).
    
    Returns (J_smooth, grad) where grad = dJ/da.
    """
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    # Restrict to top-k peaks if specified
    m = len(c)
    use_top_k = top_k is not None and top_k < m
    if use_top_k:
        k_idx = np.argpartition(c, -top_k)[-top_k:]
        c_work = c[k_idx]
    else:
        c_work = c

    # Log-sum-exp for numerical stability
    c_max = np.max(c_work)
    shifted = (c_work - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Build full-length p for convolution
    if use_top_k:
        p_full = np.zeros(2 * n - 1)
        p_full[k_idx] = p
    else:
        p_full = p

    # Gradient: grad[i] = 2n * 2*tau * conv(a_rev, p_full)[n-1+i] / S^2 - 4n*Tmax*a[i]/S^3
    a_rev = a[::-1]
    # conv(a_rev, p_full) has length n + (2n-1) - 1 = 3n-2
    conv_size = 1
    while conv_size < 3 * n - 1:
        conv_size <<= 1
    f_arev = np.fft.rfft(a_rev, conv_size)
    f_pfull = np.fft.rfft(p_full, conv_size)
    conv_result = np.fft.irfft(f_arev * f_pfull, conv_size)[:3 * n - 2]

    # dTmax/da[i] = 2*tau * conv_result[n-1+i]
    dTmax = 2.0 * tau * conv_result[n - 1:n - 1 + n]

    grad = (4.0 * n * tau / (S * S)) * conv_result[n - 1:n - 1 + n] - \
           (4.0 * n * Tmax / (S * S * S)) * a

    return J, grad


def make_init_exponential(n, alpha=1.0):
    """Exponential decay from left to right."""
    x = np.arange(n) / (n - 1)
    return np.exp(-alpha * x)


def make_init_ramp(n, beta=1.0):
    """Linear ramp."""
    x = np.arange(n) / (n - 1)
    return 1.0 + beta * x


def make_init_asym_bump(n, center=0.3, width=0.15):
    """Gaussian bump off-center."""
    x = np.arange(n) / (n - 1)
    return np.exp(-((x - center) ** 2) / (2 * width ** 2))


def make_init_piecewise(n, left=0.7, right=0.3, transition=0.1):
    """Piecewise: high left, low right with transition."""
    x = np.arange(n) / (n - 1)
    result = np.where(x < 0.5 - transition / 2, left,
                      np.where(x > 0.5 + transition / 2, right,
                               left + (right - left) * (0.5 - np.abs(x - 0.5)) / (transition / 2)))
    return np.maximum(result, 0.01)


def make_init_sinc_window(n, bandwidth=0.3):
    """Sinc-like window (truncated)."""
    x = np.arange(n) - (n - 1) / 2.0
    return np.sinc(bandwidth * x / (n / 2.0)) ** 2


def make_init_cosine_asym(n, freq=1.0, phase=0.5):
    """Asymmetric cosine."""
    x = np.arange(n) / (n - 1)
    return 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + phase * np.pi)


def run_mirror_descent(a_init, budget_s, seed, tau_start=3.0, tau_end=0.01,
                       lr_start=0.02, top_k_start=None, top_k_end=50):
    """
    Multiplicative (mirror) gradient descent with adaptive tau and top-k focusing.
    Updates: a <- a * exp(-lr * clipped_grad), then normalize.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    a = np.maximum(a, 1e-10)

    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()
    last_improve = start_time
    iter_count = 0
    lr = lr_start
    tau = tau_start
    stall_iters = 0
    max_stall = 20000

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start / (1.0 + 0.3 * elapsed / max(budget_s, 1.0))

        # Top-k: start with all, gradually focus
        if top_k_start is None:
            top_k_start = 2 * n - 1
        top_k = max(top_k_end, int(top_k_start * (1 - frac) + top_k_end * frac))

        J, grad = smooth_gradient(a, tau, top_k=top_k if top_k < 2 * n - 1 else None)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            tau = max(tau, 1.0)
            continue

        # Clip gradient to prevent explosive updates
        grad_max = np.max(np.abs(grad))
        if grad_max > 0:
            grad_clipped = np.clip(grad, -10.0 * grad_max / max(n, 1), 10.0 * grad_max / max(n, 1))
        else:
            grad_clipped = grad

        # Multiplicative update
        update = -lr * grad_clipped
        update = np.clip(update, -2.0, 2.0)  # Safety clip
        a = a * np.exp(update)
        a = np.maximum(a, 1e-15)

        # Periodic normalization (scale invariance)
        iter_count += 1
        if iter_count % 5000 == 0:
            S = np.sum(a)
            if S > 0.01:
                a = a * (n / S)
            # Check improvement
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                last_improve = time.monotonic()
                stall_iters = 0
            else:
                stall_iters += 1
                if stall_iters > max_stall:
                    # Perturbation or restart
                    cand = best_a.copy()
                    w = max(1, n // 100)
                    s = int(rng.integers(0, n - w))
                    cand[s:s + w] *= np.exp(rng.normal(0, 0.3, w))
                    cand = np.maximum(cand, 1e-10)
                    cS = np.sum(cand)
                    if cS > 0.01:
                        cand = cand * (n / cS)
                    cJ = compute_J(cand)
                    if cJ < best_J:
                        a = cand
                        best_J = cJ
                        best_a = cand.copy()
                        stall_iters = 0
                    else:
                        a = best_a.copy() * np.exp(rng.normal(0, 0.05, n))
                        a = np.maximum(a, 1e-10)
                        a = a * (n / np.sum(a))
                        stall_iters = 0

    return best_a, best_J


def run_hill_climb(a_init, budget_s, seed):
    """
    Local search with multiplicative perturbations focused on convolution peaks.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    a = np.maximum(a, 1e-10)

    best_a = a.copy()
    best_J = compute_J(a)
    step = 0.05
    iters_without_improve = 0

    while time.monotonic() < deadline:
        # Get current convolution peak location
        c = convolve_fft(a)
        peak_idx = int(np.argmax(c))

        # Strategy 1: Perturb elements that contribute to the peak
        # peak_idx corresponds to contributions from a[j]*a[peak_idx-j]
        n_perturb = int(rng.integers(max(5, n // 50), max(10, n // 10)))
        indices = rng.integers(0, n, n_perturb)

        cand = a.copy()
        for idx in indices:
            # Elements contributing to peak: a[idx] pairs with a[peak_idx - idx]
            factor = np.exp(rng.normal(0, step))
            cand[idx] *= factor
            # Also perturb the partner if in range
            partner = peak_idx - idx
            if 0 <= partner < n and partner != idx:
                cand[partner] *= np.exp(rng.normal(0, step * 0.5))

        cand = np.maximum(cand, 1e-10)
        cS = np.sum(cand)
        if cS < 0.01:
            continue
        cand = cand * (n / cS)

        cJ = compute_J(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            iters_without_improve = 0
            step = max(0.001, step * 0.97)
        else:
            iters_without_improve += 1
            step = min(0.3, step * 1.01)
            if iters_without_improve > 5000:
                # Broader restart
                a = best_a.copy()
                w = max(10, n // 20)
                s = int(rng.integers(0, n - w))
                a[s:s + w] *= np.exp(rng.normal(0, 0.2, w))
                a = np.maximum(a, 1e-10)
                a = a * (n / np.sum(a))
                iters_without_improve = 0
                step = 0.05

    return best_a, best_J


def run_subgradient_peak(a_init, budget_s, seed):
    """
    Subgradient method focusing on the active peak.
    At each step, compute the exact subgradient (using current argmax)
    and take a step in the negative subgradient direction.
    """
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy().astype(np.float64)
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    a = np.maximum(a, 1e-10)

    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.01
    iters = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        c = convolve_fft(a)
        S = np.sum(a)
        # Subgradient of max(c): use uniform weights over all argmax elements
        max_c = np.max(c)
        # Use a soft version: top-10 peaks with equal weight
        top_indices = np.argpartition(c, -10)[-10:]
        p = np.zeros(len(c))
        p[top_indices] = 1.0 / len(top_indices)

        # Compute subgradient: grad[i] = 2*sum_k p[k]*a[k-i] * 2n/S^2 - 4n*max_c*a[i]/S^3
        a_rev = a[::-1]
        conv_size = 1
        while conv_size < 3 * n - 1:
            conv_size <<= 1
        f_arev = np.fft.rfft(a_rev, conv_size)
        f_p = np.fft.rfft(p, conv_size)
        conv_result = np.fft.irfft(f_arev * f_p, conv_size)[:3 * n - 2]

        subgrad = (4.0 * n / (S * S)) * conv_result[n - 1:n - 1 + n] - \
                  (4.0 * n * max_c / (S * S * S)) * a

        # Polyak step size approximation
        # J - J* >= <grad, a - a*>
        # Use normalized step
        grad_norm = np.sqrt(np.sum(subgrad ** 2))
        if grad_norm > 0:
            step = lr / (1.0 + iters * 1e-5)
            a = a - step * subgrad / (grad_norm / n)
            a = np.maximum(a, 1e-10)
            S = np.sum(a)
            if S > 0.01:
                a = a * (n / S)

        iters += 1
        if iters % 2000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                last_improve = time.monotonic()
            elif time.monotonic() - last_improve > 30:
                # Restart with perturbation
                a = best_a.copy()
                a *= np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-10)
                a = a * (n / np.sum(a))
                lr = lr * 0.7
                last_improve = time.monotonic()

    return best_a, best_J

    remaining = total_deadline - time.monotonic() - 5.0
    if remaining > 10.0:
        print(f"\nFinal refinement ({remaining:.1f}s) on best so far (J={best_global_score:.6f})")
        res_a, res_J = run_mirror_descent(
            best_global, remaining * 0.5, seed=seed + 99999,
            tau_start=0.5, tau_end=0.005, lr_start=0.008,
            top_k_start=200, top_k_end=20
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Refined: {val:.6f}")
        else:
            print(f"  (refinement: local={res_J:.6f}, verified={val:.6f})")

        # Second refinement with hill climbing
        remaining2 = total_deadline - time.monotonic() - 5.0
        if remaining2 > 5.0:
            res_a2, res_J2 = run_hill_climb(best_global, remaining2, seed=seed + 88888)
            val2 = float(evaluate_sequence(res_a2.tolist()))
            if np.isfinite(val2) and val2 < best_global_score:
                best_global = res_a2.copy()
                best_global_score = val2
                print(f"  HC Refined: {val2:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
