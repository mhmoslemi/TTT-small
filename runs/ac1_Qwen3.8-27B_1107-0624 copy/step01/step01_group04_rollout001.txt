"""
Search for non-negative step-function sequences minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.sum())
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def mix_greedy_uniform(n, lam):
    p = lam * greedy_normalized(n) + (1.0 - lam) * np.ones(int(n), dtype=np.float64)
    return normalize_nonneg(p)


def dirichlet_start(n, alpha, rng):

def actual_score_fft(p):

    if n == 0:

    a = np.clip(a, 0.0, 1000.0)

def compute_convolution_fft(p, L):
    Pg = np.fft.rfft(p, L)
    c = np.fft.irfft(Pg * Pg, L)[:p.size * 2 - 1].real
    return np.maximum(c, 0.0)


def adam_optimize_v2(p, budget_s, seed=0, alpha_start=30.0, alpha_end=5.0,
                     lr_base=0.04, use_restart=True, restart_interval=500):
    """
    Improved Adam optimizer with:
    - Linear interpolation of alpha from alpha_start to alpha_end
    - Cosine learning rate schedule
    - Periodic restarts from perturbed best
    - Multi-peak tracking
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = lr_base * 1.5
    elif n < 256:
        lr = lr_base
    elif n < 1024:
        lr = lr_base * 0.7
    else:
        lr = lr_base * 0.5

    mu, nu, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p_local = p.copy()
    best_f_local = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 2.0)
    last_restart = 0

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))

        # Alpha: interpolate from high to low temperature
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Learning rate: cosine schedule with slight warmup at start
        if frac < 0.05:
            lr_now = lr * (frac / 0.05)
        else:
            lr_now = lr * 0.5 * (1.0 + math.cos(math.pi * (frac - 0.05) / 0.95))
        lr_now = max(lr_now, 1e-6)

        # Compute convolution
        c = compute_convolution_fft(p, L)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f_local:
            best_f_local = f_now
            best_p_local = p.copy()

        # Soft-max weights on convolution positions
        z = alpha * (c / maxc - 1.0)
        expz = np.exp(np.clip(z, -100.0, 0.0))
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        # d/dp [sum_k w_k * (p*p)_k] = 2 * w_rev * p (convolution)
        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(np.fft.rfft(p, L) * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        if not np.all(np.isfinite(grad_p)):
            break
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        # Adam update
        m = mu * m + (1.0 - mu) * grad_x
        v = nu * v + (1.0 - nu) * (grad_x * grad_x)
        mhat = m / (1.0 - mu ** t)
        vhat = v / (1.0 - nu ** t)
        step = lr_now * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -30.0, 30.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

        # Periodic restart
        if use_restart and (t - last_restart) >= restart_interval:
            last_restart = t
            p = random_perturb(best_p_local, np.random.default_rng(seed + t), 0.06)
            x = np.log(p + 1e-300)
            x -= x.mean()
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            t = 0

    final_f = actual_score_fft(p)
    if final_f < best_f_local:
        best_f_local = final_f
        best_p_local = p.copy()
    return best_p_local, best_f_local


def lp_improve_v2(p, budget_s, max_rows=None):
    """
    Cutting-plane LP direction method.
    maximizes sum(g) subject to (f*g)_k <= M for selected k, g >= 0.
    Then blends: f_new = (1-tau)*f + tau*g, rescaled.
    """
    if _scipy_opt is None:
        return p, actual_score_fft(p)

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n < 4:
        return p, actual_score_fft(p)
    budget_s = float(budget_s)
    if budget_s < 30.0:
        return p, actual_score_fft(p)

    if max_rows is None:
        if n > 4000:
            max_rows = 800
        elif n > 1500:
            max_rows = 900
        else:
            max_rows = 1000

    max_elements = 5000000
    while max_rows * n > max_elements:
        max_rows = max(128, int(max_rows * 0.85))
    max_rows = max(128, min(int(max_rows), max_elements // max(1, n)))

    S0 = math.sqrt(2.0 * n)
    f = normalize_nonneg(p) * S0
    cur_score = actual_score_fft(f)
    if not np.isfinite(cur_score):
        return p, cur_score

    best_f = f.copy()
    best_score = float(cur_score)
    M = cur_score  # target convolution level

    end = time.monotonic() + max(0.0, budget_s - 5.0)

    r0 = min(max_rows, max(64, max_rows // 3))
    rows0 = np.unique(np.linspace(0, 2 * n - 2, r0, dtype=int).astype(int))
    rows = [int(k) for k in rows0.tolist()]
    row_set = set(rows)

    L_conv = next_power_of_two(2 * n - 1)

    it = 0
    while time.monotonic() < end:
        it += 1
        rows_arr = np.asarray(rows, dtype=int)
        nr = int(rows_arr.size)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, M, dtype=np.float64)
        cobj = -np.ones(n, dtype=np.float64)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.3))
        res = None
        try:
            res = _scipy_opt.linprog(
                cobj, A_ub=A, b_ub=b,
                bounds=(0.0, None), method="highs",
                options={"time_limit": float(tl)}
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False) or getattr(res, "x", None) is None:
            break

        g = np.asarray(res.x, dtype=np.float64).ravel()
        g = np.clip(g, 0.0, None)
        sg = float(g.sum())
        if not np.isfinite(sg) or sg <= 1e-12:
            break

        # Check full constraint violations
        Pg = np.fft.rfft(g, L_conv)
        Pf = np.fft.rfft(f, L_conv)
        conv = np.fft.irfft(Pf * Pg, L_conv)[:2 * n - 1].real
        conv = np.maximum(conv, 0.0)
        tol = 1e-8 + 1e-8 * M
        violations = conv - M
        max_viol = float(np.max(violations)) if violations.size > 0 else 0.0

        if max_viol > tol:
            added = 0
            if len(rows) < max_rows:
                order = np.argsort(violations)[::-1]
                n_add = min(128, max(16, (max_rows - len(rows)) // 2))
                for ix in order[:n_add]:
                    ix = int(ix)
                    if violations[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
            if added == 0 and max_viol > 10.0 * tol:
                break
            if added > 0:
                continue

        # Scale g to match normalization
        g = g * (S0 / sg)

        # Try blending
        accepted = None
        accepted_score = np.inf
        for tau in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
            fn = (1.0 - tau) * f + tau * g
            fn = np.clip(fn, 0.0, None)
            sn = float(fn.sum())
            if sn <= 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = actual_score_fft(fn)
            if np.isfinite(sc) and sc < accepted_score:
                accepted_score = float(sc)
                accepted = fn.copy()

        if accepted is None or not np.isfinite(accepted_score):
            break
        if accepted_score >= best_score:
            break

        f = accepted
        cur_score = accepted_score
        M = cur_score
        best_f = f.copy()
        best_score = cur_score

        if it % 3 == 0:
            print("  LP n=%d iter=%d score=%.6f rows=%d" % (n, it, best_score, len(rows)))

    final_score = actual_score_fft(best_f)
    if final_score < best_score:
        best_score = final_score
    return normalize_nonneg(best_f), best_score


def multi_start_adam(base_p, budget_s, rng, n_restarts=3):
    """
    Run Adam from multiple perturbations of base_p and return the best.
    """
    best_p = base_p.copy()
    best_f = actual_score_fft(base_p)
    n = base_p.size
    per_restart = max(5.0, budget_s / max(1, n_restarts) - 2.0)

    for i in range(n_restarts):
        rem = budget_s - sum(0 for _ in range(i))  # approximate
        if i == 0:
            start = base_p.copy()
        else:
            sigma = 0.05 + 0.05 * i
            start = random_perturb(best_p, rng, sigma)
        alloc = min(per_restart, budget_s)
        p_out, f_out = adam_optimize_v2(start, alloc, seed=int(time.time()) + i)
        if np.isfinite(f_out) and f_out < best_f:
            best_f = f_out
            best_p = p_out.copy()
        budget_s -= alloc

    return best_p, best_f


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    print("=== Starting search, budget = %.0fs ===" % budget_s)

    # Phase 1: Quick baselines and known-good point (20s)
    phase1_end = t0 + 20.0

    # Known good point
    h = globals().get("height_sequence_1", None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size:
                best_p, best_score = record_candidate(arr, best_p, best_score, best_by_n)
                print("  height_sequence_1: n=%d, score=%.6f" % (arr.size, best_score))
        except Exception:
            pass

    # Quick baselines at key lengths
    base_lengths = [256, 384, 512, 768, 1024]
    for n in base_lengths:
        if time.monotonic() >= phase1_end:
            break
        p = greedy_normalized(n)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = power_normalized(n, 0.45)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = power_normalized(n, 0.50)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)
        p = power_normalized(n, 0.55)
        best_p, best_score = record_candidate(p, best_p, best_score, best_by_n)

    print("  After baselines: best_score=%.6f" % best_score)

    # Phase 2: Main Adam optimization (majority of budget)
    # Focus on the most promising lengths
    focus_lengths = [384, 256, 512, 320, 448, 768, 400, 480]
    # If we found something at a specific length, prioritize it
    if best_p is not None:
        bn = int(best_p.size)
        if bn not in focus_lengths:
            focus_lengths.insert(0, bn)
        else:
            idx = focus_lengths.index(bn)
            focus_lengths = [focus_lengths[idx]] + [x for x in focus_lengths if x != bn]

    # Deduplicate and cap
    seen = set()
    unique_lengths = []
    for ln in focus_lengths:
        if ln not in seen and ln >= 16:
            seen.add(ln)
            unique_lengths.append(ln)

    phase2_alloc = max(0.0, deadline - time.monotonic() - 300.0)
    print("  Phase 2 alloc: %.0fs, lengths: %s" % (phase2_alloc, unique_lengths[:6]))

    # Distribute time among lengths (weight towards shorter/more promising)
    weights = []
    for ln in unique_lengths[:8]:
        if ln == 384:
            weights.append(3.0)
        elif ln == 256:
            weights.append(2.5)
        elif ln == 512:
            weights.append(2.0)
        elif ln <= 500:
            weights.append(1.8)
        elif ln <= 800:
            weights.append(1.2)
        else:
            weights.append(0.8)

    total_w = sum(weights)
    for i, ln in enumerate(unique_lengths[:8]):
        rem = deadline - time.monotonic() - 300.0
        if rem < 20.0:
            break
        alloc = phase2_alloc * (weights[i] / total_w)
        alloc = min(alloc, rem - 10.0)
        if alloc < 10.0:
            continue

        # Get starting point
        rec = best_by_n.get(int(ln))
        if rec is not None:
            base = rec[1].copy()
        else:
            # Try multiple starts
            base = greedy_normalized(ln)

        # Multi-start Adam
        n_restarts = 2 if alloc > 60 else 1
        p_out, f_out = multi_start_adam(base, alloc, rng, n_restarts=n_restarts)
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("  Adam n=%d: score=%.6f (best=%.6f)" % (ln, f_out, best_score))

    # Phase 3: LP refinement (if available)
    if _scipy_opt is not None:
        rem = deadline - time.monotonic() - 150.0
        print("  Phase 3 (LP): rem=%.0fs" % rem)
        if rem > 60.0:
            # LP on the best candidate
            lp_targets = []
            if best_p is not None:
                lp_targets.append((best_score, int(best_p.size), best_p.copy()))
            for n, rec in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
                if len(lp_targets) < 3 and rec[1].size != (best_p.size if best_p is not None else -1):
                    lp_targets.append((rec[0], n, rec[1].copy()))

            for sc, n, base in lp_targets:
                rem = deadline - time.monotonic() - 100.0
                if rem < 45.0:
                    break
                alloc = min(180.0, rem - 10.0)
                print("  LP n=%d, start_score=%.6f, alloc=%.0fs" % (n, sc, alloc))
                p_out, f_out = lp_improve_v2(base, alloc)
                best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
                print("  LP result n=%d: score=%.6f (best=%.6f)" % (n, f_out, best_score))

    # Phase 4: Final polishing
    rem = deadline - time.monotonic()
    print("  Phase 4 (polish): rem=%.0fs, best=%.6f" % (rem, best_score))

    if best_p is not None and rem > 60.0:
        # Fine-tune with small alpha (focus on the peak)
        p0 = normalize_nonneg(best_p)
        alloc = min(rem - 20.0, 120.0)
        p_out, f_out = adam_optimize_v2(p0, alloc, seed=int(seed) + 777,
                                         alpha_start=50.0, alpha_end=20.0,
                                         lr_base=0.015, use_restart=False)
        best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
        print("  Polish 1: score=%.6f" % best_score)

        rem = deadline - time.monotonic()
        if rem > 40.0:
            p0 = random_perturb(best_p, rng, 0.03)
            alloc = min(rem - 10.0, 80.0)
            p_out, f_out = adam_optimize_v2(p0, alloc, seed=int(seed) + 888,
                                             alpha_start=40.0, alpha_end=15.0,
                                             lr_base=0.01, use_restart=False)
            best_p, best_score = record_candidate(p_out, best_p, best_score, best_by_n, f_out)
            print("  Polish 2: score=%.6f" % best_score)

    # Phase 5: Validate with evaluate_sequence
    candidates = []
    if best_p is not None:
        candidates.append(best_p)
    for n, rec in sorted(best_by_n.items(), key=lambda x: x[1][0])[:3]:
        candidates.append(rec[1])

    final_best_p = best_p
    final_best_score = best_score

    for cand in candidates:
        rem = deadline - time.monotonic()
        if rem < 10.0:
            break
        try:
            seq = scale_output(cand, 10.0)
            gs = try_global_eval(seq)
            if np.isfinite(gs) and gs < final_best_score:
                final_best_score = gs
                final_best_p = np.asarray(cand, dtype=np.float64).ravel()
                print("  Global eval improved: %.6f" % final_best_score)
        except Exception:
            pass

    if final_best_p is None:
        final_best_p = greedy_normalized(384)
        final_best_score = actual_score_fft(final_best_p)

    print("=== Final score: %.6f, n=%d ===" % (final_best_score, final_best_p.size))
    print("  Total time: %.1fs" % (time.monotonic() - t0))

    return scale_output(final_best_p, 10.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the evaluation."""
    try:
        return main_search(int(seed), float(budget_s))
    except Exception as e:
        print("  Exception in main_search: %s" % str(e))
        # Fallback: return a reasonable default
        p = greedy_normalized(384)
        return scale_output(p, 10.0)
