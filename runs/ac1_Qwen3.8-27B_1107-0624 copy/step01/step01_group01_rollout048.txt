import numpy as np
import time
import math

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < -1e-9:
        c = np.maximum(c, 0.0)
    return c

def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    mval = float(c.max())
    return float(2.0 * n * mval / (s * s))

def soft_max_grad(c, beta):
    """Compute soft max value and gradient weights."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    w = ex / s
    val = mval + beta * np.log(s)
    return val, w

def corr_weights(w, p):
    """h[i] = sum_j w[i+j] * p[j], for i=0..n-1."""
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n-1:2*n-1]

def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -50.0, 50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-12
    p = p / p.sum()
    return p

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())

def make_ramp(n, power=1.0, offset=0.0):
    """Create a ramp-like profile: (j+1+offset)^power normalized."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0 + offset) ** power
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    return a

            m.fill(0); v.fill(0); t = 0

        v = b2 * v + (1-b2) * (gz * gz)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)

            z = np.clip(z, -50.0, 50.0)
            z -= float(np.mean(z))
            p = softmax_from_z(z)
            c = full_convolve_fft(p)
            maxc = float(c.max())
            beta = max(1e-10, 0.3 * maxc)
            m.fill(0); v.fill(0); t = 0
            lr = base_lr
            no_imp = 0

    return best_p, best_score

def param_adam_asym(theta0, basis, deadline, rng, lr=0.25):
    """Adam in low-dim asymmetric parameter space."""
    basis = np.asarray(basis, dtype=np.float64)
    theta = np.asarray(theta0, dtype=np.float64).ravel().copy()
    dim = theta.size
    n = basis.shape[0]

    b1, b2, eps = 0.90, 0.999, 1e-8
    m = np.zeros(dim)
    v = np.zeros(dim)
    t = 0

    p = softmax_from_z(basis @ theta)
    c = full_convolve_fft(p)
    maxc = float(c.max())
    beta = max(1e-12, 0.3 * maxc)

    best_theta = theta.copy()
    best_p = p.copy()
    best_score = float(2.0 * n * maxc)
    no_imp = 0
    base_lr = lr

    while time.monotonic() < deadline:
        _, w = soft_max_grad(c, beta)
        gp = 2.0 * corr_weights(w, p)
        vp = gp + 1e-7 * p
        dot = float(np.dot(p, vp))
        gz = p * (vp - dot)
        gtheta = basis.T @ gz

        if not np.all(np.isfinite(gtheta)):
            theta = best_theta.copy(); m.fill(0); v.fill(0); t = 0
            continue

        t += 1
        m = b1*m + (1-b1)*gtheta
        v = b2*v + (1-b2)*(gtheta*gtheta)
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        theta = theta - lr * mh/(np.sqrt(vh)+eps)

        if not np.all(np.isfinite(theta)):
            theta = best_theta.copy(); m.fill(0); v.fill(0); t = 0
            continue

        theta = np.clip(theta, -8.0, 8.0)
        p = softmax_from_z(basis @ theta)
        c = full_convolve_fft(p)
        maxc = float(c.max())
        score = float(2.0 * n * maxc)

        if np.isfinite(score):
            if score < best_score - 1e-13:
                best_score = score
                best_p = p.copy()
                best_theta = theta.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 400 == 0:
                    lr = max(0.05, lr * 0.95)
        else:
            theta = best_theta.copy(); m.fill(0); v.fill(0); t = 0
            continue

        if t % 100 == 0:
            beta = max(1e-12, beta * 0.7)

        if no_imp > max(200, 5*dim):
            theta = best_theta + 0.5 * rng.normal(size=dim)
            theta = np.clip(theta, -8.0, 8.0)
            p = softmax_from_z(basis @ theta)
            c = full_convolve_fft(p)
