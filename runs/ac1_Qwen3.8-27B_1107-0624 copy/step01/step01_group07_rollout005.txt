def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def smooth_subgrad(a, beta=10.0):
    """Compute smooth subgradient of max(conv(a,a)) using log-sum-exp."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    # Softmax weights (numerically stable)
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / exp_c.sum()
    # Subgradient: subgrad[i] = sum_k weights[k] * a[k-i]
    subgrad = np.correlate(weights, a, 'full')[:n]
    return subgrad, c_max

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

            best_sc = compute_objective(a)

    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    return exp_x / exp_x.sum()

def smooth_peak_and_subgrad(a, beta=20.0):
    """Compute smoothed peak value and subgradient."""
    c = np.convolve(a, a)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / exp_c.sum()
    val = c_max + np.log(exp_c.sum()) / beta
    subgrad = np.correlate(weights, a, 'full')[:len(a)]
    return val, subgrad, c

def optimize_lbfgs(a, deadline, rng, beta=20.0):
    """Optimize using L-BFGS-B with softmax parameterization."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    # Initialize x from a
    x = np.log(a + 1e-30)
    
    best_a = a.copy()
    best_score = compute_objective(a)
    
    def obj_and_grad(x):
        a_local = softmax(x)
        val, subgrad, _ = smooth_peak_and_subgrad(a_local, beta)
        mean_sg = np.dot(subgrad, a_local)
        grad = a_local * (subgrad - mean_sg)
        return val, grad
    
    # Use L-BFGS-B
    try:
        result = minimize(obj_and_grad, x, method='L-BFGS-B',
                         options={'maxiter': 500, 'gtol': 1e-8,
                                  'maxfun': 5000, 'iprint': 0})
        if result.success or result.fun < best_score / (2.0 * n):
            a_new = softmax(result.x)
            sc = compute_objective(a_new)
            if sc < best_score:
                best_a = a_new
                best_score = sc
                a = a_new
    except Exception as e:
        print(f"    L-BFGS error: {e}")
    
    # Check time and possibly run again with higher beta
    if time.monotonic() < deadline:
        beta2 = min(100.0, beta * 3)
        x = np.log(a + 1e-30)
        try:
            result = minimize(obj_and_grad, x, method='L-BFGS-B',
                             options={'maxiter': 500, 'gtol': 1e-9,
                                      'maxfun': 5000, 'iprint': 0})
            if result.success or result.fun < best_score / (2.0 * n):

                sc = compute_objective(a_new)

def optimize_subgrad(a, deadline, rng, beta_start=5.0, beta_end=100.0):
    """Smooth subgradient descent with exponentiated gradient updates."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.05
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / 3000)
        beta = beta_start + (beta_end - beta_start) * progress ** 0.5
        
        val, subgrad, c = smooth_peak_and_subgrad(a, beta)
        
        # Exponentiated gradient update
        sg_max = subgrad.max()
        if sg_max < 1e-15:
            break
        a_new = a * np.exp(-lr * subgrad / sg_max)
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            lr *= 0.5
            continue
        a_new /= S_new
        
        sc = compute_objective(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.3, lr * 1.01)
        else:
            stall += 1
            lr *= 0.97
            if stall >= 80:
                # Perturbation
                strategy = rng.random()
                if strategy < 0.4:
                    a_pert = best_a * (1.0 + 0.2 * rng.standard_normal(n))
                elif strategy < 0.7:
                    a_pert = best_a * rng.uniform(0.3, 2.0, n)
                elif strategy < 0.85:
                    # Mix with a different smooth function
                    alt = make_cosine(n) if rng.random() < 0.5 else make_gaussian(n)
                    mix = rng.uniform(0.3, 0.7)
                    a_pert = mix * best_a + (1 - mix) * alt
                else:
                    a_pert = np.abs(rng.standard_normal(n))
                
                a_pert = np.clip(a_pert, 1e-15, None)
                a_pert /= a_pert.sum()
                sc_pert = compute_objective(a_pert)
                
                if sc_pert < best_score:
                    best_a = a_pert.copy()
                    best_score = sc_pert
                    a = a_pert
                elif rng.random() < 0.25:
                    a = a_pert
                else:
                    a = best_a.copy()
                stall = 0
                lr = 0.05
        
        if stall > 500:
            break
    
    return best_a, best_score

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_direction(a, M):
    """Solve LP to find direction g that maximizes sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None
