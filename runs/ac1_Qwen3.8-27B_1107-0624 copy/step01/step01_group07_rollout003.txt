```python
    # Unless j-i = i (i.e., j = 2i), in which case both terms are the same:
    # d/da[i] = a[i] (from first term) + a[i] (from second term) = 2*a[i]... 
    # Actually no, if j = 2i, then k = i and j-k = i, so it's a[i]*a[i] = a[i]^2
    # d/da[i] of a[i]^2 = 2*a[i]. And 2*a[j-i] = 2*a[i]. So it's consistent!
    
    # So the gradient of c[j] w.r.t. a is: grad[j][i] = 2*a[j-i] if 0<=j-i<n, else 0.
    
    # This means the subgradient of max_c is a weighted combination of shifted versions of a.
    
    # For efficiency, we can compute:
    # subgrad[i] = sum_{j: active} w_j * 2 * a[j-i] (if 0<=j-i<n)
    
    # This is like a weighted sum of shifted copies of a!
```

```python
"""
Hybrid optimization combining LP-based direction finding, subgradient methods,
and simulated annealing to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses diverse initializations, adaptive time allocation, and multiple optimization
strategies to escape local minima.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def subgradient_of_conv(a, active_mask):
    """Compute subgradient of max(conv(a,a)) given active lags.
    
    grad of c[j] w.r.t. a[i] = 2*a[j-i] if 0<=j-i<n, else 0.
    """
    n = len(a)
    subgrad = np.zeros(n)
    active_indices = np.where(active_mask)[0]
    for j in active_indices:
        # For each i in [0, n-1], contribution is 2*a[j-i] if 0<=j-i<n
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        if lo <= hi:
            k = j - np.arange(lo, hi + 1)
            subgrad[lo:hi + 1] += 2.0 * a[k]
    return subgrad


def lp_optimize(a, deadline, rng, verbose=False):
    """LP-based optimization. Returns (best_a, best_score)."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        g = lp_solve(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
            step = min(0.4, step * 1.03)
        else:
            stall += 1
            step = max(0.005, step * 0.75)
            if stall % 15 == 0 and stall > 15:
                noise = np.abs(rng.normal(0, 0.1, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 0.0, None)

                    a_new /= a_new.sum()
                    sc = compute_objective(a_new)
                    if sc < best_score:
                        best_a = a_new
                        best_score = sc
                        a = a_new
                        stall = 0
                    else:

                            a = a_new
                else:
                    break
            if stall > 120:
                break

        if verbose and it % 200 == 0:
            print(f"      LP iter {it}, score={best_score:.6f}, step={step:.4f}")

    return best_a, best_score


def subgradient_optimize(a, deadline, rng, verbose=False):
    """Subgradient method with momentum for the max-conv objective."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.05
    beta = 0.7
    momentum = np.zeros(n)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        max_c = c.max()

        # Use multiple active lags for smoother subgradient
        threshold = max_c * 0.95
        active_mask = c >= threshold
        n_active = max(1, active_mask.sum())

        subgrad = subgradient_of_conv(a, active_mask) / n_active
        # Normalize subgradient
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm > 1e-15:
            subgrad /= sg_norm

        momentum = beta * momentum + subgrad
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            lr *= 0.5
            continue
        a_new /= a_new.sum()

        sc = compute_objective(a_new)
        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.2, lr * 1.01)
        else:
            # Accept with probability based on how much worse
            delta = sc - best_score
            accept_prob = np.exp(-delta * 5.0) if delta < 0.1 else 0.0
            if rng.random() < accept_prob:
                a = a_new
            else:
                a = a_new  # still move to explore
            stall += 1
            if stall % 20 == 0:
                lr = max(0.001, lr * 0.9)
            if stall > 100:
                a = best_a.copy()
                lr = max(0.001, lr * 0.5)
                momentum *= 0.5
                stall = 50
        if stall > 300:
            break

        if verbose and it % 200 == 0:
            print(f"      SG iter {it}, score={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score


def multiplicative_optimize(a, deadline, rng, verbose=False):
    """Multiplicative update method: reduce elements that contribute to peaks."""
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        # Compute contribution of each a[i] to peaks
        threshold = 0.9 * M
        peak_mask = c > threshold
        peak_indices = np.where(peak_mask)[0]

        contrib = np.zeros(n)
        for i in range(n):
            js = peak_indices - i
            valid = (js >= 0) & (js < n)
            if valid.any():
                contrib[i] = a[js[valid]].sum()
            js2 = peak_indices + i  # from the other side
            valid2 = (js2 < 2 * n - 1) & (js2 >= n)
            if valid2.any():
                k = js2[valid2] - n + 1
                valid_k = (k >= 0) & (k < n)
                if valid_k.any():
                    contrib[i] += a[k[valid_k]].sum() * 0.5

        # Multiplicative update
        avg_c = contrib.mean() + 1e-15
        update = np.exp(-lr * contrib / avg_c)
        a_new = a * update
        a_new = np.clip(a_new, 1e-15, None)
        a_new /= a_new.sum()
        sc = compute_objective(a_new)

        if sc < best_score - 1e-12:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
            lr = min(0.3, lr * 1.02)
        else:
            stall += 1
            lr = max(0.005, lr * 0.9)
            if stall % 20 == 0 and stall > 20:
                noise = np.abs(rng.normal(0, 0.05, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.clip(a_new, 1e-15, None)
                a_new /= a_new.sum()
                sc = compute_objective(a_new)
                if sc < best_score:
                    best_a = a_new
                    best_score = sc
                    a = a_new
                    stall = 0
                else:
                    a = a_new
            if stall > 150:
                a = best_a.copy()
                stall = 75
        if stall > 400:
            break

        if verbose and it % 200 == 0:
            print(f"      MU iter {it}, score={best_score:.6f}, lr={lr:.5f}")

    return best_a, best_score


def anneal_optimize(a, deadline, rng, verbose=False):
    """Simulated annealing with diverse moves."""
    n = len(a)
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    T = 0.05
    cool = 0.995
    stall = 0
    it = 0

    while time.monotonic() < deadline and T > 0.001:
        it += 1
        # Choose a random move
        move_type = rng.integers(0, 5)

        if move_type == 0:
            # Gaussian perturbation
            a_new = a + rng.normal(0, T, n)
        elif move_type == 1:
            # Multiplicative perturbation
            a_new = a * np.exp(rng.normal(0, T, n))
        elif move_type == 2:
            # Local swap: pick two positions and adjust
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            a_new = a.copy()
            delta = rng.uniform(-T, T) * a.max()
            a_new[i] = max(0, a[i] + delta)
            a_new[j] = max(0, a[j] - delta)
        elif move_type == 3:
            # Segment perturbation
            seg_len = rng.integers(5, max(6, n // 4))
            start = rng.integers(0, max(1, n - seg_len))
            a_new = a.copy()
            a_new[start:start + seg_len] += rng.normal(0, T * 0.5, seg_len)
        else:
            # Mirror perturbation: slightly break symmetry
            half = n // 2
            a_new = a.copy()
            pert = rng.normal(0, T * 0.3, half)
            a_new[:half] += pert
            a_new[half:] -= pert[:min(half, n - half)][::-1]

        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() < 1e-10:
            continue
        a_new /= a_new.sum()
        sc = compute_objective(a_new)

        delta = sc - best_score
        if sc < best_score:
            best_a = a_new
            best_score = sc
            a = a_new
            stall = 0
        else:
            if delta < 0.01 and rng.random() < np.exp(-delta / T):
                a = a_new
            else:
                a = a_new if rng.random() < 0.3 else best_a
            stall += 1

        T *= cool
        if stall > 50:
            T = max(0.01, T * 2)
            stall = 0

        if verbose and it % 500 == 0:
            print(f"      Anneal iter {it}, score={best_score:.6f}, T={T:.5f}")

    return best_a, best_score


def make_cosine(n):
    """Raised cosine window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - (np.arange(n) - c) ** 2 / (c + 1.0) ** 2)


def make_bell(n, p=3):
    """Flat-top-like window: (1 - |2x/n - 1|^p) clipped."""
    x = np.linspace(0, 1, n)
    return np.maximum(0.0, 1.0 - np.abs(2 * x - 1) ** p)


def make_asymmetric(n, rng):
    """Asymmetric window to break palindrome constraint."""
    base = make_cosine(n)
    # Add asymmetry
    ramp = np.linspace(0.5, 1.5, n)
    a = base * ramp
    a = np.abs(a + rng.normal(0, 0.1, n))
    return a

        elif init_name == 'best' and best_seq is not None:

                return best_seq.copy()
            elif len(best_seq) > n:
                return best_seq[:n].copy()
            else:
                return np.concatenate([best_seq, np.zeros(n - len(best_seq))])
        elif init_name == 'cosine':
            return make_cosine(n)
        elif init_name == 'gaussian':
            return make_gaussian(n)
        elif init_name == 'parabolic':
            return make_parabolic(n)
        elif init_name == 'bell':
            return make_bell(n)
        elif init_name == 'asymmetric':
            return make_asymmetric(n, rng)
        elif init_name == 'random':
            return np.abs(rng.normal(1, 0.5, n))
        else:
            return np.ones(n)

    for idx, (n, init_name, method, w) in enumerate(configs):
        if time.monotonic() > deadline:
            break

        config_deadline = min(start_time + time_allocs[idx], deadline)
        if config_deadline - time.monotonic() < 5.0:
            continue

        try:
            a = get_init(n, init_name)
        except Exception:
            a = np.ones(n)

        a = np.clip(a, 1e-15, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()

        remaining = deadline - time.monotonic()
        print(f"[{idx+1}/{len(configs)}] n={n}, init={init_name}, method={method}, time_left~{remaining:.0f}s")

        try:
            if method == 'lp' and n <= 2048:
                a_opt, score = lp_optimize(a, config_deadline, rng)
            elif method == 'sg':
                a_opt, score = subgradient_optimize(a, config_deadline, rng)
            elif method == 'mu':
                a_opt, score = multiplicative_optimize(a, config_deadline, rng)
            elif method == 'anneal':
                a_opt, score = anneal_optimize(a, config_deadline, rng)
            else:
                a_opt, score = lp_optimize(a, config_deadline, rng)
        except Exception as e:
            print(f"      Error: {e}")
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  => NEW BEST: score={best_score:.6f} (n={n}, {init_name}/{method})")
        else:
            print(f"  => score={score:.6f} (best: {best_score:.6f})")

    # Final polish phase: apply best method to best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\nFinal polish on best (score={best_score:.6f}, n={len(best_seq)})...")
        polish_deadline = deadline

        # Try LP polish
        if len(best_seq) <= 2048:
            a_p, score_p = lp_optimize(best_seq, polish_deadline, rng)
            if score_p < best_score:
                best_score = score_p
                best_seq = a_p
                print(f"  After LP polish: {best_score:.6f}")

        # Try subgradient polish on result
        if time.monotonic() < deadline - 10:
            a_s, score_s = subgradient_optimize(best_seq, deadline, rng)
            if score_s < best_score:
                best_score = score_s
                best_seq = a_s
                print(f"  After SG polish: {best_score:.6f}")

        # Try multiplicative polish
        if time.monotonic() < deadline - 10:
            a_m, score_m = multiplicative_optimize(best_seq, deadline, rng)
            if score_m < best_score:
                best_score = score_m
                best_seq = a_m
                print(f"  After MU polish: {best_score:.6f}")

    if best_seq is None:
        best_seq = make_cosine(256)
        best_score = compute_objective(best_seq)

    # Verify with evaluate_sequence
    final_check = evaluate_sequence([float(x) for x in best_seq])
    if np.isfinite(final_check) and final_check < best_score:
        best_score = final_check
    elif not np.isfinite(final_check):
        print("WARNING: evaluate_sequence returned inf, adjusting...")
        best_seq = np.clip(best_seq, 0, 1000)
        if best_seq.sum() < 0.01:
            best_seq = best_seq * (0.01 / max(1e-10, best_seq.sum()))
        final_check = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(final_check):
            best_score = final_check

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")
    print(f"evaluate_sequence confirms: {final_check:.6f}")

    return [float(x) for x in best_seq]
```
