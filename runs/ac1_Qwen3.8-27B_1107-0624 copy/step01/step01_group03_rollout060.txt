def peak_penalty_objective(s, t, p=2):
    """Penalty for convolution values exceeding t."""
    conv = np.convolve(s, s)
    excess = np.maximum(0, conv - t)
    return np.sum(excess**p)

def peak_penalty_gradient(s, t, p=2):
    """Gradient of peak_penalty_objective."""
    conv = np.convolve(s, s)
    excess = np.maximum(0, conv - t)
    # d/ds[i] of conv[k] = 2*s[k-i] (as derived earlier)
    n = len(s)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if excess[k] > 0:
            weight = p * excess[k]**(p-1)
            for i in range(max(0, k-n+1), min(n, k+1)):
                grad[i] += 2.0 * weight * s[k-i]
    return grad

import numpy as np
from scipy.optimize import linprog
import time

# Time management
TOTAL_BUDGET = 1800

def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def build_conv_matrix(s):
    """Build convolution matrix A where A[k,j] = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = s
    return A

def solve_lp_direction(s, M, time_limit=2.0):
    """Solve LP: max sum(b) s.t. conv(s,b) <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def line_search_golden(s, b, current_score, tol=1e-10):
