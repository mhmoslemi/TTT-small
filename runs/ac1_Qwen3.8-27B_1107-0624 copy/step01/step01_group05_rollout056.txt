"""
LP-iterative optimizer for minimum peak self-convolution of non-negative sequences.
Minimizes 2*n*max(conv(p,p))/(sum(p))^2 by normalizing to sum(p)=1 and
iteratively solving LPs to find optimal descent directions, combined with
subgradient polishing and smooth-refinement passes.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse
try:
    from scipy.optimize import minimize as _scipy_minimize
except ImportError:
    _scipy_minimize = None

N = 4096
_FFT_LEN = 2 * N - 1

_g_best_score = float('inf')
_g_best_p = None

# Precomputed LP matrix structure
_cols_pre = np.repeat(np.arange(N, dtype=np.int64), N)
_rows_pre = _cols_pre + np.tile(np.arange(N, dtype=np.int64), N)
# Precompute sorted CSR structure
_sort_idx = np.lexsort((_cols_pre, _rows_pre))
_rows_sorted = _rows_pre[_sort_idx]
_cols_sorted = _cols_pre[_sort_idx]
_indptr = np.zeros(2 * N, dtype=np.int64)
_indptr[1:] = np.bincount(_rows_squared if False else _rows_sorted, minlength=2 * N - 1).cumsum()
_p_perm = _sort_idx % N


def _make_A(p):
    """Build constraint matrix for LP: A[k,j] = p[k-j] if 0<=k-j<N."""
    data = p[_p_perm]
    return sparse.csr_matrix((data, _cols_sorted, _indptr), shape=(2 * N - 1, N))


def _update_best(p, score):
    global _g_best_score, _g_best_p
    if np.isfinite(score) and score < _g_best_score:
        _g_best_score = float(score)
        _g_best_p = np.array(p, dtype=np.float64, copy=True)


def _self_conv(p):
    f = np.fft.rfft(p, _FFT_LEN)
    c = np.fft.irfft(f * f, _FFT_LEN).real
    np.maximum(c, 0.0, out=c)
    return c


def _score(p):
    return 2.0 * N * float(np.max(_self_conv(p)))


def _lp_solve(p, M, time_limit=20.0):
    """Solve: max sum(g) s.t. conv(p,g)_k <= M for all k, g >= 0."""
    A = _make_A(p)
    c_obj = -np.ones(N)
    b_ub = np.full(2 * N - 1, M)
    bounds = [(0.0, None)] * N
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _blend(p, q, n_coarse=100, n_fine=40):
    """Find best t in [0,1]: minimize score((1-t)*p + t*q)."""
    best_t = 0.0
    best_s = _score(p)
    best_p = p

    ts = np.linspace(0.0, 1.0, n_coarse)
    for t in ts:
        pt = (1.0 - t) * p + t * q
        s = _score(pt)
        if s < best_s - 1e-15:
            best_s = s
            best_t = t
            best_p = pt

    if 0.0 < best_t < 1.0:
        dt = 1.0 / n_coarse
        lo = max(0.0, best_t - 2.0 * dt)
        hi = min(1.0, best_t + 2.0 * dt)
        ts2 = np.linspace(lo, hi, n_fine)
        for t in ts2:
            pt = (1.0 - t) * p + t * q
            s = _score(pt)
            if s < best_s - 1e-15:
                best_s = s
                best_t = t
                best_p = pt

    return best_p, best_s


def _subgradient_step(p, step, K=32):
    """One subgradient step on max(conv(p,p))."""
    c = _self_conv(p)
    cmax = float(np.max(c))
    idx = np.argpartition(c, -K)[-K:]
    w = c[idx] + 1e-15
    w /= np.sum(w)
    L = 2 * _FFT_LEN
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    cc = np.fft.irfft(np.conj(fw) * fp, L).real
    grad = 2.0 * cc[:N]
    p_new = p - step * grad
    p_new = np.maximum(p_new, 0.0)
    s = np.sum(p_new)
    if s < 1e-10:
        return p
    return p_new / s


def _probs_from_logits(z):
    """Convert logits z (length N-1) to probability vector p (length N)."""
    m = max(0.0, float(np.max(z))) if z.size else 0.0
    expz = np.exp(np.clip(z - m, -50, 50))
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p


def _logits_from_p(p):
    p = np.maximum(p, 1e-30)
    return np.clip(np.log(p[1:]) - np.log(p[0]), -40.0, 40.0)


def _softmax_obj_and_grad(z, alpha):
    """Softmax-smoothed objective and gradient w.r.t. logits z."""
    z = np.clip(z, -40.0, 40.0)
    p = _probs_from_logits(z)
    c = _self_conv(p)
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(np.clip(u, -50.0, 50.0))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return 2.0 * N * cmax, np.zeros(N - 1)
    J = 2.0 * N * (cmax + np.log(s) / alpha)
    w = e / s
    # Gradient of max(conv(p,p)) w.r.t. p:
    L = 2 * _FFT_LEN
    fp = np.fft.rfft(p, L)
    fw = np.fft.rfft(w, L)
    cc = np.fft.irfft(np.conj(fp) * fw, L).real
    gp = 2.0 * cc[:N]
    sdot = float(np.dot(gp, p))
    gz = 2.0 * N * p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return float(J), gz


def _lbfgs_polish(p, alpha, max_iter, deadline):
    """Polish using L-BFGS-B on softmax surrogate with logit parameterization."""
    if _scipy_minimize is None:
        return p
    z0 = _logits_from_p(p)
    best_local = p.copy()
    best_score = _score(p)
    try:
        res = _scipy_minimize(
            lambda z: _softmax_obj_and_grad(z, alpha),
            z0,
            method='L-BFGS-B',
            jac=True,
            bounds=[(-40.0, 40.0)] * (N - 1),
            options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                     'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40, 'disp': False}
        )
        if res.x is not None:
            p_new = _probs_from_logits(res.x)
            s_new = _score(p_new)
            if s_new < best_score:
                best_local = p_new
                best_score = s_new
            _update_best(best_local, best_score)
    except Exception:
        pass
    return best_local
