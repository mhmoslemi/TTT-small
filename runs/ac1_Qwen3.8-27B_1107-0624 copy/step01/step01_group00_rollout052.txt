    n = a.size
    m = 1
    while m < 2 * n - 1:

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using LP-based descent + Nesterov refinement.
The LP finds the direction that maximally increases the sequence sum while
keeping the auto-convolution bounded, providing a natural "water-filling" direction.
"""

import time

import numpy as np
from scipy.optimize import linprog


def _conv_sq(a):
    n = a.size
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]


def _project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not cond.any():
        return np.zeros(n)
    rho_last = int(np.max(rho[cond]))
    theta = (cssv[rho_last - 1] - target) / rho_last
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, 1.0 / a.size)
    return a / s


def _build_A(a):
    n = a.size
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    i = k - j
    valid = (i >= 0) & (i < n)
    A = np.zeros((2 * n - 1, n))
    A[valid] = a[i[valid]]
    return A


def _lp_solve(a, M):
    n = a.size
    A = _build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={"presolve": True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _smoothed_grad(c, a, tau):
    n = a.size
    cm = c.max()
    w = np.exp((c - cm) / tau)
    w /= w.sum()
    # grad = 2 * correlate(w, a, 'full')[:n]
    # Use FFT
    m = 1
    while m < 3 * n - 2:
        m <<= 1
    FW = np.fft.rfft(w, m)
    FA = np.fft.rfft(a, m)
    xcorr = np.fft.irfft(FW * np.conj(FA), m)
    return 2.0 * xcorr[:n]


def _nesterov_phase(a, t_end, lr=0.1, tau0=0.1, momentum0=0.9):
    """Nesterov-accelerated descent on the smoothed objective."""
    n = a.size
    a = _normalize(np.abs(a))
    y = a.copy()
    v = np.zeros(n)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    best_a = a.copy()
    best_mc = _conv_sq(a).max()
    tau = tau0
    mu = momentum0
    step = 0
    while time.monotonic() < t_end - 1.0:
        c = _conv_sq(y)
        mc = c.max()
        if mc < best_mc:
            best_mc = mc
            best_a = a.copy()  # Hmm, should this be y or a?
        
        g = _smoothed_grad(c, y, tau)
        gn = np.linalg.norm(g)
        if gn > 0:
            g /= gn
        v = mu * v - lr * g
        a_new = y + v
        a_new = _project_simplex(a_new, 1.0)
        
        # Nesterov update
        t_k = (1 + math.sqrt(1 + 4 * mu * mu)) / 2
        mu_new = (1 - 1 / t_k) / (1 + 1 / t_k)
        y = a_new + (mu_new / mu - 1) * (a_new - a) if mu > 0 else a_new
        
        a = a_new
        mu = mu_new
        step += 1
        
        if step % 500 == 0:
            tau = max(0.02, tau * 0.8)
            lr = max(0.01, lr * 0.9)
    
    # Final check
    c = _conv_sq(best_a)
    return best_a, c.max()


def _lp_phase(a, t_end):
    """LP-based descent phase."""
    n = a.size
    a = _normalize(np.abs(a).astype(np.float64))
    best_a = a.copy()
    best_mc = _conv_sq(a).max()
    t_blend = 0.5
    step = 0
    no_improve = 0
    while time.monotonic() < t_end - 1.0:
        step += 1
        c = _conv_sq(a)
        M = c.max()
        
        if M < best_mc:
            best_mc = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        b = _lp_solve(a, M)
        if b is None or b.sum() < 1e-10:
            break
        
        b = _normalize(b)
        
        # Blend
        a_new = (1 - t_blend) * a + t_blend * b
        a_new = _project_simplex(a_new, 1.0)
        
        # Adaptive blend: if no improvement, reduce step
        if no_improve > 50:
            t_blend = max(0.05, t_blend * 0.9)
        
        a = a_new
        
        # Periodically restart from best if stuck
        if no_improve > 200:
            a = best_a + 0.01 * np.random.randn(n)
            a = _project_simplex(np.abs(a), 1.0)
            no_improve = 0
    
    return best_a, best_mc
