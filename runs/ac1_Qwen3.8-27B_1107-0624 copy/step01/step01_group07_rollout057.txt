def compute_objective(a):

    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def lp_polish(a, deadline, verbose=True):
    """LP-based refinement using the max-sum direction."""
    n = len(a)

        return a, np.inf
    a = a / a.sum()
    
    b = np.convolve(a, a, mode='full')
    best_score = 2.0 * n * b.max()
    best_a = a.copy()
    
    # Build convolution matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    
    step = 0.1
    stall = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() < deadline:
        it += 1
        M = b.max()
        
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        if g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            b_new = np.convolve(a_new, a_new, mode='full')
            sc = 2.0 * n * b_new.max()
            if sc < best_score - 1e-12:
                best_a = a_new
                best_score = sc
                a = a_new
                b = b_new
                # Rebuild A
                A = np.zeros((2*n-1, n))
                for j in range(n):
                    A[j:j+n, j] = a
                improved = True
                break

            stall = 0
            step = min(0.4, step * 1.02)

            step = max(0.005, step * 0.8)
            if stall % 10 == 0 and stall > 10:
                noise = np.abs(rng.normal(0, 0.03, n))
                a_new = best_a * (1.0 + noise)
                a_new = np.maximum(a_new, 0.0)
                if a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    b_new = np.convolve(a_new, a_new, mode='full')
                    sc = 2.0 * n * b_new.max()

                        best_score = sc
                        a = a_new
                        b = b_new
                        A = np.zeros((2*n-1, n))
                        for j in range(n):
                            A[j:j+n, j] = a
                        stall = 0
            if stall > 80:
                break
        
        if verbose and it % 50 == 0:
            print(f"    LP iter {it}, score={best_score:.6f}, step={step:.4f}")
    
    return best_a, best_score


def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))
