import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft, next_fast_len

def compute_objective_fast(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 using FFT for speed."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    # Use FFT for convolution
    size = next_fast_len(2 * n - 1)
    A = fft(a, size)
    C = ifft(A * A).real
    C = C[:2*n-1]  # full convolution
    return 2.0 * n * C.max() / (S * S)

def convolve_fast(a, b):
    """Fast convolution using FFT."""
    n = len(a)
    m = len(b)
    size = next_fast_len(n + m - 1)
    A = fft(a, size)
    B = fft(b, size)
    return ifft(A * B).real[:n+m-1]

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def lp_find_direction(a, M, max_iter=3):
    """
    Find direction g that maximizes sum(g) subject to conv(a,g) <= M, g >= 0.
    Uses a relaxed formulation for speed.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_line_search(a, g, best_score, n):
    """
    Find the best step size t in [0,1] for a_new = (1-t)*a + t*g.
    Uses bisection to find the largest t that improves the score.

    # Binary search for the best t

        hi = 0.5

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_t = mid
            lo = mid
        else:
            hi = mid
    
    return best_t, best_sc

def make_flat_spectrum_init(n, rng, n_iter=100):
    """
    Initialize a sequence with approximately flat spectrum using
    Gerchberg-Saxton style iteration.
    """
    a = np.abs(rng.normal(0, 1, n)) + 0.1
    a = a / a.sum()
    
    size = next_fast_len(2 * n - 1)
    
    for _ in range(n_iter):
        # Compute spectrum
        A = fft(a, size)
        # Target: flat magnitude
        target_mag = np.full(size, a.sum() / np.sqrt(size))
        # Modify phase to achieve flat magnitude
        target_A = target_mag * np.exp(1j * np.angle(A))
        # Inverse
        a_new = ifft(target_A).real[:n]
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        a = a_new
    
    return a

    """Kaiser window."""
    from scipy.special import i0
    r = np.linspace(-1, 1, n)
    return i0(beta * np.sqrt(1 - r**2)) / i0(beta)
