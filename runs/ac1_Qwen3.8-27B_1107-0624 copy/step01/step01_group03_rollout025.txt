def fast_score(s):
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)

def compute_gradient(s, tau):
    """Compute gradient of smooth max of conv(s,s)."""
    conv = np.convolve(s, s)
    conv_max = conv.max()
    exp_vals = np.exp(tau * (conv - conv_max))
    Z = exp_vals.sum()
    weights = exp_vals / Z
    # smooth_max = conv_max + log(Z) / tau
    # grad[i] = 2 * sum_k weights[k] * s[k-i]
    # = 2 * (convolve(weights, s[::-1]))[n-1 : n-1+n]
    s_flip = s[::-1]
    cross = np.convolve(weights, s_flip)
    grad = 2.0 * cross[n-1:n-1+n]
    return grad
