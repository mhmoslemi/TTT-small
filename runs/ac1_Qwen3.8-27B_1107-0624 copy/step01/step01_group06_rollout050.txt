"""
Optimizes step function coefficients to minimize J(a) = 2n*max(a*a)/S^2
using multi-start L-BFGS-B with log-sum-exp smoothing, followed by
top-k refinement and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fftpack import next_fast_len


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

    """Compute smoothed J and gradient using log-sum-exp."""

Optimizes step function coefficients to minimize J(a) = 2n*max(a*a)/S^2.
Uses multi-start L-BFGS-B with log-sum-exp smoothing (annealed tau),

import numpy as np

from scipy.fftpack import next_fast_len


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute exact J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


class SmoothObjective:
    """Log-sum-exp smoothed objective for L-BFGS."""
    def __init__(self, tau):
        self.tau = tau
        self.n = 0  # set when called
        
    def fun(self, a):
        return self.fgres(a)[0]
    
    def fgres(self, a):
        n = len(a)
        self.n = n
        c = convolve_fft(a)
        S = np.sum(a)
        if S < 0.01:
            return 1e15, np.zeros(n)
        
        c_max = np.max(c)
        shifted = (c - c_max) / self.tau
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        Tmax = c_max + self.tau * np.log(sum_exp)
        p = exp_shifted / sum_exp
        
        J = 2.0 * n * Tmax / (S * S)
        
        size = next_fast_len(3 * n - 2)
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p, size)
        cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
        
        grad = 4.0 * n * self.tau * cross / (S * S) - 4.0 * n * Tmax / (S * S * S)
        
        return J, grad

        obj.fgres,

        jac=True,
        method='L-BFGS-B',
        bounds=[(0.0, 1000.0)] * n,
        options={'maxiter': maxiter, 'maxfun': 500}
    )
    
    return res.x, compute_J(res.x)
