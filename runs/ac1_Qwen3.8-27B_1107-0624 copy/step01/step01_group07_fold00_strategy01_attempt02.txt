                new_score = evaluate_sequence(new_point)
                if np.isfinite(new_score) and L == len(new_point) and new_score < best_per_length.get(L, [np.inf])[0]:
                    best_per_length[L] = new_score
                    current = best_per_length
                    break_loop = True
                    break
            else:
                continue
            break
    # Scrape final values and validate against evaluator:
    sequence_external = convert_to_external_coord(best_per_length[512]['bounding_box'], scale=0.05)
    score_external = evaluate_sequence(sequence_external)
    if np.isfinite(score_external):
        best_sequence = {'seq': sequence_external, 'obj': score_external}
    else:
        sequence_external = ['0.0' for _ in range(512)]
        sequence_external[len(512):] = [''] * (len(best_sequence['seq']) - 512)
    return sequence_external.tolist()

def generate_uniform_sequences(length_range, n_samples):
    """Generate sequences through uniform (Guassian) sampling"""
    base_seqs = []
    L_min, L_max = length_range
    for L in range(L_min, L_max+1):
        samples = [0.5 * (1 + np.random.randn(2*L-1)**(1/2)) for _ in range(n_samples)]
        base_seqs.extend([np.array(x)[min(L-100, 100):min(L+100, 1024)] for x in samples])
    return base_seqs
