"""
Step function optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses interaction-based gradient descent, LP-guided directions, flat-top equalization,
and multi-scale refinement to beat the previous best of 1.514611.
"""

import numpy as np
import time
from scipy.optimize import linprog


def normalize_nonneg(a):
    """Normalize a 1-D array to sum 1, keeping it non-negative."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def score_normalized(a):
    """Score for a normalized sum-1 nonnegative sequence: 2*n*max(convolve(a,a))."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m)


def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    a = np.asarray(a, dtype=np.float64).ravel()
    return np.convolve(a, a)


def interaction_vector(a, k_star):
    """
    Compute the gradient of c[k_star] with respect to a[j].
    c[k_star] = sum_i a[i] * a[k_star - i]
    dc/da[j] = a[k_star - j] + a[j - k_star] (for valid indices)
    """
    n = a.size
    w = np.zeros(n, dtype=np.float64)
    for j in range(n):
        val = 0.0
        i1 = k_star - j
        if 0 <= i1 < n:
            val += a[i1]
        i2 = j - k_star
        if 0 <= i2 < n:
            val += a[i2]
        w[j] = val
    return w


def interaction_vector_fast(a, k_star):
    """Vectorized computation of the interaction vector."""
    n = a.size
    w = np.zeros(n, dtype=np.float64)
    # a[k_star - j] valid when 0 <= k_star - j < n, i.e., k_star - n < j <= k_star
    lo1 = max(0, k_star - n + 1)
    hi1 = min(n, k_star + 1)
    if hi1 > lo1:
        w[lo1:hi1] += a[k_star - (np.arange(lo1, hi1))]
    # a[j - k_star] valid when 0 <= j - k_star < n, i.e., k_star <= j < k_star + n
    lo2 = max(0, k_star)
    hi2 = min(n, k_star + n)
    if hi2 > lo2:
        w[lo2:hi2] += a[np.arange(lo2, hi2) - k_star]
    return w


def fw_step(a, c, k_star):
    """
    Frank-Wolfe / conditional gradient step.
    Move mass from highest interaction to lowest interaction position.
    """
    n = a.size
    w = interaction_vector_fast(a, k_star)
    # Weighted interaction (contribution per position)
    contrib = w * a
    donor = int(np.argmax(contrib))
    # Recipient: lowest interaction weight (preferably with some existing mass or low value)
    # To avoid creating too many non-zero entries, prefer positions with small a
    recipient = int(np.argmin(w))
    if donor == recipient:
        return None
    # Determine how much to move
    # Moving delta from donor to recipient changes c[k*] by:
    # delta * (w[recipient] - w[donor])
    # We want to reduce c[k*], so we need w[recipient] < w[donor]
    if w[recipient] >= w[donor]:
        return None
    # Try several step sizes
    max_delta = a[donor] * 0.9
    best_h = None
    best_sc = score_normalized(a)
    for frac in (0.01, 0.03, 0.08, 0.15, 0.3, 0.5):
        delta = max_delta * frac
        h = a.copy()
        h[donor] -= delta
        h[recipient] += delta
        if h[donor] < -1e-15:
            continue
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()
    return best_h


def gradient_step(a, c, k_star, lr):
    """Projected gradient descent step on the peak."""
    n = a.size
    w = interaction_vector_fast(a, k_star)
    h = a - lr * w
    h = np.clip(h, 0.0, None)
    h_s = h.sum()
    if h_s < 1e-300:
        return None
    h = h / h_s
    return h


def multi_peak_equalize(a, c, top_k=5):
    """
    Try to equalize the top-k peaks of the convolution.
    If peaks are unequal, shift mass to reduce the largest and increase the smaller ones.
    """
    n = a.size
    c = compute_conv(a)
    # Find top-k peak positions
    sorted_idx = np.argsort(c)[::-1][:top_k]
    peak_vals = c[sorted_idx]
    if peak_vals[0] <= peak_vals[-1] * 1.01:
        return None  # Already fairly equalized

    # For the largest peak, compute interaction and try to reduce it
    k_max = int(sorted_idx[0])
    w_max = interaction_vector_fast(a, k_max)

    # For a smaller peak, compute interaction and try to increase it
    k_min = int(sorted_idx[-1])
    w_min = interaction_vector_fast(a, k_min)

    # Find positions where reducing helps the max and increasing helps the min
    # Move from high w_max to low w_max AND high w_min
    score = w_max - 0.5 * w_min
    donor = int(np.argmax(score * a))
    recipient = int(np.argmin(score))
    if donor == recipient:
        return None

    best_h = None
    best_sc = score_normalized(a)
    max_delta = a[donor] * 0.5
    for frac in (0.02, 0.05, 0.1, 0.2):
        delta = max_delta * frac
        h = a.copy()
        h[donor] -= delta
        h[recipient] += delta
        h = normalize_nonneg(h)
        sc = score_normalized(h)
        if sc < best_sc:
            best_sc = sc
            best_h = h.copy()
    return best_h


def build_lp_matrix(a, rows):
    """Build the active constraint matrix for the LP."""
    n = a.size
    A = np.zeros((len(rows), n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        # c[k] = sum_i a[i]*a[k-i], dc[k]/da[j] = a[k-j] + a[j-k]
        # But for the LP constraint we need: sum_j a[k-j] * g[j] <= M
        # which means row r of A is [a[k-j] for j=0..n-1] (0 if invalid)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            A[r, lo:hi] = a[k - np.arange(lo, hi)]
    return A


def solve_lp_direction(a, M, max_rows=None):
    """
    Solve the LP: max sum(g) s.t. A g <= M, g >= 0.
    A is built from the top convolution peaks.
    """
    n = a.size
    if n > 2048:
        return None
    if M <= 1e-300:
        return None

    c = compute_conv(a)
    total = c.size  # 2n-1
    if max_rows is None:
        max_rows = total
    max_rows = min(max_rows, total)

    if max_rows < total:
        rows = np.argpartition(c, -max_rows)[-max_rows:].astype(np.int64)
    else:
        rows = np.arange(total, dtype=np.int64)

    try:
        A = build_lp_matrix(a, rows)
    except Exception:
        return None

    if not np.any(A):
        return None

    A = A / M
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(A.shape[0], dtype=np.float64)

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method='highs')
    except Exception:
        return None

    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if g.sum() > 1e-12:
            return g
    return None


def random_perturb(a, rng, scale=0.1):
    """Random multiplicative perturbation."""
    n = a.size
    h = a.copy()
    if n > 1:
        noise = rng.normal(0.0, scale, size=n)
        h = h * np.exp(noise)
    return normalize_nonneg(h)


def block_perturb(a, rng, n_blocks=8):
    """Perturb by blocks for smoother changes."""
    n = a.size
    h = a.copy()
    block_size = max(1, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, n - block_size))
        end = min(start + block_size, n)
        factor = np.exp(rng.normal(0.0, 0.3, size=end - start))
        h[start:end] *= factor
    return normalize_nonneg(h)


def resize_sequence(a, new_n):
    """Resize sequence to new length by interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == new_n:
        return normalize_nonneg(a)
    if m == 1:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, new_n)
    result = np.interp(x_new, x_old, a)
    return normalize_nonneg(np.maximum(result, 0.0))
