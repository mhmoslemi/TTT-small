    """Project vector v onto {x >= 0, sum x = target}."""
    v = np.asarray(v, dtype=np.float64)

    w = exp_c / sum_exp  # softmax weights over c
    
    # Gradient of c w.r.t. a:
    # dc_k/da_p = a[k-p] (if 0<=k-p<n) + a[p-k] (if 0<=p-k<n)
    # By symmetry of sum over k: grad_a_p = 2 * sum_k w_k * a[k-p] * [0<=k-p<n]
    grad_a = np.zeros(n, dtype=np.float64)
    # Vectorized computation using FFT-like approach:
    # grad_a = 2 * (w * a) in the convolution sense, but with a being the "kernel"
    # Actually: grad_a[p] = 2 * sum_{k} w[k] * a[k-p] * [0 <= k-p < n]
    # Let j = k-p, then k = j+p, and we need 0 <= j < n, 0 <= j+p < 2n-1
    # grad_a[p] = 2 * sum_{j=0}^{n-1} w[j+p] * a[j] * [0 <= j+p < 2n-1]
    #            = 2 * sum_{j=0}^{n-1} w[j+p] * a[j] * [j+p <= 2n-2]
    # For p >= 0: j+p <= 2n-2 means j <= 2n-2-p. Since j <= n-1 always and 2n-2-p >= n-1 when p <= n-1.
    # So for p in [0, n-1]: grad_a[p] = 2 * sum_{j=0}^{n-1} w[j+p] * a[j]
    # This is a cross-correlation! grad_a = 2 * corr(w, a) where (corr(x,y))[p] = sum_j x[j+p]*y[j]
    
    # We can compute this via FFT or directly:
    # corr(w, a)[p] = sum_{j=0}^{n-1} w[j+p] * a[j]
    # This is the "offset" inner product. We can compute it using the convolution:
    # If we define w2 = w (length 2n-1) and a2 = a (length n):
    # conv(w2, flip(a2))[p + n - 1] = sum_j w2[j] * a2[n-1-j] ... hmm this is getting complicated.
    
    # Let me just do it directly for now (O(n^2) but n is at most a few thousand):
    # Actually for n up to 2048, O(n^2) is 4M operations which is fine.
    # But let's use a smarter approach.
    
    # grad_a[p] = 2 * sum_{j=0}^{min(n-1, 2n-2-p)} w[j+p] * a[j]
    # This is equivalent to: grad_a = 2 * (a * w_full) where * denotes "truncated cross-correlation"
    # Using FFT: if we pad a to length m and w to length m, then:
    # grad_a[p] = 2 * (IFFT(FFT(a) * conj(FFT(w))))[p] ... no, that's circular correlation.
    
    # For linear cross-correlation: corr(x, y)[k] = sum_j x[j+k] * y[j]
    # This equals conv(x, flip(y))[k + len(y) - 1]
    # So grad_a[p] = 2 * conv(w, flip(a))[p + n - 1]
    
    fa = np.fft.rfft(a, m)
    fw = np.fft.rfft(np.concatenate([w, np.zeros(m - len(w))]), m)
    # cross-correlation via FFT:
    # corr(w, a)[p] = IFFT(conj(FFT(w)) * FFT(a))[p] (circular)
    # For linear: we need m >= max(len(w), len(a)) which is 2n-1, and m is already >= 2n-1.
    # corr_linear(w, a)[p] = sum_j w[j+p]*a[j] for p = 0, ..., 2n-2
    # = IFFT(conj(FFT(w)) * FFT(a))[p] (circular, with zero-padding to m)
    
    corr = np.fft.irfft(np.conj(fw) * fa, m)
    grad_a = 2.0 * corr[:n]
    
    # Now gradient w.r.t. theta:
    # a_i = exp(theta_i) / Z, da_i/dtheta_j = a_i (delta_ij - a_j)
    # grad_theta_j = sum_i grad_a_i * da_i/dtheta_j = grad_a_j * a_j - a_j * sum_i grad_a_i * a_i
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    
    return obj, grad_theta

def _softmax(theta):
    t = theta - theta.max()
    e = np.exp(t)
    return e / e.sum()

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

Approach:
- Softmax parameterization ensures a >= 0 and sum(a) = 1 automatically
- Log-sum-exp smoothing approximates the max (smooth, differentiable)
- L-BFGS-B with analytic gradients for fast convergence
- Adaptive temperature annealing (coarse to fine)
- Multiple sequence lengths and diverse initializations
- Final polish phase with very low temperature
"""

import numpy as np
import time
import math
from scipy.optimize import minimize as sp_minimize


def _fft_conv_sq(a, m):
    """Linear convolution a*a via zero-padded FFT. m >= 2*len(a)-1."""
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * a.size - 1]


def _softmax(theta):
    """Numerically stable softmax."""
    t = theta - theta.max()
    e = np.exp(t)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(theta), 1.0 / len(theta))
    return e / s


def _obj_grad(theta, n, m, tau):
    """
    Compute log-sum-exp objective and its gradient w.r.t. theta.
    a = softmax(theta), c = conv(a,a), obj = lse(c; tau).
    """
    a = _softmax(theta)
    c = _fft_conv_sq(a, m)
    
    cm = c.max()
    exp_c = np.exp((c - cm) / tau)
    sum_exp = exp_c.sum()
    obj = cm + tau * math.log(sum_exp)
    
    w = exp_c / sum_exp  # normalized weights
    
    # Gradient of max(c) w.r.t. a:
    # grad_a[p] = 2 * sum_k w[k] * a[k-p] * [0 <= k-p < n]
    # This is a cross-correlation: grad_a = 2 * corr(w, a)[:n]
    # where corr(x,y)[p] = sum_j x[j+p]*y[j]
    # Computed via FFT: corr = IFFT(conj(FFT(x)) * FFT(y))
    
    fa = np.fft.rfft(a, m)
    fw = np.fft.rfft(w, m)
    corr = np.fft.irfft(np.conj(fw) * fa, m)
    grad_a = 2.0 * corr[:n]
    
    # Gradient w.r.t. theta via chain rule (softmax parameterization):
    # grad_theta_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    
    return obj, grad_theta

    def wrapper(theta):
        return _obj_grad(theta, n, m, tau)
