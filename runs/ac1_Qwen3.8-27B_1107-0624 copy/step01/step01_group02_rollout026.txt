    # (the second term is nonzero when k-j=i, i.e., j=k-i, giving a[k-i])
    # BUT these are the same term in the sum when j=i=k-i, i.e., k=2i.
    # In that case, the term is a[i]*a[i] = a[i]^2, and its derivative is 2*a[i].
    # In all other cases, the two contributions are distinct terms in the sum, so we get 2*a[k-i].
    # Wait no. When k != 2i:
    #   First contribution: j=i term in the sum is a[i]*a[k-i], derivative wrt a[i] is a[k-i]
    #   Second contribution: j=k-i term in the sum is a[k-i]*a[i], derivative wrt a[i] is a[k-i]
    #   Total: 2*a[k-i]
    # When k = 2i:
    #   The j=i term is a[i]*a[i] = a[i]^2, derivative is 2*a[i]
    #   There's no separate "second contribution" because j=i=k-i gives the same term.
    # So: d c[k]/d a[i] = 2*a[k-i] always (with a[j]=0 for j outside [0,n-1]).
    
    # So grad[i] = sum_k p[k] * 2 * a[k-i] (with bounds)
    # = 2 * (p * a)[shifted]
    # Specifically: grad[i] = 2 * sum_k p[k] * a[k-i] where 0<=k-i<n
    # Let's compute this via cross-correlation.
    # sum_k p[k] * a[k-i] for k=0..2n-2, a indexed 0..n-1
    # = sum_k p[k] * a[k-i] (nonzero when 0<=k-i<n, i.e., i<=k<i+n)
    # This is the "anti-convolution" or cross-correlation.
    # Define: r[i] = sum_k p[k] * a[k-i]
    # = sum_j p[j+i] * a[j] (substituting j=k-i)
    # This is (p * flip(a))[i + n - 1] ... let me just use the direct formula.
    
    # r[i] = sum_k p[k] * a[k-i]
    # = sum_{j=0}^{n-1} a[j] * p[j+i]  (where k=j+i)
    # So r = cross_correlation(a, p) in a specific sense.
    # r[i] = sum_j a[j] * p[j+i]
    # This is the "valid" cross-correlation of a with p, shifted.
    # In numpy: np.correlate(p, a, mode='full')[n-1:n-1+n]
    # Actually: np.correlate(a[::-1], p, mode='full')[:n] ... let me just use FFT.
    
    # r[i] = sum_j a[j] * p[j+i]
    # Define q[m] = p[m] for m=0..2n-2, 0 otherwise.
    # r[i] = sum_j a[j] * q[j+i]
    # This is the cross-correlation (a, q)[i] = sum_j a[j]*q[j+i]
    # = (a * q_flipped)[i + len(q) - 1] ... 
    
    # Easier: use the formula directly with FFT.
    # r[i] = sum_j a[j] * p[j+i] for i=0..n-1
    # Create a new array: ext_a of length 2n-1+n = 3n-1... this is getting complex.
    # Let me just use the shift_correlation approach.
    
    # Actually, the simplest way:
    # grad[i] = 2 * sum_k p[k] * a[k-i] (with a[j]=0 for j<0 or j>=n)
    # Let's define an array a_ext of length 2n-1 (zeros outside [0,n-1])
    # Then grad[i] = 2 * sum_k p[k] * a_ext[k-i]
    # = 2 * (p * a_ext_flipped)[i + (2n-1) - 1] ... 
    
    # Simplest correct approach using FFT:
    # grad[i] = 2 * sum_{k} p[k] * a[k-i]
    # The sum is over k where 0<=k<=2n-2 and 0<=k-i<=n-1, i.e., i<=k<=i+n-1.
    # Let's define h[m] = a[m] for m=0..n-1, 0 otherwise.
    # Then grad[i]/2 = sum_k p[k] * h[k-i] = (p * h_flip)[i + n - 1 + 2n - 2 - (n-1)] ...
    
    # OK let me just do it the straightforward way:
    # grad_unscaled[i] = sum_k p[k] * h[k-i] where h[j] = a[j] if 0<=j<n else 0
    # = sum_k p[k] * h[k-i]
    # This is the cross-correlation of p with h, evaluated at lag i.
    # cross_corr(p, h)[i] = sum_k p[k] * h[k+i] ... that's the other direction.
    # Actually: sum_k p[k]*h[k-i] = cross_corr(h, p)[i] (depending on convention)
    
    # Let me just compute it directly:
    # Create array of length 2n-1 for p, and for each i, we need h[k-i] for k=0..2n-2.
    # h[k-i] is nonzero when 0<=k-i<=n-1, i.e., i<=k<=i+n-1.
    # So grad_unscaled[i] = sum_{k=max(0,i)}^{min(2n-2, i+n-1)} p[k] * a[k-i]
    
    # This is a "sliding dot product" - can be done with FFT:
    # Define A of length 2n-2+n = 3n-2, zero-padded version of a (length n)
    # Then grad_unscaled[i] = A[k-i] dot p[k] for k in range...
    
    # Actually the cleanest way: 
    # grad_unscaled[i] = sum_k p[k] * a[k-i]
    # Define q[m] = a[m] padded to length 2n-2+1 (to accommodate all shifts)
    # Then this is the "anti-correlation": sum_k p[k] * q[k-i]
    # = correlation of p and q at lag i
    # = (p * flip(q))[i + len(flipped_q) - 1]
    
    # Let me use a different approach. Define:
    # fx = np.zeros(2*(2*n-1), dtype=float); fx[:n] = a  (length n, embedded in 2n-1 context)
    # Then convolve p (length 2n-1) with a (length n) to get length 3n-2.
    # (p*a)[m] = sum_k p[k]*a[m-k]
    # We want sum_k p[k]*a[k-i] = (p*a)[k-i+k-i]... no.
    # (p * a)[s] = sum_k p[k] * a[s-k]
    # Set s-k = k-i => s = 2k-i... that doesn't work.
    
    # Let me just flip: sum_k p[k]*a[k-i] = sum_j p[j+i]*a[j] (sub j=k-i)
    # = sum_j a[j] * p[j+i]
    # This is (flip(p) * a)[(2n-2) - i] ... let me verify.
    # (flip(p) * a)[m] = sum_j flip(p)[j] * a[m-j] = sum_j p[2n-2-j] * a[m-j]
    # Let k = 2n-2-j, so j = 2n-2-k:
    # = sum_k p[k] * a[m-(2n-2-k)] = sum_k p[k] * a[m-2n+2+k]
    # Set m-2n+2+k = j (where j=k-i, so we want m-2n+2+k = k-i, m = 2n-2-i):
    # At m = 2n-2-i: (flip(p)*a)[2n-2-i] = sum_k p[k] * a[k-i]  YES!
    
    # So grad_unscaled[i] = (flip(p) * a)[2n-2-i]
    # = convolve(p[::-1], a) evaluated at index 2n-2-i
    
    p_flip = p[::-1]
    conv = fft_convolve(p_flip, a)
    # conv has length (2n-1) + n - 1 = 3n-2
    # We want conv[2n-2-i] for i=0..n-1
    idx = (2 * n - 2) - np.arange(n, dtype=np.int64)
    grad = 2.0 * conv[idx]
    
    return grad, c, M


def solve_lp_direction(f, M, active_rows):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_rows, g>=0."""
    if linprog is None:
        return None
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n > 8192:
        return None
    if M <= 1e-300:
        return None
    
    rows = np.asarray(active_rows, dtype=np.int64)
    if rows.size == 0:
        return None
    
    # Build constraint matrix
    A = np.zeros((rows.size, n), dtype=np.float64)
    for r, k in enumerate(rows):
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            idx = jlo + np.arange(jhi - jlo, dtype=np.int64)
            A[r, jlo:jhi] = f[k - idx]
    
    if not np.any(A):
        return None
    
    A = A / M
    cvec = -np.ones(n, dtype=np.float64)
    bvec = np.ones(rows.size, dtype=np.float64)
    
    try:
        res = linprog(cvec, A_ub=A, b_ub=bvec, bounds=(0.0, None), method='highs')
    except Exception:
        return None
    
    if res is not None and res.status == 0 and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        g = np.clip(g, 0.0, None)
        if float(g.sum()) > 1e-12:
            return g
    return None


def make_windowed_sinusoid(N, n_freqs=3, A_base=0.8, window_type='raised_cosine', rng=None):
    """Create a windowed multi-frequency sinusoidal sequence."""
    x = np.arange(N, dtype=np.float64) / N
    a = np.ones(N, dtype=np.float64)
    
    if window_type == 'raised_cosine':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a *= w
    elif window_type == 'fejer':
        w = 1.0 - np.abs(2.0 * x - 1.0)
        a *= w
    elif window_type == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5)))
        a *= np.clip(w, 0.01, None)
    elif window_type == 'none':
        pass
    elif window_type == 'taper':
        edge = max(1, N // 10)
        w = np.ones(N)
        ramp = np.linspace(0.1, 1.0, edge)
        w[:edge] = ramp
        w[-edge:] = ramp[::-1]
        a *= w
    else:
        pass
    
    # Add sinusoidal modulations
    if rng is not None:
        for m in range(1, n_freqs + 1):
            A = A_base / math.sqrt(m) * rng.uniform(0.3, 1.0)
            phase = rng.uniform(0.0, 2.0 * np.pi) if m > 1 else 0.0
            a += A * np.sin(2.0 * np.pi * m * (x + 0.5 / N) + phase)
    else:
        # Default: single frequency, no phase
        a += A_base * np.sin(2.0 * np.pi * (x + 0.5 / N))
    
    return normalize_nonneg(np.maximum(a, 0.0))


def make_gaussian_bump(N, rng):
    """Gaussian bump centered at N/2."""
    x = np.arange(N, dtype=np.float64) / N
    mu = 0.5 + rng.uniform(-0.15, 0.15) if rng else 0.5
    sigma = rng.uniform(0.15, 0.35) if rng else 0.25
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    # Add small sinusoidal modulation
    a *= (1.0 + 0.3 * np.sin(2 * np.pi * x))
    return normalize_nonneg(np.maximum(a, 0.0))


def generate_initial(N, kind, rng):
    """Generate initial sequence of length N based on kind."""
    if kind == 0:
        return make_windowed_sinusoid(N, n_freqs=1, A_base=0.85, window_type='raised_cosine', rng=rng)
    elif kind == 1:
        return make_windowed_sinusoid(N, n_freqs=2, A_base=0.7, window_type='taper', rng=rng)
    elif kind == 2:
        return make_windowed_sinusoid(N, n_freqs=3, A_base=0.6, window_type='hann', rng=rng)
    elif kind == 3:
        return make_windowed_sinusoid(N, n_freqs=1, A_base=1.0, window_type='taper', rng=rng)
    elif kind == 4:
        return make_gaussian_bump(N, rng)
    elif kind == 5:
        return make_windowed_sinusoid(N, n_freqs=4, A_base=0.5, window_type='raised_cosine', rng=rng)
    elif kind == 6:
        # Asymmetric: more mass on one side
        x = np.arange(N, dtype=np.float64) / N
        a = np.exp(-2.0 * (x - 0.4) ** 2) * (1.0 + 0.5 * np.sin(2 * np.pi * x))
        return normalize_nonneg(np.maximum(a, 0.0))
    elif kind == 7:
        # Linear ramp with sinusoidal
        x = np.arange(N, dtype=np.float64) / N
        a = x * (1.0 - x) * (1.0 + 0.6 * np.sin(2 * np.pi * x + 0.3))
        return normalize_nonneg(np.maximum(a, 0.0))
    else:
        return np.ones(N, dtype=np.float64) / N


def resize_sequence(a, N):
    """Resize sequence to length N using interpolation (upsample) or averaging (downsample)."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == 0:
        return np.ones(N, dtype=np.float64) / N
    if m == N:
        return normalize_nonneg(a)
    
    if m < N:
        # Upsample by linear interpolation
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, N)
        new = np.interp(x_new, x_old, a)
    else:
        # Downsample by block averaging
        new = np.zeros(N, dtype=np.float64)
        for i in range(N):
            lo = int(i * m / N)
            hi = int((i + 1) * m / N)
            if hi > lo:
                new[i] = a[lo:hi].sum()
    
    return normalize_nonneg(np.maximum(new, 0.0))


def optimize_fista_smooth(a, time_limit, rng, state):
    """
    FISTA-accelerated smooth-max projected gradient optimization.
    """
    a = normalize_nonneg(a)
    n = a.size
    deadline = time.monotonic() + time_limit
    
    c = fft_convolve(a, a)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best = a.copy()
    
    # State for FISTA
    a_prev = a.copy()
    t_k = 1.0
    temp = 0.05 * M  # initial temperature
    
    no_improve = 0
    max_no_improve = 80
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # FISTA: compute y = a + momentum
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a + ((t_k - 1.0) / t_new) * (a - a_prev)
        y = np.clip(y, 0.0, None)
        y_sum = float(y.sum())
        if y_sum < 1e-300:
            y = a.copy()
        else:
            y = y / y_sum  # normalize for gradient computation
        
        # Compute gradient at y
        grad, c, M_cur = smooth_max_gradient(y, temp)
        
        # Adaptive learning rate
        # Lipschitz constant of the gradient ~ 2 * max|a| * (2n-1) or similar
        # Use a simple adaptive scheme
        lr = temp * 0.5 / max(1.0, float(np.max(np.abs(grad))))
        lr = min(lr, 0.3)  # cap it
        
        # Gradient step
        a_new = a - lr * grad
        a_new = normalize_nonneg(a_new)
        
        # Evaluate
        c_new = fft_convolve(a_new, a_new)
        M_new = float(c_new.max())
        score_new = float(2.0 * n * M_new)
        
        if score_new < best_score - 1e-10:
            best_score = score_new
            best = a_new.copy()
            no_improve = 0
            # Cool down temperature
            temp = max(0.005 * M_cur, temp * 0.97)
            # Update global
            if state["seq"] is None or best_score < state["score"] - 1e-9:
                state["score"] = best_score
                state["seq"] = best.copy()
                if it % 100 == 1:
                    print(f"  FISTA it={it} score={best_score:.10f} n={n} temp={temp:.6f}", flush=True)
        else:
            no_improve += 1
        
        # If stuck, try different strategies
        if no_improve > max_no_improve:
            # Random perturbation
            scale = rng.uniform(0.02, 0.12)
            a = a * np.exp(rng.normal(0.0, scale, size=n))
            a = normalize_nonneg(a)
            a_prev = a.copy()
            t_k = 1.0
            no_improve = 0
            temp = max(0.02 * M_cur, temp * 1.3)
            max_no_improve = int(max_no_improve * 0.8)
            if max_no_improve < 20:
                max_no_improve = 20
        elif no_improve > max_no_improve // 2:
            # Widen temperature
            temp = min(0.2 * M_cur, temp * 1.1)
        
        a_prev = a.copy()
        a = a_new
        t_k = t_new
    
    # Update global with best
    if best_score < state["score"]:
        state["score"] = best_score
        state["seq"] = best.copy()
        print(f"  FISTA done: score={best_score:.10f} n={n}", flush=True)
    
    return best, best_score


def optimize_lp_refine(a, time_limit, rng, state):
    """LP-based refinement: find direction from covering LP, line search."""
    a = normalize_nonneg(a)
    n = a.size
    deadline = time.monotonic() + time_limit
    
    c = fft_convolve(a, a)
    M = float(c.max())
    best_score = float(2.0 * n * M)
    best = a.copy()
    
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = fft_convolve(a, a)
        M = float(c.max())
        cur_score = float(2.0 * n * M)
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
            if state["seq"] is None or best_score < state["score"] - 1e-9:
                state["score"] = best_score
                state["seq"] = best.copy()
                print(f"  LP score={best_score:.10f} n={n}", flush=True)
        else:
            no_improve += 1
        
        if no_improve > 30:
            # Perturb and try again
            a = a * np.exp(rng.normal(0.0, 0.05, size=n))
            a = normalize_nonneg(a)
            no_improve = 0
            continue
        
        # Select active constraints: peaks within 5% of max
        threshold = M * 0.95
        active = np.where(c >= threshold)[0]
        
        # If too many, take top ones
        max_constraints = min(2 * n - 1, 500)
        if active.size > max_constraints:
            # Take the largest ones
            active = active[np.argsort(c[active])[-max_constraints:]]
        
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=np.int64)
        
        # Solve LP
        g = solve_lp_direction(a, M, active)
        
        if g is not None:
            Sg = float(g.sum())
            if Sg > 1.0 + 1e-4:
                g = g / Sg
                # Line search
                best_t = 0.0
                best_sc = cur_score
                for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                    h = (1.0 - t) * a + t * g
                    h = normalize_nonneg(h)
                    sc = score_normalized(h)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t
                
                if best_t > 0 and best_sc < cur_score - 1e-10:
                    a = (1.0 - best_t) * a + best_t * g
                    a = normalize_nonneg(a)
                else:
                    no_improve += 3
            else:
                no_improve += 1
        else:
            no_improve += 5
        
        # Safety check
        if time.monotonic() > deadline:
            break
    
    if best_score < state["score"]:
        state["score"] = best_score
        state["seq"] = best.copy()
        print(f"  LP done: score={best_score:.10f} n={n}", flush=True)
    
    return best, best_score


def optimize_hybrid(a, time_limit, rng, state):
    """
    Hybrid optimizer: alternates between FISTA smooth-max and LP refinement.
    Uses adaptive switching based on progress.
    """
    a = normalize_nonneg(a)
    n = a.size
    deadline = time.monotonic() + time_limit
    
    # First: FISTA phase
    fista_time = min(time_limit * 0.6, time_limit - 10.0)
    a, sc = optimize_fista_smooth(a, fista_time, rng, state)
    
    if time.monotonic() >= deadline - 2.0:
        return a, sc
    
    # Second: LP refinement
    lp_time = min(time_limit * 0.3, deadline - time.monotonic() - 5.0)
    if lp_time > 10.0:
        a, sc = optimize_lp_refine(a, lp_time, rng, state)
    
    # Third: Final FISTA polish
    remaining = deadline - time.monotonic() - 1.0
    if remaining > 10.0:
        a, sc = optimize_fista_smooth(a, remaining, rng, state)
    
    return a, sc


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function.
    Strategy:
    1. Generate diverse initial candidates using windowed sinusoids and bumps.
    2. Run hybrid optimization (FISTA + LP) on each candidate.
    3. Progressive length scaling: upscale best result and re-optimize.
    4. Final polish at the best length found.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    hard_deadline = start + budget_s - 30  # safety margin
    
    state = {"score": float('inf'), "seq": None}
    
    # Load provided sequence if available
    provided = None
    try:
        provided = np.asarray(height_sequence_1, dtype=np.float64)
        provided = np.clip(provided, 0.0, None)
        if provided.size > 0 and provided.sum() > 1e-300:
            provided = normalize_nonneg(provided)
            ps = score_normalized(provided)
            state["score"] = ps
            state["seq"] = provided.copy()
            print(f"Provided seq: n={provided.size} score={ps:.10f}", flush=True)
        else:
            provided = None
    except Exception:
        provided = None
    
    # Phase 1: Multi-start at medium lengths (300s)
    print("=== Phase 1: Multi-start ===", flush=True)
    phase1_deadline = start + min(300.0, budget_s * 0.2)
    
    initial_configs = [
        # (N, kind, description)
        (256, 0, "256 windowed sin"),
        (256, 3, "256 strong windowed sin"),
        (512, 0, "512 windowed sin"),
        (512, 1, "512 2-freq"),
        (512, 3, "512 strong taper"),
        (512, 4, "512 gaussian"),
        (512, 6, "512 asymmetric"),
        (1024, 0, "1024 windowed sin"),
        (1024, 1, "1024 2-freq"),
        (1024, 3, "1024 strong taper"),
        (1024, 4, "1024 gaussian"),
        (1024, 5, "1024 4-freq"),
        (1024, 7, "1024 ramp+sin"),
        (2048, 0, "2048 windowed sin"),
        (2048, 1, "2048 2-freq"),
        (2048, 3, "2048 strong taper"),
        (2048, 4, "2048 gaussian"),
    ]
    
    if provided is not None and 128 <= provided.size <= 4096:
        initial_configs.insert(0, (provided.size, -1, "provided"))
    
    for N, kind, desc in initial_configs:
        if time.monotonic() > phase1_deadline - 5.0:
            break
        
        if kind == -1:
            seq = resize_sequence(provided, N)
        else:
            seq = generate_initial(N, kind, rng)
        
        seq_score = score_normalized(seq)
        print(f"  Start: {desc} score={seq_score:.6f}", flush=True)
        
        # Quick FISTA to see if this is promising
        quick_time = min(30.0, phase1_deadline - time.monotonic() - 3.0)
        if quick_time < 5.0:
            continue
        
        sub_state = {"score": seq_score, "seq": seq.copy()}
        seq, sc = optimize_fista_smooth(seq, quick_time, rng, sub_state)
        
        if sc < state["score"]:
            state["score"] = sc
            state["seq"] = sub_state["seq"].copy()
            print(f"  -> New best: {sc:.10f} from {desc}", flush=True)
    
    # Phase 2: Deep optimization on best candidates (800s)
    print("=== Phase 2: Deep optimization ===", flush=True)
    phase2_deadline = start + min(1100.0, budget_s * 0.7)
