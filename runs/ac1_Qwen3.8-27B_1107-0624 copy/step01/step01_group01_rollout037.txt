```python
from scipy.optimize import linprog
import time

def solve_lp(a, M, n):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0"""
    # conv(a,b)[k] = sum_j a[j] * b[k-j] for j where 0 <= k-j <= n-1
    # i.e., A[k,j] * b[j] where A[k,j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A @ b <= M
    # A is (2n-1) x n
    A_ineq = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max + 1):
            A_ineq[k, j] = a[j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x
```

```python
LP-alternating and gradient-based optimization for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix
import time
import sys

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = np.maximum(c, 0.0)
    return c

def score_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-30:
        return np.inf
    c = np.convolve(a, a)
    m = float(c.max())
    return float(2.0 * n * m / (s * s))

    """Build the (2n-1) x n matrix A where A[k,j] = a[j] if 0<=k-j<=n-1"""
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    """Solve: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0"""

    h = np.fft.irfft(fw * fa_rev, N)

    lr = 0.1
```

```python
                lr = max(0.02, lr * 0.95)
        w = _soft_max_weights(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h[n - 1:2 * n - 1]
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-12)
        s = float(a_new.sum())
        if s > 1e-30:
            a = a_new / s
        else:
            a = best_a.copy()
            lr = base_lr
            stall = 0
        if best_score > 0 and stall % 100 == 0:
            beta = max(1e-6, beta * 0.8)
        if stall > 500:
            a = 0.75 * best_a + 0.25 * _random_profile(n, rng)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            lr = base_lr
            beta = 0.3 * float(_convolve_fft(a).max())
            stall = 0
    return best_a, best_score


def _make_linear(n, eta):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + eta * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_symmetric(n, eta):
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 - eta * t * t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_asymmetric_cubic(n, a1, a3):
    t = np.linspace(-1.0, 1.0, n)
    a = 1.0 + a1 * t + a3 * t ** 3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return _normalize(np.interp(x1, x0, p0))


def _adam_gradient_optimize(a0, time_limit, rng, lr=0.12):
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    n = a.size
    t_start = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * float(_convolve_fft(a).max())
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    beta = 0.25 * float(_convolve_fft(a).max())
    stall = 0
    base_lr = lr
    while time.monotonic() - t_start < time_limit:
        c = _convolve_fft(a)
        M = float(c.max())
        sc = 2.0 * n * M
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 300 == 0:
                lr = max(0.03, lr * 0.96)
        w = _soft_max_weights(c, beta)
        N = _next_pow2(2 * n - 1)
        fw = np.fft.rfft(w, N)
        fa_rev = np.fft.rfft(a[::-1], N)
        h = np.fft.irfft(fw * fa_rev, N)
        grad = 2.0 * h[n - 1:2 * n - 1]
        grad -= grad.mean()
        t_adam += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t_adam)
        vh = v / (1.0 - b2 ** t_adam)
        a_new = a - lr * mh / (np.sqrt(vh) + eps)
        a_new = np.maximum(a_new, 1e-12)
        s = float(a_new.sum())
        if s > 1e-30 and np.all(np.isfinite(a_new)):
            a = a_new / s
        else:
            a = best_a.copy()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
        if t_adam % 150 == 0:
            beta = max(1e-8, beta * 0.75)
        if stall > 600:
            a = 0.7 * best_a + 0.3 * _random_profile(n, rng)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            lr = base_lr
            beta = 0.25 * float(_convolve_fft(a).max())
            stall = 0
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining LP-alternating minimization and
    Adam gradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)

    # --- Initial candidates ---
    init_profiles = []
    for n_init in (256, 512):
        for eta in (1.0, 1.5, 2.0, 2.5):
            init_profiles.append(_make_linear(n_init, eta))
        for eta in (0.5, 1.0, 1.5):
            init_profiles.append(_make_symmetric(n_init, eta))
        for a1, a3 in ((1.0, -0.5), (1.5, -1.0), (1.2, 0.5)):
            init_profiles.append(_make_asymmetric_cubic(n_init, a1, a3))
        init_profiles.append(_random_profile(n_init, rng))

    for p in init_profiles:
        sc = _score(p)
        _update(p, sc)

    # Try to use parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = _normalize(parent)
            if pn is not None:
                sc = _score(pn)
                _update(pn, sc)
                print(f"Parent score: {sc:.8f}, n={pn.size}")
    except Exception:
        pass

    print(f"Initial best: {best_score:.8f}")

    # --- Phase 1: LP optimization at n=512 ---
    n1 = 512
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: LP at n={n1}, budget={phase1_end - t_start:.0f}s")

    starts_512 = []
    for p in init_profiles:
        if p.size == n1:
            starts_512.append(p)
    if best_seq is not None and best_seq.size != n1:
        starts_512.append(_interpolate_profile(best_seq, n1))
    if best_seq is not None and best_seq.size == n1:
        starts_512.append(best_seq.copy())
    for _ in range(4):
        starts_512.append(_random_profile(n1, rng))

    # LP on best starts
    for i, s0 in enumerate(starts_512[:6]):
        if time.monotonic() > phase1_end:
            break
        per = min(phase1_end, time.monotonic() + (phase1_end - t_start) / max(1, 6 - i))
        per = min(per, 80.0)
        pa, sa = _lp_optimize(s0, per, rng)
        _update(pa, sa)
        print(f"  LP 512 [{i}]: score={sa:.8f}")

    # --- Phase 2: Gradient polish at n=512 ---
    phase2_end = min(deadline, t_start + 700.0)
    print(f"Phase 2: Gradient at n={n1}, budget={phase2_end - t_start:.0f}s")

    if best_seq is not None:
        p512 = best_seq if best_seq.size == n1 else _interpolate_profile(best_seq, n1)
    else:
        p512 = _make_linear(n1, 1.5)
    per = min(phase2_end, time.monotonic() + 120.0)
    pg, sg = _adam_gradient_optimize(p512, per, rng, lr=0.12)
    _update(pg, sg)
    print(f"  Gradient 512: score={sg:.8f}")

    # --- Phase 3: LP + Gradient at n=1024 ---
    n3 = 1024
    phase3_end = min(deadline, t_start + 1100.0)
    print(f"Phase 3: LP+Grad at n={n3}, budget={phase3_end - t_start:.0f}s")

    if best_seq is not None:
        p1024 = _interpolate_profile(best_seq, n3)
    else:
        p1024 = _make_linear(n3, 1.5)

    per_lp = min(phase3_end, time.monotonic() + 200.0)
    pa3, sa3 = _lp_optimize(p1024, per_lp, rng)
    _update(pa3, sa3)
    print(f"  LP 1024: score={sa3:.8f}")

    per_grad = min(phase3_end, time.monotonic() + 150.0)
    pg3, sg3 = _adam_gradient_optimize(pa3, per_grad, rng, lr=0.10)
    _update(pg3, sg3)
    print(f"  Grad 1024: score={sg3:.8f}")

    # --- Phase 4: Try n=256 with more iterations ---
    n4 = 256
    phase4_end = min(deadline, t_start + 1400.0)
    print(f"Phase 4: n={n4}, budget={phase4_end - t_start:.0f}s")

    p256 = _interpolate_profile(best_seq, n4) if best_seq is not None else _make_linear(n4, 1.5)
    per_lp4 = min(phase4_end, time.monotonic() + 100.0)
    pa4, sa4 = _lp_optimize(p256, per_lp4, rng)
    _update(pa4, sa4)
    per_grad4 = min(phase4_end, time.monotonic() + 80.0)
    pg4, sg4 = _adam_gradient_optimize(pa4, per_grad4, rng, lr=0.15)
    _update(pg4, sg4)
    print(f"  n=256 best: score={sg4:.8f}")

    # --- Phase 5: Additional exploration ---
    phase5_end = min(deadline, t_start + 1600.0)
    print(f"Phase 5: Exploration, budget={phase5_end - t_start:.0f}s")

    for n_try in (512, 1024):
        if time.monotonic() > phase5_end - 100.0:
            break
        for _ in range(3):
            if time.monotonic() > phase5_end - 100.0:
                break
            p_rand = _random_profile(n_try, rng)
            per_lp5 = min(phase5_end - 100.0, time.monotonic() + 60.0)
            pa5, sa5 = _lp_optimize(p_rand, per_lp5, rng)
            _update(pa5, sa5)
            if sa5 < best_score:
                per_grad5 = min(phase5_end - 100.0, time.monotonic() + 50.0)
                pg5, sg5 = _adam_gradient_optimize(pa5, per_grad5, rng, lr=0.10)
                _update(pg5, sg5)
                print(f"  Explore n={n_try}: score={sg5:.8f}")

    # --- Phase 6: Final LP rounds on best ---
    if best_seq is not None and time.monotonic() < deadline - 30.0:
        n_best = best_seq.size
        per_final = min(deadline - 30.0, time.monotonic() + 60.0)
        if per_final > 10.0:
            pa_f, sa_f = _lp_optimize(best_seq, per_final, rng)
            _update(pa_f, sa_f)
            per_grad_f = min(deadline - 30.0, time.monotonic() + 30.0)
            if per_grad_f > 5.0:
                pg_f, sg_f = _adam_gradient_optimize(pa_f, per_grad_f, rng, lr=0.08)
                _update(pg_f, sg_f)

    # --- Final evaluation ---
    print(f"Pre-eval best: {best_score:.8f}, n={best_seq.size if best_seq is not None else 0}")

    if best_seq is not None:
        final_seq = best_seq
        final_score = best_score
        # Exact rescore
        if final_seq.size <= 4096:
            sc_exact = _true_score(final_seq)
            if np.isfinite(sc_exact) and sc_exact < final_score:
                final_score = sc_exact
        # Try evaluator
        if final_seq.size <= 4096:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval) and sc_eval < final_score:
                    final_score = sc_eval
            except Exception:
                pass
        print(f"Final score: {final_score:.8f}, n={final_seq.size}")
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_linear(512, 1.5)
    return [float(x) for x in fallback]
```
