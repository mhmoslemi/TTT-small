import numpy as np
from scipy.optimize import linprog
import time


def fast_score(s):
    """Compute 2n*max(conv(s,s))/(sum(s))^2."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def build_conv_matrix(s):
    """Build (2n-1) x n matrix A where A[k,j] = s[k-j] if valid."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = s
    return A


def solve_direction(s, M, time_limit=8.0):
    """Solve LP: maximize sum(b) s.t. (s*b)[k] <= M, b >= 0."""
    n = len(s)
    A = build_conv_matrix(s)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        for k in peak_indices:

            best_sc = sc
            best_s = s_new.copy()
            s = s_new

        if h_r.sum() > 0.01:
            h_r = h_r / h_r.sum()
            candidates.append((fast_score(h_r), h_r.copy()))

    # Exponential ramp
    for alpha in [-3.0, -1.5, 0.0, 1.5, 3.0, -5.0, 5.0]:
        x = np.arange(n) / max(n - 1, 1)
        s = np.exp(alpha * (x - 0.5))
        s = np.maximum(s, 0.001)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Quadratic / Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for width in [0.1, 0.2, 0.3, 0.5]:
            x = np.arange(n) / max(n - 1, 1)
            s = np.exp(-0.5 * ((x - center) / width) ** 2)
            s = s / s.sum()
            candidates.append((fast_score(s), s.copy()))

    # Two-level
    for frac in [0.3, 0.4, 0.5, 0.6]:
        k = int(frac * n)
        for ratio in [0.3, 0.7, 2.0]:
            s = np.ones(n)
            s[:k] = ratio
            s = s / s.sum()
            candidates.append((fast_score(s), s.copy()))

    # Alternating
    for ratio in [0.3, 0.5, 0.7, 2.0]:
        s = np.ones(n)
        s[::2] = ratio
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Random
    for _ in range(6):
        s = rng.random(n) + 0.01
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    # Smooth random (Gaussian-filtered)
    for _ in range(4):
        s = rng.random(n) + 0.01
        # Simple moving average smoothing
        kernel = np.ones(5) / 5.0
        s = np.convolve(s, kernel, mode='same')
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        candidates.append((fast_score(s), s.copy()))

    candidates.sort(key=lambda x: x[0])
    return candidates


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    LP-guided local search with peak-reduction and coordinate refinement.
    Tries multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    global_best_score = float('inf')
    global_best_seq = None

    # Try to load height_sequence_1
    hseq = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size > 10 and np.isfinite(h).all() and h.sum() > 0.01:
            hseq = h
            print(f"Loaded height_sequence_1, length={h.size}, sum={h.sum():.4f}")
            print(f"  Score: {fast_score(h / h.sum()):.6f}")
    except Exception:
        pass

    # Sequence lengths to try (larger = more freedom but slower LP)
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        print(f"\n{'='*60}")
        print(f"n={n}, remaining time={remaining:.0f}s")
        print(f"{'='*60}")

        # Generate candidates
        candidates = generate_candidates(n, rng, hseq)
        print(f"Generated {len(candidates)} candidates, best initial score={candidates[0][0]:.6f}")

        # Run LP optimization on top candidates
        max_cands = 4
        time_per_cand = remaining / (max_cands * 2) - 10  # leave room for refinement

        for sc0, s0 in candidates[:max_cands]:
            if time.monotonic() > deadline - 60:
                break
            print(f"  LP optimize from score={sc0:.6f}")
            t_start = time.monotonic()

            best_s, best_sc = lp_iterate(s0, rng, deadline=time.monotonic() + time_per_cand,
                                          max_iters=600, verbose=True)
            t_elapsed = time.monotonic() - t_start
            print(f"  After LP: score={best_sc:.6f} (took {t_elapsed:.1f}s)")

            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** New global best: {global_best_score:.6f} (n={n})")

            # Peak reduction refinement
            if time.monotonic() < deadline - 60:
                best_s2, best_sc2 = peak_reduction(best_s, rng, deadline=time.monotonic() + 60,
                                                   n_iters=100)
                if best_sc2 < global_best_score:
                    global_best_score = best_sc2
                    global_best_seq = best_s2.copy()
                    print(f"  ** After peak reduction: {global_best_score:.6f}")

                # Coordinate refinement
                if time.monotonic() < deadline - 30:
                    best_s3, best_sc3 = coordinate_refine(best_s2, rng,
                                                           deadline=time.monotonic() + 45,
                                                           max_rounds=30)
                    if best_sc3 < global_best_score:
                        global_best_score = best_sc3
                        global_best_seq = best_s3.copy()
                        print(f"  ** After coord refine: {global_best_score:.6f}")

        # Random restarts if time permits
        restarts = 0
        while time.monotonic() < deadline - 90 and restarts < 3:
            r = rng.random(n) + 0.05
            r = r / r.sum()
            print(f"  Random restart {restarts+1}, initial score={fast_score(r):.6f}")
            best_s, best_sc = lp_iterate(r, rng, deadline=time.monotonic() + (deadline - time.monotonic()) / 4 - 20,
                                          max_iters=200)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Restart {restarts+1}: new best {global_best_score:.6f}")
            restarts += 1

    # Final refinement on the best sequence found
    if global_best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal refinement on best sequence (n={len(global_best_seq)}, score={global_best_score:.6f})")
        # One more round of LP
        best_s, best_sc = lp_iterate(global_best_seq, rng, deadline=time.monotonic() + 60,
                                      max_iters=100)
        if best_sc < global_best_score:
            global_best_score = best_sc
            global_best_seq = best_s.copy()
            print(f"  ** Final LP: {global_best_score:.6f}")

        # Peak reduction
        if time.monotonic() < deadline - 5:
            best_s, best_sc = peak_reduction(global_best_seq, rng, deadline=time.monotonic() + 30,
                                             n_iters=100)
            if best_sc < global_best_score:
                global_best_score = best_sc
                global_best_seq = best_s.copy()
                print(f"  ** Final peak reduction: {global_best_score:.6f}")

    # Verify with the actual evaluator
    if global_best_seq is None:
        global_best_seq = np.ones(256) / 256.0

    final_score = float(evaluate_sequence(global_best_seq.tolist()))
    print(f"\n{'='*60}")
    print(f"FINAL: verified score={final_score:.6f}, length={len(global_best_seq)}")
    print(f"{'='*60}")

    result = [float(max(0.0, min(1000.0, x))) for x in global_best_seq]
    return result
