Uses LP-based descent (SOTA approach) combined with analytical profiles
and logit-space Adam polish. Key innovation: direct LP relaxation for
principled descent directions, better initial profiles (power-law ramps
from zero), and multi-length strategy.
"""
import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        if mn > -1e-9:
            c = c - mn
        else:
            c = np.maximum(c, 0.0)
    return c


def internal_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = full_convolve_fft(a)
    if c.size == 0:
        return np.inf
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def true_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return np.inf
    c = np.convolve(a, a)
    mval = float(c.max())
    if not np.isfinite(mval):
        return np.inf
    return float(2.0 * n * mval / (s * s))


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-30:
        return None
    return a / s


def soft_mx(c, beta):
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
    mval = float(c.max())
    if beta <= 1e-14:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return mval, w
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(np.sum(ex))
    if not np.isfinite(s) or s <= 0.0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    val = mval + beta * float(np.log(s))
    w = ex / s
    return val, w


def corr_weights(w, p):
    n = len(p)
    N = next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    fp = np.fft.rfft(p[::-1], N)
    h = np.fft.irfft(fw * fp, N)
    return h[n - 1:2 * n - 1]


def softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -60.0, 60.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    p = e + 1e-15
    s = float(p.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.ones_like(e) / float(e.size)
    else:
        p = p / s
    return p


def make_profile_power(n, p):
    """a_i = (i+1)^p / sum, i=0..n-1"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a = a / s
    return a


def make_profile_ramp0(n, p=1.0):
    """a_i = (i)^p for i=0..n-1, a_0=0"""
    idx = np.arange(0, n, dtype=np.float64)
    a = idx ** p
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a = a / s
    return a


def make_profile_sine(n, freq=1.0, phase=0.0):
    t = np.linspace(0.0, np.pi, n)
    a = np.sin(freq * t + phase) ** 2
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s > 0:
        a = a / s
    return a


def make_profile_gaussian(n, sigma_frac=0.3, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if s > 0:
        a = a / s
    return a


def make_profile_quadratic_peak(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    s = float(a.sum())
    a = a / s
    return a


def make_initial_profiles(n, rng):
    """Generate diverse initial profiles of length n."""
    profiles = []
    # Power law ramps from 0
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(make_profile_ramp0(n, p))
    # Power law with offset
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(make_profile_power(n, p))
    # Sine profiles
    profiles.append(make_profile_sine(n, 1.0))
    profiles.append(make_profile_sine(n, 0.5))
    profiles.append(make_profile_sine(n, 2.0))
    # Gaussian profiles
    for sf in [0.15, 0.25, 0.4]:
        for c in [0.3, 0.5, 0.7]:
            profiles.append(make_profile_gaussian(n, sf, c))
    # Quadratic peak
    for al in [1.0, 2.0, 3.0]:
        profiles.append(make_profile_quadratic_peak(n, al))
    # Random smooth profiles
    for _ in range(15):
        coeff = rng.normal(0, 1, 8)
        t = np.linspace(0, 1, n)
        a = 1.0
        for k, c in enumerate(coeff):
            a = a + c * np.cos(np.pi * (k + 1) * t)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-6:
            profiles.append(a / s)
    # Random power combinations
    for _ in range(10):
        idx = np.linspace(0, 1, n)
        a = idx ** rng.uniform(0.3, 3.0) + (1 - idx) ** rng.uniform(0.3, 3.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-6:
            profiles.append(a / s)
    return profiles


def build_conv_matrix(a, n):
    """Build the 1023 x 512 matrix A where A[k,j] = a[k-j] if valid."""
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        for j in range(i_lo, i_hi + 1):
            A[k, j] = a[k - j]
    return A


def lp_descent_step(a, M, max_time=2.0):
    """
    Solve: maximize sum(b) subject to b >= 0, conv(a,b)[k] <= M.
    Returns optimal b or None if infeasible/failed.
    """
    n = len(a)
    m = 2 * n - 1
    t0 = time.monotonic()
    
    # Build constraint matrix
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi+1] = a[k - np.arange(i_lo, i_hi+1)]
    
    c_rp = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(m, M)
    
    try:
        res = linprog(c_rp, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': max_time})
        if res.status == 0:
            return res.x
        elif res.status == 1:  # optimal but special
            if res.x is not None and np.all(res.x >= -1e-10):
                return np.maximum(res.x, 0.0)
    except Exception:
        pass
    
    # Fallback: smaller problem or dual approach
    if time.monotonic() - t0 > max_time:
        return None
    return None


def lp_optimize(a0, budget_s, rng, verbose=True):
    """
    LP-based optimization: iteratively solve LP to find descent direction.
    """
    a = np.asarray(a0, dtype=np.float64).copy()
    n = a.size
    a = a / a.sum()
    
    c = full_convolve_fft(a)
    M = float(c.max())
    S = float(a.sum())
    score = 2.0 * n * M / (S * S)
    
    best_a = a.copy()
    best_score = score
    
    if verbose:
        print(f"  LP init: score={score:.10f}, M={M:.6e}, S={S:.6e}")
    
    no_improve = 0
    t_start = time.monotonic()
    step_count = 0
    
    while time.monotonic() - t_start < budget_s:
        c = full_convolve_fft(a)
        M = float(c.max())
        S = float(a.sum())
        
        # Solve LP
        b = lp_descent_step(a, M, max_time=1.5)
        if b is None:
            # Fallback: use gradient step
            beta = max(1e-12, 0.1 * M)
            _, w = soft_mx(c, beta)
            gp = 2.0 * corr_weights(w, a)
            a_new = a - 0.01 * gp
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = internal_score(a_new)
                if sc < best_score:
                    best_a = a_new.copy()
                    best_score = sc
                    no_improve = 0
                else:
                    no_improve += 1
                a = a_new
            else:
                break
            step_count += 1
            continue
        
        b = np.maximum(b, 0.0)
        sb = float(b.sum())
        if sb < 1e-10:
            break
        
        # Rescale b to have same sum as a
        b_rescaled = b * (S / sb)
        
        # Take a step: a_new = (1-t)*a + t*b_rescaled
        # Try several step sizes
        best_step_a = a
        best_step_score = score
        for t in [0.1, 0.2, 0.3, 0.5, 0.3]:
            a_try = (1.0 - t) * a + t * b_rescaled
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-10:
                continue
            a_try = a_try / st
            sc_try = internal_score(a_try)
            if sc_try < best_step_score:
                best_step_score = sc_try
                best_step_a = a_try.copy()
        
        if best_step_score < best_score - 1e-12:
            a = best_step_a
            best_a = a.copy()
            best_score = best_step_score
            no_improve = 0
        else:
            # Even smaller step
            a_try = 0.98 * a + 0.02 * b_rescaled
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st > 1e-10:
                a_try = a_try / st
                sc_try = internal_score(a_try)
                if sc_try < best_score:
                    a = a_try
                    best_a = a.copy()
                    best_score = sc_try
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            a = 0.95 * a + 0.05 * b_rescaled
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        step_count += 1
        if verbose and step_count == 1:
            print(f"  LP step 1: score={best_score:.10f}")
        if verbose and step_count % 20 == 0:
            print(f"  LP step {step_count}: score={best_score:.10f}")
        
        if no_improve > 50:
            # Perturb and restart
            a = best_a + 0.05 * rng.normal(0, 1, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
    
    return best_a, best_score


def adam_polish(a0, budget_s, rng, lr=0.05, reg=1e-8):
    """Adam optimization in logit space."""
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    n = a0.size
    
    z = np.log(a0)
    z -= float(np.mean(z))
    
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    adam_t = 0
    
    a = a0.copy()
    c = full_convolve_fft(a)
    M = float(c.max())
    beta = max(1e-12, 0.2 * M)
    
    best_a = a0.copy()
    best_z = z.copy()
    best_score = 2.0 * n * M
    no_imp = 0
    base_lr = lr
    
    t_start = time.monotonic()
    while time.monotonic() - t_start < budget_s:
        L, w = soft_mx(c, beta)
        gp = 2.0 * corr_weights(w, a)
        vp = gp + reg * a
        dot = float(np.dot(a, vp))
        gz = a * (vp - dot)
        gz -= float(np.mean(gz))
        
        if not np.all(np.isfinite(gz)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        adam_t += 1
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** adam_t)
        vh = v / (1 - b2 ** adam_t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        
        if not np.all(np.isfinite(z)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        z = np.clip(z, -60.0, 60.0)
        z -= float(np.mean(z))
        a = softmax_from_z(z)
        c = full_convolve_fft(a)
        M = float(c.max())
        score = 2.0 * n * M
        
        if np.isfinite(score):
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                best_z = z.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp % 300 == 0:
                    lr = max(0.01, lr * 0.98)
        else:
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            continue
        
        if adam_t % 200 == 0:
            beta = max(1e-12, beta * 0.7)
        
        if no_imp > max(800, 8 * n):
            z = 0.7 * best_z + 0.5 * rng.normal(size=n)
            z = np.clip(z, -60.0, 60.0)
            z -= float(np.mean(z))
            a = softmax_from_z(z)
            c = full_convolve_fft(a)
            M = float(c.max())
            beta = max(1e-10, 0.2 * M)
            m.fill(0.0)
            v.fill(0.0)
            adam_t = 0
            lr = base_lr
            no_imp = 0
    
    return best_a, best_score


def direct_nograd_search(a0, budget_s, rng, n_perturb=5):
    """Random perturbation search with greedy acceptance."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    n = a.size
    a = a / a.sum()
    
    c = full_convolve_fft(a)
    M = float(c.max())
    S = float(a.sum())
    score = 2.0 * n * M / (S * S)
    
    best_a = a.copy()
    best_score = score
    
    t_start = time.monotonic()
    no_imp = 0
    temp = 0.1
    step_count = 0
    
    while time.monotonic() - t_start < budget_s:
        # Random perturbation of a few elements
        idx = rng.choice(n, size=min(n_perturb, n), replace=False)
        a_new = a.copy()
        for i in idx:
            a_new[i] = max(0.0, a_new[i] * np.exp(rng.normal(0, 0.3)))
        
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        sc_new = internal_score(a_new)
        
        if sc_new < score:
            a = a_new
            score = sc_new
            if score < best_score:
                best_a = a.copy()
                best_score = score
                no_imp = 0
        else:
            no_imp += 1
            # Accept with probability exp(-(sc_new - score) / temp)
            if rng.random() < np.exp(-(sc_new - score) / max(temp, 1e-10)):
                a = a_new
                score = sc_new
        
        step_count += 1
        if step_count % 500 == 0:
            temp *= 0.95
        if no_imp > 200:
            a = best_a + 0.1 * rng.normal(0, 1, n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            score = internal_score(a)
            no_imp = 0
            temp = 0.05
    
    return best_a, best_score


def interpolate_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = p0.size
    if n0 == n_new:
        return normalize_nonneg(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    return normalize_nonneg(np.interp(x1, x0, p0))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage search:
      1. Analytical profiles (power ramps, sines, gaussians)
      2. LP-based descent optimization (SOTA approach)
      3. Adam polish in logit space
      4. Direct random perturbation search
      5. Multi-length extension and final scoring
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    candidates = []
    
    def update_best(p, sc):
        nonlocal best_seq, best_score
        if p is not None and sc is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
        if p is not None and sc is not None and np.isfinite(sc):
            if len(candidates) < 30:
                candidates.append((float(sc), np.asarray(p, dtype=np.float64).copy()))
                candidates.sort(key=lambda x: x[0])
            elif sc < candidates[-1][0]:
                candidates[-1] = (float(sc), np.asarray(p, dtype=np.float64).copy())
                candidates.sort(key=lambda x: x[0])
    
    # Phase 1: Analytical profiles at n=512
    n_main = 512
    print(f"=== Phase 1: Analytical profiles (n={n_main}) ===")
    t1 = time.monotonic()
    
    profiles = make_initial_profiles(n_main, rng)
    for p in profiles:
        sc = internal_score(p)
        update_best(p, sc)
    
    # Also try shorter lengths
    for ntry in [256, 128]:
        profiles_s = make_initial_profiles(ntry, rng)
        for p in profiles_s:
            sc = internal_score(p)
            update_best(p, sc)
    
    print(f"Phase 1 done in {time.monotonic()-t1:.1f}s, best={best_score:.10f}")
    print(f"  n={len(best_seq) if best_seq is not None else '?'}")
    
    # Check parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 0:
            pn = normalize_nonneg(parent)
            if pn is not None:
                sc = internal_score(pn)
                update_best(pn, sc)
                print(f"  parent score={sc:.10f}, n={pn.size}")
    except Exception:
        pass
    
    # Phase 2: LP-based optimization from top candidates
    print(f"\n=== Phase 2: LP optimization ===")
    t2 = time.monotonic()
    lp_budget = min(600.0, (deadline - t_start) * 0.35)
    
    # Sort candidates by score, try top few at n_main
    candidates_at_main = [c for c in candidates if c[1].size == n_main]
    if not candidates_at_main:
        candidates_at_main = [(best_score, best_seq)]
    
    for i in range(min(4, len(candidates_at_main))):
        if time.monotonic() > deadline - 300:
            break
        p0, _ = candidates_at_main[i]
        per_budget = min(lp_budget - (time.monotonic() - t2), 120.0)
        if per_budget < 10:
            break
        print(f"  LP from candidate {i} (score={candidates_at_main[i][0]:.6f}), budget={per_budget:.0f}s")
        p_best, sc = lp_optimize(p0, per_budget, rng, verbose=True)
        update_best(p_best, sc)
        print(f"  LP result: score={sc:.10f}")
    
    print(f"Phase 2 done in {time.monotonic()-t2:.1f}s, best={best_score:.10f}")
    
    # Phase 3: Adam polish
    print(f"\n=== Phase 3: Adam polish ===")
    t3 = time.monotonic()
    adam_budget = min(500.0, (deadline - t_start) * 0.3)
    
    top_candidates = sorted(candidates, key=lambda x: x[0])[:8]
    for i, (sc_c, p_c) in enumerate(top_candidates):
        if time.monotonic() > deadline - 200:
            break
        # Interpolate to n_main if needed
        p_work = p_c if p_c.size == n_main else interpolate_profile(p_c, n_main)
        if p_work is None:
            continue
        per_budget = min(adam_budget - (time.monotonic() - t3), 60.0)
        if per_budget < 5:
            break
        p_best, sc = adam_polish(p_work, per_budget, rng, lr=0.04)
        update_best(p_best, sc)
        if i < 4:
            print(f"  Adam {i}: score={sc:.10f} (n={p_best.size})")
    
    # Also polish at original lengths for shorter sequences
    for i, (sc_c, p_c) in enumerate(top_candidates[:4]):
        if time.monotonic() > deadline - 200:
            break
        n_orig = p_c.size
        if n_orig != n_main and n_orig >= 128:
            per_budget = min(30.0, (deadline - time.monotonic()) * 0.1)
            if per_budget < 3:
                break
            p_best, sc = adam_polish(p_c, per_budget, rng, lr=0.03)
            update_best(p_best, sc)
    
    print(f"Phase 3 done in {time.monotonic()-t3:.1f}s, best={best_score:.10f}")
    
    # Phase 4: Direct random perturbation search
    print(f"\n=== Phase 4: Random perturbation ===")
    t4 = time.monotonic()
    remain = deadline - time.monotonic()
    if remain > 100:
        p0 = best_seq if best_seq.size == n_main else interpolate_profile(best_seq, n_main)
        per_budget = min(200.0, remain * 0.5)
        p_best, sc = direct_nograd_search(p0, per_budget, rng, n_perturb=3)
        update_best(p_best, sc)
        print(f"  Perturbation: score={sc:.10f}")
    
    # Phase 5: Try extending to longer sequences
    print(f"\n=== Phase 5: Length extension ===")
    t5 = time.monotonic()
    remain = deadline - time.monotonic()
    if remain > 80:
        for target_n in [1024, 2048]:
            if time.monotonic() > deadline - 30:
                break
            p0 = interpolate_profile(best_seq, target_n)
            if p0 is None:
                continue
            sc0 = internal_score(p0)
            update_best(p0, sc0)
            if sc0 < 1.6:
                per_budget = min(60.0, remain * 0.3)
                if per_budget > 5:
                    p_best, sc = adam_polish(p0, per_budget, rng, lr=0.02)
                    update_best(p_best, sc)
                    print(f"  n={target_n}: score={sc:.10f}")
    
    print(f"Phase 5 done, best={best_score:.10f}")
    
    # Phase 6: Final exact scoring
    print(f"\n=== Phase 6: Final scoring ===")
    final_seq = best_seq
    final_score = best_score
    
    # Exact score for the best
    if final_seq is not None:
        if final_seq.size <= 4096:
            sc_exact = true_score(final_seq)
            if np.isfinite(sc_exact) and sc_exact < final_score:
                final_score = sc_exact
        
        # Try exact on top candidates
        for sc_c, p_c in candidates[:10]:
            if p_c.size <= 4096:
                sc_e = true_score(p_c)
                if np.isfinite(sc_e) and sc_e < final_score:
                    final_score = sc_e
                    final_seq = p_c
    
    # Call evaluator for final validation
    if final_seq is not None and final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                final_score = min(final_score, sc_eval)
        except Exception:
            pass
    
    # Final cleanup
    final_seq = normalize_nonneg(final_seq)
    if final_seq is None:
        final_seq = make_profile_ramp0(512, 1.0)
        final_score = internal_score(final_seq)
    
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0:
        final_seq = final_seq / s
    else:
        final_seq = np.ones(512) / 512.0
    
    print(f"\n=== FINAL: score={final_score:.10f}, n={final_seq.size} ===")
    print(f"  min={final_seq.min():.6e}, max={final_seq.max():.6e}")
    
    return [float(x) for x in final_seq]
