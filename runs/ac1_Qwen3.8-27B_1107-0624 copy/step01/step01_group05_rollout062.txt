N = 4096
FFT_SELF = 1 << (2*N - 2).bit_length()  # 8192
FFT_GRAD = 1 << (3*N - 2).bit_length()  # 32768

_best_score = float('inf')

def sanitize_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -40.0, 40.0)
    return z

def probs_from_z(z):
    z = sanitize_z(z)
    m = float(np.max(z)) if z.size else 0.0
    expz = np.exp(z - m)
    exp0 = np.exp(-m)
    Z = exp0 + float(np.sum(expz))
    p = np.empty(N, dtype=np.float64)
    p[0] = exp0 / Z
    p[1:] = expz / Z
    return p

def self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real
    c = c[:2*N - 1]
    np.maximum(c, 0.0, out=c)
    return c

    logp = np.log(p)

    z = sanitize(z)
    m = float(np.max(z)) if z.size else 0.0

    Z = e0 + float(np.sum(ez))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Diverse structured initializations
2. Fast multiplicative weight updates (mirror descent in entropy geometry)
3. L-BFGS-B with smooth peak surrogates (top-k weighted)
4. Direct subgradient polishing targeting active peaks
"""
import numpy as np
import time
from scipy.optimize import minimize

N = 4096
FFT_SELF = 8192       # next pow2 >= 2N-1 = 8191
FFT_FULL = 16384      # next pow2 >= 3N-2 = 12286

_best_score = float('inf')
_best_p = None
_deadline = 0.0

class TimeUp(Exception):
    pass

def check_time():
    if time.monotonic() >= _deadline:
        raise TimeUp

def to_z(p):
    p = np.maximum(p, 1e-30)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -40.0, 40.0)

def z_to_p(z):
    z = np.where(np.isfinite(z), z, 0.0)
    z = np.clip(z, -40.0, 40.0)
    m = float(np.max(z))
    ez = np.exp(z - m)
    e0 = np.exp(-m)
    Z = e0 + float(np.sum(ez))
    p = np.empty(N, dtype=np.float64)
    p[0] = e0 / Z
    p[1:] = ez / Z
    return p

def self_conv(p):
    f = np.fft.rfft(p, FFT_SELF)
    c = np.fft.irfft(f * f, FFT_SELF).real
    c = c[:2 * N - 1]
    np.maximum(c, 0.0, out=c)
    return c

def conv_grad(p, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. p.
    grad[i] = 2 * sum_j p[j] * w[i+j]
    Computed via FFT: 2 * (p_rev * w)[N-1 : 2N-1]
    """
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, FFT_FULL)
    fw = np.fft.rfft(w, FFT_FULL)
    conv = np.fft.irfft(fp_rev * fw, FFT_FULL).real
    return 2.0 * conv[N - 1:2 * N - 1]

def objective(p):
    c = self_conv(p)
    return 2.0 * N * float(np.max(c))

def update_best(p, score):
    global _best_score, _best_p
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = p.copy()

def score_and_update(p):
    s = objective(p)
    update_best(p, s)
    return s

def multiplicative_weights(p, n_iters, eta0=500.0, decay=3000, top_k=16):
    """Fast multiplicative weight (mirror descent) optimization."""
    p = p.copy()
    s = score_and_update(p)
    
    for it in range(n_iters):
        if it % 4000 == 0:
            check_time()
        
        c = self_conv(p)
        cmax = float(np.max(c))
        
        K = min(top_k, 2 * N - 1)
        idx = np.argpartition(c, -K)[-K:]
        
        # Soft exponential weighting concentrated on peaks
        temp = 0.08 * cmax + 1e-30
        ew = np.exp(c[idx] / temp)
        w = np.zeros(2 * N - 1)
        w[idx] = ew / float(np.sum(ew))
        
        gp = conv_grad(p, w)
        
        eta = eta0 / (1.0 + float(it) / decay)
        
        # Multiplicative update in simplex
        log_p = np.log(p + 1e-30) - eta * gp
        log_p -= float(np.max(log_p))
        p = np.exp(log_p)
        p_sum = float(np.sum(p))
        p = p / p_sum
        p = np.maximum(p, 1e-30)
        p = p / float(np.sum(p))
        
        if it % 200 == 0 or it == n_iters - 1:
            s = score_and_update(p)
    
    return p

def lbfgs_from_p(p0, top_k, maxfun=300, maxiter=150):
    """L-BFGS-B optimization in logit space with top-k weighted objective."""
    z0 = to_z(p0)
    
    def obj_grad(x):
        check_time()
        z = np.where(np.isfinite(x), x, 0.0)
        z = np.clip(z, -40.0, 40.0)
        p = z_to_p(z)
        c = self_conv(p)
        
        s = 2.0 * N * float(np.max(c))
        update_best(p, s)
        
        K = min(top_k, 2 * N - 1)
        idx = np.argpartition(c, -K)[-K:]
        cv = c[idx]
        # Quadratic power weighting emphasizes the max peak
        pw = cv * cv
        pw_sum = float(np.sum(pw))
        w = np.zeros(2 * N - 1)
        if pw_sum > 0:
            w[idx] = pw / pw_sum
        
        J = float(np.dot(w, c))
        
        gp = conv_grad(p, w)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gz = np.where(np.isfinite(gz), gz, 0.0)
        gz = np.clip(gz, -1e4, 1e4)
        
        return J, gz
    
    bounds = [(-40.0, 40.0)] * (N - 1)
    
    try:
        minimize(obj_grad, z0, method='L-BFGS-B', jac=True,
                 bounds=bounds,
                 options={'maxfun': maxfun, 'maxiter': maxiter,
                          'ftol': 1e-15, 'gtol': 1e-10,
                          'maxls': 50, 'disp': False})
    except TimeUp:
        pass
    except Exception:
        pass

def subgradient_polish(p, n_iters, rng):
    """Direct subgradient steps on the probability simplex."""
    p = p.copy()
    s = score_and_update(p)
    stall_count = 0
    
    for it in range(n_iters):
        check_time()
        
        c = self_conv(p)
        cmax = float(np.max(c))
        
        K = 8
        idx = np.argpartition(c, -K)[-K:]
        
        # Sharp weighting on top peaks
        w = np.zeros(2 * N - 1)
        pv = c[idx]
        sh_w = np.exp(pv / (0.04 * cmax + 1e-30))
        w[idx] = sh_w / float(np.sum(sh_w))
        
        gp = conv_grad(p, w)
        # Project to tangent space
        gp_t = gp - float(np.dot(gp, p))
        gnorm = float(np.linalg.norm(gp_t))
        
        if gnorm < 1e-15:
            break
        
        direction = gp_t / gnorm
        best_s = s
        best_p = p
        
        for step in (0.08, 0.04, 0.02, 0.01, 0.005):
            pt = p - step * direction
            pt = np.maximum(pt, 0.0)
            pt_sum = float(np.sum(pt))
            if pt_sum < 1e-10:
                continue
            pt /= pt_sum
            st = objective(pt)
            if st < best_s - 1e-16:
                best_s = st
                best_p = pt
        
        if best_s < s - 1e-16:
            p = best_p
            s = best_s
            update_best(p, s)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                # Random restart in neighborhood
                z = to_z(p) + rng.normal(0, 0.08, N - 1)
                p = z_to_p(z)
                s = score_and_update(p)
                stall_count = 0
    
    return p
