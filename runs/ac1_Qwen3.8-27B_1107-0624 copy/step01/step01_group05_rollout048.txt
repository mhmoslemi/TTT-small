def _grad_p_from_weights(p, w):
    fp = np.fft.rfft(p[::-1].copy(), FFT_GRAD)
    fw = np.fft.rfft(w, FFT_GRAD)
    conv = np.fft.irfft(fp * fw, FFT_GRAD).real
    h = conv[N - 1:2 * N - 1]
    return 2.0 * h


def _score_p(p):
    c = _self_conv(p)
    return 2.0 * float(N) * float(np.max(c))


def _update_best(z, score):
    global _best_score, _best_z
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_z = np.array(z, copy=True)


def _z_from_sequence(seq):
    s = np.asarray(seq, dtype=np.float64)
    s = np.where(np.isfinite(s), s, 0.0)
    s = np.maximum(s, 0.0)
    total = float(np.sum(s))
    if total < 1e-12:
        return np.zeros(N - 1, dtype=np.float64)
    p = s / total
    p = np.maximum(p, 1e-12)
    logp = np.log(p)
    z = logp[1:] - logp[0]
    return np.clip(z, -40.0, 40.0)


def _objective_and_grad(x, alpha):
    x = _sanitized_z(x)
    p = _probs_from_z(x)
    c = _self_conv(p)
    hard = 2.0 * float(N) * float(np.max(c))
    _update_best(x, hard)
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        J = cmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s) / alpha
        w = e / s
    gp = _grad_p_from_weights(p, w)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    return float(J), gz


def _run_lbfgsb(z0, alpha, max_fun, deadline):
    x0 = _sanitized_z(z0)
    max_fun = int(max(60, max_fun))
    max_iter = int(max(40, max_fun // 2))
    try:
        minimize(
            _objective_and_grad,
            x0,
            args=(float(alpha),),
            method="L-BFGS-B",
            jac=True,
            bounds=BOUNDS,
            options={
                "maxfun": max_fun,
                "maxiter": max_iter,
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 50,
                "disp": False,
            },
        )
    except Exception:
        pass


def _subgradient_polish(z0, deadline, rng):
    z = _sanitized_z(z0)
    p = _probs_from_z(z)
    score = _score_p(p)
    _update_best(z, score)

    noise = 0.08
    fails = 0
    iters = 0

    while time.monotonic() < deadline and iters < 20000:
        iters += 1
        if time.monotonic() >= deadline:
            break

        c = _self_conv(p)
        cmax = float(np.max(c))

        # Try multiple K values for the active peak set
        best_step_score = score
        best_step_z = z
        best_step_p = p
        best_step_dir = None
        best_step_val = 0.0

        for K in (4, 6, 8, 12, 16):
            idx = np.argpartition(c, -K)[-K:]
            w = np.zeros_like(c)
            # Use softmax weighting over the top K peaks
            cv = c[idx]
            cv_max = float(np.max(cv))
            wv = np.exp(2000.0 * (cv - cv_max))
            wv_sum = float(np.sum(wv))
            if wv_sum < 1e-30:
                wv = np.ones_like(cv) / K
            else:
                wv = wv / wv_sum
            w[idx] = wv

            gp = _grad_p_from_weights(p, w)
            sdot = float(np.dot(gp, p))
            gz = p[1:] * (gp[1:] - sdot)
            gnorm = float(np.linalg.norm(gz))
            if gnorm < 1e-14:
                continue
            direction = gz / gnorm

            # Line search: try multiple step sizes
            for step in (0.3, 0.15, 0.075, 0.037, 0.018):
                zt = z - step * direction
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < best_step_score - 1e-13:
                    best_step_score = st
                    best_step_z = zt
                    best_step_p = pt

        if best_step_score < score - 1e-13:
            z = best_step_z
            p = best_step_p
            score = best_step_score
            _update_best(z, score)
            fails = 0
            noise = min(0.2, noise * 1.15)
        else:
            # Try random perturbation
            improved = False
            for attempt in range(4):
                zt = z + rng.normal(0.0, noise, N - 1)
                zt = _sanitized_z(zt)
                pt = _probs_from_z(zt)
                st = _score_p(pt)
                if st < score - 1e-13:
                    z = zt
                    p = pt
                    score = st
                    _update_best(z, score)
                    improved = True
                    break
            if improved:
                fails = 0
                noise = min(0.2, noise * 1.2)
            else:
                fails += 1
                noise = max(0.01, noise * 0.85)
                if fails > 80:
                    break


def _peak_shaving(z0, deadline, rng):
    """Targeted peak reduction: identify peaks and directly reduce contributing p values."""
    z = _sanitized_z(z0)
    p = _probs_from_z(z)
    score = _score_p(p)
    _update_best(z, score)
    iters = 0

    while time.monotonic() < deadline and iters < 500:
        iters += 1
        c = _self_conv(p)
        cmax = float(np.max(c))

        # Find top peaks
        K = 6
        peak_idx = np.argpartition(c, -K)[-K:]
        peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1]]

        # For each peak, find the most contributing pair
        # and reduce those p values
        reduce_mask = np.zeros(N, dtype=np.float64)
        add_mask = np.zeros(N, dtype=np.float64)

        for k in peak_idx:
            # Contributing pairs: (i, k-i) for valid i
            i_min = max(0, k - N + 1)
            i_max = min(N - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                if 0 <= j < N:
                    prod = p[i] * p[j]
                    if prod > 0:
                        reduce_mask[i] += prod
                        reduce_mask[j] += prod

        # Normalize masks
        rsum = float(np.sum(reduce_mask))
        asum = float(np.sum(p))
        if rsum < 1e-20:
            break

        # Scale factor for how much to reduce
        scale = 0.05
        delta = scale * reduce_mask / rsum
        # Ensure we don't go negative
        delta = np.minimum(delta, p * 0.3)
        p_new = p - delta

        # Redistribute to positions with lowest contribution
        # Use inverse of reduce_mask as a guide for where to add
        low_contrib = 1.0 / (reduce_mask + 0.1 * float(np.max(reduce_mask)) + 1e-10)
        low_contrib = low_contrib / float(np.sum(low_contrib))
        mass_to_add = float(np.sum(delta))
        p_new += mass_to_add * low_contrib

        # Renormalize
        p_new = np.maximum(p_new, 0.0)
        p_new = p_new / float(np.sum(p_new))

        st = _score_p(p_new)
        if st < score - 1e-13:
            p = p_new
            score = st
            z = _z_from_sequence(p)
            _update_best(z, score)
        else:
            # Try smaller scale
            scale *= 0.5
            if scale < 0.001:
                break


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_z

    start_time = time.monotonic()
    budget_s = float(budget_s)
    end_time = start_time + max(1.0, budget_s) - 3.0

    _best_score = float("inf")
    _best_z = None
    rng = np.random.default_rng(seed)

    # Get starting point from height_sequence_1
    z0 = None
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            if base.size != N:
                old_x = np.linspace(0.0, 1.0, base.size)
                new_x = np.linspace(0.0, 1.0, N)
                base = np.interp(new_x, old_x, base)
            if float(np.sum(base)) > 0.01:
                base = base / float(np.sum(base))
                z0 = _z_from_sequence(base)
                sc = _score_p(base)
                _best_score = sc
                _best_z = z0.copy()
                print(f"Starting from height_sequence_1: score={sc:.6f}, n={N}")
    except Exception as e:
        print(f"Could not load height_sequence_1: {e}")

    if z0 is None:
        # Fallback: uniform
        z0 = np.zeros(N - 1, dtype=np.float64)
        p = _probs_from_z(z0)
        _best_score = _score_p(p)
        _best_z = z0.copy()
        print(f"Using uniform start: score={_best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"Initial best: {_best_score:.6f}, elapsed={elapsed:.1f}s")

    # Phase 1: Quick L-BFGS-B optimization (20% of time)
    phase1_end = start_time + budget_s * 0.15
    if time.monotonic() < phase1_end:
        alpha = 5.0e6
        max_fun = 300
        _run_lbfgsb(_best_z, alpha, max_fun, phase1_end)
        print(f"After Phase 1 (L-BFGS-B alpha={alpha:.0e}): best={_best_score:.6f}, t={time.monotonic()-start_time:.1f}s")

    # Phase 2: Subgradient polishing (30% of time)
    phase2_end = start_time + budget_s * 0.45
    if time.monotonic() < phase2_end:
        _subgradient_polish(_best_z, phase2_end, rng)
        print(f"After Phase 2 (subgradient): best={_best_score:.6f}, t={time.monotonic()-start_time:.1f}s")

    # Phase 3: Higher alpha L-BFGS-B (20% of time)
    phase3_end = start_time + budget_s * 0.65
    if time.monotonic() < phase3_end:
        alpha = 1.0e7
        rem = phase3_end - time.monotonic()
        max_fun = max(100, int(rem / 0.015))
        _run_lbfgsb(_best_z, alpha, max_fun, phase3_end)
        print(f"After Phase 3 (L-BFGS-B alpha={alpha:.0e}): best={_best_score:.6f}, t={time.monotonic()-start_time:.1f}s")

    # Phase 4: More subgradient + peak shaving (20% of time)
    phase4_end = start_time + budget_s * 0.85
    if time.monotonic() < phase4_end:
        _subgradient_polish(_best_z, min(phase4_end, start_time + budget_s * 0.77), rng)
        if time.monotonic() < phase4_end:
            _peak_shaving(_best_z, phase4_end, rng)
        print(f"After Phase 4: best={_best_score:.6f}, t={time.monotonic()-start_time:.1f}s")

    # Phase 5: Multi-restart from perturbed best (15% of time)
    phase5_end = end_time
    if time.monotonic() < phase5_end and _best_z is not None:
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)
        for r in range(n_restarts):
            if time.monotonic() >= phase5_end - 2:
                break
            noise_scale = 0.03 + 0.02 * r
            z_perturbed = _best_z + rng.normal(0.0, noise_scale, N - 1)
            z_perturbed = _sanitized_z(z_perturbed)
            alpha = 1.0e7
            max_fun = max(60, int(per_restart / 0.02))
            _run_lbfgsb(z_perturbed, alpha, max_fun, phase5_end - 2)
            if _best_score < 1.51:
                print(f"  Restart {r}: best={_best_score:.6f}")

    # Final polish
    if _best_z is not None and time.monotonic() < end_time - 2:
        _subgradient_polish(_best_z, end_time - 1, rng)

    # Prepare output
    if _best_z is None:
        z = np.zeros(N - 1, dtype=np.float64)
    else:
        z = _sanitized_z(_best_z)

    p = _probs_from_z(z)
    p = np.clip(p, 0.0, 1000.0)
    seq = [float(v) for v in p]

    try:
        score = float(evaluate_sequence(seq))
    except Exception:
        score = float("inf")

    if not np.isfinite(score) or score > 2.0000001:
        seq = [1.0 / float(N)] * N
        try:
            score = float(evaluate_sequence(seq))
        except Exception:
            score = 2.0

    # Safety: compare with inherited sequence
    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0.0, 1000.0)
            base_score = float(evaluate_sequence([float(v) for v in base]))
            if np.isfinite(base_score) and base_score < score:
                seq = [float(v) for v in base]
                score = base_score
    except Exception:
        pass

    print(f"FINAL: score={score:.8f}, n={len(seq)}, total_time={time.monotonic()-start_time:.1f}s")
    return seq
