Multi-phase search for non-negative sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based subgradient direction finding, parametric Adam optimization,
soft-max gradient descent, and coordinate descent for final polish.

import numpy as np
import time
from scipy.optimize import linprog


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 64:
        return np.convolve(a, a)
    N = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    if c.min() < 0 and c.min() > -1e-9:
        c = c - c.min()
    return c


def cross_convolve_fft(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([])
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)[:n + m - 1]
    if c.min() < 0 and c.min() > -1e-9:
        c = c - c.min()
    return c

    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([])
