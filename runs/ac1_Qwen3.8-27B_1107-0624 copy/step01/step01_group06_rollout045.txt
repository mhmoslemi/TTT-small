def smooth_J_and_grad(a, tau, top_k=50):
    """Compute smoothed objective and gradient using log-sum-exp over top-k.
    
    Uses a two-stage approach: first find top-k convolution values,
    then apply log-sum-exp smoothing only over those.
    """
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Find top-k indices
    k = min(top_k, len(c))
    # Use argpartition for efficiency
    if k < len(c):
        top_indices = np.argpartition(c, -k)[-k:]
    else:
        top_indices = np.arange(len(c))
    top_vals = c[top_indices]
    
    # Log-sum-exp over top-k
    c_max = np.max(top_vals)
    shifted = (top_vals - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    # Compute cross-correlation for gradient
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    v = np.zeros(size)
    for j in range(len(top_indices)):
        v[top_indices[j]] = p[j]
    
    a_pad = np.zeros(size)
    a_pad[:n] = a
    
    A = np.fft.rfft(a_pad)
    V = np.fft.rfft(v)
    cross = np.fft.irfft(np.conj(A) * V, size)[:n]
    
    # dc_soft/da = 2 * cross
    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S * S * S)
    
    return J, grad

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses top-k log-sum-exp smoothing for focused gradients, Adam with normalized
steps, adaptive schedules, periodic restarts, and coordinate-wise refinement.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def topk_smooth_grad(a, tau, top_k):
    """Compute J and gradient using top-k log-sum-exp smoothing.
    
    Returns (J, grad) where the gradient is of 2n*Tmax/S^2.
    """
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    k = min(top_k, 2 * n - 1)
    if k < 2 * n - 1:
        top_idx = np.argpartition(c, -k)[-k:]
    else:
        top_idx = np.arange(2 * n - 1)
    top_vals = c[top_idx]

    c_max = np.max(top_vals)
    shifted = (top_vals - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    if sum_exp <= 0:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation: cross[m] = sum_k p[k] * a[k-m]
    # Need FFT size >= (2n-1) + n - 1 = 3n - 2
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    v = np.zeros(size)
    v[top_idx] = p
    a_pad = np.zeros(size)
    a_pad[:n] = a

    A = np.fft.rfft(a_pad)
    V = np.fft.rfft(v)
    cross = np.fft.irfft(np.conj(A) * V, size)[:n]

    # dJ/da[m] = 4n*cross[m]/S^2 - 4n*Tmax/S^3
    grad = 4.0 * n * cross / (S * S) - 4.0 * n * Tmax / (S * S * S)

    return J, grad


def run_topk_gradient(a_init, budget_s, seed, tau_start=3.0, tau_end=0.05,
                      topk_start=300, topk_end=10, lr=0.008,
                      restart_interval=60.0, perturb_scale=0.15):

    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    start_time = time.monotonic()
    last_restart = start_time
    stall_counter = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))

        # Adaptive tau and top_k: smooth decay with slight nonlinearity
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        top_k = max(topk_end, int(topk_start * (topk_end / topk_start) ** (frac ** 0.5)))

        # Adaptive learning rate: decay slowly
        lr_t = lr / (1.0 + 0.3 * t / 5000.0)

        J, grad = topk_smooth_grad(a, tau, top_k)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        else:
            grad_normalized = grad

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad_normalized
        v = beta2 * v + (1.0 - beta2) * grad_normalized * grad_normalized
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)

                best_J = local_J
                best_a = a.copy()
                stall_counter = 0
            else:
                stall_counter += 1

        # Periodic restart with perturbation if stalled
        if stall_counter > 8 and (time.monotonic() - last_restart) > restart_interval:
            last_restart = time.monotonic()
            stall_counter = 0
            cand = best_a.copy()
            # Apply multiplicative noise to a random subset

            s = int(rng.integers(0, n - w))
            cand[s:s + w] *= np.exp(rng.normal(0, perturb_scale, w))
            # Also try global scaling perturbation

            cand = np.clip(cand, 0.0, 1000.0)

            else:
                a = best_a.copy()
            m = np.zeros(n)

                best_a = a.copy()
                print(f"    gd t={t} J={val:.6f} tau={tau:.4f} topk={top_k}")

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def coordinate_refine(a_init, budget_s, seed):
    """Coordinate-wise local search on sensitive indices."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)

    # Compute sensitivity: which coordinates most affect the max convolution
    c = convolve_fft(a)
    max_c = np.max(c)
    # Indices where convolution is close to max
    threshold = max_c * 0.99
    active_idx = np.where(c >= threshold)[0]

    # For each active convolution index k, the coordinates that contribute
    # are a[i] and a[k-i] for valid i. Collect all such coordinates.
    sensitive = set()
    for k in active_idx[:20]:  # limit to top 20 active indices
        for i in range(n):
            if 0 <= k - i < n:
                sensitive.add(i)
                sensitive.add(k - i)
    sensitive_list = np.array(sorted(sensitive))

    if len(sensitive_list) == 0:
        sensitive_list = np.arange(n)

    # Coordinate-wise search
    step_size = 0.02
    iters = 0
    while time.monotonic() < deadline:
        # Pick a random sensitive coordinate
        idx = sensitive_list[int(rng.integers(0, len(sensitive_list)))]

        # Try several step sizes and directions
        best_local = a[idx]
        best_local_J = best_J

        for delta in [step_size, -step_size, step_size * 0.5, -step_size * 0.5,
                      step_size * 2, -step_size * 2]:
            cand = a.copy()
            cand[idx] = max(0.0, a[idx] + delta)
            if cand[idx] <= 0 and a[idx] > 0:
                cand[idx] = 1e-6
            cJ = compute_J(cand)
            if cJ < best_local_J:
                best_local_J = cJ
                best_local = cand[idx]

        if best_local_J < best_J:
            a[idx] = best_local
            best_J = best_local_J
            best_a = a.copy()
            step_size = max(0.001, step_size * 0.97)
        else:
            step_size = min(0.1, step_size * 1.01)

        iters += 1
        if iters % 500 == 0:
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
            print(f"    coord iters={iters} J={best_J:.6f} step={step_size:.4f}")

    return best_a, best_J


def block_perturb_search(a_init, budget_s, seed):
    """Search by perturbing blocks of consecutive coordinates."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    best_a = a.copy()
    best_J = compute_J(a)
    scale = 0.05

    while time.monotonic() < deadline:
        # Random block
        block_len = int(rng.integers(max(1, n // 100), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - block_len)))
        end = min(n, start + block_len)

        cand = a.copy()
        # Multiplicative perturbation
        cand[start:end] *= np.exp(rng.normal(0, scale, end - start))
        cand = np.clip(cand, 0.0, 1000.0)

        cJ = compute_J(cand)
        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            scale = max(0.005, scale * 0.96)
        else:
            scale = min(0.2, scale * 1.02)

        # Periodic verify
        if int(time.monotonic() * 10) % 100 == 0:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()

    return best_a, best_J
