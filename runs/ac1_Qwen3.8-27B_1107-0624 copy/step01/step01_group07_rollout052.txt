    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)

def subgradient_multi_peak(a, c, frac=0.9):
    """Compute (scaled) subgradient using top peaks."""
    n = len(a)
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)
    peaks = np.where(c >= frac * M)[0]
    if len(peaks) == 0:

        if j_min <= j_max:
            g[j_min:j_max+1] += a[k - (np.arange(j_min, j_max+1))]
    return 2.0 * g / len(peaks)

"""
Hybrid optimization: Mirror descent (multiplicative weights) + LP refinement
to minimize 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based convolution, adaptive step sizes, and multiple strategies.

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def fast_convolution(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    return np.maximum(c, 0.0)


def subgradient_multi_peak(a, c, frac=0.9):
    n = len(a)
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)
    peaks = np.where(c >= frac * M)[0]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(c)])
    g = np.zeros(n)
    for k in peaks:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            g[idx] += a[k - idx]
    return 2.0 * g / len(peaks)


def mirror_descent(a, deadline, rng, eta0=0.1, eta_min=1e-5, eta_max=2.0, patience=200):
    n = len(a)
    a = np.clip(a, 1e-15, None).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    eta = eta0
    stall = 0
    it = 0
    c = fast_convolution(a)
    score_cur = 2.0 * n * c.max()

    while time.monotonic() < deadline:
        it += 1

        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        frac = max(0.5, 0.95 - 0.005 * (it // 50))
        g = subgradient_multi_peak(a, c, frac=frac)
        g_max = np.max(g)
        if g_max < 1e-15:
            break

        eta_eff = eta / g_max
        a_new = a * np.exp(-eta_eff * g)
        a_new = np.clip(a_new, 1e-15, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / S_new

        c_new = fast_convolution(a_new)
        score_new = 2.0 * n * c_new.max()

        if score_new <= score_cur:
            a = a_new
            c = c_new
            score_cur = score_new
            if stall == 0:
                eta = min(eta_max, eta * 1.01)
        else:
            eta = max(eta_min, eta * 0.6)
            if eta < eta_min * 2:
                noise = rng.normal(0, 0.03, n)
                a_pert = a * (1.0 + noise)
                a_pert = np.clip(a_pert, 1e-15, None)
                a = a_pert / a_pert.sum()
                c = fast_convolution(a)
                score_cur = 2.0 * n * c.max()
                eta = eta0 * 0.5
                stall = 0

            noise = rng.normal(0, 0.05, n)
            a_pert = best_a * (1.0 + noise)
            a_pert = np.clip(a_pert, 1e-15, None)
            a = a_pert / a_pert.sum()
            c = fast_convolution(a)
            score_cur = 2.0 * n * c.max()
            eta = eta0
            stall = 0

        if it % 2000 == 0:
            print(f"    MD iter {it}, best={best_score:.6f}, cur={score_cur:.6f}, eta={eta:.4f}, stall={stall}")

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_refine(a, deadline, rng, max_iter=500):
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    step = 0.1
    stall = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()

        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(1, (deadline - time.monotonic()) / 2)})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * shrink
            a_new = (1.0 - s) * best_a + s * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new /= a_new.sum()
            sc = compute_objective(a_new)
            if sc < best_score - 1e-13:
                best_a = a_new
                best_score = sc
                a = a_new
                improved = True
                break

            stall = 0
            step = min(0.5, step * 1.04)

            stall += 1

    return best_a, best_score


def projected_subgradient(a, deadline, rng, lr0=0.01, patience=300):
    n = len(a)
    a = np.clip(a, 1e-15, None).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    lr = lr0
    stall = 0
    it = 0
    momentum = np.zeros(n)
    beta = 0.9

    while time.monotonic() < deadline:
        it += 1
        c = fast_convolution(a)
        M = c.max()
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        frac = 0.85
        g = subgradient_multi_peak(a, c, frac=frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        momentum = beta * momentum + (1 - beta) * g
        a_new = a - lr * momentum
        # Project onto simplex {x >= 0, sum(x) = 1}
        # Find theta s.t. sum(max(a_new - theta, 0)) = 1
        sorted_a = np.sort(a_new)[::-1]

        idx = np.arange(1, n + 1)
        mask = (sorted_a - (cumulative - 1) / idx) > 0

            theta = (cumulative[rho - 1] - 1) / rho
        else:

            a = a_new

    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))
