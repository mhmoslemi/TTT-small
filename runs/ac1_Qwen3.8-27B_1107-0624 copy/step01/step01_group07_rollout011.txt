def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

        if 0 <= j < n:

"""
Hybrid LP + subgradient optimization with diverse restarts to minimize
2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve_fast(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None


def fast_subgradient(a, deadline, rng):
    """Fast subgradient with momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_objective(a)
    vel = np.zeros(n)
    lr = 0.3
    mu = 0.9
    stall = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a, mode='full')
        # Get top peaks
        n_peaks = max(2, n // 30)
        peak_lags = np.argsort(c)[-n_peaks:]
        
        grad = np.zeros(n)
        for k in peak_lags:
            w = c[k]
            j_min = max(0, k - (n-1))
            j_max = min(n-1, k)
            for i in range(j_min, j_max + 1):
                grad[i] += w * a[k - i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        vel = mu * vel + lr * grad
        a_new = a - vel
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            lr *= 0.9
            stall += 1
            continue
        a_new = a_new / s
        
        sc = compute_objective(a_new)
        if sc < best_sc - 1e-12:
            best_a = a_new.copy()
            best_sc = sc
            a = a_new
            stall = 0
            lr = min(1.0, lr * 1.02)
        else:
            stall += 1
            lr *= 0.97
            if stall % 50 == 0:
                heat = min(0.5, 0.1 + stall * 0.005)
                a = best_a * (1 + rng.normal(0, heat, n))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a /= a.sum()
                stall = 0
            if stall > 300:
                break
    
    return best_a, best_sc

def make_windows(n):
    """Generate diverse window shapes."""
    windows = {}
    # Cosine
    windows['cosine'] = 0.5 * (1 + np.cos(np.linspace(0, np.pi, n)))
    # Gaussian
    center = (n-1)/2
    for sf in [0.15, 0.25, 0.35, 0.5]:
        windows[f'gauss_{sf}'] = np.exp(-0.5 * ((np.arange(n) - center) / (sf * n))**2)
    # Parabolic
    windows['parab'] = 1 - np.abs(np.arange(n) - center) / (center + 1)
    # One-sided (exponential decay)
    for rate in [0.5, 1.0, 2.0, 3.0]:
        windows[f'decay_{rate}'] = np.exp(-rate * np.arange(n) / n)
    # Two-sided exponential
    windows['two_exp'] = np.exp(-abs(np.arange(n) - center) / (n * 0.3))
    # Bump in the middle
    x = np.arange(n)
    windows['bump'] = np.exp(-((x - center) / (n * 0.15))**2)
    return windows

Actually, let me think about the problem from a different angle. The key insight is:

For a sequence a of length n, the objective is `2n * max_k (a*a)_k / (sum a)^2`.

If we define the "autocorrelation ratio" as `R_k = (a*a)_k / (sum a)^2`, then we want to minimize `max_k R_k`.

Note that `sum_k R_k = 1` (since sum of all autocorrelation values = (sum a)^2).

So we're trying to make the autocorrelation as uniform as possible. The most uniform would be `R_k = 1/(2n-1)` for all k, giving objective `2n/(2n-1) ≈ 1`.

For non-negative sequences, the constraint is that the autocorrelation must be the "squared magnitude of the Fourier transform" (by the Wiener-Khinchin theorem, the autocorrelation is the inverse Fourier transform of the power spectrum).

So we want a non-negative sequence whose power spectrum is as flat as possible. This is related to "flat spectrum" design.

A flat power spectrum means |A(ω)|^2 ≈ constant for all ω, where A(ω) is the DFT of a. This means |A(ω)| ≈ constant, which is like a "Barker code" for real non-negative sequences.

This is actually a well-studied problem! The optimal solution for real non-negative sequences with flat spectrum is related to "complementary sequences" (Golay sequences), but for real values.

For the LP approach, the key insight is that it's finding the best "descent direction" in the space of non-negative sequences. The fact that it converges to a fixed point suggests that the fixed point is a local minimum of the max autocorrelation.

- Compute c = conv(a, a): O(n^2) or O(n log n) with FFT
- Compute p = softmax(c / t): O(n)
- Compute grad = conv(p, a) [or rather, a cross-correlation]: O(n^2)

Wait, grad_i = sum_k p_k * a_{k-i}. This is actually a convolution of p and a (with appropriate indexing). Specifically:
grad_i = sum_k p_k * a_{k-i} = (p * a)_{...}

Hmm, let me think about this more carefully.

p has length 2n-1 (indexed 0 to 2n-2).
a has length n (indexed 0 to n-1).

grad_i = sum_{k: 0<=k-i<n} p_k * a_{k-i} = sum_{j: 0<=j<n} p_{i+j} * a_j

This is a cross-correlation of p and a. We can compute it with a single loop over i:
for i in range(n):
    grad[i] = np.dot(a, p[i:i+n])

This is O(n^2) total, which is fine for n=512.

Actually, we can also use FFT for this: the cross-correlation of p and a is the inverse FFT of FFT(p) * conjugate(FFT(a)), zero-padded appropriately. But for n=512, the O(n^2) approach is fast enough.

Let me now write the final version:

"""
Soft-max gradient descent with LP refinement for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a, mode='full')
    return 2.0 * n * c.max() / (S * S)


def build_conv_matrix(a):
    """Build (2n-1 x n) matrix for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_direction(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def softmax_gradient(a, temperature):
    """Compute soft-max gradient of max(conv(a,a))."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    # Shift to avoid overflow
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / temperature)
    p = exp_c / exp_c.sum()
    
    # grad_i = sum_k p_k * a_{k-i} for valid k-i
    grad = np.zeros(n)
    for i in range(n):
        # j = k - i, so k = i + j, j in [0, n-1], k in [i, i+n-1]
        if i + n <= len(p):
            grad[i] = np.dot(a, p[i:i + n])
        else:
            valid_end = len(p) - i
            if valid_end > 0:
                grad[i] = np.dot(a[:valid_end], p[i:i + valid_end])
    return grad, c.max()


def optimize_softmax_grad(a, deadline, rng, t0=0.5, t_min=0.05):
    """Soft-max gradient descent with annealing."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_objective(a)
    t = t0
    lr = 0.5
    vel = np.zeros(n)
    momentum = 0.85
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        grad, M = softmax_gradient(a, t)
        
        # Normalize gradient to have mean 1 (so lr controls relative step)
        g_mean = grad.mean()
        if g_mean > 1e-15:
            grad = grad / g_mean
        
        vel = momentum * vel + lr * grad
        a_new = a - vel
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
            lr *= 0.9
            stall += 1
            continue
        a_new = a_new / s
        
        sc = compute_objective(a_new)
        if sc < best_sc - 1e-12:
            best_a = a_new.copy()
            best_sc = sc
            a = a_new
            stall = 0
            lr = min(2.0, lr * 1.01)
            t = max(t_min, t * 0.999)
        else:
            stall += 1
            lr = max(0.01, lr * 0.98)
            t = min(t0, t * 1.01)  # Increase temperature when stalled
            
            if stall % 40 == 0:
                # Perturbation
                heat = min(0.4, 0.05 + stall * 0.003)
                a = best_a * np.exp(rng.normal(0, heat, n))
                a = np.clip(a, 0.0, None)
                if a.sum() > 1e-10:
                    a /= a.sum()
                    vel = np.zeros(n)
                stall = 0
            
            if stall > 300:
                break
    
    return best_a, best_sc


def optimize_lp(a, deadline, rng):
    """LP-based optimization with restarts."""
    n = len(a)
    a = np.clip(a, 0.0, None).astype(np.float64)
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_objective(a)
    step = 0.15
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a, mode='full')
        M = c.max()
        
        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        improved = False
        for t in [step, step * 0.6, step * 0.4, step * 0.25, step * 0.15, step * 0.08]:
            cand = (1 - t) * best_a + t * g
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            sc = compute_objective(cand)
            if sc < best_sc - 1e-10:
                best_a = cand
                best_sc = sc
                a = cand
                improved = True
                break
        
        if improved:
            stall = 0
            step = min(0.5, step * 1.04)
        else:
            stall += 1
            step = max(0.003, step * 0.7)
            
            if stall > 20:
                r = rng.random()
                if r < 0.3:
                    # Blend with uniform
                    alpha = rng.uniform(0.1, 0.3)
                    a = (1 - alpha) * best_a + alpha / n
                elif r < 0.6:
                    # Gaussian smooth
                    from scipy.ndimage import gaussian_filter1d
                    sigma = rng.uniform(1, min(10, n // 15))
                    a = gaussian_filter1d(best_a, sigma=sigma)
                elif r < 0.8:
                    # Perturb
                    a = best_a * np.exp(rng.normal(0, 0.2, n))
                else:
                    # Reverse blend
                    a = 0.7 * best_a + 0.3 * best_a[::-1]
                
                a = np.clip(a, 0.0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                    stall = 0
                else:
                    break
            
            if stall > 150:
                break
        
        if it % 100 == 0:
            print(f"          LP it={it}, sc={best_sc:.6f}, step={step:.3f}")
    
    return best_a, best_sc


def make_initializations(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = {}
    center = (n - 1) / 2.0
    x = np.arange(n)
    
    inits['cosine'] = 0.5 * (1 + np.cos(np.linspace(0, np.pi, n)))
    inits['gauss_0.2'] = np.exp(-0.5 * ((x - center) / (0.2 * n)) ** 2)
    inits['gauss_0.35'] = np.exp(-0.5 * ((x - center) / (0.35 * n)) ** 2)
    inits['parab'] = np.maximum(0, 1 - np.abs(x - center) / (center + 1))
    inits['decay_l'] = np.exp(-1.5 * x / n) + 1e-10
    inits['decay_r'] = np.exp(-1.5 * (n - 1 - x) / n) + 1e-10
    inits['two_bump'] = np.exp(-((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['rand1'] = np.abs(rng.normal(0, 1, n))
    inits['rand2'] = (np.abs(rng.normal(0, 1, n)) ** 3)
    inits['three_bump'] = (np.exp(-((x - 0.2 * n) / (0.08 * n)) ** 2) +
                          np.exp(-((x - 0.5 * n) / (0.08 * n)) ** 2) +
                          np.exp(-((x - 0.8 * n) / (0.08 * n)) ** 2))
    
    return inits
