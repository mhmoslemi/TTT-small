"""
Optimization of step function coefficients to minimize J(a) = 2n*max(conv(a,a))/S^2
using a hybrid approach: smoothed gradient descent with adaptive tau, active-set
subgradient refinement, log-parameterized reparameterization, coordinate search,
and multi-start diversification. FFT-accelerated throughout.

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution via FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv(a,a))/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def smooth_J_grad(a, tau):
    """Smoothed objective and gradient via log-sum-exp of convolution peaks."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_sh = np.exp(shifted)
    sum_exp = np.sum(exp_sh)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_sh / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S ** 3)
    return J, grad


def active_set_subgrad(a, top_k=20):
    """Compute subgradient of max convolution using top-k active indices.
    Returns subgradient vector (unnormalized, pointing in ascent direction)."""
    n = len(a)
    c = convolve_fft(a)
    c_max = np.max(c)
    
    # Find active indices (within epsilon of max)
    threshold = c_max - 0.01 * c_max
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Limit to top_k most active for efficiency
    if len(active) > top_k:
        active_vals = c[active]
        top_idx = np.argsort(active_vals)[-top_k:]
        active = active[top_idx]
    
    # subgrad[m] = sum_{k in active} a[k-m] (if 0 <= k-m < n)
    # Each term a[k-m] is a shifted/reversed version of a
    subgrad = np.zeros(n)
    for k in active:
        # For position m in 0..n-1, a[k-m] is valid when 0 <= k-m <= n-1
        # i.e., max(0, k-n+1) <= m <= min(k, n-1)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            subgrad[lo:hi+1] += a[k - np.arange(lo, hi + 1)]
    
    return subgrad / len(active), active, c_max


def run_smoothed_gd(a_init, budget_s, seed, tau_start=3.0, tau_end=0.02, lr=0.008):
    """Smoothed gradient descent with Adam, adaptive tau, and stall recovery."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = a * (n / S)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    stall_count = 0
    stall_J = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 0.1))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr_t = lr / (1.0 + 0.05 * t / 3000.0)

        J, grad = smooth_J_grad(a, tau)

        if not np.isfinite(J) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m_buf[:] = 0
            v_buf[:] = 0
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * grad * grad
        m_hat = m_buf / (1 - beta1 ** t)
        v_hat = v_buf / (1 - beta2 ** t)

        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = np.sum(a)
        if S < 1e-10:
            a = best_a.copy()
            m_buf[:] = 0
            v_buf[:] = 0
            t = 0
            continue
        a = a * (n / S)

        if t % 3000 == 0:
            local_J = compute_J(a)
            if local_J < stall_J - 1e-8:
                stall_J = local_J
                stall_count = 0
                if local_J < best_J:
                    best_J = local_J
                    best_a = a.copy()
            else:
                stall_count += 1
                if stall_count > 4:
                    # Diversify: random multiplicative perturbation
                    w = max(3, n // 30)
                    s = int(rng.integers(0, n - w))
                    a[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                    a = np.maximum(a, 0.0)
                    S = np.sum(a)
                    if S > 1e-10:
                        a = a * (n / S)
                    m_buf[:] = 0
                    v_buf[:] = 0
                    t = 0
                    stall_count = 0
                    stall_J = compute_J(a)

        if t % 15000 == 0:
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"      gd t={t} verified_J={val:.6f} tau={tau:.4f}")

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def run_active_set_search(a_init, budget_s, seed):
    """Active-set subgradient method with projected steps on the simplex."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n)
        S = n
    a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)
    lr = 0.01
    momentum = np.zeros(n)
    mu = 0.9
    count = 0

    while time.monotonic() < deadline:
        subgrad, active, c_max = active_set_subgrad(a, top_k=30)

        # Project gradient onto tangent space of simplex: remove mean component
        # We want to minimize c_max, so we go against the subgradient
        # But we need to maintain sum(a) = n
        # Projected direction: d = -subgrad + mean(subgrad) (to keep sum constant)
        direction = -subgrad + np.mean(subgrad)

        # Apply momentum
        momentum = mu * momentum + (1 - mu) * direction
        a = a + lr * momentum
        a = np.maximum(a, 0.0)
        S = np.sum(a)
        if S < 1e-10:
            a = best_a.copy()
            momentum[:] = 0
            continue
        a = a * (n / S)

        count += 1

        if count % 1000 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                lr = min(lr * 1.05, 0.1)
            else:
                lr = max(lr * 0.9, 0.001)
                if local_J > best_J * 1.01:
                    a = best_a.copy()
                    momentum[:] = 0

        if count % 10000 == 0:
            val = float(evaluate_sequence(a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"      active t={count} verified_J={val:.6f}")

    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def run_coordinate_search(a_init, budget_s, seed):
    """Coordinate-wise multiplicative search for final refinement."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S < 1e-10:
        return a, 1e15
    a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)

    # Compute sensitivity: which coordinates matter most
    c = convolve_fft(a)
    c_max = np.max(c)
    active = np.where(c >= 0.9 * c_max)[0]

    # Build a priority list of coordinates
    # Coordinates that appear in active convolution positions are more sensitive
    sensitivity = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        sensitivity[lo:hi+1] += 1

    # Sort indices by sensitivity (descending)
    order = np.argsort(-sensitivity)

    step_size = 0.02
    iters_per_coord = 3

    while time.monotonic() < deadline:
        improved = False
        for it in range(iters_per_coord):
            for idx in order:
                if time.monotonic() > deadline - 1:
                    break
                # Try multiplicative perturbation
                factors = [1.0 + step_size, 1.0 - step_size * 0.5]
                best_factor = 1.0
                best_local_J = compute_J(a)

                for f in factors:
                    a[idx] *= f
                    if a[idx] < 0:
                        a[idx] = 0
                    S = np.sum(a)
                    if S < 1e-10:
                        a[idx] /= f
                        continue
                    a_norm = a * (n / S)
                    local_J = compute_J(a_norm)
                    if local_J < best_local_J:
                        best_local_J = local_J
                        best_factor = f
                    a[idx] /= f

                if best_factor != 1.0:
                    a[idx] *= best_factor
                    S = np.sum(a)
                    a = a * (n / S)
                    improved = True

            if not improved:
                break

        local_J = compute_J(a)
        if local_J < best_J:
            best_J = local_J
            best_a = a.copy()
        else:
            # Decay step size
            step_size *= 0.7
            if step_size < 0.001:
                break

        if int(count // 10000) != count // 10000 if 'count' in dir() else True:
            pass

    val = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(val):
        best_J = min(best_J, val)
    return best_a, best_J


def make_structured_init(n, rng, mode):
    """Create diverse structured initializations."""
    x = np.arange(n)
    if mode == 0:
        # Gaussian
        return np.exp(-0.5 * ((x - n/2) / (0.2 * n)) ** 2)
    elif mode == 1:
        # Double Gaussian
        return np.exp(-0.5 * ((x - 0.3*n) / (0.15*n)) ** 2) + np.exp(-0.5 * ((x - 0.7*n) / (0.15*n)) ** 2)
    elif mode == 2:
        # Cosine window
        return 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    elif mode == 3:
        # Random smooth
        freqs = rng.choice([1, 2, 3, 5, 8], size=3)
        phases = rng.uniform(0, 2*np.pi, 3)
        result = np.ones(n)
        for f, p in zip(freqs, phases):
            result += 0.3 * np.cos(2 * np.pi * f * x / n + p)
        return np.maximum(result, 0.01)
    elif mode == 4:
        # Block structure
        blocks = max(2, n // 500)
        result = np.ones(n)
        for b in range(blocks):
            start = int(b * n / blocks)
            end = int((b + 1) * n / blocks)
            result[start:end] = rng.uniform(0.3, 1.5)
        return result
    elif mode == 5:
        # Logistic transition
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.05, 0.2)
        return 1.0 / (1.0 + np.exp((x - center) / width)) + 0.1
    else:
        # Pure random smooth
        raw = rng.normal(0, 1, n)
        # Smooth with running average
        kernel_size = max(3, n // 200)
        kernel = np.ones(kernel_size) / kernel_size
        smooth = np.convolve(raw, kernel, mode='same')
        return np.exp(smooth * 0.5) + 0.01

    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]


def make_random_init(n, rng, seed_idx):
    """Create a random but smooth positive initialization."""
    sub_rng = np.random.default_rng(seed + seed_idx * 7)
    # Generate as sum of a few random basis functions
    n_components = sub_rng.integers(2, 6)
    result = np.zeros(n)
    for _ in range(n_components):
        center = sub_rng.uniform(0.1, 0.9) * n
        width = sub_rng.uniform(0.05, 0.3) * n
        amp = sub_rng.uniform(0.3, 1.0)
        x = np.arange(n)
        result += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    result += 0.01  # Ensure positivity
    return result
