        if not accepted:
            # Try random perturbations
            best_local = score
            best_local_z = z
            for _ in range(5):
                zt = z + rng.normal(0.0, noise, N-1)
                zt = np.clip(np.where(np.isfinite(zt), zt, 0.0), -40.0, 40.0)
                pt = probs_from_z(zt)
                st = score_p(pt)
                if st < best_local - 1e-12:
                    best_local = st
                    best_local_z = zt
            if best_local < score - 1e-12:
                z = best_local_z
                p = probs_from_z(z)
                score = best_local
                update_best(z, score)
                accepted = True
        
        if accepted:
            fails = 0
            noise = min(0.2, noise * 1.15)
        else:
            fails += 1
            noise = max(0.01, noise * 0.85)
            if fails > 150:
                break
