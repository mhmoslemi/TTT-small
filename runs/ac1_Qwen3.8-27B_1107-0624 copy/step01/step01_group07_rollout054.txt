def lp_minimize_peak(a):
    """Find g that minimizes max(conv(a,g)) subject to sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)  # shape (2n-1, n)
    # Variables: g[0], ..., g[n-1], M
    # Minimize M
    # Subject to: A @ g - M <= 0 for each row k
    #             sum(g) = 1
    #             g >= 0
    
    # In standard LP form: minimize c^T x
    # x = [g_0, ..., g_{n-1}, M]
    # c = [0, ..., 0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A @ g - M <= 0
    # In matrix form: [A, -I] @ x <= 0
    A_ub = np.zeros((2*n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2*n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

    """Find g minimizing max(conv(a,g)) s.t. sum(g)=1, g>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M
    # s.t. A@g - M <= 0 (for each of 2n-1 rows)
    #      sum(g) = 1
    #      g >= 0, M >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            j = k - i
            if 0 <= j < n:
                sg[i] += a[j]
    return sg

"""
Hybrid LP + multi-subgradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: (1) LP minimizes cross-convolution peak directly,
(2) multi-peak subgradient for smoother descent, (3) backtracking line search,
(4) wider range of sequence lengths, (5) structured restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g, mode='full')."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def lp_min_cross_peak(a):
    """
    Solve LP: minimize M s.t. conv(a,g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g, M_opt) or (None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)

    # Decision variables: x = [g_0, ..., g_{n-1}, M]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: A @ g - M <= 0  =>  [A | -1] @ x <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            g = np.clip(res.x[:n], 0.0, None)
            if g.sum() > 1e-15:
                g /= g.sum()
                return g, float(res.x[-1])
    except Exception:
        pass
    return None, None


def lp_max_sum(a, M):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g)[k] <= M for all k, g>=0.
    Returns g or None. (Original approach from previous SOTA.)
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.clip(res.x, 0.0, None)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def multi_subgradient(a, num_peaks=3):
    """
    Compute multi-peak subgradient direction.
    For each of the top peaks, accumulate the contribution vector.
    Returns a direction to subtract from a.
    """
    n = len(a)
    c = np.convolve(a, a)
    # Target the top peaks
    num_peaks = min(num_peaks, 2 * n - 1)
    top_lags = np.argsort(c)[-num_peaks:]

    # Weight peaks by their excess over the mean
    mean_c = c.mean()
    weights = np.maximum(c[top_lags] - mean_c, 0.0)
    if weights.sum() < 1e-15:
        weights = np.ones(num_peaks) / num_peaks
    else:
        weights = weights / weights.sum()

    sg = np.zeros(n)
    for w, k in zip(weights, top_lags):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += w * a[j]
    return sg

            else:

                    a = a + 1e-6 * np.abs(rng.normal(0, 1, n))
                a = np.clip(a, 1e-15, None)
                a /= a.sum()
                stall = 0
                step = 0.12

            if stall > 250:
                break

    return best_a, best_score


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian window."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_uniform(n):
    return np.ones(n)


def make_bimodal(n, rng):
    """Two Gaussian bumps at offset positions."""
    c1 = int(n * rng.uniform(0.15, 0.35))
    c2 = int(n * rng.uniform(0.65, 0.85))
    sigma = n * 0.12
    a = 0.55 * np.exp(-0.5 * ((np.arange(n) - c1) / sigma) ** 2)
    a += 0.45 * np.exp(-0.5 * ((np.arange(n) - c2) / sigma) ** 2)
    return a


def make_random(n, rng):
    """Random positive sequence."""
    return np.abs(rng.normal(0, 1, n)) + 0.05


def make_flat_center(n, rng):
    """Flat center with tapered edges."""
    a = np.ones(n)
    taper = int(n * 0.2)
    for i in range(taper):
        frac = i / taper
        a[i] = frac
        a[n - 1 - i] = frac
    return a
