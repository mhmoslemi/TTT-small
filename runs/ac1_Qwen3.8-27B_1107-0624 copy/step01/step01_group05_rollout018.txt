    S2 = C[:n]
    S1 = D[2*n-2 : n-1 : -1]  # D[2n-2], D[2n-3], ..., D[n-1]
    return S1 + S2

    gz = p[1:] * (gp[1:] - sdot)
    gz = np.nan_to_num(gz, 0.0)
    return np.clip(gz, -1e6, 1e6)

def _obj_grad(x, alpha):
    _check()
    p = _z_to_p(x)
    c = _self_conv(p)
    cmax = c.max()
    hard = 2.0 * N * cmax
    _update(p, hard)
    
    u = alpha * (c - cmax)
    u = np.clip(u, -500.0, 500.0)
    e = np.exp(u)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(2*N-1)
        w[c.argmax()] = 1.0
        J = cmax
    else:
        w = e / s
        J = cmax + np.log(s) / alpha
    
    gz = _grad_z(p, w)
    return J, gz

def _lbfgs(z0, alpha, max_fun):
    global _deadline
    try:
        minimize(_obj_grad, z0, args=(alpha,), method='L-BFGS-B',
                 jac=True, bounds=_BOUNDS,
                 options={'maxfun': max_fun, 'maxiter': max_fun//2,
                          'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 20})
    except T:
        pass
    except Exception:
        pass

def _lp_direction(p, M, top_k=200):
    """Solve relaxed LP to find direction g that reduces peaks."""
    from scipy.optimize import linprog
    c = _self_conv(p)
    # Get top_k constraint indices
    idx = np.argpartition(c, -top_k)[-top_k:]
    # LP: maximize sum(g) s.t. (p*g)[k] <= M for k in idx, g >= 0
    # (p*g)[k] = sum_i p[i]*g[k-i] for valid i,k-i
    # This is a matrix A @ g <= M where A[k,j] = p[j] if 0<=k-j<n else 0
    # A is (2n-1) x n but we only use top_k rows.
    # A[idx, j] = p[j] if 0<=idx[j]-j<n, also p[idx[j]-j]... 
    # Actually A[k,j] = p[k-j] if 0<=k-j<n, PLUS p[j-k] if 0<=j-k<n
    # Wait no. (p*g)[k] = sum_i p[i]*g[k-i] where g is defined on [0,n-1]
    # = sum_{j: 0<=j<n, 0<=k-j<n} p[k-j]*g[j]
    # So A[k,j] = p[k-j] if 0<=k-j<n else 0
    # This is a Toeplitz-like matrix (actually it's the convolution matrix)
    
    # For efficiency, I'll construct A using the structure
    # A[k,j] = p[k-j] if 0 <= k-j <= n-1
    # For k in [0, 2n-2] and j in [0, n-1]:
    # k-j in [0, n-1] means j in [k-n+1, k] ∩ [0, n-1]
    
    # This is a banded matrix. Let me just construct it directly for the selected rows.
    n = N
    A = np.zeros((len(idx), n), dtype=np.float64)
    for row_i, k in enumerate(idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[row_i, j] = p[k - j]
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(len(idx), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                  options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-8})
    
    if res.status == 0 and res.x is not None:
        g = np.clip(res.x, 0.0, None)
        if g.sum() > 0:
            return g
    return None

def _polish(z0, time_budget, rng):
    """Active-set subgradient polishing."""
    _check()
    z = np.clip(np.nan_to_num(z0, 0.0), -50.0, 50.0)
    p = _z_to_p(z)
    score = _score(p)
    _update(p, score)
    
    end = time.monotonic() + time_budget
    noise_scale = 0.1
    fails = 0
    
    while time.monotonic() < end:
        _check()
        c = _self_conv(p)
        cmax = c.max()
        n_active = min(16, 2*N-1)
        idx = np.argpartition(c, -n_active)[-n_active:]
        w = np.zeros(2*N-1)
        vals = c[idx] + 1e-15
        w[idx] = vals / vals.sum()
        
        gz = _grad_z(p, w)
        gn = np.linalg.norm(gz)
        
        accepted = False
        if gn > 1e-15:
            d = -gz / gn
            for step in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015):
                zt = z + step * d
                zt = np.clip(zt, -50, 50)
                pt = _z_to_p(zt)
                st = _score(pt)
                if st < score - 1e-13:
                    z, p, score = zt, pt, st
                    _update(p, score)
                    accepted = True
                    break
        
        if not accepted:
            # Random perturbation
            best_s, best_z = score, z
            for _ in range(5):
                zt = z + rng.normal(0, noise_scale, N-1)
                zt = np.clip(zt, -50, 50)
                pt = _z_to_p(zt)
                st = _score(pt)
                if st < best_s:
                    best_s, best_z = st, zt
            if best_s < score - 1e-13:
                z, score = best_z, best_s
                p = _z_to_p(z)
                _update(p, score)
                accepted = True
        
        if accepted:
            fails = 0
            noise_scale = min(0.3, noise_scale * 1.3)
        else:
            fails += 1
            noise_scale = max(0.01, noise_scale * 0.8)
            if fails > 200:
                break

def _taylor_window(n_samples, order, sll_db):
    """Generate a Taylor-like window (low sidelobe)."""
    x = np.linspace(-0.5, 0.5, n_samples)
    # Simplified Taylor window approximation
    # The Taylor window has controlled sidelobes
    # Using the quadratic approximation for the amplitude
    a = 10 ** (-sll_db / 20.0)
    # Taylor coefficient
    A = np.arccosh(1.0 / a) / np.pi
    # Simplified: use a raised-cosine-like shape
    w = np.ones(n_samples)
    for m in range(1, order + 1):
        w *= (1.0 + np.cos(m * np.pi * 2.0 * x)) ** 0.5
    # Normalize
    w = np.maximum(w, 0.01)
    return w

Actually, the constraint (p*g)[k] <= M is NOT the same as saying (p+tg)*p[k] <= M. Let me reconsider.

The state-of-the-art says: "maximizing sum_j b_j subject to b_j>=0 and ||f*g_0||_∞ <= M". So the constraint is on the cross-convolution f*g_0, not on the self-convolution of the new point.

The idea is: if we're at point f with ||f*f||_∞ = M, and we move to (1-t)f + t*g, then:
((1-t)f + tg) * ((1-t)f + tg) ≈ (1-t)^2 f*f + 2t(1-t) f*g + t^2 g*g
