```python
    n = len(a)
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1
    # This is a (2n-1) x n matrix
    # Column j has a[0] at row j, a[1] at row j+1, ..., a[n-1] at row j+n-1
    # So column j is a shifted version of a, starting at row j
    
    rows = []
    cols = []
    vals = []
    
    for j in range(n):
        # Column j: rows j to j+n-1 have values a[0] to a[n-1]
        r = np.arange(j, j+n)
        rows.extend(r)
        cols.extend([j]*n)
        vals.extend(a)
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
    a = a.copy()

        M = b.max()
        # Find peaks
        peaks = np.where(b >= M * 0.99)[0]
        # For each peak, reduce the contributing elements
        for k in peaks:

                reduction = 0.01 * a[j] * a[k-j] / M

                a[k-j] -= reduction

        # Renormalize

        if S > 0:
```

Actually, for a single peak k, the reduction to a[j] is proportional to a[j]*a[k-j]. So the total reduction to a[j] from peak k is proportional to a[j]*a[k-j]. If we sum over all peaks, the total reduction to a[j] is proportional to a[j] * (sum of a[k-j] over peaks k that include j).

The term sum of a[k-j] over peaks k that include j is like a "cross-correlation" of a with the peak indicator.

Let me define: p[k] = 1 if b[k] >= threshold, else 0. Then the reduction to a[j] is proportional to a[j] * sum_k p[k] * a[k-j] (with appropriate bounds).

This is: a[j] * (p * a)[j] where * is cross-correlation.

So the update is: a[j] *= (1 - lr * (p * a)[j] / M)

This can be computed with FFT!

```python
def peak_shave_fft(a, n_iters=1000, lr=0.01, thresh=0.99):
    n = len(a)
    a = a.copy().astype(float)
    for it in range(n_iters):
        b = convolve_fft(a)
        M = b.max()
        if M < 1e-10:
            break
        # Peak indicator
        p = (b >= M * thresh).astype(float)
        # Cross-correlation: (p * a)[j] = sum_k p[k] * a[k-j]
        # This is the same as the "cross" computation in the gradient code
        size = 1
        while size < 2 * max(n, len(p)) - 1:
            size <<= 1
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        cross = np.fft.irfft(np.conj(fp) * fa, size)[:n]
        # Update
        a = a * (1.0 - lr * cross / M)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S > 0:
            a = a * n / S
    return a
```

Let me also think about what makes the LP approach powerful: it's a "best response" in a game-theoretic sense. Given a, the LP finds the best b (in terms of maximizing sum) that doesn't violate the current peak constraint. This is like a "best response" update in a zero-sum game where one player chooses a and the other chooses b.

Actually, this is exactly the framework of the "Charnes-Cooper" transformation or the "Dinkelbach" method for fractional programming! The problem min max(a*a)/S^2 can be seen as a fractional program, and the LP loop is essentially the Dinkelbach iteration.

Let me formalize: We want min_x f(x)/g(x) where f(x) = max(a*a) and g(x) = S^2. The optimal value is λ*. At optimality, max(a*a) = λ* * S^2.

The LP iteration: given current a with λ = max(a*a)/S^2, find b that maximizes S_b subject to (a*b)[k] <= λ * S^2 for all k. If the optimal b has S_b > 0 and max(b*b) < λ * S_b^2, then we can improve.

Actually, I think the correct iteration is:
1. Given a, compute M = max(a*a), S = sum(a), λ = M/S^2
2. Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
3. Let b* be the solution. Compute λ_new = max(b**)/ (sum(b*))^2
4. If λ_new < λ, update a = b* (or blend)
5. Repeat

The key question is: does this monotonically decrease λ? Not necessarily in one step, but the LP gives the "best" direction.

**One more thought**: The constraint (a*b)[k] <= M defines a polytope in b-space. The LP finds the vertex of this polytope that maximizes sum(b). This vertex is determined by n active constraints (since we have n variables). So the active set LP is solving a much smaller problem.

For the full LP (all 2n-1 constraints), the active set at the optimum will be at most n constraints. So using a subset of top-m constraints (m >= n) should capture the optimum if m is large enough. In practice, m = 2*n or so should be safe, but that might be too many for speed.

```python
"""
Multi-phase optimization combining FFT-based peak shaving, LP active-set 
method (Dinkelbach-style iteration), and gradient refinement to minimize
J(a) = 2n * max(a*a) / (sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]

def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)

def peak_shave_fft(a, n_iters=5000, lr=0.05, thresh=0.95, verbose=False):
    """FFT-based peak shaving: iteratively reduce convolution peaks."""
    n = len(a)
    a = a.astype(float).copy()
    size = 1
    while size < 2 * (2 * n - 1) - 1:
        size <<= 1
    # Actually, for cross-correlation of p (length 2n-1) and a (length n),
    # we need FFT size >= (2n-1) + n - 1 = 3n - 2
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    best_a = a.copy()
    best_J = compute_J(a)
    stall = 0
    
    for it in range(n_iters):
        b = convolve_fft(a)
        M = b.max()
        if M < 1e-10:
            break
        
        p = (b >= M * thresh).astype(float)
        
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        cross = np.fft.irfft(np.conj(fp) * fa, size)[:n]
        
        # Multiplicative update
        a = a * np.exp(-lr * cross / M)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            continue
        a = a * n / S
        
        # Check improvement every 100 iterations
        if (it + 1) % 100 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if verbose and (it + 1) % 1000 == 0:
                print(f"    peak_shave it={it+1} J={J:.6f} best={best_J:.6f}")
    
    return best_a, best_J

def lp_active_set_step(a, n_top=200, method='highs'):
    """One LP step: solve max sum(b) s.t. (a*b)[k] <= M for top-k constraints, b >= 0.
    Returns the LP solution b or None if failed."""
    n = len(a)
    b_conv = convolve_fft(a)
    M = np.max(b_conv)
    
    # Get top n_top constraints
    if n_top >= 2 * n - 1:
        k_list = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(b_conv)[::-1][:n_top]
        k_list = np.sort(top_idx)
    
    n_active = len(k_list)
    
    # Build sparse constraint matrix
    # A[k_list[i], j] = a[k_list[i] - j] if 0 <= k_list[i] - j <= n-1
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(k_list):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        c = np.arange(j_min, j_max + 1)
        v = a[k - c]  # a[k-j_min], a[k-j_min-1], ..., a[k-j_max]
        # Wait: k - c = k - j_min, k - j_min - 1, ..., k - j_max
        # We want a[k-j] for j from j_min to j_max
        # k - j ranges from k-j_min down to k-j_max
        # So v = a[k-j_min : k-j_max-1 : -1] if k-j_min > k-j_max
        # Or: v = a[k - np.arange(j_min, j_max+1)]
        v = a[k - c]
        rows.append(np.full(len(c), i))
        cols.append(c)
        vals.append(v)
    
    if not rows:
        return None
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    
    A = csr_matrix((vals, (rows, cols)), shape=(n_active, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_active, M)
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
    
    if result.success and np.isfinite(result.fun):
        return result.x
    return None

def lp_iteration(a_init, n_top=200, n_iters=100, t_start=0.5, verbose=False):
    """Run LP-based active set iterations. Returns (best_a, best_J)."""
    n = len(a_init)
    a = a_init.astype(float).copy()
    S = a.sum()
    if S > 0:
        a = a * n / S
    
    best_a = a.copy()
    best_J = compute_J(a)
    t = t_start
    stall = 0
    
    for it in range(n_iters):
        b_new = lp_active_set_step(a, n_top=n_top)
        if b_new is None:
            break
        
        b_new = np.maximum(b_new, 0)
        S_new = b_new.sum()
        if S_new < 0.01:
            break
        
        # Rescale to same normalization
        b_new = b_new * n / S_new
        
        # Blend
        a_new = (1 - t) * a + t * b_new
        a_new = np.maximum(a_new, 0)
        S = a_new.sum()
        if S > 0:
            a_new = a_new * n / S
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new.copy()
            stall = 0
            t = min(0.9, t * 1.1)  # increase step
        else:
            a = a_new.copy()  # still move, but track best
            stall += 1
            t = max(0.05, t * 0.8)  # decrease step
        
        if verbose and (it + 1) % 10 == 0:
            print(f"    lp it={it+1} J={J_new:.6f} best={best_J:.6f} t={t:.3f}")
        
        if stall > 20:
            break
    
    return best_a, best_J

def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad

def gradient_refine(a_init, budget_s, seed=42, tau_start=5.0, tau_end=0.05, lr=0.01):
    """Adam gradient descent with adaptive tau."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.astype(float).copy()
    S = a.sum()
    if S > 0:
        a = a * n / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    stall = 0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - (deadline - budget_s)
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr_eff = lr / (1.0 + 0.1 * t / 1000.0)
        
        J, grad = smooth_J_and_grad(a, tau)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        if t % 500 == 0:
            J_check = compute_J(a)
            if J_check < best_J:
                best_J = J_check
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if stall > 10:
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
    
    return best_a, best_J

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))

def make_bimodal(n, sep=0.3):
    """Two Gaussian bumps."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = 0.1 * n
    center = sep * n
    return np.exp(-((x - center) ** 2) / (2 * sigma * sigma)) + \
           np.exp(-((x + center) ** 2) / (2 * sigma * sigma))

def make_cosine_bumps(n, n_bumps=3):
    """Multiple cosine bumps."""
    x = np.arange(n)
    result = np.zeros(n)
    for i in range(n_bumps):
        center = i * n // (n_bumps - 1) if n_bumps > 1 else n // 2
        sigma = n / (n_bumps * 4)
        result += np.exp(-((x - center) ** 2) / (2 * sigma * sigma))
    return result

def make_periodic(n, freq=7):
    """Periodic pattern with some randomness."""
    x = np.arange(n)
    return 1.0 + 0.7 * np.abs(np.sin(2 * np.pi * freq * x / n))
```

```python
"""
Multi-phase optimizer combining FFT-based multiplicative peak shaving,
LP active-set (Dinkelbach-style) iterations, and log-sum-exp gradient
refinement to minimize J(a) = 2n * max(a*a) / (sum(a))^2.

Key innovations over prior approach:
- FFT-based multiplicative peak shaving (O(n log n) per iter, thousands of iters)
- LP active-set method exploiting the convex structure of the sub-problem
- Adaptive thresholding for peak identification
- Phased time allocation with verification at each phase boundary
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def peak_shave_fft(a, n_iters=5000, lr=0.05, thresh=0.95, thresh_end=0.99, verbose=False):
    """FFT-based multiplicative peak shaving with adaptive threshold.
    
    At each step, identifies peaks in a*a above threshold*max, then applies
    a multiplicative reduction proportional to the peak cross-correlation.
    """
    n = len(a)
    a = a.astype(float).copy()
    
    # FFT size for cross-correlation of p (len 2n-1) with a (len n)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    best_a = a.copy()
    best_J = compute_J(a)
    stall = 0
    
    for it in range(n_iters):
        # Adaptive threshold: start wide, narrow down
        frac = min(1.0, it / max(1, n_iters))
        t_cur = thresh + (thresh_end - thresh) * frac
        
        b = convolve_fft(a)
        M = b.max()
        if M < 1e-10:
            break
        
        p = (b >= M * t_cur).astype(float)
        
        fp = np.fft.rfft(p, size)
        fa = np.fft.rfft(a, size)
        cross = np.fft.irfft(np.conj(fp) * fa, size)[:n]
        
        # Multiplicative update (exponential for stability)
        a = a * np.exp(-lr * cross / M)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 0.01:
            a = best_a.copy()
            continue
        a = a * n / S
        
        if (it + 1) % 200 == 0:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if verbose and (it + 1) % 2000 == 0:
                print(f"      peak_shave it={it+1} J={J:.6f} best={best_J:.6f} thresh={t_cur:.4f}")
            if stall > 15:
                break
    
    # Final check
    J_final = compute_J(a)
    if J_final < best_J:
        best_J = J_final
        best_a = a.copy()
    
    return best_a, best_J


def build_A_sparse(a, k_list):
    """Build sparse constraint matrix A where A[k_list[i], j] = a[k_list[i]-j] if valid."""
    n = len(a)
    n_active = len(k_list)
    rows_parts = []
    cols_parts = []
    vals_parts = []
    
    for idx, k in enumerate(k_list):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        c = np.arange(j_min, j_max + 1)
        v = a[k - c]
        rows_parts.append(np.full(len(c), idx))
        cols_parts.append(c)
        vals_parts.append(v)
    
    if not rows_parts:
        return csr_matrix((n_active, n))
    
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = np.concatenate(vals_parts)
    
    return csr_matrix((vals, (rows, cols)), shape=(n_active, n))


def lp_active_set_step(a, n_top=300):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for top-n_top constraints, b >= 0.
    Returns b or None."""
    n = len(a)
    b_conv = convolve_fft(a)
    M = np.max(b_conv)
    
    if M < 1e-15:
        return a.copy()
    
    # Get top constraints
    if n_top >= 2 * n - 1:
        k_list = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(b_conv)[::-1][:n_top]
        k_list = np.sort(top_idx)
    
    n_active = len(k_list)
    A = build_A_sparse(a, k_list)
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_active, M)
    bounds = [(0.0, 4.0 * n)] * n  # Upper bound to prevent unboundedness
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
    except Exception as e:
        return None
    
    if result.success and result.x is not None and np.isfinite(result.x).all():
        return np.maximum(result.x, 0.0)
    return None


def lp_iteration(a_init, n_top=300, n_iters=500, t_start=0.3, verbose=False):
    """Run LP-based active set iterations with adaptive step size."""
    n = len(a_init)
    a = a_init.astype(float).copy()
    S = a.sum()
    if S > 0:
        a = a * n / S
    
    best_a = a.copy()
    best_J = compute_J(a)
    t = t_start
    stall = 0
    consecutive_fail = 0
    
    for it in range(n_iters):
        b_new = lp_active_set_step(a, n_top=n_top)
        if b_new is None:
            consecutive_fail += 1
            if consecutive_fail > 5:
                break
            continue
        consecutive_fail = 0
        
        S_new = b_new.sum()
        if S_new < 0.01:
            break
        
        # Rescale to normalization
        b_new = b_new * n / S_new
        
        # Blend
        a_new = (1.0 - t) * a + t * b_new
        a_new = np.maximum(a_new, 0.0)
        S = a_new.sum()
        if S > 0:
            a_new = a_new * n / S
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            stall = 0
            t = min(0.8, t * 1.05)
        else:
            # Still accept the move if it's not too bad (exploration)
            if J_new < best_J * 1.01:
                a = a_new.copy()
            else:
                a = a_new.copy()
                t = max(0.05, t * 0.85)
            stall += 1
        
        if verbose and (it + 1) % 20 == 0:
            print(f"      lp it={it+1} J_new={J_new:.6f} best={best_J:.6f} t={t:.3f}")
        
        if stall > 30:
            # Try to escape: revert to best and reduce step
            a = best_a.copy()
            t = max(0.05, t * 0.5)
            stall = 0
    
    return best_a, best_J


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    
    J = 2.0 * n * Tmax / (S * S)
    
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad


def gradient_refine(a_init, budget_s, seed=42, tau_start=5.0, tau_end=0.05, lr=0.01):
    """Adam gradient descent with log-sum-exp smoothing and adaptive tau."""
    n = len(a_init)
    start_time = time.monotonic()
    deadline = start_time + budget_s
    
    a = a_init.astype(float).copy()
    S = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    stall = 0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** frac
        lr_eff = lr / (1.0 + 0.1 * t / 1000.0)
        
        J, grad = smooth_J_and_grad(a, tau)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        if t % 300 == 0:

                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if stall > 8:
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                stall = 0
    
    return best_a, best_J


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-x * x / (2.0 * sigma * sigma))


def make_bimodal(n, sep=0.25, width=0.08):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = width * n
    c = sep * n
    return np.exp(-((x - c) ** 2) / (2 * sigma * sigma)) + \
           np.exp(-((x + c) ** 2) / (2 * sigma * sigma))


def make_multimodal(n, n_mods=5, width=0.06):
    x = np.arange(n) - (n - 1) / 2.0
    sigma = width * n
    result = np.zeros(n)
    for i in range(n_mods):
        c = (i / (n_mods - 1) - 0.5) * n * 0.8
        result += np.exp(-((x - c) ** 2) / (2 * sigma * sigma))
    return result


def make_periodic(n, freq=7, amp=0.7):
    x = np.arange(n)
    return 1.0 + amp * np.abs(np.sin(2.0 * np.pi * freq * x / n))


def make_cosine_weight(n, freq=1.0):
    x = np.arange(n)
    return 1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)


def normalize_a(a, n):
    """Normalize a to sum = n, non-negative."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S > 0:
        a = a * n / S
    return a

    allocated = (phase3_end - time.monotonic()) / n_configs
    
    for cfg_idx, (n_top, t_start) in enumerate(lp_configs):
        if time.monotonic() > phase3_end:
            break
        remaining = min(allocated, phase3_end - time.monotonic())
        if remaining < 10.0:
            break
        
        print(f"  LP config {cfg_idx+1}: n_top={n_top}, t_start={t_start}, time={remaining:.0f}s")
        
        a3, J3 = lp_iteration(best_global, n_top=n_top, n_iters=500, t_start=t_start, verbose=False)
        
        if J3 < best_global_J:
            best_global_J = J3
            best_global = a3.copy()
            print(f"  *** LP improved: J = {J3:.6f} ***")
        
        # Also try peak shaving after LP
        a3b, J3b = peak_shave_fft(a3, n_iters=3000, lr=0.02, thresh=0.97, thresh_end=0.999)
        if J3b < best_global_J:
            best_global_J = J3b
            best_global = a3b.copy()
            print(f"  *** LP+PS improved: J = {J3b:.6f} ***")
        
        val = float(evaluate_sequence(best_global.tolist()))
        if np.isfinite(val):
            best_global_J = min(best_global_J, val)
            print(f"  Verified: {val:.6f}")
    
    # ==========================================
    # Phase 4: Gradient refinement
    # ==========================================
    print("\n=== Phase 4: Gradient refinement (450s) ===")
    phase4_end = min(time.monotonic() + 450.0, total_deadline - 60.0)
    
    gd_configs = [
        {"tau_start": 10.0, "tau_end": 0.5, "lr": 0.02, "seed": 100},
        {"tau_start": 3.0, "tau_end": 0.1, "lr": 0.01, "seed": 200},
        {"tau_start": 1.0, "tau_end": 0.02, "lr": 0.005, "seed": 300},
        {"tau_start": 0.5, "tau_end": 0.01, "lr": 0.003, "seed": 400},
    ]
    
    n_gd = len(gd_configs)
    gd_alloc = (phase4_end - time.monotonic()) / n_gd
    
    for cfg_idx, cfg in enumerate(gd_configs):
        if time.monotonic() > phase4_end:
            break
        remaining = min(gd_alloc, phase4_end - time.monotonic())
        if remaining < 20.0:
            break
        
        print(f"  GD config {cfg_idx+1}: tau={cfg['tau_start']}->{cfg['tau_end']}, lr={cfg['lr']}, time={remaining:.0f}s")
        
        a4, J4 = gradient_refine(best_global, remaining, **cfg)
        
        if J4 < best_global_J:
            best_global_J = J4
            best_global = a4.copy()
            print(f"  *** GD improved: J = {J4:.6f} ***")
        
        # Quick peak shave after GD
        a4b, J4b = peak_shave_fft(a4, n_iters=5000, lr=0.02, thresh=0.98, thresh_end=0.999)
        if J4b < best_global_J:
            best_global_J = J4b
            best_global = a4b.copy()
            print(f"  *** GD+PS improved: J = {J4b:.6f} ***")
        
        val = float(evaluate_sequence(best_global.tolist()))
        if np.isfinite(val):
            best_global_J = min(best_global_J, val)
            print(f"  Verified: {val:.6f}")
    
    # ==========================================
    # Phase 5: Final alternating rounds
    # ==========================================
    print("\n=== Phase 5: Final alternating optimization (60s) ===")
    phase5_end = min(time.monotonic() + 60.0, total_deadline - 15.0)
    
    while time.monotonic() < phase5_end:
        # LP round
        if time.monotonic() < phase5_end - 10:
            a5, J5 = lp_iteration(best_global, n_top=200, n_iters=30, t_start=0.15, verbose=False)
            if J5 < best_global_J:
                best_global_J = J5
                best_global = a5.copy()
        
        # Peak shave round
        if time.monotonic() < phase5_end - 5:
            n_it = max(500, int((phase5_end - time.monotonic() - 5) * 300))
            a5, J5 = peak_shave_fft(best_global, n_iters=n_it, lr=0.01, thresh=0.99, thresh_end=0.999)
            if J5 < best_global_J:
                best_global_J = J5
                best_global = a5.copy()
        
        # Quick gradient
        if time.monotonic() < phase5_end - 3:
            rem = min(10.0, phase5_end - time.monotonic() - 3)
            a5, J5 = gradient_refine(best_global, rem, tau_start=0.5, tau_end=0.01, lr=0.003, seed=999)
            if J5 < best_global_J:
                best_global_J = J5
                best_global = a5.copy()
    
    # ==========================================
    # Final verification
    # ==========================================
    best_global = normalize_a(best_global, n)
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_J = min(best_global_J, final_val)
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_global_J:.6f}")
    print(f"{'='*50}")
    
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
```
