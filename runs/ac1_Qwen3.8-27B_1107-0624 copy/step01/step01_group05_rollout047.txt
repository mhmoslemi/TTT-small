    fp = np.fft.rfft(p, fft_size)
    c = np.fft.irfft(fp * fp, fft_size).real[:size]
    np.maximum(c, 0, out=c)
    return c


def _objective(f):
    s = float(np.sum(f))
    if s < 0.01:
        return float('inf')
    c = _self_conv(f)
    return 2.0 * len(f) * float(np.max(c)) / (s * s)


def _update_best(f):
    global _best_score, _best_seq
    score = _objective(f)
    if score < _best_score:
        _best_score = score
        _best_seq = f.copy()


def _solve_lp(f, M, time_limit=10):
    n = len(f)
    F = _make_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=F, b_ub=b_ub, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if np.sum(g) > 1e-12:
                return g
    except Exception:
        pass
    return None


def _lp_step(f, t, time_limit=10):
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-10:
        return f, False
    f = f / s * S
    c = _self_conv(f)
    M = float(np.max(c))
    g = _solve_lp(f, M, time_limit)
    if g is None:
        return f, False
    gs = float(np.sum(g))
    if gs < 1e-10:
        return f, False
    g = g / gs * S
    return (1 - t) * f + t * g, True


def _run_lp(f0, deadline, max_iters=500):
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if s < 1e-10:
        f = np.full(N, S / N)
    else:
        f = f / s * S
    _update_best(f)

    t = 0.5
    no_improve = 0
    iter_count = 0

    for i in range(max_iters):
        if time.monotonic() >= deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 5.0:
            break

        # Adaptive time limit per LP based on remaining time
        tl = min(15.0, max(2.0, rem * 0.02))
        # In early phase allow longer solves
        if i < 30:
            tl = min(20.0, tl * 1.5)

        prev = _best_score
        f, ok = _lp_step(f, t, tl)
        if not ok:
            break
        iter_count += 1
        _update_best(f)

        if _best_score < prev - 1e-9:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                t = max(0.15, t * 0.8)
            if no_improve > 80:
                break

        if i % 50 == 0 or i < 5:
            print(f"  LP iter {i}: obj={_best_score:.6f}, t={t:.3f}, rem={rem:.0f}s")

    return f


def _subgradient_polish(f0, deadline):
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    f = f / s * S
    _update_best(f)

    step = 0.02
    best_local = _best_score
    fails = 0

    while time.monotonic() < deadline:
        c = _self_conv(f)
        cmax = float(np.max(c))
        # Top-k weights
        topk = min(16, len(c))
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * N - 1)
        wv = c[idx] + 1e-15
        w[idx] = wv / float(np.sum(wv))

        # Compute subgradient: grad_f[j] = 2 * sum_k w[k] * f[k-j]
        # Using: grad_f = 2 * (conv(w_rev, f))[N-1:2N-1][::-1]
        w_rev = w[::-1]
        fft_sz = 1 << (len(w_rev) + N - 1 - 1).bit_length()
        fw = np.fft.rfft(w_rev, fft_sz)
        ff = np.fft.rfft(f, fft_sz)
        conv = np.fft.irfft(fw * ff, fft_sz).real[:len(w_rev) + N - 1]
        grad = 2.0 * conv[N - 1:2 * N - 1][::-1]

        # Project gradient onto the constraint sum(f) = S
        gmean = np.mean(grad)
        grad_proj = grad - gmean
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-14:
            break

        # Line search
        grad_dir = -grad_proj / gnorm
        accepted = False
        for lr in (step, step * 0.5, step * 0.25, step * 0.1):
            f_try = f + lr * grad_dir
            f_try = np.maximum(f_try, 0.0)
            st = float(np.sum(f_try))
            if st < 1e-10:
                continue
            f_try = f_try / st * S
            sc = _objective(f_try)
            if sc < best_local - 1e-10:
                f = f_try
                best_local = sc
                _update_best(f)
                accepted = True
                break

        if accepted:
            fails = 0
            step = min(0.05, step * 1.1)
        else:
            fails += 1
            step = max(0.003, step * 0.7)
            if fails > 200:
                break


def _logit_objective(z, alpha):
    """Objective in logit space: soft approximation of 2n*max(p*p)."""
    z = np.clip(np.where(np.isfinite(z), z, 0.0), -50, 50)
    m = float(np.max(z))
    expz = np.exp(z - m)
    Z = 1.0 + float(np.sum(expz))
    p = np.empty(N)
    p[0] = 1.0 / Z
    p[1:] = expz / Z

    c = _self_conv(p)
    cmax = float(np.max(c))
    u = alpha * (c - cmax)
    e = np.exp(u)
    s_e = float(np.sum(e))
    if s_e < 1e-300 or not np.isfinite(s_e):
        J = cmax
        w = np.zeros(2 * N - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        J = cmax + np.log(s_e) / alpha
        w = e / s_e

    # Gradient: dJ/dz_j = dJ/dp_j * dp_j/dz_j
    # dJ/dp_j = 2 * sum_k w[k] * p[k-j] (with bounds)
    w_rev = w[::-1]
    fft_sz = 1 << (len(w_rev) + N - 1 - 1).bit_length()
    fw = np.fft.rfft(w_rev, fft_sz)
    ff = np.fft.rfft(p, fft_sz)
    conv = np.fft.irfft(fw * ff, fft_sz).real[:len(w_rev) + N - 1]
    gp = 2.0 * conv[N - 1:2 * N - 1][::-1]
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    return 2.0 * N * J, gz


def _lbfgs_polish(f0, deadline):
    """Polish using L-BFGS in logit space."""
    f = np.maximum(f0, 1e-15)
    f = f / float(np.sum(f))
    z0 = np.log(f[1:] / f[0])
    z0 = np.clip(z0, -40, 40)
    _search_dl = deadline

    def obj(z):
        if time.monotonic() >= deadline:
            raise RuntimeError("timeout")
        return _logit_objective(z, 2e6)

    try:
        res = minimize(obj, z0, jac=True, method='L-BFGS-B',
                       bounds=[(-40, 40)] * (N - 1),
                       options={'maxiter': 200, 'maxfun': 400,
                                'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            z = np.clip(res.x, -40, 40)
            m = float(np.max(z))
            expz = np.exp(z - m)
            Z = 1.0 + float(np.sum(expz))
            p = np.empty(N)
            p[0] = 1.0 / Z
            p[1:] = expz / Z
            p_scaled = p * S
            _update_best(p_scaled)
            return p_scaled
    except Exception:
        pass
    return f0 * S


def _make_initializers():
    """Generate diverse starting points."""
    inits = []
    x = np.linspace(0, 1, N, endpoint=False)
    x += 0.5 / N

    # Uniform
    inits.append(np.ones(N) * S / N)

    # Linear ramps
    for rho in (0.5, 1.0, 1.5, 2.0, 3.0):
        f = 1.0 + rho * x
        f = np.maximum(f, 0.01)
        inits.append(f / float(np.sum(f)) * S)
        f = 1.0 + rho * (1 - x)
        f = np.maximum(f, 0.01)
        inits.append(f / float(np.sum(f)) * S)

    # Power laws
    for a in (1.0, 2.0, 3.0, 5.0):
        f = np.power(x + 1e-6, a)
        inits.append(f / float(np.sum(f)) * S)
        f = np.power(1 - x + 1e-6, a)
        inits.append(f / float(np.sum(f)) * S)

    # U-shaped (high at edges)
    for a in (0.5, 1.0, 2.0):
        f = np.power(np.abs(x - 0.5) * 2, a) + 0.01
        inits.append(f / float(np.sum(f)) * S)

    # Bell-like (high in middle)
    for w in (0.1, 0.2, 0.3, 0.5):
        f = np.exp(-((x - 0.5) ** 2) / (2 * w * w))
        inits.append(f / float(np.sum(f)) * S)

    # Sine-based
    for fq in (1, 2, 3, 4):
        f = 1.0 + 0.5 * np.sin(2 * np.pi * fq * x)
        f = np.maximum(f, 0.01)
        inits.append(f / float(np.sum(f)) * S)

    return inits


def propose_candidate(seed=42, budget_s=1800.0, **kwargs):
    global _best_score, _best_seq

    start = time.monotonic()
    budget_s = float(budget_s)
    end_time = start + max(10.0, budget_s) - 3.0
    _best_score = float('inf')
    _best_seq = None

    print(f"Starting search with budget {budget_s}s, N={N}")

    # Try to use previous best as a starting point
    inits = _make_initializers()

    try:
        base = height_sequence_1
        base = np.asarray(base, dtype=np.float64)
        if base.size > 0:
            base = np.where(np.isfinite(base), base, 0.0)
            base = np.clip(base, 0, None)
            if base.size != N:
                old_x = np.linspace(0, 1, base.size)
                new_x = np.linspace(0, 1, N)
                base = np.interp(new_x, old_x, base)
            if float(np.sum(base)) > 0:
                base = base / float(np.sum(base)) * S
                inits.insert(0, base)
                print(f"Using previous best as initial, obj={_objective(base):.6f}")
    except Exception:
        pass

    # Score all initializers
    scored = []
    for f in inits:
        f = np.maximum(f, 0.0)
        s = float(np.sum(f))
        if s > 0:
            f = f / s * S
        sc = _objective(f)
        scored.append((sc, f))
    scored.sort(key=lambda x: x[0])

    # Budget allocation
    # Phase 1: LP optimization from best starts (70% of time)
    # Phase 2: LP from more starts (15% of time)
    # Phase 3: Subgradient + L-BFGS polish (15% of time)

    t1 = start + 0.65 * budget_s
    t2 = start + 0.85 * budget_s

    print(f"Phase 1 deadline: {t1 - start:.0f}s, Phase 2: {t2 - start:.0f}s")

    # Phase 1: LP from top initializers
    top_k = min(8, len(scored))
    for i in range(top_k):
        if time.monotonic() >= t1:
            break
        sc, f = scored[i]
        print(f"Phase 1, start {i}: initial obj={sc:.6f}")
        f = _run_lp(f, t1, max_iters=300)
        _update_best(f)
        print(f"  -> best so far: {_best_score:.6f}")

    # Phase 2: LP from remaining initializers (shorter time each)
    if time.monotonic() < t2:
        for i in range(top_k, min(top_k + 12, len(scored))):
            if time.monotonic() >= t2:
                break
            sc, f = scored[i]
            f = _run_lp(f, t2, max_iters=80)
            _update_best(f)
            print(f"  Phase 2 start {i}: best={_best_score:.6f}")

    # Phase 3: Polish
    if _best_seq is not None and time.monotonic() < end_time:
        print(f"Phase 3: Polishing from obj={_best_score:.6f}")
        _subgradient_polish(_best_seq, min(end_time - 5, time.monotonic() + 0.2 * budget_s))
        print(f"  After subgradient: {_best_score:.6f}")

        if time.monotonic() < end_time - 5:
            _lbfgs_polish(_best_seq, end_time)
            print(f"  After L-BFGS: {_best_score:.6f}")

    # Final verification
    if _best_seq is not None:
        seq = np.clip(_best_seq, 0, 1000.0)
        seq_list = [float(v) for v in seq]
        try:
            final_score = float(evaluate_sequence(seq_list))
        except Exception:
            final_score = float('inf')
        print(f"Final evaluated score: {final_score:.6f}")
    else:
        seq_list = [1.0 / N] * N
        final_score = 2.0
        print("Fallback to uniform")

    elapsed = time.monotonic() - start
    print(f"Total time: {elapsed:.1f}s, best score: {_best_score:.6f}")
    return seq_list
