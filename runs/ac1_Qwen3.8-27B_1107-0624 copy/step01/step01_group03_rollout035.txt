def coordinate_descent(s, max_rounds=100):
    n = len(s)
    s = s.copy()
    for round in range(max_rounds):
        improved = False
        for i in range(n):
            old_val = s[i]
            s[i] = 0  # temporarily remove
            c = np.convolve(s, s)  # convolution without s[i]
            # d[k] = 2*s_fixed[k-i]
            d = np.zeros(2*n-1)
            valid_start = max(0, i - (n-1))  # k-i >= 0 => k >= i
            valid_end = min(2*n-1, i + n)    # k-i <= n-1 => k <= i+n-1
            # Actually d[k] = 2*s[k-i] if 0 <= k-i <= n-1
            for k in range(max(0, i), min(2*n-1, i+n)):
                d[k] = 2*s[k-i]
            # e[k] = 1 if k == 2i, else 0
            e = np.zeros(2*n-1)
            if 0 <= 2*i <= 2*n-2:
                e[2*i] = 1
            
            # Minimize max_k (c[k] + d[k]*t + e[k]*t^2) over t >= 0
            # Grid search
            t_vals = np.linspace(0, 2*old_val + 0.01, 201)
            f_vals = c + t_vals[:,None]*d[None,:] + t_vals[:,None]**2*e[None,:]
            max_f = np.max(f_vals, axis=1)
            best_t = t_vals[np.argmin(max_f)]
            
            # Refine with ternary search around best_t
            lo = max(0, best_t - (2*old_val+0.01)/200)
            hi = best_t + (2*old_val+0.01)/200
            for _ in range(30):
                m1 = lo + (hi-lo)/3
                m2 = hi - (hi-lo)/3
                f1 = np.max(c + m1*d + m1**2*e)
                f2 = np.max(c + m2*d + m2**2*e)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_t = (lo + hi) / 2
            
            s[i] = max(0, best_t)
            if abs(s[i] - old_val) > 1e-10:
                improved = True
        if not improved:
            break
    return s

"""
Hybrid coordinate descent + LP polishing to minimize 2n*max(conv(s,s))/(sum(s))^2.
Uses exact 1D subproblems for coordinate updates (fast, many iterations),
then LP-guided direction for final polishing. Multiple initializations and
sequence lengths explored within the time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_score(s):
    """Compute 2n*max(convolve(s,s))/(sum(s))^2."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * conv.max() / (total * total)


def score_with_conv(s, conv):
    """Compute score given precomputed convolution."""
    n = len(s)
    total = s.sum()
    if total < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (total * total)


def coordinate_descent(s, max_rounds=200, deadline=None, grid_size=80, fine_size=150):
    """
    Coordinate descent minimizing max(conv)/(sum)^2 via exact 1D subproblems.
    For each coordinate i, solve: minimize max_k conv_t[k] / (S+t)^2 over t>=0,
    where conv_t is the convolution with s[i] replaced by t.
    """
    n = len(s)
    s = s.copy().astype(np.float64)
    conv = np.convolve(s, s).astype(np.float64)
    total_sum = float(s.sum())
    best_score = 2.0 * n * conv.max() / (total_sum * total_sum)
    no_improve_rounds = 0

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        score_before = 2.0 * n * conv.max() / (total_sum * total_sum)
        any_improved = False

        for i in range(n):
            v = s[i]
            S = total_sum - v

            if S < 1e-15:
                continue

            k_lo = max(0, i)
            k_hi = min(2 * n - 1, i + n - 1) + 1  # exclusive

            two_i_valid = (0 <= 2 * i <= 2 * n - 2)
            two_i_in_range = (k_lo <= 2 * i < k_hi)

            # Outside max (conv values that don't change with t)
            if k_lo > 0 or k_hi <= 2 * n - 1:
                parts = []
                if k_lo > 0:
                    parts.append(conv[:k_lo])
                if k_hi <= 2 * n - 1:
                    parts.append(conv[k_hi:])
                if parts:
                    outside_max = np.max(np.concatenate(parts))
                else:
                    outside_max = 0.0
            else:
                outside_max = 0.0

            L = k_hi - k_lo
            if L <= 0:
                continue

            sv = s[k_lo - i: k_hi - i]
            cv = conv[k_lo: k_hi]

            # Grid for initial search
            t_max = max(3.0 * v + 0.05, 0.05)
            t_grid = np.linspace(0.0, t_max, grid_size)

            # Linear part: intercepts + slopes * t
            intercepts = cv - 2.0 * v * sv
            slopes = 2.0 * sv

            linear_vals = intercepts[:, None] + slopes[:, None] * t_grid[None, :]

            if two_i_in_range:
                idx_2i = 2 * i - k_lo
                linear_vals[idx_2i, :] = cv[idx_2i] + t_grid ** 2 - v ** 2

            max_linear = np.max(linear_vals, axis=0)
            max_total = np.maximum(outside_max, max_linear)
            denom = (S + t_grid) ** 2
            R = max_total / denom

            best_idx = np.argmin(R)

            # Fine refinement
            if best_idx > 0:
                lo_t = t_grid[best_idx - 1]
            else:
                lo_t = 0.0
            if best_idx < grid_size - 1:
                hi_t = t_grid[best_idx + 1]
            else:
                hi_t = t_grid[-1]

            t_fine = np.linspace(lo_t, hi_t, fine_size)
            linear_fine = intercepts[:, None] + slopes[:, None] * t_fine[None, :]
            if two_i_in_range:
                linear_fine[idx_2i, :] = cv[idx_2i] + t_fine ** 2 - v ** 2
            max_linear_fine = np.max(linear_fine, axis=0)
            max_total_fine = np.maximum(outside_max, max_linear_fine)
            denom_fine = (S + t_fine) ** 2
            R_fine = max_total_fine / denom_fine
            best_t = float(t_fine[np.argmin(R_fine)])
            best_t = max(0.0, best_t)

            # Also check t=0
            R0_num = max(outside_max,
                         float(np.max(intercepts)),
                         (cv[idx_2i] - v ** 2) if two_i_in_range else -np.inf)
            R0 = R0_num / (S * S) if S > 0 else float('inf')
            R_best = max_total_fine[np.argmin(R_fine)] / denom_fine[np.argmin(R_fine)]
            if R0 < R_best:
                best_t = 0.0

            if abs(best_t - v) > 1e-12:
                delta = best_t - v
                conv[k_lo: k_hi] += 2.0 * delta * sv
                if two_i_in_range:
                    conv[2 * i] += delta * delta
                s[i] = best_t
                total_sum += delta
                any_improved = True

        score_after = 2.0 * n * conv.max() / (total_sum * total_sum)
        if score_after < best_score:
            best_score = score_after
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1

        if not any_improved:
            break
        if no_improve_rounds > 5:
            break

    return s, best_score


def build_conv_matrix(s):
    """Build (2n-1) x n matrix where A[k,j] = s[k-j] if 0<=k-j<n else 0."""
    n = len(s)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j: j + n, j] = s
    return A


def lp_polish(s, deadline, max_iters=40):
    """LP-guided local search for final polishing."""
    n = len(s)
    s = s.copy().astype(np.float64)
    best_sc = fast_score(s)
    best_s = s.copy()
    no_imp = 0

    A = build_conv_matrix(s)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = np.convolve(s, s)
        M = conv.max()

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 8.0, 'presolve': True})
            if not res.success or res.x is None:
                break
            b_dir = np.maximum(res.x, 0.0)
        except Exception:
            break

        sum_b = b_dir.sum()
        if sum_b < 1e-14:
            break

        b_dir = b_dir / sum_b

        # Line search
        best_t = 0.0
        best_ts = best_sc
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * s + t * b_dir
            cand = np.maximum(cand, 0.0)
            ts = fast_score(cand)
            if ts < best_ts - 1e-14:
                best_ts = ts
                best_t = t

        if best_t == 0.0:
            no_imp += 1
            if no_imp > 5:
                break
            continue

        s = (1.0 - best_t) * s + best_t * b_dir
        s = np.maximum(s, 0.0)
        s = s / s.sum()
        cur_sc = fast_score(s)
        no_imp = 0

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_s = s.copy()
            # Update A matrix since s changed
            A = build_conv_matrix(s)

        # Refresh A periodically
        if it % 5 == 4:
            A = build_conv_matrix(s)

    return best_s, best_sc
