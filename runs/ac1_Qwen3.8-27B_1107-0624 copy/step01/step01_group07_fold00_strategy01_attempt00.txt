                if time.monotonic() < deadline - 1 + 0.1:

                    continue
                else:

                # Step-3: Return early if deadline passed but we have a new best
                if time.monotonic() > deadline:
                    break
        
        if best_perturbation is None:

            iters_since_last_improvement += 1
            # Rate limit the LP solver calls to avoid system overload
            if iters_since_last_improvement % 10 == 0:
                new_score, new_seq = optimize_lp(best_seq, deadline - 60)  # 60 seconds buffer for final
                if new_score < best_score:
                    best_seq = new_seq
                    best_score = new_score
                    print(f"LP improvement: score {best_score}")
                    iters_since_last_improvement = 0
                elif new_score < best_score * 1.001:  # Sometimes LP finds small improvements
                    best_seq = new_seq
                    best_score = new_score
                    print(f"LP marginal improvement: score {best_score} down to {new_score}")
                    iters_since_last_improvement = 0
        
        n_iterations += 1
        
        # Intentional break if we are making progress slowly
        if n_iterations % (budget_s - 800) == 0 and iters_since_last_improvement > 50:
            # Time to burn LP in a slightly smarter way by pre-filtering peaks
            # We have to try a strategy that recaptures the LP magic but with gradient direction
            # Step-4: Perform LP based search periodically, but we already have a direct method
            # Use here some lookahead at convolution: if not improving, break out to insert more of our direct update method
            break
    
    # Final fallback if best_seq is still None
    if best_seq is None:
        best_seq = np.ones(256)
        best_score = evaluate_sequence(best_seq.tolist())
    else:
        # Final polishing
        seq_copy = np.array(best_seq)
        if time.monotonic() < deadline - 60:  # Leave 60 seconds for final optimization
            # Call LP to polish the candidate
            # Integrated the above method but with values: we don't know deadline - time_monotonic+? but one call should suffice
            best_seq, _ = optimize_gradient(best_seq, deadline, rng)
            best_score = np.inf

        if best_score >= np.inf:
            # Try gradient as fallback polishing method
            best_score = evaluate_gradient(best_seq).item()
    
    # Fixed response: return the sequence as list
    return best_seq
