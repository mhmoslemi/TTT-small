    g = np.fft.irfft(np.fft.rfft(a_rev, N) * np.fft.rfft(s, N), N)
    g = g[n - 1: 2 * n - 1]
    g /= num_active
    return g, M


def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    # Build constraint matrix: A @ g = conv(a, g)
    # A has shape (2n-1, n), A[k, j] = a[k-j] if 0<=k-j<n else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x.sum() > 1e-15:
            return res.x / res.x.sum()
    except Exception:
        pass
    return None


def mirror_descent_optimize(a, deadline, rng, use_lp=True, lp_interval=100, n_lp_total=30):
    """
    Main optimization loop using mirror descent (multiplicative updates)
    with periodic LP refinement for better directions.
    """
    n = len(a)
    a = np.clip(a, 1e-15, None).astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_obj(a)
    alpha = 0.15
    stall = 0
    it = 0
    lp_count = 0
    lp_every = lp_interval if n <= 1500 else 200
    threshold_frac = 0.90
    
    # Phase 1: Fast convergence with coarser threshold
    phase1_deadline = deadline - 120.0 if (deadline - time.monotonic()) > 240 else deadline - 60.0
    
    while time.monotonic() < phase1_deadline:
        it += 1
        c = fft_convolve(a, a)
        M = c.max()
        
        g, M = compute_subgradient(a, c, threshold_frac)
        
        # Mirror descent step
        g_mean = g.mean()
        if g_mean > 1e-15:
            a_new = a * np.exp(-alpha * g / g_mean)
        else:
            a_new = a.copy()
        a_new = np.clip(a_new, 1e-300, None)
        a_new /= a_new.sum()
        
        sc = compute_obj(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            alpha = min(0.4, alpha * 1.01)
        else:
            stall += 1
            alpha = max(0.005, alpha * 0.85)
            
            # LP refinement
            if use_lp and n <= 2048 and lp_count < n_lp_total and stall >= 15:
                g_dir = lp_direction(a, M)
                if g_dir is not None:
                    lp_count += 1
                    improved = False
                    for t in [0.5, 0.3, 0.15, 0.08, 0.04]:
                        a_try = (1 - t) * best_a + t * g_dir
                        a_try = np.clip(a_try, 1e-15, None)
                        a_try /= a_try.sum()
                        sc_try = compute_obj(a_try)
                        if sc_try < best_score - 1e-14:
                            best_a = a_try.copy()
                            best_score = sc_try
                            a = a_try
                            improved = True
                            stall = 0
                            break
                    if not improved:
                        # Perturb and continue
                        noise = np.abs(rng.normal(0, 0.05, n))
                        a = best_a * (1.0 + noise)
                        a = np.clip(a, 1e-15, None)
                        a /= a.sum()
                        stall = 0
                else:
                    noise = np.abs(rng.normal(0, 0.08, n))
                    a = best_a * (1.0 + noise)
                    a = np.clip(a, 1e-15, None)
                    a /= a.sum()
                    stall = 0
            elif stall >= 30:
                # Random escape
                scale = 0.05 + 0.1 * (stall / 100.0)
                noise = np.abs(rng.normal(0, scale, n))
                a = best_a * (1.0 + noise)
                a = np.clip(a, 1e-15, None)
                a /= a.sum()
                stall = 0
            
            if stall > 300:
                break
        
        if it % 500 == 0:
            print(f"        MD iter {it}, score={best_score:.6f}, alpha={alpha:.4f}")
    
    # Phase 2: Fine optimization with tighter threshold
    threshold_frac = 0.97
    alpha = 0.05
    stall = 0
    a = best_a.copy()
    
    while time.monotonic() < deadline:
        it += 1
        c = fft_convolve(a, a)
        M = c.max()
        
        g, M = compute_subgradient(a, c, threshold_frac)
        
        g_mean = g.mean()
        if g_mean > 1e-15:
            a_new = a * np.exp(-alpha * g / g_mean)
        else:
            a_new = a.copy()
        a_new = np.clip(a_new, 1e-300, None)
        a_new /= a_new.sum()
        
        sc = compute_obj(a_new)
        if sc < best_score - 1e-14:
            best_a = a_new.copy()
            best_score = sc
            a = a_new
            stall = 0
            alpha = min(0.2, alpha * 1.01)
        else:
            stall += 1
            alpha = max(0.001, alpha * 0.9)
            
            if stall >= 50 and use_lp and n <= 2048:
                g_dir = lp_direction(a, M)
                if g_dir is not None:
                    for t in [0.3, 0.15, 0.08, 0.04]:
                        a_try = (1 - t) * best_a + t * g_dir
                        a_try = np.clip(a_try, 1e-15, None)
                        a_try /= a_try.sum()
                        sc_try = compute_obj(a_try)
                        if sc_try < best_score - 1e-14:
                            best_a = a_try.copy()
                            best_score = sc_try
                            a = a_try
                            stall = 0
                            break
                    else:
                        noise = np.abs(rng.normal(0, 0.03, n))
                        a = best_a * (1.0 + noise)
                        a = np.clip(a, 1e-15, None)
                        a /= a.sum()
                        stall = 0
                else:
                    a = best_a
                    stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def coordinate_descent(a, max_passes=50, deadline=None):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy()
    best_score = compute_obj(a)
    
    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            # Optimize a[i] with others fixed
            # conv(a,a)[k] for k involving i changes
            # Quick: just try a few values
            orig = a[i]
            for factor in [0.7, 0.85, 1.15, 1.3, 0.5, 1.5]:
                a[i] = orig * factor
                if a[i] < 0:
                    a[i] = 0
                a_norm = a / a.sum()
                sc = compute_obj(a_norm)
                if sc < best_score:
                    best_score = sc
                    a = a_norm.copy()
                    improved = True
            a[i] = orig  # restore if not improved
    
        if not improved:
            break
    
    return a, best_score


def binom_init(n):
    """Initialize with central binomial coefficients C(2k,k)/4^k."""
    k = np.arange(n, dtype=np.float64)
    # C(2k, k) / 4^k computed via log for stability
    log_c = np.zeros(n)
    for i in range(1, n):
        # C(2k,k)/4^k = product of (2j-1)/(2j) for j=1..k
        # Log: sum_{j=1}^k log((2j-1)/(2j))
        # Or use: log(C(2k,k)) - k*log(4)
        # For large k, use Stirling or iterative
        pass
    # Iterative computation
    c = np.ones(n)
    c[0] = 1.0
    for i in range(1, n):
        c[i] = c[i-1] * (2*i - 1) / (2*i)
    return c


def cosine_init(n):
    """Asymmetric cosine initialization."""
    x = np.linspace(0, 1, n)
    return 0.5 * (1 + np.cos(np.pi * x))


def powerlaw_init(n, alpha=0.5):
    """Power-law decay initialization."""
    k = np.arange(1, n + 1, dtype=np.float64)
    return k ** (-alpha)


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian window (centered)."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def asymmetric_gaussian_init(n, sigma_frac=0.4, skew=0.3):
    """Asymmetric Gaussian to break symmetry."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    # Asymmetric: wider on one side
    left_sigma = sigma * (1 - skew)
    right_sigma = sigma * (1 + skew)
    result = np.where(x <= center,
                      np.exp(-0.5 * ((x - center) / left_sigma) ** 2),
                      np.exp(-0.5 * ((x - center) / right_sigma) ** 2))
    return result


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for optimal non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-based mirror descent with LP refinement, multiple initializations,
    and adaptive step sizes.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and (p >= 0).all():
            parent = p.copy()
    except Exception:
        pass
    
    # Define configurations: (n, init_func, description)
    configs = [
        # Binomial coefficient inits (theoretically motivated)
        (512, binom_init, "binom_512"),
        (1024, binom_init, "binom_1024"),
        (2048, binom_init, "binom_2048"),
        (4096, binom_init, "binom_4096"),
        # Asymmetric inits
        (1024, asymmetric_gaussian_init, "asym_gauss_1024"),
        (2048, asymmetric_gaussian_init, "asym_gauss_2048"),
        # Power law
        (1024, lambda n: powerlaw_init(n, 0.5), "powerlaw_1024"),
        (2048, lambda n: powerlaw_init(n, 0.5), "powerlaw_2048"),
        (2048, lambda n: powerlaw_init(n, 0.7), "powerlaw07_2048"),
        # Cosine
        (1024, cosine_init, "cosine_1024"),
        (2048, cosine_init, "cosine_2048"),
    ]
    
    # Add parent-based configs
    if parent is not None:
        pn = len(parent)
        configs.insert(0, (pn, lambda n, p=parent: p[:n].copy() if p.size >= n else np.pad(p, (0, n-p.size)), f"parent_{pn}"))
        if pn > 512:
            configs.insert(1, (1024, lambda n, p=parent: np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p), "parent_interp_1024"))
        if pn > 1024:
            configs.insert(2, (2048, lambda n, p=parent: np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p), "parent_interp_2048"))
    
    # Add a random init
    configs.append((1024, lambda n: np.abs(rng.normal(0, 1, n)), "random_1024"))
    configs.append((2048, lambda n: np.abs(rng.normal(0, 1, n)), "random_2048"))
    
    total_configs = len(configs)
    print(f"Total configs: {total_configs}, budget: {budget_s}s")
    
    # Allocate time per config - weight towards promising sizes
    weights = []
    for n, init_f, desc in configs:
        if n == 1024:
            w = 1.8
        elif n == 2048:
            w = 1.5
        elif n == 4096:
            w = 1.2
        elif n == 512:
            w = 1.0
        else:
            w = 1.0
        if 'parent' in desc:
            w *= 1.5
        weights.append(w)
    
    total_w = sum(weights)
    available = deadline - start_time - 120.0  # reserve 120s for final polish
    time_bounds = []
    cum = 0.0
    for w in weights:
        cum += available * w / total_w
        time_bounds.append(cum)
    
    for idx, (n, init_func, desc) in enumerate(configs):
        if time.monotonic() >= deadline:
            break
        
        cfg_deadline = start_time + time_bounds[idx]
        cfg_deadline = min(cfg_deadline, deadline)
        
        if cfg_deadline <= time.monotonic() + 5:
            continue
        
        remaining = deadline - time.monotonic()
        print(f"[{idx+1}/{total_configs}] n={n}, init={desc}, remaining~{remaining:.0f}s")
        
        try:
            a = init_func(n)
        except Exception:
            a = np.ones(n)
        
        a = np.clip(a, 1e-15, None).astype(np.float64)
        if a.sum() < 1e-10:
            a = np.ones(n)
        a = a / a.sum()
        
        # Initial score
        sc = compute_obj(a)
        print(f"      Initial score: {sc:.6f}")
        
        # Run optimizer
        use_lp = (n <= 2048)
        a_opt, score = mirror_descent_optimize(a, cfg_deadline, rng, use_lp=use_lp)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"      => NEW BEST: score={best_score:.6f} (n={n}, {desc})")
        else:
            print(f"      => score={score:.6f} (best: {best_score:.6f})")
    
    # Final polish on best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\nFinal polish (remaining {remaining:.0f}s):")
            n = len(best_seq)
            use_lp = (n <= 2048)
            best_seq, best_score = mirror_descent_optimize(
                best_seq, deadline, rng, use_lp=use_lp,
                lp_interval=50, n_lp_total=20
            )
            print(f"  After MD polish: {best_score:.6f}")
        
        # Coordinate descent fine-tuning on a subset of coordinates
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n = len(best_seq)
            # Only optimize every k-th coordinate for speed
            k = max(1, n // 100)
            coords_to_opt = list(range(0, n, k))
            cs_deadline = min(start_time + remaining * 0.5, deadline - 5)
            
            current_score = best_score
            for i in coords_to_opt:
                if time.monotonic() > cs_deadline:
                    break
                orig = best_seq[i]
                for factor in [0.8, 0.9, 1.1, 1.2, 0.7, 1.3, 0.95, 1.05]:
                    best_seq[i] = orig * factor
                    if best_seq[i] < 0:
                        best_seq[i] = 0
                    a_norm = best_seq / best_seq.sum()
                    sc = compute_obj(a_norm)
                    if sc < current_score:
                        current_score = sc
                        best_seq = a_norm.copy()
                # Restore if nothing improved
                if best_seq[i] != orig:
                    pass  # keep improved value
                else:
                    best_seq[i] = orig
            
            if current_score < best_score:
                best_score = current_score
                print(f"  After coord descent: {best_score:.6f}")
    
    # Safety: ensure valid output
    if best_seq is None:
        best_seq = binom_init(512)
        best_score = compute_obj(best_seq)
    
    # Ensure sum >= 0.01
    if best_seq.sum() < 0.01:
        best_seq = best_seq * (0.01 / best_seq.sum())
    
    # Clip to [0, 1000]
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE in {elapsed:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")
    
    return [float(x) for x in best_seq]
