One important detail: the projection onto the simplex. The standard algorithm is:

2. Find the threshold using the "water-filling" algorithm

Here's an efficient O(n log n) implementation:

```python
def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.where(cond)[0]
    if rho.size == 0:
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def convolve_fft(a, b):
    """Full linear convolution using FFT."""
    n, m = a.size, b.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.zeros(size, dtype=np.float64)
    fa[:n] = a
    fb = np.zeros(size, dtype=np.float64)
    fb[:m] = b
    out = np.fft.irfft(np.fft.rfft(fa, size) * np.fft.rfft(fb, size), size)
    return out[:need]


def score_sum1(f):
    """Score for a sum-1 non-negative sequence: 2*n*max(conv(f,f))."""

    n = f.size
    if n < 2:
        return np.inf
    c = convolve_fft(f, f)
    return 2.0 * n * float(c.max())
