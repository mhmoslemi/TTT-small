"""
Minimize f(seq) = 2*n*max(convolve(seq,seq)) / (sum(seq))^2
using: (1) structured initialization, (2) top-k gradient descent with
backtracking, (3) LP cutting-plane refinement, (4) coordinate descent polish.
"""

import time
import numpy as np

try:
    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def _next_pow2(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def _fft_conv_full(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n + m - 1]


def _score(a):
    """Compute 2*n*max(conv(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    c = _fft_conv_full(a)
    maxc = float(c.max())
    if not np.isfinite(maxc):
        return np.inf
    return 2.0 * n * maxc / (s * s)


def _greedy(n):
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a / a.sum()


def _power_init(n, exp):
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -exp)
    return a / a.sum()


def _mix(g, u, lam):
    return (lam * g + (1.0 - lam) * u) / (lam * g + (1.0 - lam) * u).sum()


def _cosine_taper(n, alpha):
    idx = np.arange(n, dtype=np.float64)
    t = np.cos(np.pi * idx / (2.0 * (n - 1))) ** alpha
    a = t / t.sum()
    return a


def _bump(n, width):
    center = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((idx - center) / width) ** 2)
    return a / a.sum()


def _topk_gradient(a, top_k, L):
    """Compute gradient using top-k lags average."""
    n = len(a)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    if top_k >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1, dtype=int)
    else:
        top_idx = np.argsort(c)[::-1][:top_k].astype(int)

    g = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g *= 4.0 * n / len(top_idx)
    return g, c.max()


def _adam_topk(a, budget_s, top_k_init=10, top_k_max=80, seed=0):
    """Adam optimizer with top-k gradient."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-12:
        return a, np.inf
    a = a / s

    L = _next_pow2(3 * n - 2)
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_conv_full(a).max())

    rng = np.random.default_rng(seed)
    lr = 0.03 if n < 512 else 0.02
    mu1, mu2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    start = time.monotonic()
    end = start + max(1.0, budget_s - 3.0)

    while time.monotonic() < end:
        t += 1
        frac = min(1.0, (time.monotonic() - start) / max(1e-9, budget_s))
        top_k = int(top_k_init + (top_k_max - top_k_init) * min(1.0, frac * 1.5))
        top_k = max(1, min(top_k, 2 * n - 1))

        g, maxc = _topk_gradient(a, top_k, L)
        cur_score = 2.0 * n * maxc
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Adam update in simplex space (using entropic parameterization)
        # Gradient projected to sum-zero (tangent to simplex)
        gp = float(np.dot(a, g))
        g_proj = a * (g - gp)

        m = mu1 * m + (1.0 - mu1) * g_proj
        v = mu2 * v + (1.0 - mu2) * (g_proj * g_proj)
        mhat = m / (1.0 - mu1 ** t)
        vhat = v / (1.0 - mu2 ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)

        # Apply in log-space (mirror descent style)
        log_a = np.log(a + 1e-300)
        log_a -= log_a.mean()
        log_a -= step
        log_a -= log_a.mean()

        if not np.all(np.isfinite(log_a)):
            break
        log_a = np.clip(log_a, -50, 50)
        log_a -= log_a.max()
        e = np.exp(log_a)
        se = e.sum()
        if se < 1e-300:
            break
        a = e / se

        # Backtracking line search (occasionally)
        if t % 50 == 0:
            for shrink in [0.9, 0.8, 0.7, 0.6]:
                test = a * shrink + best_a * (1 - shrink)
                ts = test.sum()
                if ts < 1e-12:
                    continue
                test = test / ts
                ts_score = 2.0 * n * float(_fft_conv_full(test).max())
                if ts_score < best_score:
                    a = test.copy()
                    best_score = ts_score
                    best_a = a.copy()
                    break

    final_score = 2.0 * n * float(_fft_conv_full(a).max())
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def _lp_refine(a, budget_s, max_rows=None):
    """LP cutting-plane: maximize sum(g) s.t. f*g <= M, g >= 0."""
    if _scipy_opt is None:
        return a, _score_low(a)

    n = len(a)
    S0 = np.sqrt(2.0 * n)
    f = (a / a.sum()) * S0
    M = float(_fft_conv_full(f).max())
    best_f = f.copy()
    best_score = 2.0 * n * M / (S0 * S0)

    if max_rows is None:
        max_rows = min(800, max(200, 400000 // max(1, n)))

    end = time.monotonic() + max(1.0, budget_s - 5.0)

    # Initial rows: evenly spaced
    n_rows_init = min(max_rows // 2, 400)
    rows = list(np.unique(np.linspace(0, 2 * n - 2, n_rows_init, dtype=int)))
    row_set = set(rows)

    while time.monotonic() < end:
        rows_arr = np.array(rows, dtype=int)
        nr = len(rows_arr)
        A = np.zeros((nr, n), dtype=np.float64)
        for ii in range(nr):
            k = int(rows_arr[ii])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]

        b = np.full(nr, M, dtype=np.float64)
        cobj = -np.ones(n)

        rem = max(1.0, end - time.monotonic() - 1.0)
        tl = max(3.0, min(60.0, rem * 0.3))
        try:
            res = _scipy_opt.linprog(cobj, A_ub=A, b_ub=b,
                                     bounds=(0, None), method='highs',
                                     options={'time_limit': tl})
        except Exception:
            break

        if res is None or not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        sg = g.sum()
        if sg < 1e-12:
            break

        # Check violations
        conv = _fft_conv_full(f, g)
        viol = conv - M
        tol = 1e-6 * M + 1e-8
        if viol.max() > tol:
            if len(rows) < max_rows:
                order = np.argsort(viol)[::-1]
                added = 0
                for ix in order:
                    if added >= 128 or added >= max_rows - len(rows):
                        break
                    ix = int(ix)
                    if viol[ix] > tol and ix not in row_set:
                        rows.append(ix)
                        row_set.add(ix)
                        added += 1
                continue
            else:
                if viol.max() > 10 * tol:
                    break

        # Rescale and line search
        g = g * (S0 / sg)
        accepted = None
        accepted_score = np.inf
        for step in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            fn = (1.0 - step) * f + step * g
            fn = np.maximum(fn, 0)
            sn = fn.sum()
            if sn < 1e-12:
                continue
            fn = fn * (S0 / sn)
            sc = 2.0 * n * float(_fft_conv_full(fn).max()) / (S0 * S0)
            if sc < accepted_score:
                accepted_score = sc
                accepted = fn.copy()

        if accepted is None or accepted_score >= best_score:
            break

        f = accepted
        M = float(_fft_conv_full(f).max())
        best_f = f.copy()
        best_score = accepted_score
        print(f"  LP: n={n} score={best_score:.6f} rows={len(rows)}")

    return best_f / S0, best_score


def _score_low(a):
    """Score assuming a is already normalized to sum sqrt(2n)."""
    n = len(a)
    S0 = np.sqrt(2.0 * n)
    return 2.0 * n * float(_fft_conv_full(a).max()) / (S0 * S0)


def _coord_descent(a, budget_s):
    """Cyclic coordinate descent with exact 1D optimization."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    L = _next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_conv_full(a).max())

    start = time.monotonic()
    end = start + max(1.0, budget_s - 2.0)
    it = 0

    while time.monotonic() < end:
        it += 1
        improved = False
        for i in range(n):
            if time.monotonic() > end:
                break
            # Optimize a[i] with others fixed
            # c_k = sum_j a_j * a_{k-j}; terms involving a_i:
            # For lag k, a_i appears when j=i (term a_i * a_{k-i}) or k-j=i (term a_j * a_i)
            # So dc_k/d a_i = 2 * a_{k-i} (for valid k-i)
            # c_k = a_i^2 * [k=0 contributes a_i*a_i] + a_i * (other terms) + (terms without a_i)
            # Let's define: c_k = a_i * B_k + C_k where B_k involves a once, C_k involves a_i 0 or 2 times
            
            # Actually: c_k = sum_j a_j * a_{k-j}
            # = a_i * a_{k-i} + sum_{j != i} a_j * a_{k-j}
            # But also if k-j = i, i.e., j = k-i, then we also get a_{k-i} * a_i
            # So there are potentially two terms with a_i:
            #   - j = i: a_i * a_{k-i}
            #   - j = k-i: a_{k-i} * a_i (if k-i != i)
            # If k-i == i (i.e., k = 2i), then both terms are the same: a_i^2
            
            # Let me just do a 1D golden section or ternary search on a[i]
            ai_old = a[i]
            # Find optimal a[i] by minimizing max_k c_k
            # c_k as a function of a[i]:
            # For each k: c_k = a[i] * (sum of a_{k-j} for j where the term involves a[i]) + rest
            
            # Simpler: just use golden section on [0, 3*ai_old + 0.1]
            lo = 0.0
            hi = 3.0 * ai_old + 0.5
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            
            if lo >= hi:
                hi = lo + 1.0
            
            c1 = lo + (1 - gr) * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            # Pre-compute the "rest" of the convolution (without a_i contributions)
            # c_k_rest = c_k - (contribution from a_i)
            # Contribution from a_i to c_k:
            #   if 0 <= k-i < n: a[i] * a[k-i] (from j=i)
            #   if 0 <= k-i < n and k-i != i: a[k-i] * a[i] (from j=k-i)
            #   if k == 2*i and 0 <= i < n: a[i]^2 (double counted, so subtract one)
            
            # This is getting complicated. Let me just evaluate directly.
            # For speed, limit to a few iterations of golden section.
            
            for _ in range(8):
                a[i] = c1
                s1 = a.sum()
                if s1 < 1e-12:
                    f1 = 1e30
                else:
                    aa = a / s1
                    f1 = float(_fft_conv_full(aa).max())
                
                a[i] = c2
                s2 = a.sum()
                if s2 < 1e-12:
                    f2 = 1e30
                else:
                    aa = a / s2
                    f2 = float(_fft_conv_full(aa).max())
                
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + (1 - gr) * (hi - lo)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
            
            a[i] = (lo + hi) / 2.0
            s_new = a.sum()
            if s_new > 1e-12:
                a = a / s_new
                sc = 2.0 * n * float(_fft_conv_full(a).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved and it > 1:
            break

    return best_a, best_score


def _stochastic_perturb(a, rng, sigma):
    """Multiplicative log-normal perturbation."""
    noise = np.exp(rng.normal(0, sigma, len(a)))
    a_new = a * noise
    s = a_new.sum()
    if s < 1e-12:
        return a
    return a_new / s


def _scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return [1.0] * 16
    b = a * (target_sum / s)
    b = np.clip(b, 0.0, 1000.0)
    return [float(x) for x in b]


def _try_global_eval(seq_list):
    try:
        val = evaluate_sequence(list(seq_list))
        val = float(val)
        return val if np.isfinite(val) else np.inf
    except Exception:
        return np.inf


def main_search(seed, budget_s):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(int(seed))

    best_p = None
    best_score = np.inf
    best_by_n = {}

    def record(p, score=None):
        nonlocal best_p, best_score
        if p is None:
            return
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            return
        if score is None:
            score = _score(a)
        if not np.isfinite(score):
            return
        if score < best_score:
            best_p = a.copy()
            best_score = float(score)
            print(f"  NEW BEST: n={a.size} score={score:.6f}")
        n = a.size
        old = best_by_n.get(n)
        if old is None or score < old[0]:
            best_by_n[n] = (float(score), a.copy())

    # Phase 1: Structured initializations
    print("Phase 1: Structured initializations")
    lengths = [64, 96, 128, 160, 192, 224, 256, 320, 384, 448, 512, 640, 768, 1024, 1280, 1536, 2048, 2560, 3072]
    
    for n in lengths:
        if time.monotonic() >= deadline - 60:
            break
        inits = [
            _greedy(n),
            _power_init(n, 0.4),
            _power_init(n, 0.5),
            _power_init(n, 0.6),
            _power_init(n, 0.7),
            _cosine_taper(n, 1.0),
            _cosine_taper(n, 2.0),
            _bump(n, n * 0.3),
            _mix(_greedy(n), np.ones(n) / n, 0.5),
            _mix(_greedy(n), np.ones(n) / n, 0.7),
            _mix(_greedy(n), np.ones(n) / n, 0.3),
        ]
        for p in inits:
            sc = _score(p)
            record(p, sc)

    # Also try the provided height_sequence_1 if available
    h = globals().get('height_sequence_1', None)
    if h is not None:
        try:
            arr = np.asarray(h, dtype=np.float64).ravel()
            arr = np.clip(arr, 0, 1000)
            if arr.size > 0 and arr.sum() > 0.01:
                record(arr)
        except Exception:
            pass

    # Phase 2: Gradient descent from promising starts
    print("Phase 2: Gradient descent")
    # Sort candidates by score
    items = [(sc, n, p) for n, (sc, p) in best_by_n.items()]
    items.sort(key=lambda x: x[0])

    # Select top candidates for gradient descent
    grad_budget_total = 600.0
    n_grad = min(len(items), 12)
    per_budget = grad_budget_total / max(1, n_grad)

    seen_sizes = set()
    count = 0
    for sc, n, p in items:
        if time.monotonic() >= deadline - 120:
            break
        if count >= n_grad:
            break
        # Avoid too many of the same size
        if n in seen_sizes and len(seen_sizes) < 8:
            continue
        seen_sizes.add(n)
        count += 1
        alloc = min(per_budget, max(10.0, deadline - time.monotonic() - 120))
        print(f"  GradDescent n={n} init_score={sc:.6f} budget={alloc:.0f}s")
        p_out, f_out = _adam_topk(p, alloc, top_k_init=5, top_k_max=50,
                                   seed=int(seed) + n)
        record(p_out, f_out)

    # Also run from random perturbations of the current best
    if best_p is not None:
        for trial in range(3):
            if time.monotonic() >= deadline - 100:
                break
            p_pert = _stochastic_perturb(best_p, rng, 0.15)
            alloc = min(80.0, max(15.0, deadline - time.monotonic() - 100))
            p_out, f_out = _adam_topk(p_pert, alloc, top_k_init=10, top_k_max=60,
                                       seed=int(seed) + 1000 + trial)
            record(p_out, f_out)

    # Phase 3: LP refinement
    print("Phase 3: LP refinement")
    if _scipy_opt is not None:
        lp_candidates = []
        for sc, n, p in items:
            if n <= 2048:
                lp_candidates.append((sc, n, p))
        # Also add best
        if best_p is not None:
            bn = len(best_p)
            if bn <= 2048:
                lp_candidates.append((best_score, bn, best_p))
        lp_candidates.sort(key=lambda x: x[0])

        lp_seen = set()
        for sc, n, p in lp_candidates:
            if time.monotonic() >= deadline - 80:
                break
            if n in lp_seen:
                continue
            if len(lp_seen) >= 4:
                break
            lp_seen.add(n)
            alloc = min(180.0, max(40.0, deadline - time.monotonic() - 80))
            print(f"  LP n={n} init_score={sc:.6f} budget={alloc:.0f}s")
            p_out, f_out = _lp_refine(p, alloc)
            record(p_out, f_out)

    # Phase 4: Coordinate descent polish
    print("Phase 4: Coordinate descent polish")
    if best_p is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            bn = len(best_p)
            if bn <= 512:
                alloc = min(rem - 15, 120.0)
                p_out, f_out = _coord_descent(best_p, alloc)
                record(p_out, f_out)
            else:
                # For longer sequences, do a partial coordinate descent
                # Just do a few passes over a subset
                alloc = min(rem - 15, 60.0)
                a = best_p.copy()
                s = a.sum()
                a = a / s
                best_cd = a.copy()
                best_cd_sc = 2.0 * a.size * float(_fft_conv_full(a).max())
                start_cd = time.monotonic()
                end_cd = start_cd + alloc
                it_cd = 0
                while time.monotonic() < end_cd:
                    it_cd += 1
                    # Random subset of coordinates
                    k = min(32, len(a))
                    idx = rng.choice(len(a), size=k, replace=False)
                    for i in idx:
                        if time.monotonic() > end_cd:
                            break
                        ai = a[i]
                        # Try a few values
                        for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]:
                            a[i] = ai * mult
                            ss = a.sum()
                            if ss < 1e-12:
                                a[i] = ai
                                continue
                            aa = a / ss
                            sc_cd = 2.0 * aa.size * float(_fft_conv_full(aa).max())
                            if sc_cd < best_cd_sc:
                                best_cd_sc = sc_cd
                                best_cd = aa.copy()
                                a = aa.copy()
                        a[i] = ai  # restore
                record(best_cd, best_cd_sc)

    # Phase 5: Final random restarts if time permits
    print("Phase 5: Final attempts")
    if best_p is not None:
        while time.monotonic() < deadline - 30:
            p_try = _stochastic_perturb(best_p, rng, 0.1)
            sc_try = _score(p_try)
            if sc_try < best_score:
                record(p_try, sc_try)
                alloc = min(60.0, max(10.0, deadline - time.monotonic() - 30))
                p_out, f_out = _adam_topk(p_try, alloc, top_k_init=8, top_k_max=40,
                                           seed=int(seed) + 9999)
                record(p_out, f_out)
            else:
                # Try a few more perturbations with larger sigma
                for sigma in [0.05, 0.2, 0.3]:
                    p_t2 = _stochastic_perturb(best_p, rng, sigma)
                    sc_t2 = _score(p_t2)
                    if sc_t2 < best_score:
                        record(p_t2, sc_t2)
                        break
                break  # Don't loop forever

    # Validate against global evaluator
    print("Validating against evaluator...")
    if best_p is None:
        best_p = _greedy(128)

    final_seq = _scale_output(best_p, 10.0)
    global_score = _try_global_eval(final_seq)
    print(f"  Local score: {best_score:.6f}, Global score: {global_score:.6f}")

    # Try a few different scalings
    best_global = global_score
    best_final_seq = final_seq
    for ts in [1.0, 5.0, 10.0, 50.0, 100.0]:
        seq = _scale_output(best_p, ts)
        gs = _try_global_eval(seq)
        if gs < best_global:
            best_global = gs
            best_final_seq = seq
            print(f"  Better scaling: sum={ts}, score={gs:.6f}")

    # Try other candidates too
    for n, (sc, p) in list(best_by_n.items())[:5]:
        for ts in [5.0, 50.0]:
            seq = _scale_output(p, ts)
            gs = _try_global_eval(seq)
            if gs < best_global:
                best_global = gs
                best_final_seq = seq
                print(f"  Best from n={n}, sum={ts}: {gs:.6f}")

    print(f"FINAL: best_global={best_global:.6f}, length={len(best_by_n.get(len(best_p//10*best_p.size, (1, np.ones(16)))[1]) if False else best_p)}")
    return best_final_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Entry point: return a list of non-negative floats minimizing the objective."""

        result = main_search(int(seed), float(budget_s))

        if result is None or len(result) == 0:
