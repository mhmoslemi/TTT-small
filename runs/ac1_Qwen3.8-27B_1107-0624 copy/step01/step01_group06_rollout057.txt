"""
Multi-refinement optimizer for minimizing J(a) = 2n*max(a*a)/(sum(a))^2.
Uses FFT-accelerated log-sum-exp smoothed gradients with Nesterov-accelerated Adam,
coordinate descent with closed-form updates, sensitivity-guided perturbations,
and multi-phase time allocation with periodic verification.

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def smooth_J_and_grad(a, tau):
    """Compute smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad


def run_nesterov_adam(a_init, budget_s, seed, tau_start, tau_end, lr_start, momentum=0.9):
    """Nesterov-accelerated Adam with adaptive tau. Returns (best_array, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau = tau_start
    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()
    stall_count = 0
    stall_best = best_J

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / max(budget_s, 1.0))
        tau = tau_start * (tau_end / tau_start) ** (frac ** 0.7)
        lr = lr_start * (0.5 + 0.5 * (1.0 - frac)) / (1.0 + 0.01 * t / 500.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Nesterov lookahead
        a_nest = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_nest = np.maximum(a_nest, 0.0)

        # Compute gradient at Nesterov point
        J_nest, grad_nest = smooth_J_and_grad(a_nest, tau)

        if not np.isfinite(J_nest) or not np.isfinite(grad_nest).all():
            a = np.maximum(a - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0)
        else:
            m = beta1 * m + (1.0 - beta1) * grad_nest
            v = beta2 * v + (1.0 - beta2) * grad_nest * grad_nest
            m_hat2 = m / (1.0 - beta1 ** t)
            v_hat2 = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat2 / (np.sqrt(v_hat2) + eps)
            a = np.maximum(a, 0.0)

        # Slight normalization to prevent drift
        if t % 500 == 0:
            S = np.sum(a)
            if S > 0 and S < 0.01:
                a = best_a.copy()
            elif S > 0:
                a = a * (n / S)

        # Periodic check
        if t % 200 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
                stall_count = 0
                stall_best = local_J
            else:
                stall_count += 1
                if stall_count > 8:
                    # Structured perturbation
                    cand = best_a.copy()
                    c = convolve_fft(cand)
                    max_idx = np.argmax(c)
                    # Perturb coordinates that contribute most to the peak
                    contrib = np.zeros(n)
                    lo = max(0, max_idx - n + 1)
                    hi = min(n - 1, max_idx)
                    for i in range(lo, hi + 1):
                        k_val = max_idx - i
                        if 0 <= k_val < n:
                            contrib[i] = abs(cand[k_val])
                    # Perturb top contributors
                    top_k = int(n * 0.1)
                    top_idx = np.argsort(contrib)[-top_k:]
                    noise = rng.normal(0, 0.02, top_k)
                    cand[top_idx] *= np.exp(noise)
                    cand = np.maximum(cand, 0)
                    cJ = compute_J(cand)
                    if cJ < stall_best:
                        a = cand
                        best_a = cand.copy()
                        best_J = cJ
                        stall_best = cJ
                    else:
                        a = best_a.copy()
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0

        # Verify with evaluate_sequence
        now = time.monotonic()
        if (now - start_time) % 10.0 < 0.1:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    nesterov t={t} J={val:.6f} tau={tau:.4f}")

    final_local = compute_J(best_a)
    return best_a, min(best_J, final_local)


def run_coordinate_descent(a_init, budget_s, seed):
    """Coordinate descent with closed-form updates. Returns (best_array, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)

    # Precompute FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    c = convolve_fft(a)
    S = np.sum(a)
    max_idx = np.argmax(c)

    passes = 0
    last_verify = 0.0

    while time.monotonic() < deadline:
        passes += 1

        # Build sensitivity ordering: prioritize coordinates with large a[k*-i]
        contrib = np.zeros(n)
        lo = max(0, max_idx - n + 1)
        hi = min(n - 1, max_idx)
        for i in range(lo, hi + 1):
            k_val = max_idx - i
            if 0 <= k_val < n:
                contrib[i] = a[k_val]

        # Process in order of sensitivity (most sensitive first)
        order = np.argsort(-contrib)

        for i in order:
            if time.monotonic() > deadline:
                break

            k_star = max_idx
            k_minus_i = k_star - i

            if k_minus_i < 0 or k_minus_i >= n:
                # c[k_star] doesn't depend on a[i], only S changes
                # Increase a[i] to increase S (decrease J)
                # But this might cause another k to become max
                # Conservative: skip or small increase
                continue

            S0 = S - a[i]
            C0 = c[k_star] - 2.0 * a[i] * a[k_minus_i]
            a_kmi = a[k_minus_i]

            if a_kmi < 1e-12:
                continue

            # Optimal: a[i]* = S0 - C0/a[k_minus_i]
            opt = S0 - C0 / a_kmi
            opt = max(0.0, min(1000.0, opt))

            delta = opt - a[i]
            if abs(delta) < 1e-14:
                continue

            # Update c: c[k] += 2*delta*a[k-i] for valid k
            delta_c = 2.0 * delta * a
            # c[k] for k in [0, 2n-2], the contribution from a[i] change:
            # c[k] += delta * a[k-i] + a[i+delta] * a[k-i] ... 
            # Actually: c[k] = sum_j a[j]*a[k-j]
            # d(c[k])/da[i] = a[k-i] + a[k-i] = 2*a[k-i] (as derived)
            # So c[k] += 2*delta*a[k-i]
            # k-i must be in [0, n-1], so k in [i, i+n-1]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            # a[k-i] for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
            c[k_lo:k_hi + 1] += 2.0 * delta * a[:k_hi + 1 - k_lo]

            a[i] = opt
            S += delta

            # Update max
            new_max_idx = np.argmax(c)
            if new_max_idx != max_idx:
                max_idx = new_max_idx
                # Update contrib is deferred to next pass

        # Full recompute to avoid numerical drift (every 10 passes)
        if passes % 10 == 0:
            c = convolve_fft(a)
            S = np.sum(a)
            max_idx = np.argmax(c)
            current_J = 2.0 * n * np.max(c) / (S * S)
            if current_J < best_J:
                best_J = current_J
                best_a = a.copy()

        # Verify periodically
        now = time.monotonic()
        if now - last_verify > 15.0:
            last_verify = now
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    coord_desc pass={passes} J={val:.6f}")

    final_local = compute_J(best_a)
    if final_local < best_J:
        best_J = final_local
    return best_a, best_J


def run_refined_gd(a_init, budget_s, seed, tau_values=None):
    """Multi-tau gradient descent: cycles through different tau values."""
    if tau_values is None:
        tau_values = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]

    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)
    start_time = time.monotonic()

    tau_cycle = 0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    while time.monotonic() < deadline:
        tau = tau_values[tau_cycle % len(tau_values)]
        lr = 0.005 / (1.0 + 0.005 * t / 1000.0)

        J, grad = smooth_J_and_grad(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Gradient step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)

        # Accept if it improves the smoothed objective
        J_new, _ = smooth_J_and_grad(a_new, tau)
        if J_new < J or (J_new < J + 0.01 * lr and rng.random() < 0.1):
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - 0.5 * step
            a_new2 = np.maximum(a_new2, 0.0)
            a = a_new2

        # Periodic full evaluation
        if t % 300 == 0:
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()

        # Switch tau occasionally
        if t % (len(tau_values) * 50) == 0:
            tau_cycle += 1

        # Verify
        now = time.monotonic()
        if (now - start_time) % 12.0 < 0.1:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                best_a = a.copy()
                print(f"    refined_gd t={t} tau={tau:.3f} J={val:.6f}")

    final_local = compute_J(best_a)
    return best_a, min(best_J, final_local)


def run_hill_climbing(a_init, budget_s, seed):
    """Adaptive hill climbing with structured perturbations."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)

    best_a = a.copy()
    best_J = compute_J(a)
    step_scale = 0.03
    c = convolve_fft(a)
    max_idx = int(np.argmax(c))
    start_time = time.monotonic()
    iterations = 0

    while time.monotonic() < deadline:
        iterations += 1

        # Structured perturbation: focus on coordinates contributing to peak
        width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
        start = int(rng.integers(0, max(1, n - width)))
        end = min(n, start + width)

        cand = a.copy()
        noise = rng.normal(0.0, step_scale, end - start)
        cand[start:end] *= np.exp(noise)

        # Also perturb near the peak-contributing region
        peak_lo = max(0, max_idx - n // 4)
        peak_hi = min(n, max_idx + 1 + n // 4)
        n_peak_perturb = int(rng.integers(1, 5))
        for _ in range(n_peak_perturb):
            idx = int(rng.integers(peak_lo, peak_hi))
            cand[idx] *= np.exp(rng.normal(0.0, step_scale * 0.5))

        cand = np.maximum(cand, 0.0)
        cS = np.sum(cand)
        if cS < 0.01:
            continue

        cJ = compute_J(cand)

        if cJ < best_J:
            best_J = cJ
            best_a = cand.copy()
            a = cand.copy()
            step_scale = max(0.001, step_scale * 0.97)
            c = convolve_fft(a)
            max_idx = int(np.argmax(c))
        else:
            step_scale = min(0.2, step_scale * 1.01)

        # Verify periodically
        if iterations % 5000 == 0:
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
                print(f"    hill_climb iter={iterations} J={val:.6f}")

    return best_a, best_J

        else:
            print(f"  Phase 2: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")
        print(f"  Phase 2 took {time.monotonic()-t_start:.1f}s")

    # === PHASE 3: Coordinate Descent (600s) ===
    p3_budget = min(600.0, total_deadline - time.monotonic() - 60.0)
    if p3_budget > 10.0:
        print(f"\nPhase 3: Coordinate Descent ({p3_budget:.0f}s)")
        t_start = time.monotonic()
        res_a, res_J = run_coordinate_descent(best_global, p3_budget, seed=seed + 3)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Phase 3 *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Phase 3: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")
        print(f"  Phase 3 took {time.monotonic()-t_start:.1f}s")

    # === PHASE 4: Fine Nesterov polish (200s) ===
    p4_budget = min(200.0, total_deadline - time.monotonic() - 60.0)
    if p4_budget > 10.0:
        print(f"\nPhase 4: Fine polish ({p4_budget:.0f}s)")
        t_start = time.monotonic()
        res_a, res_J = run_nesterov_adam(
            best_global, p4_budget, seed=seed + 4,
            tau_start=0.3, tau_end=0.01, lr_start=0.003
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Phase 4 *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Phase 4: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")
        print(f"  Phase 4 took {time.monotonic()-t_start:.1f}s")

    # === PHASE 5: Final coordinate descent polish ===
    p5_budget = total_deadline - time.monotonic() - 10.0
    if p5_budget > 10.0:
        print(f"\nPhase 5: Final coord descent ({p5_budget:.0f}s)")
        res_a, res_J = run_coordinate_descent(best_global, p5_budget, seed=seed + 5)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  Phase 5 *** NEW BEST: {val:.6f} ***")
        else:
            print(f"  Phase 5: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
