    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

        m1 = hi - gr * (hi - lo)

        a2 = (1 - m2) * a + m2 * g

        s2 = a2.sum()

        else:

Actually, one more idea: instead of the LP, I could use a simpler approach that might be faster:

**Greedy coordinate descent**: At each step, find the element `a[i]` that most reduces the peak of the auto-convolution, and adjust it.

This is O(n^2) per step (for n=512, that's 262144 operations per step, very fast). We can do millions of steps.

The challenge is finding which coordinate to adjust and in what direction. The subgradient tells us: `grad[i] = 2 * sum_{k in S} a[k-i]` where S is the set of peak indices. We want to decrease `a[i]` for large `grad[i]` and increase for small `grad[i]`.

But we also need to maintain `sum(a) = 1` and `a >= 0`. This is a constrained optimization problem.

**Projected gradient descent with averaging:**
1. Compute `c = conv(a,a)`, find peak set S
2. Compute `grad[i] = 2 * sum_{k in S} a[k-i]`
3. `a_new = a - lr * grad`
4. Project onto simplex: `a_proj = project_to_simplex(a_new)`
5. Use Polyak averaging for the iterate

The projection onto the simplex (`a >= 0, sum(a) = 1`) can be done efficiently in O(n log n).

This might be faster than LP because each step is O(n^2) (for the convolution) + O(n log n) (for the projection), and we can do many more steps.

- Switch to LP for fine-tuning when gradient steps become too small

Actually, I just realized something important. The function `f(a) = max_k conv(a,a)[k]` is NOT convex, so projected gradient descent might not work well. The function is the max of quadratic forms, and quadratic forms can be non-convex.

However, for our specific problem (non-negative sequences, sum = 1), the quadratic forms `conv(a,a)[k]` are actually **convex** in the region `a >= 0`! Let me verify:

`conv(a,a)[k] = sum_j a[j] * a[k-j]`

The Hessian of this with respect to `a` is: `H[i,i'] = a'δ_{i,k-i'} + δ_{i,i'}δ_{i',k-i}`... 

Actually the Hessian of `sum_j a[j]*a[k-j]` with respect to `a` is a symmetric matrix where `H[i,j] = δ_{i+j,k}` (i.e., H[i,j] = 1 if i+j=k, else 0). This is a permutation matrix (or a submatrix of one), which is indefinite (eigenvalues ±1).

So the individual quadratic forms are NOT convex. The max of non-convex functions is also not convex.

This means gradient-based methods might not converge to the global minimum. The LP approach (alternating linearization) is more robust.

"""
LP-based iterative optimization with golden-section line search,
multiple initializations, and periodic perturbations to minimize
2*n*max(conv(a,a))/(sum(a))^2. Key improvements over previous:
- Golden section line search for optimal step size
- Sparse LP matrices for speed
- Multiple M values to escape degeneracy
- Periodic random restarts
- Try multiple sequence lengths
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def compute_conv_fft(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = fft(a, fft_size)
    facc = ifft(fa * fa, fft_size)
    return facc[:size].real


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = compute_conv_fft(a)
    return 2.0 * n * c.max() / (S * S)


def build_sparse_conv_matrix(a):
    """Build sparse matrix A (2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # Column j: a[j] on row j, a[j-1] on row j-1, ..., a[j-(n-1)] on row j-(n-1)
        # Actually: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        # So for fixed j, A[k, j] = a[k-j] for k in [j, j+n-1]
        rr = np.arange(j, j + n)
        cc = np.full(n, j)
        vv = a.copy()
        rows.append(rr)
        cols.append(cc)
        vals.append(vv)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_find_direction(a, M, sparse_A=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns (g, dual_obj)."""
    n = len(a)
    if sparse_A is None:
        sparse_A = build_sparse_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=sparse_A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x, res.fun
    except Exception:
        pass
    return None, np.inf


def golden_section_search(a, g, a_best, best_score, n_evals_max=30):
    """Golden section search over alpha in [0,1] for f = (1-alpha)*a + alpha*g."""
    n = len(a)
    lo = 0.0
    hi = 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def eval_at(alpha):
        x = (1.0 - alpha) * a + alpha * g
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s < 1e-10:
            return np.inf
        x = x / s
        return compute_objective(x)

    # Evaluate at initial points
    f1 = eval_at(lo + gr * (hi - lo))
    f2 = eval_at(hi - gr * (hi - lo))
    evals = 2

    for _ in range(n_evals_max):
        if f1 < f2:
            hi = hi - gr * (hi - lo)
            hi_val = hi
            lo = lo
            f2 = f1
            f1 = eval_at(lo + gr * (hi - lo))
            evals += 1
        else:
            lo = lo + gr * (hi - lo)
            f1 = f2
            f2 = eval_at(hi - gr * (hi - lo))
            evals += 1
        if hi - lo < 1e-6:
            break

    alpha = (lo + hi) / 2.0
    x = (1.0 - alpha) * a + alpha * g
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-10:
        return a_best, best_score, 0.0
    x = x / s
    sc = compute_objective(x)
    if sc < best_score:
        return x, sc, alpha
    return a_best, best_score, 0.0


def optimize_lp(a0, deadline, rng, verbose=False):
    """LP-based optimization loop with golden section line search."""
    n = len(a0)
    a = np.clip(a0, 0.0, None).astype(np.float64).copy()
    if a.sum() < 1e-10:
        return a, np.inf
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_objective(a)
    stall = 0
    it = 0
    m_multiplier = 1.0  # Try M * multiplier for more/less constrained LP
    m_mult_cycle = [1.0, 1.0, 1.0, 1.01, 1.02, 1.05, 1.1]  # Cycle through multipliers
    m_mult_idx = 0

    if verbose:

        it += 1
        c = compute_conv_fft(a)

        # Cycle through M multipliers

        m_mult_idx += 1
        M_lp = M * m_mult

        g, _ = lp_find_direction(a, M_lp)
        if g is None or g.sum() < 1e-15:
            # Try with larger M
            g, _ = lp_find_direction(a, M * 1.5)

        g = g / g.sum()

            best_score = sc_new
            a = a_new

        else:

            # Try direct move to g

            sc_direct = compute_objective(a_direct)

                a = a_direct
                stall = 0

def make_cosine(n):
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_gaussian(n, sigma_frac=0.3):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_parabolic(n):
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_smoothstep(n, p=2.0):
    """Smoothstep-like window: (1 - |2x-1|^p)^(1/p) normalized."""
    x = np.linspace(0, 1, n)
    t = 1.0 - np.abs(2.0 * x - 1.0) ** p
    return np.maximum(0.0, t) ** (1.0 / p)
