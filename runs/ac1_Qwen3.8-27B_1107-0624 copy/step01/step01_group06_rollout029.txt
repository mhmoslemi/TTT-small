Top-k smoothed gradient optimization with Adam, multiple restarts,
and simulated annealing for escaping local minima.
Minimizes J(a) = 2n * max(conv(a,a)) / (sum(a))^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(conv)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def topk_gradient(a, k):

    Returns (J_approx, gradient).
    """
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    
    # Get top-k indices
    if k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -k)[-k:]

    J = 2.0 * n * T / (S * S)
    
    # Gradient: dJ/da[m] = (4n/(k*S^2)) * sum_{j in top_k} a[j-m] (valid) - (4n*T/S^3)

def run_optimization(a_init, budget_s, seed, k=10, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    """Run Adam optimization with top-k gradient. Returns (best_a, best_J)."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = np.maximum(a_init.copy(), 0.0)
    S = np.sum(a)
    if S > 0:
        a = a / S  # Normalize to sum=1 for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    
    stall_count = 0
    stall_threshold = best_J
    last_best_t = 0
    
    while time.monotonic() < deadline:
        # Adaptive learning rate: decrease over time
        elapsed_frac = (time.monotonic() - (deadline - budget_s)) / budget_s
        current_lr = lr / (1.0 + 0.1 * t / 1000.0)
        
        J, grad = topk_gradient(a, k)
        
        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a = a - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        
        # Periodically check actual J
        if t % 2000 == 0:
            actual_J = compute_J(a)
            if actual_J < best_J:
                best_J = actual_J
                best_a = a.copy()
                stall_count = 0
                stall_threshold = actual_J
                last_best_t = t
            else:
                if t - last_best_t > 20000:
                    stall_count += 1
                    if stall_count >= 5:
                        # Reset with perturbation
                        a = best_a.copy()
                        # Perturb a random block
                        w = max(1, n // 50)
                        s = int(rng.integers(0, n - w))
                        a[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                        a = np.maximum(a, 0)
                        a = a / max(np.sum(a), 0.01)
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 0
                        stall_count = 0
        
    # Final
    final_J = compute_J(best_a)
    return best_a, min(best_J, final_J)


def run_annealing(a_init, budget_s, seed, k=10):
    """Simulated annealing on top of a good starting point."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = np.maximum(a_init.copy(), 0.0)
    S = np.sum(a)
    if S > 0:
        a = a / S
    
    best_a = a.copy()
    best_J = compute_J(a)
    current_J = best_J
    temp = 0.1  # Initial temperature
    
    while time.monotonic() < deadline:
        # Generate a perturbation
        n_blocks = int(rng.integers(1, 4))
        cand = a.copy()
        for _ in range(n_blocks):
            w = int(rng.integers(max(1, n // 200), max(2, n // 10)))
            s = int(rng.integers(0, max(1, n - w)))
            e = min(n, s + w)
            cand[s:e] += rng.normal(0, temp, e - s)
        
        cand = np.maximum(cand, 0.0)
        cS = np.sum(cand)
        if cS < 0.01:
            continue
        cand = cand / cS  # Renormalize
        
        cand_J = compute_J(cand)
        delta = cand_J - current_J
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = cand
            current_J = cand_J
            if cand_J < best_J:
                best_J = cand_J
                best_a = cand.copy()
        
        # Cool down
        elapsed_frac = (time.monotonic() - (deadline - budget_s)) / budget_s
        temp = 0.1 * (0.01) ** elapsed_frac
    
    return best_a, best_J
