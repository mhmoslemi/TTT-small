Minimize 2*n*max(a*a)/(sum a)^2 for non-negative sequence a.
Uses smoothed subgradient descent with:
- Full softmax smoothing (not just top-L) for better gradient accuracy
- FISTA-style momentum for faster convergence
- Backtracking line search for adaptive step size
- Periodic peak-shaving for targeted improvement
- Multiple diverse seeds and coarse-to-fine strategy

import time
import math
import numpy as np


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _fft_conv_sq(a):
    """Compute linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)
    return c[:2 * n - 1]


def _proj_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum x = target}. O(n log n)."""
    n = len(v)
    if target <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.zeros(n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_gradient(c, a, tau):
    """
    Compute gradient of J_tau(a) = tau * log(sum_k exp(c_k/tau)) w.r.t. a.
    where c = a * a (linear convolution).
    
    grad_p = 2 * sum_k w_k * a_{k-p}
    where w_k = exp(c_k/tau) / sum_j exp(c_j/tau)
    
    Computed via FFT: grad = 2 * conv(w, flip(a))[n-1 : 2n-1]
    """
    n = len(a)
    # Softmax (numerically stable)
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    w_sum = exp_c.sum()
    if w_sum < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = exp_c / w_sum
    
    # conv(w, flip(a)) via FFT
    flip_a = a[::-1]
    m = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, m)
    ff = np.fft.rfft(flip_a, m)
    conv = np.fft.irfft(fw * ff, m)
    grad = 2.0 * conv[n - 1: n - 1 + n]
    return grad


def _fista_step(a, g, t, momentum_buf):
    """
    One FISTA step. Returns (a_new, momentum_buf_new, t_new).
    a: current iterate
    g: gradient
    t: FISTA counter
    momentum_buf: extrapolation buffer
    """
    # Extrapolation
    t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
    y = a + ((t - 1.0) / t_new) * (a - momentum_buf)
    # Step (caller should scale g appropriately)
    # We'll return y and let the caller do the step
    return y, a, t_new

            stall_count = 0

            # Take a small step in the negative direction
