"""
Hybrid optimization combining Nesterov gradient descent with top-K peak targeting,
FFT-accelerated convolution, adaptive learning rates, and structured restarts
to minimize J(a) = 2n * max(a*a) / (sum(a))^2.
"""

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    max_c = np.max(c)
    return 2.0 * n * max_c / (S * S)


def compute_grad_topK(a, K=5, tau=None):
    """Compute gradient using top-K peak weighting.
    
    If tau is None, use exact top-K (hard weighting).
    If tau is given, use soft weighting with temperature tau.
    """
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n), c
    
    if tau is not None:
        # Soft weighting
        c_max = np.max(c)
        p = np.exp((c - c_max) / tau)
        p /= np.sum(p)
        M = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
    else:
        # Hard top-K
        top_k = np.argsort(c)[-K:]
        p = np.zeros_like(c)
        p[top_k] = 1.0 / K
        M = np.max(c[top_k])
    
    # Cross-correlation: (p * a)[i] = sum_k p[k] * a[k-i]
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    # J = 2n * M / S^2
    # dJ/da[i] = 2n/S^2 * 2*cross[i] - 4n*M/S^3
    J = 2.0 * n * M / (S * S)
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * M) / (S * S * S)
    
    return J, grad, c

    S = np.sum(a)
    if S > 0:
        a = a * (2.0 * n / np.sqrt(2.0 * n))  # normalize so S = sqrt(2n), J = 2*max(c)
        # Actually let's just normalize S = n for convenience
        a = a_init.copy()
        S = np.sum(a)
        if S > 0:
            a = a * (n / S)
    
    v = np.zeros(n)  # velocity for Nesterov
    best_a = a.copy()
    best_J = compute_J_fft(a)

    last_print = 0.0

            v = np.zeros(n)
            continue

        v = momentum * v - lr * grad

def peak_coordinate_descent(a_init, budget_s, seed):
    """Coordinate descent targeting the maximum peak."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J_fft(a)
    
    start_time = time.monotonic()
    last_print = 0.0
    lr = 0.05
    
    while time.monotonic() < deadline:
        c = convolve_fft(a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        S = np.sum(a)
        
        # Gradient of c[k_star] w.r.t. a[i] is 2*a[k_star - i] (if valid)
        # So decreasing a[i] where a[k_star-i] is large will reduce c[k_star]
        # We want to minimize max(c), so reduce c[k_star]
        
        # Compute the direction that reduces c[k_star]
        # d(c[k_star])/d(a[i]) = 2 * a[k_star - i] for 0 <= k_star - i < n
        # We want to move in the direction that reduces c[k_star]:
        # delta_a[i] = -lr * 2 * a[k_star - i] (for valid i)
        
        # But we also need to check other peaks don't increase too much
        # Simple approach: just do the step and check
        
        k_shift = k_star
        # The gradient vector for c[k*]
        if k_shift < n:
            grad_c = np.zeros(n)
            grad_c[:n - k_shift] = 2.0 * a[k_shift:n]  # a[k_star - i] for i = 0..n-k_star-1
            # Wait, d(c[k])/d(a[i]) = 2*a[k-i] for 0 <= k-i < n
            # So for fixed k = k_star: d(c[k*])/d(a[i]) = 2*a[k*-i] for 0 <= k*-i < n
            # i.e., i must satisfy 0 <= i < n and 0 <= k*-i < n
            # i.e., max(0, k*-(n-1)) <= i <= min(n-1, k*)
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            for i in range(lo, hi + 1):
                grad_c[i] = 2.0 * a[k_star - i]
        else:
            # k_star >= n, so k_star - i >= n - i >= 1 for i < n
            # d(c[k])/d(a[i]) = 2*a[k-i] for 0 <= k-i < n
            # k-i < n => i > k-n
            # k-i >= 0 => i <= k (always true since i < n <= k... wait k can be up to 2n-2)
            lo = max(0, k_star - (n - 1))
            hi = min(n - 1, k_star)
            if lo > hi:
                grad_c = np.zeros(n)
            else:
                grad_c = np.zeros(n)
                for i in range(lo, hi + 1):
                    grad_c[i] = 2.0 * a[k_star - i]
        
        # Full gradient of J
        # dJ/da[i] = 2n/S^2 * grad_c[i] - 4n*M/S^3
        grad_J = (2.0 * n / (S * S)) * grad_c - (4.0 * n * M) / (S * S * S)
        
        # Nesterov-like step
        a_candidate = a - lr * grad_J
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        
        J_candidate = compute_J_fft(a_candidate)
        
        if J_candidate < best_J:
            best_J = J_candidate
            best_a = a_candidate.copy()
            a = a_candidate
            lr = min(lr * 1.05, 0.5)  # slightly increase lr on success
        else:
            lr = max(lr * 0.7, 0.001)  # decrease lr on failure
            a = best_a  # revert
        
        if time.monotonic() - last_print > 10.0:
            last_print = time.monotonic()
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            print(f"    peak_cd: elapsed={time.monotonic()-start_time:.0f}s, J={best_J:.6f}, lr={lr:.4f}")
    
    return best_a, best_J


def simulated_annealing(a_init, budget_s, seed):
    """Simulated annealing with structured perturbations."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J_fft(a)
    
    start_time = time.monotonic()
    T = 1.0  # temperature
    T_min = 0.001
    cooling = 0.999
    
    last_print = 0.0
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        T = max(T_min, T * cooling)
        
        # Generate perturbation
        r = rng.random()
        if r < 0.3:
            # Sine wave perturbation
            freq = rng.uniform(0.5, 5.0)
            amp = T * 0.5
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(n)
            perturbation = amp * np.sin(2 * np.pi * freq * x / n + phase)
            a_new = a + perturbation
        elif r < 0.6:
            # Random block perturbation
            width = int(rng.integers(max(10, n // 100), n // 10))
            start = int(rng.integers(0, n - width))
            a_new = a.copy()
            a_new[start:start+width] += rng.normal(0, T * 0.3, width)
        elif r < 0.8:
            # Cosine window perturbation
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 50, n // 5))
            a_new = a.copy()
            for i in range(max(0, center - width), min(n, center + width)):
                dist = abs(i - center) / width
                a_new[i] += T * 0.5 * np.cos(dist * np.pi) * rng.normal(0, 1)
        else:
            # Single element perturbation
            idx = int(rng.integers(0, n))
            a_new = a.copy()
            a_new[idx] *= np.exp(rng.normal(0, T))
        
        a_new = np.clip(a_new, 0.0, 1000.0)
        J_new = compute_J_fft(a_new)
        
        # Metropolis acceptance
        if J_new < best_J or rng.random() < np.exp(-(J_new - best_J) / max(T, 1e-10)):
            a = a_new
            if J_new < best_J:
                best_J = J_new
                best_a = a_new.copy()
        
        if time.monotonic() - last_print > 15.0:
            last_print = time.monotonic()
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            print(f"    SA: elapsed={elapsed:.0f}s, J={best_J:.6f}, T={T:.4f}")
    
    return best_a, best_J


def smooth_J_and_grad_v2(a, tau):
    """Improved smooth objective and gradient with better numerical stability."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]

    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)

    # Log-sum-exp with better numerical handling
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp

    J = 2.0 * n * Tmax / (S * S)

    # Cross-correlation
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)

    return J, grad

def cosine_annealing_lr(base_lr, t, T_max):

Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Combines Adam gradient descent with adaptive log-sum-exp smoothing, top-K peak
targeting, structured local search, and fine-tuning polish. Uses FFT for
fast convolution. Key improvements over baseline: smaller initial tau for
sharper gradients, cosine-annealed learning rates, peak-deflation line search,

import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J_fft(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT convolution."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S * S)


def compute_J_grad_smooth(a, tau):
    """Compute smoothed J and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 3 * n - 2:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)

    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def compute_J_grad_topK(a, K):

def normalize_sum(a, target):
    """Rescale a so that sum(a) = target."""
    S = np.sum(a)
    if S > 1e-10:
        return a * (target / S)
    return a


def phase_adam(a_init, budget_s, seed, tau_start=0.4, tau_end=0.003,
               lr_start=0.03, topK_frac=0.65, topK_K=3):
    """Adam gradient descent with adaptive tau and top-K switching."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    a = normalize_sum(a, float(n))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J_fft(a)

    start_time = time.monotonic()
    last_verify = 0.0
    stall_count = 0
    stall_thresh = best_J
    T_max = max(1, int(budget_s * 30))

    while time.monotonic() < deadline:

        frac = min(1.0, elapsed / max(budget_s, 0.1))

        use_topK = frac > topK_frac
        if use_topK:
            J, grad = compute_J_grad_topK(a, topK_K)
        else:

            J, grad = compute_J_grad_smooth(a, tau)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            continue

        t += 1
        # Cosine annealing
        lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * min(t, T_max) / T_max))

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if t % 200 == 0:
            local_J = compute_J_fft(a)
            if local_J < best_J:

                stall_thresh = local_J

                    # Perturbation restart
                    a = best_a.copy()
                    w = max(5, int(n // 30))
                    s = int(rng.integers(0, n - w))
                    a[s:s + w] *= np.exp(rng.normal(0, 0.15, w))
                    a = np.clip(a, 0, 1000)
                    m = np.zeros(n)
                    v = np.zeros(n)
                    t = 0
                    stall_count = 0

def phase_peak_deflation(a_init, budget_s, seed):
    """Peak deflation: iteratively find the worst peak and take a targeted step."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    a = normalize_sum(a, float(n))
    best_a = a.copy()
    best_J = compute_J_fft(a)
    lr = 0.02

    start_time = time.monotonic()
    last_verify = 0.0
    count = 0

    while time.monotonic() < deadline:
        c = convolve_fft(a)
        S = np.sum(a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Gradient of c[k*] w.r.t. a[i] = 2*a[k*-i] for valid k*-i
        # Use FFT to compute the cross-correlation with a delta at k*
        size = 1
        while size < n + 2 * n - 1:
            size <<= 1
        p = np.zeros(2 * n - 1)
        p[k_star] = 1.0
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p, size)
        cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]

        # Gradient of J
        grad_J = (4.0 * n * cross) / (S * S) - (4.0 * n * M) / (S * S * S)

        # Line search: try a few step sizes
        best_step_J = compute_J_fft(a)
        best_step_a = a
        for lr_try in [lr, lr * 0.5, lr * 2.0]:

            a_try = np.clip(a_try, 0.0, 1000.0)

                best_step_J = J_try
                best_step_a = a_try

        if best_step_J < best_J:
            best_J = best_step_J
            best_a = best_step_a.copy()
            a = best_step_a
            lr = min(lr * 1.08, 0.3)
        else:
            lr = max(lr * 0.7, 0.001)
            a = best_a

        count += 1
        now = time.monotonic()
        if now - last_verify > 12.0:
            last_verify = now
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            if (now - start_time) % 50 < 12:
                print(f"      peak_def: it={count}, J={best_J:.6f}, lr={lr:.5f}, k*={k_star}")

    return best_a, best_J


def phase_local_search(a_init, budget_s, seed):

            hi = min(n, center + width)
            for i in range(lo, hi):
                d = abs(i - center) / width
                a_new[i] += amp * rng.normal(0, 1) * np.cos(d * np.pi / 2) ** 2
        else:
            # Multiplicative perturbation on a few elements
            n_elem = int(rng.integers(5, 50))
            idx = rng.choice(n, size=n_elem, replace=False)
            a_new[idx] *= np.exp(amp * 0.5 * rng.normal(0, 1, n_elem))

        a_new = np.clip(a_new, 0.0, 1000.0)
        J_new = compute_J_fft(a_new)

        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            amp = max(0.005, amp * 0.97)
        else:
            amp = min(0.5, amp * 1.03)

        count += 1
        now = time.monotonic()
        if now - last_verify > 15.0:
            last_verify = now
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            if (now - start_time) % 60 < 15:
                print(f"      local_srch: it={count}, J={best_J:.6f}, amp={amp:.4f}")

    return best_a, best_J


def phase_fine_tune(a_init, budget_s, seed):
    """Very fine gradient descent with top-1 targeting and tiny steps."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    a = a_init.copy()
    a = normalize_sum(a, float(n))
    best_a = a.copy()
    best_J = compute_J_fft(a)
    lr = 0.005
    momentum = 0.95
    v = np.zeros(n)

    start_time = time.monotonic()
    last_verify = 0.0
    count = 0

    while time.monotonic() < deadline:
        J, grad = compute_J_grad_topK(a, 1)

        if not np.isfinite(J) or not np.isfinite(grad).all():
            a = best_a.copy()
            v = np.zeros(n)
            continue

        # Momentum step
        v = momentum * v - lr * grad
        a_new = np.clip(a + v, 0.0, 1000.0)

        J_new = compute_J_fft(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.02, 0.02)
        else:
            a = best_a
            v = v * 0.5
            lr = max(lr * 0.8, 0.0005)

        count += 1
        now = time.monotonic()
        if now - last_verify > 10.0:
            last_verify = now
            val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(val) and val < best_J:
                best_J = val
            if (now - start_time) % 40 < 10:
                print(f"      fine_tune: it={count}, J={best_J:.6f}, lr={lr:.6f}")

    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1 (40%): Adam GD with adaptive tau (0.4 to 0.003), cosine LR
    Phase 2 (25%): Peak deflation with line search
    Phase 3 (20%): Structured local search (sine, block, cosine perturbations)
    Phase 4 (15%): Fine-tuning with top-1 targeting and momentum

    Uses the parent sequence as primary initialization and explores
    structured alternatives for diversity.
    """
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad parent")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"n = {n}")

    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")

    # Build initializations
    inits = [("parent", parent.copy())]

    # A few structured inits for diversity
    x = np.arange(n)
    mid = (n - 1) / 2.0

    # Smooth bump
    gauss = np.exp(-((x - mid) ** 2) / (2.0 * (n * 0.3) ** 2))
    inits.append(("gauss", gauss))

    # Cosine window
    cos_win = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / n)
    inits.append(("cos_win", np.maximum(cos_win, 0.05)))

    # Score and sort
    scored = []
    for name, arr in inits:
        arr = normalize_sum(arr, float(n))
        J = compute_J_fft(arr)
        scored.append((name, J, arr))
    scored.sort(key=lambda z: z[1])
    for name, J, _ in scored:
        print(f"  init '{name}': J = {J:.6f}")

    # Time allocation
    alloc = max(1.0, (total_deadline - time.monotonic()) - 15.0)
    t1 = alloc * 0.40
    t2 = alloc * 0.25
    t3 = alloc * 0.20
    t4 = alloc * 0.15

    # Phase 1: Adam gradient descent
    print(f"\n=== Phase 1: Adam GD ({t1:.0f}s) ===")
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 5.0:
        res_a, res_J = phase_adam(
            best_global, min(t1, remaining),
            seed=seed, tau_start=0.4, tau_end=0.003, lr_start=0.03
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST after Phase 1: {val:.6f} ***")
        else:
            print(f"  Phase 1 done: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # Phase 2: Peak deflation
    print(f"\n=== Phase 2: Peak Deflation ({t2:.0f}s) ===")
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 5.0:
        res_a, res_J = phase_peak_deflation(
            best_global, min(t2, remaining), seed=seed + 1000
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST after Phase 2: {val:.6f} ***")
        else:
            print(f"  Phase 2 done: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # Phase 3: Local search
    print(f"\n=== Phase 3: Local Search ({t3:.0f}s) ===")
    remaining = total_deadline - time.monotonic() - 15.0
    if remaining > 5.0:
        res_a, res_J = phase_local_search(
            best_global, min(t3, remaining), seed=seed + 2000
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST after Phase 3: {val:.6f} ***")
        else:
            print(f"  Phase 3 done: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # Phase 4: Fine tune
    print(f"\n=== Phase 4: Fine Tune ({t4:.0f}s) ===")
    remaining = total_deadline - time.monotonic() - 8.0
    if remaining > 5.0:
        res_a, res_J = phase_fine_tune(
            best_global, remaining, seed=seed + 3000
        )
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST after Phase 4: {val:.6f} ***")
        else:
            print(f"  Phase 4 done: local={res_J:.6f}, verified={val:.6f}, best={best_global_score:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\nFinal score: {best_global_score:.6f}")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
