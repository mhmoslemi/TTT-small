    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # Bisection for threshold t such that sum(max(v-t, 0)) = 1
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(v - mid, 0.0)) > 1.0:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.maximum(v - t, 0.0)
    result /= np.sum(result)  # Normalize to sum exactly 1
    return result

def compute_gradient_smoothed(s, tau):
    """Compute gradient of log-sum-exp approximation of max(s*s)."""
    n = len(s)
    conv = np.convolve(s, s)
    max_c = np.max(conv)
    exp_c = np.exp(tau * (conv - max_c))
    w = exp_c / np.sum(exp_c)
    # g_j = 2 * sum_k w[k] * s[k-j]
    g = np.zeros(n)
    for j in range(n):
        k_lo = j
        k_hi = min(2*n-1, j+n)  # exclusive
        if k_hi > k_lo:
            g[j] = 2.0 * np.dot(w[k_lo:k_hi], s[0:k_hi-k_lo])
    return g

def subgradient_descent(s, tau, eta, n_iters, deadline, momentum=True):
    """Subgradient/FISTA method with log-sum-exp smoothing."""
    s = s.copy()
    best = s.copy()
    best_sc = fast_score(s)

        y = s.copy()
        t = 1.0
        s_prev = s.copy()

            break

            y = s + t_new * (s - s_prev)
            y = np.maximum(y, 0.0)
            y = y / np.sum(y)
            grad = compute_gradient_smoothed(y, tau)

        if sc < best_sc:
            best_sc = sc
            best = s.copy()
    
    return best, best_sc
