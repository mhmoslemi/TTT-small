from scipy.optimize import linprog

def lp_step(a, M):
    """Find g that maximizes sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    # conv(a,g)_k = sum_j a_j * g_{k-j} for valid j, k-j
    # This is a linear function of g
    # We need: for each k = 0..2n-2, sum_{j: 0<=j<n, 0<=k-j<n} a_j * g_{k-j} <= M
    # Plus g_i >= 0
    
    c = -np.ones(n)  # minimize -sum(g)
    
    # Build constraint matrix A such that A @ g <= M
    # A[k, j] = a_{k-j} if 0 <= k-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

"""
Hybrid LP + Gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

The LP-based spreading step is the key innovation: at each iteration, we solve
a linear program to find the optimal direction to spread the sequence mass
while respecting the current convolution bound. This is more targeted than
pure gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    """Full linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = c.min()
    if mn < 0:
        c = c - mn
    return c


def _score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M, method='highs'):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g or None if infeasible.
    """
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, 
                  bounds=[(0, None)] * n, method=method)
    if res.success:
        return res.x
    return None

def _smooth_max_weight(c, beta):
    """Compute softmax weights for smooth max."""
    c = np.asarray(c, dtype=np.float64)
    mval = float(c.max())
    x = (c - mval) / beta
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0:
        ex = np.zeros_like(c)
        ex[int(np.argmax(c))] = 1.0
        s = 1.0
    w = ex / s
    return mval + beta * np.log(s), w


def _gradient_step(a, beta):
    """Compute gradient of smooth-max of conv(a,a) w.r.t. a."""
    n = len(a)
    c = _fft_convolve(a)
    _, w = _smooth_max_weight(c, beta)
    
    # Gradient: 2 * h where h[j] = sum_m a_m * w[m+j]
    # h = cross-correlation of a and w
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # We want h[j] = sum_m a[m] * w[m+j]
    # This is the correlation of a with w, which is:
    # corr[j] = sum_m a[m] * w[m+j]
    # In FFT: corr = ifft(conj(result of conv(a, reversed_w)))
    # Actually: let's use the formula from the previous code
    # h[i] = sum_j w[i+j] * p[j]
    fp = np.fft.rfft(a[::-1], N)
    h_full = np.fft.irfft(fw * fp, N)
    h = h_full[n - 1:2 * n - 1]
    
    grad = 2.0 * h
    return grad, float(c.max())


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

"""
Hybrid LP + Gradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.

Key innovation: LP-based spreading - at each iteration, solve a linear program
to find the optimal direction to spread sequence mass while respecting the
current convolution bound. Combined with smooth-max gradient polish for final
refinement.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([])
    if n <= 128:
        return np.convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    mn = float(c.min())
    if mn < 0.0:
        c = c - mn
    return c


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    s = float(a.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _make_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _lp_spread(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    A = _make_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

def _smooth_max_weight(c, beta):
    c = np.asarray(c, dtype=np.float64)

        ex[int(np.argmax(c))] = 1.0
        s = 1.0

    val = mval + beta * float(np.log(s))
    return val, w


def _gradient_of_smooth_max(a, beta):

    return grad, float(c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_idx = rho[-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s

    return p / s

            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
            print(f"  Init score={sc0:.8f} -> LP={sc_lp:.10f}")

    # Phase 3: Logit-space Adam polish on best
    if best_seq is not None and time.monotonic() < deadline - 30:
        ad_budget = min(200.0, deadline - time.monotonic() - 30.0)
        if ad_budget > 10:
            ad_end = t_start + (time.monotonic() - t_start) + ad_budget
            print(f"Phase 3: Logit Adam polish, budget={ad_budget:.0f}s")
            a_ad = _logit_adam_polish(best_seq, ad_end, max_iter=20000, lr=0.05)
            sc_ad = _score(a_ad)
            if np.isfinite(sc_ad) and sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
            print(f"  Adam result: score={sc_ad:.10f}")

    # Phase 4: Final gradient polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        gp_budget = min(100.0, deadline - time.monotonic() - 15.0)
        if gp_budget > 5:
            gp_end = t_start + (time.monotonic() - t_start) + gp_budget
            a_final = _gradient_polish(best_seq, gp_end, max_iter=15000)
            sc_final = _score(a_final)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
            print(f"Phase 4: Final polish score={sc_final:.10f}")

    # Phase 5: One more LP pass on the very best
    if best_seq is not None and time.monotonic() < deadline - 10:
        lp_budget = min(60.0, deadline - time.monotonic() - 10.0)
        if lp_budget > 5:
            lp_end = t_start + (time.monotonic() - t_start) + lp_budget
            a_last = _lp_refine(best_seq, lp_end, max_iter=5000, inflate=1.0005, blend=0.2)
            sc_last = _score(a_last)
            if np.isfinite(sc_last) and sc_last < best_score:
                best_score = sc_last
                best_seq = a_last.copy()
            print(f"Phase 5: Final LP score={sc_last:.10f}")

    # Final validation and cleanup
    if best_seq is None:
        best_seq = _make_linear_profile(256, 1.5)
        best_score = _score(best_seq)

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s > 0.01:
        final_seq = final_seq / s
    else:
        final_seq = _make_linear_profile(256, 1.5)

    # Evaluate with the authoritative function
    if final_seq.size <= 4096:
        try:
            sc_eval = float(evaluate_sequence(final_seq.tolist()))
            if np.isfinite(sc_eval):
                print(f"Evaluate_sequence score: {sc_eval:.10f}")
                if sc_eval < best_score:
                    best_score = sc_eval
        except Exception:
            pass

    print(f"FINAL: score={best_score:.10f}, n={final_seq.size}")
    return [float(x) for x in final_seq]
