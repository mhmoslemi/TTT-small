    return c

def _score_p(p, n):
    c = _self_conv(p, n)
    return 2.0 * float(n) * float(np.max(c))

def _grad_conv_peak(p, w, n):
    """Approximate gradient: 2 * conv(p_reversed, w)[n-1 : 2n-1]"""
    fs = _fft_pow2(3 * n - 2)
    fp_rev = np.fft.rfft(p[::-1], fs)
    fw = np.fft.rfft(w, fs)
    conv = np.fft.irfft(fp_rev * fw, fs).real
    return 2.0 * conv[n - 1:2 * n - 1]

def _update_best(p, score, n):
    global _best_score, _best_p, _best_n
    if np.isfinite(score) and score < _best_score:
        _best_score = float(score)
        _best_p = np.array(p, copy=True)
        _best_n = n
        print(f"  New best: {score:.8f} (n={n})")

def _obj_grad_softmax(x, n, alpha, k_peaks):
    """Softmax surrogate objective and gradient w.r.t. logits z."""
    _check_time()
    p = _probs_from_z(x, n)
    c = _self_conv(p, n)
    cmax = float(np.max(c))
    hard = 2.0 * float(n) * cmax
    _update_best(p, hard, n)
    
    # Softmax weights
    u = alpha * (c - cmax)
    e = np.exp(u)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient w.r.t. p
    gp = _grad_conv_peak(p, w, n)
    
    # Gradient w.r.t. z
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    
    # Objective value (the surrogate)
    J = cmax + np.log(s) / alpha
    return float(J), gz

def _obj_grad_powermean(x, n, p_power, k_peaks):
    """Power mean surrogate."""
    _check_time()
    p = _probs_from_z(x, n)
    c = _self_conv(p, n)
    cmax = float(np.max(c))
    hard = 2.0 * float(n) * cmax
    _update_best(p, hard, n)
    
    # Power mean weights: w_k = c_k^{p-1} / sum c_j^{p-1}
    pp = p_power - 1
    c_pos = np.maximum(c, 1e-30)
    wp = c_pos ** pp
    sw = float(np.sum(wp))
    if sw <= 0 or not np.isfinite(sw):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = wp / sw
    
    gp = _grad_conv_peak(p, w, n)
    sdot = float(np.dot(gp, p))
    gz = p[1:] * (gp[1:] - sdot)
    gz = np.where(np.isfinite(gz), gz, 0.0)
    gz = np.clip(gz, -1e6, 1e6)
    
    # Power mean value
    J = float(sw) ** (1.0 / p_power)
    return J, gz

def _optimize_lbfgs(z0, n, alpha, max_fun, deadline):
    global _deadline
    _deadline = deadline
    bounds = [(-40.0, 40.0)] * (n - 1)
    max_iter = max(50, max_fun // 2)
    try:
        minimize(
            _obj_grad_softmax,
            np.array(z0, dtype=np.float64),
            args=(n, alpha),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "maxfun": max_fun,
                "maxiter": max_iter,
                "ftol": 1e-14,
                "gtol": 1e-10,
                "maxls": 50,
                "disp": False,
            },
        )
    except _Timeout:
        pass
    except Exception:
        pass

def _polish_subgrad(p0, n, deadline, rng, topk=16):
    """Active-set subgradient polishing."""
    global _best_score, _best_p, _best_n
    p = np.array(p0, copy=True)
    score = _score_p(p, n)
    _update_best(p, score, n)
    
    noise_scale = 0.1
    fails = 0
    step_mult = 1.0
    
    while time.monotonic() < deadline:
        _check_time_local(deadline)
        c = _self_conv(p, n)
        cmax = float(np.max(c))
        
        # Select top-k peaks with soft weighting
        idx = np.argpartition(c, -topk)[-topk:]
        w = np.zeros(2 * n - 1)
        vals = c[idx]
        # Use exponential weighting relative to max
        wv = np.exp(50.0 * (vals - cmax))
        w[idx] = wv / float(np.sum(wv))
        
        gp = _grad_conv_peak(p, w, n)
        sdot = float(np.dot(gp, p))
        gz = p[1:] * (gp[1:] - sdot)
        gnorm = float(np.linalg.norm(gz))
        
        accepted = False
        if gnorm > 1e-15:
            direction = gz / gnorm
            # Try multiple step sizes
            base_step = 0.5 * step_mult
            for step in (base_step, base_step*0.5, base_step*0.25, base_step*0.125, base_step*0.0625):
                # Move in logit space
                z = _p_to_z(p, n)
                zt = z - step * direction
                zt = np.clip(np.where(np.isfinite(zt), zt, 0.0), -40.0, 40.0)
                pt = _probs_from_z(zt, n)
                st = _score_p(pt, n)
                if st < score - 1e-14:
                    p = pt
                    score = st
                    _update_best(p, score, n)
                    accepted = True
                    step_mult = min(2.0, step_mult * 1.1)
                    break
        
        if not accepted:
            # Random perturbation
            best_local = score
            best_local_p = p
            for _ in range(5):
                z = _p_to_z(p, n)
                zt = z + rng.normal(0, noise_scale, n - 1)
                zt = np.clip(np.where(np.isfinite(zt), zt, 0.0), -40.0, 40.0)
                pt = _probs_from_z(zt, n)
                st = _score_p(pt, n)
                if st < best_local - 1e-14:
                    best_local = st
                    best_local_p = pt
            if best_local < score - 1e-14:
                p = best_local_p
                score = best_local
                _update_best(p, score, n)
                accepted = True
        
        if accepted:
            fails = 0
        else:
            fails += 1
            step_mult = max(0.1, step_mult * 0.8)
            noise_scale = max(0.01, noise_scale * 0.9)
            if fails > 200:
                break

def _check_time_local(deadline):
    if time.monotonic() >= deadline:
        return False
    return True

def _p_to_z(p, n):
    p = np.maximum(p, 1e-30)
    logf = np.log(p)
    z = logf[1:] - logf[0]
    return np.clip(z, -40.0, 40.0)

def _structured_z(n, rng):
    """Generate structured initial candidates."""
    x = (np.arange(n) + 0.5) / float(n)
    candidates = []
    
    # Logistic functions
    for beta in (1, 2, 3, 4, 5, 7, 10):
        f = 2.0 / (1.0 + np.exp(-beta * (x - 0.5)))
        f = np.maximum(f, 1e-12)
        candidates.append(_p_to_z(f / f.sum(), n))
        f2 = 2.0 / (1.0 + np.exp(beta * (x - 0.5)))
        f2 = np.maximum(f2, 1e-12)
        candidates.append(_p_to_z(f2 / f2.sum(), n))
    
    # Power functions
    for a in (0.5, 0.75, 1.5, 2, 3, 4):
        f = (x + 1e-6) ** a
        candidates.append(_p_to_z(f / f.sum(), n))
        f = (1 - x + 1e-6) ** a
        candidates.append(_p_to_z(f / f.sum(), n))
    
    # Gaussian bumps
    for mu in (0.3, 0.4, 0.6, 0.7):
        for sig in (0.1, 0.15, 0.2):
            f = np.exp(-0.5 * ((x - mu) / sig) ** 2)
            candidates.append(_p_to_z(f / f.sum(), n))
    
    # Bimodal
    for mu1 in (0.2, 0.3):
        for mu2 in (0.7, 0.8):
            f = np.exp(-0.5 * ((x - mu1) / 0.1) ** 2) + np.exp(-0.5 * ((x - mu2) / 0.1) ** 2)
            candidates.append(_p_to_z(f / f.sum(), n))
    
    return candidates

def _random_z(n, rng):
    """Generate a random initial candidate."""
    x = (np.arange(n) + 0.5) / float(n)
    t = x - 0.5
    z = np.zeros(n - 1)
    z += rng.uniform(-3, 3) * t
    z += rng.uniform(-2, 2) * t ** 2
    for freq in (1, 2, 3, 4, 5, 7):
        amp = rng.normal(0, 2.0 / (freq + 0.5))
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.sin(2 * np.pi * freq * x[1:] + ph)
    return np.clip(z, -40, 40)
