def normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def score(a):

    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (a.sum()**2)

    # Assume sum(a) = 1
    c = np.convolve(a, a)
    return 2.0 * a.size * c.max()

def build_lp_matrix(a):
    n = a.size
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # Column j: A[k, j] = a[k-j] if 0 <= k-j < n
        lo = j
        hi = min(m, j + n)
        A[lo:hi, j] = a[:hi-lo]
    return A

def solve_lp(a, M):

        return None
    n = a.size
    if M <= 1e-300:
        return None
    A = build_lp_matrix(a) / M
    c_obj = -np.ones(n)
    b_ub = np.ones(A.shape[0])
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.status == 0:
            g = np.clip(res.x, 0, None)
            if g.sum() > 1.0 + 1e-8:
                return g
    except:
        pass
    return None

def subgradient_step(a, c, momentum_buf=None, alpha=0.1):
    n = a.size
    k_star = int(np.argmax(c))
    # Gradient of c[k_star] w.r.t. a[j] = a[k_star - j] + a[j]
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= k_star - j < n:
            val += a[k_star - j]
        val += a[j]  # j is always in [0, n-1]
        g[j] = val
    # Step

    momentum_buf = 0.9 * momentum_buf + 0.1 * g

"""
Minimize 2n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences a.
Uses LP-based direction finding with all constraints, subgradient fallback,
and adaptive random restarts.
"""

import time
try:
    from scipy.optimize import linprog
except:
    linprog = None

def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).flatten()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.shape, 1.0/a.size)
    return a / s

def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (s*s)

def compute_score_norm(a):
    """Score assuming sum(a)=1."""
    n = a.size
    c = np.convolve(a, a)
    return 2.0 * n * c.max()

def build_full_lp_matrix(a):
    """Build the (2n-1) x n matrix for the LP."""
    n = a.size
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        lo = j
        hi = min(m, j + n)
        A[lo:hi, j] = a[:hi-lo]
    return A

def solve_lp_full(a, M):
    """Solve LP with all constraints. Returns g or None."""
    if linprog is None or M <= 1e-300:
        return None
    n = a.size
    A = build_full_lp_matrix(a)
    # Scale by M
    A /= M
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2*n-1, dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1.0 + 1e-6:
                return g
    except:
        pass
    return None

def subgradient_direction(a, k_star):
    """Compute the subgradient of c[k_star] w.r.t. a."""
    n = a.size
    g = a.copy()
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    for j in range(lo, hi):
        g[j] += a[k_star - j]
    return g

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def resize(a, N):
    """Resize sequence to length N."""
    a = np.asarray(a, dtype=np.float64)
    m = a.size
    if m == N:
        return normalize_nonneg(a)
    if m == 0:
        return np.full(N, 1.0/N)
    x_old = np.linspace(0, 1, m)
    x_new = np.linspace(0, 1, N)
    out = np.interp(x_new, x_old, a)
    return normalize_nonneg(np.maximum(out, 0))

def make_random_init(N, rng, kind=0):
    """Create a random initialization of length N."""
    if kind == 0:  # Uniform
        return np.full(N, 1.0/N)
    elif kind == 1:  # Two-block
        a = np.zeros(N)
        split = N // 2
        a[:split] = (2.0/3.0) / split
        a[split:] = (1.0/3.0) / (N - split)
        return normalize_nonneg(a)
    elif kind == 2:  # Power law
        x = np.linspace(0.01, 0.99, N)
        alpha = rng.uniform(-2, 2)
        beta = rng.uniform(-2, 2)
        a = x**max(0,alpha) * (1-x)**max(0,beta)
        if a.max() < 1e-10:
            a = x**max(0,alpha)
        return normalize_nonneg(np.maximum(a, 0))
    elif kind == 3:  # Bimodal (mass at edges)
        x = np.linspace(0, 1, N)
        sigma = rng.uniform(0.05, 0.2)
        a = np.exp(-((x-0)/sigma)**2) + np.exp(-((x-1)/sigma)**2)
        return normalize_nonneg(a)
    elif kind == 4:  # Sine
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        a = 1 + 0.8*np.sin(2*np.pi*freq*x + phase)
        return normalize_nonneg(np.maximum(a, 0))
    else:  # Random
        return normalize_nonneg(rng.random(N))

    best_score = compute_score_norm(a)

    update_global(state, best_a, best_score)

        M = c.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_score - 1e-10:
