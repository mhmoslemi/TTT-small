"""
Optimize non-negative step-function sequence to minimize
f(seq) = 2*n*max(convolve(seq,seq))/sum(seq)^2.

Strategy:
1. Multi-length initialization with greedy/power-based starts.
2. Adam gradient descent with smoothed max objective.
3. LP-based direction finding with cutting planes.
4. Pair-coordinate local optimization (exploiting linearity of convolution).
5. Final polishing.

import time
import numpy as np

    import scipy.optimize as _scipy_opt
except Exception:
    _scipy_opt = None


def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s


def greedy_normalized(n):
    n = int(n)
    if n <= 0:
        n = 16
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return normalize_nonneg(a)


def power_normalized(n, s):
    n = int(n)
    if n <= 0:
        n = 16
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(idx, -float(s))
    return normalize_nonneg(a)


def actual_score_fft(p):
    a = np.asarray(p, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-10:
        return np.inf
    if n <= 512:
        c = np.convolve(a, a)
    else:
        L = next_power_of_two(2 * n - 1)
        P = np.fft.rfft(a, L)
        c = np.fft.irfft(P * P, L)[:2 * n - 1].real
    c = np.maximum(c, 0.0)
    maxc = float(np.max(c))
    if not np.isfinite(maxc):
        return np.inf
    return float(2.0 * n * maxc / (s * s))


def scale_output(p, target_sum=10.0):
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(16, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    b = a * (float(target_sum) / s)
    b = np.clip(b, 0.0, 1000.0)
    bs = float(b.sum())
    if bs < 0.01:
        b = b * (0.01 / bs)
    return [float(x) for x in b]


def adam_optimize(p, budget_s, seed=0):
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return p, np.inf
    p = normalize_nonneg(p)
    budget_s = float(budget_s)
    if budget_s <= 1.0:
        return p, actual_score_fft(p)

    x = np.log(p + 1e-300)
    x -= x.mean()

    if n < 64:
        lr = 0.05
    elif n < 256:
        lr = 0.04
    elif n < 1024:
        lr = 0.025
    else:
        lr = 0.018

    mu, nu, eps = 0.9, 0.999, 1e-8
    m_vec = np.zeros(n, dtype=np.float64)
    v_vec = np.zeros(n, dtype=np.float64)
    t = 0

    L = next_power_of_two(2 * n - 1)
    best_p = p.copy()
    best_f = actual_score_fft(p)

    start = time.monotonic()
    end = start + max(0.0, budget_s - 1.0)

    while time.monotonic() < end:
        t += 1
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(1e-9, budget_s))
        alpha = 10.0 + 40.0 * np.exp(-3.0 * frac)

        Pg = np.fft.rfft(p, L)
        c = np.fft.irfft(Pg * Pg, L)[:2 * n - 1].real
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        if not np.isfinite(maxc) or maxc <= 0.0:
            break

        f_now = float(2.0 * n * maxc)
        if np.isfinite(f_now) and f_now < best_f:
            best_f = f_now
            best_p = p.copy()

        z = alpha * (c / maxc - 1.0)
        z = np.clip(z, -50.0, 50.0)
        expz = np.exp(z)
        sz = float(expz.sum())
        if not np.isfinite(sz) or sz <= 0.0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[int(np.argmax(c))] = 1.0
        else:
            w = expz / sz

        W = np.fft.rfft(w[::-1], L)
        Cfull = np.fft.irfft(Pg * W, L)
        C = Cfull[n - 1:2 * n - 1][::-1].real
        C = np.maximum(C, 0.0)

        grad_p = (4.0 * n) * C
        gp = float(np.dot(p, grad_p))
        grad_x = p * (grad_p - gp)

        m_vec = mu * m_vec + (1.0 - mu) * grad_x
        v_vec = nu * v_vec + (1.0 - nu) * (grad_x * grad_x)
        mhat = m_vec / (1.0 - mu ** t)
        vhat = v_vec / (1.0 - nu ** t)
        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.clip(step, -50.0, 50.0)
        x = x - step
        x -= x.mean()

        if not np.all(np.isfinite(x)):
            break
        x = np.clip(x, -80.0, 80.0)
        x -= x.mean()
        x = x - x.max()
        e = np.exp(x)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            break
        p = e / se

    final_f = actual_score_fft(p)
    if final_f < best_f:
        best_f = final_f
        best_p = p.copy()
    return best_p, best_f


def lp_improve(p, budget_s):

    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size

    budget_s = float(budget_s)
    if budget_s < 15.0:

    max_rows = min(1200, max(200, 5000000 // n))

    best_f = f.copy()
    best_score = float(cur_score)

    # Initial rows: spread evenly

                break

            break

        if it % 3 == 0:

def pair_optimize(p, budget_s, seed=0):
    """
    Local optimization by optimizing pairs of coordinates.
    For a pair (i,j) with i != j, the convolution is linear in x = a[i]
    (except at k=i+j where there's a quadratic term), so we can find
    the optimal x by 1D search.
    """
    rng = np.random.default_rng(int(seed))
    p = np.asarray(p, dtype=np.float64).ravel().copy()
    n = p.size
    if n < 2:
        return p, actual_score_fft(p)
    
    p = normalize_nonneg(p)
    current_score = actual_score_fft(p)
    
    if budget_s < 1.0:
        return p, current_score
    
    deadline = time.monotonic() + budget_s
    
    # Precompute the convolution coefficients for fast pair updates
    # (a*a)[k] = sum_m a[m]*a[k-m]
    # When we change a[i] from old to new, (a*a)[k] changes by:
    #   (new - old) * a[k-i]  (for each valid k-i)
    # When we change a[j] from old to new, similarly.
    
    best_p = p.copy()
    best_score = current_score
    
    n_iter = 0
    while time.monotonic() < deadline:
        n_iter += 1
        
        # Pick a pair
        # Strategy: pick i near the peak, j random or also near the peak
        c = np.convolve(p, p) if n <= 512 else np.fft.irfft(np.fft.rfft(p, next_power_of_two(2*n-1))**2, next_power_of_two(2*n-1))[:2*n-1].real
        c = np.maximum(c, 0.0)
        
        # Find peak location
        peak_k = int(np.argmax(c))
        
        # The peak is at k, which involves pairs (m, k-m) for valid m.
        # Pick two coordinates that contribute most to the peak.
        lo = max(0, peak_k - n + 1)
        hi = min(peak_k, n - 1)
        if hi - lo + 1 >= 2:
            # Pick two random indices from [lo, hi]
            candidates = np.arange(lo, hi + 1)
            idx = rng.choice(candidates, size=min(2, len(candidates)), replace=False)
            i, j = int(idx[0]), int(idx[1])
            if i == j:
                j = (j + 1) % n
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n
        
        # Optimize pair (i, j)
        s = p[i] + p[j]
        if s <= 0:
            continue
        
        # Compute base convolution with a[i]=0, a[j]=0
        # Then (a*a)[k] = base[k] + x*a[k-i] + (s-x)*a[k-j] for k != i+j
        #              = base[k] + x*a[k-i] + (s-x)*a[k-j] + x*(s-x) - x*(s-x) ...

1. Multi-length initialization with greedy/power-based starts.

def next_power_of_two(m):
    L = 1
    while L < m:
        L <<= 1
    return L


def normalize_nonneg(u):
    a = np.asarray(u, dtype=np.float64).ravel()

    a = np.clip(a, 0.0, 1000.0)

        s = float(a.size)

    n = int(n)
    if n <= 0:

    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):

    return normalize_nonneg(a)

    a = np.power(idx, -float(s))
