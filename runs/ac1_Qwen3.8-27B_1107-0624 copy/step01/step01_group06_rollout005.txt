    # Hmm, that's sum_j p[j] * a[j-i] which is what I want!
    # Wait: sum_k a[k] * p[k+i]. Let j = k+i, k = j-i.
    # = sum_j a[j-i] * p[j] = sum_j p[j] * a[j-i]. Yes!
    
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    
    return J, grad


def multiplicative_subgradient_optimize(a_init, budget_s, seed=42,
                                         tau_frac=0.01, 
                                         eta_start=None,
                                         eta_min=None,
                                         use_symmetry=False):
    """Optimize using multiplicative updates with smoothed gradient."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S_init = np.sum(a)
    if S_init > 0:
        a = a / S_init  # Normalize to sum=1
    
    if use_symmetry:
        # Make symmetric
        a = (a + a[::-1]) / 2.0
        a = a / np.sum(a)
    
    best_a = a.copy()
    best_J = compute_J(a)
    
    if eta_start is None:
        eta_start = n * 0.5
    if eta_min is None:
        eta_min = n * 0.001
    
    eta = eta_start
    patience = 0
    max_patience = 200
    n_iter = 0
    start_time = time.monotonic()
    last_print = 0
    
    while time.monotonic() < deadline:
        tau = tau_frac * max(1e-10, max(c := convolve_fft(a)))
        # Actually let's use a fixed tau relative to the scale
        # After normalization S=1, c values are ~1/n, so max(c) ~ 1/n * factor
        # tau should be small relative to max(c)
        c = convolve_fft(a)
        c_max = np.max(c)
        tau = max(1e-8, tau_frac * c_max)
        
        J_smooth, grad = smoothed_gradient(a, tau)
        
        # The gradient of J w.r.t. a (with S=1 constraint)
        # We project onto the constraint sum(a)=1 by subtracting the mean
        grad_mean = np.mean(grad)
        grad_proj = grad - grad_mean  # Project onto hyperplane sum=0
        
        # Multiplicative update
        # a_i *= exp(-eta * grad_proj_i / S)
        # But S=1, so: a_i *= exp(-eta * grad_proj_i)
        # Scale eta appropriately
        # grad_proj is of order ~ n * max(c) / S^2 ~ n * (1/n) = 1
        # So eta should be O(1/n) for small updates
        eta_eff = eta / n
        
        log_update = -eta_eff * grad_proj
        log_update = np.clip(log_update, -5, 5)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        S_new = np.sum(a_new)
        if S_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / S_new
        
        if use_symmetry:
            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = a_new / np.sum(a_new)
        
        J_new = compute_J(a_new)
        
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                # Try a perturbation
                pert = np.exp(rng.normal(0, 0.05, n))
                a_pert = a * pert
                if use_symmetry:
                    a_pert = (a_pert + a_pert[::-1]) / 2.0
                a_pert = np.maximum(a_pert, 0)
                Sp = np.sum(a_pert)
                if Sp > 0:
                    a_pert = a_pert / Sp
                J_pert = compute_J(a_pert)
                if J_pert < best_J:
                    best_J = J_pert
                    best_a = a_pert.copy()
                    a = a_pert
                    patience = 0
            if patience > max_patience:
                # Increase eta (anneal)
                eta = min(eta_start * 4, eta * 1.5)
                patience = 0
                # Reset to best
                a = best_a.copy()
        
        # Decay eta slowly
        eta = max(eta_min, eta * 0.9995)
        
        n_iter += 1
        if n_iter % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    iter={n_iter}, J={best_J:.6f}, eta={eta:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_J


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function."""
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)
    
    # Get base sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)
    
    n = parent.size
    print(f"n = {n}")
    
    # Initial score
    best_global = parent.copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial score: {best_global_score:.6f}")
    
    # Prepare initializations
    inits = []
    inits.append(("parent", parent.copy()))
    inits.append(("parent_sym", (parent + parent[::-1]) / 2.0))
    
    # Add some structured initializations
    x = np.arange(n)
    mid = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((x - mid) / (0.2 * n))**2)
    inits.append(("gauss", gauss))
    
    cos1 = 1.0 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(("cos1", cos1))
    
    cos2 = 1.0 + 0.3 * np.cos(4 * np.pi * x / n) + 0.2 * np.cos(6 * np.pi * x / n)
    inits.append(("cos2", cos2))
    
    rand = np.exp(rng.normal(0, 0.5, n))
    inits.append(("rand", rand))
    
    # Score all initializations
    scored_inits = []
    for name, arr in inits:
        arr = np.maximum(arr, 0)
        S = np.sum(arr)
        if S > 0:
            arr = arr / S
        J = compute_J(arr)
        scored_inits.append((name, J, arr))
    
    scored_inits.sort(key=lambda x: x[1])
    for name, J, _ in scored_inits:
        print(f"  init '{name}': J = {J:.6f}")
    
    # Time allocation
    remaining = total_deadline - time.monotonic() - 20.0
    # Give most time to the best initialization
    # Top 1: 50%, next 2: 20% each, rest: 5% each
    
    strategies = []
    if remaining > 0:
        strategies.append(("main", scored_inits[0], remaining * 0.5, False))
        if len(scored_inits) > 1:
            strategies.append(("alt1", scored_inits[1], remaining * 0.2, True))
        if len(scored_inits) > 2:
            strategies.append(("alt2", scored_inits[2], remaining * 0.15, False))
        if len(scored_inits) > 3:
            strategies.append(("alt3", scored_inits[3], remaining * 0.1, True))
        if len(scored_inits) > 4:
            strategies.append(("alt4", scored_inits[4], remaining * 0.1, False))
    
    # Execute strategies
    for name, (init_name, init_J, init_arr), budget, use_sym in strategies:
        if time.monotonic() >= total_deadline - 20.0:
            break
        budget = min(budget, total_deadline - time.monotonic() - 15.0)
        if budget < 5.0:
            break
        
        print(f"\nRunning '{name}' from '{init_name}' (budget={budget:.1f}s, sym={use_sym})")
        
        res_a, res_J = multiplicative_subgradient_optimize(
            init_arr, budget, seed=seed + hash(init_name) % 100000,

            best_global_score = val
