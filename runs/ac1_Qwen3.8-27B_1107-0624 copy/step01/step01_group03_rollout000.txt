    """Compute 2n*max(conv(s,s))/sum(s)^2 efficiently."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)


def verify_uniform_optimal(n, num_trials=100):
    """Verify that random non-negative sequences do not beat the uniform score of 2."""
    uniform_score = fast_score(np.ones(n))
    rng = np.random.default_rng(12345)
    min_found = uniform_score
    for _ in range(num_trials):
        # Various random non-negative sequences
        s = rng.random(n)
        s = s / s.sum() * n  # Normalize to same total as uniform
        sc = fast_score(s)
        if sc < min_found:
            min_found = sc
    return uniform_score, min_found


def try_structured_search(n, deadline):
    """Try structured families to see if anything beats 2.0."""
    best_score = 2.0
    best_seq = np.ones(n)

    # Linear ramp
    for beta in np.linspace(0, 0.95, 19):
        s = 1.0 + beta * (2.0 * np.arange(n) / (n - 1) - 1.0)
        s = np.maximum(s, 0.01)
        sc = fast_score(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = s.copy()

    # Exponential
    for alpha in np.linspace(-3, 3, 13):
        s = np.exp(alpha * np.arange(n) / (n - 1))
        sc = fast_score(s)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = s.copy()

    # Two-level
    for frac in [0.3, 0.5, 0.7]:
        k = int(frac * n)
        for ratio in [0.5, 0.7, 1.5, 2.0]:
            s = np.zeros(n)
            s[:k] = ratio
            s[k:] = 1.0
            sc = fast_score(s)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = s.copy()

    return best_score, best_seq
