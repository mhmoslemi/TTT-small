"""
FFT-based smooth-max gradient optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp approximation of the max, multiplicative updates via FFT-computed gradients,
with adaptive temperature, multiple initializations, and optional LP polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    N = 1
    while N < n_out:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_out]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (S * S)


def smooth_max_gradient(a, tau):
    """
    Compute the smooth max of conv(a,a) and its gradient w.r.t. a.
    Returns (convolution, gradient).
    """
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    # Softmax weights for smooth max
    tc = tau * c
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    w = exp_tc / exp_tc.sum()

    # Gradient: grad_i = 2 * sum_k w[k] * a[k-i] (cross-correlation)
    fw = np.fft.rfft(w, N)
    fa2 = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa2), N)
    grad = 2.0 * cross[:n]

    return c, grad


def optimize_smooth_grad(a, deadline, rng, tau_init=5.0, tau_max=500.0, eta_init=0.3):
    """
    FFT-based smooth max gradient optimization with multiplicative updates.
    """
    n = len(a)

    best_score = compute_objective(a)
    tau = tau_init
    eta = eta_init
    stall = 0
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        c, grad = smooth_max_gradient(a, tau)
        current_max = c.max()

        # Multiplicative update
        # Scale grad so that average reduction is ~eta
        grad_scale = grad.max() + 1e-15

        a_new = np.clip(a_new, 0.0, None)

def optimize_peak_shave(a, deadline, rng):

        if len(peak_indices) == 0:
            break

        # a[i] contributes a[i]*a[k-i] to c[k] for each peak k

        a_new = np.clip(a_new, 0.0, None)
        S_new = a_new.sum()
        if S_new < 1e-10:
            break

        sc = compute_objective(a_new)

            best_score = sc

def build_conv_matrix(a):
    """Build matrix A (shape 2n-1 x n) such that A @ g = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(g) s.t. convolve(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def make_gaussian(n, sigma_frac=0.3):
    """Gaussian window centered in the middle."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)


def make_cosine(n):
    """Raised cosine (Hann-like) window."""
    return 0.5 * (1.0 + np.cos(np.linspace(0, np.pi, n)))


def make_parabolic(n):
    """Parabolic (triangular) window."""
    c = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - np.abs(np.arange(n) - c) / (c + 1.0))


def make_bi_gaussian(n, sigma1_frac=0.2, sigma2_frac=0.4, weight=0.5):
    """Sum of two Gaussians with different widths."""
    center = (n - 1) / 2.0
    s1 = sigma1_frac * n
    s2 = sigma2_frac * n
    x = np.arange(n) - center
    return weight * np.exp(-0.5 * (x / s1) ** 2) + (1 - weight) * np.exp(-0.5 * (x / s2) ** 2)


def make_flat_with_edges(n, edge_frac=0.1):
    """Nearly flat with slightly reduced edges."""
    a = np.ones(n)
    edge = int(edge_frac * n)
    for i in range(edge):
        decay = 1.0 - i / (edge + 1)
        a[i] = decay
        a[n - 1 - i] = decay
    return a

    elapsed_total = time.monotonic() - start_time
    print(f"\nDONE in {elapsed_total:.1f}s. Final score={best_score:.6f}, n={len(best_seq)}")

    return [float(x) for x in best_seq]
