def smooth_obj_grad(a, tau):

def make_periodic(n, period, duty=0.5, phase=0):
    """Create a periodic sequence with given period and duty cycle."""
    x = np.arange(n)
    pattern = np.zeros(period)
    high_count = int(duty * period)
    pattern[:high_count] = 1.0
    # Extend pattern
    reps = n // period + 1
    full = np.tile(pattern, reps)[:n]
    # Apply sinusoidal envelope
    envelope = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    return full * (0.3 + 0.7 * envelope) + 0.1


def make_bimodal(n, frac=0.3):

    x = np.arange(n)
    sigma = 0.1 * n
    p1 = frac * n
    p2 = (1 - frac) * n
    return np.exp(-(x - p1)**2 / (2*sigma*sigma)) + np.exp(-(x - p2)**2 / (2*sigma*sigma))


def make_gap(n, gap_frac=0.2):

"""
Multi-phase optimizer for minimizing J(a) = 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated log-sum-exp smoothed gradients with Adam optimizer,
adaptive tau cooling, peak-targeted coordinate transport, and multi-start
with structured initializations.

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J = 2n*max(a*a)/S^2 using FFT."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_grad(a, tau):
    """Compute smoothed objective and gradient via log-sum-exp."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = np.sum(a)
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = np.max(c)
    shifted = (c - c_max) / tau
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    if sum_exp < 1e-300:
        return 1e15, np.zeros(n)
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_s / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S)
    return J, grad


def norm_n(a):
    """Normalize so sum(a) = len(a)."""
    S = np.sum(a)
    if S < 1e-10:
        return a
    return a * len(a) / S


def gd_run(a_init, budget, seed, tau_hi=2.0, tau_lo=0.01, lr0=0.03):
    """Adam gradient descent with cosine tau cooling and adaptive restarts."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget
    a = norm_n(a_init.astype(np.float64).copy())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_J = compute_J(a)
    start = time.monotonic()
    stall = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / max(budget, 1.0))
        # Fast cosine cooling: reach small tau in first 50% of time
        if frac < 0.5:
            tau_frac = frac * 2.0
        else:
            tau_frac = 1.0
        tau = tau_lo + (tau_hi - tau_lo) * 0.5 * (1.0 + np.cos(np.pi * tau_frac))
        lr = lr0 / (1.0 + 0.5 * t / 3000.0)

        J, grad = smooth_obj_grad(a, tau)
        if not np.isfinite(J) or not np.all(np.isfinite(grad)):
            a = best_a.copy()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a = norm_n(a)

            local_J = compute_J(a)

                stall = 0

                    c = convolve_fft(best_a)

def peak_transport_descent(a_init, budget, seed):
    """Iteratively transport mass from peak contributors to non-peak positions."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget
    a = norm_n(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_J = compute_J(a)
    alpha = 0.08
    count = 0

    while time.monotonic() < deadline:
        count += 1
        c = convolve_fft(a)
        k_star = int(np.argmax(c))
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        idxs = np.arange(lo, hi + 1)
        contribs = a[k_star - idxs]

        top_k = min(100, len(idxs))
        top_idx = np.argsort(contribs)[-top_k:]
        total_mass = alpha * np.sum(contribs[top_idx])

        if total_mass < 1e-10:
            alpha = max(0.01, alpha * 0.9)
            continue

        a_new = a.copy()
        a_new[idxs[top_idx]] -= total_mass / top_k * 2.0
        a_new = np.maximum(a_new, 0.0)

        non_peak = np.setdiff1d(np.arange(n), idxs)
        if len(non_peak) > top_k:
            add_pos = rng.choice(non_peak, top_k, replace=True)

            a = a_new.copy()
            alpha = min(0.15, alpha * 1.01)
        else:
            a = best_a.copy()
            alpha = max(0.005, alpha * 0.92)

def make_periodic(n, period, duty=0.5, phase=0):
    """Periodic sequence with sinusoidal envelope."""
    x = np.arange(n)
    pattern = np.zeros(period)
    high_cnt = max(1, int(duty * period))
    pattern[phase:phase + high_cnt] = 1.0
    if phase + high_cnt > period:
        pattern[:phase + high_cnt - period] = 1.0
    reps = n // period + 1
    full = np.tile(pattern, reps)[:n]
    env = 0.4 + 0.6 * (0.5 + 0.5 * np.cos(2.0 * np.pi * x / n))
    return full * env + 0.05


def make_bimodal(n, frac=0.25, sigma_frac=0.12):
    """Two Gaussian peaks."""
    x = np.arange(n)
    sigma = sigma_frac * n
    p1 = frac * n
    p2 = (1.0 - frac) * n
    return np.exp(-(x - p1) ** 2 / (2 * sigma * sigma)) + np.exp(-(x - p2)
