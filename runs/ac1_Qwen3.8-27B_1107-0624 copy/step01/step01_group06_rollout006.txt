    J = 2.0 * n * Tmax / (S * S)

import numpy as np
import time


def convolve_fft(a):
    """Fast self-convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa * fa, size)
    return result[:2 * n - 1]


def compute_J(a):
    """Compute exact J = 2n*max(a*a)/S^2."""
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * float(np.max(c)) / (S * S)


def cross_correlation_fft(p, a):
    """Compute cross[i] = sum_k p[k] * a[k-i] for i in [0, n-1]."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    return cross


def smooth_obj_grad(a, tau):
    """Compute smoothed J and gradient using log-sum-exp."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    S = float(np.sum(a))
    if S < 0.01:
        return 1e15, np.zeros(n)
    c_max = float(np.max(c))
    shifted = (c - c_max) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    Tmax = c_max + tau * np.log(sum_exp)
    p = exp_shifted / sum_exp
    J = 2.0 * n * Tmax / (S * S)
    fp = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(fa) * fp, size)[:n]
    grad = (4.0 * n * cross) / (S * S) - (4.0 * n * Tmax) / (S * S * S) * a
    return J, grad

def run_fixed_point(a_init, budget_s, seed, n_active_frac=0.05, damping=0.3):

    no_improve = 0

def run_coordinate_peak(a_init, budget_s, seed):
    """Coordinate descent that directly targets reducing the max convolution."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    a = a_init.copy()

    if S > 0:
        a = a * (n / S)
    best_a = a.copy()
    best_J = compute_J(a)
    c = convolve_fft(a)
    iter_count = 0

    while time.monotonic() < deadline:
        # Find peak
        k_star = int(np.argmax(c))
        c_max = c[k_star]
        # Find second max (excluding positions affected by single coordinate changes)
        # For coordinate i, c[k] changes for k in [i, i+n-1]
        # So k_star is affected by i if i <= k_star <= i+n-1, i.e., k_star-n+1 <= i <= k_star
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        # Try reducing a[i] for i in [i_lo, i_hi]
        # The contribution of a[i] to c[k_star] is: a[i]*a[k_star-i]
        # Reducing a[i] by delta reduces c[k_star] by delta*a[k_star-i]
        # But it also affects other c[k] values
        best_delta = 0
        best_new_J = best_J
        # Try a few random candidates from the contributing indices
        n_try = min(20, i_hi - i_lo + 1)
        indices = list(range(i_lo, i_hi + 1))
        rng.shuffle(indices)
        indices = indices[:n_try]

        for i in indices:
            contrib = a[i] * a[k_star - i]
            if contrib <= 0 or a[i] <= 0:
                continue
            # Try several reduction fractions
            for frac in [0.5, 0.3, 0.2, 0.1, 0.05]:
                delta = frac * a[i]
                a[i] -= delta
                if a[i] < 0:
                    a[i] = 0
                # Quick check: recompute c at k_star and a few neighbors
                # Full recompute is expensive, so just check k_star
                new_c_ks = c[k_star] - delta * (a[k_star - i] + (a[i - k_star] if 0 <= i - k_star < n else 0))
                # Actually this is approximate. Let's just do a full check for the top candidates
                a[i] += delta  # restore
                # Do full evaluation for the most promising
                if frac == 0.3:
                    a[i] -= delta
                    c_new = convolve_fft(a)
                    J_new = 2.0 * n * float(np.max(c_new)) / (S * S)
                    if J_new < best_new_J:
                        best_new_J = J_new
                        best_delta = delta
                        best_i = i
                        best_frac = frac
                    a[i] += delta

        if best_delta > 0 and best_new_J < best_J - 1e-12:
            a[best_i] -= best_delta
            a[best_i] = max(0, a[best_i])
            S_new = np.sum(a)
            if S_new < 0.01:
                a[best_i] += best_delta
                break
            c = convolve_fft(a)
            # Re-normalize
            a = a * (n / S_new)
            c = convolve_fft(a)
            local_J = compute_J(a)
            if local_J < best_J:
                best_J = local_J
                best_a = a.copy()
            iter_count += 1

    return best_a, best_J

        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise ValueError("bad")
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    n = parent.size
    print(f"Sequence length n = {n}")

    # Build diverse initializations
    inits = {}
    inits["parent"] = parent.copy()
    inits["ones"] = np.ones(n)
    inits["gauss_narrow"] = make_gaussian(n, 0.15)
    inits["gauss_mid"] = make_gaussian(n, 0.3)
    inits["gauss_wide"] = make_gaussian(n, 0.45)
    inits["edge"] = make_edge_conc(n, 2.0)
    inits["edge_sharp"] = make_edge_conc(n, 4.0)
    inits["cos1"] = make_cosine(n, 1, 0.4)
    inits["cos2"] = make_cosine(n, 2, 0.4)
    inits["cos3"] = make_cosine(n, 3, 0.3)
    inits["bimodal"] = make_bimodal(n, 0.05)
    inits["comb"] = make_comb(n, 3)
    inits["rand_s1"] = make_random_pos(n, rng, 0.4)
    inits["rand_s2"] = make_random_pos(n, rng, 0.8)
    inits["rand_s3"] = make_random_pos(n, rng, 1.2)
    # Blend parent with edge
    inits["parent_edge"] = 0.5 * parent + 0.5 * make_edge_conc(n, 3.0)
    inits["parent_gauss"] = 0.5 * parent + 0.5 * make_gaussian(n, 0.35)

    # Normalize and score all initializations
    scored = []
    for name, arr in inits.items():
        arr = np.maximum(arr, 0.0)
        S = np.sum(arr)
        if S < 0.01:
            continue
        arr = arr * (n / S)
        J = compute_J(arr)
        scored.append((J, name, arr))
        print(f"  init '{name}': J = {J:.6f}")

    scored.sort(key=lambda x: x[0])

    # Initial best
    best_global = scored[0][2].copy()
    best_global_score = float(evaluate_sequence(best_global.tolist()))
    print(f"Initial best score: {best_global_score:.6f}")

    # Time allocation
    elapsed = time.monotonic() - (total_deadline - float(budget_s) + 8.0)
    remaining = total_deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]

    # Phase 1: Fixed-point on top 5 (40% of time)
    fp_time = remaining * 0.35
    phase1_deadline = time.monotonic() + fp_time
    print(f"\n=== Phase 1: Fixed-point optimization ({fp_time:.0f}s) ===")
    for idx in range(min(5, len(scored))):
        if time.monotonic() >= phase1_deadline:
            break
        J0, name, arr = scored[idx]
        t_budget = min(phase1_deadline - time.monotonic(), 30.0)
        if t_budget < 3:
            break
        print(f"  FP on '{name}' (budget={t_budget:.1f}s, J0={J0:.6f})")
        res_a, res_J = run_fixed_point(arr, t_budget, seed=seed + 100 + idx)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"    *** NEW BEST (FP): {val:.6f} ***")
        else:
            print(f"    FP result: local={res_J:.6f}, verified={val:.6f}")

    # Phase 2: Adam GD on top 3 (35% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        gd_time = remaining * 0.45
        phase2_deadline = time.monotonic() + gd_time
        print(f"\n=== Phase 2: Adam GD optimization ({gd_time:.0f}s) ===")
        # Use best results from phase 1 as starting points for GD
        gd_inits = []
        # Top scored inits
        for idx in range(min(3, len(scored))):
            gd_inits.append(scored[idx][2])
        # Also include the current global best
        gd_inits.append(best_global.copy())

        for idx, arr in enumerate(gd_inits):
            if time.monotonic() >= phase2_deadline:
                break
            t_budget = min(phase2_deadline - time.monotonic(), 60.0)
            if t_budget < 5:
                break
            # Adaptive tau based on current best
            tau_s = max(10.0, n * 0.05)
            tau_e = 0.5
            lr_s = 0.01
            print(f"  GD init {idx} (budget={t_budget:.1f}s)")
            res_a, res_J = run_adam(arr, t_budget, seed=seed + 500 + idx,
                                     tau_start=tau_s, tau_end=tau_e, lr_start=lr_s)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"    *** NEW BEST (GD): {val:.6f} ***")
            else:
                print(f"    GD result: local={res_J:.6f}, verified={val:.6f}")

    # Phase 3: More fixed-point with tighter parameters on global best (15% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        fp2_time = remaining * 0.35
        phase3_deadline = time.monotonic() + fp2_time
        print(f"\n=== Phase 3: Tight fixed-point on best ({fp2_time:.0f}s) ===")
        for attempt in range(3):
            if time.monotonic() >= phase3_deadline:
                break
            t_budget = min(phase3_deadline - time.monotonic(), 25.0)
            if t_budget < 3:
                break
            # Perturb the best slightly for each attempt
            start = best_global.copy()
            if attempt > 0:
                w = max(1, n // 10)
                s = int(rng.integers(0, max(1, n - w)))
                start[s:s+w] *= np.exp(rng.normal(0, 0.1, w))
                start = np.maximum(start, 0)
                SS = np.sum(start)
                if SS > 0.01:
                    start = start * (n / SS)
            active_frac = 0.02 + 0.01 * attempt
            res_a, res_J = run_fixed_point(start, t_budget, seed=seed + 900 + attempt,
                                            n_active_frac=active_frac, damping=0.2)
            val = float(evaluate_sequence(res_a.tolist()))
            if np.isfinite(val) and val < best_global_score:
                best_global = res_a.copy()
                best_global_score = val
                print(f"  *** NEW BEST (FP2): {val:.6f} ***")
            else:
                print(f"  FP2 attempt {attempt}: local={res_J:.6f}, verified={val:.6f}")

    # Phase 4: Adam GD with very sharp tau on global best (10% of time)
    remaining = total_deadline - time.monotonic()
    if remaining > 15:
        gd2_time = remaining * 0.6
        print(f"\n=== Phase 4: Sharp GD refinement ({gd2_time:.0f}s) ===")
        res_a, res_J = run_adam(best_global, gd2_time, seed=seed + 2000,
                                 tau_start=5.0, tau_end=0.1, lr_start=0.005)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST (sharp GD): {val:.6f} ***")
        else:
            print(f"  Sharp GD: local={res_J:.6f}, verified={val:.6f}")

    # Phase 5: Final coordinate peak reduction
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        coord_time = remaining * 0.7
        print(f"\n=== Phase 5: Coordinate peak reduction ({coord_time:.0f}s) ===")
        res_a, res_J = run_coordinate_peak(best_global, coord_time, seed=seed + 3000)
        val = float(evaluate_sequence(res_a.tolist()))
        if np.isfinite(val) and val < best_global_score:
            best_global = res_a.copy()
            best_global_score = val
            print(f"  *** NEW BEST (coord): {val:.6f} ***")
        else:
            print(f"  Coord: local={res_J:.6f}, verified={val:.6f}")

    # Final verification
    final_val = float(evaluate_sequence(best_global.tolist()))
    if np.isfinite(final_val):
        best_global_score = min(best_global_score, final_val)

    print(f"\n=== FINAL SCORE: {best_global_score:.6f} ===")
    return [float(x) for x in np.clip(best_global, 0.0, 1000.0)]
