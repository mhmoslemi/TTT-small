Let me also think about the simplex projection more carefully. We need to project onto {s ≥ 0, sum(s) = 1}.

The projection algorithm (Duchi et al., 2011):
Given v ∈ R^n, find the closest point on the probability simplex.
1. Sort v in decreasing order: v_(1) ≥ v_(2) ≥ ... ≥ v_(n)
2. Find the largest ρ such that sum_{i=1}^{ρ} (v_(i) - τ) > 0 where τ = (sum_{i=1}^{ρ} v_(i) - 1) / ρ
3. The projection is: proj(v)_i = max(v_i - τ, 0)

This is O(n log n) per projection.

For n=2048, this is fast.

def project_simplex(v):
    """Project v onto the probability simplex {s >= 0, sum(s) = 1}."""
    n = len(v)

def fast_score(s):
    """Compute the objective value."""
    n = len(s)
    total = np.sum(s)
    if total < 1e-15:
        return float('inf')
    conv = np.convolve(s, s)
    return 2.0 * n * np.max(conv) / (total * total)

def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1)
    rho = np.max(ind[cond]) if np.any(cond) else 0
    if rho == 0:
        return np.ones(n) / n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def softmax(x, tau=1.0):
    """Stable softmax with temperature."""
    x = tau * x
    x_max = np.max(x)
    e = np.exp(x - x_max)
    return e / np.sum(e)

def compute_grad_lse(s, tau):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k(s))) wrt s.
    Uses FFT-based convolution for efficiency.
    """
    n = len(s)
    conv = np.convolve(s, s)
    
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    exp_vals = np.exp(tau * conv_shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[i] = 2 * sum_j s[j] * p[i+j]
    # = 2 * (s_rev * p)[n-1+i] where s_rev = s[::-1]
    s_rev = s[::-1]
    conv_sp = np.convolve(s_rev, p)
    grad = 2.0 * conv_sp[n - 1 : 2 * n - 1]
    
    return grad, conv

def peak_smash(s, n_peaks=3, strength=0.1):
    """
    Directly reduce the top convolution peaks by redistributing mass.
    """
    n = len(s)
    conv = np.convolve(s, s)
    
    # Find top peaks
    peak_indices = np.argsort(conv)[::-1][:n_peaks]
    
    s_new = s.copy()
    for k in peak_indices:
        # Find the contributing pairs
        # conv[k] = sum_j s[j]*s[k-j]
        total_val = conv[k]
        if total_val < 1e-15:
            continue
        
        # Find the pair with the largest contribution
        best_j = -1
        best_prod = 0
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            prod = s[j] * s[k - j]
            if prod > best_prod:
                best_prod = prod
                best_j = j
        
        if best_j < 0:
            continue
        
        # Reduce s[best_j] and s[k - best_j] slightly
        # Transfer mass to a position that doesn't contribute to any peak
        j1 = best_j
        j2 = k - j1
        
        # Find a "safe" position to move mass to
        # A position that minimizes its contribution to all peaks
        contributions = np.zeros(n)
        for kp in peak_indices:
            for j in range(max(0, kp - n + 1), min(n, kp + 1)):
                contributions[j] += s[kp - j]
        
        # Move mass from the peak-contributing positions to low-contribution positions
        safe_positions = np.argsort(contributions)[:max(1, n // 10)]
        
        # Move a small amount of mass
        delta = strength * min(s[j1], s[j2])
        s_new[j1] -= delta / 2
        s_new[j2] -= delta / 2
        # Distribute to safe positions
        for sp in safe_positions:
            s_new[sp] += delta / len(safe_positions)
    
    s_new = np.maximum(s_new, 0.0)
    s_new = s_new / s_new.sum()
    return s_new

def mirror_descent_optimize(s, max_iters=5000, deadline=None, verbose=False):

        # Clip for numerical stability
