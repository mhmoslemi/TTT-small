```python
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def convolve_fft(a):
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    result = np.fft.irfft(fa*fa, size)
    return result[:2*n-1]

def compute_J(a):
    n = len(a)
    c = convolve_fft(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e15
    return 2.0 * n * np.max(c) / (S*S)

def build_conv_matrix(a):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = len(a)
    # Each diagonal d (d=0,...,n-1) has constant value a[d] and length n
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    A = sp.diags(diag_data, offsets, shape=(2*n-1, n), format='csr')
    return A

def lp_step(a, M):
    """Solve the LP: maximize sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_conv_matrix(a)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None

def lp_based_optimize(a_init, budget_s, seed=42):
    """Run the LP-based optimization."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    a = a_init.copy()
    S = np.sum(a)
    if S > 0:
        a = a * (n / S)  # Normalize to sum = n
    
    best_a = a.copy()
    best_J = compute_J(a)
    print(f"  LP start: J = {best_J:.6f}")
    
    iteration = 0
    stagnant = 0
    
    while time.monotonic() < deadline - 30:
        iteration += 1
        t_iter_start = time.monotonic()
        
        # Compute current peak
        c_a = convolve_fft(a)
        M = np.max(c_a)
        
        # Solve LP
        b = lp_step(a, M)
        if b is None or not np.isfinite(b).any():
            stagnant += 1
            if stagnant > 5:
                # Perturb and restart
                a = np.maximum(a * (1 + 0.1 * rng.normal(0, 1, n)), 0)
                a = a / np.sum(a) * n
                stagnant = 0
            continue
        
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            stagnant += 1
            continue
        
        # Check if b is proportional to a (stagnation)
        if sum_b > 0:
            ratio = b / np.maximum(a, 1e-10)
            if np.std(ratio) / (np.mean(ratio) + 1e-10) < 0.01:
                # b is approximately proportional to a
                stagnant += 1
                if stagnant > 3:
                    a = np.maximum(a * (1 + 0.05 * rng.normal(0, 1, n)), 0)
                    a = a / np.sum(a) * n
                    stagnant = 0
                continue
        
        # Normalize b to have same sum as a
        g = b * (np.sum(a) / sum_b)
        
        # Line search for t
        best_t = 0
        best_t_J = best_J
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = (1-t)*a + t*g
            if np.any(a_new < 0):
                continue
            J_new = compute_J(a_new)
            if J_new < best_t_J:
                best_t_J = J_new
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*g
            stagnant = 0
            if best_t_J < best_J:
                best_J = best_t_J
                best_a = a.copy()
                if iteration % 10 == 0:
                    print(f"  LP iter {iteration}: J = {best_J:.6f}, t={best_t:.2f}, time={time.monotonic()-t_iter_start:.1f}s")
        else:
            stagnant += 1
            if
