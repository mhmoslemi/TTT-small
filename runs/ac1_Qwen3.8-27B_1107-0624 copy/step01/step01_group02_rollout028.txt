import numpy as np
import time
from scipy.optimize import linprog

# Global state
best_score = float('inf')
best_seq = None

def normalize(a):
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def score_norm(a):
    """Score for sum-1 nonnegative sequence: 2*n*max(conv(a,a))"""
    n = len(a)
    c = np.convolve(a, a)
    return 2.0 * n * c.max()

def build_A(a, rows):
    """Build constraint matrix rows for given lag indices."""
    n = len(a)
    A = np.zeros((len(rows), n))
    for i, k in enumerate(rows):
        k = int(k)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[i, j] = a[idx]
    return A

def solve_lp(a, M, rows):
    """Solve LP: max sum(g) s.t. A(g) <= M, g >= 0."""
    n = len(a)
    A = build_A(a, rows)
    c = -np.ones(n)
    b = np.full(len(rows), M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        g = np.clip(res.x, 0, None)
        return g
    return None

def optimize_lp(a, time_limit, verbose=False):
    """Main LP optimization loop."""
    global best_score, best_seq
    a = normalize(a.copy())
    n = len(a)
    deadline = time.monotonic() + time_limit
    best_local = score_norm(a)
    best_a = a.copy()

        c = np.convolve(a, a)
        M = c.max()
        cur_score = 2.0 * n * M
        
        if cur_score < best_local:
            best_local = cur_score
            best_a = a.copy()

                best_score = best_local
                best_seq = best_a.copy()
        else:
            no_improve += 1

            # Use more constraints for precision
            threshold = M * 0.95

            # Use fewer constraints for speed
            k_active = min(len(c), max(50, n // 2))
            rows = np.argpartition(c, -k_active)[-k_active:]
        
        rows = rows.astype(int)
        
        g = solve_lp(a, M, rows)

            Sg = g.sum()
            if Sg > 1.0 + 1e-6:

                best_t = 0

using LP-based local search with adaptive active sets.

import numpy as np
import time
from scipy.optimize import linprog

def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones_like(a) / a.size
    return a / s

def _score_normalized(a):
    """Returns 2*n*max_conv for a sum-1 nonnegative sequence."""
    n = a.size
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())

def _build_constraint_matrix(a, rows):
    """Build LP constraint matrix for given lag indices."""
    n = a.size
    A = np.zeros((rows.size, n), dtype=np.float64)
    for i in range(rows.size):
        k = int(rows[i])
        jlo = max(0, k - n + 1)
        jhi = min(n, k + 1)
        if jhi > jlo:
            A[i, jlo:jhi] = a[k - np.arange(jhi - jlo)[::-1]]
    return A

def _solve_lp(a, M, rows):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for k in rows, g >= 0."""
    n = a.size
    if M <= 0:
        return None
    try:
        A = _build_constraint_matrix(a, rows)
        c_obj = -np.ones(n, dtype=np.float64)
        b_vec = np.full(rows.size, M, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)
            g = np.clip(g, 0.0, None)
            if g.sum() > 0:
                return g
    except Exception:
        pass
    return None

        c = np.convolve(a, a)
        M = float(c.max())
        cur_score = float(2.0 * n * M)
        
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_a = a.copy()
            no_improve = 0
            if best_local_score < global_state['score']:
                global_state['score'] = best_local_score
                global_state['seq'] = best_a.copy()
                print(f"  Improved: {best_local_score:.10f} (n={n})", flush=True)
        else:
            no_improve += 1
        
        # Determine phase based on progress
        if no_improve < 5:
            phase = 0  # Active improvements, use more constraints
        elif no_improve < 15:
            phase = 1  # Getting stuck, use fewer constraints
        else:
            phase = 2  # Stuck, need perturbation
        
        # Choose active set
        if phase == 0:
            # Use all constraints (or most) for precision
            if n <= 1024:
                rows = np.arange(2*n-1)
            else:
                thresh = M * 0.99
                rows = np.where(c >= thresh)[0]
                if rows.size > 1500:
                    rows = np.argpartition(c, -1500)[-1500:]
        elif phase == 1:
            # Top ~n constraints
            k = min(2*n-1, max(100, n))
            rows = np.argpartition(c, -k)[-k:]
        else:
            # Top ~n/2 constraints
            k = min(2*n-1, max(50, n // 2))
