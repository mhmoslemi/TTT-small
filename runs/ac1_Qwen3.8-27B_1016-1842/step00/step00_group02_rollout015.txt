import numpy as np

def _npo2(m):
    N = 1
    while N < m:
        N <<= 1
    return N


def _max_conv(a):
    n = len(a)
    N = _npo2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return float(c.max())


def _soft_max_grad(a, T):
    n = len(a)
    N = _npo2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    ex = np.exp((conv - mc) / T)

    C = np.fft.irfft(FW * np.conj(fa), N)[:n]
