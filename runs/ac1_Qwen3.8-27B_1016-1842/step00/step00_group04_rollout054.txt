def compute_conv(f):
    """Compute linear convolution of f with itself using FFT."""
    n = len(f)
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    C = np.fft.irfft(F * F, fft_size)[:size]
    return C

def subgradient_step(f, k_star, n):
    """Compute the subgradient of c[k_star] w.r.t. f."""
    g = np.zeros(n)
    # First term: f[k_star - i] for valid i
    i_start = max(0, k_star - n + 1)
    i_end = min(n - 1, k_star)
    if i_start <= i_end:
        g[i_start:i_end+1] += f[k_star - i_end:k_star - i_start - 1:-1]
    # Second term: f[i - k_star] for valid i
    i_start2 = max(0, k_star)
    i_end2 = min(n - 1, k_star + n - 1)
    if i_start2 <= i_end2:

from scipy.optimize import linprog

def convolve_fft(f):
    n = len(f)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_n)
    return np.fft.irfft(F * F, fft_n)[:size]

def cross_convolve_fft(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_n = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_n)
    G = np.fft.rfft(g, fft_n)
    return np.fft.irfft(F * G, fft_n)[:size]
