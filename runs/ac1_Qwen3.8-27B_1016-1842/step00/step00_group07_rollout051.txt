import numpy as np
import time


def _score_arr(a):
    """Exact score using the same formula as evaluate_sequence."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (s * s))


def _radial_step(a, m_star, lr):
    """Reduce a[i] paired with a[m_star - i] via multiplicative update."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, m_star - n + 1)
    hi = min(n - 1, m_star)
    for i in range(lo, hi + 1):
        j = m_star - i
        g[i] += a[j]
        if i != j:
            g[j] += a[i]
        else:
            g[i] += a[i]
    g *= lr
    a = np.clip(a - g, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return None
    return a / s


def _cos_init(n, k, A, phase, rng):
    idx = np.arange(n)
    a = 1.0 + A * np.cos(2.0 * np.pi * k * idx / n + phase)
    a[rng.integers(0, n, size=n // 4)] *= np.exp(rng.normal(0, 0.05, size=n // 4))
    a = np.clip(a, 0.01, None)
    return a / a.sum()
