    k_max = int(np.argmax(b))
    b_max = b[k_max]
    idx = k_max - np.arange(len(a))
    valid = (idx >= 0) & (idx < len(a))
    a_shifted = np.zeros(len(a))
    a_shifted[valid] = a[idx[valid]]
    c1 = 4.0 * len(a) / (S * S)

    score = 2.0 * len(a) * b_max / (S * S)
    grad = c1 * a_shifted - c2
    return score, grad
