            for bt in (0.5, 0.25, 0.12, 0.05):
                fn2 = np.clip((1.0 - bt) * f + bt * g, 0.0, 1000.0)
                s2 = evaluate_sequence(fn2.tolist())
                if np.isfinite(s2) and s2 < best_sc:
                    best = fn2.copy()
                    best_sc = s2
                    f = fn2
                    ok = True
                    break
        if not ok:
            stall += 1
            if stall > 50:
                break
            w = max(1, n // 80)
            s = int(rng.integers(0, max(1, n - w)))
            f[s:s + w] *= np.exp(rng.normal(0.0, 0.12))
            f = np.clip(f, 0.0, 1000.0)
            s2 = evaluate_sequence(f.tolist())
            if np.isfinite(s2) and s2 < best_sc:
                best = f.copy()
                best_sc = s2
                stall = 0
        else:
            stall = 0
    if not np.isfinite(evaluate_sequence(best.tolist())):
        best = par.copy()
    return [float(x) for x in best]
