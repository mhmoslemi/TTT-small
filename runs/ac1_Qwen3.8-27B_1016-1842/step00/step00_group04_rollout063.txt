import numpy as np
import time


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def compute_gradient(a, k, M, S):
    n = len(a)
    grad = np.zeros(n)
    start = max(0, k - n + 1)
    end = min(n - 1, k) + 1
    if end > start:
        seg = a[k - end + 1: k - start + 1][::-1].copy()
        grad[start:end] = seg - M / S
    return grad


def evaluate_fast(a):
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf, 0, 0.0
    b = fft_convolve(a, a)
    k = int(np.argmax(b))
    M = float(b[k])
    return float(2 * len(a) * M / (S * S)), k, M


def make_profiles(n, rng):
    profiles = []
    for eps in [0.2, 0.25, 0.3, 1.0/3, 0.35, 0.4, 0.5, 0.6]:
        a = np.empty(n)
        m = n // 2
        a[:m] = 1.0 + eps
        a[m:] = max(0.0, 1.0 - eps)
        profiles.append(a)
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append((1.0 - x) ** p)
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append(np.exp(-alpha * x))
    for beta in [0.5, 1.0, 2.0, 3.0, 5.0, 10.0, 20.0]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append(1.0 / (1.0 + np.exp(beta * (x - 0.5))))
    for c in [0.5, 0.7, 0.9, 1.2, 1.5]:
        x = np.linspace(0.0, 1.0, n)
        a = np.cos(c * np.pi * x / 2.0)

    for d in [0.3, 0.4, 0.5, 0.6, 0.8]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append(1.0 + d * np.cos(np.pi * x))
    for d in [0.3, 0.4, 0.5, 0.6, 0.8]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append(1.0 + d * np.cos(2.0 * np.pi * x))
    for d in [0.3, 0.5, 0.7]:
        x = np.linspace(0.0, 1.0, n)
        profiles.append(1.0 + d * np.cos(3.0 * np.pi * x))
    for m_frac in [0.3, 0.4, 0.6, 0.7]:
        for eps in [0.3, 0.5, 0.7]:
            a = np.empty(n)
            m = int(m_frac * n)
            a[:m] = 1.0 + eps
            a[m:] = max(0.0, 1.0 - eps)
            profiles.append(a)
    for trial in range(8):
        a = np.random.dirichlet(np.ones(n) * 0.5, rng) * n
        profiles.append(a)
    return profiles


def gradient_descent(a0, deadline, n):
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(np.sum(a))
    if S < 0.01:
        return a0
    scales = np.ones(n)
    scales[:] = S / n
    lr = 0.5
    best_local = a.copy()
    best_score_local = np.inf
    stagnant = 0
    while time.monotonic() < deadline:
        b = fft_convolve(a, a)
        k = int(np.argmax(b))
        M = float(b[k])
        S = float(np.sum(a))
        if S < 0.01:
            break
        score = 2.0 * n * M / (S * S)
        if score < best_score_local:
            best_score_local = score
            best_local = a.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                a = best_local.copy()
                stagnant = 0
            lr *= 0.95
        grad = compute_gradient(a, k, M, S)
        step = lr * grad / max(1.0, np.sqrt(np.sum(grad * grad)) / n)
        a = a - step
        a = np.clip(a, 0.0, 1000.0)
        if stagnant % 100 == 0:
            a = np.clip(a + 0.01 * (np.random.randn(n) * S / n), 0.0, 1000.0)
            lr = max(0.01, lr * 2.0)
    return best_local
