def solve_lp_for_g(f, M, deadline=None):

    # A is (2n-1) x n
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def build_sp_conv(f):
    """Sparse matrix A with (A @ g) ≈ f * g (linear convolution)."""
    n = f.size
    ii = np.tile(np.arange(n), n)
    jj = np.repeat(np.arange(n), n)
    return sp.csr_matrix((f[ii], (ii + jj, jj)), shape=(2 * n - 1, n))


def lp_dir(f, conv, M, dl, nc):
    """Solve affinity LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = f.size
    nm = min(len(conv), nc)
    ks = np.argpartition(conv, -nm)[:nm]
    try:
        A = build_sp_conv(f).tocsr()[ks]
        c = -np.ones(n)
        b = M * np.ones(nm)
        bnds = [(0.0, None)] * n
        tl = dl - time.monotonic()
        if tl < 5:
            return None
        res = linprog(c, A_ub=A, b_ub=b, bounds=bnds, method='highs',
                      options={'time_limit': max(1.0, tl - 20)})
        if res.success:
            g = np.clip(res.x, 0.0, None)

    except Exception:
        pass
    return None
