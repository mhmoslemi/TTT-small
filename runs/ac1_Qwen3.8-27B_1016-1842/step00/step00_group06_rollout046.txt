import numpy as np
import time

def simplex_project(v, target_sum=1.0):
    """Project v onto {x: x >= 0, sum(x) = target_sum}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target_sum)
    rho = np.max(rho[cond])
    theta = (cssv[rho-1] - target_sum) / rho
    return np.maximum(v - theta, 0)

import numpy as np

def simplex_project(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.arange(1, n+1)[::-1]  # n, n-1, ..., 1
    cond = u * rho > (cv - s)
    # Find largest rho satisfying condition
    # Actually standard algorithm:
    # u sorted descending, cv = cumsum(u)
    # rho* = max{k: u_k > (cv_k - s)/k}
    # theta = (cv_{rho*} - s) / rho*
    # result = max(v - theta, 0)
    
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    w = np.arange(1, n+1, dtype=float)
    cond = (cv - s) / w < u
    if not np.any(cond):
        return np.full(n, s/n)
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cv[rho-1] - s) / rho
    return np.maximum(v - theta, 0.0)
