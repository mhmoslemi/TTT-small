import numpy as np
import time

def fast_score(a):
    """Fast local score computation (approximate, for pre-filtering)."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    # Use FFT for convolution (O(n log n) instead of O(n^2))
    size = 2 * n
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2*n-1]
    return 2.0 * n * np.max(b) / (s * s)

def make_perturbation(base, rng, mode):
    n = len(base)
    candidate = base.copy()
    if mode == 0:
        # Block multiplicative
        width = int(rng.integers(max(1, n // 100), max(2, n // 8)))
        start = int(rng.integers(0, n - width + 1))
        candidate[start:start+width] *= float(rng.uniform(0.5, 1.5))
    elif mode == 1:
        # Notch (reduce a block)
