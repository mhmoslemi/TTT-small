def _fft_convolve(a):

    fft_size = 1 << (target - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:target]

    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0][-1]

def _fft_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    t = 2 * n - 1
    f = 1 << (t - 1).bit_length()
    A = np.fft.rfft(a, f)
    return np.fft.irfft(A * A, f)[:t]

def _simplex_proj(v):
    """Project v onto {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > cs - 1.0)[0][-1]
    return np.maximum(v - (cs[rho] - 1.0) / (rho + 1.0), 0.0)

    idx = k_star - j
    mask = (idx >= 0) & (idx < n)
    g = np.zeros(n)
    g[mask] = 2.0 * a[idx[mask]]
