import numpy as np
import scipy.optimize as so
import scipy.sparse as sp
import time


def _build_toeplitz(f):
    n = len(f)
    m = 2 * n - 1
    diags = [np.full(n, f[v]) for v in range(n)]
    offsets = list(range(0, -n, -1))
    return sp.diags(diags, offsets, shape=(m, n), format='csr')


def _lp_solve(f, M, tl):
    try:
        A = _build_toeplitz(f)
        c = -np.ones(len(f))
        res = so.linprog(c, A_ub=A, b_ub=np.full(2 * len(f) - 1, M),
                         bounds=[(0.0, None)] * len(f), method='highs',
                         options={'time_limit': tl})
        if res.success and res.x is not None and res.x.sum() > 1e-12:
            return res.x
    except Exception:
        pass
    return None


def _block_perturb(x, rng, sig):
    n = len(x)
    c = x.copy()
    w = int(rng.integers(max(2, n // 60), max(3, n // 6)))
    s = int(rng.integers(0, max(1, n - w)))
    c[s:s + w] *= float(np.exp(rng.normal(0.0, sig)))
    return np.clip(c, 0.0, 1000.0)


def _swap_perturb(x, rng):
    n = len(x)
    c = x.copy()
    i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
    tot = c[i] + c[j]
    if tot < 1e-12:
        return c
    r = float(np.exp(rng.normal(0.0, 0.8)))
    c[i] = tot * r / (1.0 + r)
    c[j] = tot - c[i]
    return np.clip(c, 0.0, 1000.0)

    if np.isfinite(sc) and sc < best_score:

        best_score = sc

    t = np.linspace(0.0, 1.0, n)
    for alpha in [0.1, 0.3, 0.5]:
        for mode in ['ramp', 'cos']:
            if mode == 'ramp':

            else:

            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:

                best_score = sc

                no_improve = 0
