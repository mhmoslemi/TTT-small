def build_conv_matrix(f, n):
    """Build sparse matrix A where A @ b = conv(f, b)."""
    diagonals = []
    offsets = []
    for d in range(n):
        length = n - d
        diagonals.append(np.full(length, f[d]))
        offsets.append(d)
    return scipy.sparse.diags(diagonals, offsets, shape=(2*n-1, n), format='csc')

    diags = [np.full(n - d, f[d]) for d in range(n)]
