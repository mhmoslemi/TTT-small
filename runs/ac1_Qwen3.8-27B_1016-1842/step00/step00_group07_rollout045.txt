import numpy as np
import time


def fast_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    n = a.size
    return float(2.0 * n * float(np.convolve(a, a).max()) / (s * s))


def _grad_lag(a, k, g):
    n = a.size
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if hi > lo:
        g[lo:hi] += a[k - np.arange(lo, hi)]
    lo2 = max(0, k)
    hi2 = min(n, k + n)
    if hi2 > lo2:
        g[lo2:hi2] += a[np.arange(lo2, hi2) - k]
