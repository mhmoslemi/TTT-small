import numpy as np
import time

def simplex_project(z):
    n = z.size
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    idx = np.nonzero(u * np.arange(1, n + 1) > cssv - 1.0)[0]
    if idx.size == 0:
        return np.maximum(z - (z.sum() - 1.0) / n, 0.0)
    rho = idx[-1]
    return np.maximum(z - (cssv[rho] - 1.0) / (rho + 1.0), 0.0)

def fourier_init(n, rng):
    N = 2 * n - 1
    F = np.zeros(N, dtype=np.complex128)
    F[0] = 1.0
    c = 1.0 / np.sqrt(N - 1)
    half = (N - 1) // 2
    ph = rng.random(half)
    F[1:half + 1] = c * np.exp(2j * np.pi * ph)
    F[N - half:N] = np.conj(F[1:half + 1][::-1])
    f = np.fft.irfft(F, N)[:n]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return f / s
