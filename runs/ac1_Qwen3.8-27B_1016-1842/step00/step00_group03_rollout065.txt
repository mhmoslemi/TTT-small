def _clip_nonneg(x, n):
    x = np.maximum(x, 0.0)
    x = np.minimum(x, 1000.0)

def _simplex_project(v):
    """Project v onto {x >= 0, sum x = 1} using water-filling. O(n log n)."""
    n = v.size
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0]) if v[0] > 0 else np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _internal_score(a):
    """Exact score for sum(a)==1: 2*n*max(conv(a,a))."""
    c = np.convolve(a, a, mode="full")
    return 2.0 * a.size * float(np.max(c))


def _make_exp(n, alpha, rng):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * x / max(1, n - 1))
    a += 1e-9
    s = a.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s


def _make_poly(n, p, rng):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = (1.0 - x) ** p
    a += 1e-9
    s = a.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return a / s

            a = cand

                cnd = a * np.exp(rng.normal(0.0, 0.15, n))
                a = _simplex_project(cnd)
                stall = 0
        it += 1
    return best_a, best_sc
