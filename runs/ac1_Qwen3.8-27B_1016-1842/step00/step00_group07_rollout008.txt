import numpy as np
import time


def fft_conv(a):
    n = len(a)
    n_out = 2 * n - 1
    n_fft = 1 << (n_out - 1).bit_length()
    A = np.fft.rfft(a, n_fft)
    return np.fft.irfft(A * A, n_fft)[:n_out]


def quick_score(a):
    S = a.sum()
    if S < 0.01:
        return np.inf
    C = fft_conv(a)
    return 2.0 * len(a) * C.max() / (S * S)


def apply_ramp(base, gamma, kind):
    n = base.size
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    if kind == 0:
        mult = 1.0 + gamma * (2.0 * t - 1.0)
    elif kind == 1:
        mult = np.exp(gamma * (t - 0.5))
    elif kind == 2:
        mult = 1.0 + gamma * np.sin(2.0 * np.pi * t)
    else:
        mult = 1.0 + gamma * (t * t - t + 0.25)
    return np.clip(base * mult, 0.0, 1000.0)


def peak_perturb(base, rng):
    n = base.size
    C = fft_conv(base)
    k_st = int(np.argmax(C))
    j_lo = max(0, k_st - n + 1)
    j_hi = min(n - 1, k_st)
    js = np.arange(j_lo, j_hi + 1)
    paired = k_st - js
    a_j = base[js]
    a_p = base[paired]
    cand = base.copy()
    eps = 0.005 + 0.03 * rng.random()
    mx = a_p.max()
    if mx > 1e-15:
        fac = np.maximum(1.0 - eps * a_p / mx, 0.05)
        cand[js] = a_j * fac
    if rng.random() < 0.5:
        w = int(rng.integers(max(1, n // 300), max(2, n // 10)))
        st = int(rng.integers(0, n - w))
        cand[st:st + w] *= np.float64(np.exp(rng.normal(0.0, 0.06, w)))
    return np.clip(cand, 0.0, 1000.0)


def random_block(base, rng):
    n = base.size
    cand = base.copy()
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        w = int(rng.integers(max(1, n // 500), max(2, n // 8)))
        st = int(rng.integers(0, n - w))
        rng_val = float(np.exp(rng.normal(0.0, 0.1, w)))
        cand[st:st + w] *= rng_val
    if rng.random() < 0.3:
        w2 = int(rng.integers(max(1, n // 500), max(2, n // 8)))
        st2 = int(rng.integers(0, n - w2))
        cand[st2:st2 + w2] *= float(np.exp(rng.normal(0.0, 0.1, w2)))
    return np.clip(cand, 0.0, 1000.0)
