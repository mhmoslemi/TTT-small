import numpy as np
from scipy import sparse

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a, b):
    """Linear convolution via FFT, O(n log n) memory and time."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def build_constraint_rows(a, active):
    """Build dense A_ub rows for active peak indices. O(m*n) storage."""
    n = len(a)
    m = len(active)
    A = np.zeros((m, n), dtype=np.float64)
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]
    return A


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 5.0)

    n = len(height_sequence_1)
    if n == 0:
        return [1.0]

    initial = np.asarray(height_sequence_1, dtype=np.float64).copy()

    best_score = float(evaluate_sequence(best.tolist()))

    a = initial.copy()
    a += 0.10 * rng.standard_normal(n)
    a = np.clip(a, 1e-3, 1000.0)

        a += 0.15 * rng.standard_normal(n)
        a = np.clip(a, 1e-3, 1000.0)

    for outer in range(200):

            break
        remaining = deadline - time.monotonic()
        if remaining < 15.0:
            break

        b = fft_conv(a, a)
        M = float(b.max())
        S = float(a.sum())
        if S < 0.01 or M <= 1e-12:

        # LP: maximise sum(g)  s.t.  g >= 0,  (a * g)_k <= M  for all k
