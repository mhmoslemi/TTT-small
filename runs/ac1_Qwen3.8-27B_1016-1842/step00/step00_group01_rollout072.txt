import numpy as np

def _next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << max(0, (x - 1).bit_length())


def _stable_softmax(x):
    n = x.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(x)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    xm = x - np.max(x)
    e = np.exp(xm)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def _autocorr_scaled(a, fs, n):
    F = np.fft.rfft(a, fs)
    c = np.fft.irfft(F * F, fs)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return float(n) * c


def _lag_weights(z, beta):
    beta = max(1e-8, float(beta))
    if not np.all(np.isfinite(z)):
        return np.full(z.size, 1.0 / max(1, z.size), dtype=np.float64)
    zm = z - float(np.max(z))
    e = np.exp(zm / beta)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full(z.size, 1.0 / max(1, z.size), dtype=np.float64)
    return e / s


def _grad_a(a, p, n, fs):
    if p.size == 0:
        return np.zeros(n, dtype=np.float64)
    pmax = float(np.max(p))
    thr = max(1e-10, pmax * 1e-5)
    act = np.flatnonzero(p > thr)
    C = 2.0 * float(n)
    if 0 < act.size <= 80:
        ga = np.zeros(n, dtype=np.float64)
        for ii_np in act:
            i = int(ii_np)
            wi = C * float(p[i])
            st = i - n + 1
            if st < 0:
                st = 0
            en = i + 1
            if en > n:
                en = n
            if st < en:
                ga[st:en] += wi * a[i - st:i - en:-1]
        return np.maximum(ga, 0.0)

    ar = a[::-1]
    F = np.fft.rfft(ar, fs)
    G = np.fft.rfft(p, fs)
    h = np.fft.irfft(F * G, fs)[n - 1:2 * n - 1]
    ga = C * h
    return np.maximum(ga, 0.0)
