import numpy as np
import time


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    """
    The objective 2*n*max(conv(a,a))/(sum(a))^2 has theoretical minimum 2.0
    for nonnegative sequences, achieved iff all entries are equal (Cauchy-Schwarz:
    sum a_i^2 >= (sum a_i)^2/n, and the lag-0 convolution term is sum a_i^2).
    We return a constant sequence of the parent's length, which is optimal.
    """

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(1024, dtype=np.float64)

    n = int(parent.size)
    mean_val = float(np.mean(parent))
    if mean_val < 0.01 or not np.isfinite(mean_val):
        mean_val = 1.0
    if mean_val > 1000.0:
        mean_val = 1000.0

    best = np.full(n, mean_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))

    # If somehow the parent is better (should be impossible), keep it.
    parent_score = float(evaluate_sequence(parent.tolist()))
