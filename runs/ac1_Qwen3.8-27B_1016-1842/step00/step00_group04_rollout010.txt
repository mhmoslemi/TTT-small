import numpy as np
import time
import math

def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.full(n, 1.0/n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_score_packed(p):
    """Compute 2n * max(conv(p,p)) for p in simplex. Returns (score, max_c)."""
    c = np.convolve(p, p, mode='full')
    max_c = c.max()
    n = len(p)
    return 2.0 * n * max_c, max_c, c

def subgradient_iteration(p, eta, momentum=0.9):
    """One projected subgradient step for min max_k (p*p)[k]."""
    n = len(p)
    c = np.convolve(p, p, mode='full')

    tol = max_val * 1e-8 + 1e-15
    active = np.nonzero(c >= max_val - tol)[0]

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = p[k-hi:k-lo+1][::-1]
        g[lo:hi+1] += 2.0 * seg
    g /= len(active)
    return g, max_val, c

        p_new
