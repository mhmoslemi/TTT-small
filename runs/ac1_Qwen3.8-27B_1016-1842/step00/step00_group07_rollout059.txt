import numpy as np
import time


def _score(a):
    """Exact score via the authoritative evaluator."""
    return float(evaluate_sequence(a.tolist()))


def _power_law(n, alpha):
    """Generate a power-law ramp of length n."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, alpha)
    return np.clip(a, 0.0, 1000.0)


def _block_perturb(base, rng, n_blocks):
    """Apply random block multiplicative perturbations."""
    n = base.size
    c = base.copy()
    for _ in range(n_blocks):
        lo_w = max(1, n // 400)
        hi_w = max(lo_w + 1, n // 8)
        width = int(rng.integers(lo_w, hi_w))
        start = int(rng.integers(0, n - width + 1))
        scale = float(np.exp(rng.normal(0.0, 0.12)))
        c[start:start + width] *= scale
    return np.clip(c, 0.0, 1000.0)


def _shape_perturb(base, rng, n_blocks):
    """Perturb by blending with a slightly different power law in a region."""
    n = base.size
    c = base.copy()
    for _ in range(n_blocks):
        width = int(rng.integers(max(2, n // 40), max(3, n // 4)))
        start = int(rng.integers(0, n - width + 1))
        # Fit local slope and perturb it
        seg = c[start:start + width]
        if seg.min() > 0.01:
            log_seg = np.log(np.maximum(seg, 1e-6))
            x = np.arange(width, dtype=np.float64)
            slope = float(rng.normal(0.0, 0.02))
            log_seg = log_seg + slope * (x / width)
            c[start:start + width] = np.exp(log_seg)
    return np.clip(c, 0.0, 1000.0)

    deadline = time.monotonic() + max(1.0, float(budget_s) - 8.0)

    parent = None
    try:

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all():
            parent = np.clip(p, 0.0, 1000.0)
            n = parent.size
    except Exception:
        pass

    best = None
    best_score = np.inf

    # Score parent if valid
    if parent is not None:
        s = _score(parent)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = parent.copy()

    # Try power-law initializations (scale-invariant, so no need to tune amplitude)
    alphas = list(np.arange(0.30, 0.79, 0.025))
    for alpha in alphas:
        a = _power_law(n, alpha)
        if a.sum() < 0.01:
            continue
        s = _score(a)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = a.copy()

    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = _score(best)
        if not np.isfinite(best_score):
            best = np.ones(1, dtype=np.float64)
            best_score = _score(best)

        else:
