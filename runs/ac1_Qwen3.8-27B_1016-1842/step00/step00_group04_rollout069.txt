from scipy import sparse
from scipy.optimize import linprog


def build_conv_matrix(a, n):
    j = np.arange(n)
    rows = (j[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(j, n)
    data = np.tile(a, n)
    return sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def simplex_project(v, s):
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.arange(1, v.size + 1)
    cond = v_sorted > (cumsum - s) / rho
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(v.size, s / v.size)
    j = idx[0]
    theta = (cumsum[j] - s) / (j + 1)
    return np.maximum(v - theta, 0.0)


def subgrad_step(a, k_star, lr, n):

    grad = np.zeros(n)
