        conv = fftconvolve(a, a)

        if s > 1e-10:

        else:
            a = np.ones(n) / n
            break

            best = a.copy()

        if rng.random() < 0.1:

from scipy.signal import fftconvolve
import time

def _normalize(a):
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s

def _score_from_conv(conv, n):
    return 2.0 * n * np.max(conv)

def _grad_for_peaks(a, peaks, n):
    g = np.zeros(n)
    for k in peaks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return g

        parent = np.ones(n)

    a = parent / parent.sum()

    best_score = float(evaluate_sequence(best.tolist()))

    T = min(5, n)  # number of active peaks
    lr = 1.0 / n
    stall = 0

        conv = fftconvolve(a, a)
        # Find top T peaks
        peaks_idx = np.argpartition(conv, -T)[-T:]
        peak_vals = conv[peaks_idx]
        # Weight peaks by their value
        weights = peak_vals / (peak_vals.sum() + 1e-15)
        
        g = np.zeros(n)
        for k, w in zip(peaks_idx, weights):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)

            conv_try = fftconvolve(a_try, a_try)
            score_try = _score_from_conv(conv_try, n)

                stall = 0
                cur_score = score_try
