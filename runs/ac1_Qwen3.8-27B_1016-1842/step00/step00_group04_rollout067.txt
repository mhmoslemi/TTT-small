import numpy as np
import time

def compute_score_fast(a):
    """Fast score computation without calling evaluate_sequence."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sa = a.sum()
    if sa < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sa * sa)

def find_max_lag(a):
    """Find the lag where the convolution maximum occurs."""
    b = np.convolve(a, a)
    return int(np.argmax(b)), float(b.max())
