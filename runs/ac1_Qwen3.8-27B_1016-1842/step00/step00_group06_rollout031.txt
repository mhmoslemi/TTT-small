from scipy.sparse import csr_matrix

def solve_lp(a, n):
    # Primal: max 1^T g s.t. A g <= 1, g >= 0
    # A[k,j] = a[k-j] if 0 <= k-j <= n-1
    
    # Build sparse A
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j+n):
            if k < 2*n-1:
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
