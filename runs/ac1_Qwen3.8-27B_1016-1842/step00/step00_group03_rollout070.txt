import numpy as np
import time


def _next_pow2(m):
    m = max(1, int(m))
    n = 1
    while n < m:
        n <<= 1
    return n


def _smooth_grad(a, N, beta_scale):
    n = a.size
    F = np.fft.rfft(a, N)
    b = np.fft.irfft(F * np.conj(F), N)
    lim = 2 * n - 1
    b = b[:lim]
    if not np.isfinite(b).all():
        return np.zeros(n), 0.0, 0, 1.0
    mb = float(np.max(b))
    if mb <= 1e-300:
        return np.zeros(n), 0.0, 0, 1.0
    bt = beta_scale / mb
    wb = np.exp(bt * (b - mb))
    wsum = float(np.sum(wb))
    wfull = np.zeros(N)
    wfull[:lim] = wb
    W = np.fft.rfft(wfull, N)
    C = np.fft.irfft(W * np.fft.rfft(a[::-1], N), N)
    g = C[(n - 1):(2 * n - 1)]
    g = (2.0 * g) / max(wsum, 1e-300)
    return g, mb, int(np.argmax(b)), wsum

    pscore = 2.0 * n * pm if pm > 0 else 2.0
    best = [parent.copy(), pscore if np.isfinite(pscore) else 2.0]

    configs = [(0.5, 0.5), (0.3, 0.5), (1.0, 0.5), (0.5, 0.4),
               (1.0, 0.6), (2.0, 0.7), (0.5, 0.7), (3.0, 0.5)]
    starts = []
    for cc, aa in configs:
        vct = _build(cc, aa, n)
        m = _maxb(vct, N, n)
        sc = 2.0 * n * m / (float(np.sum(vct)) ** 2)
        starts.append((vct, sc, (cc, aa)))
    starts.append((parent, pscore if np.isfinite(pscore) else 2.0, None))
    starts.sort(key=lambda z: z[1])

    # small local refine of the best parametric start
    if starts[0][2] is not None:
        bc, ba = starts[0][2]
        bc0, sc0 = starts[0][1]
        for cc in (bc * 0.7, bc, bc * 1.3):
            for aa in (ba - 0.05, ba, ba + 0.05):
                cand = _build(cc, aa, n)
                m = _maxb(cand, N, n)
                sc = 2.0 * n * m / (float(np.sum(cand)) ** 2)
                if np.isfinite(sc) and sc < sc0:
                    starts[0] = (cand, sc, (cc, aa))

    if np.isfinite(starts[0][1]) and starts[0][1] < best[1]:
        best[1] = starts[0][1]
        best[0] = starts[0][0].copy()

    head = max(6.0, 0.03 * float(budget_s))
    t_opt = t_start + max(1.0, float(budget_s) - head)
    dtot = max(1.0, t_opt - t_start)
    t1 = t_start + 0.70 * dtot
    t2 = t_start + 0.88 * dtot
    _trail(starts[0][0], n, N, t1, rng, 0.6, 0.6, best)
    if len(starts) > 1:
        _trail(starts[1][0], n, N, t2, rng, 0.9, 0.5, best)
    if len(starts) > 2:
        _trail(starts[2][0], n, N, t_opt, rng, 1.2, 0.4, best)

    # light smoothing pass on current best (kept only if it improves)
    cur = best[0]
    for k in (5, 11, 21, 51):
        if k >= n - 1:
            continue
        kern = np.ones(k) / float(k)
        sm = np.convolve(cur, kern, mode="same")
        sm = np.maximum(sm, 0.0)
        ss = float(np.sum(sm))
        if not np.isfinite(ss) or ss < 1e-9:
            continue
        sm = sm / ss
        m = _maxb(sm, N, n)
        sc = 2.0 * n * m
        if np.isfinite(sc) and sc < best[1]:
            best[1] = sc
            best[0] = sm
            cur = sm

    # authoritative confirmation of the returned candidate
    final = np.clip(best[0], 0.0, 1000.0)
    try:
        fs = float(evaluate_sequence(final.tolist()))
    except Exception:
        fs = np.inf
    if np.isfinite(fs):
        best_out = [float(x) for x in final]
    else:
        fp = float(evaluate_sequence(parent.tolist()))
        best_out = [float(x) for x in parent] if np.isfinite(fp) else [float(x) for x in np.ones(n)]

    return best_out
