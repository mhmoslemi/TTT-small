def do_conv(a):
    n = len(a)
    size = 2 * n
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def grad_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    j1 = max(0, k - n + 1)
    j2 = min(k, n - 1)

    return g
