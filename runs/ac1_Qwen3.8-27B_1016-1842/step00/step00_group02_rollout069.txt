def score_params(c, basis, n, tanh_limit):
    z = np.dot(basis, c)
    seq = 1.0 + tanh_limit * np.tanh(z)
    # Because tanh_limit <1, seq positive; no clip needed but guard
    s = seq.sum()
    if s <= 0.0: return np.inf
    # scale to sum n to keep numeric stable; score invariant. Avoid overflow
    seq = seq * (float(n) / s)
    # convolve via FFT
    m = 2*n - 1
    L = 1 << int((m - 1).bit_length())
    f = np.fft.rfft(seq, L)
    b = np.fft.irfft(f * f, L)[:m]
    maxb = float(np.max(b))
    if maxb < 0: maxb = 0
    if not np.isfinite(maxb): return np.inf
    return 2.0 * float(n) * maxb / (float(n)*float(n))  # since sum=n after scale

def _fft_len(m):
    if m <= 1: return 1
    return 1 << int((m - 1).bit_length())

def _fast_score(seq, n, fft_len):
    n=int(n)
    if n <= 0: return float('inf')
    f=np.fft.rfft(seq, fft_len)
    conv=np.fft.irfft(f*f, fft_len)
    end=2*n-1
    if conv.size < end:
       end=conv.size
    if end <=0: return float('inf')
    mb=float(np.max(conv[:end]))
    if mb <0: mb=0.0
    if not np.isfinite(mb): return float('inf')
    return 2.0*float(n)*mb/float(n*n)

def _project_to_basis(x, basis):
    P=basis.shape[1]
    x=np.asarray(x, dtype=np.float64)
    x=x - float(np.mean(x))

class _SearchTimeout(Exception):

def _fft_len(m):
    if m <= 1:
        return 1
    return 1 << int((m - 1).bit_length())


def _fast_score(seq, n, fft_len):
    n = int(n)
    if n <= 0:
        return float("inf")
    f = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(f * f, fft_len)
    end = 2 * n - 1
    if conv.size < end:
        end = conv.size
    if end <= 0:
        return float("inf")
    mb = float(np.max(conv[:end]))
    if mb < 0.0:
        mb = 0.0
    if not np.isfinite(mb):
        return float("inf")
    return 2.0 * float(n) * mb / (float(n) * float(n))


def _sequence_from_c(c, basis, n):
    n = int(n)
    z = basis.dot(c)
    z = np.clip(z, -50.0, 50.0)
    seq = 1.0 + np.tanh(z)
    if not np.all(np.isfinite(seq)):
        seq = np.ones(n, dtype=np.float64)
    else:
        seq = np.clip(seq, 1e-12, 1000.0)
    s = float(seq.sum())
    if s <= 1e-12:
        seq = np.ones(n, dtype=np.float64)
    else:
        seq = seq * (float(n) / s)
    return seq


def _state_score(c, basis, n, fft_len, state):
    if time.monotonic() > state[2]:
        raise _SearchTimeout()

        return float("inf")

    Amat = Amat + 1e-10 * np.eye(bvec.size)
    try:
        c = np.linalg.solve(Amat, bvec)

        c = np.linalg.lstsq(Amat, bvec, rcond=None)[0]
    c = np.asarray(c, dtype=np.float64)
    nrm = float(np.linalg.norm(c))
    if nrm > 2.0:
        c *= 2.0 / nrm
    if not np.all(np.isfinite(c)):
        c = np.zeros_like(c)
    return c

            return float("inf")
        return float(evaluate_sequence([float(x) for x in a]))
