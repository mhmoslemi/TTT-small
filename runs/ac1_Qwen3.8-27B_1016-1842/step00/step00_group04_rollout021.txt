import numpy as np
import time


def make_power_ramp(n, p, shift=0.0):
    j = np.arange(n, dtype=np.float64)
    return ((j + shift) / (n - 1 + shift)) ** p


def make_sym_ramp(n, p_dir, p_und):
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1)
    a = t ** p_dir
    # multiply by a decay from the right
    a *= (1 - t) ** p_und
    return a


def make_bump(n, p, center=0.5):
    j = np.arange(n, dtype=np.float64)
    t = np.abs(j / (n - 1) - center) * 2
    t = np.clip(t, 0, 1)
    return (1 - t) ** p


def block_perturb(base, rng, n):
    cand = base.copy()
    n_width = max(1, n // 200)
    width = int(rng.integers(n_width, max(n_width + 1, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    log_f = rng.normal(0.0, 0.08)
    cand[start:stop] *= np.exp(log_f)
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    if parent.size == 0 or not np.isfinite(parent).all():

        n = parent.size

    best_score = float(evaluate_sequence(best.tolist()))

    for p in [2, 3, 4, 5, 6, 7, 8, 10, 15]:
        for shift in [0.0, 0.5, 1.0, 2.0, 5.0]:

            candidates.append(a)
    for p_d in [1, 2, 3]:

            a = make_sym_ramp(n, p_d, p_u)
            candidates.append(a)
    for p in [1, 2, 3, 4, 5]:

            candidates.append(a)

import numpy as np
import time


def make_ramp(n, p, shift=0.0):
    j = np.arange(n, dtype=np.float64)
    return ((j + shift) / (n - 1.0 + shift)) ** p


def make_double_ramp(n, p1, p2):
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1.0)
    return t ** p1 * (1.0 - t) ** p2


def make_asym_bump(n, p, peak_frac):
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1.0)
    a = np.where(t < peak_frac,
                 (t / peak_frac) ** p,
                 ((1.0 - t) / (1.0 - peak_frac)) ** p)
    return a


def block_perturb(base, rng, n, sigma):
    cand = base.copy()
    lo = max(1, n // 300)
    hi = max(lo + 1, n // 6)
    width = int(rng.integers(lo, hi))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, sigma))
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    n = len(height_sequence_1)

    if parent.size == 0 or not np.isfinite(parent).all():
        n = 128
        parent = np.ones(n, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    inits = []
    for p in [2, 3, 4, 5, 6, 8, 10, 14, 20]:
        for sh in [0.0, 1.0, 5.0, 50.0, 500.0]:
            inits.append(make_ramp(n, p, sh))
    for p1 in [1, 2, 3, 4]:
        for p2 in [1, 2, 3, 5]:
            inits.append(make_double_ramp(n, p1, p2))
    for p in [2, 3, 4, 5, 7]:
        for pf in [0.3, 0.4, 0.5, 0.6, 0.7]:
            inits.append(make_asym_bump(n, p, pf))

    for a in inits:
        if time.monotonic() > deadline - 20.0:

        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
