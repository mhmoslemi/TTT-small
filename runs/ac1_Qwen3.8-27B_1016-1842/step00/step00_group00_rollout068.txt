def _autocorr(x):
    n = len(x)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X.conj(), N)[:2 * n - 1]

def _subgrad_peak(a, k):
