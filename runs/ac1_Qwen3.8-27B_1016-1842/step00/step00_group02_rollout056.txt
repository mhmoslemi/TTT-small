import time

def _fc(a):
    n = a.size
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)

def _pj(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.size, 1.0 / a.size)
    return a / s

def _fs(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return 1e18
    c = _fc(a)
    m = float(c.max())
    if m < 0.0:
        m = 0.0
    return 2.0 * a.size * m / (s * s)

def _sg(a, c, tk):
    n = a.size
    mx = float(c.max())
    if tk >= c.size:
        act = np.arange(c.size)
    else:
        act = np.argpartition(c, -tk)[-tk:]
    sg = np.zeros(n)
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        al = k - hi
        ah = k - lo
        sg[lo:hi + 1] += a[al:ah + 1][::-1]
    sg /= float(len(act))
    return sg, mx
