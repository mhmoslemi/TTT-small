def compute_subgradient(a, c, M, tol=1e-10):
    n = len(a)
    grad = np.zeros(n)
    active = np.where(c >= M - tol * M)[0]
    for k in active:
        # Contribution from a[k-j]: j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a[k-j] for j in [j_lo, j_hi]: indices k-j_hi to k-j_lo
            idx = k - j_hi + np.arange(j_hi - j_lo + 1)
            grad[j_lo:j_hi+1] += a[idx]
        
        # Contribution from a[j-k]: j ranges from max(0, k) to min(n-1, k+n-1)
        j_lo2 = max(0, k)

            grad[j_lo2:j_hi2+1] += a[idx2]
