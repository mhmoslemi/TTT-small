            break
        w = int(rng.integers(3, max(4, n // 80)))
        s_pos = int(rng.integers(0, n - w))
        a2 = a.copy()
        a2[s_pos:s_pos + w] *= np.exp(rng.normal(0.0, 0.07, w))
        a2 = np.maximum(a2, 0.0)
        ts = float(a2.sum())
        if ts < 1e-12:
            continue
        a2 = a2 / ts
        new_maxb = float(fft_full_convolve(a2).max())
        if new_maxb < best_maxb:
            a = a2
            best_maxb = new_maxb
            best = a2 * orig_sum
        elif rng.random() < 0.03:
            a = a2

    # Phase 3: Symmetrize and re-optimize if helpful
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    if a_sym.sum() > 1e-12:
        a_sym = a_sym / a_sym.sum()
        sym_maxb = float(fft_full_convolve(a_sym).max())
        if sym_maxb < best_maxb:
            best = a_sym * orig_sum
            best_maxb = sym_maxb

    # Final validation
    final_score = float(evaluate_sequence(best.tolist()))
    orig_score = float(evaluate_sequence(parent.tolist()))
    if (not np.isfinite(final_score)) or final_score >= orig_score:
        best = parent

    return [float(x) for x in best]
