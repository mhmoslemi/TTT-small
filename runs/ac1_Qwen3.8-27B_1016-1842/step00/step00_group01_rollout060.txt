        if a.sum() < 1e-6:
            continue
        a = a / a.sum()
        c = _fft_conv(a, a, fft_n)
        sc = 2.0 * n * c.max()
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if best is None:
        best = np.ones(n) / n
        best_sc = 2.0

    # Soft-gradient descent with Nesterov momentum
    a = best.copy()
    v_mom = np.zeros(n)
    lr = 0.3
    beta = 200.0
    mom = 0.85
    stagnant = 0
    best_a = best.copy()
    best_fft_sc = best_sc

    for it in range(800000):
        if time.monotonic() > deadline:
            break

        # Convolve a with a
        ca = np.fft.rfft(a, fft_n)
        c = np.fft.irfft(ca * ca, fft_n)[:nconv]
        cmax = c.max()

        # Soft-max weights
        cs = c - cmax
        w = np.exp(beta * cs)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(nconv)
            w[int(cmax == c).nonzero()[0][0]] = 1.0
        else:
            w /= ws

        # Gradient: grad[j] = cross[j] + a[j]
        # cross[j] = sum_m a[m]*w[m+j] (zero-padded)
        cw = np.fft.rfft(w, fft_n)
        cross = np.fft.irfft(np.conj(ca) * cw, fft_n)[:n]
        grad = cross + a

        # Momentum step
        v_mom = mom * v_mom - lr * grad
        a_new = a + v_mom
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        # Evaluate new point
        cn_arr = np.fft.rfft(a_new, fft_n)
        cn = np.fft.irfft(cn_arr * cn_arr, fft_n)[:nconv]
        cnmax = cn.max()

        if cnmax < cmax * 1.0 - 1e-15:
            a = a_new
            sc_new = 2.0 * n * cnmax
            if sc_new < best_fft_sc - 1e-10:
                best_fft_sc = sc_new
                best_a = a.copy()
                stagnant = 0
            else:
                stagnant += 1
        else:
            lr *= 0.75
            v_mom *= 0.3
            stagnant += 1

        # Adapt parameters
        if stagnant > 300:
            beta = min(2000.0, beta * 1.4)
            lr = max(0.005, lr * 0.5)
            v_mom *= 0.1
            stagnant = 0
            if lr < 0.003:
                break

        # Periodic exact check
        if it % 2000 == 0 and it > 0:
            sc_exact = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_exact) and sc_exact < best_sc - 1e-10:
                best_sc = sc_exact
                best = a.copy()

    # Final exact evaluation of candidates
    sc1 = float(evaluate_sequence(best_a.tolist()))
    sc2 = float(evaluate_sequence(best.tolist()))
    if np.isfinite(sc1) and (not np.isfinite(sc2) or sc1 < sc2):
        final = best_a
    else:
        final = best

    if not np.isfinite(float(evaluate_sequence(final.tolist()))):
        return [1.0 / n] * n
    return [float(x) for x in final]
