import numpy as np
import time

def compute_conv_and_subgrad(a):
    """Compute max of conv(a,a) and corresponding subgradient of max_b/S^2."""
    n = len(a)
    c = np.convolve(a, a, mode='full')
    k_star = int(np.argmax(c))
    max_b = c[k_star]
    S = np.sum(a)
    
    # Subgradient of max_b / S^2
    grad = np.full(n, -2.0 * max_b / (S ** 3))
    m_min = max(0, k_star - (n - 1))
    m_max = min(n - 1, k_star)
    if m_min <= m_max:
        idx = np.arange(m_min, m_max + 1)
        shifted = k_star - idx  # this is k_star - m
        grad[idx] = 2.0 * (a[shifted] * S - max_b) / (S ** 3)
    
    return max_b, S, grad
