def fft_conv_sq(a):
    n = len(a)
    L = np.fft.next_fast_len(2 * n - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]


def compute_h(a, w):
    n = len(a)
    L = np.fft.next_fast_len(3 * n)
    fa = np.fft.rfft(a[::-1], L)
    fw = np.fft.rfft(w, L)
    conv = np.fft.irfft(fa * fw, L)
