- element-wise update: O(n)

Let me design the algorithm:

1. Start from `height_sequence_1`, normalize to sum = sqrt(2n).
2. Apply a small monotone (or normed-noise) perturbation to break symmetry.
3. Iterate:
   - compute c = convolve(f, f)
   - k* = argmax(c)
   - compute grad[i] = f[k*-i] - c[k*]/S (where f[m]=0 if m not in [0,n-1])
   - f_new = f - eta * grad; clamp to >=0
   - if sum(f_new) < tiny: break
   - renormalize f_new *= sqrt(2n)/sum(f_new)
   - if time: break
   - evaluate score occasionally or after improvement accepted (since evaluation is similar cost to step, we can use a surrogate)
   - actually, since our update minimizes the *surrogate* (the peak coordinate at k*), we can evaluate every K steps to track best.
   
Use a learning rate schedule: start eta moderate, decay slowly. 

Simplest: after each step, check time, save best whenever the surrogate (score at current k*) decreases. We can periodically call evaluate_sequence to keep exact best.

The number of evaluations is expensive — each is O(n log n). Let me just compute c once per step and use it as both the gradient info AND the score at current k*. Since score = 2n*c[k*]/S², and S is fixed (sqrt(2n)), score = 2n*c[k*]/(2n) = c[k*]. Oh nice!

import numpy as np
import time


def _safe_sequence(x):
    a = np.asarray(x, dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 5.0)

    f0 = _safe_sequence(height_sequence_1)
    n = f0.size
    if n == 0:
        f0 = np.ones(4, dtype=np.float64)
        n = 4

    target_sum = float(np.sqrt(2.0 * n))

    def normalize(f):
        s = float(f.sum())
        if s < 1e-12:
            return None
        return f * (target_sum / s)

    # Fallback: uniform
    uni = np.full(n, target_sum / n, dtype=np.float64)
    cu = np.convolve(uni, uni, mode='full')
    best_m = float(cu.max())
    best_f = uni.copy()

    # Start from parent if valid; otherwise uniform
    if f0.sum() > 1e-9:
        cur = normalize(f0)
        if cur is not None:
            cc = np.convolve(cur, cur, mode='full')
            if float(cc.max()) < best_m:
                best_m = float(cc.max())
                best_f = cur.copy()
            f = cur.copy()
        else:
            f = uni.copy()
    else:
        f = uni.copy()

    # Helper: one gradient step on current f, using precomputed k* and peak.
    # Returns updated f (may renormalize). Also returns peak before step.
    def step(f, k, peak, eta):
        S = target_sum
        idx_p = k - np.arange(n)
        g = np.zeros(n)
        valid = (idx_p >= 0) & (idx_p < n)
        g[valid] = f[idx_p[valid]]
        g -= peak / S
        fn = f - (2.0 * eta) * g
        fn = np.maximum(0.0, fn)
        fn = np.minimum(1000.0, fn)
        rn = normalize(fn)
        if rn is None:
            return None
        return rn

    def perturb(f, amp):
        # Asymmetric perturbation: ramp + noise
        idx = np.arange(n, dtype=np.float64)
        ramp = (idx / max(1.0, float(n - 1))) - 0.5
        noise = rng.normal(0.0, 1.0, size=n)
        f2 = f + amp * (0.6 * ramp + 0.4 * noise)
        return normalize(np.maximum(0.0, np.minimum(1000.0, f2)))

    def try_accept(f):
        nonlocal best_f, best_m
        if f is None:
            return
        cc = np.convolve(f, f, mode='full')
        m = float(cc.max())
        if m < best_m - 1e-12:
            best_m = m
            best_f = f.copy()

    # Phase 1: initial perturbation from uniform-like state
    amp0 = 0.35 * max(1.0, float(np.std(f)))
    f = perturb(f, amp0)
    try_accept(f)

    # Phase 2: gradient descent with restarts
    eta = 0.4
    decay = 0.99999
    min_eta = 1e-4
    stall = 0
    STALL_LIMIT = 600
    last_restart_score = best_m

    last_t = time.monotonic()
    step_count = 0
    while time.monotonic() < deadline:
        c = np.convolve(f, f, mode='full')
        k = int(np.argmax(c))
        peak = float(c[k])

        if peak < best_m - 1e-12:
            best_m = peak
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        fn = step(f, k, peak, eta)
        if fn is None:
            f = best_f.copy()
            eta = max(min_eta, eta * 0.5)
            continue
        f = fn
        eta = max(min_eta, eta * decay)
        step_count += 1

        if stall > STALL_LIMIT:
            # Restart: jump back to best, add fresh perturbation, reset eta
            f = best_f.copy()
            amp = 0.15 * max(1.0, float(np.std(f)))
            fp = perturb(f, amp)
            if fp is not None:
                f = fp
            try_accept(f)
            eta = 0.4
            stall = 0
            # If we've been stuck with same best for a while, try bigger kick
            if time.monotonic() - last_t > 60.0 and best_m == last_restart_score:
                amp = 0.5 * max(1.0, float(np.std(f)))
                fp = perturb(f, amp)
                if fp is not None:
                    f = fp
                try_accept(f)
                last_t = time.monotonic()
                last_restart_score = best_m

    # Final exact evaluation of the best candidate
    cand = [float(x) for x in best_f]
    sc = float(evaluate_sequence(cand))
    if not np.isfinite(sc):
        cand = [float(x) for x in uni]
    return cand
