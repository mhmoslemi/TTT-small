import numpy as np
import time

def simplex_project(v, s):
    """Project v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # Binary search for the threshold
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - s) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[0]
    theta = (cumsum[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def compute_subgrad(a, k_star):
    """Compute subgradient of max autocorrelation at lag k_star."""
    n = a.size
    # grad[m] = 2*a[k_star - m] for valid indices, with correction
    idx_hi = k_star - np.arange(n)
    valid = (idx_hi >= 0) & (idx_hi < n)
    grad = np.zeros(n)
    j = idx_hi[valid]
    i = np.where(valid)[0]
    grad[i] = 2.0 * a[j]
    # Correct diagonal: if k_star = 2*m, grad[m] should be a[m] not 2*a[m]
    if k_star % 2 == 0:
        m = k_star // 2
        if 0 <= m < n:
            grad[m] = a[m]
    return grad

    """Compute autocorrelation and return (peak, argmax)."""
    b = np.convolve(a, a)
    k_star = int(np.argmax(b))
    return float(b[k_star]), k_star
