```python
import numpy as np
import time


def fft_conv_self(x, p):
    X = np.fft.rfft(x, p)
    return np.fft.irfft(X * X, p)


def fft_xcorr(h, f, p2):
    H = np.fft.rfft(h, p2)
    F = np.fft.rfft(f, p2)
    return np.fft.irfft(H * np.conjugate(F), p2)


def softmax_grad(f, c, n, T):
    shifted = c - c.max()
    h = np.exp(shifted / T)
    hsum = h.sum()
    if hsum < 1e-300:
        return np.zeros(n)
    h = h / hsum
    p2 = 1
    while p2 < 3 * n:
        p2 *= 2
    xc = fft_xcorr(h, f, p2)
    return xc[:n]


def score_from_c(c, n):
    return 2.0 * n * float(c.max())


def make_init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def make_init_sinusoid(n, freq, amp):
    t = np.arange(n, dtype=np.float64)
    return (1.0 + amp * np.sin(2.0 * np.pi * freq * t / n)) / n


def make_init_gauss(n, width):
    t = (np.arange(n, dtype=np.float64) - 0.5 * n) / n
    return np.exp(-width * t * t)
    # will be normalized later


def make_init_chirp(n, amp):
    t = np.arange(n, dtype=np.float64)
    return 1.0 + amp * np.cos(2.0 * np.pi * t * t / n)


def normalize(f):
    s = f.sum()
    if s < 1e-12:
```
