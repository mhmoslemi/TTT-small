    # Work in normalized space (sum=1) since objective is scale-invariant
    s = np.sum(best)
    if s < 1e-12:
        best_n = np.ones(n) / n
    else:
        best_n = best / s

    # Phase 1: structured initializations
    structs = []
    structs.append(np.arange(1, n + 1, dtype=np.float64))
    structs.append(np.power(np.arange(1, n + 1, dtype=np.float64), 0.5))
    structs.append(np.power(np.arange(1, n + 1, dtype=np.float64), 2.0))
    structs.append(np.exp(-np.arange(n, dtype=np.float64) / (n * 0.3)) + 0.05)
    structs.append(np.exp(-np.arange(n - 1 - np.arange(n), dtype=np.float64) / (n * 0.3)) + 0.05)
    structs.append(np.abs(np.sin(np.pi * np.arange(n, dtype=np.float64) / n)) + 0.1)
    structs.append(np.cos(np.pi * np.arange(n, dtype=np.float64) / (2.0 * n + 1)) ** 2 + 0.05)
    tl = np.ones(n, dtype=np.float64)
    tl[:n // 3] = 2.0
    tl[n - n // 3:] = 0.3
    structs.append(tl)
    structs.append(tl[::-1].copy())
    for c in structs:
        c = np.maximum(c, 0.0)
        sc = np.sum(c)
        if sc < 1e-12:
            continue
        cn = c / sc
        score = _eval(cn)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_n = cn.copy()

    # Phase 2: local search with simulated annealing
    T = 0.015
    it = 0
    while time.monotonic() < deadline:
        it += 1
        x = best_n.copy()
        mv = rng.integers(0, 6)

        if mv == 0:
            w = int(rng.integers(max(2, n // 40), n // 3 + 1))
            st = int(rng.integers(0, n - w))
            x[st:st + w] *= np.exp(rng.normal(0.0, 0.35))
        elif mv == 1:
            f = int(rng.integers(1, max(2, n // 5) + 1))
            amp = float(rng.uniform(0.03, 0.4))
            ph = float(rng.uniform(0, 6.2832))
            x *= 1.0 + amp * np.cos(2.0 * np.pi * f * np.arange(n) / n + ph)
        elif mv == 2:
            w1 = int(rng.integers(max(2, n // 200), n // 5 + 1))
            w2 = int(rng.integers(max(2, n // 200), n // 5 + 1))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            bl = float(rng.uniform(0.2, 0.8))
            tmp = x[s1:s1 + w1].copy()
            x[s1:s1 + w1] = bl * tmp + (1.0 - bl) * x[s2:s2 + w2]
            x[s2:s2 + w2] = (1.0 - bl) * tmp + bl * x[s2:s2 + w2]
        elif mv == 3:
            k = int(rng.integers(1, min(300, n // 20) + 1))
            idx = rng.integers(0, n, size=k)
            x[idx] += rng.normal(0.0, 0.2, size=k)
        elif mv == 4:
            # multi-scale
            for width in (n // 10, n // 40, max(1, n // 160)):
                w2 = max(1, width)
                st2 = int(rng.integers(0, max(1, n - w2)))
                x[st2:min(n, st2 + w2)] *= np.exp(rng.normal(0.0, 0.2))
        else:
            # pull toward uniform + boost peaks strategy
            w = int(rng.integers(max(2, n // 100), n // 8 + 1))
            st = int(rng.integers(0, n - w))
            x[st:st + w] = (1.0 - bl2) * x[st:st + w] + bl2 / w * np.sum(x[st:st + w])
            bl2 = 0.0  # unused, keep branch simple
            x[st:st + w] *= float(rng.uniform(0.5, 1.8))

        x = np.maximum(x, 0.0)
        sx = np.sum(x)
        if sx < 1e-12:
            continue
        x = x / sx
        score = _eval(x)
        if not np.isfinite(score):
            continue

        if score < best_score:
            best_score = score
            best_n = x.copy()
        else:
            delta = score - best_score
            if delta < 0.5 and rng.random() < np.exp(-delta / max(T, 1e-8)):
                best_n = x.copy()

        if it % 50 == 0:
            T = max(0.0005, T * 0.995)

    # Scale back to parent's magnitude for output
    s_parent = np.sum(parent)
    s_best = np.sum(best_n)
    if s_parent > 0 and s_best > 0:
        out = best_n * (s_parent / s_best)
    else:
        out = best_n
    out = np.clip(out, 0.0, 1000.0)
    # Verify
    final_score = _eval(out)
    if not np.isfinite(final_score) or final_score > best_score:
        out = best_n
        out = np.clip(out, 0.0, 1000.0)
    return [float(v) for v in out]
