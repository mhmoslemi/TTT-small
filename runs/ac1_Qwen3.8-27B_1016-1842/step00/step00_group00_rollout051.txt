import numpy as np
import time

def _autocorr(x):
    """Full linear autocorrelation via FFT."""
    n = len(x)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X.conj(), N)[:2 * n - 1]

def _subgrad(a, C, peaks, S):
    """
    Compute (sub)gradient for multi-peak objective.
    peaks: list of (k, weight) tuples.
    Returns gradient direction to subtract.
    """
    n = len(a)
    g = np.zeros(n)
    for k, w in peaks:
        # a[k-i] term
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += w * a[k - lo :: -1][:hi - lo + 1]
        # a[i-k] term
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            g[lo2:hi2+1] += w * a[lo2 - k:hi2 - k + 1]
    # Subtract 2*Cmax/S term (the S-derivative)
    total_w = sum(w for _, w in peaks)
    cmax_w = sum(C[k] * w for k, w in peaks) / total_w
    g -= 2.0 * cmax_w / S
    return g
