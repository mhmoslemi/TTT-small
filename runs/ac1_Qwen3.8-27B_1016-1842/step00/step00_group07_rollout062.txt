def _normalize(x, floor=1e-18):
    y = np.asarray(x, dtype=np.float64).copy()
    y = np.nan_to_num(y, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(y, floor, 1000.0, out=y)
    s = float(y.sum())
    if not np.isfinite(s) or s <= floor * max(1, y.size):
        return None
    y /= s

def _autoconv(a):
    n = a.size
    m = 2 * n - 1
    if m <= 1:
        size = 1
    else:
        size = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:m]


def _score_from_c(c, n, s=1.0):
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    return float(2.0 * n * m / (s * s))

    if s is None:
        return np.inf
    c = _autoconv(s)
    np.maximum(c, 0.0, out=c)
    return _score_from_c(c, s.size)

def _pressure_top(a, act, w):
    n = a.size
    p = np.zeros(n, dtype=np.float64)
    for i in range(act.size):
        k = int(act[i])
        ww = float(w[i])
        if ww <= 0.0:
            continue
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo > hi:
            continue
        if hi == lo:
            p[lo] += ww * a[k - lo]
        else:
            start = k - lo
            stop = k - hi
            if stop > 0:
                p[lo:hi + 1] += ww * a[start:stop - 1:-1]
            else:
                p[lo:hi + 1] += ww * a[start::-1]
    pm = float(p.max())
    if not np.isfinite(pm) or pm <= 0.0:
        p.fill(1.0 / max(1, n))
    else:
        p /= pm
    return p

            c = _autoconv(a)
            np.maximum(c, 0.0, out=c)
            sc = _score_from_c(c, n)
            prev_sc = sc
            eta = min(2.0, max(0.25, eta))
            stall = 0
            if np.isfinite(sc) and sc < best_sc:
                best = a.copy()
                best_sc = sc

    return best


def _polish(a, n, deadline, rng):
    a = _normalize(a)
    if a is None:
        a = np.ones(n, dtype=np.float64)
        a /= float(n)

    best = a.copy()
    c = _autoconv(best)
    np.maximum(c, 0.0, out=c)
    best_sc = _score_from_c(c, n)

    current = best.copy()
    stale = 0

    while time.monotonic() < deadline:
        width = int(rng.integers(max(1, n // 200), max(2, min(n, n // 8) + 1)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)

        cand = current.copy()
        if stop > start:
            cand[start:stop] *= np.exp(rng.normal(0.0, 0.40, stop - start))

        if rng.random() < 0.35:
            width2 = int(rng.integers(max(1, n // 500), max(2, min(n, n // 20) + 1)))
            s2 = int(rng.integers(0, max(1, n - width2 + 1)))
            e2 = min(n, s2 + width2)
            if e2 > s2:
                cand[s2:e2] *= np.exp(rng.normal(0.0, 0.45, e2 - s2))

        cand = _normalize(cand)
        if cand is None:
            continue

        c2 = _autoconv(cand)
        np.maximum(c2, 0.0, out=c2)
        sc2 = _score_from_c(c2, n)

        if np.isfinite(sc2) and sc2 < best_sc - 1e-12:
            best = cand.copy()
            best_sc = sc2
            current = cand.copy()
            stale = 0
        else:
            stale += 1
            if stale > 25:
                current = _normalize(best * np.exp(0.5 * rng.standard_normal(n)))
                if current is None:
                    current = best.copy()
                stale = 0

    return best


def _make_seeds(base, n, rng):
    seeds = []
    b = base / float(base.sum())
    seeds.append(b.copy())

    t = np.arange(n, dtype=np.float64) / max(1, n - 1)

    for sc in (0.25, 0.60, 1.10, 1.90):
        s = b * np.exp(sc * rng.standard_normal(n))
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)

    for r in (0.7, 1.4, 2.6, 4.5):
        s = np.exp(-r * t)
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)
        s = np.exp(-r * (1.0 - t))
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)

    eps = 1e-3
    for p in (0.35, 0.70, 1.50, 2.70):
        s = (t + eps) ** p
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)
        s = (1.0 - t + eps) ** p
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)

    for m in (1, 2, 3, 5):
        for amp in (0.25, 0.45, 0.62):
            s = 1.0 + amp * np.sin(2.0 * np.pi * float(m) * t + 0.55)
            nz = _normalize(s)
            if nz is not None:
                seeds.append(nz)
            s = 1.0 + amp * np.cos(2.0 * np.pi * float(m) * t - 0.35)
            nz = _normalize(s)
            if nz is not None:
                seeds.append(nz)

    for f in (0.18, 0.33, 0.52):
        s = np.ones(n, dtype=np.float64)
        c1 = max(1, int(n * f))
        c2 = max(c1 + 1, int(n * (1.0 - f)))
        s[:c1] = 3.0
        s[c2:] = 2.0
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)

        s2 = np.ones(n, dtype=np.float64)
        s2[c1:c2] = 3.0
        nz = _normalize(s2)
        if nz is not None:
            seeds.append(nz)

    for c in (0.8, 1.8, 3.2):
        s = b * (1.0 + c * (t - 0.5))
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)
        s = b * np.exp(2.0 * c * (t - 0.5))
        nz = _normalize(s)
        if nz is not None:
            seeds.append(nz)

    return seeds


def _exact_finite(seq):
    try:
        v = float(evaluate_sequence(seq))
    except Exception:
        return np.inf
    return v if np.isfinite(v) else np.inf


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(int(seed))
    base = _base_array()
    n = base.size
    if n <= 0:
        base = np.ones(64, dtype=np.float64)
        n = 64

    now = time.monotonic()
    deadline = now + max(60.0, float(budget_s) - 60.0)

    best = _normalize(base)
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best /= float(n)
    best_sc = _fft_score_array(best)

    seeds = _make_seeds(base, n, rng)
    scored = []
    for s in seeds:
        if s is None:
            continue
        sc = _fft_score_array(s)
        if np.isfinite(sc):
            scored.append((sc, len(scored), s.copy()))

    scored.sort()
    if scored:
        best = scored[0][2].copy()
        best_sc = scored[0][0]

    unit = max(10.0, (deadline - time.monotonic()) / 12.0)

    for pi in range(min(4, len(scored))):
        if time.monotonic() > deadline - 5.0:
            break
        limit = min(135.0, unit * 1.45)
        pdl = min(deadline, time.monotonic() + limit)
        eta0 = 0.35 + 0.18 * pi
        cur = _pressure_phase(scored[pi][2], n, pdl, rng, eta0, 12)
        if cur is not None:
            sc = _fft_score_array(cur)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = sc

    for ri in range(2):
        if time.monotonic() > deadline - 5.0:
            break
        limit = min(120.0, unit * 1.15)
        pdl = min(deadline, time.monotonic() + limit)
        start = best.copy()
        if ri:
            start = _normalize(start * np.exp((0.10 + 0.12 * ri) * rng.standard_normal(n)))
            if start is None:
                start = best.copy()
        cur = _pressure_phase(start, n, pdl, rng, 0.30 + 0.12 * ri, 16)
        if cur is not None:
            sc = _fft_score_array(cur)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = sc

    if deadline - time.monotonic() > 25.0:
        pdl = min(deadline - 5.0, time.monotonic() + max(20.0, (deadline - time.monotonic()) * 0.50))
        cur = _pressure_phase(best, n, pdl, rng, 0.30, 16)
        if cur is not None:
            sc = _fft_score_array(cur)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = cur.copy()
                best_sc = sc

    if deadline - time.monotonic() > 6.0:
        pdl = min(deadline - 2.0, time.monotonic() + max(3.0, (deadline - time.monotonic()) * 0.25))
        best = _polish(best, n, pdl, rng)

    best = _normalize(best)
    if best is None:
        best = _normalize(base)
    if best is None:
        best = np.ones(n, dtype=np.float64)
        best /= float(n)

    out = [float(x) for x in best]
    if _exact_finite(out) < np.inf:
        return out

    base_out = [float(x) for x in _normalize(base)]
    if base_out:
        return base_out
    return [float(x) for x in best]
