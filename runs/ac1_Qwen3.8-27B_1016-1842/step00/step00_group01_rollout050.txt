```python
def project_simplex(x):
    """Project x onto {y >= 0, sum(y) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]  # decreasing
    # Find m
    cssv = np.cumsum(u)
    # Condition: u_j > (cssv[j] - 1) / (j+1)  (0-indexed)
    rho = np.where(u > (cssv - 1.0) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    m = rho[-1] + 1  # 1-indexed
    theta = (cssv[m-1] - 1.0) / m
    return np.maximum(x - theta, 0.0)
```

```python
import numpy as np
import time


def fft_convolve(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    p2 = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, p2)
    return np.fft.irfft(A * A, p2)[:size]


def project_simplex(x):
    """Euclidean projection onto {y>=0, sum(y)=1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / idx
    m = int(np.sum(cond))
    if m == 0:
        return np.full(n, 1.0 / n)
    theta = (cssv[m - 1] - 1.0) / m
    return np.maximum(x - theta, 0.0)


def ac1_bound(a, n):
    """Compute 2n*max(conv)/sum^2 scale-invariant."""
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * float(np.max(c)) / (s * s)


def subgrad_step(a, k_star, eta):
    """One subgradient step toward reducing c[k_star]."""
    n = len(a)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    i0 = k_star - hi
    i1 = k_star - lo
    a_sym = 0.5 * (a + a[::-1])
    a_sym = project_simplex(a_sym)
    c = fft_convolve(a_sym)
    a_new = a_sym - eta * np.zeros(n)
    lo2 = max(0, k_star - n + 1)
    hi2 = min(n - 1, k_star)
    j0 = k_star - hi2
    j1 = k_star - lo2
    a_new[lo2:hi2+1] -= 2.0 * eta * a_sym[j0:j1+1][::-1]
    a_new = project_simplex(np.maximum(a_new, 0.0))
    return a_new


def try_candidate(a, deadline, n):
    """Refine a via subgradient descent; return refined result and score."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = np.sum(a)

        return a, np.inf
    a = a / s

    it = 0
    for eta in [0.05, 0.01, 0.002, 0.0004, 0.00008]:

        prev = best_local_score
        while time.monotonic() < deadline:
            c = fft_convolve(a)
            k_star = int(np.argmax(c))
            cur = 2.0 * n * float(c[k_star])

                best_local_score = cur
                best_local = a.copy()
                stall = 0
            elif stall < 3:
                stall += 1
            if prev - cur > 1e-6:
                stall = 0
            prev = cur
            if stall > 80:
                break
            a = subgrad_step(a, k_star, eta)
            it += 1
    return best_local, best_local_score
```
