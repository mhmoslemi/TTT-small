import scipy.sparse as sp
from scipy.optimize import linprog

def solve_lp(a, k_active, M, n):
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_active):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            b_idx = k - i
            rows.append(idx)
            cols.append(b_idx)
            vals.append(a[i])
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(len(k_active), n))
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    res = linprog(c, A_ub=A, b_ub=np.full(len(k_active), M), 
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None

    cols = []
    vals = []
    for idx, k in enumerate(k_active):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        b_range = k - i_range
        rows.extend([idx] * len(i_range))
        cols.extend(b_range.tolist())
        vals.extend(a[i_range].tolist())

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(len(k_active), M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:

    return None
