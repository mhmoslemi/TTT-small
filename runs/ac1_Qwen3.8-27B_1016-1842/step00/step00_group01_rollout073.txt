import numpy as np
import time


def normalize_nonneg(y):
    y = np.asarray(y, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    y = np.maximum(y, 0.0)
    s = float(np.sum(y))
    if y.size == 0:
        return np.ones(1, dtype=np.float64)
    if s < 1e-15:
        return np.full(y.size, 1.0 / y.size, dtype=np.float64)
    return y / s


def next_pow2(m):
    m = int(max(1, m))
    L = 1
    while L < m:
        L <<= 1
    return L


def fft_conv_pair(a, L):
    n = int(a.size)
    m = 2 * n - 1
    fa = np.fft.rfft(a, L)
    b = np.fft.irfft(fa * fa, L)
    b = b[:m].copy()
    b = np.where(np.isfinite(b), b, 0.0)
    np.maximum(b, 0.0, out=b)
    s = float(np.sum(b))
    if s < 1e-300 or not np.isfinite(s):
        return fa, np.full(m, 1.0 / m, dtype=np.float64)
    b /= s
    return fa, b


def corr_direction(fa, a, u, L):
    n = int(a.size)
    m = 2 * n - 1
    fu = np.fft.rfft(u[::-1], L)
    hconv = np.fft.irfft(fa * fu, L)
    h = hconv[m - 1 :: -1].copy()
    h = np.where(np.isfinite(h), h, 0.0)
    np.maximum(h, 0.0, out=h)
    return h


def make_weights(b, beta, K):
    m = int(b.size)
    if beta <= 1e-12:
        return b.copy()
    bmax = float(np.max(b))
    if bmax <= 0.0 or not np.isfinite(bmax):
        w = b.copy()
    else:
        T = float(K) / bmax
        x = (b - bmax) * T
        y = np.exp(x)
        s = float(np.sum(y))
        if s <= 0.0 or not np.isfinite(s):
            w = np.zeros(m, dtype=np.float64)
            w[int(np.argmax(b))] = 1.0
        else:
            w = y / s
    if beta >= 1.0:
        return w
    return (1.0 - beta) * b + beta * w


def multiplicative_update(a, h, eta):
    d = h - float(np.mean(h))
    d_scale = float(np.max(np.abs(d)))
    if d_scale <= 1e-300:
        return a.copy()
    max_log = 1.0
    eff_eta = float(eta)
    if eff_eta * d_scale > max_log:
        eff_eta = max_log / d_scale
    z = -eff_eta * d
    z = np.clip(z, -1.2, 1.2)
    y = a * np.exp(z)
    return normalize_nonneg(y)

    cands.append(normalize_nonneg(0.25 + np.abs(x - 0.5)))

    for _ in range(6):
        y = np.ones(n, dtype=np.float64)
        for f in (1, 2, 3, 5, 8):
            coeff = float(rng.uniform(0.10, 0.70))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            y += coeff * np.sin(2.0 * np.pi * f * x + phase)
        cands.append(normalize_nonneg(y))

    for _ in range(4):
        c = float(rng.uniform(0.20, 0.80))
        w = float(rng.uniform(0.05, 0.25))
        y = 0.2 + np.exp(-((x - c) ** 2) / (2.0 * w * w))
        cands.append(normalize_nonneg(y))

    for _ in range(5):
        y = np.ones(n, dtype=np.float64)
        for pw in range(1, 8):
            coeff = float(rng.normal(0.0, 0.35))
            y += coeff * (x ** pw)
        cands.append(normalize_nonneg(y))

    for f in (1, 2, 3):
        for amp in (0.5, 0.9):
            for ph in (0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi):
                y = 1.0 + amp * np.sin(2.0 * np.pi * f * x + ph)
                cands.append(normalize_nonneg(y))

    return cands


def local_search(a, L, n, rng, deadline, snapshots):
    a = normalize_nonneg(a)
    fa, b = fft_conv_pair(a, L)
    score = 2.0 * float(n) * float(np.max(b))
    best_score = score
    best_a = a.copy()
    snapshot_score = best_score

    eta = 2.0
    stuck = 0
    iters = 0

    while time.monotonic() < deadline:
        if iters > 200000:
            break

        temp = 0.008 + 0.05 * float(np.exp(-iters / 4000.0))
        prog = min(1.0, iters / 6000.0)
        beta = 0.2 + 0.8 * prog
        K = 8.0 + 250.0 * prog

        u = make_weights(b, beta, K)
        h = corr_direction(fa, a, u, L)
        a_try = multiplicative_update(a, h, eta)

        fa_try, b_try = fft_conv_pair(a_try, L)
        score_try = 2.0 * float(n) * float(np.max(b_try))

        accept = False
        if np.isfinite(score_try):
            if score_try < score - 1e-15:
                accept = True
            else:
                delta = score_try - score
                if delta < 50.0 * temp:
                    accept = bool(rng.random() < np.exp(-delta / temp))

        if accept:
            a = a_try
            fa = fa_try
            b = b_try
            score = score_try

            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                if best_score < snapshot_score - 1e-5:
                    snapshots.append(a.copy())
                    snapshot_score = best_score
                    if len(snapshots) > 128:
                        del snapshots[0]
                stuck = 0
                eta = min(eta * 1.07, 40.0)
            else:
                if score > best_score + 0.15:
                    a = best_a.copy()
                    fa, b = fft_conv_pair(a, L)
                    score = 2.0 * float(n) * float(np.max(b))
                eta = max(eta * 0.85, 0.02)
                stuck += 1
        else:
            eta = max(eta * 0.55, 0.01)
            stuck += 1
            if stuck > 60:
                a_rand = random_perturb(best_a, rng)
                fa_rand, b_rand = fft_conv_pair(a_rand, L)
                score_rand = 2.0 * float(n) * float(np.max(b_rand))
                if np.isfinite(score_rand) and (score_rand < best_score + 0.25 or rng.random() < 0.25):
                    a = a_rand
                    fa = fa_rand
                    b = b_rand
                    score = score_rand
                    if score < best_score - 1e-13:
                        best_score = score
                        best_a = a.copy()
                        if best_score < snapshot_score - 1e-5:
                            snapshots.append(a.copy())
                            snapshot_score = best_score
                            if len(snapshots) > 128:
                                del snapshots[0]
                else:
                    a = best_a.copy()
                    fa, b = fft_conv_pair(a, L)
                    score = 2.0 * float(n) * float(np.max(b))
                eta = 2.0
                stuck = 0

        iters += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(0.0, float(budget_s))
    reserve = max(5.0, min(40.0, 0.05 * total))
    search_budget = max(0.0, total - reserve)
    final_deadline = start + max(0.0, total - 2.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(6121, dtype=np.float64)

    if parent.ndim != 1 or parent.size < 1:
        parent = np.ones(6121, dtype=np.float64)
    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    n = int(parent.size)
    base = normalize_nonneg(parent)
    L = next_pow2(3 * n - 2)

    snapshots = [base.copy()]
    local_bests = []
    top = []

    if search_budget > 1.0:
        cands = make_initial_candidates(base, n, rng)
        scored = []
        cutoff = start + search_budget - max(10.0, 0.02 * total)
        for cand in cands:
            if time.monotonic() > cutoff:
                break
            fa_c, b_c = fft_conv_pair(normalize_nonneg(cand), L)
            sc = 2.0 * float(n) * float(np.max(b_c))
            if np.isfinite(sc):
                scored.append([sc, normalize_nonneg(cand)])

        while len(scored) > 0 and len(top) < 3:
            min_idx = 0
            for i in range(1, len(scored)):
                if scored[i][0] < scored[min_idx][0]:
                    min_idx = i
            top.append(scored.pop(min_idx))

    if len(top) == 0:
        top = [[None, base]]
    elif len(top) == 1:
        fracs = [1.0]
    elif len(top) == 2:
        fracs = [0.65, 1.0]
    else:
        fracs = [0.55, 0.82, 1.0]

    search_deadline = start + search_budget
    for idx, frac in enumerate(fracs):
        limit = start + search_budget * frac
        if time.monotonic() >= limit - 1.0:
            continue
        arr0 = top[idx][1]
        snapshots.append(arr0.copy())
        if len(snapshots) > 128:
            del snapshots[0]
        best_arr, _ = local_search(arr0, L, n, rng, limit, snapshots)
        local_bests.append(best_arr)
        snapshots.append(best_arr.copy())
        if len(snapshots) > 128:
            del snapshots[0]

    eval_arrays = [base]
    for _, arr in top:
        eval_arrays.append(arr)
    eval_arrays.extend(local_bests)
    eval_arrays.extend(snapshots[-48:])

    seen_ids = set()
    unique_arrays = []
    for arr in eval_arrays:
        if id(arr) not in seen_ids:
            seen_ids.add(id(arr))
            unique_arrays.append(arr)

    best_seq = base.tolist()
    best_val = float(evaluate_sequence(best_seq))

    for arr in unique_arrays:
        if time.monotonic() > final_deadline:
            break
        seq = np.clip(np.asarray(arr, dtype=np.float64), 0.0, 1000.0).tolist()
        val = float(evaluate_sequence(seq))
        if np.isfinite(val) and val < best_val:
            best_val = val
            best_seq = seq

    if not np.isfinite(best_val):
        best_seq = [1.0] * max(1, n)
        best_val = float(evaluate_sequence(best_seq))

    return best_seq
