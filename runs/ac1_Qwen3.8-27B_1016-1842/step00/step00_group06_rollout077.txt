            return
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc

    if n <= 64:
        if n > 1 and time.monotonic() < deadline - 20.0:
            b = n
            prof, s = optimize_small(b, min(220.0, deadline - time.monotonic() - 20.0), int(seed))
            ps = float(evaluate_sequence(prof.tolist()))
            if np.isfinite(ps) and ps < small_score:
                small_profile, small_score = prof, ps
            consider(prof)
    else:
        for b in (32, 64):
            if time.monotonic() >= deadline - 35.0:
                break
            if b == 64 and best_score < 1.42:
                break
            alloc = min(260.0, deadline - time.monotonic() - 35.0)
            if alloc <= 20.0:
                break
            prof, s = optimize_small(b, alloc, int(seed) + b)
            ps = float(evaluate_sequence(prof.tolist()))
            if np.isfinite(ps) and ps < small_score:
                small_profile, small_score = prof, ps
            if n > b:
                cand = upsample_profile(prof, n)
            else:
                cand = prof.copy()
            consider(cand)

    if time.monotonic() < deadline - 8.0 and best.size == n:
        if best_score < 1.45:
            local_budget = 45.0
        elif best_score < TARGET:
            local_budget = 120.0
        else:
            local_budget = 220.0
        local_budget = min(max(0.0, deadline - time.monotonic() - 8.0), local_budget)
        if local_budget > 3.0:
            cand, _ = local_n(best, int(seed) + 991, time.monotonic() + local_budget)
            consider(cand)

    if (not np.isfinite(best_score) or best_score > TARGET) and small_profile is not None:
        ps = float(evaluate_sequence(small_profile.tolist()))
        if np.isfinite(ps) and ps < best_score:
            best = small_profile
            best_score = ps

    out = np.asarray(best, dtype=np.float64).ravel()
    if out.size == 0 or not np.all(np.isfinite(out)):
        out = np.ones(256, dtype=np.float64)
    out = np.clip(out, 0.0, 1000.0)
    return [float(v) for v in out]
