def _quick_score(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = float(b.max())
    S = float(a.sum())
    if S < 0.01:
        return np.inf, b, max_b, S
    return 2.0 * n * max_b / (S * S), b, max_b, S


def _partner(a, k):
    n = len(a)
    p = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        p[lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return p


def _make_inits(a, n, rng):

    return inits
