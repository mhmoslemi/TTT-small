    fb = np.fft.rfft(b, N)

        w = np.exp(tc)
        w /= w.sum()

        # 4. Gradient g[i] = 2 * sum_k w_k * a[k-i]  (via FFT convolution)
        #    g[i] = 2 * (conv(a, w_rev))[2n-2-i]
        h = _fftconv(a, w[::-1])
        g = 2.0 * h[2 * n - 2 : n - 1 : -1]

        # 5. Gradient step + projection onto {a>=0, sum=1}
        a = a - lr * g
        a = _proj(a, 1.0)

        # 6. Periodic exact-eval verification
        if it % 3000 == 0:
            s = float(evaluate_sequence(a.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a.copy()

    # Final verification
    s = float(evaluate_sequence(a.tolist()))
    if np.isfinite(s) and s < best_score:
        best_a = a.copy()

    return [float(x) for x in best_a]
