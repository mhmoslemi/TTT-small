    """Solve the dual LP greedily. Returns y and the dual objective value."""
    n = len(f)
    y = np.zeros(2*n - 1)
    # Sliding window sum for efficiency
    window_sum = 0.0
    # Process constraints j = 0, 1, ..., n-1
    for j in range(n):
        if f[j] <= 1e-15:
            continue
        c_j = 1.0 / f[j]
        # Window is [j, j+n-1], sum of y over this window
        # We need to maintain window_sum = sum(y[j:j+n])
        # When we move from j to j+1, we add y[j+n] and remove y[j]
