import numpy as np

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = 1.0 + rho * u > cs
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    r = int(idx[-1]) + 1
    theta = (cs[r - 1] - 1.0) / r
    return np.maximum(v - theta, 0.0)


def autocorr(a):
    n = len(a)
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A)[:need]


def fista_step(a, ct, t_k, step, n, top_k=1):
    b = autocorr(a)
    if top_k <= 1:
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            idx = k_star - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi + 1] = 2.0 * a[idx]
    else:

        sorted_idx = np.argsort(b)[::-1][:top_k]
        grad = np.zeros(n)
        for ki in sorted_idx:
            k = int(ki)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                idx = k - np.arange(j_lo, j_hi + 1)
                grad[j_lo:j_hi + 1] += 2.0 * a[idx]

        v[:int(p * n)] = 1.6
        v = np.maximum(v, 1e-8)
        v /= v.sum()
        inits.append(v)
    return inits


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n == 0:
        return [1.0]
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    inits = make_inits(parent, n, rng)
    n_inits = len(inits)
    time_per_iter = max(8.0, (float(budget_s) - 40.0) / max(1, n_inits))
    for ai in inits:
        if time.monotonic() > total_deadline - 10.0:
            break
        dl = min(time.monotonic() + time_per_iter, total_deadline - 10.0)
        a_opt = fista_run(ai, n, dl, step=0.4, top_k=1, max_it=300000)
        a_opt = np.clip(a_opt, 0.0, 1000.0)
        if a_opt.sum() < 0.01:
            continue
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_opt.copy()
            best_score = score
    if time.monotonic() < total_deadline - 5.0:
        s = best.sum()
        if s > 1e-6:
            a_norm = best / s
            a_opt = fista_run(a_norm, n, total_deadline - 3.0, step=0.25, top_k=3, max_it=400000)
            a_opt = np.clip(a_opt, 0.0, 1000.0)
            if a_opt.sum() >= 0.01:
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = a_opt.copy()
                    best_score = score
    if time.monotonic() < total_deadline - 3.0:
        s = best.sum()
        if s > 1e-6:
            a_norm = best / s
            a_opt = fista_run(a_norm, n, total_deadline - 1.0, step=0.15, top_k=5, max_it=400000)
            a_opt = np.clip(a_opt, 0.0, 1000.0)
            if a_opt.sum() >= 0.01:
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = a_opt.copy()
                    best_score = score
    return [float(v) for v in best]
