import numpy as np
import time

def self_conv(a):
    n = 2 * len(a) - 1
    sz = 1
    while sz < n:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:n]

def make_gaussian(n, w):
    x = np.arange(n)
    return np.exp(-0.5 * ((x - (n - 1) / 2.0) / (w * n / 2.0)) ** 2)

    return np.clip(1.0 - np.abs(np.arange(n) - mid) / (mid + 1), 0, 1)
