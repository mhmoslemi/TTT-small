def fft_conv(a):
    n = a.size
    sz = 2 * n - 1

    return np.fft.irfft(A * A, N)[:sz]

def score_unit(a):

    c = fft_conv(a)
    return 2.0 * a.size * c.max()

def grad_peak(a):
    """Gradient of max(conv(a)) w.r.t. a (at the argmax)."""
    c = fft_conv(a)
    k = int(np.argmax(c))
    n = a.size
    i0 = max(0, k - n + 1)
    i1 = min(n - 1, k)

    idx = np.arange(i0, i1 + 1)
