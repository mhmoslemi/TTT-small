            g[idx] += a[ki - idx]
    return g / len(active_lags)


def do_subgradient(a, rng, lr, active_frac, max_active):
    n = a.size
    c = fftconvolve(a, a, mode='full')
    M = c.max()
    threshold = M * active_frac
    active_mask = c >= threshold
    if not active_mask.any():
        active_mask = np.zeros(c.size, dtype=bool)
        active_mask[np.argmax(c)] = True
    active_lags = np.where(active_mask)[0]
    if len(active_lags) > max_active:
        vals = c[active_lags]
        top = np.argsort(vals)[::-1][:max_active]
        active_lags = active_lags[top]
    g = compute_subgrad(a, active_lags)
    a_new = a - lr * g
    return a_new, score_from_conv(c, a)


def score_from_conv(c, a):
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return float('inf')
    return 2.0 * n * c.max() / (S * S)
