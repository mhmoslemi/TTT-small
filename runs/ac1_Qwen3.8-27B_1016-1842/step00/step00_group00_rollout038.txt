import numpy as np

def compute_gradient(a, n):
    """Compute gradient of 2n*max(conv(a,a))/sum(a)^2 w.r.t. a."""
    S = np.sum(a)
    conv = np.fft.irfft(np.fft.rfft(a, 2*n-1) ** 2, 2*n-1)
    M = np.max(conv)
    # Find active modes (within tolerance)
    tol = 1e-10 * max(M, 1.0)
    active = np.where(conv >= M - tol)[0]
    K = len(active)
    
    # Compute sum over active modes of a[k-j] for valid k-j
    # For each active mode k, contribution is 2*a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    mode_sum = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        # a[k-j] for j in [lo, hi-1] means indices k-lo to k-(hi-1) = k-lo to n-1 (reversed)
        # k-j ranges from k-lo to k-(hi-1)
        # We want a[k-j] for j in [lo, hi-1], which is a[k-lo ... k-(hi-1)] reversed
        idx_start = k - hi + 1
        idx_end = k - lo + 1

    grad = (2.0 * n / S**3) * (S * (2.0 / K) * mode_sum - 2.0 * M * a)

import numpy as np
import time


def _grad_normalized(a, n, Nfft):
    """Return (unit_grad, M, S) for objective 2n*max(ac)/S^2."""
    S = float(np.sum(a))
    if S < 1e-10:
        return np.zeros(n), 0.0, S
    fa = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(fa * fa, Nfft)
    M = float(conv.max())
    if M < 1e-30:
        return np.zeros(n), M, S
    tol = 1e-8 * M + 1e-12
    active = np.where(conv >= M - tol)[0]
    K = len(active)
    if K > 200:
        t2 = M * (1.0 - 1e-5)
        active = np.where(conv >= t2)[0]
        K = max(len(active), 1)
    ms = np.zeros(n)
    for k in active[:200]:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            ms[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g = S * ms / K - M * a
    gn = float(np.sqrt(np.dot(g, g)))
    if gn < 1e-15:
        return np.zeros(n), M, S
    return g / gn, M, S
