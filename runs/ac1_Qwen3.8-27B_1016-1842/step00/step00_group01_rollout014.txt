import numpy as np
import time

def fft_convolve(a):
    """Full linear convolution a*a via FFT in O(n log n)."""
    n = a.size
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_subgrad(a, k, n):
    """Subgradient of max_k (a*a)_k w.r.t. a at index k."""
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    sg = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    sg[idx] = 2.0 * a[k - idx]
    return sg

    best = np.clip(a * total, 0.0, 1000.0)

    prev_max = np.inf
    best_max_b = np.inf
    stall = 0

    for it in range(10000000):
        if time.monotonic() > deadline:
            break

        conv = fft_convolve(a)
        max_b = float(conv.max())

        # Use top-2 peaks for smoother descent direction
        flat_idx = np.argpartition(-conv, 2)[:2]
        v1, v2 = conv[flat_idx[0]], conv[flat_idx[1]]
        w1 = np.exp(v1 / (max_b + 1e-15))
