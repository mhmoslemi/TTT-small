def _fft_autocorr(a):
    """Full linear autocorrelation via FFT. O(n log n)."""
    n = a.size

    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    n = a.size
    m = int(c.argmax())
    # a[i] appears in c[m] with weight 2*a[m-i] (for valid i)
    # Multiplicative: a[i] *= exp(-lr * 2 * a[m-i]) for valid i

    factors = np.ones(n)

    a_new = a * factors

import numpy as np
import time


def _fft_autocorr(a):
    n = a.size
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _grad_c_m(a, m):
    n = a.size
    g = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[m - hi:m - lo + 1][::-1]
    return g


def _fast_score(a, c):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * a.size * c.max() / (s * s)


def _run_descent(a, rng, deadline):
    n = a.size
    a = a / max(a.sum(), 1e-15)
    lr = 0.02
    mom = np.zeros(n)
    beta = 0.8
    patience = 0
    best_local = a.copy()
    best_local_sc = _fast_score(a, _fft_autocorr(a))

    for it in range(300000):
        if time.monotonic() > deadline:
            break
        c = _fft_autocorr(a)
        sc = _fast_score(a, c)
        if sc < best_local_sc:
            best_local_sc = sc
            best_local = a.copy()

        # Top-3 lag gradient for smoother direction
        top = np.argsort(c)[-3:][::-1]
        g = np.zeros(n)
        for m in top:
            g += _grad_c_m(a, int(m))
        g /= 3.0

        mom = beta * mom - lr * g
        a_t = a + mom
        a_t = np.maximum(a_t, 0.0)
        st = a_t.sum()
        if st < 1e-15:
            lr *= 0.5
            continue
        a_t = a_t / st

        c_t = _fft_autocorr(a_t)
        sc_t = _fast_score(a_t, c_t)

        if sc_t <= sc * (1.0 + 1e-9):
            a = a_t
            patience = 0
            lr = min(lr * 1.003, 0.1)
        else:
            patience += 1
            lr *= 0.85
            mom *= 0.5
            if patience > 400:
                a = a + rng.normal(0.0, 0.04 / np.sqrt(n), n)
                a = np.maximum(a, 1e-6 / n)
                a = a / a.sum()
                lr = 0.02
                mom = np.zeros(n)
                patience = 0
    return best_local, best_local_sc
