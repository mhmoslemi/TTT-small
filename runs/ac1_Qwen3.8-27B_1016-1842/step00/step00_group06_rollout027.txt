import time

        parent = np.ones(100)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    idx = np.arange(n, dtype=np.float64)
    inits = [
        1.0 + 999.0 * idx / max(1, n - 1),
        1.0 + 999.0 * (idx / max(1, n - 1)) ** 2,
        1.0 + 999.0 * np.exp(idx / max(1, n - 1)) / np.e,
        0.5 + 0.5 * np.exp(2.0 * idx / max(1, n - 1)),
        100.0 * (1.0 + np.sin(np.pi * idx / max(1, n - 1))),
        1.0 + 0.8 * np.exp(idx / max(1, n - 1)) / np.e,
    ]
    for c in inits:
        c = np.clip(c, 0.0, 1000.0)
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s

    iters = 0
    while time.monotonic() < deadline:
        iters += 1
        candidate = best.copy()
        r = rng.random()

        if r < 0.25:
            w = int(rng.integers(max(2, n // 400), max(3, n // 25)))
            s = int(rng.integers(0, n - w))
            candidate[s:s + w] *= np.exp(rng.normal(0.0, 0.06))
        elif r < 0.45:
            conv = np.convolve(candidate, candidate)
            k = int(np.argmax(conv))
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            if hi > lo:
                ii = np.arange(lo, hi)
                jj = k - ii
                valid = (jj >= 0) & (jj < n)
                if np.any(valid):
                    pv = candidate[jj[valid]]
                    mv = max(1e-10, np.max(candidate))
                    candidate[ii[valid]] *= np.exp(-0.04 * pv / mv)
            if lo > 0:
                candidate[:lo] *= 1.008
            if hi < n:
                candidate[hi:] *= 1.008
        elif r < 0.60:
            freq = int(rng.integers(1, max(2, n // 40)))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            amp = float(rng.uniform(0.02, 0.08)) * max(1.0, np.mean(candidate))
            candidate += amp * np.sin(2.0 * np.pi * freq * idx / n + phase)
        elif r < 0.75:
            nc = int(rng.integers(3, max(4, n // 25)))
            ci = rng.integers(0, n, size=nc)
            candidate[ci] *= np.exp(rng.normal(0.0, 0.18, size=nc))
        elif r < 0.85:
            grad = idx / max(1.0, n - 1) - 0.5
            candidate += 0.02 * np.mean(candidate) * grad
        elif r < 0.92:
            mean_f = np.mean(candidate)
            candidate += 0.015 * (mean_f - candidate)
        else:
            w = int(rng.integers(n // 25, max(n // 25 + 1, n // 6)))
            s = int(rng.integers(0, n - w))
            base = max(0.01, 0.4 * np.mean(best))
            t = np.arange(w, dtype=np.float64)
            candidate[s:s + w] = base * (1.0 + 0.5 * np.sin(2.0 * np.pi * rng.uniform(1, 4) * t / w))

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

    return [float(x) for x in best]
