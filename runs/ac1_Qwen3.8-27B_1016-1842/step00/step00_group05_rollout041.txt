    # Theoretical minimum: for non-negative x,
    #   max(conv(x,x)) >= sum(x^2) >= (sum(x))^2 / n   (Cauchy-Schwarz)
    # so score = 2*n*max_b/sum^2 >= 2.
    # Equality iff all x[i] are equal (constant vector).
    # A constant vector of ones achieves exactly 2.0.

    const_seq = np.ones(n, dtype=np.float64)
    const_score = float(evaluate_sequence(const_seq.tolist()))
    parent_score = float(evaluate_sequence(parent.tolist()))

    if np.isfinite(parent_score) and parent_score <= const_score:
        best = parent
    else:
        best = const_seq

    # In the unlikely case neither is at exactly 2.0 (floating-point),
    # try a few tiny perturbations of the constant to see if we improve.
    best_score = min(parent_score if np.isfinite(parent_score) else np.inf,
                     const_score if np.isfinite(const_score) else np.inf)
    if best_score > 2.0 + 1e-12 and time.monotonic() < deadline:
        rng = np.random.default_rng(seed)
        for _ in range(5000):
            if time.monotonic() >= deadline:
                break
            w = int(rng.integers(1, max(2, n // 4)))
            start = int(rng.integers(0, n - w + 1))
            candidate = best.copy()
            candidate[start:start + w] *= float(np.exp(rng.normal(0.0, 0.02)))
            candidate = np.clip(candidate, 0.0, 1000.0)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score

    # Final safety: ensure we return a valid finite list
    result = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if not np.isfinite(result).all() or result.sum() < 0.01:
        result = np.ones(n, dtype=np.float64)

    return [float(x) for x in result]
