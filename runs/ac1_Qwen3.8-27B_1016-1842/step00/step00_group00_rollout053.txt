def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    # Round up to next power of 2 for FFT efficiency
    import math
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C

import numpy as np
import time

def compute_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smooth_gradient(a, C, tau):
    n = len(a)
    S = a.sum()
    # Softmax weights
    scaled_C = tau * C / (S * S)
    max_val = scaled_C.max()
    exp_C = np.exp(scaled_C - max_val)
    w = exp_C / exp_C.sum()
    smooth_max = (scaled_C.max() + np.log(exp_C.sum())) / tau  # = log(sum exp) / tau
    
    # Cross-correlation: r[i] = sum_j a[j] * w[i+j] for i=0..n-1
    # Using FFT:
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    b = a[::-1]  # reversed
    B = np.fft.rfft(b, fft_size)
    W = np.fft.rfft(w, fft_size)
    r = np.fft.irfft(B * W, fft_size)[n-1 : n-1+n]
    
    grad = (2.0 * n / (S * S)) * (2.0 / tau) * r - (4.0 * n * smooth_max / (S**3)) * np.ones(n)
    return grad, smooth_max
