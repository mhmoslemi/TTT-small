    a[:p] = 0.5 * (1.0 + np.cos(np.pi * xs / max(1, p - 1)))
    xs2 = np.arange(q, dtype=np.float64)
    a[p + gap:p + gap + q] = 0.5 * (1.0 + np.cos(np.pi * xs2 / max(1, q - 1)))
    return a


def sanitize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    a = np.minimum(a, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    S = a.sum()
    if S < 0.01:
        a = np.ones_like(a)
        S = a.sum()
    return a / S


def run_adam(a, n, deadline, rng):
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.02
    no_improve = 0
    best_a = a.copy()
    best_sc = fast_score(a)
    best_a = np.maximum(a, 0.0)
    best_a = best_a / best_a.sum()
    best_sc = fast_score(best_a)
    while time.monotonic() < deadline:
        t += 1
        b = fft_conv(a)
        k_star = int(np.argmax(b))
        S = a.sum()
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if hi >= lo:
            g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
        direction = -g * S + 2.0 * b[k_star] * np.ones(n)
        m = 0.9 * m + 0.1 * direction
        v = 0.999 * v + 0.001 * direction * direction
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-10:
            a = best_a.copy()
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr = max(0.001, lr * 0.5)
            continue
        a = a / S
        sc = fast_score(a)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.01, 0.15)
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.85
                if lr < 1e-6:
                    break
                no_improve = 0
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(2535)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(2535)
    n = parent.size
    parent = np.clip(parent, 0.0, 1000.0)

    inits = []
    sp = parent.sum()
    if sp > 0.01:
        inits.append(parent / sp)

    for gf in (1.0 / 3.0, 0.32, 0.34, 0.36, 0.40, 0.45):
        a = make_two_block(n, gf)
        if a.sum() > 0.01:
            inits.append(a / a.sum())
    for gf in (1.0 / 3.0, 0.34, 0.40):
        a = make_smoothed_two_block(n, gf)
        if a.sum() > 0.01:
            inits.append(a / a.sum())

    best = None
    best_score = np.inf
    for ini in inits:
        s = fast_score(ini)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = ini.copy()

    real = float(evaluate_sequence(best.tolist()))
    if np.isfinite(real):
        best_score = real

    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        refined, ref_sc = run_adam(best.copy(), n, deadline, rng)
        if ref_sc < best_score:
            best = refined
            best_score = ref_sc
            real2 = float(evaluate_sequence(best.tolist()))
            if np.isfinite(real2):
                best_score = real2

    final_real = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_real):
        best = sanitize(best)
    parent_real = float(evaluate_sequence(parent.tolist()))
    if np.isfinite(parent_real) and parent_real < best_score:
        best = parent / max(parent.sum(), 1e-10)

    result = np.asarray(best, dtype=np.float64)
    result = np.maximum(result, 0.0)
    result = np.minimum(result, 1000.0)
    result = np.where(np.isfinite(result), result, 0.0)
    if result.sum() < 0.01:
        result = np.ones(n)
    return [float(x) for x in result]
