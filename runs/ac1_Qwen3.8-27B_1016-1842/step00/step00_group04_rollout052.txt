   def build_conv_matrix(a):
       """Build the sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
       n = len(a)
       # A[k, j] = a[k-j] if 0 <= k-j < n
       rows = []
       cols = []
       vals = []
       for j in range(n):
           # a[k-j] is nonzero for k-j in [0, n-1], i.e., k in [j, j+n-1]
           k_start = j
           k_end = min(j + n - 1, 2*n - 2)

               cols.append(j)
               vals.append(a[k - j])
       A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
       return A

       n = len(a)

       c = -np.ones(n)  # minimize -sum(b)

       res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
