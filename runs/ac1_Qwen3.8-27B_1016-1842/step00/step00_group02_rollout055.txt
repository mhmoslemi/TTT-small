import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = 1.0 + rho * u > cs
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    r = int(idx[-1]) + 1
    theta = (cs[r - 1] - 1.0) / r
    return np.maximum(v - theta, 0.0)


def autocorr(a):
    """Full autocorrelation of a via FFT."""
    n = len(a)
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A)[:need]


def compute_grad(a, k_star, n):
    """Subgradient of max_k (a*a)[k] at the given k_star."""
    grad = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n - 1, k_star)
    if j_lo <= j_hi:
        idx = k_star - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi + 1] = 2.0 * a[idx]
    return grad


def fista_run(a0, n, deadline, step=0.5, top_k=1):
    """Run FISTA to minimize max autocorrelation on the simplex."""
    a = a0.copy()
    ct = a.copy()
    t_k = 1.0
    it = 0
    while time.monotonic() < deadline:
        b = autocorr(a)
        # Find top-k lags for better subgradient
        if top_k > 1:
            # Use average of top few gradients
            c = np.clip(b, 0, None)  # b should be nonnegative for a>=0
            # Get top_k indices (handle ties by using all above threshold)
            threshold = np.partition(b, -top_k)[-top_k]
            top_idx = np.where(b >= threshold)[0]
            grad = np.zeros(n)
            for ki in top_idx:
                k_star = int(ki)
                j_lo = max(0, k_star - n + 1)
                j_hi = min(n - 1, k_star)
                if j_lo <= j_hi:
                    idx = k_star - np.arange(j_lo, j_hi + 1)
                    grad[j_lo:j_hi + 1] += 2.0 * a[idx]
            grad /= len(top_idx)
        else:
            k_star = int(np.argmax(b))
            grad = compute_grad(a, k_star, n)

        a_new = project_simplex(y)

        t_k = t_new
        a = a_new
        it += 1
        # Safety: break if stuck
        if it > 1000000:

    return a
