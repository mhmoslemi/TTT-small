            if sc < best_score:
                best_score = sc
                best = a_dir.copy()
                a = a_dir
                stale = 0
            else:
                # try mirror mix
                a_mix = 0.5 * (a + a[::-1])
                a_mix = normalize_to(a_mix, target)
                sc = score_of(a_mix)
                if sc < best_score:
                    best_score = sc
                    best = a_mix.copy()
                    a = a_mix
                    stale = 0
                elif stale >= 8:
                    # big random kick
                    a_kick = np.exp(rng.normal(0.0, 1.0, size=n))
                    a_kick = normalize_to(a_kick, target)
                    sc = score_of(a_kick)
                    if sc < best_score:
                        best_score = sc
                        best = a_kick.copy()
                    a = a_kick
                    stale = 0

    # Final safety: ensure best is valid
    if not np.isfinite(best).all() or best.size == 0 or best.sum() < 0.01:
        best = np.ones(n) * target / n
    return [float(x) for x in np.clip(best, 0.0, 1000.0)]
