def compute_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)

    # Scale so sum = S
    mu = 0.0
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    mu = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - mu, 0.0)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n == 0:
        return [1.0]
    
    # Phase 1: Find a good initial non-constant sequence
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Try analytical initializations
    inits = []
    # Asymmetric ramp
    inits.append(np.linspace(0.5, 1.5, n))
    # One-sided taper
    inits.append(np.concatenate([np.linspace(0, 1, n//2), np.ones(n - n//2)]))
    # Sine
    inits.append(np.sin(np.pi * (np.arange(n) + 0.5) / n))
    # Raised cosine
    inits.append(0.5 * (1 + np.cos(2*np.pi*np.arange(n)/(n-1+1e-9))))
    # Asymmetric window: linear up, cosine down
    x_up = np.arange(n//2) / (n//2)
    x_dn = np.arange(n - n//2) / (n - n//2)
    inits.append(np.concatenate([x_up, np.cos(np.pi * x_dn / 2)]))
    
    for init in inits:
        init = np.clip(init, 0.0, 1000.0)
        s = float(evaluate_sequence(init.tolist()))
        if np.isfinite(s) and s < best_score:
            best = init.copy()
            best_score = s
    
    # Phase 2: Gradient descent on the AC1 objective
    # Fix sum(a) = S_best for scale invariance
    S_best = best.sum()
    lr = 0.01
    for it in range(5000):
        if time.monotonic() > deadline - 120:
            break
        a = best
        b = fft_convolve(a, a)
        max_b = b.max()
        k_star = int(np.argmax(b))
        S = a.sum()
        
        # Subgradient of max(b)/S^2 w.r.t. a
        # ∂f/∂a[i] = (2n/S^2)*(2*a[k*-i]) - (4n*max_b/S^2)  [for valid i]
        # = (2n/S^2) * (2*a[k*-i] - 2*max_b/S)
        grad_conv = subgradient_max_conv(a, b, k_star)
        grad = (2.0*n / (S*S)) * (grad_conv - 2.0*max_b/S)
        
        # Normalized gradient step
        # Scale lr by the typical magnitude
        g_scale = np.sum(grad**2) ** 0.5
        if g_scale < 1e-15:
            break
        step = lr * grad / g_scale * S  # scale by S for proper normalization
        
        a_new = a - step
        a_new = simplex_project(a_new, S_best)
        a_new = np.clip(a_new, 0.0, 1000.0)
        
        score = float(evaluate_sequence(a_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_new
            best_score = score
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
    
    # Phase 3: LP-based refinement
    t = 0.2
    K = 30
    for it in range(50):
        if time.monotonic() > deadline - 60:
            break
        a = best
        b_a = fft_convolve(a, a)
        max_b = b_a.max()
        S = a.sum()
        
        # Build LP: max 1^T g s.t. conv(a,g)[k] <= max_b for top K, g >= 0
        top_idx = np.argsort(b_a)[::-1][:K]
        rows, cols, vals = [], [], []
        for ii, k in enumerate(top_idx):
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            js = np.arange(j_lo, j_hi + 1)
            rows.extend([ii] * len(js))
            cols.extend(js.tolist())
            vals.extend((a[k - js]).tolist())
        
        if len(vals) == 0:
            break
        A_ub = sparse.csc_matrix((vals, (rows, cols)), shape=(K, n))
        
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=np.full(K, max_b),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 20, 'presolve': True})
        except Exception:
            break
        
        if not res.success:
            K = max(10, K // 2)
            continue
        
        g = res.x
        Sg = g.sum()
        if Sg < 1e-10:
            break
        g = (S / Sg) * g
        
        a_new = (1-t)*a + t*g
        a_new = np.clip(a_new, 0.0, 1000.0)
        score = float(evaluate_sequence(a_new.tolist()))
        
        if np.isfinite(score) and score < best_score:
            best = a_new
            best_score = score
        elif t > 0.01:
            a_new2 = (1-t*0.5)*a + t*0.5*g
            a_new2 = np.clip(a_new2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(a_new2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = a_new2
                best_score = score2
