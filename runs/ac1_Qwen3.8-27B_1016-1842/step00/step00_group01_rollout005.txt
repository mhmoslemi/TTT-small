def fft_convolve(a):
    """Compute linear self-convolution of a via FFT. O(n log n)."""
    n = len(a)

    P = 1
    while P < conv_len:
        P *= 2
    A = np.fft.rfft(a, P)
    c = np.fft.irfft(A * A, P)
    return c[:conv_len]
