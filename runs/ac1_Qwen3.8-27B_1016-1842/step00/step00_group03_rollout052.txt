import time

def fft_convolve(a):
    """Compute full linear convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n  # next power of 2 or just 2n
    f = np.fft.rfft(a, size)
    b = np.fft.irfft(f * f, size)
    return b[:2*n-1]

    """Compute the AC1 score for a non-negative sequence."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    b = fft_convolve(a)
    return 2.0 * n * np.max(b) / (S * S)
