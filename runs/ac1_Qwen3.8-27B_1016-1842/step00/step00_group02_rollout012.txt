                    trial[m] += delta
                    s = score_from_params(trial)
                    if s < best_local_score:
                        best_local = trial
                        best_local_score = s
                        step = max(0.001, step * 0.7)
                        improved = True
                        break
                else:
                    break
            best_params = best_local
            best_score = best_local_score
        if not improved:
            break
    
    # Decode final
    best_a = np.ones(n) + basis @ best_params
    best_a = np.clip(best_a, 0.0, 1000.0)
    
    # Verify with exact evaluator
    final_score = float(evaluate_sequence(best_a.tolist()))
    
    # Also check parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    parent_score = float(evaluate_sequence(parent.tolist()))
    
    if parent_score < final_score:
        best_a = parent
        final_score = parent_score
    
    return [float(x) for x in best_a]
