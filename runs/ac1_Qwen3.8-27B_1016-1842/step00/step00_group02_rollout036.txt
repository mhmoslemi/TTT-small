def project_simplex(x):
    # x has length n, project to {a >= 0, sum(a) = 1}
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u - (cssv - 1) / np.arange(1, n+1)
    found = np.where(cond > 0)[0]
    if len(found) == 0:
        return np.ones(n) / n
    rho = found[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(x - theta, 0)

This is O(n log n) due to sorting. For n=6962, this is fast.

def project_simplex_fast(x):
    """Project x onto {a >= 0, sum(a)=1} using simple clip + renormalize."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        return np.ones(x.shape[0]) / x.shape[0]
    return x / s

def full_conv(a, N):
    """Full linear convolution via FFT. N >= 2*len(a)-1."""
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)

def compute_score_normalized(a):
    """Score for sum(a)=1: 2n * max(conv)."""
    n = a.size
    conv = full_conv(a, 2*n-1)
    return 2.0 * n * conv.max()

def subgradient_step(a, alpha, momentum_arr, mu):
    """One subgradient step with momentum. Returns (new_a, new_momentum, max_val, k_star)."""
    n = a.size
    N = 2*n - 1
    conv = full_conv(a, N)
    conv = np.maximum(conv, 0)  # remove numerical noise
    max_val = conv.max()
    
    # Active set: lags within 90% of max (tunable)
    threshold = max_val * 0.90
    active = np.where(conv >= threshold)[0]
    
    # If too few active, expand
    if len(active) < 10:
        # Take top 10
        idx = np.argsort(conv)[::-1][:10]
        active = idx
    
    # Compute subgradient: sg[j] = mean over active k of a[k-j]
    sg = np.zeros(n)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            sg[idx] += a[k - idx]
    sg /= len(active)
    
    # Center the subgradient (to respect sum constraint)
    sg = sg - sg.mean()
    
    # Momentum (Nesterov-like)
    velocity = mu * momentum_arr - alpha * sg
    momentum_arr = velocity
    a_new = project_simplex_fast(a + velocity)
    
    return a_new, momentum_arr, max_val
