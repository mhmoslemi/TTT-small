def _lp_dir(a, n, iters):
    y = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    cdf = np.empty(2 * n)
    ae = np.arange(n)
    ie = np.minimum(ae + n, 2 * n - 1)
    for _ in range(iters):
        cdf[0] = 0.0
        np.cumsum(y, out=cdf[1:])
        W = cdf[ie] - cdf[ae]

        jstar = int(viol.argmin())
