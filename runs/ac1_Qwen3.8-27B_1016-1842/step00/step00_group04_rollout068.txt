from scipy.optimize import linprog


def build_conv_matrix(a):
    n = len(a)
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = np.tile(np.arange(n, dtype=np.int64), n) + col_idx
    val = np.tile(a, n).astype(np.float64)
    return sparse.csr_matrix((val, (row_idx, col_idx)), shape=(2 * n - 1, n))


def project_sum_nonneg(v, target):
    lo = float(v.min()) - 1.0

            lo = mid
        else:
            hi = mid
    return np.maximum(v - 0.5 * (lo + hi), 0.0)


def compute_conv(a):

def subgradient_at_max(a, conv):

    k_hi = min(n - 1, k)
    j_idx = np.arange(k_lo, k_hi + 1)
    g[j_idx] = 2.0 * a[k - j_idx]
    return g


def lp_direction(a, M, budget_check):

                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': max(1.0, budget_check())})
    if not res.success or res.x is None:
        return None
    s = res.x.sum()
    if s < 1e-6:
