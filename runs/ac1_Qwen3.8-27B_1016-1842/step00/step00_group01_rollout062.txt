def _make_inits(n, rng):

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    phases = rng.uniform(0.0, 2.0 * np.pi, n)
    inits.append(np.abs(np.fft.ifft(np.exp(1j * phases))).real)
    return inits

    n = len(height_sequence_1)
    rng = np.random.default_rng(seed)

    # Best initialization
    inits = _make_inits(n, rng)
    best_raw = np.ones(n)
    best_score = np.inf

        a0 = np.clip(a0, 0.0, 1000.0)

            a0 = np.ones(n)
        sc = float(evaluate_sequence(a0.tolist()))
        if np.isfinite(sc) and sc < best_score:

    b = best_raw.copy()

    # FFT size for cross-correlation (w has length 2n-1, b has length n)
    fft_size = 1
    need = (2 * n - 1) + n - 1  # = 3n - 2
    while fft_size < need:

    step = 0.5
    momentum = 0.3
    v = np.zeros(n, dtype=np.float64)
    no_improve = 0
    target = 1.50274349
    soft_param = 0.005  # fraction of max_c for softmax

    while time.monotonic() < deadline:
        c = np.convolve(b, b, mode='full')
        max_c = c.max()
        score = 2.0 * n * max_c

            best_score = score
            best_raw = b.copy()
            no_improve = 0

                break
        else:

            if s < 1e-15:

            else:

            v *= 0.0
            step = 0.5
