def fft_conv(a):
    n = int(a.size)
    if n == 0: return np.zeros(0, dtype=np.float64)
    m = 2*n - 1
    L = 1
    while L < m:
        L <<= 1
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)[:m]
    np.maximum(b, 0.0, out=b)
    return b

import time
import math
import numpy as np


def sanitize_parent(parent, fallback_n=6733):
    try:
        arr = np.asarray(parent, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(fallback_n, dtype=np.float64)
    if arr.size == 0:
        arr = np.ones(fallback_n, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    return arr / s


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        fallback = 6733
        return np.ones(fallback, dtype=np.float64) / float(fallback)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def fft_conv(a):
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)[:m]
    np.maximum(b, 0.0, out=b)
    return b


def score_fast(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    n = int(arr.size)
    if n == 0:
        return float("inf")
    s = float(arr.sum())
    if s <= 1e-15:
        return float("inf")
    b = fft_conv(arr)
    mx = float(np.max(b))
    if not math.isfinite(mx):
        return float("inf")
    return 2.0 * float(n) * mx / (s * s)


def exact_score(p):
    try:
        return float(evaluate_sequence([float(x) for x in p]))
    except Exception:
        return float("inf")


def softmax_vector(x):
    x = np.asarray(x, dtype=np.float64)
    n = int(x.size)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    y = x - float(np.max(x))
    p = np.exp(y)
    z = float(np.sum(p))
    if not math.isfinite(z) or z <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / z


def top_lags(b, q):
    q = int(q)
    m = int(b.size)
    if m <= 1:
        return np.arange(m, dtype=np.int64)
    if q <= 1:
        return np.array([int(np.argmax(b))], dtype=np.int64)
    if q >= m:
        return np.arange(m, dtype=np.int64)
    return np.argpartition(b, -q)[-q:].astype(np.int64, copy=False)


def lag_weights(t):
    t = np.asarray(t, dtype=np.float64)
    if t.size == 0:
        return np.array([], dtype=np.float64)
    mx = float(np.max(t))
    if not math.isfinite(mx) or mx <= 0.0:
        return np.ones(t.size, dtype=np.float64) / float(t.size)
    if t.size == 1:
        return np.array([1.0], dtype=np.float64)
    mn = float(np.min(t))
    temp = mx - mn
    if temp <= 1e-14:
        temp = max(1e-14, 1e-3 * abs(mx))
    e = np.exp((t - mx) / temp)
    s = float(np.sum(e))
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(t.size, dtype=np.float64) / float(t.size)
    return e / s


def weighted_reverse_push(p, idx, weights):
    n = int(p.size)
    h = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(idx, weights):
        k = int(kk)
        i0 = k - (n - 1)
        if i0 < 0:
            i0 = 0
        i1 = k
        if i1 >= n:
            i1 = n - 1
        if i0 > i1:
            continue
        start = k - i0
        last = k - i1
        if last >= 0:
            stop = last - 1
            if stop >= 0:
                seg = p[start:stop:-1]
            else:
                seg = p[start::-1]
        else:
            seg = p[start::-1]
        h[i0:i1 + 1] += float(ww) * seg
    return h


def optimize_sequence(p_init, rng, stop_time):
    p = normalize_positive(p_init)
    n = int(p.size)
    if n == 0:
        return None, float("inf")
    best_p = p.copy()
    best_score = score_fast(p)
    eps = 1e-15
    x = np.log(np.maximum(p, eps))
    x -= float(np.mean(x))
    lr = 0.2
    no_success = 0
    jitter = 0.12
    it = 0
    qcycle = (2, 4, 8, 16, 32)
    while time.monotonic() < stop_time:
        q = qcycle[min(len(qcycle) - 1, it // 500)]
        p = softmax_vector(x)
        b = fft_conv(p)
        s = float(p.sum())
        if s <= 1e-15:
            break
        score = 2.0 * float(n) * float(np.max(b)) / (s * s)
        if not math.isfinite(score):
            break
        if score < best_score:
            best_score = score
            best_p = p.copy()
        idx = top_lags(b, q)
        w = lag_weights(b[idx])
        h = weighted_reverse_push(p, idx, w)
        g = 2.0 * h
        dot = float(np.dot(p, g))
        grad = p * (g - dot)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            x = x + rng.normal(0.0, jitter, n)
            x -= float(np.mean(x))
            jitter *= 0.8

        accepted = False

            mn = float(np.min(x_new))
            if mx - mn > 120.0:
                x_new = np.clip(x_new, -60.0, 60.0)
            p_new = softmax_vector(x_new)
            b_new = fft_conv(p_new)
            s_new = float(p_new.sum())
            if s_new <= 1e-15:
                continue
            score_new = 2.0 * float(n) * float(np.max(b_new)) / (s_new * s_new)
            if math.isfinite(score_new) and score_new < score - 1e-12:
                x = x_new
                p = p_new
                b = b_new
                score = score_new
                accepted = True
                break
        if accepted:
            no_success = 0
            if lr < 0.8:
                lr *= 1.05
        else:
            no_success += 1
            lr *= 0.6
            if lr < 1e-4 or no_success > 18:
                x = x + rng.normal(0.0, jitter, n)
                x -= float(np.mean(x))
                jitter *= 0.7
                if jitter < 1e-4:
                    jitter = 0.1
                lr = 0.2
                no_success = 0
        it += 1
    return best_p, best_score


def unit_direction(v):
    v = np.asarray(v, dtype=np.float64)
    nv = float(np.linalg.norm(v))
    if nv <= 1e-14:
        return np.zeros(v.shape, dtype=np.float64)
    return v / nv


def harmonic_candidate(amps, comps):
    if len(comps) == 0:
        n = 6733
        return np.ones(n, dtype=np.float64) / float(n)
    n = int(comps[0].size)
    g = np.zeros(n, dtype=np.float64)
    for a, s in zip(amps, comps):
        a = float(a)
        if a != 0.0:
            g += a * s
    if not np.all(np.isfinite(g)):
        return np.ones(n, dtype=np.float64) / float(n)
    if float(np.max(g)) - float(np.min(g)) <= 1e-14:
        return np.ones(n, dtype=np.float64) / float(n)
    if float(np.min(g)) >= 0.0:
        g = -g
    mn = float(np.min(g))
    if mn >= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    p = 1.0 + (-1.0 / mn) * g
    if not np.all(np.isfinite(p)):
        return np.ones(n, dtype=np.float64) / float(n)
    p = np.maximum(p, 0.0)
    ss = float(p.sum())
    if ss <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return p / ss


def harmonic_eval(amps, comps):
    p = harmonic_candidate(amps, comps)
    return score_fast(p), p


def harmonic_search(n, comps, rng, stop_time):
    M = len(comps)
    if M == 0:
        return None, float("inf"), normalize_positive(np.ones(n) / float(n)) if n > 0 else np.ones(1)
    best_amps = np.zeros(M, dtype=np.float64)
    best_score = float("inf")
    best_p = np.ones(n, dtype=np.float64) / float(n)
    evals = 0
    max_evals = 8000

    base_dirs = []
    for decay in (0.75, 0.5):
        v = np.zeros(M, dtype=np.float64)
        for m in range(M):
            sign = 1.0 if (m % 2 == 0) else -1.0
            v[m] = sign * (decay ** m)
        base_dirs.append(unit_direction(v))
    v = np.zeros(M, dtype=np.float64)
    v[0] = 1.0
    base_dirs.append(unit_direction(v))
    v = np.ones(M, dtype=np.float64)
    base_dirs.append(unit_direction(v))

    for v in base_dirs:
        if time.monotonic() >= stop_time or evals >= max_evals:
            break
        sc, pp = harmonic_eval(v, comps)
        evals += 1
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amps = v.copy()
            best_p = pp.copy()

    for _ in range(80):
        if time.monotonic() >= stop_time or evals >= max_evals:
            break
        v = rng.standard_normal(M)
        if rng.random() < 0.5:
            v = -v
        v = unit_direction(v)
        sc, pp = harmonic_eval(v, comps)
        evals += 1
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amps = v.copy()
            best_p = pp.copy()

    step = 0.35
    stuck = 0
    while time.monotonic() < stop_time and evals < max_evals:
        improved = False
        order = rng.permutation(M)
        for dim in order:
            if time.monotonic() >= stop_time or evals >= max_evals:
                break
            for mult in (0.7, -0.7, 0.35, -0.35, 0.15, -0.15):
                if time.monotonic() >= stop_time or evals >= max_evals:
                    break
                trial = best_amps.copy()
                trial[dim] += step * mult
                trial = unit_direction(trial)
                sc, pp = harmonic_eval(trial, comps)
                evals += 1
                if math.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_amps = trial
                    best_p = pp.copy()
                    improved = True
                    break
        if not improved:
            step *= 0.75
            stuck += 1
            if step < 0.02 or stuck > 30:
                jump = best_amps + rng.normal(0.0, 0.5, M)
                jump = unit_direction(jump)
                sc, pp = harmonic_eval(jump, comps)
                evals += 1
                if math.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_amps = jump
                    best_p = pp.copy()
                    stuck = 0
                    step = 0.3
                else:
                    step = 0.18
                stuck = 0
    return best_amps, best_score, best_p


def zero_small(p, threshold):
    arr = np.asarray(p, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr[arr < threshold] = 0.0
    s = float(arr.sum())
    if s <= 1e-15:
        return arr
    return arr / s


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    budget = max(1.0, float(budget_s))
    hard = t0 + max(0.0, budget - 5.0)
    deadline = t0 + max(0.0, budget - 45.0)
    rng = np.random.default_rng(seed)

    try:
        base = sanitize_parent(height_sequence_1)
    except Exception:
        base = sanitize_parent(np.ones(6733, dtype=np.float64))

    n = int(base.size)
    best_p = base.copy()
    best_score = score_fast(best_p)

    if time.monotonic() < deadline - 30.0:
        sc0 = exact_score(best_p)
        if math.isfinite(sc0) and sc0 < 1.50274349:
            best_score = min(best_score, sc0)

    M = 5
    if n > 16 and time.monotonic() < deadline - 120.0:
        idx = np.arange(n, dtype=np.float64) / float(n)
        comps = []
        twopi = 2.0 * math.pi
        for m in range(1, M + 1):
            comps.append(np.sin(twopi * float(m) * idx))
        harm_stop = min(deadline - 120.0, t0 + min(260.0, 0.28 * budget))
        if harm_stop > t0 + 5.0:
            amps, harm_score, harm_p = harmonic_search(n, comps, rng, harm_stop)
            if harm_p is not None and math.isfinite(harm_score) and harm_score < best_score:
                best_score = harm_score
                best_p = harm_p

    full_stop = deadline - 30.0
    if time.monotonic() < full_stop - 60.0:
        opt_p, opt_score = optimize_sequence(best_p, rng, full_stop)
        if opt_p is not None and math.isfinite(opt_score) and opt_score < best_score:
            best_score = opt_score
            best_p = opt_p

    final_p = np.asarray(best_p, dtype=np.float64)
    final_p = np.where(np.isfinite(final_p), final_p, 0.0)
    final_p = np.clip(final_p, 0.0, 1000.0)
    ss = float(final_p.sum())
    if ss <= 1e-8:
        final_p = np.ones(n, dtype=np.float64) / float(n)
    else:
        final_p = final_p / ss

    candidates = [final_p]
    pz = zero_small(final_p, 1e-11)
    if int(pz.size) > 0 and float(pz.sum()) > 1e-8:
        candidates.append(pz)

    chosen = final_p
    if time.monotonic() < hard - 75.0:
        best_sc = float("inf")
        for cand in candidates:
            sc = exact_score(cand)
            if math.isfinite(sc) and sc < best_sc:
                best_sc = sc
                chosen = cand

    chosen = np.asarray(chosen, dtype=np.float64)
    chosen = np.where(np.isfinite(chosen), chosen, 0.0)
    chosen = np.clip(chosen, 0.0, 1000.0)
    cz = float(chosen.sum())
    if cz <= 1e-8:
        chosen = np.ones(n, dtype=np.float64) / float(n)
    else:
        chosen = chosen / cz

    return [float(x) for x in chosen]
