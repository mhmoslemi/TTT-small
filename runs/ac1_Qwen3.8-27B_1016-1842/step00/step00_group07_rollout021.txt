import numpy as np
import time

def lse_grad_and_peak(a, tau):
    """Returns (gradient, max_conv_value, peak_lag)."""
    n = a.size
    b = np.convolve(a, a)
    peak_lag = int(np.argmax(b))
    max_b = b[peak_lag]
    # LSE weights
    b_shift = b - max_b
    p = np.exp(b_shift / tau)
    p_sum = p.sum()
    p /= p_sum
    # Gradient via correlation
    grad = np.correlate(p, a, mode='valid')
    return grad, max_b, peak_lag

```python
from scipy import sparse
from scipy.optimize import linprog

def solve_dual_lp(f, M):
    """Solve the dual LP to find g maximizing sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Build sparse constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j+n):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```
