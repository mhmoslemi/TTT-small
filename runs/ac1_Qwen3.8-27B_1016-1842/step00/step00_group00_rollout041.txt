import time


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n == 0:
        return [1.0]

    # For any non-negative a, the AC1 score 2*n*max(conv(a,a))/sum(a)^2 >= 2
    # by Cauchy-Schwarz: max(conv) >= conv[0] = sum(a^2) >= (sum(a))^2/n.
    # Equality holds iff all entries are equal. So the global optimum is uniform.
    s = float(np.sum(parent))
    if s < 0.01:
        c = max(1.0, 0.01 / n)
    else:
        c = s / n
    uniform = np.full(n, c, dtype=np.float64)

    best = uniform.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    par_score = float(evaluate_sequence(parent.tolist()))
    if (not np.isfinite(best_score)) or (np.isfinite(par_score) and par_score < best_score):
        best = parent.copy()
        best_score = par_score

    # Tiny refinement loop: in theory uniform is optimal, but allow the
    # evaluator's floating-point arithmetic to confirm or (rarely) beat it.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)
    no_improve = 0
    while time.monotonic() < deadline and no_improve < 200:

        idx = rng.choice(n, size=k, replace=False)
        cand = best.copy()
        cand[idx] *= np.exp(rng.normal(0.0, 1e-5, size=k))

        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            no_improve = 0

            no_improve += 1

    uni_score = float(evaluate_sequence(uniform.tolist()))
    if np.isfinite(uni_score) and uni_score < best_score:
