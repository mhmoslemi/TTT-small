def _fft_convolve(a):

    while size < 2 * n - 1:

    fc = np.fft.irfft(fa * fa, size)
    return fc[:2 * n - 1]

def _score(a, n):
    s = a.sum()
    if s < 0.01:
        return 1e18
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_init(kind, n, rng):
    j = np.arange(n, dtype=np.float64)
    if kind == 0:

import numpy as np
import time

def _fft_convolve(a):
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fc = np.fft.irfft(fa * fa, size)
    return fc[:2 * n - 1]

def _score(a, n):
    s = a.sum()
    if s < 0.01:
        return 1e18
    c = _fft_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _make_init(kind, n, rng):
    j = np.arange(n, dtype=np.float64)

    elif kind == 3:
