import numpy as np
import time

def simplex_project(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in O(n log n)."""
    n = v.size
    if n == 0:
        return v
    sv = np.sort(v)[::-1]
    cs = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    w = np.where(sv * rho > cs - s)[0]
    if w.size == 0:
        return np.full(n, s / n)
    r = int(w[-1])
    theta = (cs[r] - s) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_conv(x):
    """Full convolution x*x via FFT. O(n log n)."""
    n = x.size
    N = 1
    target = 2 * n - 1
    while N < target:
        N <<= 1
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X, N)[:target]

def fft_xcorr(c, x):
    """Cross-correlation: T[j] = sum_k c[k]*x[k-j] for j=0..n-1 via FFT."""
    n = x.size
    m = c.size
    N = 1
    target = m + n - 1
    while N < target:
        N <<= 1
    C = np.fft.rfft(c, N)
    X = np.fft.rfft(x, N)
    return np.fft.irfft(C * np.conj(X), N)[:n]

import numpy as np
import time

def simp_proj(v, s=1.0):
    """Project v onto the probability simplex {x>=0, sum(x)=s}."""
    n = v.size
    sv = np.sort(v)[::-1]
    cs = np.cumsum(sv)
    w = np.where(sv * np.arange(1, n+1) > cs - s)[0]
    if w.size == 0:
        return np.full(n, s / n)
    r = int(w[-1])
    return np.maximum(v - (cs[r] - s) / (r + 1.0), 0.0)

def fconv(x):
    """Full self-convolution via FFT."""
    n = x.size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X, N)[:2 * n - 1]

def fxcorr(c, x):
    """T[j]=sum_k c[k]*x[k-j], j=0..n-1, via FFT."""
    n = x.size
    N = 1
    while N < c.size + n - 1:
        N <<= 1
    return np.fft.irfft(
