import numpy as np
import time


def _score(a):
    """Compute the AC1 score for a nonnegative array."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * b.max() / (s * s))


def _subgrad_direction(a, k_star):
    """Return the subgradient of the score w.r.t. a at the maximizing lag."""
    n = a.size
    s = a.sum()
    mb = np.convolve(a, a)[k_star]
    idx = np.arange(n)
    kmi = k_star - idx
    mask = (kmi >= 0) & (kmi < n)
    grad = np.zeros(n)
    grad[mask] = 2.0 * a[kmi[mask]]
    return (2.0 * n / (s * s)) * grad - (4.0 * n * mb / (s * s * s))

    best_score = float(evaluate_sequence(best.tolist()))
    for c in cands:
        if c.size != n or not np.isfinite(c).all() or c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = c.copy()

    # ---- main search loop ----
    step = 0.05
    stuck = 0
    max_stuck = 300
    slow = 0.9992
    fast = 1.04

    while time.monotonic() < deadline:
        a = best
        b_full = np.convolve(a, a)
        k_star = int(np.argmax(b_full))

        if np.isfinite(best_score) and best_score < 1.503:
            # very good; fine-tune with tiny steps
            step = min(step, 0.01)

        # subgradient step
        sg = _subgrad_direction(a, k_star)
        a_new = a - step * sg

        # random block perturbation (exploration)
        if rng.random() < 0.6:
            w_lo = max(1, n // max(200, int(n ** 0.55)))
            w_hi = max(w_lo + 1, n // max(15, int(n ** 0.35)))
            a_new = _block_perturb(a_new, rng, (w_lo, w_hi), 0.15)

        # occasional large jump to escape local minima
        if stuck > max_stuck:
            w_lo = max(2, n // 50)
            w_hi = max(w_lo + 1, n // 4)
            a_new = _block_perturb(best, rng, (w_lo, w_hi), 0.6)
            stuck = 0

        a_new = np.clip(a_new, 0.0, 1000.0)
        sc = float(evaluate_sequence(a_new.tolist()))

        if np.isfinite(sc) and sc < best_score:
            best = a_new
            best_score = sc
            stuck = 0
            step = min(step * fast, 0.5)
        else:
            stuck += 1
            step = max(step * slow, 1e-5)

    return [float(x) for x in best]
