import time


def _conv_peak(a):
    """Compute max of linear convolution a*a using FFT (O(n log n))."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return float(np.max(c))

    """Fast approximate score without full evaluator call."""
    s = float(np.sum(a))
    if s < 0.01:
        return 1e30
    return 2.0 * a.size * _conv_peak(a) / (s * s)
