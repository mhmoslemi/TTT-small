    n = len(a) + len(b) - 1
    size = 1 << (n-1).bit_length()  # next power of 2
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n]

def fast_conv(a):
    """Compute a * a using FFT. Returns full convolution."""
    n = len(a)
    size = 1 << (2*n-1-1).bit_length()  # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def make_block_perturb(base, rng, block_frac=0.1):
    n = len(base)
    c = base.copy()
    width = max(2, int(n * block_frac))
    start = rng.integers(0, n - width)
    c[start:start+width] *= np.exp(rng.normal(0, 0.3))
    return np.clip(c, 0, 1000)

def make_element_perturb(base, rng, frac=0.1):
    n = len(base)
    c = base.copy()
    n_pert = max(1, int(n * frac))
    idx = rng.choice(n, n_pert, replace=False)
