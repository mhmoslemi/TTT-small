    C = fft_convolve(a)
    M = C.max()
    S = a.sum()
    
    # Find top-k peaks
    k = min(20, 2*n-1)
    top_c = np.sort(C)[-k:][::-1]  # descending
    top_idx = np.argsort(C)[-k:][::-1]
    
    # Softmax weights
    weights = np.exp((top_c - top_c[0]) * temp)

    # M_soft approximation (we use actual M for the score)

    # Gradient using soft weights

def fft_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C


def score_fft(a):
    n = len(a)
    C = fft_convolve(a)
    S = a.sum()
    return 2.0 * n * float(C.max()) / (S * S)

    n = len(a)
    C = fft_convolve(a)
    k = min(top_k, 2 * n - 1)
