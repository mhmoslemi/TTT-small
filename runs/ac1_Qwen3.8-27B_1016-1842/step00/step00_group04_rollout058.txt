    if np.isfinite(fs):
        best_sc = min(best_sc, fs)

    for attempt in range(25):
        if time.monotonic() > dl - 60:
            break

        # Diverse initializations
        if attempt == 0:
            a = np.ones(n) + 0.4 * rng.standard_normal(n)
        elif attempt == 1:
            f = rng.integers(2, 15)
            a = 1.0 + 0.6 * np.sin(2.0 * np.pi * f * np.arange(n) / n) + 0.1 * rng.standard_normal(n)
        elif attempt == 2:
            p = np.linspace(0, 1, n)
            a = 0.3 + 1.4 * np.sin(np.pi * p) ** 2 + 0.1 * rng.standard_normal(n)
        elif attempt == 3:
            bsz = max(1, n // 12)
            a = np.ones(n)
            for bi in range(0, n, bsz):
                a[bi:min(n, bi + bsz)] *= rng.uniform(0.2, 2.5)
        elif attempt == 4:
            f1 = rng.integers(1, 8)
            f2 = rng.integers(f1 + 1, f1 + 10)
            a = 1.0 + 0.4 * np.sin(2.0 * np.pi * f1 * np.arange(n) / n) + 0.4 * np.cos(2.0 * np.pi * f2 * np.arange(n) / n)
        else:
            a = best.copy() + 0.3 * rng.standard_normal(n)

        a = np.clip(a, 0.0, 1000.0)
        a = proj_simplex(a, np.float64(n))
        sc0 = score_of(a)
        if sc0 < best_sc:
            best_sc = sc0
            best = a.copy()

        # Subgradient with momentum and multi-peak
        eta = 0.4
        mom = np.zeros(n, dtype=np.float64)
        beta = 0.6
        best_ph = a.copy()
        best_ph_sc = sc0
        patience = 0

        for it in range(80000):
            if time.monotonic() > dl - 30:
                break

            c = fast_conv(a)
            # Use top-3 peaks for balanced subgradient
            nn = len(c)
            topk = min(3, nn)
            peaks = np.argpartition(c, -topk)[-topk:]
            peak_max = c[peaks].max()
            # Only use peaks close to the max
            active = peaks[c[peaks] > 0.7 * peak_max]

            g = np.zeros(n, dtype=np.float64)
            for kp in active:
                kp = int(kp)
                lo = max(0, kp - n + 1)
                hi = min(n - 1, kp)
                ia = np.arange(lo, hi + 1)
                g[ia] += a[kp - ia]
            g /= float(len(active)) if len(active) > 0 else 1.0

            mom = beta * mom + g
            gnorm = float(np.linalg.norm(mom))
            if gnorm < 1e-12:
                break
            step = eta * n / gnorm
            a = a - step * mom
            a = proj_simplex(a, np.float64(n))

            if it % 80 == 79:
                sc = score_of(a)
                if sc < best_ph_sc:
                    best_ph_sc = sc
                    best_ph = a.copy()
                    patience = 0
                else:
                    patience += 1
                    if patience > 30:
                        eta = max(0.05, eta * 0.9)
                    if patience > 100:
                        # Random kick to escape
                        kick_mask = rng.random(n) < 0.08
                        a[kick_mask] *= np.exp(rng.normal(0.0, 0.6, int(kick_mask.sum())))
                        a = np.clip(a, 0.0, 1000.0)
                        a = proj_simplex(a, np.float64(n))
                        patience = 0
                        eta = 0.3
                    if patience > 300:
                        break

        sc = best_ph_sc
        if sc < best_sc:
            best_sc = sc
            best = best_ph.copy()

    # Local refinement: block scale perturbations
    for it in range(80000):
        if time.monotonic() > dl - 10:
            break
        a = best.copy()
        w = int(rng.integers(4, max(5, n // 15)))
        st = int(rng.integers(0, max(1, n - w + 1)))
        a[st:st + w] *= np.exp(rng.normal(0.0, 0.2, w))
        a = np.clip(a, 0.0, 1000.0)
        sc = score_of(a)
        if sc < best_sc:
            best_sc = sc
            best = a.copy()

    # Final validation with evaluator
    ev = evaluate_sequence(best.tolist())
    if not np.isfinite(ev):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
