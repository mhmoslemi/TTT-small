import numpy as np
import time


def _evaluate(arr):
    """Wrapper around the authoritative evaluator for NumPy arrays."""
    return float(evaluate_sequence([float(x) for x in arr]))


def _block_perturb(base, rng, n):
    """O(n) nonnegative multiplicative block perturbation."""
    c = base.copy()
    width = int(rng.integers(max(1, n // 400), max(2, n // 15)))
    start = int(rng.integers(0, max(1, n - width)))
    c[start:start + width] *= float(rng.normal(1.0, 0.6))
    return np.clip(c, 0.0, 1000.0)


def _smooth_perturb(base, rng, n):
    """O(n) additive smooth bump perturbation."""
    c = base.copy()
    center = float(rng.integers(n // 4, 3 * n // 4 + 1))
    sigma = float(rng.integers(max(2, n // 30), max(3, n // 4)))
    idx = np.arange(n, dtype=np.float64)
    bump = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    amp = float(rng.normal(0.0, 0.15 * np.mean(base) + 1e-6))
    c += amp * bump
    return np.clip(c, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)

    # --- Initialization ---
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    best = parent.copy()
    best_score = _evaluate(best)

    # Try a pure constant (theoretical optimum = 2.0)
    const = np.full(n, 1.0, dtype=np.float64)
    cs = _evaluate(const)
    if np.isfinite(cs) and cs < best_score:
        best = const
        best_score = cs

    # --- Local search (will not improve below 2 by Cauchy–Schwarz,
    #     but serves as a safeguard against implementation edge-cases) ---
    stall = 0
    max_stall = 500  # stop after this many non-improving tries

    while time.monotonic() < deadline and stall < max_stall:
        # Pick perturbation type
        if rng.random() < 0.7:
            candidate = _block_perturb(best, rng, n)
        else:
            candidate = _smooth_perturb(best, rng, n)

        if not np.isfinite(candidate).all() or candidate.sum() < 0.01:
            stall += 1
            continue

        score = _evaluate(candidate)
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            stall = 0
        else:
            stall += 1

    # Final safety: make sure we return a valid finite non-empty list
    result = [float(x) for x in best]
    if not result or any(not np.isfinite(x) for x in result):
        result = [1.0] * n
    return result
