        best_score = float(evaluate_candidate(best))
    
    # Local search
    while time.monotonic() < deadline:
        # Generate perturbation
        candidate = best.copy()
        # Block perturbation
        block_size = rng.integers(max(1, n//200), max(2, n//20))
        start = rng.integers(0, n - block_size)
        candidate[start:start+block_size] *= np.exp(rng.normal(0, 0.05, size=block_size))

        if candidate.sum() < 0.01:

        score = float(evaluate_candidate(candidate))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    
    return [float(x) for x in best]
