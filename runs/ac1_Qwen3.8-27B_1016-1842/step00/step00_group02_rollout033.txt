def _fft_autocorr(a):
    """Full linear autocorrelation via FFT. O(n log n)."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]
