import time

    a /= a.sum()

    a += 0.01 * np.cos(2*np.pi*np.arange(n)/n) + 0.01*rng.standard_normal(n)
    a = np.clip(a, 0, None)
    a /= a.sum()
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Adam-like optimizer state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    lr = 0.01
    
    eval_interval = max(1, int(50))  # evaluate every 50 steps
    
    while time.monotonic() < deadline:
        # Compute subgradient
        conv = np.convolve(a, a)
        S = a.sum()
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        g = np.full(n, -4.0*n*M/S**3)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                g[i] = 4.0*n*a[j]/(S*S) - 4.0*n*M/S**3
        
        # Check if all zero (converged)
        if np.abs(g).max() < 1e-12:
            break
        
        # Adam update
        t_step += 1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*g**2
        m_hat = m/(1-beta1**t_step)
        v_hat = v/(1-beta2**t_step)
        
        a_new = a - lr*m_hat/(np.sqrt(v_hat)+eps)
        a_new = np.clip(a_new, 0, 1000)
        if a_new.sum() < 0.01:
            a_new = a
        else:
            a_new /= a_new.sum()
        a = a_new
        
        # Periodic evaluation
        if t_step % eval_interval == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = score

            w = max(2, n // 40)
            s = int(rng.integers(0, n - w))
            a[s:s + w] *= np.exp(rng.normal(0.0, 0.12, size=w))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() > 0.01:
                a /= a.sum()

        # If stuck, try a bigger restart perturbation
        if stall_count > stall_limit:
            for trial in range(5):
                p = a.copy()
                p += 0.15 * rng.standard_normal(n)
                p = np.clip(p, 0.0, None)
                if p.sum() > 0.01:
                    p /= p.sum()
                    sc = float(evaluate_sequence(p.tolist()))
                    if np.isfinite(sc) and sc < best_score:
                        best = p.copy()
                        best_score = sc
                        a = p.copy()
                        stall_count = 0
                        break

    # Final exact evaluation
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best = a.copy()
        best_score = sc

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.full(n, 0.02 / n)
    return [float(x) for x in best]
