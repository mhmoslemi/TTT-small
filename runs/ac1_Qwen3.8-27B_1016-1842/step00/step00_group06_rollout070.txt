import numpy as np
import time

def compute_score(a):
    """Compute the score without calling evaluate_sequence (for speed)."""
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * len(a) * max_b / (sum_a ** 2)

def subgradient_step(a, lr):
    """One subgradient step to reduce max convolution."""
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    n = len(a)
    # gradient: g[i] = a[k_star - i] if 0 <= k_star - i < n
    k_min = max(0, k_star - n + 1)
    k_max = min(n - 1, k_star)
    indices = np.arange(k_min, k_max + 1)
    grad = np.zeros(n)
    grad[indices] = a[k_star - indices]
    # descent
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def block_perturb(a, rng, width=None):
    """Perturb a random block multiplicatively."""
    n = len(a)
    if width is None:

    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0, 0.1, size=width))
    a_new = a.copy()
    a_new[start:start+width] *= factor
    return np.clip(a_new, 0, 1000)

import time

def fast_score(a):
    conv = np.convolve(a, a)
    mb = conv.max()
    sa = a.sum()
    if sa < 1e-10:
        return np.inf
    return 2.0 * len(a) * mb / (sa * sa)

def subgradient(a, lr):
    n = len(a)
    conv = np.convolve(a, a)
    ks = int(np.argmax(conv))
    lo = max(0, ks - n + 1)
    hi = min(n - 1, ks)
    idx = np.arange(lo, hi + 1)
    g = np.zeros(n)
    g[idx] = a[ks - idx]
    a2 = a - lr * g
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s < 1e-10:
        return a.copy()
    return a2 / s

def block_pert(a, rng, wlo=5, whi=None):
    n = len(a)
    if whi is None:
        whi = max(10, n // 5)

    f = np.exp(rng.normal(0, 0.15, w))
    a2 = a.copy()
    a2[st:st+w] *= f
    return np.clip(a2, 0, 1000.0)
