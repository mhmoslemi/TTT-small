def _conv(a):

    n = len(a)
    size = 1

    b = _conv(a)
    max_b = b.max()
    sum_a = a.sum()

import time

def _fft_conv(a):
    n = a.size
    size = 2 * n - 1
    # Next power of 2 for FFT efficiency
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

def _score_from_conv(a, b):
    sum_a = a.sum()
    max_b = b.max()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * a.size * max_b / (sum_a * sum_a)

def _grad_from_conv(a, b):
    n = a.size
    sum_a = a.sum()
    max_b = b.max()
    # Use all near-argmax positions for a smoother subgradient
    # Find positions where b[k] > max_b * (1 - 1e-6)
    threshold = max_b * (1 - 1e-8)
    peaks = np.where(b >= threshold)[0]
    
    grad = np.zeros(n)
    for k in peaks:
        # grad[j] += 2*a[k-j] for valid j
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # j in [lo, hi], k-j in [k-hi, k-lo]
            # a[k-j] for j=lo..hi = a[k-hi..k-lo] reversed
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    
    # Average over peaks
    npeaks = len(peaks)
    grad /= npeaks
    # Multiply by 2 (from the subgradient formula: 2*a[k-j])
    grad *= 2.0
    
    # Full score subgradient:
    # ∂score/∂a[j] = 2n * ∂max_b/∂a[j] / sum^2 - 4n*max_b/sum^3
    # = 2n * 2*a_avg[k-j] / sum^2 - 4n*max_b/sum^3
    S2 = sum_a * sum_a
    S3 = S2 * sum_a
    grad = (2.0 * n / S2) * grad - 4.0 * n * max_b / S3
    return grad
