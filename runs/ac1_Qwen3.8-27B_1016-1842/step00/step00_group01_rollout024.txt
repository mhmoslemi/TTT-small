import numpy as np
import time


def _fft_conv(a):
    n = a.size
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, 2 * n - 1)


def _score(a):
    n = a.size
    c = _fft_conv(a)
    m = float(c.max())
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return 2.0 * n * m / (s * s)


def _subgrad(c, a):
    n = a.size
    m = float(c.max())
    tol = max(1e-7 * m, 1e-14)
    act = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in act:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g /= len(act)
    return g, m
