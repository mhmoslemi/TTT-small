import numpy as np
import time


def fft_conv(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def top_peaks_subgrad(conv, a, n, K):
    """Compute subgradient direction from top-K peaks."""
    K = min(K, len(conv))
    vals, idx = np.partition(conv, -K)[-K:], np.argpartition(conv, -K)[-K:]
    # Sort by value descending for weighting
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    # Weights: normalize to sum to 1
    w = vals / vals.sum() if vals.sum() > 0 else np.ones(len(vals)) / len(vals)
    
    direction = np.zeros(n)
    for k, wk in zip(idx, w):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            # direction[lo:hi+1] += wk * 2 * a[k-hi : k-lo+1][::-1]
            # a[k-i] for i from lo to hi: a[k-lo], a[k-lo-1], ..., a[k-hi]
            # That's a[k-hi : k-lo+1] reversed
            direction[lo:hi+1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]
    return direction

import numpy as np
import time


def fft_conv(a):
    """Full linear autocorrelation via FFT, O(n log n)."""
    n = len(a)
    size = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def subgrad_direction(conv, a, n, K):
    """Weighted subgradient from top-K peak positions."""
    K = min(K, len(conv))
    idx = np.argpartition(conv, -K)[-K:]
    vals = conv[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    wsum = vals.sum()
    if wsum <= 0:
        w = np.ones(K) / K
    else:
        w = vals / wsum
    direction = np.zeros(n)
    for k, wk in zip(idx, w):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            direction[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]
    return direction

    else:
        a = a / s

    a = a / a.sum()

    stagnation = 0

        K = K_base + (stagnation // 200) * 2
        K = min(K, n)
