import numpy as np
import time


def _subgrad(f, conv, top_k=5):
    """Subgradient of max(conv) w.r.t. f. O(n*top_k) time, O(n) space."""
    n = f.size
    mx = conv.max()
    idx = np.where(conv >= mx - 1e-9)[0]
    if idx.size == 0:
        idx = np.array([np.argmax(conv)])
    if idx.size > top_k:
        vals = conv[idx]
        ord_ = np.argsort(vals)[::-1][:top_k]
        idx = idx[ord_]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        seg = f[k - hi : k - lo + 1][::-1]
        grad[lo:hi + 1] += seg
    return 2.0 * grad


def _score_from_f(f):
    """Compute score assuming f is already normalized to sum=1."""
    conv = np.convolve(f, f)
    return 2.0 * f.size * conv.max(), conv
