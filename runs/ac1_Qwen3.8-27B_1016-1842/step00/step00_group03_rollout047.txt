    # For each column i (0..n-1), nonzeros at rows i..i+n-1 with values f[0..n-1]

    rows = i_idx + j_idx  # row index

    vals = np.tile(f, n)   # values
    from scipy.sparse import coo_matrix
    return coo_matrix((vals, (rows, cols)), shape=(nrows, n)).tocsr()
