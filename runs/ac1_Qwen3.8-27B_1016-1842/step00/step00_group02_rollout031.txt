```python
def project_simplex(y):
    """Project y onto {a >= 0, sum(a) = 1}."""
    n = len(y)
    if n == 0:
        return y
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(y - theta, 0)
```

For the cutting plane approach:
- Active set: a set of lags $K \subset \{0, ..., 2n-2\}$
- Reduced LP: $\max \sum b_j$ s.t. $\sum_j f_{k-j} b_j \leq M$ for $k \in K$, $b \geq 0$
- This reduced LP has only $|K|$ inequality constraints
- After solving, check all $2n-1$ constraints (via FFT): if any are violated, add to $|K|$

```python
def solve_lp_cutting_plane(f, M, n, max_iter=100):
    """Solve max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Uses cutting plane with FFT for constraint checks."""
    N = 1
    while N < 2*n - 1:
        N *= 2
    Fp = np.fft.rfft(f, N)
    
    # Start with a few constraints
    active_lags = [0, n-1, 2*n-2]  # edge and center
    active_lags = list(set(k for k in active_lags if 0 <= k < 2*n-1))
    
    b = np.ones(n) / n  # initial feasible point (or zeros)
    
    for _ in range(max_iter):
        if len(active_lags) >= 2*n-1:
            break
        # Build constraint matrix for active lags
        # Row k: (f*b)_k = sum_j f[k-j]*b[j] for valid j
        A_rows = []
        for k in active_lags:
            row = np.zeros(n)
            lo = max(0, k - (n-1))
            hi = min(n, k + 1)
            # row[j] = f[k-j] for j in [lo, hi)
            # f[k-j] for j in [lo, hi): f indices from k-lo to k-(hi-1)
            # k-lo = k - max(0,k-n+1) = min(k, n-1)
            # k-(hi-1) = k - min(n,k+1) + 1 = max(0, k-n+1)
            # So f[max(0,k-n+1) : min(k,n-1)+1] reversed
            f_idx = np.arange(lo, hi)
            row[lo:hi] = f[k - f_idx]
            A_rows.append(row)
        A_mat = np.array(A_rows)  # shape (len(active_lags), n)
        
        # Solve LP: max sum(b) s.t. A_mat @ b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        res = scipy.optimize.linprog(c, A_ub=A_mat, b_ub=M, bounds=[(0, None)]*n, method='highs')
        
        if res.status != 0:
            break
        b_new = res.x
        
        # Check all constraints via FFT
        Bp = np.fft.rfft(b_new, N)
        all_convs = np.fft.irfft(Fp * Bp, N)[:2*n-1]
        violated = np.where(all_convs > M * (1 + 1e-10))[0]
        
        if len(violated) == 0:
            return b_new  # feasible for all constraints, done
        
        # Add violated constraints to active set
        new_lags = [int(v) for v in violated[:10]]  # add at most 10 new
        for kl in new_lags:
            if kl not in active_lags:
                active_lags.append(kl)
    
    return b
```
