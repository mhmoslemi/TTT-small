import numpy as np
import time


def _try_candidate(seq, best, best_score):
    """Evaluate and return (seq, score) if better, else (best, best_score)."""
    score = float(evaluate_sequence([float(x) for x in seq]))
    if np.isfinite(score) and score < best_score:
        return seq, score
    return best, best_score


def _constant_seq(n, val):
    return np.full(n, val, dtype=np.float64)


def _block_perturb(base, rng, strength):
    n = base.size
    out = base.copy()
    width = max(2, n // 64)
    start = int(rng.integers(0, n - width + 1))
    out[start:start + width] *= float(np.exp(rng.normal(0.0, strength)))
    return np.clip(out, 0.0, 1000.0)
