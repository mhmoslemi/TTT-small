def fast_score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2 for valid a."""
    b = np.convolve(a, a)
    max_b = float(np.max(b))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:

    return 2.0 * len(a) * max_b / (sum_a * sum_a)

            elif np.isfinite(sc_new) and sc_new < cur_sc:
                a = a_new
                improved = True
                stalled = 0
                break
            else:
                eta *= 0.5

        if not improved:
            stalled += 1
            if stalled > 300:
                n_p = max(1, n // 40)
                for _ in range(n_p):
                    a[int(rng.integers(0, n))] *= float(rng.uniform(0.3, 3.0))
                a = np.clip(a, 0.0, 1000.0)
                s = float(a.sum())
                if s >= 0.01:
                    a = a / s * n
                eta = 0.1
                beta_sm = min(beta_sm * 1.5, 100.0)
                stalled = 0
        elif eta < 0.5:
            eta *= 1.04
        elif best_score < 1.6 and beta_sm > 2.0:
            beta_sm *= 0.97

    # Authoritative validation
    out = [float(v) for v in best]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs):
        out = [float(v) for v in parent]
    return out
