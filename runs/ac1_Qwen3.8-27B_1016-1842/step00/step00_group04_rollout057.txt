            p_best = bp

    # Local refinement via block perturbations
    while time.monotonic() < dead:
        width = int(rng.integers(max(1, n // 400), max(2, n // 15)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        p_try = p_best.copy()
        p_try[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))
        p_try = _simplex_project(p_try)
        c_try = _conv_full(p_try)
        s_try = _score_from_c(c_try, n)
        if s_try < score_best:
            score_best = s_try
            p_best = p_try
        else:
            p_try2 = p_best.copy()
            w2 = max(1, width // 3)
            s2 = int(rng.integers(start, max(start + 1, stop - w2 + 1)))
            e2 = min(n, s2 + w2)
            p_try2[s2:e2] *= float(np.exp(rng.normal(0.0, 0.04)))
            p_try2 = _simplex_project(p_try2)
            c2 = _conv_full(p_try2)
            s2v = _score_from_c(c2, n)
            if s2v < score_best:
                score_best = s2v
                p_best = p_try2

        # Periodic soft-grad boost
        if int((time.monotonic() + 5.0 - (dead))) < 0:
            break
        elapsed = time.monotonic() - (dead - (float(budget_s) - 5.0))
        if int(elapsed * 10) % 50 == 0:
            if time.monotonic() < dead - 1.0:
                mom = np.zeros(n)
                p = p_best.copy()
                for t in range(30):
                    if time.monotonic() > dead:
                        break
                    beta = 80.0 * (1.0 + 0.3 * t)
                    et = 0.003 / (1.0 + 0.1 * t)
                    c = _conv_full(p)
                    g, _ = _soft_max_grad(c, beta, p, n)
                    mom = 0.9 * mom + g
                    p = _simplex_project(p - et * mom)
                c = _conv_full(p)
                s = _score_from_c(c, n)
                if s < score_best:
                    score_best = s
                    p_best = p

    # Convert back: use sum matching parent for scale-invariance
    a_out = p_best * sp
    a_out = np.clip(a_out, 0.0, 1000.0)
    if float(a_out.sum()) < 0.01:
        a_out = np.ones(n, dtype=np.float64)
    result = [float(x) for x in a_out]
    sv = float(evaluate_sequence(result))
    if not np.isfinite(sv):
        return [float(x) for x in parent]
    return result
