from scipy.optimize import linprog
import time

def build_conv_matrix(a, n):
    """Build the (2n-1) x n sparse matrix C where (a*b)_k = (C @ b)_k."""
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    C = lil_matrix((2*n-1, n), dtype=np.float64)
    # Actually, let's use COO
    C = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return C

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def build_conv_matrix(a, n):
    total = n * n
    rows = np.empty(total, dtype=np.int64)
    cols = np.empty(total, dtype=np.int64)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        j = np.arange(lo, hi + 1)
        rows[idx:idx+len(j)] = k
        cols[idx:idx+len(j)] = j
        vals[idx:idx+len(j)] = a[k - j]
        idx += len(j)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
