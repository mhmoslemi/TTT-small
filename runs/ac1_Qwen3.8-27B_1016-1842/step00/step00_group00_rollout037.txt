import numpy as np
import time


def fft_conv(a, b):
    """Linear convolution via FFT, O(n log n)."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def sm_p(c, tau):
    """Softmax weights approximating the argmax distribution."""
    cm = c.max()
    e = np.exp(np.clip(tau * (c - cm), -80, 80))
    return e / e.sum()


def gr_sm(a, p):
    """Gradient of smooth-max(conv(a,a)): cross-correlation a with p."""
    n = len(a)
    pr = p[::-1]
    t = n + 2 * n - 2
    N = 1 << (t - 1).bit_length()
    cv = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(pr, N), N)[:t]
    return cv[n - 1 : 2 * n - 1][::-1].copy()


def gd(a, rng, t_end):
    """FFT-based gradient descent minimising max(convolve(a,a)) with sum(a)=1."""
    n = len(a)
    a = np.clip(a, 0.0, 1000.0).astype(np.float64)
    if a.sum() < 0.01:
        return a
    a /= a.sum()
    ba = a.copy()
    bc = fft_conv(a, a).max()
    tau, lr, st = 100.0, 0.5, 0
    while time.monotonic() < t_end:
        c = fft_conv(a, a)
        cm = c.max()
        if cm < bc:
            bc, ba, st = cm, a.copy(), 0
            tau = max(50.0, tau * 0.99)
        else:
            st += 1
        if st > 200:
            a = np.maximum(a + rng.normal(0.0, 0.05, n), 0.0)
            s = a.sum()
            a = a / s if s > 0.01 else ba.copy()
            st, tau = 0, min(3000.0, tau * 1.05)
        p = sm_p(c, tau)
        g = gr_sm(a, p)
        g -= g.mean()
        t = lr
        ok = False
        for _ in range(8):
            a2 = np.maximum(a - t * g, 0.0)
            s = a2.sum()

                    a, ok = a2, True
                    break
            t *= 0.5
        lr = min(2.0, lr + 0.01) if ok else max(0.01, lr * 0.9)
    return ba
