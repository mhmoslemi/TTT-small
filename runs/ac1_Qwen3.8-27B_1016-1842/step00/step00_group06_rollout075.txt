import numpy as np
import time

def _next_pow2(m):
    size=1
    while size < m:
        size <<= 1
    return size

def _fft_conv(a,b):
    if len(a)==0 or len(b)==0:
        return np.zeros(0, dtype=np.float64)
    need = len(a)+len(b)-1
    size = _next_pow2(need)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    out = np.fft.irfft(fa*fb, size)
    return out[:need]

def _self_conv(f):
    C = _fft_conv(f, f)
    C = np.ascontiguousarray(C[:2*f.size-1])

def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_raw(a, target=1.0):
    if a is None:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)

    np.minimum(a, 1000.0, out=a)

        return None
    a = (a / s) * target
    np.maximum(a, 0.0, out=a)

def _central(n):
    c = np.empty(int(n), dtype=np.float64)
    if n == 0:
        return c
    c[0] = 1.0
    if n > 1:
        j = np.arange(1, n, dtype=np.float64)
        c[1:] = np.cumprod((2.0 * j - 1.0) / (2.0 * j))
    return c


def _self_conv(f):
    n = f.size
    need = 2 * n - 1
    size = _next_pow2(need)
    fa = np.fft.rfft(f, size)
    out = np.fft.irfft(fa * fa, size)
    out = np.ascontiguousarray(out[:need])
    np.maximum(out, 0.0, out=out)
    return out


def _grad_conv(f, w):
    n = f.size
    target = 2 * n - 1
    if w.size != target:
        tmp = np.zeros(target, dtype=np.float64)
        m = min(w.size, target)
        tmp[:m] = w[:m]
        w = tmp
    need = n + target - 1
    size = _next_pow2(need)
    fa = np.fft.rfft(f, size)
    wr = np.ascontiguousarray(w[::-1])
    fw = np.fft.rfft(wr, size)
    out = np.fft.irfft(fa * fw, size)
    return out[n - 1:2 * n - 1][::-1]


def _logsum_weights(c, frac):
    cmax = float(np.max(c))
    L = c.size
    if cmax <= 0.0:
        return np.ones(L, dtype=np.float64) / float(L)
    beta = max(cmax * frac, 1e-300)
    z = (c - cmax) / beta
    w = np.exp(z)
    wm = float(np.max(w))
    if not np.isfinite(wm) or wm <= 0.0:
        w = np.zeros(L, dtype=np.float64)
        if L:
            w[int(np.argmax(c))] = 1.0
    else:
        w = w / wm
    return w


def _mirror_candidate(f, h, alpha):

def _proj_candidate(f, h, alpha):

def _perturb_candidate(f, rng, strength):
