def fast_convolve(a):

import numpy as np
import time


def fast_convolve(a):
    """O(n log n) linear convolution via FFT. O(n) memory."""
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    return np.fft.irfft(fa * fa, fft_size)[:size]


def subgrad_at_max(a, s_star):
    """Subgradient of max-conv at position s_star: g[j]=2*a[s_star-j] if valid."""
    n = a.size
    g = np.zeros(n)
    lo = max(0, s_star - n + 1)
    hi = min(n - 1, s_star)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[s_star - hi:s_star - lo + 1][::-1]
    return g


def score_fast(a):
    """Compute score using FFT convolution (fast, for inner loop)."""
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * a.size * conv.max() / (s * s)


def normalize_to_mean1(a, n):
    """Rescale so sum equals n (mean 1)."""
    s = a.sum()
    if s < 1e-15:
        return np.ones(n)
    return a * (n / s)
