import numpy as np

def _fft_convolve(a, b):
    """Full linear convolution via FFT, O(n log n)."""
    out_len = len(a) + len(b) - 1
    m = 1
    while m < out_len:
        m *= 2
    A = np.fft.rfft(a, m)
    B = np.fft.rfft(b, m)
    return np.fft.irfft(A * B, m)[:out_len]


def _ac1_score_fast(a):
    """AC1 score via FFT convolution (same formula as evaluator)."""
    n = a.size
    s = a.sum()
    if s < 0.01:

    conv = _fft_convolve(a, a)
    return float(2.0 * n * conv.max() / (s * s))
