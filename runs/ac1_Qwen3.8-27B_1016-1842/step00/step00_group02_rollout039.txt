import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    n = len(height_sequence_1)
    
    def score_seq(a):
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            return float('inf')
        return evaluate_sequence(a.tolist())
    
    # Try various initializations
    inits = []
    inits.append(np.clip(height_sequence_1, 0, 1000).astype(float))  # parent
    inits.append(np.arange(1, n+1, dtype=float))  # linear
    inits.append(np.arange(1, n+1, dtype=float)**2)  # quadratic
    inits.append(np.exp(np.linspace(0, 2, n)))  # exp ramp
    inits.append(np.exp(np.linspace(2, 0, n)))  # exp decay
    
    best = None
    best_score = float('inf')
    for init in inits:
        s = score_seq(init)
        if s < best_score:
            best_score = s
            best = init.copy()
    
    # Local search: perturb random blocks multiplicatively
    step_size = 0.1  # log-space perturbation std
    while time.monotonic() < deadline:
        candidate = best.copy()
        width = rng.integers(max(3, n//1000), max(4, n//10))
        start = rng.integers(0, max(1, n - width))
        end = min(n, start + width)
        
        # Log-space perturbation
        mask = end > 0
        candidate[start:end] = np.clip(
            candidate[start:end] * np.exp(rng.normal(0, step_size, size=end-start)),
            0, 1000
        )
        
        s = score_seq(candidate)
        if np.isfinite(s) and s < best_score:
            best = candidate
            best_score = s
            
            # Adaptive step: reduce if we improve a lot
            if best_score < 1.51:
                step_size = max(0.01, step_size * 0.95)
