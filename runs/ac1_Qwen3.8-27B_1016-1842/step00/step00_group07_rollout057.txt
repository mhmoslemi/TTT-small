import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    # Get parent sequence
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = a.size
    if n == 0 or not np.all(np.isfinite(a)):
        a = np.ones(256, dtype=np.float64)
        n = 256
    
    # Normalize to mean 1
    S = a.sum()
    if S < 1e-6:
        a[:] = 1.0
    else:
        a *= n / S
    
    # Break symmetry with small noise
    a += 0.005 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    a *= n / max(a.sum(), 1e-6)
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # FFT size
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size <<= 1
    
    # Helper: compute convolution
    def conv_of(x):
        fx = np.fft.rfft(x, fft_size)
        return np.fft.irfft(fx * fx, fft_size)[:2 * n - 1]
    
    # Main optimization loop
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.7
    stall = 0
    eval_count = 0
    
    while time.monotonic() < deadline:
        b = conv_of(a)
        M = float(b.max())
        
        # Top positions
        thresh = M * 0.98
        top_idx = np.where(b >= thresh)[0]
        if len(top_idx) > 15:
            top_idx = top_idx[np.argsort(b[top_idx])[-15:]]
        
        # Effective gradient
        g = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        g /= max(len(top_idx), 1)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * g
        g_use = momentum
        
        # Step
        a_new = a - lr * g_use
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-6:
            a_new = a
        else:
            a_new *= n / S_new
        
        # Check
        b_new = conv_of(a_new)
        M_new = float(b_new.max())
        
        if M_new < M - 1e-12 * M:
            a = a_new
            # Periodic exact eval
            eval_count += 1
            if eval_count % 2000 == 0 or (M_new / M < 0.999):
                sc = float(evaluate_sequence(a.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                stall = 0
            if lr < 1e-10:
                # Restart with perturbation
                a = best + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a *= n / max(a.sum(), 1e-6)
                momentum[:] = 0
                lr = 0.01
                stall = 0
    
    # Final
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best = a.copy()
        best_score = sc
    
    return [float(x) for x in best]
