    obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    res = linprog(
        obj, A_ub=A, b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method='highs',
        options={'time_limit': max(1.0, time_limit), 'presolve': True},
    )
    if res.success and res.x is not None:
        g = np.asarray(res.x, dtype=np.float64)
        if np.all(np.isfinite(g)) and g.sum() > 1e-14:
            return g
    return None


def safe_score(arr):
    """Evaluate a numpy array via the authoritative evaluator."""
    try:
        return float(evaluate_sequence(arr.tolist()))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --- Initialization ---
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
        parent = np.ones(256, dtype=np.float64)
    n = parent.size
    idx = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1.0)

    inits = {}
    inits['flat'] = parent.copy()
    inits['cos2'] = np.sin(np.pi * (idx + 0.5) / n) ** 2
    inits['cos4'] = np.sin(np.pi * (idx + 0.5) / n) ** 4
    inits['gauss'] = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits['tri'] = np.maximum(0.0, 1.0 - 2.0 * np.abs(idx - center) / np.maximum(1.0, n - 1.0))

    best_score = np.inf
    best = inits['flat'].copy()
    for key, arr in inits.items():
        arr = np.clip(arr, 0.0, 1000.0)
        s = safe_score(arr)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = arr.copy()

    if n < 2:
        return [1.0, 1.0]

    # --- LP-based iterations ---
    rows, cols = build_conv_structure(n)

    for key in ['cos2', 'gauss', 'cos4', 'tri', 'cos2']:
        if time.monotonic() > deadline - 30.0:
            break
        a = np.clip(inits[key], 0.0, 1000.0).copy()
        a_sum = float(np.sum(a))
        if a_sum < 0.01:
            continue

        for it in range(40):
            t_left = deadline - time.monotonic() - 10.0
            if t_left < 15.0:
                break
            b = np.convolve(a, a)
            M = float(np.max(b))
            if M <= 0:
                break
            lp_time = min(120.0, t_left - 8.0)
            g = lp_step(a, M, rows, cols, lp_time)
            if g is None:
                break
            g_sum = float(np.sum(g))
            if g_sum < 1e-12:
                break
            g = g * (a_sum / g_sum)

            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.15]:
                a_new = (1.0 - alpha) * a + alpha * g
                a_new = np.clip(a_new, 0.0, 1000.0)
                s_new = safe_score(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best = a_new.copy()
                if np.isfinite(s_new):
                    a = a_new
                    improved = s_new <= M * 2.0 * n / (a_sum ** 2) + 1e-8
                    break
            if not improved:
                break

        s_end = safe_score(a)
        if np.isfinite(s_end) and s_end < best_score:
            best_score = s_end
            best = a.copy()

    # --- Local search polishing ---
    step_scale = 0.04
    stall = 0
    while time.monotonic() < deadline - 2.0:
        a2 = best.copy()
        width = max(2, n // 150)
        start = int(rng.integers(0, max(1, n - width)))
        a2[start:start + width] *= np.exp(rng.normal(0.0, step_scale))
        if rng.random() < 0.3:
            a2 = 0.5 * (a2 + a2[::-1])
        a2 = np.clip(a2, 0.0, 1000.0)
        s2 = safe_score(a2)
        if np.isfinite(s2) and s2 < best_score - 1e-10:
            best_score = s2
            best = a2.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step_scale = max(0.003, step_scale * 0.85)

    # --- Final sanity check ---
    final_score = safe_score(best)
    if not np.isfinite(final_score) or final_score >= best_score:
        best = np.clip(inits['flat'], 0.0, 1000.0)
        best_score = np.min(final_score, 2.0)

    result = [float(x) for x in best]
    assert len(result) > 0 and all(np.isfinite(x) and x >= 0 for x in result)
    return result
