```python
import numpy as np
import time

def fast_conv(a):
    """Compute full linear convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    ca = np.fft.irfft(fa * fa, fsize)[:size]
    return ca

def fast_score(a):
    """Fast computation of the AC1 score."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fast_conv(a)
    mb = c.max()
    return 2 * n * mb / (s * s)
```
