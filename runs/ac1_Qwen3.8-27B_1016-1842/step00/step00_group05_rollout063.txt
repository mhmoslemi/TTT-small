```python
def project_simplex(a):
    n = len(a)
    # Find tau such that sum(max(a - tau, 0)) = 1
    # Sort a in descending order
    a_sorted = np.sort(a)[::-1]
    # Find the threshold
    # sum_{i: a_sorted[i] > tau} (a_sorted[i] - tau) = 1
    # Let k be the number of elements > tau
    # sum_{i=0}^{k-1} a_sorted[i] - k*tau = 1
    # tau = (sum_{i=0}^{k-1} a_sorted[i] - 1) / k
    # We need a_sorted[k-1] > tau >= a_sorted[k] (with a_sorted[n] = -inf)
    
    # Cumulative sums
    cumsum = np.cumsum(a_sorted)
    # For each possible k (1 to n):
    # tau_k = (cumsum[k-1] - 1) / k
    # Valid if a_sorted[k-1] > tau_k >= (a_sorted[k] if k < n else -inf)
    
    k_vals = np.arange(1, n+1)
    tau_candidate = (cumsum[:-1] - 1.0) / k_vals
    # Check validity
    valid_upper = a_sorted[:-1] > tau_candidate
    valid_lower = np.ones(n, dtype=bool)
    if n > 1:
        valid_lower[:-1] = tau_candidate[:-1] >= a_sorted[1:]
    # Hmm this is getting complicated. Let me just use binary search.
    
    lo = np.min(a) - 1
    hi = np.max(a) + 1
    for _ in range(50):  # 50 iterations for double precision
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(a - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(a - tau, 0)
```
