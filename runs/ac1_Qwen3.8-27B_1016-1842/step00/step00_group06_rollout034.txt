def project_simplex(x, target_sum):
    """Project x onto {y >= 0, sum(y) = target_sum}."""
    x = np.clip(x, 0, None)
    s = x.sum()
    if s < 1e-15:
        n = len(x)
        return np.full(n, target_sum / n)
    # Scale first (if all positive, this is the projection onto sum constraint)
    # But we also need non-negativity, so use the standard simplex projection
    # Actually, since we clip to 0 first and then scale, it's not exactly projection
    # Let me use the proper algorithm
    
    # Standard: find t such that sum(max(x_i - t, 0)) = target_sum
    # Then projection is max(x_i - t, 0)
    # Binary search for t
    x_sorted = np.sort(x)[::-1]  # decreasing
    n = len(x)
    # Find the right number of active components
    cumsum = np.cumsum(x_sorted)
    rho = np.argmax(x_sorted * np.arange(1, n+1) > (cumsum - target_sum)) 
    # Actually the standard formula:
    # t = (sum_{i=1}^{rho} x_i - target_sum) / rho
    # projection = max(x_i - t, 0)

    if s > 0:

    else:
        x_clipped = np.full(n, target_sum / n)
    return x_clipped

def compute_grad_at_peaks(x, peak_lags, top_m):
    """Compute subgradient of max(conv) at given peak lags."""
    n = len(x)
    grad = np.zeros(n)
    count = 0

def compute_subgrad(x, k, n):
    """Gradient of conv(x,x)[k] w.r.t. x, O(n)."""
    grad = np.zeros(n)
