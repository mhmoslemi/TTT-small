    if f.size == 0 or f.sum() < 0.01:

    n = len(f)

    momentum = np.zeros_like(f)
    base_lr = 1.0 / np.sqrt(n)
    beta = 0.9

        k_star = int(np.argmax(corr))

        sg -= sg.mean()

        lr = base_lr / (1.0 + 0.01 * t)
        
        # Momentum update
        momentum = beta * momentum + lr * sg
        f = f - momentum

        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-10:

            if score < best_score:

                best = f.copy()
    
    # Final check

def _autocorr(f):
    """Linear autocorrelation via FFT. O(n log n)."""
    n = len(f)
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    F = np.fft.rfft(f, fft_len)
    return np.fft.irfft(F * F, fft_len)[:size]

    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    sg[idx] = 2.0 * f[k - idx]
