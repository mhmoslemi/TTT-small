import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(256)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256)
    
    n = parent.size
    if n == 0:
        return [1.0]
    
    idx = np.arange(n, dtype=np.float64)
    j1 = idx + 1.0
    
    # Generate diverse initializations
    cands = [parent.copy(), j1.copy(), j1 ** 0.5, j1 ** 1.5, j1 ** 2.0, j1 ** 3.0]
    for c in [0.5, 1.0, 2.0, 4.0]:
        cands.append(np.exp(c * idx / n))
    t = idx / max(1.0, n - 1)
    for c in [1.0, 5.0, 20.0, 100.0]:
        cands.append(1.0 + c * (2.0 * t - 1.0) ** 2)
    for f in [1, 2, 3, 5, 8]:
        cands.append(np.clip(1.0 + 0.9 * np.cos(2.0 * np.pi * f * idx / n), 0.01, None))
    
    # Evaluate initializations
    best = None
    best_score = np.inf
    for c in cands:
        c = np.clip(c.astype(np.float64), 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        sc = evaluate_sequence(c.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = c.copy()
    
    if best is None:
        return [1.0] * n
    
    # Iterative refinement
    eta = 0.3
    stall = 0
    niter = 0
    
    while time.monotonic() < deadline:
        niter += 1
        if stall > 10000:
            break
        
        b = np.convolve(best, best)
        k_star = int(np.argmax(b))
        M = float(b[k_star])
        S = float(np.sum(best))
        ms = M / max(S, 1e-15)
        
        # Compute gradient of the score
        grad = np.full(n, -ms)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        if hi > lo:
            jj = np.arange(lo, hi)
            grad[lo:hi] = best[k_star - jj] - ms
        
        gmax = float(np.max(np.abs(grad)))
        amax = float(np.max(best))
        
        improved = False
        if gmax > 1e-15 and amax > 1e-15:
            for trial in [1.0, 0.3, 0.1]:
                cand = best - eta * trial * (grad / gmax) * amax
                cand = np.clip(cand, 0.0, 1000.0)
                cs = float(np.sum(cand))
                if cs < 0.01:
                    continue
                b2 = np.convolve(cand, cand)
                sc = 2.0 * n * float(np.max(b2)) / (cs * cs)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = cand
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            # Random multiplicative perturbation
            w = max(2, int(n * (0.002 + 0.008 * min(1.0, stall / 2000.0))))
            s0 = int(rng.integers(0, max(1, n - w)))
            sig = 0.02 + 0.08 * min(1.0, stall / 3000.0)
            cand = best.copy()
            cand[s0:s0 + w] *= float(np.exp(rng.normal(0.0, sig)))
            cand = np.clip(cand, 0.0, 1000.0)
            cs = float(np.sum(cand))
            if cs >= 0.01:
                b2 = np.convolve(cand, cand)
                sc = 2.0 * n * float(np.max(b2)) / (cs * cs)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best = cand
                    stall = 0
                    continue
            stall += 1
        
        # Periodic exact verification
        if niter % 200 == 0:
            sc_exact = evaluate_sequence([float(x) for x in best])
            if np.isfinite(sc_exact) and sc_exact < best_score:
                best_score = sc_exact
            elif not np.isfinite(sc_exact):
                break
    
    # Final verification
    fs = evaluate_sequence([float(x) for x in best])
    if not np.isfinite(fs):
        return [1.0] * n
    return [float(x) for x in best]
