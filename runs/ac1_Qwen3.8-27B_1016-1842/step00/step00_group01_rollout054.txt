def fast_autocorr(a):
    """Compute full linear autocorrelation of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1

def _fft_autocorr(a):
    """Full linear autocorrelation via FFT, O(n log n)."""
    n = a.size

    A = np.fft.rfft(a, fft_len)
    ac = np.fft.irfft(A * A, fft_len)[:size]

import numpy as np
import time


def _fft_autocorr(a):
    """Full linear autocorrelation via FFT, O(n log n) memory O(n)."""
    n = a.size
    size = 2 * n - 1
    fft_len = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:size]


def _fast_score(a):
    """Quick score using FFT autocorrelation."""
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return 1e30
    ac = _fft_autocorr(a)
    return float(2.0 * n * ac.max() / (S * S))


def _subgradient_step(a, lr, rng):
    """One subgradient step; returns updated a and the fast score."""
    n = a.size
    ac = _fft_autocorr(a)
    S = a.sum()
    k_star = int(np.argmax(ac))
    M = float(ac[k_star])
    score = 2.0 * n * M / (S * S)

    # Gradient of AC[k*] w.r.t a: 2*a[k*-j] for valid j
    grad_ac = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    idx = k_star - np.arange(lo, hi + 1)
    grad_ac[lo:hi + 1] = 2.0 * a[idx]

    # df/da = 2n*grad_ac/S^2 - 4n*M/S^3
    subgrad = 2.0 * n * grad_ac / (S * S) - 4.0 * n * M / (S * S * S)

    # Normalized descent step
    sn = np.linalg.norm(subgrad)
    if sn < 1e-18:
        return a, score, True
    a_new = a - lr * subgrad / sn
    a_new = np.clip(a_new, 1e-8, 1000.0)
    if a_new.sum() < 0.01:
        a_new += (0.01 - a_new.sum()) / n
    return a_new, score, False

            break
        a, score, converged = _subgradient_step(a, lr, rng)

            break
        if score < best_local_score:

            best_local = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.85

                    a = np.clip(a, 1e-8, 1000.0)
                    lr = max(lr, 0.01)
                    stall = 0
    return best_local, best_local_score
