def _conv_fft(a, N):
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * a.size - 1]
    return c.real

def _corr_fft(a, p, N):
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    corr = np.fft.irfft(fp * np.conjugate(fa), N)[:a.size]

    c = _conv_fft(a, N)

    cmax = float(c.max())

        return cmax
    z = (c - cmax) / tau
    # clip z to <= 0 to reduce overflow from roundoff
    z = np.minimum(z, 0.0)
    return cmax / tau + float(np.log(np.sum(np.exp(z))))

def _lp_direction(a, K, M, bound):
    if linprog is None:
        return None
    if K.size == 0:
        return None
    n = a.size
    m = K.size
    # If constraints too many, downsample? Already m small.
    A = np.zeros((m, n), dtype=np.float64)
    for r in range(m):
        k = int(K[r])
        start = max(0, k - (n - 1))
        end = min(n, k + 1)
        if end > start:
            A[r, start:end] = a[k - np.arange(start, end)]
    # if all rows zero, no useful
    if not np.any(A):
        return None
    c_obj = -np.ones(n, dtype=np.float64)
    b = np.full(m, float(M), dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0.0, float(bound)), method='highs', options={'presolve': True})
    except Exception:
        return None
    if res is None:
        return None
    ok = hasattr(res, 'success') and res.success
    if not ok:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if g.size != n or not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s

def _fast_score(a,N):
    if a.size==0: return np.inf
    fa=np.fft.rfft(a,N)
    c=np.fft.irfft(fa*fa,N)[:2*a.size-1].real
    m=float(c.max()); s=float(a.sum())

    return 2.0*a.size*m/(s*s)

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _fast_score(a, N):
    if a.size == 0:
        return np.inf
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * a.size - 1].real
    m = float(c.max())
    s = float(a.sum())
    if not np.isfinite(m) or s <= 1e-12:
        return np.inf
    return 2.0 * a.size * m / (s * s)


def _focus(c, tau):
    cmax = float(c.max())
    if tau <= 1e-300 or not np.isfinite(tau):
        return cmax
    z = (c - cmax) / tau
    z = np.minimum(z, 0.0)
    return cmax / tau + float(np.log(np.sum(np.exp(z))))


def _normalize(a, target=1.0):
    s = float(a.sum())
    if s <= 1e-300:
        return np.full_like(a, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def _initial_candidates(n, base):

def _block_perturb(a, rng, sigma):
    n = a.size
    if n <= 1:
        return a.copy()
    width = int(np.exp(rng.normal(math.log(max(2, n // 32)), 0.7)))
    width = min(max(1, width), n)
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = a.copy()
    cand[start:start + width] *= np.exp(rng.normal(0.0, sigma, size=width))
    cand = np.clip(cand, 1e-300, None)
    return _normalize(cand)


def _top_indices(c, m):
    L = c.size
    m = min(int(m), L)
    if m <= 0:
        return np.empty(0, dtype=np.int64)
    if m >= L:
        return np.arange(L, dtype=np.int64)
    idx = np.argpartition(c, -m)[-m:]
    return idx.astype(np.int64)


def _lp_direction(a, K, M, bound):
    if linprog is None or K.size == 0:
        return None
    n = a.size
    m = K.size
    A = np.zeros((m, n), dtype=np.float64)
    for r in range(m):
        k = int(K[r])
        start = max(0, k - (n - 1))
        end = min(n, k + 1)
        if end > start:
            A[r, start:end] = a[k - np.arange(start, end)]
    if not np.any(A):
        return None
    c_obj = -np.ones(n, dtype=np.float64)
    b = np.full(m, float(M), dtype=np.float64)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=(0.0, float(bound)), method='highs')
    except Exception:
        return None
    if res is None or not getattr(res, 'success', False):
        return None
    g = np.asarray(res.x, dtype=np.float64)
    if g.size != n or not np.all(np.isfinite(g)):
        return None
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s

        p = np.exp((c - M) / tau)
        psum = float(p.sum())
        if psum <= 0.0 or not np.isfinite(psum):
            p = np.ones_like(c, dtype=np.float64) / float(c.size)
        else:
            p = p / psum

        fp = np.fft.rfft(p, N)
        corr = np.fft.irfft(fp * np.conj(fa), N)[:n].real
        g = 2.0 * corr
        if not np.all(np.isfinite(g)):
            g = np.zeros_like(a, dtype=np.float64)

        mom = 0.75 * mom + 0.25 * g
        step = mom - float(mom.mean())
        scale = float(step.max() - step.min())
        if scale <= 1e-18:
            cand = _block_perturb(a, rng, 0.25)
        else:
            step = step / scale
            factor = np.clip(-lr * step, -4.0, 4.0)
            cand = a * np.exp(factor)
            cand = np.maximum(cand, floor)
            s = float(cand.sum())
            if s <= 1e-12:
                cand = a.copy()
            else:
                cand = cand / s

        fanew = np.fft.rfft(cand, N)
        cnew = np.fft.irfft(fanew * fanew, N)[:two].real
        Mnew = float(cnew.max())
        Fnew = _focus(cnew, tau)
        fsnew = 2.0 * n * Mnew

        accept = False
        if np.isfinite(Fnew):
            if Fnew <= Fcur + 1e-12:
                accept = True
            elif np.isfinite(fsnew) and fsnew < cur_fast - 1e-12:
                accept = True

        if accept:
            a = cand
            cur_fast = fsnew
            lr = min(0.45, lr * 1.02)
            fail = 0
            no_best = 0
            if np.isfinite(fsnew) and fsnew < best_fast:
                best_fast = fsnew
                best = cand.copy()
                if it - last_exact_it >= 150:
                    es = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(es) and es < best_exact:
                        best_exact = es
                        best = cand.copy()
                    last_exact_it = it
            if tau_frac > tau_min_frac:
                tau_frac *= 0.9995
        else:
            fail += 1
            no_best += 1
            if fail > 25:
                lr = max(0.02, lr * 0.65)
                fail = 0

        if it > 0 and it % 300 == 100:
            cand = _block_perturb(a, rng, 0.6)
            ftmp = _fast_score(cand, N)
            if np.isfinite(ftmp) and (ftmp < cur_fast or ftmp < best_fast * 1.2):
                a = cand
                cur_fast = ftmp
            if time.monotonic() > deadline - 40:
                break

        if it % 350 == 175 and linprog is not None and lp_done < max_lp and time.monotonic() < deadline - 80:
            K = _top_indices(c, 18)
            gd = _lp_direction(a, K, M, 1.0)
            lp_done += 1
            if gd is not None:
                for tt in (0.3, 0.1, 0.5, 0.05):
                    cand = _blend(a, gd, tt)
                    fstmp = _fast_score(cand, N)
                    if np.isfinite(fstmp) and fstmp < best_fast:
                        a = cand
                        cur_fast = fstmp
                        best = cand.copy()
                        best_fast = fstmp
                        no_best = 0
                        break
                    if np.isfinite(fstmp) and fstmp < cur_fast:
                        a = cand
                        cur_fast = fstmp
                        break

        if it - last_exact_it >= 800:
            es = float(evaluate_sequence(a.tolist()))
            if np.isfinite(es) and es < best_exact:
                best_exact = es
                best = a.copy()
            last_exact_it = it

        if no_best > 3000:
            a = best.copy()
            cur_fast = best_fast
            lr = 0.18
            tau_frac = max(tau_frac, 0.2)
            mom = np.zeros_like(a, dtype=np.float64)
            no_best = 0
            a = _block_perturb(a, rng, 0.1)
            cur_fast = _fast_score(a, N)

        it += 1

    final_exact = best_exact
    final_arr = best

    if time.monotonic() < deadline - 10:
        ramp = np.arange(n, dtype=np.float64)
        rss = float(ramp.sum())
        if rss > 0.0:
            ramp = ramp / rss
            es = float(evaluate_sequence(ramp.tolist()))
            if np.isfinite(es) and es < final_exact:
                final_exact = es
                final_arr = ramp.copy()

    for arr in (a, best):
        if arr is None:
            continue
        es = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(es) and es < final_exact:
            final_exact = es
            final_arr = arr.copy()

    if final_arr is None or final_arr.size == 0 or not np.all(np.isfinite(final_arr)):
        final_arr = np.ones(256, dtype=np.float64)

    final_arr = np.clip(final_arr, 0.0, 1000.0)
    s = float(final_arr.sum())
    if s < 0.01:
        final_arr = np.ones(256, dtype=np.float64)

    return [float(x) for x in final_arr]
