def compute_grad_at(a, k_star):
    n = len(a)
    grad = np.zeros(n)
    # Term 1: grad[j] += a[k_star - j] for valid j
    j1_lo = max(0, k_star - n + 1)
    j1_hi = min(n - 1, k_star)
    if j1_hi >= j1_lo:
        # j goes from j1_lo to j1_hi
        # k_star - j goes from k_star - j1_lo to k_star - j1_hi
        # which is from (k_star - j1_lo) down to (k_star - j1_hi)
        a_slice = a[k_star - j1_hi : k_star - j1_lo + 1]  # this is a[hi] to a[lo]
        grad[j1_lo:j1_hi+1] += a_slice[::-1]
    
    # Term 2: grad[j] += a[j - k_star] for valid j
    j2_lo = max(0, k_star)
    j2_hi = min(n - 1, k_star + n - 1)
    if j2_hi >= j2_lo:
        a_slice = a[j2_lo - k_star : j2_hi - k_star + 1]
        grad[j2_lo:j2_hi+1] += a_slice
    
    return grad
