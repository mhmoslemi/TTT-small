    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa)

    sum_a = a.sum()
    return 2 * n * max_conv / sum_a**2

def fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
