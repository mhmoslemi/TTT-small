def _subgrad_direction(a, a_rev, k_star, n, peaks):
    """Compute mean of a[k-i] over peak positions, O(|peaks|*n) memory."""
    h = np.zeros(n, dtype=np.float64)
    for k in peaks:
        m = n - 1 - int(k)
        lo = max(0, -m)
        hi = min(n, n - m)
        if hi > lo:
            h[lo:hi] += a_rev[m + lo:m + hi]
    h /= len(peaks)
    return h
