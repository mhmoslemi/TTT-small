import numpy as np
import time

def compute_convolution(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n  # next power of 2 >= 2n-1
    # Actually, let's use 2*n-1 or the next power of 2
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    C = A * A
    c = np.fft.irfft(C, fft_size)[:conv_size]
    return c

def simplex_project(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    # v is 1D array
    # Standard projection onto probability simplex (scaled)
    # First project onto {x >= 0, sum(x) = 1} then scale
    v = np.maximum(v, 0)
    # Bisection for the threshold
    # Sort v in descending order
    # Find the unique threshold
    n = len(v)
    # Simple approach: iteratively adjust
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find largest rho such that v_sorted[rho] > (cumsum[rho] - 1) / (rho + 1)
    rho_mask = (np.arange(1, n+1) * v_sorted) > (cumsum - 1)
    rho = np.sum(rho_mask)
    if rho == 0:
        return np.full(n, s / n)
    theta = (cumsum[rho-1] - 1) / rho
    result = np.maximum(v - theta, 0)
    result *= s / result.sum() if result.sum() > 0 else 0
    return result

            K = K[np.argsort(c[K])[::-1][:20]]
        g = subgrad_of_max(a, c, K)
        v_mom = mu * v_mom + (1.0 - mu) * g
        dd = float(v_mom.dot(v_mom))
        if dd < 1e-30:
            break
        step = lr
        accepted = False
        for _ in range(10):
            a_try = project_simplex(a - step * v_mom, 1.0)
            c_try = fft_conv_sq(a_try)
            f_try = c_try.max()
            if f_try <= mc - c1 * step * dd:
                a = a_try
                mc = f_try
                accepted = True
                break
            step *= 0.5
        if accepted:
            stall = 0
        else:
            stall += 1
            lr *= 0.6
            v_mom *= 0.5
            if stall > 50:
                break
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    j = np.arange(n, dtype=np.float64)

    inits = [parent,
             np.cos(np.pi * j / (2.0 * n)) ** 2,
             0.5 * (1.0 + np.cos(2.0 * np.pi * j / n)) + 1e-8,
             1.0 - np.abs(2.0 * j / (n - 1) - 1.0) + 0.05,
             np.exp(-2.0 * (j / n - 0.5) ** 2)]

    best = None
    best_score = np.inf
    best_norm = None
    for init in inits:
        init = np.clip(init.astype(np.float64), 0.0, 1000.0)
        s = init.sum()
        if s < 0.01:
            continue
        sc = float(evaluate_sequence(init.tolist()))
        if sc < best_score:
            best_score = sc
            best = init.copy()
            best_norm = init / s

    if best is None:
        best = np.ones(n)
        best_score = 2.0
        best_norm = best / n

    # Main gradient optimization on simplex
    a_opt = optimize_simplex(best_norm, n, 200000, deadline, lr_init=0.5)
    sc_opt = float(evaluate_sequence((a_opt * best.sum()).tolist()))
    if sc_opt < best_score:
        best_score = sc_opt
        best = a_opt * best.sum()
        best_norm = a_opt.copy()

    # Phase 2: refine with smaller lr
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a_opt2 = optimize_simplex(best_norm, n, 200000, deadline, lr_init=0.05)
        sc2 = float(evaluate_sequence((a_opt2 * best.sum()).tolist()))
        if sc2 < best_score:
            best_score = sc2
            best = a_opt2 * best.sum()
            best_norm = a_opt2.copy()

    # Phase 3: random restarts from perturbed best
    remaining = deadline - time.monotonic()
    S = best.sum()
    for _ in range(500):
        if time.monotonic() > deadline:
            break
        pct = 0.05 + 0.15 * rng.random()
        a_pert = best_norm * (1.0 + pct * rng.standard_normal(n))
        a_pert = project_simplex(a_pert, 1.0)
        a_try = optimize_simplex(a_pert, n, 20000, deadline, lr_init=0.3)
        sc_try = float(evaluate_sequence((a_try * S).tolist()))
        if sc_try < best_score:
            best_score = sc_try
            best = a_try * S
            best_norm = a_try.copy()

    # Final validation
    final = np.clip(best, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(n)
    fs = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(fs):
        final = np.ones(n)
    return [float(x) for x in final]
