def _cheb_matrix(x, M):
    B = np.empty((len(x), M), dtype=np.float64)
    if M==0: return B
    T0 = np.ones_like(x)
    if M>0: B[:,0]=x  # T1? Excluding constant
    if M==1: return B
    T1 = x.copy()
    for k in range(2, M+1):
       T2 = 2*x*T1 - T0
       B[:, k-1] = T2
       T0,T1 = T1,T2
    return B

def _next_pow2(length):

def _self_conv(a,N):
    m = 2*a.size - 1
    fa = np.fft.rfft(a,N)
    c = np.fft.irfft(fa*fa,N)
    c = c[:m]
    if c.size:

    return c

def _smooth_F(q, gamma):

    v = np.exp(gamma*(q-m))

def _z_to_a(z,n):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.mean(z))
    mx = float(np.max(z))
    if not np.isfinite(mx):
        return np.ones(n, dtype=np.float64)/n
    a = np.exp(z - mx)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64)/n
    return a/s

def _cheb_basis(x,M):

def _full_search(z,n,N,deadline,rng,gamma_start,gamma_end,eta0,target_q):

    z = z - float(np.mean(z))

    gamma=gamma_start

import numpy as np
import time
from operator import itemgetter


def _next_pow2(length):
    n = 1
    while n < length:
        n <<= 1
    return n


def _self_conv(a, N):
    a = np.asarray(a, dtype=np.float64)
    if a.size <= 512:
        return np.maximum(np.convolve(a, a), 0.0)
    m = 2 * a.size - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return np.maximum(c[:m], 0.0)


def _smooth_F(q, gamma):
    m = float(np.max(q))
    v = np.exp(gamma * (q - m))
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return m, m
    return float(m + np.log(s) / gamma), m


def _softmax_p(q, gamma):
    m = float(np.max(q))
    v = np.exp(gamma * (q - m))
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(q.size, 1.0 / q.size)
    return v / s


def _corr_p_a(p, a, N):
    fp = np.fft.rfft(p, N)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fp * np.conj(fa), N)
    return np.maximum(r[:a.size], 0.0)


def _z_to_a(z, n):
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.mean(z))
    mx = float(np.max(z))
    if not np.isfinite(mx):
        return np.ones(n, dtype=np.float64) / n
    a = np.exp(z - mx)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _score_from_a(a, n, N):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return float("inf")
    c = _self_conv(a, N)
    cm = float(c.max()) if c.size else 0.0
    if not np.isfinite(cm):
        return float("inf")
    return float(2.0 * n * cm / (s * s))


def _cheb_basis(x, M):
    n = x.size
    B = np.empty((n, M), dtype=np.float64)
    if M <= 0:
        return B
    T0 = np.ones_like(x, dtype=np.float64)
    T1 = np.asarray(x, dtype=np.float64).copy()
    B[:, 0] = T1
    for k in range(2, M + 1):
        T2 = 2.0 * x * T1 - T0
        B[:, k - 1] = T2
        T0, T1 = T1, T2
    return B


def _add_smooth_noise(z, rng, scale):
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if scale <= 0.0 or n < 2:
        return z
    t = 2.0 * np.linspace(0.0, 1.0, n) - 1.0
    noise = np.zeros(n, dtype=np.float64)
    for k in range(1, 7):
        amp = scale / float(k)
        c = amp * rng.standard_normal()
        s = amp * rng.standard_normal()
        noise += c * np.cos(2.0 * np.pi * k * t) + s * np.sin(2.0 * np.pi * k * t)
    return z + noise
