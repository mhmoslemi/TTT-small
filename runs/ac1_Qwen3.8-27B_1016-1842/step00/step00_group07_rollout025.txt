import numpy as np
import time


def _local_search(best, rng, deadline, max_iters=200000):
    """Greedy block-mean local search; O(n) per iteration."""
    n = best.size
    best_score = float(evaluate_sequence(best.tolist()))
    small = max(2, n // 200)
    large = max(small + 1, n // 10)
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        cand = best.copy()
        w = int(rng.integers(small, large + 1))
        s = int(rng.integers(0, n - w + 1))
        mean = cand[s:s + w].mean()
        alpha = float(rng.uniform(0.05, 0.5))
        cand[s:s + w] = (1.0 - alpha) * cand[s:s + w] + alpha * mean

        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:

            best_score = sc
