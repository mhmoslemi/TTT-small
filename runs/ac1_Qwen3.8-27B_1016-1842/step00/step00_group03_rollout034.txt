from scipy import sparse
import scipy.optimize as opt

def solve_lp_step(f, n, M):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    # Build sparse convolution matrix (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            rows.extend(np.full(len(idx), k))
            cols.extend(idx)
            vals.extend(f[idx])
    C = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Add nonnegativity: -I g <= 0
    # A_ub = [C; -I], b_ub = [M*1; 0]
    I = sparse.eye(n, format='csr')
    A_ub = sparse.vstack([C, -I], format='csr')
    b_ub = np.concatenate([M * np.ones(2*n-1), np.zeros(n)])
    
    # c = -1 (minimize -sum(g))
    c_vec = -np.ones(n)
    
    res = opt.linprog(c_vec, A_ub=A_ub, b_ub=b_ub, method='highs', 
                       options={'time_limit': 60})
    if res.status == 0:
        return res.x
    return None

def build_conv_matrix(f, n):
    """Build sparse (2n-1) x n convolution matrix."""
    # Row k, col i: f[i] if max(0, k-n+1) <= i <= min(k, n-1)
    rows_parts = []
    cols_parts = []
    vals_parts = []

        i_hi = min(k, n - 1)
