import numpy as np
import time


def _fast_auto_max(a):
    """Compute max of full linear autocorrelation via FFT in O(n log n)."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    ac = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return float(ac.max())


def _score_fast(a):
    """Quick score estimate using FFT; exact up to roundoff."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    mb = _fast_auto_max(a)
    return float(2.0 * n * mb / (s * s))
