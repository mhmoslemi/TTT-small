import numpy as np
import time

def _compute_score_and_grad(a, n):
    """Compute score, max position, and gradient. O(n log n)."""
    c = np.convolve(a, a)
    S = a.sum()
    if S < 1e-10:
        return np.inf, -1, np.zeros(n)
    M = c.max()
    j_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    
    # Gradient
    inv_S3 = 1.0 / (S * S * S)
    grad = np.full(n, -4.0 * n * M * inv_S3)
    lo = max(0, j_star - n + 1)
    hi = min(n, j_star + 1)
    if hi > lo:
        idx = np.arange(lo, hi)
        mirror = j_star - idx
        grad[idx] = 4.0 * n * (a[mirror] * S - M) * inv_S3
    return score, j_star, grad

def _grad_descent(a, n, rng, deadline, best, best_score):
    """Run gradient descent from initial point a."""
    lr = 0.05
    stale = 0
    a = a.copy()
    
    while time.monotonic() < deadline:
        score, j_star, grad = _compute_score_and_grad(a, n)
        if not np.isfinite(score):

            best[:] = a
            stale = 0

            stale += 1

            scale = 0.5 * max(1.0, np.std(a))
            a += rng.normal(0, scale, n)
            a = np.clip(a, 0.0, 1000.0)

            stale = 0
