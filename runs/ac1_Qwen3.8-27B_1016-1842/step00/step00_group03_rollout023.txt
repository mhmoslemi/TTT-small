    M = conv.max()

    active = np.where(conv >= threshold)[0]

        # j in [max(0, k-n+1), min(n-1, k)]

        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += a[k - j_hi : k - j_lo + 1][::-1]

        # j in [k, min(n-1, k+n-1)] but also j>=0, so j in [max(0,k), min(n-1, k+n-1)]
        j_lo2 = max(0, k)

        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] += a[0:j_hi2 - j_lo2 + 1]
