def lowpass_noise(n, rng, frac):
    r = rng.normal(size=n)
    R = np.fft.rfft(r)
    k = np.arange(R.size, dtype=np.float64)
    cutoff = max(1.0, float(R.size)*frac)
    mask = 1.0 / (1.0 + (k/cutoff)**2)
    out = np.fft.irfft(R*mask, n)
    out -= out.mean()
    std = float(out.std())
    if std > 1e-12: out /= std
    return out
```

def run_search(a, n, S, size_c, size_x, deadline, rng):
    a = make_valid(a, S).copy()
    c = self_conv(a, size_c)
    cur = float(c.max())
    best = a.copy()
    best_score = cur
    z = np.log(a)
    z -= z.mean()
    T = min(0.35, max(0.04, 0.12 * cur))
    lr = 0.18
    beta = 0.55
    maxstep = 0.7
    mom = np.zeros(n, dtype=np.float64)
    stall = 0
    eps = 1e-12 * max(1.0, S)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
        p, _ = lse_weights(c, T)
        v = corr_pa(p, a, size_x, n)
        g = a * (2.0 * T * v)
        g -= g.mean()
        scale = float(np.mean(np.abs(g)))
        if not np.isfinite(scale) or scale < 1e-15:
            # perturb
            pert = lowpass_noise(n, rng, 0.2)
            z += 0.35 * pert
            z -= z.mean()
            ex = np.exp(z - z.max())
            a = make_valid(S * ex / max(1e-300, ex.sum()), S)
            c = self_conv(a, size_c)
            cur = float(c.max())
            z = np.log(a); z -= z.mean()
            mom = np.zeros(n); lr=0.18; T=max(0.1,T)
            continue
        g /= scale
        mom = beta * mom - lr * g
        mom -= mom.mean()
        np.clip(mom, -maxstep, maxstep, out=mom)
        z_new = z + mom
        z_new -= z_new.mean()
        np.clip(z_new, -50.0, 50.0, out=z_new)
        ex = np.exp(z_new - z_new.max())
        denom = float(ex.sum())
        if not np.isfinite(denom) or denom <= 0:
            lr *= 0.7
            continue
        a_new = S * ex / denom
        a_new = np.clip(a_new, eps, 1000.0)
        a_new *= S / float(a_new.sum())
        c_new = self_conv(a_new, size_c)
        new = float(c_new.max())
        if np.isfinite(new) and new < cur - 1e-12:
            a = a_new
            c = c_new
            cur = new
            z = np.log(a)
            z -= z.mean()
            stall = 0
            lr = min(0.8, lr * 1.03)
            T = max(0.02, T * 0.9997)
        else:
            lr *= 0.88
            mom *= 0.5
            stall += 1
            if stall > 25:
                pert = lowpass_noise(n, rng, 0.2)
                z += 0.35 * pert
                z -= z.mean()
                ex = np.exp(z - z.max())
                a = make_valid(S * ex / max(1e-300, ex.sum()), S)
                c = self_conv(a, size_c)
                cur = float(c.max())
                z = np.log(a); z -= z.mean()
                mom = np.zeros(n, dtype=np.float64)
                lr = 0.18
                T = max(T, 0.1)
                stall = 0
            T = max(0.02, T * 0.999)
        if it % 2000 == 0 and cur < best_score - 1e-12:
            best_score = cur; best = a.copy()
    if cur < best_score - 1e-12:
        best_score = cur; best = a.copy()
    return best, best_score
```

```python
import numpy as np
import time


def next_pow2(m):
    m = max(1, int(m))
    s = 1
    while s < m:
        s <<= 1
    return s


def self_conv(a, size):
    a = np.asarray(a, dtype=np.float64)
    m = 2 * a.size - 1
    F = np.fft.rfft(a, size)
    c = np.fft.irfft(F * F, size)
    c = c[:m]
    np.maximum(c, 0.0, out=c)
    return c


def lse_weights(c, T):
    T = float(max(1e-6, T))
    cmax = float(c.max())
    x = (c - cmax) / T
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros_like(c, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        p = ex / s
    return p


def corr_pa(p, a, size, n):
    # v[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
    ar = a[::-1]
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(ar, size)
    h = np.fft.irfft(P * A, size)
    return h[n - 1:2 * n - 1]


def make_valid(a, S):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.size)
    a = float(S) * a / s
    eps = 1e-12 * max(1.0, float(S))
    a = np.maximum(a, eps)
    a = float(S) * a / float(a.sum())
    return a


def lowpass_noise(n, rng, frac):
    r = rng.normal(size=n)
    R = np.fft.rfft(r)
    k = np.arange(R.size, dtype=np.float64)
    cutoff = max(1.0, float(R.size) * float(frac))
    mask = 1.0 / (1.0 + (k / cutoff) * (k / cutoff))
    out = np.fft.irfft(R * mask, n)
    out -= float(out.mean())
    sd = float(out.std())
    if sd > 1e-12:
        out /= sd
    return out


def golay_pair_cropped(n):
    n = max(1, int(n))
    L = 1
    while L < n:
        L <<= 1
    u = np.array([1.0], dtype=np.float64)
    v = np.array([1.0], dtype=np.float64)
    while u.size < L:
        u_next = np.concatenate((u, v))
        v_next = np.concatenate((u, -v))
        u, v = u_next, v_next
    return u[:n], v[:n]


def quick_score(a, size):
    c = self_conv(a, size)
    return float(c.max())


def run_search(a, n, S, size_c, size_x, deadline, rng):
    a = make_valid(a, S).copy()
    c = self_conv(a, size_c)
    cur = float(c.max())
    best = a.copy()
    best_score = cur

    z = np.log(a)
    z -= float(z.mean())

    T = float(min(0.35, max(0.04, 0.12 * cur)))
    lr = 0.18
    beta = 0.55
    maxstep = 0.7
    mom = np.zeros(n, dtype=np.float64)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()

        p = lse_weights(c, T)
        v = corr_pa(p, a, size_x, n)
        g = a * (2.0 * T * v)
        g -= float(g.mean())
        scale = float(np.mean(np.abs(g)))
        if not np.isfinite(scale) or scale < 1e-15:
            pert = lowpass_noise(n, rng, 0.2)
            z += 0.35 * pert
            z -= float(z.mean())
            ex = np.exp(z - float(z.max()))
            a = make_valid(S * ex / max(1e-300, float(ex.sum())), S)
            c = self_conv(a, size_c)
            cur = float(c.max())
            z = np.log(a)
            z -= float(z.mean())
            mom = np.zeros(n, dtype=np.float64)
            lr = 0.18
            T = float(max(0.1, T))
            continue

        g /= scale
        mom = beta * mom - lr * g
        mom -= float(mom.mean())
        np.clip(mom, -maxstep, maxstep, out=mom)

        z_new = z + mom
        z_new -= float(z_new.mean())
        np.clip(z_new, -50.0, 50.0, out=z_new)

        ex = np.exp(z_new - float(z_new.max()))
        denom = float(ex.sum())
        if not np.isfinite(denom) or denom <= 0.0:
            lr *= 0.7
            continue

        a_new = S * ex / denom
        a_new = np.clip(a_new, 1e-12 * max(1.0, S), 1000.0)
        a_new *= S / float(a_new.sum())

        c_new = self_conv(a_new, size_c)
        new = float(c_new.max())

        accept = np.isfinite(new) and (
            new < cur - 1e-12 or
            (stall > 30 and new < cur + 0.1 and float(rng.random()) < 0.15)
        )

        if accept:
            a = a_new
            c = c_new
            cur = new
            z = np.log(a)
            z -= float(z.mean())
            stall = 0
            if new < cur or True:
                lr = float(min(0.8, lr * 1.03))
                T = float(max(0.02, T * 0.9997))
            else:
                lr = float(lr * 0.85)
                T = float(max(0.06, T))
        else:
            lr = float(lr * 0.88)
            mom *= 0.5
            stall += 1
            T = float(max(0.02, T * 0.999))
            if stall > 25:
                pert = lowpass_noise(n, rng, 0.2)
                z += 0.35 * pert
                z -= float(z.mean())
                ex = np.exp(z - float(z.max()))
                a = make_valid(S * ex / max(1e-300, float(ex.sum())), S)
                c = self_conv(a, size_c)
                cur = float(c.max())
                z = np.log(a)
                z -= float(z.mean())
                mom = np.zeros(n, dtype=np.float64)
                lr = 0.18
                T = float(max(0.1, T))
                stall = 0

        if it % 2000 == 0 and cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()

    if cur < best_score - 1e-12:
        best_score = cur
        best = a.copy()
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    now = time.monotonic()
    deadline = now + max(1.0, float(budget_s) - 25.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n = int(parent.size)
    except Exception:
        parent = np.ones(1, dtype=np.float64)
        n = 1
    if n < 1:
        n = 1
        parent = np.ones(1, dtype=np.float64)

    S = float(np.sqrt(2.0 * float(n)))
    size_c = next_pow2(2 * n - 1)
    size_x = next_pow2(3 * n)

    parent_v = make_valid(parent, S)

    inits = []

    noise_p = lowpass_noise(n, rng, 0.05)
    inits.append(parent_v + 0.25 * noise_p)

    u, v = golay_pair_cropped(n)
    inits.append((u + v + 2.0) / 4.0)
    inits.append((u - v + 2.0) / 4.0)
    inits.append((u + np.roll(v, 1) + 2.0) / 4.0)
    inits.append((u + 1.0) / 2.0)
    inits.append((v + 1.0) / 2.0)

    r = lowpass_noise(n, rng, 0.15)
    inits.append(1.0 + 0.5 * r)

    idx = np.arange(n, dtype=np.int64)
    L = next_pow2(n)
    ph = (idx * idx) % int(L)
    ch = np.cos(2.0 * np.pi * ph / float(L))
    inits.append(0.5 + 0.5 * ch)

    inits.append(np.ones(n, dtype=np.float64))

    scored = []
    for init in inits:
        a = make_valid(init, S)
        sc = quick_score(a, size_c)
        if np.isfinite(sc):
            scored.append((sc, a))

    if not scored:
        a = make_valid(np.ones(n, dtype=np.float64), S)
        scored.append((quick_score(a, size_c), a))

    scores = np.empty(len(scored), dtype=np.float64)
    for i, s in enumerate(scored):
        scores[i] = s[0]
    order = np.argsort(scores)
    scored = [scored[i] for i in order]

    global_best = scored[0][1].copy()
    global_score = float(scored[0][0])

    weights = np.array([0.32, 0.20, 0.12, 0.10, 0.09, 0.08, 0.04, 0.03, 0.02],
                       dtype=np.float64)
    if len(scored) > weights.size:
        extra = np.full(len(scored) - weights.size, 0.005, dtype=np.float64)
        weights = np.concatenate((weights, extra))
    weights = weights[:len(scored)]
    wsum = float(weights.sum())
    if not np.isfinite(wsum) or wsum <= 0.0:
        weights = np.ones(len(scored), dtype=np.float64)
    else:
        weights = weights / wsum

    for i in range(len(scored)):
        t = time.monotonic()
        avail = deadline - 12.0 - t
        if avail <= 3.0:
            break
        rem_w = float(np.sum(weights[i:]))
        if rem_w <= 0.0:
            break
        frac = min(0.98, float(weights[i]) / rem_w)
        sub_deadline = t + max(2.0, avail * frac - 2.0)
        sub_deadline = min(sub_deadline, deadline - 10.0)

        best_a, best_sc = run_search(
            scored[i][1].copy(), n, S, size_c, size_x, sub_deadline, rng
        )
        if np.isfinite(float(best_sc)) and float(best_sc) < global_score:
            global_score = float(best_sc)
            global_best = best_a.copy()

    if time.monotonic() < deadline - 10.0:
        best_a, best_sc = run_search(
            global_best.copy(), n, S, size_c, size_x, deadline - 5.0, rng
        )
        if np.isfinite(float(best_sc)) and float(best_sc) < global_score:
            global_score = float(best_sc)
            global_best = best_a.copy()

    arr = np.asarray(global_best, dtype=np.float64).ravel()
    if arr.size == 0 or not np.isfinite(arr).all():
        arr = np.ones(n, dtype=np.float64)

    best_list = []
    for x in arr:
        best_list.append(float(x))

    if n <= 8000 and time.monotonic() < deadline - 2.0:
        try:
            exact = float(evaluate_sequence(best_list))
            if not np.isfinite(exact):
                best_list = [1.0] * n
        except Exception:
            pass

    return best_list
```
