import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]


def _score(a):
    s = float(a.sum())
    if s < 1e-9:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * len(a) * float(c.max()) / (s * s)


def _grad_top(a, k):
    n = len(a)
    c = _fft_conv(a)
    idx = np.argpartition(c, -k)[-k:]
    w = np.exp(c[idx] / 0.05 - np.max(c[idx]))
    w /= w.sum()
    g = np.zeros(n)
    for ii, lag in enumerate(idx):
        lo_j = max(0, lag - n + 1)
        hi_j = min(n - 1, lag)
        segment = np.arange(lo_j, hi_j + 1)
        vals = a[np.clip(lag - segment, 0, n - 1)]
        coef = 2.0 * w[ii] / k
        np.add.at(g, segment, coef * vals)
    s = float(a.sum())
    cmax = float(c.max())
    return 2.0 * n * (s * g - 2.0 * cmax / s) / (s * s), cmax
