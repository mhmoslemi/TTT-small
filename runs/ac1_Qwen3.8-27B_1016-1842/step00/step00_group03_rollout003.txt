def _normalize(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:

    return a / s


def _subgradient(a, conv):
    """Compute subgradient of max(conv) w.r.t. a, in O(n)."""
    n = a.size
    mx = conv.max()
    # Find positions where conv is within tolerance of max
    tol = mx * 1e-10 + 1e-15
    active = np.where(conv >= mx - tol)[0]
    g = np.zeros(n)
    for m in active:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[m - np.arange(lo, hi + 1)]
    g /= len(active)
    return g

    if s < 0.01:
        return np.inf
