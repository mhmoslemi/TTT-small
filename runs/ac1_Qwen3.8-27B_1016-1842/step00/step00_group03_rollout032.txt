def fft_convolve_sq(a):
    """Compute a*a convolution via FFT. O(n log n)."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:2*n-1]
    return b

def normalize(a):
    s = np.sum(a)
    if s < 1e-10:

import numpy as np
import time
import math


def fft_conv_sq(a):
    """Linear autocorrelation via FFT, O(n log n)."""
    n = a.size
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def topk_subgrad(a, ntop=4):
    """Smoothed subgradient: weighted avg of gradients from top-n peaks. O(n log n)."""
    n = a.size
    b = fft_conv_sq(a)
    k = min(ntop, 2 * n - 1)
    top = np.argpartition(b, -k)[-k:]
    top = top[np.argsort(b[top])[::-1]]
    w = b[top].astype(np.float64)
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(k):
        ki = int(top[idx])
        i0 = max(0, ki - n + 1)
        i1 = min(n - 1, ki)
        g[i0:i1 + 1] += w[idx] * 2.0 * a[ki - i1:ki - i0 + 1][::-1]
    return g, b
