import numpy as np
import time

def compute_score(a):
    """Fast surrogate score using FFT convolution."""
    n = len(a)
    # Use FFT for fast convolution
    size = 2 * n - 1
    # Round up to power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(fa * fa, fsize)[:size]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)
