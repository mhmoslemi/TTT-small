def compute_score(a):
    """Fast computation of the AC1 score."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    n = a.size
    # Linear convolution via FFT
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)

    n = a.size
