import numpy as np
import time

def compute_score(a):
    """Compute the AC1 upper bound score for sequence a."""
    a = np.clip(a.astype(np.float64), 0.0, 1000.0)
    n = len(a)
    if a.sum() < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (a.sum()**2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)
    
    # Initial parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    best = parent.copy()
    best_score = compute_score(best)
    
    # Try linear ramps as structured initializations
    # a[i] = low + (high-low) * i/(n-1)
    for low, high in [(1, 1000), (500, 1000), (100, 1000)]:
        t = np.linspace(0, 1, n)
        candidate = low + (high - low) * t
        score = compute_score(candidate)
        if score < best_score:
            best = candidate.copy()
            best_score = score
    
    # Local search: perturb blocks
    n_blocks = 32
    block_size = n // n_blocks
    while time.monotonic() < deadline:
        # Random block perturbation
        idx = rng.integers(0, n_blocks)
        start = idx * block_size
        end = min(n, (idx+1) * block_size)
        candidate = best.copy()
        candidate[start:end] *= np.exp(rng.normal(0, 0.05))
        candidate = np.clip(candidate, 0, 1000)
        score = compute_score(candidate)
        if score < best_score:
            best = candidate
            best_score = score
    
    return [float(x) for x in np.clip(best, 0, 1000)]
