import time
from scipy.optimize import minimize

_x_grid = None
_degree = None

def fast_conv_score(a):
    n = int(a.size)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(fa * fa, fsize)[:size]
    max_b = float(np.max(conv))
    s = float(np.sum(a))
    if s < 0.01:
        return 1e10
    return 2.0 * n * max_b / (s * s)

    a = np.zeros(_x_grid.size, dtype=np.float64)

        a += params[k] * (_x_grid ** k)
    a = np.clip(a, 0.0, 1000.0)

    cand = base.copy()
    w = max(2, int(n * block_frac))
    start = int(rng.integers(0, n - w))
    cand[start:start + w] *= np.exp(rng.normal(0.0, sigma, size=w))

        cand[start2:start2 + w2] *= np.exp(rng.normal(0.0, sigma * 0.5, size=w2))
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand
