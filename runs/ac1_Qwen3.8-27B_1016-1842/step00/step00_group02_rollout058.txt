import numpy as np
import time
from scipy.optimize import minimize


def conv_full(x):
    """Full (linear) convolution of x with itself via FFT. O(n log n)."""
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    size = 1
    need = 2 * n - 1
    while size < need:
        size <<= 1
    fx = np.fft.rfft(x, size)
    out = np.fft.irfft(fx * fx, size)
    return out[:need]


def score_of(a):
    """AC1 upper bound for a nonnegative finite array."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    mb = float(np.max(conv_full(a)))
    return 2.0 * n * mb / (s * s)


def poly_shape(coef, t):
    """Evaluate polynomial in t from coefficient list (ascending power)."""
    a = np.zeros_like(t)
    for k, c in enumerate(coef):
        a = a + c * (t ** k)
    return np.clip(a, 0.0, 1000.0)


def make_candidates(n):

    pool = [np.ones(n, dtype=np.float64)]
    for c in np.linspace(0.5, 12.0, 16):

        pool.append(c1 + c2 * t + 0.5 * c2 * t * t)

        pool.append(c1 - c2 * x * x)

    def fun(p):
        a = np.zeros_like(t)

        return score_of(a)
    x0 = list(init[:degree + 1]) + [1.0] * max(0, degree + 1 - len(init))
    bounds = [(0.0, 1000.0)] * (degree + 1)

                   options={"maxiter": 60, "maxls": 30, "ftol": 1e-10})
    p = res.x if res.success else x0
    a = np.zeros_like(t)
    for k in range(degree + 1):
        a = a + p[k] * (t ** k)
    return np.clip(a, 1e-6, 1000.0), float(fun(p))
