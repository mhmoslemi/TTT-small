import time

def parametric_seq(n, shape, params):
    """Generate a parametric sequence."""
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1)  # 0 to 1
    if shape == 'linear':
        h, floor = params
        return floor + (h - floor) * (1 - t)
    elif shape == 'quad':
        h, floor = params
        return floor + (h - floor) * (1 - t)**2
    elif shape == 'power':
        h, floor, p = params
        return floor + (h - floor) * (1 - t)**p
    elif shape == 'exp':
        h, floor, alpha = params
        return floor + (h - floor) * np.exp(-alpha * j)
    else:
        return np.ones(n)
