        w = int(rng.integers(1, max(2, n // 30)))
        st = int(rng.integers(0, n - w))
        f[st:st + w] *= np.exp(rng.normal(0.0, 0.005, w))
        np.maximum(f, 0.0, out=f)
        s = f.sum()
        f *= 1.0 / s if s > 0 else 0
        
        if it % 200 == 0:
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
            else:
                f = best_f.copy()

    return [float(x) for x in best_f]
