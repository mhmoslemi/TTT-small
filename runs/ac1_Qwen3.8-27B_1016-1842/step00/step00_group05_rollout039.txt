import time

def _score(a):
    """Compute the AC1 score without calling evaluate_sequence."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    c = np.convolve(a, a)
    return 2.0 * n * c.max() / (S * S)

def _grad_direction(a):
    """Compute the subgradient direction for f(a) = 2n*max(c)/S²."""
    n = a.size
    S = a.sum()
    c = np.convolve(a, a)
    m = int(c.argmax())
    cm = c[m]
    
    # dc[m]/da[i]
    i_lo = max(0, m - n + 1)
    i_hi = min(n - 1, m)
    idx = np.arange(i_lo, i_hi + 1)
    j = m - idx
    dcda = a[j].copy()
    diag_mask = (j == idx)
