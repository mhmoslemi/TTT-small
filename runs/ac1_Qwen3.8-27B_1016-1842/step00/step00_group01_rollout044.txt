def fast_convolve_max(a):
    """Compute max of full linear convolution using FFT. O(n log n)."""
    n = a.size
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return float(np.max(conv[:2*n-1]))

        return float('inf')
