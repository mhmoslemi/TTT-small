        return None
    return a / s

def _optimize(a, deadline, rng, eta0, tau):
    n = a.size
    S = float(a.sum())
    if S < 1e-12:
        return a, float('inf')
    a = a / S
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta0
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        conv = _fft_conv(a, a)
        M = float(conv.max())
        if M <= 0:
            break
        L = min(5, conv.size)
        top_idx = np.argpartition(conv, -L)[-L:]
        top_vals = conv[top_idx]
        w = np.exp(tau * (top_vals - top_vals.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(L):
            k = int(top[j](top_idx)) if False else int(top_idx[j])
            wi = float(w[j])
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            if lo <= hi:
                grad[lo:hi + 1] += wi * a[k - hi:k - lo + 1][::-1]
        grad *= 4.0 * n
        grad -= 4.0 * n * M * a
        upd = np.clip(-eta * grad, -18.0, 18.0)
        a *= np.exp(upd)
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-15:
            a = np.ones(n) / n
            continue
        a /= S
        it += 1
        if it <= 3 or it % 5 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        if it % 400 == 0:
            eta *= 0.7
        if stall > 100:
            a += rng.normal(0.0, 2.0 / n, n)
            a = np.clip(a, 0.0, 1000.0)
            S = float(a.sum())
            if S > 1e-15:
                a /= S
            eta = eta0
            stall = 0
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 8.0)
    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
        a0 = np.ones(256)
    n = a0.size
    S0 = float(a0.sum())
    if S0 < 0.01:
        a0 = np.ones(n)
        S0 = float(n)
    a0 = a0 / S0
    best_a = a0.copy()
    best_sc = float(evaluate_sequence(best_a.tolist()))
    inits = [a0.copy()]
    for m in range(1, 8):
        ia = _init_mode(n, rng, m)
        if ia is not None:
            inits.append(ia)
    per = max(8.0, (float(budget_s) - 25.0) / len(inits))
    for ia in inits:
        if time.monotonic() > deadline - 12.0:
            break
        dl = min(deadline - 5.0, time.monotonic() + per)
        ra, rs = _optimize(ia, dl, rng, 1.0, 30.0)
        if rs < best_sc:
            best_sc = rs
            best_a = ra.copy()
    if time.monotonic() < deadline - 12.0:
        ra, rs = _optimize(best_a, deadline - 4.0, rng, 0.4, 80.0)
        if rs < best_sc:
            best_sc = rs
            best_a = ra.copy()
        fs = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(fs) and fs < best_sc:
            best_sc = fs
    fs = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(fs):
        return [1.0] * n
    return [float(x) for x in best_a]
