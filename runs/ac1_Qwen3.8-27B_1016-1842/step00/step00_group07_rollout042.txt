import numpy as np

def _block_perturb(base, rng):
    """Return a O(n)-memory multiplicative block perturbation of base."""
    cand = base.copy()
    n = cand.size
    w = int(rng.integers(max(1, n // 400), max(2, n // 15)))
    s = int(rng.integers(0, n - w))
    e = s + w
    cand[s:e] *= float(np.exp(rng.normal(0.0, 0.04)))
    if rng.random() < 0.25:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
