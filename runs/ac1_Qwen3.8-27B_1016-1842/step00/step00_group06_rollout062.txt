import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):

def _two_block(n, scale=1.0):
    """Construct a two-block sequence: 1s at edges, 0s in middle."""
    seq = np.zeros(n, dtype=np.float64)
    if n < 4:
        seq[:] = scale
        return seq
    gap = 1 if n % 2 == 1 else 2
    L = (n - gap) // 2
    if L < 1:
        seq[:] = scale
        return seq
    seq[:L] = scale
    seq[-L:] = scale
    return seq
