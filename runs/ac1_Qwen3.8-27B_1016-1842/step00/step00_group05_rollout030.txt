import numpy as np
import time

def smooth_perturbation(n, rng):
    """Generate a random smooth multiplicative perturbation."""
    x = np.linspace(0, 1, n)
    n_harmonics = rng.integers(1, 8)
    pert = np.zeros(n)
    for _ in range(n_harmonics):
        k = rng.integers(1, n_harmonics + 1)
        amp = rng.exponential(0.05)
        phase = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * x + phase)
    return np.exp(pert)

def block_perturbation(n, rng, base):
    """Generate a block multiplicative perturbation of base."""
    cand = base.copy()
    width = rng.integers(2, max(3, n // 40))
    start = rng.integers(0, n - width)
    cand[start:start+width] *= np.exp(rng.normal(0, 0.08, width))
    return np.clip(cand, 0, 1000)

def evaluate_shape(a, n):
    """Evaluate a shape (assumed nonnegative, will be normalized)."""
    a = np.clip(a, 0, 1000)
    if a.sum() < 0.01:
        return np.inf, a
    s = float(evaluate_sequence(a.tolist()))
    return s, a

            amp = rng.exponential(0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * k * x + phase)

        a = a / a.sum() * 100
        
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:

            best = a.copy()
