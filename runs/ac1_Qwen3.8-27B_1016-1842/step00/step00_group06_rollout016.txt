```python
def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for threshold
    mu = np.sort(v)[::-1]
    sv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n+1) > (sv - target_sum))[0][-1]
    theta = (sv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```
