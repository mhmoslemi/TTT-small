import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    T = 0.05
    T_min = 0.001
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        x = best.copy()
        move_type = rng.integers(0, 4)
        
        if move_type == 0:

            w = int(rng.integers(max(2, n//100), n//5 + 1))
            start = int(rng.integers(0, n - w))
            x[start:start+w] *= np.exp(rng.normal(0, 0.2 + 0.5*T))
        elif move_type == 1:

            amp = rng.uniform(0.02, 0.3)
            x *= 1.0 + amp * np.cos(2*np.pi*freq*np.arange(n)/n + rng.uniform(0, 2*np.pi))
        elif move_type == 2:
            # Two-block swap/blend
            w1 = int(rng.integers(max(2, n//200), n//10 + 1))
            w2 = int(rng.integers(max(2, n//200), n//10 + 1))
            s1 = int(rng.integers(0, n - w1))
            s2 = int(rng.integers(0, n - w2))
            blend = float(rng.uniform(0.2, 0.8))
            tmp = x[s1:s1+w1].copy()
            x[s1:s1+w1] = blend*x[s1:s1+w1] + (1-blend)*x[s2:s2+w2]
            x[s2:s2+w2] = (1-blend)*tmp + blend*x[s2:s2+w2]
        else:
            # Random coordinate perturbation
            k = int(rng.integers(1, min(50, n//10) + 1))
            idx = rng.integers(0, n, size=k)
            x[idx] += rng.normal(0, 0.1, size=k)
        
        x = np.maximum(x, 0.0)
        # Renormalize
        sx = np.sum(x)
        if sx < 1e-10:
            continue
        x = x / sx  # scale doesn't matter
        
        score = float(evaluate_sequence(x.tolist()))
        if not np.isfinite(score):
            continue
        
        if score < best_score:
            best = x.copy()
            best_score = score
        else:
            # Simulated annealing acceptance
            d = score - best_score
            if d > 0 and rng.random() < np.exp(-d / (T + 1e-10)):
                best = x.copy()
        
        # Cool down
        T = max(T_min, T * 0.9995)
    
    return [float(v) for v in np.clip(best, 0.0, 1000.0)]
