import numpy as np
import time

def compute_score_parts(a):
    """Return (b, M, T, k_positions) where b=conv, M=max, T=sum, k_positions=argmax."""
    b = np.convolve(a, a)
    T = float(np.sum(a))
    M = float(np.max(b))
    # Find positions where b is close to M (within numerical tolerance)
    tol = 1e-10 * max(1.0, abs(M))
    k_pos = np.where(b >= M - tol)[0]
    return b, M, T, k_pos

def compute_subgrad(a, M, T, k_pos, n):
    """Compute subgradient of score = 2n*M/T^2."""
    g = np.full(n, -4.0 * n * M / (T * T))
    coeff = 2.0 * n / (T * len(k_pos))
    for k in k_pos:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

            # a[k-i] for i in [lo, hi]
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi+1] += coeff * 2.0 * a[idx]
    return g

import numpy as np
import time


def _score_parts(a):
    b = np.convolve(a, a)
    T = float(np.sum(a))
    M = float(np.max(b))
    tol = 1e-9 * max(1.0, abs(M))
    k_pos = np.where(b >= M - tol)[0]
    return b, M, T, k_pos


def _subgrad(a, M, T, k_pos, n):
    g = np.full(n, -4.0 * n * M / (T * T))
    coeff = 2.0 * n / (T * len(k_pos))
    for k in k_pos:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi + 1] += coeff * 2.0 * a[idx]
    return g


def _perturb(a, rng, scale=0.08):
    a2 = a.copy()
    n = a2.size
    nb = int(rng.integers(2, max(3, n // 500)))
    for _ in range(nb):
        w = int(rng.integers(max(1, n // 200), max(2, n // 20)))
        s = int(rng.integers(0, n - w))
        a2[s:s + w] *= float(np.exp(rng.normal(0, 0.25)))
    a2 += scale * np.mean(a) * rng.standard_normal(n)
    return np.clip(a2, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a.ndim != 1 or a.size == 0 or not np.all(np.isfinite(a)):
        a = np.ones(256, dtype=np.float64)
    n = a.size

    # Initial perturbation to break symmetry (uniform is a critical point)
    a = _perturb(a, rng, scale=0.05)

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    lr = 0.5
    m1 = 0.9
    m2 = 0.999
    beta1 = np.zeros(n)
    beta2 = np.zeros(n)
    t_step = 0
    no_improve = 0
    eval_count = 0

    while time.monotonic() < deadline:
        b, M, T, k_pos = _score_parts(a)
        if T < 0.01:
            a = np.ones(n) * 0.1
            continue

        # Direct score (avoids extra convolve)
        cur_score = 2.0 * n * M / (T * T)

        if np.isfinite(cur_score) and cur_score < best_score:
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
        else:
            no_improve += 1

        g = _subgrad(a, M, T, k_pos, n)
        t_step += 1
        beta1 = m1 * beta1 + (1 - m1) * g
        beta2 = m2 * beta2 + (1 - m2) * (g * g)
        b1c = beta1 / (1.0 - m1 ** t_step)
        b2c = beta2 / (1.0 - m2 ** t_step)

            lr = min(lr * 1.01, 2.0)
