import numpy as np
import time


def _score_arr(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return float('inf')
    m = np.convolve(a, a).max()
    return 2.0 * n * m / (s * s)


def _grad_dir(a, k):
    n = a.size
    s = a.sum()
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    rev = a[k - hi : k - lo + 1][::-1]
    conv_k = np.dot(a[lo:hi + 1], rev)
    d = np.zeros(n)
    d[lo:hi + 1] = conv_k / s - rev
    return d


def _make_inits(parent, n, seed):
    j = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]
    inits.append(1.0 + 1.5 * (j / float(n) - 0.3))
    inits.append(np.clip(np.exp(-1.2 * j / float(n)), 0.01, None))
    inits.append(np.clip(np.exp(-1.2 * (n - 1 - j) / float(n)), 0.01, None))
    inits.append(np.clip(0.3 + 1.2 * np.cos(np.pi * j / (float(n) + 1.0)), 0.01, None))
    inits.append(np.clip(1.0 + 0.8 * np.sin(2.0 * np.pi * j / float(n)), 0.01, None))
    rr = np.random.default_rng(seed + 13)
    inits.append(np.clip(1.0 + rr.normal(0, 0.5, n), 0.01, None))
    inits.append(np.clip(2.0 * np.exp(-4.0 * (j / float(n) - 0.4) ** 2), 0.01, None))
    inits.append(np.clip(1.0 + 0.7 * ((j % 7) / 7.0 - 0.5) * (j / float(n)), 0.01, None))
    inits.append(np.clip(1.5 * np.exp(-2.0 * j / float(n)) + 0.3, 0.01, None))
    return inits


def _block_perturb(a, rng, n):
    w = max(1, int(np.random.default_rng(int(rng.integers(0, 2**31))).integers(
        max(1, n // 40), n // 4)))
    st = int(rng.integers(0, max(1, n - w)))
    a[st:st + w] *= float(np.exp(rng.normal(0, 0.4)))
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _full_perturb(a, rng, n):
    w = max(1, int(n * rng.uniform(0.1, 0.5)))
    st = int(rng.integers(0, max(1, n - w)))
    a[st:st + w] *= float(np.exp(rng.normal(0, 0.6)))
    if st + w < n:
        w2 = max(1, int(n * rng.uniform(0.05, 0.3)))
        st2 = int(rng.integers(max(0, st + w), max(1, n - w2)))
        a[st2:st2 + w2] *= float(np.exp(rng.normal(0, 0.4)))
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 6.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)
    n = parent.size

    inits = _make_inits(parent, n, seed)
    best = None
    best_score = float('inf')
    for xi in inits:
        xi = np.clip(xi, 0.0, 1000.0)
        sc = _score_arr(xi)
        if sc < best_score:
            best_score = sc
            best = xi.copy()

    if best is None:
        best = parent.copy()
        best_score = _score_arr(best)

    step_fracs = [0.02, 0.05, 0.1, 0.2, 0.5, 0.08, 0.15]
    sf_idx = 0
    stuck = 0
    consecutive_no_imp = 0

    while time.monotonic() < deadline:
        a = best
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        s = a.sum()
        if s < 0.01:
            break

        d = _grad_dir(a, k)
        dn = np.linalg.norm(d)
        typ = float(np.median(a))

        improved = False
        if dn > 1e-14 and typ > 1e-14:
            for fi in range(len(step_fracs)):
                idx = (sf_idx + fi) % len(step_fracs)
                step = step_fracs[idx] * typ
                new_a = a + step * (d / dn)
                np.clip(new_a, 0.0, 1000.0, out=new_a)
                ns = new_a.sum()
                if ns < 0.01:
                    continue
                nc = np.convolve(new_a, new_a).max()
                nsc = 2.0 * n * nc / (ns * ns)
                if nsc < best_score:
                    best = new_a
                    best_score = nsc
                    sf_idx = fi
                    improved = True
                    break

        if improved:
            consecutive_no_imp = 0
            stuck = 0
        else:
            consecutive_no_imp += 1
            stuck += 1
            if consecutive_no_imp > 15:
                a2 = best.copy()
                _block_perturb(a2, rng, n)
                sc2 = _score_arr(a2)
                if sc2 < best_score:
                    best = a2
                    best_score = sc2
                consecutive_no_imp = 0
            if stuck > 60:
                a2 = best.copy()
                _full_perturb(a2, rng, n)
                sc2 = _score_arr(a2)
                if sc2 < best_score:
                    best = a2
                    best_score = sc2
                elif rng.random() < 0.3:
                    a2 = np.clip(1.0 + rng.normal(0, 0.5, n), 0.01, 1000.0)
                    sc2 = _score_arr(a2)
                    if sc2 < best_score:
                        best = a2
                        best_score = sc2
                stuck = 0
            sf_idx = (sf_idx + 1) % len(step_fracs)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if not np.isfinite(final_score) or final_score > 10.0:
        best = np.ones(min(n, 256), dtype=np.float64)
        final_score = float(evaluate_sequence([float(x) for x in best]))

    return [float(x) for x in best]
