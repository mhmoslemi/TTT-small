import numpy as np
import time


def normalize_seq(x, target):
    s = x.sum()
    if s < 1e-14:
        return None
    return x * (target / s)


def subgrad_at_lag(x, k, n):
    grad = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        lo = k - j_hi
        hi = k - j_lo + 1
        grad[j_lo:j_hi + 1] = x[lo:hi][::-1]
    return grad


def make_ramp(n, slope):
    if n <= 1:
        return np.ones(n)
    return 1.0 + slope * np.arange(n, dtype=np.float64) / (n - 1)


def run_subgradient(x, rng, n, deadline, best_ref, best_score_ref):
    target = np.sqrt(2.0 * n)
    x = x.copy()
    alpha = 0.6
    stall = 0
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        conv = np.convolve(x, x)
        k_star = int(conv.argmax())
        max_v = conv[k_star]

        grad = subgrad_at_lag(x, k_star, n)
        n_active = 1
        for dk in (-1, 1, -2, 2):
            k2 = k_star + dk
            if 0 <= k2 < len(conv) and conv[k2] > 0.92 * max_v:
                grad += 0.4 * subgrad_at_lag(x, k2, n)
                n_active += 1
        grad /= n_active

            alpha *= 0.85
