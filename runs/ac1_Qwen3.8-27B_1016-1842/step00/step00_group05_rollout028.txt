def compute_autocorr(a):
    """Compute full autocorrelation using FFT. O(n log n)."""
    n = len(a)

def autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    F = np.fft.rfft(a, size)
    return np.fft.irfft(np.abs(F)**2, size)

def grad_at_lags(a, lags):
    """Compute gradient of sum of b[k] for k in lags, w.r.t. a."""
    n = len(a)

        hi = min(n, k + 1)
        if lo < hi:

    return grad
