            break

        # Gradient: direction[i] = M - a[k*-i] (proportional to -∂f/∂a[i], S=1)
        a_shifted = np.zeros(n)
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        if i_max >= i_min:
            a_shifted[i_min:i_max + 1] = a[k_star - i_max:k_star - i_min - 1:-1]

        direction = M - a_shifted
        a = a + lr * direction
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-14:
            break
        a /= S

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    a_parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a_parent.ndim != 1 or a_parent.size == 0 or not np.isfinite(a_parent).all():
        a_parent = np.ones(2535)
    n = len(a_parent)

    global_best_a = a_parent.copy()
    global_best_score = 2.0 * n * fft_conv(a_parent).max() / a_parent.sum()**2

    # Build diverse initializations
    inits = []
    inits.append(a_parent.copy())
    inits.append((rng.random(n) > 0.45).astype(np.float64))
    inits.append((rng.random(n) > 0.35).astype(np.float64))
    # Periodic patterns
    for period in [3, 4, 5, 7, 11]:
        pat = np.zeros(period)
        pat[0:max(1, period // 3 + 1)] = 1.0
        inits.append(np.tile(pat, n // period + 1)[:n])
    # Smooth oscillatory
    t = np.linspace(0, 8 * np.pi, n)
    inits.append(np.abs(np.sin(t) + 0.6 * np.sin(2.3 * t + 1.1)))
    inits.append(0.5 + 0.5 * np.cos(3.7 * t + 0.4) * np.sin(1.3 * t))
    # Parent modulated
    inits.append(np.maximum(a_parent * np.exp(rng.normal(0, 1.0, n)), 0))
    inits.append(np.maximum(a_parent * (0.5 + rng.random(n)), 0))

    for idx, init in enumerate(inits):
        if time.monotonic() > deadline:
            break
        ba, bs = optimize_one(init, rng, deadline, n)
        if ba is not None:
            # Verify with exact evaluator
            s = float(evaluate_sequence(ba.tolist()))
            if np.isfinite(s) and s < global_best_score:
                global_best_score = s
                global_best_a = ba.copy()
        # Any additional short polish runs on best found
        if time.monotonic() < deadline and idx % 3 == 0:
            ba2, bs2 = optimize_one(global_best_a * np.exp(rng.normal(0, 0.1, n)), rng, deadline, n)
            if ba2 is not None:
                s2 = float(evaluate_sequence(ba2.tolist()))
                if np.isfinite(s2) and s2 < global_best_score:
                    global_best_score = s2
                    global_best_a = ba2.copy()

    # Long final polish on the best
    if time.monotonic() < deadline:
        ba, bs = optimize_one(global_best_a, rng, deadline, n)
        if ba is not None:
            s = float(evaluate_sequence(ba.tolist()))
            if np.isfinite(s) and s < global_best_score:
                global_best_score = s
                global_best_a = ba.copy()

    # Final safety check
    final_score = float(evaluate_sequence(global_best_a.tolist()))
    if not np.isfinite(final_score):
        global_best_a = a_parent
    return [float(x) for x in global_best_a]
