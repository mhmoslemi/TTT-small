    """Compute AC1 using FFT convolution. Returns (score, max_conv, sum_a)."""
    n = len(a)
    if n == 0 or np.sum(a) < 0.01:
        return np.inf, 0, 0
    conv = np.fft.fftconvolve(a, a, mode='full')
    # fftconvolve might give tiny imaginary parts
    conv = conv.real
    # Clip small negative values from numerical error
    conv = np.maximum(conv, 0)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    score = 2.0 * n * max_conv / (sum_a * sum_a)
    return score, max_conv, sum_a
