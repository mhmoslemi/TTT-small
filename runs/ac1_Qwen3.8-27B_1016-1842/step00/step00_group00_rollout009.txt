def proj_simplex(v):
    """Project v onto {x: x>=0, sum(x)=1}."""
    n = v.size
    sv = np.sort(v)[::-1]
    cv = np.cumsum(sv)
    r = np.arange(1, n + 1, dtype=np.float64)
    ok = (sv - (cv - 1.0) / r) > 0.0
    if not ok.any():
        return np.full(n, 1.0 / n)
    rho = int(np.max(r[ok]))
    theta = (cv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

    best_a = parent / psum
    best_mb = float(fft_conv(best_a).max())

    # Diverse initializations (all on the simplex)
    inits = [parent / psum, np.ones(n) / n]

    t = np.arange(n, dtype=np.float64) / n
    for freq in (2, 3, 5, 8):
        mod = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t)
        inits.append(mod / mod.sum())

    # Bimodal
    bi = np.full(n, 0.1, dtype=np.float64)
    bi[n // 5: n // 5 + n // 10] = 1.0
    bi[4 * n // 5: 4 * n // 5 + n // 10] = 1.0
    inits.append(bi / bi.sum())

    # Anti-symmetric perturbation of uniform
    anti = np.ones(n, dtype=np.float64)
    anti[:n // 2] += 0.3
    anti[n // 2:] -= 0.15
    anti = np.clip(anti, 0.01, None)
    inits.append(anti / anti.sum())

    for a0 in inits:
        if time.monotonic() > deadline - 3.0:
            break
        ba, bmb = optimize_from(a0, rng, deadline)
        if bmb < best_mb:
            best_mb = bmb
            best_a = ba.copy()

    # Scale back to original magnitude and validate
    result = best_a * psum
    result = np.clip(result, 0.0, 999.0)
    if result.sum() < 0.01:
        result = np.clip(parent, 0.0, 999.0)

    sc = float(evaluate_sequence(result.tolist()))
    if not np.isfinite(sc):
        result = np.clip(parent, 0.0, 999.0)
        sc = float(evaluate_sequence(result.tolist()))
        if not np.isfinite(sc):
            result = np.ones(64, dtype=np.float64)

    return [float(x) for x in result]
