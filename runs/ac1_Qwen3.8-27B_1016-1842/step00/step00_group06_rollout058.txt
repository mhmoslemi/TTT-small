def fast_score(a):
    n = a.size
    s = a.sum()
    if s < 1e-30:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)

def make_alt(n, ratio):
    a = np.ones(n, dtype=np.float64)
    a[1::2] = ratio
    return a

def make_mod(n, p):
    idx = np.arange(n)
    return ((idx % p).astype(np.float64))
