def _normalize(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _score_norm(a):
    n = len(a)
    conv = np.convolve(a, a)
    return 2.0 * n * float(conv.max())


def _subgrad_dir(a, k_max):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    if hi >= lo:
        g[lo:hi + 1] = a[k_max - hi: k_max - lo + 1][::-1] + a[lo:hi + 1]
    return g - g.mean()
