def _subgrad_of_maxlag(a, k_star):
    """Subgradient of C[k_star] wrt a, where C = conv(a,a)."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    idx = np.arange(lo, hi + 1)
    grad[idx] = 2.0 * a[k_star - idx]
    if k_star % 2 == 0:
        mid = k_star // 2
        if lo <= mid <= hi:
            grad[mid] = a[mid]
    return grad

def _run_subgrad_descent(a, n, rng, deadline, best, best_score, max_iters=20000):
    """Run subgradient descent from initialization a. Return best found."""
    S = a.sum()
    if S < 1e-10:
        return best, best_score
    a = a / S  # normalize to sum = 1
    
    eta = 1.0 / (n * np.sqrt(n))  # initial step size
    eta0 = eta
    
    for it in range(max_iters):

            break

        k_star = int(np.argmax(C))

        actual_score = 2.0 * n * M  # since S=1

            best = a.copy()

        # (dropping the factor 1/S^2 since S=1)
        grad = _subgrad_of_maxlag(a, k_star)

        a = a - eta * direction
        a = np.maximum(0.0, a)

        if S_new < 1e-10:
            a = np.ones(n)
            S_new = n
        a = a / S_new

        eta = eta0 / np.sqrt(it + 1)

        if it % 2000 == 1000:

            a = a + noise
            a = np.maximum(0.0, a)

    score = float(evaluate_sequence(best.tolist()))
