import time

def make_ramp(n):
    return (np.arange(1, n+1, dtype=np.float64)) / n

def make_super_ramp(n, p):
    return ((np.arange(1, n+1, dtype=np.float64)) / n) ** p

def make_parabola(n):
    i = np.arange(n, dtype=np.float64)
    return i * (n - 1 - i) / (0.5 * n) ** 2  # peak at center

def make_superposition(n, alpha):
    # Blend ramp and reversed ramp
    ramp = (np.arange(1, n+1, dtype=np.float64)) / n
    rev_ramp = 1.0 - (np.arange(n, dtype=np.float64)) / n
    return alpha * ramp + (1 - alpha) * rev_ramp

def block_perturb(a, rng, n_blocks=1):
    """Perturb random blocks of the sequence."""
    n = len(a)

    for _ in range(n_blocks):

        # Or add noise

    return np.clip(c, 0.0, 1000.0)
