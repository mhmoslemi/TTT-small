def _fft_conv(a):
    """Full convolution of a with itself using FFT. O(n log n)."""
    n = a.size
    N = 1
    size = 2 * n - 1
    while N < size:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    center2 = n * 0.7
    for w in [n*0.05, n*0.1, n*0.15]:
        g = np.exp(-0.5*((x-center1)/w)**2) + 0.8*np.exp(-0.5*((x-center2)/w)**2)
        if g.sum() > 0.01:
            candidates.append(g / g.sum())
    
    best = candidates[0].copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    for c in candidates[1:]:
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s
    
    # Subgradient descent
    lr = 1.0 / (n ** 0.5) * 5.0
    momentum = 0.7
    vel = np.zeros(n)
    stall = 0
    
    while time.monotonic() < deadline:
        b = _fft_conv(best)
        k = int(np.argmax(b))
        g = _subgradient(best, k)
        
        # Also include subgradients from near-max indices for smoothing
        thresh = b[k] * 0.99
        near = np.where(b >= thresh)[0]
        if len(near) > 1:
            g_avg = np.zeros(n)
            for kk in near[:20]:  # Cap to avoid O(n*n)
                lo = max(0, kk - n + 1)
                hi = min(n - 1, kk)
                if lo <= hi:
                    g_avg[lo:hi+1] += best[kk - lo : kk - hi - 1 : -1]
            g_avg /= len(near[:20])
            g = 0.5 * g + 0.5 * g_avg  # Blend
        
        # Momentum update
        vel = momentum * vel + lr * g
        candidate = _project_simplex(best - vel)
        
        # Evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            prev_score = best_score
            best = candidate
            best_score = score
            stall = 0
            # Nudge lr up slightly on improvement
            lr = min(lr * 1.01, 50.0 / n)
        else:
            stall += 1
            lr *= 0.95
            momentum = max(0.3, momentum * 0.99)
        
        if stall > 50:
            # Random restart perturbation
            idx = rng.integers(0, n)
            best[idx] *= 1.5
            best = _project_simplex(best)
            lr = 1.0 / (n ** 0.5) * 3.0
            stall = 0
            best_score = float(evaluate_sequence(best.tolist()))
        
        if lr < 1e-6:
            lr = 1.0 / (n ** 0.5)
            momentum = 0.7
            stall = 0
    
    return [float(v) for v in best]
