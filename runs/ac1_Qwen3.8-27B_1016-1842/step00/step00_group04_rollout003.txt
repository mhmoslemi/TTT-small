import numpy as np
import time


def evaluate_local(seq):
    """Thin wrapper around the authoritative evaluator with safety."""
    try:
        s = float(evaluate_sequence(seq.tolist()))
        return s if np.isfinite(s) else float('inf')
    except Exception:
        return float('inf')


def perturb_block(base, rng, width_frac=0.05, noise=0.05):
    """O(n) block multiplicative perturbation."""
    n = base.size
    w = max(2, int(n * width_frac))
    s = int(rng.integers(0, max(1, n - w)))
    cand = base.copy()
    cand[s:s + w] *= np.exp(rng.normal(0.0, noise, size=w))
    return np.clip(cand, 0.0, 1000.0)


def perturb_pair(base, rng):
    """Move mass between two random positions (keeps sum roughly constant)."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))

        return base.copy()

    cand[i] = max(0.0, cand[i] - amt)
    cand[j] = min(1000.0, cand[j] + amt)
    return cand
