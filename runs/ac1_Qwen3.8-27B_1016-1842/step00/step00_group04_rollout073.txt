import numpy as np
import time

def make_power(a_n, p):
    idx = np.arange(1, a_n + 1, dtype=np.float64)
    a = idx ** p
    mx = a.max()
    if mx > 0:
        a /= mx
    return a

def make_exp(a_n, rate):
    a = np.exp(rate * np.arange(a_n, dtype=np.float64) / float(a_n))
    mx = a.max()
    if mx > 0:
        a /= mx
    return a

def make_smooth_perturb(base, rng):
    n = base.size
    a = base.copy()
    # Add a smooth random perturbation (sum of random sinusoids)
    n_modes = 3

        freq = rng.integers(1, max(2, n // 4))

        amp = rng.uniform(-0.05, 0.05)
        a += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / float(n) + phase)
    return np.clip(a, 0.0, 1000.0)

import time

def gen_pwr(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    m = a[-1]
    return a / m if m > 0 else a

def gen_exp(n, r):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(r * x / max(1.0, float(n - 1)))
    m = a[-1]
    return a / m if m > 0 else a

def perturb_block(a, rng, wmin, wmax, sigma):
    n = a.size
    width = int(rng.integers(wmin, max(wmin + 1, wmax)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    a[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
    return np.clip(a, 0.0, 1000.0)

def perturb_points(a, rng, cnt, sigma):
    n = a.size
    ids = rng.integers(0, n, size=cnt)
    for i in ids:
        a[int(i)] = max(0.0, a[int(i)] * float(np.exp(rng.normal(0.0, sigma))))
    return np.clip(a, 0.0, 1000.0)
