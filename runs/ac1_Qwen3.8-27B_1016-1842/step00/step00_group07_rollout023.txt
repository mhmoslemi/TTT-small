import time

def fft_convolve_sq(a):
    """Compute a*a (linear convolution) via FFT."""
    n = len(a)
    N = 1
    while N < 2*n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)
    return C[:2*n-1]

def compute_score(a):
    """Compute the AC1 upper bound score."""
    n = len(a)
    if np.sum(a) < 0.01:
        return float('inf')
    c = fft_convolve_sq(a)
    max_c = np.max(c)
    return 2.0 * n * max_c / (np.sum(a) ** 2)

def gradient_peel(a, c, k_star, lr):
    """One gradient step to reduce c[k_star]."""
    n = len(a)
    # Gradient of c[k_star] w.r.t. a[j] is a[k_star - j] for valid j
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = a[idx]
    # Projected gradient step with sum constraint
    g = g - np.mean(g)  # zero-mean to maintain sum
    a_new = a - lr * g
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 0:
        a_new = a_new / a_new.sum()  # re-normalize
    else:
        a_new = a / 2.0
    return a_new
