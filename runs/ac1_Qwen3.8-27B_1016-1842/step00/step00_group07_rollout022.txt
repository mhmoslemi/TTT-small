For the peak at lag $k^*$, the contribution is $\sum_i a_i a_{k^*-i}$. To reduce this, we can:

import numpy as np
import time

def _autoconv(a):
    """Linear autocorrelation via FFT, O(n log n)."""
    n = a.size
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:m]

def _eval_score(a):
    """Score = 2n * max(a*a) / (sum(a))^2."""
    n = a.size
    b = _autoconv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def _subgrad(a, b, k):
    """Subgradient of (a*a)_k at a. O(n)."""
    n = a.size
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g
