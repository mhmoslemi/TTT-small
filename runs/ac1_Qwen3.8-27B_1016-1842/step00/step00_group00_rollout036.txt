def simplex_project(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Binary search for lambda
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rho = np.where(v_sorted > (cumsum - 1) / np.arange(1, n+1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cumsum[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

import numpy as np

def fft_conv(a):
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]


def quick_project(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v / s


def simplex_project(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    ar = np.arange(1, n + 1)
    cond = u > (css - 1.0) / ar
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (css[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def subgrad_at(a, C, n):
    M = float(C.max())
    k_star = int(C.argmax())
    g = np.zeros(n)

    return g, M
