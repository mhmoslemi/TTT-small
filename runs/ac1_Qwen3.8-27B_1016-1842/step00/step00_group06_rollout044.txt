def compute_score(a):
    """Compute the exact score for a nonnegative sequence."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2 * n * max_b / (sum_a ** 2))

    a = np.arange(1.0, n + 1.0)

        s = evaluate_sequence(init.tolist())
