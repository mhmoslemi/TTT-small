import time

def compute_conv_and_score(a):
    """Compute convolution, its max, and the score. O(n log n) or O(n^2)."""
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    M = conv[k_star]
    S = np.sum(a)
    score = 2.0 * len(a) * M / (S * S)
    return score, conv, k_star, M, S

def make_initial_shape(n, p=2.0):
    """Create a parametric initial shape: (j+1)^p normalized to sum 1."""
    j = np.arange(1, n+1, dtype=np.float64)
    f = j ** p
    f /= f.sum()
    return f

    n = len(a)
