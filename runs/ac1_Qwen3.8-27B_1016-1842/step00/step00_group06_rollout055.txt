```python
    n = len(f)
    f_new = f.copy()

    lo = max(0, k_star - (n-1))
    hi = min(n-1, k_star)
    for i in range(lo, hi + 1):
        j = k_star - i
        if i == j:

        if f_new[i] > f_new[j]:

            delta = alpha * (f_new[i] - f_new[j]) * 0.5
            f_new[i] -= delta
            f_new[j] += delta

            delta = alpha * (f_new[j] - f_new[i]) * 0.5
            f_new[j] -= delta
            f_new[i] += delta
```

```python
def project_simplex(v, S):
    """Project v onto {x >= 0 : sum(x) = S}."""
    n = len(v)
    # Sort v in descending order
    u_idx = np.argsort(v)[::-1]
    u = v[u_idx]
    # Find rho: largest index such that sum(u[:rho+1])/n - u[rho] > 0 ... 
    # Standard algorithm:
    # cumsum_j = sum_{i=1}^{j} u_i
    # rho = max{j : u_j > (sum_{i=1}^{j} u_i - S) / j}
    # theta = (sum_{i=1}^{rho} u_i - S) / rho
    cumsum = np.cumsum(u)
    rho = np.sum(np.maximum(u * np.arange(1, n+1) - (cumsum - S), 0) > 0)
    if rho == 0:
        return np.full(n, S / n)  # or just return clipped v
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0)
```
