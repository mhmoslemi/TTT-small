def build_near_sidon(n, target_size, max_mult=2, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    S = set()
    sum_counts = {}  # sum -> count of (unordered) pairs achieving it
    
    candidates = list(range(n))
    rng.shuffle(candidates)
    
    for x in candidates:
        if len(S) >= target_size:
            break
        # Check if adding x would violate max_mult
        new_sum_ok = True
        # New pairs: (x, x) and (x, s) for s in S
        # (x, x): sum 2x
        c_2x = sum_counts.get(2*x, 0) + 1
        if c_2x > max_mult:
            new_sum_ok = False
        # (x, s) for s in S, s != x
        if new_sum_ok:
            for s in S:
                if s == x:
                    continue
                c = sum_counts.get(x + s, 0) + 1
                if c > max_mult:
                    new_sum_ok = False
                    break
        if new_sum_ok:
            # Add x
            S.add(x)
            sum_counts[2*x] = sum_counts.get(2*x, 0) + 1
            for s in S:
                if s != x:
                    sum_counts[x + s] = sum_counts.get(x + s, 0) + 1

```python
def build_set(n, target_k, max_t, rng):
    """Build a subset of {0,...,n-1} of size >= target_k with max sum multiplicity <= max_t."""
    best_S = set()
    best_freq = {}
    
    for attempt in range(50):  # multiple restarts
        S = set()
        freq = {}  # sum -> count of unordered pairs
        order = list(range(n))
        rng.shuffle(order)
        
        for x in order:
            if len(S) >= target_k:
                break
            # Check feasibility
            ok = True
            # Check diagonal: 2x
            if freq.get(2*x, 0) + 1 > max_t:
                ok = False
            # Check off-diagonal: x+s for s in S
            if ok:
                for s in S:
                    if freq.get(x+s, 0) + 1 > max_t:
                        ok = False
                        break
            if ok:
                S.add(x)
                if 2*x in freq:
                    freq[2*x] += 1
                else:
                    freq[2*x] = 1
                for s in S:
                    if s != x:
                        if x+s in freq:
                            freq[x+s] += 1
                        else:
                            freq[x+s] = 1
        
        if len(S) > len(best_S):
            best_S = S
            best_freq = freq
    
    return best_S
```

This should work. With n=3809, target_k=150, max_t=2, the greedy should find 150 elements (or close). Multiple restarts help.

Time complexity: each attempt is O(n * k) = O(3809 * 150) ≈ 572K. With 50 attempts: 29M. In Python, this might take 30-60 seconds. That's acceptable.

After building the set:
1. Create the 0/1 array
2. Evaluate
3. Local search: try adding/removing/swapping elements

For local search, I can:
- Try adding a random element not in S (check if it improves score)
- Try removing a random element from S
- Try swapping (remove one, add another)
- Evaluate each candidate

Each evaluation is one `evaluate_sequence` call, which is O(n²) ≈ 10ms. In the remaining time (say 900 seconds), I can do ~90000 evaluations. That's plenty.
