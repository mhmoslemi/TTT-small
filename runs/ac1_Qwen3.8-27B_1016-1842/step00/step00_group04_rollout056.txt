import numpy as np
import time


def fast_convolve(a):
    """Linear autocorrelation a*a of length 2n-1 via FFT (near-linear in n)."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]


def project_simplex(v):
    """Project v onto {x >= 0, sum x == 1} (Euclidean)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def metric_of(a):
    """Search objective: max of (a*a) for sum-normalized a (== score / 2n)."""
    return float(fast_convolve(a).max())


def make_geom(n, r):
    g = np.power(r, np.arange(n, dtype=np.float64))
    g = g / g.sum()
    return g


def block_perturb(a, rng, sigma):
    n = a.size
    width = int(rng.integers(max(2, n // 200), max(3, n // 8)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    c = a.copy()
    c[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    if rng.random() < 0.3 and stop > 1:
        c[start:stop] = c[start:stop][::-1]
    return project_simplex(c)


def subgradient_step(a, conv, rng):
    """One descent direction for max_k (a*a)_k, refined by line search."""
    n = a.size
    M = conv.max()
    thresh = M * 0.999 - 1e-15 * M
    order = np.argsort(conv)[::-1]
    active = order[conv[order] >= thresh]
    active = active[:200]
    if active.size == 0:
        return a
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
    grad /= active.size
    gn = np.linalg.norm(grad)
    if gn < 1e-16:
        return a
    return a, grad / gn
