def _conv_full(a):
    """Full linear convolution via FFT."""
    n = len(a)

    while fs < size:
        fs <<= 1
    A = np.fft.rfft(a, fs)
    return np.fft.irfft(A * A, fs)[:size]

    """Project v onto {x >= 0, sum(x) = 1}. O(n log n)."""
