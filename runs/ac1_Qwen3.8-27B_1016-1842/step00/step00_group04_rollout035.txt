def _spectral_step(a, alpha, S):

    N = 1
    while N < 2 * n:
        N *= 2
    ap = np.zeros(N)

            if np.isfinite(s_s) and s_s < cur_score:
                a = a_s
            momentum[:] = 0.0

        # Periodic random kick
        if it % 5000 == 4999:
            a += 0.04 * S * rng.standard_normal(n)
            a = _normalize(a, S)
            lr = 1.5
            momentum[:] = 0.0

        # Anneal
        lr = max(0.15, lr * 0.9999)
        beta_m = min(0.95, 0.85 + 0.00001 * it)

    # Final verification
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        best = a0.copy()
        final_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
