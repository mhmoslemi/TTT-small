def fast_conv(a):
    """Self-convolution via FFT, O(n log n)."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score_fast(a):
    """Score matching evaluate_sequence."""
    n = a.size
    sa = a.sum()
    if sa < 0.01:
        return 1e30
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (sa * sa))

def gradient_direction(a, ks, n):
    """Descent direction for active lag ks."""
    S = a.sum()
    js = np.arange(max(0, ks - n + 1), min(n, ks + 1))
    bks = float(np.dot(a[js], a[ks - js]))
    lam = 2.0 * bks / S
    idx = ks - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    d = np.zeros(n)
    d[mask] = a[idx[mask]]
    d -= lam * a
    # Zero out mean to preserve sum
    d -= d.sum() / n
    return d

def score_fast(a):
    n = a.size
    sa = a.sum()
    if sa < 0.01:
        return 1e30
    b = np.convolve(a, a)
    return float(2.0 * n * b.max() / (sa * sa))


def grad_dir(a, ks, n):
    S = a.sum()
    lo = max(0, ks - n + 1)
    hi = min(n, ks + 1)
    js = np.arange(lo, hi)
    bks = float(np.dot(a[js], a[ks - js]))
    lam = 2.0 * bks / S
    idx = ks - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    d = np.zeros(n)
    d[mask] = a[idx[mask]]
    d -= lam * a
    d -= d.sum() / n
    return d
