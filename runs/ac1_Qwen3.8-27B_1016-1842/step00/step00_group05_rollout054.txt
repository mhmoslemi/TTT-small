import numpy as np
import time


def compute_gradient(a, n, S, M, k_star):
    """Compute gradient of 2*n*max_b/S^2 w.r.t. a at the maximizing lag."""
    grad = np.full(n, -4.0 * n * M / (S ** 3))
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 4.0 * n * (S * a[k_star - idx] - M) / (S ** 3)
    return grad


def gradient_descent(a, n, deadline, rng, best_state):
    """Run projected gradient descent; update best_state in place."""
    a = np.clip(a.astype(np.float64), 0.0, 1000.0).copy()
    a /= max(a.sum(), 1e-10)
    lr = 0.1
    stall = 0
    for _ in range(200000):
        if time.monotonic() > deadline:
            return a
        S = a.sum()
        if S < 0.01:
            a[:] = 1.0 / n
            continue
        b = np.convolve(a, a)
        M = float(b.max())
        k_star = int(np.argmax(b))
        score = 2.0 * n * M / (S * S)
        if score < best_state[0]:
            best_state[0] = score
            best_state[1] = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            a += 0.005 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
            a /= max(a.sum(), 1e-10)
            stall = 0
        grad = compute_gradient(a, n, S, M, k_star)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

            a = np.clip(a, 0.0, 1000.0)
