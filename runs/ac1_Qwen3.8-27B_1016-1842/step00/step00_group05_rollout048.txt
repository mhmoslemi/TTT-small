import numpy as np
import time

def compute_conv_fft(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def subgradient_score(a):
    """Compute score and subgradient."""
    n = len(a)
    conv = compute_conv_fft(a)
    max_b = conv.max()
    S = a.sum()
    k_star = conv.argmax()
    score = 2.0 * n * max_b / (S * S)
    # Subgradient
    grad = np.full(n, -2.0 * n * max_b / (S ** 3))
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    grad[valid] += 4.0 * n * a[idx[valid]] / (S * S)
    return score, grad, conv
