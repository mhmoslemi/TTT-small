    # A[k, j] = a[j] if max(0, k-n+1) <= j <= min(k, n-1)

    cols = []

def compute_gradient_at_argmax(a, m):
    """Compute subgradient of (a*a)_m w.r.t. a at the point a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo <= hi:
        # grad[lo:hi+1] = 2 * a[m - lo : m - hi + 1][::-1]
        paired = m - np.arange(lo, hi + 1)
        grad[lo:hi+1] = 2.0 * a[paired]
    return grad
