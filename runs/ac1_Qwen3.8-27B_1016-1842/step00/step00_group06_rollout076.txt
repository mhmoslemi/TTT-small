import numpy as np

def safe_evaluate(x):
    try:
        return float(evaluate_sequence(np.asarray(x).tolist()))
    except Exception:
        return np.inf


def normalize_nonneg(x):
    x = np.asarray(x, dtype=np.float64)
    if x.ndim != 1:
        x = x.reshape(-1)
    if x.size == 0:
        return np.ones(1, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1000.0)
    s = float(x.sum())
    if s <= 1e-12:
        return np.ones(x.size, dtype=np.float64) / x.size
    return x / s


def estimate_score(x):
    x = normalize_nonneg(x)
    n = int(x.size)
    if n == 0:
        return np.inf
    L = 2 * n - 1
    fpad = np.zeros(L, dtype=np.float64)
    fpad[:n] = x
    F = np.fft.rfft(fpad)
    c = np.fft.irfft(F * F, n=L)
    peak = float(np.max(c))
    s = float(np.sum(x))
    if s <= 0.0 or not np.isfinite(peak):
        return np.inf
    return 2.0 * float(n) * peak / (s * s)


def odd_opt(x, limit):
    x = int(x)
    if x < 1:
        x = 1
    lim = int(limit) if limit is not None else x
    if x >= lim:
        x = max(1, lim // 2)
    if x < 3:
        x = 3
    if x % 2 == 0:
        x += 1
    if x >= lim:
        x = max(1, lim - 1 if lim > 1 else 1)
        if x % 2 == 0:
            x -= 1
    return max(1, x)


def init_sine(n, m, b):

        if step < 1e-9:
            step = 1e-9
        fn = f - step * grad
        fn[fn < 0.0] = 0.0
        snew = float(fn.sum())
        if snew <= 1e-12 or not np.isfinite(snew):
            f = smooth_perturb(f, rng)
            no_improve = 0
            continue
        f = fn / snew
        if not np.all(np.isfinite(f)):
            f = best_f.copy()

    if use_exact:
        ex = safe_evaluate(best_f)
        if np.isfinite(ex) and ex < best_exact:
            best_exact = ex
    if np.isfinite(best_exact):
        return best_f, best_exact
    return best_f, float(best_est)


def make_small_init(n_full, rng, limit):
    m = 512
    mh = odd_opt(m // 60, m)
    f0 = init_sine(m, mh, 0.85)
    limit = max(float(limit), time.monotonic() + 0.5)
    f, _ = optimize_sequence(f0, rng, limit, use_exact=False)
    x = np.linspace(0.0, m - 1, int(n_full))
    return normalize_nonneg(np.interp(x, np.arange(m), f))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    budget = max(1.0, float(budget_s))
    headroom = min(45.0, max(5.0, 0.05 * budget))
    limit = start + max(1.0, budget - headroom)
    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None
    if parent is None or parent.size == 0:
        n = 5582
        parent = np.ones(n, dtype=np.float64)
    else:
        parent = normalize_nonneg(parent)
        n = int(parent.size)

    inits = []
    if n > 0:
        inits.append(parent)

        denoms = (260, 180, 130, 90)
        bs = (0.80, 0.85, 0.90, 0.78)
        for d, b in zip(denoms, bs):
            m = odd_opt(n // d, n)
            inits.append(init_sine(n, m, b))

        if budget > 240.0 and n > 1500:
            small_time = min(15.0, max(5.0, 0.02 * budget))
            small_limit = min(limit, start + small_time)
            try:
                inits.append(make_small_init(n, rng, small_limit))
            except Exception:
                pass

        inits.append(init_fourier(n, rng, terms=4))

    scored = []
    for idx, arr in enumerate(inits):
        arr = normalize_nonneg(arr)
        sc = estimate_score(arr)
        if np.isfinite(sc):
            scored.append((float(sc), idx, arr))
    if not scored:
        scored = [(2.0, 0, np.ones(max(1, n), dtype=np.float64) / max(1, n))]
    scored.sort()

    global_f = scored[0][2].copy()
    global_score = safe_evaluate(global_f)
    if not np.isfinite(global_score):
        global_f = normalize_nonneg(np.ones(max(1, n), dtype=np.float64))
        global_score = safe_evaluate(global_f)

    base_fracs = (0.28, 0.20, 0.16, 0.14, 0.12, 0.10, 0.10, 0.10)
    fracs = []
    for i in range(len(scored)):
        if i < len(base_fracs):
            fracs.append(float(base_fracs[i]))
        else:
            fracs.append(0.10)
    sf = float(sum(fracs))
    if sf <= 0.0:
        fracs = [1.0 / max(1, len(scored))]
    else:
        fracs = [v / sf for v in fracs]

    elapsed = time.monotonic() - start
    work_total = max(10.0, (limit - start) - elapsed)

    allocated = start
    for pos, item in enumerate(scored):
        if not np.isfinite(item[0]):
            continue
        frac = fracs[pos] if pos < len(fracs) else 0.10 / max(1, len(fracs))
        allocated = min(limit, allocated + frac * work_total)
        if allocated <= time.monotonic() + 2.0:
            continue
        try:
            bf, bs = optimize_sequence(item[2], rng, allocated, use_exact=True)
        except Exception:
            bf, bs = item[2], item[0]
        if np.isfinite(bs) and bs < global_score:
            global_score = bs
            global_f = normalize_nonneg(bf)
        if limit - time.monotonic() < 30.0:
            break

    final_f = normalize_nonneg(global_f)
    final_score = safe_evaluate(final_f)
    if not np.isfinite(final_score):
        fallback = parent if parent is not None and parent.size > 0 else np.ones(max(1, n), dtype=np.float64)
        final_f = normalize_nonneg(fallback)
        final_score = safe_evaluate(final_f)
    if not np.isfinite(final_score):
        final_f = normalize_nonneg(np.ones(max(1, n), dtype=np.float64))
        final_score = safe_evaluate(final_f)

    final_f = np.clip(np.asarray(final_f, dtype=np.float64), 0.0, 1000.0)
    if final_f.size == 0:
        final_f = np.ones(1, dtype=np.float64)
    return [float(v) for v in final_f]
