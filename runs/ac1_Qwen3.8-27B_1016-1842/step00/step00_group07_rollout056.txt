from scipy.signal import fftconvolve, correlate


def _correlate_valid(w, a):
    """Compute correlate(w, a, 'valid') for w length 2n-1, a length n -> length n."""
    return correlate(w, a, mode='valid')


def _project(a, S):
    """Project a onto a>=0, sum(a)=S."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a *= S / s
    return a
