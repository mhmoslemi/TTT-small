import time

def compute_gradient(a, m):
    """Compute gradient of b_m w.r.t. a, where m is the argmax lag."""
    n = len(a)
    # (grad b_m)_i = 2 * a[m-i] if 0 <= m-i < n, else 0
    grad = np.zeros(n)
    # m-i is in [0, n-1] means i is in [m-n+1, m]
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    grad[lo:hi+1] = 2 * a[m - np.arange(lo, hi+1)]
    return grad

import numpy as np
import time
from scipy.signal import fftconvolve


def _local_score(x, n):
    sx = x.sum()
    if sx < 1e-15:
        return np.inf
    bx = fftconvolve(x, x)
    return float(2.0 * n * bx.max() / (sx * sx))


def _soft_autocorr_grad(a, b, n, top_k=40):
    bmean = b.sum() / b.size
    excess = np.maximum(b - bmean, 0.0)
    s = excess.sum()
    if s < 1e-30:
        m = int(np.argmax(b))
        grad = np.zeros(n)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        grad[lo:hi + 1] = 2.0 * a[m - np.arange(lo, hi + 1)]
        return grad
    if excess.size > top_k:
        idx = np.argpartition(-excess, top_k)[:top_k]
    else:
        idx = np.arange(excess.size)
    grad = np.zeros(n)
    for mi in idx:
        mi = int(mi)
        w = excess[mi] / s
        if w < 1e-14:
            continue
        lo = max(0, mi - n + 1)
        hi = min(n - 1, mi)
        seg = mi - np.arange(lo, hi + 1)
        grad[lo:hi + 1] += 2.0 * w * a[seg]
    return grad


def _gd_from(x, n, eta0, deadline, rng):
    a = x.copy()
    best = a.copy()
    bs = _local_score(a, n)
    eta = eta0
    stall = 0
    while time.monotonic() < deadline:
        b = fftconvolve(a, a)
        g = _soft_autocorr_grad(a, b, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        for _ in range(12):
            cand = a - eta * g
            cand = np.maximum(cand, 0.0)
            cs = cand.sum()
            if cs < 1e-12:
                eta *= 0.5
                continue
            cand *= np.sqrt(2.0 * n) / cs
            sc = _local_score(cand, n)
            if np.isfinite(sc) and sc < bs:
                best = cand.copy()
                bs = sc
                a = cand
                improved = True
                break
            eta *= 0.5
        if improved:
            stall = 0
            eta = min(eta * 1.1, 5.0)
        else:
            stall += 1
            eta = eta0
            if stall > 40:
                sig = 0.8 / np.sqrt(n)
                a = best * np.exp(rng.normal(0.0, sig, n))
                a = np.maximum(a, 0.0)
                a = a / a.sum() * np.sqrt(2.0 * n)
                stall = 0
    return best, bs
