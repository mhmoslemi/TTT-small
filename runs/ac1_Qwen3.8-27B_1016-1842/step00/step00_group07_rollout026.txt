def autocorr(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    m = 1 << (2*n - 2).bit_length()  # next power of 2 >= 2n-1
    A = np.fft.rfft(a, m)
    return np.fft.irfft(A * A, m)[:2*n-1]

    best_score = np.inf
    for init in inits:
        sc = evaluate_sequence(init.tolist())
        if sc < best_score:
            best_score = sc
            best = init.copy()

    target_sum = np.sqrt(2.0 * n)
    a = best.copy()
    stagnation = 0
    
    while time.monotonic() < deadline - 10:

        peak_k = int(np.argmax(b))

        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:

        improved = False

            a_try = normalize(a_try, target_sum)
            b_try = autocorr(a_try)

                a = a_try
                improved = True
                break

            if sc < best_score:

                best = a.copy()
            stagnation = 0
        else:

            # Random perturbation every few failures

                    best_score = sc
                    best = a_try.copy()
                    a = a_try
                    stagnation = 0

        if stagnation > 20:
            break
    
    return [float(x) for x in best]

import numpy as np
import time


def autocorr(a):
    """Linear autocorrelation via FFT. O(n log n)."""
    n = len(a)
    m = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, m)
    return np.fft.irfft(A * A, m)[:2 * n - 1]


def normalize(a, target_sum):
    """Clip to nonnegative and scale to target sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a)
        s = float(a.size)
    a = a * (target_sum / s)
    return np.minimum(a, 1000.0)


def grad_peak(a, k):
    """Gradient of b_k = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def block_perturb(a, rng, n_blocks=3, scale=0.07):
    """Multiplicative perturbation on random blocks."""
    n = len(a)
    c = a.copy()
    for _ in range(n_blocks):
        w = int(rng.integers(1, max(2, n // 25)))
        st = int(rng.integers(0, n - w))
        c[st:st + w] *= np.exp(rng.normal(0.0, scale))
    ts = np.sqrt(2.0 * n)
    return normalize(c, ts)


def mirror_perturb(a, rng, scale=0.05):

    half = n // 2
    c = a.copy()

    st = int(rng.integers(0, max(1, half - w + 1)))
    m = np.exp(rng.normal(0.0, scale))
    c[st:st + w] *= m
    c[n - st - w:n - st] *= m
    ts = np.sqrt(2.0 * n)
    return normalize(c, ts)

    else:
        a = 1.0 + 0.35 * np.abs(t)
    ts = np.sqrt(2.0 * n)
    return normalize(a, ts)
