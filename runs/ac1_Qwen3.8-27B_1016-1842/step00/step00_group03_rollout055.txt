        if np.isfinite(score) and score < best_loop_score:
            best_loop_score = score
            best_loop = a.copy()

        # Subgradient of M = conv[m] w.r.t. a
        valid = (idx <= m) & (m - idx < n)
        dM = np.zeros(n)
        j = m - idx[valid]
        dM[valid] = 2.0 * a[j]
        if m % 2 == 0:
            km = m // 2
            if 0 <= km < n:
                dM[km] = a[km]

        # Gradient of f = 2n*M/S^2
        S3 = S * S * S
        grad = 2.0 * n * (S * dM - 2.0 * M) / S3

        # Momentum update
        momentum = beta * momentum + grad
        a = a - lr * momentum
        a = np.clip(a, 0.0, 1000.0)

        # Renormalize (scale-invariance of objective)
        mx = np.max(a)
        if mx > 0:
            a = 100.0 * a / mx
        else:
            break

        # Learning-rate schedule and occasional perturbation
        if it % 2000 == 0 and it > 0:
            lr *= 0.95
        if it % 15000 == 0 and it > 0:
            a = a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            mx = np.max(a)
            if mx > 0:
                a = 100.0 * a / mx

    # Accept loop result if better
    s_loop = float(evaluate_sequence(best_loop.tolist()))
    if np.isfinite(s_loop) and s_loop < best_score:
        best_score = s_loop
        best = best_loop.copy()

    # Final validation
    final = np.clip(best, 0.0, 1000.0)
    s_final = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(s_final) or s_final > best_score:
        final = parent
    s_check = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(s_check):
        final = np.ones(n, dtype=np.float64)

    return [float(x) for x in final]
