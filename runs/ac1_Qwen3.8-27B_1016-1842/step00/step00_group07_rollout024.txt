def make_shape_power(n, p, c):
    """a[i] = (1 + c*i/(n-1))^p, then normalize sum=1."""
    x = np.arange(n, dtype=np.float64)
    base = 1.0 + c * (x / max(1, n - 1))
    seq = np.power(np.maximum(base, 1e-10), p)
    s = seq.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64)
    return seq / s


def make_shape_exp(n, c):
    """a[i] = exp(c * i/(n-1)), normalized."""
    x = np.arange(n, dtype=np.float64)
    seq = np.exp(c * x / max(1, n - 1))
    s = seq.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64)
    return seq / s


def make_shape_quad(n, a, b):
    """a[i] = 1 + a*t + b*t^2 where t = 2*i/(n-1) - 1, normalized."""
    t = 2.0 * np.arange(n, dtype=np.float64) / max(1, n - 1) - 1.0
    seq = 1.0 + a * t + b * t * t
    seq = np.maximum(seq, 1e-10)
    s = seq.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64)
    return seq / s


def subgradient_step(a, step):
    """One projected subgradient step to reduce max(autoconv) with sum(a)=1."""
    n = len(a)
    conv = fft_autoconv(a)
    M = conv.max()
    if M < 1e-30:
        return a, 0.0
    # Use top peaks for stability
    npeaks = min(7, 2 * n - 1)
    peaks = np.argsort(conv)[-npeaks:]
    grad = np.zeros(n, dtype=np.float64)
    for k in peaks:
        j_idx = k - np.arange(n)
        valid = (j_idx >= 0) & (j_idx < n)
        g = np.zeros(n, dtype=np.float64)
        g[valid] = 2.0 * a[j_idx[valid]]
        grad += g
    grad /= npeaks
    # Center the gradient to approximately preserve sum
    grad -= grad.mean()
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, M
    return a_new / s, M


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1: Parametric shape search (fast FFT evaluation)
    candidates = []
    # Power-law shapes
    for p in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        for c in [0.5, 1.0, 2.0, 4.0, 8.0]:
            if c > 20:
                break
            candidates.append(make_shape_power(n, p, c))
            candidates.append(make_shape_power(n, p, -c))
    # Exponential shapes
    for c in [0.5, 1.0, 2.0, 3.0, 5.0, 8.0]:
        candidates.append(make_shape_exp(n, c))
        candidates.append(make_shape_exp(n, -c))
    # Quadratic shapes
    for a in [-0.8, -0.5, -0.3, 0.3, 0.5, 0.8]:
        for b in [-0.5, -0.2, 0.0, 0.2, 0.5]:
            candidates.append(make_shape_quad(n, a, b))

    for shape in candidates:
        if time.monotonic() > deadline - 300:
            break
        conv = fft_autoconv(shape)
        score_est = 2.0 * n * conv.max()
        if score_est < best_score * 1.01:
            score = evaluate_sequence(shape.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = shape.copy()

    # Phase 2: Subgradient refinement
    a = best.copy()
    if a.sum() > 0:
        a = a / a.sum()
    step = 0.002
    stall = 0
    prev_M = fft_autoconv(a).max()

    while time.monotonic() < deadline - 200:
        a, cur_M = subgradient_step(a, step)
        if cur_M < prev_M * (1 - 1e-10):
            stall = 0
            step = min(step * 1.02, 0.05)
        else:
            stall += 1
            step *= 0.93
            if stall > 300:
                break
        prev_M = cur_M
        # Periodic authoritative check
        if stall % 50 == 0:
            sc = evaluate_sequence(a.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()

    # Phase 3: Local perturbation search with authoritative evaluator
    scale = max(1.0, best.sum())
    while time.monotonic() < deadline - 3:
        cand = best.copy()
        width = max(1, int(n * rng.uniform(0.0005, 0.015)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        cand[start:stop] *= np.exp(rng.normal(0.0, 0.04))
        if rng.random() < 0.2:
            width2 = max(1, int(n * rng.uniform(0.001, 0.005)))
            start2 = int(rng.integers(0, max(1, n - width2)))
            cand[start2:start2 + width2] *= np.exp(rng.normal(0.0, 0.08))
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand.copy()

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = parent
    return [float(x) for x in best]
