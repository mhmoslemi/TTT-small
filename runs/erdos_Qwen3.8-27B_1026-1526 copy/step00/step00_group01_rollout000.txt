    return np.max(corr)

def project_to_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0, 1)
    # Iteratively adjust to satisfy sum constraint
    for _ in range(100):
        s = h.sum()
        target = n / 2.0
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Add diff to elements that have room
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            room = h
            mask = room > 1e-10
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    # Final exact adjustment
    h = np.clip(h, 0, 1)
    diff = n / 2.0 - h.sum()
    if abs(diff) > 1e-10:

        if diff > 0:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmin(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def smooth_c5_obj(h, n, tau, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if tau >= 1000:
        obj = m
    else:
        shifted = (corr - m) * tau
        shifted = np.clip(shifted, -500, 500)
        lse = m * tau + np.log(np.sum(np.exp(shifted)))
        obj = lse / tau
    penalty = lam * (h.sum() - n / 2.0) ** 2
    return obj + penalty


def grad_smooth_c5_obj(h, n, tau, lam):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    m = np.max(corr)
    if tau >= 1000:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    else:
        shifted = (corr - m) * tau
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        w = weights[k] * tau
        if w < 1e-15:
            continue
        if k < n:
            lo = 0
            hi = min(n - k, n)
            for i in range(lo, hi):
                grad[i + k] += w * (1.0 - h[i])
                grad[i] -= w * h[i + k]
        else:
            shift = k - (n - 1)
            lo = max(0, -shift)
            hi = min(n, n - shift)
            for i in range(lo, hi):
                grad[i] += w * (1.0 - h[i + shift])
                grad[i + shift] -= w * h[i]
    grad = grad * dx
    penalty_grad = 2.0 * lam * (h.sum() - n / 2.0)
    return grad + penalty_grad


def perturb_and_eval(h, n, rng, step_size):
    h_pert = h + step_size * rng.randn(n)
    h_pert = project_feasible(h_pert, n)
    return h_pert
