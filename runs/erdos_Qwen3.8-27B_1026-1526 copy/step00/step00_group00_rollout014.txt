def corr_grad_k(h, k):
    """Subgradient of corr[k] w.r.t. h. k is 0..2n-2."""
    n = len(h)
    m = k - n + 1  # actual shift
    grad = np.zeros(n)

        b = j - m
        if 0 <= b < n:
            grad[j] -= h[b]

def smooth_obj_and_grad(h, n, t):
    h = np.clip(h, 0, 1)
    corr = get_corr(h)
    m = np.max(corr)
    # log-sum-exp
    exp_vals = np.exp((corr - m) / t)
    log_sum = m + t * np.log(np.sum(exp_vals))
    obj = log_sum * 2.0 / n
    
    # Gradient of log-sum-exp
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w > 1e-15:
            grad += (w / t) * corr_grad_k(h, k)
    
    # Scale by dx
    grad *= 2.0 / n
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

_opt = {}

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def get_c5(h, n):
    return float(np.max(get_corr(h)) * 2.0 / n)

def corr_grad_k(h, k):
    n = len(h)
    mv = k - n + 1
    grad = np.zeros(n)
    for j in range(n):
        a = j + mv
        if 0 <= a < n:
            grad[j] += 1.0 - h[a]
        b = j - mv
        if 0 <= b < n:
            grad[j] -= h[b]
    return grad

def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_obj_val(h, n, t):
    h = np.clip(h, 0.0, 1.0)
    corr = get_corr(h)
    m = np.max(corr)
    ev = np.exp((corr - m) / t)
    s = np.sum(ev)
    if s < 1e-300:
        s = 1e-300
    return (m + t * np.log(s)) * 2.0 / n

def smooth_grad_val(h, n, t):
    h = np.clip(h, 0.0, 1.0)
    corr = get_corr(h)
    m = np.max(corr)
    ev = np.exp((corr - m) / t)
    s = np.sum(ev)
    if s < 1e-300:
        s = 1e-300
    w = ev / s
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-14:
            grad += w[k] * corr_grad_k(h, k)
    return grad * 2.0 / n

def _obj_fn(x):
    return smooth_obj_val(x, _opt['n'], _opt['t'])

def _jac_fn(x):
    return smooth_grad_val(x, _opt['n'], _opt['t'])

def _con_fun(x):

    _opt = {'n': n, 't': t, 'target': target}
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_fn, h0, jac=_jac_fn, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if hasattr(res, 'x') and res.x is not None:
            return project_feasible(res.x, target)
    except Exception:
        pass
    return h0

    lr = 0.05
    best = get_c5(h, n)
    best_h = h.copy()

    while time.time() < t_end:
        corr = get_corr(h)
        k_star = int(np.argmax(corr))
        g = corr_grad_k(h, k_star)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h = h - lr * g

            lr = min(lr * 1.01, 0.25)
        else:
            no_imp += 1
            if no_imp > 300:
                lr *= 0.7
                no_imp = 0
    return best_h, best
