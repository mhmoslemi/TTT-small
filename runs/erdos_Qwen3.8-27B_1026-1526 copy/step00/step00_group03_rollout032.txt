        for i in range(n):
            old_val = h[i]
            candidates = [0.0, 1.0, 0.5]
            for delta in [0.3, 0.15, 0.05, 0.01, 0.003]:
                lo = max(0.0, old_val - delta)
                hi = min(1.0, old_val + delta)
                candidates.append(lo)
                candidates.append(hi)
            best_c5_local = c5
            best_snapshot = h.copy()
            for c in candidates:
                if abs(c - old_val) < 1e-12:
                    continue
                h[i] = c
                h = repair(h, target_sum)
                new_c5 = compute_c5(h, n)
                if new_c5 < best_c5_local - 1e-14:
                    best_c5_local = new_c5
                    best_snapshot = h.copy()
            h = best_snapshot
            if best_c5_local < c5 - 1e-14:
                c5 = best_c5_local
                improved = True
        if not improved:
            break
    return h, c5


def binary_local_search(h, n, time_limit=None, start_time=None, seed=0):
    rng = np.random.default_rng(seed)
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    for it in range(500000):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[int(rng.integers(len(ones)))])
        j = int(zeros[int(rng.integers(len(zeros)))])
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


def sim_anneal_binary(h, n, time_limit=None, start_time=None, seed=0):
    rng = np.random.default_rng(seed)
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    temp = 0.004
    while True:
        if time_limit is not None and start_time is not None:
            remaining = time_limit - (time.time() - start_time)
            if remaining < 0.05:
                break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[int(rng.integers(len(ones)))])
        j = int(zeros[int(rng.integers(len(zeros)))])
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - c5
        if delta < 0 or (temp > 1e-6 and rng.random() < np.exp(-delta / temp)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        temp = max(1e-5, temp * 0.99998)
    return best_h, best_c5


def fix_binary_count(h, target_ones):
    n = len(h)
    n_ones = int(np.sum(h > 0.5))
    if n_ones == target_ones:
        return h
    if n_ones > target_ones:
        oi = np.where(h > 0.5)[0]
        rm = np.random.default_rng(12345).choice(oi, size=n_ones - target_ones, replace=False)
        h[rm] = 0.0
    else:
        zi = np.where(h <= 0.5)[0]
        ad = np.random.default_rng(12345).choice(zi, size=target_ones - n_ones, replace=False)
        h[ad] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = 1.0
    best_n = 70

    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 70

    if h0 is not None and n0 > 10:
        n = n0
        target_sum = n / 2.0
        h = repair(h0.copy(), target_sum)
    else:
        n = 70
        target_sum = 35.0
        h = np.full(n, 0.5)

    # Phase 1: Coordinate descent from initial (30% budget)
    t_end1 = start_time + budget_s * 0.30
    h, c5 = coordinate_descent(h, n, target_sum, max_sweeps=10000, time_limit=t_end1, start_time=start_time)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

    # Phase 2: Binary local search (15% budget)
    t_end2 = start_time + budget_s * 0.45
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = fix_binary_count(h_bin, int(target_sum))
    h_bin, c5_bin = binary_local_search(h_bin, n, time_limit=t_end2, start_time=start_time, seed=seed + 1)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
        best_n = n

    # Short CD from best binary
    t_end2b = start_time + budget_s * 0.55
    h_cd = h_bin.copy()
    h_cd, c5_cd = coordinate_descent(h_cd, n, target_sum, max_sweeps=3000, time_limit=t_end2b, start_time=start_time)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
        best_n = n

    # Phase 3: Simulated annealing (20% budget)
    t_end3 = start_time + budget_s * 0.75
    h_sa = (best_h > 0.5).astype(np.float64)
    h_sa = fix_binary_count(h_sa, int(best_n / 2.0))
    h_sa, c5_sa = sim_anneal_binary(h_sa, best_n, time_limit=t_end3, start_time=start_time, seed=seed + 2)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()
        best_n = best_n

    # Short CD from SA result
    t_end3b = start_time + budget_s * 0.80
    h_cd2 = h_sa.copy()
    h_cd2, c5_cd2 = coordinate_descent(h_cd2, best_n, best_n / 2.0, max_sweeps=2000, time_limit=t_end3b, start_time=start_time)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 4: Multi-start binary with different seeds (10% budget)
    t_end4 = start_time + budget_s * 0.90
    for s in range(5):
        if time.time() > t_end4:
            break
        h_rand = np.zeros(n)
        positions = rng.choice(n, size=int(target_sum), replace=False)
        h_rand[positions] = 1.0
        h_rand, c5_r = binary_local_search(h_rand, n,
                                            time_limit=min(time.time(), t_end4 - (time.time() - start_time + (t_end4 - start_time)) * 0),
                                            start_time=start_time, seed=seed + 100 + s * 17)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_rand.copy()
            best_n = n

    # Phase 5: Higher resolution attempts (5% budget)
    t_end5 = start_time + budget_s * 0.95
    for n_try in [100, 140]:
        if time.time() > t_end5:
            break
        n_t = n_try
        ts_t = n_t / 2.0
        if best_h is not None and len(best_h) != n_t:
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_t, endpoint=False)
            h_init = np.interp(x_new, x_old, best_h)
        else:
            h_init = np.full(n_t, 0.5)
        h_b = np.where(h_init > 0.5, 1.0, 0.0)
        h_b = fix_binary_count(h_b, int(ts_t))
        h_b, c5_t = binary_local_search(h_b, n_t, time_limit=t_end5, start_time=start_time, seed=seed + 500 + n_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_b.copy()
            best_n = n_t
        if time.time() < t_end5:
            h_t = h_b.copy()
            h_t, c5_t2 = coordinate_descent(h_t, n_t, ts_t, max_sweeps=1000, time_limit=t_end5, start_time=start_time)
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t.copy()
                best_n = n_t

    # Final polish
    if time.time() < start_time + budget_s - 5 and best_h is not None:
        n_f = best_n
        ts_f = n_f / 2.0
        h_f = repair(best_h.copy(), ts_f)
        h_f, c5_f = coordinate_descent(h_f, n_f, ts_f, max_sweeps=2000,
                                       time_limit=start_time + budget_s - 2, start_time=start_time)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_f

    final_h = repair(best_h, best_n / 2.0)
    final_c5 = compute_c5(final_h, best_n)
    return (final_h, final_c5, best_n)
