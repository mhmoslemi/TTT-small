    """Project v onto {x : 0 ≤ x_i ≤ 1, sum(x) = target}."""
    n = len(v)
    # Find τ such that sum(clamp(v_i - τ, 0, 1)) = target
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
