            if global_stuck > 20000:

                h = project_feasible(h, target_sum, n)
                velocity = np.zeros(n)
                lr = lr_init
                global_stuck = 0
                stuck = 0

    return best_h, best_c5

def lbfgs_phase(h_current, n, dx, target_sum, duration, rng, n_tries=40):
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    start = time.time()
    bounds = [(0.0, 1.0)] * n

    for attempt in range(n_tries):
        if time.time() - start > duration:
            break
        scale = 0.04 * np.exp(-0.08 * attempt)
        h_s = best_h + rng.normal(0, scale, n)
        h_s = project_feasible(h_s, target_sum, n)

        try:
            res = minimize(
                obj_scipy,
                h_s,
                args=(n, dx, target_sum),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-9}
            )
            h_opt = project_feasible(res.x, target_sum, n)
            c5_val = compute_c5(h_opt, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        except Exception:
            pass

    return best_h, best_c5

def fine_subgrad_phase(h_current, n, dx, target_sum, duration, rng):
    h = h_current.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start = time.time()
    lr = 0.002
    stuck = 0

    while time.time() - start < duration:
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = int(np.argmax(corr))
        grad = grad_at_lag(h, top_idx, n)

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum, n)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 2000:
                lr *= 0.5
                stuck = 0
                if lr < 1e-10:
                    break
            elif stuck > 500:
                h = best_h + rng.normal(0, 0.01, n)
                h = project_feasible(h, target_sum, n)
                stuck = 0

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = np.ones(70) * 0.5

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    h_init = project_feasible(h_init, target_sum, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)

    time_limit = budget_s - 20

    remaining = time_limit - (time.time() - start_time)
    if remaining > 10:
        t1 = remaining * 0.40
        best_h, best_c5 = subgradient_phase(best_h, n, dx, target_sum, 0.025, t1, rng, top_m=5, momentum=0.85)

    remaining = time_limit - (time.time() - start_time)
    if remaining > 10:
        t2 = remaining * 0.40
        best_h, best_c5 = lbfgs_phase(best_h, n, dx, target_sum, t2, rng, n_tries=50)

    remaining = time_limit - (time.time() - start_time)
    if remaining > 10:
        t3 = remaining * 0.45
        best_h, best_c5 = subgradient_phase(best_h, n, dx, target_sum, 0.008, t3, rng, top_m=3, momentum=0.8)

    remaining = time_limit - (time.time() - start_time)
    if remaining > 5:
        best_h, best_c5 = lbfgs_phase(best_h, n, dx, target_sum, remaining * 0.6, rng, n_tries=30)

    remaining = time_limit - (time.time() - start_time)
    if remaining > 3:
        best_h, best_c5 = fine_subgrad_phase(best_h, n, dx, target_sum, remaining * 0.8, rng)

    best_h = project_feasible(best_h, target_sum, n)
    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n
