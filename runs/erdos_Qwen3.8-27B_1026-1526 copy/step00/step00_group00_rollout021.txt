import numpy as np
import time
from scipy.optimize import minimize
import math

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def repair_sum(h, target):
    h = np.clip(h, 0, 1).astype(float).copy()
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n_m = int(np.sum(mask))
        if n_m == 0:
            break
        h[mask] += diff / n_m
        h = np.clip(h, 0, 1)
    return h

def smoothed_c5(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    if max_c < -1e-15:
        return dx * max_c
    shifted = T * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    return dx * (max_c + np.log(np.sum(np.exp(shifted))) / T)

def optimize_at_n(h_start, n, max_iter=1000):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_best = h_start.copy()
    c5_best = compute_c5(h_best, n)
    
    def obj(x):
        return compute_c5(x, n)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    
    try:
        res = minimize(obj, h_best, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        if res.x is not None:
            h_try = repair_sum(res.x, target)
            c5_try = compute_c5(h_try, n)
            if c5_try < c5_best:
                h_best = h_try
                c5_best = c5_try
    except Exception:
        pass
    
    return h_best, c5_best

def anneal_optimize(h_start, n, t_remaining):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    c5_cur = compute_c5(h_cur, n)
    T_schedule = [5, 20, 100, 500, 2000]
    
    for T in T_schedule:
        if t_remaining() < 15:
            break
        def obj_smooth(x, T_val=T):
            return smoothed_c5(x, n, T_val)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
        try:
            res = minimize(obj_smooth, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-10})
            if res.x is not None:
                h_try = repair_sum(res.x, target)
                c5_try = compute_c5(h_try, n)
                if c5_try < c5_cur:
                    h_cur = h_try
                    c5_cur = c5_try
        except Exception:
            pass
    
    return h_cur, c5_cur

def penalty_optimize(h_start, n, max_iter=2000):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_best = h_start.copy()
    c5_best = compute_c5(h_best, n)
    
    for penalty in [10, 100, 1000]:
        def obj_pen(x):
            return compute_c5(x, n) + penalty * (np.sum(x) - target) ** 2
        try:
            res = minimize(obj_pen, h_best, method='L-BFGS-B', bounds=bounds,

            if res.x is not None:
                h_try = repair_sum(res.x, target)
                c5_try = compute_c5(h_try, n)
                if c5_try < c5_best:
                    h_best = h_try
                    c5_best = c5_try
        except Exception:
            pass
    
    return h_best, c5_best
