        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo, hi = -2.0, 2.0

        else:

    if k >= 0:

    return obj, grad

    return best_h, best_c5


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    return project_onto_set(h_new, n_new / 2.0)


def random_feasible(n, rng):
    h = rng.uniform(0, 1, n)
    return project_onto_set(h, n / 2.0)


def local_refine(h, n, time_limit, start_time):
    S = n / 2.0
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for it in range(200):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = np.argmax(corr)
        grad = compute_grad_component(h, k_star, n)
        step = 0.5 / n
        for trial in range(5):
            h_try = project_onto_set(h - step * grad, S)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                break
            step *= 0.5
        else:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.ones(45) * 0.5

    n0 = len(h0)
    best_h = project_onto_set(h0, n0 / 2.0)
    best_c5 = evaluate_c5(best_h)

    def elapsed():
        return time.time() - start_time

    # Phase 1: SLSQP with smoothing on initial resolution
    if elapsed() < budget_s * 0.12:
        h1, c5_1 = slsqp_smooth(best_h, n0, time_limit=budget_s * 0.15, start_time=start_time)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Subgradient + local refine
    if elapsed() < budget_s * 0.3:
        h2, c5_2 = subgradient_optimize(best_h, n0, max_iters=20000,
                                        time_limit=budget_s * 0.25, start_time=start_time)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    if elapsed() < budget_s * 0.35:
        h3, c5_3 = local_refine(best_h, n0, time_limit=budget_s * 0.35, start_time=start_time)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 3: Try higher resolutions
    n_candidates = [60, 80, 100, 125, 150, 200]
    phase3_start = budget_s * 0.35
    phase3_end = budget_s * 0.9
    per_n = max(1.0, (phase3_end - phase3_start) / len(n_candidates))

    for n_new in n_candidates:
        if elapsed() > phase3_end:
            break
        h_up = upsample(best_h, n_new)
        c5_up = evaluate_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

        t_limit = min(phase3_end, start_time + elapsed() + per_n * 0.5)
        h_opt, c5_opt = slsqp_smooth(best_h, n_new, time_limit=t_limit, start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if elapsed() < phase3_end - 5:
            h_sub, c5_sub = subgradient_optimize(best_h, n_new, max_iters=30000,
                                                 time_limit=phase3_end - 5, start_time=start_time)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()

        if elapsed() < phase3_end - 2:
            h_lr, c5_lr = local_refine(best_h, n_new, time_limit=phase3_end - 2, start_time=start_time)
            if c5_lr < best_c5:
                best_c5 = c5_lr
                best_h = h_lr.copy()

    # Phase 4: Random restarts if time remains
    if elapsed() < budget_s * 0.85:
        n_try = max(n0, 60)
        t_limit = budget_s * 0.85
        for restart in range(3):
            if elapsed() > t_limit:
                break
            h_rand = random_feasible(n_try, rng)
            h_r, c5_r = slsqp_smooth(h_rand, n_try, time_limit=t_limit, start_time=start_time)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            h_r2, c5_r2 = subgradient_optimize(h_r, n_try, max_iters=10000,
                                               time_limit=t_limit, start_time=start_time)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

    # Final polish
    if elapsed() < budget_s * 0.97:
        h_final, c5_final = local_refine(best_h, len(best_h), time_limit=budget_s * 0.97, start_time=start_time)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_onto_set(best_h, len(best_h) / 2.0)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
