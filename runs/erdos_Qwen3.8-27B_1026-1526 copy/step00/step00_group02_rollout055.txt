```python
import numpy as np
from scipy.optimize import minimize_scalar, minimize
import time

def compute_corr(h):
    n = len(h)
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return corr

def compute_c5(h):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    return np.max(corr) * dx

def get_coefficients(h, i):
    """For coordinate i, compute the coefficient and constant for each correlation element."""
    n = len(h)
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    # For each k in 0..2n-2, find coefficient of h[i] in corr[k]
    # corr[k] = sum_j h[j] * (1-h[j+lag]) where lag = k - (n-1)
    # h[i] appears with coefficient:
    #   (1-h[i+lag]) if 0 <= i+lag < n
    #   -h[i-lag] if 0 <= i-lag < n
    coeffs = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - (n-1)
        c = 0.0
        if 0 <= i + lag < n:
            c += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            c -= h[i - lag]
        coeffs[k] = c
    # constants: corr[k] - coeffs[k] * h[i]
    consts = corr - coeffs * h[i]
    return coeffs, consts

def optimize_coordinate(h, i):
    """Optimize coordinate i given all others fixed."""
    n = len(h)
    dx = 2.0 / n
    coeffs, consts = get_coefficients(h, i)
    
    def obj(x):
        return np.max(coeffs * x + consts) * dx
    
    result = minimize_scalar(obj, bounds=(0, 1), method='bounded')
    new_val = result.x
    h[i] = new_val
    return h

def coordinate_descent(h, max_iters=100, tol=1e-8):
    n = len(h)
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        improved = False
        for i in range(n):
            old_val = h[i].copy()
            h = optimize_coordinate(h, i)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - tol:
                best_c5 = new_c5
                improved = True
            elif abs(h[i] - old_val) < tol:
                h[i] = old_val
        if not improved:
            break
    return h, best_c5
```

```python
import numpy as np
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return np.max(corr) * dx

def optimize_single_coord_full(h, i, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    
    coeffs = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        lag = k - (n - 1)
        c = 0.0
        if 0 <= i + lag < n:
            c += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            c -= h[i - lag]
        coeffs[k] = c
    
    consts = corr - coeffs * h[i]
    sum_others = np.sum(h) - h[i]
    target = n / 2.0
    
    def obj(x):
        vals = coeffs * x + consts
        return np.max(vals) * dx + penalty_weight * (sum_others + x - target) ** 2
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
    return float(result.x)

    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)

        max_change = 0.0

            max_change = max(max_change, abs(new_val - h[i]))
            h[i] = new_val
```

```python
def optimize_single_coord_vec(h, i, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    
    k = np.arange(2*n - 1)
    # First term: (1 - h[i+lag]) where lag = k-(n-1)
    # i+lag = i + k - (n-1) = k + i - n + 1
    idx1 = k + i - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    # Second term: h[i-lag] where lag = k-(n-1)
    # i-lag = i - k + (n-1) = n + i - 1 - k
    idx2 = n + i - 1 - k
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    coeffs = np.zeros(2*n - 1)
    coeffs[valid1] = (1.0 - h[idx1[valid1]])
    coeffs[valid2] -= h[idx2[valid2]]
    
    consts = corr - coeffs * h[i]
    sum_others = np.sum(h) - h[i]
    target = n / 2.0
    
    def obj(x):
        return np.max(coeffs * x + consts) * dx + penalty_weight * (sum_others + x - target) ** 2
    
    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
    return float(result.x)
```

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def optimize_single_coord(h, i, penalty_weight):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    k = np.arange(2 * n - 1)
    idx1 = k + i - n + 1
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = n + i - 1 - k
    valid2 = (idx2 >= 0) & (idx2 < n)
    coeffs = np.zeros(2 * n - 1)
    coeffs[valid1] = (1.0 - h[idx1[valid1]])
    coeffs[valid2] -= h[idx2[valid2]]
    consts = corr - coeffs * h[i]
    sum_others = np.sum(h) - h[i]
    target = n / 2.0

    def obj(x):
        return float(np.max(coeffs * x + consts) * dx + penalty_weight * (sum_others + x - target) ** 2)

    result = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-9})
    return float(result.x)

def coord_descent(h, max_sweeps=100, penalty_weight=1000.0):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            new_val = optimize_single_coord(h, i, penalty_weight)
            max_change = max(max_change, abs(new_val - h[i]))
            h[i] = new_val
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if max_change < 1e-9:
            break
    return h.copy(), best_c5

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def random_start(n, rng):
    h = rng.rand(n)
    return make_feasible(h, n)

def block_start(n, rng):
    h = np.zeros(n)
    nblocks = 4 + rng.randint(0, 12)
    block_len = max(1, n // nblocks)
    fill_ratio = rng.uniform(0.3, 0.7)
    for b in range(0, n, block_len):
        fill = max(1, int(block_len * fill_ratio))
        h[b:b + fill] = 1.0
    perm = rng.permutation(n)
    h = h[perm]
    return make_feasible(h, n)

def cosine_start(n, rng):
    x = np.linspace(0, 2, n)
    freq = 1.0 + rng.randint(0, 6)
    phase = rng.rand() * 2.0 * np.pi
    h = 0.5 + 0.5 * np.cos(np.pi * freq * x + phase)
    return make_feasible(h, n)

def step_start(n, rng):
    h = np.zeros(n)
    cut = max(1, int(n * rng.uniform(0.25, 0.75)))
    h[:cut] = 1.0
    remainder = n / 2.0 - cut
    if 0 < remainder < 1 and cut < n:
        h[cut] = remainder
    perm = rng.permutation(n)
    return make_feasible(h[perm], n)

def two_step_start(n, rng):
    h = np.zeros(n)
    split = max(1, int(n * rng.uniform(0.3, 0.7)))
    a = rng.uniform(0.5, 1.0)
    b = rng.uniform(0.0, 0.5)
    h[:split] = a
    h[split:] = b
    s = np.sum(h)
    if s > 0:
        h *= (n / 2.0) / s
    h = np.clip(h, 0, 1)
    perm = rng.permutation(n)
    return make_feasible(h[perm], n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = random_start(45, rng)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def time_left():
        return budget_s - (time.time() - start_time)

    h_opt, c5 = coord_descent(h0, max_sweeps=300, penalty_weight=2000.0)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_opt.copy()

    generators = [random_start, block_start, cosine_start, step_start, two_step_start]
    n_candidates = sorted(set([n0] + list(range(10, 10, 10)) + list(range(20, 301, 10))))
    n_candidates = [n for n in n_candidates if n >= 8]

    for n_try in n_candidates:
        if time_left() < 45:
            break
        est_per_sweep = 0.01 * n_try * n_try / 1000.0 + 0.5
        max_sweeps = min(100, max(10, int(time_left() * 0.6 / (est_per_sweep * 3))))
        n_starts = min(15, max(3, int(1500.0 / (n_try * 1.5))))

        for s in range(n_starts):
            if time_left() < 12:
                break
            gen = generators[s % len(generators)]
            h = gen(n_try, rng)
            if n_try == n0 and s >= len(generators):
                r = rng.rand()
                if r < 0.4:
                    h = best_h + rng.randn(n_try) * 0.1
                elif r < 0.7:
                    h = best_h * (0.5 + 0.5 * rng.rand(n_try))
                else:
                    mask = rng.rand(n_try) < 0.2
                    h = best_h.copy()
                    h[mask] = rng.rand(mask.sum())
                h = make_feasible(h, n_try)
            if n_try != len(best_h) and s == 0:
                h_interp = np.interp(np.linspace(0, n_try - 1, n_try), np.linspace(0, len(best_h) - 1, len(best_h)), best_h)
                h_interp = make_feasible(h_interp, n_try)
                if rng.rand() < 0.5:
                    h = h_interp
            h_opt_local, c5_local = coord_descent(h, max_sweeps=max_sweeps, penalty_weight=2000.0)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_opt_local.copy()

    if time_left() > 60:
        for _ in range(5):
            if time_left() < 30:
                break
            pert = best_h + rng.randn(len(best_h)) * 0.05
            pert = make_feasible(pert, len(best_h))
            h_opt_local, c5_local = coord_descent(pert, max_sweeps=100, penalty_weight=2000.0)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_opt_local.copy()

    if time_left() > 20:
        h_fine, c5_fine = coord_descent(best_h, max_sweeps=300, penalty_weight=50000.0)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    n_final = len(best_h)
    target = n_final / 2.0
    for _ in range(100):
        diff = target - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = best_h < 1.0 - 1e-15
        else:
            mask = best_h > 1e-15
        if not mask.any():
            break
        best_h[mask] += diff / mask.sum()
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
```
