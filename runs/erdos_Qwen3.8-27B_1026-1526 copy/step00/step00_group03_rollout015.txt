def repair(h, n):
    """Repair h to satisfy constraints: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h.copy(), 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if s == 0:
        h = np.full(n, 0.5)
        return h
    if abs(s - target) < 1e-10:
        return h
    # Scale to fix the sum
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Additional fix: adjust free variables
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        diff = target - s
        # Adjust elements that aren't at bounds
        free = (h > 1e-6) & (h < 1 - 1e-6)
        if np.sum(free) > 0:
            adj = diff / np.sum(free)
            h[free] += adj
            h[free] = np.clip(h[free], 0, 1)
        # Final fix
        s = np.sum(h)
        if abs(s - target) > 1e-10:
            # Spread across all
            h += (target - s) / n
            h = np.clip(h, 0, 1)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if s > 0 and abs(s - target) > 1e-12:
        # Proportional scaling
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    # Fix remaining error
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to increase; add to elements not at 1
            mask = h < 1.0 - 1e-10
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
            else:
                h += diff / n
        else:
            # Need to decrease; subtract from elements not at 0
            mask = h > 1e-10
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
            else:
                h += diff / n
        h = np.clip(h, 0.0, 1.0)

    s = np.sum(h)

                grad[j] += w * (1.0 - h[j - m])
            for j in range(n - m):
                grad[j] -= w * h[j + m]

    return grad

def optimize_coordinate(h, i, n, dx):
    """Find optimal value of h[i] given all other coordinates fixed."""
    n_lags = 2 * n - 1
    # Compute current correlations
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_vals = corr * dx  # shape (2n-1,)
    
    # For coordinate i, compute alpha and beta for each lag
    # c_k(t) = alpha_k * t + beta_k
    # where t = h[i]
    
    alphas = np.zeros(n_lags)
    
    for k in range(n_lags):
        lag = k - (n - 1)
        if lag >= 0:
            # h[i] appears as:
            # 1. h[i]*(1-h[i+lag]) if i <= n-1-lag
            # 2. -h[i]*h[i-lag] if i >= lag (from h[i-lag]*(1-h[i]))
            a = 0.0
            if i <= n - 1 - lag and i + lag < n:
                a += (1.0 - h[i + lag])
            if i >= lag and i - lag >= 0:
                a -= h[i - lag]
            alphas[k] = dx * a
        else:
            m = -lag  # m > 0
            # For lag = -m: c = dx * sum_{j=m}^{n-1} h[j]*(1-h[j-m])
            # h[i] appears as:
            # 1. h[i]*(1-h[i-m]) if i >= m and i-m >= 0
            # 2. -h[i]*h[i+m] if i+m < n and i >= 0 (from h[i+m]*(1-h[i]))
            a = 0.0
            if i >= m:
                a += (1.0 - h[i - m])
            if i + m < n:
                a -= h[i + m]
            alphas[k] = dx * a
    
    # beta_k = c5_vals[k] - h[i] * alphas[k]
    h_i = h[i]
    betas = c5_vals - h_i * alphas
    
    # Now find t in [0,1] that minimizes max_k (alphas[k]*t + betas[k])
    # Evaluate at t=0, t=1, and at all pairwise intersections in [0,1]
    
    best_t = h_i  # default
    best_val = np.max(alphas * h_i + betas)
    
    # Check boundaries
    for t_check in [0.0, 1.0]:
        val = np.max(alphas * t_check + betas)
        if val < best_val:
            best_val = val
            best_t = t_check
    
    # Check intersection points
    # For pairs (p, q): t = (betas[q] - betas[p]) / (alphas[p] - alphas[q])
    # Only consider if alphas[p] != alphas[q] and t in [0,1]
    for p in range(n_lags):
        for q in range(p + 1, n_lags):
            da = alphas[p] - alphas[q]
            if abs(da) < 1e-14:
                continue
            t_int = (betas[q] - betas[p]) / da
            if 0.0 <= t_int <= 1.0:
                val = np.max(alphas * t_int + betas)
                if val < best_val:
                    best_val = val
                    best_t = t_int
    
    return best_t, best_val

                    candidates.append(np.clip(t, 0.0, 1.0))

    # Shape: (n_lags, len(candidates))
    f_matrix = alphas[:, None] * candidates[None, :] + betas[:, None]
    f_vals = np.max(f_matrix, axis=0)
    best_idx = np.argmin(f_vals)
    best_t = candidates[best_idx]
    best_val = f_vals[best_idx]
    
    return best_t, best_val
