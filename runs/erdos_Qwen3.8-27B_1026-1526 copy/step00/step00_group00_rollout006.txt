    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP with moderate smoothing
    T_list = [50.0, 80.0, 120.0]
    for T in T_list:
        for attempt in range(8):
            if elapsed() > budget_s * 0.5:
                break
            if attempt == 0:
                h_start = best_h.copy()
            else:
                scale = max(0.05, 0.3 / (1.0 + 0.3 * attempt))
                h_start = best_h + np.random.normal(0, scale, n)
                h_start = project_onto_constraint(h_start, target_sum)

            def obj_func(h, T=T):
                return smooth_max_obj(h, dx, T)

            def jac_func(h, T=T):
                return smooth_max_grad(h, dx, T)

            try:
                res = minimize(obj_func, h_start, jac=jac_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-14, 'disp': False})
                h_trial = project_onto_constraint(res.x, target_sum)
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
            except Exception:
                pass

    # Phase 2: Local perturbation refinement
    if elapsed() < budget_s - 60:
        h_local, c5_local = local_perturb_improve(best_h, dx, target_sum, n_iter=3000, step_size=0.08)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local

    # Phase 3: Fine-tuning SLSQP with high T
    for attempt in range(10):
        if elapsed() > budget_s - 20:
            break
        if attempt == 0:
            h_start = best_h.copy()
        else:
            scale = 0.02 + 0.02 / (1.0 + attempt * 0.5)
            h_start = best_h + np.random.normal(0, scale, n)
            h_start = project_onto_constraint(h_start, target_sum)

        try:
            res = minimize(lambda h: smooth_max_obj(h, dx, 200.0), h_start,
                           jac=lambda h: smooth_max_grad(h, dx, 200.0),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'disp': False})
            h_trial = project_onto_constraint(res.x, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        except Exception:
            pass

    # Phase 4: Final local perturbation
    if elapsed() < budget_s - 5:
        h_local, c5_local = local_perturb_improve(best_h, dx, target_sum, n_iter=2000, step_size=0.03)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local

    # Final local perturbation with very small steps
    if elapsed() < budget_s - 2:
        h_local, c5_local = local_perturb_improve(best_h, dx, target_sum, n_iter=1000, step_size=0.01)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local

    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
