```python
import numpy as np
import time
import math

def _compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-12:
            break
        # Bisection on shift
        lo = -1.0
        hi = 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            proj = np.clip(h - mid, 0.0, 1.0)
            if np.sum(proj) > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    return h

def _smooth_max_value(corr, alpha):
    vmax = np.max(corr)
    if alpha < 1e-10:
        return float(vmax)
    stable = corr - vmax
    return float(vmax + np.log(np.sum(np.exp(alpha * stable))) / alpha)

def _smooth_gradient(h, n, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    p = exp_vals / np.sum(exp_vals)
    
    # Vectorized gradient computation
    # p has length 2n-1
    # For coordinate i, grad[i] = dx * (sum_A[i] - sum_B[i])
    # sum_A[i] = sum_{j=0}^{n-1} p[j + n - 1 - i] * (1 - h[j])
    # sum_B[i] = sum_{j=0}^{n-1} p[j + i - n + 1] * h[j] (with masking)
    
    nm1 = n - 1
    j_idx = np.arange(n)
    
    # For A: p_index = j + nm1 - i, ranges from (0+nm1-(n-1))=0 to ((n-1)+nm1-0)=2n-2
    p_idx_A = j_idx[np.newaxis, :] + nm1 - j_idx[:, np.newaxis]  # shape (n,n)
    P_A = p[p_idx_A]  # shape (n,n), no masking needed as indices are always valid
    sum_A = P_A.dot(1.0 - h)  # shape (n,)
    
    # For B: p_index = j + i - nm1, ranges from (0+0-nm1)=-(n-1) to ((n-1)+(n-1)-nm1)=n-1
    p_idx_B = j_idx[np.newaxis, :] + j_idx[:, np.newaxis] - nm1  # shape (n,n)
    mask_B = (p_idx_B >= 0) & (p_idx_B <= 2 * nm1)
    p_idx_B_clipped = np.clip(p_idx_B, 0, 2 * nm1)
    P_B = np.where(mask_B, p[p_idx_B_clipped], 0.0)
    sum_B = P_B.dot(h)  # shape (n,)
    
    return dx * (sum_A - sum_B)

def _gradient_descent(h0, n, dx, target_sum, max_iter, alpha_start, alpha_end, checkpoint_time=None):
    h = h0.copy()
    c5 = _compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    t_check = time.time() if checkpoint_time else None
    
    for it in range(max_iter):
        # Check time
        if checkpoint_time and (time.time() - t_check > checkpoint_time):
            break
        
        frac = it / max(float(max_iter), 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        grad = _smooth_gradient(h, n, dx, alpha)
        
        # Backtracking line search
        lr = 0.01
        f_curr = _smooth_max_value(np.correlate(h, 1.0 - h, mode='full') * dx, alpha)
        
        improved = False
        for _ in range(20):
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_feasible(h_new, target_sum)
            f_new = _smooth_max_value(np.correlate(h_new, 1.0 - h_new, mode='full') * dx, alpha)
            if f_new < f_curr:
                improved = True
                break
            lr *= 0.5
        
        if improved:
            h = h_new
        else:
            # Try smaller step
            h = h - 0.001 * grad
            h = np.clip(h, 0.0, 1.0)
            h = _project_feasible(h, target_sum)
        
        # Update best
        c5_new = _compute_c5(h, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

def _coordinate_descent(h0, n, dx, target_sum, max_sweeps, checkpoint_time=None):
    h = h0.copy()
    c5 = _compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    t_check = time.time() if checkpoint_time else None
    
    for sweep in range(max_sweeps):
        if checkpoint_time and (time.time() - t_check > checkpoint_time):
            break
        
        for i in range(n):
```

```python
    vmax = np.max(corr)

    return float(vmax + np.log(np.sum(np.exp(alpha * stable))) / alpha)
```
