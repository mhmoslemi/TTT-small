from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_function(h, n, beta, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    smax = logsumexp(beta * corr) / beta
    pen = lam * (np.sum(h) - n * 0.5) ** 2
    return smax + pen

    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm
