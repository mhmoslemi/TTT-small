import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def obj_func(x, n, target):
    h = project_to_feasible(x, target)
    return compute_c5(h)

def optimize_local(h_start, n, target, maxiter=5000, maxfev=10000):
    try:
        result = minimize(obj_func, h_start, args=(n, target), method='Nelder-Mead',
                          options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-10,
                                   'maxfev': maxfev, 'adaptive': True})
        h = project_to_feasible(result.x, target)
        c5 = compute_c5(h)
        return h, c5

        try:

            h = project_to_feasible(result.x, target)
            c5 = compute_c5(h)
            return h, c5
        except Exception:
            return h_start, compute_c5(h_start)
