    corr = np.correlate(h, 1-h, mode='full') * dx
    # Smooth max approximation
    shifted = beta * (corr - np.max(corr))

def smooth_max_grad(h, beta):

    n = len(h)
    corr = np.correlate(h, 1-h, mode='full') * dx

    g = np.zeros(2*n-1)  # will accumulate
    grad_h = np.zeros(n)

        else:

            gm = np.zeros(n)
            gm[:n-mp] += (-h[mp:]) * dx
            gm[mp:] += (1 - h[:n-mp]) * dx
        grad_h += weights[k] * gm
    return grad_h

Since the objective is f(h) = max_k c_k(h) where each c_k is smooth (quadratic in h), I can use the following:

**Projected gradient with line search:**
1. Compute g = ∇c_{k*}(h) where k* = argmax c_k
2. Try step sizes α, α/2, α/4, ... until f(h - α*g) < f(h)
3. If no step works, stop (or use a different direction)

But this can get stuck. A better approach:

**Frank-Wolfe (conditional gradient) style:**
1. Compute g = ∇c_{k*}(h)
2. Find s = argmin_{s ∈ feasible set} <g, s> (this is a simpler optimization)
3. Move: h_new = (1-t)*h + t*s for some t
4. The feasible set is {0 ≤ x ≤ 1, sum(x) = n/2}

The linear minimization over the feasible set:
min <g, s> subject to 0 ≤ s ≤ 1, sum(s) = n/2

This can be solved by sorting: the optimal s has s[i] = 1 for the |n/2 - count| smallest g[i], s[i] = 0 for the largest, and one fractional value.

Wait, more precisely: we want to minimize sum g[i]*s[i] subject to 0 ≤ s[i] ≤ 1 and sum s[i] = n/2.

The optimal solution: sort g in ascending order. Set s[i] = 1 for the smallest n/2 values (approximately), s[i] = 0 for the rest, with one fractional value to make the sum exactly n/2.

Actually, for sum(s) = n/2 where n/2 might not be integer:
- Sort g ascending: g[0] ≤ g[1] ≤ ... ≤ g[n-1]
- Set s[i] = 1 for i = 0, 1, ..., floor(n/2)-1
- Set s[floor(n/2)] = n/2 - floor(n/2) (fractional part)
- Set s[i] = 0 for i > floor(n/2)

If n/2 is integer (n even), all values are 0 or 1.
If n/2 is half-integer (n odd), one value is 0.5.

This is the Frank-Wolfe (conditional gradient) step.

def compute_all_corr(h, n):
    """Compute all correlation values c[m] * dx for m = -(n-1) to (n-1)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr  # length 2n-1

def compute_c5(corr):
    return np.max(corr)

def project_to_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using water-filling."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def fw_step(g, n):
    """Frank-Wolfe step: find s = argmin <g, s> over {0<=s<=1, sum(s)=n/2}."""
    target = n / 2.0
    order = np.argsort(g)
    s = np.zeros(n)
    # Fill in 1s for the smallest g values
    full = int(np.floor(target))
    frac = target - full
    s[order[:full]] = 1.0
    if frac > 1e-10:
        s[order[full]] = frac
    return s

def grad_shift(h, n, m, dx):
    """Gradient of c[m]*dx w.r.t. h for shift m."""
    g = np.zeros(n)
    if m == 0:
        g[:] = (1 - 2 * h) * dx
    elif m > 0:
        g[:n-m] += (1 - h[m:]) * dx
        g[m:] -= h[:n-m] * dx
    else:
        mp = -m
        g[:n-mp] += (-h[mp:]) * dx
        g[mp:] += (1 - h[:n-mp]) * dx
    return g

import numpy as np
import time
import math

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_corr(h, n)))

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fw_vertex(g, n):
    target = n / 2.0
    order = np.argsort(g)
    s = np.zeros(n)
    full = int(math.floor(target))
    frac = target - full
    s[order[:full]] = 1.0
    if frac > 1e-10:
        s[order[full]] = frac
    return s

def weighted_grad(h, n, weights, dx):
    g = np.zeros(n)
    w0 = weights[n - 1]
    if w0 > 1e-12:
        g += (1.0 - 2.0 * h) * w0
    for m in range(1, n):
        wp = weights[n - 1 + m]
        if wp > 1e-8:
            g[:n - m] += (1.0 - h[m:]) * wp
            g[m:] -= h[:n - m] * wp
        wn = weights[n - 1 - m]
        if wn > 1e-8:
            g[:n - m] += (-h[m:]) * wn
            g[m:] += (1.0 - h[:n - m]) * wn
    return g * dx

def optimize_fw(h, n, max_iter, time_deadline, rng):
    dx = 2.0 / n
    beta = 8.0
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    stall = 0

    for it in range(max_iter):
        if time.time() > time_deadline:
            break

        corr = compute_all_corr(h, n)
        cv = float(np.max(corr))

        if cv < best_val - 1e-10:
            best_val = cv
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        shifted = beta * (corr - np.max(corr))
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        if wsum > 1e-30:
            weights /= wsum
        else:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0

        g = weighted_grad(h, n, weights, dx)
        s = fw_vertex(g, n)

        t = 2.0 / (it + 2)
        if stall > 100:
            t = min(t, 0.03)

        h_new = (1.0 - t) * h + t * s
        vn = compute_c5(h_new, n)

        if vn < cv:
            h = h_new
        else:
            h_new2 = (1.0 - t * 0.5) * h + t * 0.5 * s
            vn2 = compute_c5(h_new2, n)
            if vn2 < cv:
                h = h_new2
            elif stall > 300:
                break
            elif stall > 100 and stall % 50 == 0:
                h = h + 0.02 * (rng.random(n) - 0.5)
                h = project_to_feasible(h, n)

        if it % 300 == 0 and it > 0:
            if stall > 20:
                beta = max(3.0, beta * 0.8)
            else:
                beta = min(40.0, beta * 1.3)

    return best_h, best_val

def smooth_max_obj(x, n, beta):
    corr = compute_all_corr(x, n)
    mv = np.max(corr)
    shifted = beta * (corr - mv)
    shifted -= np.max(shifted)
    return mv + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad_fn(x, n, beta):
    corr = compute_all_corr(x, n)
    mv = np.max(corr)
    shifted = beta * (corr - mv)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws > 1e-30:
        w /= ws
    else:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    dx = 2.0 / n
    return weighted_grad(x, n, w, dx)

def slsqp_constraint_fun(x, n):
    return np.sum(x) - n / 2.0

def slsqp_constraint_jac(x, n):
    return np.ones(len(x))

def optimize_slsqp(h, n, beta, max_iter, time_deadline):
    from scipy.optimize import minimize

    def obj(x):
        if time.time() > time_deadline:
            return 1e10
        return smooth_max_obj(x, n, beta)

    def grad(x):
        if time.time() > time_deadline:
            return np.zeros(len(x))
        return smooth_max_grad_fn(x, n, beta)

    def cfun(x):
        if time.time() > time_deadline:
            return 0.0
        return slsqp_constraint_fun(x, n)

    def cjac(x):
        if time.time() > time_deadline:
            return np.zeros(len(x))
        return slsqp_constraint_jac(x, n)

    cons = {'type': 'eq', 'fun': cfun, 'jac': cjac}
    bounds = [(0.0, 1.0)] * n

    try:
        result = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                          bounds=bounds, constraints=[cons],
                          options={'maxiter': max_iter, 'ftol': 1e-12})
        h_opt = project_to_feasible(result.x, n)
    except Exception:
        h_opt = project_to_feasible(h, n)

    return h_opt, compute_c5(h_opt, n)
