                best_h, best_c5 = h_pol, c5_pol

    # Final polish
    if elapsed() < budget_s - 5:
        h, c5 = polish_slsqp(best_h, n, n / 2.0, t0, budget_s)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, len(best_h)
