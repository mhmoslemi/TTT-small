    comp = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(comp) * dx)

    return x.sum() - _target_sum

    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mu = (lo + hi) * 0.5
        h_p = np.clip(h - mu, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) * 0.5
    return np.clip(h - mu, 0.0, 1.0)

def _subgradient(h, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    c0 = compute_c5(h)
    for i in range(n):
        h_p = h.copy()
        h_p[i] += eps
        h_m = h.copy()
        h_m[i] -= eps

        h_m = np.clip(h_m, 0.0, 1.0)
        grad[i] = (compute_c5(h_p) - compute_c5(h_m)) / (2.0 * eps)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_target_sum = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(comp) * dx)

def _sum_eq(x):
    return x.sum() - _target_sum

def _project(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mu = (lo + hi) * 0.5
        s = np.clip(h - mu, 0.0, 1.0).sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _corr_grad(h, k):
    n = len(h)
    off = k - (n - 1)
    grad = np.zeros(n)
    i_range = np.arange(n)
    j2 = i_range + off
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] += (1.0 - h[j2[mask2]])
    j1 = i_range - off
    mask1 = (j1 >= 0) & (j1 < n)
    grad[mask1] -= h[j1[mask1]]
    return grad

def _proj_grad_step(h, target, lr0, n_steps):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for s in range(n_steps):
        comp = np.correlate(h, 1.0 - np.ones(n) * 0 + 1 - h, mode="full")
        k_star = int(np.argmax(comp))
        grad = _corr_grad(h, k_star) * dx
        lr = lr0 * (1.0 + s / n_steps) ** (-2.0)
        h = _project(h - lr * grad, target)
    return h

def _try_slsqp(h_start, n, target, maxiter, ftol):
    global _target_sum
    _target_sum = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        res = minimize(
            compute_c5, h_start, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
    except Exception:
        return h_start.copy(), compute_c5(h_start)
    h_out = _project(np.clip(res.x, 0.0, 1.0), target)
    return h_out, compute_c5(h_out)

def _make_alternating(n, target):
    h = np.zeros(n)
    h[::2] = 1.0
    h = _project(h, target)
    return h

def _make_periodic(n, target, period):
    h = np.zeros(n)
    for p in range(period):
        if p < period // 2:
            h[p::period] = 1.0
        else:
            h[p::period] = 0.0
    h = _project(h, target)
    return h

def _make_random(n, target, rng):
    h = rng.standard_normal(n)
    h = _project(h, target)
    return h
