import time

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    
    m = np.max(corr)
    alpha_corr = alpha * (corr - m)
    exp_vals = np.exp(alpha_corr)
    total_exp = np.sum(exp_vals)
    softmax = exp_vals / total_exp
    smooth_val = m + np.log(total_exp) / alpha
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for i in range(n):
        g = 0.0
        for m_idx in range(n_lags):
            k = m_idx - (n - 1)
            w = softmax[m_idx]
            contrib = 0.0
            if 0 <= i - k < n:
                contrib += (1.0 - h[i - k])
            if 0 <= i + k < n:
                contrib -= h[i + k]
            g += w * contrib
        grad[i] = g * dx

def project_to_constraint(h, target_sum):
    h = h.copy()
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_free = np.sum(free)
        if n_free > 0:
            h[free] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
        else:
            # Try to adjust boundary elements
            if diff > 0:
                candidates = np.argsort(h)
                for idx in candidates:
                    if h[idx] < 1.0:
                        add = min(diff, 1.0 - h[idx])
                        h[idx] += add
                        diff -= add
                        if abs(diff) < 1e-12:
                            break
            else:
                candidates = np.argsort(-h)
                for idx in candidates:
                    if h[idx] > 0.0:
                        sub = min(-diff, h[idx])
                        h[idx] -= sub
                        diff += sub
                        if abs(diff) < 1e-12:

def coordinate_descent(h, n, max_passes=100, time_limit=None, start_time=None):
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    
    for p in range(max_passes):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for i in range(n):
            # Try different values of h[i], adjusting another coordinate to maintain sum
            old_val = best_h[i]
            for v in [0.0, 0.25, 0.5, 0.75, 1.0, old_val]:
                if v == old_val:
                    continue
                h_try = best_h.copy()

                diff = target_sum - np.sum(h_try)

                    free = [j for j in range(n) if j != i and 0 < best_h[j] < 1]
                    if free:
                        # Distribute the adjustment
                        h_try[free] += diff / len(free)
                        h_try = np.clip(h_try, 0, 1)
                    else:
                        continue
                    # Check if sum is still satisfied
                    if abs(np.sum(h_try) - target_sum) > 1e-6:
                        h_try = project_to_constraint(h_try, target_sum)
                        if abs(np.sum(h_try) - target_sum) > 1e-6:

                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    
    m = np.max(corr)
    alpha_corr = alpha * (corr - m)
    exp_vals = np.exp(alpha_corr)
    total_exp = np.sum(exp_vals)
    softmax = exp_vals / total_exp
    smooth_val = m + np.log(total_exp) / alpha
    
    # Gradient: d(smooth_c5)/d(h[i]) = sum_k softmax[k] * d(corr[k])/d(h[i])
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for i in range(n):
        g = 0.0
        for m_idx in range(n_lags):
            k = m_idx - (n - 1)
            w = softmax[m_idx]
            if w < 1e-15:
                continue
            contrib = 0.0
            if 0 <= i - k < n:
                contrib += (1.0 - h[i - k])
            if 0 <= i + k < n:
                contrib -= h[i + k]
            g += w * contrib
        grad[i] = g * dx
    
    return smooth_val, grad
