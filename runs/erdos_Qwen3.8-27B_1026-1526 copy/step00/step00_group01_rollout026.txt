import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def projection_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Iteratively adjust
    for _ in range(100):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

def gradient_c5(h, n):
    """Compute gradient of C5 = max_k corr[k]*dx wrt h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    
    # Numerical gradient of the active component
    grad = np.zeros(n)
    eps = 1e-6
    for j in range(n):
        hp = h.copy()
        hp[j] = min(hp[j] + eps, 1.0)
        hn = h.copy()
        hn[j] = max(hn[j] - eps, 0.0)
        gp = np.correlate(hp, 1.0 - hp, mode='full')[max_idx] * dx
        gn = np.correlate(hn, 1.0 - hn, mode='full')[max_idx] * dx
        grad[j] = (gp - gn) / (eps)
    return grad, max_idx

def subgradient_optimize(h0, n, budget_s, start_time, lr0=0.05):
    h = h0.copy().astype(float)
    h = projection_to_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    
    while time.time() - start_time < budget_s - 10:
        grad, max_idx = gradient_c5(h, n)
        h_new = h - lr * grad
        h_new = projection_to_feasible(h_new, n)

    k = max_idx - (n - 1)
    grad = np.zeros(n)

            grad[j] += (1 - h[j - k])

            grad[j] += (1 - h[j + l])

            grad[j] += -h[j - l]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def projection_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def analytic_gradient(h, n, max_idx):
    dx = 2.0 / n
    k = max_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            grad[j] += (1.0 - h[j - k])
        for j in range(0, n - k):
            grad[j] += -h[j + k]
    else:
        l = -k
        for j in range(0, n - l):
            grad[j] += (1.0 - h[j + l])
        for j in range(l, n):
            grad[j] += -h[j - l]
    return grad * dx

def subgradient_optimize(h0, n, max_time, start_time, lr0=0.01):
    h = h0.copy().astype(float)
    h = projection_to_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    
    while time.time() - start_time < max_time - 5:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = analytic_gradient(h, n, max_idx)
        
        h_new = h - lr * grad
        h_new = projection_to_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_h, best_c5

def smooth_max_objective(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max
    max_val = np.max(corr)
    shifted = corr - max_val
    return max_val + np.log(np.sum(np.exp(T * shifted))) / T

def slsqp_optimize(h0, n, max_time, start_time, T=50):
    h = h0.copy().astype(float)
    h = projection_to_feasible(h, n)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        return smooth_max_objective(x, n, T)
    
    bounds = [(0, 1)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': 500, 'ftol': 1e-10})
    
    h_opt = projection_to_feasible(result.x, n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

    """Optimize h[j] given all other h fixed"""
    # For each lag k, f_k = a_k * h[j] + b_k where a_k, b_k are computed from other variables
    # We want to minimize max_k (a_k * h[j] + b_k) over h[j] ∈ [0,1]
    
    # Compute a_k and b_k for all k
    # corr[k] = sum_i h[i+k]*(1-h[i]) (approximately, with index handling)
    # The dependence on h[j] is:
    # - If j appears as the first argument h[i+k]=h[j], i.e., i=j-k: contributes (1-h[j-k]) to the coefficient... 
    #   wait, it's h[j] * (1-h[i]) where i = j-k, so coefficient of h[j] is (1-h[j-k])
    # - If j appears as the second argument h[i]=h[j], i.e., i=j: contributes h[i+k] * (-1) = -h[j+k]
    
    # So f_k = (contribution involving h[j]) + (contribution not involving h[j])
    # f_k = (1-h[j-k]) * h[j] - h[j+k] * h[j] + rest_k (if both j and j+k are in range for lag k)
    #     = h[j] * ((1-h[j-k]) - h[j+k]) + rest_k (when applicable)
    
    # More carefully:
    # For lag k (where corr index = k + (n-1)):
    # corr[k+(n-1)] = sum_i h[i+k]*(1-h[i]) (valid i)
    # h[j] appears when:
    #   i+k = j → i = j-k (valid if 0 ≤ j-k < n): term is h[j]*(1-h[j-k]), coeff of h[j] is (1-h[j-k])
    #   i = j (valid if 0 ≤ j < n): term is h[j+k]*(1-h[j]) = -h[j+k]*h[j] + h[j+k], coeff of h[j] is -h[j+k]
    
    n_corr = 2 * n - 1
    a = np.zeros(n_corr)  # coefficient of h[j]
    b = np.zeros(n_corr)  # constant term (independent of h[j])
    
    # Save original h[j]
    hj = h[j]
    h_save = h.copy()
    h[j] = 0  # temporarily set to 0 to compute b
    
    for m in range(n_corr):
        k = m - (n - 1)
        # Compute corr[m] with h[j] = 0
        # This is the "b" part (constant in h[j])
        b[m] = np.correlate(h, 1.0 - h, mode='full')[m]
    
    h[j] = 1  # set to 1 to compute a
    for m in range(n_corr):
        k = m - (n - 1)
        a[m] = np.correlate(h, 1.0 - h, mode='full')[m] - b[m]
    
    h[j] = hj
    
    # Now minimize max_m (a[m] * x + b[m]) * dx over x ∈ [0,1]
    # This is a 1D minimax problem
    # Use bisection or ternary search
    
    dx = 2.0 / n
    
    def obj(x):
        return np.max(a * x + b) * dx
    
    # Evaluate at endpoints and find minimum
    # For a linear-in-x function, the max of linear functions is convex in x
    # So we can use ternary search or bisection on the derivative
    
    # Actually, max of linear functions is convex, so ternary search works
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2
    return x_opt

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def projection_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_sweep(h, n):
    """One full sweep of coordinate descent"""
    dx = 2.0 / n
    for j in range(n):
        h0 = h.copy()
        h0[j] = 0.0
        b = np.correlate(h0, 1.0 - h0, mode='full')
        h1 = h.copy()
        h1[j] = 1.0
        ab = np.correlate(h1, 1.0 - h1, mode='full')
        a = ab - b
        
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(a * m1 + b) < np.max(a * m2 + b):
                hi = m2
            else:
                lo = m1
        h[j] = (lo + hi) / 2.0
    
    # Reproject to satisfy sum constraint
    h = projection_to_feasible(h, n)
    return h

def coordinate_descent(h0, n, max_time, start_time, n_sweeps_max=100000):
    h = h0.copy().astype(float)
    h = projection_to_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve_count = 0
    
    for sweep in range(n_sweeps_max):
        if time.time() - start_time > max_time - 10:

        h = coordinate_sweep(h, n)
        c5 = compute_c5(h)
        
        if c5 < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 50:
                break
    
    return best_h, best_c5

def perturb_h(h, n, seed):
    """Create a perturbed version of h"""
    rng = np.random.RandomState(seed)
    h_pert = h + rng.randn(n) * 0.1

def make_random_binary(n, seed):
    """Create a random binary-like vector with sum = n/2"""
    rng = np.random.RandomState(seed)
