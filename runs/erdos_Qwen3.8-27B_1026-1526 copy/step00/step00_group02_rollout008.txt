```python
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_objective(h, dx, tau):
    c = compute_correlations(h)
    # Shift to avoid overflow
    c_shifted = c - np.max(c)
    log_sum = np.max(c) + np.log(np.sum(np.exp(tau * c_shifted)))
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def corr_gradient_component(h, k):
    """Gradient of c_k with respect to h, where k is the actual lag (not index)."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # (1-h_{i-k}) for i in [k, n)
        if n > k:
            grad[k:] += (1.0 - h[:n-k])
        # -h_{i+k} for i in [0, n-k)
        if n > k:
            grad[:n-k] += -h[k:]
    else:
        kk = -k
        # (1-h_{i+kk}) for i in [0, n-kk)
        if n > kk:
            grad[:n-kk] += (1.0 - h[kk:])
        # -h_{i-kk} for i in [kk, n)
        if n > kk:
            grad[kk:] += -h[:n-kk]
    return grad

def lse_obj(h, dx, tau):
    c = compute_corr(h)
    c_max = np.max(c)
    shifted = c - c_max
    return (c_max + np.log(np.sum(np.exp(tau * shifted))) / tau) * dx

def lse_grad(h, dx, tau):
    c = compute_corr(h)
    c_max = np.max(c)
    shifted = c - c_max
    exp_vals = np.exp(tau * shifted)
    weights = exp_vals / exp_vals.sum()
    
    n = len(h)
    grad = np.zeros(n)
    for idx, w in enumerate(weights):
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        grad += w * corr_gradient_component(h, k)
    return grad * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h, 0, 1)
```

```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def corr_grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] += -h[k:]
    else:
        kk = -k
        if kk < n:
            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] += -h[:n-kk]
    return g

def sm_obj_and_grad(h, dx, tau):
    c = compute_corr(h)
    c_max = c.max()
    shifted = c - c_max
    exp_c = np.exp(tau * shifted)
    w = exp_c / exp_c.sum()
    obj = (c_max + np.log(exp_c.sum()) / tau) * dx
    
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        grad += w[idx] * corr_grad_at_lag(h, k)
    grad *= dx
    return obj, grad

def project_feasible(h, target_sum):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h_init)
    bounds = [(0.0, 1.0)] * n

             'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

        return sm_obj_and_grad(x, dx, tau)[0]
    
    def obj_jac(x):
        return sm_obj_and_grad(x, dx, tau)[1]
    
    res = minimize(
        fun=obj_fun, x0=h_init, jac=obj_jac,
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    if res.x is not None:
        return project_feasible(res.x, target_sum)
    return project_feasible(h_init, target_sum)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def evaluate_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)


def corr_grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n - k] += (1.0 - h[k:])
            g[k:] += -h[:n - k]
    else:
        kk = -k
        if kk < n:
            g[kk:] += (1.0 - h[:n - kk])
            g[:n - kk] += -h[kk:]
    return g


def lse_obj(h, dx, tau):
    c = compute_corr(h)
    c_max = c.max()
    shifted = c - c_max
    exp_sum = np.sum(np.exp(tau * shifted))
    return (c_max + np.log(exp_sum) / tau) * dx


def lse_gradient(h, dx, tau):
    c = compute_corr(h)
    c_max = c.max()
    shifted = c - c_max
    exp_vals = np.exp(tau * shifted)
    w = exp_vals / exp_vals.sum()
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(c)):
        if w[idx] < 1e-14:
            continue
        k = idx - (n - 1)
        grad += w[idx] * corr_grad_at_lag(h, k)
    return grad * dx


def project_feasible(h, target_sum):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(120):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h_init, dx, target_sum, tau, maxiter):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj_fun(x):
        return lse_obj(x, dx, tau)

    def obj_jac(x):
        return lse_gradient(x, dx, tau)

    try:
        res = minimize(
            fun=obj_fun, x0=h_init.copy(), jac=obj_jac,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, target_sum)
    except Exception:
        pass
    return project_feasible(h_init, target_sum)


def projected_subgradient(h_init, dx, target_sum, n_steps, lr_init):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    n = len(h)
    lr = lr_init
    for step in range(n_steps):
        c = compute_corr(h)
        top_k_val = np.sort(c)[-3:].max()
        active = c > top_k_val - 0.5 * (c.max() - c.min())
        if not active.any():
            active = c == c.max()
        g = np.zeros(n)
        n_active = active.sum()
        for idx in range(len(c)):
            if not active[idx]:
                continue
            k = idx - (n - 1)
            g += corr_grad_at_lag(h, k)
        g /= n_active
        g *= dx
        h_new = h - lr * g
        h_new = project_feasible(h_new, target_sum)
        c5 = evaluate_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr *= 0.998
    return best_h, best_c5

    t0 = time.time()
    rng = np.random.default_rng(seed)

    n = 45
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = np.array(initial_h_values, dtype=np.float64).flatten()
    if len(h0) != n:
        h0 = np.full(n, target_sum / n)
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_feasible(h0, target_sum)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, dx)

    def remaining():
        return budget_s - (time.time() - t0)

    def try_improve(h_candidate):
        nonlocal best_h, best_c5
        c5 = evaluate_c5(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    # Phase 1: SLSQP continuation
    taus = [5, 10, 20, 50, 100, 200, 400, 800, 1500]
    for tau in taus:
        if remaining() < 120:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, tau, 300)
        try_improve(h_new)
        if remaining() < 120:
            break
        h_new2 = slsqp_optimize(best_h, dx, target_sum, tau, 200)
        try_improve(h_new2)

    # Phase 2: Projected subgradient refinement
    if remaining() > 90:
        h_pd, c5_pd = projected_subgradient(best_h, dx, target_sum, 800, 0.003)
        try_improve(h_pd)

    # Phase 3: More SLSQP at high tau from best
    if remaining() > 90:
        for tau in [2000, 3000, 5000]:
            if remaining() < 60:
                break
            h_new = slsqp_optimize(best_h, dx, target_sum, tau, 200)
            try_improve(h_new)

    # Phase 4: Random restarts
    n_restarts = 30
    for r in range(n_restarts):
        if remaining() < 120:
            break
        scale = rng.uniform(0.01, 0.1)
        h_pert = best_h + rng.standard_normal(n) * scale
        h_pert = project_feasible(h_pert, target_sum)
        for tau in [50, 200, 800]:
            if remaining() < 60:

            h_new = slsqp_optimize(h_pert, dx, target_sum, tau, 150)
            try_improve(h_new)

    # Phase 5: Final CD/alternating refinement on active lags
    if remaining() > 60:
        for _ in range(10):
            if remaining() < 60:
                break
            c = compute_corr(best_h)
            max_idx = int(np.argmax(c))
            k_active = max_idx - (n - 1)
            g_active = corr_grad_at_lag(best_h, k_active) * dx
            shrink = 0.05
            h_try = best_h - shrink * g_active
            h_try = project_feasible(h_try, target_sum)
            c5_try = evaluate_c5(h_try, dx)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            else:
                shrink *= 0.5
                h_try = best_h + shrink * g_active

                c5_try = evaluate_c5(h_try, dx)

                    best_c5 = c5_try

    if remaining() > 30:
        h_new = slsqp_optimize(best_h, dx, target_sum, 5000, 100)
        try_improve(h_new)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, target_sum)
    best_c5 = evaluate_c5(best_h, dx)

    return best_h, best_c5, n
```
