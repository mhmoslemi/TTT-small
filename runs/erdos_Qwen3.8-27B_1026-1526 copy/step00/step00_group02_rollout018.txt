    return m + np.log(np.sum(np.exp(shifted))) / alpha


def corr_dx(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def c5_value(h, n):
    return float(np.max(corr_dx(h, n)))


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] += per
        h = np.clip(h, 0.0, 1.0)
    return h


def make_restart(h_best, n, target_sum, rng, strength):
    h_new = h_best + strength * rng.standard_normal(n)
    return project_feasible(h_new, target_sum)


def slsqp_solve(h0, n, target_sum, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        cx = corr_dx(h, n)
        pm = smooth_max(cx, alpha)
        sv = np.sum(h) - target_sum
        return pm + 50.0 * sv * sv

    def obj_grad(h):
        cx = corr_dx(h, n)
        w = np.exp(np.clip(alpha * (cx - np.max(cx)), -50.0, 50.0))
        w = w / np.sum(w)
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        grad = np.zeros_like(h)
        for k in range(2 * n - 1):
            wk = w[k]
            for i in range(max(0, k - (n - 1)), min(n, n + k - (n - 1) + 1)):
                j = i + k - (n - 1)
                if 0 <= i < n and 0 <= j < n:
                    grad[i] += wk * dx * (1.0 - h[j])
                    grad[j] -= wk * dx * h[i]
        sv = np.sum(h) - target_sum
        grad += 100.0 * sv * np.ones(n)
        return grad

    def cons_fun(h):
        return np.sum(h) - target_sum

    def cons_jac(h):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0


def lbfgsb_solve(h0, n, target_sum, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    pw = 500.0

    def obj(h):
        cx = corr_dx(h, n)
        pm = smooth_max(cx, alpha)
        sv = np.sum(h) - target_sum
        return pm + pw * sv * sv

    def obj_grad(h):
        cx = corr_dx(h, n)
        m = np.max(cx)
        w = np.exp(np.clip(alpha * (cx - m), -50.0, 50.0))
        w = w / np.sum(w)
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        grad = np.zeros_like(h)
        for k in range(2 * n - 1):
            wk = w[k]
            lo = max(0, k - (n - 1))
            hi = min(n, n + k - (n - 1) + 1)
            for i in range(lo, hi):
                j = i + k - (n - 1)
                if 0 <= i < n and 0 <= j < n:
                    grad[i] += wk * dx * (1.0 - h[j])
                    grad[j] -= wk * dx * h[i]
        sv = np.sum(h) - target_sum
        grad += 2.0 * pw * sv * np.ones(n)
        return grad

    try:
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h0


def coordinate_refine(h, n, target_sum, budget_check, time_limit, frac):
    c5_cur = c5_value(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    for ss in steps:
        for sweep in range(30):
            if budget_check() > time_limit * frac:
                return h, c5_cur
            improved = False
            order = list(range(n))
            for i in order:
                if budget_check() > time_limit * frac:
                    return h, c5_cur
                old = h[i]
                for dv in [-ss, -ss * 0.5, ss * 0.5, ss]:
                    nv = old + dv
                    if nv < -1e-9 or nv > 1.0 + 1e-9:
                        continue
                    h_test = h.copy()
                    h_test[i] = max(0.0, min(1.0, nv))
                    h_test = project_feasible(h_test, target_sum)
                    c5_test = c5_value(h_test, n)
                    if c5_test < c5_cur - 1e-10:
                        h = h_test
                        c5_cur = c5_test
                        improved = True
                        break
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    h_best = np.array(initial_h_values, dtype=np.float64).copy()
    target_sum = n0 / 2.0
    c5_best = c5_value(h_best, n0)

    # Phase 1: SLSQP warm start with increasing alpha
    alphas = [20, 50, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if elapsed() > budget_s * 0.30:
            break
        h_cand = slsqp_solve(h_best, n0, target_sum, alpha, 2000, 1e-14)
        h_cand = project_feasible(h_cand, target_sum)
        c5_cand = c5_value(h_cand, n0)
        if c5_cand < c5_best - 1e-12:
            h_best = h_cand
            c5_best = c5_cand

    # Phase 2: L-BFGS-B with penalty (multiple restarts from perturbed best)
    n_restarts = 40
    for r in range(n_restarts):
        if elapsed() > budget_s * 0.65:
            break
        strength = 0.03 + 0.25 * (r / max(1, n_restarts - 1))
        h_start = make_restart(h_best, n0, target_sum, rng, strength)
        alpha = 1000 if r < 20 else 3000
        h_cand = lbfgsb_solve(h_start, n0, target_sum, alpha, 500)
        h_cand = project_feasible(h_cand, target_sum)
        c5_cand = c5_value(h_cand, n0)
        if c5_cand < c5_best - 1e-12:
            h_best = h_cand
            c5_best = c5_cand

    # Phase 3: SLSQP from best with high alpha
    if elapsed() < budget_s * 0.75:
        for alpha in [3000, 10000, 30000]:
            if elapsed() > budget_s * 0.80:
                break
            h_cand = slsqp_solve(h_best, n0, target_sum, alpha, 3000, 1e-14)
            h_cand = project_feasible(h_cand, target_sum)
            c5_cand = c5_value(h_cand, n0)
            if c5_cand < c5_best - 1e-12:
                h_best = h_cand
                c5_best = c5_cand

    # Phase 4: Coordinate descent refinement
    h_best, c5_best = coordinate_refine(h_best, n0, target_sum, elapsed, budget_s, 0.92)

    # Phase 5: Try larger n if time permits
    if elapsed() < budget_s * 0.5:
        for n_try in [90, 180]:
            if elapsed() > budget_s * 0.70:
                break
            # Interpolate from best solution
            h_interp = np.interp(np.linspace(0, 2, n_try, endpoint=False),
                                 np.linspace(0, 2, n0, endpoint=False),
                                 h_best)
            target_sum_try = n_try / 2.0
            h_interp = project_feasible(h_interp, target_sum_try)
            c5_try = c5_value(h_interp, n_try)

            # Optimize with L-BFGS-B
            for alpha in [500, 2000]:
                if elapsed() > budget_s * 0.85:
                    break
                h_cand = lbfgsb_solve(h_interp, n_try, target_sum_try, alpha, 300)
                h_cand = project_feasible(h_cand, target_sum_try)
                c5_cand = c5_value(h_cand, n_try)
                if c5_cand < c5_try:
                    h_interp = h_cand
                    c5_try = c5_cand

            if c5_try < c5_best:
                h_best = h_interp
                c5_best = c5_try
                n0 = n_try
                target_sum = target_sum_try

    # Final projection and evaluation
    h_best = project_feasible(h_best, target_sum)
    c5_final = c5_value(h_best, n0)

    return h_best, c5_final, n0
