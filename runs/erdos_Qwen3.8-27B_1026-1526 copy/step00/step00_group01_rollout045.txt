    # Phase 5: Higher resolution
    for n_try in [94, 142, 188]:
        if not time_left(0.9):
            break
        n_old = len(best_h)
        x_old = np.linspace(0.5 / n_old, 2.0 - 0.5 / n_old, n_old)
        x_new = np.linspace(0.5 / n_try, 2.0 - 0.5 / n_try, n_try)
        f_interp = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
        h_high = np.clip(f_interp(x_new), 0.0, 1.0)
        h_high = project_feasible(h_high, n_try / 2.0)

        h_new, c5_new = run_lbfgs(h_high, 200, max_iter=500, ftol=1e-9)
        if c5_new < best_c5 - 1e-8:
            best_c5 = c5_new
            best_h = h_new
            h_new2, c5_new2 = run_slsqp(h_new, 500, max_iter=200, ftol=1e-10)
            if c5_new2 < best_c5 - 1e-8:
                best_c5 = c5_new2
                best_h = h_new2
        h_bin = binary_snap(h_new, n_try / 2.0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 6: Final local search with small perturbations
    if time_left(0.97):
        n_final = len(best_h)
        for i in range(10):
            if not time_left(0.97):
                break
            h0 = best_h + rng.normal(0, 0.005, n_final)
            h0 = project_feasible(h0, n_final / 2.0)
            h_new, c5_new = run_slsqp(h0, 1000, max_iter=200, ftol=1e-12)
            if c5_new < best_c5 - 1e-9:
                best_c5 = c5_new
                best_h = h_new
        h_bin = binary_snap(best_h, n_final / 2.0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
