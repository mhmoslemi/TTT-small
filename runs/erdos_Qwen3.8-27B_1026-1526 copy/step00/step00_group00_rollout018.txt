import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exp_terms = np.exp(beta * (corr - max_corr))
    log_sum = max_corr + np.log(np.sum(exp_terms)) / beta
    return dx * log_sum


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            mask = (h > 0) | (h < 1)
            if not np.any(mask):
                break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_with_penalty(h, beta, penalty):
    n = len(h)
    target = n / 2.0
    val = smooth_max_c5(h, beta)
    val += penalty * (np.sum(h) - target) ** 2
    return val

    x_src = np.arange(n_src) * (2.0 / n_src)

    h_dst = np.interp(x_dst, x_ext, h_ext)
    return h_dst
