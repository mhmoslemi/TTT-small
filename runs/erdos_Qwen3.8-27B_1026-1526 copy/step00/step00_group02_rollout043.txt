def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).copy()
    n = int(y.size)
    if n == 0:
        return y
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(p.sum())

    if res > 1e-10:
        idx = np.where(p < 1.0)[0]
        if idx.size:
            room = 1.0 - p[idx]
            j = int(np.argmax(room))
            add = min(float(room[j]), res)
            p[idx[j]] += add
    elif res < -1e-10:
        idx = np.where(p > 0.0)[0]
        if idx.size:
            vals = p[idx]
            j = int(np.argmin(vals))
            sub = min(float(vals[j]), -res)
            p[idx[j]] -= sub

    return p

def upsample_h(h, n_new):

    n = int(h.size)

def snap_to_grid(h, n, denom=1 << 30):
    h = np.asarray(h, dtype=float)
    n = int(n)
    denom = int(denom)
    a = np.rint(h * float(denom)).astype(np.int64)
    a = np.clip(a, 0, denom)
    target_int = (n * denom) // 2
    diff = int(target_int - int(a.sum()))

    if diff > 0:
        order = np.argsort(denom - a)[::-1]
        for idx in order:
            if diff <= 0:
                break
            add = int(min(int(denom - a[idx]), diff))
            if add > 0:
                a[idx] += add
                diff -= add
    elif diff < 0:
        order = np.argsort(a)
        for idx in order:
            if diff >= 0:
                break
            sub = int(min(int(a[idx]), -diff))
            if sub > 0:
                a[idx] -= sub
                diff += sub

    return a.astype(float) / float(denom)


def weighted_gradient(h, W):
    h = np.asarray(h, dtype=float)
    W = np.asarray(W, dtype=float)
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    if W.size != 2 * n - 1:
        tmp = np.zeros(2 * n - 1, dtype=float)
        m = min(W.size, tmp.size)
        tmp[:m] = W[:m]
        W = tmp
    if not np.any(W):
        return np.zeros(n, dtype=float)

    center = n - 1
    ones = np.ones(n, dtype=float)
    Wr = W[::-1]

    s1 = np.convolve(W, ones, mode="full")[center:center + n]
    hm = np.convolve(W, h, mode="full")[center:center + n]
    s2 = np.convolve(Wr, ones, mode="full")[center:center + n]
    hp = np.convolve(Wr, h, mode="full")[center:center + n]

    return (s1 - hm) - (s2 - hp)

def penalty_fun(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return 0.0
    c = _get_c(x)
    T = float(_OPT.get("T", 10.0))
    mu = float(_OPT.get("mu", 1.0))
    target = float(_OPT.get("target", 0.5 * float(x.size)))

    m = float(np.max(c))
    z = T * (c - m)
    exps = np.exp(z)
    s = float(np.sum(exps))
    if s <= 0.0 or not math.isfinite(s):
        return 1e100

    val = (m + math.log(s)) / T
    err = float(np.sum(x) - target)
    val += 0.5 * mu * err * err
    if not math.isfinite(val):
        return 1e100
    return float(val)


def penalty_grad(x):
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return np.zeros(0, dtype=float)

    c = _get_c(x)
    T = float(_OPT.get("T", 10.0))
    mu = float(_OPT.get("mu", 1.0))
    target = float(_OPT.get("target", 0.5 * float(n)))

    m = float(np.max(c))
    z = T * (c - m)
    exps = np.exp(z)
    s = float(np.sum(exps))
    if s <= 0.0 or not math.isfinite(s):
        return np.zeros(n, dtype=float)

    W = exps / s
    grad = weighted_gradient(x, W)
    err = float(np.sum(x) - target)
    grad += mu * err
    return grad

def fallback_descent(h0, n, T, maxiter):
    n = int(n)
    target = 0.5 * float(n)
    _OPT["T"] = float(T)
    _OPT["x"] = None
    _OPT["c"] = None

    h = make_feasible(h0, n)
    current = max_c(h)
    best_c = current
    best_h = h.copy()
    step = 0.08

    for _ in range(int(maxiter)):
        if time.time() >= _OPT.get("deadline", 1e100):
            break
        if best_c - current > 0.05:
            h = best_h.copy()
            current = best_c

        c = _get_c(h)
        m = float(np.max(c))
        z = float(T) * (c - m)
        exps = np.exp(z)
        s = float(np.sum(exps))
        if s <= 0.0 or not math.isfinite(s):
            break

        W = exps / s
        g = weighted_gradient(h, W)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            a = step * factor / gn
            y = h - a * g
            p = project_box_sum(y, target)
            val = max_c(p)
            if val < current - 1e-12:
                h = p
                current = val
                accepted = True
                if current < best_c:
                    best_c = current
                    best_h = h.copy()
                break

        if not accepted:
            step *= 0.8
            if step < 1e-7:
                break

    return best_h

def update_best(best_h, best_c, candidate, n):
    if candidate is None:
        return best_h, best_c
    n = int(n)

    c1 = make_feasible(candidate, n)
    v1 = float(c5_of(c1, n))

def hill_climb(h, n, rng, time_limit):
    n = int(n)
    time_limit = float(time_limit)
    if n <= 0 or time_limit <= 0.0:
        return make_feasible(h, n)

    h = make_feasible(h, n).copy()
    target = 0.5 * float(n)

    c = np.correlate(h, 1.0 - h, mode="full")
    maxc = float(np.max(c))
    best_c = maxc
    best_h = h.copy()

    step = 0.15
    no_improve = 0
    K = max(1, min(10, n))
    outer_trials = 18 if n <= 80 else 8
    t_start = time.time()

    while time.time() - t_start < time_limit:
        if best_c - maxc > 0.05:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full")
            maxc = float(np.max(c))

        top_count = min(5, int(c.size))
        idx_top = np.argsort(c)[::-1][:top_count]
        temp = max(0.1, 0.05 * float(maxc))
        w = np.exp((c[idx_top] - c[idx_top[0]]) / temp)
        wsum = float(w.sum())
        if wsum > 0.0 and math.isfinite(wsum):
            w = w / wsum

        W = np.zeros(c.size, dtype=float)
        W[idx_top] = w
        g = weighted_gradient(h, W)

        if K < n:
            high_idx = np.argpartition(g, -K)[-K:]
            low_idx = np.argpartition(g, K)[:K]
        else:
            high_idx = np.argsort(g)[-K:]
            low_idx = np.argsort(g)[:K]

        improved = False
        for _ in range(outer_trials):
            if time.time() - t_start >= time_limit:
                break

            if rng.random() < 0.85:
                src = int(high_idx[int(rng.integers(K))])
                dst = int(low_idx[int(rng.integers(K))])
            else:
                src = int(rng.integers(n))
                dst = int(rng.integers(n))

            if src == dst:
                continue

            maxd = min(float(h[src]), float(1.0 - h[dst]))
            if maxd <= 1e-14:
                continue

            base = min(step, maxd)
            deltas = []
            for f in (1.0, 0.6, 0.3, 0.15, 0.08):
                d = base * f
                if d > 1e-14 and d <= maxd:
                    deltas.append(d)
            for d in (0.002, 0.001, 0.0005, 0.0002):
                if d < maxd:
                    deltas.append(d)
            if len(deltas) < 8:
                d = base * float(rng.uniform(0.2, 1.0))
                if 0.0 < d < maxd:
                    deltas.append(d)

            for d in deltas[:10]:
                cand = h.copy()
                cand[dst] += d
                cand[src] -= d
                val = max_c(cand)
                if val < maxc - 1e-12:
                    h = cand
                    maxc = val
                    improved = True
                    break

            if improved:
                break

        if improved:
            no_improve = 0
            hm = mirror_project(h, n)
            vm = max_c(hm)
            if vm < maxc:
                h = hm
                maxc = vm
            if maxc < best_c:
                best_c = maxc
                best_h = h.copy()
            if abs(float(h.sum()) - target) > 1e-7:
                h = make_feasible(h, n)
                maxc = max_c(h)
        else:
            no_improve += 1
            step *= 0.85
            if step < 1e-5 or no_improve > 12:
                sigma = 0.03 if n <= 80 else 0.015
                h = make_feasible(best_h + sigma * rng.standard_normal(n), n)
                c = np.correlate(h, 1.0 - h, mode="full")
                maxc = float(np.max(c))
                step = 0.1
                no_improve = 0
                if maxc > best_c + 0.1:
                    h = best_h.copy()
                    maxc = best_c

    return best_h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    global_deadline = t0 + max(1.0, budget - 5.0)

    rng = np.random.default_rng(int(seed))
    phase1_end = t0 + min(450.0, max(60.0, 0.30 * budget))

    schedule1 = (
        (1.0, 1e3),

        (40.0, 1e5),
        (100.0, 1e5),
    )

    for T, mu in schedule1:
        if time.time() >= phase1_end or time.time() >= global_deadline:
            break
        rem = budget - (time.time() - t0)
        if rem < 20.0:
            break
        mi = choose_maxiter(n0, rem)
        if mi <= 0:
            break

        _OPT["deadline"] = min(global_deadline, time.time() + call_limit_for(n0, rem))
        h_opt = optimize_one(best_h, n0, T, mu, mi)
        best_h, best_c = update_best(best_h, best_c, h_opt, n0)

        rem = budget - (time.time() - t0)
        if time.time() < phase1_end and rem > 60.0:
            h_pert = make_feasible(best_h + 0.04 * rng.standard_normal(n0), n0)
            mi2 = max(20, mi // 2)
            _OPT["deadline"] = min(
                global_deadline,
                time.time() + call_limit_for(n0, rem) * 0.6,
            )
            h_opt2 = optimize_one(h_pert, n0, T, mu, mi2)
            best_h, best_c = update_best(best_h, best_c, h_opt2, n0)

    for i in range(8):
        if time.time() >= phase1_end or time.time() >= global_deadline:
            break
        rem = budget - (time.time() - t0)
        if rem < 30.0:
            break
        h_rand = random_feasible(n0, rng, i % 4)
        mi = max(20, choose_maxiter(n0, rem) // 2)
        _OPT["deadline"] = min(
            global_deadline,
            time.time() + call_limit_for(n0, rem) * 0.7,
        )
        h_opt = optimize_one(h_rand, n0, 20.0, 1e5, mi)
        best_h, best_c = update_best(best_h, best_c, h_opt, n0)

    rem = budget - (time.time() - t0)
    if rem > 30.0 and time.time() < global_deadline:
        tlim = hill_time_limit(n0, rem)
        if tlim > 5.0:
            h_hill = hill_climb(best_h, n0, rng, tlim)
            best_h, best_c = update_best(best_h, best_c, h_hill, n0)

    n_cands = []
    for f in (2, 4, 8, 16):
        n2 = int(n0 * f)
        if n2 > n0 and n2 <= 720 and n2 not in n_cands:
            n_cands.append(n2)

    for n2 in n_cands:
        rem = budget - (time.time() - t0)
        if rem < 30.0 or time.time() >= global_deadline:
            break

        h2 = upsample_h(best_h, n2)
        h2m = mirror_project(h2, n2)
        if float(c5_of(h2m, n2)) < float(c5_of(h2, n2)):
            h2 = h2m

        schedule2 = (
            (1.0, 1e3),
            (10.0, 1e4),
            (40.0, 1e5),
        )

        for T, mu in schedule2:
            rem = budget - (time.time() - t0)
            if rem < 20.0 or time.time() >= global_deadline:
                break
            mi = choose_maxiter(n2, rem)
            if mi <= 0:
                break

            _OPT["deadline"] = min(global_deadline, time.time() + call_limit_for(n2, rem))
            h_opt = optimize_one(h2, n2, T, mu, mi)
            best_h, best_c = update_best(best_h, best_c, h_opt, n2)

            c_opt = float(c5_of(make_feasible(h_opt, n2), n2))
            h2f = make_feasible(h2, n2)
            c_cur = float(c5_of(h2f, n2))
            if c_opt < c_cur:
                h2 = make_feasible(h_opt, n2)
            h2m = mirror_project(h2, n2)
            if float(c5_of(h2m, n2)) < float(c5_of(h2, n2)):
                h2 = h2m

        rem = budget - (time.time() - t0)
        if rem > 50.0 and time.time() < global_deadline:
            h_rand = random_feasible(n2, rng, 1)
            mi = max(20, choose_maxiter(n2, rem) // 2)
            _OPT["deadline"] = min(
                global_deadline,
                time.time() + call_limit_for(n2, rem) * 0.7,
            )
            h_opt = optimize_one(h_rand, n2, 20.0, 1e5, mi)
            best_h, best_c = update_best(best_h, best_c, h_opt, n2)
            if float(c5_of(make_feasible(h_opt, n2), n2)) < float(c5_of(h2, n2)):
                h2 = make_feasible(h_opt, n2)

        rem = budget - (time.time() - t0)
        if n2 <= 360 and rem > 30.0 and time.time() < global_deadline:
            tlim = hill_time_limit(n2, rem)
            if tlim > 3.0:
                h_hill = hill_climb(h2, n2, rng, tlim)
                if float(c5_of(make_feasible(h_hill, n2), n2)) < float(c5_of(h2, n2)):
                    h2 = make_feasible(h_hill, n2)
                best_h, best_c = update_best(best_h, best_c, h_hill, n2)

    n_best = int(best_h.size)
    rem = budget - (time.time() - t0)
    if n_best <= 180 and rem > 20.0 and time.time() < global_deadline:
        tlim = min(120.0, max(5.0, rem * 0.4))
        h_hill = hill_climb(best_h, n_best, rng, tlim)
        best_h, best_c = update_best(best_h, best_c, h_hill, n_best)
        n_best = int(best_h.size)

    h_final = make_feasible(best_h, n_best)
    h_snap = snap_to_grid(h_final, n_best)
    c_snap = float(c5_of(h_snap, n_best))
    c_orig = float(c5_of(h_final, n_best))

    h_out = h_snap
    c_out = float(c5_of(h_out, n_best))
    if not math.isfinite(c_out):
        h_out = h_final
        c_out = float(c5_of(h_out, n_best))

    return h_out, c_out, n_best
