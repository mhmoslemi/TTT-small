import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgradient(h, n):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    k_star = int(np.argmax(corr))
    # subgradient of corr[k_star] w.r.t. h
    g = np.zeros(n)
    # c[k] = sum_n h[n+k] * (1-h[n])
    # dc[k]/dh[j] = (1-h[j-k]) * 1[0 <= j-k < n] - h[j+k] * 1[0 <= j+k < n]
    for j in range(n):
        jm = j - k_star
        jp = j + k_star
        if 0 <= jm < n:
            g[j] += (1.0 - h[jm])
        if 0 <= jp < n:
            g[j] -= h[jp]
    return g * dx, k_star

def project(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_subgradient(h0, max_iters, budget_s, start_time):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adaptive step size
    step = 0.1
    patience = 0
    best_patience_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s:
            break
        
        c5_val, g = compute_subgradient(h, n)

        h_new = project(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)

    """Smooth approximation to max"""
    shifted = T * corr - np.max(T * corr)
    return np.max(T * corr) / T + np.log(np.sum(np.exp(shifted))) / T
