import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_hypersimplex(h, n):
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(h - alpha, 0.0, 1.0)
    return result

def smooth_obj_grad(h, T, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    c_max = np.max(corr)
    shifted = np.clip(T * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (c_max + np.log(sum_exp) / T) * dx
    w = T * exp_s / sum_exp
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)
    grad_h = np.zeros(n)
    for k in range(n_corr):
        j = lags[k]
        wk = w[k]
        if wk < 1e-14:
            continue
        if j == 0:
            grad_h += wk * (1.0 - 2.0 * h)
        elif j > 0:
            grad_h[:n - j] += wk * (1.0 - h[j:])
            grad_h[j:] -= wk * h[:n - j]
        else:
            ja = -j
            grad_h[ja:] += wk * (1.0 - h[:n - ja])
            grad_h[:n - ja] -= wk * h[ja:]
    grad_h *= dx
    return obj, grad_h

def subgradient_step(h, dx, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    c_max = np.max(corr)
    max_indices = np.where(corr >= c_max - 1e-12)[0]
    n_corr = 2 * n - 1

    grad_h = np.zeros(n)

        j = lags[idx]
        if j == 0:
            grad_h += (1.0 - 2.0 * h) / n_max
        elif j > 0:
            grad_h[:n - j] += (1.0 - h[j:]) / n_max
            grad_h[j:] -= h[:n - j] / n_max
        else:
            ja = -j
            grad_h[ja:] += (1.0 - h[:n - ja]) / n_max
            grad_h[:n - ja] -= h[ja:] / n_max
    grad_h *= dx
    return h - lr * grad_h

    start_ls = time.time()

            break
        n_change = int(rng.integers(1, 5))
        idx = rng.choice(n, size=n_change, replace=False)
        delta = rng.uniform(-0.08, 0.08, size=n_change)
        new_h = best_h.copy()
        new_h[idx] += delta
        new_h = project_hypersimplex(new_h, n)
        c5 = compute_c5(new_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
    return best_h, best_c5
