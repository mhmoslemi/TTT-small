def subgradient_step(h, active_shifts, n_points):
    n = n_points
    grad = np.zeros(n)
    for k in active_shifts:
        # c_k = sum_{n valid} h[n+k] * (1 - h[n])
        # d(c_k)/d(h[j]) = (1-h[j-k]) * I[0<=j-k<n] + (-h[j+k]) * I[0<=j+k<n]
        # First term: for j where j-k is valid index
        valid_first = np.arange(max(0, k), min(n, n+k))  # j values where j-k is in [0,n-1]
        # Actually: j-k >= 0 and j-k < n, so j >= k and j < n+k, intersected with [0, n-1]
        # So j in [max(0,k), min(n-1, n+k-1)]
        j_start = max(0, k)
        j_end = min(n, n + k)  # exclusive
        for j in range(j_start, j_end):
            grad[j] += (1.0 - h[j - k])
        
        # Second term: j+k valid, so j >= -k and j < n-k
        j_start2 = max(0, -k)
        j_end2 = min(n, n - k)
        for j in range(j_start2, j_end2):
            grad[j] += (-h[j + k])
    
    return grad

            best_obj = obj
            best_h = h.copy()

        grad = compute_gradient(h, corr, n, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            step_size *= 0.9
            h = best_h.copy()
            velocity = np.zeros(n)
        else:
            direction = -grad / gnorm
            velocity = momentum * velocity + direction
            h_try = project_to_constraint(h + step_size * velocity, target_sum)
            try_obj, _ = compute_c5_and_corr(h_try, n)
            if try_obj <= obj:
                h = h_try
                step_size = min(0.08, step_size * 1.02)
            else:
                h_try2 = project_to_constraint(h + 0.3 * step_size * velocity, target_sum)
                try_obj2, _ = compute_c5_and_corr(h_try2, n)
                if try_obj2 <= obj:
                    h = h_try2
                step_size *= 0.8

        tau = min(60.0, tau * 1.0003)

        iters += 1
        if iters % 3000 == 0:
            h = best_h.copy()
            velocity = np.zeros(n)
            step_size = 0.04

    return best_h, best_obj


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n_points = 70
    target_sum = 35.0

    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
    except (NameError, TypeError):
        h = np.full(n_points, 0.5)

    if len(h) != n_points:
        x_old = np.linspace(0, 1, len(h))
        x_new = np.linspace(0, 1, n_points)
        h = np.interp(x_new, x_old, h)

    h = project_to_constraint(h, target_sum)
    best_h = h.copy()
    best_obj, _ = compute_c5_and_corr(h, n_points)

    # Phase 1: Gradient descent (40% of budget)
    phase1_end = start_time + 0.40 * budget_s
    best_h, best_obj = gradient_phase(h, target_sum, n_points, phase1_end)

    # Phase 2: Coordinate + pair-swap refinement (30% of budget)
    phase2_end = start_time + 0.70 * budget_s
    h = best_h.copy()
    while time.time() < phase2_end:
        h, obj = coordinate_refine(h, target_sum, n_points, max_rounds=3,
                                   deadline=phase2_end)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        h, obj = pair_swap_refine(h, target_sum, n_points, rng, max_rounds=2,
                                  deadline=phase2_end)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()

    # Phase 3: Stochastic search (remaining minus buffer)
    phase3_end = start_time + budget_s - 20.0
    if time.time() < phase3_end:
        best_h, best_obj = stochastic_search(best_h, target_sum, n_points, rng, phase3_end)

    # Final polish
    final_h, final_obj = coordinate_refine(best_h, target_sum, n_points, max_rounds=5)
    if final_obj < best_obj:
        best_h = final_h
        best_obj = final_obj

    final_h, final_obj = pair_swap_refine(best_h, target_sum, n_points, rng, max_rounds=3)
    if final_obj < best_obj:
        best_h = final_h
        best_obj = final_obj

    best_h = project_to_constraint(best_h, target_sum)
    best_obj, _ = compute_c5_and_corr(best_h, n_points)

    return (best_h, best_obj, n_points)
