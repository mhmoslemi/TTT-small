            h_new = h - lr * grad
            h_new = project_onto_simplex(h_new, n / 2.0)
            c = evaluate(h_new, n)
            if c < best_local - 1e-12:
                h = h_new
                best_local = c
                stall = 0
                lr = min(lr * 1.02, 0.5)
                update_best(h)
            else:
                stall += 1
                lr *= 0.9
                if stall > 100:
                    break
        
        return h
    
    def optimize_cyclic(h, n, max_sweeps=100):
        """Cyclic coordinate descent with 1D subproblems."""
        h = h.copy().astype(float)
        h = project_onto_simplex(h, n / 2.0)
        update_best(h)
        dx = 2.0 / n
        target = n / 2.0
        no_improve_sweeps = 0
        
        for sweep in range(max_sweeps):
            if time_left() < 10:
                break

            for j in range(n):
                if time_left() < 5:
                    break

                corr = np.correlate(h, 1.0 - h, mode="full") * dx
                A = np.zeros(2 * n - 1)
                for k_idx in range(2 * n - 1):
                    shift = (n - 1) - k_idx
                    val = 0.0
                    if 0 <= j + shift < n:
                        val += (1.0 - h[j + shift])
                    if 0 <= j - shift < n:
                        val -= h[j - shift]
                    A[k_idx] = dx * val
                B = corr - A * h[j]
                
                # Minimize max_k (A_k * t + B_k) over t in [0, 1]
                def obj(t):
                    return np.max(A * t + B)
                
                res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                                      options={'maxiter': 50})
                new_j = res.x
                h[j] = new_j
                h = project_onto_simplex(h, target)
            
            new_best = evaluate(h, n)
            if new_best < sweep_best - 1e-10:
                no_improve_sweeps = 0
                update_best(h)
            else:
                no_improve_sweeps += 1
                if no_improve_sweeps >= 5:
                    break
        
        return h
    
    # Phase 1: Subgradient from initial
    if h0 is not None:
        h1 = h0.copy().astype(float)
        h1 = project_onto_simplex(h1, len(h1) / 2.0)
        h1 = optimize_subgrad(h1, len(h1), max_iters=300)
        h1 = optimize_cyclic(h1, len(h1), max_sweeps=30)
    
    # Phase 2: Random starts at various resolutions
    for n in [30, 45, 60, 90, 120]:

        if time_left() > 20:
            h_polish = optimize_subgrad(h_polish, len(h_polish), max_iters=200)
    
    if best_h is None:
        h_final = np.full(45, 0.5)
    else:
        h_final = best_h.copy()
    
    c5 = evaluate(h_final, len(h_final))
    return h_final, c5, len(h_final)

import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_onto_simplex(h, target_sum):
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    return np.clip(h - theta, 0.0, 1.0)

def compute_grad_topk(h, n, top_k_count=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    top_k = np.argsort(corr)[::-1][:top_k_count]
    grad = np.zeros(n)
    for k_idx in top_k:
        w = 1.0 / len(top_k)
        shift = (n - 1) - k_idx
        for j in range(n):
            t1 = 0.0
            if 0 <= j + shift < n:
                t1 = 1.0 - h[j + shift]
            t2 = 0.0
            if 0 <= j - shift < n:
                t2 = h[j - shift]
            grad[j] += w * dx * (t1 - t2)
    return grad, corr
