def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_constraint(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    n = len(h)
    for _ in range(200):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            count = np.sum(mask)
            if count == 0:
                break
            step = diff / count
            h[mask] = np.maximum(h[mask] + step, 0.0)
        diff = target_sum - np.sum(h)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))

def smooth_max_gradient(h, T=50.0):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = corr - max_c
    weights = np.exp(T * shifted)
    total_w = np.sum(weights)
    if total_w < 1e-30:
        weights = np.zeros_like(corr)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights = weights / total_w
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)

            R = n - 1 - k
        else:

            R = n - 1

            continue
