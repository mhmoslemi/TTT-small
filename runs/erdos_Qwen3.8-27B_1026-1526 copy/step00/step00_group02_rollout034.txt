import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary_h(n, rng):
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    if n % 2 == 1:
        h[half] = 0.5
    idx = rng.permutation(n)
    return h[idx].copy()

def generate_move(h, n, rng, step_size):
    h_new = h.copy()
    move_type = rng.randint(5)
    if move_type == 0:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            return None
        delta = rng.uniform(-step_size, step_size)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < -1e-9 or hi > 1 + 1e-9 or hj < -1e-9 or hj > 1 + 1e-9:
            return None
        h_new[i] = hi
        h_new[j] = hj
    elif move_type == 1:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            return None
        h_new[i], h_new[j] = h[j], h[i]
    elif move_type == 2:
        block_len = rng.randint(1, max(2, n // 6))
        if n - block_len < 2:
            return None
        start = rng.randint(0, n - block_len)
        if start + block_len < n and rng.rand() < 0.5:
            h_new[start + 1:start + block_len + 1] = h[start:start + block_len]
            h_new[start] = h[start + block_len]
        elif start > 0:
            h_new[start - 1:start + block_len - 1] = h[start:start + block_len]
            h_new[start + block_len - 1] = h[start - 1]
        else:
            return None
    elif move_type == 3:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            return None
        delta = rng.uniform(-step_size, step_size)
        hi = h[i] + delta
        hj = h[j] - delta
        if hi < -1e-9 or hi > 1 + 1e-9 or hj < -1e-9 or hj > 1 + 1e-9:
            return None
        h_new[i] = hi
        h_new[j] = hj
        k = rng.randint(n)
        l = rng.randint(n)
        if k == l or k == i or k == j or l == i or l == j:
            return None
        h_new[k], h_new[l] = h[l], h[k]
    else:
        block_len = rng.randint(2, max(3, n // 5))
        if n - block_len + 1 < 2:
            return None
        start = rng.randint(0, n - block_len + 1)
        h_new[start:start + block_len] = h[start:start + block_len][::-1]
    return h_new

def sa_optimize(h0, n, rng, t_end):
    h = h0.copy()
    cur = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = cur
    T = 0.015
    min_T = 1e-8
    cool = 0.99998
    step_size = 0.25
    stall = 0
    max_stall = 3000

    while time.time() < t_end and T > min_T:
        h_new = generate_move(h, n, rng, step_size)
        if h_new is None:
            T *= cool
            continue
        new_c5 = evaluate_c5(h_new)
        if new_c5 <= cur + 1e-12:
            h = h_new
            cur = new_c5
            stall = 0
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
            T = min(T * 1.005, 0.02)
        else:
            diff = new_c5 - cur
            if diff < 8.0 * T + 0.002:
                if rng.rand() < np.exp(-diff / T):
                    h = h_new
                    cur = new_c5
            stall += 1
        T *= cool
        step_size = max(0.03, 0.25 * (T / 0.015))
        if stall > max_stall:
            h = best_h.copy()
            cur = best_c5
            T = max(T, 0.01)
            step_size = 0.2
            stall = 0

    hc_end = min(time.time() + 15, t_end)
    while time.time() < hc_end:
        improved = False
        for _ in range(300):
            h_new = generate_move(h, n, rng, 0.05)
            if h_new is None:
                continue
            new_c5 = evaluate_c5(h_new)
            if new_c5 < cur - 1e-12:
                h = h_new
                cur = new_c5
                improved = True
                if cur < best_c5:
                    best_c5 = cur
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def smooth_c5(v, n, beta):
    dx = 2.0 / n
    corr = np.correlate(v, 1.0 - v, mode="full") * dx
    m = np.max(corr)
    if beta <= 0:
        return m
    return m + np.log(np.sum(np.exp(corr - m))) / beta

def constraint_fn(v, n):
    return v.sum() - n / 2.0

def polish_scipy(h_init, n, t_end):
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn, 'args': (n,)}]
    best = h_init.copy()
    best_val = evaluate_c5(h_init)
    for beta in [20.0, 50.0, 100.0]:
        if time.time() >= t_end:
            break
        def obj(v):
            return smooth_c5(v, n, beta)
        res = minimize(obj, best, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        if res.x is not None:
            cand = np.clip(res.x, 0.0, 1.0)
            cand_val = evaluate_c5(cand)
            if cand_val < best_val:
                best = cand
                best_val = cand_val
    return best, best_val
