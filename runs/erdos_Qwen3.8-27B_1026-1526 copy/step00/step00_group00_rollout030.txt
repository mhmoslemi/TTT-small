def project(h, target_sum):
    """Project h to feasible set: 0<=h[i]<=1, sum(h)=target_sum"""
    h = np.clip(h, 0.0, 1.0)

def _project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if not np.any(mask):
            return h
        frac = diff / max(1, np.sum(~h)) 
        h[mask] = np.minimum(h[mask] + frac, 1.0)
    else:
        mask = h > 0.0
        if not np.any(mask):
            return h
        frac = -diff / max(1, np.sum(h))
        h[mask] = np.maximum(h[mask] - frac, 0.0)
