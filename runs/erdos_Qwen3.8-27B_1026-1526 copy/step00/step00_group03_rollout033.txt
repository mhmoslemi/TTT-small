```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    """Returns (c5_value, all_lags)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr), corr

def compute_grad_lag(h, k, n):
    """Gradient of f_k w.r.t. h, where k is the actual lag (can be negative)"""
    # f_k = sum_{j: 0<=j<n, 0<=j+k<n} h[j] * (1-h[j+k])
    # In full correlation array, this is at index k + n - 1
    # But let's compute directly
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1 - 2 * h
    else:
        # h[i] appears as:
        # - In h[i] factor when j=i: contributes (1-h[i+k]) if 0<=i+k<n
        # - In (1-h[j+k]) factor when j+k=i, i.e., j=i-k: contributes -h[i-k] if 0<=i-k<n
        # Combined: (1-h[i+k])_valid - h[i-k]_valid
        if k > 0:
            # h[i] in first factor: i+k < n means i < n-k
            grad[:n-k] += 1 - h[k:]
            # h[i] in second factor (as 1-h[i]): i-k >= 0 means i >= k
            grad[k:] -= h[:n-k]
        else:
            # k < 0, let kk = -k > 0
            kk = -k
            # h[i] in first factor: i+k < n always (since k<0, i+k < i < n), and i+k >= 0 means i >= -k = kk
            grad[kk:] += 1 - h[:n-kk]
            # h[i] in second factor: i-k >= 0 always (since k<0), and i-k < n means i < n+k = n-kk
            grad[:n-kk] -= h[kk:]
    return grad

def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Need to adjust sum
    if s > target:
        # Reduce: take from elements > 0 proportionally
        excess = s - target
        mask = h > 0
        total = h[mask].sum()
        if total > 0:
            h[mask] -= excess * h[mask] / total
    else:
        # Increase: add to elements < 1 proportionally
        deficit = target - s
        mask = h < 1
        total_space = (1 - h[mask]).sum()
        if total_space > 0:
            h[mask] += deficit * (1 - h[mask]) / total_space
    h = np.clip(h, 0, 1)
    # Final adjustment for any remaining discrepancy
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        diff = target - s
        # Spread equally
        mask = (h > 0) | (h < 1)
        # Better: spread to elements that have room
        if diff > 0:
            mask = h < 1
            space = 1 - h[mask]
            if space.sum() > 0:
                h[mask] += diff * space / space.sum()
        else:
            mask = h > 0
            room = h[mask]
            if room.sum() > 0:
                h[mask] += diff * room / room.sum()
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    else:
        n = 70
        h = np.full(n, 0.5)
    
    best_h = h.copy()
    best_c5, _ = compute_c5_full(best_h)
    
    # Subgradient optimization
    alpha = 0.01
    patience = 0
    max_patience = 500
    
    while time.time() - t0 < budget_s - 30:
        c5, lags = compute_c5_full(h)
        
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Reset h to best and reduce alpha
            h = best_h.copy()
            alpha *= 0.5
            patience = 0
            max_patience = int(max_patience * 1.2)
        
        # Find active lag(s)
        active_lags = np.where(lags >= best_c5 - 1e-4)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(lags)])
        
        # Average gradient over active lags
        grad = np.zeros(n)
        for lag_idx in active_lags:
            k = lag_idx - (n - 1)  # Convert to actual lag
            grad += compute_grad_lag(h, k, n)
        grad /= len(active_lags)
        
        # Gradient step
        h_new = project_feasible(h - alpha * grad, n)
        
        # Check if improved
        new_c5, _ = compute_c5_full(h_new)
        if new_c5 < c5:
            h = h_new
        else:
            # Try smaller step

            new_c5, _ = compute_c5_full(h_new)
            if new_c5 < c5:

        if patience > max_patience // 2:
            perturb = np.random.randn(n) * 0.05

    return best_h, best_c5, n
```

```python
def project_sum_box(h, n, target):
    """Project h onto {0<=h_i<=1, sum h_i = target}"""
    # Binary search on lambda
    lo, hi = -max(h) - 1, max(1 - h) + 1

        return np.sum(np.clip(h - lam, 0, 1))
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)
```

For a binary sequence, a swap (move a 1 from position i to position j) changes the autocorrelation as follows:
- For each lag k, the autocorrelation changes by:

  - +1 for lags where h[j+k]=1 or h[j-k]=1 (adding a 1 at j)
  - Net change in autocorr_k = (# of 1s at distance k from j) - (# of 1s at distance k from i)

So I can precompute for each position the "autocorrelation profile" and then evaluate a swap in O(1) per lag.

```python
def swap_gain(h, i, j, n):
    """Gain in C5 from moving 1 at position i to position j (where h[i]=1, h[j]=0)"""
    # Change in autocorr_k for each k:
    # removing 1 at i: -1 for each k where h[i+k]=1 (k>0) or h[i-k]=1 (k<0) or k=0 contributes -1
    # adding 1 at j: +1 for each k where h[j+k]=1 or h[j-k]=1
    # Net change in autocorr_k: count_ones_near(j, k) - count_ones_near(i, k)
    # where count_ones_near(p, k) = h[p+k] + h[p-k] (with bounds)
    
    # New C5 = max over k of (35 - new_autocorr_k) * dx
    # = max over k of (35 - old_autocorr_k - delta_k) * dx
    # where delta_k = count_ones_near(j,k) - count_ones_near(i,k)
    
    # We want this to be < current C5
    # i.e., for all k: 35 - old_autocorr_k - delta_k <= old_max / dx - epsilon
```

```python
def compute_c5_full(h):
```

```python
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

        else:

def binary_eval(h, n):
    """Evaluate C5 for binary h"""
    dx = 2.0 / n
    n_half = n // 2
    autocorr = np.correlate(h, h, mode="full")
    # C5 = max_k (n_half - autocorr_k) * dx

    # And f_k = sum_i h[i]*(1-h[i+k]) = (number of i where h[i]=1, h[i+k]=0)

    # = n_half - autocorr_k (for valid i)

    gaps = n_half - autocorr

def binary_swap_delta(h, i, j, n):

    # delta in autocorr_k = change in autocorr
    # delta in gap_k = -delta in autocorr_k
    d = np.zeros(2*n-1)

            dc += h[j+k]
        if 0 <= j-k < n and j-k != i:
            dc += h[j-k]

        if 0 <= i+k < n:
            dc -= h[i+k]
        if 0 <= i-k < n:
            dc -= h[i-k]
        d[n-1+k] = dc
```

```python
import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_sum_box(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_lag(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        kk = -lag
        g[kk:] += 1.0 - h[:n - kk]
        g[:n - kk] -= h[kk:]
    return g

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 70

            for p in order[:num_ones - n_half]:

        elif num_ones < n_half:

                h_bin[zero_idx[p]] = 1.0
        
        # Verify

                    best_bin[j] = 1
                    c5 = compute_c5_full(best_bin)[0]
                    best_bin[i] = 1
                    best_bin[j] = 0
                    
                    if c5 < best_move_c5 - 1e-12:

                best_c5 = c5_try
                best_h = h_try
    
    return best_h, best_c5, n
```
