    return np.correlate(h, 1 - h, mode="full")

def smooth_max(vals, temp):
    shifted = vals / temp - np.max(vals) / temp
    exps = np.exp(shifted)
    return temp * (np.max(vals) / temp + np.log(np.sum(exps)))

def smooth_max_grad(vals, temp):
    shifted = vals / temp - np.max(vals) / temp
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    return weights

def c5_gradient(h, dx, temp):
    n = len(h)
    c = compute_correlation(h) * dx
    weights = smooth_max_grad(c, temp)

    # grad c[k] / dh[p] = dx * [(1-h[p-k+n-1]) * I(valid) - h[p+k-n+1] * I(valid)]
