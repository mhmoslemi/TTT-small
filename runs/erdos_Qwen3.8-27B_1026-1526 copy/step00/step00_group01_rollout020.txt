    return np.correlate(h, 1.0 - h, mode='full')

def c5_value(h, dx):
    corr = compute_corr(h) * dx
    return np.max(corr)

def soft_max_c5_and_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr(h) * dx
    max_c = np.max(corr)
    # Soft max
    shifted = corr - max_c
    exp_vals = np.exp(tau * shifted)
    sum_exp = np.sum(exp_vals)
    c5_val = max_c + np.log(sum_exp) / tau
    weights = exp_vals / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for L in range(2 * n - 1):
        if weights[L] < 1e-12:
            continue
        w = weights[L] * dx
        m = L - (n - 1)  # offset
        if m >= 0:
            # c_L = dx * sum_{i=0}^{n-1-m} h[i] * (1 - h[i+m])
            for i in range(n - m):
                grad[i] += w * (1.0 - h[i + m])
                if m > 0:
                    grad[i + m] += w * (-h[i])
        else:
            p = -m

            for i in range(p, n):
                grad[i] += w * (1.0 - h[i - p])
                grad[i - p] += w * (-h[i])
    return c5_val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_hypersimplex(h, alpha):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > alpha:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return proj

def soft_max_c5_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_vals = np.exp(tau * shifted)
    sum_exp = np.sum(exp_vals)
    c5_val = max_c + np.log(sum_exp) / tau
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for L in range(2 * n - 1):
        w = weights[L]
        if w < 1e-10:
            continue
        wdx = w * dx
        m = L - (n - 1)
        if m >= 0:
            start = np.arange(0, n - m)
            grad[start] += wdx * (1.0 - h[start + m])
            if m > 0:
                grad[start + m] -= wdx * h[start]
        else:
            p = -m
            start = np.arange(p, n)
            grad[start] += wdx * (1.0 - h[start - p])
            grad[start - p] -= wdx * h[start]
    return c5_val, grad

def binary_c5_numpy(h_bin, n):
    dx = 2.0 / n
    corr = np.correlate(h_bin.astype(np.float64), (1.0 - h_bin).astype(np.float64), mode='full')
    return np.max(corr) * dx

def local_search_binary(S, n, max_iters=5000, rng=None):
    S = set(S)
    h = np.zeros(n, dtype=np.int8)
    for i in S:
        h[i] = 1
    best_c5 = binary_c5_numpy(h, n)
    not_S = set(range(n)) - S
    for it in range(max_iters):
        improved = False
        best_swap = None
        best_swap_c5 = best_c5
        S_list = list(S)
        NS_list = list(not_S)
        for i in S_list:
            for j in NS_list:
                h[i] = 0
                h[j] = 1
                c5 = binary_c5_numpy(h, n)
                h[j] = 0
                h[i] = 1
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            S.discard(i)
            S.add(j)
            not_S.discard(j)
            not_S.add(i)
            best_c5 = best_swap_c5
            improved = True
        if not improved:
            break
    return S, best_c5, h

def simulated_annealing_binary(S, n, max_iters=20000, T_start=0.05, T_end=0.001, rng=None):
    S = set(S)
    h = np.zeros(n, dtype=np.int8)
    for i in S:
        h[i] = 1
    current_c5 = binary_c5_numpy(h, n)
    best_c5 = current_c5
    best_h = h.copy()
    not_S = set(range(n)) - S
    n_trials = 100
    for it in range(max_iters):
        progress = it / max(max_iters - 1, 1)
        T = T_start * (T_end / T_start) ** progress
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h[i] = 1 - h[i]
            h[j] = 1 - h[j]
            if np.sum(h) != n // 2:
                h[i] = 1 - h[i]
                h[j] = 1 - h[j]
                continue
            new_c5 = binary_c5_numpy(h, n)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                current_c5 = new_c5
                S.discard(i)
                S.add(j)
                not_S.discard(j)
                not_S.add(i)
                S.discard(j)
                S.add(i)
                not_S.discard(i)
                not_S.add(j)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1 - h[i]
                h[j] = 1 - h[j]
    return best_h, best_c5

def continuous_refine(h0, n, dx, budget_per_call=60.0):
    target_sum = n / 2.0
    start = time.time()
    h = project_hypersimplex(h0.astype(np.float64), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for tau in [20.0, 50.0, 100.0, 200.0]:
        if time.time() - start > budget_per_call:
            break
        def obj(x):
            hh = project_hypersimplex(x, target_sum)
            val, _ = soft_max_c5_grad(hh, dx, tau)
            return val
        def obj_grad(x):
            hh = project_hypersimplex(x, target_sum)
            val, g = soft_max_c5_grad(hh, dx, tau)
            return val, g
        try:
            res = minimize(obj_grad, h, method='L-BFGS-B', options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10})
            h_new = project_hypersimplex(res.x, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5
