def generate_msequence(degree, poly):
    """Generate m-sequence using LFSR"""
    # poly is the tap positions (0-indexed from the right)
    # e.g., for x^6 + x + 1: taps at positions 0 and 1 (feedback to bit 0 and bit 1)
    n = 2**degree - 1
    # Initialize LFSR
    state = 1  # non-zero initial state
    seq = []
    for _ in range(n):
        seq.append(state & 1)
        # Compute feedback
        fb = 0
        for tap in poly:
            fb ^= (state >> tap) & 1
        state = (state >> 1) | (fb << (degree - 1))
    return np.array(seq)
