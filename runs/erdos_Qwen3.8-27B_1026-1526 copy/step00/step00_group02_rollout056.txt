        val = compute_c5(self.h) + self.penalty_w * (s - self.target) ** 2
        self.h[self.i] = old
        return val

        penalty_w = 10.0 + 490.0 * frac

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return dx * np.max(corr)

class C5Objective:
    def __init__(self, h, i, target, penalty_w):
        self.h = h
        self.i = i
        self.target = target
        self.penalty_w = penalty_w
    def __call__(self, t):
        old = self.h[self.i]
        self.h[self.i] = t
        s = np.sum(self.h)
        val = compute_c5(self.h) + self.penalty_w * (s - self.target) ** 2
        self.h[self.i] = old
        return val
