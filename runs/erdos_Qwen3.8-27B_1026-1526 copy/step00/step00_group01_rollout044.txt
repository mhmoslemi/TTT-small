def project_feasible(h, target_sum):

            adj = min(adj, (1.0 - proj[free]).min() - 1e-12)
            proj[free] += adj
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def c5_gradient_vec(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
