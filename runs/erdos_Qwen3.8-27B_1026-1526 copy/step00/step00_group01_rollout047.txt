                    best_c5 = c5
                    improved = True
                    pairs_done += 1
                    break
            if improved:
                break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()

    if parent is not None:
        best_h = parent.copy()
        best_n = len(best_h)
        best_c5 = evaluate_c5(best_h)
    else:
        best_n = 47
        best_h = make_binary_start(best_n, rng)
        best_c5 = evaluate_c5(best_h)

    def remaining():
        return budget_s - 25 - (time.time() - t0)

    if remaining() > 50:
        h_opt, c5_v = optimize_slsqp(best_h, best_n, beta=15, max_iter=200)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt

    if remaining() > 50:
        h_opt, c5_v = optimize_slsqp(best_h, best_n, beta=60, max_iter=200)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt

    if remaining() > 30:
        h_opt, c5_v = optimize_slsqp(best_h, best_n, beta=200, max_iter=100)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt

    if remaining() > 15:
        h_bin, c5_bin = try_binary_round(best_h, best_n)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        h_ls, c5_ls = local_flip_search(best_h, best_n, max_flips=80)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    n_list = [94, 93, 95, 187, 189, 191]
    for n_try in n_list:
        if remaining() < 50:
            break
        h0 = upsample(best_h, n_try)
        h0 = project_sum(h0, n_try)
        h_opt, c5_v = optimize_slsqp(h0, n_try, beta=40, max_iter=120)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt
            best_n = n_try
        if remaining() > 20:
            h_bin, c5_bin = try_binary_round(best_h, len(best_h))
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin
        if remaining() > 15:
            h_ls, c5_ls = local_flip_search(best_h, len(best_h), max_flips=50)
            if c5_ls < best_c5:
                best_h, best_c5 = h_ls, c5_ls

    n_cur = len(best_h)
    restart_count = 0
    while remaining() > 35 and restart_count < 30:
        h0 = best_h + rng.randn(n_cur) * 0.06
        h0 = project_sum(h0, n_cur)
        h_opt, c5_v = optimize_slsqp(h0, n_cur, beta=80, max_iter=80)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt
        h_bin, c5_bin = try_binary_round(best_h, n_cur)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
        if remaining() > 10:
            h_ls, c5_ls = local_flip_search(best_h, n_cur, max_flips=40)
            if c5_ls < best_c5:
                best_h, best_c5 = h_ls, c5_ls
        restart_count += 1

    for _ in range(10):
        if remaining() < 30:
            break
        h_rand = make_binary_start(n_cur, rng)
        h_opt, c5_v = optimize_slsqp(h_rand, n_cur, beta=60, max_iter=100)
        if c5_v < best_c5:
            best_c5, best_h = c5_v, h_opt

    if remaining() > 20:
        h_ls, c5_ls = local_pair_search(best_h, len(best_h), max_pairs=50)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_h = project_sum(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
