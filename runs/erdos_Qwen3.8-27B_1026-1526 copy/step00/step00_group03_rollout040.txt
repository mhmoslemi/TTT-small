    target = n_points / 2.0

    for _ in range(100):

        if abs(s - target) < 1e-10:

        h = np.clip(h + diff, 0, 1)
    # Final adjustment: distribute remaining mass
    s = np.sum(h)

    if remaining > 0:

        total_slack = np.sum(slack)

    upper = t_hi
    # For a > 0: upper bound on t

Now let me think about the coordinate descent more carefully.

For coordinate j, the lags ℓ range from -(n-1) to (n-1), total 2n-1 lags.

The coefficient a_ℓ of h[j] in f_ℓ is:
- (1-h[j+ℓ])*dx if 0 ≤ j+ℓ < n, else dx (since h[out]=0, so 1-0=1)
- minus h[j-ℓ]*dx if 0 ≤ j-ℓ < n, else 0

So a_ℓ = [(1-h[j+ℓ]) if in_range(j+ℓ) else 1] * dx - [h[j-ℓ] if in_range(j-ℓ) else 0] * dx

And the constant c_ℓ = f_ℓ - a_ℓ * h[j] (current value)

Let me precompute f_ℓ for all ℓ using np.correlate, then for each j, compute a_ℓ and c_ℓ.

Actually, I realize there's a subtlety. When I update h[j], the f_ℓ values change for all ℓ (since h[j] appears in f_ℓ for every ℓ). So after each coordinate update, I need to update all f_ℓ.

f_ℓ new = f_ℓ old + a_ℓ * (h[j]_new - h[j]_old)

This is O(n) per coordinate update (to update all 2n-1 f_ℓ values).

So the total cost per pass is:
- Initial computation of all f_ℓ: O(n²) (using correlate)
- For each of n coordinates:
  - Compute a_ℓ: O(n)
  - Solve 1D problem: O(n * 50) ≈ O(n) (with vectorized feasibility check)
  - Update f_ℓ: O(n)
- Total: O(n² + n² * 50) ≈ O(50 * n²)

For n=70: 50 * 4900 = 245,000 operations per pass. With 1000 passes, that's 245 million operations. In numpy, this should be well under a second per pass, so 1000 passes would take maybe 1-2 seconds.

import time

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_feasible(h, n):
    target = n / 2.0

            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_all_f(h, n):
    """Compute f_l for all lags l = -(n-1), ..., n-1"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # shape (2n-1,), index corresponds to l = idx - (n-1)

def coordinate_descent(h, n, max_passes=1000, time_limit=None, start_time=0):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    best_val = evaluate_c5(h, n)
    h = h.copy()
    
    for pass_num in range(max_passes):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Compute all f_l
        f = compute_all_f(h, n)
        
        improved = False
        for j in range(n):
            # Compute a_l for all lags
            # a_l = [(1-h[j+l])*dx if in_range(j+l) else dx] - [h[j-l]*dx if in_range(j-l) else 0]
            # lags l range from -(n-1) to (n-1)
            
            # For lag l (0-indexed in the f array as l + n - 1):
            # a[l_idx] where l = l_idx - (n-1)
            
            # Let me compute using lags from -(n-1) to (n-1)
            lags = np.arange(-(n-1), n)  # shape (2n-1,)
            
            # j + l: valid if 0 <= j+l < n
            jp = j + lags  # shape (2n-1,)
            jn = j - lags  # shape (2n-1,)
            
            # term1: (1 - h[j+l]) if 0 <= j+l < n, else 1
            jp_valid = (jp >= 0) & (jp < n)
            term1 = np.ones(n_lags)
            term1[jp_valid] = 1.0 - h[jp[jp_valid]]
            
            # term2: h[j-l] if 0 <= j-l < n, else 0
            jn_valid = (jn >= 0) & (jn < n)
            term2 = np.zeros(n_lags)
            term2[jn_valid] = h[jn[jn_valid]]
            
            a = (term1 - term2) * dx  # shape (2n-1,)
            c = f - a * h[j]  # shape (2n-1,)
            
            # Solve: minimize max_l (a_l * t + c_l) over t in [0, 1]
            t_opt, V_opt = solve_1d_minimax(a, c)
            
            if abs(t_opt - h[j]) > 1e-10:
                # Update h[j]
                delta = t_opt - h[j]
                h[j] = t_opt
                f = f + a * delta
                improved = True
        
        # After full pass, check if sum constraint is violated
        # (coordinate descent should maintain it if we're careful, but let's check)
        current_val = float(np.max(f))
        if current_val < best_val - 1e-10:
            best_val = current_val
        
        if not improved:
            break
    
    return h, best_val

    """Minimize max_i (a[i]*t + c[i]) over t in [t_lo, t_hi]
    Returns (t_opt, V_opt)"""

    lo_vals = np.where(a >= 0, a * t_lo + c, a * t_hi + c)
    V_lo = float(np.max(lo_vals))
    # Upper bound
    V_hi = float(np.max(np.maximum(a * t_lo + c, a * t_hi + c)))

    for _ in range(60):

        # Feasibility check: exists t in [t_lo, t_hi] s.t. a_i*t + c_i <= V_mid for all i
        lower = t_lo
        upper = t_hi
        
        pos = a > 1e-14

            V_lo = V_mid
        else:
            V_hi = V_mid
            t_result = (lower + upper) * 0.5

Wait, I realize there's an issue with the coordinate descent and the sum constraint. When I update h[j], the sum changes. I need to either:
1. After each update, re-project (expensive)
2. Incorporate the sum constraint into the 1D subproblem

Option 2 is better. The 1D subproblem becomes:
minimize max_ℓ (a_ℓ * t + c_ℓ)
subject to: 0 ≤ t ≤ 1
            sum_{i≠j} h[i] + t = n/2  →  t = n/2 - sum_{i≠j} h[i]

Wait, that fixes t entirely! So coordinate descent doesn't work directly with the sum constraint.

Hmm, that's a problem. Let me think about alternatives.

Option A: Drop the sum constraint during optimization and project at the end.
- This might work if the natural optimum satisfies sum ≈ n/2
- But it's risky

Option B: Use a Lagrange multiplier for the sum constraint.
- Penialty: add λ * (sum(h) - n/2) to the objective
- Or use augmented Lagrangian

Option C: Use a different parameterization that automatically satisfies the sum constraint.
- E.g., h = P(f) where P is a projection

Option D: Use pair swaps instead of individual coordinate updates.
- Swap: increase h[j] by δ and decrease h[k] by δ
- This maintains the sum constraint
- The 2D subproblem: minimize max_ℓ f_ℓ(h[j]+δ, h[k]-δ) over δ
- This is still a 1D problem in δ!

Option D looks promising! Let me go with pair swaps.

For a pair (j, k) and a swap amount δ:
h[j] → h[j] + δ
h[k] → h[k] - δ

Constraints: 0 ≤ h[j]+δ ≤ 1, 0 ≤ h[k]-δ ≤ 1
→ δ ∈ [max(-h[j], h[k]-1), min(1-h[j], h[k])]

The change in f_ℓ:
Δf_ℓ = a_ℓ^j * δ + a_ℓ^k * (-δ) = (a_ℓ^j - a_ℓ^k) * δ

where a_ℓ^j is the coefficient of h[j] in f_ℓ (computed as before).

So the 1D subproblem is:
minimize max_ℓ (f_ℓ + (a_ℓ^j - a_ℓ^k) * δ)
subject to δ ∈ [δ_min, δ_max]

This is the same type of problem as before! I can use the same solve_1d_minimax function.

The advantage is that the sum is automatically preserved.

Algorithm:
1. For each pair (j, k) [or a random subset of pairs]:
   a. Compute a^j and a^k (coefficients of h[j] and h[k] in each f_ℓ)
   b. d = a^j - a^k
   c. Solve 1D minimax: minimize max_ℓ (f_ℓ + d_ℓ * δ) over δ ∈ [δ_min, δ_max]
   d. If improvement, update h[j] += δ, h[k] -= δ, f += d * δ
2. Repeat

For efficiency, I don't need to try all O(n²) pairs. I can:

- Or try "greedy" pairs: for the active lag ℓ*, find the pair that most reduces f_ℓ*

def subgradient_descent(h, n, max_iters=10000, time_limit=None, start_time=0):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    h = h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h, n)
    
    step = 0.5  # initial step size
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        f = compute_all_f(h, n)
        max_f = np.max(f)
        
        if max_f < best_val:
            best_val = max_f
            best_h = h.copy()
        
        # Find active lags
        tol = 1e-6
        active = np.where(f >= max_f - tol)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for l_idx in active:
            l = lags[l_idx]
            for j in range(n):
                # a_l^j = (1-h[j+l])*dx (if valid else dx) - h[j-l]*dx (if valid else 0)
                jp = j + l
                jn = j - l
                t1 = (1.0 - h[jp]) if (0 <= jp < n) else 1.0
                t2 = h[jn] if (0 <= jn < n) else 0.0
                g[j] += (t1 - t2) * dx
        g /= len(active)
        
        # Take step
        h = h - step * g
        h = project_onto_feasible(h, n)
        
        # Decrease step size
        step *= 0.999
        
        if step < 1e-6:
            break
    
    return best_h, best_val

import numpy as np
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, n):
    target = n / 2.0
    lo = -float(np.max(h)) - 1.0
    hi = float(np.max(1.0 - h)) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_all_f(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def solve_1d_minimax(a, c, t_lo, t_hi):
    """Minimize max_i (a[i]*t + c[i]) over t in [t_lo, t_hi].
    Returns (t_opt, V_opt)."""
    if t_lo >= t_hi:
        return t_lo, float(np.max(a * t_lo + c))
    
    # Bounds on V
    # For each line a_i*t + c_i, min over [t_lo, t_hi]:
    #   if a_i >= 0: min at t_lo → a_i*t_lo + c_i
    #   if a_i < 0: min at t_hi → a_i*t_hi + c_i
    lo_vals = np.where(a >= 0, a * t_lo + c, a * t_hi + c)
    V_lo = float(np.max(lo_vals))
    hi_vals = np.maximum(a * t_lo + c, a * t_hi + c)
    V_hi = float(np.max(hi_vals))
    
    if V_hi - V_lo < 1e-14:
        t_mid = (t_lo + t_hi) * 0.5
        return t_mid, V_lo
    
    t_result = (t_lo + t_hi) * 0.5
    for _ in range(50):
        V_mid = (V_lo + V_hi) * 0.5
        lower = t_lo
        upper = t_hi
        
        pos = a > 1e-15
        neg = a < -1e-15
        zero = ~pos & ~neg
        
        if np.any(zero) and float(np.max(c[zero])) > V_mid + 1e-15:
            V_lo = V_mid
            continue
        
        if np.any(pos):
            ub = float(np.min((V_mid - c[pos]) / a[pos]))
            if ub < upper:
                upper = ub
        
        if np.any(neg):
            lb = float(np.max((V_mid - c[neg]) / a[neg]))
            if lb > lower:
                lower = lb
        
        if lower > upper + 1e-15:
            V_lo = V_mid
        else:
            V_hi = V_mid
            t_result = (lower + upper) * 0.5
    
    return t_result, V_hi

def pair_swap_optimize(h, n, max_passes=200, time_limit=None, start_time=0, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)
    h = h.copy()
    f = compute_all_f(h, n)
    best_val = float(np.max(f))
    best_h = h.copy()
    
    def compute_A(h):
        A = np.zeros((n_lags, n))
        js = np.arange(n)
        for l_idx in range(n_lags):
            l = lags[l_idx]
            jp = js + l
            jn = js - l
            jp_valid = (jp >= 0) & (jp < n)
            jn_valid = (jn >= 0) & (jn < n)
            term1 = np.ones(n)
            term1[jp_valid] = 1.0 - h[jp[jp_valid]]
            term2 = np.zeros(n)
            term2[jn_valid] = h[jn[jn_valid]]
            A[l_idx, :] = (term1 - term2) * dx
        return A
    
    A = compute_A(h)
    
    for pass_num in range(max_passes):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        improved = False
        n_tries = n * 3
        for _ in range(n_tries):
            # Pick a lag to focus on (weighted by how "active" it is)
            max_f = float(np.max(f))
            # Use all lags above some threshold
            threshold = max_f - 0.01 * max_f
            candidates = np.where(f >= threshold)[0]
            if len(candidates) == 0:
                candidates = np.array([np.argmax(f)])
            l_idx = candidates[rng.randint(len(candidates))]
            
            # Find j with min A[l_idx, j] and k with max
            sorted_idx = np.argsort(A[l_idx, :])
            j = int(sorted_idx[0])
            k = int(sorted_idx[-1])
            if j == k:
                continue
            
            d = A[:, j] - A[:, k]
            
            # Feasible range
            delta_min = max(-h[j], h[k] - 1.0)
            delta_max = min(1.0 - h[j], h[k])
            if delta_min >= delta_max - 1e-12:
                continue
            
            t_opt, V_opt = solve_1d_minimax(d, f, delta_min, delta_max)
            
            if abs(t_opt) > 1e-10:
                h[j] += t_opt
                h[k] -= t_opt
                f += d * t_opt
                # Update A (just recompute the affected parts)
                # For simplicity, recompute A periodically
                improved = True
        
        # Recompute A every pass
        A = compute_A(h)
        
        current_val = float(np.max(f))
        if current_val < best_val - 1e-12:
            best_val = current_val
            best_h = h.copy()
    
    return best_h, best_val

            jn_valid = (jn >= 0) & (jn < n)
            term1 = np.ones(n)
            term1[jp_valid] = 1.0 - h[jp[jp_valid]]
            term2 = np.zeros(n)
            term2[jn_valid] = h[jn[jn_valid]]
            a_col = (term1 - term2) * dx  # a_l^j for all j
            
            # Pick j from bottom, k from top
            sorted_idx = np.argsort(a_col)
            # Pick from top 3 / bottom 3 for diversity
            j = int(sorted_idx[rng.randint(3)])
            k = int(sorted_idx[-1 - rng.randint(3)])
            if j == k:
                continue
            
            # Full d vector: change in each f_l per unit δ
            # d[l2] = a_{l2}^j - a_{l2}^k
            # Need to compute this for all l2
            # This requires computing a_{l2}^j and a_{l2}^k for all l2
            # a_{l2}^j = (1-h[j+l2])*dx (if valid) - h[j-l2]*dx (if valid)
            
            # Vectorized over l2:
            l2 = lags  # shape (n_lags,)
            jp_j = j + l2  # j + l2 for each lag l2
            jn_j = j - l2
            jp_j_valid = (jp_j >= 0) & (jp_j < n)
            jn_j_valid = (jn_j >= 0) & (jn_j < n)
            t1_j = np.ones(n_lags)
            t1_j[jp_j_valid] = 1.0 - h[jp_j[jp_j_valid]]
            t2_j = np.zeros(n_lags)
            t2_j[jn_j_valid] = h[jn_j[jn_j_valid]]
            a_j = (t1_j - t2_j) * dx
            
            jp_k = k + l2
            jn_k = k - l2
            jp_k_valid = (jp_k >= 0) & (jp_k < n)
            jn_k_valid = (jn_k >= 0) & (jn_k < n)
            t1_k = np.ones(n_lags)
            t1_k[jp_k_valid] = 1.0 - h[jp_k[jp_k_valid]]
            t2_k = np.zeros(n_lags)
            t2_k[jn_k_valid] = h[jn_k[jn_k_valid]]
            a_k = (t1_k - t2_k) * dx
            
            d = a_j - a_k
            
            delta_min = max(-h[j], h[k] - 1.0)
            delta_max = min(1.0 - h[j], h[k])
            if delta_min >= delta_max - 1e-12:
                continue
            
            t_opt, V_opt = solve_1d_minimax(d, f, delta_min, delta_max)
            
            if abs(t_opt) > 1e-10:
                h[j] += t_opt
                h[k] -= t_opt
                f += d * t_opt
                improved = True
        
        current_val = float(np.max(f))
        if current_val < best_val - 1e-12:
            best_val = current_val
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_val
