    """Optimize h[i] and h[j] while keeping h[i]+h[j] constant."""
    s = h[i] + h[j]
    if s <= 0 or s >= 2:
        return h
    
    t_low = max(0, s - 1)
    t_high = min(1, s)
    
    if t_low == t_high:
        return h
    
    # For each correlation index k, compute the coefficient of t
    # c_k = dx * sum_j' h[j'] * (1 - h[j'+offset])

    # The change in c_k is: dx * [contribution from i + contribution from j]

    #            + h[i_new]*(1-h[i_new+off if valid]) ... this is getting complicated

    # and pick the best
    
    n = len(h)
    dx = 2.0 / n
    best_t = h[i]
    best_val = np.inf
    
    # Grid search for t
    n_grid = 50

        h[i] = t
        h[j] = s - t
        val = max_overlap(h, dx)
        if val < best_val:

            best_t = t

    h[j] = s - best_t

    corr = np.correlate(h, 1 - h, mode='full')

import numpy as np

from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def repair_h(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(len(h), target_sum / len(h))
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h, max_iter=100):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
        return repair_h(res.x, target_sum)
    except Exception:
        return h.copy()


def paired_cd(h, max_time=600.0, n_grid=30, seed=42):

    n = len(h)

    start = time.time()
    stagnant = 0
    max_stagnant = 40
    grid_size = n_grid

    while time.time() - start < max_time and stagnant < max_stagnant:
        sweep_best = best
        order = rng.permutation(n)
        for idx in range(n):
            i = order[idx]
            j = rng.randint(n)
            while j == i:
                j = rng.randint(n)
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            grid = np.linspace(lo, hi, grid_size)
            if h[i] < lo + 1e-12 or h[i] > hi - 1e-12:
                grid = np.concatenate([[h[i]], grid])
            best_t = h[i]
            best_v = sweep_best
            for t in grid:
                h[i] = t
                h[j] = s - t
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_t = t
            h[i] = best_t
            h[j] = s - best_t
            if best_v < sweep_best - 1e-14:

        if sweep_best < best - 1e-13:
            best = sweep_best
            stagnant = 0

        else:
            stagnant += 1
            grid_size = max(20, grid_size - 5)
