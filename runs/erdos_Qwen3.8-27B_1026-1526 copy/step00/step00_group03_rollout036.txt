```python
def build_lp_constraint_matrix(n, g):
    """Build the constraint matrix for the LP step"""
    # For each lag k (0 to 2n-2), the constraint is:
    # sum_i h[i] * g[i - m_k] <= t
    # where m_k = k - (n-1)
    # Variable vector: x = [h_0, h_1, ..., h_{n-1}, t]
    # Constraint row: [g[0-m_k] (if valid), g[1-m_k] (if valid), ..., g[(n-1)-m_k] (if valid), -1]
    
    n_rows = 2 * n - 1
    A = np.zeros((n_rows, n + 1))
    for k in range(n_rows):
        m = k - (n - 1)
        for i in range(n):
            jm = i - m
            if 0 <= jm < n:
                A[k, i] = g[jm]
        A[k, n] = -1.0  # coefficient of t
    return A
```

def project_to_feasible(v, n):
    target = n / 2.0
    v = np.asarray(v, dtype=float)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = np.max(corr)
    tol = max(1e-10, 0.005 * F)
    active = np.where(corr >= F - tol)[0]
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in active:

        vm = idx - m
        valid1 = (vm >= 0) & (vm < n)
        grad[valid1] += (1.0 - h[vm[valid1]])
        vm2 = idx + m
        valid2 = (vm2 >= 0) & (vm2 < n)
        grad[valid2] += -h[vm2[valid2]]
    grad /= max(len(active), 1)
    return grad * dx
