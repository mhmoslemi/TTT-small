                a[mask2] += -h[np.clip(j2, 0, n-1)]

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h):
    n = len(h)
    h = h.copy()
    target = n / 2.0
    for _ in range(100):
        deficit = target - np.sum(h)
        if abs(deficit) < 1e-13:
            break
        if deficit > 0:
            mask = (h < 1.0 - 1e-14)
        else:
            mask = (h > 1e-14)
        if not np.any(mask):
            break
        step = deficit / float(np.sum(mask))
        h[mask] += step
        np.clip(h, 0.0, 1.0, out=h)
    return h

def coordinate_descent(h, max_sweeps=10000, time_budget=100.0, start_time=None):
    n = len(h)
    h = h.copy()
    old_c5 = compute_c5(h)
    lags = np.arange(2 * n - 1) - (n - 1)
    two_n_minus_1 = 2 * n - 1

    for sweep in range(max_sweeps):
        if start_time is not None and time.time() - start_time > time_budget:
            break
        improved = False
        for i in range(n):
            w = 1.0 - h
            corr = np.correlate(h, w, mode='full')
            j1 = i - lags
            j2 = i + lags
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            a = np.zeros(two_n_minus_1)
            if np.any(mask1):
                a[mask1] = 1.0 - h[np.clip(j1, 0, n - 1)]
            if np.any(mask2):
                a[mask2] += -h[np.clip(j2, 0, n - 1)]
            b = corr - a * h[i]
            candidates = np.array([0.0, 1.0])
            mask = np.abs(a) > 1e-14
            if np.any(mask):
                new_cands = -b[mask] / a[mask]
                new_cands = new_cands[(new_cands >= -1e-9) & (new_cands <= 1.0 + 1e-9)]
                if len(new_cands) > 0:
                    new_cands = np.clip(new_cands, 0.0, 1.0)
                    candidates = np.concatenate([candidates, new_cands])
            vals = np.max(a[None, :] * candidates[:, None] + b[None, :], axis=1)
            best_x = candidates[np.argmin(vals)]
            if abs(best_x - h[i]) > 1e-13:
                h[i] = best_x
                improved = True
        h = project_to_feasible(h)
        new_c5 = compute_c5(h)
        if not improved and abs(new_c5 - old_c5) < 1e-14:
            break
        if new_c5 < old_c5 - 1e-15:
            old_c5 = new_c5
    return h, compute_c5(h)

def smoothed_obj_grad(x, tau, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(tau * (corr - m))
    Z = np.sum(exp_vals)
    p = exp_vals / Z
    obj = m + np.log(Z) / tau + 1000.0 * (np.sum(x) - n / 2.0) ** 2
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    penalty_grad = 2000.0 * (np.sum(x) - n / 2.0)
    for i in range(n):
        j1 = i - lags
        j2 = i + lags
        mask1 = (j1 >= 0) & (j1 < n)
        mask2 = (j2 >= 0) & (j2 < n)
        term = np.zeros(2 * n - 1)
        if np.any(mask1):
            term[mask1] = 1.0 - x[j1[mask1]]
        if np.any(mask2):
            term[mask2] -= x[j2[mask2]]
        grad[i] = tau * dx * np.dot(p, term) + penalty_grad
    return obj, grad

def lbfgs_polish(h, time_budget, start_time):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for tau in [20.0, 60.0, 150.0, 400.0]:
        if time.time() - start_time > time_budget:
            break
        def obj_func(x):
            o, _ = smoothed_obj_grad(x, tau, n)
            return o
        def grad_func(x):
            _, g = smoothed_obj_grad(x, tau, n)
            return g
        res = minimize(obj_func, best_h, method='L-BFGS-B', jac=grad_func,
                      bounds=bounds, options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_to_feasible(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5

def generate_binary_start(n, rng):
    target = n / 2.0
    h = np.zeros(n)
    num_ones = int(target)
    remainder = target - num_ones
    indices = rng.permutation(n)
    h[indices[:num_ones]] = 1.0
    if remainder > 0 and num_ones < n:
        h[indices[num_ones]] = remainder
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n_init = len(initial_h_values)
    best_h = initial_h_values.copy().astype(float)
    best_c5 = compute_c5(best_h)
    n = n_init

    remaining = budget_s - (time.time() - t0)
    h_cd, c5_cd = coordinate_descent(best_h, max_sweeps=100000, time_budget=remaining * 0.35, start_time=t0)
    if c5_cd < best_c5:
        best_h, best_c5 = h_cd, c5_cd

    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_pol, c5_pol = lbfgs_polish(best_h, remaining * 0.25, t0)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            h_cd2, c5_cd2 = coordinate_descent(best_h, max_sweeps=50000, time_budget=remaining * 0.4, start_time=t0)
            if c5_cd2 < best_c5:
                best_h, best_c5 = h_cd2, c5_cd2

    remaining = budget_s - (time.time() - t0)
    attempt = 0
    while remaining > 120 and attempt < 20:
        if attempt % 4 == 0:
            h_start = generate_binary_start(n, rng)
        elif attempt % 4 == 1:
            h_start = (rng.random(n) > 0.5).astype(float)
            h_start = project_to_feasible(h_start)
        elif attempt % 4 == 2:
            h_start = best_h + rng.normal(0, 0.03, n)
            h_start = np.clip(h_start, 0, 1)
            h_start = project_to_feasible(h_start)
        else:
            h_start = best_h.copy()
            k = rng.randint(n)
            h_start[k] = 1.0 - h_start[k]
            h_start = project_to_feasible(h_start)

        h_try, c5_try = coordinate_descent(h_start, max_sweeps=20000,
                                            time_budget=min(90.0, remaining * 0.12), start_time=t0)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            remaining = budget_s - (time.time() - t0)
            if remaining > 30:
                h_p, c5_p = lbfgs_polish(best_h, remaining * 0.15, t0)
                if c5_p < best_c5:
                    best_h, best_c5 = h_p, c5_p
        attempt += 1
        remaining = budget_s - (time.time() - t0)

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_final, c5_final = coordinate_descent(best_h, max_sweeps=100000,
                                                time_budget=remaining * 0.85, start_time=t0)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return best_h, best_c5, n
