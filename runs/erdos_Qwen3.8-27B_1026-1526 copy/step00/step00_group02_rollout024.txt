    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)
    val = float(corr[idx] * dx)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        j1 = i - lag
        if 0 <= j1 < n:
            g += (1.0 - h[j1])
        j2 = i + lag
        if 0 <= j2 < n:
            g -= h[j2]
        grad[i] = g
    return val, grad * dx


def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    log_sum = m + np.log(np.sum(np.exp(shifted))) / beta
    return float(log_sum * dx)


def smooth_max_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    val = float((m + np.log(np.sum(exp_shifted)) / beta) * dx)
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-8:
            continue
        lag = idx - (n - 1)
        for i in range(n):
            g = 0.0
            j1 = i - lag
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            j2 = i + lag
            if 0 <= j2 < n:
                g -= h[j2]
            grad[i] += w * g
    return val, grad * dx


def project_sum(h, target_sum):
    """Project h onto the set {x : sum(x) = target_sum, 0 <= x <= 1}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    # Final adjustment: distribute any residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        # Find a coordinate to adjust
        if residual > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + residual)
    return h

        return np.sum(x) - target_sum
    def eq_jac(x):
        return np.ones(n)
    return [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
