        s = np.sum(proj)
        if s > target:

        else:

    # Final adjustment
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:

            proj[mask] += diff / np.sum(mask)

def softmax_gradient(h, n, beta):
    """Compute gradient of smoothed max using softmax."""
    corr = np.correlate(h, 1 - h, mode="full")
    # Softmax weights
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    weights /= weights.sum()
    
    # Compute gradient as weighted sum of individual gradients
    # For lag k (0-indexed in correlate output), m = k - (n-1)
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-10:
            continue
        m = k - (n - 1)
        for j in range(n):
            if 0 <= j - m < n:
                grad[j] += w * (1.0 - h[j - m])
            if 0 <= j + m < n:
                grad[j] -= w * h[j + m]
    return grad

def softmax_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    exp_vals = np.exp(beta * (corr - c_max))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        for j in range(n):
            jm = j - m
            jp = j + m
            if 0 <= jm < n:
                grad[j] += w * (1.0 - h[jm])
            if 0 <= jp < n:
                grad[j] -= w * h[jp]
    return grad

def sa_phase(h, n, rng, deadline, target):
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    n_iter = 0
    T = 0.005
    total_iters = max(1, int((deadline - time.time()) * 5000))
    cool_rate = (1e-7 / max(T, 1e-7)) ** (1.0 / max(total_iters, 1))
    
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.08, 0.08)
        h[i] += delta
        h[j] -= delta
        if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
            h[i] -= delta
            h[j] += delta
            T *= cool_rate
            continue
        c5_new = compute_c5(h, n)
        de = c5_new - c5
        if de < 0 or rng.uniform() < np.exp(-de / T):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        T *= cool_rate
        n_iter += 1
        # Every 10000 iters, reheat if stuck
        if n_iter % 20000 == 0:
            h[:] = best_h
            c5 = best_c5
            T = 0.002
            cool_rate = (1e-7 / T) ** (1.0 / max(total_iters // 10, 1))
    
    return best_h, best_c5

Also, I'm worried about the `softmax_grad` function being slow for large n. Let me optimize it:

For the gradient computation, instead of looping over all k and all j, I can vectorize:

For a given m (lag), the gradient contribution to position j is:
- (1 - h[j-m]) if 0 ≤ j-m < n
- -h[j+m] if 0 ≤ j+m < n

This can be computed as:
grad[j] = (1 - h[j-m]) * I[0≤j-m<n] - h[j+m] * I[0≤j+m<n]

For all j simultaneously:
- Term 1: for j in range(max(0, m), min(n, n+m)): grad[j] += 1 - h[j-m]
  (when m ≥ 0: j ranges from m to n-1; when m < 0: j ranges from 0 to n-1+m)
- Term 2: for j in range(max(0, -m), min(n, n-m)): grad[j] -= h[j+m]
  (when m ≥ 0: j ranges from 0 to n-1-m; when m < 0: j ranges from -m to n-1)

Let me vectorize this:

```python
def softmax_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    exp_vals = np.exp(beta * (corr - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        # Term 1: (1 - h[j-m]) for valid j
        # j-m in [0, n-1] => j in [m, m+n-1] intersect [0, n-1]
        j_start = max(0, m)
        j_end = min(n, m + n)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += w * (1.0 - h[j_start - m : j_end - m])
        # Term 2: -h[j+m] for valid j
        # j+m in [0, n-1] => j in [-m, -m+n-1] intersect [0, n-1]
        j_start2 = max(0, -m)
        j_end2 = min(n, -m + n)  # exclusive
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w * h[j_start2 + m : j_end2 + m]
    
    return grad
```
