import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def make_random_binary(n, rng):
    h = np.zeros(n)
    k = n // 2
    idx = rng.choice(n, k, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[rng.choice(zeros)] = 0.5
    return h


def snap_to_binary(h_val, n, rng):
    h = h_val.copy().astype(float)
    if n % 2 == 0:
        k = n // 2
        idx = np.argsort(-h)[:k]
        result = np.zeros(n)
        result[idx] = 1.0
        return result
    else:
        k = (n - 1) // 2
        idx = np.argsort(-h)
        result = np.zeros(n)
        result[idx[:k]] = 1.0
        remaining = np.where(result == 0)[0]
        best_rem = remaining[np.argmax(h[remaining])]
        result[best_rem] = 0.5
        return result

def _corr_gradient(x, n):
    grad = np.zeros(n)
    one_minus = 1.0 - x
    corr = np.correlate(x, one_minus, mode='full')
    for i in range(n):
        for k in range(-n + 1, n):
            j = i - k
            if 0 <= j < n:
                grad[i] += 2.0 * corr[n - 1 + k] * one_minus[j]
                grad[j] += 2.0 * corr[n - 1 + k] * (-x[i])
    return grad
