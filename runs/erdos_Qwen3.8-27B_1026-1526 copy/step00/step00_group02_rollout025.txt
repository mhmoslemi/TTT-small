def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        else:

    cons = [{'type': 'eq', 'fun': SumEq(target_sum)}]

    try:
        res = minimize(C5Obj(dx), h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_feasible(res.x.copy(), target_sum)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5(h_opt, dx)
