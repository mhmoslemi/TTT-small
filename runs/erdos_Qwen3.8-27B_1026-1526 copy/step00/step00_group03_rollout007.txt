So there's an inherent lower bound. The best we can do is distribute the "boundary" (elements of S vs not S) as evenly as possible.

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = h.astype(np.float64).copy()
    tgt = n * 0.5
    for _ in range(300):
        s = h.sum()
        if s < 1e-14:
            h.fill(tgt / n)
            return h
        h *= tgt / s
        np.clip(h, 0.0, 1.0, out=h)
        if abs(h.sum() - tgt) < 1e-13:
            return h
    return h

def _c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _set_coord(h, n, i, v):
    tgt = n * 0.5
    old_v = h[i]
    need = tgt - v
    rest_sum = tgt - old_v
    if rest_sum < 1e-14 or need < -1e-14:
        ht = h.copy()
        ht[i] = v
        return _repair(ht, n)
    scale = need / rest_sum
    if scale < 0.0 or scale > 1.05:
        ht = h.copy()
        ht[i] = v
        return _repair(ht, n)
    ht = h.copy()
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    ht[mask] *= scale
    ht[i] = v
    np.clip(ht, 0.0, 1.0, out=ht)
    if abs(ht.sum() - tgt) > 1e-10:
        ht = _repair(ht, n)
    return ht

def _sweep(h, n, gsize):
    bh = h.copy()
    bc = _c5(bh, n)
    for i in range(n):
        v0 = bh[i]
        lo = max(0.0, v0 - 0.5)
        hi = min(1.0, v0 + 0.5)
        g = np.unique(np.concatenate((np.linspace(lo, hi, gsize), np.array([0.0, 1.0]))))
        best = bc
        best_h = None
        for v in g:
            if abs(float(v) - v0) < 1e-10:
                continue
            ht = _set_coord(bh, n, i, float(v))
            c = _c5(ht, n)
            if c < best - 1e-14:
                best = c
                best_h = ht
        if best_h is not None:
            bh = best_h
            bc = best
    return bh, bc

def _sweep_fine(h, n):
    bh = h.copy()
    bc = _c5(bh, n)
    for i in range(n):
        v0 = bh[i]
        for delta in (-0.02, -0.01, -0.005, 0.005, 0.01, 0.02):
            v = v0 + delta
            if v < 0 or v > 1:
                continue
            ht = _set_coord(bh, n, i, v)
            c = _c5(ht, n)
            if c < bc - 1e-14:
                bh = ht
                bc = c
                v0 = v
    return bh, bc

def _binarize(h, n):
    order = np.argsort(-h)
    hb = np.zeros(n)
    hb[order[:n // 2]] = 1.0
    return hb

def _gen_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    return h

def _gen_blocks(n, nblocks):
    h = np.zeros(n)
    block_size = n // nblocks
    for b in range(0, nblocks, 2):
        start = b * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return h

def _gen_random(n, rng):
    return rng.rand(n) * 1.6 - 0.3

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    n = 70

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        bh = _gen_alternating(n)
    bh = _repair(bh, n)
    bc = _c5(bh, n)

    def time_left():
        return budget_s - (time.time() - t0)

    stagnation = 0

    while time_left() > 90:

        bh, bc = _sweep(bh, n, 21)
        if bc < old_bc - 1e-12:
            stagnation = 0
        else:

            elif r < 0.5:

            elif r < 0.7:
                idx = rng.choice(n, n // 3 + 1, replace=False)
                bh[idx] = rng.rand(len(idx))
            else:
                perm = rng.permutation(n)
                bh = bh[perm]
            bh = _repair(bh, n)
            bc = _c5(bh, n)
            stagnation = 0
            max_stagnation = min(max_stagnation + 3, 60)

    if time_left() > 60:
        bh = _binarize(bh, n)
        bh, bc_bin = _sweep(bh, n, 21)
        if bc_bin < bc - 1e-12:
            bh, bc = _sweep(bh, n, 21)
        bh, bc = _sweep_fine(bh, n)

    if time_left() > 45:

        starts.append(_gen_alternating(n))
        starts.append(_gen_blocks(n, 2))
        starts.append(_gen_blocks(n, 4))
        starts.append(_gen_blocks(n, 5))
        starts.append(_gen_blocks(n, 6))
        starts.append(_gen_random(n, rng))
        starts.append(_gen_random(n, rng))

        for si in range(len(starts)):
            sarr[si] = starts[si]
        for si in range(len(starts)):
            if time_left() < 45:
                break

                bh = hs.copy()
                bc = cs
                bh, bc = _sweep(bh, n, 21)
                bh, bc = _sweep_fine(bh, n)

    if time_left() > 20:
        bh, bc = _sweep_fine(bh, n)

    bh = _repair(bh, n)
    bc = _c5(bh, n)
    return (bh, bc, n)
