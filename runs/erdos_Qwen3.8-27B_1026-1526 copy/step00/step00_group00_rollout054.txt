    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Water-filling projection
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        adj = diff / np.sum(free)
        h_new = h.copy()
        h_new[free] += adj
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - np.sum(h_new)
        h = h_new
    return h

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target):
    h = np.clip(h.astype(np.float64), 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
        else:
            free = h > 1e-15
        if not np.any(free):
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h


def cde_step(h, target, dx, deltas):
    n = len(h)
    c5 = compute_c5(h, dx)
    improved_any = False
    for i in range(n):
        for jo in [1, 2, 5, 11, n // 2]:
            j = (i + jo) % n
            if i == j:
                continue
            for d in deltas:
                hn = h.copy()
                hn[i] += d
                hn[j] -= d
                if 0.0 <= hn[i] <= 1.0 and 0.0 <= hn[j] <= 1.0:
                    if abs(np.sum(hn) - target) < 1e-10:
                        c5n = compute_c5(hn, dx)
                        if c5n < c5 - 1e-14:
                            h = hn
                            c5 = c5n
                            improved_any = True
                            break
    return h, c5, improved_any


def slsqp_optimize(h, n, max_iter=1000, ftol=1e-15):
    dx = 2.0 / n
    target = n / 2.0
    h = project_sum(h, target)
    c5 = compute_c5(h, dx)

    def obj(x):
        return compute_c5(x, dx)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        hx = project_sum(np.clip(res.x, 0.0, 1.0), target)
        c5x = compute_c5(hx, dx)
        if c5x < c5:
            return hx, c5x
    except Exception:
        pass
    return h, c5


def nm_optimize(h, n, max_iter=3000):
    dx = 2.0 / n
    target = n / 2.0
    h = project_sum(h, target)
    c5 = compute_c5(h, dx)

    def obj(x):
        xx = project_sum(x, target)
        return compute_c5(xx, dx)

    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-12,
                                'adaptive': True})
        hx = project_sum(np.clip(res.x, 0.0, 1.0), target)
        c5x = compute_c5(hx, dx)
        if c5x < c5:
            return hx, c5x
    except Exception:
        pass
    return h, c5
