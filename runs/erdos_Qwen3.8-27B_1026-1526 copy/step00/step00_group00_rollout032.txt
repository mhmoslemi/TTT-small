        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def logsumexp(values, alpha):
    m = np.max(values)
    if alpha == 0:
        return m, np.full(len(values), 1.0/len(values))
    shifted = alpha * (values - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    lse = m + np.log(sum_exp) / alpha
    probs = exp_vals / sum_exp
    return lse, probs

def smooth_gradient(h, n_points, alpha):
    """Compute gradient of smoothed objective (log-sum-exp of correlations)"""
    dx = 2.0 / n_points
    n = n_points
    # Compute all correlation values (without dx)
    corr_raw = np.correlate(h, 1 - h, mode="full")
    # Lag index: corr_raw[j] corresponds to lag l = j - (n-1)
    
    # Compute log-sum-exp
    m = np.max(corr_raw)
    shifted = alpha * (corr_raw - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    probs = exp_vals / sum_exp  # weighted probabilities
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        l = j - (n - 1)  # lag
        if l == 0:
            g = 1 - 2 * h
        elif l > 0:
            g = np.zeros(n)
            g[l:] += (1 - h[:n-l])
            g[:n-l] -= h[l:]
        else:
            ll = -l
            g = np.zeros(n)
            g[:n-ll] += (1 - h[ll:])
            g[ll:] -= h[:n-ll]
        grad += probs[j] * g
    
    return grad * dx, m * dx + np.log(sum_exp) / alpha

def project_onto_feasible(h, target_sum):
    n = len(h)
    lo = h.min() - 2
    hi = h.max() + 2
    for _ in range(200):
        tau = (lo + hi) / 2
        h_proj = np.clip(h - tau, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_smooth_grad(h, n_points, alpha):
    n = n_points
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr_raw)
    shifted = alpha * (corr_raw - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    probs = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        l = k - (n - 1)
        p = probs[k]
        if p < 1e-10:
            continue
        if l == 0:
            grad += p * (1 - 2 * h)
        elif l > 0:
            grad[l:] += p * (1 - h[:n-l])
            grad[:n-l] -= p * h[l:]
        else:
            ll = -l
            grad[:n-ll] += p * (1 - h[ll:])
            grad[ll:] -= p * h[:n-ll]
    
    obj = m * dx + np.log(sum_exp) / alpha
    return grad * dx, obj

        h_cur, obj = pgd_optimize(h_rand, n, 30, 80, target_sum, 0.3, 0.5, deadline - 5)
        h_cur, obj = pgd_optimize(h_cur, n, 5, 80, target_sum, 0.1, 0.4, deadline - 5)
        if obj < best_c5:
            best_c5 = obj
            best_h = h_cur.copy()

    best_h = project_onto_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, n)

    return (best_h, best_c5, n)
