So f(h) = (1/t) * log(Σ_k exp(t*c[k]))

∇f(h) = (1/t) * Σ_k exp(t*c[k]) * t * ∇c[k] / Σ_k exp(t*c[k])

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    m = max_idx - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j + m < n:
            val += (1.0 - h[j + m])
        if 0 <= j - m < n:
            val -= h[j - m]
        grad[j] = val * dx
    return np.max(corr) * dx, grad


def smooth_c5_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_vals = np.exp(t * shifted)
    sum_exp = exp_vals.sum()
    fval = cmax + np.log(sum_exp) / t
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = weights[k]
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j + m < n:
                val += (1.0 - h[j + m])
            if 0 <= j - m < n:
                val -= h[j - m]
            grad[j] += wk * val * dx
    return fval, grad


def project_feasible(h, target_sum):
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        n_active = np.sum(mask)
        if n_active == 0:
            break
        h[mask] += diff / n_active
        h = np.clip(h, 0.0, 1.0)
    return h


class SumEq:
    def __init__(self, target, n):
        self.target = target
        self.n = n

    def fun(self, x):
        return np.sum(x) - self.target

    def jac(self, x):
        return np.ones(self.n)


class SmoothObj:
    def __init__(self, t):
        self.t = t

    def __call__(self, h):
        return smooth_c5_and_grad(h, self.t)[0]


class SmoothObjJac:
    def __init__(self, t):
        self.t = t

    def __call__(self, h):
        return smooth_c5_and_grad(h, self.t)[1]


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)

    best_h = project_feasible(h0, target_sum)

    # Phase 1: SLSQP with log-sum-exp smoothing (coarse to fine)
    temps = [3, 10, 30, 100, 300, 1000, 3000, 10000]
    for t in temps:
        if time.time() - start_time > budget_s * 0.25:
            break
        bounds = [(0.0, 1.0)] * n
        se = SumEq(target_sum, n)

        obj_fn = SmoothObj(t)
        jac_fn = SmoothObjJac(t)
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', jac=jac_fn,
                           bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target_sum)

                    h_new = project_feasible(res.x, ts_alt)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n_alt
                except Exception:
                    pass

        n_f = len(best_h)
        ts_f = n_f / 2.0
        bounds_f = [(0.0, 1.0)] * n_f
