    # Phase 2: Multiple restarts with diverse initializations
    restarts = [
        'random', 'thue', 'ring0', 'ring1', 'sigma7', 'sigma13', 'sigma3', 'clusters'
    ]
    for r, label in enumerate(restarts):
        if time.time() > deadline - 20:
            break
        if label == 'random':
            h_r = (rng.random(n_points) < 0.5).astype(float)
        elif label == 'thue':
            h_r = _thue_morse(n_points)
        elif label == 'ring0':
            h_r = _ring_pattern(n_points, rng, 0)
        elif label == 'ring1':
            h_r = _ring_pattern(n_points, rng, 1)
        elif label == 'sigma7':
            h_r = _sigma_pattern(n_points, 7, 1)
        elif label == 'sigma13':
            h_r = _sigma_pattern(n_points, 13, 3)
        elif label == 'sigma3':
            h_r = _sigma_pattern(n_points, 3, 0)
        else:
            h_r = np.zeros(n_points)
            block = max(3, n_points // 12)
            for b in range(12):
                s = b * block
                e = min(s + block, n_points)
                if b % 3 != 1:
                    h_r[s:e] = 1.0

        h_r = _fix_binary_sum(h_r, n_points, rng)
        remaining = deadline - time.time()
        iters = max(50000, int(remaining * 40000 / max(1, len(restarts) - r)))
        _, _, best_h, best_c5 = _sa_swap(h_r.copy(), n_points, rng, iters, t0, deadline, best_h, best_c5)

    # Phase 3: More SA refinement from best
    if time.time() < deadline - 15:
        iters = int((deadline - time.time() - 15) * 60000)
        _, _, best_h, best_c5 = _sa_swap(best_h.copy(), n_points, rng, iters, t0, deadline, best_h, best_c5)

    # Phase 4: Continuous refinement via SLSQP
    if time.time() < deadline - 10:
        h_cont = best_h.astype(float).copy()
        h_cont += rng.random(n_points) * 0.03
        h_cont = np.clip(h_cont, 0.0, 1.0)
        diff = n_points / 2.0 - np.sum(h_cont)
        h_cont += diff / n_points
        h_cont = np.clip(h_cont, 0.0, 1.0)

        try:
            res = minimize(
                _scipy_obj,
                h_cont,
                jac=_scipy_jac,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n_points,
                constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            if res.fun < best_c5:
                best_h = np.clip(res.x, 0.0, 1.0)
                best_c5 = _evaluate_c5(best_h)
        except Exception:
            pass

    # Final repair
    best_h = np.clip(best_h, 0.0, 1.0)
    diff = n_points / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        mask = (best_h > 0.005) & (best_h < 0.995)
        if np.sum(mask) > 0:
            best_h[mask] += diff / np.sum(mask)
        else:
            best_h += diff / n_points
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _evaluate_c5(best_h)

    return best_h, best_c5, n_points
