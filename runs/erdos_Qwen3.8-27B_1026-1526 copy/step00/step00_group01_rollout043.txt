For arrays a and b of length n:
np.correlate(a, b, mode='full') returns an array c of length 2n-1.
c[k] = sum_i a[i] * b[i - (k - (n-1))]  (with appropriate bounds)

Actually, the numpy documentation says:
"cross-correlation of 1-dimensional sequences. The result has length 2*N-1, where N is the length of the shorter input. The sequence v is reversed and conjugated."

So c[k] for k=0,...,2n-2 represents:
c[k] = sum_{m} a[m] * conj(b[m + k - (n-1)])

where the sum is over m such that both m and m+k-(n-1) are valid indices.

For our case: a = h, b = 1-h.
c[k] = sum_m h[m] * (1-h)[m + k - (n-1)]

Let L = k - (n-1). Then:
c[n-1+L] = sum_m h[m] * (1-h)[m+L]

This is the number of positions m where h[m]=1 and h[m+L]=0 (for binary h).

The max over k of c[k] gives the maximum overlap at any shift.

This matches the problem statement. Good.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def local_search(h, time_limit=30, rng=None):
    n = len(h)
    h = h.astype(float).copy()
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = np.max(compute_corr(h)) * dx
    t0 = time.time()
    no_improve = 0
    while time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        mids = np.setdiff1d(np.arange(n), np.concatenate([ones, zeros])) if len(ones) + len(zeros) < n else np.array([], dtype=int)
        groups = [(ones, zeros), (ones, mids), (zeros, mids)]
        for ga, gb in groups:
            if len(ga) == 0 or len(gb) == 0:
                continue
            idx_a = list(ga)
            idx_b = list(gb)
            rng.shuffle(idx_a)
            rng.shuffle(idx_b)
            for i in idx_a:
                for j in idx_b:
                    vi, vj = h[i], h[j]
                    if vi == vj:
                        continue
                    h[i], h[j] = vj, vi
                    nc = np.max(compute_corr(h)) * dx
                    if nc < best_c5 - 1e-15:
                        best_c5 = nc
                        improved = True
                    else:
                        h[i], h[j] = vi, vj
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    return h, best_c5

def simulated_annealing(h, time_limit=60, rng=None):
    n = len(h)
    h = h.astype(float).copy()
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    current_c5 = np.max(compute_corr(h)) * dx
    best_c5 = current_c5
    best_h = h.copy()
    t0 = time.time()
    T = 0.05
    cooling = 0.9999
    while time.time() - t0 < time_limit and T > 0.0005:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        mids = np.setdiff1d(np.arange(n), np.concatenate([ones, zeros])) if len(ones) + len(zeros) < n else np.array([], dtype=int)
        all_diff = np.concatenate([ones, mids])
        all_opp = np.concatenate([zeros, mids])
        if len(all_diff) == 0 or len(all_opp) == 0:
            break
        i = int(rng.choice(all_diff))
        j = int(rng.choice(all_opp))
        if h[i] == h[j]:
            T *= cooling
            continue
        vi, vj = h[i], h[j]
        h[i], h[j] = vj, vi
        nc = np.max(compute_corr(h)) * dx
        d = nc - current_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            current_c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            h[i], h[j] = vi, vj
        T *= cooling
    return best_h, best_c5

def make_random_h(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h

def make_block_h(n, rng):
    h = np.zeros(n)
    block = max(2, n // 5)
    pos = 0
    fill = True
    while pos < n:
        if fill:
            h[pos:pos + block] = 1.0
            pos += block
        else:
            pos += block
        fill = not fill
    s = np.sum(h)
    target = n / 2.0
    if s > target:
        excess = int(s - target)
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        h[ones[:excess]] = 0.0
    elif s < target:
        deficit = int(target - s)
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        h[zeros[:deficit]] = 1.0
    return h

def kick_sequence(h, n_kicks, rng):
    n = len(h)
    hk = h.copy()
    for _ in range(n_kicks):
        a, b = int(rng.randint(n)), int(rng.randint(n))
        if hk[a] != hk[b]:
            hk[a], hk[b] = hk[b], hk[a]
    return hk
