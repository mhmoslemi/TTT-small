```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    """Ensure h is in [0,1] with sum = n/2."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                step = diff / cnt
                h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                step = diff / cnt
                h[mask] = np.maximum(h[mask] + step, 0.0)
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    # Exact sum fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Add to index with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def golden_section_1d(f, a, b, tol=1e-10, max_iter=40):
    """Minimize unimodal f on [a, b] using golden section."""
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2, min(fc, fd)
```

```python
def soft_c5_and_grad(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    M = np.max(corr)
    exps = np.exp(beta * (corr - M))
    w = exps / np.sum(exps)
    soft_max = M + np.log(np.sum(exps)) / beta
    c5 = soft_max * dx
    
    # Compute gradient
    # grad[t] = dx * (sum_k w[k] * dcorr_k/dh_t - sum_k w[k] * dcorr_k/dh_{n-1})
    # for t = 0, ..., n-2
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        for t in range(n):
            # d(corr[k])/d(h[t])
            if s == 0:
                d = 1.0 - 2.0 * h[t]
            else:
                d = 0.0
                if 0 <= t + s < n:
                    d += (1.0 - h[t + s])
                if 0 <= t - s < n:
                    d -= h[t - s]
            grad[t] += wk * d
    
    # Now reduce: for free variables t = 0,...,n-2
    # effective_grad[t] = grad[t] - grad[n-1]
    eff_grad = grad[:n-1] - grad[n-1]
    
    return c5, eff_grad
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    M = np.max(corr)
    exps = np.exp(beta * (corr - M))
    w = exps / np.sum(exps)
    soft_max = M + np.log(np.sum(exps)) / beta
    c5 = soft_max * dx
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif s > 0:
            # d(corr)/d(h[t]) = (1-h[t+s]) for t in [0, n-1-s] minus h[t-s] for t in [s, n-1]
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] -= wk * h[:n-s]
        else:  # s < 0
            s2 = -s
            # d(corr)/d(h[t]) = (1-h[t-s2]) for t in [s2, n-1] minus h[t+s2] for t in [0, n-1-s2]
            grad[s2:] += wk * (1.0 - h[:n-s2])
            grad[:n-s2] -= wk * h[s2:]
    
    eff_grad = grad[:n-1] - grad[n-1]
    return c5, eff_grad
```

def make_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):

def golden_section_step(h, t, n, lo, hi, max_iter=25):
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h[t] += c; h[n-1] -= c
    fc = compute_c5(h)
    h[t] -= c; h[n-1] += c

    for _ in range(max_iter):

            break
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            h[t] += c; h[n-1] -= c
