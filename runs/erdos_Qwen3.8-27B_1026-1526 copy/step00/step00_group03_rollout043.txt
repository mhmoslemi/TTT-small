def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def best_coord_value(h, idx, n):
    m = 2 * n - 1
    curr = np.correlate(h, 1.0 - h, mode='full')
    p = idx
    ks = np.arange(m) - (n - 1)
    idx_pk = p + ks
    idx_pmk = p - ks
    h_pk = np.zeros(m)
    vp = (idx_pk >= 0) & (idx_pk < n)
    h_pk[vp] = h[idx_pk[vp]]
    h_pmk = np.zeros(m)
    vm = (idx_pmk >= 0) & (idx_pmk < n)
    h_pmk[vm] = h[idx_pmk[vm]]
    B = 1.0 - h_pk - h_pmk
    u_min = -h[p]
    u_max = 1.0 - h[p]
    if u_min >= u_max - 1e-15:
        return h[p]
    ia = np.arange(m)
    I, J = np.meshgrid(ia, ia, indexing='ij')
    dB = B[I] - B[J]
    dC = curr[J] - curr[I]
    valid_db = np.abs(dB) > 1e-14
    u_vals = np.full(m * m, np.inf)
    u_vals[valid_db] = dC[valid_db] / dB[valid_db]
    in_range = valid_db & (u_vals >= u_min) & (u_vals <= u_max)
    bps = np.unique(u_vals[in_range])
    all_cands = np.unique(np.concatenate([bps, [u_min, u_max]]))
    vals = curr[None, :] + all_cands[:, None] * B[None, :]
    g_vals = np.max(vals, axis=1)
    best_u = all_cands[np.argmin(g_vals)]
    return np.clip(h[p] + best_u, 0.0, 1.0)


def coord_descent(h, n, rng, max_sweeps=10):
    h = h.copy()
    for _ in range(max_sweeps):
        order = rng.permutation(n)
        for idx in order:
            h[idx] = best_coord_value(h, idx, n)
        h = repair(h, n)
    return h


def to_binary(h, n):
    target = n // 2
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:target]] = 1.0
    return h_bin


def binary_local_search(h_bin, n, rng, max_rounds=50):
    h = h_bin.copy()
    best_c5 = c5_value(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for a in ones:
            for b in zeros:
                h[a] = 0.0
                h[b] = 1.0
                c5 = c5_value(h)
                if c5 < best_c5 - 1e-15:
                    best_c5 = c5
                    improved = True
                    break
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n = 70
        h = np.full(n, 0.5)

    n = len(h)
    h = repair(h, n)
    best_h = h.copy()
    best_c5 = c5_value(h)

    best_h = coord_descent(best_h, n, rng, max_sweeps=30)
    best_c5 = c5_value(best_h)

    h_bin = to_binary(best_h, n)
    h_bin, c5_bin = binary_local_search(h_bin, n, rng, max_rounds=100)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin

    h_cont = coord_descent(best_h, n, rng, max_sweeps=30)
    if c5_value(h_cont) < best_c5:
        best_c5 = c5_value(h_cont)
        best_h = h_cont

    stuck = 0
    max_stuck = 80
    while time.time() - start_time < budget_s - 30:
        strategy = rng.integers(0, 6)

        if strategy == 0:
            h_pert = best_h + rng.normal(0, 0.03, n)
        elif strategy == 1:
            h_pert = best_h.copy()
            n_flip = int(rng.integers(2, 6))
            idxs = rng.choice(n, n_flip, replace=False)
            h_pert[idxs] = 1.0 - h_pert[idxs]
        elif strategy == 2:
            h_pert = best_h.copy()
            L = int(rng.integers(3, min(12, n // 5)))
            b1 = int(rng.integers(0, n - L))
            b2 = int(rng.integers(0, n - L))
            if abs(b1 - b2) >= L:
                tmp = h_pert[b1:b1 + L].copy()
                h_pert[b1:b1 + L] = h_pert[b2:b2 + L]
                h_pert[b2:b2 + L] = tmp
            else:
                h_pert = best_h + rng.normal(0, 0.1, n)
        elif strategy == 3:
            h_bin2 = to_binary(best_h, n)
            h_bin2, c5_b2 = binary_local_search(h_bin2, n, rng, max_rounds=50)
            h_pert = h_bin2
        elif strategy == 4:
            h_pert = best_h + rng.normal(0, 0.15, n)
        else:
            h_pert = best_h.copy()
            n_swap = int(rng.integers(2, 4))
            for _ in range(n_swap):
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                h_pert[i1], h_pert[i2] = h_pert[i2], h_pert[i1]

        h_pert = repair(h_pert, n)
        h_pert = coord_descent(h_pert, n, rng, max_sweeps=8)

        h_bin3 = to_binary(h_pert, n)
        h_bin3, c5_b3 = binary_local_search(h_bin3, n, rng, max_rounds=30)
        h_cont3 = coord_descent(h_bin3, n, rng, max_sweeps=10)

        c5_pert = c5_value(h_pert)
        c5_candidate = min(c5_pert, c5_b3, c5_value(h_cont3))

        if c5_pert <= c5_b3 and c5_pert <= c5_value(h_cont3):
            h_cand = h_pert
        elif c5_b3 <= c5_value(h_cont3):
            h_cand = h_bin3
        else:
            h_cand = h_cont3

        if c5_candidate < best_c5 - 1e-13:
            best_h = h_cand
            best_c5 = c5_candidate
            stuck = 0
        else:
            stuck += 1
            if stuck >= max_stuck:
                for _ in range(10):
                    if time.time() - start_time > budget_s - 30:
                        break
                    h_rand = rng.uniform(0, 1, n)
                    h_rand = repair(h_rand, n)
                    h_rand = coord_descent(h_rand, n, rng, max_sweeps=15)
                    h_rand_bin = to_binary(h_rand, n)
                    h_rand_bin, _ = binary_local_search(h_rand_bin, n, rng, max_rounds=50)
                    c5_rand = min(c5_value(h_rand), c5_value(h_rand_bin))
                    if c5_rand < best_c5:
                        best_h = h_rand if c5_value(h_rand) < c5_value(h_rand_bin) else h_rand_bin
                        best_c5 = c5_rand
                        stuck = 0
                        break
                if stuck >= max_stuck:
                    stuck = 0

    best_h = repair(best_h, n)
    best_c5 = c5_value(best_h)
    return best_h, best_c5, n
