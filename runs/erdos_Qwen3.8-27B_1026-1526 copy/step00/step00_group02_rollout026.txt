def project_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    # Use bisection to find threshold
    lo, hi = 0.0, 1.0
    for _ in range(100):

        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection on threshold."""
    n = len(h)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    # The function sum(clip(h - tau, 0, 1)) is continuous and decreasing in tau
    lo = np.min(h) - 1.0  # at this point, sum = n (all 1s)
    hi = np.max(h) + 1.0  # at this point, sum = 0 (all 0s)
    
    # Actually: clip(h - tau, 0, 1)
    # When tau is very negative: h - tau > 1 for all, so clip = 1, sum = n
    # When tau is very positive: h - tau < 0 for all, so clip = 0, sum = 0
    # We want sum = target_sum
    
    if target_sum >= n:
        return np.ones(n)
    if target_sum <= 0:
        return np.zeros(n)
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid  # need to increase tau to decrease sum
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    
    # Verify sum is close enough
    # If not, do a tiny adjustment
    diff = target_sum - result.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference to non-boundary elements
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.any():
            per = diff / mask.sum()
            result[mask] = np.clip(result[mask] + per, 0.0, 1.0)
    
    return result

def eval_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, mode='full')[k] * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def softmax_obj(h, alpha, dx):
    """Compute soft-max approximation of C5."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    # For numerical stability
    if alpha * max_c > 500:
        return max_c
    return np.max(corr) + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (corr - max_c))))

def compute_grad_softmax(h, alpha, dx):
    """Compute gradient of soft-max of correlation values w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        i_lo = max(0, -m)
        i_hi = min(n, n - m)
        for i in range(i_lo, i_hi):
            j = i + m
            grad[i] += w * dx * (1.0 - h[j])
            grad[j] -= w * dx * h[i]
    
    return grad

def polish_with_slsqp(h0, target_sum, n, dx):
    """Polish solution using SLSQP."""
    def objective(h):
        return softmax_obj(h, 100.0, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(objective, h0, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
        if result.success:
            h_sol = np.clip(result.x, 0.0, 1.0)
            h_sol = project_to_feasible(h_sol, target_sum)
            return h_sol
        else:
            # Still try the result
            h_sol = np.clip(result.x, 0.0, 1.0)
            h_sol = project_to_feasible(h_sol, target_sum)
            if eval_c5(h_sol, dx) < eval_c5(h0, dx):
                return h_sol
    except:
        pass
    return None

def generate_random_feasible(n, target_sum, rng):
    """Generate a random feasible point."""
    # Start with random values and project
    h = rng.random(n)
    h = project_to_feasible(h, target_sum)
    return h

def projected_gradient_optimize(h0, target_sum, n, dx, time_limit, rng, max_iter=10000):

                alpha = 30.0
                stall_count = 0
    
    return best_h_local, best_local
