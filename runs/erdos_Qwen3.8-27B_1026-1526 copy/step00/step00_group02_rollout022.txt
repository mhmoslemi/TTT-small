    corr_shifted = corr - np.max(corr)

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(h, ts):
    h = h.astype(np.float64).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

    n = len(h)
    m = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(corr.max())
    e = np.exp(beta * (corr - cm))
    w = e / e.sum()
