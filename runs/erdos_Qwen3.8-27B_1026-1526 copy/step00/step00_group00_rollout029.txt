import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr), corr

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        excess = np.sum(h) - target
        if abs(excess) < 1e-14:
            break
        at_zero = h <= 1e-15
        at_one = h >= 1.0 - 1e-15
        free = ~(at_zero | at_one)
        nfree = np.sum(free)
        if nfree == 0:
            break
        h[free] -= excess / nfree
        h = np.clip(h, 0.0, 1.0)
    return h
