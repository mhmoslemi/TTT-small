    diff = target - float(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def optimize_chain(h0, n, target, alphas, max_iter=400, ftol=1e-14):
    h_cur = h0.copy()
    for alpha in alphas:
        if _time_left() < 10:
            break
        lam = 50.0
        res = minimize(
            corr_obj, h_cur.copy(), args=(alpha, n, lam),
            method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-8}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = evaluate_c5(h_new)
        c5_cur = evaluate_c5(h_cur)
        if c5_new < c5_cur:
            h_cur = h_new
    return h_cur

def _time_left():
    return _budget - (time.time() - _t0)

_budget = 1300.0
_t0 = 0.0

def coordinate_refine(h, n, max_sweeps=200):
    global _budget, _t0
    target = n / 2.0
    c5_cur = evaluate_c5(h)
    for sweep in range(max_sweeps):
        if _time_left() < 5:
            break
        improved = False
        for i in range(n):
            best_val = h[i]
            best_c5 = c5_cur
            for trial in np.linspace(0.0, 1.0, 13):
                if abs(trial - h[i]) < 1e-6:
                    continue
                h_trial = h.copy()
                h_trial[i] = trial
                diff = target - float(np.sum(h_trial))
                if abs(diff) > 1e-12:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    n_mask = int(np.sum(mask))
                    if n_mask > 0:
                        h_trial[mask] += diff / n_mask
                    h_trial = np.clip(h_trial, 0.0, 1.0)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    best_val = trial
                    h = h_trial.copy()
                    c5_cur = best_c5
            if abs(h[i] - best_val) > 1e-8:
                h[i] = best_val
                h = fix_sum(h, n)
                c5_cur = evaluate_c5(h)
                improved = True
        if not improved:
            break
    return h, c5_cur

def random_perturb_search(h, n, n_iters=3000):
    global _budget, _t0
    target = n / 2.0
    c5_cur = evaluate_c5(h)
    scale = 0.05
    for it in range(n_iters):
        if _time_left() < 5:
            break
        if it % 500 == 500:
            scale = max(0.01, scale * 0.7)
        h_trial = h.copy()
        n_pert = np.random.randint(1, 6)
        idx = np.random.choice(n, n_pert, replace=False)
        for i in idx:
            h_trial[i] += np.random.uniform(-scale, scale)
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = fix_sum(h_trial, n)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_cur - 1e-10:
            c5_cur = c5_trial
            h = h_trial
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    global _budget, _t0

    _t0 = time.time()
    np.random.seed(seed)

    try:
        h_best = np.array(initial_h_values, dtype=np.float64).copy()

        h_best = np.full(45, 0.5)

    n = n_best

    h_cur = optimize_chain(h_best, n, target, [5.0, 10.0, 20.0, 50.0, 100.0, 200.0])

    if c5_cur < c5_best:
        c5_best = c5_cur

    h_cur, c5_cur = coordinate_refine(h_cur, n, max_sweeps=100)
    if c5_cur < c5_best:
        c5_best = c5_cur
        h_best = h_cur.copy()

    try_n_list = [45, 51, 63, 75, 87, 105]
    for n_try in try_n_list:
        if _time_left() < 60:
            break
        if n_try == n_best and _time_left() < 150:
            continue
        t_try = n_try / 2.0
        x_old = np.linspace(0, 2, len(h_cur), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_cur)
        h_interp = np.clip(h_interp, 0.0, 1.0)
        h_interp = fix_sum(h_interp, n_try)
        h_opt = optimize_chain(h_interp, n_try, t_try, [10.0, 30.0, 80.0, 150.0])
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt.copy()
            n_best = n_try
            h_cur = h_opt.copy()

    for n_try in [45, 63, 87]:
        if _time_left() < 40:
            break
        t_try = n_try / 2.0
        for _ in range(3):
            if _time_left() < 20:
                break
            h_r = np.random.rand(n_try)
            h_r = fix_sum(h_r, n_try)
            h_opt = optimize_chain(h_r, n_try, t_try, [20.0, 60.0, 150.0], max_iter=200)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < c5_best:
                c5_best = c5_opt
                h_best = h_opt.copy()
                n_best = n_try

    if _time_left() > 30:
        n = n_best
        h_cur, c5_cur = coordinate_refine(h_best.copy(), n, max_sweeps=200)
        if c5_cur < c5_best:
            c5_best = c5_cur
            h_best = h_cur.copy()

    if _time_left() > 20:
        n = n_best
        h_best, c5_cur = random_perturb_search(h_best, n, n_iters=2000)
        if c5_cur < c5_best:
            c5_best = c5_cur

    if _time_left() > 15:
        n = n_best
        h_best, c5_cur = coordinate_refine(h_best, n, max_sweeps=100)
        if c5_cur < c5_best:
            c5_best = c5_cur

    c5_final = evaluate_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = fix_sum(h_best, n_best)
    c5_final = evaluate_c5(h_best)

    return (h_best, c5_final, n_best)
