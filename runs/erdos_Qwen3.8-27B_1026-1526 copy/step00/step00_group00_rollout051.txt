    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    if m == 0:
        return 0.0
    x = beta * (corr - m)
    x = np.clip(x, -100, 100)

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)

def slp_optimize(h_init, start_time, time_limit, max_iter=100):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Compute current correlation and find which k's are "active"
        corr = np.correlate(h, 1 - h, mode="full")
        # Get gradients for each k
        # Linearize: corr[k](h+δ) ≈ corr[k](h) + g_k · δ
        # where g_k[j] = (1-h[j-s_k]) * 1_{valid} - h[j+s_k] * 1_{valid}
        
        # Solve LP:
        # minimize t
        # s.t. corr[k] + dx * g_k · (h_new - h) <= t for all k
        #      0 <= h_new <= 1
        #      sum(h_new) = n/2
        
        # In standard form for linprog:
        # Variables: [h_new (n), t (1)] → total n+1 variables
        # Minimize: t → c = [0,...,0,1]
        # Constraints:
        #   dx * g_k · h_new - dx * g_k · h <= t for each k
        #   → dx * g_k · h_new - t <= dx * g_k · h
        #   -h_new <= 0
        #   h_new <= 1
        #   sum(h_new) = n/2 (equality)
        
        # Build constraint matrix
        A_ub = []
        b_ub = []
        
        for k in range(2*n-1):
            s = k - (n-1)
            g = np.zeros(n)
            if s >= 0:
                g[s:] = 1.0 - h[:n-s]
                g[:n-s] -= h[s:]
            else:
                ms = -s
                g[:n-ms] = 1.0 - h[ms:]
                g[ms:] -= h[:n-ms]
            
            # dx * g · h_new - t <= dx * g · h
            row = np.zeros(n+1)
            row[:n] = dx * g
            row[n] = -1.0
            A_ub.append(row)
            b_ub.append(dx * np.dot(g, h))
        
        # 0 <= h_new: -h_new <= 0
        for i in range(n):
            row = np.zeros(n+1)
            row[i] = -1.0
            A_ub.append(row)
            b_ub.append(0.0)
        
        # h_new <= 1: h_new <= 1
        for i in range(n):
            row = np.zeros(n+1)
            row[i] = 1.0
            A_ub.append(row)
            b_ub.append(1.0)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Equality: sum(h_new) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        c = np.zeros(n+1)
        c[n] = 1.0  # minimize t
        
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=[(0,1)]*n + [(0, None)], method='highs')
        
        if res.success:
            h_new = res.x[:n]

            # Check actual C5 (not the linearized one)
            c5_actual = compute_c5(h_new)
            if c5_actual < best_c5:
                best_c5 = c5_actual
                best_h = h_new.copy()
            h = h_new
        else:
            break
    
    return best_h, best_c5

from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_constraint(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def slp_optimize(h_init, start_time, time_limit, max_iter=200, step_init=0.3):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1 - h, mode="full")
        
        # Build LP
        n_vars = n + 1
        A_ub_rows = []
        b_ub_vals = []
        
        for k in range(2 * n - 1):
            s = k - (n - 1)
            g = np.zeros(n)
            if s >= 0:
                g[s:] = 1.0 - h[:n-s]
                g[:n-s] -= h[s:]
            else:
                ms = -s
                g[:n-ms] = 1.0 - h[ms:]
                g[ms:] -= h[:n-ms]
            
            row = np.zeros(n_vars)
            row[:n] = dx * g
            row[n] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(dx * np.dot(g, h))
        
        # Box constraints
        for i in range(n):
            row = np.zeros(n_vars)
            row[i] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(-h[i] + step)  # h_new[i] >= h[i] - step
            
            row = np.zeros(n_vars)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(h[i] + step)  # h_new[i] <= h[i] + step
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        c = np.zeros(n_vars)
        c[n] = 1.0
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(0, None)], method='highs')
        
        if res.success:
            h_new = res.x[:n]
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Accept if the linearized objective improved
            t_new = res.x[n]
            t_old = compute_c5(h)
            if t_new < t_old - 1e-10:
                h = h_new
                # Increase step
                step = min(step * 1.1, 0.5)
            else:
                # Decrease step
                step *= 0.7
        else:
            step *= 0.5
        
        if step < 1e-6:
            break
    
    return best_h, best_c5
