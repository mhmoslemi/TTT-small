import numpy as np
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def to_binary(h, n_points):
    # Convert to 0-1 valued, maintaining sum = n/2
    h = np.clip(h, 0, 1)
    # Sort by value, keep top n/2 as 1
    threshold = np.sort(h)[::-1][n_points // 2 - 1]
    h_bin = (h >= threshold).astype(float)
    # Fix if sum is wrong
    while np.sum(h_bin) < n_points // 2:
        idx = np.where(h_bin == 0)[0]
        # Pick the one with highest original value
        best = idx[np.argmax(h[idx])]
        h_bin[best] = 1
    while np.sum(h_bin) > n_points // 2:
        idx = np.where(h_bin == 1)[0]
        best = idx[np.argmin(h[idx])]
        h_bin[best] = 0
    return h_bin

def local_search_binary(h, n_points, budget_s, start_time):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    current_c5 = evaluate_c5(h, n_points)
    best_c5 = current_c5
    best_h = h.copy()
    
    while time.time() - start_time < budget_s:
        improved = False
        best_swap = None
        best_swap_c5 = current_c5
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = evaluate_c5(h, n_points)
                h[i] = 1
                h[j] = 0
                if c5 < best_swap_c5 - 1e-10:
                    best_swap_c5 = c5
                    best_swap = (i, j)
        
        if best_swap is not None:

            h[i] = 0

            zeros = np.where(h == 0)[0]
            current_c5 = best_swap_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
