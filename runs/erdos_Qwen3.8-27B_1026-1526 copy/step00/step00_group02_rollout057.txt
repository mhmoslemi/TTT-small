import numpy as np
from scipy.optimize import minimize

def _project_sum(h, target):
    h = h.astype(float).copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def _calc_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _subgrad_c5(h, dx):
    eps = 1e-6
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        hp = h.copy()
        hm = h.copy()
        hp[j] = min(1.0, h[j] + eps)
        hm[j] = max(0.0, h[j] - eps)
        d = hp[j] - hm[j]
        if d > 1e-14:
            g[j] = (_calc_c5(hp, dx) - _calc_c5(hm, dx)) / d
        else:
            g[j] = 0.0
    return g


def _lnbnbfgs_obj(h, dx, target_sum, mu):
    c5 = _calc_c5(h, dx)
    pen = mu * (np.sum(h) - target_sum) ** 2
    return c5 + pen


def _lnbnbfgs_grad(h, dx, target_sum, mu):

def _nm_obj(h, dx, target_sum, mu):
