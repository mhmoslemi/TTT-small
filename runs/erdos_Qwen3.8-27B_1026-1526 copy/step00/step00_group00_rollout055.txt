def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum_box(h, n):
    target = n * 0.5
    for _ in range(40):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            h = np.full(n, 0.5)
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)


def subgrad_vec(h, m):
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        cnt = n - m
        if cnt > 0:
            g[:cnt] += (1.0 - h[m:])
        if m < n:
            g[m:] -= h[:cnt]
    else:
        mm = -m
        cnt = n - mm
        g[mm:] += (1.0 - h[:cnt])
        g[:cnt] -= h[mm:]
    return g

    return h
