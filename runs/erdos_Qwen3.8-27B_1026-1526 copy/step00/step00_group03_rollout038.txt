import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target_sum) < 1e-9:
        return h
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        s = float(clipped.sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    e = np.exp(alpha * (corr - m))
    return m + float(np.log(np.sum(e))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    e = np.exp(alpha * (corr - m))
    total = float(np.sum(e))
    probs = e / total
    grad = np.zeros(n)

        p = float(probs[lag + n - 1]) * dx
        if p < 1e-14:

        lo = max(0, -lag)
        hi = min(n, n - lag)
        grad[lo + lag:hi + lag] += (1.0 - h[lo:hi]) * p
        grad[lo:hi] += -h[lo + lag:hi + lag] * p
    return grad

def sum_eq_obj(h):
    return float(h.sum()) - len(h) / 2.0

def sum_eq_jac(h):
    return np.ones(len(h))

def lse_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(corr))
    e = np.exp(alpha * (corr - m))
    return m + float(np.log(np.sum(e))) / alpha

def lse_grad(h, alpha):
