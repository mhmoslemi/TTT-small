    current_c5 = np.max(corr)
    
    for it in range(max_iter):
        improved = False
        # Try all pairs (or a subset)
        i, j = np.random.randint(0, N, 2)
        if i == j:
            continue
        
        a, b = h[i], h[j]
        # h[i] = a + t, h[j] = b - t
        # Need 0 ≤ a+t ≤ 1 and 0 ≤ b-t ≤ 1
        # So t ∈ [max(-a, b-1), min(1-a, b)]
        t_lo = max(-a, b - 1.0)
        t_hi = min(1.0 - a, b)
        if t_lo >= t_hi:
            continue
        
        # Binary search or golden section for optimal t
        # Objective: compute_c5(h with h[i]=a+t, h[j]=b-t)
        best_t = 0.0
        best_val = current_c5
        
        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        lo, hi = t_lo, t_hi
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        
        h[i], h[j] = a + c, b - c
        fc = compute_c5(h)

from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def get_corr_dx(h):
    """Get all correlation values multiplied by dx"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_obj_and_grad(h, beta=50.0):
    """Compute smooth approximation of max(corr_dx) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr_dx(h)
    
    # Weights (softmax)
    weights = np.exp(beta * (corr - np.max(corr)))
    weights /= weights.sum()
    
    obj = np.max(corr) + np.log(weights.sum()) / beta  # = logsumexp(beta*corr)/beta
    
    # Gradient: sum_l w_l * dg_l/dh
    # g_l(h) = dx * sum_{valid n} h[n+l](1-h[n]) where l = idx - (n-1)
    # dg_l/dh[j] = dx * [(1-h[j-l]) * 1_{valid} - h[j+l] * 1_{valid}]
    
    grad = np.zeros(n)
    # l ranges from -(n-1) to (n-1), indexed by corr index 0 to 2n-2
    # corr index i corresponds to lag l = i - (n-1)
    
    for i in range(2 * n - 1):
        l = i - (n - 1)  # lag
        w = weights[i]
        if w < 1e-15:
            continue
        # dg_l/dh[j] = dx * [(1-h[j-l]) * 1_{0<=j-l<n} - h[j+l] * 1_{0<=j+l<n}]
        # For each j:
        # Term 1: (1-h[j-l]) if 0 <= j-l < n, i.e., l <= j <= l+n-1
        # Term 2: -h[j+l] if 0 <= j+l < n, i.e., -l <= j <= n-1-l
        
        j_min1 = max(0, l)
        j_max1 = min(n - 1, n - 1 + l)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1+1] += w * dx * (1.0 - h[j_min1 - l : j_max1 - l + 1])
        
        j_min2 = max(0, -l)
        j_max2 = min(n - 1, n - 1 - l)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= w * dx * h[j_min2 + l : j_max2 + l + 1]
    
    return obj, grad

def coordinate_descent(h, N, max_sweeps=30, seed=42, tol=1e-10):

    rng = np.random.RandomState(seed)
    current_c5 = compute_c5(h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        # Generate pair order
        pairs = [(i, j) for i in range(N) for j in range(i + 1, N)]
        rng.shuffle(pairs)

        for i, j in pairs:
            a, b = h[i], h[j]
            t_lo = max(-a, b - 1.0) + 1e-12
            t_hi = min(1.0 - a, b) - 1e-12
            if t_lo >= t_hi:
                continue
            
            lo, hi = t_lo, t_hi
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            h[i], h[j] = a + c, b - c
            fc = compute_c5(h)
            h[i], h[j] = a + d, b - d
            fd = compute_c5(h)

                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    h[i], h[j] = a + c, b - c
                    fc = compute_c5(h)
                else:

                    c = d

def project_to_feasible(h, N):
    """Project h to satisfy sum(h) = N/2 and 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust to satisfy sum constraint
    for _ in range(100):

        if abs(diff) < 1e-12:

        if diff > 0:

            if not mask.any():
