def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        # Water-filling projection
        if diff > 0:
            # Need to decrease sum by diff
            # Decrease the largest values first
            order = np.argsort(h)[::-1]
            remaining = diff
            for idx in order:
                dec = min(h[idx], remaining)
                h[idx] -= dec
                remaining -= dec
                if remaining < 1e-12:
                    break
        else:
            # Need to increase sum by -diff
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                inc = min(1.0 - h[idx], remaining)
                h[idx] += inc
                remaining -= inc
                if remaining < 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, target_sum, n_passes=50, grid_size=200, penalty_w=50.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for p in range(n_passes):
        changed = False
        for i in range(n):
            old = h[i]
            best_t = old
            best_val = np.inf
            # Fine grid evaluation
            for t in np.linspace(0.0, 1.0, grid_size):
                h[i] = t
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                val = np.max(corr) + penalty_w * (np.sum(h) - target_sum) ** 2
                if val < best_val:
                    best_val = val
                    best_t = t
            h[i] = best_t
            if abs(best_t - old) > 1e-12:
                changed = True
        if not changed:
            break

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            order = np.argsort(h)[::-1]
            remaining = diff
            for idx in order:
                dec = min(h[idx], remaining)
                h[idx] -= dec
                remaining -= dec
                if remaining < 1e-12:
                    break
        else:
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                inc = min(1.0 - h[idx], remaining)
                h[idx] += inc
                remaining -= inc
                if remaining < 1e-12:
                    break
    return np.clip(h, 0.0, 1.0)


def coord_descent(h, target_sum, n_passes=50, grid_size=200, penalty_w=50.0):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    for p in range(n_passes):
        changed = False
        current_sum = np.sum(h)
        for i in range(n):
            old = h[i]
            best_t = old
            best_val = np.inf
            for t in grid:
                h[i] = t
                new_sum = current_sum - old + t
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                val = np.max(corr) + penalty_w * (new_sum - target_sum) ** 2
                if val < best_val:
                    best_val = val
                    best_t = t
            h[i] = best_t
            current_sum += (best_t - old)
            if abs(best_t - old) > 1e-12:
                changed = True
        if not changed:
            break
    return h


def slsqp_optimize(h, target_sum, maxiter=300):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    def obj(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x)

    try:
        res = minimize(obj, np.clip(h, 0, 1), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and not np.any(np.isnan(res.x)):
            return res.x
    except Exception:
        pass
    return h


def nm_optimize(h, target_sum, maxiter=5000):
    n = len(h)

    def obj(x):
        x = np.clip(x, 0, 1)
        x = project_feasible(x, target_sum)
        return compute_c5(x)

    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': 1e-13})
        if res.x is not None and not np.any(np.isnan(res.x)):
            return project_feasible(res.x, target_sum)
    except Exception:
        pass
    return h
