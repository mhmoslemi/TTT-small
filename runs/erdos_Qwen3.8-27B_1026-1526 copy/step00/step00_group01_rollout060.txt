                h_new[i] += delta

                mask[i] = False

                # Pick the one with most room
                room = np.minimum(h_new[candidates], 1.0 - h_new[candidates])

            break

            c = compute_c5(h)
            h[pos] -= rem
            if c < best_c:
                best_c = c
                best_pos = pos
        h[best_pos] += rem
    return project_h(h)

def slsqp_opt(h0, T, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_smooth, h0, args=(T,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cons_sum, 'jac': cons_sum_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_h(res.x)

def cdescent(h, niter, npair, seed):
    rng = np.random.default_rng(seed)
    n = len(h)
    bc = compute_c5(h)
    ds = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for _ in range(niter):
        imp = False
        for _ in range(npair):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            oh, oj = h[i], h[j]
            for d in ds:
                hi, hj = oh + d, oj - d
                if hi < -1e-9 or hi > 1.0 + 1e-9 or hj < -1e-9 or hj > 1.0 + 1e-9:
                    continue
                h[i], h[j] = hi, hj
                c = compute_c5(h)
                if c < bc - 1e-13:
                    bc = c
                    imp = True
                    break
                h[i], h[j] = oh, oj
            if imp:
                break
        if not imp:
            break
    return project_h(h), bc
