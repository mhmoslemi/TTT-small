```python
    # corr_k = sum_j h[j] * (1-h[j+k]) for valid j, j+k

    #   j = i: term is h[i]*(1-h[i+k]) for lag k where i+k < n
    #   j+k = i: term is h[j]*(1-h[i]) = h[i-k]*(1-h[i]) for lag k where i-k >= 0

    #   If i+k < n: corr_k has term h[i]*(1-h[i+k]) → coefficient (1-h[i+k]) for h[i]
    #   If i-k >= 0: corr_k has term h[i-k]*(1-h[i]) → coefficient -h[i-k] for h[i]

    #   a_k = (1-h[i+k]) if i+k < n else 0
    #          + (-h[i-k]) if i-k >= 0 else 0
    #   b_k = corr_k without the h[i] terms
```

```python
        # Find active lag at t_mid
        vals = intercepts + slopes * t_mid
        k_star = np.argmax(vals)
        # Unconstrained min of lam*(c+t)² + slopes[k_star]*t + intercepts[k_star]
        # f'(t) = 2*lam*(c+t) + slopes[k_star] = 0
        t_unc = -c - slopes[k_star] / (2.0 * lam)
        if sorted_bps[idx] <= t_unc <= sorted_bps[idx + 1]:
            all_candidates.add(t_unc)
    
    # Evaluate all candidates
    candidates = np.array(list(all_candidates))
    # f(t) = max_k(intercept[k] + slope[k]*t) + lam*(c+t)²
    # = max of (intercept[k] + slope[k]*t) for each k, then add penalty
    # Vectorized: for each candidate t, compute max over k
    # candidates shape: (N,), slopes shape: (m,)
    # matrix of slopes[k]*t for all k, t: (m, N)
    vals_matrix = intercepts[:, None] + slopes[:, None] * candidates[None, :]
    max_vals = np.max(vals_matrix, axis=0)
    penalty = lam * (c + candidates) ** 2
    f_vals = max_vals + penalty
    
    best_idx = np.argmin(f_vals)
    best_t = candidates[best_idx]
    
    h[i] = best_t
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _c5_value(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0.0, 1.0)
    return h

def _cd_step(h, i, target_sum, lam, dx):
    n = len(h)
    old_h_i = h[i]
    sum_others = np.sum(h) - old_h_i
    c = sum_others - target_sum

    h[i] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = 1.0
    corr1 = np.correlate(h, 1.0 - h, mode='full') * dx
    h[i] = old_h_i

    slopes = corr1 - corr0
    intercepts = corr0
    m = len(slopes)

    bps = [0.0, 1.0]
    for k in range(m):
        for j in range(k + 1, m):
            denom = slopes[k] - slopes[j]
            if abs(denom) > 1e-14:
                t = (intercepts[j] - intercepts[k]) / denom
                if 0.0 <= t <= 1.0:
                    bps.append(t)

    sorted_bps = sorted(set(bps))
    all_candidates = set(sorted_bps)
    for idx in range(len(sorted_bps) - 1):
        t_lo = sorted_bps[idx]
        t_hi = sorted_bps[idx + 1]
        if t_hi - t_lo < 1e-12:
            continue
        t_mid = (t_lo + t_hi) / 2.0
        vals = intercepts + slopes * t_mid
        k_star = int(np.argmax(vals))
        t_unc = -c - slopes[k_star] / (2.0 * lam)
        if t_lo <= t_unc <= t_hi:
            all_candidates.add(float(t_unc))

    candidates = np.array(sorted(all_candidates))
    if len(candidates) == 0:
        candidates = np.array([0.0, 1.0])

    vals_mat = intercepts[:, None] + slopes[:, None] * candidates[None, :]
    max_vals = np.max(vals_mat, axis=0)
    penalty = lam * (c + candidates) ** 2
    f_vals = max_vals + penalty

    best_idx = int(np.argmin(f_vals))
    best_t = float(candidates[best_idx])
    h[i] = best_t
    return h[i]

def _coordinate_descent(h, n, target_sum, lam, max_sweeps, start_time, budget_s):
    dx = 2.0 / n
    h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - start_time > budget_s * 0.85:
            break
        max_diff = 0.0
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            old = h[i]
            _cd_step(h, i, target_sum, lam, dx)
            max_diff = max(max_diff, abs(h[i] - old))
        if max_diff < 1e-10:
            break
    return h

def _slsqp_polish(h, n, target_sum):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return float(np.max(corr))

    def cons_eq(x):
        return np.sum(x) - target_sum

    constraints = [{'type': 'eq', 'fun': cons_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-14})
        h_out = np.clip(res.x, 0.0, 1.0)
        h_out = _project_feasible(h_out, target_sum)
        return h_out
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_val = float('inf')
    best_n = 45

    # Phase 1: Optimize from parent construction at n=45
    n = 45
    target_sum = n / 2.0
    h_init = np.array(initial_h_values, dtype=float)

    h_cd = _coordinate_descent(h_init, n, target_sum, lam=80.0,
                               max_sweeps=300, start_time=start_time, budget_s=budget_s)
    h_cd = _project_feasible(h_cd, target_sum)
    val = _c5_value(h_cd, n)
    if val < best_val:
        best_val = val
        best_h = h_cd.copy()
        best_n = n

    h_slsqp = _slsqp_polish(h_cd, n, target_sum)
    val = _c5_value(h_slsqp, n)
    if val < best_val:
        best_val = val
        best_h = h_slsqp.copy()
        best_n = n

    # Phase 2: Random restarts at n=45
    remaining = budget_s - (time.time() - start_time)
    n_restarts = max(1, int(remaining / 20))
    for r in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate diverse starting points
        strategy = r % 4
        if strategy == 0:
            h_r = best_h.copy()
            mask = np.random.random(n) < 0.25
            h_r[mask] += np.random.uniform(-0.4, 0.4, int(np.sum(mask)))
        elif strategy == 1:
            h_r = best_h.copy()
            idx = np.random.permutation(n)[:10]
            h_r[idx] = 1.0 - h_r[idx]
        elif strategy == 2:
            h_r = np.random.dirichlet(np.ones(n) * 2.0) * 2.0
            h_r = np.clip(h_r, 0.0, 1.0)
        else:
            h_r = np.zeros(n)
            k_half = n // 2
            perm = np.random.permutation(n)
            h_r[perm[:k_half]] = 1.0
            h_r[perm[k_half]] = n / 2.0 - k_half

        h_r = _project_feasible(h_r, target_sum)
        if np.any(h_r < -1e-12) or np.any(h_r > 1 + 1e-12):
            continue

        h_cd_r = _coordinate_descent(h_r, n, target_sum, lam=80.0,
                                     max_sweeps=150, start_time=start_time, budget_s=budget_s)
        h_cd_r = _project_feasible(h_cd_r, target_sum)
        val_r = _c5_value(h_cd_r, n)

        if time.time() - start_time < budget_s * 0.85:
            h_slsqp_r = _slsqp_polish(h_cd_r, n, target_sum)
            val_r2 = _c5_value(h_slsqp_r, n)
            if val_r2 < val_r:
                val_r = val_r2
                h_cd_r = h_slsqp_r

        if val_r < best_val:
            best_val = val_r
            best_h = h_cd_r.copy()
            best_n = n

    # Phase 3: Try finer grid n=90 if time permits
    if time.time() - start_time < budget_s * 0.5:
        n2 = 90
        target_sum2 = n2 / 2.0

        # Upsample best solution
        h_up = np.zeros(n2)
        ratios = np.arange(n2) * n / n2
        for j in range(n2):
            idx0 = int(ratios[j])
            if idx0 >= n:
                idx0 = n - 1
            idx1 = min(idx0 + 1, n - 1)
            frac = ratios[j] - idx0
            h_up[j] = (1 - frac) * best_h[idx0] + frac * best_h[idx1]

        h_up = _project_feasible(h_up, target_sum2)
        h_cd2 = _coordinate_descent(h_up, n2, target_sum2, lam=80.0,
                                    max_sweeps=100, start_time=start_time, budget_s=budget_s)
        h_cd2 = _project_feasible(h_cd2, target_sum2)
        val2 = _c5_value(h_cd2, n2)
        if time.time() - start_time < budget_s * 0.8:
            h_cd2 = _slsqp_polish(h_cd2, n2, target_sum2)
            val2 = _c5_value(h_cd2, n2)

        if val2 < best_val:
            best_val = val2
            best_h = h_cd2.copy()
            best_n = n2

        # A few restarts at n=90
        for r in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            h_r2 = best_h.copy()
            mask = np.random.random(n2) < 0.2
            h_r2[mask] += np.random.uniform(-0.3, 0.3, int(np.sum(mask)))
            h_r2 = _project_feasible(h_r2, target_sum2)
            h_cd_r2 = _coordinate_descent(h_r2, n2, target_sum2, lam=80.0,
                                          max_sweeps=80, start_time=start_time, budget_s=budget_s)
            h_cd_r2 = _project_feasible(h_cd_r2, target_sum2)
            val_r2 = _c5_value(h_cd_r2, n2)
            if val_r2 < best_val:
                if time.time() - start_time < budget_s * 0.9:
                    h_cd_r2 = _slsqp_polish(h_cd_r2, n2, target_sum2)
                    val_r2 = _c5_value(h_cd_r2, n2)
                if val_r2 < best_val:
                    best_val = val_r2
                    best_h = h_cd_r2.copy()
                    best_n = n2

    # Final: ensure feasibility
    target_sum_final = best_n / 2.0
    best_h = _project_feasible(best_h, target_sum_final)
    final_val = _c5_value(best_h, best_n)

    if final_val > best_val:
        best_val = final_val

    return best_h, best_val, best_n
```
