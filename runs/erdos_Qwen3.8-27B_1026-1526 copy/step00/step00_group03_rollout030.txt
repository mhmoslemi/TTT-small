```python
def compute_grad(corr_weights, h, n):
    """
    corr_weights: array of length 2n-1, the softmax weights
    Returns gradient of sum_k weights[k] * corr[k] w.r.t. h
    """
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = corr_weights[j]
        if w < 1e-14:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            # First factor: m in [lag, n-1], contributes (1-h[m-lag])
            # h[m-lag] for m=lag..n-1 is h[0..n-1-lag]
            grad[lag:] += w * (1.0 - h[:n - lag])
            # Second factor: m in [0, n-1-lag], contributes -h[m+lag]
            # h[m+lag] for m=0..n-1-lag is h[lag..n-1]
            grad[:n - lag] += w * (-h[lag:])
        else:
            L = -lag  # L > 0
            # First factor: m in [0, n-1-L], contributes (1-h[m+L])
            # h[m+L] for m=0..n-1-L is h[L..n-1]
            grad[:n - L] += w * (1.0 - h[L:])
            # Second factor: m in [L, n-1], contributes -h[m-L]
            # h[m-L] for m=L..n-1 is h[0..n-1-L]
            grad[L:] += w * (-h[:n - L])
    return grad
```

```python
def project_polytope(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

        T *= 0.75
        step *= 0.85
        if T < 0.002:
            T = 0.002
        if step < 0.0005:
            step = 0.0005

        if stall_count > 500:
            h = best_h + 0.02 * np.random.randn(n_points)
            h = project_polytope(h, target_sum)
            v = np.zeros(n_points)
            stall_count = 0

    # Phase 2: Nelder-Mead polishing
    if time.time() - t_start < elapsed_limit_phase2:
        cap_time = budget_s - (time.time() - t_start) - budget_s * 0.20
        if cap_time > 30:
            def nm_obj(x):
                x = project_polytope(x, target_sum)
                return compute_c5(x, n_points)

            pert = 0.005
            simplex = [best_h + pert * np.random.randn(n_points)]
            simplex = [project_polytope(s, target_sum) for s in simplex]
            for i in range(n_points):
                s = best_h.copy()
                s[i] += pert
                simplex.append(project_polytope(s, target_sum))

            res = minimize(nm_obj, best_h, method='Nelder-Mead',
                           options={'maxiter': 3000, 'fatol': 1e-10,
                                   'xatol': 1e-10, 'initial_simplex': np.array(simplex),
                                   'adaptive': True})
            h_nm = project_polytope(res.x, target_sum)
            c5_nm = compute_c5(h_nm, n_points)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()

    # Phase 3: Local search with random restarts
    if time.time() - t_start < elapsed_limit_phase3:
        T_local = 0.005
        step_local = 0.005
        v_local = np.zeros(n_points)
        for restart in range(50):
            if time.time() - t_start > elapsed_limit_phase3:
                break
            h_local = best_h + 0.03 * np.random.randn(n_points)
            h_local = project_polytope(h_local, target_sum)
            v_local = np.zeros(n_points)
            for it in range(300):
                if time.time() - t_start > elapsed_limit_phase3:
                    break
                _, grad = smoothObjectiveAndGrad(h_local, n_points, T_local)
                v_local = 0.85 * v_local - step_local * grad
                h_try = h_local + v_local
                h_try = project_polytope(h_try, target_sum)
                obj_cur = compute_c5(h_local, n_points)
                obj_try = compute_c5(h_try, n_points)
                if obj_try < obj_cur:
                    h_local = h_try
                else:
                    v_local *= 0.5
                    step_local *= 0.95
                    if step_local < 1e-5:
                        step_local = 0.005
                        break
            c5_local = compute_c5(h_local, n_points)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()

    # Final refinement with very small steps
    if time.time() - t_start < budget_s - 10:
        T_fin = 0.001
        step_fin = 0.001
        v_fin = np.zeros(n_points)
        for _ in range(500):
            if time.time() - t_start > budget_s - 10:
                break
            _, grad = smoothObjectiveAndGrad(best_h, n_points, T_fin)
            v_fin = 0.8 * v_fin - step_fin * grad
            h_try = best_h + v_fin
            h_try = project_polytope(h_try, target_sum)
            c5_try = compute_c5(h_try, n_points)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    best_h = project_polytope(best_h, target_sum)
    best_c5 = compute_c5(best_h, n_points)
    return best_h, best_c5, n_points
```
