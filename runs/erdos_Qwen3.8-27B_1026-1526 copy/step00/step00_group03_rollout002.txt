import numpy as np
from scipy.optimize import minimize
import time


def _project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    mid = 0.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + mid, 0.0, 1.0)


def _c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _sfun(x, ts):
    return np.sum(x) - ts


def _sjac(x, ts):
    return np.ones(x.shape[0])


def _sobj(x, dx):
    return _c5(x, dx)


def _nmobj(x, dx, ts, pw):
    xc = np.clip(x, 0.0, 1.0)
    return _c5(xc, dx) + pw * (np.sum(xc) - ts) ** 2


def _tleft(t0, budget):
    return budget - (time.time() - t0)


def _pair_cd(h, n, dx, t0, budget, max_sweeps):
    best = _c5(h, dx)
    for sweep in range(max_sweeps):
        if _tleft(t0, budget) < 30:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                tlo = max(-h[i], h[j] - 1.0)
                thi = min(1.0 - h[i], h[j])
                if thi - tlo < 1e-10:
                    continue
                grid = np.linspace(tlo, thi, 13)
                lb = best
                lt = 0.0
                for t in grid:
                    h[i] += t
                    h[j] -= t
                    v = _c5(h, dx)
                    if v < lb:
                        lb = v
                        lt = t
                    h[i] -= t
                    h[j] += t
                if lt != 0.0:
                    h[i] += lt
                    h[j] -= lt
                    best = lb
                    improved = True
        if not improved:
            break
    return h, best


def _nm_run(h0, dx, ts, pw, n, maxit, xatol, fatol):
    try:
        r = minimize(_nmobj, h0, method='Nelder-Mead',
                     args=(dx, ts, pw),
                     options={'maxiter': maxit, 'xatol': xatol,
                              'fatol': fatol, 'adaptive': True})
        return _project(r.x, n)
    except Exception:
        return h0


def _slsqp_run(h0, dx, ts, n, maxit):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sfun, 'jac': _sjac, 'args': (ts,)}]
    try:
        r = minimize(_sobj, h0, method='SLSQP', bounds=bnds,
                     constraints=cons, args=(dx,),
                     options={'maxiter': maxit, 'ftol': 1e-12})
        return _project(r.x, n)
    except Exception:
        return h0
