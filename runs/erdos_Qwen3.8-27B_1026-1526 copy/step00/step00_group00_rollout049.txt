        if c5_try < current_c5:
            h = h_try
            accepted = True
        else:
            for f in [0.5, 0.25, 0.1]:
                h_try = project_simplex(h - lr * f * grad, target)
                if compute_c5(h_try) < current_c5:
                    h = h_try
                    accepted = True
                    break
        if not accepted and frac > 0.3:
            noise = rng.normal(0, 0.01 * (1.0 - frac), n)
            h = project_simplex(h + noise, target)

    return best_h, best_c5

def opt_anneal(h_init, n, duration_s, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    target = n / 2.0
    start = time.time()
    T_init = 0.04
    T_end = 0.0003

    while time.time() - start < duration_s:
        elapsed = time.time() - start
        frac = min(elapsed / duration_s, 1.0)
        T = T_end + (T_init - T_end) * (1.0 - frac) ** 2

        batch = 300
        for _ in range(batch):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = rng.uniform(0.005, 0.25)
            if rng.random() < 0.5:
                sign = 1.0
            else:
                sign = -1.0

            h_new = h.copy()
            h_new[i] += sign * delta
            h_new[j] -= sign * delta
            if h_new[i] < -1e-9 or h_new[i] > 1.0 + 1e-9 or h_new[j] < -1e-9 or h_new[j] > 1.0 + 1e-9:
                continue

            c5_new = compute_c5(h_new)
            diff = c5_new - current_c5
            if diff < 0 or (T > 1e-12 and rng.random() < np.exp(-diff / T)):
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()

    return best_h, best_c5

def make_alternating(n):
    h = np.zeros(n)
    h[::2] = 1.0
    target = n / 2.0
    s = np.sum(h)
    if s > target:
        excess = s - target
        ones = np.where(h == 1.0)[0]
        h[ones[0]] -= excess
    elif s < target:
        deficit = target - s
        zeros = np.where(h == 0.0)[0]
        h[zeros[0]] += deficit
    return project_simplex(h, target)

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_simplex(h_dst, n_dst / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = make_alternating(45)

    n = len(h_init)
    best_h = np.clip(h_init, 0.0, 1.0)
    best_h = project_simplex(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    h_alt = make_alternating(n)
    c5_alt = compute_c5(h_alt)
    if c5_alt < best_c5:
        best_h, best_c5 = h_alt, c5_alt

    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        d = min(remaining * 0.30, 300)
        h_s, c5_s = opt_subgradient(best_h, n, d, rng)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        d = min(remaining * 0.25, 250)
        h_a, c5_a = opt_anneal(best_h, n, d, rng)
        if c5_a < best_c5:
            best_h, best_c5 = h_a, c5_a

    n_candidates = [n + 1, 50, 60, 80, 100, 120, 150]
    n_candidates = [x for x in n_candidates if x != n and x >= 10]

    for n_try in n_candidates:
        remaining = budget_s - (time.time() - t0)
        if remaining < 120:
            break

        h_try = interpolate_h(h_init, n, n_try)
        h_alt2 = make_alternating(n_try)
        if compute_c5(h_alt2) < compute_c5(h_try):
            h_try = h_alt2

        d1 = min(remaining * 0.12, 200)
        h_opt1, c5_opt1 = opt_subgradient(h_try, n_try, d1, rng)

        remaining = budget_s - (time.time() - t0)
        if remaining > 40:
            d2 = min(remaining * 0.08, 100)
            h_opt2, c5_opt2 = opt_anneal(h_opt1, n_try, d2, rng)
            if c5_opt2 < c5_opt1:
                h_opt1, c5_opt1 = h_opt2, c5_opt2

        if c5_opt1 < best_c5:
            best_h, best_c5 = h_opt1, c5_opt1

    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_final_try = len(best_h)
        d = min(remaining * 0.6, 300)
        h_f, c5_f = opt_anneal(best_h, n_final_try, d, rng)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_final_try = len(best_h)
        d = min(remaining * 0.5, 200)
        h_f2, c5_f2 = opt_subgradient(best_h, n_final_try, d, rng)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2

    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_simplex(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    elapsed = time.time() - t0
    if elapsed > budget_s:
        pass

    return (best_h, best_c5, n_final)
