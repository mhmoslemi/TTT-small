    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_onto_box_sum(v, n):
    """Project v onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    v = np.array(v, dtype=float).copy()
    # Bisection on lambda: find lambda s.t. sum(clip(v - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining error
    diff = target - np.sum(v)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if np.any(mask):
            v[mask] += diff / np.sum(mask)
            v = np.clip(v, 0.0, 1.0)
    return v

    """Compute gradient of c[j_star] w.r.t. h, where c = correlate(h, 1-h, 'full')"""
    k = j_star - n + 1  # the actual shift
    grad = np.zeros(n)

def projected_subgradient(h, n, max_iter=5000, lr_init=0.01):

    dx = 2.0 / n
    h = h.copy()

def grad_corr_comp(h, j, n):
    """Gradient of c[j] (before dx scaling) w.r.t. h"""
    s = j - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    # First term: (1-h[t+s]) for 0 <= t+s < n
    if s >= 0:
        idx = np.arange(0, n - s)
        grad[idx] += (1.0 - h[idx + s])
    else:
        idx = np.arange(-s, n)
        grad[idx] += (1.0 - h[idx + s])
    # Second term: -h[t-s] for 0 <= t-s < n
    if s >= 0:
        idx = np.arange(s, n)
        grad[idx] -= h[idx - s]
    else:
        idx = np.arange(0, n + s)
        grad[idx] -= h[idx - s]
    return grad * dx

def soft_max_grad(h, n, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*c_j))) where c = correlate(h,1-h,'full')*dx"""
    c5_val, c_vals = compute_c5(h)
    # Soft max weights
    shifted = beta * c_vals - beta * np.max(c_vals)
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Gradient = sum_j w_j * grad c[j]
    grad = np.zeros(n)
    # For efficiency, only compute grad for top contributors
    top_k = min(50, len(c_vals))
    top_idx = np.argsort(c_vals)[-top_k:]
    for j in top_idx:
        w = weights[j]
        if w > 1e-6:
            grad += w * grad_corr_comp(h, j, n)
    return grad, c5_val

        return x[-1]

        t = x[-1]

        return t - c
    
    def eq(x):
        h = x[:-1]
        return np.sum(h) - 0.5 * len(h)
    
    def eq_jac(x):
        n_h = len(x) - 1
        jy = np.zeros(len(x))
        jy[:n_h] = 1.0
        return jy
    
    x0 = np.concatenate([h_init, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.01, c5_init * 0.8), 2.0)]
    constraints = [

        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    
    if result.x is not None:
        h_out = project_box_sum(result.x[:-1], n)
        return h_out, compute_c5(h_out)[0]
    return h_init, c5_init

I think the coordinate descent idea is interesting. If I fix all h[j] for j ≠ i, then:
f_k(h) = sum_m h[m]*(1-h[m+s_k])
The terms involving h[i] are:

We want to minimize max_k (b_k + a_k * x) over x ∈ [0,1].

This is: find x that minimizes the upper envelope of lines {b_k + a_k * x : k = 0, ..., 2n-2}.

The minimum of the upper envelope of lines occurs at:
- A boundary (x=0 or x=1), or
- An intersection of two lines where the upper envelope changes

To find this:
1. Compute all b_k + a_k * x at x=0 and x=1
2. The minimum of the max at x=0 is max_k b_k
3. The minimum of the max at x=1 is max_k (b_k + a_k)
4. The minimum might be at an interior point where two lines intersect

For the interior point: we need to find the x where two lines cross and both are on the upper envelope.

A simpler approach: ternary search on x ∈ [0,1] since the function max_k(b_k + a_k*x) is convex in x (max of linear functions is convex).

Ternary search: O(n log(1/ε)) evaluations, each O(n). Total: O(n² log(1/ε)).

def coordinate_descent_step(h, i, n):
    """Optimize h[i] by coordinate descent (ignoring sum constraint for now)"""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # current correlation
    
    # Compute a_k and b_k for each k
    # f_k = b_k + a_k * h[i]
    # a_k = (1 - h[i+s_k]) - h[i-s_k] (with bounds checking)
    # b_k = f_k(0) = c[k] - a_k * h[i] (i.e., f_k with h[i] removed)
    
    # Actually, let's compute directly
    # c[k] = sum_m h[m] * (1-h[m+s_k])
    # The terms involving h[i]:
    #   m=i: h[i] * (1-h[i+s_k])
    #   m=i-s_k: h[i-s_k] * (1-h[i])
    # So c[k] = (c[k] - h[i]*(1-h[i+s_k]) - h[i-s_k]*(1-h[i])) + h[i]*(1-h[i+s_k]) + h[i-s_k]*(1-h[i])
    #          = c[k] - h[i]*(1-h[i+s_k]) - h[i-s_k] + h[i-s_k]*h[i] + h[i]*(1-h[i+s_k]) + h[i-s_k] - h[i-s_k]*h[i]
    # Hmm, this is circular. Let me just compute b_k and a_k directly.
    
    # For shift s = k - (n-1):
    # c[k] = sum_m h[m] * (1-h[m+s]) for valid m
    # Terms with h[i]:
    #   m=i: contributes h[i] * (1-h[i+s])
    #   m+s=i, i.e., m=i-s: contributes h[i-s] * (1-h[i])
    
    s_vals = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to n-1
    
    # a_k: coefficient of h[i] in c[k]
    # From m=i: (1 - h[i+s_k]) if 0 <= i+s_k < n
    # From m=i-s_k: -h[i-s_k] if 0 <= i-s_k < n
    a = np.zeros(2*n-1)
    for idx, s in enumerate(s_vals):
        coeff = 0.0
        ip = i + s
        im = i - s
        if 0 <= ip < n:
            coeff += (1.0 - h[ip])
        if 0 <= im < n:
            coeff -= h[im]
        a[idx] = coeff
    
    # b_k: c[k] evaluated at h[i] = 0
    # c[k] = sum_m h[m]*(1-h[m+s]) with h[i] removed
    # = c[k] - a[k]*h[i] (since c[k] = b_k + a_k*h[i])
    b = c - a * h[i]
    
    # Now minimize max_k (b_k + a_k * x) over x in [0,1]
    # Use ternary search (the function is convex)
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(b + a * m1)
        f2 = np.max(b + a * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    x_opt = 0.5 * (lo + hi)
    return x_opt

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c


def project_box_sum(v, n):
    target = 0.5 * n
    v = np.array(v, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(v)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = v < 1.0
        else:
            mask = v > 0.0
        if np.any(mask):
            v[mask] += diff / np.sum(mask)
            v = np.clip(v, 0.0, 1.0)
    return v


def slsqp_solve(h_init, n, maxiter=2000, tol_extra=0.01):
    c5_init, _ = compute_c5(h_init)

    def obj_func(x):
        return x[-1]

    def ineq_func(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
        return t - c

    def eq_func(x):
        h = x[:-1]
        return np.sum(h) - 0.5 * len(h)

    def eq_jac_func(x):
        n_h = len(x) - 1
        jy = np.zeros(len(x))
        jy[:n_h] = 1.0
        return jy

    x0 = np.concatenate([h_init, [c5_init + tol_extra]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.85), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]

    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

    if result.x is not None:
        h_out = project_box_sum(result.x[:-1], n)
        c5_out = compute_c5(h_out)[0]
        return h_out, c5_out
    return h_init, c5_init


def coordinate_descent(h, n, max_sweeps=200, time_limit=None, t_start=None):
    s_vals = np.arange(2 * n - 1) - (n - 1)
    h = h.copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        any_change = False
        for i in range(n):
            c = np.correlate(h, 1.0 - h, mode='full')
            a = np.zeros(2 * n - 1)
            ip = i + s_vals
            im = i - s_vals
            mask_p = (ip >= 0) & (ip < n)
            mask_m = (im >= 0) & (im < n)
            a[mask_p] += (1.0 - h[ip[mask_p]])
            a[mask_m] -= h[im[mask_m]]
            b = c - a * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(b + a * m1)
                f2 = np.max(b + a * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_opt = 0.5 * (lo + hi)
            if abs(x_opt - h[i]) > 1e-12:
                h[i] = x_opt
                any_change = True

        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, n)
        c5_now = compute_c5(h)[0]
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if not any_change:
            break

    return best_h, best_c5


def soft_max_slsqp(h_init, n, beta, maxiter=500):
    dx = 2.0 / n

    def obj_func(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        vals = beta * c * dx
        vals = vals - np.max(vals)
        smax = np.log(np.sum(np.exp(vals))) / beta
        return t + 1e6 * max(0, smax - t)

    def ineq_func(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full')
        vals = beta * c * dx
        vals = vals - np.max(vals)
        smax = np.log(np.sum(np.exp(vals))) / beta
        return t - smax

    def eq_func(x):
        h = x[:-1]
        return np.sum(h) - 0.5 * len(h)

    def eq_jac_func(x):
        n_h = len(x) - 1
        jy = np.zeros(len(x))
        jy[:n_h] = 1.0
        return jy

    c5_init, _ = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.01, c5_init * 0.85), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]

    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})

    if result.x is not None:
        h_out = project_box_sum(result.x[:-1], n)
        return h_out, compute_c5(h_out)[0]
    return h_init, c5_init
