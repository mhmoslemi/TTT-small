    h = h.astype(float).copy()
    lags = compute_lags(h)
    dx = 2.0 / n
    best_obj = float(np.max(lags) + mu * (float(np.sum(h)) - n / 2.0) ** 2)

    for sweep in range(max_sweeps):
        max_change = 0.0
        S = float(np.sum(h)) - n / 2.0

        for j in range(n):
            old_val = h[j]
            lags_save = lags.copy()
            S_save = S

            ks = np.arange(1, n)
            mask1 = ks <= j
            mask2 = ks <= (n - 1 - j)

            if np.any(mask1):
                ks1 = ks[mask1]
                idx1 = j - ks1
                coef1 = (1.0 - h[idx1]) * dx
            else:
                ks1 = np.array([], dtype=int)
                coef1 = np.array([])

            if np.any(mask2):
                ks2 = ks[mask2]
                idx2 = j + ks2
                coef2 = -h[idx2] * dx
            else:
                ks2 = np.array([], dtype=int)
                coef2 = np.array([])

            f1d = partial(_f1d_func, old_val=old_val, lags_save=lags_save,
                          ks1=ks1, ks2=ks2, coef1=coef1, coef2=coef2,
                          dx=dx, mu=mu, S_save=S_save)
            res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-14})
            new_val = float(res.x)
            change = abs(new_val - old_val)
            max_change = max(max_change, change)

            diff = new_val - old_val
            lags[0] += (new_val * (1.0 - new_val) - old_val * (1.0 - old_val)) * dx
            if len(ks1) > 0:
                lags[ks1] += diff * coef1
            if len(ks2) > 0:
                lags[ks2] += diff * coef2
            h[j] = new_val
            S += diff

        obj = float(np.max(lags) + mu * S ** 2)
        if obj < best_obj:
            best_obj = obj
        if max_change < tol:
            break

    return h, float(np.max(compute_lags(h)))


def _slsqp_obj(x, n):
    lags = compute_lags(x)
    return float(np.max(lags))


def _slsqp_grad(x, n):
    lags = compute_lags(x)
    k_star = int(np.argmax(lags))
    dx = 2.0 / n
    grad = np.zeros(n)
    k = k_star
    if k == 0:
        grad[:] = (1.0 - 2.0 * x) * dx
    else:
        for j in range(n):
            g = 0.0
            if j >= k:
                g += (1.0 - x[j - k])
            if j < n - k:
                g -= x[j + k]
            grad[j] = g * dx
    return grad


def _slsqp_eq(x, n):
    return float(np.sum(x) - n / 2.0)


def _slsqp_eq_jac(x, n):
    return np.ones(n)


def slsqp_polish(h_init, n, maxiter=1500):
    h = h_init.copy()
    c5_init = compute_c5_from_h(h)
    obj_p = partial(_slsqp_obj, n=n)
    grad_p = partial(_slsqp_grad, n=n)
    eq_p = partial(_slsqp_eq, n=n)
    eq_jac_p = partial(_slsqp_eq_jac, n=n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_p, 'jac': eq_jac_p}]
    result = minimize(obj_p, h, jac=grad_p, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_box_sum(result.x, n / 2.0)
        return h_out, compute_c5_from_h(h_out)
    return h, c5_init


def _smooth_max_obj(x, n, alpha):
    lags = compute_lags(x)
    m = np.max(lags)
    if m < -1000:
        return m
    return float(m + np.log(np.sum(np.exp(alpha * (lags - m)))) / alpha)


def _smooth_max_grad(x, n, alpha):
    lags = compute_lags(x)
    m = np.max(lags)
    weights = np.exp(alpha * (lags - m))
    weights /= np.sum(weights)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        if weights[k] < 1e-15:
            continue
        wk = weights[k]
        if k == 0:
            grad[:] += wk * (1.0 - 2.0 * x) * dx
        else:
            for j in range(n):
                g = 0.0
                if j >= k:
                    g += (1.0 - x[j - k])
                if j < n - k:
                    g -= x[j + k]
                grad[j] += wk * g * dx
    return grad


def lbfgsb_smooth(h_init, n, alpha=50.0, maxiter=300):
    x0 = h_init.copy()
    obj_p = partial(_smooth_max_obj, n=n, alpha=alpha)
    grad_p = partial(_smooth_max_grad, n=n, alpha=alpha)
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_p, x0, jac=grad_p, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_box_sum(result.x, n / 2.0)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    n = len(h0)
    h0 = project_box_sum(h0, n / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    if time.time() - t0 < budget_s - 200:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, max_sweeps=40, mu=3.0)
        h_cd = project_box_sum(h_cd, n_best / 2.0)
        c5_cd = compute_c5_from_h(h_cd)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    if time.time() - t0 < budget_s - 150:
        h_sp, c5_sp = slsqp_polish(best_h, n_best, maxiter=2000)
        if c5_sp < best_c5:
            best_h = h_sp
            best_c5 = c5_sp
            n_best = len(best_h)

    if time.time() - t0 < budget_s - 120:
        h_lb, c5_lb = lbfgsb_smooth(best_h, n_best, alpha=80.0, maxiter=500)
        if c5_lb < best_c5:
            best_h = h_lb
            best_c5 = c5_lb
            n_best = len(best_h)

    resolutions = [200, 270, 360, 480]
    for n_alt in resolutions:
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_alt = np.clip(f_interp(x_new), 0.0, 1.0)
        h_alt = project_box_sum(h_alt, n_alt / 2.0)

        if time.time() - t0 < budget_s - 80:
            h_cd, c5_cd = coordinate_descent(h_alt, n_alt, max_sweeps=20, mu=3.0)
            h_cd = project_box_sum(h_cd, n_alt / 2.0)
            c5_cd = compute_c5_from_h(h_cd)

            if time.time() - t0 < budget_s - 50:
                h_sp, c5_sp = slsqp_polish(h_cd, n_alt, maxiter=1000)
                if c5_sp < c5_cd:
                    h_cd = h_sp
                    c5_cd = c5_sp

            if c5_cd < best_c5:
                best_h = h_cd
                best_c5 = c5_cd
                n_best = n_alt

    if time.time() - t0 < budget_s - 100:
        for ri in range(8):
            if time.time() - t0 > budget_s - 40:
                break
            scale = 0.03 + 0.04 * ri / 8.0
            h_pert = best_h + scale * rng.standard_normal(n_best)
            h_pert = project_box_sum(h_pert, n_best / 2.0)
            h_cd, c5_cd = coordinate_descent(h_pert, n_best, max_sweeps=15, mu=3.0)
            h_cd = project_box_sum(h_cd, n_best / 2.0)
            c5_cd = compute_c5_from_h(h_cd)
            if c5_cd < best_c5:
                best_h = h_cd
                best_c5 = c5_cd
                n_best = len(best_h)

    if time.time() - t0 < budget_s - 60:
        for try_alpha in [100.0, 200.0]:
            if time.time() - t0 > budget_s - 40:
                break
            h_lb, c5_lb = lbfgsb_smooth(best_h, n_best, alpha=try_alpha, maxiter=300)
            if c5_lb < best_c5:
                best_h = h_lb
                best_c5 = c5_lb
                n_best = len(best_h)

    if time.time() - t0 < budget_s - 30:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, max_sweeps=30, mu=2.0)
        h_cd = project_box_sum(h_cd, n_best / 2.0)
        c5_cd = compute_c5_from_h(h_cd)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    if time.time() - t0 < budget_s - 20:
        h_sp, c5_sp = slsqp_polish(best_h, n_best, maxiter=1500)
        if c5_sp < best_c5:
            best_h = h_sp
            best_c5 = c5_sp
            n_best = len(best_h)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
