def compute_c5(h):

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total_room = room.sum()
            if total_room <= 0:
                break
            frac = min(1.0, diff / total_room)
            h[mask] += frac * room
        else:
            mask = h > 0.0
            if not np.any(mask):

            room = h[mask]
            total_room = room.sum()
