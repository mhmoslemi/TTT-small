    except Exception:
        return h

def generate_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)
    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
            starts.append(np.clip(parent_h + rng.randn(n) * 0.05, 0, 1))
            starts.append(np.clip(parent_h + rng.randn(n) * 0.15, 0, 1))
        else:
            x_p = np.linspace(0, 2, len(parent_h), endpoint=False)
            f = interp1d(x_p, parent_h, kind='cubic', fill_value='extrapolate')
            hi_interp = np.clip(f(x), 0, 1)
            starts.append(hi_interp)
            starts.append(np.clip(hi_interp + rng.randn(n) * 0.08, 0, 1))
    for freq in [1.3, 1.7, 2.1, 2.5, 3.1, 3.7, 4.3]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.28, 0.46)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), 0, 1))
    for _ in range(4):
        starts.append(rng.rand(n))
    h_saw = np.mod(ar / float(n) * 3.0, 1.0)
    starts.append(h_saw)
    c1, c2 = 0.3 * n, 0.7 * n
    w = 0.15 * n
    b1 = np.exp(-0.5 * ((ar - c1) / w) ** 2)
    b2 = np.exp(-0.5 * ((ar - c2) / w) ** 2)
    starts.append(np.clip(0.1 + 0.8 * 0.5 * (b1 + b2), 0, 1))
    h_lin1 = np.linspace(0.05, 0.95, n)
    starts.append(h_lin1)
    starts.append(rng.rand(n) * 0.6 + 0.2)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_list = [59, 61, 63, 57, 55, 65, 53, 67, 49, 51, 69, 71, 73, 75]
    if parent_h is not None:
        n_list.remove(len(parent_h)) if len(parent_h) in n_list else None
        n_list.insert(0, len(parent_h))

    t_p1 = start_time + budget_s * 0.45
    t_p2 = start_time + budget_s * 0.82
    t_p3 = start_time + budget_s * 0.97

    for n in n_list:
        if time.time() > t_p1:
            break
        starts = generate_starts(n, parent_h, rng)
        for h0 in starts:
            if time.time() > t_p1:
                break
            h0 = project_feasible(h0, n)
            current_h = h0.copy()
            taus = [0.1, 0.04, 0.015]
            for tau in taus:
                if time.time() > t_p1:
                    break
                current_h = slsqp_optimize(current_h, n, tau, maxiter=50)
            current_h, c5 = coord_desc(current_h, n, max_rounds=5,
                                       time_budget=t_p1, start_time=start_time)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()

    if best_h is None:
        if parent_h is not None:
            best_h = project_feasible(parent_h.copy(), len(parent_h))
            best_c5 = evaluate_c5(best_h)
        else:
            best_h = project_feasible(np.full(59, 0.5), 59)
            best_c5 = evaluate_c5(best_h)

    n = len(best_h)
    best_h, best_c5 = coord_desc(best_h, n, max_rounds=20, time_budget=t_p2 * 0.5, start_time=start_time)

    pert_scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35]
    it = 0
    while time.time() < t_p2:
        it += 1
        progress = (time.time() - start_time - (t_p1 - start_time)) / max(1, t_p2 - t_p1)
        if progress < 0.3:
            scale = pert_scales[rng.randint(2, len(pert_scales))]
        elif progress < 0.7:
            scale = pert_scales[rng.randint(0, len(pert_scales))]
        else:
            scale = pert_scales[rng.randint(0, 4)]
        pert = best_h + rng.randn(n) * scale
        pert = project_feasible(pert, n)
        pert, c5 = coord_desc(pert, n, max_rounds=6, time_budget=t_p2, start_time=start_time)
        if c5 < best_c5:
            best_c5 = c5
            best_h = pert.copy()

    if time.time() < t_p3:
        for n_try in [n + 2, n - 2, n + 4, n + 6, n - 4]:
            if n_try < 30 or n_try > 120:
                continue
            if time.time() > t_p3:
                break
            x_orig = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f = interp1d(x_orig, best_h, kind='cubic', fill_value='extrapolate')
            h_try = np.clip(f(x_new), 0, 1)
            h_try = project_feasible(h_try, n_try)
            h_try, c5 = coord_desc(h_try, n_try, max_rounds=25, time_budget=t_p3, start_time=start_time)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_try.copy()
                n = n_try

    if time.time() < start_time + budget_s * 0.99:
        n = len(best_h)
        best_h, best_c5 = coord_desc(best_h, n, max_rounds=200,
                                      time_budget=start_time + budget_s * 0.99, start_time=start_time)

    return best_h, best_c5, len(best_h)
