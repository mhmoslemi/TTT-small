        # a_k = (1-h[j+k]) * [j+k valid] - h[j-k] * [j-k valid]
        # Note: in correlate "full" mode, lag k in the output corresponds to shift (k-(n-1))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    target = n / 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = result < 1.0 - 1e-12
            if mask.any():
                idx = np.where(mask)[0]
                add = diff / len(idx)
                result[idx] = np.minimum(result[idx] + add, 1.0)
            else:
                result[np.argmin(result)] += diff
        else:
            mask = result > 1e-12
            if mask.any():
                idx = np.where(mask)[0]
                sub = -diff / len(idx)
                result[idx] = np.maximum(result[idx] - sub, 0.0)
            else:
                result[np.argmax(result)] += diff
    return result

def minimize_1d_max(a, b):
    """Minimize max_i (a_i * t + b_i) for t in [0, 1]."""
    def f(t):
        return np.max(a * t + b)
    
    def subgrad_condition(t):
        vals = a * t + b
        mv = np.max(vals)
        active = vals >= mv - 1e-12
        return np.min(a[active]), np.max(a[active])
    
    lo, hi = 0.0, 1.0
    # Check boundaries
    mn_a, mx_a = subgrad_condition(lo)
    if mn_a > 0:  # increasing at lo
        return lo
    mn_a, mx_a = subgrad_condition(hi)
    if mx_a < 0:  # decreasing at hi
        return hi
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        mn_a, mx_a = subgrad_condition(mid)
        if mx_a < 0:  # still decreasing, go left
            hi = mid
        elif mn_a > 0:  # increasing, go right
            lo = mid
        else:
            break  # 0 is in subgradient, optimal
    return (lo + hi) / 2.0

def coord_descent(h, n, n_sweeps, rng, verbose=False):
    """Coordinate descent with exact 1D line search."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    shifts = np.arange(-(n-1), n)  # 2n-1 shifts
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        for j in order:
            # Compute a_k for each shift
            js = j - shifts
            jp = j + shifts
            a = np.zeros(2*n-1)
            mask_neg = (js >= 0) & (js < n)
            mask_pos = (jp >= 0) & (jp < n)
            a[mask_neg] = 1.0 - h[js[mask_neg]]
            a[mask_pos] -= h[jp[mask_pos]]
            
            b = corr - a * h[j]
            
            t_opt = minimize_1d_max(a, b)
            
            if t_opt != h[j]:
                delta = t_opt - h[j]
                h[j] = t_opt
                # Update corr incrementally
                # Δc[k] = delta * a_k (since a_k was computed with current h, 
                # and the only change is h[j])
                # Wait, a_k = (1-h[j-s])*[valid] - h[j+s]*[valid]
                # Δc[k] = delta * (1-h[j-s])*[valid] + h[j-s]*(-delta)*[valid]... 
                # Actually Δc[k] = delta * a_k is NOT correct because a_k depends on h[j] through h[j-s] or h[j+s] when s makes those equal to j.
                # For s=0: a_0 = (1-h[j]) - h[j] = 1-2h[j]
                # Δc_0 = h_new*(1-h_new) - h_old*(1-h_old) = delta*(1 - h_new - h_old)
                # So Δc is NOT simply delta * a_k.
                # Let me just recompute corr.
                corr = np.correlate(h, 1.0 - h, mode="full")
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5

def subgradient_polish(h, n, n_iters, rng):

    h = h.copy()
    h_prev = h.copy()
    dx = 2.0 / n
    
    for t in range(1, n_iters + 1):
        y = h + (t-1)/(t+2) * (h - h_prev)  # Nesterov momentum
        y = np.clip(y, 0, 1)
        
        corr = np.correlate(y, 1.0 - y, mode="full")

        beta = 100.0
        shifted = corr - max_val
        weights = np.exp(beta * shifted)
        weights /= weights.sum()
        
        # Gradient
        shifts = np.arange(-(n-1), n)
        grad = np.zeros(n)
        for idx, s in enumerate(shifts):
            if weights[idx] < 1e-10:
                continue
            js = np.arange(n) - s
            jp = np.arange(n) + s
            a = np.zeros(n)
            mask_neg = (js >= 0) & (js < n)
            mask_pos = (jp >= 0) & (jp < n)
            a[mask_neg] = 1.0 - y[js[mask_neg]]
            a[mask_pos] -= y[jp[mask_pos]]

        h_new = project_feasible(h_new, n)
        
        h_prev = h
        h = h_new

        # Sum of top-k
        topk = np.partition(vals, -k)[-k:]
        return np.sum(topk)

        threshold = np.partition(vals, -k)[-k]

- If i+s = j, i.e., i = j-s: the term h[j-s]*(1-h[j]) changes by -delta*h[j-s]

So the change in c[k] is:
Δc[k] = delta * (1-h[j+s]) * [0 ≤ j+s < n] - delta * h[j-s] * [0 ≤ j-s < n]

Note: I'm using the OLD values of h[j+s] and h[j-s] (which are unchanged since only h[j] changed).

This is exactly delta * a_k where a_k is what I computed before! So the incremental update IS valid:
corr[k] += delta * a_k

        h_new, c5_new = coord_descent(h_start, best_n, n_sw, rng, use_topk=use_topk, topk_val=15)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time.time() - t0 < budget_s - 200:
        t_remaining = budget_s - (time.time() - t0)
        n_iters = max(200, min(2000, int(t_remaining / 0.005)))
        h_new, c5_new = subgradient_polish(best_h, best_n, n_iters, rng)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    for n_try in [285, 380]:
        if time.time() - t0 > budget_s - 200:
            break
        x_old = np.linspace(0.0, 2.0, best_n)
        x_new = np.linspace(0.0, 2.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        t_remaining = budget_s - (time.time() - t0)
        n_sw = max(100, min(400, int(t_remaining / 0.015)))
        h_new, c5_new = coord_descent(h_interp, n_try, n_sw, rng, use_topk=False)
        if c5_new < best_c5:
            best_h, best_c5, best_n = h_new, c5_new, n_try

    if time.time() - t0 < budget_s - 80:
        h_new, c5_new = slsqp_refine(best_h, best_n, 400, 60.0)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time.time() - t0 < budget_s - 40:
        t_remaining = budget_s - (time.time() - t0)
        n_sw = max(100, int(t_remaining / 0.005))
        h_new, c5_new = coord_descent(best_h, best_n, n_sw, rng, use_topk=False)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time.time() - t0 < budget_s - 20:
        h_new, c5_new = slsqp_refine(best_h, best_n, 300, 200.0)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
