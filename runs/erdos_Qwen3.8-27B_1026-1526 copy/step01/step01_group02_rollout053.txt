    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # soft max
    c_shifted = c - np.max(c)  # for numerical stability
    weights = np.exp(beta * c_shifted)
    weights /= np.sum(weights)
    return np.max(c) + np.log(np.sum(weights)) / beta  # approximation to max

            return x[-1]

from scipy.optimize import minimize
from scipy.interpolate import interp1d
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def _active_indices():
    return _global_active

_global_active = None

def _ineq_active(x):
    h_x = x[:-1]
    t_x = x[-1]
    n_x = len(h_x)
    dx_x = 2.0 / n_x
    c_x = np.correlate(h_x, 1.0 - h_x, mode='full') * dx_x
    return t_x - c_x[_global_active]

def _eq_sum(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0

def _obj_t(x):
    return x[-1]

def slsqp_active(h_init, n, maxiter=2000, K=10):
    global _global_active
    h = h_init.copy()
    current_c5 = compute_c5(h)
    for outer in range(6):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        t = float(np.max(c))
        top_k = np.argsort(c)[::-1][:K]
        _global_active = top_k
        bounds = [(0.0, 1.0)] * n + [(max(0.001, t * 0.5), 2.0)]
        x0 = np.concatenate([h, [t + 0.01]])
        cons = [
            {'type': 'ineq', 'fun': _ineq_active},
            {'type': 'eq', 'fun': _eq_sum}
        ]
        result = minimize(_obj_t, x0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': max(100, maxiter // 6), 'ftol': 1e-15})
        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5(h_new)
            if c5_new < current_c5 - 1e-12:
                h = h_new
                current_c5 = c5_new
            else:
                break
        else:
            break
    return h, current_c5

def slsqp_full(h_init, n, maxiter=2000):
    global _global_active
    h = h_init.copy()
    current_c5 = compute_c5(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    t = float(np.max(c))
    _global_active = np.arange(len(c))
    bounds = [(0.0, 1.0)] * n + [(max(0.001, t * 0.5), 2.0)]
    x0 = np.concatenate([h, [t + 0.02]])
    cons = [
        {'type': 'ineq', 'fun': _ineq_active},
        {'type': 'eq', 'fun': _eq_sum}
    ]
    result = minimize(_obj_t, x0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_new = project_feasible(result.x[:-1], n)
        c5_new = compute_c5(h_new)
        if c5_new < current_c5:
            return h_new, c5_new
    return h, current_c5

def soft_max_obj(x, beta, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    z = np.sum(exp_c)
    sm = c_max + np.log(z) / beta
    sd = np.sum(x) - n / 2.0
    return sm + lam * sd * sd

def soft_max_grad(x, beta, lam):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        i_lo = max(0, n - 1 - m)
        i_hi = min(n - 1, m)
        for i in range(i_lo, i_hi + 1):
            j1 = i + m - (n - 1)
            j2 = i
            grad[j1] += w * (1.0 - x[j2])
            grad[j2] += w * (-x[j1])
    grad *= dx
    sd = np.sum(x) - n / 2.0
    grad += 2.0 * lam * sd
    return grad

def lbfgs_soft_max(h_init, beta, lam, maxiter=500):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(soft_max_obj, h_init, jac=soft_max_grad,
                      method='L-BFGS-B', bounds=bounds,
                      args=(beta, lam),
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5(h_out)
    return h_init, compute_c5(h_init)

def perturb_and_optimize(h_base, n, rng, scale, beta, lam):
    h_pert = h_base + scale * rng.standard_normal(n)
    h_pert = project_feasible(h_pert, n)
    h_new, c5_new = lbfgs_soft_max(h_pert, beta, lam, maxiter=200)
    if c5_new < compute_c5(h_base):
        h_new, c5_new = slsqp_active(h_new, n, maxiter=1000, K=8)
    return h_new, c5_new

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 135
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def time_left(margin=30):
        return time() - t0 < budget_s - margin

    # Phase 1: Active-set SLSQP from initial
    if time_left(90):
        best_h, best_c5 = slsqp_active(best_h, n, maxiter=4000, K=12)

    # Phase 2: Soft-max L-BFGS-B for smoothing
    if time_left(60):
        h_sm, c5_sm = lbfgs_soft_max(best_h, beta=40, lam=200.0, maxiter=400)
        if c5_sm < best_c5:
            best_h, best_c5 = h_sm, c5_sm
            best_h, best_c5 = slsqp_active(best_h, n, maxiter=2000, K=10)

    # Phase 3: Full SLSQP for final convergence
    if time_left(90):
        best_h, best_c5 = slsqp_full(best_h, n, maxiter=3000)

    # Phase 4: Multiple perturbations with different strategies
    n_perturb = 15
    if time_left(300):
        for attempt in range(n_perturb):
            if not time_left(40):
                break
            frac = attempt / max(1, n_perturb)
            if attempt % 3 == 0:
                scale = 0.04 * (1.0 - frac) + 0.005
            elif attempt % 3 == 1:
                # Structured perturbation: shift-like
                shift = int(rng.integers(1, max(2, n // 10)))
                h_struct = best_h.copy()
                h_struct[shift:] = best_h[:-shift]
                h_struct[:shift] = best_h[-shift:]
                h_struct = project_feasible(h_struct, n)
                h_struct += 0.02 * rng.standard_normal(n)
                h_struct = project_feasible(h_struct, n)
                h_new, c5_new = slsqp_active(h_struct, n, maxiter=1000, K=8)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                continue
            else:
                # Frequency-domain perturbation
                h_freq = best_h.copy()
                H = np.fft.rfft(h_freq)
                freq_mask = np.ones(len(H))
                freq_mask[1:3] = 0
                H_pert = H + 0.02 * rng.standard_normal(len(H)) * freq_mask
                h_freq = np.fft.irfft(H_pert, n=n)
                h_freq = project_feasible(h_freq, n)
                h_new, c5_new = slsqp_active(h_freq, n, maxiter=1000, K=8)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                continue

            h_new, c5_new = perturb_and_optimize(best_h, n, rng, scale, 50, 200.0)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                best_h, best_c5 = slsqp_active(best_h, n, maxiter=1500, K=10)

    # Phase 5: Gradient-based descent (projected subgradient on max)
    if time_left(120):
        h_grad = best_h.copy()
        lr = 0.005
        for step in range(200):
            if not time_left(30):
                break
            c = np.correlate(h_grad, 1.0 - h_grad, mode='full') * (2.0 / n)
            k_max = np.argmax(c)
            # Subgradient of c[k_max] w.r.t. h
            n_local = n
            m = k_max
            subgrad = np.zeros(n_local)
            i_lo = max(0, n_local - 1 - m)
            i_hi = min(n_local - 1, m)
            for i in range(i_lo, i_hi + 1):
                j1 = i + m - (n_local - 1)
                j2 = i
                subgrad[j1] += (1.0 - h_grad[j2])
                subgrad[j2] += (-h_grad[j1])
            subgrad *= (2.0 / n_local)
            # Projected step
            h_trial = h_grad - lr * subgrad
            h_trial = project_feasible(h_trial, n_local)
            c5_trial = compute_c5(h_trial)
            if c5_trial < compute_c5(h_grad) - 1e-12:
                h_grad = h_trial
                lr = min(lr * 1.1, 0.02)
            else:
                lr *= 0.7
        c5_grad = compute_c5(h_grad)
        if c5_grad < best_c5:
            best_h = h_grad
            best_c5 = c5_grad
            best_h, best_c5 = slsqp_active(best_h, n, maxiter=1000, K=10)

    # Phase 6: Multi-resolution
    for n_try in [270, 180, 90]:
        if not time_left(120):
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = slsqp_active(h_interp, n_try, maxiter=2000, K=10)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            # Refine back to original resolution
            h_back = interpolate_h(best_h, len(best_h), n)
            h_back = project_feasible(h_back, n)
            h_back, c5_back = slsqp_active(h_back, n, maxiter=2000, K=10)
            if c5_back < best_c5:
                best_h, best_c5 = h_back, c5_back

    # Phase 7: Final polish
    if time_left(20):
        best_h, best_c5 = slsqp_active(best_h, len(best_h), maxiter=2000, K=15)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
