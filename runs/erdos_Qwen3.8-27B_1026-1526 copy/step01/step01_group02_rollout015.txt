import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total_room = room.sum()
            if total_room < 1e-15:
                break
            frac = diff / total_room
            h[mask] += frac * room
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            total = h[mask].sum()
            if total < 1e-15:
                break
            frac = (-diff) / total
            h[mask] -= frac * h[mask]
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

def c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def smooth_obj(x, beta, n, lam):
    h = x
    c = c5_values(h)
    cmax = np.max(c)
    sh = c - cmax
    ec = np.exp(beta * sh)
    Z = np.sum(ec)
    val = cmax + (1.0 / beta) * np.log(Z)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return val + penalty

def smooth_jac(x, beta, n, lam):
    h = x
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    sh = c - cmax
    ec = np.exp(beta * sh)
    Z = np.sum(ec)
    p = ec / Z
    
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        off = n - 1 - k
        lo1 = max(0, -off)
        hi1 = min(n - 1, n - 1 - off)
        if lo1 <= hi1:
            A[lo1:hi1 + 1] += pk * h[lo1 + off:hi1 + 1 + off]
        lo2 = max(0, off)
        hi2 = min(n - 1, n - 1 + off)
        if lo2 <= hi2:
            B[lo2:hi2 + 1] += pk * h[lo2 - off:hi2 + 1 - off]
    
    grad = dx * (1.0 - A - B)
    penalty_grad = 2.0 * lam * (np.sum(h) - n / 2.0)
    grad += penalty_grad
    
    return grad
