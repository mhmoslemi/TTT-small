import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask.astype(float))
            step = min(step, (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask.astype(float))
            step = min(step, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h_init, n, maxiter=2000):
    h_init = np.array(h_init, dtype=float)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * (2.0 / n)

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j

    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]

    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def coordinate_polish(h, n, max_sweeps=20):
    h = h.copy()
    best_c = compute_c5_from_h(h)
    delta = 0.02
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for sign in [1, -1]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h_trial[i] + sign * delta, 0, 1)
                h_trial = project_feasible(h_trial, n)
                c_trial = compute_c5_from_h(h_trial)
                if c_trial < best_c - 1e-12:
                    h = h_trial
                    best_c = c_trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)

    if time.time() - t0 < budget_s - 400:

        best_h, best_c5 = coordinate_polish(best_h, n, max_sweeps=10)

    x_grid = np.linspace(0, 2, n, endpoint=False)
    candidates = []

    for freq in range(1, 12):
        for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0, np.pi, 4.0 * np.pi / 3.0, 5.0 * np.pi / 3.0]:
            for amp in [0.25, 0.35, 0.42]:
                h_c = 0.5 + amp * np.sin(np.pi * freq * x_grid + phase)
                h_c = project_feasible(h_c, n)

    for _ in range(30):

        candidates.append(h_c)

        h_c = project_feasible(h_c, n)
        candidates.append(h_c)

            h_c = best_h + scale * rng.standard_normal(n)

        n_refine += 1

        if c5_ref < best_c5 - 1e-10:
            best_h = h_ref
            best_c5 = c5_ref

        x_old = np.linspace(0, 2, len(best_h), endpoint=False)

        h_new = np.clip(f_interp(x_new), 0, 1)

        if c5_ref < best_c5 - 1e-10:

            best_c5 = c5_ref

        frac = attempt / 60.0

        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        h_ref, c5_ref = slsqp_refine(h_pert, n, maxiter=600)
        h_ref, c5_ref = coordinate_polish(h_ref, n, max_sweeps=3)
        if c5_ref < best_c5 - 1e-10:

            best_c5 = c5_ref

        best_h, best_c5 = coordinate_polish(best_h, n, max_sweeps=15)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
