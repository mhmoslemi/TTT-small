from scipy.optimize import differential_evolution, minimize
from scipy.interpolate import interp1d

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def get_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            count = int(np.sum(mask))
            room = 1.0 - h[mask]
            step = min(diff / count, room.min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            count = int(np.sum(mask))
            room = h[mask]
            step = min(-diff / count, room.min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def _obj_slack(x):
    return x[-1]

def _ineq_full(x):
    h = x[:-1]
    t = x[-1]
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

    return np.sum(x[:-1]) - len(x[:-1]) / 2.0

def _eq_jac(x):
    n_vars = len(x) - 1
    j = np.zeros(len(x))
    j[:n_vars] = 1.0
    return j

        {'type': 'eq', 'fun': _eq_func, 'jac': _eq_jac}
    ]
    try:
        result = minimize(_obj_slack, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol})
        if result.x is not None:
            h_out = project_feasible(result.x[:-1], n)
            return h_out, compute_c5_from_h(h_out)
    except:
        pass
    return h_init, c5_init
