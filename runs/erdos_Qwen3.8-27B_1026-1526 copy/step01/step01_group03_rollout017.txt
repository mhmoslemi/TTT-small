```python
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # distribute the difference to elements that can absorb it
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            per = diff / np.sum(mask)
            result[mask] = np.clip(result[mask] + per, 0.0, 1.0)
            # fix any residual
            diff2 = target - np.sum(result)
            if abs(diff2) > 1e-10:
                idx = int(np.argmin(np.abs(result - 0.5)))
                result[idx] = np.clip(result[idx] + diff2, 0.0, 1.0)
    return result

def compute_smooth_max_grad(h, n, tau):
    """Compute gradient of smooth_max(corr*dx) w.r.t. h"""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    vals = corr * dx
    
    # softmax weights (numerically stable)
    v = vals / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    weights = exp_v / np.sum(exp_v)
    
    # For each lag l (indexed 0 to 2n-2 in "full" mode),
    # the lag in terms of offset is l - (n-1)
    # corr[j] = sum_i h[i] * g[i + j - (n-1)]
    # where g = 1 - h
    # So for lag offset = j - (n-1):
    #   corr[j] = sum_i h[i] * (1-h[i + offset])
    #   valid when 0 <= i < n and 0 <= i+offset < n
    
    grad = np.zeros(n)
    
    # For each lag j, the gradient contribution to h[i] is:
    # d/dh[i] [sum_k h[k]*(1-h[k+offset])] = (1-h[i+offset]) - h[i-offset]
    # (where the first term is i appearing as h[i], second as h[i+offset] = h[i] when k=i-offset)

    # For offset > 0: i ranges from 0 to n-1-offset
    #   h[i] appears as first factor: contributes (1-h[i+offset])
    #   h[i] appears as second factor when i-offset >= 0, i.e., i >= offset:
    #     h[i] = h[(i-offset)+offset], so i = k+offset, k = i-offset
    #     contributes -h[i-offset]
    
    # For offset < 0: let m = -offset > 0
    #   i ranges from m to n-1
    #   h[i] appears as first factor: contributes (1-h[i-m])
    #   h[i] appears as second factor when i+m < n, i.e., i < n-m:
    #     contributes -h[i+m]
    
    # Actually let me just compute this directly for efficiency
    # For lag offset o (can be negative), the overlap is:
    #   sum over i where both i and i+o are valid: h[i] * (1-h[i+o])
    
    # I'll vectorize over lags
    # For each lag j (0 to 2n-2), offset = j - (n-1)
    # The gradient at h[i] from lag j is:
    #   (1-h[i+offset]) if 0<=i<n and 0<=i+offset<n, else 0
    #   minus h[i-offset] if 0<=i-offset<n, else 0
    
    # This is O(n^2) which for n~60-80 is fine (3600-6400 operations)
    
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-15:

        offset = j - (n - 1)
        # h[i] as first factor: valid when 0 <= i < n and 0 <= i+offset < n

        i_end = min(n, n - offset)
        if i_start < i_end:
            idx = np.arange(i_start, i_end)
            grad[idx] += w * (1.0 - h[idx + offset])
        # h[i] as second factor: h[i] = h[(i-offset)+offset], valid when 0 <= i-offset < n
        # i.e., offset <= i < n+offset
        i_start2 = max(0, offset)
        i_end2 = min(n, n + offset)
        if i_start2 < i_end2:
            idx2 = np.arange(i_start2, i_end2)
            grad[idx2] += w * (-h[idx2 - offset])
    
    return grad * dx
```

For a fixed h with h[i] = t, the overlap at lag k is:
overlap_k(t) = t * (1-h_{i+k}) + (1-t) * h_{i-k} + C_k
where C_k is the contribution from all other terms (constant in t).
= t * (1 - h_{i+k} - h_{i-k}) + h_{i-k} + C_k

Wait, that's not quite right. Let me be more careful.

overlap_k = sum_j h[j] * (1-h[j+k]) * dx

If I change h[i] from its current value to t:
- Terms where j = i: t * (1-h[i+k]) * dx (was h[i] * (1-h[i+k]) * dx)
  Change: (t - h[i]) * (1-h[i+k]) * dx
- Terms where j+k = i, i.e., j = i-k: h[i-k] * (1-t) * dx (was h[i-k] * (1-h[i]) * dx)
  Change: h[i-k] * (1-t - (1-h[i])) * dx = h[i-k] * (h[i] - t) * dx

So the change in overlap_k when h[i] changes to t:
Δ_k = (t - h[i]) * (1-h[i+k]) * dx - h[i-k] * (t - h[i]) * dx
    = (t - h[i]) * dx * ((1-h[i+k]) - h[i-k])
    = (t - h[i]) * dx * (1 - h[i+k] - h[i-k])

This is linear in t! So the optimal t for minimizing the max overlap is... well, it's still a min-max problem but each overlap is linear in t. The optimal t is where the maximum overlap is minimized, which for linear functions means finding the t where the two most critical overlaps cross.

This is a bisection problem! For each coordinate i, I can find the optimal t by bisection on the max overlap as a function of t.

Let me optimize: instead of calling evaluate_c5 each time, I can compute the max overlap more efficiently by only updating the affected terms.

When h[i] changes, only the overlaps at lags k where i or i+k or i-k is involved change. Specifically, for each lag k, the overlap changes by:
Δ_k = (t - old_t) * dx * (1 - h[i+k] - h[i-k]) (with appropriate boundary handling)

So I can precompute the "sensitivity" for each coordinate and each lag, and then the overlap as a function of t is:
overlap_k(t) = base_k + (t - old_t) * sensitivity_k

where sensitivity_k = dx * (1 - h[i+k] - h[i-k]) (with boundary handling).

Then finding the optimal t is finding the t that minimizes the max of (base_k + (t - old_t) * sensitivity_k) over k.

This is a 1D convex optimization problem! The function f(t) = max_k (a_k + b_k * t) is piecewise linear and convex. Its minimum is at the point where two of the lines cross (or at a boundary).

I can find this by:
1. Compute a_k = overlap_k(old_t) - old_t * sensitivity_k for each k
2. f(t) = max_k (a_k + sensitivity_k * t)
3. Find t* = argmin_{t in [0,1]} f(t)

For a set of lines, the minimum of the max is at a vertex of the upper envelope, which is where two lines intersect. I can find this by:
- Computing all pairwise intersection points
- Keeping only those in [0,1]
- Evaluating f at each candidate point and at the boundaries
- Returning the best

But with 2n-1 lines, that's O(n^2) intersection points, which is O(n^2) total. For n=60, that's 3600 intersection points, which is fine.

Actually, there's a simpler approach: since f(t) is convex (max of linear functions), I can use ternary search or bisection on the derivative. The subgradient of f at t is sensitivity_k* where k* = argmax_k (a_k + sensitivity_k * t). So I can bisection search for the point where the subgradient changes sign.

Let me implement this more efficiently:

```python
def optimize_coordinate(h, n, i):
    """Find optimal value of h[i] given all other coordinates fixed"""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    base_overlaps = corr * dx  # shape (2n-1,)
    
    old_t = h[i]
    
    # Sensitivity of each overlap w.r.t. h[i]
    sens = np.zeros(2 * n - 1)
    for j in range(2 * n - 1):
        offset = j - (n - 1)
        # h[i] as first factor: contributes (1-h[i+offset]) if valid
        if 0 <= i + offset < n:
            sens[j] += (1.0 - h[i + offset])
        # h[i] as second factor: contributes -h[i-offset] if valid
        if 0 <= i - offset < n:
            sens[j] += (-h[i - offset])
    sens *= dx
    
    # overlap_j(t) = base_overlaps[j] + (t - old_t) * sens[j]
    # = (base_overlaps[j] - old_t * sens[j]) + t * sens[j]
    a = base_overlaps - old_t * sens  # intercept
    b = sens  # slope
    
    # Find t in [0,1] that minimizes max_j (a_j + b_j * t)
    # Use bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a + b * mid
        max_idx = np.argmax(vals)
        # Subgradient at mid is b[max_idx]
        # But we need to be more careful: the subgradient is the set of
        # convex combinations of b_j for all j achieving the max
        # For bisection, we use: if b[max_idx] > 0, decrease t (move left)
        # if b[max_idx] < 0, increase t (move right)
        if b[max_idx] > 1e-10:
            hi = mid
        elif b[max_idx] < -1e-10:
            lo = mid
        else:
            break
    
    return 0.5 * (lo + hi)
```

        if w < 1e-14:

        offset = j - (n - 1)

        i_end = min(n, n - offset)

        if i_start2 < i_end2:
