def compute_corr_grad(h, n):
    """Compute the Jacobian of corr w.r.t. h. Shape: (2n-1, n)"""
    J = np.zeros((2*n-1, n))
    g = 1.0 - h
    for k in range(n):  # positive shifts including 0
        m = n - 1 + k
        # d corr[m] / d h[j] for j = k, ..., n-1: (1-h[j-k])
        J[m, k:] = g[:n-k]  # g[j-k] for j=k,...,n-1 → g[0], g[1], ..., g[n-1-k]
        # d corr[m] / d h[j] for j = 0, ..., n-1-k: -h[j+k]
        J[m, :n-k] -= h[k:]  # h[j+k] for j=0,...,n-1-k → h[k], h[k+1], ..., h[n-1]
    for kp in range(1, n):  # negative shifts, k' = 1, ..., n-1
        m = n - 1 - kp
        # d corr[m] / d h[j] for j = 0, ..., n-1-kp: (1-h[j+kp])
        J[m, :n-kp] = g[kp:]  # g[j+kp] for j=0,...,n-kp-1 → g[kp], ..., g[n-1]
        # d corr[m] / d h[j] for j = kp, ..., n-1: -h[j-kp]
        J[m, kp:] -= h[:n-kp]  # h[j-kp] for j=kp,...,n-1 → h[0], ..., h[n-1-kp]
    return J

        # (-h[j-k]) for j in [k, n-1] → h[:n-k] has length n-k, corresponds to j=k..n-1 with h[j-k]
        if n - k > 0:
            grad[k:] += w * (-h[:n-k])
    
    # Negative shifts k' = 1, ..., n-1 (m = n-1-k')
    for kp in range(1, n):
        m = n - 1 - kp
        w = weights[m] * scale
        # (1-h[j-kp]) for j in [kp, n-1] → g[:n-kp] has length n-kp, corresponds to j=kp..n-1 with g[j-kp]
        if n - kp > 0:
            grad[kp:] += w * g[:n-kp]
        # (-h[j+kp]) for j in [0, n-1-kp] → h[kp:] has length n-kp, corresponds to j=0..n-1-kp with h[j+kp]
        if n - kp > 0:
            grad[:n-kp] += w * (-h[kp:])
    
    s = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * s
    return grad
