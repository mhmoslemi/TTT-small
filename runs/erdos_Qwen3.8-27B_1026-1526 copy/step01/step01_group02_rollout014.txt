from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask.astype(float))
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask.astype(float))
            h[mask] = np.maximum(0.0, h[mask] - step)
    return np.clip(h, 0.0, 1.0)

def project_feasible_lambda(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2} using bisection on lambda."""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = -1.0, 1.0
    for _ in range(100):

        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:

    projected = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

        if np.any(mask):
            projected[mask] += diff / np.sum(mask.astype(float))

    return projected

def smooth_max(c, dx, alpha):
    """Compute smooth max of c*dx using log-sum-exp."""
    vals = c * dx
    max_val = np.max(vals)
    if max_val > 500 / alpha:
        return max_val
    shifted = alpha * (vals - max_val)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return max_val + log_sum / alpha

def smooth_max_grad_weights(c, dx, alpha):
    """Compute softmax weights for the smooth max."""
    vals = c * dx
    max_val = np.max(vals)
    shifted = alpha * (vals - max_val)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights

def smooth_max_gradient(h, alpha):
    """Compute gradient of smooth max of correlation for given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    weights = smooth_max_grad_weights(c, dx, alpha)
    # Gradient: dx * sum_p weights[p] * dc(p)/dh
    # dc(p)/dh[j] = 1 - h[j+l] (if valid) - h[j-l] (if valid), where l = p-(n-1)
    grad = np.zeros(n)
    for p in range(2*n-1):
        l = p - (n-1)
        w = weights[p] * dx
        if w < 1e-15:
            continue
        # dc/dh[j] = 1 - h[j+l]*(valid) - h[j-l]*(valid)
        # This is 1 - (h shifted by l) - (h shifted by -l)
        shifted_pos = np.zeros(n)
        if l >= 0:
            shifted_pos[:n-l] = h[l:]

            shifted_neg[l:] = h[:n-l]
        else:
            shifted_neg[:n+l] = h[-l:]
        grad += w * (1.0 - shifted_pos - shifted_neg)
    return grad

            for l in range(-(n - 1), n):
                if l >= 0:
                    if i < n - l:
                        hp = h[i + l] if i + l < n else 0.0
                        hm = h[i - l] if i - l >= 0 else 0.0
                        change = -1.0 + hp + hm
                        delta[l + n - 1] += change
                else:
                    if i >= -l and i < n:
                        hp = h[i + l] if 0 <= i + l < n else 0.0
                        hm = h[i - l] if 0 <= i - l < n else 0.0
                        change = -1.0 + hp + hm
                        delta[l + n - 1] += change
            new_c = c + delta
            new_c5 = float(np.max(new_c) * dx)
            delta_c5 = new_c5 - c5
            if delta_c5 < -1e-12:
                h[i] = 0.0
                h[(h == 0.5).argmax() if np.any(h == 0.5) else np.argmin(h)] = 0.0  # placeholder
                # Actually need to find a 0 to flip to 1
                zeros = np.where(h == 0.0)[0]
                if len(zeros) == 0:
                    h[i] = 1.0
                    continue
                j = zeros[0]
                h[j] = 1.0
                c = new_c.copy()
                # Actually this is getting complicated. Let me just recompute.
                c = np.correlate(h, 1.0 - h, mode="full")
                c5 = float(np.max(c) * dx)
                improved = True
                no_improve = 0
            elif T > 0 and np.random.random() < np.exp(-delta_c5 / T):
                zeros = np.where(h == 0.0)[0]
                if len(zeros) == 0:
                    continue
                j = zeros[np.random.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c = np.correlate(h, 1.0 - h, mode="full")
                c5 = float(np.max(c) * dx)
                improved = True
        if not improved:
            no_improve += 1
        if no_improve > 50:
            break
    return h.copy(), c5


def binary_hill_climb(h_bin, n, max_rounds=500):
    h = h_bin.copy().astype(float)
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    c5 = float(np.max(c) * dx)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            # Try swapping i (1) with each zero j
            best_j = -1
            best_new_c5 = c5
            for j in zeros:
                # Compute new c5 for swap
                h[i] = 0.0
                h[j] = 1.0
                new_c = np.correlate(h, 1.0 - h, mode="full")
                new_c5 = float(np.max(new_c) * dx)
                h[i] = 1.0
                h[j] = 0.0
                if new_c5 < best_new_c5 - 1e-12:
                    best_new_c5 = new_c5
                    best_j = j
            if best_j >= 0:
                h[i] = 0.0
                h[best_j] = 1.0
                c = np.correlate(h, 1.0 - h, mode="full")
                c5 = best_new_c5
                improved = True
                break
        if not improved:
            break
    return h, c5


def make_slsqp_constraints():
    return [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - len(x) / 2.0},
    ]


def slsqp_polish(h_init, maxiter=500):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    c_total = np.correlate(h_init, 1.0 - h_init, mode="full")
    dx = 2.0 / n
    c5_vals = c_total * dx

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        cv = c * (2.0 / len(x))
        return np.max(cv)

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        cv = c * (2.0 / len(x))
        active = cv >= np.max(cv) - 1e-10
        active_count = np.sum(active)
        if active_count == 0:
            return np.zeros(len(x))
        grad_sum = np.zeros(len(x))
        for p in range(2 * len(x) - 1):
            if not active[p]:
                continue
            l = p - (len(x) - 1)
            gp = np.zeros(len(x))
            if l >= 0:
                gp[:len(x) - l] = 1.0
                gp[:len(x) - l] -= x[l:]
                gp[l:] -= x[:len(x) - l]
            else:
                gp[-l:] = 1.0
                gp[-l:] -= x[:len(x) + l]
                gp[:len(x) + l] -= x[-l:]
            grad_sum += gp
        return grad_sum / active_count * (2.0 / len(x))

    result = minimize(obj, h_init, method='SLSQP', jac=obj_grad,
                      bounds=[(0, 1)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Phase 0: Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)
    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    best_n = n

    def elapsed():
        return time.time() - t0

    # Phase A: Projected gradient with smooth max (annealing alpha)
    if elapsed() < budget_s - 200:
        for phase in range(3):
            if elapsed() > budget_s - 200:
                break
            alpha_start = 20 + phase * 40
            alpha_end = 100 + phase * 200
            n_steps = 800
            h_phase, c5_phase = projected_gradient_optimize(
                best_h, alpha_start, alpha_end, n_steps, 1.0)
            if c5_phase < best_c5:
                best_h = h_phase
                best_c5 = c5_phase
            best_h = project_feasible(best_h, n)
            best_c5 = compute_c5_from_h(best_h)

    # Phase B: Binary local search
    if elapsed() < budget_s - 150:
        h_bin = (best_h > 0.5).astype(float)
        ones_count = int(np.sum(h_bin))
        target_ones = n // 2
        if ones_count > target_ones:
            ones_idx = np.where(h_bin == 1.0)[0]
            rng.shuffle(ones_idx)
            for i in range(ones_count - target_ones):
                h_bin[ones_idx[i]] = 0.0
        elif ones_count < target_ones:
            zeros_idx = np.where(h_bin == 0.0)[0]
            rng.shuffle(zeros_idx)
            for i in range(target_ones - ones_count):
                h_bin[zeros_idx[i]] = 1.0
        c5_bin = compute_c5_from_h(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin

        # Hill climb on binary
        if elapsed() < budget_s - 100:
            for attempt in range(5):
                if elapsed() > budget_s - 100:
                    break
                h_try, c5_try = binary_hill_climb(h_bin, n, max_rounds=200)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    h_bin = h_try.copy()
                else:
                    # Perturb and try again
                    h_pert = h_bin.copy()
                    for _ in range(5):
                        i = rng.integers(0, n)
                        j = rng.integers(0, n)
                        if h_pert[i] == 1.0 and h_pert[j] == 0.0:
                            h_pert[i] = 0.0
                            h_pert[j] = 1.0
                            break
                    h_bin = h_pert

    # Phase C: Multi-start with different structures
    if elapsed() < budget_s - 100:
        structures = []
        # Symmetric structure
        half = n // 2
        h_sym = np.zeros(n)
        h_sym[:half] = 1.0
        r = np.arange(n)
        # Alternating blocks
        block_size = max(1, n // 12)
        h_alt = np.zeros(n)
        for bi in range(0, n, block_size * 2):
            h_alt[bi:bi + block_size] = 1.0
        h_alt = project_feasible(h_alt, n)
        structures.append(h_alt)
        # Random binary
        h_rand_bin = np.zeros(n)
        for _ in range(n // 2):
            h_rand_bin[rng.integers(0, n)] = 1.0
        structures.append(h_rand_bin)
        # Smooth wave
        t_wave = np.linspace(0, 2 * np.pi, n, endpoint=False)
        h_wave = np.clip(0.5 + 0.3 * np.cos(t_wave), 0, 1)
        h_wave = project_feasible(h_wave, n)
        structures.append(h_wave)

        for h_struct in structures:
            if elapsed() > budget_s - 100:
                break
            h_struct = project_feasible(h_struct, n)
            c5_s = compute_c5_from_h(h_struct)
            if c5_s > best_c5 + 0.1:
                continue
            h_try, c5_try = projected_gradient_optimize(
                h_struct, 30, 300, 400, 1.0)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                best_n = n

    # Phase D: Try higher resolution
    if elapsed() < budget_s - 120:
        for n_new in [270]:
            if elapsed() > budget_s - 120:
                break
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate", bounds_error=False)
            h_interp = np.clip(f_interp(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_new)
            c5_interp = compute_c5_from_h(h_interp)
            if c5_interp < best_c5:
                h_interp, c5_interp = projected_gradient_optimize(
                    h_interp, 30, 300, 300, 1.0)
            if c5_interp < best_c5:
                best_h = h_interp
                best_c5 = c5_interp
                best_n = n_new

    # Phase E: SLSQP polish
    if elapsed() < budget_s - 60:
        try:
            h_polished, c5_polished = slsqp_polish(best_h, maxiter=300)
            if c5_polished < best_c5:
                best_h = h_polished
                best_c5 = c5_polished
        except Exception:
            pass

    # Phase F: Final binary refinement
    if elapsed() < budget_s - 30:
        h_bin_final = (best_h > 0.5).astype(float)
        ones_count = int(np.sum(h_bin_final))
        target_ones = len(best_h) // 2
        if ones_count > target_ones:
            idx = np.where(h_bin_final == 1.0)[0]
            np.random.default_rng(seed + 1).shuffle(idx)
            for i in range(ones_count - target_ones):
                h_bin_final[idx[i]] = 0.0
        elif ones_count < target_ones:
            idx = np.where(h_bin_final == 0.0)[0]
            np.random.default_rng(seed + 1).shuffle(idx)
            for i in range(target_ones - ones_count):
                h_bin_final[idx[i]] = 1.0
        h_bin_final, c5_final = binary_hill_climb(h_bin_final, len(best_h), max_rounds=300)
        if c5_final < best_c5:
            best_h = h_bin_final
            best_c5 = c5_final

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
