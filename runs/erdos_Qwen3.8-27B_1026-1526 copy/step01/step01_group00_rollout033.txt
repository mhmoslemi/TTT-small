    for freq in [1, 2, 3]:
        hp = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(hp, ts))
    h_blk = np.zeros(n)
    seg = n // 6
    for phase in range(4):
        s0 = phase * (n // 4)
        h_blk[s0:s0 + seg // 2] = 1.0
    starts.append(project_to_sum(h_blk, ts))
    h_bin = (rng.rand(n) > 0.5).astype(float)
    h_bin = project_to_sum(h_bin, ts)
    smoothed = np.convolve(h_bin, np.ones(3) / 3.0, mode='same')
    starts.append(project_to_sum(smoothed, ts))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            init = np.array(initial_h_values, dtype=float)
            n_init = len(init)
            ts_init = n_init / 2.0
            init = project_to_sum(init, ts_init)
            best_h = init.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_init
        except Exception:
            pass

    if best_h is None:
        best_n = 40
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Multi-resolution (60% budget)
    phase1_end = t_start + 0.58 * budget_s

    res_list = [24, 48, 96, 144, 192, 240, 320]
    if best_n not in res_list:
        res_list.append(best_n)
    res_list = sorted(set([r for r in res_list if r % 2 == 0 and r >= 16]))

    for n in res_list:
        elapsed = time.time() - t_start
        if elapsed > phase1_end - 30:
            break
        if remaining() < 40:
            break
        if n % 2 != 0:
            continue
        ts = n // 2
        if ts * 2 != n:
            continue

        n_phase1_left = len([r for r in res_list if r >= n and r % 2 == 0])
        time_budget = max(20, min(100, (phase1_end - time.time()) / max(1, n_phase1_left)))

        starts = generate_starts(n, ts, rng, best_h, best_n)
        max_starts = min(len(starts), max(3, int(time_budget / 18)))

        for si in range(max_starts):
            if time.time() - t_start > phase1_end - 15:
                break
            if remaining() < 25:
                break
            h_start = project_to_sum(starts[si], ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n

            h_cur = h_start.copy()
            alphas = [5.0, 20.0, 80.0, 300.0, 1000.0]
            time_per_alpha = max(5, (phase1_end - time.time()) / (len(alphas) * 1.5) / max_starts)

            for ai, alpha in enumerate(alphas):
                if time.time() - t_start > phase1_end - 10:
                    break
                if remaining() < 15:
                    break
                t_alloc = min(time_per_alpha, remaining() * 0.4)
                if t_alloc < 3:
                    break
                lr0 = 0.02 / max(1.0, alpha * 0.005)
                h_cur, c5_cur = pgd_optimize(h_cur, ts, alpha, 500, lr0, t_alloc, True)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n

    # Phase 2: Fine-tuning (30% budget)
    phase2_end = t_start + 0.92 * budget_s

    if best_h is not None:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha in [500.0, 2000.0, 8000.0]:
            if time.time() - t_start > phase2_end - 20:
                break
            if remaining() < 20:
                break
            lr0 = 0.005 / max(1.0, alpha * 0.001)
            t_alloc = min(50, remaining() * 0.3)
            h_pol, c5_pol = pgd_optimize(best_h, ts_f, alpha, 5000, lr0, t_alloc, False)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        # Try higher resolution
        if time.time() - t_start < phase2_end - 30 and remaining() > 40:
            n_high = max(360, 2 * best_n)
            n_high = (n_high // 2) * 2
            ts_high = n_high / 2.0
            h_high = upsample(best_h, n_high)
            h_high = project_to_sum(h_high, ts_high)

            for alpha in [100.0, 500.0, 3000.0]:
                if time.time() - t_start > phase2_end - 10:
                    break
                if remaining() < 20:
                    break
                lr0 = 0.003 / max(1.0, alpha * 0.001)
                t_alloc = min(35, remaining() * 0.25)
                h_high, c5_high = pgd_optimize(h_high, ts_high, alpha, 5000, lr0, t_alloc, False)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()
                    best_n = n_high
                    ts_f = ts_high
                    n_f = n_high

        # Also try a slightly different resolution
        if remaining() > 30 and time.time() - t_start < phase2_end:
            n_alt = best_n + 20
            if n_alt % 2 != 0:
                n_alt += 1
            ts_alt = n_alt / 2.0
            h_alt = upsample(best_h, n_alt)
            h_alt = project_to_sum(h_alt, ts_alt)
            for alpha in [500.0, 3000.0]:
                if remaining() < 15:
                    break
                lr0 = 0.003
                t_alloc = min(25, remaining() * 0.3)
                h_alt, c5_alt = pgd_optimize(h_alt, ts_alt, alpha, 3000, lr0, t_alloc, False)
                if c5_alt < best_c5:
                    best_c5 = c5_alt
                    best_h = h_alt.copy()
                    best_n = n_alt

    # Phase 3: Coordinate descent polish
    if best_h is not None and remaining() > 15:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_cd = best_h.copy()
        cd_time = min(40, remaining() * 0.5)
        cd_start = time.time()

        for rnd in range(200):
            if time.time() - cd_start > cd_time:
                break
            if remaining() < 10:
                break
            improved = False
            order = rng.permutation(n_f)
            for i in order:
                for delta in [0.002, -0.002, 0.001, -0.001, 0.005, -0.005, 0.0005, -0.0005]:
                    h_try = h_cd.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                    h_try = project_to_sum(h_try, ts_f)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-12:
                        h_cd = h_try
                        best_c5 = c5_try
                        best_h = h_cd.copy()
                        improved = True
                        break
            if not improved:
                break

    # Final validation
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
