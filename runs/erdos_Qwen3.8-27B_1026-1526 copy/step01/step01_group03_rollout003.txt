    sorted_corr = np.sort(corr)
    top = sorted_corr[-k:]
    base = smooth_max(corr, tau) * dx
    penalty = np.mean(np.maximum(0, top - base / dx) ** 2)
    return base + 0.5 * penalty * dx


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def project_feasible(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result


def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    f = interp1d(x_old, h_old, kind='cubic', fill_value='extrapolate')
    return np.clip(f(x_new), 0.0, 1.0)


def coord_descent(h, n, max_rounds, tol):
    best_c = evaluate_c5(h)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, -0.02, -0.01, -0.005, -0.002, -0.001]
    for rd in range(max_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                h_trial = h.copy()
                h_trial[i] = h[i] + d
                h_trial = project_feasible(h_trial, n)
                c = evaluate_c5(h_trial)
                if c < best_c - tol:
                    h = h_trial.copy()
                    best_c = c
                    improved = True
        if not improved:
            break
    return h, best_c


def slsqp_optimize(h0, n, tau, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    res = minimize(obj_smooth, h0, method='SLSQP', args=(n, tau),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)


def generate_perturbed(h, n, rng, scales):
    starts = []
    for s in scales:
        pert = h + rng.randn(n) * s
        starts.append(project_feasible(pert, n))
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if parent_h is not None:
        best_h = project_feasible(parent_h, len(parent_h))
        best_c5 = evaluate_c5(best_h)
    else:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    n_base = len(best_h)

    phase1_end = t0 + budget_s * 0.30
    phase2_end = t0 + budget_s * 0.55
    phase3_end = t0 + budget_s * 0.80
    phase4_end = t0 + budget_s * 0.95

    # Phase 1: SLSQP from parent at decreasing tau
    n = n_base
    for tau in [0.06, 0.03, 0.015, 0.008, 0.004]:
        if time.time() > phase1_end:
            break
        h_cur = slsqp_optimize(best_h, n, tau, 120, 1e-14)
        c = evaluate_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            best_h = h_cur.copy()

    # Phase 2: Try nearby n values and more SLSQP
    n_candidates = []
    for dn in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16, 24, 32]:
        nn = n_base + dn
        if 35 <= nn <= 160 and nn not in n_candidates:
            n_candidates.append(nn)

    for n_try in [n_base] + n_candidates:
        if time.time() > phase2_end:
            break

        h_cur = resample_h(best_h, n_try)
        h_cur = project_feasible(h_cur, n_try)

        for tau in [0.04, 0.02, 0.01, 0.005]:
            if time.time() > phase2_end:
                break
            h_cur = slsqp_optimize(h_cur, n_try, tau, 100, 1e-14)
            c = evaluate_c5(h_cur)
            if c < best_c5:
                best_c5 = c
                best_h = h_cur.copy()
                n_base = n_try

    # Phase 3: Coordinate descent polishing
    if time.time() < phase3_end:
        n = len(best_h)
        remaining_cd = phase3_end - time.time()
        rounds = max(10, int(remaining_cd * 8))
        h_cd, c_cd = coord_descent(best_h, n, rounds, 1e-13)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 4: Ultra-fine SLSQP + multi-restart
    if time.time() < phase4_end:
        n = len(best_h)
        for tau in [0.003, 0.002, 0.001, 0.0005]:
            if time.time() > phase4_end:
                break
            h_pol = slsqp_optimize(best_h, n, tau, 250, 1e-15)
            c = evaluate_c5(h_pol)
            if c < best_c5:
                best_c5 = c
                best_h = h_pol.copy()

        # Multi-restart
        scales = [0.004, 0.002, 0.006, 0.003, 0.001, 0.005]
        for sc in scales:
            if time.time() > phase4_end:
                break
            h_pert = best_h + rng.randn(n) * sc
            h_pert = project_feasible(h_pert, n)
            h_pol = slsqp_optimize(h_pert, n, 0.002, 150, 1e-15)
            c = evaluate_c5(h_pol)
            if c < best_c5:
                best_c5 = c
                best_h = h_pol.copy()

    # Final quick coord descent
    if time.time() < t0 + budget_s * 0.98:
        n = len(best_h)
        h_cd, c_cd = coord_descent(best_h, n, 20, 1e-13)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Affine transform check: try h' = 1 - h (should give same C5 but verify)
    if time.time() < t0 + budget_s * 0.99:
        h_flip = project_feasible(1.0 - best_h, len(best_h))
        c_flip = evaluate_c5(h_flip)
        if c_flip < best_c5:
            best_c5 = c_flip
            best_h = h_flip.copy()

    return best_h, best_c5, len(best_h)
