            if time() - t0 > budget_s - 10:
                break
            s = i % 6
            if s == 0:
                h_p = best_h + 0.02 * rng.standard_normal(n)
            elif s == 1:
                h_p = best_h + 0.06 * rng.standard_normal(n)
            elif s == 2:
                h_p = best_h.copy()
                idx = rng.choice(n, size=max(1, n // 5), replace=False)
                h_p[idx] += 0.1 * rng.standard_normal(len(idx))
            elif s == 3:
                h_p = 1.0 - best_h
            elif s == 4:
                xs = np.linspace(0, 2 * np.pi, n)
                freq = rng.uniform(2, 8)
                h_p = best_h + 0.05 * np.sin(freq * xs + rng.uniform(0, 6.28))
            else:
                h_p = rng.random(n)
            h_p = project(h_p)
            h_new, c_new = slsqp_minimize(h_p, maxiter=1000)
            if c_new < best_c - 1e-10:
                best_h, best_c = h_new, c_new

    # Phase 6: Higher resolution attempts
    for n_factor in [2, 3]:
        if time() - t0 > budget_s - 80:
            break
        n2 = n * n_factor
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        f_int = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate')
        h2 = project(np.clip(f_int(x_new), 0.0, 1.0))
        h2, c2 = slsqp_minimize(h2, maxiter=3000)
        if c2 < best_c - 1e-8:
            best_h, best_c = h2, c2
            n = n2

    # Final small polish
    if time() - t0 < budget_s - 10:
        best_h, best_c = slsqp_minimize(best_h, maxiter=2000)

    best_h = project(best_h)
    best_c = compute_c5(best_h)
    return best_h, best_c, len(best_h)
