import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(v, target_sum):
    """Project v onto {h: sum(h) = target_sum, 0 <= h <= 1}"""
    n = len(v)
    # Binary search for lambda such that sum(clip(v + lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(v + (lo + hi) / 2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_alpha(corr, alpha):
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    return m + np.log(np.sum(exp_vals)) / alpha, w

def compute_grad_smooth_max(h, w, alpha):
    """
    Compute gradient of smooth_max(corr, alpha) w.r.t. h.
    corr = np.correlate(h, 1-h, mode='full')
    smooth_max = m + log(sum(exp(alpha*(corr-m))))/alpha
    d(smooth_max)/d(h[j]) = (1/alpha) * sum_k alpha * exp(alpha*(corr[k]-m))/sum * d(corr[k])/d(h[j])
    = sum_k w[k] * d(corr[k])/d(h[j])
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        m = k - (n - 1)

        grad[j1] += w[k] * (1.0 - h[j1 + m])
        j2_start = max(0, m)
        j2_end = min(n - 1, n - 1 + m)
        if j2_start <= j2_end:
            j2 = np.arange(j2_start, j2_end + 1)
            grad[j2] -= w[k] * h[j2 - m]
    return grad

def projected_gradient_descent(h0, n, target_sum, alpha, max_iters, lr, t_start, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_val = float('inf')
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit - 30:
            break
        
        corr = compute_corr(h)
        sm_val, w = smooth_max_alpha(corr, alpha)
        c5_actual = compute_c5(h)
        if c5_actual < best_val:
            best_val = c5_actual
            best_h = h.copy()
        
        grad = compute_grad_smooth_max(h, w, alpha)
        
        # Adaptive step size
        if it == 0:
            step = lr
        else:
            # Simple backtracking
            pass
        
        # Line search (backtracking)
        for _ in range(20):
            h_new = h - step * grad
            h_new = project_to_sum(h_new, target_sum)
            corr_new = compute_corr(h_new)
            sm_new, _ = smooth_max_alpha(corr_new, alpha)
            if sm_new < sm_val - 1e-12:

            step *= 0.5
        else:
            # Couldn't improve, try smaller step
            h_new = h - step * grad
            h_new = project_to_sum(h_new, target_sum)
            h = h_new

def project_to_sum(v, target_sum):
    n = len(v)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(v + lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)

def compute_grad_corr(h, w):
    """Gradient of smooth_max of correlation w.r.t. h, given softmax weights w."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        m = k - (n - 1)
        s_start = max(0, -m)
        s_end = min(n - 1, n - 1 - m)
        if s_start > s_end:
            continue
        j1 = np.arange(s_start, s_end + 1)
        grad[j1] += wk * (1.0 - h[j1 + m])
        j2_start = max(0, m)
        j2_end = min(n - 1, n - 1 + m)
        if j2_start <= j2_end:
            j2 = np.arange(j2_start, j2_end + 1)
            grad[j2] -= wk * h[j2 - m]
    return grad
