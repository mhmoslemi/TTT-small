    n = len(h)
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode="full")
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h
    
    for _ in range(max_passes):
        M = np.max(C)
        improved = False
        
        # Compute all coefficients A
        A = np.zeros((n, 2*n - 1))
        for j in range(n):
            k = np.arange(2*n - 1)
            s = k - (n - 1)
            idx1 = j - s
            idx2 = j + s
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            A[j, valid1] += (1.0 - h[idx1[valid1]])
            A[j, valid2] -= h[idx2[valid2]]
        
        best_gain = 0
        best_pair = None
        
        for i in ones:
            for j in zeros:
                new_C_max = np.max(C - A[i] + A[j])
                gain = M - new_C_max
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            C = np.correlate(h, 1.0 - h, mode="full")
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = True
        
        if not improved:
            break
    
    return h
