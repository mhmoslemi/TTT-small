def compute_smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    m_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - m_max))
    w = exp_corr / np.sum(exp_corr)
    
    n_full = len(corr)  # 2n-1
    grad = np.zeros(n)
    
    for m in range(n):
        g = 0.0
        for k in range(-n+1, n):
            j = k + n - 1  # index into corr array
            if j < 0 or j >= n_full:
                continue

            mk = m - k
            if 0 <= mk < n:

            mk2 = m + k
            if 0 <= mk2 < n:

        grad[m] = g
    
    return grad, m_max + np.log(np.sum(exp_corr)) / alpha

def project_to_simplex(h, s):
    """Project h onto {x: sum(x)=s, 0<=x[i]<=1} using bisection on lambda"""
    h = h.astype(float).copy()
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = s
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

                    best_h = h_cur.copy()
                    best_n = n

            for pert_scale in [0.03, 0.01, 0.005]:
                elapsed = time.time() - t_start
                if elapsed > budget_s - 25:
                    break
                h_pert = project_to_simplex(h_cur + rng.normal(0, pert_scale, n), target_sum)
                h_pert = optimize_slsqp(h_pert, n, target_sum, 200.0, 200)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()
                    best_n = n
                    h_cur = h_pert

    elapsed = time.time() - t_start
    if elapsed < budget_s - 20 and best_h is not None:
        ts_f = best_n / 2.0
        h_pol = optimize_slsqp(best_h, best_n, ts_f, 500.0, 500, 1e-14)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    elapsed = time.time() - t_start
    if elapsed < budget_s - 5 and best_h is not None:
        remaining = budget_s - elapsed - 2
        if remaining > 5:
            best_h, best_c5 = coordinate_refine(best_h, best_n / 2.0, remaining)
            best_n = len(best_h)

    best_h = project_to_simplex(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
