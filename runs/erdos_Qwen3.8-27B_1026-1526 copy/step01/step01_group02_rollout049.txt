        h_interp = project_feasible(h_interp, n_high)
        h_new, c5_new = slsqp_refine(h_interp, maxiter=3000, ftol=1e-14)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new

    # Stage 6: Final basin hopping at highest resolution
    nv = len(best_h)
    n_final_pert = 20
    for i in range(n_final_pert):
        if budget_s - (time.time() - t0) < 25:
            break
        scale = 0.03 * (1.0 - i / max(1, n_final_pert)) + 0.003
        mode = i % 3
        pert = make_pert(rng, nv, scale, mode)
        h_pert = best_h + pert
        h_pert = project_feasible(h_pert, nv)
        h_new, c5_new = slsqp_refine(h_pert, maxiter=600, ftol=1e-13)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new

    # Stage 7: Final polish
    if budget_s - (time.time() - t0) > 20:
        best_h, best_c5 = slsqp_refine(best_h, len(best_h), maxiter=5000, ftol=1e-15)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
