    h = h_init.copy()
    c5 = compute_c5(h)
    dx = 2.0 / n
    
    for outer in range(10):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        max_c = np.max(c)
        # Active constraints: those within active_tol of max
        active_mask = c > max_c - active_tol
        active_idx = np.where(active_mask)[0]
        
        # Define constraints for active set only
        def ineq_active(x):
            h_x = x[:-1]
            t = x[-1]
            c_x = np.correlate(h_x, 1.0 - h_x, mode="full") * (2.0/len(h_x))
            return t - c_x[active_idx]
        
        def eq_func(x):
            return np.sum(x[:-1]) - len(x[:-1])/2.0
        
        x0 = np.concatenate([h, [max_c + 0.01]])
        bounds = [(0.0, 1.0)]*n + [(0.0, 2.0)]
        constraints = [
            {'type': 'ineq', 'fun': ineq_active},
            {'type': 'eq', 'fun': eq_func}
        ]
        
        result = minimize(lambda x: x[-1], x0, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': maxiter//10, 'ftol': 1e-14})
        
        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                active_tol = max(0.001, active_tol * 0.5)  # Tighten
            else:
                break
    
    return h, c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            count = np.sum(mask).astype(float)
            max_add = float((1.0 - h[mask]).min())
            step = min(diff / count, max_add)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            count = np.sum(mask).astype(float)
            max_sub = float(h[mask].min())
            step = min(-diff / count, max_sub)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h_init, n, maxiter=1000, time_limit=60):
    h = h_init.copy()
    c5 = compute_c5(h)
    dx = 2.0 / n
    t0 = time.time()
    
    # Start with all constraints, then narrow to active set
    active_tol = 0.05
    h_work = h.copy()
    c5_work = c5
    
    for outer in range(20):
        if time.time() - t0 > time_limit:
            break
        
        c_vals = np.correlate(h_work, 1.0 - h_work, mode="full") * dx
        max_c = float(np.max(c_vals))
        active_mask = c_vals > max_c - active_tol
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) == 0:
            active_idx = [np.argmax(c_vals)]
        
        def ineq_fun(x):
            h_x = x[:-1]
            t = x[-1]
            c_x = np.correlate(h_x, 1.0 - h_x, mode="full") * (2.0/len(h_x))
            return t - c_x[active_idx]
        
        def eq_fun(x):
            return float(np.sum(x[:-1])) - len(x[:-1])/2.0
        
        x0 = np.concatenate([h_work, [max_c + 0.01]])
        bounds = [(0.0, 1.0)]*n + [(max(0.1, max_c*0.8), 1.0)]
        constraints = [
            {'type': 'ineq', 'fun': ineq_fun},
            {'type': 'eq', 'fun': eq_fun}
        ]
        
        result = minimize(lambda x: x[-1], x0, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': max(100, maxiter//20), 'ftol': 1e-14})
        
        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_work - 1e-10:
                h_work = h_new
                c5_work = c5_new
                active_tol = max(0.001, active_tol * 0.5)
            else:
                active_tol = min(0.5, active_tol * 1.5)
                if active_tol > 0.1:
                    break
    
    return h_work, c5_work

def de_objective(x, n):
    h = x.copy()
    # Penalty for constraint violation
    diff = float(np.sum(h)) - n/2.0
    penalty = 10.0 * abs(diff)
    h_clipped = np.clip(h, 0, 1)
    c5_val = compute_c5(h_clipped)
    return c5_val + penalty

    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            count = float(np.sum(mask))
            max_add = float((1.0 - h[mask]).min())
            step = min(diff / count, max_add)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            count = float(np.sum(mask))
            max_sub = float(h[mask].min())
