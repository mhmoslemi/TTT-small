def compute_c5(h):

def compute_grad_value(h, beta):
    """Compute smooth max value and its gradient w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    value = (m + np.log(np.sum(exp_vals)) / beta) * (2.0 / n)
    
    # Gradient computation (vectorized)
    lags = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    
    idx1 = j_idx - lags[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
    
    idx2 = j_idx + lags[None, :]

    term2 = (weights[None, :] * h2).sum(axis=1)
    
    grad = (term1 - term2) * (2.0 / n)
    return value, grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    ev = np.exp(shifted)
    w = ev / ev.sum()
    val = (m + np.log(ev.sum()) / beta) * (2.0 / n)
    lags = np.arange(-(n - 1), n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    idx1 = j[:, None] - lags[None, :]
    v1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(v1, h[np.clip(idx1, 0, n - 1)], 0.0)
    idx2 = j[:, None] + lags[None, :]
    v2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
    grad = ((w[None, :] * (1.0 - h1)).sum(axis=1) - (w[None, :] * h2).sum(axis=1)) * (2.0 / n)
    return val, grad
