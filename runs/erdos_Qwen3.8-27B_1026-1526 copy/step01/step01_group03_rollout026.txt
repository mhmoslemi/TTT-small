This is a reasonable approach but let me think about what could make it even better...

**Key improvement ideas:**

1. **Adaptive τ**: Instead of a fixed schedule, adapt τ based on the "spread" of the correlation values. If the top few lags are close, we can use a smaller τ. If they're spread out, we need a larger τ to keep the max smooth.

2. **Lag-focused perturbation**: Instead of random perturbation, perturb in a direction that specifically reduces the worst lag. The gradient of f_{k*} (the worst lag) gives such a direction.

3. **Oscillation detection**: If the optimizer oscillates (goes back and forth between two solutions), increase the random component.

4. **Better initialization**: Use the structure of the problem to create better starting points. For example, a function that alternates between high and low values in a structured way.

5. **Multi-objective view**: Consider not just the max, but also the second-worst, third-worst, etc. A solution that has a lower max might have a higher second-worst. The smooth max with small τ handles this, but we can be more explicit.

Let me implement the **lag-focused gradient** idea:

For the worst lag k*, the overlap is:
f_{k*} = ∑_{i} h[i] * (1-h[i+k*]) * dx (for the appropriate range)

The gradient of f_{k*} with respect to h[m] is:
- From the direct term (h[m] appears as the first factor): (1-h[m+k*]) * dx
- From the indirect term (h[m] appears as the second factor): -h[m-k*] * dx (when applicable)

So the gradient points in a direction that, if we move h against it, reduces f_{k*}.

But we also need to maintain the constraint sum(h) = n/2. So we should project the gradient to the constraint hyperplane.

Let me implement this:

```python
def lag_gradient(h, n, k):
    """Compute gradient of the overlap at lag k with respect to h."""
    dx = 2.0 / n
    g = np.zeros(n)
    
    # Forward: for lag k (positive), f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
    # d/ dh[m] for m in [0, n-k]: (1-h[m+k]) * dx
    # d/ dh[m] for m in [k, n-1]: from h[m-k]*(1-h[m]), contributes -h[m-k] * dx
    
    for m in range(n - k):
        g[m] += (1 - h[m+k]) * dx
    for m in range(k, n):
        g[m] += -h[m-k] * dx
    
    return g
```

Wait, I need to be more careful about the indexing. Let me use the full correlation definition.

Actually, for the full correlation np.correlate(h, g, mode="full") where g = 1-h:
- The output has length 2n-1
- Output[j] = sum_i h[i] * g[i + j - (n-1)]
  where the sum is over all i such that 0 ≤ i < n and 0 ≤ i+j-(n-1) < n

For j = n-1 (lag 0): Output[n-1] = sum_i h[i] * g[i] = sum_i h[i]*(1-h[i])
For j = n-1+k (positive lag k): Output[n-1+k] = sum_i h[i] * g[i+k] = sum_i h[i]*(1-h[i+k])
  where i ranges from 0 to n-1-k
For j = n-1-k (negative lag k): Output[n-1-k] = sum_i h[i] * g[i-k] = sum_i h[i]*(1-h[i-k])
  where i ranges from k to n-1

So for positive lag k:
f_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) * dx

Gradient wrt h[m]:
- If m ∈ [0, n-1-k]: term h[m]*(1-h[m+k]) contributes (1-h[m+k])*dx
- If m ∈ [k, n-1]: term h[m-k]*(1-h[m]) contributes -h[m-k]*dx

So:
grad[m] = dx * (1-h[m+k]) [if m ≤ n-1-k] - dx * h[m-k] [if m ≥ k]

def overlap_gradient(h, n, k):

2. Compute the gradient of f_{k*}

This is like a **cyclical coordinate descent** on the lags.

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    shift = 0.5 * (lo + hi)
    result = np.clip(h + shift, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - result
        else:
            room = result
        sorted_idx = np.argsort(-room)
        n_fix = max(1, int(abs(diff) * n * 0.1))
        per = diff / min(n_fix, n)
        for i in range(min(n_fix, n)):
            idx = sorted_idx[i]
            result[idx] = np.clip(result[idx] + per, 0, 1)
        residual = target - np.sum(result)
        if abs(residual) > 1e-14:
            idx = int(np.argmin(np.abs(result - 0.5)))
            result[idx] = np.clip(result[idx] + residual, 0, 1)
    return result

def smooth_max_value(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_max_weights(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    return w

def optimize_lbfgsb(h, n, tau, max_iter, time_limit):

    dx = 2.0 / n
    h = project_feasible(h.copy(), n)

        corr = np.correlate(h, g, mode="full")
        sm = smooth_max_value(corr, tau)
        # Add penalty for constraint violation
        penalty = 100.0 * (np.sum(h) - n/2.0)**2
        return sm * dx + penalty
    
    def obj_grad(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full")
        w = smooth_max_weights(corr, tau)
        # Gradient of smooth max wrt corr
        # Then chain rule to h
        # This is complex; use numerical gradient as fallback
        return None
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj, h, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8}
        )
        h = project_feasible(res.x, n)
    except:
        pass
    return h

def optimize_slsqp(h, n, tau, max_iter, time_limit):
    """Optimize using SLSQP with smooth max."""
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full")
        return smooth_max_value(corr, tau) * dx
    
    def constraint_fun(h):
        return np.sum(h) - n / 2.0
    
    try:
        res = minimize(
            obj, h, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_fun}],
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h = project_feasible(res.x, n)
    except:
        pass
    return h

def lag_descent_step(h, n, k, lr):
    """Take a step to reduce overlap at lag k."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k < n:
        grad[:n-k] += (1 - h[k:]) * dx
        grad[k:] += -h[:n-k] * dx
    # Project gradient to constraint hyperplane (remove mean component)
    grad = grad - np.mean(grad)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

def smooth_max_gradient(h, n, tau):
    """Compute gradient of smooth_max(correlation) wrt h."""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")  # length 2n-1
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for m in range(n):
        # Lag 0
        grad[m] += w[n-1] * (1 - 2*h[m])
        
        # Positive lags k = 1, ..., n-1
        for k in range(1, n):
            # From lag +k (index n-1+k):
            # dc/dh[m] = (1-h[m-k]) if k<=m<=n-1, minus h[m+k] if 0<=m<=n-1-k

            if 0 <= m <= n-1-k:
                grad[m] += -w[n-1+k] * h[m+k]

            if k <= m <= n-1:

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_max_weights(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    return w

def evaluate_objective(h, n, tau):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    dx = 2.0 / n
    return smooth_max_value(corr, tau) * dx

def smooth_max_gradient(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    w = smooth_max_weights(corr, tau)
    grad = np.zeros(n)
    for m in range(n):
        grad[m] += w[n-1] * (1.0 - 2.0 * h[m])
        for k in range(1, n):
            if k <= m:
                grad[m] += w[n-1+k] * (1.0 - h[m-k])
            if 0 <= m <= n-1-k:
                grad[m] -= w[n-1+k] * h[m+k]
            if 0 <= m <= n-1-k:
                grad[m] += w[n-1-k] * (1.0 - h[m+k])
            if k <= m:
                grad[m] -= w[n-1-k] * h[m-k]
    return grad * dx

def project_feasible(h, n):

    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)

            hi = mid
        else:
            lo = mid
    shift = 0.5 * (lo + hi)
    result = np.clip(h + shift, 0.0, 1.0)

    if abs(diff) > 1e-10:

            room = 1.0 - result
        else:
            room = result
        sorted_idx = np.argsort(-room)
        n_fix = max(1, int(abs(diff) * 2))

        per = diff / n_fix
        for i in range(n_fix):

            idx = int(np.argmin(np.abs(result - 0.5)))

def slsqp_optimize(h, n, tau, max_iter, time_limit):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n

        corr = np.correlate(h, g, mode="full")
        return smooth_max_value(corr, tau) * dx

        return np.sum(h) - n / 2.0

            obj_func, h, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_func}],
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )

            h_pert = project_feasible(best_h + rng.randn(n) * noise_scale, n)
            h_pert, c5_pert = projected_gradient_optimize(h_pert, n, 0.01, 400, phase2_limit, lr_init=0.03)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
    if best_h is not None and time.time() < phase3_limit:
        n = len(best_h)
        h_cur = best_h.copy()
        for tau in [0.005, 0.003, 0.002, 0.001, 0.0005]:
            if time.time() > phase3_limit:
                break
            h_cur = slsqp_optimize(h_cur, n, tau, 200, phase3_limit)
            c5_val = evaluate_c5(h_cur)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_cur.copy()
        for attempt in range(8):
            if time.time() > phase3_limit:
                break
            noise = rng.randn(n) * 0.003
            h_pert = project_feasible(best_h + noise, n)
            h_pert = slsqp_optimize(h_pert, n, 0.001, 150, phase3_limit)
            c5_val = evaluate_c5(h_pert)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_pert.copy()
        for dn in [-2, 2]:
            if time.time() > phase3_limit:
                break
            n_new = len(best_h) + dn
            if n_new < 40 or n_new > 80:
                continue
            x_p = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            f = interp1d(x_p, best_h, kind='cubic', fill_value='extrapolate')
            h_interp = np.clip(f(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_new)
            h_interp, c5_interp = projected_gradient_optimize(h_interp, n_new, 0.01, 500, phase3_limit, lr_init=0.03)
            h_interp = slsqp_optimize(h_interp, n_new, 0.002, 100, phase3_limit)
            c5_val = evaluate_c5(h_interp)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_interp.copy()
    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 59
            best_h = np.full(n, 0.5)
            best_h = project_feasible(best_h, n)
            best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
