def compute_logsumexp_gradient(h, beta, dx):
    n = len(h)
    # Compute c values (all shifts)
    c = np.correlate(h, 1.0 - h, mode="full") * dx  # length 2n-1
    
    # Log-sum-exp for numerical stability
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        for shift_idx in range(2*n-1):
            shift = shift_idx - (n-1)
            # First term: (1-h[j+shift]) if 0<=j+shift<n
            if 0 <= j+shift < n:
                grad[j] += w[shift_idx] * (1.0 - h[j+shift])
            # Second term: -h[j-shift] if 0<=j-shift<n
            if 0 <= j-shift < n:
                grad[j] -= w[shift_idx] * h[j-shift]
    grad *= dx
    return grad, (max_c + np.log(np.sum(exp_c))) / beta  # gradient and objective value

def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    for s_idx, s in enumerate(shifts):
        if s >= 0:
            grad[:n-s] += w[s_idx] * (1.0 - h[s:])
            grad[s:] -= w[s_idx] * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w[s_idx] * (1.0 - h[:n-ss])
            grad[:n-ss] -= w[s_idx] * h[ss:]
    grad *= dx
    obj_val = (max_c + np.log(np.sum(exp_c))) / beta
    return grad, obj_val, max_c

def adam_step(params, grad, t):
    lr = params['lr'] / (1.0 + t * 1e-4)
    params['m'] = params['beta1'] * params['m'] + (1 - params['beta1']) * grad
    params['v'] = params['beta2'] * params['v'] + (1 - params['beta2']) * grad**2
    m_hat = params['m'] / (1 - params['beta1']**(t+1))
    v_hat = params['v'] / (1 - params['beta2']**(t+1))
    step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    return step

    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(h0)

        'beta1': 0.9,
        'beta2': 0.999,
        'm': np.zeros(n),
        'v': np.zeros(n)
    }
    
    # β schedule: start at 10, end at 100
    total_steps = 500
    t = 0
    while t < total_steps and time.time() - t_start < budget_s:
        beta = 10.0 + (100.0 - 10.0) * t / total_steps
        grad, lse_val, max_c = compute_lse_gradient(h, beta, dx)
        step = adam_step(params, grad, t)
        h_new = h - step
        h_new = project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[np.argmax(h < 1.0)] += diff
        else:
            mask = h > 0.0
            if np.any(mask):
                h[np.argmin(h > 0.0)] += diff
    return np.clip(h, 0.0, 1.0)

def compute_jacobian_cols(h):
    """Compute J[k, j] = dc_k/dh[j] for all k, j. Returns (2n-1, n) array."""
    n = len(h)
    J = np.zeros((2*n-1, n))
    for s_idx in range(2*n-1):
        s = s_idx - (n-1)
        if s >= 0:
            # J[s_idx, j] = (1-h[j+s]) for j in [0, n-1-s] minus h[j-s] for j in [s, n-1]
            J[s_idx, :n-s] += (1.0 - h[s:])
            J[s_idx, s:] -= h[:n-s]
        else:
            ss = -s
            # J[s_idx, j] = (1-h[j-ss]) for j in [ss, n-1] minus h[j+ss] for j in [0, n-1-ss]
            J[s_idx, ss:] += (1.0 - h[:n-ss])
            J[s_idx, :n-ss] -= h[ss:]
    return J

def coordinate_descent(h, n, max_sweeps=100, grid_size=50):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    J = compute_jacobian_cols(h)
    
    best_c5 = np.max(c) * dx
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            slopes = J[:, j]  # how c_k changes with h[j]
            # f(x) = max_k (c_k + (x - h[j]) * slopes_k)
            # Evaluate on grid
            x0 = h[j]
            xs = np.linspace(0.0, 1.0, grid_size)
            # f_vals[g] = max_k(c + (xs[g]-x0)*slopes)
            # Vectorized: for each x, compute c + (x-x0)*slopes, take max
            f_vals = np.max(c[None, :] + (xs[:, None] - x0) * slopes[None, :], axis=1)
            best_idx = np.argmin(f_vals)
            x_new = xs[best_idx]
            
            if abs(x_new - x0) > 1e-10:
                h[j] = x_new
                # Update c and J for changed coordinate
                # c_k changes by (x_new - x0) * slopes_k
                c += (x_new - x0) * slopes
                # J changes: for all k, J[k, j] stays the same
                # But J[k, l] for l≠j might change because h[j] changed
                # J[k, l] = (1-h[l+s])*1_{valid} - h[l-s]*1_{valid}
                # Only changes if l+s or l-s equals j
                # This is complex to update incrementally. Let's recompute J each sweep.
                improved = True
        
        if not improved:
            break
        
        # Recompute full c and J at end of sweep
        c = np.correlate(h, 1.0 - h, mode="full")
        J = compute_jacobian_cols(h)
        c5_cur = np.max(c) * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        J = compute_jacobian_cols(h)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[np.where(mask)[0][0]] += diff
        else:
            mask = h > 0.0
            if np.any(mask):
                h[np.where(mask)[0][0]] += diff
    return np.clip(h, 0.0, 1.0)


def compute_jacobian(h):
    n = len(h)
    J = np.zeros((2 * n - 1, n))
    for s_idx in range(2 * n - 1):
        s = s_idx - (n - 1)
        if s >= 0:
            J[s_idx, :n - s] += (1.0 - h[s:])
            J[s_idx, s:] -= h[:n - s]
        else:
            ss = -s
            J[s_idx, ss:] += (1.0 - h[:n - ss])
            J[s_idx, :n - ss] -= h[ss:]
    return J


def coordinate_descent(h, n, max_sweeps=200, grid_size=60):
    dx = 2.0 / n
    h_work = h.copy()
    c = np.correlate(h_work, 1.0 - h_work, mode="full")
    J = compute_jacobian(h_work)
    best_c5 = np.max(c) * dx
    best_h = h_work.copy()

    for sweep in range(max_sweeps):
        J = compute_jacobian(h_work)
        improved = False
        for j in range(n):
            slopes = J[:, j]
            x0 = h_work[j]
            xs = np.linspace(0.0, 1.0, grid_size)
            f_vals = np.max(c[None, :] + (xs[:, None] - x0) * slopes[None, :], axis=1)
            best_idx = np.argmin(f_vals)
            x_new = xs[best_idx]
            if abs(x_new - x0) > 1e-10:
                c += (x_new - x0) * slopes
                h_work[j] = x_new
                improved = True
        if not improved:
            break
        h_work = project_feasible(h_work, n)
        c = np.correlate(h_work, 1.0 - h_work, mode="full")
        c5_cur = np.max(c) * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_work.copy()
    return best_h, best_c5


def compute_lse_gradient(h, beta, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(c)
    exp_c = np.exp(np.clip(beta * (c - max_c), -50, 50))
    w = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for s_idx in range(2 * n - 1):
        s = s_idx - (n - 1)
        if s >= 0:
            grad[:n - s] += w[s_idx] * (1.0 - h[s:])
            grad[s:] -= w[s_idx] * h[:n - s]
        else:
            ss = -s
            grad[ss:] += w[s_idx] * (1.0 - h[:n - ss])
            grad[:n - ss] -= w[s_idx] * h[ss:]
    grad *= dx
    obj_val = (max_c + np.log(np.sum(exp_c))) / beta
    return grad, obj_val, max_c


def adam_optimize(h0, n, max_steps, t_start, deadline):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(h0)
    lr = 0.05
    b1 = 0.9
    b2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)

    for t in range(max_steps):
        if time.time() > deadline:
            break
        beta = 5.0 + 45.0 * t / max(max_steps - 1, 1)
        grad, lse_val, max_c = compute_lse_gradient(h, beta, dx)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        step = np.clip(step, -0.05, 0.05)
        h_new = h - step
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    hh = x[:-1]
    tt = x[-1]
    nn = len(hh)
    dd = 2.0 / nn
    cc = np.correlate(hh, 1.0 - hh, mode="full")
    return tt - cc * dd


def slsqp_eq(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0


def slsqp_eq_jac(x):
    jj = np.zeros(len(x))
    jj[:-1] = 1.0
    return jj


def slsqp_polish(h, n, maxiter=2000):
    c5_init = compute_c5_from_h(h)
    x0 = np.concatenate([h, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.7), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h, c5_init
