    val = 0.0
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = p * (c - c_max)

    sum_diff = np.sum(x) - n / 2.0

    return val

        args=(p, penalty, n),
        bounds=bounds,

    )
    h_out = project_feasible(result.x)
