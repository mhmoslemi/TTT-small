    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    v -= v.max()
    exp_v = np.exp(v)

    grad = np.zeros(n)
    for m in range(2 * n - 1):

        pm = p[m]
        if pm < 1e-14:

        term = np.ones(n)
        if k >= 0:
            if k > 0:
                term[:n-k] -= h[k:]
                term[k:] -= h[:n-k]
        else:
            q = -k
            term[q:] -= h[:n-q]
            term[:n-q] -= h[q:]
        grad += pm * term
    return dx * grad

    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    v -= v.max()
    exp_v = np.exp(v)
    p = exp_v / exp_v.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        pm = p[m]
        if pm < 1e-14:
            continue
        term = np.ones(n)
        if k >= 0:
            if k > 0:
                term[:n - k] -= h[k:]
                term[k:] -= h[:n - k]
        else:
            q = -k
            term[q:] -= h[:n - q]
            term[:n - q] -= h[q:]
        grad += pm * term
    return dx * grad


def obj_and_grad(h, n, tau, mu):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    sm = smooth_max_lse(corr, tau)
    s = np.sum(h) - n / 2.0
    f = sm * dx + mu * s * s
    grad = grad_smooth_max(h, n, tau)
    if mu > 0.0:
        grad = grad + 2.0 * mu * s
    return f, grad


def fun_obj(x, n, tau, mu):
    f, _ = obj_and_grad(x, n, tau, mu)
    return f


def fun_jac(x, n, tau, mu):
    _, g = obj_and_grad(x, n, tau, mu)
    return g


def constraint_func(h, n):
    return np.sum(h) - n / 2.0


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        cand = np.clip(h + mid, 0.0, 1.0)
        if np.sum(cand) > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(proj - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - proj[idx])
                proj[idx] += add
                diff -= add
            else:
                sub = min(-diff, proj[idx])
                proj[idx] -= sub
                diff += sub
    return proj


def do_trust_constr(h0, n, tau, bounds, lc, maxiter):
    try:
        res = minimize(
            fun=fun_obj,
            x0=h0,
            jac=fun_jac,
            args=(n, tau, 0.0),
            bounds=bounds,
            constraints=[lc],
            method='trust-constr',
            options={'maxiter': maxiter, 'gtol': 1e-8, 'xtol': 1e-12}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0


def do_slsqp(h0, n, tau, bounds, cons_slsqp, maxiter):
    try:
        res = minimize(
            fun_obj, h0,
            method='SLSQP',
            args=(n, tau, 0.0),
            bounds=bounds,
            constraints=cons_slsqp,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0


def compute_structured_noise(h, n, scale):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    k_star = int(np.argmax(corr) - (n - 1))
    noise = np.zeros(n)
    for i in range(n):
        grad_val = 1.0
        if 0 <= i + k_star < n:
            grad_val -= h[i + k_star]
        if 0 <= i - k_star < n:
            grad_val -= h[i - k_star]
        noise[i] = -grad_val
    norm = np.max(np.abs(noise))
    if norm > 1e-15:
        noise = scale * noise / norm
    return noise


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.full(59, 0.5)

    n = len(best_h)
    best_c5 = evaluate_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    lc = LinearConstraint(np.ones((1, n)), lb=[n / 2.0], ub=[n / 2.0])
    cons_slsqp = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]

    # Phase 1: Coarse-to-fine trust-constr optimization
    tau_schedule = [0.08, 0.05, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002, 0.001]
    current_h = best_h.copy()

    for tau in tau_schedule:
        if time.time() - start_time > budget_s * 0.35:
            break
        maxiter = 200 if tau > 0.01 else 300
        current_h = do_trust_constr(current_h, n, tau, bounds, lc, maxiter)
        c5 = evaluate_c5(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()

    # Phase 2: Multi-restart perturbation with trust-constr
    elapsed = time.time() - start_time
    remaining = budget_s * 0.72 - elapsed
    if remaining > 20:
        n_restarts = min(40, int(remaining / 3.0))
        for r in range(n_restarts):
            if time.time() - start_time > budget_s * 0.74:
                break

            if r < 6:
                noise = rng.randn(n) * 0.015
            elif r < 12:
                noise = rng.randn(n) * 0.04
            elif r < 18:
                noise = compute_structured_noise(best_h, n, 0.025)
            elif r < 24:
                x_grid = np.linspace(0, 2 * np.pi, n)
                freq1 = rng.uniform(0.8, 4.0)
                freq2 = rng.uniform(3.0, 8.0)
                noise = 0.02 * (np.sin(freq1 * x_grid + rng.uniform(0, 6.28)) +
                                0.5 * np.sin(freq2 * x_grid + rng.uniform(0, 6.28)))
            elif r < 30:
                mask = rng.rand(n) < 0.25
                noise = np.zeros(n)
                noise[mask] = rng.choice([-0.04, 0.04], size=int(np.sum(mask)))
            else:
                noise = rng.randn(n) * (0.01 + 0.01 * rng.rand())

            perturbed = project_feasible(best_h + noise, n)

            tau_phase2 = 0.005 if r % 2 == 0 else 0.002
            cand = do_trust_constr(perturbed, n, tau_phase2, bounds, lc, 200)
            c5 = evaluate_c5(cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand

    # Phase 3: SLSQP polish for exact constraint handling
    if time.time() - start_time < budget_s * 0.88:
        tau_polish = [0.001, 0.0005, 0.0003, 0.0002]
        for tau in tau_polish:
            if time.time() - start_time > budget_s * 0.95:
                break
            cand = do_slsqp(best_h, n, tau, bounds, cons_slsqp, 200)
            c5 = evaluate_c5(cand)
            if c5 < best_c5:
                best_c5 = c5
                best_h = cand

    # Phase 4: Try a few different n values if budget allows
    if time.time() - start_time < budget_s * 0.80:
        for n_try in [57, 61, 63]:
            if time.time() - start_time > budget_s * 0.92:
                break
            x_orig = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_orig, best_h, kind='cubic', fill_value='extrapolate')
            h_try = np.clip(f_interp(x_new), 0.0, 1.0)
            h_try = project_feasible(h_try, n_try)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
            else:
                bounds_t = [(0.0, 1.0)] * n_try
                lc_t = LinearConstraint(np.ones((1, n_try)), lb=[n_try / 2.0], ub=[n_try / 2.0])
                cand = do_trust_constr(h_try, n_try, 0.005, bounds_t, lc_t, 150)
                c5_try = evaluate_c5(cand)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = cand

    # Phase 5: Final ultra-fine polish
    if time.time() - start_time < budget_s * 0.97:
        n_final = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_final
        lc_f = LinearConstraint(np.ones((1, n_final)), lb=[n_final / 2.0], ub=[n_final / 2.0])
        cand = do_trust_constr(best_h, n_final, 0.0008, bounds_f, lc_f, 400)
        c5_final = evaluate_c5(cand)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = cand

    return best_h, best_c5, len(best_h)
