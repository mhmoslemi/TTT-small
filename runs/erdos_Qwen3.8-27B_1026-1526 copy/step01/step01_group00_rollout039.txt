Let me think of a better approach. Actually, for a single coordinate change, I can compute the effect on all overlaps more efficiently.

If I change h[i] from v_old to v_new, how does each overlap change?

overlap_k = dx * sum_j h[j]*(1-h[j+k_mod])

For a given shift k (= idx - (n-1)):
- If i is in the "first" position (j = i), the term h[i]*(1-h[i+k]) changes by (v_new - v_old)*(1-h[i+k])
- If i is in the "second" position (j+k = i, i.e., j = i-k), the term h[i-k]*(1-h[i]) changes by -h[i-k]*(v_new - v_old)

So the change in overlap_k is:
Δoverlap_k = dx * (v_new - v_old) * [(1-h[i+k]) - h[i-k]] (with appropriate bounds)

This is LINEAR in (v_new - v_old)! So the change is the same for all shifts (scaled by a shift-dependent factor).

This means: if I increase h[i] by δ, the change in overlap_k is:
δ * dx * [(1-h[i+k]) - h[i-k]] (with bounds)

So I can precompute the "sensitivity" vector s_i where s_i[k] = dx * [(1-h[i+k]) - h[i-k]], and then the change in overlap_k from changing h[i] by δ is δ * s_i[k].

def optimal_pair_move(h, i, j, n, overlaps):
    """Find the best transfer δ from j to i (h[i]+=δ, h[j]-=δ) that minimizes max overlap"""
    sens_i = sensitivity_vector(h, i, n)
    sens_j = sensitivity_vector(h, j, n)
    diff_sens = sens_i - sens_j  # change in overlap_k per unit
    
    # We want to minimize max_k (overlaps_k + δ * diff_sens_k)
    # subject to 0 <= h[i] + δ <= 1 and 0 <= h[j] - δ <= 1
    # i.e., -h[i] <= δ <= 1-h[i] and h[j]-1 <= δ <= h[j]
    
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_lo >= delta_hi:

        # sens_j[k_max] should be small/negative (increasing h[j] helps)

        # sens_i[k_max] = dx * [(1-h[i+shift]) - h[i-shift]] (with bounds)

        if shift_max >= 0:
            for i in range(n - shift_max):
                sens_at_kmax[i] += (1.0 - h[i + shift_max])
            if shift_max > 0:
                for i in range(shift_max, n):
                    sens_at_kmax[i] -= h[i - shift_max]
        else:
            s = -shift_max
            for i in range(s, n):
                sens_at_kmax[i] += (1.0 - h[i - s])
            for i in range(n - s):
                sens_at_kmax[i] -= h[i + s]
        sens_at_kmax *= (2.0 / n)
        
        # Best sources: high sens (decreasing helps) AND high h value (can decrease)
        # Best targets: low sens (increasing helps) AND low h value (can increase)
        # Combine: source_score = sens * h (benefit of decreasing)
        #           target_score = -sens * (1-h) (benefit of increasing)
        
        # Actually, the change in overlap at k_max from transferring δ from i to j is:
        # δ * (sens_j[k_max] - sens_i[k_max])
        # We want this to be negative, so sens_j < sens_i
        # The amount we can transfer is min(h[i], 1-h[j])
        
        # Greedy: find i with max sens_at_kmax (and h[i] > 0)
        #         find j with min sens_at_kmax (and h[j] < 1)
        #         transfer from i to j
        
        # But we should also consider the effect on OTHER shifts!
        # So let's use the full sensitivity vectors for the top candidates
        
        # Top-5 sources (highest sens at k_max)
        source_candidates = np.argsort(-sens_at_kmax)[:10]
        source_candidates = [s for s in source_candidates if h[s] > 0.01]

        target_candidates = np.argsort(sens_at_kmax)[:10]
        target_candidates = [t for t in target_candidates if h[t] < 0.99]

        best_move = None
        
        for si in source_candidates:
            sens_si = sensitivity_vector(h, si, n)
            for tj in target_candidates:
                sens_tj = sensitivity_vector(h, tj, n)
                diff_sens = sens_tj - sens_si  # change per unit transfer from si to tj
                # We want max_k (overlaps_k + δ * diff_sens_k) to decrease
                # Find optimal δ via ternary search
                delta_lo = max(-h[si], h[tj] - 1.0)  # Wait, transfer from si to tj means h[si]-=δ, h[tj]+=δ
                # So δ > 0 means h[si] decreases
                # Constraint: 0 <= h[si]-δ <= 1 and 0 <= h[tj]+δ <= 1
                # => δ <= h[si] and δ <= 1-h[tj]
                # => 0 <= δ <= min(h[si], 1-h[tj])
                delta_lo = 0.0
                delta_hi = min(h[si], 1.0 - h[tj])
                
                if delta_hi < 1e-8:
                    continue
                
                # Ternary search for optimal δ
                lo, hi = 0.0, delta_hi
                for _ in range(40):
                    m1 = lo + (hi - lo) / 3
                    m2 = hi - (hi - lo) / 3
                    f1 = float(np.max(overlaps + m1 * diff_sens))
                    f2 = float(np.max(overlaps + m2 * diff_sens))
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                
                delta_star = (lo + hi) / 2
                new_max = float(np.max(overlaps + delta_star * diff_sens))
                improvement = c5 - new_max
                
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_move = (si, tj, delta_star)
        
        if best_move is not None and best_improvement > 1e-13:
            si, tj, delta = best_move
            h[si] -= delta
            h[tj] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, n / 2.0)
            overlaps = compute_all_overlaps(h)
            c5 = float(np.max(overlaps))
        else:
            break
    
    return h, c5

    center = n - 1
    sens = np.zeros(2 * n - 1)
    
    # Positive shifts: k = 0, 1, ..., n-1
    k = np.arange(n)  # 0 to n-1

    # (1-h[i-s]) where i-s >= 0

    sens[idx_neg[mask3]] += (1.0 - h[i - s[mask3]])
    # -h[i+s] where i+s < n
    mask4 = (i + s) < n
    sens[idx_neg[mask4]] -= h[i + s[mask4]]

The function f(δ) = max_k (overlaps_k + δ * diff_sens_k) is convex and piecewise linear. Its minimum occurs at a point where two lines cross (or at a boundary). I can find the minimum by:
1. Evaluating f at δ=0 and δ=δ_max
2. If f(δ_max) < f(0), the minimum is to the right of 0
3. Use bisection on the subgradient: the subgradient at δ is {min_k diff_sens_k : k in argmax} to {max_k diff_sens_k : k in argmax}. If 0 is in the subgradient, we're at the minimum.

Actually, for a max of linear functions, the minimum can be found by solving a small LP. But for our purposes, the ternary search with 20 iterations gives precision of (δ_max/3)^20 ≈ 0 (since δ_max < 1). So 20 iterations is plenty.

```python
import numpy as np
import time

def compute_c5(h):
    """Compute the C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    """Compute all overlap values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    """Project h to satisfy sum constraint and box constraints."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def sensitivity_vector(h, i, n):
    """Compute the vector of partial derivatives of all overlaps w.r.t. h[i]."""
    dx = 2.0 / n
    center = n - 1
    sens = np.zeros(2 * n - 1)
    k = np.arange(n)
    idx_pos = center + k
    mask1 = (i + k) < n
    if np.any(mask1):
        sens[idx_pos[mask1]] += (1.0 - h[i + k[mask1]])
    mask2 = (k > 0) & (i - k >= 0)
    if np.any(mask2):
        sens[idx_pos[mask2]] -= h[i - k[mask2]]
    s = np.arange(1, n)
    idx_neg = center - s
    mask3 = (i - s) >= 0
    if np.any(mask3):
        sens[idx_neg[mask3]] += (1.0 - h[i - s[mask3]])
    mask4 = (i + s) < n
    if np.any(mask4):
        sens[idx_neg[mask4]] -= h[i + s[mask4]]
    return sens * dx

def targeted_greedy(h0, n, target_sum, max_iters=40000, t_start=None, time_limit=None, 
                    n_src=8, n_tgt=8, n_ternary=20):
    """Greedy optimization using targeted pair moves."""
    h = h0.copy()
    overlaps = compute_all_overlaps(h)
    c5 = float(np.max(overlaps))
    center = n - 1
    
    for it in range(max_iters):
        if t_start is not None and time.time() - t_start > time_limit - 5:
            break
        
        # Find the worst shift
        k_max = int(np.argmax(overlaps))
        shift_max = k_max - center
        
        # Compute sensitivity at k_max for all positions (vectorized)
        sens_kmax = np.zeros(n)
        if shift_max >= 0:
            valid = np.arange(n - shift_max)
            if np.any(valid):
                sens_kmax[valid] += (1.0 - h[valid + shift_max])
            if shift_max > 0:
                valid2 = np.arange(shift_max, n)
                if np.any(valid2):
                    sens_kmax[valid2] -= h[valid2 - shift_max]
        else:
            s_val = -shift_max
            valid = np.arange(s_val, n)
            if np.any(valid):
                sens_kmax[valid] += (1.0 - h[valid - s_val])
            valid2 = np.arange(n - s_val)
            if np.any(valid2):
                sens_kmax[valid2] -= h[valid2 + s_val]
        sens_kmax *= (2.0 / n)
        
        # Pick source candidates (high sens = decreasing h helps most)
        # and target candidates (low sens = increasing h helps most)
        src_idx = np.argsort(-sens_kmax)[:n_src * 2]
        tgt_idx = np.argsort(sens_kmax)[:n_tgt * 2]
        
        # Filter: sources should have h > small, targets should have h < 1-small
        src_candidates = [int(s) for s in src_idx if h[s] > 0.005][:n_src]
        tgt_candidates = [int(t) for t in tgt_idx if h[t] < 0.995][:n_tgt]
        
        if not src_candidates or not tgt_candidates:
            break
        
        # Try all (source, target) pairs
        best_improvement = 1e-14
        best_si = -1
        best_tj = -1
        best_delta = 0.0
        
        for si in src_candidates:
            sens_si = sensitivity_vector(h, si, n)
            for tj in tgt_candidates:
                if tj == si:
                    continue
                sens_tj = sensitivity_vector(h, tj, n)
                diff_sens = sens_tj - sens_si
                delta_hi = min(h[si], 1.0 - h[tj])
                if delta_hi < 1e-8:
                    continue
                
                # Find optimal delta via ternary search
                lo = 0.0
                hi = delta_hi
                for _ in range(n_ternary):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f1 = float(np.max(overlaps + m1 * diff_sens))
                    f2 = float(np.max(overlaps + m2 * diff_sens))
                    if f1 < f2:
                        hi = m2
                    else:
                        lo = m1
                delta_star = (lo + hi) / 2.0
                if delta_star < 1e-10:
                    continue
                new_max = float(np.max(overlaps + delta_star * diff_sens))
                imp = c5 - new_max
                if imp > best_improvement:
                    best_improvement = imp
                    best_si = si
                    best_tj = tj
                    best_delta = delta_star
        
        if best_si >= 0:
            h[best_si] -= best_delta
            h[best_tj] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            overlaps = compute_all_overlaps(h)
            c5 = float(np.max(overlaps))
        else:
            break
    
    return h, c5

def coord_descent_fast(h0, n, target_sum, max_rounds=200, t_start=None, time_limit=None):
    """Fast coordinate descent using sensitivity vectors."""
    h = h0.copy()
    overlaps = compute_all_overlaps(h
