np.correlate(a, b, mode='full') returns an array of length len(a) + len(b) - 1.
Element k (0-indexed) is: sum_{i} a[i] * b[i + k - (len(b) - 1)]

So for a = h, b = 1-h, both of length n:
corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)])

Let lag = k - (n-1). Then:
corr = sum_i h[i] * (1 - h[i + lag]) where 0 ≤ i < n and 0 ≤ i+lag < n.

d(corr)/d(h[j]):

- j appears in h[i+lag] when i+lag = j, i.e., i = j-lag: contributes h[j-lag] * (-1) if 0 ≤ j-lag < n

So:

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, corr, top_k=3):
    """Compute average subgradient from top_k lag values."""
    n = len(h)
    # Find top k indices
    top_indices = np.argsort(corr)[::-1][:top_k]
    # Weight by softmax for smoothness
    top_vals = corr[top_indices]
    weights = np.exp(100 * (top_vals - top_vals[0]))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        lag = idx - (n - 1)
        indices = np.arange(n)
        valid_first = (indices + lag >= 0) & (indices + lag < n)
        valid_second = (indices - lag >= 0) & (indices - lag < n)
        grad[valid_first] += w * (1.0 - h[indices[valid_first] + lag])
        grad[valid_second] -= w * h[indices[valid_second] - lag]
    
    return grad

def local_search(h, n, target_sum, max_iters=300, alpha0=0.05):
    """Subgradient local search with adaptive step size."""
    h = project_to_sum(h.copy(), target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5_and_corrs(h)
    alpha = alpha0
    no_improve = 0
    
    for it in range(max_iters):
        _, corr = compute_c5_and_corrs(h)
        grad = subgradient_step(h, corr, top_k=5)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        h_new = h - alpha * grad / grad_norm
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new, _ = compute_c5_and_corrs(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

        else:
            no_improve += 1
            alpha = max(alpha * 0.85, 0.001)

            break

                h_ls, c5_ls = local_search_smooth(population[idx], n, target_sum,
                                                  max_iters=200, alpha_sm=50.0, step0=0.02)
                population[idx] = h_ls
                c5_vals[idx] = c5_ls
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

        new_pop = np.zeros((pop_size, n))
        elite_idx = np.argsort(c5_vals)[:elite_count]
        for i, idx in enumerate(elite_idx):
            new_pop[i] = population[idx].copy()

        fill_idx = elite_count
        mu_rate = 0.1 + 0.05 * (1.0 - gen / max(1, gen + 50))
        mu_std = 0.03 + 0.02 * (1.0 - min(1.0, gen / 100.0))
        while fill_idx < pop_size:
            p1 = tournament_select(population, c5_vals, rng, 3)
            p2 = tournament_select(population, c5_vals, rng, 3)

            cx_mask = rng.random(n) < 0.5
            c1 = np.where(cx_mask, p1, p2)
            c2 = np.where(cx_mask, p2, p1)

            for c in [c1, c2]:
                mut_mask = rng.random(n) < mu_rate
                n_mut = np.sum(mut_mask)
                if n_mut > 0:
                    c[mut_mask] += rng.normal(0, mu_std, n_mut)
                c = np.clip(c, 0.0, 1.0)
                c = project_to_sum(c, target_sum)

            new_pop[fill_idx] = c1
            fill_idx += 1
            if fill_idx < pop_size:
                new_pop[fill_idx] = c2
                fill_idx += 1

        population = new_pop
        c5_vals = np.array([compute_c5(population[i]) for i in range(pop_size)])
        gen += 1

        if gen % 30 == 0:
            worst_i = np.argmax(c5_vals)
            population[worst_i] = generate_diverse_individual(rng, n, target_sum, best_h, rng.randint(5))
            c5_vals[worst_i] = compute_c5(population[worst_i])

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    allocated = {}
    allocated[best_n] = 0.5 * budget_s
    other_ns = [120, 180, 240, 360]
    remaining = 0.45 * budget_s
    for nn in other_ns:
        if nn != best_n:
            allocated[nn] = remaining / len([x for x in other_ns if x != best_n])
        if nn % 2 != 0:
            allocated.pop(nn, None)

    for n in sorted(allocated.keys()):
        if time_left() < 30:
            break
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-6:
            continue

        n_budget = min(allocated.get(n, 60), time_left() * 0.9)
        if n_budget < 20:
            continue

        if n != best_n and best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_sum(h_interp, ts)
        else:
            h_interp = best_h.copy() if n == best_n else np.full(n, 0.5)

        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n

        pop_size = max(30, min(80, int(n_budget / 15)))
        if n > 300:
            pop_size = max(20, min(40, int(n_budget / 20)))

        alloc_ls = min(n_budget * 0.7, time_left() * 0.8)
        h_ga, c5_ga = memetic_ga(h_interp, n, ts, rng, alloc_ls, pop_size=pop_size)
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga.copy()
            best_n = n

        if time_left() > 15:
            h_cd, c5_cd = coordinate_pair_search(h_ga, n, ts, rng, max_rounds=20)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

    if best_h is not None and time_left() > 10:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha_sm in [30.0, 80.0, 200.0, 500.0]:
            if time_left() < 8:
                break
            h_pol, c5_pol = local_search_smooth(best_h, n_f, ts_f,
                                                max_iters=300, alpha_sm=alpha_sm, step0=0.01)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 8:
            h_cd, c5_cd = coordinate_pair_search(best_h, n_f, ts_f, rng, max_rounds=40)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
