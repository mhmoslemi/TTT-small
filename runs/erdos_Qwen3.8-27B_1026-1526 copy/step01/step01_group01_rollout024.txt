For coordinate j, with all other h fixed, the objective (ignoring sum constraint) is:
f(t) = max_k (corr[k] + (t - h[j]) * coeff[k])

where coeff[k] = d corr[k] / d h[j]

This is a convex piecewise linear function of t. The minimum over t in [0,1] can be found by:
- Computing f(t) at t=0, t=1
- Finding where the subgradient changes sign (i.e., where the slope goes from negative to positive)
- The slope of f at t is: max_{k: active at t} coeff[k]
  - Actually, the subgradient of f at t is the convex hull of {coeff[k] : k is active at t}
  - f is minimized at t* where 0 is in the subgradient, i.e., min active coeff <= 0 <= max active coeff

For a simpler approach, since f is convex and piecewise linear with at most (2n-1) pieces, I can:
1. Evaluate f at many points in [0,1]
2. Use the minimum as a starting point
3. Refine with the subgradient condition

Or even simpler: since the function is 1D and convex, I can use golden section search or ternary search.

    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_full(h)
    obj = smooth_max(corr, beta) * (2.0 / n)
    sum_viol = np.sum(h) - n / 2.0
    obj += penalty * sum_viol * sum_viol
    return obj


def penalty_grad(x, n, beta, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_full(h)
    dx = 2.0 / n
    
    # Gradient of smooth max
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Gradient of corr with respect to h
    # For lag m_k = k - (n-1):
    # d corr[k] / d h[j] = (1 - h[j - m_k]) * 1{0 <= j - m_k < n} - h[j + m_k] * 1{0 <= j + m_k < n}

    # grad_h[j] = sum_k weights[k] * [(1-h[j-m_k]) - h[j+m_k]] (where valid)
    
    # Let's define:
    # A[j] = sum_k weights[k] * (1-h[j-m_k]) for valid j-m_k
    # B[j] = sum_k weights[k] * h[j+m_k] for valid j+m_k
    # grad_h[j] = A[j] - B[j]
    
    # A[j] = sum_{p=0}^{n-1} weights[j-p+n-1] * (1-h[p]) where 0 <= j-p+n-1 < 2n-1
    #       = sum_{p} (1-h[p]) * w[j-p+n-1] where w = weights
    # This is a cross-correlation.
    
    # Using the definition: (a * b)[n] = sum_k a[k] * b[n-k]
    # A[j] = sum_p (1-h[p]) * w[n-1+j-p]

def coord_descent_func(h, n, max_sweeps=20):
    """Coordinate descent: for each j, find optimal t in [0,1] minimizing max_k corr.
    Then project to satisfy sum constraint.
    """
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            if old <= 1e-15 or old >= 1.0 - 1e-15:
                continue
            
            corr = compute_corr_full(h)
            
            # Compute coefficient of h[j] in each corr[k]
            coeffs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                mk = k - (n - 1)
                c = 0.0
                jm = j - mk
                if 0 <= jm < n:
                    c += (1.0 - h[jm])
                jp = j + mk
                if 0 <= jp < n:
                    c -= h[jp]
                coeffs[k] = c
            
            # f(t) = max_k (corr[k] + (t - old) * coeffs[k])
            # = max_k ((corr[k] - old * coeffs[k]) + t * coeffs[k])
            a_vals = corr - old * coeffs
            b_vals = coeffs
            
            # Ternary search for min of f(t) over [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(80):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a_vals + m1 * b_vals)
                f2 = np.max(a_vals + m2 * b_vals)
                if f1 < f2:

                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0

            # If projection made it worse, we might want to revert

    corr = compute_corr_full(h)

    # m_k = k - (n-1)

    # Create matrix C[j, k] = coeff(k, j)
    # Then grad_h = C.T @ weights

    jp_clipped = np.clip(jp, 0, n-1)

    grad_h = coeff.T @ weights  # (n,)
    
    sum_viol = np.sum(h) - n / 2.0
    grad = dx * grad_h + 2.0 * penalty * sum_viol
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def penalty_obj(x, n, beta, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_full(h)
    obj = smooth_max(corr, beta) * (2.0 / n)
    sv = np.sum(h) - n / 2.0
    return obj + penalty * sv * sv


def penalty_grad(x, n, beta, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_full(h)
    dx = 2.0 / n
    
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / exp_vals.sum()
    
    js = np.arange(n)
    ms = np.arange(2 * n - 1) - (n - 1)
    jm = js[:, None] - ms[None, :]
    jp = js[:, None] + ms[None, :]
    valid_m = (jm >= 0) & (jm < n)
    valid_p = (jp >= 0) & (jp < n)
    jm_c = np.clip(jm, 0, n - 1)
    jp_c = np.clip(jp, 0, n - 1)
    coeff = (1.0 - h[jm_c]) * valid_m.astype(float) - h[jp_c] * valid_p.astype(float)
    grad_h = coeff.T @ weights
    
    sv = np.sum(h) - n / 2.0
    return dx * grad_h + 2.0 * penalty * sv


def slsqp_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr_full(h)
    return smooth_max(corr, beta) * (2.0 / n)


def slsqp_cons(x, n):
    return np.sum(x) - n / 2.0


def coord_descent(h, n, max_sweeps=20):
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            if old <= 1e-15 or old >= 1.0 - 1e-15:
                continue
            
            corr = compute_corr_full(h)
            ms = np.arange(2 * n - 1) - (n - 1)
            jm = j - ms
            jp = j + ms
            vm = (jm >= 0) & (jm < n)
            vp = (jp >= 0) & (jp < n)
            jm_c = np.clip(jm, 0, n - 1)
            jp_c = np.clip(jp, 0, n - 1)
            coeffs = (1.0 - h[jm_c]) * vm.astype(float) - h[jp_c] * vp.astype(float)
            
            a_vals = corr - old * coeffs
            b_vals = coeffs
            
            lo, hi = 0.0, 1.0
            for _ in range(80):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a_vals + m1 * b_vals)
                f2 = np.max(a_vals + m2 * b_vals)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            
            if abs(t_opt - old) > 1e-12:
                h[j] = t_opt
                improved = True
        
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved and sweep > 2:

    return h, compute_c5(h)

    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)
