import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_con_fun(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * i / period + phase)
    h += rng.normal(0.0, 0.02, n)
    return project_feasible(h, n)

def binary_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return project_feasible(h, n)

        h += amp * np.sin(2.0 * np.pi * i / freq + phase)
    h += rng.normal(0.0, 0.01, n)
    return project_feasible(h, n)

def slsqp(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None
