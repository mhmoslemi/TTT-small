import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(h[idx] + diff / len(idx), 1.0)
        else:
            mask = h > 1e-15
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(h[idx] + diff / len(idx), 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def subgradient_direction(h, n):
    ov = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(ov)
    shift = k_max - (n - 1)
    
    # Find all k that are close to the max (within 1e-6) to average gradients
    threshold = ov[k_max] - 1e-6
    k_active = np.where(ov >= threshold)[0]
    
    grad = np.zeros(n)
    for k in k_active:
        s = k - (n - 1)
        # d/dh[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    
    grad /= len(k_active)
    return grad, ov, k_max

def subgradient_optimize(h0, n, time_limit, max_iter=100000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.85  # momentum
    t_start = time.time()
    
    # Polyak-like step size
    best_f = best_c5
    f_ref = best_f + 0.01  # reference value for step size
    
    lr = 0.05
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        grad, ov, k_max = subgradient_direction(h, n)
        
        # Polyak step size: lr = (f(h) - f_ref) / ||grad||^2
        f_curr = np.max(ov) * (2.0 / n)
        grad_norm_sq = np.dot(grad, grad)
        if grad_norm_sq > 1e-10:
            lr_polyak = (f_curr - best_f) / (grad_norm_sq * (2.0/n)) * 2.0
            lr_polyak = np.clip(lr_polyak, 0.001, 0.5)
        else:
            lr_polyak = lr
        
        # Blend with fixed lr for stability
        lr_use = 0.5 * lr + 0.5 * lr_polyak
        
        v = mu * v - lr_use * grad
        h_new = project_feasible(h + v, n)
        
        c5_new = compute_c5(h_new)
        h = h_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.2)
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr = max(0.005, lr * 0.8)
            if no_improve > 500:
                # Reset momentum and try fresh
                v = np.zeros(n)
                lr = 0.1
                no_improve = 0
    
    return best_h, best_c5

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    ov = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(ov, beta) * (2.0 / n)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def slsqp_optimize(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def make_perturbed(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)

def simulated_annealing(h0, n, time_limit, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    T = 0.01
    T_min = 0.0001
    
    while time.time() - t_start < time_limit:
        T *= 0.9995
        if T < T_min:
            T = T_min
        
        # Random perturbation
        n_perturb = rng.randint(max(1, n//50), n//5 + 1)
        idx = rng.choice(n, n_perturb, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.normal(0.0, 0.05, n_perturb)
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - compute_c5(h)
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] = np.minimum(h[idx] + diff / len(idx), 1.0)
        else:
            mask = h > 1e-15
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            h[idx] = np.maximum(h[idx] + diff / len(idx), 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_smooth(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    ov = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(ov, beta) * (2.0 / n)


def obj_grad_smooth(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    ov = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(ov)
    shifted = np.clip(beta * (ov - m), -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    norm = np.sum(exp_shifted)
    w = exp_shifted / norm
    f = (m + np.log(norm) / beta) * (2.0 / n)
    grad = np.zeros(n)
    for k in range(len(ov)):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wk * h[lo2 + s:hi2 + s]
    grad *= (2.0 / n)
    return f, grad


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def slsqp_optimize(h0, n, max_iter, beta, use_grad=True):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        if use_grad:
            res = minimize(obj_grad_smooth, h0, jac=True, method='SLSQP',
                           bounds=bounds, constraints=cons, args=(n, beta),
                           options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        else:
            res = minimize(obj_smooth, h0, method='SLSQP', bounds=bounds,
                           constraints=cons, args=(n, beta),
                           options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def subgradient_direction(h, n):
    ov = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(ov)
    threshold = ov[k_max] - 1e-8
    k_active = np.where(ov >= threshold)[0]
    grad = np.zeros(n)
    for k in k_active:
        s = k - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    grad /= float(len(k_active))
    return grad


def subgradient_optimize(h0, n, time_limit, max_iter=500000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    mu = 0.80
    t_start = time.time()
    lr = 0.08
    no_improve = 0
    f_curr = best_c5

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        grad = subgradient_direction(h, n)
        gn = np.dot(grad, grad)
        if gn > 1e-14:
            lr_polyak = (f_curr - best_c5 + 0.002) / gn
            lr_polyak = np.clip(lr_polyak, 0.002, 0.3)
        else:
            lr_polyak = lr
        lr_use = 0.4 * lr + 0.6 * lr_polyak
        v = mu * v - lr_use * grad
        h = project_feasible(h + v, n)
        f_curr = compute_c5(h)
        if f_curr < best_c5:
            best_c5 = f_curr
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.25)
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr = max(0.005, lr * 0.75)
            if no_improve > 400:
                v = np.zeros(n)
                lr = 0.1
                no_improve = 0
    return best_h, best_c5


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_perturbed(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def simulated_annealing(h0, n, time_limit, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    T = 0.008
    T_min = 0.00005
    while time.time() - t_start < time_limit:
        T = max(T * 0.9997, T_min)
        n_perturb = rng.randint(max(1, n // 40), n // 4 + 1)
        idx = rng.choice(n, n_perturb, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.normal(0.0, 0.04, n_perturb)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5


def make_structured(n, period, amplitude, phase, rng):
    i = np.arange(n, dtype=float)
    h = 0.5 + amplitude * np.sin(2.0 * np.pi * i / period + phase)
    h = h + rng.normal(0.0, 0.02, n)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
