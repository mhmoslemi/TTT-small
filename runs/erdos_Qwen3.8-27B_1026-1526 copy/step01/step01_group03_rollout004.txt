        x_new = np.linspace(0, 2, nc, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, nc)

        # Try a few different initializations at this resolution
        inits = [h_try]
        inits.append(project_feasible(h_try + rng.randn(nc) * 0.05, nc))
        inits.append(project_feasible(h_try + rng.randn(nc) * 0.1, nc))

        for h_init in inits:
            if remaining(0.78) < 8:
                break
            h_r, c5_r = slsqp_refine(
                h_init, nc,
                taus=[0.05, 0.03, 0.015, 0.008],
                maxiter_per_tau=60,
                time_limit_fn=lambda: remaining(0.78)
            )
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

        if remaining(0.78) > 8:
            h_sg, c5_sg = subgradient_descent(best_h, len(best_h), n_iter=200, top_k_count=6)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg

    # Phase 4: Extensive final polishing
    if remaining(0.94) > 10:
        n_fin = len(best_h)
        bounds_f = [(0.0, 1.0)] * n_fin
        cons_f = [{'type': 'eq', 'fun': constraint_eq, 'args': (n_fin,)}]

        for tau in [0.004, 0.002, 0.001, 0.0006, 0.0003]:
            if remaining(0.97) < 4:
                break
            for attempt in range(10):
                if remaining(0.97) < 2:
                    break
                scale = 0.002 + 0.003 * attempt
                pert = project_feasible(best_h + rng.randn(n_fin) * scale, n_fin)
                try:
                    res = minimize(obj_smooth, pert, args=(tau,), method='SLSQP',
                                  bounds=bounds_f, constraints=cons_f,
                                  options={'maxiter': 250, 'ftol': 1e-15})
                    h_pol = project_feasible(res.x, n_fin)
                    c5_pol = evaluate_c5(h_pol)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol
                except Exception:
                    pass

        # Final subgradient pass
        if remaining(0.97) > 5:
            h_fin, c5_fin = subgradient_descent(best_h, n_fin, n_iter=800, top_k_count=12)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
            # One more SLSQP after subgradient
            h_final, c5_final = slsqp_refine(
                best_h, n_fin,
                taus=[0.001, 0.0005],
                maxiter_per_tau=150,
                time_limit_fn=lambda: remaining(0.99)
            )
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final

    # Ensure final feasibility
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_out
