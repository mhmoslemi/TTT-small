```python
def one_lp_iteration(h_current, n, target_sum):
    """One iteration of the outer approximation (linearization) method."""
    from scipy.optimize import linprog
    
    # Linearize g_k at h_current
    # g_k(h) ≈ g_k(h*) + sum_j [grad_k[j]] * (h[j] - h*[j])
    # = [g_k(h*) - sum_j grad_k[j] * h*[j]] + sum_j grad_k[j] * h[j]
    
    # Compute gradients
    grad_all = np.zeros((2*n-1, n))
    for lag in range(-(n-1), n):
        k = lag + n - 1
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        if i_start < i_end:
            grad_all[k, i_start:i_end] = (1.0 - h_current[i_start + lag:i_end + lag])
            grad_all[k, i_start + lag:i_end + lag] -= h_current[i_start:i_end]
    
    # g_k(h*) values
    lags = compute_c5_all_lags(h_current)
    
    # Constant term: g_k(h*) - grad_k . h*
    const = lags - np.sum(grad_all * h_current, axis=1)
    
    # LP: min t
    # s.t. t - sum_j grad_k[j] * h[j] >= const_k  for all k
    #      0 <= h[j] <= 1
    #      sum h[j] = target_sum
    #
    # In standard form for linprog: min c^T x
    # x = [t, h_0, ..., h_{n-1}]
    # c = [1, 0, ..., 0]
    
    n_vars = 1 + n
    c = np.zeros(n_vars)
    c[0] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # t - sum grad_k[j]*h[j] >= const_k
    # => -t + sum grad_k[j]*h[j] <= -const_k
    A_ub = np.zeros((2*n-1, n_vars))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = grad_all
    b_ub = -const
    
    # Equality constraints: sum h = target_sum
    A_eq = np.zeros((1, n_vars))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([target_sum])
    
    # Bounds: t is free, h in [0,1]
    bounds = [(None, None)] + [(0, 1)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return np.clip(res.x[1:], 0, 1)
    else:
        return h_current
```

from scipy.optimize import minimize, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_objective_and_grad(h, alpha):
    n = len(h)
    lags = compute_c5_all_lags(h)
    m = np.max(lags)
    shifted = alpha * (lags - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    log_sum = np.max(shifted) + np.log(np.sum(exp_vals))
    val = m + log_sum / alpha
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        w = weights[lag + n - 1]
        if w < 1e-14:
            continue
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        if i_start >= i_end:
            continue
        grad[i_start:i_end] += w * (1.0 - h[i_start + lag:i_end + lag])
        grad[i_start + lag:i_end + lag] -= w * h[i_start:i_end]
    return val, grad

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        return smooth_max_objective_and_grad(x, alpha)[0]
    def grad(x):

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
