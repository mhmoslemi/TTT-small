        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15

            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:

            if not np.any(mask):
                break
            sub = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - sub, 0.0)

    def obj(x):
        return x[-1]

        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * (2.0 / len(h))

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            add = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            sub = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    return np.clip(h, 0.0, 1.0)


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * (2.0 / n)


def slsqp_eq(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_refine(h, n, maxiter=3000, ftol=1e-14):
    c5_init = compute_c5(h)
    x0 = np.concatenate([h, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h, c5_init

    h_full = h_template.copy()

    dx = 2.0 / n
    return np.max(c * dx)
