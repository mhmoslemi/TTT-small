import numpy as np
from scipy.optimize import minimize, fminbound
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-13
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Add to entries with most room
            order = np.argsort(1.0 - h)[::-1]
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-13:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if diff > -1e-13:
                    break
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_slsqp(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return smooth_max(corr, beta)

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def optimize_slsqp(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(obj_slsqp, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except:
        return None, None

def projected_subgradient_step(h, n, lr):
    """One step of projected subgradient descent on the max of correlations."""
    dx_val = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx_val
    max_val = np.max(corr)
    # Active set: correlations within small tolerance of max
    active = np.where(corr > max_val - 0.01)[0]
    
    # Compute subgradient: average gradient over active set
    # grad of corr_k w.r.t. h: 
    # corr_k = sum_i h[i+k]*(1-h[i]) for the positive lag k
    # d(corr_k)/d(h[j]) involves contributions from j as both i and i+k
    
    # Actually, let's compute the gradient numerically for the active max
    # Or analytically:
    # For the "full" correlation, index m in corr corresponds to lag k = m - (n-1)
    # corr[m] = sum_i h[min(i+k, n-1)] * (1-h[max(i-k, 0)]) ... this is getting complex
    
    # Let me just do it numerically with the specific active lags
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)  # actual lag
        if k >= 0:
            # corr[m] = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
            # d/dh[j]: 
            #   if j = i+k for some i in [0, n-1-k]: contributes (1-h[i]) = (1-h[j-k])
            #   if j = i for some i in [0, n-1-k]: contributes -h[i+k] = -h[j+k]
            for i in range(n - k):
                if i + k < n:
                    grad[i + k] += (1 - h[i])
                if i < n:
                    grad[i] -= h[i + k]
        else:
            m2 = -k  # positive magnitude
            # corr[m] = sum_{i=m2}^{n-1} h[i] * (1-h[i-m2])
            for i in range(m2, n):
                grad[i] += (1 - h[i - m2])
                grad[i - m2] -= h[i]
    
    grad /= len(active)
    grad *= dx_val  # scale by dx since corr already includes dx
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new)

def local_search(h, n, rng, n_steps):

For k ≥ 0: c_k = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])

∂c_k/∂h[j]:
- h[j] appears in the first factor h[i+k] when i+k = j, i.e., i = j-k. This is valid when 0 ≤ j-k ≤ n-1-k, i.e., k ≤ j ≤ n-1.
  Contribution: (1-h[i]) = (1-h[j-k])
- h[j] appears in the second factor (1-h[i]) when i = j. This is valid when 0 ≤ j ≤ n-1-k.
  Contribution: -h[i+k] = -h[j+k]

So:
[∂c_k/∂h[j]] = (1-h[j-k]) * 1[k≤j≤n-1] - h[j+k] * 1[0≤j≤n-1-k]

In vector form:
grad_k = zeros(n)
grad_k[k:n] += (1 - h[0:n-k])  # (1-h[j-k]) for j from k to n-1
grad_k[0:n-k] -= h[k:n]  # -h[j+k] for j from 0 to n-1-k

= sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])  [substituting j = i-m]

This is the same as c_m! So the gradient is the same.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_half_corrs(h):
    """Compute correlations for lags 0 to n-1 (the independent half)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # The lags k=0,...,n-1 correspond to indices n-1, n, ..., 2n-2
    return corr[n-1:] * dx  # shape (n,)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()

    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)

        if diff > 0:

                if add <= 0: break

                diff += sub

        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except:
        return None, None

def compute_subgrad_h(h, n):
    """Compute subgradient of max correlation w.r.t. h."""
    half_corr = compute_half_corrs(h)  # shape (n,), lags 0..n-1
    max_val = np.max(half_corr)
    # Active set: lags within tolerance
    tol = 0.005
    active = np.where(half_corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        if k == 0:
            # c_0 = sum_i h[i]*(1-h[i]), grad = 1-2h[i]
            grad += 1.0 - 2.0 * h
        else:
            # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # grad[j] += (1-h[j-k]) for j=k..n-1
            # grad[j] -= h[j+k] for j=0..n-1-k
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
    grad /= max(1, len(active))
    grad *= (2.0 / n)
    return grad

            best_c5 = c5
        h = h_new
        lr *= lr_decay

    half = corr[n - 1:]
    max_val = np.max(half)
    tol = 0.003
    active = np.where(half >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        if k == 0:
            grad += 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
    grad /= max(1, len(active))
    grad *= dx_val
    return grad


def subgradient_descent(h, n, max_iters, lr_init, lr_decay, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t_start = time.time()
    plateau = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        grad = compute_subgrad(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-8:
            best_h = h_new.copy()
            best_c5 = c5
            plateau = 0
        else:
            plateau += 1
            if plateau > 200:
                lr *= 0.9
                plateau = 0
        h = h_new
        lr *= lr_decay
    return best_h, best_c5


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_sinusoidal_perturb(h, n, period, amplitude, phase, rng):
    i = np.arange(n, dtype=float)
    pert = amplitude * np.sin(2.0 * np.pi * i / period + phase)
    h_p = h + pert + rng.normal(0.0, 0.02, n)
    return project_feasible(h_p, n)


def local_search_push(h, n, rng, n_tries):
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_tries):
        idx = rng.choice(n, size=2, replace=False)
        for delta in [0.05, -0.05, 0.1, -0.1]:
            h_tr = best.copy()
            h_tr[idx[0]] = np.clip(h_tr[idx[0]] + delta, 0.0, 1.0)
            h_tr[idx[1]] = np.clip(h_tr[idx[1]] - delta, 0.0, 1.0)
            h_tr = project_feasible(h_tr, n)
            c5 = compute_c5(h_tr)
            if c5 < best_c5 - 1e-7:
                best = h_tr.copy()
                best_c5 = c5
                break
    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals():
        try:
            h_init = np.array(initial_h_values, dtype=float).copy()
            n_init = len(h_init)
            h_init = project_feasible(h_init, n_init)
            best_h = h_init
            best_c5 = compute_c5(h_init)
            best_n = n_init
        except Exception:
            pass

    if best_h is None:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    # Phase 1: Initial SLSQP polish
    if time.time() - t0 < budget_s - 200:
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, 800, 100.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Subgradient descent on true max
    elapsed = time.time() - t0
    if elapsed < budget_s - 400:
        sg_budget = min(200.0, budget_s - 400 - elapsed)
        h_sg, c5_sg = subgradient_descent(best_h, best_n, 5000, 0.004, 0.9995, sg_budget)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, 500, 120.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Structured multi-start (sinusoidal perturbations)
    periods_to_try = [3, 4, 5, 6, 7, 8, 10, 12, 15, 19, 24, 30, 47, 95]
    for period in periods_to_try:
        if time.time() - t0 > budget_s - 250:
            break
        for amp in [0.08, 0.15, 0.25]:
            if time.time() - t0 > budget_s - 250:
                break
            for phase in [0.0, np.pi / 2.0, np.pi]:
                if time.time() - t0 > budget_s - 250:
                    break
                h_pert = make_sinusoidal_perturb(best_h, best_n, period, amp, phase, rng)
                h_opt, c5_opt = optimize_slsqp(h_pert, best_n, 350, 100.0)
                if h_opt is not None and c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Random multi-start SLSQP
    no_improve = 0
    while time.time() - t0 < budget_s - 180:
        sigma = 0.06 + 0.06 * (0.9 ** min(no_improve, 35))
        h_pert = best_h + rng.normal(0, sigma, best_n)
        h_pert = project_feasible(h_pert, best_n)
        if rng.random() < 0.1:
            h_pert = best_h + rng.normal(0, 0.35, best_n)
            h_pert = project_feasible(h_pert, best_n)
        h_opt, c5_opt = optimize_slsqp(h_pert, best_n, 300, 100.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 35:
            break

    # Phase 5: Multi-resolution (up then down)
    for n_hi in [380, 570]:
        if time.time() - t0 > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, n_hi)
        h_interp = project_feasible(h_interp, n_hi)
        h_opt, c5_opt = optimize_slsqp(h_interp, n_hi, 400, 60.0)
        if h_opt is not None:
            h_down = interpolate_h(h_opt, n_hi, best_n)
            h_down = project_feasible(h_down, best_n)
            c5_down = compute_c5(h_down)
            if c5_down < best_c5:
                best_h, best_c5 = h_down, c5_down
            h_opt2, c5_opt2 = optimize_slsqp(h_down, best_n, 300, 100.0)
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 6: Local search (pair pushes)
    if time.time() - t0 < budget_s - 60:
        best_h, best_c5 = local_search_push(best_h, best_n, rng, 300)

    # Phase 7: Final SLSQP polish with high beta
    if time.time() - t0 < budget_s - 25:
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, 600, 300.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 8: Ultra-fine subgradient
    if time.time() - t0 < budget_s - 10:
        h_sg, c5_sg = subgradient_descent(best_h, best_n, 2000, 0.001, 0.999, 15.0)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
