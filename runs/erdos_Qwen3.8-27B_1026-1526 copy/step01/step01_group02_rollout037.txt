    grad = np.zeros(n)
    for k in range(m):
        if weights[k] < 1e-10:
            continue
        shift = k - (n - 1)
        # First term: (1-h[j+shift]) for j where 0<=j+shift<n, i.e., -shift <= j <= n-1-shift
        j_min1 = max(0, -shift)
        j_max1 = min(n - 1, n - 1 - shift)
        if j_min1 <= j_max1:
            grad[j_min1:j_max1+1] += weights[k] * (1.0 - h[j_min1 + shift:j_max1 + shift + 1])
        # Second term: -h[j-shift] for j where 0<=j-shift<n, i.e., shift <= j <= n-1+shift

        j_max2 = min(n - 1, n - 1 + shift)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= weights[k] * h[j_min2 - shift:j_max2 - shift + 1]
    
    grad *= dx
    return grad
