    c = np.correlate(h, 1.0 - h, mode="full")

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max(vals, T):
    m = np.max(vals)
    if T < 1e-12:
        return m
    shifted = (vals - m) / T
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)

def compute_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

def compute_lags(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def soft_max(vals, T):
    m = np.max(vals)
    if T < 1e-12:
        return m
    shifted = np.clip((vals - m) / T, -50, 50)
    return m + T * np.log(np.sum(np.exp(shifted)))


def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max(vals, T):
    m = np.max(vals)
    if T < 1e-12:
        return m
    shifted = np.clip((vals - m) / T, -50.0, 50.0)
    return m + T * np.log(np.sum(np.exp(shifted)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = float(np.sum(mask))
            room = (1.0 - h[mask]).min()
            h[mask] += min(diff / cnt, room)
        else:
            mask = h > 0.0

                break
            cnt = float(np.sum(mask))
            room = h[mask].min()
