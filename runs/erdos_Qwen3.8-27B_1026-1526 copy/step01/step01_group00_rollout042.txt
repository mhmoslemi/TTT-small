def gradient_descent(h0, n, target_sum, alpha, lr, max_iters, time_limit):
    """Projected gradient descent with adaptive step size."""
    h = h0.copy()
    t_start = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        smax, weights = smooth_max_and_weights(corr, alpha)
        grad = compute_gradient(h, weights)
        
        # Project gradient onto the hyperplane sum = 0 (for the linear constraint)
        # The feasible set is {h: 0<=h<=1, sum(h)=S}
        # Gradient step: h + alpha * (grad - mean(grad)) to stay on the sum constraint
        # (approximately, before the projection)
        grad_centered = grad - np.mean(grad)
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = h - step * grad_centered
            h_projected = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_projected)
            if c5_new < smax - 1e-10 * step:
                break
            step *= 0.5
        
        h = h_projected
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            h_best = h.copy()
    
    return h_best, best_c5

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from n_old points to n_new points on [0,2)."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Periodic interpolation
    h_new = np.zeros(n_new)
    for i in range(n_new):
        # Find the position in the old grid (periodic)
        pos = x_new[i] * n_old / 2.0
        i0 = int(pos) % n_old
        i1 = (i0 + 1) % n_old
        frac = pos - int(pos)
        h_new[i] = (1-frac) * h_old[i0] + frac * h_old[i1]
    target = n_new / 2.0
    return project_to_sum(h_new, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        clamped = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(clamped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-9:
        mask = (result > 1e-7) & (result < 1.0 - 1e-7)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

    m = np.max(corr)

    shifted = np.clip(shifted, -400, 400)
    ev = np.exp(shifted)
    se = np.sum(ev)
    smax = m + np.log(se) / alpha
    w = ev / se
    return smax, w

def compute_gradient(h, weights):
    n = len(h)
    ft = np.convolve(weights, 1.0 - h, mode='full')
    first_term = ft[n-1:2*n-1]
    second_term = np.zeros(n)
    for s in range(-(n-1), n):
        ws = weights[s + n - 1]
        if ws < 1e-13:
            continue
        j0 = max(0, -s)
        j1 = min(n-1, n-1-s)
        if j0 <= j1:
            second_term[j0:j1+1] += ws * h[j0+s:j1+1+s]
    return first_term - second_term

    """Gradient using only the active (near-max) shifts."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    active = corr >= m - tol
    if not np.any(active):
        active = np.zeros_like(corr, dtype=bool)
        active[np.argmax(corr)] = True
    w = np.zeros_like(corr)
    w[active] = 1.0 / np.sum(active)
    return compute_gradient(h, w), m

    h_best = h.copy()
    if lr is None:
        lr = 1.0 / (alpha + 1.0)
    step = lr
    momentum = np.zeros(n) if use_momentum else None
    beta = 0.7
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        smax, w = get_smooth_max_stuff(corr, alpha)
        grad = compute_gradient(h, w)
        grad -= np.mean(grad)
        
        if momentum is not None:
            momentum = beta * momentum + grad
            direction = momentum
        else:

            if c5_new < smax - 1e-12:
                best_h_new = h_new
                best_c5_new = c5_new
                break
            step *= 0.5
            if step < 1e-10:

            best_c5_new = c5_new

            step = min(step * 1.2, lr * 2)  # cautiously increase

            # corr[k] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
            val = np.dot(h[s:], one_minus_h[:n-s])
        else:
            # s < 0, let t = -s
            t = -s
            # corr[k] = sum_{i=t}^{n-1} h[i] * (1-h[i-t])
            val = np.dot(h[t:], one_minus_h[:n-t])
        if val > max_val:
            max_val = val
    return max_val * dx

def find_active(corr, tolerance=0.01):
    """Find indices of near-max correlation values."""
    m = np.max(corr)
    return np.where(corr >= m - tolerance)[0]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-10:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        clamped = np.clip(h - mid, 0.0, 1.0)
        if np.sum(clamped) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-9:
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def get_smooth_max_stuff(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400, 400)
    ev = np.exp(shifted)
    se = np.sum(ev)
    smax = m + np.log(se) / alpha
    w = ev / se
    return smax, w

def compute_gradient(h, weights):
    n = len(h)
    ft = np.convolve(weights, 1.0 - h, mode='full')
    first_term = ft[n - 1:2 * n - 1]
    second_term = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = weights[s + n - 1]
        if ws < 1e-14:
            continue
        j0 = max(0, -s)
        j1 = min(n - 1, n - 1 - s)
        if j0 <= j1:
            second_term[j0:j1 + 1] += ws * h[j0 + s:j1 + 1 + s]
    return first_term - second_term

def compute_corr_at(h, k):
    n = len(h)
    m = n - 1
    s = k - m
    if s >= 0:
        return np.dot(h[s:], 1.0 - h[:n - s])
    else:
        t = -s
        return np.dot(h[t:], 1.0 - h[:n - t])

def compute_c5_active(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    max_val = 0.0
    for k in active_indices:
        val = compute_corr_at(h, k)
        if val > max_val:
            max_val = val
    return max_val * dx

def find_active(corr, tolerance=0.005):
    m = np.max(corr)
    return np.where(corr >= m - tolerance)[0].tolist()

def pgd(h0, n, target_sum, alpha, max_iters, time_limit, use_momentum=False):
    h = h0.astype(float).copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_best = h.copy()
    lr = 1.0 / (alpha * 0.5 + 1.0)
    step = lr
    momentum = np.zeros(n) if use_momentum else None
    beta = 0.8

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        smax, w = get_smooth_max_stuff(corr, alpha)
        grad = compute_gradient(h, w)
        grad -= np.mean(grad)

        if momentum is not None:
            momentum = beta * momentum + grad
            direction = momentum
        else:
            direction = grad

        h_new = h - step * direction
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new > smax:
            h_new2 = h - step * 0.3 * direction
            h_new2 = project_to_sum(h_new2, target_sum)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 > smax:
                h_new2 = h - step * 0.1 * direction
                h_new2 = project_to_sum(h_new2, target_sum)
                c5_new2 = compute_c5(h_new2)
            if c5_new2 < c5_new:
                h_new = h_new2
                c5_new = c5_new2
            step *= 0.5
        else:
            step = min(step * 1.1, lr * 3.0)

        h = h_new
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            h_best = h.copy()

    return h_best, best_c5

def swap_search(h0, n, target_sum, rng, max_iters, time_limit):
    h = h0.astype(float).copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = find_active(corr, 0.01)

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        a, b = rng.choice(n, size=2, replace=False)
        if abs(h[a] - h[b]) < 0.01:
            continue
        h[a], h[b] = h[b], h[a]
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5_active(h, active)
        if c5_new < best_c5 - 1e-10:
            best_c5 = compute_c5(h)
            best_h = h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full')
            active = find_active(corr, 0.01)
        else:
            h[a], h[b] = h[b], h[a]
            h = project_to_sum(h, best_c5 / (2.0 / n) / n * n, target_sum)
            h[a], h[b] = best_h[a], best_h[b]
            h = h.copy()

    h = best_h.copy()
    return h, best_c5

def interpolate_to(n_from, h_from, n_to):
    x_from = np.arange(n_from) / n_from * 2.0
    x_to = np.arange(n_to) / n_to * 2.0
    h_to = np.zeros(n_to)
    for i in range(n_to):
        pos = x_to[i] / 2.0 * n_from
        i0 = int(pos) % n_from
        i1 = (i0 + 1) % n_from
        frac = pos - int(pos)
        h_to[i] = (1.0 - frac) * h_from[i0] + frac * h_from[i1]
    return project_to_sum(h_to, n_to / 2.0)

def multispectral_init(n, rng, freqs):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for f, amp in freqs:
        h += amp * np.sin(2.0 * np.pi * f * x / 2.0)
    return project_to_sum(h, n / 2.0)
