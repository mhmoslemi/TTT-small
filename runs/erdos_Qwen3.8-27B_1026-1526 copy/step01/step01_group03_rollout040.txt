        ig = np.zeros(n)

        corr = compute_corr(hh)
        w = smooth_max_grad_coeff(corr, tau)
        ig = np.zeros(n)
        for i in range(n):
            idx1 = i + s
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - s
            valid2 = (idx2 >= 0) & (idx2 < n)
            t1 = np.zeros(narr)
            t1[valid1] = 1.0 - hh[idx1[valid1]]
            t2 = np.zeros(narr)
            t2[valid2] = hh[idx2[valid2]]
            ig[i] = dx * (np.sum(w * t1) - np.sum(w * t2))
        return ig
    
    def cons_fun(hh):
        return np.sum(hh) - target
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def interpolate_to_n(parent, n_target, rng):
    n_src = len(parent)
    if n_src == n_target:
        return parent.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_target, endpoint=False)
    if n_src >= 4:
        from scipy.interpolate import interp1d
        f = interp1d(x_src, parent, kind='cubic', fill_value='extrapolate')
        h = f(x_tgt)
    else:
        f = interp1d(x_src, parent, kind='linear', fill_value='extrapolate')
        h = f(x_tgt)
    return np.clip(h, 0.0, 1.0)

def generate_perturbed(h, n, rng, scale=0.1):
    h_pert = h + rng.randn(n) * scale
    return project_feasible(h_pert, n)

def generate_structured_starts(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)
    starts = []
    
    for freq in [1.2, 1.4, 1.6, 1.8, 2.0, 2.2, 2.5]:
        for phase in [0, np.pi/2, np.pi, 3*np.pi/4]:
            amp = 0.45
            h = np.clip(0.5 + amp * np.sin(2*np.pi*freq*x + phase), 0, 1)
            starts.append(h)
    
    for freq in [1.3, 1.7, 2.1, 2.7]:
        h = np.clip(0.5 + 0.4*np.cos(2*np.pi*freq*x) + 0.1*np.cos(2*np.pi*2.3*freq*x), 0, 1)
        starts.append(h)
    
    for _ in range(3):
        h = 0.5 + rng.uniform(0.2, 0.45) * np.sin(2*np.pi*rng.uniform(1,3)*x + rng.uniform(0, 2*np.pi))
        h = np.clip(h, 0, 1)
        starts.append(h)
    
    h_tri = np.mod(ar / n * 3.0, 2.0)
    h_tri = np.where(h_tri < 1.0, h_tri, 2.0 - h_tri)
    starts.append(np.clip(h_tri * 0.8 + 0.1, 0, 1))
    
    h_saw = ar / float(n)
    h_saw = 0.3 + 0.4 * h_saw
    starts.append(np.clip(h_saw, 0, 1))
    
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    def elapsed():
        return time.time() - start_time
    
    def time_left_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)
    
    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
    
    n_values = [41, 47, 53, 59, 61, 65, 69, 73, 79, 85, 91, 97]
    if parent_h is not None:
        n_values.insert(0, len(parent_h))
        n_values = list(dict.fromkeys(n_values))
    
    def try_optimize(h0, n, tau_start, n_passes_per_tau, lambda_sum):
        nonlocal best_h, best_c5
        h = h0.copy()
        taus = [tau_start * (0.5**k) for k in range(6)]
        taus = [max(t, 0.001) for t in taus]
        
        for tau in taus:
            if elapsed() > budget_s * 0.80:
                return
            npasses = n_passes_per_tau
            h, _ = coord_descent(h, n, tau, npasses, rng, lambda_sum)
            h = project_feasible(h, n)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        return h
    
    for n in n_values:
        if elapsed() > budget_s * 0.65:
            break
        
        starts = []
        
        if parent_h is not None:
            h_interp = interpolate_to_n(parent_h, n, rng)
            starts.append(h_interp)
            starts.append(project_feasible(1.0 - h_interp, n))
            starts.append(generate_perturbed(h_interp, n, rng, 0.08))
            starts.append(generate_perturbed(h_interp, n, rng, 0.15))
            starts.append(generate_perturbed(h_interp, n, rng, 0.25))
        
        struct_starts = generate_structured_starts(n, rng)
        starts.extend(struct_starts[:6])
        
        starts.append(rng.rand(n))
        starts.append(np.clip(0.5 + rng.uniform(0.2, 0.4)*np.sin(2*np.pi*rng.uniform(1,3)*np.linspace(0,2,n,endpoint=False)), 0, 1))
        
        for h0 in starts:
            if elapsed() > budget_s * 0.65:
                break
            h0 = project_feasible(h0, n)
            h_opt = try_optimize(h0, n, 0.15, 40, 0.5)
        
        if h_opt is not None and elapsed() < budget_s * 0.75:
            h_lbg = lbfgs_smooth(h_opt, n, 0.005, maxiter=150)
            c5 = evaluate_c5(h_lbg)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_lbg.copy()
    
    if best_h is None:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)
    
    n_best = len(best_h)
    h_best = best_h.copy()
    
    if elapsed() < budget_s * 0.85:
        h_best = lbfgs_smooth(h_best, n_best, 0.003, maxiter=300)
        h_best = project_feasible(h_best, n_best)
        c5 = evaluate_c5(h_best)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_best.copy()
    
    if elapsed() < budget_s * 0.92:
        h_best = slsqp_polish(best_h, n_best, 0.002, maxiter=100)
        c5 = evaluate_c5(h_best)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_best.copy()
    
    if elapsed() < budget_s * 0.97:
        for _ in range(3):
            if elapsed() > budget_s * 0.98:
                break
            pert = best_h + rng.randn(n_best) * 0.02
            pert = project_feasible(pert, n_best)
            h_tmp = lbfgs_smooth(pert, n_best, 0.002, maxiter=100)
            h_tmp = project_feasible(h_tmp, n_best)
            c5 = evaluate_c5(h_tmp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tmp.copy()
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)
