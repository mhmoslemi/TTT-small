    """Compute subgradient of C5 at h (using active constraints)."""
    n = len(h)
    lags = compute_all_lags(h)
    c5 = np.max(lags)
    # Active lags: those within a small tolerance of the max
    tol = 0.01
    active = np.where(lags >= c5 - tol)[0]
    
    # For each active lag, compute gradient
    # Lag index k in correlate corresponds to offset k - (n-1)
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual lag
        if k >= 0:
            # f_k(h) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j] = (1-h[j+k]) if 0<=j<=n-1-k, minus h[j-k] if k<=j<=n-1
            for j in range(0, n - k):
                grad[j] += (1.0 - h[j + k])
            for j in range(k, n):
                grad[j] -= h[j - k]
        else:
            # k < 0, let k = -j, j > 0
            # c[-j] = sum_{m=0}^{n-1-j} h[m]*(1-h[m+j])
            # This is the same formula as for positive lag j
            for m in range(0, n + k):  # k is negative, so n+k = n-j
                grad[m] += (1.0 - h[m - k])  # h[m+j]
            for m in range(-k, n):
                grad[m] -= h[m + k]  # h[m-j]
    
    # Average over active constraints
    if len(active) > 0:
        grad /= len(active)
    
    return grad, c5

1. Start with x (the unconstrained minimizer or current point)

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    
    # Find lambda such that sum(clamp(h + lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    
    # Exact adjustment for floating point
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        # Distribute among non-bound points
        mask = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    
    return h_proj
```

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h + lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    max_c = np.max(c)
    exp_s = np.exp(tau * (c - max_c))
    w = exp_s / np.sum(exp_s)
    val = max_c + np.log(np.sum(exp_s)) / tau
    
    grad = np.zeros(n)
    for lag in range(n):
        wk = w[n - 1 + lag]
        if wk < 1e-10:
            continue
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
    for lag_abs in range(1, n):
        wk = w[n - 1 - lag_abs]
        if wk < 1e-10:
            continue
        grad[:n - lag_abs] += wk * (1.0 - h[lag_abs:])
        grad[lag_abs:] -= wk * h[:n - lag_abs]
    
    grad *= dx
    return val, grad


def projected_gradient_optimize(h, tau, lr_init, max_iter, time_limit):
    n = len(h)
    h = h.copy()
    lr = lr_init
    best_val = compute_c5_from_h(h)
    
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        val, grad = smooth_max_grad(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_val - 1e-13:

            best_val = c5_new
            lr = min(lr * 1.08, 0.5)
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
    
    return h, best_val


def slsqp_polish(h, n, maxiter=2000):
    c5_init = compute_c5_from_h(h)
    x0 = np.concatenate([h, [c5_init + 0.05]])
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h_v = x[:-1]
        t = x[-1]
        n_v = len(h_v)
        dx = 2.0 / n_v
        c = np.correlate(h_v, 1.0 - h_v, mode="full")
        return t - c * dx
    
    def eq_fun(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0
    
    def eq_jac(x):
        n_v = len(x) - 1
        j = np.zeros(len(x))
        j[:n_v] = 1.0
        return j
    
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.7), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h, c5_init

            break
        val, grad = smooth_max_grad(h, tau)
        step_dir = (beta * v + (1 - beta) * grad) if use_momentum else grad
        h_new = h - lr * step_dir
        h_new = project_feasible(h_new, n)
        val_new, _ = smooth_max_grad(h_new, tau)
        if val_new < best_smooth - 1e-14:
            if use_momentum:
                v = step_dir
            h = h_new
            best_smooth = val_new
            best_true = compute_c5_from_h(h_new)
            lr = min(lr * 1.03, 0.3)
            stagnant = 0
        else:
            stagnant += 1
            lr *= 0.65
            if use_momentum:
                v *= 0.5
            if lr < 1e-10 or stagnant > 50:
                break
    return h, best_true


def slsqp_polish(h, n, maxiter=2000):
    c5_init = compute_c5_from_h(h)

    def obj_func(x):
        return x[-1]

    def ineq_func(x):
        h_v = x[:-1]
        t = x[-1]
        n_v = len(h_v)
        dx_v = 2.0 / n_v
        c = np.correlate(h_v, 1.0 - h_v, mode="full")
        return t - c * dx_v

    def eq_func(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

    def eq_jac_func(x):
        n_v = len(x) - 1
        j = np.zeros(len(x))
        j[:n_v] = 1.0
        return j

    x0 = np.concatenate([h, [c5_init + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h, c5_init


def low_res_search(n_low, rng, time_limit):
    best_h = None
    best_c5 = 1.0
    for attempt in range(30):
        if time.time() > time_limit:
            break
        h_rand = rng.random(n_low)
        h_rand = project_feasible(h_rand, n_low)
        time_sub = min(time_limit, time.time() + 5)
        h_cur, c5_cur = projected_gradient_optimize(h_rand, 30, 0.02, 200, time_sub)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    if best_h is None:
        best_h = np.full(n_low, 0.5)
        best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5


def upscale(h_from, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    f_interp = interp1d(x_old, h_from, kind='linear', fill_value="extrapolate")
    h_new = np.clip(f_interp(x_new), 0.0, 1.0)
    h_new = project_feasible(h_new, n_to)
    return h_new


def downscale(h_from, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    f_interp = interp1d(x_old, h_from, kind='linear', fill_value="extrapolate")
    h_new = np.clip(f_interp(x_new), 0.0, 1.0)
    h_new = project_feasible(h_new, n_to)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        h0 = project_feasible(h0, n)
    else:
        n = 135
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    time_limit = t0 + budget_s

    # Phase 1: Projected gradient with tau annealing on current resolution
    tau_schedule = [5, 10, 20, 50, 100, 200, 400]
    for tau in tau_schedule:
        if time.time() > time_limit - 180:
            break
        time_sub = min(time_limit - 180, time.time() + 80)
        h_cur, c5_cur = projected_gradient_optimize(
            best_h, tau, 0.01, 500, time_sub, use_momentum=(tau >= 50))
        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur

    # Phase 2: SLSQP polish
    if time.time() < time_limit - 120:
        h_new, c5_new = slsqp_polish(best_h, n_best, maxiter=3000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Low-resolution exploration and transfer
    if time.time() < time_limit - 150:
        for n_low in [24, 36, 48]:
            if time.time() > time_limit - 150:
                break
            h_low, c5_low = low_res_search(n_low, rng, min(time_limit - 150, time.time() + 30))
            h_up = upscale(h_low, n_low, n_best)
            time_sub = min(time_limit - 150, time.time() + 40)
            h_cur, c5_cur = projected_gradient_optimize(h_up, 30, 0.01, 300, time_sub)
            if c5_cur < best_c5:
                best_h = h_cur
                best_c5 = c5_cur

    # Phase 4: Perturbation + re-optimization
    n_perturb = 25
    for i in range(n_perturb):
        if time.time() > time_limit - 80:
            break
        scale = 0.04 * (1.0 - i / max(1, n_perturb)) + 0.003
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        time_sub = min(time_limit - 80, time.time() + 25)
        h_cur, c5_cur = projected_gradient_optimize(h_pert, 50, 0.005, 200, time_sub)
        if c5_cur < best_c5 and time.time() < time_limit - 80:
            h_new, c5_new = slsqp_polish(h_cur, n_best, maxiter=1000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        elif c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur

    # Phase 5: Multi-resolution transfer (upscale then optimize)
    for n_try in [180, 270, 360]:
        if time.time() > time_limit - 120:
            break
        h_up = upscale(best_h, n_best, n_try)
        time_sub = min(time_limit - 120, time.time() + 50)
        h_cur, c5_cur = projected_gradient_optimize(h_up, 20, 0.008, 300, time_sub)
        if c5_cur < best_c5 and time.time() < time_limit - 120:
            h_new, c5_new = slsqp_polish(h_cur, n_try, maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n_best = n_try

    # Phase 6: Downscale-upscale cycle to escape local minima
    for n_ds in [60, 90]:
        if time.time() > time_limit - 90:
            break
        h_ds = downscale(best_h, n_best, n_ds)
        h_up = upscale(h_ds, n_ds, n_best)
        time_sub = min(time_limit - 90, time.time() + 30)
        h_cur, c5_cur = projected_gradient_optimize(h_up, 80, 0.01, 300, time_sub)
        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur

    # Phase 7: Final SLSQP polish with tight tolerance
    if time.time() < time_limit - 20:
        h_new, c5_new = slsqp_polish(best_h, n_best, maxiter=3000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 8: One more round of small perturbations
    for i in range(10):
        if time.time() > time_limit - 10:
            break
        h_pert = best_h + 0.01 * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        time_sub = min(time_limit - 10, time.time() + 10)
        h_cur, c5_cur = projected_gradient_optimize(h_pert, 100, 0.003, 100, time_sub)
        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
