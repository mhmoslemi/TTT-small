        h_new = pgd_step(h, n, lr, tau, k_top, rng)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < old_c5 - 1e-12:
            h = h_new
            no_improve = 0
            lr = min(lr * 1.03, 0.02)
        else:
            no_improve += 1
            lr *= 0.6
            if no_improve > 30 or lr < 1e-7:
                h = project_feasible(h + rng.randn(n) * 0.004, n)
                no_improve = 0
                lr = 0.005
    return h

def slsqp_refine(h, n, tau, k_top, bounds, maxiter):
    def obj_func(h_in):
        return smooth_max_obj(h_in, n, tau, k_top)

    def jac_func(h_in):
        return smooth_max_grad(h_in, n, tau, k_top)

    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    try:
        res = minimize(
            obj_func, h,
            jac=jac_func,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h

def generate_diverse_start(n, parent, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)
    starts = []
    if parent is not None:
        starts.append(parent)
    for freq in [1.2, 1.8, 2.5, 3.1, 3.7, 4.3]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.15, 0.35)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase), 0, 1))
    for _ in range(3):
        starts.append(rng.rand(n))
    saw = np.mod(ar.astype(float) / n * 6.0, 2.0)
    saw = np.where(saw < 1.0, saw, 2.0 - saw)
    starts.append(np.clip(saw * 0.8 + 0.1, 0, 1))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        best_h = project_feasible(parent_h.copy(), len(parent_h))
        best_c5 = evaluate_c5(best_h)
        current_h = best_h.copy()
    else:
        n_init = 60
        current_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_h = current_h.copy()
        best_c5 = evaluate_c5(best_h)

    n0 = len(current_h)
    target_resolutions = sorted(set([n0, n0 + 10, n0 + 20, int(n0 * 1.5), int(n0 * 2), int(n0 * 2.5)]))
    target_resolutions = [r for r in target_resolutions if r <= 250]

    for n_target in target_resolutions:
        if time.time() - start_time > budget_s * 0.82:
            break

        if len(current_h) != n_target:
            current_h = interp_up(current_h, n_target)
            current_h = project_feasible(current_h, n_target)

        n = n_target
        bounds = [(0.0, 1.0)] * n

        tau_schedule = [0.15, 0.10, 0.06, 0.035, 0.02, 0.01, 0.005, 0.002, 0.001]
        k_schedule = [None, None, 50, 30, 20, 12, 8, 5, 3]
        iters_schedule = [60, 60, 80, 80, 100, 100, 120, 150, 200]

        for i in range(len(tau_schedule)):
            if time.time() - start_time > budget_s * 0.82:
                break
            tau = tau_schedule[i]
            k_top = k_schedule[i]
            maxit = iters_schedule[i]
            current_h = slsqp_refine(current_h, n, tau, k_top, bounds, maxit)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()

        if time.time() - start_time < budget_s * 0.78:
            time_left = max(5.0, (budget_s * 0.80 - (time.time() - start_time)))
            current_h = pgd_refine(current_h, n, rng, time_left)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()

    if best_h is None:
        n = 60
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    if time.time() - start_time < budget_s * 0.88:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        for tau in [0.0008, 0.0004, 0.0002]:
            if time.time() - start_time > budget_s * 0.95:
                break
            for attempt in range(6):
                if time.time() - start_time > budget_s * 0.95:
                    break
                noise_scale = 0.002 + 0.002 * attempt
                h_start = project_feasible(best_h + rng.randn(n) * noise_scale, n)
                h_pol = slsqp_refine(h_start, n, tau, 3, bounds, 250)
                c5 = evaluate_c5(h_pol)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_pol.copy()

    if time.time() - start_time < budget_s * 0.90:
        n = len(best_h)
        time_left = max(3.0, (budget_s * 0.93 - (time.time() - start_time)))
        h_pgd = pgd_refine(best_h, n, rng, time_left)
        c5 = evaluate_c5(h_pgd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pgd

    if best_h is None:
        n = 60
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
