```python
    max_c = np.max(c)

    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / beta
    # Penalty for integral constraint
    sum_pen = mu * (np.sum(h) - n / 2.0) ** 2
    return log_sum + sum_pen

def smooth_gradient(h, beta, mu, n):

    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    # Compute gradient of softmax approx
    # grad[j] = sum_k w_k * dc[k]/dh[j] * dx
    # dc[k]/dh[j] where k = L + (n-1), L is the lag
    # dc/dh[j] = (1-h[j+L])*[valid] + (-h[j-L])*[valid]

    for j in range(n):
        g = 0.0
        for L in range(-(n-1), n):
            k = L + (n - 1)
            w = weights[k]
            # First term: h[j] appears as first factor when j is in range and j+L is in range
            if 0 <= j < n and 0 <= j + L < n:

            # Second term: h[j] appears as second factor when j-L is in range
            if 0 <= j - L < n:
                g += w * (-h[j - L])
        grad[j] = g * dx
    
    # Add gradient of penalty
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    
    return grad
```

```python
def smooth_grad(x):
    n = _g_n
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    max_c = np.max(c)
    shifted = _g_beta * (c - max_c)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    # Vectorized computation
    # For term 1: grad1[j] = sum_i weights[i] * (1-x[j-L_i]) * [0<=j-L_i<n]
    # where L_i = i - (n-1)
    
    # Build index matrix
    j_idx = np.arange(n)[:, None]  # (n, 1)
    i_idx = np.arange(2*n-1)  # (2n-1,)
    L_vals = i_idx - (n-1)  # (2n-1,)
    
    # term1: j - L = j - (i - (n-1)) = j - i + n - 1
    idx1 = j_idx - (i_idx - (n-1))  # (n, 2n-1)
    mask1 = (idx1 >= 0) & (idx1 < n)  # (n, 2n-1)
    # term1[j] = sum_i weights[i] * (1-x[idx1[j,i]]) * mask1[j,i]
    # We need (1-x[idx1]) but only where mask is True
    idx1_clipped = np.clip(idx1, 0, n-1)
    term1 = np.sum(mask1 * weights[None, :] * (1.0 - x[idx1_clipped]), axis=1)
    
    # term2: j + L = j + i - (n-1)
    idx2 = j_idx + (i_idx - (n-1))  # (n, 2n-1)
    mask2 = (idx2 >= 0) & (idx2 < n)  # (n, 2n-1)
    idx2_clipped = np.clip(idx2, 0, n-1)
    term2 = np.sum(mask2 * weights[None, :] * x[idx2_clipped], axis=1)
    
    grad = dx * (term1 - term2)
    grad += 2.0 * _g_mu * (np.sum(x) - n / 2.0)
    return grad
```
