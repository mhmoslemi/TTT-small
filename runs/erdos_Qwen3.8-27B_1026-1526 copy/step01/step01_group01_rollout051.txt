from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask]).max() if cnt > 0 else 0
            # Distribute evenly
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

        h_clip = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h_clip, 1.0 - h_clip, mode="full")
        sm = smooth_max(corr, beta)
        penalty = mu * (np.sum(h_clip) - n / 2.0) ** 2
        return sm * (2.0 / n) + penalty
    return obj

def lbfgs_optimize(h0, n, beta_start, beta_end, mu_start, mu_end, 
                    max_iter, time_limit, rng):

    h = h0.copy()
    n_steps = max_iter
    best_h = h.copy()
    best_val = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    
    # Anneal beta and mu
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        mu = mu_start * (mu_end / mu_start) ** frac
        
        h = project_feasible(h, n)
        res = minimize(make_objective(n, beta, mu), h, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 30, 'gtol': 1e-6})
        h = project_feasible(res.x, n)
        val = compute_c5(h)
        if val < best_val:

            best_h = h.copy()
        
        if step % 10 == 0 and time.time() > time_limit:

    return best_h, best_val

        h_resized = h[:ratio * n_new].reshape(n_new, ratio).mean(axis=1)

    return project_feasible(h_new, n_new)

def binary_c5(h):
    """Compute C5 for a binary (0-1) array. Faster since no floating point issues."""
    n = len(h)
    dx = 2.0 / n
    # For binary h, corr[n-1+k] = number of i with h[i]=0 and h[i+k]=1
    # We can compute this efficiently
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def binary_local_search(h, n, max_flips, rng, time_limit):
    """Local search: try flipping pairs of bits (one 1->0 and one 0->1) to maintain sum."""
    best_h = h.copy()
    best_c5 = binary_c5(best_h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(max_flips):
        if time.time() > time_limit:
            break
        improved = False
        # Try random swaps
        for _ in range(50):
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            if best_h[ones[i]] != 1 or best_h[zeros[j]] != 0:
                continue
            # Swap
            best_h[ones[i]] = 0
            best_h[zeros[j]] = 1
            new_c5 = binary_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                improved = True
                # Update lists
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
            else:
                # Undo
                best_h[ones[i]] = 1
                best_h[zeros[j]] = 0
        
        if not improved:
            # Try more aggressive: flip 2 pairs
            for _ in range(20):
                i1 = rng.randint(len(ones))
                i2 = (i1 + 1) % len(ones)
                j1 = rng.randint(len(zeros))
                j2 = (j1 + 1) % len(zeros)
                if len(ones) < 2 or len(zeros) < 2:
                    break
                if best_h[ones[i1]] != 1 or best_h[ones[i2]] != 1:
                    continue
                if best_h[zeros[j1]] != 0 or best_h[zeros[j2]] != 0:
                    continue
                best_h[ones[i1]] = 0
                best_h[ones[i2]] = 0
                best_h[zeros[j1]] = 1
                best_h[zeros[j2]] = 1
                new_c5 = binary_c5(best_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
                    ones = np.where(best_h == 1)[0]
                    zeros = np.where(best_h == 0)[0]
                else:
                    best_h[ones[i1]] = 1
                    best_h[ones[i2]] = 1
                    best_h[zeros[j1]] = 0
                    best_h[zeros[j2]] = 0
            
            if not improved:
                break
    
    return best_h, best_c5

from scipy.optimize import minimize, fftpack
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_fft(h):
    """Faster C5 computation using FFT for binary arrays."""
    n = len(h)
    dx = 2.0 / n
    if len(h) > 0 and np.all((h == 0) | (h == 1)):
        # Binary case: use FFT for autocorrelation
        N = 2 * n
        H = fftpack.rfft(h.astype(float), N)
        autocorr = fftpack.irfft(H * np.conj(H), N)[:n]
        # corr(n-1+k) = sum(h[k:]) - autocorr[k]
        suffix_sums = np.cumsum(h[::-1])[::-1]  # suffix_sums[k] = sum(h[k:])

  Original: h[p]=1, h[q]=0
  After swap: h[p]=0, h[q]=1
  
  autocorr(k) = sum_i h[i+k]*h[i]
  
  Change in autocorr(k) from changing h[p]: 
  - In the sum, h[p] appears as h[i] when i=p (contribution h[p+k]*h[p]) and as h[i+k] when i=p-k (contribution h[p]*h[p-k])
  - Changing h[p] from 1 to 0:
    - Removes h[p+k]*1 from term i=p: change = -h[p+k]
    - Removes 1*h[p-k] from term i=p-k: change = -h[p-k]
  - Total change from h[p]: -(h[p+k] + h[p-k])
  
  Change in autocorr(k) from changing h[q]:
  - Changing h[q] from 0 to 1:
    - Adds h[q+k]*1 to term i=q: change = +h[q+k]
    - Adds 1*h[q-k] to term i=q-k: change = +h[q-k]
  - Total change from h[q]: +(h[q+k] + h[q-k])
  
  Total change in autocorr(k): h[q+k] + h[q-k] - h[p+k] - h[p-k]

And change in sum(h[k:n]):
- If p >= k: sum decreases by 1 (h[p] was 1, now 0)
- If q >= k: sum increases by 1 (h[q] was 0, now 1)
- Net change: (1 if q >= k else 0) - (1 if p >= k else 0)

So the change in corr(n-1+k) = change in sum(h[k:n]) - change in autocorr(k)

This is O(1) per lag! So the total change in C₅ (which is the max) can be computed in O(n) per swap (checking all lags). But we only care about whether the max decreases, so we can early-terminate.
