from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_first = (first_idx >= 0) & (first_idx < n)
    valid_second = (second_idx >= 0) & (second_idx < n)
    first_idx_c = np.clip(first_idx, 0, n-1)
    second_idx_c = np.clip(second_idx, 0, n-1)
    contrib_f = valid_first * (1.0 - h[first_idx_c]) * w[None, :]
    contrib_s = valid_second * h[second_idx_c] * w[None, :]

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n-1)
    sc = np.clip(second_idx, 0, n-1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def parametric_h(n, params, rng):
    x = np.linspace(0, 2.0, n, endpoint=False)
    a, f1, phi1, b, f2, phi2 = params
    h = 0.5 + a * np.sin(2.0 * np.pi * f1 * x / 2.0 + phi1) + b * np.sin(2.0 * np.pi * f2 * x / 2.0 + phi2)
    return project_to_sum(h, n / 2.0)

def make_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    starts.append(np.full(n, 0.5))
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7]:
        s = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(s, target_sum))
    for nblks in [2, 3, 4, 5, 6, 7, 8]:
        blk = n // nblks
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    for _ in range(4):
        a = rng.uniform(0.1, 0.45)
        b = rng.uniform(-0.3, 0.3)
        f1 = rng.choice([1, 2, 3, 4, 5])
        f2 = rng.choice([2, 3, 4, 5, 6, 7, 8])
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        starts.append(parametric_h(n, (a, f1, p1, b, f2, p2), rng))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), target_sum))
    hb = (rng.rand(n) > 0.5).astype(float)
    attempts = 0
    while abs(hb.sum() - target_sum) > 1e-10 and attempts < 10000:
        if hb.sum() > target_sum:
            idx = rng.choice(np.where(hb > 0)[0])
            hb[idx] = 0.0
        else:
            idx = rng.choice(np.where(hb < 1)[0])
            hb[idx] = 1.0
        attempts += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())
    return starts

def binary_flip_search(h, n, target_sum, rng, max_rounds=50):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_trials = min(200, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-10:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        c5_cur = best_c5
        improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Continuation on parent solution
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [3, 8, 20, 50, 150, 500, 1500]:
        if time_left() < 25:
            break
        h_new = slsqp_local(best_h, n1, ts1, alpha, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n1

    # Phase 2: Multi-start at multiple resolutions
    resolutions = [20, 30, 40, 50, 60, 72, 90, 120, 144, 180, 200, 240]
    for n_tr in resolutions:
        if time_left() < 80:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        starts = make_starts(n_tr, ts_tr, rng, best_h, best_n)
        for h_start in starts:
            if time_left() < 25:
                break
            h_start = project_to_sum(h_start, ts_tr)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_tr
            for alpha in [10, 50, 300]:
                if time_left() < 15:
                    break
                h_opt = slsqp_local(h_start, n_tr, ts_tr, alpha, maxiter=600)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                h_start = h_opt

    # Phase 3: Shake and re-optimize
    n_best = best_n
    ts_best = n_best / 2.0
    for trial in range(30):
        if time_left() < 30:
            break
        noise = 0.15 * (1.0 - trial / 30.0) + 0.03
        h_shake = best_h + rng.uniform(-noise, noise, best_n)
        h_shake = project_to_sum(h_shake, ts_best)
        for alpha in [30, 150, 800]:
            if time_left() < 20:
                break
            h_opt = slsqp_local(h_shake, n_best, ts_best, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_best
            h_shake = h_opt

    # Phase 4: Binary flip search on best solution
    if time_left() > 60:
        h_bin = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin.sum() - ts_best) > 1e-10 and attempts < 5000:
            if h_bin.sum() > ts_best:
                idx = rng.choice(np.where(h_bin > 0)[0])
                h_bin[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin < 1)[0])
                h_bin[idx] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts_best) < 1e-10:
            h_bf, c5_bf = binary_flip_search(h_bin, best_n, ts_best, rng, max_rounds=80)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
            h_smooth = slsqp_local(h_bf, best_n, ts_best, 200, maxiter=800)
            c5_sm = compute_c5(h_smooth)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_smooth.copy()

    # Phase 5: Final high-alpha polish
    if time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [1000, 5000, 20000]:
            if time_left() < 25:
                break
            h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
