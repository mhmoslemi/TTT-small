    """One step of projected subgradient descent with momentum."""
    dx = 2.0 / n
    # Compute correlation to find which k achieves max
    c = np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using water-filling."""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h-lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        flat = np.where((h_proj > 1e-12) & (h_proj < 1.0 - 1e-12))[0]
        if len(flat) > 0:
            h_proj[flat[0]] += diff
    h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_lbfgsb(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    c = corr_values(h)
    val = smooth_max(c, beta) + mu * (np.sum(h) - n/2.0)**2
    return val

def subgradient_of_corr(h, k_idx):
    """Compute gradient of corr[k_idx] w.r.t. h (unmultiplied by dx)."""
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # c = sum_{i=lag}^{n-1} h[i] * (1-h[i-lag])
        for j in range(n):
            g = 0.0
            if j >= lag:
                g += 1.0 - h[j - lag]
            if j + lag < n:
                g -= h[j + lag]
            grad[j] = g

        for j in range(n):
            g = 0.0
            if j < n - lp:
                g += 1.0 - h[j + lp]
            if j >= lp:
                g -= h[j - lp]
            grad[j] = g

    k_max = np.argmax(c)

    h_new = h + mom_buf
    h_new = project_feasible(h_new, n)
    return h_new, mom_buf, c[k_max]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        flat = np.where((h_p > 1e-10) & (h_p < 1.0 - 1e-10))[0]
        if len(flat) > 0:
            h_p[flat[0]] += diff
    return np.clip(h_p, 0.0, 1.0)


def smooth_max(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(sh))) / beta


def lbfgsb_obj(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    c = corr_values(h)
    return smooth_max(c, beta) + mu * (np.sum(h) - n / 2.0) ** 2


def lbfgsb_grad(x, n, beta, mu):
    h = np.clip(x, 0.0, 1.0)
    c = corr_values(h)
    sh = np.clip(beta * (c - np.max(c)), -500.0, 500.0)
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    for ki in range(len(c)):
        if w[ki] < 1e-12:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            jlo, jhi = lag, n
            valid_j = np.arange(jlo, jhi)
            grad[valid_j] += w[ki] * (1.0 - h[valid_j - lag])
            if lag > 0:
                valid_j2 = np.arange(0, n - lag)
                grad[valid_j2] += w[ki] * (-h[valid_j2 + lag])
        else:
            lp = -lag
            valid_j = np.arange(0, n - lp)
            grad[valid_j] += w[ki] * (1.0 - h[valid_j + lp])
            valid_j2 = np.arange(lp, n)
            grad[valid_j2] += w[ki] * (-h[valid_j2 - lp])
    grad *= (2.0 / n)
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad


def run_lbfgsb(h0, n, beta, mu, max_iter, max_time):
    t_start = time.time()
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lbfgsb_obj, h0, jac=lbfgsb_grad, method='L-BFGS-B',
                       bounds=bounds, args=(n, beta, mu),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8,
                                'maxfun': 5000, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def subgradient_step(h, n, step_size, mom_buf, momentum, top_frac):
    c = corr_values(h)
    nk = len(c)
    top_k = max(3, int(top_frac * nk))
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:top_k]
    weights = np.exp((c[top_idx] - c[top_idx[0]]) * 30.0)
    weights /= weights.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for t in range(top_k):
        ki = int(top_idx[t])
        wt = weights[t]
        lag = ki - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                grad[j] += wt * (1.0 - h[j - lag])
            for j in range(0, n - lag):
                grad[j] -= wt * h[j + lag]
        else:
            lp = -lag
            for j in range(0, n - lp):
                grad[j] += wt * (1.0 - h[j + lp])
            for j in range(lp, n):
                grad[j] -= wt * h[j - lp]
    grad *= dx
    mom_buf = momentum * mom_buf + step_size * grad
    h_new = h - mom_buf
    h_new = project_feasible(h_new, n)
    return h_new, mom_buf, c[np.argmax(c)]


def freq_perturbation(h, n, rng, n_freqs, amplitude):
    h_p = h.copy()
    target_energy = amplitude * amplitude * n / 4.0
    for _ in range(n_freqs):
        f = rng.randint(1, n // 2 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = amplitude / np.sqrt(n_freqs)
        idx = np.arange(n)
        h_p += amp * np.sin(2 * np.pi * f * idx / n + phase)
    return project_feasible(h_p, n)


def block_perturbation(h, n, rng, n_blocks, amplitude):
    h_p = h.copy()
    block_size = max(3, n // n_blocks)
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        sign = 1.0 if rng.random() < 0.5 else -1.0
        amp = amplitude * rng.uniform(0.5, 1.5)
        h_p[start:start + block_size] += sign * amp
    return project_feasible(h_p, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)
