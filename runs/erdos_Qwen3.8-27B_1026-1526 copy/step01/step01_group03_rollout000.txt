import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    if vmax > 50:
        return float(vmax) * tau
    exp_sum = np.sum(np.exp(v - vmax))
    return float(tau * (vmax + np.log(exp_sum)))


def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result


def compute_corr(h):
    g = 1.0 - h
    return np.correlate(h, g, mode="full")


def gradient_at_lag(h, lag_idx):
    n = len(h)
    shift = lag_idx - (n - 1)
    grad = np.zeros(n)
    j_range = np.arange(n)
    i_first = j_range - shift
    valid_first = (i_first >= 0) & (i_first < n)
    grad[valid_first] += (1.0 - h[i_first[valid_first]])
    j_second = j_range + shift
    valid_second = (j_second >= 0) & (j_second < n)
    grad[valid_second] -= h[j_second[valid_second]]
    return grad


def obj_func(h, n, tau):
    corr = compute_corr(h)
    return smooth_max(corr, tau) * (2.0 / n)


def constraint_func(h, n):
    return np.sum(h) - n / 2.0


def subgradient_optimize(h, n, n_iter, rng, time_limit_fn):
    step_size = 0.03
    for it in range(n_iter):
        if time_limit_fn():
            break
        corr = compute_corr(h)
        top_k = min(4, len(corr))
        idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for li in idx:
            grad += gradient_at_lag(h, li)
        grad /= float(top_k)
        gn = np.linalg.norm(grad)
        if gn > 1e-12:
            grad = grad / gn
        h = project_feasible(h - step_size * grad, n)
        step_size *= 0.99
    return h


def generate_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)

    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
        else:
            x_p = np.linspace(0, 2, len(parent_h), endpoint=False)
            f = interp1d(x_p, parent_h, kind='cubic', fill_value='extrapolate')
            starts.append(f(x).clip(0, 1))
        hp = parent_h + rng.randn(len(parent_h)) * 0.04
        hp = hp.clip(0, 1)
        if len(parent_h) == n:
            starts.append(hp.copy())
        else:
            x_p = np.linspace(0, 2, len(parent_h), endpoint=False)
            f2 = interp1d(x_p, hp, kind='cubic', fill_value='extrapolate')
            starts.append(f2(x).clip(0, 1))

    for _ in range(3):
        starts.append(rng.rand(n))

    for freq in [1.3, 1.8, 2.3, 2.8, 3.4, 4.1]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.28, 0.48)
        starts.append((0.5 + amp * np.sin(2 * np.pi * freq * x + phase)).clip(0, 1))

    for freq in [5.0, 6.0, 7.3, 8.7]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.25, 0.45)
        starts.append((0.5 + amp * np.cos(2 * np.pi * freq * x + phase)).clip(0, 1))

    starts.append(np.linspace(0.03, 0.97, n))
    starts.append(np.linspace(0.97, 0.03, n))

    c1, c2 = 0.28 * n, 0.72 * n
    w = 0.14 * n
    b1 = np.exp(-0.5 * ((ar - c1) / w) ** 2)
    b2 = np.exp(-0.5 * ((ar - c2) / w) ** 2)
    starts.append((0.12 + 0.78 * 0.5 * (b1 + b2)).clip(0, 1))

    golden = 0.6180339887498949
    h_golden = np.array([1.0 if (np.floor((i + 1) * golden) - np.floor(i * golden)) > 0.45 else 0.15 for i in range(n)])
    starts.append(h_golden)

    h_saw = np.mod(ar / n * 3.0, 2.0)
    h_saw = np.where(h_saw < 1.0, h_saw, 2.0 - h_saw)
    starts.append(h_saw.clip(0, 1))

    h_bin = (rng.rand(n) > 0.5).astype(float)
    starts.append((h_bin * 0.82 + 0.09).clip(0, 1))

    return [project_feasible(s, n) for s in starts]


def optimize_start(h0, n, bounds, cons, rng, time_limit_fn):
    current_h = h0.copy()

    if not time_limit_fn():
        current_h = subgradient_optimize(current_h, n, 40, rng, time_limit_fn)

    for tau in [0.05, 0.02, 0.008, 0.003]:
        if time_limit_fn():
            break
        try:
            res = minimize(
                obj_func, current_h,
                method='SLSQP',
                args=(n, tau),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-13}
            )
            current_h = project_feasible(res.x, n)
        except Exception:
            pass

    return current_h

    np.random.seed(seed)

    start_time = time.time()

    best_h = None
