def adam_step(params, grads, m, v, beta1, beta2, lr, t, eps=1e-8):
    """Single Adam step, returns updated params, m, v."""
    m = beta1 * m + (1 - beta1) * grads
    v = beta2 * v + (1 - beta2) * grads ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    params = params - lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m, v

def compute_gradient(h, n, alpha):

    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)  # for numerical stability

    smax = m + np.log(np.sum(exp_vals)) / alpha
    dx = 2.0 / n
    
    grad = np.zeros(n)
    # For each lag s (from -(n-1) to n-1), k = s + (n-1)
    # s >= 0: d(corr)/dh[j] = [s<=j]<=n-1]*(1-h[j-s]) - [0<=j<=n-1-s]*h[j+s]
    # s < 0: t = -s
    #   d(corr)/dh[j] = [0<=j<=n-1-t]*(1-h[j+t]) - [t<=j<=n-1]*h[j-t]
    
    for s in range(-(n-1), n):
        wk = w[s + n - 1]
        if wk < 1e-15:
            continue
        if s >= 0:
            # Term 1: +[s<=j<=n-1]*(1-h[j-s])
            if s < n:
                j_range = range(s, n)
                grad[s:n] += wk * (1.0 - h[:n-s])
            # Term 2: -[0<=j<=n-1-s]*h[j+s]

                grad[:n-s] -= wk * h[s:n]
        else:

            # Term 1: +[0<=j<=n-1-t]*(1-h[j+t])

            # Term 2: -[t<=j<=n-1]*h[j-t]
            if t < n:
                grad[t:n] -= wk * h[:n-t]
    
    return smax * dx, grad * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)

import numpy as np
from scipy.ndimage import uniform_filter1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    smax = (m + np.log(np.sum(exp_vals)) / alpha) * dx
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        wk = w[s + n - 1] * dx
        if wk < 1e-16:
            continue
        if s >= 0:
            if s < n:
                grad[s:] += wk * (1.0 - h[:n - s])
            if n - s > 0:
                grad[:n - s] -= wk * h[s:]
        else:
            t = -s
            if n - t > 0:
                grad[:n - t] += wk * (1.0 - h[t:])
            if t < n:
                grad[t:] -= wk * h[:n - t]
    return smax, grad

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return project_to_sum(np.interp(x_dst, x_src, h_src), n_dst / 2.0)

def adam_optimize(h0, n, ts, alpha_schedule, iters_per_alpha, base_lr, t_init=0):
    h = h0.copy()
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = t_init
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    for alpha in alpha_schedule:
        lr = base_lr / np.sqrt(alpha)
        for _ in range(iters_per_alpha):
            t += 1
            _, grad = obj_and_grad(h, alpha)
            mv = 0.9 * mv + 0.1 * grad
            vv = 0.999 * vv + 0.001 * grad * grad
            mhat = mv / (1.0 - 0.9 ** t)
            vhat = vv / (1.0 - 0.999 ** t)
            h = h - lr * mhat / (np.sqrt(vhat) + 1e-8)
            h = project_to_sum(h, ts)
            c5 = compute_c5(h)
            if c5 < best_c5_local:
                best_c5_local = c5
                best_local = h.copy()
    return best_local, best_c5_local, t

def momentum_optimize(h0, n, ts, alpha, iters, lr, momentum):
    h = h0.copy()
    h_prev = h.copy()
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    for _ in range(iters):
        _, grad = obj_and_grad(h, alpha)
        h_new = h - lr * grad + momentum * (h - h_prev)
        h_prev = h
        h = project_to_sum(h_new, ts)
        c5 = compute_c5(h)
        if c5 < best_c5_local:
            best_c5_local = c5
            best_local = h.copy()
    return best_local, best_c5_local

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            initial_h = np.array(initial_h_values, dtype=float)
            n_init = len(initial_h)
            ts = n_init / 2.0
            initial_h = project_to_sum(initial_h, ts)
            best_h = initial_h.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_init
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_list = [120, 180, 240, 300, 360, 420, 480, 540, 600]

    for n in n_list:
        if time_left() < 70:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        starts = []
        starts.append(interp_h(best_h, n))
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb.copy())
        starts.append(project_to_sum(uniform_filter1d(hb, max(3, n // 30)), ts))
        xv = np.linspace(0, 2, n, endpoint=False)
        for fmax in [2, 4, 6]:
            hs = np.full(n, 0.5)
            for f in range(1, fmax + 1):
                amp = rng.uniform(0.1, 0.3) / f
                hs += amp * np.sin(2 * np.pi * f * xv + rng.uniform(0, 2 * np.pi))
            starts.append(project_to_sum(hs, ts))
        for period in [4, 6, 8, 12, 16, 20, 24, 30]:
            half = period // 2
            pat = np.tile(np.concatenate([np.ones(half), np.zeros(half)]), n // period + 1)[:n]
            starts.append(project_to_sum(pat, ts))
        starts.append(project_to_sum(rng.uniform(0, 1, n), ts))

        scored = []
        for hs in starts:
            hs = project_to_sum(hs, ts)
            c5s = compute_c5(hs)
            if c5s < best_c5:
                best_c5 = c5s
                best_h = hs.copy()
                best_n = n
            scored.append((c5s, hs))
        scored.sort(key=lambda x: x[0])

        n_opt = min(3, len(scored))
        for idx in range(n_opt):
            if time_left() < 50:
                break
            h_start = scored[idx][1].copy()
            rem = time_left()
            alphas = [5, 20, 80, 300, 1000]
            total_iters = min(4000, max(400, int(rem * 2.5)))
            ipa = total_iters // len(alphas)
            h_res, c5_res, _ = adam_optimize(h_start, n, ts, alphas, ipa, 0.01)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n
            if time_left() > 40:
                h_res2, c5_res2 = momentum_optimize(h_res, n, ts, 1000,
                                                     min(2000, max(200, int(time_left() * 1.5))),
                                                     0.003, 0.5)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2.copy()
                    best_n = n

    if best_h is not None and time_left() > 20:
        nfs = best_n
        tsf = nfs / 2.0
        for alpha in [500, 1500, 5000]:
            if time_left() < 12:
                break
            iters = min(3000, max(100, int(time_left() * 2)))
            h_r, c5_r, _ = adam_optimize(best_h, nfs, tsf, [alpha], iters, 0.005)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
        if time_left() > 15:
            h_r2, c5_r2 = momentum_optimize(best_h, nfs, tsf, 5000,
                                             min(2000, max(100, int(time_left() * 1.5))),
                                             0.002, 0.7)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
