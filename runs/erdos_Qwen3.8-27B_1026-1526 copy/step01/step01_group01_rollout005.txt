    while time_left() > 400 and no_improve < 40:
        trial += 1
        if trial <= 4:
            sigma = 0.06
        elif trial <= 8:
            sigma = 0.12
        elif trial <= 14:
            sigma = 0.2
        elif trial <= 20:
            sigma = 0.35
        else:
            sigma = 0.5

        h_start = best_h + rng.normal(0.0, sigma, best_n)
        h_start = project_feasible(h_start, best_n)

        h_new, c5_new = opt_lfgs(h_start, best_n, 70.0, 5e3, max_iter=300, gtol=1e-7)
        if h_new is not None and c5_new < best_c5:
            h_new2, c5_new2 = opt_slsqp(h_new, best_n, 180.0, max_iter=250)
            if h_new2 is not None and c5_new2 < c5_new:
                best_h, best_c5 = h_new2, c5_new2
            else:
                best_h, best_c5 = h_new, c5_new
            no_improve = 0
        else:
            no_improve += 1

    # Phase 4: Structured restarts with Fourier patterns
    if time_left() > 250:
        for freq in [3, 5, 8, 13]:
            if time_left() < 120:
                break
            h_struct = make_fourier_init(best_n, freq, 0.25, rng)
            h_new, c5_new = opt_lfgs(h_struct, best_n, 60.0, 5e3, max_iter=250)
            if h_new is not None and c5_new < best_c5:
                h_new2, c5_new2 = opt_slsqp(h_new, best_n, 150.0, max_iter=200)
                if h_new2 is not None and c5_new2 < c5_new:
                    best_h, best_c5 = h_new2, c5_new2
                else:
                    best_h, best_c5 = h_new, c5_new

    # Phase 5: Ramp-based restarts
    if time_left() > 150:
        h_ramp = make_ramp_init(best_n, rng)
        h_new, c5_new = opt_lfgs(h_ramp, best_n, 50.0, 5e3, max_iter=200)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Resolution upsample to n=284
    if time_left() > 250:
        h_up = interp_h(best_h, best_n, 284)
        h_up = project_feasible(h_up, 284)
        for beta in [40.0, 80.0]:
            if time_left() < 120:
                break
            h_new, c5_new = opt_lfgs(h_up, 284, beta, 5e3, max_iter=200, gtol=1e-7)
            if h_new is not None and c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, 284

    # Phase 7: Resolution upsample to n=380
    if time_left() > 200:
        h_up = interp_h(best_h, best_n, 380)
        h_up = project_feasible(h_up, 380)
        for beta in [35.0, 70.0]:
            if time_left() < 100:
                break
            h_new, c5_new = opt_lfgs(h_up, 380, beta, 5e3, max_iter=200, gtol=1e-7)
            if h_new is not None and c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, 380

    # Phase 8: Final SLSQP with very high beta
    if time_left() > 80:
        h_new, c5_new = opt_slsqp(best_h, best_n, 350.0, max_iter=400)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 9: Ultra-high beta L-BFGS for final tightening
    if time_left() > 40:
        h_new, c5_new = opt_lfgs(best_h, best_n, 600.0, 1e5, max_iter=300, gtol=1e-10)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final SLSQP if time permits
    if time_left() > 25:
        h_new, c5_new = opt_slsqp(best_h, best_n, 500.0, max_iter=200)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
