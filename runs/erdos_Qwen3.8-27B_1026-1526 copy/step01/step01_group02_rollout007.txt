_DE_N = 135

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x):
    return x[-1]

def _slsqp_ineq(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")

    return np.sum(x[:-1]) - n / 2.0

    jac[:n] = 1.0
    return jac

import numpy as np
from scipy.optimize import minimize, differential_evolution
from scipy.interpolate import interp1d
import time

_DE_N = 135
_ACTIVE_K = np.arange(20)


def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(diff / cnt, (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def _obj(x):
    return x[-1]


def _ineq_active(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return x[-1] - c[_ACTIVE_K] * dx


def _eq(x):
    n = len(x) - 1
    return np.sum(x[:-1]) - n / 2.0


def _eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def _de_obj(x):
    h = project_feasible(np.clip(x, 0.0, 1.0), _DE_N)
    return compute_c5(h)


def slsqp_active(h, n, outer_iters=40, inner_maxiter=300, top_k=20):
    global _ACTIVE_K
    h = project_feasible(h.copy(), n)
    c5_val = compute_c5(h)
    for outer in range(outer_iters):
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        top_idx = np.argsort(c)[-min(top_k, len(c)):]
        _ACTIVE_K = top_idx
        c5_now = float(c.max())
        x0 = np.concatenate([h, [c5_now + 0.002]])
        bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_now * 0.92), 1.0)]
        constraints = [
            {'type': 'ineq', 'fun': _ineq_active},
            {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
        ]
        result = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': inner_maxiter, 'ftol': 1e-14})
        if result.x is not None:
            h_new = project_feasible(result.x[:-1], n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_val - 1e-12:
                h = h_new
                c5_val = c5_new
            else:
                if c5_new < c5_val:
                    h = h_new
                    c5_val = c5_new
                break
        else:
            break
    return h, compute_c5(h)


def slsqp_full(h, n, maxiter=1000):
    c5_init = compute_c5(h)
    x0 = np.concatenate([h, [c5_init + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.95), 1.0)]
    dx = 2.0 / n
    c_all = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argsort(c_all)[-min(30, len(c_all)):]
    _ACTIVE_K_save = _ACTIVE_K.copy()
    _ACTIVE_K = top_idx
    constraints = [
        {'type': 'ineq', 'fun': _ineq_active},
        {'type': 'eq', 'fun': _eq, 'jac': _eq_jac}
    ]
    result = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    _ACTIVE_K = _ACTIVE_K_save
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h.copy(), c5_init


def upscale(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f_interp = interp1d(x_old, h, kind='linear', fill_value="extrapolate", assume_sorted=True)
    h_new = np.clip(f_interp(x_new), 0.0, 1.0)
    h_new = project_feasible(h_new, n_new)
    return h_new


def structured_starts(n, rng):
    starts = []
    h_step = np.zeros(n)
    h_step[:n // 2] = 1.0
    starts.append(project_feasible(h_step, n))
    h_two = np.zeros(n)
    h_two[:n // 4] = 1.0
    h_two[n // 2:3 * n // 4] = 1.0
    starts.append(project_feasible(h_two, n))
    h_noise = rng.random(n)
    h_noise = project_feasible(h_noise, n)
    starts.append(h_noise)
    h_smooth = np.linspace(0.8, 0.2, n)
    h_smooth = project_feasible(h_smooth, n)
    starts.append(h_smooth)
    h_osc = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * np.arange(n) / n)
    h_osc = project_feasible(h_osc, n)
    starts.append(h_osc)
    return starts
