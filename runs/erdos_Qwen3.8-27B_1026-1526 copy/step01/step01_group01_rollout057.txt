def compute_smooth_max_grad(h, n, beta):

    g = 1.0 - h
    corr = np.correlate(h, g, 'full') * dx
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    p = np.exp(np.clip(shifted, -500, 500))
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        p = np.zeros(2*n - 1)
        p[np.argmax(corr)] = 1.0
    else:
        p /= p_sum
    
    m_vals = np.arange(2*n - 1) - (n - 1)
    j_vals = np.arange(n)
    
    jm = j_vals[:, None] - m_vals[None, :]
    jp = j_vals[:, None] + m_vals[None, :]
    
    mask_jm = (jm >= 0) & (jm < n)
    mask_jp = (jp >= 0) & (jp < n)
    
    h_jm = np.where(mask_jm, h[np.clip(jm, 0, n-1)], 0.0)
    term1 = (1.0 - h_jm) * mask_jm
    
    h_jp = np.where(mask_jp, h[np.clip(jp, 0, n-1)], 0.0)
    term2 = h_jp * mask_jp
    
    grad = dx * (term1 - term2) @ p
    return grad

            best_c5 = c5
        
        h = h_new
        t_coeff = t_new
    
    return best_h, best_c5
