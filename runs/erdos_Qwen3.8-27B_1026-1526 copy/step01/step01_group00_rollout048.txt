                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_fn():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            parent = np.array(initial_h_values, dtype=float)
            n_p = len(parent)
            ts_p = n_p / 2.0
            if abs(ts_p - round(ts_p)) < 1e-6:
                ts_p = float(round(ts_p))
            parent = project_to_sum(parent, ts_p)
            parent = np.clip(parent, 0.0, 1.0)
            best_h = parent.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_p
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = float(int(round(n / 2.0)))

    # Phase 1: Projected gradient from parent with various k_top
    for k_top in [3, 5, 8, 12, 20]:
        if time_fn() < 40:
            break
        lr = 0.02 / np.sqrt(k_top)
        h_pg, c5_pg = proj_grad_momentum(best_h, n, ts, k_top, 600, lr, 0.9, time_fn)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 2: Binary swap search
    if time_fn() > 60:
        h_bin, c5_bin = binary_swap_search(best_h, n, ts, rng, 4000, 0.03)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Continuous refinement from best
    for k_top in [3, 5, 8]:
        if time_fn() < 30:
            break
        lr = 0.01 / np.sqrt(k_top)
        h_pg, c5_pg = proj_grad_momentum(best_h, n, ts, k_top, 400, lr, 0.92, time_fn)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg

    # Phase 4: Block perturbation + re-optimize (iterated local search)
    for iteration in range(5):
        if time_fn() < 80:
            break
        h_pert, c5_pert = block_perturbation(best_h, n, ts, rng, 50)
        for k_top in [5, 8]:
            if time_fn() < 30:
                break
            lr = 0.015 / np.sqrt(k_top)
            h_pg, c5_pg = proj_grad_momentum(h_pert, n, ts, k_top, 400, lr, 0.9, time_fn)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg

    # Phase 5: Multi-resolution exploration
    for n_try in [90, 120, 150, 200, 240, 270]:
        if time_fn() < 80:
            break
        ts_try = float(int(round(n_try / 2.0)))
        if ts_try != n_try / 2.0:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, ts_try)

        starts = [h_interp]
        starts.append(project_to_sum(rng.uniform(0.1, 0.9, n_try), ts_try))
        h_pat = np.zeros(n_try)
        per = max(2, n_try // 30)
        for pi in range(0, n_try, per * 2):
            h_pat[pi:min(pi + per, n_try)] = 1.0
        starts.append(project_to_sum(h_pat, ts_try))

        for h_s in starts:
            if time_fn() < 40:
                break
            for k_top in [5, 10]:
                if time_fn() < 25:
                    break
                lr = 0.02 / np.sqrt(k_top)
                h_pg, c5_pg = proj_grad_momentum(h_s, n_try, ts_try, k_top, 300, lr, 0.9, time_fn)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg
                    best_n = n_try
                    n = n_try
                    ts = ts_try
                    h_s = h_pg
            if time_fn() > 40:
                h_bin2, c5_bin2 = binary_swap_search(h_s, n_try, ts_try, rng, 2000, 0.02)
                if c5_bin2 < best_c5:
                    best_c5 = c5_bin2
                    best_h = h_bin2
                    best_n = n_try
                    n = n_try
                    ts = ts_try

    # Phase 6: If we found a better solution at different n, transfer back to original n
    if best_n != 180:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, 180, endpoint=False)
        h_trans = np.interp(x_new, x_old, best_h)
        h_trans = project_to_sum(h_trans, 90.0)
        c5_trans = compute_c5(h_trans)
        for k_top in [3, 5, 8]:
            if time_fn() < 20:
                break
            lr = 0.01 / np.sqrt(k_top)
            h_pg, c5_pg = proj_grad_momentum(h_trans, 180, 90.0, k_top, 300, lr, 0.9, time_fn)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg
                best_n = 180
                n = 180
                ts = 90.0

    # Phase 7: Final fine-tuning
    n = best_n
    ts = float(int(round(n / 2.0)))
    if time_fn() > 20:
        h_ft, c5_ft = fine_tune(best_h, n, ts, rng, 100, time_fn)
        if c5_ft < best_c5:
            best_c5 = c5_ft
            best_h = h_ft

    if time_fn() > 15:
        for k_top in [3, 5]:
            if time_fn() < 10:
                break
            lr = 0.005 / k_top
            h_pg, c5_pg = proj_grad_momentum(best_h, n, ts, k_top, 300, lr, 0.95, time_fn)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    ts_final = float(int(round(best_n / 2.0)))
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
