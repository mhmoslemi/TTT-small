    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        else:

            if cnt == 0:

        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def subgradient_minimax(h, n, iters, alpha_init, top_k=5, decay=0.98):
    """Subgradient method for minimax."""
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(iters):
        corr_vals = compute_corr_vals(h)
        # Find top-k active constraints
        idx = np.argsort(corr_vals)[-top_k:]
        
        g = np.zeros(n)
        dx = 2.0 / n
        for k_idx in idx:
            shift = k_idx - (n - 1)
            j = np.arange(n)
            jp = j + shift
            valid_p = (jp >= 0) & (jp < n)
            g[valid_p] += (1.0 - h[jp[valid_p]]) * dx
            jm = j - shift
            valid_m = (jm >= 0) & (jm < n)
            g[valid_m] -= h[jm[valid_m]] * dx

        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        alpha *= decay
    
    return best_h, best_c5

def coordinate_descent(h, n, max_sweeps=10):

    h = h.copy()
    for _ in range(max_sweeps):
        for i in range(n):
            best_val = h[i]

                else:

def slsqp_optimize_free(h, n, max_iter=300, beta=100.0):
    """SLSQP only on free variables (those not at 0 or 1)."""
    tol = 1e-6
    free_mask = (h > tol) & (h < 1.0 - tol)
    free_idx = np.where(free_mask)[0]
    n_free = len(free_idx)
    
    if n_free < 2:
        return h, compute_c5(h)
    
    h_free_init = h[free_idx].copy()
    fixed_part = h.copy()
    target_sum = n / 2.0
    fixed_sum = float(fixed_part[~free_mask].sum())
    free_target = target_sum - fixed_sum
    
    bounds = [(0.0, 1.0)] * n_free
    
    def obj_free(x_free):
        h_full = fixed_part.copy()
        h_full[free_idx] = x_free
        return smooth_max(compute_corrs(h_full), beta)
    
    def constr_free(x_free):
        return float(np.sum(x_free)) - free_target
    
    try:

                      constraints=[{'type': 'eq', 'fun': constr_free}],

        h_new[free_idx] = res.x
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return None, None

Actually, the problem says:
$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

The integral is over x such that both x and x+k are in [0,2]. For k ≥ 0, x ranges over [0, 2-k]. For k < 0, x ranges over [-k, 2].

In the discrete setting with n points on [0,2], the shift k corresponds to shifting by m grid points (m = k/dx). The valid range of m is from -(n-1) to (n-1), which is exactly what `np.correlate` with mode="full" gives.

So yes, we need to consider all 2n-1 shifts. The max is over all of them.

But wait: for very large shifts (|m| close to n-1), the overlap region is very small (only 1 or 2 points), so the contribution is tiny. The significant shifts are those with |m| << n. In practice, the max will be achieved at moderate shifts.

For the subgradient method, I only compute the gradient for the top-k shifts, which are the active constraints. This is efficient.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h_raw, n):
    h = np.asarray(h_raw, dtype=float)
    target = n / 2.0
    lo = 0.0 - float(np.max(h)) - 1.0
    hi = 1.0 - float(np.min(h)) + 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - float(result.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - result))
            result[idx] = min(result[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(result))
            result[idx] = max(result[idx] + diff, 0.0)
    return result

def smooth_max(x, beta):
    m = float(np.max(x))
    shifted = np.clip(beta * (x - m), -400.0, 400.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / beta

def subgradient_step(h, n, alpha, top_k):
    corrs = compute_corrs(h)
    idx = np.argsort(corrs)[-top_k:]
    g = np.zeros(n)
    dx = 2.0 / n
    for k_idx in idx:
        shift = int(k_idx) - (n - 1)
        j = np.arange(n)
        jp = j + shift
        valid_p = (jp >= 0) & (jp < n)
        if np.any(valid_p):
            g[valid_p] += (1.0 - h[jp[valid_p]]) * dx
        jm = j - shift
        valid_m = (jm >= 0) & (jm < n)
        if np.any(valid_m):
            g[valid_m] -= h[jm[valid_m]] * dx
    g /= float(len(idx))
    h_new = h - alpha * g
    return project_feasible(h_new, n)

def subgradient_loop(h, n, max_iters, alpha_init, decay, top_k, patience):
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        h_new = subgradient_step(h, n, alpha, top_k)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            alpha = min(alpha * 1.02, 80.0)
        else:
            no_improve += 1
        h = h_new
        alpha *= decay
        if no_improve > patience:
            alpha = max(alpha * 0.5, 0.05)
            no_improve = 0
            h = best_h.copy()
            if alpha < 0.01:
                break
        if alpha < 0.005:
            break
    return best_h, best_c5

def slsqp_obj(x, n, beta):
    h_q = np.clip(x, 0.0, 1.0)
    return smooth_max(compute_corrs(h_q), beta)

def slsqp_constr(x, n):
    return float(np.sum(x)) - n / 2.0

def slsqp_optimize(h, n, max_iter, beta):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constr, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, args=(n, beta),
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        if np.isfinite(c5_res):
            return h_res, c5_res
        return None, None
    except Exception:
        return None, None

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

def perturb_h(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)

def make_binary_pattern(n, rng, pattern_type):
    h = np.zeros(n)
    if pattern_type == 0:
        idx = rng.permutation(n)
        h[idx[:n // 2]] = 1.0
    elif pattern_type == 1:
        period = max(2, n // 10)
        for i in range(n):
            if (i % period) < period // 2:
                h[i] = 1.0
    elif pattern_type == 2:
        blocks = 5
        per_block = n // blocks
        for b in range(blocks):
            if b % 2 == 0:
                h[b * per_block:(b + 1) * per_block] = 1.0
    elif pattern_type == 3:
        freq = rng.choice([3, 5, 7, 11])
        t = np.linspace(0, freq * 2 * np.pi, n)
        h = 0.5 + 0.5 * np.sin(t)
        h = (np.sort(h)[::-1] > 0.5).astype(float)
    else:
        h[:] = 0.5
    return project_feasible(h, n)
