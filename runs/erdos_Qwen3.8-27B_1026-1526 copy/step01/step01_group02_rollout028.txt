def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project to [0,1] and sum = n/2
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smoothed_max_objective(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    # Shift to avoids overflow
    m = np.max(vals)
    if alpha * m > 700:
        m = 700.0 / alpha
    shifted = vals - m
    log_sum = np.log(np.sum(np.exp(alpha * shifted))) + alpha * m
    obj = log_sum / alpha
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    obj += mu * s * s
    return obj

def smoothed_max_gradient(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    m = np.max(vals)
    if alpha * m > 700:
        m = 700.0 / alpha
    shifted = vals - m
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    # Compute weighted sum of gradients
    grad = np.zeros(n)
    for k in range(2 * n - 1):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj_and_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    vals = c * dx
    m = np.max(vals)
    threshold = 50.0 / alpha
    if m > threshold:
        m = threshold
    shifted = vals - m
    exp_vals = np.exp(alpha * shifted)
    total = np.sum(exp_vals)
    w = exp_vals / total
    obj = (np.log(total) + alpha * m) / alpha
    s = np.sum(h) - n / 2.0
    obj += mu * s * s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                grad[:n - d] += wk * (1.0 - h[d:])
                grad[d:] += -wk * h[:n - d]
        else:
            dp = -d
            if dp < n:
                grad[:n - dp] += wk * (1.0 - h[dp:])
                grad[dp:] += -wk * h[:n - dp]
    grad += 2.0 * mu * s
    return obj, grad


def make_smooth_obj(alpha, mu):
    def obj(x):
        o, _ = smooth_max_obj_and_grad(x, alpha, mu)
        return o
    return obj


def make_smooth_grad(alpha, mu):
    def grad(x):
        _, g = smooth_max_obj_and_grad(x, alpha, mu)
        return g
    return grad


def optimize_smooth(h_start, alpha, mu, n, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    obj_fn = make_smooth_obj(alpha, mu)
    grad_fn = make_smooth_grad(alpha, mu)
    try:
        result = minimize(obj_fn, h_start, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'gtol': 1e-7, 'ftol': 1e-14})
        if result.x is not None:
            return result.x
    except Exception:
        pass
    return h_start
