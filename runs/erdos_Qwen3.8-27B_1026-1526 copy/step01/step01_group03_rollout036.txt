    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            sroom = np.sum(room)
            if sroom < 1e-15:
                break
            frac = min(1.0, diff / sroom)
            h = h + frac * room
        else:
            room = h
            sroom = np.sum(room)
            if sroom < 1e-15:
                break
            frac = min(1.0, -diff / sroom)
            h = h - frac * room
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    c = corr_full(h)
    return smooth_max(c, tau) * dx

def eq_con_fun(h, n):
    return np.sum(h) - n / 2.0

def eq_con_jac(h, n):
    return np.ones(n)

def obj_active_max(h, n, lags):
    c = corr_full(h)
    return np.max(c[lags])

def ineq_con_fun(h, n, lag, M_target):
    dx = 2.0 / n
    c = corr_full(h)
    return M_target / dx - c[lag]

def make_ineq_con(n, lag, M_target):
    return {'type': 'ineq', 'fun': lambda h: ineq_con_fun(h, n, lag, M_target), 'args': ()}

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
    except Exception:
        best_h = None

    if best_h is None:
        n = 59
        best_h = np.full(n, 0.5)
        best_h = project_feasible(best_h, n)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def time_left():
        return budget_s - (time.time() - start_time)

    def do_minimize(h0, obj_fn, args, cons, maxiter=200, ftol=1e-14):
        try:
            res = minimize(
                obj_fn, h0,
                method='SLSQP',
                args=args,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            return project_feasible(res.x, n)
        except Exception:
            return None

    # Phase 1: Smooth max optimization with tau annealing
    cons_std = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]

    taus = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001]
    current_h = best_h.copy()

    for tau in taus:
        if time_left() < 60:
            break
        for attempt in range(2):
            if time_left() < 30:
                break
            noise_scale = 0.008 * (attempt + 1)
            h_init = current_h + rng.randn(n) * noise_scale
            h_init = project_feasible(h_init, n)
            h_new = do_minimize(h_init, obj_smooth, (n, tau), cons_std, maxiter=300, ftol=1e-14)
            if h_new is not None:
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    current_h = h_new.copy()

    # Phase 2: Active set refinement - minimize max over top lags
    if time_left() > 120:
        corr = corr_full(best_h)
        max_corr = np.max(corr)

        for topk in [3, 5, 8, 12, 20, 35]:
            if time_left() < 60:
                break
            top_lags = np.argsort(corr)[-topk:]

            cons_eq = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]
            for attempt in range(3):
                if time_left() < 30:
                    break
                noise_scale = 0.004 * (attempt + 1)
                h_init = best_h + rng.randn(n) * noise_scale
                h_init = project_feasible(h_init, n)

                def obj_top(h, nl=top_lags):
                    c = corr_full(h)
                    return np.max(c[nl])

                h_new = do_minimize(h_init, obj_top, (), cons_eq, maxiter=400, ftol=1e-14)
                if h_new is not None:
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        corr = corr_full(best_h)
                        max_corr = np.max(corr)
                        top_lags = np.argsort(corr)[-topk:]

    # Phase 3: Binary search on M with active constraint feasibility
    if time_left() > 150:
        M_high = best_c5 + 0.0005
        M_low = best_c5 - 0.002

        for _bs in range(8):
            if time_left() < 60:
                break
            M_mid = 0.5 * (M_low + M_high)

            corr_cur = corr_full(best_h)
            max_c = np.max(corr_cur)
            active_threshold = max_c * 0.985
            active_lags = np.where(corr_cur >= active_threshold)[0]

            cons_bs = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]
            for lag in active_lags:
                con = {'type': 'ineq', 'fun': ineq_con_fun, 'args': (n, lag, M_mid)}
                cons_bs.append(con)

            h_init = best_h.copy()
            h_new = do_minimize(h_init, lambda x: 0.0, (), cons_bs, maxiter=300, ftol=1e-14)

            if h_new is not None:
                c5 = evaluate_c5(h_new)
                if c5 <= M_mid + 1e-8:
                    M_low = M_mid
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                else:
                    M_high = M_mid
            else:
                M_high = M_mid

    # Phase 4: Fine polish with multiple small perturbations
    if time_left() > 40:
        cons_pf = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n,)}]
        for attempt in range(30):
            if time_left() < 15:
                break
            noise = rng.randn(n) * 0.001
            h_init = best_h + noise
            h_init = project_feasible(h_init, n)
            h_new = do_minimize(h_init, obj_smooth, (n, 0.0005), cons_pf, maxiter=400, ftol=1e-15)
            if h_new is not None:
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

    # Phase 5: Try slightly different n values if time permits
    if time_left() > 200:
        for n_try in [n - 2, n + 2, n - 4, n + 4]:
            if time_left() < 120:
                break
            if n_try < 20 or n_try > 100:
                continue
            dx_try = 2.0 / n_try
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate')
            h_try = np.clip(f(x_new), 0, 1)
            h_try = project_feasible(h_try, n_try)
            c5_try = evaluate_c5(h_try)

            if c5_try < best_c5 + 0.001:
                bounds_try = [(0.0, 1.0)] * n_try
                cons_try = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac, 'args': (n_try,)}]
                for tau in [0.01, 0.005, 0.002, 0.001]:
                    if time_left() < 60:
                        break
                    h_init = h_try + rng.randn(n_try) * 0.005
                    h_init = project_feasible(h_init, n_try)
                    try:
                        res = minimize(
                            obj_smooth, h_init,
                            method='SLSQP',
                            args=(n_try, tau),
                            bounds=bounds_try,
                            constraints=cons_try,
                            options={'maxiter': 200, 'ftol': 1e-14}
                        )
                        h_pol = project_feasible(res.x, n_try)
                        c5_pol = evaluate_c5(h_pol)
                        if c5_pol < best_c5:
                            best_c5 = c5_pol
                            best_h = h_pol.copy()
                    except Exception:
                        pass

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
