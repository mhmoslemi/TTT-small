Let me check: (t-h0)(1-2h0) = t - 2th0 - h0 + 2h0²
And t-t²-h0+h0².

These differ by: (t - 2th0 - h0 + 2h0²) - (t - t² - h0 + h0²) = -2th0 + t² + h0² = (t-h0)². 

So the sum of the two linear approximations gives the actual change plus (t-h0)². This makes sense because the actual change is quadratic (t(1-t) is quadratic in t), while my linear decomposition misses the cross term.

For p≠n-1, the two conditions give different terms and I can add the linear changes.

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def coord_descent_pass(h, corr, n, beta, rng, n_passes=1):

    dx = 2.0 / n
    for pass_idx in range(n_passes):
        order = rng.permutation(n)
        for i0 in order:

            p1_min = n - 1 - i0
            p1_max = 2 * (n - 1) - i0
            p2_min = i0 - (n - 1)
            p2_max = i0
            
            # Precompute change coefficients
            p1_all = np.arange(p1_min, p1_max + 1)
            p1_mask = p1_all != (n - 1)

                coeff1 = (1.0 - h[idx1])  # these don't include i0

            p2 = p2_all[p2_mask]
            if len(p2) > 0:
                idx2 = i0 - p2 + (n - 1)
                coeff2 = h[idx2]  # these don't include i0
            
            def obj_1d(t):
                new_corr = corr.copy()
                if len(p1) > 0:
                    new_corr[p1] += (t - h0) * coeff1
                if len(p2) > 0:
                    new_corr[p2] += coeff2 * (h0 - t)
                new_corr[n - 1] += t * (1.0 - t) - h0 * (1.0 - h0)
                m = np.max(new_corr)
                shifted = np.clip(beta * (new_corr - m), -500.0, 500.0)
                return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

            if new_val != h0:

                if len(p1) > 0:
                    corr[p1] += (new_val - h0) * coeff1
                if len(p2) > 0:

                corr[n - 1] += new_val * (1.0 - new_val) - h0 * (1.0 - h0)
        # Project to maintain sum constraint
        h = project_feasible(h, n)
        corr = full_correlation(h)
    return h, corr
```

```python
def golden_min(f, a, b, tol=1e-12, max_iter=100):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2.0
```

```python
    # Phase 3: Perturbation + coordinate descent

        h_pert = h + rng.normal(0, sigma, n)
        h_pert = project_feasible(h_pert, n)

                h = h_pert.copy()
                no_improve = 0
```
