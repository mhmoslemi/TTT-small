def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):

            break
        if diff > 0:

            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14

def obj_smooth_max(x, n, beta, active_indices):

    m = np.max(vals)

def obj_penalty(x, n, beta, penalty):
