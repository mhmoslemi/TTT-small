import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_topk(values, tau, k):
    if len(values) <= k:
        v = values / tau
        vmax = np.max(v)
        return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))
    topk_idx = np.argpartition(values, -k)[-k:]
    topk = values[topk_idx]
    v = topk / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - result[idx])
            else:
                adj = max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return result

def constraint_func(h, n):
    return np.sum(h) - n / 2.0

def obj_smooth(h, n, tau, k):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max_topk(corr, tau, k) * dx

def obj_active(h, n, tau, active_idx):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    active_vals = corr[active_idx]
    v = active_vals / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx

def coord_eval(val, h_base, n, i):
    h_temp = h_base.copy()
    h_temp[i] = val
    h_temp = project_feasible(h_temp, n)
    return evaluate_c5(h_temp)

def generate_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n)

    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
            starts.append(project_feasible(parent_h + rng.randn(n) * 0.06, n))
            starts.append(project_feasible(parent_h + rng.randn(n) * 0.12, n))
        else:
            x_p = np.linspace(0, 2, len(parent_h), endpoint=False)
            f = interp1d(x_p, parent_h, kind='cubic', fill_value='extrapolate')
            starts.append(project_feasible(f(x), n))
            h_pert = parent_h + rng.randn(len(parent_h)) * 0.1
            f2 = interp1d(x_p, np.clip(h_pert, 0, 1), kind='cubic', fill_value='extrapolate')
            starts.append(project_feasible(f2(x), n))

    for freq in [1.1, 1.5, 1.9, 2.4, 2.9]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.28, 0.44)
        starts.append(project_feasible(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), n))

    for freq1, freq2 in [(1.3, 2.1), (1.7, 3.1), (2.2, 3.5)]:
        ph1, ph2 = rng.uniform(0, 2 * np.pi), rng.uniform(0, 2 * np.pi)
        a1, a2 = 0.25, 0.2
        sig = a1 * np.sin(2 * np.pi * freq1 * x + ph1) + a2 * np.sin(2 * np.pi * freq2 * x + ph2)
        starts.append(project_feasible(0.5 + sig, n))

    starts.append(project_feasible(rng.rand(n), n))
    starts.append(project_feasible(0.5 + 0.3 * np.cos(2 * np.pi * x), n))

    c1, c2 = 0.3 * n, 0.7 * n
    w = 0.15 * n
    b1 = np.exp(-0.5 * ((ar - c1) / w) ** 2)
    b2 = np.exp(-0.5 * ((ar - c2) / w) ** 2)
    starts.append(project_feasible(0.15 + 0.8 * 0.5 * (b1 + b2), n))

    h_saw = np.mod(ar * 1.0 / n * 3.0, 2.0)
    h_saw = np.where(h_saw < 1.0, h_saw, 2.0 - h_saw)
    starts.append(project_feasible(h_saw, n))

    return starts
