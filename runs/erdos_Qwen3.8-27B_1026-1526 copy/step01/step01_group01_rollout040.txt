```python
def spectral_shaping(h, n_iters, rng):
    n = len(h)
    h = h.copy()
    for it in range(n_iters):
        # Compute FFT
        H = fft(h)
        # Get magnitudes
        mag = np.abs(H)
        phase = np.angle(H)
        # Target: make magnitudes more uniform (except DC)
        # DC component: H[0] = sum(h) = n/2
        # Target magnitude for non-DC: mean of current magnitudes
        target_mag = np.mean(mag[1:])
        # Interpolate between current and target
        alpha = 0.1  # mixing parameter
        new_mag = np.copy(mag)
        new_mag[0] = n/2.0  # keep DC fixed
        new_mag[1:] = mag[1:] + alpha * (target_mag - mag[1:])
        # Reconstruct
        new_H = new_mag * np.exp(1j * phase)
        h_new = np.real(ifft(new_H))
        # Project
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    x_max = np.max(x)
    shifted = np.clip(beta * (x - x_max), -500.0, 500.0)
    return x_max + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            # Apply to all
            h += diff / n
        else:
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def slsqp_optimize(h, n, beta, max_iter=300, ftol=1e-14):
    def obj(x):
        h_local = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h_local, 1.0 - h_local, mode="full")
        return smooth_max(corr, beta) * (2.0 / n)
    
    def con(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con}],
                      options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, None

def subgradient_descent(h, n, n_iters, step_decay=0.999, base_step=0.02):
    """Direct subgradient method focusing on worst lags."""
    h = h.copy()
    current_c5 = compute_c5(h)
    step = base_step
    momentum = np.zeros(n)
    
    for it in range(n_iters):
        corr_arr = compute_corr_array(h)
        # Find top 5 lags
        top_k = 5
        top_idx = np.argsort(corr_arr)[-top_k:]
        
        # Compute weighted gradient
        # Weights: use softmax of top correlation values
        top_vals = corr_arr[top_idx]
        weights = np.exp(top_vals / (0.01)) / np.sum(np.exp(top_vals / (0.01)))
        
        grad = np.zeros(n)
        for w, k_idx in zip(weights, top_idx):
            k = k_idx - (n - 1)
            # grad[j] from lag k:
            # (1 - h[j+k]) if 0 <= j+k < n, minus h[j-k] if 0 <= j-k < n
            if k > 0:
                # (1-h[j+k]) contribution: for j in [0, n-k-1]
                grad[:n-k] += w * (1.0 - h[k:n])
                # -h[j-k] contribution: for j in [k, n-1]
                grad[k:] += w * (-h[:n-k])
            elif k < 0:
                jk = -k
                # (1-h[j-k]) = (1-h[j+jk]): for j in [0, n-jk-1]
                grad[:n-jk] += w * (1.0 - h[jk:n])
                # -h[j+k] = -h[j-jk]: for j in [jk, n-1]
                grad[jk:] += w * (-h[:n-jk])
            else:
                # k=0: grad[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
                grad += w * (1.0 - 2.0 * h)
        
        # Normalize
        gn = np.linalg.norm(grad)
        if gn > 1e-10:
            grad /= gn
        
        # Momentum (Nesterov-like)
        momentum = 0.9 * momentum + grad
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm > 1e-10:
            momentum /= momentum_norm
        
        # Update

        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

            h = h_new
            current_c5 = c5_new
        else:
            # Rejected, reduce step
            step *= 0.95
        
        step = max(step * step_decay, 0.001)
    
    return h, current_c5
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    x_max = np.max(x)
    shifted = np.clip(beta * (x - x_max), -500.0, 500.0)
    return x_max + np.log(np.sum(np.exp(shifted))) / beta

def compute_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_scaled = corr * dx
    corr_max = np.max(corr_scaled)
    shifted = np.clip(beta * (corr_scaled - corr_max), -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += pm * (1.0 - h[k:n])
            if n > k:
                grad[k:] += pm * (-h[:n-k])
        else:
            ka = -k
            if n - ka > 0:
                grad[ka:] += pm * (1.0 - h[:n-ka])
                grad[:n-ka] += pm * (-h[ka:n])
    return grad

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            h += diff / n
        else:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def slsqp_optimize(h, n, beta, max_iter=300, ftol=1e-14):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        return smooth_max(corr, beta) * (2.0 / n)
    
    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        return compute_gradient(hc, n, beta)
    
    def con(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                      options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, None

def lbfgs_penalty(h, n, beta, max_iter=500, lam=10.0):
    """L-BFGS with penalty for sum constraint."""
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        sm = smooth_max(corr, beta) * (2.0 / n)
        pen = lam * (np.sum(hc) - n/2.0)**2
        return sm + pen
    
    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        g = compute_gradient(hc, n, beta)
        g += 2.0 * lam * (np.sum(hc) - n/2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, None

def subgradient_momentum(h, n, n_iters, base_step=0.005, momentum_rate=0.8):
    """Subgradient method with Nesterov momentum, focusing on worst lags."""
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    momentum = np.zeros(n)
    step = base_step
    dx = 2.0 / n
    
    for it in range(n_iters):
        # Compute correlation and find worst lags
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        # Use top 10 lags with softmax weights
        top_k = 10
        sorted_idx = np.argsort(corr)
        top_idx = sorted_idx[-top_k:]
        top_vals = corr[top_idx]
        # Softmax with temperature
        temp = 0.005
        w = np.exp((top_vals - np.max(top_vals)) / temp)
        w /= np.sum(w)
        
        grad = np.zeros(n)
        for i, m in enumerate(top_idx):
            wi = w[i]
            k = m - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += wi * (1.0 - h[k:n])
                    grad[k:] += wi * (-h[:n-k])
            else:
                ka = -k
                if n - ka > 0:
                    grad[ka:] += wi * (1.0 - h[:n-ka])
                    grad[:n-ka] += wi * (-h[ka:n])
        
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn
        
        # Nesterov momentum
        momentum = momentum_rate * momentum + (1 - momentum_rate) * grad
        mn = np.linalg.norm(momentum)
        if mn > 1e-15:
            momentum /= mn
        
        h_new = h - step * momentum
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < current_c5:
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
            current_c5 = c5_new
            step = min(step * 1.01, 0.05)
        else:
            step *= 0.9
        
        # Anneal step
        step = max(step * 0.9999, 0.001)
    
    return best_h, best_c5

def simulated_annealing(h, n, n_iters, T0=0.003, Tf=0.00003, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    x = np.arange(n, dtype=float)
    
    for it in range(n_iters):
        T = T0 * (Tf / T0) ** (float(it) / float(n_iters))
        pert_type = rng.randint(4)
        h_pert = h.copy()
        
        if pert_type == 0:
            center = rng.uniform(0, n)
            width = rng.uniform(3.0, 10.0)
            amp = rng.uniform(0.03, 0.15) * (1.0 if rng.random() > 0.5 else -1.0)
            h_pert += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        elif pert_type == 1:
            freq = rng.uniform(1.0, 8.0)
            amp = rng.uniform(0.02, 0.08)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif pert_type == 2:
            bs = rng.randint(3, 12)
            start = rng.randint(0, max(1, n - bs))
            shift = rng.uniform(0.02, 0.1) * rng.choice([-1, 1])
            window = np.linspace(1, 0, bs)
            h_pert[start:start+bs] += shift * window
        else:
            h_pert += rng.normal(0, rng.uniform(0.02, 0.08), n)
        
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        
        delta = c5_pert - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_pert
            current_c5 = c5_pert
            if c5_pert < best_c5:
                best_h = h.copy()
                best_c5 = c5_pert
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    x_max = np.max(x)
    shifted = np.clip(beta * (x - x_max), -500.0, 500.0)
    return x_max + np.log(np.sum(np.exp(shifted))) / beta


def compute_gradient(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_scaled = corr * dx
    corr_max = np.max(corr_scaled)
    shifted = np.clip(beta * (corr_scaled - corr_max), -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += pm * (1.0 - h[k:n])
                grad[k:] += pm * (-h[:n-k])
        else:
            ka = -k
            if n - ka > 0:
                grad[ka:] += pm * (1.0 - h[:n-ka])
                grad[:n-ka] += pm * (-h[ka:n])
    return grad


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            h += diff / n
        else:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def slsqp_optimize(h, n, beta, max_iter=300, ftol=1e-14):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        return smooth_max(corr, beta) * (2.0 / n)

    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        return compute_gradient(hc, n, beta)

    def con(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, None


def lbfgs_penalty(h, n, beta, max_iter=500, lam=10.0):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        sm = smooth_max(corr, beta) * (2.0 / n)
        pen = lam * (np.sum(hc) - n / 2.0) ** 2
        return sm + pen

    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        g = compute_gradient(hc, n, beta)
        g += 2.0 * lam * (np.sum(hc) - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return None, None


def subgradient_momentum(h, n, n_iters, base_step=0.004, momentum_rate=0.85):
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    momentum = np.zeros(n)
    step = base_step
    dx = 2.0 / n

    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        top_k = 8
        sorted_idx = np.argsort(corr)
        top_idx = sorted_idx[-top_k:]
        top_vals = corr[top_idx]
        temp = 0.003
        w = np.exp((top_vals - np.max(top_vals)) / temp)
        w /= np.sum(w)

        grad = np.zeros(n)
        for i, m in enumerate(top_idx):
            wi = w[i]
            k = m - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += wi * (1.0 - h[k:n])
                    grad[k:] += wi * (-h[:n-k])
            else:
                ka = -k
                if n - ka > 0:
                    grad[ka:] += wi * (1.0 - h[:n-ka])
                    grad[:n-ka] += wi * (-h[ka:n])

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad /= gn

        momentum = momentum_rate * momentum + (1.0 - momentum_rate) * grad
        mn = np.linalg.norm(momentum)
        if mn > 1e-15:
            momentum /= mn

        h_new = h - step * momentum
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < current_c5:
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
            current_c5 = c5_new
            step = min(step * 1.005, 0.04)
        else:
            step *= 0.92

        step = max(step * 0.99995, 0.001)

    return best_h, best_c5


def simulated_annealing(h, n, n_iters, T0=0.002, Tf=0.00002, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    x = np.arange(n, dtype=float)

    for it in range(n_iters):
        T = T0 * (Tf / T0) ** (float(it) / float(n_iters))
        pert_type = rng.randint(5)
        h_pert = h.copy()

        if pert_type == 0:
            center = rng.uniform(0, n)
            width = rng.uniform(2.0, 12.0)
            amp = rng.uniform(0.02, 0.12) * (1.0 if rng.random() > 0.5 else -1.0)
            h_pert += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        elif pert_type == 1:
            freq = rng.uniform(0.5, 10.0)
            amp = rng.uniform(0.01, 0.06)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert += amp * np.sin(2 * np.pi * freq * x / n + phase)
        elif pert_type == 2:
            bs = rng.randint(2, 15)
            start = rng.randint(0, max(1, n - bs))
            shift = rng.uniform(0.01, 0.08) * rng.choice([-1, 1])
            window = np.cos(np.pi * np.arange(bs) / max(1, bs - 1)) ** 2
            h_pert[start:start+bs] += shift * window
        elif pert_type == 3:
            n_bumps = rng.randint(1, 3)
            for _ in range(n_bumps):
                c = rng.uniform(0, n)
                w = rng.uniform(1.0, 5.0)
                a = rng.uniform(0.01, 0.06) * rng.choice([-1, 1])
                h_pert += a * np.exp(-0.5 * ((x - c) / w) ** 2)
        else:
            sigma = rng.uniform(0.01, 0.05)
            h_pert += rng.normal(0, sigma, n)

        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)

        delta = c5_pert - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_pert
            current_c5 = c5_pert
            if c5_pert < best_c5:
                best_h = h.copy()
                best_c5 = c5_pert

    return best_h, best_c5


def spectral_shaping(h, n, n_iters=80, alpha=0.03, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(n_iters):
        H = fft(h)
        mag = np.abs(H)
        phase = np.angle(H)

        target_mag = np.mean(mag[1:])
        new_mag = np.copy(mag)
        new_mag[0] = n / 2.0
        for fi in range(1, n):
            if mag[fi] > 1.5 * target_mag:
                new_mag[fi] = 1.2 * target_mag + 0.3 * (mag[fi] - 1.5 * target_mag)
            else:
                new_mag[fi] = mag[fi] + alpha * (target_mag - mag[fi])

        new_mag = np.maximum(new_mag, 0)
        new_mag[0] = n / 2.0

        new_H = new_mag * np.exp(1j * phase)
        h_new = np.real(ifft(new_H))
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new

    return best_h, best_c5
```
