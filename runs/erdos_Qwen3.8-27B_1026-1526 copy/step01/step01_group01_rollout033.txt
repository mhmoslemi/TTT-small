```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            order = np.argsort(1.0 - h)[::-1]
            remaining = diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = 1.0 - h[idx]
                add = min(remaining, room)
                h[idx] += add
                remaining -= add
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = h[idx]
                sub = min(remaining, room)
                h[idx] -= sub
                remaining -= sub
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def obj_smooth_max(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def subgradient_polish(h, n, steps, alpha_init):
    """Polish using subgradient descent on the max lag."""
    dx = 2.0 / n
    for s in range(steps):
        alpha = alpha_init * (0.95 ** s)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = int(np.argmax(corr))
        d = k_star - (n - 1)
        
        # Gradient of sum_i h[i]*(1-h[i+d]) w.r.t. h
        grad = np.zeros(n)
        if d >= 0:
            # h[i] * (1-h[i+d]): grad[i] gets (1-h[i+d]) for i where i+d < n
            # (1-h[i]) * h[i-d]: grad[i] gets -h[i-d] for i where i-d >= 0
            if d == 0:
                grad[:] = 0  # For binary this would be 0, but for general h...
                # Actually for d=0: sum h[i]*(1-h[i]), grad[i] = 1-2*h[i]
                grad[:] = 1.0 - 2.0 * h
            else:
                # Term 1: h[i]*(1-h[i+d]) for i=0..n-1-d
                grad[0:n-d] += (1.0 - h[d:n])
                # Term 2: h[i-d]*(1-h[i]) for i=d..n-1
                grad[d:n] -= h[0:n-d]
        else:
            dd = -d
            if dd == 0:
                grad[:] = 1.0 - 2.0 * h
            else:
                # h[i]*(1-h[i-dd]) for i=dd..n-1
                grad[dd:n] += (1.0 - h[0:n-dd])
                # h[i+dd]*(1-h[i]) for i=0..n-1-dd
                grad[0:n-dd] -= h[dd:n]
        
        h_new = h - alpha * grad
        h = project_feasible(h_new, n)
    
    return h

def slsqp_optimize(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_smooth_max, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            order = np.argsort(1.0 - h)[::-1]
            remaining = diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = 1.0 - h[idx]
                add = min(remaining, room)
                h[idx] += add
                remaining -= add
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = h[idx]
                sub = min(remaining, room)
                h[idx] -= sub
                remaining -= sub
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def subgradient_polish(h, n, steps, alpha_init):
    dx = 2.0 / n
    for s in range(steps):
        alpha = alpha_init / (1.0 + 0.05 * s)
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        d = k_star - (n - 1)
        
        grad = np.zeros(n)
        if d >= 0:
            if d < n:
                grad[d:] += (1.0 - h[0:n-d])
                grad[0:n-d] -= h[d:]
        else:
            dd = -d
            if dd < n:
                grad[0:n-dd] += (1.0 - h[dd:])
                grad[dd:] -= h[0:n-dd]
        
        h_new = h - alpha * grad
        h = project_feasible(h_new, n)
    return h

def slsqp_optimize(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        if np.isfinite(c5):
            return h_res, c5
    except Exception:
        pass
    return None, None

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```

**Better approach: Direct worst-lag optimization**

Instead of smooth max, directly optimize the worst lag:
1. Find the worst lag k*
2. Minimize f(h) = ∑ h[i]*(1-h[i-k*]) subject to:
   - ∑ h[i]*(1-h[i-k]) ≤ f(h) + ε for all other k (to not make others worse)
   - 0 ≤ h[i] ≤ 1
   - ∑ h[i] = n/2

This is a non-convex problem (bilinear objective + bilinear constraints), but we can solve it with SLSQP.

Actually, even simpler: just minimize the worst lag directly, and add a small penalty for other lags exceeding the current max:

f(h) = ∑ h[i]*(1-h[i-k*]) + λ * ∑_k max(0, ∑ h[i]*(1-h[i-k]) - current_max)^2

The penalty term is smooth and encourages other lags to stay below the current max.

```python
def obj_focused_worst(h_flat, n, k_star_idx, current_max_corr, lam):
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            order = np.argsort(1.0 - h)[::-1]
            remaining = diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = 1.0 - h[idx]
                add = min(remaining, room)
                h[idx] += add
                remaining -= add
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            order = np.argsort(h)
            remaining = -diff
            for idx in order:
                if remaining <= 1e-14:
                    break
                room = h[idx]
                sub = min(remaining, room)
                h[idx] -= sub
                remaining -= sub
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def obj_smooth_max(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def subgradient_polish(h, n, steps, alpha_init):
    for s in range(steps):
        alpha = alpha_init / (1.0 + 0.08 * s)
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        d = k_star - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            if d < n:
                grad[d:] += (1.0 - h[0:n - d])
                grad[0:n - d] -= h[d:]
        else:
            dd = -d
            if dd < n:
                grad[0:n - dd] += (1.0 - h[dd:])
                grad[dd:] -= h[0:n - dd]
        h_new = h - alpha * grad
        h = project_feasible(h_new, n)
    return h


def slsqp_optimize(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_smooth_max, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        if np.isfinite(c5):
            return h_res, c5
    except Exception:
        pass
    return None, None


def alternating_worst_lag(h, n, max_rounds, lam, max_iter_per_round):
    current_max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for _ in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))

        def obj_focused(x):

            main = corr_x[k_star]
            excess = np.maximum(corr_x - current_max_corr, 0.0)
            return (main + lam * np.sum(excess * excess)) * (2.0 / n)

            h_res = project_feasible(res.x, n)

            new_max_corr = np.max(np.correlate(h_res, 1.0 - h_res, mode="full"))

                h = h_res
                current_max_corr = new_max_corr
```
