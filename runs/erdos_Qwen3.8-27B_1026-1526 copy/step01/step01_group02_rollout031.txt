def solve_lp(h0, n):
    import cvxpy as cp
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    dx = 2.0 / n
    obj = cp.Minimize(t)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for k in range(2*n - 1):
        lag = k - (n-1)
        # Linearize g_k(h) around h0
        # g_k(h) = sum_i h[i+lag]*(1-h[i]) * dx (for valid indices)
        # Linearization: g_k(h0) + sum_j (partial g_k / partial h_j)|_{h0} * (h_j - h0_j)
        
        # Compute g_k(h0) and gradient at h0
        g_k_h0 = 0.0
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + lag < n:
                g_k_h0 += h0[i+lag] * (1 - h0[i]) * dx
                grad[i+lag] += (1 - h0[i]) * dx
                grad[i] += -h0[i+lag] * dx
        
        # Linearized constraint: g_k_h0 + grad^T (h - h0) <= t
        constraints.append(g_k_h0 + grad @ (h_var - h0) <= t)
    
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLOP)  # or ECOS, SCS
    
    if h_var.value is not None:
        return np.array(h_var.value), t.value
    return h0, np.inf

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_all_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx


def project_feasible_fast(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_weighted_gradient(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if abs(w) < 1e-16:
            continue
        lag = k - (n - 1)
        if lag > 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0:
            la = -lag
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]
        else:
            grad[:] += w * (1.0 - 2.0 * h)
    return grad


def smoothed_grad_step(h, n, t_smooth, alpha, momentum, beta):
    corr = compute_all_c5(h)
    scaled = t_smooth * corr
    scaled -= np.max(scaled)
    exp_scaled = np.exp(scaled)
    weights = exp_scaled / np.sum(exp_scaled)
    grad = compute_weighted_gradient(h, n, weights)
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad = grad / gnorm
    momentum = beta * momentum + (1.0 - beta) * grad
    h_new = h - alpha * momentum
    h_new = project_feasible_fast(h_new, n)
    return h_new, momentum, np.max(corr)


def active_set_slsqp(h, n, top_k=20, maxiter=600):
    corr = compute_all_c5(h)
    c_max = float(np.max(corr))
    top_idx = np.argpartition(corr, -top_k)[-top_k:]

    def obj_fn(x):
        return x[-1]

    def ineq_fn(x):
        hx = x[:-1]
        t = x[-1]
        c = np.correlate(hx, 1.0 - hx, mode="full") * (2.0 / n)
        return t - c[top_idx]

    def eq_fn(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac_fn(x):
        jac = np.zeros(len(x))
        jac[:-1] = 1.0
        return jac

    x0 = np.concatenate([h, [c_max + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(max(0.1, c_max * 0.85), c_max + 0.5)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_fn},
        {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}
    ]
    result = minimize(obj_fn, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible_fast(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h.copy(), c_max


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(135, 0.5)
    n = len(h)
    h = project_feasible_fast(h, n)

    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)

    phase1_end = budget_s * 0.35
    phase2_end = budget_s * 0.55
    phase3_end = budget_s * 0.80

    # Phase 1: Smoothed gradient descent with momentum, increasing t_smooth
    for phase in range(8):
        if time.time() - t0 > phase1_end:
            break
        t_smooth = 40.0 * (phase + 1)
        alpha0 = 0.04 / (1.0 + 0.3 * phase)
        beta = 0.85 + 0.02 * min(phase, 5)
        momentum = np.zeros(n)
        stall_count = 0

        for it in range(4000):
            if time.time() - t0 > phase1_end:
                break
            h_new, momentum, _ = smoothed_grad_step(h, n, t_smooth, alpha0, momentum, beta)
            c_new = compute_c5_from_h(h_new)
            if c_new < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = c_new
                stall_count = 0
            else:
                stall_count += 1
            h = h_new
            if stall_count > 500:
                break

    best_h = project_feasible_fast(best_h, n)
    best_c5 = compute_c5_from_h(best_h)

    # Phase 2: Active-set SLSQP refinement
    if time.time() - t0 < phase2_end:
        for attempt in range(15):
            if time.time() - t0 > phase2_end:
                break
            h_try = best_h + 0.005 * rng.standard_normal(n) if attempt > 0 else best_h
            h_try = project_feasible_fast(h_try, n)
            h_new, c5_new = active_set_slsqp(h_try, n, top_k=20, maxiter=1000)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_feasible_fast(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    # Phase 3: Multi-resolution
    for n_target in [270, 540]:
        if time.time() - t0 > phase3_end - 100:
            break
        n_curr = len(best_h)
        x_old = np.linspace(0, 2, n_curr, endpoint=False)
        x_new = np.linspace(0, 2, n_target, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_high = np.clip(f_interp(x_new), 0, 1)
        h_high = project_feasible_fast(h_high, n_target)
        c_high = compute_c5_from_h(h_high)

        # Smoothed gradient at higher resolution
        for phase in range(4):
            if time.time() - t0 > phase3_end - 50:
                break
            t_smooth = 50.0 * (phase + 1)
            alpha0 = 0.03
            beta = 0.9
            momentum = np.zeros(n_target)

            for it in range(2000):
                if time.time() - t0 > phase3_end - 50:
                    break
                h_high, momentum, _ = smoothed_grad_step(h_high, n_target, t_smooth, alpha0, momentum, beta)
                c_new = compute_c5_from_h(h_high)
                if c_new < best_c5 - 1e-12:
                    best_h = h_high.copy()
                    best_c5 = c_new
                    n_curr = n_target

        # Active-set SLSQP at higher resolution
        if time.time() - t0 < phase3_end - 30:
            h_new, c5_new = active_set_slsqp(h_high, n_target, top_k=25, maxiter=800)
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                n_curr = n_target

    # Phase 4: Final fine-tuning with small perturbations + SLSQP
    n_final = len(best_h)
    if time.time() - t0 < budget_s - 30:
        for attempt in range(40):
            if time.time() - t0 > budget_s - 20:
                break
            scale = 0.008 * (1.0 + attempt * 0.1)
            h_pert = best_h + scale * rng.standard_normal(n_final)
            h_pert = project_feasible_fast(h_pert, n_final)
            h_new, c5_new = active_set_slsqp(h_pert, n_final, top_k=12, maxiter=500)
            if c5_new < best_c5 - 1e-11:
                best_h = h_new
                best_c5 = c5_new

    # Final projected gradient polish
    if time.time() - t0 < budget_s - 15:
        n_fin = len(best_h)
        momentum = np.zeros(n_fin)
        for it in range(2000):
            if time.time() - t0 > budget_s - 15:
                break
            best_h, momentum, _ = smoothed_grad_step(best_h, n_fin, 200.0, 0.01, momentum, 0.9)
            c_new = compute_c5_from_h(best_h)
            best_c5 = min(best_c5, c_new)

    best_h = project_feasible_fast(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
