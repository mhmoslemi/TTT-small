    # Compute current correlations
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    num_k = 2 * n - 1
    
    # For coordinate m, compute A_k = d(corr[k])/d(h[m]) for each k
    k_index = np.arange(num_k)  # 0 to 2n-2
    actual_k = k_index - (n - 1)  # -(n-1) to (n-1)
    
    # A_k = (1 - h[m+k]) * [0 <= m+k < n] - h[m-k] * [0 <= m-k < n]
    m_plus_k = m + actual_k  # array of m+k for each k
    m_minus_k = m - actual_k  # array of m-k for each k
    
    valid_plus = (m_plus_k >= 0) & (m_plus_k < n)
    valid_minus = (m_minus_k >= 0) & (m_minus_k < n)
    
    A = np.zeros(num_k)
    A[valid_plus] = 1.0 - h[m_plus_k[valid_plus]]
    A[valid_minus] -= h[m_minus_k[valid_minus]]
    
    # C_k = corr[k] - h[m] * A_k
    C = corr - h[m] * A
    
    # Now minimize max_k (C_k + A_k * t) for t in [0, 1]
    # Use fine grid + refinement
    t_grid = np.linspace(0, 1, 200)
    f_vals = np.max(C[None, :] + A[None, :] * t_grid[:, None], axis=1)
    
    # Find minimum
    t_best = t_grid[np.argmin(f_vals)]
    f_best = f_vals[np.argmin(f_vals)]
    
    # Refine with golden section
    if 0 < t_best < 1:
        # Narrow the bracket
        idx = np.argmin(f_vals)
        if idx > 0 and idx < len(t_grid)-1:
            t_lo = t_grid[max(0, idx-2)]
            t_hi = t_grid[min(len(t_grid)-1, idx+2)]
        else:
            t_lo, t_hi = 0.0, 1.0
        
        # Golden section refinement
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(50):
            t1 = t_hi - gr * (t_hi - t_lo)
            t2 = t_lo + gr * (t_hi - t_lo)
            f1 = np.max(C + A * t1)
            f2 = np.max(C + A * t2)
            if f1 < f2:
                t_hi = t2
            else:
                t_lo = t1
        t_best = 0.5 * (t_lo + t_hi)
        f_best = np.max(C + A * t_best)
    
    return t_best

def coordinate_descent(h, n, max_sweeps=1000, tol=1e-10):
    """Run coordinate descent to minimize max correlation."""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for sweep in range(max_sweeps):

        if old_max - new_max < tol:
            break

    return h

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                adj = min(diff, 1.0 - result[idx])
            else:
                adj = max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-13:
                break
        if abs(diff) > 1e-10:
            j = int(np.argmin(np.abs(result - 0.5)))
            result[j] = np.clip(result[j] + diff, 0.0, 1.0)
    return result

def coord_best_val(h, m, n):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    nk = 2 * n - 1
    actual_k = np.arange(nk) - (n - 1)
    mpk = m + actual_k
    mkk = m - actual_k
    vp = (mpk >= 0) & (mpk < n)
    vm = (mkk >= 0) & (mkk < n)
    A = np.zeros(nk)
    A[vp] = g[mpk[vp]]
    A[vm] -= h[mkk[vm]]
    C = corr - h[m] * A
    t_grid = np.linspace(0.0, 1.0, 80)
    f = np.max(C[None, :] + A[None, :] * t_grid[:, None], axis=1)
    idx = int(np.argmin(f))
    if idx == 0:
        return 0.0
    if idx == len(t_grid) - 1:
        return 1.0
    t_lo = t_grid[max(0, idx - 1)]
    t_hi = t_grid[min(len(t_grid) - 1, idx + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(30):
        t1 = t_hi - gr * (t_hi - t_lo)
        t2 = t_lo + gr * (t_hi - t_lo)
        f1 = np.max(C + A * t1)
        f2 = np.max(C + A * t2)
        if f1 < f2:
            t_hi = t2
        else:
            t_lo = t1
    return 0.5 * (t_lo + t_hi)

def coord_descent(h, n, max_sweeps=3000, tol=1e-11, project_every=5):
    h = h.copy()
    for sweep in range(max_sweeps):
        old_corr = np.correlate(h, 1.0 - h, mode="full")
        old_max = float(np.max(old_corr))
        for m in range(n):
            h[m] = coord_best_val(h, m, n)
        if sweep % project_every == 4:
            h = project_feasible(h, n)
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_corr))
        if old_max - new_max < tol:
            break
    h = project_feasible(h, n)
    return h

def coord_descent_random_order(h, n, max_sweeps=2000, tol=1e-11, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    for sweep in range(max_sweeps):
        old_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        order = rng.permutation(n)
        for m in order:
            h[m] = coord_best_val(h, m, n)
        if sweep % 5 == 4:
            h = project_feasible(h, n)
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        if old_max - new_max < tol:
            break
    h = project_feasible(h, n)
    return h

def smooth_max_obj(h, n, tau):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * (2.0 / n)

def slsqp_polish(h, n, maxiter=300):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        g = 1.0 - x
        corr = np.correlate(x, g, mode="full")
        tau = 0.003
        v = corr / tau
        vmax = np.max(v)
        return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * (2.0 / n)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
