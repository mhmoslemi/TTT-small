import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def compute_c5_vector(h):
    """Return the full correlation vector scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def compute_subgrad_at_lag(h, k):
    """Compute subgradient of C_{n-1+k} w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:n] -= h[:n-k]
    else:
        kk = -k
        if n - kk > 0:
            g[kk:] += (1.0 - h[:kk])
            g[:n-kk] -= h[kk:]
    return g * dx


def subgradient_descent(h, n_iters=5000, active_eps=0.001, lr_init=0.001):
    """Subgradient descent on the max correlation."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    
    for it in range(n_iters):
        C = compute_c5_vector(h)
        cmax = np.max(C)
        if cmax < best_c5:
            best_c5 = cmax
            best_h = h.copy()
        
        # Active set
        active_mask = C >= cmax - active_eps
        n_active = np.sum(active_mask)
        if n_active == 0:
            active_mask[np.argmax(C)] = True
            n_active = 1
        
        # Average subgradient over active lags
        grad = np.zeros(n)
        for m in np.where(active_mask)[0]:
            k = m - (n - 1)
            grad += compute_subgrad_at_lag(h, k)
        grad /= n_active
        
        # Adaptive step size
        lr = lr_init / (1.0 + it * 0.01)
        
        # Projected step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5_from_h(h_new)
        if c5_new < cmax * (1 + 1e-6):  # Allow small increase
            h = h_new
        else:
            # Halve learning rate
            lr_init *= 0.5
            h_new = h - lr_init / (1.0 + it * 0.01) * grad
            h_new = project_feasible(h_new, n)
            h = h_new
    
    return best_h, best_c5


def slsqp_polish(h, n, maxiter=2000, ftol=1e-14):
    """Polish with SLSQP."""
    c5_init = compute_c5_from_h(h)
    
    def obj_func(x):
        return x[-1]
    
    def ineq_func(x):
        h_v = x[:-1]
        t = x[-1]
        n_v = len(h_v)
        dx = 2.0 / n_v
        c = np.correlate(h_v, 1.0 - h_v, mode="full")
        return t - c * dx
    
    def eq_func(x):
        h_v = x[:-1]
        return np.sum(h_v) - len(h_v) / 2.0
    
    def eq_jac_func(x):
        n_v = len(x) - 1
        jac = np.zeros(len(x))
        jac[:n_v] = 1.0
        return jac
    
    x0 = np.concatenate([h, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]
    
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h, c5_init


def structured_perturb(h, n, rng, mode='sinusoidal'):
    """Generate structured perturbations."""
    if mode == 'sinusoidal':
        freq = rng.choice([2, 3, 4, 5, 6, 8, 10])
        x = np.arange(n)
        amp = rng.uniform(0.01, 0.05)
        phase = rng.uniform(0, 2*np.pi)
        pert = amp * np.sin(2*np.pi*freq*x/n + phase)
    elif mode == 'block':
        block_size = rng.choice([3, 5, 10, 15])
        pert = np.zeros(n)
        center = rng.integers(0, n)
        pert[max(0,center-block_size//2):min(n,center+block_size//2)] = rng.uniform(-0.05, 0.05)
    else:
        pert = rng.standard_normal(n) * 0.02
    return pert

For the first term: w_k * (1-h[j+k]) for k such that 0≤j+k<n, i.e., -j ≤ k ≤ n-1-j.
Let m = j+k, then k = m-j, and m ranges from 0 to n-1.
Sum = ∑_{m=0}^{n-1} w_{m-j} * (1-h[m])

For the second term: -w_k * h[j-k] for k such that 0≤j-k<n, i.e., j-n+1 ≤ k ≤ j.
Let m = j-k, then k = j-m, and m ranges from 0 to n-1.
Sum = -∑_{m=0}^{n-1} w_{j-m} * h[m]

So the weighted gradient at position j is:
dx * [∑_{m=0}^{n-1} w_{m-j} * (1-h[m]) - ∑_{m=0}^{n-1} w_{j-m} * h[m]]

This is a convolution! w_{m-j} is w flipped and shifted by j. This can be computed as:
First term: convolve(1-h, w_flipped) evaluated at appropriate positions
Second term: -convolve(h, w) evaluated at appropriate positions

Actually, let me define w_k for k = -(n-1), -(n-2), ..., 0, ..., n-2, n-1.
In array form, w has length 2n-1, indexed as w[0] corresponds to k=-(n-1), w[n-1] to k=0, w[2n-2] to k=n-1.

The first term for position j: ∑_{m=0}^{n-1} w_{m-j} * (1-h[m])
where w_k = weights[k+(n-1)] in the correlation indexing.

This is (1-h) * w_flipped convolved at position... hmm, it's getting complicated with the indexing. Let me just use the loop for now since n=135 is small enough.

Actually, for n=135, the inner loop over 269 lags with O(n) work each is 269*135 ≈ 36k operations. In Python, this is maybe 1-2ms per gradient evaluation. For 2000 iterations, that's 2-4 seconds. Fine.

But wait, I can vectorize the inner loop! Instead of looping over lags, I can compute the weighted gradient for all j simultaneously using numpy:

```python
def weighted_subgradient(h, weights):
    """Compute weighted sum of subgradients over all lags."""
    n = len(h)
    dx = 2.0 / n
    # weights has length 2n-1, weights[m] for lag k = m-(n-1)
    
    grad = np.zeros(n)
    for m in range(len(weights)):
        if weights[m] < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[:n-k] += weights[m] * (1.0 - h[k:])
            grad[k:n] -= weights[m] * h[:n-k]
        else:
            kk = -k
            grad[kk:] += weights[m] * (1.0 - h[:kk])
            grad[:n-kk] -= weights[m] * h[kk:]
    return grad * dx
```

This is still O(n²) but the inner operations are vectorized (numpy slicing). For 269 lags, each with two O(n) operations, that's 269 * 2 * 135 ≈ 73k basic ops, done in numpy. Should be fast.

One more thought: the problem has a natural structure related to the Fourier transform. The autocorrelation of h is related to |F(h)|². Minimizing the max of the "complementary" autocorrelation (autocorrelation of h*(1-h)) is related to designing a signal with flat spectrum. This connects to Zadoff-Chu sequences and other low-autocorrelation sequences.

For the discrete problem, if we define a[j] = h[j] - 0.5 (mean zero, since ∑h = n/2), then:
h[j](1-h[j+k]) = (0.5+a[j])(0.5-a[j+k]) = 0.25 - 0.5*a[j+k] + 0.5*a[j] - a[j]*a[j+k]

The sum over j (with appropriate range) gives:
C_k * dx ≈ 0.25 * (overlap_length) - 0.5 * ∑a[j+k] + 0.5 * ∑a[j] - ∑a[j]*a[j+k]

The middle terms might not vanish for partial overlaps. The last term is the autocorrelation of a.

To minimize max C_k, we want the autocorrelation of a to be as "flat" as possible (spread out), while keeping the bias terms controlled.

This is essentially a code design problem! The optimal h corresponds to a sequence a with:
- Mean 0 (from the constraint)
- Elements in [-0.5, 0.5] (from h ∈ [0,1])
- Minimum peak sidelobe of the autocorrelation

This is the "minimum peak sidelobe" (MPS) problem in radar signal design! And there are well-known algorithms for it (e.g., the Failure-free adaptive algorithm, PSOS, etc.).

However, our problem is slightly different because:
1. The "autocorrelation" is of the complementary sequence (we have a[j]*a[j+k] subtracted, not just a[j]*a[j+k])
2. There's the bias term from the non-uniform overlap

But the connection suggests that spread-spectrum-like sequences (with flat autocorrelation) should be good starting points.

Let me try a Zadoff-Chu-like initialization:
a[j] = (1/√n) * exp(2πi * j² / 2n) for cyclic sequences... but our sequences are non-cyclic and real-valued.

For real sequences, Kasami or m-sequences might give good starting points.

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def lag_subgradient(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:n] -= h[:n-k]
    else:
        kk = -k
        if n - kk > 0:
            g[kk:] += (1.0 - h[:kk])
            g[:n-kk] -= h[kk:]
    return g * dx

def smooth_max_gradient(h, tau):
    """Gradient of smooth max of correlations."""
    n = len(h)
    C = compute_c5_vector(h)
    C_shifted = C - C.max()
    w = np.exp(tau * C_shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        grad += w[m] * lag_subgradient(h, k)
    return grad, C.max()

def smooth_max_descent(h, n_iters=3000, tau=300, lr_init=0.002, momentum=0.9):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    moment = np.zeros(n)
    lr = lr_init
    
    for it in range(n_iters):
        grad, cmax = smooth_max_gradient(h, tau)
        moment = momentum * moment - lr * grad
        h_new = h + moment
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adapt learning rate
        if c5_new > cmax * 1.001 + 1e-6:
            lr *= 0.5
        elif it % 100 == 0:
            lr = min(lr * 1.01, lr_init)
        
        h = h_new
    
    return best_h, best_c5

def slsqp_polish(h, n, maxiter=2000, ftol=1e-14):
    c5_init = compute_c5_from_h(h)
    
    def obj_func(x):
        return x[-1]
    
    def ineq_func(x):
        h_v = x[:-1]
        t = x[-1]
        n_v = len(h_v)
        dx = 2.0 / n_v
        c = np.correlate(h_v, 1.0 - h_v, mode="full")
        return t - c * dx
    
    def eq_func(x):
        h_v = x[:-1]
        return np.sum(h_v) - len(h_v) / 2.0
    
    def eq_jac_func(x):
        n_v = len(x) - 1
        jac = np.zeros(len(x))
        jac[:n_v] = 1.0
        return jac
    
    x0 = np.concatenate([h, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]
    
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': ftol})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h.copy(), c5_init
