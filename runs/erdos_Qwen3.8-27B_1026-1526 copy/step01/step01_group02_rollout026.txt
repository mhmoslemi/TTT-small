    sum_exp = np.sum(exp_s)
    obj = c_max + np.log(sum_exp) / t
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    m = 2 * n - 1
    for k in range(m):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        # First term: (1-h[j-s]) for valid j
        if s >= 0:
            grad[s:n] += w * dx * (1.0 - h[0:n-s])
        else:
            grad[0:n+s] += w * dx * (1.0 - h[-s:])
        # Second term: -h[j+s] for valid j
        if s >= 0:
            grad[0:n-s] -= w * dx * h[s:]
        else:
            grad[-s:] -= w * dx * h[0:n+s]
    return obj, grad

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    nn = len(h)
    dx = 2.0 / nn
    c = np.correlate(h, 1.0 - h, mode="full")
    return x[-1] - np.max(c * dx)

def slsqp_eq(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0

def slsqp_eq_jac(x):
    nn = len(x) - 1
    j = np.zeros(len(x))
    j[:nn] = 1.0
    return j
