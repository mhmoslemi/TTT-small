def compute_grad(h, alpha):

    corr = get_corr(h)
    _, w = smooth_max_grad(corr, alpha)
    
    grad = np.zeros(n)
    
    for lag in range(0, n):
        # Positive lag
        k_pos = lag + n - 1
        w_val = w[k_pos]
        if w_val > 1e-12:

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        obj, w = smooth_max_grad(get_corr(h), alpha)
        grad = compute_grad(h, alpha)
        
        # Backtracking line search
        for _ in range(20):
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            obj_new, _ = smooth_max_grad(get_corr(h_new), alpha)
            if obj_new < obj - 1e-4 * lr * np.dot(grad, h_new - h):
                break
            lr *= 0.5
        
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    return val, w

    corr = get_corr(h)
    _, w = smooth_max_and_weights(corr, alpha)
    grad = np.zeros(n)
    
    n1 = n - 1
    for lag in range(0, n):

        wk = w[k]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    return val, w

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max_and_weights(corr, alpha)
    grad = np.zeros(n)
    n1 = n - 1
    for lag in range(n):
        k = lag + n1
        wk = w[k]
        if wk > 1e-14:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        if lag > 0:
            k2 = n1 - lag
            wk2 = w[k2]
            if wk2 > 1e-14:
                grad[:n - lag] += wk2 * (1.0 - h[lag:])
                grad[lag:] -= wk2 * h[:n - lag]
    return grad

def projected_gradient_step(h, target_sum, alpha, lr, max_backtrace=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj_cur, _ = smooth_max_and_weights(corr, alpha)
    grad = compute_grad(h, alpha)
    
    for bt in range(max_backtrace):
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        obj_new, _ = smooth_max_and_weights(corr_new, alpha)
        if obj_new <= obj_cur - 1e-4 * lr * np.dot(grad, h_new - h):
            return h_new, obj_new, lr
        lr *= 0.5
    return h_new, obj_new, lr

def run_projected_gradient(h0, target_sum, alphas, n_iters, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    lr = 0.01
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        for it in range(n_iters):
            if time.time() - t_start > time_limit:
                break
            h, obj, lr = projected_gradient_step(h, target_sum, alpha, lr)
            lr = min(lr * 1.05, 0.1)  # slowly increase lr
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    return val, w

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max_and_weights(corr, alpha)
    grad = np.zeros(n)
    n1 = n - 1
    for lag in range(n):
        k = lag + n1
        wk = w[k]
        if wk > 1e-14:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        if lag > 0:
            k2 = n1 - lag
            wk2 = w[k2]
            if wk2 > 1e-14:
                grad[:n - lag] += wk2 * (1.0 - h[lag:])
                grad[lag:] -= wk2 * h[:n - lag]
    return grad

def run_pgd(h0, target_sum, alphas, n_iters_per_alpha, time_limit, t_ref):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02

    for alpha in alphas:
        if time.time() - t_ref > time_limit:
            break
        for it in range(n_iters_per_alpha):
            if time.time() - t_ref > time_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            obj_cur, _ = smooth_max_and_weights(corr, alpha)
            grad = compute_grad(h, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break

            accepted = False
            for bt in range(40):
                h_new = h - lr * grad
                h_new = project_to_sum(h_new, target_sum)
                corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
                obj_new, _ = smooth_max_and_weights(corr_new, alpha)
                ls_c = lr * np.dot(grad, h_new - h)
                if obj_new <= obj_cur - 1e-5 * max(ls_c, 1e-10):
                    h = h_new
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                h = project_to_sum(h - lr * grad, target_sum)
            lr = min(lr * 1.1, 0.2)

            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5

def local_search(h0, target_sum, rng, n_iters, t_ref, time_limit):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    max_stall = 2000

    for it in range(n_iters):

        h_new = h.copy()

        if move_type < 0.3:

            delta = rng.uniform(-0.04, 0.04)
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        elif move_type < 0.6:
            i, j = rng.choice(n, size=2, replace=False)
            amt = rng.uniform(-0.03, 0.03)
            h_new[i] = np.clip(h_new[i] + amt, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - amt, 0.0, 1.0)
        elif move_type < 0.85:
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = h[j], h[i]
        else:
            i1, i2 = sorted(rng.choice(n, size=2, replace=False))
            length = min(i2 - i1, rng.randint(1, 5))
            blk = h_new[i1:i1+length].copy()
            h_new[i1:i1+length] = h_new[i1+1:i1+length+1] if i1+length+1 <= n else h_new[i1:i1+length]
            h_new[i1+length] = blk[0] if i1+length < n else blk[0]

        h_new = project_to_sum(h_new, target_sum)

        if c5_new < best_c5 - 1e-12:

            stall = 0
        elif c5_new < compute_c5(h) - 1e-12:
            h = h_new
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def fourier_init(n, rng, n_modes=15):
    h = np.full(n, 0.5)
    indices = np.arange(n)
    for m in range(1, n_modes + 1):
        freq = 2.0 * np.pi * m / n
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(freq * indices) + b * np.sin(freq * indices)
    h = np.clip(h, 0.0, 1.0)
    return h

def sinusoidal_init(n, rng, freq_range=(1, 8)):
    h = np.full(n, 0.5)
    x = np.linspace(0, 2.0, n, endpoint=False)
    n_harm = rng.randint(1, 4)

        f = rng.uniform(freq_range[0], freq_range[1])
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        h += amp * np.sin(2 * np.pi * f * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src, period=2.0)
    return project_to_sum(h_dst, n_dst / 2.0)
