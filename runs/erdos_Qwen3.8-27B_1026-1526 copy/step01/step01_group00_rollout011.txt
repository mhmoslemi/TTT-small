import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_set(v, target_sum):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    h = np.asarray(h, dtype=float)
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / exp_c.sum()
    f_val = m + np.log(exp_c.sum() / len(exp_c)) / alpha
    
    s_range = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)[:, None]
    s = s_range[None, :]
    
    v1 = (j - s >= 0) & (j - s < n)
    v2 = (j + s >= 0) & (j + s < n)
    
    i1 = np.clip(j - s, 0, n - 1)
    i2 = np.clip(j + s, 0, n - 1)
    
    t1 = ((1.0 - h)[i1] * v1) @ w
    t2 = (h[i2] * v2) @ w
    grad = t1 - t2
    
    return f_val, grad

def slsqp_step(h0, n, target_sum, alpha, maxiter=100):
    h0 = np.asarray(h0, dtype=float).copy()
    
    def obj(x):
        return smooth_max_and_grad(x, alpha)[0]
    
    def grad(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    res = minimize(
        obj, h0, method='SLSQP', jac=grad,
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                     'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project_onto_set(res.x, target_sum)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_onto_set(v, target_sum):
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
            h = np.clip(h, 0.0, 1.0)
        else:

    return h

def smooth_max_and_grad(h, alpha):
    h = np.asarray(h, dtype=float)
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / exp_c.sum()
    f_val = m + np.log(exp_c.sum() / len(exp_c)) / alpha
    s_range = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)[:, None]
    s = s_range[None, :]
    v1 = (j - s >= 0) & (j - s < n)
    v2 = (j + s >= 0) & (j + s < n)
    i1 = np.clip(j - s, 0, n - 1)
    i2 = np.clip(j + s, 0, n - 1)
    t1 = ((1.0 - h)[i1] * v1) @ w
    t2 = (h[i2] * v2) @ w
    grad = t1 - t2
    return f_val, grad

def _eq_jacobian(x):
    return np.ones(len(x))

        constraints=[{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jacobian}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return project_onto_set(res.x, n / 2.0)
