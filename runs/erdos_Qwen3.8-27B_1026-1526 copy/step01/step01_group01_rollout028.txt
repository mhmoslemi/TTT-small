def project_feasible(h, n):

                h = project_feasible(h, n)

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()

    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)

        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def obj_lbfgsb(x, n, beta, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    smax = m + np.log(np.sum(np.exp(shifted))) / beta
    smax *= (2.0 / n)
    sq_diff = (np.sum(x) - n / 2.0) ** 2
    return smax + penalty * sq_diff


def obj_slsqp(x, n, beta):
