            h[mask] = np.maximum(h[mask] - (-diff) / np.sum(mask), 0.0)

def smooth_max_and_probs(c, tau):
    c_max = np.max(c)
    shifted = np.clip((c - c_max) / tau, -100.0, 0.0)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    probs = exp_c / sum_exp
    val = c_max + tau * np.log(sum_exp)
    return val, probs


def grad_smooth_corr(h, probs):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(probs)):
        p = probs[k]
        if p < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                g = (1.0 - 2.0 * h) * dx
            else:
                g = np.zeros(n)
                g[:n - lag] += 1.0 - h[lag:]
                g[lag:] -= h[:n - lag]
                g *= dx
        else:
            L = -lag
            g = np.zeros(n)
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
            g *= dx
        grad += p * g
    return grad


def obj_and_grad(h, tau, mu):
    c = compute_correlations(h)
    val, probs = smooth_max_and_probs(c, tau)
    grad = grad_smooth_corr(h, probs)
    s = np.sum(h) - len(h) / 2.0
    val += mu * s * s
    grad = grad + 2.0 * mu * s
    return val, grad

def slsqp_polish(h_init, n, maxiter=3000):

    def obj(x):
        return x[-1]
    
    def ineq(x):
        h_x = x[:-1]
        t = x[-1]
        c = compute_correlations(h_x)
        return t - np.max(c)
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    def eq_jac(x):
        jac = np.zeros(len(x))
        jac[:n] = 1.0
        return jac
    
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]

                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None and result.success or (hasattr(result, 'x') and result.x is not None):
        h_out = project_feasible(result.x[:-1], n)
        if compute_c5_from_h(h_out) <= c5_init + 1e-10:
            return h_out
    return h_init

    c = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):

            break

            mask = h < 1.0 - 1e-15
            if not np.any(mask):

        else:

            if not np.any(mask):

            h[mask] = np.maximum(h[mask] - (-diff) / np.sum(mask), 0.0)

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += final_diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_probs(c, tau):
    c_max = np.max(c)
    shifted = np.clip((c - c_max) / tau, -100.0, 0.0)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    probs = exp_c / sum_exp
    val = c_max + tau * np.log(sum_exp)
    return val, probs


def grad_smooth_corr(h, probs):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(probs)):
        p = probs[k]
        if p < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                g = (1.0 - 2.0 * h) * dx
            else:
                g = np.zeros(n)
                g[:n - lag] += 1.0 - h[lag:]
                g[lag:] -= h[:n - lag]
                g *= dx
        else:
            L = -lag
            g = np.zeros(n)
            g[L:] += 1.0 - h[:n - L]
            g[:n - L] -= h[L:]
            g *= dx
        grad += p * g
    return grad


def obj_func(h):
    c = compute_correlations(h)
    val, _ = smooth_max_and_probs(c, _curr_tau)
    s = np.sum(h) - len(h) / 2.0
    val += _curr_mu * s * s
    return float(val)


def grad_func(h):
    c = compute_correlations(h)
    _, probs = smooth_max_and_probs(c, _curr_tau)
    grad = grad_smooth_corr(h, probs)
    s = np.sum(h) - len(h) / 2.0
    grad = grad + 2.0 * _curr_mu * s
    return grad


def do_lbfgsb(h, tau, mu, maxiter=300):
    global _curr_tau, _curr_mu
    _curr_tau = tau
    _curr_mu = mu
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    return project_feasible(result.x, n)


def slsqp_polish(h_init, n, maxiter=3000):
    c5_init = compute_c5_from_h(h_init)

    def obj_slsqp(x):
        return x[-1]

    def ineq_slsqp(x):
        h_x = x[:-1]
        t = x[-1]
        c = compute_correlations(h_x)
        return t - np.max(c)

    def eq_slsqp(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac_slsqp(x):
        jac = np.zeros(len(x))
        jac[:n] = 1.0
        return jac

    x0 = np.concatenate([h_init, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.9), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_slsqp},
        {'type': 'eq', 'fun': eq_slsqp, 'jac': eq_jac_slsqp}
    ]
    result = minimize(obj_slsqp, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 1e-10:
            return h_out
    return h_init


def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f_interp = interp1d(x_old, h_old, kind='linear', fill_value="extrapolate")
    h_new = np.clip(f_interp(x_new), 0, 1)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    else:
        h_best = np.full(135, 0.5)
        n_best = 135

    h_best = project_feasible(h_best, n_best)
    best_c5 = compute_c5_from_h(h_best)

    tau_schedule = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003, 0.002,
                    0.0015, 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]

    for i_tau, tau in enumerate(tau_schedule):
        if time.time() - t0 > budget_s - 200:
            break
        remaining_frac = 1.0 - (i_tau / len(tau_schedule))
        maxiter = int(200 + 300 * remaining_frac)
        mu = 500.0 if i_tau < len(tau_schedule) // 2 else 100.0
        h_cur = do_lbfgsb(h_best, tau, mu, maxiter=maxiter)
        c5_cur = compute_c5_from_h(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            h_best = h_cur

    if time.time() - t0 > budget_s - 200:
        h_best = project_feasible(h_best, n_best)
        return h_best, compute_c5_from_h(h_best), n_best

    h_polished = slsqp_polish(h_best, n_best, maxiter=3000)
    c5_polished = compute_c5_from_h(h_polished)
    if c5_polished < best_c5:
        best_c5 = c5_polished
        h_best = h_polished

    for n_try in [270, 405]:
        if time.time() - t0 > budget_s - 300:
            break
        h_interp = interpolate_h(h_best, n_try)
        c5_interp = compute_c5_from_h(h_interp)

        tau_fine = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
        for i_tau, tau in enumerate(tau_fine):
            if time.time() - t0 > budget_s - 150:
                break
            maxiter = int(150 + 200 * (1 - i_tau / len(tau_fine)))
            mu = 500.0 if i_tau < len(tau_fine) // 2 else 100.0
            h_cur = do_lbfgsb(h_interp, tau, mu, maxiter=maxiter)
            c5_cur = compute_c5_from_h(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                h_best = h_cur
                n_best = n_try
            h_interp = h_cur

        h_polished = slsqp_polish(h_best, n_best, maxiter=2000)
        c5_polished = compute_c5_from_h(h_polished)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            h_best = h_polished

    n_perturb = 50
    for i in range(n_perturb):
        if time.time() - t0 > budget_s - 40:
            break

        strategy = i % 5
        frac = i / max(1, n_perturb)

        if strategy == 0:
            scale = 0.04 * (1.0 - frac) + 0.005
            h_pert = h_best + scale * rng.standard_normal(n_best)
        elif strategy == 1:
            h_pert = h_best.copy()
            strength = 0.08 * (1.0 - frac) + 0.01
            mask_up = h_pert > 0.5
            h_pert[mask_up] = np.minimum(1.0, h_pert[mask_up] + strength)
            mask_dn = ~mask_up
            h_pert[mask_dn] = np.maximum(0.0, h_pert[mask_dn] - strength)
        elif strategy == 2:
            h_pert = h_best.copy()
            n_swaps = rng.integers(2, 6)
            for _ in range(n_swaps):
                i1, i2 = rng.choice(n_best, size=2, replace=False)
                h_pert[i1], h_pert[i2] = h_pert[i2], h_pert[i1]
        elif strategy == 3:
            h_pert = h_best.copy()
            center = rng.integers(3, n_best - 3)
            r = rng.integers(3, 8)
            for j in range(max(0, center - r), min(n_best, center + r + 1)):
                h_pert[j] += 0.04 * rng.standard_normal()
        else:
            h_pert = h_best.copy()
            n_regions = rng.integers(2, 5)
            for _ in range(n_regions):
                start = rng.integers(0, n_best - 5)
                length = rng.integers(3, 10)
                shift = 0.03 * rng.standard_normal()
                end = min(n_best, start + length)
                h_pert[start:end] += shift

        h_pert = project_feasible(h_pert, n_best)

        tau = 0.005 + 0.005 * frac
        h_cur = do_lbfgsb(h_pert, tau, 500.0, maxiter=200)
        c5_cur = compute_c5_from_h(h_cur)
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            h_best = h_cur

    if time.time() - t0 < budget_s - 20:
        for _ in range(3):
            if time.time() - t0 > budget_s - 10:
                break
            h_polished = slsqp_polish(h_best, n_best, maxiter=2000)
            c5_polished = compute_c5_from_h(h_polished)
            if c5_polished < best_c5:
                best_c5 = c5_polished
                h_best = h_polished
            else:
                break

    h_best = project_feasible(h_best, n_best)
    best_c5 = compute_c5_from_h(h_best)
    return h_best, best_c5, n_best
