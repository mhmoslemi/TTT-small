def smooth_max_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    log_sum_exp = c_max + np.log(np.sum(np.exp(beta * (c - c_max))))
    smooth_max = log_sum_exp / beta
    penalty = lam * (np.sum(h) - n/2.0) ** 2
    return smooth_max + penalty

def smooth_max_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = np.max(c)
    # Softmax weights
    exp_vals = np.exp(beta * (c - c_max))
    p = exp_vals / np.sum(exp_vals)
    # Gradient of f_k w.r.t. h[j]:
    # f_k = sum_i h[i+k] * (1-h[i]) * dx
    # d f_k / d h[j] = dx * [(1-h[j-k]) if j-k valid - h[j+k] if j+k valid]
    
    grad = np.zeros(n)
    for k_idx in range(len(c)):  # k_idx corresponds to shift
        k = k_idx - (n - 1)  # actual shift
        if p[k_idx] < 1e-10:
            continue
        # For shift k: f_k = sum_i h[i+k]*(1-h[i])*dx
        # h[j] appears as:
        #   h[i+k] when i+k = j => i = j-k, contributes (1-h[j-k])*dx
        #   (1-h[i]) when i = j => contributes -h[j+k]*dx

            ik = j - k  # i when h[i+k] = h[j]
            jk = j + k  # index when (1-h[i]) has i=j, need h[j+k]
            g = 0.0
            if 0 <= ik < n:
                g += (1.0 - h[ik]) * dx
            if 0 <= jk < n:
                g -= h[jk] * dx
            grad[j] += p[k_idx] * g
    
    # Penalty gradient
    grad += 2.0 * lam * (np.sum(h) - n/2.0) * np.ones(n)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            # Distribute evenly among those with room
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, beta, lam, n):
    h = x
    c = compute_correlations(h)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    lse = c_max + np.log(exp_c.sum())
    return lse / beta + lam * (h.sum() - n / 2.0) ** 2

def smooth_grad(x, beta, lam, n):
    h = x
    c = compute_correlations(h)
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    dx = 2.0 / n
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            # h[j+shift] term: j from 0 to n-1-shift
            if n - 1 - shift >= 0:
                j = np.arange(0, n - shift)
                grad[j] -= pk * h[j + shift] * dx
            # (1-h[j-shift]) term: j from shift to n-1
            if shift <= n - 1:
                j = np.arange(shift, n)
                grad[j] += pk * (1.0 - h[j - shift]) * dx
        else:
            s = -shift
            # h[j+shift] = h[j-s]: j from s to n-1
            j = np.arange(s, n)
            grad[j] -= pk * h[j - s] * dx
            # (1-h[j-shift]) = (1-h[j+s]): j from 0 to n-1-s
            j = np.arange(0, n - s)
            grad[j] += pk * (1.0 - h[j + s]) * dx
    
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def lbfgs_smooth(h_init, n, beta, lam, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj, h_init, method='L-BFGS-B',
        jac=smooth_grad, bounds=bounds,
        args=(beta, lam, n),
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50}
    )
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)

def anneal(h_init, n, betas, lam, maxiter_per_beta=300):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    for beta in betas:
        h, c5 = lbfgs_smooth(h, n, beta, lam, maxiter_per_beta)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    return np.clip(h, 0.0, 1.0)


def smooth_obj_and_grad(x, beta, lam, n):
    h = x
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(beta * (c - c_max))
    lse = c_max + np.log(exp_c.sum())
    obj = lse / beta + lam * (h.sum() - n / 2.0) ** 2
    p = exp_c / exp_c.sum()
    c_rev = p[::-1]
    a = 1.0 - 2.0 * h
    corr = np.correlate(c_rev, a, mode='full')
    grad = dx * corr[:n][::-1]
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return obj, grad


def lbfgs_smooth(h_init, n, beta, lam, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smooth_obj_and_grad, h_init, method='L-BFGS-B',
            bounds=bounds, args=(beta, lam, n),
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40}
        )
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    except Exception:
        return h_init, compute_c5_from_h(h_init)


def annealed_optimize(h_init, n, betas, lam, maxiter_per_beta=400, time_limit=None, t0=None):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    for beta in betas:
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        h, c5 = lbfgs_smooth(h, n, beta, lam, maxiter_per_beta)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def generate_structured_start(n, rng):
    x = np.arange(n)
    kind = rng.integers(0, 5)
    if kind == 0:
        f = rng.uniform(0.5, 5.0)
        A = rng.uniform(0.2, 0.45)
        h = 0.5 + A * np.sin(2 * np.pi * f * x / n)
    elif kind == 1:
        f1 = rng.uniform(0.5, 3.0)
        f2 = rng.uniform(3.0, 8.0)
        A1 = rng.uniform(0.15, 0.35)
        A2 = rng.uniform(0.05, 0.2)
        h = 0.5 + A1 * np.sin(2 * np.pi * f1 * x / n) + A2 * np.sin(2 * np.pi * f2 * x / n)
    elif kind == 2:
        h = 0.5 + 0.3 * np.cos(2 * np.pi * rng.uniform(1, 4) * x / n) * np.cos(2 * np.pi * rng.uniform(5, 10) * x / n)
    elif kind == 3:
        h = 0.5 + 0.25 * np.tanh(3.0 * np.sin(2 * np.pi * rng.uniform(1, 5) * x / n))
    else:
        h = 0.5 + 0.3 * np.sign(np.sin(2 * np.pi * rng.uniform(2, 6) * x / n)) * np.exp(-0.5 * (x - n / 2) ** 2 / (n * 0.3))
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
        h0 = project_feasible(h0, n)
    else:
        n = 135
        h0 = project_feasible(np.full(n, 0.5) + 0.1 * rng.standard_normal(n), n)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - 5 - elapsed()

    beta_schedule = [5, 15, 40, 100, 250, 600, 1500]
    lam = 8.0

    if remaining() > 120:
        best_h, best_c5 = annealed_optimize(best_h, n_best, beta_schedule, lam,
                                            maxiter_per_beta=500, time_limit=budget_s - 60, t0=t0)
        n_best = len(best_h)

    ichain_betas = [20, 80, 300, 1000]
    n_ils = 0
    while remaining() > 200:
        scale = 0.03 * (1.0 + 0.02 * n_ils) * np.exp(-0.005 * n_ils)
        scale = max(scale, 0.005)
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        h_new, c5_new = annealed_optimize(h_pert, n_best, ichain_betas, lam,
                                          maxiter_per_beta=250, time_limit=budget_s - 20, t0=t0)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new
            best_c5 = c5_new
            n_best = len(best_h)
        n_ils += 1
        if n_ils > 300:
            break

    if remaining() > 150:
        n_restarts = 0
        while remaining() > 80 and n_restarts < 200:
            rtype = rng.integers(0, 4)
            if rtype == 0:
                h_start = project_feasible(rng.random(n_best), n_best)
            elif rtype == 1:
                h_bin = (rng.random(n_best) < 0.5).astype(float)
                h_start = project_feasible(h_bin + 0.05 * rng.standard_normal(n_best), n_best)
            elif rtype == 2:
                h_start = generate_structured_start(n_best, rng)
            else:
                h_start = project_feasible(best_h + 0.15 * rng.standard_normal(n_best), n_best)
            h_new, c5_new = annealed_optimize(h_start, n_best, [30, 150, 600], 8.0,
                                              maxiter_per_beta=200, time_limit=budget_s - 20, t0=t0)
            if c5_new < best_c5 - 1e-9:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)
            n_restarts += 1

    if remaining() > 100:
        n_alt = n_best * 2
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate')
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_alt)
        h_new, c5_new = annealed_optimize(h_interp, n_alt, [20, 80, 300, 1000], 8.0,
                                          maxiter_per_beta=400, time_limit=budget_s - 30, t0=t0)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new
            best_c5 = c5_new
            n_best = len(best_h)

    if remaining() > 20:
        n_final = len(best_h)
        h_fin, c5_fin = lbfgs_smooth(best_h, n_final, 2000, 20.0, maxiter=500)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
