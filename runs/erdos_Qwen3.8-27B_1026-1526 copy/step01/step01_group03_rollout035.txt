    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    weights = smooth_max_weights(corr, tau)

    # grad[i] = dx * sum_m weights[m] * d_c[m]/dh[i]

    # Phase 1: Coarse-to-fine SLSQP from parent
    tau_schedule = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015]
    for tau in tau_schedule:
        if time.time() - t0 > budget_s * 0.20:
            break
        best_h, best_c5 = slsqp_optimize(best_h, n, tau, max_iter=300, ftol=1e-12)

    # Phase 2: Multi-start exploration
    starts = generate_diverse_starts(n, parent_h, rng)
    for h0 in starts:
        if time.time() - t0 > budget_s * 0.45:
            break
        h0 = project_feasible(h0, n)
        for tau in [0.1, 0.04, 0.015]:
            if time.time() - t0 > budget_s * 0.45:
                break
            h0, c5_tmp = slsqp_optimize(h0, n, tau, max_iter=200, ftol=1e-11)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h0.copy()

    # Phase 3: ES + SLSQP hybrid
    base_noise = 0.08
    no_improve = 0
    es_count = 0
    while time.time() - t0 < budget_s * 0.68:
        es_count += 1
        if no_improve > 30:
            base_noise = min(0.4, base_noise * 1.4)
            no_improve = 0
        elif no_improve > 0:
            base_noise = max(0.005, base_noise * 0.97)
        else:
            base_noise = max(0.003, base_noise * 0.94)

        h_mut = np.clip(best_h + rng.randn(n) * base_noise, 0, 1)
        h_mut = project_feasible(h_mut, n)
        c5_mut = evaluate_c5(h_mut)

        if c5_mut < best_c5:
            best_h = h_mut
            best_c5 = c5_mut
            no_improve = 0
            if es_count % 15 == 0:
                best_h, best_c5 = slsqp_optimize(best_h, n, 0.02, max_iter=80, ftol=1e-10)
        else:
            no_improve += 1

        if es_count % 40 == 0:
            h_pert = project_feasible(best_h + rng.randn(n) * 0.06, n)
            h_pert, c5_pert = slsqp_optimize(h_pert, n, 0.04, max_iter=150, ftol=1e-10)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert

        if es_count % 60 == 0:
            h_pert = project_feasible(best_h + rng.randn(n) * 0.12, n)
            h_pert, c5_pert = slsqp_optimize(h_pert, n, 0.08, max_iter=100, ftol=1e-10)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert

    # Phase 4: Subgradient polishing
    if time.time() - t0 < budget_s * 0.85:
        step_size = 0.05
        eps = 0.02
        for it in range(500):
            if time.time() - t0 > budget_s * 0.85:
                break
            h_new, cmax_val, grad = subgrad_step(best_h, n, step_size, eps)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                step_size = max(0.001, step_size * 0.99)
            else:
                step_size = max(0.0005, step_size * 0.95)
            if step_size < 0.001:
                break

    # Phase 5: Coordinate descent fine-tuning
    if time.time() - t0 < budget_s * 0.93:
        for it in range(2000):
            if time.time() - t0 > budget_s * 0.93:
                break
            idx = it % n
            for delta in [0.005, -0.005, 0.002, -0.002, 0.01, -0.01]:
                h_pert = best_h.copy()
                h_pert[idx] += delta
                h_pert = project_feasible(h_pert, n)
                c5_pert = evaluate_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                    break

    # Phase 6: Try nearby n values
    if best_h is not None and time.time() - t0 < budget_s * 0.95:
        bn = len(best_h)
        n_tries = [bn - 2, bn + 2, bn - 4, bn + 4]
        for n_try in n_tries:
            if time.time() - t0 > budget_s * 0.96:
                break
            if n_try < 20 or n_try == bn:
                continue
            x_old = np.linspace(0, 2, bn, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate')
            h_try = np.clip(f_interp(x_new), 0, 1)
            h_try = project_feasible(h_try, n_try)
            for tau in [0.03, 0.01]:
                if time.time() - t0 > budget_s * 0.96:
                    break
                h_try, c5_try = slsqp_optimize(h_try, n_try, tau, max_iter=150, ftol=1e-12)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try

    # Phase 7: Final SLSQP polish
    if best_h is not None and time.time() - t0 < budget_s * 0.97:
        n_final = len(best_h)
        for tau in [0.008, 0.004, 0.002]:
            if time.time() - t0 > budget_s * 0.98:
                break
            best_h, best_c5 = slsqp_optimize(best_h, n_final, tau, max_iter=400, ftol=1e-14)

    if best_h is None:
        best_h = np.full(n_base, 0.5)
        best_c5 = evaluate_c5(best_h)

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
