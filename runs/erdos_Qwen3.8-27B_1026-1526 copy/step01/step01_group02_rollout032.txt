So c_arr[k_idx] = Σᵢ h[i + (k_idx - (n-1))] * (1-h[i])

Let s = k_idx - (n-1) (the shift).
c_arr[k_idx] = Σᵢ h[i+s] * (1-h[i])

The valid range of i is: 0 ≤ i < n AND 0 ≤ i+s < n.
So i is in [max(0, -s), min(n, n-s) - 1].

Now, ∂c_arr[k_idx]/∂h[j]:
h[j] appears in c_arr[k_idx] in two ways:
1. As h[i+s] = h[j], so i = j-s. Valid when 0 ≤ j-s < n.
   Contribution: (1-h[j-s])
2. As (1-h[i]) where i=j, so h[i] = h[j] = h[i]. 
   Contribution: -h[j+s] (from h[i+s] = h[j+s])
   Valid when 0 ≤ j+s < n.

So ∂c_arr[k_idx]/∂h[j] = (1-h[j-s]) * 1_{0≤j-s<n} - h[j+s] * 1_{0≤j+s<n}

And the gradient of the smoothed objective:
grad[j] = dx * Σ_{k_idx} w[k_idx] * [(1-h[j-s]) * 1_{0≤j-s<n} - h[j+s] * 1_{0≤j+s<n}]

where s = k_idx - (n-1).

Let me reparameterize: let t = s (the shift), ranging from -(n-1) to n-1.
w_t = w[t + (n-1)]

grad[j] = dx * Σ_t w_t * [(1-h[j-t]) * 1_{0≤j-t<n} - h[j+t] * 1_{0≤j+t<n}]

Now let me vectorize. For a fixed t:
- Term 1: (1-h[j-t]) for j such that 0 ≤ j-t < n, i.e., t ≤ j < t+n.
  Combined with 0 ≤ j < n: max(0,t) ≤ j < min(n, t+n).
  If t ≥ 0: j in [t, n-1], h[j-t] = h[0:n-t]
  If t < 0: j in [0, n-1+t], h[j-t] = h[-t:n]

- Term 2: h[j+t] for j such that 0 ≤ j+t < n, i.e., -t ≤ j < n-t.
  Combined with 0 ≤ j < n: max(0,-t) ≤ j < min(n, n-t).
  If t ≥ 0: j in [0, n-1-t], h[j+t] = h[t:n]
  If t < 0: j in [-t, n-1], h[j+t] = h[0:n+t]

So:
```python
def smoothed_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    obj = c_max + np.log(exp_c.sum()) / tau
    
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        t = k_idx - (n - 1)
        wt = w[k_idx] * dx
        if wt < 1e-15:
            continue
        if t >= 0:
            # Term 1: j in [t, n-1], add wt*(1-h[j-t])
            if t < n:
                grad[t:] += wt * (1.0 - h[:n-t])
            # Term 2: j in [0, n-1-t], subtract wt*h[j+t]
            if t < n:
                grad[:n-t] -= wt * h[t:]
        else:
            tt = -t
            # Term 1: j in [0, n-1-tt], add wt*(1-h[j+tt])
            if tt < n:
                grad[:n-tt] += wt * (1.0 - h[tt:])
            # Term 2: j in [tt, n-1], subtract wt*h[j-tt]
            if tt < n:
                grad[tt:] -= wt * h[:n-tt]
    
    return obj, grad
```

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    # Euclidean projection onto {0<=h<=1, sum(h)=target}
    # Find lambda: sum(clip(h-lambda, 0, 1)) = target
    def f(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0)) - target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    # f(lo) >= 0 (all h-lo >= 1, so clip gives 1, sum = n > target = n/2)
    # f(hi) <= 0 (all h-hi <= 0, so clip gives 0, sum = 0 < target)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    def obj_func(x):
        return x[-1]
    
    def ineq_func(x):
        h_x = x[:-1]
        t = x[-1]
        n_x = len(h_x)
        dx = 2.0 / n_x
        c = np.correlate(h_x, 1.0 - h_x, mode="full")
        return t - c * dx

        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

        jac = np.zeros(len(x))
        jac[:n_x] = 1.0
        return jac

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def smoothed_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum
    obj = c_max + np.log(exp_sum) / tau
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        t = k_idx - (n - 1)
        wt = w[k_idx] * dx
        if wt < 1e-14:
            continue
        if t >= 0:
            if t < n:
                grad[t:] += wt * (1.0 - h[:n - t])
                grad[:n - t] -= wt * h[t:]
        else:
            tt = -t
            if tt < n:
                grad[:n - tt] += wt * (1.0 - h[tt:])
                grad[tt:] -= wt * h[:n - tt]
    return obj, grad


def gradient_descent_anneal(h, n, tau_start, tau_end, n_iters, lr_init, deadline):
    h = h.copy()
    tau = tau_start
    if n_iters > 1:
        tau_ratio = (tau_end / tau_start) ** (1.0 / n_iters)
    else:
        tau_ratio = tau_end / tau_start

        obj, grad = smoothed_obj_grad(h, tau)
        tau *= tau_ratio
        grad_proj = grad - np.mean(grad)
        h_new = project_feasible(h - lr * grad_proj, n)

        if c5_check < best_obj:
            best_obj = c5_check
            best_h = h.copy()
        if lr < 1e-12:
            break
    return best_h


def slsqp_polish(h, n, maxiter=2000, deadline=None):
    c5_init = compute_c5_from_h(h)
    
    def obj_func(x):
        return x[-1]
    
    def ineq_func(x):
        h_x = x[:-1]
        t_val = x[-1]
        n_x = len(h_x)
        dx = 2.0 / n_x
        c = np.correlate(h_x, 1.0 - h_x, mode="full")
        return t_val - c * dx
    
    def eq_func(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0
    
    def eq_jac(x):
        n_x = len(x) - 1
        jac = np.zeros(len(x))
        jac[:n_x] = 1.0
        return jac
    
    x0 = np.concatenate([h, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac}
    ]
    
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h.copy(), c5_init

def slsqp_eq_jac(x):

    jac = np.zeros(len(x))
    jac[:n_x] = 1.0
    return jac


def slsqp_polish(h, n, maxiter=2000):
    c5_init = compute_c5_from_h(h)
    x0 = np.concatenate([h, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h.copy(), c5_init


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    deadline = t0 + budget_s

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 135
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    if time.time() < deadline - 400:
        h_poly, c5_poly = slsqp_polish(h0, n, maxiter=5000)
        if c5_poly < best_c5:
            best_h = h_poly
            best_c5 = c5_poly
            n_best = len(best_h)

    n_rounds = 10
    for r in range(n_rounds):
        if time.time() > deadline - 250:
            break
        scale = 0.015 * (r + 1)
        h_start = best_h + scale * rng.standard_normal(n_best)
        h_start = project_feasible(h_start, n_best)
        h_annealed = gradient_descent_anneal(
            h_start, n_best,
            tau_start=10, tau_end=15000,
            n_iters=600, lr_init=0.1,
            deadline=deadline - 250
        )
        c5_a = compute_c5_from_h(h_annealed)
        if c5_a < best_c5 - 1e-10:
            best_h = h_annealed
            best_c5 = c5_a
            n_best = len(best_h)
            if time.time() < deadline - 200:
                h_poly, c5_poly = slsqp_polish(best_h, n_best, maxiter=2000)
                if c5_poly < best_c5:
                    best_h = h_poly
                    best_c5 = c5_poly
                    n_best = len(best_h)

    x = np.linspace(0, 2, n_best, endpoint=False)
    patterns = []
    patterns.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    patterns.append(0.5 + 0.3 * np.sin(4 * np.pi * x))
    patterns.append(0.5 + 0.4 * np.cos(2 * np.pi * x))
    patterns.append(0.5 + 0.3 * np.sin(3 * np.pi * x))
    patterns.append(0.5 + 0.35 * np.cos(2 * np.pi * 1.5 * x))
    patterns.append(np.where((x % 0.5) < 0.25, 0.9, 0.1))
    patterns.append(np.where((x % 0.4) < 0.2, 0.85, 0.15))
    for _ in range(4):
        patterns.append(rng.random(n_best))

    for pat in patterns:
        if time.time() > deadline - 150:
            break
        h_pat = project_feasible(pat, n_best)
        c5_pat = compute_c5_from_h(h_pat)
        if c5_pat > 0.5:
            continue
        h_annealed = gradient_descent_anneal(
            h_pat, n_best,
            tau_start=10, tau_end=8000,
            n_iters=400, lr_init=0.1,
            deadline=deadline - 150
        )
        c5_a = compute_c5_from_h(h_annealed)
        if c5_a < best_c5 - 1e-8:
            best_h = h_annealed
            best_c5 = c5_a
            n_best = len(best_h)
            if time.time() < deadline - 100:
                h_poly, c5_poly = slsqp_polish(best_h, n_best, maxiter=1500)
                if c5_poly < best_c5:
                    best_h = h_poly
                    best_c5 = c5_poly
                    n_best = len(best_h)

    for n_try in [200, 270, 360]:
        if time.time() > deadline - 200:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        h_annealed = gradient_descent_anneal(
            h_interp, n_try,
            tau_start=10, tau_end=10000,
            n_iters=400, lr_init=0.05,
            deadline=deadline - 100
        )
        c5_a = compute_c5_from_h(h_annealed)
        if c5_a < best_c5 - 1e-8:
            best_h = h_annealed
            best_c5 = c5_a
            n_best = n_try
            if time.time() < deadline - 80:
                h_poly, c5_poly = slsqp_polish(h_annealed, n_try, maxiter=2500)
                if c5_poly < best_c5:
                    best_h = h_poly
                    best_c5 = c5_poly
                    n_best = len(best_h)

    if time.time() < deadline - 60:
        for r in range(6):
            if time.time() > deadline - 40:
                break
            h_start = best_h + 0.005 * rng.standard_normal(n_best)
            h_start = project_feasible(h_start, n_best)
            h_annealed = gradient_descent_anneal(
                h_start, n_best,
                tau_start=50, tau_end=50000,
                n_iters=500, lr_init=0.02,
                deadline=deadline - 40
            )
            c5_a = compute_c5_from_h(h_annealed)
            if c5_a < best_c5:
                best_h = h_annealed
                best_c5 = c5_a

    if time.time() < deadline - 20:
        h_poly, c5_poly = slsqp_polish(best_h, n_best, maxiter=5000)
        if c5_poly < best_c5:
            best_h = h_poly
            best_c5 = c5_poly
            n_best = len(best_h)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
