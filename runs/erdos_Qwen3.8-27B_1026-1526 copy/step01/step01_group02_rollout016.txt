from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            # Proportional distribution
            room = 1.0 - h[mask]
            total_room = np.sum(room)
            if total_room < 1e-15:
                break
            frac = min(1.0, diff / total_room)
            h[mask] += room * frac
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            total = np.sum(h[mask])
            if total < 1e-15:
                break
            frac = min(1.0, -diff / total)
            h[mask] -= h[mask] * frac
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Put remainder in the variable with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h

def correlation_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def smooth_objective(h, temp):

        # Stable computation
        shifted = vals - max_v
        return max_v + np.log(np.sum(np.exp(temp * shifted))) / temp
    else:
        return max_v

def smooth_grad(h, temp):
    vals = correlation_values(h)
    max_v = np.max(vals)
    shifted = vals - max_v
    # Clip for numerical stability
    exp_vals = np.exp(np.clip(temp * shifted, -500, 0))
    weights = exp_vals / np.sum(exp_vals)
    
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each lag m (0 to 2n-2), compute gradient of out[m] w.r.t. h
    # out[m] = sum_i h[i + m - n + 1] * (1 - h[i])
    # Let shift = m - n + 1
    # out[m] = sum_i h[i + shift] * (1 - h[i])
    # where valid i: 0 <= i <= n-1 and 0 <= i+shift <= n-1
    
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        shift = m - n + 1
        # Valid range of i
        i_min = max(0, -shift)
        i_max = min(n - 1, n - 1 - shift)
        if i_min > i_max:
            continue
        for i in range(i_min, i_max + 1):
            j = i + shift
            # ∂out[m]/∂h[j] += (1 - h[i])
            # ∂out[m]/∂h[i] += -h[j]
            grad[j] += w * dx * (1.0 - h[i])
            grad[i] += w * dx * (-h[j])
    
    return grad

def slsqp_on_smooth(h_init, temp, maxiter=500):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    
    def obj(x):
        return smooth_objective(x, temp)
    
    def grad(x):
        return smooth_grad(x, temp)
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun}]
    
    result = minimize(obj, h_init, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def slsqp_original(h_init, maxiter=2000):
    n = len(h_init)
    dx = 2.0 / n
    c5_init = compute_c5_from_h(h_init)
    
    def obj_and_t(x):
        h = x[:-1]
        t = x[-1]
        vals = np.correlate(h, 1.0 - h, mode="full") * dx
        return t
    
    def ineq_fun(x):
        h = x[:-1]
        t = x[-1]
        vals = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - vals
    
    def eq_fun(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

    bounds = [(0.0, 1.0)] * n + [(0.001, 1.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun},
        {'type': 'eq', 'fun': eq_fun}
    ]
    
    result = minimize(obj_and_t, x0, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init
