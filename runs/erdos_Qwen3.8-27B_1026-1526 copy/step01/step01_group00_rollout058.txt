def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    n = len(h)
    if abs(np.sum(h) - target_sum) < 1e-12:
        return h
    # Bisection on lambda: find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo = -max(h) - 0.1
    hi = 1.0 - min(h) + 0.1

        if s < target_sum:
            lo = mid
        else:

    diff = target_sum - np.sum(h)

        if np.any(mask):

            h_new = project_to_sum(h_new, target_sum)
        elif r < 0.7:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.04, 0.04)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        else:
            # Block move: change a small consecutive block
            block_size = rng.randint(2, 5)
            start = rng.randint(0, n - block_size)
            deltas = rng.uniform(-0.03, 0.03, block_size)
            deltas -= np.mean(deltas)  # Preserve sum approximately
            h_new = h.copy()
            h_new[start:start+block_size] = np.clip(h_new[start:start+block_size] + deltas, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or (c5_new > c5_cur and rng.random() < np.exp(-(c5_new - c5_cur) / T)):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T = max(T * 0.99995, 0.0001)
    
    return best_h, best_c5

def coord_descent(h, n, target_sum, t_start, budget_s, max_rounds=100):
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if budget_s - (time.time() - t_start) < 10:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t_start) < 8:
                break
            for delta in (-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02):
                h_new = h.copy()
                h_new[i] += delta
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur
