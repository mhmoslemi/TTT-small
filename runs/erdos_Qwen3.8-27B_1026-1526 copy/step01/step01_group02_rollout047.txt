from scipy.optimize import minimize
from scipy.signal import fftconvolve
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = fftconvolve(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            n_mask = int(mask.sum())
            if n_mask == 0:
                break
            step = diff / n_mask
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            n_mask = int(mask.sum())
            if n_mask == 0:
                break
            step = -diff / n_mask
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return h

def equity_fun(x):
    return np.sum(x) - len(x) / 2.0

def smooth_max_obj(h, tau, n):
    dx = 2.0 / n
    c = fftconvolve(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    shifted = tau * (c - max_c)
    # Use log-sum-exp for stability
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return max_c + (1.0 / tau) * log_sum

def slsqp_smooth_max(h0, n, tau, maxiter=2000):
    """SLSQP with smooth max objective"""
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': equity_fun}]
    
    def obj(x):
        return smooth_max_obj(x, tau, n)
    
    result = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out
    return project_feasible(h0, n)

            if time.time() - t0 > time_limit:

        h = slsqp_smooth_max(h, n, tau, maxiter=1500)

        perturb += coeff * np.cos(2 * np.pi * freq * x + phase)

        h_high = np.clip(f_int(x_new), 0, 1)
        h_high = project_feasible(h_high, n_high)

        for tau in [50, 200, 500]:
            if time.time() - t0 > budget_s * 0.92:
                break
            h_high = slsqp_refine(h_high, n_high, tau, maxiter=600)

        c5_high = compute_c5_from_h(h_high)
        if c5_high < best_c5 - 1e-8:
            best_h = h_high
            best_c5 = c5_high
            n = n_high

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
