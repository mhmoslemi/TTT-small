    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:

Let me also add the **LP outer approximation** as a separate phase:

The idea: for a target value c, check if the feasible set {h : f_k(h) ≤ c ∀k, 0≤h≤1, sum(h)=n/2} is non-empty.

We can approximate this with a sequence of LPs by linearizing the bilinear terms. At each iteration:
1. Take current h*
2. Linearize f_k at h*: f_k(h) ≈ f_k(h*) + ∇f_k(h*)ᵀ (h - h*)
3. Solve LP: min 0 subject to linearized constraints, 0≤h≤1, sum(h)=n/2
4. If the LP solution satisfies the original constraints, feasible
5. Otherwise, add the most violated original constraint as a new linear constraint (using its gradient at the LP solution)
6. Repeat

This is essentially **Benders decomposition** and can escape local optima.

import cvxpy as cp

def lp_round(h_star, n, c_target):
    """Check if there exists h with f_k(h) <= c_target for all k."""
    h = cp.Variable(n)
    
    # Linearize f_k at h_star
    # f_k(h) = sum_i h[i] * (1 - h[i+k])
    # grad f_k / dh[j] = 1 - h_star[j+k] - h_star[j-k] (with appropriate boundary)
    
    dx = 2.0 / n
    corr_h_star = np.correlate(h_star, 1.0 - h_star, mode="full")
    
    # Compute gradient of each f_k w.r.t. h
    # f_k = sum_i h[i] * (1-h[i+k]) for valid i
    # df_k/dh[j] = (1-h[j+k]) - h[j-k] (when valid)
    
    # For the full correlation, the offset d = m - (n-1) where m is the index
    # f_d = sum_{i: 0<=i<n, 0<=i+d<n} h[i] * (1-h[i+d])
    
    # grad_d[j] = sum over valid i of d/dh[j] [h[i](1-h[i+d])]
    # = (1-h[j+d]) [when j is valid as i] - h[j-d] [when j-d is valid as i]
    # = (1-h[j+d]) * I(0<=j<n, 0<=j+d<n) - h[j-d] * I(0<=j-d<n, 0<=j<n)
    
    # For the linearized constraint:
    # f_d(h) ≈ f_d(h*) + sum_j grad_d[j] * (h[j] - h*[j])
    # = f_d(h*) - sum_j grad_d[j] * h*[j] + sum_j grad_d[j] * h[j]
    
    # We need: [f_d(h*) - sum_j grad_d[j]*h*[j] + sum_j grad_d[j]*h[j]] * dx <= c_target
    # i.e., sum_j grad_d[j] * h[j] <= c_target/dx - f_d(h*) + sum_j grad_d[j]*h*[j]
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    for d in range(-(n-1), n):
        # Compute grad_d
        grad = np.zeros(n)
        # (1-h[j+d]) for valid j
        valid_upper = (0 <= d)  # j+d < n means j < n-d
        # For offset d:
        # Term 1: (1-h[j+d]) for j such that 0<=j<n and 0<=j+d<n
        # Term 2: -h[j-d] for j such that 0<=j-d<n and 0<=j<n (i.e., d<=j<n+d)
        
        for j in range(n):
            g = 0.0
            # Term 1: j valid as i, j+d valid
            if 0 <= j + d < n:
                g += 1.0 - h_star[j + d]
            # Term 2: j-d valid as i, j valid
            if 0 <= j - d < n:
                g -= h_star[j - d]
            grad[j] = g
        
        # RHS
        rhs = c_target / dx - corr_h_star[d + (n-1)] + np.dot(grad, h_star)
        constraints.append(cp.dot(grad, h) <= rhs)
    
    prob = cp.Problem(cp.Minimize(0), constraints)
    try:
        prob.solve(solver=cp.GLPK)
        if h.value is not None:
            return h.value
    except:
        pass
    return None

def aug_lag_optimize(h0, n, beta, max_outer=50, max_inner=200):
    h = h0.copy()
    lam = 0.0
    rho = 1e4
    
    for outer in range(max_outer):
        # Inner: minimize smooth_max + lam*(sum(h)-n/2) + 0.5*rho*(sum(h)-n/2)^2
        def inner_obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            dx = 2.0 / n
            m = corr.max()
            sh = np.clip(beta * (corr - m), -500.0, 500.0)
            val = (m + np.log(np.exp(sh).sum()) / beta) * dx
            s = x.sum() - n * 0.5
            val += lam * s + 0.5 * rho * s * s
            return val
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(inner_obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_inner, 'ftol': 1e-14})
        h = res.x
        
        # Update multiplier
        s = h.sum() - n * 0.5
        lam += rho * s
        
        if abs(s) < 1e-10:
            break
    
    return project_feasible(h, n)
