import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(vals, tau):
    v = vals / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_max_grad(corr, h, tau):
    n = len(h)
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    weights = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    ar = np.arange(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-6:
            continue
        offset = k - (n - 1)
        valid_f = (ar + offset >= 0) & (ar + offset < n)
        valid_s = (ar - offset >= 0) & (ar - offset < n)
        hf = h[np.clip(ar + offset, 0, n - 1)]
        hs = h[np.clip(ar - offset, 0, n - 1)]
        grad += w * ((1.0 - hf) * valid_f - hs * valid_s)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(result - 0.5))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def pgd_smooth_max(h, n, tau_start, n_steps, decay_rate, deadline):
    h_curr = h.copy()
    tau = tau_start
    best_h = h_curr.copy()
    best_c5 = evaluate_c5(h_curr)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_curr)
        sm = smooth_max(corr, tau)
        grad = smooth_max_grad(corr, h_curr, tau)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        alpha = 1.0 / (gn * gn + 1e-8)
        for _ in range(40):
            h_new = project_feasible(h_curr - alpha * grad, n)
            sm_new = smooth_max(compute_corr(h_new), tau)
            if sm_new < sm - 1e-14:
                h_curr = h_new
                break
            alpha *= 0.5
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
        tau *= decay_rate
    return best_h, best_c5

def objective_fn(h, tau):
    return smooth_max(compute_corr(h), tau)

def constraint_fn(h, n):
    return np.sum(h) - n / 2.0

def slsqp_refine(h, n, tau, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn, 'args': (n,)}]
    try:
        res = minimize(objective_fn, h, args=(tau,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def generate_starts(n, parent_h, rng, best_h):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
        else:
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            starts.append(np.clip(np.interp(x, xp, parent_h), 0, 1))
        hp = parent_h + rng.randn(len(parent_h)) * 0.04
        if len(parent_h) == n:
            starts.append(np.clip(hp, 0, 1))
        else:
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            starts.append(np.clip(np.interp(x, xp, hp), 0, 1))
    if best_h is not None and len(best_h) != n:
        xb = np.linspace(0, 2, len(best_h), endpoint=False)
        starts.append(np.clip(np.interp(x, xb, best_h), 0, 1))
    elif best_h is not None:
        starts.append(best_h.copy())
    for freq in [1.2, 1.5, 1.8, 2.1, 2.5, 2.9, 3.3, 3.8]:
        amp = 0.28 + 0.12 * rng.rand()
        phase = rng.uniform(0, 2 * np.pi)

    for _ in range(4):
        starts.append(rng.rand(n))
    for period in [0.35, 0.5, 0.65, 0.8, 1.1]:
        saw = np.mod(x / period, 2.0)
        saw = np.where(saw < 1.0, saw, 2.0 - saw)
        starts.append(np.clip(0.15 + 0.7 * saw, 0, 1))
    knots = np.sort(rng.uniform(0.1, 1.9, size=4))
    vals = rng.uniform(0.1, 0.9, size=4)
    starts.append(np.clip(np.interp(x, knots, vals), 0, 1))
    return starts
