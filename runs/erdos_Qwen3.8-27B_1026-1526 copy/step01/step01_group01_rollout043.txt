def compute_c5(h):

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0, 1)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - result))
            result[idx] = min(result[idx] + diff, 1.0)
        else:

            result[idx] = max(result[idx] + diff, 0.0)

    return m + np.log(np.sum(np.exp(shifted))) / beta

    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    j = k_star - (n - 1)

        coeffs[n - jj:] = 1.0

    half_n = n // 2

    if n % 2 != 0:

        h_new[sorted_idx[half_n]] = 0.5

    if j >= 0:

        coeffs[j:] = 1.0 - h[:n - j]
    else:
        jj = -j
        coeffs = np.ones(n)
        coeffs[:n - jj] = 1.0 - h[jj:]

    half_n = n // 2
    sorted_idx = np.argsort(coeffs)
    h_new = np.zeros(n)
    h_new[sorted_idx[:half_n]] = 1.0
    if n % 2 != 0:
        h_new[sorted_idx[half_n]] = 0.5
    return h_new


def damped_best_response(h, n, alpha):
    h_br = best_response_step(h, n)
    h_new = alpha * h_br + (1.0 - alpha) * h
    return project_feasible(h_new, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 47

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float)
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.full(47, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 47

    n = best_n

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP polish from initial
    if elapsed() < budget_s - 150:
        h_opt, c5_opt = slsqp_optimize(best_h, n, 60.0, 400)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: FISTA with multiple betas
    if elapsed() < budget_s - 350:
        for beta_val in [40, 80, 150, 300, 600]:
            if elapsed() > budget_s - 320:
                break
            h_cur = best_h.copy()
            for sub in range(3):
                if elapsed() > budget_s - 310:
                    break
                h_opt, c5_opt = fista_optimize(h_cur, n, beta_val, 300, n / 3.0)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                h_cur = best_h.copy()

    # Phase 3: Multi-start with perturbations and best response
    no_improve = 0
    attempt = 0
    while elapsed() < budget_s - 400 and no_improve < 50:
        attempt += 1
        if rng.random() < 0.25:
            sigma = 0.25
        else:
            sigma = 0.04 + 0.08 * (0.92 ** min(no_improve, 20))
        h_pert = project_feasible(best_h + rng.normal(0, sigma, n), n)

        h_br = damped_best_response(h_pert, n, 0.15)
        c5_br = compute_c5(h_br)
        if c5_br < best_c5:
            best_h, best_c5 = h_br, c5_br
            no_improve = 0
            continue

        h_opt, c5_opt = fista_optimize(h_pert, n, 120, 200, n / 3.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

    # Phase 4: Resolution changes
    if elapsed() < budget_s - 250:
        for n_try in [47, 94, 284, 380]:
            if elapsed() > budget_s - 200:
                break
            if n_try == best_n:
                continue
            h_interp = interpolate_h(best_h, best_n, n_try)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c5_opt = slsqp_optimize(h_interp, n_try, 50.0, 250)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try
                n = n_try

    # Phase 5: Additional FISTA rounds at current resolution
    if elapsed() < budget_s - 100:
        for beta_val in [200, 400, 800]:
            if elapsed() > budget_s - 70:
                break
            h_opt, c5_opt = fista_optimize(best_h, n, beta_val, 400, n / 3.0)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s - 40:
        h_opt, c5_opt = slsqp_optimize(best_h, n, 200.0, 500)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if elapsed() < budget_s - 20:
        h_opt, c5_opt = slsqp_optimize(best_h, n, 500.0, 300)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
