import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-15:
                break
            alpha = min(diff / total, 1.0)
            h[mask] += alpha * room
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            total = room.sum()
            if total < 1e-15:
                break
            alpha = min(-diff / total, 1.0)
            h[mask] -= alpha * room
    return np.clip(h, 0.0, 1.0)


def smooth_max_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    shifted = tau * (c - m)
    shifted -= shifted.max()
    return m + np.log(np.sum(np.exp(shifted))) / tau


def smooth_max_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    m = c.max()
    logits = tau * (c - m)
    logits -= logits.max()
    w = np.exp(logits)
    w /= w.sum()
    grad = np.zeros(n)
    active = w > 1e-12
    for k in np.where(active)[0]:
        s = k - (n - 1)
        wk = w[k]
        if s == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            for i in range(n):
                j = i - s
                if 0 <= j < n:
                    grad[i] += wk * (1.0 - h[j])
                j2 = i + s
                if 0 <= j2 < n:
                    grad[i] -= wk * h[j2]
    grad *= dx
    return grad


def fista_smooth_max(h0, tau, n_iter, lr0=0.5, lam=50.0):
    n = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t_mom = 1.0
    for it in range(n_iter):
        if it % 5000 == 0 and n > 50:
            lr = lr0 / (1.0 + 0.01 * it)
        else:
            lr = lr0 / (1.0 + 0.01 * it)
        grad = smooth_max_gradient(h, tau)
        csum = h.sum() - n / 2.0
        grad += 2.0 * lam * csum
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        obj_new = smooth_max_value(h_new, tau) + lam * (h_new.sum() - n / 2.0) ** 2
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        h_mom = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        h_mom = project_feasible(h_mom, n)
        obj_mom = smooth_max_value(h_mom, tau) + lam * (h_mom.sum() - n / 2.0) ** 2
        if obj_mom <= obj_new + 1e-12:
            h = h_mom
        else:
            h = h_new
            t_mom = 1.0
        t_mom = t_new
    return h


def slsqp_polish(h_init, n, c5_bound, maxiter=2000):
    c5_init = compute_c5_from_h(h_init)
    t_val = max(c5_init, c5_bound) + 0.02
    x0 = np.concatenate([h_init, [t_val]])
    nvars = n

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * dx

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    def eq_jac(x):
        j = np.zeros(nvars + 1)
        j[:nvars] = 1.0
        return j

    bounds = [(0.0, 1.0)] * nvars + [(0.001, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def binary_local_search(h_init, n, max_swaps=5000):
    h = h_init.copy()
    h = (h > 0.5).astype(float)
    if n % 2 == 0:
        target_ones = n // 2
    else:
        target_ones = n // 2
    while np.sum(h > 0.5) > target_ones:
        idx = np.where(h == 1.0)[0]
        h[idx[np.argmax(h[idx] - np.random.default_rng(0).random(len(idx)))]] = 0.0
    while np.sum(h > 0.5) < target_ones:
        idx = np.where(h == 0.0)[0]
        if len(idx) == 0:
            break
        h[idx[0]] = 1.0
    if n % 2 == 1:
        h[np.argmin((h - 0.5) ** 2)] = 0.5
    h = project_feasible(h, n)
    c5_cur = compute_c5_from_h(h)
    for _ in range(max_swaps):
        ones = np.where(h >= 0.99)[0]
        zeros = np.where(h <= 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.default_rng(0).integers(0, len(ones))]
        j = zeros[np.random.default_rng(0).integers(0, len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5_from_h(h)
        if c5_new < c5_cur - 1e-10:
            c5_cur = c5_new
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5_cur


def lfbgsb_smooth_max(h0, tau, n, maxiter=500, lam=100.0):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        m = c.max()
        s = tau * (c - m)
        s -= s.max()
        val = m + np.log(np.sum(np.exp(s))) / tau
        val += lam * (x.sum() - n / 2.0) ** 2
        return val

    def grad(x):
        g = smooth_max_gradient(x, tau)
        g += 2.0 * lam * (x.sum() - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14,
                                           'gtol': 1e-10})
    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h0, compute_c5_from_h(h0)
