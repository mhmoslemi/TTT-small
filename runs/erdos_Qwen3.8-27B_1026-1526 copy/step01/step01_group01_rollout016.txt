    else:
        kk = -k
        for i in range(n - kk):
            grad[i] += (1.0 - h[i + kk]) * dx
            grad[i + kk] -= h[i] * dx
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def projected_subgradient(h, n, n_steps, step_scale):
    h_cur = h.copy()
    best_local = compute_c5(h_cur)
    best_local_h = h_cur.copy()
    step = step_scale
    for it in range(n_steps):
        h_new = subgradient_step(h_cur, n, step)
        c5_new = compute_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            best_local_h = h_new.copy()
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.9
            if step < 1e-6:
                step = 1e-6
        h_cur = subgradient_step(h_cur, n, step)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_local:
            best_local = c5_cur
            best_local_h = h_cur.copy()
    return best_local_h, best_local


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.ones(200) * 0.5
        best_n = 200
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    def check_time(remaining):
        return elapsed() < budget_s - remaining

    # Phase 1: Quick optimization at initial resolution
    if check_time(80):
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, 400, 30.0, 200.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Subgradient refinement
    if check_time(100):
        h_sg, c5_sg = projected_subgradient(best_h, best_n, 2000, 0.02)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 3: Multi-resolution sweep
    resolutions = sorted(set([
        max(best_n, 100),
        150, 200, 250, 300, 350, 400, 450, 500
    ]))

    for n_try in resolutions:
        if not check_time(120):
            break

        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)

        starts = [h_interp]
        starts.append(make_binary_start(n_try, rng))
        starts.append(make_block_start(n_try, rng, rng.randint(6, 16)))
        starts.append(make_periodic_start(n_try, rng.randint(5, 20), rng.uniform(0, 2), rng))
        starts.append(make_pulse_start(n_try, rng, rng.randint(8, 24)))
        noise = rng.normal(0, 0.2, n_try)
        starts.append(project_feasible(h_interp + noise, n_try))

        for si, h_start in enumerate(starts):
            if not check_time(60):
                break
            beta_val = 40.0 if si == 0 else 30.0
            pen_val = 500.0
            h_opt, c5_opt = optimize_lbfgs(h_start, n_try, 350, beta_val, pen_val)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 4: Subgradient on best solution
    if check_time(80):
        h_sg, c5_sg = projected_subgradient(best_h, best_n, 3000, 0.015)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Phase 5: Higher resolution push
    if check_time(150):
        n_high = min(best_n + 100, 600)
        h_interp = interpolate_h(best_h, best_n, n_high)
        h_interp = project_feasible(h_interp, n_high)
        h_opt, c5_opt = optimize_lbfgs(h_interp, n_high, 400, 50.0, 1000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_high

    # Phase 6: Final polishing with increasing beta
    if check_time(60):
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, 400, 80.0, 2000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if check_time(40):
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, 300, 150.0, 5000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: One more subgradient pass
    if check_time(30):
        h_sg, c5_sg = projected_subgradient(best_h, best_n, 2000, 0.01)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg, c5_sg

    # Final check
    if check_time(15):
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, 200, 200.0, 10000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
