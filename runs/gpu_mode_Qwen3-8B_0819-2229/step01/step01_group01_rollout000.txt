import torch
import triton
import triton.language as tl
from torch import Tensor
from typing import Tuple, Dict

def custom_kernel(data: Tuple[Tensor, Tensor, Dict[str, Tensor], Dict]) -> Tensor:
    """
    Triton implementation of TriMul for AlphaFold3.

    This kernel efficiently implements the "outgoing" version of the TriMul operation,
    which is a core operation in protein structure prediction models. The kernel
    leverages Triton's capabilities to efficiently handle the N^3 computation, 
    optimizing memory access and computation for high performance on GPU.

    Args:
        data: Tuple of (input: Tensor, mask: Tensor, weights: Dict, config: Dict)
            - input: [bs, seq_len, seq_len, dim]
            - mask: [bs, seq_len, seq_len]
            - weights: Dictionary with model weights
            - config: Dictionary with model configuration

    Returns:
        output: Processed tensor [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    
    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Initialize output tensor
    output = torch.zeros(bs, seq_len, seq_len, dim, device=device)

    # Define Triton kernel
    @triton.jit
    def triton_kernel(
        input_ptr, mask_ptr, left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, to_out_norm_bias, to_out_weight,
        out_ptr, bs, seq_len, dim, hidden_dim, BLOCK: tl.constexpr, 
        mask_val: tl.constexpr
    ):
        # Calculate grid shape
        pid = tl.program_id(0)
        m_idx = pid // (seq_len // BLOCK)
        n_idx = pid % (seq_len // BLOCK)
        i = m_idx * BLOCK + tl.arange(0, BLOCK)
        j = n_idx * BLOCK + tl.arange(0, BLOCK)
        mask_offsets = i[:, None] * seq_len + j[None, :]
        mask = tl.load(mask_ptr + mask_offsets, mask_val)

        # Compute left and right projections
        left = tl.load(input_ptr + i[:, None, None] * seq_len * seq_len + j[None, :, None] * seq_len + tl.arange(0, dim), mask=mask, other=0.0)
        left = tl.dot(left, left_proj_weight)
        right = tl.load(input_ptr + i[:, None, None] * seq_len * seq_len + j[None, :, None] * seq_len + tl.arange(0, dim), mask=mask, other=0.0)
        right = tl.dot(right, right_proj_weight)

        # Compute gates
        left_gate = tl.load(input_ptr + i[:, None, None] * seq_len * seq_len + j[None, :, None] * seq_len + tl.arange(0, dim), mask=mask, other=0.0)
        left_gate = tl.dot(left_gate, left_gate_weight).sigmoid()
        right_gate = tl.load(input_ptr + i[:, None, None] * seq_len * seq_len + j[None, :, None] * seq_len + tl.arange(0, dim), mask=mask, other=0.0)
        right_gate = tl.dot(right_gate, right_gate_weight).sigmoid()
        out_gate = tl.load(input_ptr + i[:, None, None] * seq_len * seq_len + j[None, :, None] * seq_len + tl.arange(0, dim), mask=mask, other=0.0)
        out_gate = tl.dot(out_gate, out_gate_weight).sigmoid()

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # Compute out
        out = tl.einsum('ikd,jkd->ijd', left, right)
        out = tl.dot(out, to_out_norm_weight) + to_out_norm_bias
        out = out * out_gate
        out = tl.dot(out, to_out_weight)

        # Write back to output
        out_offsets = i[:, None] * seq_len * seq_len + j[None, :] * seq_len + tl.arange(0, dim)
        tl.store(out_ptr + out_offsets, out)

    # Launch kernel
    BLOCK = 128
    grid = (triton.cdiv(seq_len * seq_len, BLOCK),)
    triton_kernel[grid](
        input_tensor, mask.unsqueeze(-1), left_proj_weight, right_proj_weight,
        left_gate_weight, right_gate_weight, out_gate_weight, to_out_norm_weight, 
        to_out_norm_bias, to_out_weight, output,
        bs, seq_len, dim, hidden_dim, BLOCK, 0.0
    )

    # Return output
    return output
