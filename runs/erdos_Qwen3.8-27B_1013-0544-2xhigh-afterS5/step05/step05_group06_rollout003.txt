def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    """Compute Jacobian F[i, m] = d(g[m])/d(h[i])"""
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for i in range(n):
        for m_idx in range(2 * n - 1):
            d = dm[m_idx]
            # h[i] contributes to g[m] through:
            # (1 - h[i+d]) if i+d in [0, n)  -> coefficient (1-h[i+d])
            # -h[i-d] if i-d in [0, n)       -> coefficient -h[i-d]
            j1 = i + d
            j2 = i - d
            if 0 <= j1 < n:
                F[i, m_idx] += (1.0 - h[j1])
            if 0 <= j2 < n:
                F[i, m_idx] -= h[j2]
    return F

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)  # (2n-1,)
    # For each i and each d:
    # F[i, m] = (1-h[i+d]) * [0<=i+d<n] - h[i-d] * [0<=i-d<n]
    # where d = dm[m]
    
    i_idx = np.arange(n)[:, None]  # (n, 1)
    d_idx = dm[None, :]  # (1, 2n-1)
    
    j1 = i_idx + d_idx  # (n, 2n-1)
    j2 = i_idx - d_idx  # (n, 2n-1)
    
    F = np.zeros((n, 2 * n - 1))

    F[mask1] += (1.0 - h[j1[mask1]])
    F[mask2] -= h[j2[mask2]]
    return F

    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(10000):
        if time.time() > deadline:
            break

        F = compute_F(h, n)

        g_top = g[top_idx]

    F[r[v1], m1[v1]] -= delta
    F[r[v2], m2[v2]] -= delta

def batch_pair_descent(h, n, deadline, B=1000, K=10):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    F = compute_F(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    n_lags = 2 * n - 1
    
    stall = 0
    while time.time() < deadline:
        # Pick B random pairs
        i_idx = np.random.randint(0, n, B)
        j_idx = np.random.randint(0, n, B)
        # Exclude i == j
        same = i_idx == j_idx
        if same.any():
            i_idx[same] = (i_idx[same] + 1) % n
            # Re-check
            same2 = i_idx == j_idx
            i_idx[same2] = (i_idx[same2] + 1) % n
        
        S = F[i_idx] - F[j_idx]  # (B, n_lags)
        
        # Top K lags
        top = np.argsort(g)[::-1][:K]
        g_top = g[top]  # (K,)
        S_top = S[:, top]  # (B, K)
        
        # Find optimal delta for each pair
        d_lo = np.maximum(-h[i_idx], h[j_idx] - 1.0)  # (B,)
        d_hi = np.minimum(1.0 - h[i_idx], h[j_idx])  # (B,)
        
        # Evaluate at boundaries
        vals_lo = np.max(g_top[None, :] + d_lo[:, None] * S_top, axis=1)  # (B,)
        vals_hi = np.max(g_top[None, :] + d_hi[:, None] * S_top, axis=1)  # (B,)
        
        best_deltas = np.where(vals_lo < vals_hi, d_lo, d_hi)
        best_vals = np.minimum(vals_lo, vals_hi)
        
        # Breakpoints between top K values
        # dc[b, k1, k2] = (g_top[k2] - g_top[k1]) / (S_top[b,k1] - S_top[b,k2])
        g_diff = g_top[None, None, :] - g_top[None, :, None]  # (1, K, K)
        S_diff = S_top[:, :, None] - S_top[:, None, :]  # (B, K, K)
        valid = np.abs(S_diff) > 1e-15
        dc = np.where(valid, g_diff / np.where(valid, S_diff, 1.0), 0.0)  # (B, K, K)
        
        # Check range
        lo = d_lo[:, None, None]
        hi = d_hi[:, None, None]
        in_range = valid & (dc > lo) & (dc < hi)
        
        # Evaluate f at breakpoints: f[b,k1,k2,k3] = g_top[k3] + dc[b,k1,k2]*S_top[b,k3]
        # Shape: (B, K, K, K)
        dc_e = dc[:, :, :, None]  # (B, K, K, 1)
        S_e = S_top[:, None, None, :]  # (B, 1, 1, K)
        g_e = g_top[None, None, None, :]  # (1, 1, 1, K)
        f_vals = g_e + dc_e * S_e  # (B, K, K, K)
        f_max = np.max(f_vals, axis=3)  # (B, K, K)
        f_max = np.where(in_range, f_max, np.inf)
        
        # Best breakpoint for each b
        flat = f_max.reshape(B, -1)
        best_k = np.argmin(flat, axis=1)
        bp_vals = flat[np.arange(B), best_k]
        
        dc_flat = dc.reshape(B, -1)
        bp_deltas = dc_flat[np.arange(B), best_k]
        
        improve = bp_vals < best_vals
        best_vals[improve] = bp_vals[improve]
        best_deltas[improve] = bp_deltas[improve]
        
        # Apply best update
        best_b = np.argmin(best_vals)
        if best_vals[best_b] < best_c - 1e-12:
            d = float(best_deltas[best_b])
            ii = int(i_idx[best_b])
            jj = int(j_idx[best_b])
            h[ii] += d
            h[jj] -= d
            g += d * S[best_b]
            update_F(F, ii, d, n, dm)
            update_F(F, jj, -d, n, dm)
            best_c = float(g.max())
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
    
    h = project_feasible(h, n)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    g = compute_g(h_bin, n)
    F = compute_F(h_bin, n)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    while time.time() < deadline:
        improved = False
        # Try all swaps
        best_swap_c = best_c
        best_swap = None
        for z in zeros:
            for o in ones:
                S = -F[o] + F[z]  # change in g when h[z]=1, h[o]=0
                new_max = np.max(g + S)
                if new_max < best_swap_c - 1e-14:
                    best_swap_c = new_max
                    best_swap = (z, o)
        if best_swap:
            z, o = best_swap
            h_bin[z] = 1.0
            h_bin[o] = 0.0
            g += -F[o] + F[z]
            # Update F
            dm = np.arange(2 * n - 1) - (n - 1)
            update_F(F, z, 1.0, n, dm)
            update_F(F, o, -1.0, n, dm)
            best_c = best_swap_c
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            improved = True
        if not improved:
            break
    
    return h_bin, best_c

def softmax_slsqp(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    np.random.seed(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP (15% time)
    p1 = t0 + 0.15 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [1000, 5000, 20000, 100000, 500000]:
        if time.time() > p1:
            break
        h = softmax_slsqp(h, n, alpha, p1, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 2: Batch pair descent (55% time)
    p2 = t0 + 0.70 * (budget_s - 3)
    h, c = batch_pair_descent(best_h.copy(), n, p2, B=1000, K=10)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    # Phase 3: Binary polish (15% time)
    p3 = t0 + 0.85 * (budget_s - 3)
    if time.time() < p3 - 10:
        hb, cb = binary_polish(best_h.copy(), n, p3 - 10)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    
    # Phase 4: SLSQP polish (10% time)
    if time.time() < dl - 30:
        h = softmax_slsqp(best_h.copy(), n, 1000000, dl - 30, maxiter=300)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Phase 5: Final pair descent
    if time.time() < dl - 5:
        h, c = batch_pair_descent(best_h.copy(), n, dl - 3, B=2000, K=15)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
