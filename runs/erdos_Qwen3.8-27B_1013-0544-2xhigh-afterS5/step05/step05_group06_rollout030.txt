import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    F[j - dm[v4], m_idx[v4]] += delta

def slsqp_pass(h, n, alphas, deadline, maxiter_per=300):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    current = h.copy()
    best_c = compute_c5(current, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x, a=alpha):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            shifted = a * (g - m)
            sm = shifted.max()
            return m + (sm + np.log(np.exp(shifted - sm).sum())) / a
        try:
            res = minimize(obj, current, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_per, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            c = compute_c5(new_h, n)
            if c < best_c:
                best_c = c
                current = new_h
        except Exception:
            pass
    return current, best_c

def pair_descent(h, n, deadline, max_sweeps=500, K=8, rng=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx_arr = np.arange(2 * n - 1)
    stall_count = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        sweep_improved = 0
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for ii in range(n):
            i = int(order[ii])
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                candidates = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    d = float(np.clip(d, d_lo, d_hi))
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-13:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    update_F(F, i, j, best_delta, n, dm)
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    sweep_improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if sweep_improved == 0:
            stall_count += 1
            if stall_count > 4:
                break
        else:
            stall_count = 0
    return best_h, best_c

def binary_polish(h, n, deadline, max_rounds=1000):
    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for z in zeros[:]:
            if time.time() > deadline:
                break
            for o in ones[:]:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(o)
                    ones.append(z)
                    zeros.remove(z)
                    zeros.append(o)
                    improved = True
                    break
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
