def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    s_max = sh.max()
    exp_sh = np.exp(sh - s_max)
    p = exp_sh / exp_sh.sum()  # softmax weights, length 2n-1
    
    # Gradient: sum_k p[k] * dc_k/dx
    # dc_k/dx[j] = dx * [(1-x[j+offset]) * 1[valid] - x[j-offset] * 1[valid]]
    # where offset = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        w = p[k] * dx
        # Term 1: (1-x[j+offset]) for j where 0<=j+offset<n
        j_lo = max(0, -offset)
        j_hi = min(n, n - offset)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1 - x[j_lo + offset:j_hi + offset])
        # Term 2: -x[j-offset] for j where 0<=j-offset<n
        j_lo2 = max(0, offset)
        j_hi2 = min(n, n + offset)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * x[j_lo2 - offset:j_hi2 - offset]
    
    f = m + (s_max + np.log(exp_sh.sum())) / alpha
    return f, grad

    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = soft_obj_jac(h, n, alpha)
        # Sort indices by gradient (ascending)
        order = np.argsort(g)
        improved = False
        # Try moving from high-gradient to low-gradient
        n_try = min(25, n)
        for ii in range(n_try):
            if time.time() > deadline:
                return h, best_c
            i = order[ii]
            for jj in range(n - 1, max(n - 1 - n_try, ii), -1):
                j = order[jj]
                if i == j:
                    continue
                for d in [0.01, 0.005, 0.002]:
                    if h[i] + d <= 1.0 + 1e-9 and h[j] - d >= -1e-9:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                            break
                        else:
                            h[i] -= d
                            h[j] += d
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Try reverse direction
            for ii in range(n - 1, max(n - 1 - n_try, -1), -1):
                if time.time() > deadline:
                    return h, best_c
                i = order[ii]
                for jj in range(n_try):
                    j = order[jj]
                    if i == j:
                        continue
                    for d in [0.005, 0.002, 0.001]:
                        if h[i] - d >= -1e-9 and h[j] + d <= 1.0 + 1e-9:
                            h[i] -= d
                            h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                                break
                            else:
                                h[i] += d
                                h[j] -= d
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c

def multi_start_refine(h0, n, deadline, rng, n_starts=4):
    """Try several perturbations and keep the best."""
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    for r in range(n_starts):
        if time.time() > deadline - 20:
            break
        scale = 0.02 + 0.04 * r
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(80, (deadline - time.time()) * 0.3)
        if alloc < 15:
            break
        for a in [1000, 10000]:
            if time.time() > time.time() + alloc - 5:
                break
            ht = slsqp_opt(hp, n, a, maxiter=1000)
            c = compute_c5(ht, n)
            if c < compute_c5(hp, n):
                hp = ht
        hp, c = guided_pair_search(hp, n, time.time() + min(20, alloc / 2))
        if c < best_c:
            best_c = c
            best_h = hp
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    except Exception:
        best_h = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation with analytical gradients
    alphas = [50, 200, 1000, 5000, 20000]
    for a in alphas:
        if time.time() > deadline - 80:
            break
        remaining = deadline - 80 - time.time()
        maxit = min(3000, max(200, int(remaining * 2)))
        h_new = slsqp_opt(best_h, n, a, maxiter=maxit)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new

    # Phase 2: Guided local search
    if time.time() < deadline - 40:
        best_h, best_c = guided_pair_search(best_h, n, deadline - 40)

    # Phase 3: Higher-alpha SLSQP polish
    if time.time() < deadline - 30:
        for a in [50000, 200000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_opt(best_h, n, a, maxiter=2000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new

    # Phase 4: Multi-start refinement
    if time.time() < deadline - 40:
        best_h, best_c = multi_start_refine(best_h, n, deadline - 10, rng, n_starts=3)

    # Phase 5: Final SLSQP + guided search
    if time.time() < deadline - 10:
        h_new = slsqp_opt(best_h, n, 500000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 2:
        best_h, best_c = guided_pair_search(best_h, n, deadline - 1)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
