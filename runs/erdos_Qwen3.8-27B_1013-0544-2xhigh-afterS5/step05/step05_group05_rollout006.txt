```python
def compute_delta(h, n, a, b):
    dx = 2.0 / n
    js = np.arange(2 * n - 1)
    ks = js - (n - 1)
    
    # For a (0 -> 1):
    a_plus_k = a + ks  # a + k for each lag
    a_minus_k = a - ks  # a - k for each lag
    
    delta = np.zeros(2 * n - 1)
    
    # a as first factor: valid if 0 <= a+k < n
    mask = (a_plus_k >= 0) & (a_plus_k < n)
    delta[mask] += (1.0 - h[a_plus_k[mask]])
    
    # a as second factor: valid if 0 <= a-k < n
    mask = (a_minus_k >= 0) & (a_minus_k < n)
    delta[mask] -= h[a_minus_k[mask]]
    
    # b (1 -> 0):
    b_plus_k = b + ks
    b_minus_k = b - ks
    
    # b as first factor: valid if 0 <= b+k < n
    mask = (b_plus_k >= 0) & (b_plus_k < n)
    delta[mask] -= (1.0 - h[b_plus_k[mask]])
    
    # b as second factor: valid if 0 <= b-k < n
    mask = (b_minus_k >= 0) & (b_minus_k < n)
    delta[mask] += h[b_minus_k[mask]]
    
    return dx * delta
```
