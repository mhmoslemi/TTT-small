```python
                    improved = True
                    break

            break
```

Let me also reconsider the approach. Instead of precomputing the full V matrix (which is O(n²) memory), I can compute the delta for a specific swap (a,b) in O(n) time:

Let me write a cleaner version:

```python
def swap_delta(h, a, b, n):
    """Compute delta in correlation when h[a] flips 1->0 and h[b] flips 0->1."""
    # delta[k] = V[b,k] - V[a,k]
    # V[x,k] = 1 - h[x-(k-n+1)] - h[x+(k-n+1)] (with h out of range = 0)
    # delta[k] = h[b-(k-n+1)] + h[b+(k-n+1)] - h[a-(k-n+1)] - h[a+(k-n+1)]
    k = np.arange(2*n-1)
    m = k - (n-1)  # m ranges from -(n-1) to (n-1)
    
    # h[b-m]: valid when 0 <= b-m <= n-1
    idx_bm = b - m
    idx_bp = b + m
    idx_am = a - m
    idx_ap = a + m
    
    delta = np.zeros(2*n-1)
    valid = (idx_bm >= 0) & (idx_bm < n)
    delta[valid] += h[idx_bm[valid]]
    valid = (idx_bp >= 0) & (idx_bp < n)
    delta[valid] += h[idx_bp[valid]]
    valid = (idx_am >= 0) & (idx_am < n)
    delta[valid] -= h[idx_am[valid]]
    valid = (idx_ap >= 0) & (idx_ap < n)
    delta[valid] -= h[idx_ap[valid]]
    
    return delta
```

This is O(n) per swap evaluation, which is much faster than O(n²) for full recompute.

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def raw_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def swap_delta(h, a, b, n):
    k = np.arange(2 * n - 1)
    m = k - (n - 1)
    delta = np.zeros(2 * n - 1)
    for idx, sign in [(b - m, 1), (b + m, 1), (a - m, -1), (a + m, -1)]:
        valid = (idx >= 0) & (idx < n)
        delta[valid] += sign * h[idx[valid]]
    return delta

def fast_sa(h_start, n, deadline, seed=42, t_start=0.1, t_end=0.0001, max_iter=500000):
    h = to_binary(h_start, n).copy()
    corr = raw_corr(h).astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max_iter)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = swap_delta(h, i, j, n)
        new_corr = corr + delta
        new_max = new_corr.max()
        
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(T, 1e-15)):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        
    return best_h, best_max

def fast_local_search(h_start, n, deadline, seed=123, max_sweeps=50000):

    corr = raw_corr(h).astype(float)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)

                        best_h = h.copy()
                    improved = True
```

And j ≥ max(0, x-n+1) = 0 (since x ≥ 0, x-n+1 ≤ 0 for x ≤ n-1).
So j ∈ [0, n-1], meaning LB[x] is non-zero for k = x+n-1-j where j ∈ [0,n-1], i.e., k ∈ [x, x+n-1].

So LB[x] has n non-zero entries at positions k = x, x+1, ..., x+n-1 (clamped to [0, 2n-2]).

Similarly, LP[x][k] = h[x+k-n+1]. Let j = x+k-n+1, so k = j-x+n-1. As j goes from 0 to n-1, k goes from n-1-x to 2n-2-x.
LP[x] is non-zero for k ∈ [n-1-x, 2n-2-x], which has n entries.

So each of LB[x] and LP[x] is a sparse vector with exactly n non-zero entries (out of 2n-1 positions). The delta for a swap has at most 4n non-zero entries (but typically less due to overlaps and cancellations).

For the local search, I can precompute LB and LP for all positions (O(n²) memory and time), and then the delta for any swap is just a vector subtraction of 4 precomputed vectors, which is O(n) but with very low constant (just array subtraction).

Actually, even better: I can precompute S[x] = LB[x] + LP[x] for all x. Then delta = S[b] - S[a], which is a single vector subtraction.

S[x] has length 2n-1 and is the sum of two sparse vectors, each with n non-zeros. So S[x] has at most 2n non-zeros (but typically around n since they overlap in the middle).

Memory: n * (2n-1) = O(n²). For n=250: 250 * 499 ≈ 125K floats = 1MB. Fine.

```python
def precompute_S(h, n):
    """Precompute S[x] = LB[x] + LP[x] for all x."""
    S = np.zeros((n, 2*n-1))
    for x in range(n):
        # LB[x][k] = h[x+n-1-k] if valid
        for k in range(2*n-1):
            j = x + n - 1 - k
            if 0 <= j < n:
                S[x, k] += h[j]
            j2 = x + k - n + 1
            if 0 <= j2 < n:
                S[x, k] += h[j2]
    return S
```

This is O(n²) but with a small constant. For n=250, it's about 250*500 = 125K iterations, which is fast.

```python
    return h_bin

    S = np.zeros((n, 2*n-1))
    for x in range(n):
        # S[x][k] = h[x+n-1-k] (if valid) + h[x+k-n+1] (if valid)
        # First term: j = x+n-1-k, valid when 0<=j<n, i.e., n-1-x < k <= x+n-1... 
        # j >= 0 → x+n-1-k >= 0 → k <= x+n-1
        # j <= n-1 → x+n-1-k <= n-1 → k >= x
        # So k ranges from x to min(x+n-1, 2n-2)
        k_lo = x
        k_hi = min(x + n - 1, 2*n - 2)
        kk = np.arange(k_lo, k_hi + 1)
        jj = x + n - 1 - kk  # these are the indices into h
        S[x, kk] += h[jj]
        # Second term: j2 = x+k-n+1, valid when 0<=j2<n
        # j2 >= 0 → x+k-n+1 >= 0 → k >= n-1-x
        # j2 <= n-1 → x+k-n+1 <= n-1 → k <= 2n-2-x
        k_lo2 = max(0, n - 1 - x)
        k_hi2 = min(2*n - 2, 2*n - 2 - x)
        kk2 = np.arange(k_lo2, k_hi2 + 1)
        jj2 = x + kk2 - n + 1
        S[x, kk2] += h[jj2]
    return S

def update_S(S, h, x, delta, n):
    """Update S after h[x] changes by delta."""
    for xi in range(n):
        # S[xi] changes at positions where xi+n-1-k = x or xi+k-n+1 = x
        k1 = xi + n - 1 - x  # where first term has h[x]
        k2 = x - xi + n - 1  # where second term has h[x]
        if 0 <= k1 < 2*n-1:
            S[xi, k1] += delta
        if 0 <= k2 < 2*n-1:
            S[xi, k2] += delta

    best_h = h.copy()

    S = precompute_S(h, n)
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]

        T = t_start * (t_end / t_start) ** (it / max_iter)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        new_max = (corr + delta).max()

            h[i] = 0.0

            corr = corr + delta

            update_S(S, h, i, -1.0, n)
            update_S(S, h, j, 1.0, n)
            # Update ones and zeros

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def precompute_S(h, n):
    S = np.zeros((n, 2 * n - 1))
    for x in range(n):
        k_lo = x
        k_hi = min(x + n - 1, 2 * n - 2)
        kk = np.arange(k_lo, k_hi + 1)
        S[x, kk] += h[x + n - 1 - kk]
        k_lo2 = max(0, n - 1 - x)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - x)
        kk2 = np.arange(k_lo2, k_hi2 + 1)
        S[x, kk2] += h[x + kk2 - n + 1]
    return S

def update_S_all(S, x, dv, n):
    for xi in range(n):
        k1 = xi + n - 1 - x
        k2 = x - xi + n - 1
        if 0 <= k1 < 2 * n - 1:
            S[xi, k1] += dv
        if 0 <= k2 < 2 * n - 1:
            S[xi, k2] += dv
```
