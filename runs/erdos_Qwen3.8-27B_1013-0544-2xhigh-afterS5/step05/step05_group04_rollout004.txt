def compute_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    if max_s > 500:
        LSE = (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / alpha
    else:
        LSE = np.log(np.sum(np.exp(shifted))) / alpha
    return float(m + LSE)

    # corr[m] = sum_i h[i] * (1-h[i+s]) * dx where s = m - (n-1)

            # j-s in [0, n-1] means j in [0, n-1+s]
            grad[:n+s] -= w * x[-s:]
    return grad

    return float(x.sum() - n / 2.0)

    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=lambda x: obj_grad(x, n, alpha),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_step(h, n, lr):
    """One subgradient step with projection."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    # Active set: lags where c is close to max
    tol = 1e-8
    active = np.where(c >= max_c - tol)[0]
    
    # Compute subgradient as average gradient over active lags
    grad = np.zeros(n)
    for m in active:

        if s >= 0:

            grad[s:] -= h[:n-s]
        else:
            grad[-s:] += (1 - h[:s])
            grad[:n+s] -= h[-s:]
    grad *= dx / len(active)
    
    h_new = h - lr * grad
    return project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lse_and_weights(c, m, alpha):
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 50:
        w = np.exp(shifted - mx)
        lse = (mx + np.log(w.sum())) / alpha
    else:
        w = np.exp(shifted)
        lse = np.log(w.sum()) / alpha
    return lse, w / w.sum()

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    lse, _ = _lse_and_weights(c, m, alpha)
    return float(m + lse)

def obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    _, w = _lse_and_weights(c, m, alpha)
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wi = w[m_idx] * dx
        if wi < 1e-15:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * (1.0 - x[s:])
            grad[s:] -= wi * x[:n - s]
        else:
            t = -s
            grad[t:] += wi * (1.0 - x[:n - t])
            grad[:n - t] -= wi * x[t:]
    return grad

def _sum_fun(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       jac=lambda x: obj_grad(x, n, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_descent(h, n, deadline, lr0=0.005):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    stale = 0
    for _ in range(10000):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        mx = c.max()
        active = np.where(c >= mx - 1e-10)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        grad = np.zeros(n)
        for m_idx in active:
            s = m_idx - (n - 1)
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] -= h[:n - s]
            else:
                t = -s
                grad[t:] += (1.0 - h[:n - t])
                grad[:n - t] -= h[t:]
        h = project_feasible(h - lr * dx * grad / len(active), n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 100:
                lr *= 0.7
                stale = 0
                if lr < 1e-7:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=60):

    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True

            if not improved:

    return h, best_c
