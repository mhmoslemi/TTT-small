def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_of_lag(h, k, n):
    """Compute gradient of g_k(h) = sum_{valid i} h[i](1-h[i+k]) with respect to h."""
    grad = np.zeros(n)
    if k == 0:
        # g_0(h) = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
        # For continuous h in [0,1], grad = 1 - 2*h
        return 1.0 - 2.0 * h
    if k > 0:
        # Valid i: 0 <= i <= n-1-k
        # g = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
        # d/dh[i] for i in [0, n-1-k]: (1 - h[i+k])
        # d/dh[i] for i in [k, n-1]: -h[i-k]
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        m = -k
        # Valid i: m <= i <= n-1
        # g = sum_{i=m}^{n-1} h[i](1-h[i-m])
        # d/dh[i] for i in [m, n-1]: (1 - h[i-m])
        # d/dh[i] for i in [0, n-1-m]: -h[i+m]
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad

    grad = gradient_of_lag(h, k, n)

    return project_feasible(h_new, n)

        to_add = rng.choice(zeros, target - n_ones, replace=False)

def sdpr_solution(n, deadline):
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    h = cp.diag(X)  # diagonal of X
    
    constraints = [
        X >> 0,  # PSD
        cp.sum(h) == n / 2,
        h >= 0,
        h <= 1,
    ]
    
    # Add correlation constraints
    for k in range(-(n-1), n):
        if k == 0:
            # g_0 = sum h[i](1-h[i]) = sum h[i] - sum X[i,i] = 0 (always)
            continue

            idx = np.arange(valid_n)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, k, n):
    """Gradient of g_k(h) = sum_{valid i} h[i]*(1-h[i+k])"""
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        m = n - k
        grad[:m] += (1.0 - h[k:])
        grad[k:] -= h[:m]
    else:
        m = -k
        mm = n - m
        grad[m:] += (1.0 - h[:mm])
        grad[:mm] -= h[m:]
    return grad

def subgradient_descent(h, n, deadline, top_k=3, eta0=0.1, decay=0.9999):

    best_h = h.copy()

    dx = 2.0 / n
    k_range = np.arange(-(n-1), n)
    eta = eta0

        corrs = np.correlate(h, 1 - h, mode='full')
        cur_c = corrs.max() * dx
        if cur_c < best_c:

            best_h = h.copy()
            stall_count = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, k, n):
    grad = np.zeros(n)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        m = n - k
        grad[:m] += (1.0 - h[k:])
        grad[k:] -= h[:m]
    else:
        m = -k
        mm = n - m
        grad[m:] += (1.0 - h[:mm])
        grad[:mm] -= h[m:]
    return grad

def subgrad_momentum(h, n, deadline, top_k=5, eta0=0.08, mu=0.9):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    eta = eta0
    vel = np.zeros(n)
    h_curr = h.copy()
    avg_h = h.copy()
    count = 0
    stall = 0
    
    while time.time() < deadline:
        corrs = np.correlate(h_curr, 1 - h_curr, mode='full')
        cur_c = corrs.max() * dx
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1
        
        top_idxs = np.argsort(corrs)[-top_k:]
        g = np.zeros(n)
        for idx in top_idxs:
            g += grad_lag(h_curr, k_range[idx], n)
        g /= top_k
        
        vel = mu * vel - eta * g
        h_new = h_curr + vel
        h_curr = project_feasible(h_new, n)
        
        count += 1
        avg_h = (count * avg_h + h_curr) / (count + 1)
        
        eta *= 0.9998
        if stall > 5000:
            eta = max(eta, 0.05)
            mu = max(mu * 0.95, 0.7)
    
    c_avg = compute_c5(avg_h, n)
    if c_avg < best_c:
        best_c = c_avg
        best_h = avg_h.copy()
    return best_h, best_c

    rng = np.random.RandomState(seed)

    return tm
