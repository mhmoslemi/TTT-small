from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    """Compute the change in the full correlation vector when h[pos] changes by delta."""
    dx = 2.0 / n
    # c_full[k] = sum_i h[i] * (1-h[i-(k-(n-1))]) * dx
    # For lag l = k-(n-1), c_k involves pairs (i, i-l)
    # Changing h[pos] affects:
    # 1. Terms where pos is the first index: h[pos]*(1-h[pos-l]) for each l
    # 2. Terms where pos is the second index: h[pos+l]*(1-h[pos]) for each l
    
    # In the full correlation array of size 2n-1:
    # Index k corresponds to lag l = k - (n-1)
    # c_k = sum over valid i of h[i] * (1-h[i-l])
    
    # Effect of changing h[pos] by delta:
    # For lag l, if pos and pos-l are both valid: delta * (1-h[pos-l])
    # For lag l, if pos and pos+l are both valid: -delta * h[pos+l]
    
    delta_c = np.zeros(2*n-1)
    l_min = -(n-1)
    l_max = n-1
    
    for l in range(l_min, l_max+1):
        k = l + (n-1)
        dc = 0.0
        # pos as first index: h[pos]*(1-h[pos-l])
        if 0 <= pos-l < n:

        # pos as second index: h[pos+l]*(1-h[pos])
        if 0 <= pos+l < n:
            dc += -delta * h[pos+l]
        delta_c[k] = dc * dx
    
    return delta_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_v_matrix(h, n):
    """Build v[pos, idx] = (1-h[pos-k]) - h[pos+k] where k = idx-(n-1)."""
    # k ranges from -(n-1) to (n-1), idx = k + (n-1)
    ks = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each pos, compute pos-k and pos+k
    # pos_minus_k[pos, idx] = pos - ks[idx]
    # pos_plus_k[pos, idx] = pos + ks[idx]
    pos_idx = np.arange(n)[:, None]  # shape (n, 1)
    pos_minus_k = pos_idx - ks[None, :]  # shape (n, 2n-1)
    pos_plus_k = pos_idx + ks[None, :]   # shape (n, 2n-1)
    
    mask1 = (pos_minus_k >= 0) & (pos_minus_k < n)
    mask2 = (pos_plus_k >= 0) & (pos_plus_k < n)
    
    # h[pos_minus_k] where mask1, else 0
    term1 = np.where(mask1, 1.0 - h[np.clip(pos_minus_k, 0, n-1)], 0.0)
    term2 = np.where(mask2, h[np.clip(pos_plus_k, 0, n-1)], 0.0)
    
    return term1 - term2

def vectorized_local_search(h, n, deadline, steps, max_iters=100000):
    """Find best single-pair move using vectorized computation."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    iters = 0
    
    for step in steps:
        while time.time() < deadline and iters < max_iters:
            iters += 1
            c_full = np.correlate(h, 1 - h, mode='full') * dx
            c_max = c_full.max()
            
            v = build_v_matrix(h, n)  # shape (n, 2n-1)
            
            best_improvement = 1e-12
            best_i, best_j = -1, -1
            
            for i in range(n):
                if h[i] + step > 1.0:
                    continue
                # For this i, try all j
                # new_c[j] = c_full + step*dx*(v[i] - v[j])
                # max_new_c[j] = max(new_c[j])
                sv = step * dx
                candidates = c_full[None, :] + sv * (v[i][None, :] - v)  # (n, 2n-1)
                maxes = candidates.max(axis=1)
                # Valid j: h[j] >= step and j != i
                valid = (h >= step)
                valid[i] = False
                if not valid.any():
                    continue
                maxes_mod = np.where(valid, maxes, np.inf)
                j_best = int(maxes_mod.argmin())
                if maxes_mod[j_best] < best_c - 1e-14:
                    best_improvement = best_c - maxes_mod[j_best]
                    best_i, best_j = i, j_best
            
            if best_i < 0:
                break
            
            h[best_i] += step
            h[best_j] -= step
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            else:
                # Shouldn't happen much, but revert if it does
                h[best_i] -= step
                h[best_j] += step
                h = project_feasible(h, n)
                break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_v_matrix(h, n):
    ks = np.arange(-(n - 1), n)
    pos_idx = np.arange(n)[:, None]
    pmk = pos_idx - ks[None, :]
    ppk = pos_idx + ks[None, :]
    m1 = (pmk >= 0) & (pmk < n)
    m2 = (ppk >= 0) & (ppk < n)
    t1 = np.where(m1, 1.0 - h[np.clip(pmk, 0, n - 1)], 0.0)
    t2 = np.where(m2, h[np.clip(ppk, 0, n - 1)], 0.0)
    return t1 - t2

            v = build_v_matrix(h, n)

            best_i, best_j = -1, -1
            for i in range(n):
                if h[i] + step > 1.0:
                    continue

                maxes = cands.max(axis=1)
                valid = (h >= step)
                valid[i] = False

                    continue
