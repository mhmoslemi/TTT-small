One more idea: **subgradient method with averaging**. Since we're minimizing max_k c_k, we can use the Frank-Wolfe (conditional gradient) method:

- Find the h' in the feasible set that minimizes the subgradient direction

The subgradient of max_k c_k at the active k* is ∇c_{k*}. The conditional gradient step would find:
h' = argmin_{h' in feasible} <∇c_{k*}, h'>

This is a linear program! And the feasible set is a simplex (box constraints + sum constraint). The LP can be solved in closed form:
h'[i] = 0 if ∇c_{k*}[i] > 0 (and we need to decrease h[i])
h'[i] = 1 if ∇c_{k*}[i] < 0 (and we need to increase h[i])
Then adjust to meet the sum constraint.

Let me think about the FW step more carefully:
- ∇c_{k*}[j] = dx * [(1-h[j+k*]) - h[j-k*]] (with boundary terms = 0)
- We want to minimize sum_j ∇c_{k*}[j] * h'[j] subject to 0 <= h'[j] <= 1, sum(h') = n/2
- This is: minimize sum_j g[j] * h'[j] where g[j] = ∇c_{k*}[j]
- The LP solution: set h'[j] = 1 for the (n/2) smallest values of g[j], h'[j] = 0 for the rest

```python
def fw_step(h, n, k_star):
    """Frank-Wolfe step: compute the conditionally optimal direction."""
    dx = 2.0 / n
    s = k_star - (n - 1)  # actual shift
    g = np.zeros(n)  # subgradient of c_{k_star}
    for j in range(n):
        val = 0.0
        if 0 <= j + s < n:
            val += (1.0 - h[j + s])
        if 0 <= j - s < n:
            val -= h[j - s]
        g[j] = dx * val
    # LP: minimize sum g[j]*h'[j] s.t. 0<=h'<=1, sum(h')=n/2
    # Solution: h'[j]=1 for smallest g[j] values
    target = n / 2
    order = np.argsort(g)
    h_lp = np.zeros(n)
    h_lp[order[:int(target)]] = 1.0
    # Handle fractional part
    frac = target - int(target)
    if frac > 0:
        h_lp[order[int(target)]] = frac
    return h_lp
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h + lam, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
    return np.clip(proj, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    w = exps / exps.sum()
    obj = m + np.log(exps.sum()) / alpha
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        mk1 = (j + shifts >= 0) & (j + shifts < n)
        if mk1.any():
            grad[j] += (w[mk1] * (1.0 - h[j + shifts[mk1]])).sum()
        mk2 = (j - shifts >= 0) & (j - shifts < n)
        if mk2.any():
            grad[j] -= (w[mk2] * h[j - shifts[mk2]]).sum()
    grad *= dx
    return obj, grad

def nesterov_descent(h, n, deadline, alpha_start=50.0, alpha_end=30000.0,
                     n_steps=10000, lr=0.004, momentum=0.9):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stuck = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(n_steps, 1))
        obj, grad = smooth_grad(h, n, alpha)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 500:
                v = v * 0.5
                stuck = 0
        h = h_new
    return best_h, best_c

def fw_step(h, n, k_star):
    dx = 2.0 / n
    s = k_star - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j - s < n:
            val += (1.0 - h[j - s])
        if 0 <= j + s < n:
            val -= h[j + s]
        g[j] = dx * val
    target = n / 2.0
    order = np.argsort(g)
    h_lp = np.zeros(n)
    h_lp[order[:int(target)]] = 1.0
    frac = target - int(target)
    if frac > 1e-10:
        h_lp[order[int(target)]] = frac
    return h_lp

def fw_explode(h, n, deadline, n_iters=3000, gamma=None):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if gamma is None:
        gamma = [2.0 / 3.0] * n_iters
    for it in range(n_iters):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        k_star = c.argmax()
        h_lp = fw_step(h, n, k_star)
        t = gamma[it] if it < len(gamma) else 2.0 / (it + 2)
        h_new = (1.0 - t) * h + t * h_lp
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def targeted_descent(h, n, deadline, max_top=25, deltas=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    dx = 2.0 / n
    for delta in deltas:
        if time.time() > deadline:
            break
        for _pass in range(800):
            if time.time() > deadline:
                return h, best_c
            improved = False
            c = np.correlate(h, 1 - h, mode='full') * dx
            k_star = c.argmax()
            s = k_star - (n - 1)
            scores = np.zeros(n)
            for m in range(n):
                if 0 <= m + s < n:
                    scores[m] += h[m + s]
                if 0 <= m - s < n:
                    scores[m] += h[m - s]
            order = np.argsort(scores)
            low_pos = order[:max_top]
            high_pos = order[-max_top:]
            for i in low_pos:
                if improved:
                    break
                for j in high_pos:
                    if i == j or h[j] < delta or h[i] > 1.0 - delta:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                        break
                    else:
                        h[i] -= delta
                        h[j] += delta
            if not improved:
                break
    return h, best_c

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
```
