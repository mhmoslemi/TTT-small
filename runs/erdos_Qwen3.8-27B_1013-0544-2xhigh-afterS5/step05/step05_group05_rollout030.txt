import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = g - m
        return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def binary_ls(h, n, deadline, seed=123, max_rounds=500):
    best = h.copy()
    best_c = compute_c5(best, n)
    rng = np.random.RandomState(seed)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                return best, best_c
            for j in ones:
                best[i], best[j] = 1.0, 0.0
                c = compute_c5(best, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                best[i], best[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return best, best_c

def ils(h, n, deadline, seed=42, max_iters=3000):
    rng = np.random.RandomState(seed)
    curr = h.copy()
    curr_c = compute_c5(curr, n)
    best = curr.copy()
    best_c = curr_c
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ls_deadline = min(deadline, time.time() + 2.0)
        curr, curr_c = binary_ls(curr, n, ls_deadline, seed=seed + it * 7 + 1, max_rounds=50)
        if curr_c < best_c:
            best_c = curr_c
            best = curr.copy()
        k = rng.randint(2, 8)
        ones = np.where(curr == 1)[0]
        zeros = np.where(curr == 0)[0]
        for _ in range(min(k, len(ones), len(zeros))):
            if len(ones) < 1 or len(zeros) < 1:
                break
            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            curr[a], curr[b] = 1.0, 0.0
            ones = np.where(curr == 1)[0]
            zeros = np.where(curr == 0)[0]
    return best, best_c

def sa_multi(h0, n, deadline, seed=999):
    rng = np.random.RandomState(seed)
    curr = h0.copy()
    curr_c = compute_c5(curr, n)
    best = curr.copy()
    best_c = curr_c
    ones = np.where(curr == 1)[0]
    zeros = np.where(curr == 0)[0]
    T = 0.03
    Tmin = 5e-6
    cool = 0.9998
    n_iter = 0
    while time.time() < deadline:
        n_iter += 1
        r = rng.random()
        k = 1 if r < 0.6 else (2 if r < 0.85 else 3)
        k = min(k, len(ones), len(zeros))
        if k < 1:
            break
        mo = rng.choice(ones, size=k, replace=False)
        mz = rng.choice(zeros, size=k, replace=False)
        new = curr.copy()
        new[mo] = 0.0
        new[mz] = 1.0
        nc = compute_c5(new, n)
        diff = nc - curr_c
        if diff < 0 or (n_iter < 500000 and rng.random() < np.exp(-diff / T)):
            curr = new
            curr_c = nc
            ones = np.where(curr == 1)[0]
            zeros = np.where(curr == 0)[0]
            if nc < best_c:
                best_c = nc
                best = curr.copy()
        T = max(T * cool, Tmin)
    return best, best_c

def subgrad_polish(h0, n, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                best_h = ht
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
    return best_h, best_c
