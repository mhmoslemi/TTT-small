            nh = project_feasible(h - stp * g, n)
            if not np.allclose(nh, h):
                nc = compute_c5(nh, n)
                if nc < cur - 1e-15:
                    h = nh
                    improved = True
                    break
            stp *= 0.5
        if improved:
            continue
        
        # Single exchange with line search
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:
            break
        best_stp = 0
        for fr in [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]:
            mv = min(h[i], 1.0 - h[j]) * fr
            if mv < 1e-12:
                continue
            nh = h.copy()
            nh[i] -= mv
            nh[j] += mv
            nh = project_feasible(nh, n)
            nc = compute_c5(nh, n)
            if nc < cur - 1e-15:
                best_stp = mv
                h = nh
                break
        if best_stp == 0:
            # Try random exchange as escape
            if stall < 400:
                i2, j2 = np.random.randint(0, n, 2)
                if i2 != j2:
                    mv = min(h[i2], 1.0 - h[j2]) * 0.1
                    nh = h.copy()
                    nh[i2] -= mv
                    nh[j2] += mv
                    nh = project_feasible(nh, n)
                    nc = compute_c5(nh, n)
                    # Accept with probability based on improvement (simulated annealing)
                    if nc < cur or np.random.rand() < np.exp(-(nc - cur) * 50):
                        h = nh
            else:
                break
    
    return best_h, best_c

def optimize(h0, n, deadline):
