import numpy as np
from scipy.linalg import null_space
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def get_coefficients(h_old, n):
    """Get coefficient matrix for all lags.
    Returns A where A[s, i] is the coefficient of h[i] in lag s's overlap.
    """
    result = np.correlate(h_old, 1 - h_old, mode='full')
    num_lags = len(result)  # 2n-1
    A = np.zeros((num_lags, n))
    for s in range(num_lags):
        d = s - (n - 1)
        # coefficient of h[i] is (1 - h_old[i-d]) if 0 <= i-d < n
        for i in range(n):
            j = i - d
            if 0 <= j < n:
                A[s, i] = 1.0 - h_old[j]
    return A

def active_set_lp(h, n, n_active=20, alpha_margin=0.0):
    """Solve LP with active set constraints.
    Returns new h that minimizes the max over active lags (linearized).
    """
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    
    # Select active lags: those within margin of max
    threshold = max_corr - alpha_margin
    active = np.where(corr >= threshold)[0]
    
    if len(active) == 0:
        active = [np.argmax(corr)]
    if len(active) > n_active:
        # Take top n_active
        active = active[np.argsort(corr[active])[-n_active:]]
    
    # Build coefficient matrix for active lags
    A = np.zeros((len(active), n))
    for idx, s in enumerate(active):
        d = s - (n - 1)
        for i in range(n):
            j = i - d
            if 0 <= j < n:
                A[idx, i] = 1.0 - h[j]
    
    # LP: min t
    # s.t. A @ h - t <= 0  =>  [A, -1] @ [h; t] <= 0
    #      sum(h) = n/2
    #      0 <= h <= 1
    #      t >= 0
    n_var = n + 1  # h[0..n-1], t
    
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: A @ h - t <= 0 for each active lag
    A_ineq = np.zeros((len(active), n_var))
    A_ineq[:, :n] = A
    A_ineq[:, n] = -1.0
    b_ineq = np.zeros(len(active))
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], active
    return h, active

def alternating_lp(h, n, deadline, max_iter=200, n_active_max=30):
    """Alternating linearization: freeze (1-h), solve LP, repeat."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Solve LP with active set
        h_new, active = active_set_lp(h, n, n_active=n_active_max)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Check convergence
        if abs(c_new - compute_c5(h, n)) < 1e-10:
            h = h_new
            break
        
        # Damped update for stability
        h = 0.5 * h + 0.5 * h_new
        h = project_feasible(h, n)
    
    return best_h, best_c

                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def bangbang_round(h, n):
    hb = (h > 0.5).astype(float)
    excess = hb.sum() - n / 2.0
    if excess > 0:
        ones = np.where(hb == 1.0)[0]
        order = ones[np.argsort(h[ones])]
        for idx in order[:int(excess)]:
            hb[idx] = 0.0
    elif excess < 0:
        zeros = np.where(hb == 0.0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for idx in order[:int(-excess)]:
            hb[idx] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(120, 0.5)
        n = 120
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)

    # Phase 1: Alternating LP
    alloc1 = min(350, (deadline - t0) * 0.45)
    h_lp, c_lp = alternating_lp(h_init, n, t0 + alloc1)
    if c_lp < best_c:
        best_c = c_lp
        best_h = h_lp

    # Phase 2: Bang-bang trial
    if time.time() < deadline - 100:
        hb = bangbang_round(best_h, n)
        hb = project_feasible(hb, n)
        c_hb = compute_c5(hb, n)
        if c_hb < best_c:
            best_c = c_hb
            best_h = hb
        # Try LP from bang-bang
        h_ll, c_ll = alternating_lp(hb, n, time.time() + min(60, deadline - t0 - 200))
        if c_ll < best_c:
            best_c = c_ll
            best_h = h_ll

    # Phase 3: SLSQP progressive
    if time.time() < deadline - 80:
        for alpha in [2000, 10000, 50000, 200000, 500000]:
            if time.time() > deadline - 60:
                break
            h_new = slsqp_optimize(best_h, n, alpha, maxiter=400)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new

    # Phase 4: Pair search
    if time.time() < deadline - 20:
        best_h, best_c = pair_search(best_h, n, deadline - 8,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    # Phase 5: Multi-restart
    if time.time() < deadline - 50:
        remaining = deadline - time.time() - 30
        n_restarts = max(2, int(remaining / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.015 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 20) * 0.45)
            if alloc < 20:
                break
            h_r, c_r = alternating_lp(hp, n, time.time() + alloc * 0.5, max_iter=100, max_k=30)
            if c_r < best_c:
                best_c = c_r
                best_h = h_r
            if time.time() < deadline - 15:
                h_r2 = slsqp_optimize(h_r, n, 100000, maxiter=300)
                c_r2 = compute_c5(h_r2, n)
                if c_r2 < best_c:
                    best_c = c_r2
                    best_h = h_r2
                if time.time() < deadline - 10:
                    best_h, best_c = pair_search(best_h, n, deadline - 5,
                                                 [0.002, 0.001, 0.0005, 0.0002])

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h_new = slsqp_optimize(best_h, n, 1000000, maxiter=1500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
