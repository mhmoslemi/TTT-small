def eval_move_fast(c, h, n, i, j, delta):
    """Evaluate max correlation after h[i] += delta, h[j] -= delta, given current correlation c."""
    # The change in correlation at position m (shift k = m-(n-1)):
    # c[m] = sum over valid i' of h[i'] * (1-h[i'+k])
    # After change, the terms involving h[i] or h[j] change
    
    # For position m with k = m-(n-1):
    # h[i] appears in c[m] if:
    #   - as first factor: i + k >= 0 and i + k < n, i.e., m-(n-1)+i is in [0,n) → contributes h[i]*(1-h[i+k])
    #   - as second factor: i - k >= 0 and i - k < n, i.e., contributes h[i-k]*(1-h[i])
    
    # Change in c[m] from h[i] → h[i]+delta:
    #   Factor 1: if 0 <= i+k < n: add delta*(1-h[i+k])
    #   Factor 2: if 0 <= i-k < n: add -delta*h[i-k] (because (1-h[i]) changes by -delta)
    # Change in c[m] from h[j] → h[j]-delta:
    #   Factor 1: if 0 <= j+k < n: add -delta*(1-h[j+k])
    #   Factor 2: if 0 <= j-k < n: add delta*h[j-k]
    
    # So for each m (equivalently, for each k from -(n-1) to n-1):
    # change = delta * [1[0<=i+k<n]*(1-h[i+k]) - 1[0<=i-k<n]*h[i-k] 
    #                      - 1[0<=j+k<n]*(1-h[j+k]) + 1[0<=j-k<n]*h[j-k]]
    
    # k ranges from -(n-1) to n-1
    # Let's vectorize:
    ks = np.arange(-(n-1), n)  # all shifts
    # For index i:
    ipk = i + ks  # i+k
    imk = i - ks  # i-k
    # For index j:
    jpk = j + ks
    jmk = j - ks
    
    change = delta * (
        np.where((ipk >= 0) & (ipk < n), 1.0 - h[ipk[np.where((ipk >= 0) & (ipk < n))[0]]], 0)
        - np.where((imk >= 0) & (imk < n), h[imk[np.where((imk >= 0) & (imk < n))[0]]], 0)
        - np.where((jpk >= 0) & (jpk < n), 1.0 - h[jpk[np.where((jpk >= 0) & (jpk < n))[0]]], 0)
        + np.where((jmk >= 0) & (jmk < n), h[jmk[np.where((jmk >= 0) & (jmk < n))[0]]], 0)
    )
    
    c_new = c + change
    return np.max(c_new) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx, np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n+k]
    return grad * dx

def projected_grad_descent(h, n, deadline, rng):
    """Projected gradient with momentum and adaptive step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.1
    mom = 0.9
    for it in range(100000):
        if time.time() > deadline:
            break
        g = subgrad_at_max(h, n)
        g -= g.mean()  # project to tangent space
        v = mom * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.95
            v *= 0.5
            if lr < 1e-8:
                # Restart with noise
                h = project_feasible(best_h + 0.01 * rng.randn(n), n)
                v = np.zeros(n)
                lr = 0.1
    return best_h, best_c

def lbfgs_smooth(h, n, alpha, maxiter):
    """L-BFGS-B on smooth max approximation."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - target) ** 2
        return val
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        # Weighted sum of subgradients
        g = np.zeros(n)
        for idx in range(len(w)):
            if w[idx] < 1e-8:
                continue
            k = idx - (n - 1)
            if k >= 0:
                g[k:] += w[idx] * (1.0 - x[:n-k])
                g[:n-k] += -w[idx] * x[k:]
            else:
                g[:n+k] += w[idx] * (1.0 - x[-k:])
                g[-k:] += -w[idx] * x[:n+k]
        g *= dx
        g += 2e5 * (x.sum() - target)
        return g
    
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_swap_cd(h, n, deadline, rng):
    """Coordinate descent with random pair swaps and ternary search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    n_iter = 0
    no_improve = 0
    for it in range(500000):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Find optimal t: h[i] += t, h[j] -= t
        lo = max(-h[i], -(1.0 - h[j]))  # h[i]+t >= 0 and h[j]-t >= 0
        hi = min(1.0 - h[i], h[j])       # h[i]+t <= 1 and h[j]-t <= 1
        if hi <= lo:
            continue
        # Ternary search
        a, b = lo, hi
        for _ in range(15):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            h[i] += m1; h[j] -= m1
            c1 = compute_c5(h, n)
            h[i] -= m1; h[j] += m1
            h[i] += m2; h[j] -= m2
            c2 = compute_c5(h, n)
            h[i] -= m2; h[j] += m2
            if c1 < c2:
                b = m2
            else:
                a = m1
        t_opt = (a + b) / 2.0
        h[i] += t_opt; h[j] -= t_opt
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] -= t_opt; h[j] += t_opt
            no_improve += 1
            if no_improve > 500:
                # Add small perturbation
                h = project_feasible(best_h + 0.005 * rng.randn(n), n)
                no_improve = 0
    return best_h, best_c

def binary_annealing(h, n, deadline, rng, max_iter=50000):
    """Simulated annealing on binary solutions."""
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly target ones
    ones = np.where(h_bin == 1)[0]
    if len(ones) > target:
        extra = rng.choice(ones, len(ones) - target, replace=False)
        h_bin[extra] = 0
    elif len(ones) < target:
        zeros = np.where(h_bin == 0)[0]
        need = target - len(ones)
        add = rng.choice(zeros, need, replace=False)
        h_bin[add] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    temp = 0.02
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones_list = np.where(h_bin == 1)[0]
        zeros_list = np.where(h_bin == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        delta = c_new - best_c
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
            temp *= 0.999
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
        temp = max(temp, 1e-7)
    return best_h, best_c

    # Pick the n_free coordinates closest to 0.5 (most "uncertain")
    sorted_idx = np.argsort(np.abs(h - 0.5))
    free_idx = sorted_idx[:n_free]

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    k = m_star - (n - 1)

        grad[:n-k] += -h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n+k]
    return grad * dx

    target = n / 2.0

        sh = alpha * (c - m)
