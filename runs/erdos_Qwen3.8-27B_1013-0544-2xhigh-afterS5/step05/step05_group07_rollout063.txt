def project_feasible(h, n):

            if s < 1e-14:
                break
            h = h + diff / s * h

def build_lp_matrices(h, n):
    """Build the LP matrices for the successive linearization."""
    n_corr = 2 * n - 1
    A_ub = np.zeros((n_corr, n + 1))
    b_ub = np.zeros(n_corr)

            b_k = np.dot(h[:n-pk], h[pk:])

        b_ub[m] = -b_k

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

            if s < 1e-14:
                break
            h = h + diff / s * room

            s = h.sum()
            if s < 1e-14:
                break

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def successive_lp(h, n, deadline, max_iter=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    n_corr = 2 * n - 1
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        A_ub = np.zeros((n_corr, n + 1))
        b_ub = np.zeros(n_corr)
        for m in range(n_corr):
            k = m - (n - 1)
            if k >= 0:
                A_ub[m, k:n] = 1.0 - h[:n - k]
                A_ub[m, :n - k] -= h[k:]
                b_ub[m] = -np.dot(h[k:], h[:n - k])
            else:
                pk = -k
                A_ub[m, :n - pk] = 1.0 - h[pk:]
                A_ub[m, pk:] -= h[:n - pk]
                b_ub[m] = -np.dot(h[:n - pk], h[pk:])
            A_ub[m, n] = -1.0
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status != 0:
            break
        h_lp = res.x[:n]
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_trial = project_feasible((1 - alpha) * best_h + alpha * h_lp, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-14:
                best_c = c_trial
                best_h = h_trial.copy()
                improved = True
                break
        h = best_h
        if not improved:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0
    return best_h, best_c

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_vals = np.exp(shifted - sm)
    weights = exp_vals / exp_vals.sum()
    val = m + (sm + np.log(exp_vals.sum())) / alpha
    val += 1e4 * (x.sum() - n / 2.0) ** 2
    g = np.zeros(n)
    for m_idx in range(len(c)):
        if weights[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        w = weights[m_idx]
        if k >= 0:
            g[k:] += (1.0 - x[:n - k]) * w
            g[:n - k] += -x[k:] * w
        else:
            pk = -k
            g[:n - pk] += (1.0 - x[pk:]) * w
            g[pk:] += -x[:n - pk] * w
    g *= dx
    g += 2e4 * (x.sum() - n / 2.0)
    return val, g

def smooth_lbfgs(h, n, alpha, maxiter, deadline):
    def wrapper(x):
        return smooth_obj_grad(x, n, alpha)
    try:
        res = minimize(wrapper, h, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h

    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:

    if c1 < best_c:

    p2 = t0 + 0.65 * (budget_s - 3)
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000]:
        if time.time() > p2 - 5:

        h_new = smooth_lbfgs(best_h.copy(), n, alpha, 100, p2)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 3: LP again from improved point (15%)
    p3 = t0 + 0.80 * (budget_s - 3)
    h3, c3 = successive_lp(best_h.copy(), n, p3, max_iter=100)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    # Phase 4: Perturbed restarts + LP (20%)
    for restart in range(8):
        if time.time() > deadline - 15:
            break
        sigma = 0.005 + 0.02 * restart
        h_pert = project_feasible(best_h + sigma * rng.randn(n), n)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c:
            best_c, best_h = c_pert, h_pert.copy()
        h_r, c_r = successive_lp(h_pert, n, deadline - 5, max_iter=60)
        if c_r < best_c:
            best_c, best_h = c_r, h_r.copy()

    # Final polish
    if time.time() < deadline - 10:
        h_f = smooth_lbfgs(best_h.copy(), n, 5000000, 80, deadline)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
