```python
def binary_sa_fast(h, n, deadline, rng):
    """Fast binary SA using incremental correlation updates"""
    h_b = h.copy()
    # Round to binary
    idx = np.argsort(h_b)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    
    # Precompute correlation
    dx = 2.0 / n
    c = np.correlate(h_b, 1 - h_b, mode='full') * dx
    current_max = c.max()
    best_c = current_max
    best_h = h_b.copy()
    
    ones = set(np.where(h_b == 1)[0].tolist())
    zeros = set(np.where(h_b == 0)[0].tolist())
    temp = 0.002
    
    iterations = 0
    while time.time() < deadline:
        # Pick random pair
        i = list(zeros)[rng.randint(len(zeros))]
        j = list(ones)[rng.randint(len(ones))]
        
        # Compute delta for all k (O(n) but with simple operations)
        # Δc_k = h[j-k] - h[i-k] + h[j+k] - h[i+k] (with 0 for out of range)
        # We need max_k (c[k] + Δc_k)
        
        # For efficiency, compute delta only for k where it might be non-zero
        # But for simplicity, let's vectorize
        # k ranges from -(n-1) to (n-1)
        ks = np.arange(-(n-1), n)
        
        # h[idx] for various idx arrays
        def safe_h(idx_arr):
            """Return h values, 0 for out of range"""
            valid = (idx_arr >= 0) & (idx_arr < n)
            result = np.zeros(len(idx_arr))
            result[valid] = h_b[idx_arr[valid]]
            return result
        
        di_k = safe_h(i - ks)  # h[i-k] for all k
        dj_k = safe_h(j - ks)  # h[j-k] for all k
        di_pk = safe_h(i + ks)  # h[i+k] for all k
        dj_pk = safe_h(j + ks)  # h[j+k] for all k
        
        delta_c = (dj_k - di_k + dj_pk - di_pk) * dx
        
        new_c_max = (c + delta_c).max()
        delta = new_c_max - current_max
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-9)):
            # Accept
            h_b[i] = 1.0
            h_b[j] = 0.0
            c += delta_c
            current_max = c.max()
            
            ones.discard(j); ones.add(i)
            zeros.discard(i); zeros.add(j)
            
            if current_max < best_c:
                best_c = current_max
                best_h = h_b.copy()
        else:
            pass  # reject, nothing to do
        
        temp = max(temp * 0.99995, 1e-7)
        iterations += 1
        if iterations % 10000 == 0:
            pass  # could add logging
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def full_correlation(h, n):
    """Compute full correlation vector"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except:
        n = 250
        h0 = np.full(n, 0.5)
    
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Fast continuous hill climbing with O(n) per move
    # This is the main innovation: use the delta formula for O(n) evaluation per move
    p1 = t0 + 0.50 * (budget_s - 5)
    h = best_h.copy()
    c = full_correlation(h, n)  # c[k+n-1] is the correlation at lag k
    cur_max = c.max()
    temp = 0.001
    delta_scale = 0.01
    
    iters = 0
    while time.time() < p1:
        # Pick random pair (i, j) and step size
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = rng.choice([1, -1])
        d = delta_scale * rng.uniform(0.3, 1.0)
        
        # Check bounds
        hi_new = h[i] + s * d
        hj_new = h[j] - s * d
        if hi_new < -0.01 or hi_new > 1.01 or hj_new < -0.01 or hj_new > 1.01:
            continue
        
        # Compute delta for all lags
        # k ranges from -(n-1) to (n-1), stored at index k + n - 1
        ks = np.arange(-(n-1), n)
        
        # Compute h[j+k], h[j-k], h[i+k], h[i-k] with 0 for OOB
        jk = j + ks
        jmk = j - ks
        ik = i + ks
        imk = i - ks
        
        hjk = np.where((jk >= 0) & (jk < n), h[jk[(jk >= 0) & (jk < n)]], 0.0)
        hjmk = np.where((jmk >= 0) & (jmk < n), h[jmk[(jmk >= 0) & (jmk < n)]], 0.0)
        hik = np.where((ik >= 0) & (ik < n), h[ik[(ik >= 0) & (ik < n)]], 0.0)
        himk = np.where((imk >= 0) & (imk < n), h[imk[(imk >= 0) & (imk < n)]], 0.0)
        
        delta_c = d * s * (hjk + hjmk - hik - himk) * dx
        
        new_c_max = (c + delta_c).max()
        delta = new_c_max - cur_max
        
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            h[i] += s * d
            h[j] -= s * d
```

```python
def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            room = 1.0 - h

            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h

def full_corr(h, n):

def safe_idx(arr, idx):

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 1e4 * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    m_star = int(np.argmax(c))
```
