```python
def update_corr(corr, h, n, i, j, di, dj):
    """Update correlation after h[i] += di, h[j] += dj."""
    # For each k, C_k changes by:
    # di * (1 - h[i+k-n+1]) - di * h[i-k+n-1]  (for valid indices)
    # + dj * (1 - h[j+k-n+1]) - dj * h[j-k+n-1]
    # But we need to use the OLD h values (before the change)
    # Actually, since we're computing the change, and the change is linear in di, dj
    # (the second-order term di*dj is zero since i≠j), we can use old h.
    for k in range(2*n-1):
        delta = 0
        lo = max(0, n-1-k)
        hi = min(n-1, 2*n-2-k)
        # Effect of h[i] += di
        if lo <= i <= hi:
            j1 = i + k - n + 1
            if 0 <= j1 < n:
                delta += di * (1 - h[j1])
        i2_lo = max(0, k-n+1)
        i2_hi = min(n-1, k)
        if i2_lo <= i <= i2_hi:
            j2 = i - k + n - 1
            delta -= di * h[j2]
        # Effect of h[j] += dj
        if lo <= j <= hi:
            j1 = j + k - n + 1
            if 0 <= j1 < n:
                delta += dj * (1 - h[j1])
        if i2_lo <= j <= i2_hi:
            j2 = j - k + n - 1
            delta -= dj * h[j2]
        corr[k] += delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, n, k):
    """Analytical gradient of the k-th raw correlation w.r.t. h."""
    g = np.zeros(n)
    lo = max(0, n - 1 - k)
    hi = min(n - 1, 2 * n - 2 - k)
    # Term 1: i in [lo, hi], j1 = i + k - n + 1
    j1 = np.arange(lo, hi + 1) + k - n + 1
    g[lo:hi + 1] += 1.0 - h[j1] if len(j1) > 0 else 0
    # Term 2: i in [max(0,k-n+1), min(n-1,k)], j2 = i - k + n - 1
    i2_lo = max(0, k - n + 1)
    i2_hi = min(n - 1, k)
    if i2_lo <= i2_hi:
        j2 = np.arange(i2_lo, i2_hi + 1) - k + n - 1
        g[i2_lo:i2_hi + 1] -= h[j2]
    return g

def smooth_gradient(h, n, alpha, top_k=30):
    """Gradient of the alpha-softmax smoothing of max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    shifted = alpha * (c - c.max())
    exps = np.exp(shifted)
    p = exps / exps.sum()
    idx = np.argpartition(p, -top_k)[-top_k:]
    mask = p[idx] > 1e-12
    active_idx = idx[mask]
    g = np.zeros(n)
    for k in active_idx:
        g += p[k] * grad_corr_k(h, n, k)
    return dx * g

def slsqp_opt(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    def grad(x):
        return smooth_gradient(x, n, alpha)
    def cfun(x):
        return x.sum() - n / 2.0
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_descent(h, n, deadline, alpha=100000, lr_init=0.005, momentum=0.9):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    for it in range(20000):
        if time.time() > deadline:
            break
        g = smooth_gradient(h, n, alpha)
        g = g - g.mean()  # project to satisfy sum constraint
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            best_c = c_new
            best_h = h.copy()
            lr = min(lr * 1.05, lr_init * 2)
        else:
            v = 0.5 * v
            lr *= 0.7
            if lr < 1e-9:
                break
    return best_h, best_c

def grad_guided_pair(h, n, deadline, alpha=50000, n_sweeps=50):
    """Coordinate descent guided by the smooth gradient."""
    best_c = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        g = smooth_gradient(h, n, alpha)
        g = g - g.mean()
        improved = False
        order = np.argsort(np.abs(g))[::-1]
        for i in order:
            if time.time() > deadline:
                return h, best_c
            gi = g[i]
            if gi > 1e-8:
                # Want to decrease h[i], increase some j with g[j] < 0
                candidates = np.where(g < -0.3 * abs(gi))[0]
                candidates = candidates[candidates != i]
                if len(candidates) == 0:
                    candidates = np.argsort(g)[:5]
                    candidates = [c for c in candidates if c != i]
            elif gi < -1e-8:
                # Want to increase h[i], decrease some j with g[j] > 0
                candidates = np.where(g > -0.3 * abs(gi))[0]
                candidates = candidates[candidates != i]
                if len(candidates) == 0:
                    candidates = np.argsort(g)[-5:]
                    candidates = [c for c in candidates if c != i]
            else:
                continue
            for j in candidates[:5]:
                for d in [0.03, 0.01, 0.005, 0.001, 0.0003]:
                    if gi > 0:
                        if h[i] - d >= 0 and h[j] + d <= 1:
                            h[i] -= d; h[j] += d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                                break
                            h[i] += d; h[j] -= d
                            break
                    else:
                        if h[i] + d <= 1 and h[j] - d >= 0:
                            h[i] += d; h[j] -= d
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                                break
                            h[i] -= d; h[j] += d
                            break
    return h, best_c

def sim_anneal(h, n, deadline, iters=50000):
    """Simulated annealing on pairs."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    temp = 0.002
    for it in range(iters):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = np.random.uniform(0.001, 0.05) * (1 if np.random.rand() < 0.5 else -1)
        h[i] += d; h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= d; h[j] += d
            continue
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c; best_h = h.copy()
        elif np.random.rand() > np.exp((best_c - c) / temp):
            h[i] -= d; h[j] += d
        else:
            pass  # accept
        if it % 5000 == 0:
            temp *= 0.85
    return best_h, best_c

def full_optimize(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # SLSQP with increasing alpha
    for a in [2000, 10000, 50000, 200000]:
        if time.time() > deadline - 15:
            break
        ht = slsqp_opt(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c, best_h = c, ht
        else:
            break
    h = best_h
    # Gradient descent
    if time.time() < deadline - 20:
        alloc = min(60, (deadline - time.time()) * 0.4)
        h2, c2 = gradient_descent(h, n, time.time() + alloc,
                                  alpha=100000, lr_init=0.003)
        if c2 < best_c - 1e-14:
            best_c, best_h = c2, h2
    # Gradient-guided pair search
    if time.time() < deadline - 10:
        h3, c3 = grad_guided_pair(best_h, n, time.time() + min(40, (deadline - time.time()) * 0.4),
                                  alpha=50000)
        if c3 < best_c - 1e-14:
            best_c, best_h = c3, h3
    # Final SLSQP polish
    if time.time() < deadline - 3:
        ht = slsqp_opt(best_h, n, 500000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

        ht, ct = grad_guided_pair(best_h, n, deadline - 2, alpha=200000, n_sweeps=30)
        if ct < best_c:
            best_c, best_h = ct, ht
        ht = slsqp_opt(best_h, n, 1000000, maxiter=3000)
        ct = compute_c5(ht, n)
        if ct < best_c:
            best_c, best_h = ct, ht
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
