```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, deadline, steps):
    """Fast coordinate descent: try small moves at each position"""
    best_c = compute_c5(h, n)
    for step in steps:
        for _ in range(50):  # max sweeps
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                for d in [step, -step]:
                    if 0 <= h[i] + d <= 1:
                        # Move mass from i to spread across all others proportionally
                        # Simple: move to i+1 and i-1
                        for j in [(i+1) % n, (i-1) % n]:
                            if i == j:
                                continue
                            nd_i = h[i] + d
                            nd_j = h[j] - d
                            if 0 <= nd_i <= 1 and 0 <= nd_j <= 1:
                                h[i] = nd_i
                                h[j] = nd_j
                                c = compute_c5(h, n)
                                if c < best_c - 1e-15:
                                    best_c = c
                                    improved = True
                                    break
                                else:
                                    h[i] -= d
                                    h[j] += d
            if not improved:
                break
    return h, best_c

def pair_search_focused(h, n, deadline, steps, top_M=5):
    """Pair search focusing on active correlations"""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for step in steps:
        no_improve = 0
        while time.time() < deadline and no_improve < 3:
            corr = np.correlate(h, 1-h, mode='full') * dx
            top_idx = np.argsort(corr)[-top_M:]
            improved = False
            for k in top_idx:
                lag = k - (n-1)
                if lag < 0:
                    continue
                for i in range(n - lag):
                    j = i + lag
                    # Try moving mass i->j and j->i
                    for di_dj in [(step, -step), (-step, step)]:
                        nd_i = h[i] + di_dj[0]
                        nd_j = h[j] + di_dj[1]
                        if 0 <= nd_i <= 1 and 0 <= nd_j <= 1:
                            h[i] = nd_i
                            h[j] = nd_j
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                                # Don't revert, keep the move
                            else:
                                h[i] -= di_dj[0]
                                h[j] -= di_dj[1]
                    if time.time() > deadline:
                        return h, best_c
            if not improved:
                no_improve += 1
    return h, best_c

def full_pair_search(h, n, deadline, steps, radius=0):
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgradient_step(h, n, lr, threshold=0.005):

            if 0 <= j_plus < n:

            grad[j] += gj

                    if i == j:
                        continue
                    for di, dj in [(step, -step), (-step, step)]:
                        if -1e-12 <= h[i] + di <= 1 + 1e-12 and -1e-12 <= h[j] + dj <= 1 + 1e-12:
                            h[i] += di
                            h[j] += dj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= di
                                h[j] -= dj
            if not improved:
                no_improve += 1
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = project_feasible(h_init, n_init).copy()
    best_c = compute_c5(best_h, n_init)
    h = best_h.copy()

    # Phase 1: SLSQP with progressive refinement
    alphas = [300, 1000, 3000, 10000, 30000, 100000, 300000]
    for a in alphas:
        if time.time() > deadline - 120:
            break
        ht = slsqp(h, n_init, a, maxiter=2500, ftol=1e-15)
        c = compute_c5(ht, n_init)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
            h = ht

    # Phase 2: Subgradient descent
    if time.time() < deadline - 80:
        sg_deadline = min(deadline - 80, time.time() + 100)
        h_sg, c_sg = subgradient_descent(h, n_init, sg_deadline, lr0=0.001, decay=300, threshold=0.003)
        if c_sg < best_c - 1e-14:
            best_c = c_sg
            best_h = h_sg.copy()
            h = h_sg

    # Phase 3: SLSQP from subgradient result
    if time.time() < deadline - 60:
        for a in [50000, 200000, 500000]:
            if time.time() > deadline - 50:
                break
            ht = slsqp(h, n_init, a, maxiter=3000, ftol=1e-15)
            c = compute_c5(ht, n_init)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht.copy()
                h = ht

    # Phase 4: Focused pair search on active correlations
    if time.time() < deadline - 40:
        h, best_c = focused_pair(h, n_init, min(deadline - 20, time.time() + 80),
                                 [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001], top_M=5)
        best_h = h.copy()

    # Phase 5: SLSQP polish
    if time.time() < deadline - 25:
        ht = slsqp(h, n_init, 1000000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n_init)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
            h = ht

    # Phase 6: Final fine pair search
    if time.time() < deadline - 5:
        if n_init <= 150:
            h, best_c = local_pair(h, n_init, deadline - 2,
                                    [0.001, 0.0005, 0.0001], radius=n_init)
        else:
            h, best_c = local_pair(h, n_init, deadline - 2,
                                    [0.001, 0.0005, 0.0002], radius=40)
        best_h = h.copy()

    # Phase 7: Try other n values
    if time.time() < deadline - 50:
        for n_try in [100, 150, 200]:
            if time.time() > deadline - 20:
                break
            if n_try == n_init:
                continue
            x_new = np.linspace(0, 2, n_try)
            x_orig = np.linspace(0, 2, len(best_h))
            h_try = np.interp(x_new, x_orig, best_h)
            h_try = project_feasible(h_try, n_try)
            c_try = compute_c5(h_try, n_try)
            for a in [500, 5000, 50000]:
                if time.time() > deadline - 15:
                    break
                ht = slsqp(h_try, n_try, a, maxiter=2000, ftol=1e-15)
                c = compute_c5(ht, n_try)
                if c < c_try - 1e-14:
                    c_try = c
                    h_try = ht
            if n_try <= 120 and time.time() < deadline - 10:
                h_try, c_try = local_pair(h_try, n_try, min(deadline - 8, time.time() + 30),
                                           [0.001, 0.0005, 0.0001], radius=n_try)
            if c_try < best_c - 1e-14:
                best_c = c_try
                best_h = h_try.copy()

    # Final subgradient squeeze if time remains
    if time.time() < deadline - 10:
        n_best = len(best_h)
        h_s, c_s = subgradient_descent(best_h, n_best, deadline - 5, lr0=0.0005, decay=200, threshold=0.002)
        if c_s < best_c - 1e-14:
            best_c = c_s
            best_h = h_s

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
