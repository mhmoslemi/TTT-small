                for j1 in ones:
                    for j2 in ones:
                        if j2 <= j1:
                            continue
                        hb[i1], hb[i2], hb[j1], hb[j2] = 1.0, 1.0, 0.0, 0.0
                        c_new = np.correlate(hb, 1 - hb, mode='full')
                        nm = c_new.max()
                        imp = best_max - nm
                        if imp > best_imp:
                            best_imp = imp
                            best_moves = (i1, i2, j1, j2)
                        hb[i1], hb[i2], hb[j1], hb[j2] = 0.0, 0.0, 1.0, 1.0
        if best_moves is None:
            break
        i1, i2, j1, j2 = best_moves
        hb[i1] = 1.0; hb[i2] = 1.0; hb[j1] = 0.0; hb[j2] = 0.0
        c = np.correlate(hb, 1 - hb, mode='full')
        best_max = c.max()
        ones.remove(j1); ones.remove(j2)
        ones.append(i1); ones.append(i2)
        zeros.remove(i1); zeros.remove(i2)
        zeros.append(j1); zeros.append(j2)
    return hb, best_max * 2.0 / n

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.permutation(n)
    hb = np.zeros(n)
    hb[idx[:n//2]] = 1.0
    return hb

def structured_binary(n, seed):
    # Use modular arithmetic based structure
    rng = np.random.RandomState(seed)
    # Try golden ratio based
    phi = (1 + 5**0.5) / 2
    offset = rng.randint(n)
    hb = np.zeros(n)
    for i in range(n):
        if int((i + offset) * phi) % 2 == 0:
            hb[i] = 1.0
    # Ensure exactly n/2 ones
    if hb.sum() != n // 2:
        if hb.sum() > n // 2:
            ones_idx = np.where(hb == 1)[0]
            to_remove = rng.choice(ones_idx, size=int(hb.sum() - n//2), replace=False)
            hb[to_remove] = 0.0
        else:
            zeros_idx = np.where(hb == 0)[0]
            to_add = rng.choice(zeros_idx, size=int(n//2 - hb.sum()), replace=False)
            hb[to_add] = 1.0
    return hb

def optimize_for_n(h_start, n, deadline, seed=42):
    hb = to_binary(h_start, n)
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    
    # Greedy from rounded solution
    if time.time() < deadline - 5:
        hb, c = greedy_binary(hb, n, deadline - 5 * (len([1])) // 1, max_sweeps=200)
        if c < best_c:
            best_c = c
            best_hb = hb.copy()
    
    # 2-swap greedy
    if time.time() < deadline * 0.6:
        hb2, c2 = two_swap_greedy(best_hb, n, deadline * 0.6, max_rounds=30)
        if c2 < best_c:
            best_c = c2
            best_hb = hb2.copy()
    
    # SA
    if time.time() < deadline * 0.85:
        hb3, c3 = sa_binary(best_hb, n, deadline * 0.85, seed=seed)
        if c3 < best_c:
            best_c = c3
            best_hb = hb3.copy()
    
    # Final greedy polish
    if time.time() < deadline - 2:
        hb4, c4 = greedy_binary(best_hb, n, deadline - 2, max_sweeps=100)
        if c4 < best_c:
            best_c = c4
            best_hb = hb4.copy()
    
    # Try from random starts
    if time.time() < deadline * 0.9:
        for rs in range(3):
            if time.time() > deadline * 0.9:
                break
            hb_r = random_binary(n, seed + 100 + rs)
            hb_r, c_r = greedy_binary(hb_r, n, deadline * 0.9, max_sweeps=50)
            if c_r < best_c:
                best_c = c_r
                best_hb = hb_r.copy()
    
    return best_hb, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)

        n0 = 250
        h0 = np.full(n0, 0.5)
    
    best_c = compute_c5(to_binary(h0, n0), n0)
    best_h = to_binary(h0, n0).copy()
    best_n = n0
    
    # Primary optimization on n0
    primary_deadline = t0 + 0.55 * (budget_s - 10)
    hb, c = optimize_for_n(h0, n0, primary_deadline, seed=seed)
    if c < best_c:
        best_c = c
        best_h = hb.copy()
        best_n = n0
    
    # Try other n values
    n_options = [200, 220, 240, 260, 280, 300, 320]
    if n0 not in n_options:
        n_options.insert(0, n0)
    
    remaining = dl - time.time() - 30
    per_n = remaining / len(n_options)
    
    for ni in n_options:
        if time.time() > dl - 30:
            break
        if ni == n0:
            continue
        n_dl = min(time.time() + per_n, dl - 30)
        if n_dl <= time.time() + 2:
            continue
        if ni % 2 != 0:
            ni += 1
        h_start = np.full(ni, 0.5)
        # Use structured start
        h_s = structured_binary(ni, seed + ni)
        hb_i, c_i = optimize_for_n(h_s, ni, n_dl, seed=seed + ni)
        if c_i < best_c:
            best_c = c_i
            best_h = hb_i.copy()
            best_n = ni
    
    # Final SA polish on best
    if time.time() < dl - 5:
        hb_f, c_f = sa_binary(best_h, best_n, dl - 5, seed=seed + 999)
        if c_f < best_c:
            best_c = c_f
            best_h = hb_f.copy()
    
    # Quick greedy polish
    if time.time() < dl - 2:
        hb_g, c_g = greedy_binary(best_h, best_n, dl - 2, max_sweeps=50)
        if c_g < best_c:
            best_c = c_g
            best_h = hb_g.copy()
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n // 2
    if best_h.sum() != target:
        best_h = to_binary(best_h, best_n)
    
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
