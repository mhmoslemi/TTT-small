import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    """Compute ∂g_m/∂h[i] for all m, i. Returns matrix of shape (2n-1, n)."""
    dm = np.arange(2*n-1) - (n-1)
    a = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        d = int(dm[m])
        if d >= 0:
            if d < n:
                a[m, d:] += (1.0 - h[:n-d])
                a[m, :n-d] -= h[d:]
        else:
            dd = -d
            if dd < n:
                a[m, :n-dd] += (1.0 - h[dd:])
                a[m, dd:] -= h[:n-dd]
    return a

def outer_approx_phase(h, n, deadline, max_iters=200):
    """Outer approximation method for min max_k g_k."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    h_curr = h.copy()
    t = best_c
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_g(h_curr, n)
        # Verify actual C5
        actual_c = g.max()
        if actual_c < best_c:
            best_c = actual_c
            best_h = h_curr.copy()
        
        # Compute gradients at current point
        A = compute_grad_matrix(h_curr, n)
        
        # Solve LP: min 0 s.t. g_k(h_curr) + A[k,:]·(h - h_curr) ≤ t, 0≤h≤1, sum(h)=n/2
        h_var = cp.Variable(n)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
        for m in range(2*n-1):
            lin_val = g[m] + A[m] @ (h_var - h_curr)
            constraints.append(lin_val <= t)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, tol=1e-10, max_iters=500)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                break
        
        if h_var.value is None:
            # Infeasible, increase t
            t *= 1.01
            continue
        
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        
        # Check actual C5 of new solution
        c_new = compute_c5(h_new, n)
        if c_new <= t + 1e-10:
            # Feasible! Can try to reduce t
            t = max(c_new * 0.999, c_new - 0.001)
        else:
            # Linearization was too optimistic, update linearization point
            t = max(t, c_new * 1.001)
        
        h_curr = h_new
        # Dampening: if h changed a lot, step back
        if np.abs(h_curr - h).sum() > n * 0.5:
            h_curr = 0.5 * (h_curr + h)
            h_curr = project_feasible(h_curr, n)
            h = h_curr
    
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        d = int(dm[m])
        if d >= 0 and d < n:
            A[m, d:] += (1.0 - h[:n - d])
            A[m, :n - d] -= h[d:]
        elif d < 0:
            dd = -d
            if dd < n:
                A[m, :n - dd] += (1.0 - h[dd:])
                A[m, dd:] -= h[:n - dd]
    return A

def solve_lp_feasibility(h_curr, n, t, g_curr, A):
    A_ub = A.copy()
    b_ub = t - g_curr + A @ h_curr
    A_eq = np.ones((1, n))
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n
    try:
        res = linprog(np.zeros(n), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def outer_approx(h, n, deadline, max_iters=500):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    t = best_c * 1.001
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        g = compute_g(h_curr, n)
        actual_c = float(g.max())
        if actual_c < best_c - 1e-14:
            best_c = actual_c
            best_h = h_curr.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                break
        A = compute_grad_matrix(h_curr, n)
        h_new = solve_lp_feasibility(h_curr, n, t, g, A)
        if h_new is None:
            t *= 1.005
            continue
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new <= t + 1e-10:
            t = max(c_new * 0.9995, c_new - 0.0005)
        else:
            t = max(t * 1.002, c_new * 0.999)
        step = np.linalg.norm(h_new - h_curr) / np.sqrt(n)
        if step > 0.3:
            h_curr = 0.5 * (h_curr + h_new)
            h_curr = project_feasible(h_curr, n)
        else:
            h_curr = h_new
    return best_h, best_c

def slsqp_optimize(h, n, alpha, deadline, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(g)[::-1][:K]
        A = compute_grad_matrix(h, n)
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = A[:, i] - A[:, j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                lo, hi = d_lo, d_hi
                for _ in range(25):
                    mid = (lo + hi) / 2
                    vals = g + mid * s
                    new_max = vals.max()
                    mid_plus = mid + (hi - lo) * 0.25
                    mid_minus = mid - (hi - lo) * 0.25
                    if mid_plus <= d_hi:
                        if np.max(g + mid_plus * s) < new_max:
                            lo = mid
                        else:
                            hi = mid
                    else:
                        hi = mid
                    if mid_minus >= d_lo:
                        if np.max(g + mid_minus * s) < new_max:
                            hi = mid
                        else:
                            lo = mid
                    else:
                        lo = mid
                best_delta = (lo + hi) / 2
                best_val = float(np.max(g + best_delta * s))
                if best_val < current_max - 1e-13:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * s
                    current_max = g.max()
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
    return best_h, best_c

def random_perturb(h, n, scale, rng):
    h_new = h.copy()
    mask = rng.random(n) < 0.3
    h_new[mask] += rng.randn(mask.sum()) * scale
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm start
    p1_end = t0 + 0.12 * (budget_s - 8)
    h = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > p1_end:
            break
        h = slsqp_optimize(h, n, alpha, p1_end, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Outer approximation (LP-based)
    p2_end = t0 + 0.45 * (budget_s - 8)
    h, c = outer_approx(best_h.copy(), n, p2_end, max_iters=400)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Pair descent
    p3_end = t0 + 0.78 * (budget_s - 8)
    h, c = pair_descent(best_h.copy(), n, p3_end, max_sweeps=800)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 4: SLSQP polish
    if time.time() < dl - 60:
        h = slsqp_optimize(best_h.copy(), n, 500000, dl - 60, maxiter=300)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 5: Final pair descent
    if time.time() < dl - 5:
        h, c = pair_descent(best_h.copy(), n, dl - 3, max_sweeps=300)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 6: Try random perturbations if time remains
    if time.time() < dl - 30:
        for attempt in range(5):
            if time.time() > dl - 30:
                break
            h_pert = random_perturb(best_h, n, 0.05, rng)
            h_pert, c_pert = pair_descent(h_pert, n, dl - 30, max_sweeps=50)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
