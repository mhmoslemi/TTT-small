def subgradient_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    C = corr  # array of length 2n-1
    max_C = C.max()
    active = C >= max_C - 1e-10  # or some tolerance
    # Subgradient: average of gradients of active C_k
    # Or just the steepest one
    grad = np.zeros(n)
    n_active = active.sum()
    if n_active == 0:
        return h
    for k in range(len(C)):
        if not active[k]:
            continue
        shift = k - n + 1
        lo1 = max(0, -shift)
        hi1 = min(n - 1, n - 1 - shift)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1 - h[lo1+shift:hi1+shift+1])
        lo2 = max(0, shift)
        hi2 = min(n - 1, n - 1 + shift)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2-shift:hi2-shift+1]
    grad *= dx / n_active
    # Projected gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

One known result: for the continuous version, the optimal C_5 is related to the minimum of max_k ∫ h(x)h(x+k) dx (which is slightly different from our problem). Actually our problem is ∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx = 1 - ∫ h(x)h(x+k) dx (by the constraint ∫h = 1).

Wait! That's a key insight:
∫_0^∞ h(x)(1-h(x+k)) dx

Hmm, but the integral is over what domain? Let me re-read the problem.

The problem states: C₅ = max_k ∫ h(x)(1 - h(x+k)) dx

And h: [0,2] → [0,1]. So h is defined on [0,2] and we're integrating over x such that both x and x+k are in [0,2]? Or is h extended to be 0 outside [0,2]?

Looking at the discretization: we have n_points samples over [0,2], and we compute np.correlate(h, 1-h, mode='full'). The "full" correlation gives all possible shifts, and the values outside the overlap region are 0 (since one of the arrays is 0 there). So effectively, h is zero-padded.

So C_k = dx * sum_{i: valid} h[i] * (1 - h[i+k-n+1])

And the integral ∫ h(x)(1-h(x+k)) dx ≈ C_k (with the zero extension).

Now, ∫_all h(x) dx = 1 (by constraint). But ∫_all h(x+h) dx is only over the overlap region.

So C_k = ∫_overlap h(x) dx - ∫_overlap h(x)h(x+k) dx

This is not simply 1 - autocorrelation because the integration domain changes with k.

OK so the key point is that for large |k|, the overlap region is small, so C_k is small. The maximum is achieved for small k.

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sh_max = sh.max()
    if sh_max > 500:
        return float(m + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_eq(x, n):
    return float(x.sum() - n / 2.0)

def _sum_eq_jac(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_subgrad(h, n, tol_frac=0.02):
    """Compute subgradient of max correlation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    active = corr >= max_c - tol_frac * max_c
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if not active[k]:
            continue
        shift = k - n + 1
        lo1 = max(0, -shift)
        hi1 = min(n - 1, n - 1 - shift)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1+shift:hi1+shift+1])
        lo2 = max(0, shift)
        hi2 = min(n - 1, n - 1 + shift)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2-shift:hi2-shift+1]
    n_active = max(1, int(active.sum()))
    grad *= dx / n_active
    return grad, max_c

def subgrad_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = 0
    patience = 0
    while time.time() < deadline:
        t += 1
        grad, max_c = compute_subgrad(h, n, tol_frac=0.01)
        step = 0.5 / (1.0 + 0.05 * t)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            h = h_new  # still move, might find a better region later
            patience += 1
            if patience > 500:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def near_pair_search(h, n, deadline, deltas, radius=50):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:

                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
            if not improved:
                break
    return h, best_c

def optimize_single(h_start, n, deadline, rng):

    best_c = compute_c5(h, n)
    
    # Phase A: Subgradient (fast global)
    t_sg_end = time.time() + min(60, (deadline - time.time()) * 0.2)
    h_sg, c_sg = subgrad_descent(h, n, t_sg_end)
    if c_sg < best_c:
        best_c = c_sg
        h = h_sg

    alphas = [200, 1000, 5000, 20000, 100000]

            h = ht

                                     radius=60)

    best_c = compute_c5(h, n)
    return h, best_c
