from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n, rng):

    target = n // 2

    # Sort by value, keep top n/2 as 1
    idx = np.argsort(h_copy)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0

            sm = sh.max()
            soft = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        else:
            soft = m

        return x.sum() - n / 2.0

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def project_feasible(h, n):
