            break

            if s < 1e-14:

            if s < 1e-14:

        im = i - dm

                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(-h, kind='mergesort')[:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for i in range(n):
        ip = i + dm
        im = i - dm
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        F[i, vp] = 1.0 - h[ip[vp]]
        F[i, vm] -= h[im[vm]]
    return F

def update_F_single(F, pos, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    j1 = pos - dm
    v1 = (j1 >= 0) & (j1 < n)
    F[j1[v1], m_idx[v1]] -= delta
    j2 = pos + dm
    v2 = (j2 >= 0) & (j2 < n)
    F[j2[v2], m_idx[v2]] -= delta

def binary_local_search(h, n, deadline, rng, max_sweeps=50000):
    """Global-best-swap local search on binary solutions."""
    h_bin = h.copy()
    g = compute_g(h_bin, n)
    F = compute_F(h_bin, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    best_c = g.max()
    ones = np.where(h_bin == 1)[0].copy()
    zeros = np.where(h_bin == 0)[0].copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        # Find best swap: min over (i in ones, j in zeros) of max_k(g[k] + F[j][k] - F[i][k])
        M = g[None, None, :] + F[zeros][None, :, :] - F[ones][:, None, :]
        new_maxes = M.max(axis=2)
        min_pos = np.argmin(new_maxes)
        min_val = new_maxes.flat[min_pos]
        
        if min_val < best_c - 1e-14:
            i_idx, j_idx = np.unravel_index(min_pos, new_maxes.shape)
            i = ones[i_idx]
            j = zeros[j_idx]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            g += F[j] - F[i]
            update_F_single(F, i, -1.0, n, dm)
            update_F_single(F, j, 1.0, n, dm)
            ones[i_idx] = j
            zeros[j_idx] = i
            best_c = g.max()
        else:
            break
    
    return h_bin, best_c
