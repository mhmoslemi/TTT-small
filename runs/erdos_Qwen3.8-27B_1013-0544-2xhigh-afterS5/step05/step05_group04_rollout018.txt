import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_smoothed(h, n, alpha):
    """Compute gradient of the alpha-smoothed max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    p = np.exp(sh - mx)
    p /= p.sum()
    
    # A[j] = convolve(p[:n], 1-h, 'full')[:n]
    A = np.convolve(p[:n], 1 - h, 'full')[:n]
    
    # B[j] = sum_{k=0}^{n-1-j} p[k] * h[j+k]
    B = np.zeros(n)
    for j in range(n):
        B[j] = np.dot(p[:n-j], h[j:])
    
    grad = dx * (A - B)
    return grad, m

def subgradient_optimize(h, n, deadline, alpha_start=100, alpha_end=50000, n_iters=None):
    """Projected subgradient method with adaptive learning rate."""
    if n_iters is None:
        n_iters = 200000
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    # Adaptive alpha schedule
    n_total = n_iters
    step = 1.0
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(n_total):
        if time.time() > deadline:
            break
        # Adaptive alpha
        progress = it / n_total
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        grad, m = compute_grad_smoothed(h, n, alpha)
        
        # Momentum update
        momentum = beta * momentum - step * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        # Adaptive step size (reduced if objective worsens)
        try:
            if c > m + 0.01:  # heuristic
                step *= 0.95
            else:
                step = min(step * 1.02, 0.1)
        except:
            pass
        
        h = h_new
        
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    """SLSQP optimization with smoothed objective."""
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        p = np.exp(sh - mx)
        p /= p.sum()
        A = np.convolve(p[:n], 1 - x, 'full')[:n]
        B = np.zeros(n)
        for j in range(n):
            B[j] = np.dot(p[:n-j], x[j:])
        return dx * (A - B)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, args=(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    js = list(range(i+1, n))
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                    js = [j for j in range(lo, hi) if j != i]
                for j in js:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    out = np.correlate(h, 1-h, mode='full') * dx
    m = out.max()
    sh = alpha * (out - m)
    mx = sh.max()
    p = np.exp(sh - mx)
    p /= p.sum()
    A = np.convolve(p[n-1:], 1-h, 'full')[:n]
    B = np.zeros(n)
    for j in range(n):
        B[j] = np.dot(p[n-1-j : 2*n-1-j], h)
    return dx * (A - B), m

def grad_optimize(h, n, deadline, alpha_start=50, alpha_end=100000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.05
    momentum = np.zeros(n)
    beta = 0.8
    n_bad = 0
    total_iters = 0
    
    while time.time() < deadline:
        total_iters += 1
        progress = min(1.0, total_iters / 100000)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        grad, m = smooth_grad(h, n, alpha)
        momentum = beta * momentum - step * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            n_bad = 0
            step = min(step * 1.03, 0.2)
        else:
            n_bad += 1
            if n_bad > 200:
                step *= 0.8
                n_bad = 0
                momentum *= 0.5
        
        h = h_new
    return best_h, best_c
```
