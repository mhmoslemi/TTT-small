import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def simulated_annealing(hb, n, deadline, T0=0.1, T1=0.001, n_iter=500000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    T = T0
    cool_rate = (T1 / T0) ** (1.0 / max(n_iter, 1))
    
    ones = np.where(cur_h == 1)[0].copy()
    zeros = np.where(cur_h == 0)[0].copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 50 == 0 and ones.size != n // 2:
            # Safety check
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
        
        # Random swap
        zi = rng.randint(zeros.size)
        oi = rng.randint(ones.size)
        a = zeros[zi]
        b = ones[oi]
        
        cur_h[a], cur_h[b] = 1.0, 0.0
        new_c = compute_c5(cur_h, n)
        
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c = new_c
            # Update sets
            zeros[zi] = b
            ones[oi] = a
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[a], cur_h[b] = 0.0, 1.0
        
        T *= cool_rate
    
    return best_h, best_c

def greedy_local_search(hb, n, deadline, max_sweeps=200):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = set(np.where(hb == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for a in sorted(zeros):
            if time.time() > deadline:
                break
            for b in sorted(ones):
                best_h[a] = 1.0
                best_h[b] = 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(b)
                    ones.add(a)
                    zeros.discard(a)
                    zeros.add(b)
                    improved = True
                    break
                else:
                    best_h[a] = 0.0
                    best_h[b] = 1.0
        if not improved:
            break
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def simulated_annealing(hb, n, deadline, T0, T1, n_iter, rng):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    T = T0
    cool_rate = (T1 / T0) ** (1.0 / max(n_iter, 1))
    
    ones_list = list(np.where(cur_h == 1)[0])
    zeros_list = list(np.where(cur_h == 0)[0])
    
    no_improve = 0
    kick_interval = max(n_iter // 10, 10000)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % kick_interval == kick_interval - 1 and no_improve > kick_interval // 2:
            # Kick: random perturbation
            for _ in range(n // 10):
                a = zeros_list[rng.randint(len(zeros_list))]
                b = ones_list[rng.randint(len(ones_list))]
                cur_h[a], cur_h[b] = 1.0, 0.0
                zeros_list[zeros_list.index(a)] = b
                ones_list[ones_list.index(b)] = a
            cur_c = compute_c5(cur_h, n)
            no_improve = 0
            continue
        
        a_idx = rng.randint(len(zeros_list))
        b_idx = rng.randint(len(ones_list))
        a = zeros_list[a_idx]
        b = ones_list[b_idx]
        
        cur_h[a] = 1.0
        cur_h[b] = 0.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c = new_c
            zeros_list[a_idx] = b
            ones_list[b_idx] = a
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur_h[a] = 0.0
            cur_h[b] = 1.0
        
        T *= cool_rate
    
    return best_h, best_c

def greedy_local_search(hb, n, deadline):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = list(np.where(hb == 0)[0].tolist())
    
    while time.time() < deadline:
        improved = False
        for a in zeros:
            if time.time() > deadline:
                return best_h, best_c
            best_h[a] = 1.0
            for b in sorted(ones):
                best_h[b] = 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(b)
                    zeros.remove(a)
                    ones.add(a)
                    zeros.append(b)
                    improved = True
                    break
                best_h[b] = 1.0
            if improved:
                break
            best_h[a] = 0.0
        if not improved:
            break
    
    return best_h, best_c

One idea: instead of SLSQP (which is slow for n=250), use a custom projected subgradient method for the minimax problem. The subgradient of max_k c[k] with respect to h is just c[argmax_k c[k]] (the subgradient of the max).

But the problem is that c[k] = ∫ h(x)(1-h(x+k)) dx, and dc[k]/dh[i] depends on the structure.

Let me compute: c[k] = sum_j h[j] * (1-h[j-(k-(n-1))])

dc[k]/dh[i] = (1-h[i-(k-(n-1))]) - h[i+(k-(n-1))]  (for valid indices)

Wait, let me be more careful:
c[m] = sum_j h[j] * (1-h[j - lag]) where lag = m - (n-1)

dc[m]/dh[i] = contribution of i as j: (1-h[i-lag]) 
             + contribution of i as j-lag (i.e., j = i+lag): -h[i+lag]

So dc[m]/dh[i] = (1-h[i-lag]) - h[i+lag] (with terms that are 0 if index is out of range)

For the minimax objective C = max_m c[m], the subgradient is:
dC/dh[i] = sum_{m: c[m] = C} w_m * dc[m]/dh[i]

where w_m are the weights (summing to 1) for the active components. A simple choice is the uniform weight over active components, or just pick the single maximizing m.

For the projected subgradient method:
h_{t+1} = Project(h_t - alpha * g_t)

where g_t is the subgradient and Project clips to [0,1] and adjusts to satisfy the sum constraint.

This is similar to what the previous code was doing with the "analytic_grad" function, but using the exact subgradient instead of the softmax approximation.

Actually, for the subgradient, I can compute it more efficiently:

g[i] = dx * sum_{active m} w_m * [(1-h[i-(m-(n-1))]) - h[i+(m-(n-1))]]

If there's only one active m (which is common), then:
g[i] = dx * [(1-h[i-lag]) - h[i+lag]]

where lag = m - (n-1).

This is just: g = dx * [(1-h) shifted by lag, minus h shifted by -lag]

Let me redo: c[m] = sum_i h[i] * (1-h[i-lag])

dc/dh[j] = (1-h[j-lag]) + sum_i h[i] * d(1-h[i-lag])/dh[j]

d(1-h[i-lag])/dh[j] = -delta(i-lag, j) = -delta(i, j+lag)

So the second term: sum_i h[i] * (-delta(i, j+lag)) = -h[j+lag] (if j+lag is in range)

So dc/dh[j] = (1-h[j-lag]) - h[j+lag] (with terms 0 if index out of range)

For lag > 0:
- (1-h[j-lag]): j-lag ≥ 0 means j ≥ lag. So for j ≥ lag: (1-h[j-lag]), for j < lag: 0
- h[j+lag]: j+lag < n means j < n-lag. So for j < n-lag: h[j+lag], for j ≥ n-lag: 0

g[j] = (1-h[j-lag]) * [j ≥ lag] - h[j+lag] * [j < n-lag]

Let me now think about the overall algorithm more carefully. I think the most promising approach is:

1. Start from the parent binary solution
2. Run simulated annealing with a good cooling schedule
3. Polish with greedy local search
4. Repeat with different random seeds / perturbations

The key insight is that for this problem, the binary solutions are what matter (the continuous relaxation is just a way to find good binary solutions). So I should focus most of my time on the binary search.

Let me also think about the cooling schedule. For simulated annealing to work well:
- T0 should be high enough to accept most moves initially (exploration)
- T1 should be low enough to only accept improvements at the end (exploitation)
- The cooling rate should be slow enough that the system has time to equilibrate at each temperature

For our problem, the "energy" scale is C₅ ≈ 0.38. The difference between a good move and a bad move is typically small (maybe 0.001-0.01 in C₅). So T should be on the scale of 0.001-0.01.

Let me set T0 = 0.01 and T1 = 0.0001, with geometric cooling over N iterations.

If N = 100,000 iterations:
cool_rate = (T1/T0)^(1/N) = 0.01^(1/100000) = e^(ln(0.01)/100000) = e^(-0.000046) ≈ 0.99995

So the temperature changes very slowly. That's fine.

At T = 0.01, a move that worsens C by 0.001 is accepted with probability exp(-0.001/0.01) = exp(-0.1) ≈ 0.90.
At T = 0.001, same move is accepted with probability exp(-0.001/0.001) = exp(-1) ≈ 0.37.
At T = 0.0001, same move is accepted with probability exp(-0.001/0.0001) = exp(-10) ≈ 0.00005.

So the cooling schedule transitions from "accept almost everything" to "accept only improvements" over the course of the run. Good.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def sa_run(hb, n, deadline, T0, T1, rng):
    """Simulated annealing on binary solution."""
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    T = T0
    
    ones = list(np.where(cur_h == 1)[0])
    zeros = list(np.where(cur_h == 0)[0])
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    it = 0
    no_improve = 0
    max_it = 2000000
    
    while time.time() < deadline:
        a_idx = rng.randint(n_zeros)
        b_idx = rng.randint(n_ones)
        a = zeros[a_idx]
        b = ones[b_idx]
        
        cur_h[a] = 1.0
        cur_h[b] = 0.0
        new_c = compute_c5(cur_h, n)
        d = new_c - cur_c
        
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_c = new_c
            zeros[a_idx] = b
            ones[b_idx] = a
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            cur_h[a] = 0.0
            cur_h[b] = 1.0
        
        T = T0 * (T1 / T0) ** (it / max_it)
        it += 1
    
    return best_h, best_c

def greedy_search(hb, n, deadline):
    """Greedy local search: try all single swaps, accept first improvement."""
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    
    while time.time() < deadline:
        ones = sorted(np.where(best_h == 1)[0].tolist())
        zeros = sorted(np.where(best_h == 0)[0].tolist())
        improved = False
        for a in zeros:
            if time.time() > deadline:
                return best_h, best_c
            best_h[a] = 1.0
            for b in ones:
                best_h[b] = 0.0
                c = compute_c5(best_h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    break
                best_h[b] = 1.0
            if improved:
                break
            best_h[a] = 0.0
        if not improved:
            break
    
    return best_h, best_c

def subgrad_step(h, n, deadline, alpha_init=0.1, max_iter=100000):
    """Subgradient method for the continuous relaxation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    dx = 2.0 / n
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        active = np.where(np.abs(c - m) < 1e-10)[0]
        
        g = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            wk = 1.0 / len(active)
            if lag > 0:
                t1 = np.zeros(n); t1[lag:] = 1.0 - h[:n-lag]
                t2 = np.zeros(n); t2[:n-lag] = h[lag:]
            elif lag < 0:
                ll = -lag
                t1 = np.zeros(n); t1[:n-ll] = 1.0 - h[ll:]
                t2 = np.zeros(n); t2[ll:] = h[:n-ll]
            else:
                t1 = np.zeros(n); t2 = np.zeros(n)
            g += wk * (t1 - t2)
        g *= dx
        
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            alpha = min(alpha * 1.02, 5.0)
        else:
            alpha *= 0.95
            stall += 1
            if stall > 300 or alpha < 1e-10:
                break
    
    return best_h, best_c

```python
def compute_subgrad(h, n):
    """Compute the subgradient of C(h) = max_k c_k(h) at point h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    active = np.where(c >= m - 1e-12)[0]
    
    g = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        wk = 1.0 / len(active)
        if lag > 0:
            t1 = np.zeros(n); t1[lag:] = 1.0 - h[:n-lag]
            t2 = np.zeros(n); t2[:n-lag] = h[lag:]
        elif lag < 0:
            ll = -lag
            t1 = np.zeros(n); t1[:n-ll] = 1.0 - h[ll:]
            t2 = np.zeros(n); t2[ll:] = h[:n-ll]
        else:
            continue  # lag=0: c[0] = sum h[i](1-h[i]) = 0, gradient is 0
        g += wk * (t1 - t2)
    return g * dx, m * dx
```
