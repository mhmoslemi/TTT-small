        cur_c = corr.max() * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12: break

            s = h.sum()
            if s < 1e-12: break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_hill_climb(h, n, deadline, max_sweeps=200, subset_size=25, seed=42):
    m_range = np.arange(-(n - 1), n, dtype=np.int64)
    n_lags = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        ns = min(subset_size, len(zeros))
        z_sample = rng.choice(zeros, size=ns, replace=False)
        for a in z_sample:
            am_p = a + m_range
            am_m = a - m_range
            A = np.zeros(n_lags)
            mp = (am_p >= 0) & (am_p < n)
            A[mp] -= h[am_p[mp]]
            mm = (am_m >= 0) & (am_m < n)
            A[mm] += 1.0 - h[am_m[mm]]
            bp_p = ones[:, None] + m_range[None, :]
            bp_m = ones[:, None] - m_range[None, :]
            mp2 = (bp_p >= 0) & (bp_p < n)
            mm2 = (bp_m >= 0) & (bp_m < n)
            B = np.zeros((len(ones), n_lags))
            B[mp2] = h[bp_p[mp2]]
            B[mm2] -= (1.0 - h[bp_m[mm2]])
            new_max = (corr[None, :] + A[None, :] + B).max(axis=1)
            bi = np.argmin(new_max)
            nc = new_max[bi] * dx
            if nc < cur_c - 1e-15:
                h[a], h[ones[bi]] = 1.0, 0.0
                corr += A + B[bi]
                cur_c = nc
                ones = np.where(h == 1)[0]
                improved = True
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def sa_anneal(h, n, deadline, T0=0.4, T1=0.005, n_iter=200000, seed=42):
    m_range = np.arange(-(n - 1), n, dtype=np.int64)
    n_lags = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    T = T0
    cooldown = (T1 / T0) ** (1.0 / max(n_iter, 10))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooldown
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        a = rng.choice(zeros)
        b = rng.choice(ones)
        am_p = a + m_range
        am_m = a - m_range
        A = np.zeros(n_lags)
        mp = (am_p >= 0) & (am_p < n)
        A[mp] -= h[am_p[mp]]
        mm = (am_m >= 0) & (am_m < n)
        A[mm] += 1.0 - h[am_m[mm]]
        bm_p = b + m_range
        bm_m = b - m_range
        Bv = np.zeros(n_lags)
        mp2 = (bm_p >= 0) & (bm_p < n)
        Bv[mp2] = h[bm_p[mp2]]
        mm2 = (bm_m >= 0) & (bm_m < n)
        Bv[mm2] -= (1.0 - h[bm_m[mm2]])
        dv = A + Bv
        nc = (corr + dv).max() * dx
        dc = nc - cur_c
        if dc < 0 or rng.random() < np.exp(-min(-dc, 40.0) / max(T, 1e-12)):
            h[a], h[b] = 1.0, 0.0
            corr += dv
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def smart_sa(h, n, deadline, T0=0.3, T1=0.003, n_iter=200000, seed=42):
    """SA with biased selection: prefer moves near high-correlation lags."""
    m_range = np.arange(-(n - 1), n, dtype=np.int64)
    n_lags = 2 * n - 1
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').copy()
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    T = T0
    cooldown = (T1 / T0) ** (1.0 / max(n_iter, 10))
    # Track "hot" positions: indices where h contributes most to max correlation
    hot_mask = np.zeros(n)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T *= cooldown
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Bias: pick zero from hot positions with 50% probability
        if rng.random() < 0.5:
            hw = hot_mask[zeros]
            tw = hw.sum()
            if tw > 1e-10:
                probs = hw / tw
                a = rng.choice(zeros, p=probs)
            else:
                a = rng.choice(zeros)
        else:
            a = rng.choice(zeros)
        b = rng.choice(ones)
        am_p = a + m_range
        am_m = a - m_range
        A = np.zeros(n_lags)
        mp = (am_p >= 0) & (am_p < n)
        A[mp] -= h[am_p[mp]]
        mm = (am_m >= 0) & (am_m < n)
        A[mm] += 1.0 - h[am_m[mm]]
        bm_p = b + m_range
        bm_m = b - m_range
        Bv = np.zeros(n_lags)
        mp2 = (bm_p >= 0) & (bm_p < n)
        Bv[mp2] = h[bm_p[mp2]]
        mm2 = (bm_m >= 0) & (bm_m < n)
        Bv[mm2] -= (1.0 - h[bm_m[mm2]])
        dv = A + Bv
        nc = (corr + dv).max() * dx
        dc = nc - cur_c
        if dc < 0 or rng.random() < np.exp(-min(-dc, 40.0) / max(T, 1e-12)):
            h[a], h[b] = 1.0, 0.0
            corr += dv
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        # Update hot mask every 100 iterations
        if it % 100 == 99:
            corr_max = corr.max()
            hot_mask = np.zeros(n)
            active = np.where(corr > corr_max * 0.95)[0]
            for ki in active:
                lo = max(0, (ki - (n - 1)))
                hi = min(n, ki + 1)
                for j in range(lo, hi):
                    lag = ki - (n - 1)
                    # h[j+lag]*(1-h[j]) contributes; h[j] contributes as second factor
                    if 0 <= j + lag < n:
                        hot_mask[j + lag] += 1
                    hot_mask[j] += 1
            hs = hot_mask.sum()
            if hs > 0:
                hot_mask /= hs
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def optimize_at_n(h_init, n, deadline, seed=42):
    """Full optimization pipeline at a given resolution."""
    rng = np.random.RandomState(seed)
    h = make_binary(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Greedy
    t1 = min(deadline - 0.3 * (deadline - time.time()), time.time() + 8)
    hb, cb = greedy_hill_climb(h, n, t1, max_sweeps=150, subset_size=25, seed=seed)
    if cb < best_c:
        best_c, best_h = cb, hb
    # SA
    t2 = min(deadline - 0.15 * (deadline - time.time()), time.time() + 15)
    hb, cb = sa_anneal(best_h, n, t2, T0=0.4, T1=0.005, n_iter=150000, seed=seed + 5)
    if cb < best_c:
        best_c, best_h = cb, hb
    # Smart SA
    t3 = min(deadline, time.time() + 15)
    hb, cb = smart_sa(best_h, n, t3, T0=0.3, T1=0.003, n_iter=150000, seed=seed + 10)
    if cb < best_c:
        best_c, best_h = cb, hb
    # Final greedy
    hb, cb = greedy_hill_climb(best_h, n, deadline, max_sweeps=80, subset_size=30, seed=seed + 20)
    if cb < best_c:
        best_c, best_h = cb, hb
    return best_h, best_c
