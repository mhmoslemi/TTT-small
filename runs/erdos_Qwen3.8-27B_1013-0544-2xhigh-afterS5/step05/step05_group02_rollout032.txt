def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj(x, n, alpha):
    dx = 2.0 / n
    hx = np.clip(x, 0.0, 1.0)
    c = np.correlate(hx, 1 - hx, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    return float(x.sum() - n / 2.0)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        step = lr * grad / grad_norm
        
        # Backtracking
        t = 1.0
        f_curr = m  # approximate
        while t > 1e-6:
            h_new = h - t * step
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < f_curr - 0.1 * t * grad_norm / (2.0/n):
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                break
            t *= 0.5
        else:
            # Try a smaller fixed step
            h_new = h - 0.001 * step
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    hx = np.clip(x, 0.0, 1.0)
    c = np.correlate(hx, 1 - hx, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def slsqp_round(h, n, alpha, iters=4000):
    b = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'args': (n,)}]
    try:
        r = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                    bounds=b, constraints=cons,
                    options={'maxiter': iters, 'ftol': 1e-18})
        return project_feasible(r.x, n)
    except Exception:
        return h.copy()

def thue_morse(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    for kid in range(2 * n - 1):
        if w[kid] < 1e-11:
            continue
        s = kid - (n - 1)
        wi = w[kid]
        for j in range(n):
            g = 1.0
            if 0 <= j + s < n:
                g -= h[j + s]
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] += wi * g
    return grad

def proj_grad_method(h, n, deadline, alpha=20000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.7
    lr = 0.02
    stuck = 0
    while time.time() < deadline:
        grad = smooth_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        v = beta * v - lr * grad / gn
        h_try = project_feasible(h + v, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try.copy()
            h = h_try
            stuck = 0
            lr = min(lr * 1.05, 0.05)
        else:
            h = h + 0.5 * (h_try - h)
            h = project_feasible(h, n)
            stuck += 1
            if stuck > 200:
                beta = 0.5
                lr = 0.01
                v = np.zeros(n)
                stuck = 0
                break
    return best_h, best_c

def binary_sa(h_bin, n, deadline, seed_val=7):
    h = h_bin.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed_val)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    T = 0.3
    it = 0
    while time.time() < deadline and T > 0.005:
        it += 1
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        nc = compute_c5(h, n)
        if nc < best_c or rng.random() < np.exp(-(nc - best_c) / T):
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0
        if it % 500 == 0:
            T *= 0.97
    return best_h, best_c

def pair_polish(h, n, deadline, seed_val=99):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    rng = np.random.RandomState(seed_val)
    stall = 0
    while time.time() < deadline and stall < 3000:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-14:
            best_c = curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 1.0
            if 0 <= j + s < n:
                g -= h[j + s]
            if 0 <= j - s < n:
                g -= h[j - s]
            grad[j] = g
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            continue
        d = 0.004
        d = min(d, h[i_src] * 0.9, (1.0 - h[i_dst]) * 0.9)
        if d < 1e-8:
            continue
        h[i_src] -= d
        h[i_dst] += d
        nc = compute_c5(h, n)
        if nc >= curr + 1e-15:
            h[i_src] += d
            h[i_dst] -= d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    try:

    except Exception:

    best_c = compute_c5(best_h, n0)

        ht = slsqp_round(best_h, n0, alpha, iters=3000)
        ct = compute_c5(ht, n0)
        if ct < best_c:

            best_h = ht

    # Phase 2: Try other n values (20%)
    p2_end = t0 + (deadline - t0) * 0.50

        if time.time() > p2_end:
            break
        if n_try == n0:
            continue
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_try)
        hi = np.interp(x_new, x_old, best_h)
        if n_try == 256:
            tm = thue_morse(256)
            hi = 0.6 * hi + 0.4 * tm
        hi = project_feasible(hi, n_try)
        for alpha in [10000, 100000]:

            ct = compute_c5(ht, n_try)
            if ct < best_c:
                best_c = ct
                best_h = ht
                n0 = n_try

    # Phase 3: Projected gradient with momentum (20%)
    p3_end = t0 + (deadline - t0) * 0.70
    n_cur = len(best_h)
    best_h, best_c = proj_grad_method(best_h, n_cur, p3_end, alpha=30000)

    # Phase 4: Binary SA (15%)
    p4_end = t0 + (deadline - t0) * 0.85
    h_bin = (best_h > 0.5).astype(float)
    if h_bin.sum() != n_cur // 2:
        if h_bin.sum() > n_cur // 2:
            idx = np.where(h_bin == 1)[0]
            for k in range(int(h_bin.sum()) - n_cur // 2):
                h_bin[idx[k]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            for k in range(n_cur // 2 - int(h_bin.sum())):
                h_bin[idx[k]] = 1
    hb, cb = binary_sa(h_bin, n_cur, p4_end, seed_val=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb

    # Phase 5: Final polish
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)

    return best_h, best_c, len(best_h)
