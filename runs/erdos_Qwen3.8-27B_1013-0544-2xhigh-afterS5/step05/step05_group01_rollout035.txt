    m = c.max()

        return float(np.sum(x) - n / 2.0)

        return np.ones(n)

    try:

            block = h[start:end].copy()
            block_n = len(block)

                h_full = h.copy()
                h_full[start:end] = x
                return smooth_obj(h_full, n, alpha)
            
            bounds = [(0.0, 1.0)] * block_n

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n, max_iter=500):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h * (1.0 + diff / tot), 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    es = np.exp(sh - mx)
    w = es / es.sum()
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(-(n-1), n):
            wi = w[k + n - 1]
            m1 = i - k
            if 0 <= m1 < n:
                s += wi * (1.0 - x[m1])
            m2 = i + k
            if 0 <= m2 < n:
                s -= wi * x[m2]
        grad[i] = dx * s
    return grad

def obj_with_grad(x, n, alpha):
    return smooth_obj(x, n, alpha), smooth_grad(x, n, alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_with_grad, h, args=(n, alpha), method='SLSQP',
                       jac=True, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_descent(h, n, deadline, alpha=100000):
    """Gradient descent with adaptive step size."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    for _ in range(5000):
        if time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        g_proj = g - np.mean(g)
        h_new = h - step * g_proj
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            if step < 0.1:
                step *= 1.1
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return h, best_c

def focused_search(h, n, deadline, top_m=40):
    """Local search focused on coordinates affecting top lags."""
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                break
            dx = 2.0 / n
            corr = np.correlate(h, 1 - h, mode='full') * dx
            top_idx = np.argsort(corr)[-top_m:]
            active = set()
            for k in top_idx:
                ki = int(k)
                if ki < n:
                    active.update(range(max(0, n - 1 - ki), n))
                else:
                    sh = ki - (n - 1)
                    active.update(range(0, min(n, n - sh)))
            alist = sorted(active)
            improved = False
            for i in range(len(alist)):
                if time.time() > deadline:
                    return h, best_c
                ii = alist[i]
                for j in range(i+1, len(alist)):
                    jj = alist[j]
                    for si, sj in [(1,-1),(-1,1)]:
                        hi = h[ii] + si * d
                        hj = h[jj] + sj * d
                        if 0.0 <= hi <= 1.0 and 0.
