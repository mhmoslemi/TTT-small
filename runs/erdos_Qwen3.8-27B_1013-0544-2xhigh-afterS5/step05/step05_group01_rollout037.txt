        idx_s = i + n - 1 - hi4
        idx_e = i + n - 1 - lo4
        a[lo4:hi4 + 1] -= h[idx_s:idx_e + 1][::-1]
    return a

def golden_section_min(c, a, d_min, d_max, max_iter=25):
    if d_max - d_min < 1e-14:
        return 0.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo, a_hi = d_min, d_max
    b = a_hi - gr * (a_hi - a_lo)
    cc = a_lo + gr * (a_hi - a_lo)
    f_b = float((c + b * a).max())
    f_cc = float((c + cc * a).max())
    for _ in range(max_iter):
        if f_b < f_cc:
            a_hi = cc
            cc = b
            f_cc = f_b
            b = a_hi - gr * (a_hi - a_lo)
            f_b = float((c + b * a).max())
        else:
            a_lo = b
            b = cc
            f_b = f_cc
            cc = a_lo + gr * (a_hi - a_lo)
            f_cc = float((c + cc * a).max())
    d_opt = (a_lo + a_hi) / 2.0
    return d_opt

def fast_local_search(h, n, deadline, max_sweeps=500):
    h = h.copy()
    best_c = compute_c5(h, n)
    sweeps_done = 0
    stagnant = 0
    while time.time() < deadline and sweeps_done < max_sweeps:
        c = correlate_full(h, n)
        c_max = c.max()
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                hi = h[i]
                hj = h[j]
                d_min = max(-hi, hj - 1.0)
                d_max = min(1.0 - hi, hj)
                if d_max - d_min < 1e-12:
                    continue
                a = compute_delta_a(i, j, h, n)
                if np.all(np.abs(a) < 1e-15):
                    continue
                d_opt = golden_section_min(c, a, d_min, d_max)
                if abs(d_opt) < 1e-13:
                    continue
                f_new = float((c + d_opt * a).max())
                if f_new < c_max - 1e-12:
                    h[i] += d_opt
                    h[j] -= d_opt
                    c = c + d_opt * a
                    c_max = c.max()
                    improved = True
        if not improved:
            stagnant += 1
            if stagnant > 2:
                break
        else:
            stagnant = 0
        sweeps_done += 1
    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    return h, final_c

def random_local_search(h, n, deadline, rng, n_random=50000):
    h = h.copy()
    best_c = compute_c5(h, n)
    count = 0
    while time.time() < deadline and count < n_random:
        c = correlate_full(h, n)
        c_max = c.max()
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi = h[i]
        hj = h[j]
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_max - d_min < 1e-12:
            continue
        a = compute_delta_a(i, j, h, n)
        if np.all(np.abs(a) < 1e-15):
            count += 1
            continue
        d_opt = golden_section_min(c, a, d_min, d_max)
        if abs(d_opt) < 1e-13:
            count += 1
            continue
        f_new = float((c + d_opt * a).max())
        if f_new < c_max - 1e-12:
            h[i] += d_opt
            h[j] -= d_opt
        count += 1
    h = project_feasible(h, n)
    final_c = compute_c5(h, n)
    return h, final_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=2000, ftol=1e-15)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    if time.time() < deadline - 30:
        best_h, best_c = fast_local_search(best_h, n, deadline - 30, max_sweeps=300)
    if time.time() < deadline - 10:
        ht, ct = random_local_search(best_h, n, deadline - 5, rng, n_random=100000)
        if ct < best_c:
            best_c = ct
            best_h = ht
    if time.time() < deadline - 3:
        best_h, best_c = fast_local_search(best_h, n, deadline - 1, max_sweeps=100)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    if h_init is not None:
        alloc1 = min(400, (deadline - t0) * 0.55)
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [125, 200, 400]:
            if n_try == n_init or time.time() > deadline - 60:
                continue
            remaining = deadline - time.time()
            alloc = min(120, remaining * 0.3)
            if alloc < 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(3, int(remaining / 80))
        scales = [0.003, 0.008, 0.02, 0.04, 0.06, 0.1]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = fast_local_search(best_h, n, deadline - 2, max_sweeps=200)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
