def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = np.minimum(1.0, h + diff / s * room)
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = np.maximum(0.0, h + diff / s * h)
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0

_alpha_global = 1000.0

def _obj_func(x):

    m = g.max()
    sh = _alpha_global * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / _alpha_global

    return x.sum() - len(x) / 2.0
