def softmax_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_shifted = alpha * (c - c.max())
    exp_c = np.exp(c_shifted)
    weights = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        a = abs(s)
        if a >= n:  # no overlap
            continue
        w = weights[k] * dx

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mx = c.max()
    c_shift = alpha * (c - mx)
    exp_c = np.exp(c_shift)
    return mx + np.log(exp_c.sum()) / alpha

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mx = c.max()
    c_shift = alpha * (c - mx)
    exp_c = np.exp(c_shift)
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        a = abs(s)
        if a >= n:
            continue
        wk = w[k] * dx
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:n])
            grad[s:] -= wk * h[:n-s]
        else:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] -= wk * h[t:]
    return grad

        return softmax_grad(x, n, alpha)

        return x.sum() - n / 2.0

        return np.ones(n)
