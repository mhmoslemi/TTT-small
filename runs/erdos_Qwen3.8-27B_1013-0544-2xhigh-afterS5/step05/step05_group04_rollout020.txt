    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

            grad[shifted] += (-h[shifted + kk]) * dx
    
    # Average over active set (Dubins-Walsh step)
    grad /= len(active)
    return grad, max_val

    Since sum constraint couples everything, we do pair moves."""

    num_lags = 2 * n - 1
    
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        # Shuffle for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.time() > deadline:
                return h, best_c
            # Compute gradient of corr w.r.t. h[i]
            gi = np.zeros(num_lags)
            for k in range(-(n-1), n):
                ki = k + n - 1
                if 0 <= i + k < n:
                    gi[ki] += (1 - h[i + k]) * dx
                if 0 <= i - k < n:
                    gi[ki] += (-h[i - k]) * dx
            
            # Find the j that gives best improvement
            # The change in corr when we do h[i] += d, h[j] -= d is: d * (gi - gj)
            # We want to minimize max_k(corr_k + d*(gi_k - gj_k))
            # Best j is one where gj has large components in the direction of gi's "excess"
            
            # Simple heuristic: try j where h[j] is most different from h[i]
            # and where the gradient difference is favorable
            candidates = np.where(np.abs(h - h[i]) > 0.01)[0]
            candidates = candidates[candidates != i]
            if len(candidates) == 0:
                continue
            # Take top 5 by |h[j] - h[i]|
            top = candidates[np.argsort(np.abs(h[candidates] - h[i]))][-5:]
            
            for j in top:
                # Compute gj
                gj = np.zeros(num_lags)
                for k in range(-(n-1), n):
                    ki = k + n - 1
                    if 0 <= j + k < n:
                        gj[ki] += (1 - h[j + k]) * dx
                    if 0 <= j - k < n:
                        gj[ki] += (-h[j - k]) * dx
                
                g = gi - gj
                corr = compute_corr(h, n)
                
                # f(d) = max_k(corr_k + d * g_k), d in [-h[i], min(h[j], 1-h[i])]
                d_lo = max(-h[i], 0)  # d >= 0 for this direction
                d_hi = min(h[j], 1 - h[i])
                if d_hi - d_lo < 1e-12:
                    continue
                
                # Golden section
                alpha = (np.sqrt(5) - 1) / 2
                c1 = d_hi - alpha * (d_hi - d_lo)
                c2 = d_lo + alpha * (d_hi - d_lo)
                fc1 = np.max(corr + c1 * g)
                fc2 = np.max(corr + c2 * g)
                for _ in range(40):
                    if abs(d_hi - d_lo) < 1e-12:
                        break
                    if fc1 < fc2:
                        d_hi = c2; c2 = c1; fc2 = fc1
                        c1 = d_hi - alpha * (d_hi - d_lo)
                        fc1 = np.max(corr + c1 * g)
                    else:
                        d_lo = c1; c1 = c2; fc1 = fc2
                        c2 = d_lo + alpha * (d_hi - d_lo)
                        fc2 = np.max(corr + c2 * g)
                
                opt_d = (d_lo + d_hi) / 2
                new_c = np.max(corr + opt_d * g)
                if new_c < best_c - 1e-12:
                    h[i] += opt_d
                    h[j] -= opt_d
                    best_c = new_c
                    improved = True
                    gi = gi  # gi doesn't change since only h[i], h[j] changed
                    # Actually gi changes! Need to recompute. But for simplicity, 
                    # let's just continue with the sweep.
        
        if not improved:
            break
    
    return h, best_c

2. Precompute the gradient matrix G[i,k] = dc_k/dh[i]
3. Use subgradient descent: at each step, compute active set, take step, project
4. Alternating: use "top-lag equalization" - identify the top M lags and solve a small subproblem to equalize them
5. Fine pair search at the end

The key insight for step 4: if the top M lags all achieve value ≈ C, we want to find a direction d such that all top M lags decrease. This means: G_active @ d < 0 for all active lags. This is a linear feasibility problem!

If such a direction exists (and it's projected to be feasible), we take a step in that direction.

This is essentially what the subgradient method does, but we can be more aggressive by solving the LP:
minimize t
subject to G_active @ d + c_active <= t
||d||_2 <= 1
sum(d) = 0 (to maintain constraint)
d >= -h, -d <= 1-h (box constraints, approximately)

This is a small LP (M constraints + 1 + n variables) that we can solve quickly.

    return best_h, best_c, len(best_h)
