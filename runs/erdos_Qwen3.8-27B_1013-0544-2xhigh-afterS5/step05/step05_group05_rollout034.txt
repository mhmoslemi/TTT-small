def thue_morse_init(n):
    """Generate Thue-Morse sequence of length n, made binary with n/2 ones."""
    # Thue-Morse: t(k) = popcount(k) % 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Ensure exactly n//2 ones
    target = n // 2
    ones = np.where(tm == 1)[0]
    zeros = np.where(tm == 0)[0]
    if len(ones) > target:
        # Remove excess ones
        excess = len(ones) - target
        tm[ones[-excess:]] = 0
    elif len(ones) < target:
        # Add needed ones
        needed = target - len(ones)
        tm[zeros[:needed]] = 1
    return tm

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

            idx = np.where(tm == 0)[0]

            rng.shuffle(idx)
            tm[idx[:target - int(tm.sum())]] = 1

    p = len(pattern)
    h = np.tile(pattern, n // p + 1)[:n].astype(float)

            break

        zeros = list(np.where(h == 0)[0])

        m = g.max()
