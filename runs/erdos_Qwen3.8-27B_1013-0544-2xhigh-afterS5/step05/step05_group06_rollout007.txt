def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smoothed_max_val(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smoothed_weights(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    return w / w.sum()

def smooth_obj(h, n, alpha):
    return smoothed_max_val(compute_g(h, n), alpha)

def smooth_grad(h, n, alpha):
    g = compute_g(h, n)
    w = smoothed_weights(g, alpha)
    grad = np.zeros(n)
    dm = np.arange(2 * n - 1) - (n - 1)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        d = int(dm[k])
        if d >= 0:
            if n - d > 0:
                grad[:n-d] += wk * (1.0 - h[d:])
                grad[d:] -= wk * h[:n-d]
        else:
            dd = -d
            if n - dd > 0:
                grad[dd:] += wk * (1.0 - h[:n-dd])
                grad[:n-dd] -= wk * h[dd:]
    return (2.0 / n) * grad

def slsqp_analytic(h, n, alpha, deadline, maxiter=150):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, jac=lambda x: smooth_grad(x, n, alpha),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, max_sweeps=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    mi = np.arange(2 * n - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = np.zeros((n, 2 * n - 1))
        for mk in range(2 * n - 1):
            d = int(dm[mk])
            ilo = max(0, d)
            ihi = min(n, n + d)
            if ilo < ihi:
                F[ilo:ihi, mk] += (1.0 - h[ilo-d:ihi-d])
            jlo = max(0, -d)
            jhi = min(n, n - d)
            if jlo < jhi:
                F[jlo:jhi, mk] -= h[jlo+d:jhi+d]
        g = compute_g(h, n)
        cmake = g.max()
        K = min(6, 2 * n - 1)
        top = np.argsort(g)[::-1][:K]
        imps = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                dlo = max(-h[i], h[j] - 1.0)
                dhi = min(1.0 - h[i], h[j])
                if dhi - dlo < 1e-14:
                    continue
                cands = [dlo, dhi]
                for a in range(K):
                    for b in range(K):
                        if top[a] == top[b]:
                            continue
                        ds = s[top[b]] - s[top[a]]
                        if abs(ds) > 1e-15:
                            dc = (g[top[a]] - g[top[b]]) / ds
                            if dlo < dc < dhi:
                                cands.append(dc)
                bd = 0.0
                bv = cmake
                for dd in cands:
                    dd = float(np.clip(dd, dlo, dhi))
                    vv = float(np.max(g + dd * s))
                    if vv < bv - 1e-13:
                        bv = vv
                        bd = dd
                if bd != 0.0:
                    h[i] += bd
                    h[j] -= bd
                    g += bd * s
                    v1 = (i + dm >= 0) & (i + dm < n)
                    F[i + dm[v1], mi[v1]] -= bd
                    v2 = (i - dm >= 0) & (i - dm < n)
                    F[i - dm[v2], mi[v2]] -= bd
                    v3 = (j + dm >= 0) & (j + dm < n)
                    F[j + dm[v3], mi[v3]] += bd
                    v4 = (j - dm >= 0) & (j - dm < n)
                    F[j - dm[v4], mi[v4]] += bd
                    cmake = g.max()
                    top = np.argsort(g)[::-1][:K]
                    imps += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if imps == 0:
            stall += 1
            if stall > 4:
                break
        else:
            stall = 0
    return best_h, best_c

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    bc = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = set(np.where(hb == 0)[0].tolist())
    for _ in range(40):
        if time.time() > deadline:
            break
        imp = False
        for z in sorted(zeros):
            for o in sorted(ones):
                hb[z] = 1.0
                hb[o] = 0.0
                c = compute_c5(hb, n)
                if c < bc - 1e-14:
                    bc = c
                    ones.discard(o)
                    ones.add(z)
                    zeros.discard(z)
                    zeros.add(o)
                    imp = True
                else:
                    hb[z] = 0.0
                    hb[o] = 1.0
                if imp:
                    break
            if imp:
                break
        if not imp:
            break
    return hb, bc

def make_start(n, mode, rng):
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif mode == 2:
        h = np.zeros(n)
        q = n // 4
        h[:q] = 1.0
        h[2*q:3*q] = 1.0
    elif mode == 3:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x * 1.5)
    elif mode == 4:
        h = rng.random(n)
    elif mode == 5:
        h = np.zeros(n)
        for k in range(n // 4):
            h[2*k] = 1.0
    else:
        h = np.full(n, 0.5)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    # Resolution list
    res_list = [n0]
    for nt in [128, 160, 200, 250, 320]:
        if nt % 2 == 0 and nt not in res_list:
            res_list.append(nt)
    res_list = sorted(set(res_list))

    n_res = len(res_list)
    slot_time = (dl - t0) / n_res

    for ri, n in enumerate(res_list):
        if time.time() > dl - 8:
            break
        slot_end = t0 + slot_time * (ri + 1)
        slot_end = min(slot_end, dl - 5)

        # Initial point for this resolution
        if n == n0:
            h = h0.copy()
        else:
            x0 = np.linspace(0, 2, n0, endpoint=False)
            x1 = np.linspace(0, 2, n, endpoint=False)
            h = project_feasible(np.interp(x1, x0, h0), n)

        # Also try a couple of structured starts
        starts = [h]
        if n <= 320:
            starts.append(make_start(n, 2, rng))
            starts.append(make_start(n, 4, rng))

        for h_init in starts:
            if time.time() > slot_end - 10:
                break
            h = h_init.copy()
            c = compute_c5(h, n)

            # Phase A: SLSQP with analytical gradient
            pa_end = t0 + (slot_end - t0) * 0.45 + t0
            pa_end = min(pa_end, slot_end - 30)
            for alpha in [100, 500, 2000, 10000, 50000]:
                if time.time() > pa_end:
                    break
                h_new = slsqp_analytic(h, n, alpha, pa_end, maxiter=80)
                c_new = compute_c5(h_new, n)
                if c_new < c:
                    c = c_new
                    h = h_new

            # Phase B: Pair descent
            pb_end = t0 + (slot_end - t0) * 0.80 + t0
            pb_end = min(pb_end, slot_end - 15)
            h2, c2 = pair_descent(h, n, pb_end, max_sweeps=150)
            if c2 < c:
                c = c2
                h = h2

            # Phase C: Binary polish
            if time.time() < slot_end - 3:
                hb, cb = binary_polish(h, n, slot_end - 2)
                if cb < c:
                    c = cb
                    h = hb

            if c < best_c:
                best_c = c
                best_h = h.copy()

    # Final polish
    if time.time() < dl - 5:
        nb = len(best_h)
        h2, c2 = pair_descent(best_h.copy(), nb, dl - 3, max_sweeps=100)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
