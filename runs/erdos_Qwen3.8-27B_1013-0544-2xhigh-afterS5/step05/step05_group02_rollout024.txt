def binary_pattern(n, period, duty=0.5, offset=0):
    """Create a binary pattern with given period and duty cycle."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i + offset) % period
        if phase < int(period * duty):
            h[i] = 1.0
    # Adjust to get exact sum = n/2
    # Currently sum = count of 1s
    target = n // 2
    current = int(h.sum())
    if current < target:
        # Need to flip some 0s to 1s
        zeros = np.where(h == 0)[0]
        for j in range(target - current):
            h[zeros[j]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        for j in range(current - target):
            h[ones[j]] = 0.0
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, n, alpha):
    """Compute value and gradient of (1/alpha)*log(sum exp(alpha*c_k))."""
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    s = exp_c.sum()
    obj = m + np.log(s) / alpha
    p = exp_c / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = n - 1 - k  # shift
        # term 1: (1-h[i+sh]) for 0<=i<n, 0<=i+sh<n
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1+sh:hi1+sh])
        # term 2: -h[i-sh] for 0<=i<n, 0<=i-sh<n
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2-sh:hi2-sh]
    
    grad *= (1.0 / alpha)
    return obj, grad

def fista(h0, n, alpha, n_iter, deadline, step=None):
    """FISTA optimization with softmax penalty."""
    if step is None:
        step = alpha / (2.0 * (n + 1))
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = softmax_obj_grad(z, n, alpha)
        # Gradient step
        y = z - step * grad
        # Project
        x_new = project_feasible(y, n)
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        moment = (t - 1.0) / t_new
        z = x_new + moment * (x_new - x)
        z = np.clip(z, 0.0, 1.0)
        x = x_new
        t = t_new
    
    return x

def directed_polish(h, n, deadline, max_stall=200):
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        sh = n - 1 - k_star
        g = np.zeros(n)
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1+sh:hi1+sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2-sh:hi2-sh]
        
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            idx = np.argsort(g)
            i_src = int(idx[-1])
            i_dst = int(idx[0])
            if i_src == i_dst:
                break
        
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-10:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue

        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:

            best_h = h.copy()
            c = new_c
            stall = 0

            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(steps) - 1)

    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    obj_val = float('inf')
    h_val = h0.copy()
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Compute the "other" factors
        # c_k = sum_i h_i * (1 - h_{i+s_k}) where s_k = n-1-k
        # g_i_k = 1 - h_{i+s_k}

        cons_lp = list(constraints)

            # g_i = 1 - h_val[i+s] for valid i

            lo = max(0, -s)
            hi = min(n, n-s)
            g[lo:hi] = 1.0 - h_val[lo+s:hi+s]
            cons_lp.append(cp.sum(h * g) <= t)
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, cons_lp)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            try:

            except Exception:
                break
        
        if h.value is not None:
            h_new = np.array(h.value).flatten()
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)

                h_val = h_new
            else:
                # Add a small perturbation to avoid cycling
                h_val = project_feasible(h_new + 0.01 * np.random.randn(n), n)
    
    return h_val, obj_val

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    s = exp_c.sum()
    obj = m + np.log(s) / alpha
    p = exp_c / s
    grad = np.zeros(n)
    shifts = np.arange(2*n - 1) - (n - 1)
    for idx in range(2 * n - 1):
        pk = p[idx]
        if pk < 1e-15:
            continue
        sh = shifts[idx]
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1+sh:hi1+sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2-sh:hi2-sh]
    grad /= alpha
    return obj, grad

def fista(h0, n, alpha, n_iter, deadline):
    step = alpha / (2.0 * (n + 1))
    x = h0.copy()
    z = h0.copy()
    t = 1.0
    for i in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = softmax_obj_grad(z, n, alpha)
        y = z - step * grad
        x_new = project_feasible(y, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        moment = (t - 1.0) / t_new
        z = x_new + moment * (x_new - x)
        z = np.clip(z, 0.0, 1.0)
        x = x_new
        t = t_new
    return x

def directed_polish(h, n, deadline):
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        sh = n - 1 - k_star
        g = np.zeros(n)
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1+sh:hi1+sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2-sh:hi2-sh]
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            idx = np.argsort(g)
            i_src, i_dst = int(idx[-1]), int(idx[0])
            if i_src == i_dst:
                break
        d = min(steps[step_idx], h[i_src]*0.99, (1.0-h[i_dst])*0.99)
        if d < 1e-10:
            step_idx = min(step_idx+1, len(steps)-1)
            stall += 1
            if stall > 200: break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx-1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx+1, len(steps)-1)
            if stall > 200: break
    return best_h, best_c

def multi_lag_polish(h, n, deadline, num_lags=5):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0

        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:

            lo1 = max(0, -sh)
            hi1 = min(n, n - sh)
            if lo1 < hi1:
                g[lo1:hi1] += (1.0 - h[lo1+sh:hi1+sh])
            lo2 = max(0, sh)
            hi2 = min(n, n + sh)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2-sh:hi2-sh]
        if g.max() < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = min(steps[step_idx], h[i_src]*0.99, (1.0-h[i_dst])*0.99)
        if d < 1e-10:
            step_idx = min(step_idx+1, len(steps)-1)
            stall += 1
            if stall > 100: break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx-1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx+1, len(steps)-1)
            if stall > 150: break
    return best_h, best_c
