```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted - mx))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_shift_matrix(h, n):
    """Build A[i,m] = h[i+m-(n-1)] if valid else 0. Shape (n, 2n-1)."""
    nc = 2 * n - 1
    A = np.zeros((n, nc))
    for m in range(nc):
        idx = np.arange(n) + m - (n - 1)
        valid = (idx >= 0) & (idx < n)
        A[np.arange(n)[valid], m] = h[idx[valid]]
    return A

def coord_descent(h, n, max_passes=200, grid_size=40, deadline=None,
                  project_every=10):
    """Free coordinate descent on max-correlation with periodic projection."""
    h = h.copy()
    nc = 2 * n - 1
    c_base = np.correlate(h, 1 - h, mode='full').copy()
    A = build_shift_matrix(h, n)
    A_rev = A[:, ::-1]
    # coeff_i[m] = 1 - A[i,m] - A[i,nc-1-m] = 1 - A[i,m] - A_rev[i,m]
    coeffs = 1.0 - A - A_rev  # shape (n, nc)
    best_c = float(np.max(c_base))
    pass_count = 0

    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        pass_count += 1
        any_imp = False
        for i in range(n):
            ci = coeffs[i]
            old = h[i]
            best_v = old
            best_val = float(np.max(c_base))
            # Grid
            for v in np.linspace(0.0, 1.0, grid_size):
                val = float(np.max(c_base + (v - old) * ci))
                if val < best_val - 1e-15:
                    best_val = val
                    best_v = v
            # Refine
            if 0.0 < best_v < 1.0:
                step = 1.0 / grid_size
                lo = max(0.0, best_v - step)
                hi = min(1.0, best_v + step)
                for v in np.linspace(lo, hi, grid_size):
                    val = float(np.max(c_base + (v - old) * ci))
                    if val < best_val - 1e-15:
                        best_val = val
                        best_v = v
            if abs(best_v - old) > 1e-14:
                delta = best_v - old
                c_base += delta * ci
                h[i] = best_v
                # Update row of coeffs
                idx1 = i - (n - 1) + np.arange(nc)
                idx2 = i + (n - 1) - np.arange(nc)
                # Only need to update the row we just changed
                v1 = (idx1 >= 0) & (idx1 < n)
                v2 = (idx2 >= 0) & (idx2 < n)
                coeffs[i][v1] -= delta  # h[i+m-(n-1)] changed by delta
                coeffs[i][v2] -= delta  # h[i-m+(n-1)] changed by delta
                # Also update other rows that reference index i
                # Row j: A[j,m] = h[j+m-(n-1)]. This equals h[i] when j+m-(n-1)=i
                # i.e., m = i-j+(n-1). Also A_rev[j,m] = h[j-m+(n-1)] = h[i] when m=j-i+(n-1)
                m1 = i - np.arange(n) + (n - 1)  # for A: j+m-(n-1)=i => m=i-j+(n-1)
                m2 = np.arange(n) - i + (n - 1)  # for A_rev: j-m+(n-1)=i => m=j-i+(n-1)
                v1m = (m1 >= 0) & (m1 < nc)
                v2m = (m2 >= 0) & (m2 < nc)
                rows1 = np.arange(n)[v1m]
                rows2 = np.arange(n)[v2m]
                coeffs[rows1, m1[v1m]] -= delta
                coeffs[rows2, m2[v2m]] -= delta
                any_imp = True
        if not any_imp:
            break
        if pass_count % project_every == 0:
            h = project_feasible(h, n)
            c_base = np.correlate(h, 1 - h, mode='full').copy()
            A = build_shift_matrix(h, n)
            A_rev = A[:, ::-1]
            coeffs = 1.0 - A - A_rev

    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def local_pair_polish(h, n, deadline, radius=30, deltas=None):
    """Fine pair search for final polish."""
    if deltas is None:
        deltas = [0.002, 0.001, 0.0005, 0.0002]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                jlo = max(0, i - radius)
                jhi = min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    for sign in (1, -1):
                        hi = h[i] + sign * d
                        hj = h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-14:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - sign * d, hj + sign * d
            if not improved:
                break
    return h, best_c

    # Phase 1: SLSQP warm-up
    t1 = time.time()
    if t1 < deadline - 60:
        h1 = slsqp_optimize(best_h.copy(), n_init, alpha=3000, maxiter=2000)
        c1 = compute_c5(h1, n_init)
        if c1 < best_c:
            best_c, best_h = c1, h1

    # Phase 2: Coordinate descent (main workhorse)
    t2 = time.time()
    if t2 < deadline - 60:
        alloc = min(300, deadline - t2 - 60)
        h2, c2 = coord_descent(best_h, n_init, max_passes=400,
                               grid_size=50, deadline=t2 + alloc, project_every=15)
        if c2 < best_c:
            best_c, best_h = c2, h2

    # Phase 3: SLSQP with high alpha
    t3 = time.time()
    if t3 < deadline - 40:
        h3 = slsqp_optimize(best_h.copy(), n_init, alpha=50000, maxiter=4000)
        c3 = compute_c5(h3, n_init)
        if c3 < best_c:
            best_c, best_h = c3, h3

    # Phase 4: More coordinate descent with finer grid
    t4 = time.time()
    if t4 < deadline - 30:
        alloc = min(250, deadline - t4 - 30)
        h4, c4 = coord_descent(best_h, n_init, max_passes=500,
                               grid_size=80, deadline=t4 + alloc, project_every=20)
        if c4 < best_c:
            best_c, best_h = c4, h4

    # Phase 5: Try other resolutions
    n_curr = len(best_h)
    for n_try in [150, 200, 125, 300]:
        if time.time() > deadline - 80:
            break
        if n_try == n_curr:
            continue
        x_old = np.linspace(0, 2, n_curr)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        # Quick SLSQP
        ht = slsqp_optimize(h_interp.copy(), n_try, alpha=5000, maxiter=2000)
        ct = compute_c5(ht, n_try)
        # Coord descent
        if time.time() < deadline - 40:
            ht, ct = coord_descent(ht, n_try, max_passes=200,
                                   grid_size=50, deadline=deadline - 40)
        if ct < best_c:
            best_c = ct
            best_h = ht
            n_curr = n_try

    # Phase 6: Final SLSQP + coord descent on best
    t6 = time.time()
    n_best = len(best_h)
    if t6 < deadline - 20:
        hf = slsqp_optimize(best_h.copy(), n_best, alpha=200000, maxiter=5000)
        cf = compute_c5(hf, n_best)
        if cf < best_c:
            best_c, best_h = cf, hf
    if time.time() < deadline - 10:
        hfin, cfin = coord_descent(best_h, n_best, max_passes=300,
                                   grid_size=100, deadline=deadline - 5,
                                   project_every=25)
        if cfin < best_c:
            best_c, best_h = cfin, hfin

    # Phase 7: Pair polish
    if time.time() < deadline - 2:
        np_, cnp = local_pair_polish(best_h, len(best_h), deadline - 1,
                                     radius=40)
        if cnp < best_c:
            best_c, best_h = cnp, np_

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
