```python
    """
    Compute new max correlation after swapping h[i](0→1) and h[j](1→0).
    c is the current raw correlation array (length 2n-1), indexed as c[k+(n-1)] for lag k.
    Returns (new_max, delta array or None).

    # Δc[k] = -h[i+k]*I(0≤i+k<n) + (1-h[i-k])*I(0≤i-k<n) + h[j+k]*I(0≤j+k<n) - (1-h[j-k])*I(0≤j-k<n)

    t1 = np.zeros(2*n-1)

    t1[mask1] = -h[i + k[mask1]]

    mask2 = (i - k >= 0) & (i - k < n)
    t2[mask2] = (1.0 - h[i - k[mask2]])
    
    # Term 3: h[j+k] where 0 ≤ j+k < n
    t3 = np.zeros(2*n-1)
    mask3 = (j + k >= 0) & (j + k < n)
    t3[mask3] = h[j + k[mask3]]
    
    # Term 4: -(1-h[j-k]) where 0 ≤ j-k < n
    t4 = np.zeros(2*n-1)
    mask4 = (j - k >= 0) & (j - k < n)
    t4[mask4] = -(1.0 - h[j - k[mask4]])
    
    delta = t1 + t2 + t3 + t4
    new_c = c + delta
    new_max = new_c.max()
    return new_max, delta
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def golden_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    target = n // 2
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    offset = rng.random() * n
    positions = np.round(np.arange(target) * n * phi + offset) % n
    positions = positions.astype(int)
    # Handle duplicates
    h = np.zeros(n)
    count = 0
    for p in positions:
        if h[p] == 0 and count < target:
            h[p] = 1.0
            count += 1
    if count < target:
        zeros = np.where(h == 0)[0]
        remaining = target - count
        h[zeros[:remaining]] = 1.0
    return h

def random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def sa_binary(h_init, n, deadline, t_start=10.0, t_end=0.01, seed=42, patience=30000):
    h = h_init.copy()
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    rng = np.random.RandomState(seed)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    n_iter = 0
    last_improve = 0
    
    while time.time() < deadline:
        T = t_start * (t_end / t_start) ** (min(n_iter, 300000) / 300000.0)
        
        ii = rng.randint(len(zeros))
        jj = rng.randint(len(ones))
        i = zeros[ii]
        j = ones[jj]
        
        h[i] = 1.0
        h[j] = 0.0
        new_c = compute_c5(h, n)
        n_iter += 1
        
        if new_c < cur_c or (new_c - cur_c) < T * abs(np.log(rng.random() + 1e-30)):
            cur_c = new_c
            zeros[ii] = j
            ones[jj] = i
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
                last_improve = n_iter
        else:
            h[i] = 0.0
            h[j] = 1.0
        
        if n_iter - last_improve > patience:
            # Restart from best
            h = best_h.copy()
            cur_c = best_c
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            last_improve = n_iter
    
    return best_h, best_c
```

Also, I realize that for binary sequences, the change in C₅ from a single swap is quantized (it's either 0 or ±1 in the raw correlation, which translates to ±dx in C₅). So the temperature should be on the order of dx for the SA to be meaningful.

For n=250, dx = 0.008. So T should be around 0.008 for the annealing to work well. Starting at T=0.05 and cooling to T=0.001 might work.

Actually, let me think about this more carefully. When we do a swap:
- The raw correlation c[k] changes by Δc[k] ∈ {-2,-1,0,1,2} for each k
- C₅ = max(c) * dx
- So ΔC₅ = Δ(max(c)) * dx

The maximum can change by -2, -1, 0, 1, or 2 (in raw units), so ΔC₅ ∈ {-2dx, -dx, 0, dx, 2dx}.

For n=250, dx=0.008, so ΔC₅ ∈ {-0.016, -0.008, 0, 0.008, 0.016}.

For the Metropolis acceptance: P(accept worse) = exp(-ΔC₅/T).
- If ΔC₅ = 0.008 and T = 0.01: P = exp(-0.8) ≈ 0.45
- If ΔC₅ = 0.016 and T = 0.01: P = exp(-1.6) ≈ 0.20

So T around 0.01-0.05 gives reasonable acceptance rates for worsening moves. Let me use t_start=0.05, t_end=0.002.

Now let me also think about the fractional refinement. After finding a good binary solution, I can try to improve it by allowing small fractional perturbations.

The idea: start from the binary solution and do a few steps of subgradient descent or SLSQP to see if we can get below the binary C₅.

For the subgradient approach:
- Compute g[k] = c[k] for all k
- The subgradient with respect to h[i] for the maximum lag k* is:
  ∂c[k*]/∂h[i] = (1-h[i+k*])*I(0≤i+k*<n) - h[i-k*]*I(0≤i-k*<n) - (1-h[i-k*])*I(0≤i-k*<n) + h[i+k*]*I(0≤i+k*<n)
  
  Wait, let me be more careful. c[k] = ∑_m h[m+k]*(1-h[m])
  
  ∂c[k]/∂h[i] = (1-h[i-k])*I(0≤i-k<n) - h[i+k]*I(0≤i+k<n)
  
  (The first term is from h[m+k] where m+k=i, i.e., m=i-k; the second term is from (1-h[m]) where m=i.)

So the subgradient of the max is the sum of subgradients at the maximizing lags (or a weighted combination for the smooth max).

For a single maximizing lag k*:
∇_h c[k*] has components:
  ∂c[k*]/∂h[i] = (1-h[i-k*])*I(0≤i-k*<n) - h[i+k*]*I(0≤i+k*<n)

This is a vector with at most 2n non-zero entries (n from each term).

Let me implement a simple subgradient descent:

```python
def subgrad_step(h, n, deadline, lr=0.01, max_iter=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = corr.max()
        k_star = int(np.argmax(corr)) - (n - 1)  # actual lag
        
        # Compute subgradient at k_star
        grad = np.zeros(n)
        # Term 1: (1-h[i-k*]) for i-k* in [0, n)
        if k_star >= 0:
            # i-k* >= 0 means i >= k_star
            # i-k* < n means i < n + k_star (always true for i < n)
            grad[k_star:] += (1.0 - h[:n-k_star])
        else:
            to = -k_star
            # i-k* >= 0 means i >= k_star (always true since k_star < 0)
            # i-k* < n means i < n + k_star = n - to
            grad[:n-to] += (1.0 - h[to:])
        
        # Term 2: -h[i+k*] for i+k* in [0, n)
        if k_star >= 0:
            # i+k* < n means i < n - k_star
            grad[:n-k_star] -= h[k_star:]
        else:
            to = -k_star
            # i+k* >= 0 means i >= -k_star = to
            grad[to:] -= h[:n-to]
        
        # Step
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_h, best_c
```
