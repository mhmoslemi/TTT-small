dc_k/dh[t] = -dR_k/dh[t]

R_k = sum_j h[j]h[j+k]
dR_k/dh[t] = h[t+k] (from j=t term, if 0<=t+k<n) + h[t-k] (from j+k=t, i.e., j=t-k, if 0<=t-k<n)

So dc_k/dh[t] = -(h[t+k]*I(0<=t+k<n) + h[t-k]*I(0<=t-k<n))

Total gradient: g[t] = dx * sum_k w_k * dc_k/dh[t]
= -dx * sum_k w_k * (h[t+k]*I(0<=t+k<n) + h[t-k]*I(0<=t-k<n))

For k >= 0:
- h[t+k] is valid when t+k <= n-1, i.e., t <= n-1-k. So t ranges from 0 to n-1-k.
  Contribution to g[t] from this term: w_k * h[t+k] for t=0..n-1-k
- h[t-k] is valid when t-k >= 0, i.e., t >= k. So t ranges from k to n-1.
  Contribution to g[t] from this term: w_k * h[t-k] for t=k..n-1

For k < 0, let k' = -k > 0:
- h[t+k] = h[t-k'] is valid when t-k' >= 0, i.e., t >= k'. So t ranges from k' to n-1.
  Contribution: w_{k'} * h[t-k'] for t=k'..n-1
- h[t-k] = h[t+k'] is valid when t+k' <= n-1, i.e., t <= n-1-k'. So t ranges from 0 to n-1-k'.
  Contribution: w_{k'} * h[t+k'] for t=0..n-1-k'

So for k > 0 (k = 1, 2, ..., n-1):
g[0:n-k] += w_k * h[k:n]     (from h[t+k] term)
g[k:n] += w_k * h[0:n-k]     (from h[t-k] term)

For k = 0:
g[0:n] += w_0 * h[0:n]       (both terms give h[t])
g[0:n] += w_0 * h[0:n]       (same, so actually 2*w_0*h[t])

Wait, for k=0: dR_0/dh[t] = h[t] + h[t] = 2h[t]. So dc_0/dh[t] = -2h[t].

For k < 0 (let k = -k', k' > 0):
g[k':n] += w_{k'} * h[0:n-k']    (from h[t+k] = h[t-k'] term)
g[0:n-k'] += w_{k'} * h[k':n]    (from h[t-k] = h[t+k'] term)

So the final gradient is:
g = -dx * [for each k: contributions]

def softmax_grad(h, n, alpha, top_k=200):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    n_c = len(c_raw)  # 2n-1
    m = c_raw.max()  # we can work with raw values, alpha scaling is the same
    shifted = alpha * (c_raw - m)
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    if top_k < n_c:
        top_indices = np.argsort(c_raw)[-top_k:]
        mask = np.zeros(n_c)
        mask[top_indices] = 1.0
        weights = weights * mask
        wsum = weights.sum()
        if wsum > 0:
            weights = weights / wsum
    
    g = np.zeros(n)
    for k_idx in range(n_c):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            # h[t+k] valid for t=0..n-1-k
            g[:n-k] += w * h[k:]
            # h[t-k] valid for t=k..n-1
            g[k:] += w * h[:n-k]
        else:
            kp = -k
            # h[t+k] = h[t-kp] valid for t=kp..n-1
            g[kp:] += w * h[:n-kp]
            # h[t-k] = h[t+kp] valid for t=0..n-1-kp
            g[:n-kp] += w * h[kp:]
    g *= -dx
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    s = alpha * (c - m)
    ms = s.max()
    return (m + ms + np.log(np.exp(s - ms).sum())) / alpha * dx

def softmax_grad(h, n, alpha, top_k=100):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    n_c = len(c_raw)
    m = c_raw.max()
    s = alpha * (c_raw - m)
    ms = s.max()
    es = np.exp(s - ms)
    weights = es / es.sum()
    if top_k < n_c:
        top_idx = np.argsort(c_raw)[-top_k:]
        mask = np.zeros(n_c)
        mask[top_idx] = 1.0
        weights *= mask
        wsum = weights.sum()
        if wsum > 1e-15:
            weights /= wsum
    g = np.zeros(n)
    for k_idx in range(n_c):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            g[:n - k] += w * h[k:]
            g[k:] += w * h[:n - k]
        else:
            kp = -k
            g[kp:] += w * h[:n - kp]
            g[:n - kp] += w * h[kp:]
    g *= -dx
    return g

def lbfgs_run(h, n, alpha, maxiter, deadline):
    def obj(x):
        return softmax_obj(x, n, alpha)
    def grad(x):
        return softmax_grad(x, n, alpha)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa(h, n, deadline, rng, max_iter=200000):
    target = int(n / 2)
    h_bin = (h >= 0.5).astype(float)
    s = int(h_bin.sum())
    if s > target:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        for i in range(s - target):
            h_bin[idx[i]] = 0
    elif s < target:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        for i in range(target - s):
            h_bin[idx[i]] = 1
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    cur_c = best_c
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    temp = 0.015
    zo_arr = np.array(list(zeros))
    on_arr = np.array(list(ones))
    for it in range(max_iter):
        if time.time() > deadline:
            break
        i = int(zo_arr[rng.randint(len(zo_arr))])
        j = int(on_arr[rng.randint(len(on_arr))])
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        if c_new <= cur_c:
            cur_c = c_new
            ones.add(i); ones.discard(j); zeros.discard(i); zeros.add(j)
            zo_arr[i] = -1
            on_arr[j] = i
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
        else:
            delta = c_new - cur_c
            if temp > 1e-10 and rng.random() < np.exp(-delta / temp):
                ones.add(i); ones.discard(j); zeros.discard(i); zeros.add(j)
                zo_arr[i] = -1
                on_arr[j] = i
                cur_c = c_new
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
        temp *= 0.9998
        if temp < 1e-9:
            temp = 1e-9
    return best_h, best_c

def round_and_local(h, n, deadline, rng):
    # Round to binary
    h_bin = (h >= 0.5).astype(float)
    target = int(n / 2)
    s = int(h_bin.sum())
    if s > target:
        idx = np.where(h_bin == 1)[0]
        order = idx[np.argsort(h[idx])[::-1]]
        for i in range(s - target):
            h_bin[order[i]] = 0
    elif s < target:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(h[idx])]
        for i in range(target - s):
            h_bin[order[i]] = 1
    # Local binary improvement
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for i in list(zeros):
            if time.time() > deadline:
                break
            for j in list(ones):
                h_bin[i], h_bin[j] = 1.0, 0.0
                c_new = compute_c5(h_bin, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h_bin.copy()
                    ones.add(i); ones.discard(j); zeros.discard(i); zeros.add(j)
                    improved = True
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0.copy(), n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS smooth optimization with increasing alpha (35%)
    p1 = t0 + 0.35 * budget_s
    alphas = [500, 2000, 10000, 50000, 200000]
    for alpha in alphas:
        if time.time() > p1 - 15:
            break
        h_new = lbfgs_run(best_h.copy(), n, alpha, 400, p1)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 2: Round + local binary search (20%)
    p2 = t0 + 0.55 * budget_s
    h_r, c_r = round_and_local(best_h.copy(), n, p2, rng)
    if c_r < best_c:
        best_c, best_h = c_r, h_r.copy()

    # Phase 3: Binary SA from current best (20%)
    p3 = t0 + 0.75 * budget_s
    hb, cb = binary_sa(best_h.copy(), n, p3, rng, max_iter=200000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 4: L-BFGS with very high alpha (15%)
    p4 = t0 + 0.90 * budget_s
    for alpha in [500000, 2000000]:
        if time.time() > p4 - 10:
            break
        h_new = lbfgs_run(best_h.copy(), n, alpha, 300, p4)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()

    # Phase 5: Final round + local + short SA (10%)
    h_r2, c_r2 = round_and_local(best_h.copy(), n, deadline - 20, rng)
    if c_r2 < best_c:
        best_c, best_h = c_r2, h_r2.copy()
    if time.time() < deadline - 10:
        hb2, cb2 = binary_sa(best_h.copy(), n, deadline - 2, rng, max_iter=50000)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
