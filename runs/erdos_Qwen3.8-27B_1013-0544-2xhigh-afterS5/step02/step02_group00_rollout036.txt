- At w, find which lags are active (achieve the max)
- If max active coeff > 0: w is too large, move left
- If min active coeff < 0: w is too small, move right
- Otherwise: optimal

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    s = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(x)) / beta

def compute_coeffs(h, j):
    n = len(h)
    lags = np.arange(2*n-1) - (n-1)
    i1 = j + lags
    i2 = j - lags
    m1 = (i1 >= 0) & (i1 < n)
    m2 = (i2 >= 0) & (i2 < n)
    coeff = np.zeros(2*n-1)
    coeff[m1] += (1.0 - h[i1[m1]])
    coeff[m2] -= h[i2[m2]]
    return coeff

        fmax[idx0] = C_full[idx0] + (w - v) * (1.0 - w - v)
        fm = np.max(fmax)
        if fm < best_f:
