def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.astype(float).copy()
    lo = -h.max() - 0.01
    hi = 1.0 - h.min() + 0.01
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent(h, n_moves, rng, n):
    """Random pairwise coordinate descent"""
    h = h.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    
    for mv in range(n_moves):
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        
        # Determine max move in each direction
        d_max_pos = min(h[i], 1.0 - h[j])  # h[i] decreases, h[j] increases
        d_max_neg = min(h[j], 1.0 - h[i])  # h[j] decreases, h[i] increases
        
        best_local = best_c5
        best_d = 0.0
        best_dir = 0
        
        # Try positive direction (mass i -> j)
        if d_max_pos > 1e-8:
            for frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
                d = d_max_pos * frac
                h[i] -= d
                h[j] += d
                c5 = compute_c5(h)
                h[i] += d
                h[j] -= d
                if c5 < best_local:
                    best_local = c5
                    best_d = -d  # negative means h[i] decreased
                    best_dir = 1
        
        # Try negative direction (mass j -> i)
        if d_max_neg > 1e-8:

                h[j] += d
                h[i] -= d

                    best_d = d
                    best_dir = -1

                h[i] += best_d
                h[j] -= best_d
            else:
                h[j] += best_d
                h[i] -= best_d
            best_c5 = best_local

        return smooth_max(corr, beta) * dx

        return np.sum(h) - target_sum
    
    def con_jac(h):
        return np.ones(n_pts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.astype(float).copy()
    lo = -h.max() - 0.01
    hi = 1.0 - h.min() + 0.01
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_opt(h0, n_pts, dx, target_sum, beta, maxiter=200, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n_pts
    def obj_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    def con_func(h):
        return np.sum(h) - target_sum
    def con_jac(h):
        return np.ones(n_pts)
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_func, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(res.x, target_sum)
    except Exception:
        return project_h(h0, target_sum)

def coord_descent(h, n_moves, rng, n, use_anneal=False, temp=0.001):
    h = h.copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    t = temp
    for mv in range(n_moves):
        if use_anneal:
            t = max(0.0001, temp * (1.0 - 0.002 * mv))
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        d_pos = min(h[i], 1.0 - h[j])
        d_neg = min(h[j], 1.0 - h[i])
        best_local = best_c5
        best_d = 0.0
        if d_pos > 1e-8:
            for frac in [0.15, 0.4, 0.7, 1.0]:
                d = d_pos * frac
                h[i] -= d
                h[j] += d
                c5 = compute_c5(h)
                h[i] += d
                h[j] -= d
                if c5 < best_local or (use_anneal and c5 < best_local + t * rng.exponential()):
                    if c5 < best_local:
                        best_local = c5
                        best_d = (-d, i, j)
        if d_neg > 1e-8:
            for frac in [0.15, 0.4, 0.7, 1.0]:
                d = d_neg * frac
                h[j] -= d
                h[i] += d
                c5 = compute_c5(h)
                h[j] += d
                h[i] -= d
                if c5 < best_local or (use_anneal and c5 < best_local + t * rng.exponential()):
                    if c5 < best_local:
                        best_local = c5
                        best_d = (d, j, i)
        if best_d != 0.0:
            d_val, src, dst = best_d
            h[src] += d_val
            h[dst] -= d_val
            if best_local < best_c5:
                best_c5 = best_local
    h = project_h(h, target_sum)
    return h, best_c5

def active_lag_opt(h0, n_pts, dx, target_sum, top_k, penalty_beta, maxiter=150):
    bounds = [(0.0, 1.0)] * n_pts
    def obj_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        top = corr[idx]
        m = np.max(top)
        val = m + np.log(np.sum(np.exp(penalty_beta * (top - m)))) / penalty_beta
        return val * dx
    def con_func(h):
        return np.sum(h) - target_sum
    def con_jac(h):
        return np.ones(n_pts)
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_func, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_h(res.x, target_sum)
    except Exception:
        return project_h(h0, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5)
    n_parent = len(h0)

    n_list = [n_parent]
    for delta in [2, -2, 4, -4, 6, -6]:
        nn = n_parent + delta
        if 48 <= nn <= 110 and nn % 2 == 0 and nn not in n_list:
            n_list.append(nn)

    total_avail = budget_s - 80
    time_parent = total_avail * 0.55
    time_other = total_avail * 0.45 / max(1, len(n_list) - 1)

    for n_pts in n_list:
        t0 = time.time()
        time_limit = time_parent if n_pts == n_parent else time_other
        if time.time() - start_time > budget_s - 40:
            break
        target_sum = n_pts / 2.0
        dx = 2.0 / n_pts

        if n_pts == n_parent:
            h_curr = project_h(h0, target_sum)
        else:
            xo = np.linspace(0, 2, n_parent, endpoint=False)
            xn = np.linspace(0, 2, n_pts, endpoint=False)
            h_curr = np.interp(xn, xo, h0, period=2.0)
            h_curr = project_h(h_curr, target_sum)
        c5_curr = compute_c5(h_curr)

        # Phase 1: SLSQP with increasing smoothness
        for beta in [40, 80, 150, 250, 400]:
            if time.time() - t0 > time_limit * 0.35:
                break
            h_new = slsqp_opt(h_curr, n_pts, dx, target_sum, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new

        # Phase 2: Active lag focused optimization
        if time.time() - t0 < time_limit * 0.5:
            for top_k in [3, 5]:
                if time.time() - t0 > time_limit * 0.5:
                    break
                h_new = active_lag_opt(h_curr, n_pts, dx, target_sum, top_k, 100, maxiter=150)
                c5_new = compute_c5(h_new)
                if c5_new < c5_curr:
                    h_curr = h_new
                    c5_curr = c5_new

        # Phase 3: Coordinate descent (exploit)
        if time.time() - t0 < time_limit * 0.75:
            n_cd = 8000 if n_pts == n_parent else 3000
            h_curr, c5_curr = coord_descent(h_curr, n_cd, rng, n_pts)

        # Phase 4: SLSQP polish with high beta
        if time.time() - t0 < time_limit * 0.85:
            h_new = slsqp_opt(h_curr, n_pts, dx, target_sum, 500, maxiter=300, ftol=1e-14)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new

        # Phase 5: More coordinate descent
        if time.time() - t0 < time_limit * 0.95:
            h_curr, c5_curr = coord_descent(h_curr, 5000, rng, n_pts)

        if c5_curr < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_curr
            best_n = n_pts

    # Try annealed CD on best to escape local minima
    if best_h is not None and time.time() - start_time < budget_s - 30:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        for attempt in range(3):
            if time.time() - start_time > budget_s - 20:
                break
            h_anneal = best_h + rng.randn(n_f) * 0.02
            h_anneal = project_h(h_anneal, ts)
            h_anneal, c5_anneal = coord_descent(h_anneal, 6000, rng, n_anneal if False else n_f, use_anneal=True, temp=0.0005)
            h_anneal = slsqp_opt(h_anneal, n_f, dx_f, ts, 300, maxiter=200)
            c5_anneal = compute_c5(h_anneal)
            if c5_anneal < best_c5:
                best_h = h_anneal.copy()
                best_c5 = c5_anneal

    # Final polish
    if best_h is not None and time.time() - start_time < budget_s - 10:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        h_new = slsqp_opt(best_h, n_f, dx_f, ts, 600, maxiter=500, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        if time.time() - start_time < budget_s - 5:
            best_h, best_c5 = coord_descent(best_h, 10000, rng, best_n)

    return best_h, best_c5, best_n
