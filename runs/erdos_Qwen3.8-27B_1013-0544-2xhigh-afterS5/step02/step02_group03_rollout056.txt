    return h_local, best_local


def generate_step_candidate(n, rng):
    """Generate a random step-like candidate with the right integral."""
    h = np.zeros(n)
    n_ones = n // 2
    perm = rng.permutation(n)
    h[perm[:n_ones]] = 1.0
    return project_feasible(h, n / 2.0)


def generate_smooth_candidate(n, rng):
    """Generate a smooth candidate using sine + random perturbation."""
    x = np.linspace(0, 2, n, endpoint=False)
    phases = rng.uniform(0, 2 * np.pi, 3)
    freqs = rng.choice([2, 3, 4, 5], 3)
    h = 0.5 + 0.3 * np.sin(freqs[0] * np.pi * x + phases[0])
    h += 0.15 * np.sin(freqs[1] * np.pi * x + phases[1])
    h += 0.1 * np.sin(freqs[2] * np.pi * x + phases[2])
    h = np.clip(h + rng.randn(n) * 0.05, 0, 1)
    return project_feasible(h, n / 2.0)


def slsqp_obj(x, dx, beta):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx


def slsqp_eq(x, target_sum):
    return x.sum() - target_sum


def slsqp_eq_jac(target_sum, n):
    return np.ones(n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    # Initialize from parent or default
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 0: Explore diverse candidates at smaller resolutions
    if tl() > 100:
        candidates = []
        for nc in [30, 40, 50]:
            for gen_fn in [generate_step_candidate, generate_smooth_candidate]:
                h_c = gen_fn(nc, rng)
                c5_c = compute_c5(h_c)
                candidates.append((h_c, c5_c, nc))
        # Also try upsampled initial
        for nc in [40, 50]:
            h_c = project_feasible(resample_h(h_init, nc), nc / 2.0)
            c5_c = compute_c5(h_c)
            candidates.append((h_c, c5_c, nc))
        candidates.sort(key=lambda t: t[1])
        # Optimize the best few coarse candidates
        for h_c, c5_c, nc in candidates[:6]:
            if tl() < 50:
                break
            dx_c = 2.0 / nc
            ts_c = nc / 2.0
            bounds_c = Bounds(np.zeros(nc), np.ones(nc))
            for beta in [50, 500]:
                if tl() < 25:
                    break
                try:
                    res = minimize(lbfgs_obj, h_c.copy(),
                                   args=(dx_c, ts_c, beta),
                                   jac=lambda x, dx_=dx_c, ts_=ts_c, b_=beta, n_=nc: lbfgs_grad(x, dx_, ts_, b_, n_),
                                   method='L-BFGS-B', bounds=bounds_c,
                                   options={'maxiter': 2000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_c)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        # Upsample to our resolution
                        h_up = project_feasible(resample_h(h_new, n), target_sum)
                        c5_up = compute_c5(h_up)
                        if c5_up < best_c5:
                            best_h = h_up.copy()
                            best_c5 = c5_up
                        elif c5_new < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5_new
                            n = nc
                            dx = dx_c
                            target_sum = ts_c
                            bounds = bounds_c
                except Exception:
                    pass

    # Phase 1: L-BFGS-B continuation at current resolution
    for beta in [10, 50, 200, 800, 3000, 10000, 30000]:
        if tl() < 30:
            break
        for trial in range(3):
            if tl() < 15:
                break
            x0 = best_h.copy()
            if trial == 1:
                x0 += rng.randn(n) * 0.008
                x0 = project_feasible(x0, target_sum)
            elif trial == 2:
                x0 += rng.randn(n) * 0.003
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(lbfgs_obj, x0,
                               args=(dx, target_sum, beta),
                               jac=lambda x, dx_=dx, ts_=target_sum, b_=beta, n_=n: lbfgs_grad(x, dx_, ts_, b_, n_),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-14, 'maxls': 50})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Momentum subgradient refinement
    if tl() > 60:
        h_sg, c5_sg = subgrad_momentum(best_h, target_sum, t0, budget_s, n_iters=60000)
        if c5_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c5_sg

    # Phase 3: SLSQP polish with tight constraints
    if tl() > 50:
        for beta in [5000, 20000, 80000]:
            if tl() < 25:
                break
            cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac,
                     'args': (target_sum, n)}]
            try:
                res = minimize(slsqp_obj, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Try higher resolution
    if tl() > 80:
        for n_try in [n + 40, n + 80, 100, 140]:
            if n_try < 30 or tl() < 40:
                continue
            h_t = project_feasible(resample_h(best_h, n_try), n_try / 2.0)
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            bounds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [200, 2000, 10000]:
                if tl() < 20:
                    break
                try:
                    res = minimize(lbfgs_obj, h_t.copy(),
                                   args=(dx_t, ts_t, beta),
                                   jac=lambda x, dx_=dx_t, ts_=ts_t, b_=beta, n_=n_try: lbfgs_grad(x, dx_, ts_, b_, n_),
                                   method='L-BFGS-B', bounds=bounds_t,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bounds_t
                except Exception:
                    pass

    # Phase 5: Final SLSQP at best resolution
    if tl() > 30:
        n = len(best_h)
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [50000, 200000]:
            if tl() < 15:
                break
            cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac,
                     'args': (target_sum, n)}]
            try:
                res = minimize(slsqp_obj, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
