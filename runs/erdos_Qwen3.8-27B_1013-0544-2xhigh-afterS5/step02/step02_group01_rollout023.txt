    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Adjust only free variables
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0 and abs(diff) <= free.sum() * 0.5:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Second pass
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free2 = (h > 0) & (h < 1.0)
            if free2.sum() > 0:

    )

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):

    x = h.copy().astype(float)

    best_x = x.copy()

    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.maximum(corr * dx, 1e-16)
    log_sp = np.sum(p * np.log(c)) / p
    sp = np.exp(log_sp)
    val = np.exp(log_sp / p)
    sum_diff = h.sum() - n / 2.0
    total = val + lam * sum_diff * sum_diff
    j = np.arange(n)
    # grad of p-norm: sp^(1/p - 1) * dx * sum_k c_k^(p-1) * dcorr_k/dh_i
    log_sp_precise = np.sum(np.log(c))
    log_coeff_base = (1.0 / p - 1.0) * log_sp_precise + np.log(dx)
    grad = np.zeros(n)
    for k in range(len(corr)):
        log_w = log_coeff_base + (p - 1.0) * np.log(c[k])
        if log_w < -40:
            continue
        w = np.exp(log_w)
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += w * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += w * (-h[jp[mp]])
    grad += 2.0 * lam * sum_diff
    return total, grad


def lbfgsb_run(h, p, lam, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pnorm_obj_grad, h, args=(p, lam), jac=True,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return res.x
    except Exception:
        return h.copy()


def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def perturb_h(h, sigma, rng):
    h_p = h + sigma * rng.randn(len(h))
    return project_feasible(h_p, len(h) / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: L-BFGS-B with p-norm at current resolution, increasing p
    for p in [16, 32, 64, 128, 256]:
        if tl() < 40:
            break
        lam = max(0.5, n * 0.02)
        h_new = lbfgsb_run(h_best, p, lam, maxiter=400)
        h_new = project_feasible(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: Nesterov with growing top_k
    for top_k in [4, 6, 10, 15, 20]:
        if tl() < 25:
            break
        h_n, c_n = nesterov_subgrad(h_best, 4000, 0.06, 0.9996, top_k)
        if c_n < best_c5:
            h_best, best_c5 = h_n, c_n
        if tl() < 20:
            break
        h_n2, c_n2 = nesterov_subgrad(h_best, 2000, 0.02, 0.9998, top_k)
        if c_n2 < best_c5:
            h_best, best_c5 = h_n2, c_n2

    # Phase 3: L-BFGS-B polish after Nesterov
    for p in [64, 128, 256, 512]:
        if tl() < 20:
            break
        lam = max(0.1, n * 0.005)
        h_new = lbfgsb_run(h_best, p, lam, maxiter=300)
        h_new = project_feasible(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 4: Perturbation restarts to escape local optima
    for sigma in [0.02, 0.05, 0.01, 0.03]:
        if tl() < 30:
            break
        h_pert = perturb_h(h_best, sigma, rng)
        h_new = lbfgsb_run(h_pert, 64, n * 0.01, maxiter=200)
        h_new = project_feasible(h_new, n / 2.0)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 5: Higher resolution
    for n_new in [300, 400, 500]:
        if tl() < 80:
            break
        h_interp = interp_to_n(h_best, n_new)
        h_cur = h_interp
        c_cur = compute_c5(h_interp)
        improved = False
        for p in [32, 64, 128, 256]:
            if tl() < 30:
                break
            lam = max(0.5, n_new * 0.01)
            h_t = lbfgsb_run(h_cur, p, lam, maxiter=300)
            h_t = project_feasible(h_t, n_new / 2.0)
            c_t = compute_c5(h_t)
            if c_t < c_cur:
                h_cur, c_cur = h_t, c_t
                improved = True
        if tl() > 40:
            h_n, c_n = nesterov_subgrad(h_cur, 3000, 0.03, 0.9997, top_k=8)
            if c_n < c_cur:
                h_cur, c_cur = h_n, c_n
                improved = True
        if tl() > 25:
            h_t = lbfgsb_run(h_cur, 128, n_new * 0.003, maxiter=300)
            h_t = project_feasible(h_t, n_new / 2.0)
            c_t = compute_c5(h_t)
            if c_t < c_cur:
                h_cur, c_cur = h_t, c_t
                improved = True
        if improved and c_cur < best_c5:
            h_best, best_c5, n = h_cur, c_cur, n_new

    # Phase 6: Final aggressive polish
    for p in [128, 256, 512, 1024]:
        if tl() < 15:
            break
        lam = n * 0.001
        h_f = lbfgsb_run(h_best, p, lam, maxiter=500)
        h_f = project_feasible(h_f, n / 2.0)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    # Final Nesterov micro-tune
    if tl() > 10:
        h_f2, c_f2 = nesterov_subgrad(h_best, 2000, 0.01, 0.9999, top_k=5)
        if c_f2 < best_c5:
            h_best, best_c5 = h_f2, c_f2

    return h_best, best_c5, len(h_best)
