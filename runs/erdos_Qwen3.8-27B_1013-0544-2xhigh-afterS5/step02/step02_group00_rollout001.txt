            for beta in [50, 200, 800]:
                if remaining() < 8:
                    break
                try:
                    res = minimize(
                        lambda x: period_obj(x, n_points, p, beta),
                        init, method='SLSQP', bounds=bounds_p,
                        constraints=[con_p],
                        options={'maxiter': 2000, 'ftol': 1e-15})
                    h_new = make_periodic_h(res.x, n_points, p)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        best_n = n_points
                except Exception:
                    pass

    # Phase 2: Full-space L-BFGS-B refinement
    if remaining() > 50:
        cur = best_h.copy()
        cur_c5 = best_c5
        n_cur = len(cur)
        tgt = n_cur / 2.0
        bounds_full = Bounds(np.zeros(n_cur), np.ones(n_cur))

        for beta in [100, 300, 1000, 3000]:
            if remaining() < 30:
                break
            try:
                res = minimize(
                    lambda x: full_obj(x, n_cur, beta),
                    cur, method='L-BFGS-B', bounds=bounds_full,
                    options={'maxiter': 5000, 'gtol': 1e-14, 'ftol': 1e-16})
                h_new = project_h(res.x, tgt)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    cur = h_new
                    cur_c5 = c5_new
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_cur
            except Exception:
                pass

        # SLSQP polish with very high beta
        if remaining() > 20:
            con_full = {'type': 'eq', 'fun': lambda x: full_con(x, tgt),
                        'jac': lambda x: full_con_jac(x)}
            try:
                res = minimize(
                    lambda x: full_obj(x, n_cur, 2000.0),
                    cur, method='SLSQP', bounds=bounds_full,
                    constraints=[con_full],
                    options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h(res.x, tgt)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5, best_n = h_new.copy(), c5_new, n_cur
            except Exception:
                pass

    # Phase 3: Subgradient descent with momentum on active lags
    if remaining() > 40:
        cur = best_h.copy()
        cur_c5 = best_c5
        n_cur = len(cur)
        tgt = n_cur / 2.0
        momentum = np.zeros(n_cur)
        alpha = 0.1

        for iteration in range(20000):
            if remaining() < 5:
                break
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            # Top 3 active lags
            top_k = np.argsort(corr)[::-1][:3]
            # Subgradient
            grad = np.zeros(n_cur)
            for lag_idx in top_k:
                lag = int(lag_idx) - (n_cur - 1)
                for j in range(n_cur):
                    i1 = j - lag
                    i2 = j + lag
                    if 0 <= i1 < n_cur:
                        grad[j] += (1.0 - cur[i1])
                    if 0 <= i2 < n_cur:
                        grad[j] -= cur[i2]
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            grad /= gnorm
            # Momentum update
            momentum = 0.9 * momentum + 0.1 * grad
            momentum /= (np.linalg.norm(momentum) + 1e-15)
            step = alpha / (1.0 + 0.001 * iteration)
            h_trial = project_h(cur - step * momentum, tgt)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_cur
            else:
                alpha *= 0.95

    # Phase 4: Coordinate descent
    if remaining() > 30:
        cur = best_h.copy()
        cur_c5 = best_c5
        n_cur = len(cur)
        tgt = n_cur / 2.0
        stuck_count = 0

        for iteration in range(30000):
            if remaining() < 5:
                break
            i = iteration % n_cur
            old_val = cur[i]
            # Try a grid of values
            candidates = [0.0, 1.0, 0.5,
                          old_val - 0.1, old_val - 0.02,
                          old_val + 0.02, old_val + 0.1]
            best_val = old_val
            best_c = cur_c5
            for cv in candidates:
                cv = float(np.clip(cv, 0.0, 1.0))
                if abs(cv - old_val) < 1e-10:
                    continue
                h_t = cur.copy()
                h_t[i] = cv
                h_t = project_h(h_t, tgt)
                c = compute_c5(h_t)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cv
            if best_val != old_val:
                cur[i] = best_val
                cur = project_h(cur, tgt)
                cur_c5 = best_c
                stuck_count = 0
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_cur
            else:
                stuck_count += 1
                if stuck_count > n_cur * 3:
                    # Random restart perturbation
                    h_t = cur + np.random.normal(0, 0.03, n_cur)
                    h_t = project_h(h_t, tgt)
                    c_t = compute_c5(h_t)
                    if c_t < cur_c5:
                        cur = h_t
                        cur_c5 = c_t
                        stuck_count = 0
                    else:
                        break

    return best_h, best_c5, best_n
