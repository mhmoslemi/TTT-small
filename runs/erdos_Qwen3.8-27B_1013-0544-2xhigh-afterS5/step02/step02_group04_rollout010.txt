    n_points = best_n
    target_sum = n_points / 2.0
    best_h = project_h(best_h, target_sum)

    attempt = 0
    stagnant = 0
    while time_left() > 40:
        attempt += 1
        h0 = best_h.copy()
        strat = attempt % 10

        if strat == 0:
            scale = 0.002 + 0.012 * np.exp(-attempt / 25.0)
            h0 += rng.randn(n_points) * scale
        elif strat == 1:
            corr = np.correlate(h0, 1 - h0, mode="full")
            active = np.argmax(corr)
            lag = active - (n_points - 1)
            grad = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= j + lag < n_points:
                    grad[j] += abs(1.0 - h0[j + lag])
                if 0 <= j - lag < n_points:
                    grad[j] += abs(h0[j - lag])
            top_idx = np.argsort(grad)[-12:]
            h0[top_idx] += rng.randn(len(top_idx)) * 0.015
        elif strat == 2:
            blen = rng.randint(3, 8)
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] = 1.0 - h0[bs:bs + blen]
        elif strat == 3:
            k = rng.randint(1, 6)
            t = np.linspace(0, 2 * np.pi * k, n_points)
            h0 += 0.015 * np.cos(t + rng.uniform(0, 6.28))
        elif strat == 4:
            kernel = np.ones(5) / 5.0
            h0 = 0.8 * h0 + 0.2 * np.convolve(h0, kernel, mode='same')
            h0 += rng.randn(n_points) * 0.004
        elif strat == 5:
            h0 = 0.6 * h0 + 0.4 * h0[::-1]
            h0 += rng.randn(n_points) * 0.008
        elif strat == 6:
            for _ in range(rng.randint(2, 5)):
                i, j = rng.choice(n_points, 2, replace=False)
                d = 0.025 * rng.randn()
                h0[i] += d
                h0[j] -= d
        elif strat == 7:
            block_size = rng.randint(5, 15)
            bs = rng.randint(0, n_points - block_size)
            old_vals = h0[bs:bs + block_size].copy()
            h0[bs:bs + block_size] = rng.uniform(0.1, 0.9, block_size)
            h0[bs:bs + block_size] = old_vals + h0[bs:bs + block_size] - old_vals.mean()
        elif strat == 8:
            # Anti-correlation push: reduce overlap at active lag
            corr = np.correlate(h0, 1 - h0, mode="full")
            top_lags = np.argsort(corr)[-3:]
            for tl in top_lags:
                lag = tl - (n_points - 1)
                for j in range(min(5, n_points)):
                    if 0 <= j + lag < n_points:
                        d = 0.005 * (1.0 - 2.0 * h0[j])
                        h0[j] += d
                    if 0 <= j - lag < n_points:
                        d = 0.005 * (h0[j] - 0.5)
                        h0[j] += d
        else:
            h0 += rng.randn(n_points) * 0.03

        h0 = project_h(h0, target_sum)
        beta = 800 if attempt % 3 == 0 else 300
        h_new, c5 = slsqp_optimize(h0, n_points, target_sum, beta, 4000, 1e-15)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                # Aggressive restart
                h0 = best_h + rng.randn(n_points) * 0.08
                h0 = project_h(h0, target_sum)
                h_new, c5 = slsqp_optimize(h0, n_points, target_sum, 200, 5000, 1e-14)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
                stagnant = 0

    # Phase 4: Final high-precision polish
    for beta in [3000, 8000, 20000]:
        if time_left() < 10:
            break
        h_new, c5 = slsqp_optimize(best_h, best_n, best_n / 2.0, beta, 10000, 1e-16)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
