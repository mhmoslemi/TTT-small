from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta):
    """Return (objective, gradient) for smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    obj = m + np.log(np.sum(exp_vals)) / beta
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if abs(w) < 1e-13:
            continue
        d = k - (n - 1)

        j1_lo = max(0, -d)
        j1_hi = min(n-1, n-1-d)

        # Second factor: -h[j-d] for j-d valid

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k] * dx
        if abs(w) < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                idx = np.arange(d, n)
                grad[idx] += w * (1.0 - h[idx - d])
                idx = np.arange(0, n-d)
                grad[idx] += w * (-h[idx + d])
        else:
            ad = -d
            idx = np.arange(0, n-ad)
            grad[idx] += w * (1.0 - h[idx + ad])
            idx = np.arange(ad, n)
            grad[idx] += w * (-h[idx - ad])

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

        h_parent = np.array(initial_h_values, dtype=float).copy()

    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    preferred_n = n_parent
    
    # Resolution sweep
    n_list = []

        n = n_parent + delta
        if 40 <= n <= 250 and n not in n_list:
            n_list.append(n)
    
    def opt_cycle(h0, n, target_sum, deadline):
        """Run optimization cycles on h0 with n points until deadline."""
        bounds = Bounds(np.zeros(n), np.ones(n))
        def cf(x): return np.sum(x) - target_sum
        def cj(x): return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        
        bh = h0.copy()
        bc = compute_c5(h0)
        
        for beta in [50, 100, 200, 400, 800]:
            if time.time() > deadline:

            for method in ['SLSQP', 'trust-constr']:
                if time.time() > deadline:
                    break

                def g(h): return smooth_max_obj_grad(h, beta)[1]
                try:
                    if method == 'SLSQP':
                        res = minimize(f, bh, jac=g, method=method, bounds=bounds,
                                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
                    else:
                        res = minimize(f, bh, jac=g, method=method, bounds=bounds,
                                      constraints=cons, options={'maxiter': 200, 'gtol': 1e-10, 'xtol': 1e-14})
                    h_new = project_h(res.x, target_sum)
                    c = compute_c5(h_new)
                    if c < bc:
                        bh, bc = h_new.copy(), c
                except Exception:

        return bh, bc
    
    # Time allocation
    total_budget = budget_s - 100  # leave 100s for final polish

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    sum_exp = np.sum(exp_vals)
    obj = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k] * dx
        if abs(w) < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                ia = np.arange(d, n)
                grad[ia] += w * (1.0 - h[ia - d])
                ib = np.arange(0, n - d)
                grad[ib] += w * (-h[ib + d])
        else:
            ad = -d
            ia = np.arange(0, n - ad)
            grad[ia] += w * (1.0 - h[ia + ad])
            ib = np.arange(ad, n)
            grad[ib] += w * (-h[ib - ad])
    return obj, grad

def topk_obj_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    idx = np.argsort(corr)[::-1][:top_k]
    ct = corr[idx]
    m = np.max(ct)
    ev = np.exp(beta * (ct - m))
    se = np.sum(ev)
    obj = m + np.log(se) / beta
    wt = ev / se
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        w = wt[i] * dx
        if abs(w) < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            if d < n:
                ia = np.arange(d, n)
                grad[ia] += w * (1.0 - h[ia - d])
                ib = np.arange(0, n - d)
                grad[ib] += w * (-h[ib + d])
        else:
            ad = -d
            ia = np.arange(0, n - ad)
            grad[ia] += w * (1.0 - h[ia + ad])
            ib = np.arange(ad, n)
            grad[ib] += w * (-h[ib - ad])
    return obj, grad

def single_corr_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    d = k - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            ia = np.arange(d, n)
            grad[ia] = (1.0 - h[ia - d]) * dx
            ib = np.arange(0, n - d)
            grad[ib] -= h[ib + d] * dx
    else:
        ad = -d
        ia = np.arange(0, n - ad)
        grad[ia] = (1.0 - h[ia + ad]) * dx
        ib = np.arange(ad, n)
        grad[ib] -= h[ib - ad] * dx
    return grad

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    xs = np.linspace(0, 2, n_src, endpoint=False)
    xd = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xd, xs, h_src)

def subgrad_step(h, n, target_sum, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    g = single_corr_grad(h, k_max)
    h_new = h - step_size * g
    h_new = project_h(h_new, target_sum)
    return h_new

def optimize_h(h0, n, target_sum, deadline):
    bounds = Bounds(np.zeros(n), np.ones(n))
    def cf(x):
        return np.sum(x) - target_sum
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    bh = h0.copy()
    bc = compute_c5(h0)

    for beta in [50, 100, 200, 400]:
        if time.time() > deadline:
            break
        def f(h):
            return smooth_max_obj_grad(h, beta)[0]
        def g(h):
            return smooth_max_obj_grad(h, beta)[1]
        try:
            res = minimize(f, bh, jac=g, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 600, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c = compute_c5(h_new)
            if c < bc:
                bh, bc = h_new.copy(), c
        except Exception:
            pass

    if time.time() > deadline - 5:
        return bh, bc

    for beta in [200, 500]:
        if time.time() > deadline:
            break
        def f2(h):
            return topk_obj_grad(h, beta, 8)[0]

            return topk_obj_grad(h, beta, 8)[1]
        try:
            res = minimize(f2, bh, jac=g2, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c = compute_c5(h_new)
            if c < bc:
                bh, bc = h_new.copy(), c
        except Exception:
            pass

    if time.time() > deadline - 5:
        return bh, bc

    for it in range(20):
        if time.time() > deadline:
            break
        h_sub = subgrad_step(bh, n, target_sum, 0.01 / (it + 1))
        c_sub = compute_c5(h_sub)
        if c_sub < bc:
            bh, bc = h_sub.copy(), c_sub
        else:
            break

    if time.time() > deadline - 5:
        return bh, bc

    for beta in [600, 1000]:
        if time.time() > deadline:
            break
        def f3(h):
            return smooth_max_obj_grad(h, beta)[0]
        def g3(h):
            return smooth_max_obj_grad(h, beta)[1]
        try:
            res = minimize(f3, bh, jac=g3, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 800, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c = compute_c5(h_new)
            if c < bc:

        except Exception:
            pass

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)

    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    best_n = n_parent

    n_list = []
    for delta in [0, 10, 20, -10, 30, -20, 40, 50]:
        nv = n_parent + delta
        if 40 <= nv <= 300 and nv not in n_list:
            n_list.append(nv)

    reserve = 150
    total_avail = budget_s - reserve - 10
    per_n = max(50, total_avail / len(n_list))

    for ni, n in enumerate(n_list):
        elapsed = time.time() - start_time
        if elapsed > budget_s - reserve - 20:
            break
        deadline = min(time.time() + per_n * 1.2, start_time + budget_s - reserve - 10)
        target_sum = n / 2.0

        starts = []
        if n == n_parent:
            starts.append(h_parent.copy())
        else:
            starts.append(project_h(interpolate_h(h_parent, n), target_sum))
        if len(best_h) == n:
            starts.append(best_h.copy())
        else:
            starts.append(project_h(interpolate_h(best_h, n), target_sum))

        base_for_pert = starts[0] if starts else np.full(n, 0.5)
        perturbed = base_for_pert + np.random.randn(n) * 0.08
        starts.append(project_h(perturbed, target_sum))

        for h0 in starts[:3]:
            if time.time() > deadline:
                break
            h0 = project_h(h0, target_sum)
            h_opt, c5_opt = optimize_h(h0, n, target_sum, deadline)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt.copy(), c5_opt, n

    if time.time() < start_time + budget_s - 30:
        target_sum = best_n / 2.0
        deadline = start_time + budget_s - 10
        h_f, c5_f = optimize_h(best_h, best_n, target_sum, deadline)
        if c5_f < best_c5:
            best_h, best_c5 = h_f.copy(), c5_f

    return best_h, best_c5, len(best_h)
