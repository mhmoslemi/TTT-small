- g_{k+1} = subgradient at y_{k+1}
- t_{k+1} = 2/(k+2)
- Output: x̄ = (1/(T+1)) ∑_{k=0}^{T} x_k (the averaged iterate)

This gives O(1/k) convergence for convex problems, which is better than the O(1/√k) of plain subgradient.

import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, ts):
    return h.sum() - ts


def con_jac(h, ts):
    return np.ones(len(h))


def subgrad_for_lag(h, k_worst, n):
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def coord_descent_pass(h, n, target_sum):
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    for j in range(n):
        corr_cur = get_corr(h)
        B = np.zeros(2 * n - 1)
        v1 = (lags >= -j) & (lags <= n - 1 - j)
        v2 = (lags >= j - (n - 1)) & (lags <= j)
        if v1.any():
            B[v1] += (1.0 - h[j + lags[v1]])
        if v2.any():
            B[v2] -= h[j - lags[v2]]
        A = corr_cur - B * h[j]

        def f1d(t):
            return np.max(A + B * t)

        res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
                              options={'xatol': 1e-9})
        h[j] = res.x
        h = project_feasible(h, target_sum)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def sla_step(h, n, target_sum, rng):
    k = rng.randint(1, 4)
    idxs = rng.choice(n, size=k, replace=False)
    for j in idxs:
        d = rng.uniform(-0.15, 0.15)
        h[j] = np.clip(h[j] + d, 0.0, 1.0)
    h = project_feasible(h, target_sum)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation
    for beta in [10, 50, 200, 800, 3000, 10000]:
        if tl() < 50:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for trial in range(3):
            if tl() < 25:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += rng.randn(n) * (0.03 / trial)
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 2: Coordinate descent
    if tl() > 150:
        h = best_h.copy()
        for it in range(300):
            if tl() < 30:
                break
            h = coord_descent_pass(h, n, target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5

    # Phase 3: Simulated annealing
    if tl() > 200:
        h = best_h.copy()
        T = 0.05
        best_sa = best_c5
        for it in range(20000):
            if tl() < 50:
                break
            T *= 0.9995
            h_new = sla_step(h, n, target_sum, rng)
            c5_new = compute_c5(h_new)
            if c5_new < best_sa or rng.rand() < np.exp(-(c5_new - best_sa) / max(T, 1e-8)):
                h = h_new
                if c5_new < best_c5:
                    best_h, best_c5 = h.copy(), c5_new
                    best_sa = c5_new
            if it % 500 == 0:
                h_sa = sla_step(h, n, target_sum, rng)
                c5_sa = compute_c5(h_sa)
                if c5_sa < best_c5:
                    best_h, best_c5 = h_sa.copy(), c5_sa

    # Phase 4: Subgradient with averaging
    if tl() > 150:
        h = best_h.copy()
        avg_h = h.copy()
        for it in range(15000):
            if tl() < 40:
                break
            corr = get_corr(h)
            k_worst = np.argmax(corr)
            grad = subgrad_for_lag(h, k_worst, n)
            t_k = 2.0 / (it + 2)
            lr = 0.5 * t_k / np.sqrt(np.sum(grad**2) + 1e-10)
            lr = min(lr, 0.05)
            h = h - lr * grad
            h = project_feasible(h, target_sum)
            avg_h += (h - avg_h) / (it + 1)
            if it % 500 == 499:
                c5 = compute_c5(avg_h)
                if c5 < best_c5:
                    best_h, best_c5 = avg_h.copy(), c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 5: Multi-resolution
    if tl() > 200:
        for n_try in [50, 55, 60, 65, 75, 80, 85, 90, 100, 120]:
            if tl() < 40 or n_try == n:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
            for beta in [500, 5000, 20000]:
                if tl() < 20:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5, n = h_new.copy(), c5, n_try
                        dx, target_sum, bounds = dx_t, ts_t, bnds_t
                except Exception:
                    pass
            # Quick coord descent at this resolution
            if tl() > 30:
                h_cd = project_feasible(res.x, ts_t) if 'res' in dir() and res.success else h_t
                for _ in range(30):
                    if tl() < 10:
                        break
                    h_cd = coord_descent_pass(h_cd, n_try, ts_t)
                c5 = compute_c5(h_cd)
                if c5 < best_c5:
                    best_h, best_c5, n = h_cd.copy(), c5, n_try
                    dx, target_sum, bounds = dx_t, ts_t, bnds_t

    # Phase 6: Final polish on best resolution
    if tl() > 20:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [50000, 200000]:
            if tl() < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final coord descent pass
    if tl() > 15:
        h = best_h.copy()
        h = coord_descent_pass(h, n, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    return best_h, best_c5, len(best_h)
