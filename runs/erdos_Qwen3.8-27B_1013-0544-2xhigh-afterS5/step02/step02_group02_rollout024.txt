def compute_c5(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:

            if c < best_c5 - 1e-14:
                best_h = hp.copy()
                best_c5 = c
                break

    # Phase 9: Ultimate polish
    if rem() > 15:
        hp = slsqp_opt(best_h, n, 10000000, 6000)
        c = compute_c5(hp, n)
        if c < best_c5:
            best_h = hp.copy()
            best_c5 = c
        if rem() > 10:
            h, c = pair_descent(best_h, n, max_rounds=20, time_limit=t0 + budget_s - 5)
            if c < best_c5:
                best_h = h.copy()
                best_c5 = c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
