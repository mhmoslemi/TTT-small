def binary_snap(h, target_sum):
    """Push h toward binary, maintaining sum constraint."""
    h_bin = (h >= 0.5).astype(float)
    diff = int(round(target_sum)) - int(h_bin.sum())
    if diff > 0:
        # Need more ones: flip lowest zeros
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros])[::-1]  # zeros closest to 0.5 first
        for i in range(min(diff, len(order))):
            h_bin[zeros[order[i]]] = 1
    elif diff < 0:
        # Need fewer ones: flip highest ones
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])  # ones closest to 0.5 first
        for i in range(min(-diff, len(order))):
            h_bin[ones[order[i]]] = 0
    return h_bin

                if delta < 0 or np.random.random() < np.exp(-min(delta / max(T, 1e-9), 400)):
                    h_cur = h_new
                    c5_cur = c5_new
                    if c5_cur < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_cur
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

                T = max(T * cooling, T_min)
                if no_improve > 400:
                    T = max(T, 0.003)
                    no_improve = 0

        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur

    # Final high-precision polish
    if tleft() > 15:
        n_best = len(best_h)
        target_sum = n_best / 2.0
        dx = 2.0 / n_best
        bounds = Bounds(np.zeros(n_best), np.ones(n_best))
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [3000, 8000, 20000]:
            if tleft() < 10:
                break
            try:
                res = minimize(obj_func, best_h, args=(beta, dx),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
