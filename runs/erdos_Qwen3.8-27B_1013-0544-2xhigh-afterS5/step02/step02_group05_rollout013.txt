    n = len(h)
    n_corr = 2 * n - 1
    h_j = h[j]
    
    # Compute coefficients and constants for each lag
    coeffs = np.zeros(n_corr)
    consts = np.zeros(n_corr)
    
    for k_idx in range(n_corr):
        k_shift = k_idx - (n - 1)
        # Valid range for n_idx: max(0, -k_shift) to min(n, n - k_shift) - 1
        lo = max(0, -k_shift)
        hi = min(n, n - k_shift)
        if lo >= hi:

        a_indices = np.arange(lo, hi) + k_shift
        b_indices = np.arange(lo, hi)

        x_fine = np.linspace(x_lo, x_hi, 80)
        vals_f = corr_base[None, :] + coeffs_quad[None, :] * x_fine[:, None]**2 + coeffs_lin[None, :] * x_fine[:, None]
        x_best = x_fine[np.argmin(np.max(vals_f, axis=1))]
    return x_best

def coordinate_descent_sweep(h, n_points, max_sweeps=25, time_fn=None, min_time=15):
    best_local_c5 = float('inf')
    for sweep in range(max_sweeps):
        if time_fn is not None and time_fn() < min_time:
            break
        for j in range(n_points):
            h[j] = optimize_single_coord(h, j)
        h = project_sum(h, n_points / 2.0)
        c5 = compute_c5(h)
        if c5 < best_local_c5:
            best_local_c5 = c5
    return h, best_local_c5

def two_block_h(n, frac):
    h = np.zeros(n)
    a = int(round(frac * n))
    b_start = int(round((1.0 + frac) * n))
    for i in range(n):
        x = 2.0 * i / n
        if x < frac or x >= 1.0 + frac:
            h[i] = 1.0
    return project_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def tl():
        return budget_s - (time.time() - start_time)

    n_list = [70, 80, 90]

    for n_points in n_list:
        if tl() < 50:
            break
        target_sum = n_points / 2.0
        dx = 2.0 / n_points

        # Build starting structures
        starts = []
        if 'initial_h_values' in globals() and initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=float)
            n_parent = len(parent)
            if n_points == n_parent:
                starts.append(parent.copy())
            else:
                xo = np.linspace(0, 2, n_parent, endpoint=False)
                xn = np.linspace(0, 2, n_points, endpoint=False)
                starts.append(project_sum(np.interp(xn, xo, parent), target_sum))
        starts.append(two_block_h(n_points, 0.35))
        starts.append(two_block_h(n_points, 0.25))
        starts.append(two_block_h(n_points, 0.45))
        starts.append(project_sum(np.random.random(n_points), target_sum))
        # A "combed" structure
        h_comb = np.zeros(n_points)
        step = n_points // 10
        for blk in range(10):
            s = blk * step + step // 4
            e = s + step // 2
            h_comb[s:e] = 2.0
        starts.append(project_sum(h_comb, target_sum))

        for h0 in starts:
            if tl() < 30:
                break
            h = project_sum(h0.copy(), target_sum)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h.copy(), n_points

            # Coordinate descent
            h, c5_cd = coordinate_descent_sweep(h, n_points, max_sweeps=30, time_fn=tl, min_time=30)
            if c5_cd < best_c5:
                best_c5, best_h, best_n = c5_cd, h.copy(), n_points

            # SLSQP polish
            if tl() > 20:
                bounds = Bounds(np.zeros(n_points), np.ones(n_points))
                for beta in [150, 300]:
                    if tl() < 10:
                        break
                    def obj_sm(hv):
                        return smooth_max(np.correlate(hv, 1.0 - hv, mode="full"), float(beta)) * dx
                    try:
                        res = minimize(obj_sm, h, method='SLSQP', bounds=bounds,
                                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                                      'jac': lambda x: np.ones(n_points)}],
                                       options={'maxiter': 300, 'ftol': 1e-14})
                        h_new = project_sum(res.x, target_sum)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5_cd:
                            h, c5_cd = h_new, c5_new
                            if c5_cd < best_c5:
                                best_c5, best_h, best_n = c5_cd, h.copy(), n_points
                    except Exception:
                        pass

    # Final intensive polish on best solution
    if best_h is not None and tl() > 20:
        nf = best_n
        ts = nf / 2.0
        dxf = 2.0 / nf
        bounds_f = Bounds(np.zeros(nf), np.ones(nf))
        # More coordinate descent sweeps
        best_h, _ = coordinate_descent_sweep(best_h, nf, max_sweeps=40, time_fn=tl, min_time=20)
        best_c5 = compute_c5(best_h)
        # High-beta SLSQP
        for beta in [300, 500, 800]:
            if tl() < 10:
                break
            def obj_fin(hv):
                return smooth_max(np.correlate(hv, 1.0 - hv, mode="full"), float(beta)) * dxf
            try:
                res = minimize(obj_fin, best_h, method='SLSQP', bounds=bounds_f,
                               constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                                              'jac': lambda x: np.ones(nf)}],
                               options={'maxiter': 500, 'ftol': 1e-15})
                h_new = project_sum(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
            except Exception:
                pass
        # One more coordinate descent pass
        best_h, _ = coordinate_descent_sweep(best_h, nf, max_sweeps=20, time_fn=tl, min_time=5)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
