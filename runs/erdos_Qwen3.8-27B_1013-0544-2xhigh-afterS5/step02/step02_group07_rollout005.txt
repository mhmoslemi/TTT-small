    """Project h to simplex-like constraint: 0<=h[i]<=1, sum(h)=target_sum"""
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            break
        step = diff / nf
        # Clamp to not violate bounds
        if step > 0:
            max_step = (1.0 - h[free]).min()
            step = min(step, max_step - 1e-15)
        else:
            min_step = h[free].min()
            step = max(step, -min_step + 1e-15)
        if abs(step) < 1e-16:
            break
        h[free] += step
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, beta, dx):
    """Compute smooth max objective and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = smooth_max_val(corr, beta) * dx
    
    # Get softmax weights
    w = smooth_max_weights(corr, beta)
    
    # Compute gradient
    # lags go from -(n-1) to (n-1), indexed 0 to 2n-2 in corr
    # corr[k] corresponds to lag d = k - (n-1)
    lags = np.arange(2 * n - 1) - (n - 1)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        d = lags[k]
        if d >= 0:
            # dc_k/dh[j] = (1-h[j+d]) for j in [0, n-d-1]
            #              - h[j-d] for j in [d, n-1]
            j1 = np.arange(0, n - d)
            grad[j1] += w[k] * dx * (1.0 - h[j1 + d])
            j2 = np.arange(d, n)
            grad[j2] -= w[k] * dx * h[j2 - d]
        else:
            dd = -d
            # dc_k/dh[j] = (1-h[j-dd]) for j in [dd, n-1]  [term i=j, h[i+d]=h[j-dd]]
            #              - h[j+dd] for j in [0, n-dd-1]   [term i=j+dd, (1-h[i+d])=(1-h[j])]
            j1 = np.arange(dd, n)
            grad[j1] += w[k] * dx * (1.0 - h[j1 - dd])
            j2 = np.arange(0, n - dd)
            grad[j2] -= w[k] * dx * h[j2 + dd]
    
    return f, grad

    """L-BFGS-B optimization with smooth max objective and projection for sum constraint."""
    n = len(h0)
    dx = 2.0 / n

        f, g = obj_grad(h, beta, dx)
        # Add penalty for sum constraint
        s = h.sum()
        penalty = 1000.0 * (s - target_sum) ** 2
        g_pen = 2000.0 * (s - target_sum) * np.ones(n)
        return f + penalty, g + g_pen
    
    res = minimize(penalty_obj, h0, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': maxiter, 'ftol': tol, 'gtol': tol, 'maxls': 50})
    h_opt = project_to_sum(np.clip(res.x, 0, 1), target_sum)
    return h_opt, compute_c5(h_opt)

                    for i in idx:
                        if h_cur[i] > 0.5:
                            h_new[i] += 0.05
                        else:
                            h_new[i] -= 0.05
            elif mt < 0.7:
                blen = rng.randint(2, min(8, n // 5 + 1))
                i = rng.randint(0, n - blen + 1)
                j = rng.randint(0, n - blen + 1)
                tr = 0
                while abs(j - i) < blen and tr < 15:
                    j = rng.randint(0, n - blen + 1)
                    tr += 1
                if abs(j - i) >= blen:
                    tmp = h_new[i:i + blen].copy()
                    h_new[i:i + blen] = h_new[j:j + blen]
                    h_new[j:j + blen] = tmp
            elif mt < 0.85:
                i, j = rng.choice(n, 2, replace=False)
                h_new[i], h_new[j] = h_new[j], h_new[i]
            else:
                n_pts = rng.randint(3, min(12, n // 3))
                idx = rng.choice(n, n_pts, replace=False)
                h_new[idx] += rng.randn(n_pts) * 0.06

            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)

            if c5_new < c5_cur or rng.random() < np.exp(-min(400.0, (c5_new - c5_cur) / max(T, 1e-10))):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                stagnant += 1

            T = max(T * cooling, T_min)
            if stagnant > 500:
                T = max(T, 0.006)
                stagnant = 0

    # Phase 5: Final high-precision polish
    if tleft() > 30:
        for beta in [1000, 2000, 4000]:
            if tleft() < 15:
                break
            try:
                h_new, c5_new = lbfgsb_optimize(best_h, beta, target_sum, maxiter=5000, pen=10000.0)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
            except Exception:
                pass

    # Phase 6: Try slightly different n by interpolation if time allows
    if tleft() > 100:
        for n2 in [n * 2, n + 20]:
            if tleft() < 50 or n2 > 400:
                break
            x_orig = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_interp = np.interp(x_new, x_orig, best_h, period=2)
            h_interp = project_to_sum(h_interp, n2 / 2.0)
            c5_interp = compute_c5(h_interp)
            if c5_interp < best_c5:
                try:
                    h_opt, c5_opt = lbfgsb_optimize(h_interp, 100, n2 / 2.0, maxiter=3000)
                    if c5_opt < best_c5:
                        best_h, best_c5, n = h_opt, c5_opt, n2
                        target_sum = n / 2.0
                except Exception:
                    pass

    best_h = project_to_sum(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
