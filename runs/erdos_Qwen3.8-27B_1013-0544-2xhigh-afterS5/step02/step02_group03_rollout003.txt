import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    exp_s = np.exp(shifted)
    return exp_s / np.sum(exp_s)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_grad(h, dx, beta):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max(corr, beta) * dx
    weights = smooth_max_weights(corr, beta) * dx
    grad = np.zeros(n)
    n_corr = len(corr)
    for m in range(n_corr):
        k = m - (n - 1)
        w = weights[m]
        if w < 1e-16:
            continue
        if k >= 0:
            # j in [k, n-1]: add w*(1-h[j-k])
            # j in [0, n-1-k]: subtract w*h[j+k]
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            la = -k
            # j in [0, n-1-la]: add w*(1-h[j+la])
            # j in [la, n-1]: subtract w*h[j-la]
            grad[:n - la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n - la]
    return val, grad

    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

        def cf(h):
            return h.sum() - ts
        return cf

    def con_jac_fn(h):
        return np.ones(len(h))

    # Phase 1: SLSQP with exact gradient, increasing beta
    beta_schedule = [5, 20, 80, 300, 1000, 3000]

            break

            if trial > 0:

            def fun(h):
                return obj_grad(h, dx, beta)[0]

                return obj_grad(h, dx, beta)[1]
