    dx = 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_subgrad(h, beta=100):
    """Compute subgradient of smooth max of correlations w.r.t. h."""
    n = len(h)
    corr = compute_corr(h)
    # Soft max weights
    m = np.max(corr)
    w = np.exp(beta * (corr - m)) / np.exp(beta * (corr - m)).sum()
    
    # Gradient of each c[d] w.r.t. h
    # c[d] for d = k - (n-1), where k = 0, ..., 2n-2
    grad = np.zeros(n)
    for k in range(2*n - 1):
        d = k - (n - 1)
        if abs(w[k]) < 1e-10:
            continue
        if d == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        elif d > 0:
            # c[d] = sum_{i=d}^{n-1} h[i]*(1-h[i-d])
            # ∂c[d]/∂h[j] = (1-h[j-d]) for j in [d, n-1] - h[j+d] for j in [0, n-1-d]
            for j in range(d, n):
                grad[j] += w[k] * (1.0 - h[j-d])
            for j in range(0, n - d):
                grad[j] -= w[k] * h[j+d]
        else:
            e = -d
            # c[-e] = sum_{i=0}^{n-1-e} h[i]*(1-h[i+e])
            # ∂c[-e]/∂h[j] = (1-h[j+e]) for j in [0, n-1-e] - h[j-e] for j in [e, n-1]
            for j in range(0, n - e):
                grad[j] += w[k] * (1.0 - h[j+e])
            for j in range(e, n):
                grad[j] -= w[k] * h[j-e]
    return grad

def coordinate_descent(h, target_sum, n_sweeps=10):

    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            def obj_1d(x_i):
                h[i] = x_i
                corr = np.correlate(h, 1.0 - h, mode="full")
                return np.max(corr)
            res = minimize_scalar(obj_1d, bounds=(0, 1), method='bounded', options={'xatol': 1e-8})
            h[i] = res.x
            # Fix sum constraint
            diff = target_sum - h.sum()
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
            # Re-fix sum
            diff = target_sum - h.sum()
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                improved = True
            else:
                h[i] = old_val
                diff = target_sum - h.sum()
                h += diff / n
                h = np.clip(h, 0.0, 1.0)
        if not improved:
            break
    return h, best_c5

                        if h_cur[i] > 0.4 and h_cur[j] < 0.6:
                            h_new[i] -= amt
                            h_new[j] += amt
                else:
                    for _ in range(n_adj):
                        i = np.random.randint(n_pts)
                        if h_cur[i] > 0.5:
                            h_new[i] -= 0.06
                        else:
                            h_new[i] += 0.06
            elif r < 0.62:
                # Block operations
                blen = np.random.randint(2, min(8, n_pts // 5 + 1))
                i = np.random.randint(0, n_pts - blen + 1)
                if np.random.random() < 0.4:
                    h_new[i:i + blen] = 1.0 - h_new[i:i + blen]
                else:
                    shift = np.random.randint(-3, 4)
                    if shift != 0:
                        block = h_new[i:i + blen].copy()
                        h_new[i:i + blen] = np.roll(block, shift)
            elif r < 0.78:
                # Gaussian perturbation subset
                frac = 0.1 + 0.25 * (T / 0.010)
                mask = np.random.random(n_pts) < frac
                nm = int(mask.sum())
                if nm > 0:
                    h_new[mask] += np.random.randn(nm) * (0.02 + 0.06 * T / 0.010)
            elif r < 0.90:
                # Pair transfers
                n_pairs = np.random.randint(2, 6)
                for _ in range(n_pairs):
                    i, j = np.random.choice(n_pts, 2, replace=False)
                    amt = 0.02 + 0.06 * np.random.random()
                    if h_cur[i] >= h_cur[j]:
                        h_new[i] -= amt
                        h_new[j] += amt
                    else:
                        h_new[i] += amt
                        h_new[j] -= amt
            else:
                # Large kick
                n_k = np.random.randint(8, 25)
                idx = np.random.choice(n_pts, n_k, replace=False)
                h_new[idx] += np.random.randn(n_k) * 0.12

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(300.0, delta / max(T, 1e-12))):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    h_best = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            T = max(T * cooling, T_min)
            if no_improve > 600:
                T = max(T, 0.007)
                no_improve = 0
        c5_cur = compute_c5(project_h(h_cur, target_sum))
        if c5_cur < best_c5:
            h_best = project_h(h_cur, target_sum)
            best_c5 = c5_cur

    # Phase 3: Coordinate descent
    if (budget_s - (time.time() - t0)) > 60:
        h_cd, c5_cd = coordinate_descent(h_best.copy(), target_sum, n_sweeps=6)
        if c5_cd < best_c5:
            h_best, best_c5 = h_cd, c5_cd

    # Phase 4: SLSQP high-precision polish
    if (budget_s - (time.time() - t0)) > 40:
        for beta in [400, 800, 1500]:
            if (time.time() - t0) > budget_s - 15:
                break
            try:
                res = minimize(smooth_max_obj, h_best, args=(beta,),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    h_best, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Multi-start SLSQP from perturbations
    if (budget_s - (time.time() - t0)) > 50:
        for attempt in range(5):
            if (time.time() - t0) > budget_s - 15:
                break
            h0 = h_best.copy()
            n_pert = np.random.randint(4, 12)
            idx = np.random.choice(n_pts, n_pert, replace=False)
            h0[idx] += np.random.randn(n_pert) * 0.05
            h0 = project_h(h0, target_sum)
            try:
                res = minimize(smooth_max_obj, h0, args=(500,),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 2500, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    h_best, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 6: Final coordinate descent + SLSQP
    if (budget_s - (time.time() - t0)) > 25:
        h_cd, c5_cd = coordinate_descent(h_best.copy(), target_sum, n_sweeps=4)
        if c5_cd < best_c5:
            h_best, best_c5 = h_cd, c5_cd
        if (budget_s - (time.time() - t0)) > 10:
            try:
                res = minimize(smooth_max_obj, h_best, args=(2000,),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    h_best, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    h_best = project_h(h_best, target_sum)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, n_pts
