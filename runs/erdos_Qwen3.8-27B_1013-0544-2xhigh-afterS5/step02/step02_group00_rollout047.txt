    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_convex(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, beta):
    """Return (value, gradient weights) for smooth max."""
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_vals = np.exp(x_shift)
    weights = exp_vals / exp_vals.sum()
    value = m + np.log(np.sum(exp_vals) / len(x)) / beta
    return value, weights

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_convex(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_value(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_subgrad(h, beta, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[::-1][:top_k]
        corr_use = corr[idx]
        m = np.max(corr_use)
        x_shift = np.clip(beta * (corr_use - m), -500.0, 500.0)
        exp_vals = np.exp(x_shift)
        weights_full = np.zeros(len(corr))
        weights_full[idx] = exp_vals / exp_vals.sum()
    else:
        m = np.max(corr)
        x_shift = np.clip(beta * (corr - m), -500.0, 500.0)
        exp_vals = np.exp(x_shift)
        weights_full = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    active = np.where(weights_full > 1e-9)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        w = weights_full[idx]
        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - k:j_hi + 1 - k])
        j_lo2 = max(0, -k)
        j_hi2 = min(n - 1, n - 1 - k)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= w * h[j_lo2 + k:j_hi2 + 1 + k]
    return grad, corr

def _slsqp_obj(h_in):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max_value(c, 400.0) * (2.0 / len(h_in))

def _slsqp_con(x_in):
    return np.sum(x_in) - _global_target

def _slsqp_jac(x_in):
    return np.ones(x_in.shape)

def subgrad_descent(h, target_sum, n_steps, beta, start_alpha, top_k=None):
    c5 = compute_c5(h)
    for step in range(n_steps):
        grad, corr = compute_subgrad(h, beta, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        alpha = start_alpha / (1.0 + 0.001 * step)
        for attempt in range(4):
            h_trial = project_convex(h - alpha * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5 - 1e-15:
                h = h_trial
                c5 = c5_trial
                break
            alpha *= 0.4
    return h, c5

def try_resolution(h_parent, n_points, target_sum, budget_remaining):
    x_old = np.linspace(0, 2, len(h_parent), endpoint=False)
    x_new = np.linspace(0, 2, n_points, endpoint=False)
    h = np.interp(x_new, x_old, h_parent)
    h = project_convex(h, target_sum)
    c5 = compute_c5(h)
    if budget_remaining() < 10:

    h, c5 = subgrad_descent(h, target_sum, 4000, 80, 0.15, top_k=None)

    _global_target = target_sum

    constraints = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-14})
        h_new = project_convex(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    except Exception:
        pass
    if budget_remaining() < 5:

    if budget_remaining() < 5:

    h, c5 = subgrad_descent(h, target_sum, 5000, 600, 0.03, top_k=2)

def run(seed=42, budget_s=1100, **kwargs):
