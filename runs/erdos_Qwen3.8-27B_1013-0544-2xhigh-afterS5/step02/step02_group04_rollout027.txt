        if time_left() < 35:
            break
        h_new, c5 = slsqp_optimize(best_h, n_points, target_sum, bounds, constraints, beta, 5000, 1e-16)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 3: Targeted worst-lag reduction + re-optimize
    for attempt in range(50):
        if time_left() < 25:
            break
        h0 = best_h.copy()
        strat = attempt % 6
        if strat == 0:
            h0 = targeted_perturb(h0, n_points, target_sum, rng)
        elif strat == 1:
            h0 += rng.randn(n_points) * 0.004
        elif strat == 2:
            i, j = rng.choice(n_points, 2, replace=False)
            d = 0.03 * rng.randn()
            h0[i] += d
            h0[j] -= d
        elif strat == 3:
            blen = rng.randint(2, 6)
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] += rng.randn(blen) * 0.015
        elif strat == 4:
            h0 = 0.7 * h0 + 0.3 * h0[::-1]
            h0 += rng.randn(n_points) * 0.005
        else:
            # Flatten peaks slightly
            h0 = 0.95 * h0 + 0.05 * np.mean(h0)

        h0 = proj_sum_box(h0, target_sum)
        beta = 150 + 100 * (attempt % 5)
        h_new, c5 = slsqp_optimize(h0, n_points, target_sum, bounds, constraints, beta, 3000, 1e-15)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 4: Binary-like exploration
    if time_left() > 60:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(h_bin.sum())
        if n_ones > n_points // 2:
            max_idx = np.where(h_bin == 1)[0]
            to_rm = max_idx[np.argsort(best_h[max_idx])[:n_ones - n_points // 2]]
            h_bin[to_rm] = 0
        elif n_ones < n_points // 2:
            min_idx = np.where(h_bin == 0)[0]
            to_ad = min_idx[np.argsort(-best_h[min_idx])[:n_points // 2 - n_ones]]
            h_bin[to_ad] = 1
        h_blend = 0.4 * best_h + 0.6 * h_bin
        h_blend = proj_sum_box(h_blend, target_sum)
        h_new, c5 = slsqp_optimize(h_blend, n_points, target_sum, bounds, constraints, 200, 3000, 1e-15)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 5: Final high-precision polish
    for beta in [1000, 2000, 5000]:
        if time_left() < 12:
            break
        h_new, c5 = slsqp_optimize(best_h, n_points, target_sum, bounds, constraints, beta, 8000, 1e-17)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Final Nesterov polish at high beta
    if time_left() > 15:
        best_h, best_c5 = nesterov_phase(best_h, n_points, target_sum, 500.0, 1000.0, 500, time_left, 5)

    best_h = proj_sum_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
