```python
def grad_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        e = np.exp(sh - mx)
    else:
        e = np.exp(sh)
    w = e / e.sum()
    
    # Compute gradient efficiently
    # first[j] = sum_k w[k] * (1-h[j+offsets[k]]) for valid
    # second[j] = sum_k w[k] * h[j-offsets[k]] for valid
    # 
    # key insight: first = correlate(w_reversed, 1-h) aligned properly
    # Actually, let's just use the direct O(n^2) approach since n is small
    
    offsets = np.arange(-(n-1), n)  # length 2n-1
    A = 1.0 - h
    B = h
    first = np.zeros(n)
    second = np.zeros(n)
    for k in range(2*n - 1):
        off = offsets[k]
        wk = w[k]
        if wk < 1e-15:
            continue
        if off >= 0:
            # A[off:n] has length n-off, corresponds to first[0:n-off]
            first[:n-off] += wk * A[off:]
            # B[0:n-off] corresponds to second[off:n]
            second[off:] += wk * B[:n-off]
        else:
            ao = -off
            # A[0:n-ao] corresponds to first[ao:n]
            first[ao:] += wk * A[:n-ao]
            # B[ao:n] corresponds to second[0:n-ao]
            second[:n-ao] += wk * B[ao:]
    
    return (first - second) / alpha
```

This is O(n * 2n) = O(n^2) which for n=80 is about 12800 operations. Very fast!

            L = n - off
            first[:L] += wk * A[off:]
            second[off:] += wk * B[:L]
        else:
            ao = -off
            L = n - ao
            first[ao:] += wk * A[:L]
            second[:L] += wk * B[ao:]
    return (first - second) / alpha

def slsqp(h, n, alpha, maxiter=2000, ftol=1e-16):
    def f(x): return obj_lse(x, n, alpha)
    def g(x): return grad_lse(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def gen_mseq(n):
    state = 0x7F
    seq = np.zeros(n)
    for i in range(min(n, 63)):
        seq[i] = state & 1
        fb = (state & 1) ^ ((state >> 6) & 1)
        state = (state >> 1) | (fb << 6)
    cur = seq.sum()
    tgt = n / 2.0
    for i in range(min(n, 63), n):
        if cur < tgt - 1e-10:
            seq[i] = 1.0
            cur += 1.0
    return project_feasible(seq, n)
