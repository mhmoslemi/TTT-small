    starts = [(h_best.copy(), n_best)]
    for trial in range(40):
        if tl() < 150:
            break
        n = n_best
        target = n / 2.0
        s = trial % 8
        if s == 0:
            h = 0.5 + 0.35 * np.random.randn(n)
        elif s == 1:
            h = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            h[idx] = 1.0
            h += 0.12 * np.random.randn(n)
        elif s == 2:
            x = np.arange(n) * 2.0 / n
            h = 0.5
            for _ in range(np.random.randint(2, 5)):
                w = np.random.uniform(0.5, 6.0) * 2 * np.pi
                p = np.random.uniform(0, 2 * np.pi)
                a = np.random.uniform(0.05, 0.3)
                h += a * np.sin(w * x + p)
        elif s == 3:
            # Nonlinear transform of sine
            x = np.arange(n) * 2.0 / n
            base = np.sin(2 * np.pi * np.random.randint(1, 4) * x + np.random.uniform(0, 6.28))
            h = 0.5 + 0.4 * np.sign(base) * np.abs(base) ** np.random.uniform(0.3, 1.5)
        elif s == 4:
            # Quadratic phase
            x = np.arange(n) * 2.0 / n
            c = np.random.uniform(1, 8)
            h = 0.5 + 0.3 * np.cos(c * x ** 2 + np.random.uniform(0, 6.28))
        elif s == 5:
            # Chebyshev-like
            x = np.arange(n) * 2.0 / n
            h = 0.5 + 0.3 * np.cos(np.random.uniform(2, 12) * x) * np.cos(np.random.uniform(2, 12) * x)
        elif s == 6:
            # Perturbed best
            h = h_best.copy() if len(h_best) == n else np.full(n, 0.5)
            h += 0.2 * np.random.randn(n)
        else:
            # Step function with random offsets
            nblocks = np.random.randint(3, 12)
            h = np.zeros(n)
            for b in range(nblocks):
                start = np.random.randint(0, n)
                length = np.random.randint(2, max(3, n // nblocks))
                h[start:start + length] = np.random.choice([0.3, 0.7, 1.0, 0.8])
        
        h = project_box_sum(h, target)
        starts.append((h, n))

    # Coordinate descent from each start
    for h_start, n in starts:
        if tl() < 20:
            break
        h = h_start.copy()
        max_sweeps = 500 if n == n_best else 200
        for s in range(max_sweeps):
            if tl() < 10:
                break
            # Alternate forward and backward sweeps
            if s % 2 == 0:
                h = cd_sweep(h, n)
            else:
                h = cd_sweep_reversed(h, n)
            c5 = compute_c5(h)
            if n == n_best and c5 < c5_best - 1e-12:
                c5_best = c5
                h_best = h.copy()
            elif c5 < c5_best - 1e-12:
                pass  # Track for this n
        
        # Polish
        if tl() > 30:
            for _ in range(200):
                if tl() < 15:
                    break
                h = cd_sweep(h, n)
                c5 = compute_c5(h)
                if n == n_best and c5 < c5_best:
                    c5_best = c5
                    h_best = h.copy()

    # Intensive polish on best solution
    if tl() > 30:
        h = h_best.copy()
        n = n_best
       停滞 = 0
        for _ in range(3000):
            if tl() < 5:
                break
            h_new = cd_sweep(h, n)
            h_new2 = cd_sweep_reversed(h_new, n)
            c5 = compute_c5(h_new2)
            if c5 < c5_best - 1e-14:
                c5_best = c5
                h_best = h_new2.copy()
                h = h_new2
                停滞 = 0
            else:
                h = h_new2
                停滞 += 1
                if 停滞 > 100:
                    break

    # Try different n values via interpolation
    for n_try in [50, 60, 80, 90, 100, 120, 140]:
        if tl() < 90:
            break
        if n_try == n_best:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best, period=2.0)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        
        for _ in range(300):
            if tl() < 45:
                break
            if np.random.rand() < 0.5:
                h_interp = cd_sweep(h_interp, n_try)
            else:
                h_interp = cd_sweep_reversed(h_interp, n_try)
        
        c5_try = compute_c5(h_interp)
        if c5_try < c5_best:
            c5_best = c5_try
            h_best = h_interp.copy()
            n_best = n_try

    # Final: also try random restarts from best at same n
    if tl() > 60:
        for trial in range(10):
            if tl() < 30:
                break
            h = h_best + np.random.randn(n_best) * 0.05
            h = project_box_sum(h, n_best / 2.0)
            for _ in range(200):
                if tl() < 15:
                    break
                h = cd_sweep(h, n_best)
            c5 = compute_c5(h)
            if c5 < c5_best:
                c5_best = c5
                h_best = h.copy()

    h_best = project_box_sum(h_best, len(h_best) / 2.0)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
