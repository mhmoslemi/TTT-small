```python
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    g = np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # For each lag k (0 to 2n-2), the shift is k - (n-1)
    # grad_k[j] = (1-h[j-shift]) if valid - h[j+shift] if valid
    for k in range(2*n-1):
        shift = k - (n-1)
        if abs(w[k]) < 1e-15:
            continue
        for j in range(n):
            val = 0.0
            i1 = j - shift
            if 0 <= i1 < n:
                val += (1.0 - h[i1])
            i2 = j + shift
            if 0 <= i2 < n:
                val -= h[i2]
            g[j] += w[k] * val
    return g
```

```python
def _objective_and_grad(h, n, target_sum, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    # Smooth max
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (cmax + np.log(sum_exp) / beta) * dx + lam * (h.sum() - target_sum)**2
    
    # Gradient of smooth max
    weights = exp_vals / sum_exp  # softmax weights
    # d/dh[j] of corr[k]
    g = np.zeros(n)
    for k in range(2*n-1):
        shift = k - (n-1)
        wk = weights[k] * dx
        if abs(wk) < 1e-15:
            continue
        for j in range(n):
            val = 0.0
            i1 = j - shift
            if 0 <= i1 < n:
                val += (1.0 - h[i1])
            i2 = j + shift
            if 0 <= i2 < n:
                val -= h[i2]
            g[j] += wk * val
    # Gradient of penalty
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _shift_right(arr, d, n):
    result = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            result[d:] = arr[:n-d]
    else:
        if n + d > 0:
            result[:n+d] = arr[-d:]
    return result

def _shift_left(arr, d, n):
    result = np.zeros(n)
    if d >= 0:
        if n - d > 0:
            result[:n-d] = arr[d:]
    else:
        if n + d > 0:
            result[-d:] = arr[:n+d]
    return result
```

```python
def _grad_corr_weighted(h, weights, n):
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        shift = k - (n - 1)
        wk = weights[k]
        if abs(wk) < 1e-18:
            continue
        g += wk * _shift_right(omh, shift, n)
        g -= wk * _shift_left(h, shift, n)
    return g

def _obj_grad(h, n, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -200, 200)
    ev = np.exp(shifted)
    se = ev.sum()
    obj = (cmax + np.log(se) / beta) * dx + lam * (h.sum() - n / 2.0) ** 2
    w = (ev / se) * dx
    g = _grad_corr_weighted(h, w, n)
    g += 2.0 * lam * (h.sum() - n / 2.0)

            c = compute_c5(h_new)

            pass
```

```python
        best_h, best_c5 = _optimize(best_h, n, ts, bnds, t0, budget_s, 5)

    # Phase 4: Very tight final stages
    if _remaining(t0, budget_s) > 10:
        for beta, lam, maxiter in [(5000, 3000, 2000), (10000, 5000, 3000)]:
            if _remaining(t0, budget_s) < 8:
                break
            try:
                r = minimize(_obj_grad, best_h, method='L-BFGS-B',
                            bounds=bnds, jac=True, args=(n, beta, lam),
                            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
