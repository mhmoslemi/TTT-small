def project_h(h, target_sum):

        if abs(diff) < 1e-14:

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()

def smooth_max_val(corr, beta):
    m = np.max(corr)
    if beta < 1:
        return m
    shifted = beta * (corr - m)
    mx = np.max(shifted)
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / beta


def softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    mx = np.max(shifted)
    e = np.exp(shifted - mx)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return e / s


def compute_grad(h, beta):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, beta)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-16:

        k = m - (n - 1)
        wm = w[m]
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if hi < lo:

        j1s = lo + k
