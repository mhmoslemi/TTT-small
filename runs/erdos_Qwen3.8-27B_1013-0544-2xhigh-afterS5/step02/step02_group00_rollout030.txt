import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_pair_coeff(h, i, j):
    n = len(h)
    s = np.arange(-(n - 1), n)
    coeff = np.zeros(2 * n - 1)
    idx_s_i = i + s
    m1 = (idx_s_i >= 0) & (idx_s_i < n)
    coeff[m1] += (1.0 - h[idx_s_i[m1]])
    idx_ms_i = i - s
    m2 = (idx_ms_i >= 0) & (idx_ms_i < n)
    coeff[m2] -= h[idx_ms_i[m2]]
    idx_s_j = j + s
    m3 = (idx_s_j >= 0) & (idx_s_j < n)
    coeff[m3] -= (1.0 - h[idx_s_j[m3]])
    idx_ms_j = j - s
    m4 = (idx_ms_j >= 0) & (idx_ms_j < n)
    coeff[m4] += h[idx_ms_j[m4]]
    return coeff


def pair_swap(h, i, j, dx, n_grid=25):
    n = len(h)
    d = compute_pair_coeff(h, i, j)
    corr = np.correlate(h, 1.0 - h, mode="full")
    d_lo = max(0.0, -h[i])
    d_hi = min(1.0 - h[i], h[j])
    if d_hi <= d_lo + 1e-15:
        return h, False
    deltas = np.linspace(d_lo, d_hi, n_grid)
    best_c5 = np.inf
    best_delta = 0.0
    for delta in deltas:
        val = np.max(corr + delta * d) * dx
        if val < best_c5:
            best_c5 = val
            best_delta = delta
    if abs(best_delta) > 1e-15:
        h[i] += best_delta
        h[j] -= best_delta
        return h, True
    return h, False


def smooth_max_obj(h, dx, beta):
    overlap = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(overlap)
    x_shift = np.clip(beta * (overlap - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(overlap)) / beta) * dx


def slsqp_optimize(h, target_sum, dx, maxiter, beta, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x_in):
        return smooth_max_obj(x_in, dx, beta)

    def con(x_in):
        return np.sum(x_in) - target_sum

    def con_jac(x_in):
        return np.ones(x_in.shape)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(res.x, target_sum)
    except Exception:
        return h.copy()


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([60, 80, 100, 120, 140, 160, 200, 240, 280, 320, 400]))
    if len(h_parent) in n_list:
        n_list.remove(len(h_parent))
    n_list.insert(0, len(h_parent))

    for n_points in n_list:
        if time_left() < 40:
            break
        target_sum = n_points / 2.0
        dx = 2.0 / n_points

        x_b = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_points)
        h_cur = project_h(np.interp(x_n, x_b, best_h), target_sum)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

        # Also try a fresh structured init at this resolution
        alt = np.where(np.arange(n_points) % 3 == 0, 0.6, 0.4)
        alt = 0.5 + 0.1 * (alt - 0.5) * 2
        h_alt = project_h(alt, target_sum)
        c5_alt = compute_c5(h_alt)
        if c5_alt < c5_cur:
            h_cur = h_alt
            c5_cur = c5_alt
            if c5_alt < best_c5:
                best_h, best_c5 = h_cur.copy(), c5_alt

        # SLSQP phase 1
        if time_left() > 30:
            h_new = slsqp_optimize(h_cur, target_sum, dx, 500, 100, time_left)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur

        # Pair swap coordinate descent
        if time_left() > 15:
            no_improve = 0
            max_iters = 30000 if n_points <= 200 else 15000
            for it in range(max_iters):
                if time_left() < 5:
                    break
                i, j = np.random.choice(n_points, 2, replace=False)
                h_cur, improved = pair_swap(h_cur, i, j, dx, n_grid=20)
                if improved:
                    c5_cur = compute_c5(h_cur)
                    if c5_cur < best_c5:
                        best_h, best_c5 = h_cur.copy(), c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 5000:
                        # Perturb to escape
                        idx = np.random.choice(n_points, size=min(5, n_points), replace=False)
                        h_cur[idx] += np.random.uniform(-0.05, 0.05, size=len(idx))
                        h_cur = project_h(h_cur, target_sum)
                        c5_cur = compute_c5(h_cur)
                        no_improve = 0

        # SLSQP phase 2 (higher beta)
        if time_left() > 20:
            h_new = slsqp_optimize(h_cur, target_sum, dx, 1000, 300, time_left)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), best_c5 if False else c5_new

            # More pair swaps after SLSQP
            if time_left() > 15:
                for it in range(20000):
                    if time_left() < 5:
                        break
                    i, j = np.random.choice(n_points, 2, replace=False)
                    h_cur, improved = pair_swap(h_cur, i, j, dx, n_grid=15)
                    if improved:
                        c5_cur = compute_c5(h_cur)
                        if c5_cur < best_c5:
                            best_h, best_c5 = h_cur.copy(), c5_cur

    # Final polish on best solution
    if time_left() > 10 and best_h is not None:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        dx_fin = 2.0 / n_fin
        h_fin = best_h.copy()
        h_new = slsqp_optimize(h_fin, t_fin, dx_fin, 3000, 600, time_left)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

        for it in range(20000):
            if time_left() < 3:
                break
            i, j = np.random.choice(n_fin, 2, replace=False)
            best_h, improved = pair_swap(best_h, i, j, dx_fin, n_grid=15)
            if improved:
                c5_tmp = compute_c5(best_h)
                if c5_tmp < best_c5:
                    best_c5 = c5_tmp

    return best_h, best_c5, len(best_h)
