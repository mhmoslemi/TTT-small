from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def corr_lag_gradient(h, d):
    """Gradient of correlation at lag d (where d = m - (n-1)) w.r.t h."""
    n = len(h)
    grad = np.zeros(n)
    # dc/dh[j] = (1-h[j+d]) if 0<=j+d<n, minus h[j-d] if 0<=j-d<n
    # First term: (1-h[j+d]) for j where 0<=j<n and 0<=j+d<n
    if d >= 0:
        # j+d < n means j < n-d
        valid = np.arange(n - d)
        grad[valid] += (1.0 - h[valid + d])
        # j-d >= 0 means j >= d
        valid2 = np.arange(d, n)
        grad[valid2] -= h[valid2 - d]
    else:
        da = -d
        # j+d < n always true for j < n (since d < 0, j+d < j < n)
        # but need j+d >= 0, i.e., j >= da
        valid = np.arange(da, n)
        grad[valid] += (1.0 - h[valid + d])
        # j-d < n means j < n+d = n-da
        # j-d >= 0 means j >= d (always true since d < 0)
        valid2 = np.arange(n - da)
        grad[valid2] -= h[valid2 - d]
    return grad

def obj_smooth_max(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def obj_grad_smooth_max(h, beta, dx):

    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    # Gradient of smooth_max = dot(weights, dc/dh) for each h component
    grad = np.zeros(len(h))
    n = len(h)

        if weights[m_idx] < 1e-10:
            continue
        d = m_idx - (n - 1)
        grad += weights[m_idx] * corr_lag_gradient(h, d)
    return grad * dx
