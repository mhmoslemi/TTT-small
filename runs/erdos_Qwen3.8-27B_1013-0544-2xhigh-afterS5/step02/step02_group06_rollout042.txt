import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            # If all at bounds, adjust the least extreme ones
            if d > 0:
                # Need to increase sum, decrease the smallest 1s
                idx = np.argsort(h)[:abs(d)]
                h[idx] = np.maximum(0, h[idx] + d / max(1, len(idx)))
            else:
                idx = np.argsort(h)[-abs(d):]
                h[idx] = np.minimum(1, h[idx] + d / max(1, len(idx)))
            h = np.clip(h, 0.0, 1.0)
        else:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def binary_energy(b, n):
    """Compute max correlation for binary sequence b (0/1 values)."""
    corr = np.correlate(b.astype(float), 1.0 - b.astype(float), mode="full")
    return float(np.max(corr)) * (2.0 / n), corr

def binary_local_search(b, n, t0, budget_s, max_iter=100000):
    """Local search on binary sequences with exactly n/2 ones."""
    ts = n // 2
    h = b.astype(float).copy()
    curr_c5, curr_corr = binary_energy(h, n)
    best_h = h.copy()
    best_c5 = curr_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Temperature schedule
    T_start = 0.1
    T_end = 0.001
    n_iter = max_iter
    
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 30:
            break
        
        # Decay temperature
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        # Random swap: one 1 -> 0, one 0 -> 1
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Try the swap
        h[i] = 0
        h[j] = 1
        new_c5, _ = binary_energy(h, n)
        delta = new_c5 - curr_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            curr_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def swap_delta(h, n_pts, i, j):
    """Compute change in corr when h[i]: 1->0 and h[j]: 0->1. Returns array of length 2*n_pts-1."""
    delta = np.zeros(2 * n_pts - 1)
    # Affected n values for each k
    # n such that n+k or n is in {i, j}
    # i.e., n ∈ {i-k, i, j-k, j} for each k
    # But we need to iterate over k and check which affected n are in S_k
    
    # For efficiency, let's iterate over the affected n values and determine which k they affect
    # n affects corr[k] when: n ∈ S_k, i.e., 0≤n<n_pts and 0≤n+k<n_pts
    # So for a given n, k must satisfy: -n ≤ k ≤ n_pts-1-n
    # And the term at (n, k) is h[n+k]*(1-h[n])
    
    # Affected (n, k) pairs:
    # For n = i-k: h[n+k] = h[i] (changed), so all k where n=i-k is in S_k → all k where 0≤i-k<n_pts and 0≤i<n_pts → k in [i-n_pts+1, i]
    #   But also need the term: h[i]*(1-h[i-k])
    # For n = i: h[n] = h[i] (changed), so all k where n=i is in S_k → k in [-i, n_pts-1-i]
    #   Term: h[i+k]*(1-h[i])
    # For n = j-k: h[n+k] = h[j] (changed), k in [j-n_pts+1, j]
    #   Term: h[j]*(1-h[j-k])
    # For n = j: h[n] = h[j] (changed), k in [-j, n_pts-1-j]
    #   Term: h[j+k]*(1-h[j])
    
    # This is getting complex. Let me just do it the straightforward way.
    # For all k in range(-(n_pts-1), n_pts):
    #   Find affected n values in S_k
    #   Compute old and new terms
    
    for k in range(-(n_pts - 1), n_pts):
        d = 0.0
        # n values that are affected and in S_k
        for n_val in set([i - k, i, j - k, j]):
            if 0 <= n_val < n_pts and 0 <= n_val + k < n_pts:
                # This n is in S_k and is affected
                nk = n_val + k
                # Old term
                old_term = h[nk] * (1 - h[n_val])
                # New term
                new_nk_h = h[nk]
                if nk == i:
                    new_nk_h = 0.0
                elif nk == j:
                    new_nk_h = 1.0
                new_n_h = h[n_val]
                if n_val == i:
                    new_n_h = 0.0
                elif n_val == j:
                    new_n_h = 1.0
                new_term = new_nk_h * (1 - new_n_h)
                d += new_term - old_term
        if d != 0:
            delta[k + (n_pts - 1)] = d
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        m = (h > 1e-15) & (h < 1.0 - 1e-15)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def make_objective(n, target_sum, beta):
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, beta) * (2.0 / n) + 1000.0 * (h.sum() - target_sum) ** 2
    return obj

def continuous_polish(h, n, t_start, time_limit):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in [50, 100, 200, 500, 1000, 2000]:
        if time.time() - t_start > time_limit:
            break
        obj = make_objective(n, ts, beta)
        try:
            r = minimize(obj, best_h, method='L-BFGS-B', bounds=bnds,
                        options={'maxiter': 1000, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass
    
    return best_h, best_c5

    """Local search on binary sequences."""
    h = h_bin.astype(float).copy()
    ts = n // 2
    # Ensure exactly ts ones
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    while len(ones) > ts:

        h[idx] = 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    while len(ones) < ts:
        idx = zeros[np.random.randint(len(zeros))]
        h[idx] = 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    curr_max = float(np.max(corr))
    best_h = h.copy()
    best_max = curr_max
    
    # Phase 1: Steepest descent
    improved = True
    while improved and time.time() - t_start < time_limit * 0.3:
        improved = False
        best_swap = None
        best_delta = 0
        for ii in range(len(ones)):
            for jj in range(len(zeros)):
                i, j = ones[ii], zeros[jj]
                h[i] = 0
                h[j] = 1
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                delta = new_max - curr_max
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0

            i, j = best_swap
            h[i] = 0
            h[j] = 1
            curr_max += best_delta
            ones[ones == i] = j
            zeros[zeros == j] = i
            if curr_max < best_max:
                best_max = curr_max
                best_h = h.copy()
            improved = True

    T = max(1.0, abs(best_max) * 0.1)

        h[i] = 0

            ones[ones == i] = j

                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def obj_continuous(h, n, ts, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + 1000.0 * (h.sum() - ts) ** 2
