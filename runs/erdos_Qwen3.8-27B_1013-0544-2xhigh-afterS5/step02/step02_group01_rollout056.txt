```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def grad_smoothmax(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    j = np.arange(n)
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
    else:
        idx = np.arange(len(corr))
    c_sub = corr[idx]
    m = np.max(c_sub)
    exps = np.exp(beta * (c_sub - m))
    sw = np.sum(exps)
    w = exps / sw
    grad = np.zeros(n)
    for ki in range(len(idx)):
        wk = w[ki]
        if wk < 1e-12:
            continue
        s = idx[ki] - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return grad

def smoothmax_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

def smoothmax_jac(x, beta):
    return grad_smoothmax(x, beta)

def pen_obj(x, beta, target, pen):
    return smoothmax_obj(x, beta) + pen * (x.sum() - target) ** 2

def pen_jac(x, beta, target, pen):
    g = grad_smoothmax(x, beta)
    return g + 2.0 * pen * (x.sum() - target) * np.ones(len(x))

def sum_eq_fun(x, target):
    return x.sum() - target

def sum_eq_jac(x, target):
    return np.ones(len(x))

def lbfgs_refine(h, maxiter, beta, pen=500.0):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(pen_obj, h, jac=pen_jac, method='L-BFGS-B',
                       bounds=bounds, args=(beta, target, pen),
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        res = minimize(smoothmax_obj, h, jac=smoothmax_jac,
                       args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def annealed_gd(h, max_time, beta_start=5.0, beta_end=200.0,
                top_k_start=50, top_k_end=5, seed=42, n_stages=20):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    t0 = time.time()
    momentum = 0.85
    v = np.zeros(n)
    for stage in range(n_stages):
        frac = stage / max(n_stages - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        top_k = max(5, int(top_k_start * (1 - frac) + top_k_end * frac))
        lr = 0.03 / (1.0 + 0.02 * beta)
        for it in range(400):
            if time.time() - t0 > max_time:
                cv = compute_c5(x)
                if cv < best_v:
                    best_v, best_x = cv, x.copy()
                return best_x, best_v
            grad = grad_smoothmax(x, beta, top_k)
            v = momentum * v - lr * grad
            x = project_feasible(x + v, target)
            if it % 100 == 99:
                cv = compute_c5(x)
                if cv < best_v:
                    best_v, best_x = cv, x.copy()
    cv = compute_c5(x)
    if cv < best_v:
        best_v, best_x = cv, x.copy()
    return best_x, best_v
