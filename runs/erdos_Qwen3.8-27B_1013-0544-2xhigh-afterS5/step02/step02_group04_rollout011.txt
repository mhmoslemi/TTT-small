import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def grad_row(h, idx):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    j1 = idx - lags
    j2 = idx + lags
    m1 = (j1 >= 0) & (j1 < n)
    m2 = (j2 >= 0) & (j2 < n)
    g = np.zeros(2 * n - 1)
    g[m1] += (1.0 - h[j1[m1]])
    g[m2] -= h[j2[m2]]
    return g

def one_d_min(c, g, lo, hi, n_evals=40):
    if hi - lo < 1e-12:
        return 0.0
    a, b = lo, hi
    for _ in range(n_evals):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(c + m1 * g) < np.max(c + m2 * g):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def coord_descent(h, target_sum, n_sweeps, n_evals, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)
        for idx in order:
            lo = -h[idx]
            hi = 1.0 - h[idx]
            if hi - lo < 1e-12:
                continue
            c = corr_full(h)
            g = grad_row(h, idx)
            delta = one_d_min(c, g, lo, hi, n_evals)
            if abs(delta) > 1e-10:
                h_new = h.copy()
                h_new[idx] += delta
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
        if not improved and sweep > 5:
            break
    return h, best_c5

def pairwise_cd(h, target_sum, n_iters, n_evals, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h)
    for itr in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        c = corr_full(h)
        gi = grad_row(h, i)
        gj = grad_row(h, j)
        dg = gi - gj
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if hi - lo < 1e-12:
            continue
        delta = one_d_min(c, dg, lo, hi, n_evals)
        if abs(delta) > 1e-11:
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick coordinate descent polish on initial
    if time_left() > 200:
        h, c5 = coord_descent(best_h, target_sum, 100, 40, seed=seed)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Try different n_points with diverse starts
    try_n_points = [60, 65, 70, 75, 80, 90, 100]
    candidates = []

    for np_ in try_n_points:
        if time_left() < 150:
            break
        ts = np_ / 2.0
        # Try a few random starts
        for attempt in range(3):
            if time_left() < 100:
                break
            if attempt == 0:
                # Smooth random
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                t = np.linspace(0, 2 * np.pi, np_)
                h_try = 0.5 + 0.3 * np.sin(freq * t + phase)
            elif attempt == 1:
                # Block structure
                nblocks = rng.randint(3, 8)
                h_try = np.zeros(np_)
                for b in range(nblocks):
                    s = int(b * np_ / nblocks)
                    e = int((b + 1) * np_ / nblocks)
                    h_try[s:e] = rng.uniform(0.1, 0.9)
            else:
                # Near-uniform with noise
                h_try = 0.5 + rng.randn(np_) * 0.1
            h_try = project_h(h_try, ts)
            # Quick coordinate descent
            h_res, c5_res = coord_descent(h_try, ts, 50, 30, seed=seed + attempt * 100 + np_)
            if c5_res < best_c5 - 1e-6:
                candidates.append((c5_res, h_res.copy()))
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    # Sort candidates and keep top few
    candidates.sort(key=lambda x: x[0])

    # Phase 3: Simulated annealing on best solution
    temperature = 0.04
    n_pts = len(best_h)
    ts = n_pts / 2.0
    stagnant = 0

    while time_left() > 80:
        h_pert = best_h + rng.randn(n_pts) * temperature
        h_pert = project_h(h_pert, ts)
        h_new, c5_new = coord_descent(h_pert, ts, 40, 35, seed=seed + int(time.time() * 7919) % 100000)
        if c5_new < best_c5 - 1e-10:
            best_h, best_c5 = h_new, c5_new
            stagnant = 0
            temperature = max(0.003, temperature * 0.85)
        else:
            stagnant += 1
            temperature = max(0.003, temperature * 0.93)

        # Also try pairwise CD periodically
        if stagnant % 3 == 0 and time_left() > 80:
            h_pw, c5_pw = pairwise_cd(best_h, ts, 2000, 30, seed=seed + stagnant)
            if c5_pw < best_c5:
                best_h, best_c5 = h_pw, c5_pw

    # Phase 4: High-precision final polish
    n_pts = len(best_h)
    ts = n_pts / 2.0
    if time_left() > 30:
        h, c5 = coord_descent(best_h, ts, 300, 80, seed=seed + 999)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Pairwise final polish
    if time_left() > 20:
        h, c5 = pairwise_cd(best_h, ts, 5000, 50, seed=seed + 7777)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # One more coordinate descent at very high precision
    if time_left() > 10:
        h, c5 = coord_descent(best_h, ts, 200, 100, seed=seed + 12345)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = project_h(best_h, ts)
    best_c5 = compute_c5(best_h)
    n_pts = len(best_h)
    return best_h, best_c5, n_pts
