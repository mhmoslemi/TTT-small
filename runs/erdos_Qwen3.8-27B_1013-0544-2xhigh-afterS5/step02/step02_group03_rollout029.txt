import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def compute_grad_smmax(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = np.clip(beta * (corr - M), -700.0, 700.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        g = np.ones(n)
        if s >= 0:
            v1 = np.arange(0, n - s)
            g[v1] -= h[v1 + s]
            v2 = np.arange(s, n)
            g[v2] -= h[v2 - s]
        else:
            la = -s
            v1 = np.arange(la, n)
            g[v1] -= h[v1 - la]
            v2 = np.arange(0, n - la)
            g[v2] -= h[v2 + la]
        grad += w[k] * g
    return grad, smooth_max_val(corr, beta)

def obj_and_grad(h, dx, beta):
    grad, val = compute_grad_smmax(h, beta)
    return val * dx, grad * dx

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    def tl():
        return budget_s - (time.time() - t0)
    
    # SLSQP with analytic gradient
    for beta in [10, 50, 200, 500, 2000, 5000, 10000]:
        if tl() < 30:
            break
        for trial in range(3):
            if tl() < 15:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * (0.03 / trial)
                x0 = project_feasible(x0, target_sum)
            try:
                cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(obj_and_grad, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    
    # Subgradient with adaptive step size

    shifted = np.clip(beta * (corr - M), -700.0, 700.0)
    w = np.exp(shifted)
    w /= w.sum()
    grad_h = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        grad_h += w[k] * grad_corr_shift(h, s)
    val = smooth_max_val(corr, beta)
    # Constraint: sum(h) = n/2, i.e., 2*sum(z) = n/2, sum(z) = n/4
    s_con = 2.0 * z.sum() - n / 2.0  # This is sum(h) - n/2
    penalty = (rho / 2.0) * s_con ** 2 + mu * s_con
    grad_h_con = (rho * s_con + mu) * np.ones(n)
    grad_z = sym_grad_to_z(grad_h + grad_h_con, n)
    # Wait, I need to be more careful with the constraint gradient in z-space
    # s_con = sum(h) - n/2 = 2*sum(z) - n/2
    # d s_con / dz[j] = 2 for all j
    # So the constraint gradient in z-space is: 2 * (rho * s_con + mu)
    # But grad_h already has the constraint gradient added...
    # Let me redo this properly.
    
    # Objective in z-space: f(z) = smooth_max(corr(sym_expand(z))) * dx
    # df/dz[j] = sum_i (df/dh[i]) * (dh[i]/dz[j])
    # dh[i]/dz[j] = 1 if i==j or i==n-1-j (and j < m)
    # So df/dz[j] = df/dh[j] + df/dh[n-1-j]
    
    # Total grad of objective w.r.t. z:
    grad_obj_z = sym_grad_to_z(grad_h * dx, n)
    
    # Gradient of constraint penalty w.r.t. z:
    # penalty = (rho/2)*(2*sum(z) - n/2)^2 + mu*(2*sum(z) - n/2)
    # d penalty/dz[j] = 2*(rho*(2*sum(z)-n/2) + mu) * 2 = 4*(rho*s_con + mu)
    # Wait: d/dz[j] (2*sum(z) - n/2) = 2
    # d/dz[j] (rho/2 * s_con^2) = rho * s_con * 2 = 2*rho*s_con
    # d/dz[j] (mu * s_con) = 2*mu
    # Total: 2*(rho*s_con + mu)
    grad_pen_z = 2.0 * (rho * s_con + mu) * np.ones(len(z))
    
    return val * dx + penalty, grad_obj_z + grad_pen_z

def con_eq_fun(x, target_sum):
    return x.sum() - target_sum

def con_eq_jac(x, n):
    return np.ones(n)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def to_sym_z(h, n):
    """Extract symmetric half from h (for even n)"""
    m = n // 2
    # Average the symmetric pairs to enforce symmetry
    z = 0.5 * (h[:m] + h[m:][::-1])
    return z
