    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_simplex_box(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = h.copy()
    for _ in range(50):  # bisection on lambda
        lo, hi = -2.0, 2.0
        for _ in range(60):
            mid = (lo + hi) / 2
            proj = np.clip(h - mid, 0, 1)
            s = proj.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        proj = np.clip(h - (lo+hi)/2, 0, 1)
        if abs(proj.sum() - target_sum) < 1e-12:

    return proj

def compute_subgradient(h, weights):
    """Compute subgradient of weighted sum of correlations.
    weights correspond to the lags in np.correlate(h, 1-h, mode='full') output."""
    n = len(h)
    # The full correlation has length 2n-1
    # Index m in the correlation corresponds to lag m - (n-1)
    # r(lag) = sum_i h[i+lag] * (1-h[i]) for valid i
    # dr(lag)/dh[j] = (1-h[j-lag]) if j-lag valid, minus h[j+lag] if j+lag valid
    
    g = np.zeros(n)
    lags = np.arange(-(n-1), n)  # all possible lags
    
    for idx, lag in enumerate(lags):
        w = weights[idx]
        if w < 1e-15:
            continue

        # and in (1-h[i]) when i = j, valid if 0 <= j < n, contributing -h[j+lag] if valid

            i = j - lag
            if 0 <= i < n:
                term1 = 1.0 - h[i]
            term2 = 0.0
            i2 = j + lag
            if 0 <= i2 < n:
                term2 = h[i2]
            g[j] += w * (term1 - term2)
    
    return g

def compute_subgradient_fast(h, weights):
    n = len(h)
    g = np.zeros(n)
    lags = np.arange(-(n-1), n)
    
    for idx, lag in enumerate(lags):
        w = weights[idx]
        if w < 1e-15:
            continue
        # term1: for j in range(n), if j-lag in [0,n): 1 - h[j-lag]
        # term2: for j in range(n), if j+lag in [0,n): h[j+lag]
        
        if lag >= 0:
            # j-lag >= 0 means j >= lag
            # j-lag < n means j < n+lag (always true since j < n)
            term1 = np.zeros(n)
            term1[lag:] = 1.0 - h[:n-lag]
            # j+lag < n means j < n-lag
            term2 = np.zeros(n)
            term2[:n-lag] = h[lag:]
        else:
            # lag < 0, let la = -lag > 0
            la = -lag
            # j-lag = j+la >= 0 (always), j+la < n means j < n-la
            term1 = np.zeros(n)
            term1[:n-la] = 1.0 - h[la:]
            # j+lag = j-la >= 0 means j >= la, j-la < n (always)
            term2 = np.zeros(n)
            term2[la:] = h[:n-la]
        
        g += w * (term1 - term2)
    
    return g

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, corr=None):
    n = len(h)
    dx = 2.0 / n
    if corr is None:
        corr = compute_corr(h)
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_onto_constraint(h, target_sum):
    """Project onto {0<=h<=1, sum=h=target_sum} using bisection"""
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def subgradient_smooth(h, beta):
    """Gradient of smooth_max(correlations, beta)"""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # softmax weights
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    
    g = np.zeros(n)
    lags = np.arange(-(n-1), n)
    
    for idx, lag in enumerate(lags):
        wi = w[idx]
        if wi < 1e-10:
            continue
        if lag >= 0:
            term1 = np.zeros(n)
            if n - lag > 0:
                term1[lag:] = 1.0 - h[:n-lag]
            term2 = np.zeros(n)
            if n - lag > 0:
                term2[:n-lag] = h[lag:]
        else:
            la = -lag
            term1 = np.zeros(n)
            term1[:n-la] = 1.0 - h[la:]
            term2 = np.zeros(n)
            term2[la:] = h[:n-la]
        g += wi * (term1 - term2)
    
    return g, corr

    best_c5 = compute_c5(best_h)

    h_bin = project_onto_constraint(h_bin, target_sum)
    c5_bin = compute_c5(h_bin)
    
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    
    # Local search on binary pattern
    h_ls = best_h.copy()
    improved = True
    temperature = 0.05
    while improved and (time.time() - start_time) < budget_s * 0.15:
        improved = False
        c5_curr = compute_c5(h_ls)
        np.random.shuffle(range(n))
        for j in range(n):
            if (time.time() - start_time) > budget_s * 0.15:
                break
            # Try flipping
            h_try = h_ls.copy()
            h_try[j] = 1.0 - h_try[j]
            h_try = project_onto_constraint(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr - 1e-10:
                h_ls = h_try.copy()
                c5_curr = c5_try

                if c5_curr < best_c5:
                    best_h = h_ls.copy()
                    best_c5 = c5_curr
            elif np.random.random() < np.exp((c5_curr - c5_try) / temperature):

                c5_curr = c5_try
                improved = True
                if c5_curr < best_c5:

                    best_c5 = c5_curr

    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

    """Objective and gradient for smooth max of correlations"""
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    obj = smooth_max_val(corr, beta)

        else:
            la = -lag
            k = n - la
            g[:k] += wi * (1.0 - h[la:])
            g[la:] -= wi * h[:k]
    
    return obj, g, corr, w

Let me also add one more optimization technique: **coordinate descent**. For each coordinate j, we can find the optimal value of h[j] (given all other coordinates fixed) by solving a 1D problem. This is because the objective as a function of h[j] alone is a smooth max of linear functions in h[j] (since each correlation is linear in h[j] when other coordinates are fixed... wait, no, it's not linear because of the h[j]·h[i] cross terms).

Actually, r(ℓ) = Σ h[i+ℓ](1-h[i]). If we fix all h[i] except h[j], then:
- r(ℓ) as a function of h[j]:
  - h[j] appears as h[i+ℓ] when i = j-ℓ: term is h[j]·(1-h[j-ℓ])
  - h[j] appears in (1-h[i]) when i = j: term is h[j+ℓ]·(1-h[j])
  
So r(ℓ) = [h[j]·(1-h[j-ℓ]) if valid] + [h[j+ℓ]·(1-h[j]) if valid] + [constant]

This is a quadratic in h[j]! Specifically:
r(ℓ) = a(ℓ)·h[j] + b(ℓ)·h[j]² + c(ℓ)

where:
- If j-ℓ valid: a += (1-h[j-ℓ]), b += 0 (it's linear in h[j])
  Wait, h[j]·(1-h[j-ℓ]) is linear in h[j] (since h[j-ℓ] is fixed).
- If j+ℓ valid: h[j+ℓ]·(1-h[j]) = h[j+ℓ] - h[j+ℓ]·h[j], which is linear in h[j] with coefficient -h[j+ℓ].

So r(ℓ) is actually LINEAR in h[j] (given other coordinates fixed)! Let me recheck:
r(ℓ) = Σ_i h[i+ℓ]·(1-h[i])

Terms involving h[j]:
1. When i+ℓ = j, i.e., i = j-ℓ (if valid): h[j]·(1-h[j-ℓ]) → linear in h[j]
2. When i = j (if valid): h[j+ℓ]·(1-h[j]) = h[j+ℓ] - h[j+ℓ]·h[j] → linear in h[j]

Other terms don't involve h[j].

So r(ℓ) = h[j]·(1-h[j-ℓ])·1_{valid} + h[j+ℓ]·(1-h[j])·1_{valid} + const
         = h[j]·[(1-h[j-ℓ])·1_{j-ℓ valid} - h[j+ℓ]·1_{j+ℓ valid}] + const

This is linear in h[j]! So the max of a set of linear functions in h[j] is a piecewise linear convex function in h[j]. We can find its minimum on [0,1] by checking the breakpoints.

This means coordinate descent is very efficient: for each j, the optimal h[j] is either 0, 1, or a breakpoint where two lines cross.

Even better: since we're minimizing a convex piecewise-linear function of h[j] on [0,1], the minimum is at a breakpoint or at the boundary. The breakpoints occur where two of the linear functions r(ℓ) are equal.

But there are O(n) lags, so O(n) breakpoints per coordinate, and O(n²) total for all coordinates. For n=70, that's about 5000 operations per full coordinate descent sweep, which is very fast.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_PARAMS = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, corr=None):
    n = len(h)
    dx = 2.0 / n
    if corr is None:
        corr = compute_corr(h)
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def obj_grad_smmax(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    g = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    for idx in range(len(lags)):
        wi = w[idx]
        if wi < 1e-9:
            continue
        lag = lags[idx]
        if lag >= 0:
            k = n - lag
            if k > 0:
                g[lag:] += wi * (1.0 - h[:k])
                g[:k] -= wi * h[lag:]
        else:
            la = -lag
            k = n - la
            g[:k] += wi * (1.0 - h[la:])
            g[la:] -= wi * h[:k]
    return smooth_max_val(corr, beta), g, corr, w

        h_try = project_feasible(h_try, ts)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur or rng.random() < np.exp(-(c5_try - c5_cur) / temp):
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best = h_cur.copy()
                best_c5 = c5_cur
        temp *= 0.997
    return best, best_c5

def subgrad_loop(h0, n, ts, t_limit, rng, init_beta, init_alpha):
    h = h0.copy()
    c5 = compute_c5(h)
    obj_best = smooth_max_val(compute_corr(h), init_beta)
    alpha = init_alpha

    no_imp = 0
