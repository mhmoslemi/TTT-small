```python
        # Overlap at lag k: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # To reduce this, we want to move h[i] down where h[i+k] is up,
        # and move h[i+k] up where h[i] is up.
        # Gradient descent direction: -dO/dh
        k = abs(worst_lag)
        if k == 0:
            # O = sum h[i](1-h[i]), gradient = 1-2h[i]
            # To minimize, decrease O by moving h in direction (2h-1)
            # But we need to maintain sum constraint
            direction = 2 * best_h - 1
        elif worst_lag > 0:
            # O = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # dO/dh[j] = (1-h[j+k]) for j in [0, n-1-k] (j is in first factor)
            #           -h[j-k] for j in [k, n-1] (j is in second factor)
            direction = np.zeros(n)
            for j in range(n - k):
                direction[j] += (1 - best_h[j + k])
            for j in range(k, n):
                direction[j] -= best_h[j - k]
        else:

            for j in range(k2, n):
                direction[j] -= best_h[j - k2]

        direction -= direction.mean()
```

```python
def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)
    w /= w.sum()
    
    # w has length 2n-1, indexed 0 to 2n-2
    # w[idx] corresponds to lag = idx - (n-1)
    # By symmetry, w[idx] = w[2(n-1)-idx]
    # We only need lags 0 to n-1: indices n-1 to 2n-2
    # But by symmetry, we can just use the positive lag weights
    
    # For efficiency, compute using the positive lags only
    # Positive lags are at indices n-1+k for k=0,...,n-1
    pos_w = w[n-1:]  # weights for lags 0, 1, ..., n-1
    # But we also need the negative lags which have the same weights
    # Actually since full correlation is symmetric, c[idx] = c[2n-2-idx]
    # So w[idx] = w[2n-2-idx]
    # The positive lags (idx = n-1+k) and negative lags (idx = n-1-k) have same weight
    # So the total weight for "lag magnitude k" is w[n-1+k] + w[n-1-k] for k>0, and w[n-1] for k=0
    
    # Gradient computation:
    grad = np.zeros(n)
    for k in range(n):
        # Weight for lag k (combining positive and negative)
        if k == 0:
            wk = w[n-1]
        else:
            wk = w[n-1+k] + w[n-1-k]
        
        # dc_k/dh[j] for lag k:
        # For j in [0, n-1-k]: (1-h[j+k])
        # For j in [k, n-1]: -h[j-k]
        if k == 0:
            # dc_0/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
            grad += wk * dx * (1 - 2*h)
        else:
            grad[:n-k] += wk * dx * (1 - h[k:])
            grad[k:] += wk * dx * (-h[:-k])
    
    return grad
```
