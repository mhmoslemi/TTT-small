import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def quick_project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, dx, beta, target_sum):
    h_c = np.clip(h, 0, 1)
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    pen = 100.0 * (h_c.sum() - target_sum) ** 2
    return smooth_max(corr, beta) * dx + pen

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def coord_descent(h, target_sum, max_sweeps=50):
    n = len(h)
    best_c5 = compute_c5(h)
    deltas = [0.005, 0.01, 0.02, 0.04]
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for delta in deltas:
                h_trial = h.copy()
                h_trial[i] = min(1.0, h[i] + delta)
                h_trial = quick_project(h_trial, target_sum)
                c5 = compute_c5(h_trial)
                if c5 < best_c5 - 1e-13:
                    h = h_trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                continue
            for delta in deltas:
                h_trial = h.copy()
                h_trial[i] = max(0.0, h[i] - delta)
                h_trial = quick_project(h_trial, target_sum)
                c5 = compute_c5(h_trial)
                if c5 < best_c5 - 1e-13:
                    h = h_trial
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def slsqp_round(h_start, n, t0, budget_s, betas=None, max_iter=8000):
    if betas is None:
        betas = [10, 100, 1000, 10000]
    if budget_s - (time.time() - t0) < 20:
        return h_start, compute_c5(h_start)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_cur = project_feasible(h_start.copy(), target_sum)
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if budget_s - (time.time() - t0) < 15:
            break
        for trial in range(2):
            if budget_s - (time.time() - t0) < 10:
                break
            x0 = h_cur.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.004
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta, target_sum),
                             method='SLSQP', bounds=bounds,
                             options={'maxiter': max_iter, 'ftol': 1e-15})
                h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
                c5 = compute_c5(h_new)
                if c5 < c5_cur - 1e-10:
                    h_cur = h_new
                    c5_cur = c5
            except Exception:
                pass
    return h_cur, c5_cur

def pairwise_refine(h, target_sum, max_sweeps=5):
    n = len(h)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n - 1):
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-5:
                    continue
                h_trial = h.copy()
                h_trial[i], h_trial[j] = h[j], h[i]
                c5 = compute_c5(h_trial)
                if c5 < best_c5 - 1e-12:
                    h = h_trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5
