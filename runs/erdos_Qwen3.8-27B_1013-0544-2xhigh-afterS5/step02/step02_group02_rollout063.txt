from scipy.optimize import minimize, differential_evolution

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def g_to_h(g):
    return np.concatenate([g, 1.0 - g])

def ap_project(h, n):
    half = n // 2
    return (h[:half] + (1.0 - h[half:])) / 2.0

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def obj_g(g, alpha):
    h = g_to_h(g)
    return obj_smooth(h, len(h), alpha)

def grad_g(g, alpha):
    n = len(g)
    eps = 1e-8
    grad = np.zeros(n)
    f0 = obj_g(g, alpha)
    for i in range(n):
        if g[i] <= eps:
            gp = g.copy()
            gp[i] = g[i] + eps
            grad[i] = (obj_g(gp, alpha) - f0) / eps
        elif g[i] >= 1.0 - eps:
            gm = g.copy()
            gm[i] = g[i] - eps
            grad[i] = (f0 - obj_g(gm, alpha)) / eps
        else:
            gp = g.copy()
            gm = g.copy()
            gp[i] = g[i] + eps
            gm[i] = g[i] - eps
            grad[i] = (obj_g(gp, alpha) - obj_g(gm, alpha)) / (2.0 * eps)
    return grad

                      bounds=bounds,

                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_g(g, alpha, maxiter):
    bounds = [(0.0, 1.0)] * len(g)

                      bounds=bounds, jac=grad_g,

    except Exception:
