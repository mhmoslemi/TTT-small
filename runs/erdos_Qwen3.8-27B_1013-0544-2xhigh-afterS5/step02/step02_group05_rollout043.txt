```python
def project_onto_simplex_box(x, S):
    """Project x onto {0 <= h <= 1, sum(h) = S}"""
    n = len(x)
    # Find tau such that sum(clamp(x - tau, 0, 1)) = S
    lo = x.min() - 1
    hi = x.max() + 1
    
    def f(tau):
        return np.sum(np.clip(x - tau, 0, 1)) - S
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(x - tau, 0, 1)
```

```python
    c = np.correlate(h, 1-h, mode="full")
    max_c = c.max()

    threshold = max_c - active_frac * (max_c - c.min())
    active = np.where(c >= threshold)[0]
    
    # Compute subgradient
    # Element m of correlation corresponds to lag k = m - (n-1)
    grad = np.zeros(n)
```

The left side is a quadratic form in h. Specifically, it's h^T A_k h where A_k is a symmetric matrix with 1/2 at positions (i, i+k) and (i+k, i).

Therefore, cvxpy cannot directly handle this. We'd need to use a non-convex solver or reformulate.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def exact_project(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def obj_smooth(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, float(beta)) * dx

def obj_penalty(h, n, beta, pen):
    ts = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, float(beta)) * dx + pen * (h.sum() - ts) ** 2

def const_eq(h, n):
    return h.sum() - n / 2.0

def const_eq_jac(h, n):
    return np.ones(n)

def optimize_at_n(n, h_start, t_start, t_limit, rng):
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_curr = exact_project(h_start, ts)
    c_curr = compute_c5(h_curr)
    constraint = {'type': 'eq', 'fun': const_eq, 'jac': const_eq_jac, 'args': (n,)}

    # Phase A: SLSQP with increasing beta
    for beta in [30, 60, 120, 250, 500]:
        if time.time() - t_start > t_limit * 0.5:
            break
        try:
            res = minimize(obj_smooth, h_curr, args=(n, beta), method='SLSQP',
                           bounds=bounds, constraints=[constraint],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = exact_project(res.x, ts)
            c_new = compute_c5(h_new)
            if c_new < c_curr:
                h_curr, c_curr = h_new.copy(), c_new
        except Exception:
            pass

    # Phase B: L-BFGS-B with penalty (no explicit constraint)
    if time.time() - t_start < t_limit * 0.7:
        for pen in [200.0, 5000.0]:

            try:

                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 10000, 'ftol': 1e-20, 'gtol': 1e-14})
                h_new = exact_project(res.x, ts)

    best_h = h_init.copy()

    # Phase 1: Optimize at initial resolution (generous time)
    t1 = min(250.0, budget_s * 0.3)

    elapsed = time.time() - t_start

        n_f = best_n
        ts_f = n_f / 2.0

                if c_new < best_c5:

            except Exception:
```
