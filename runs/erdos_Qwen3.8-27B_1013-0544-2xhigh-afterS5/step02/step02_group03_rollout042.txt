import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def slsqp_optimize(h0, n, beta, maxiter=5000, ftol=1e-15):
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, beta) * dx
    cons = [{'type': 'eq', 'fun': lambda h: h.sum() - target_sum,
            'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0.copy()

def pair_swap_local_search(h, n, max_rounds=500, budget_check=None):
    """Local search using pair swaps along the worst lag."""
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if budget_check is not None and budget_check() < 15:
            break
        
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        k_worst = np.argmax(corr)
        lag = abs(k_worst - (n - 1))
        
        if lag == 0:
            # Self-overlap: h[i]*(1-h[i]), try to make h more binary
            # Swap between a high and low value
            i = np.argmax(best_h)
            j = np.argmin(best_h)
            pairs = [(i, j)]
        else:
            pairs = [(i, i + lag) for i in range(n - lag)]
            if k_worst < n - 1:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
