import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(h, n, alpha):
    """Log-sum-exp smooth max of correlation * dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_sum_con(h, n):
    return float(np.sum(h) - n / 2.0)

def obj_sum_jac(h, n):
    return np.ones(n)

def lbbf_optimize(h, n, alpha, maxiter, ftol):
    """L-BFGS-B optimization with penalty for sum constraint."""
    def penalized_obj(v, penalty):
        c = obj_lse(v, n, alpha)
        s_con = np.sum(v) - n / 2.0
        return c + penalty * s_con * s_con
    
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for penalty in [10, 100, 1000, 10000]:
        res = minimize(penalized_obj, best_h, args=(penalty,), method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, n)
        c = compute_c5(h_new, n)
        if c < best_c5:
            best_h = h_new
            best_c5 = c
    
    return best_h, best_c5

    def obj(v):
        return obj_lse(v, n, alpha)
    
    cons = [{'type': 'eq', 'fun': obj_sum_con, 'jac': obj_sum_jac, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    
    try:

                      options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def delta_corr(h, n, i0, j0, d):
    """First-order change in correlation vector for h[i0]+=d, h[j0]-=d."""
    nc = 2 * n - 1
    delta = np.zeros(nc)
    
    # h[i0] += d:
    # Case B (first factor): j in [i0, i0+n-1], delta += d*(1-h[i0-j+n-1])
    jb_start = i0
    jb_end = min(i0 + n - 1, nc - 1)
    if jb_start <= jb_end:
        js = np.arange(jb_start, jb_end + 1)
        idx = i0 - js + n - 1
        delta[jb_start:jb_end+1] += d * (1.0 - h[idx])
    # Case A (second factor): j in [n-1-i0, 2n-2-i0], delta += -d*h[i0+j-n+1]
    ja_start = n - 1 - i0
    ja_end = min(2 * n - 2 - i0, nc - 1)
    if ja_start <= ja_end:
        js = np.arange(ja_start, ja_end + 1)
        idx = i0 + js - n + 1
        delta[ja_start:ja_end+1] -= d * h[idx]
    
    # h[j0] -= d (i.e., h[j0] changes by -d):
    # Case B: j in [j0, j0+n-1], delta += (-d)*(1-h[j0-j+n-1])
    jb_start = j0
    jb_end = min(j0 + n - 1, nc - 1)

        js = np.arange(jb_start, jb_end + 1)
        idx = j0 - js + n - 1
        delta[jb_start:jb_end+1] += (-d) * (1.0 - h[idx])
    # Case A: j in [n-1-j0, 2n-2-j0], delta += -(-d)*h[j0+j-n+1] = d*h[j0+j-n+1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_opt(h, n, alpha, maxiter=2000, ftol=1e-16):
    def obj(v):
        return obj_lse(v, n, alpha)
    def con_fun(v):
        return float(np.sum(v) - n / 2.0)
    def con_jac(v):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except:
        return h

def fast_local_search(h, n, t0, budget_s, rng, min_remaining=30):
    """Fast hill-climbing with incremental correlation updates."""
    dx = 2.0 / n
    nc = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full')  # unscaled
    current_c5 = float(np.max(corr))  # unscaled max
    best_c5 = current_c5
    best_h = h.copy()
    
    # Step size schedule
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for step_size in step_sizes:
        if budget_s - (time.time() - t0) < min_remaining:
            break
        n_stuck = 0
        max_stuck = 5000
        while n_stuck < max_stuck:
            if budget_s - (time.time() - t0) < min_remaining:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = step_size * rng.choice([1.0, -1.0])
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                n_stuck += 1
                continue
            
            # Compute delta corr
            delta = np.zeros(nc)
            # h[i] += d
            s1 = i
            e1 = min(i + n - 1, nc - 1)
            js = np.arange(s1, e1 + 1)
            delta[s1:e1+1] += d * (1.0 - h[i - js + n - 1])
            s2 = n - 1 - i
            e2 = min(2*n-2-i, nc-1)
            js = np.arange(s2, e2 + 1)
            delta[s2:e2+1] -= d * h[i + js - n + 1]
            # h[j] -= d
            s3 = j
            e3 = min(j + n - 1, nc - 1)
            js = np.arange(s3, e3 + 1)
            delta[s3:e3+1] += (-d) * (1.0 - h[j - js + n - 1])
            s4 = n - 1 - j
            e4 = min(2*n-2-j, nc-1)
            js = np.arange(s4, e4 + 1)
            delta[s4:e4+1] += d * h[j + js - n + 1]
            
            new_corr = corr + delta
            new_max = float(np.max(new_corr))
            
            if new_max < current_c5 - 1e-15:
                h[i] = hi
                h[j] = hj
                corr = new_corr
                current_c5 = new_max
                n_stuck = 0
                if new_max < best_c5:
                    best_c5 = new_max
                    best_h = h.copy()
            else:
                n_stuck += 1
    
    return best_h, best_c5 * dx
