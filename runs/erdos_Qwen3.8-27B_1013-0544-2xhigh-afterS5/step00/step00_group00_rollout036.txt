    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return np.max(compute_c5_vector(h))

def project_to_feasible(h, target_sum):
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def smooth_max(vals, tau):
    m = vals.max()
    return m + np.log(np.sum(np.exp(tau * (vals - m)))) / tau

def obj_func(x, n, target_sum, tau, penalty):
    c = compute_c5_vector(x)
    sm = smooth_max(c, tau)
    sv = x.sum() - target_sum
    return sm + penalty * sv * sv

    best_c = compute_c5(h)

        if time.time() - t0 > max_time or time.time() - start_time > budget_s - 60:
            break

                       method='L-BFGS-B', bounds=[(0, 1)] * n,
                       options={'maxiter': 200, 'ftol': 1e-15})

        c_new = compute_c5(h_new)

            best_h = h_new.copy()
        h = h_new.copy()
    return best_h, best_c
