import numpy as np
from scipy.optimize import minimize

def project_onto_set(g, target):
    """Project g onto {h : 0<=h_i<=1, sum(h)=target}"""
    lo = g.min() - target
    hi = g.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)

def get_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smax_grad(h, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*F_k)))"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    Fm = F - F.max()
    w = np.exp(beta * Fm)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:n])
            grad[s:] += -wk * h[:n-s]
        else:
            sp = -s
            grad[sp:] += wk * (1.0 - h[:n-sp])
            grad[:n-sp] += -wk * h[sp:]
    return dx * grad

import numpy as np
import time

def project_set(g, target):
    """Project g onto {h : 0<=h<=1, sum(h)=target} using bisection."""
    lo = g.min() - target
    hi = g.max()
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)

    """Compute C5 value for given h."""
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smax_grad(h, beta):
    """Gradient of soft-max approximation to C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    Fm = F - F.max()
    w = np.exp(beta * Fm)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += -wk * h[s:]
        else:
            sp = -s
            grad[:n-sp] += wk * (1.0 - h[sp:])
            grad[sp:] += -wk * h[:n-sp]
    return dx * grad

def get_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    Fm = F - F.max()
    w = np.exp(beta * Fm)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += -wk * h[s:]
        else:
            sp = -s
            grad[:n-sp] += wk * (1.0 - h[sp:])
            grad[sp:] += -wk * h[:n-sp]
    return dx * grad
