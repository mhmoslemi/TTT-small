import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    # Distribute proportionally
    if diff > 0:
        # Need to increase sum
        # Add to elements not at 1
        mask = h < 1.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
    else:
        # Need to decrease sum
        mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def refine_slsqp(h0, n, maxiter=300):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 
             'jac': lambda x: np.ones(n)}]

                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    h = project_onto_feasible(res.x, n)
    return h, compute_c5(h)

    """Project h onto {h: 0<=h<=1, sum(h)=n/2} using bisection on threshold."""
    h = h.astype(float)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return np.clip(h, 0, 1)
    
    def sum_clipped(tau):
        return np.clip(h - tau, 0, 1).sum()

    lo, hi = -1.0, 2.0

        mid = (lo + hi) / 2
        if sum_clipped(mid) > target:
            lo = mid
        else:

Let me add one more strategy: a "smoothed" version of the problem where I replace max with a log-sum-exp, and use BFGS or L-BFGS-B on that. The log-sum-exp is smooth and differentiable, so gradient-based methods work well.

For the log-sum-exp:
g(h) = (1/β) * log(sum_k exp(β * c_k(h)))

where c_k(h) = dx * corr[k] and corr = np.correlate(h, 1-h, mode="full").

The gradient of g with respect to h[j] can be computed:
∂g/∂h[j] = sum_k (β * ∂c_k/∂h[j] * exp(β * c_k)) / (β * sum_k exp(β*c_k))
          = sum_k (∂c_k/∂h[j] * p_k)

where p_k = exp(β*c_k) / sum exp(β*c_k) (softmax probabilities).

And ∂c_k/∂h[j] = dx * (1 - h[j+shift_k]) - dx * h[j-shift_k] (approximately, from the correlation formula).

def compute_c5_smooth(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    # Log-sum-exp for numerical stability
    m = vals.max()
    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def refine_smooth(h0, n, beta=50.0, maxiter=300):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: compute_c5_smooth(x, beta), h0, method='SLSQP',
                   bounds=bnds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    h = project_feasible(res.x, n)
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-14:
        return np.clip(h, 0, 1)
    def sum_clipped(tau):
        return np.clip(h - tau, 0, 1).sum()
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def compute_c5_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = vals.max()
    return float(m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta)

def refine_slsqp(h0, n, maxiter=500):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(compute_c5, h0, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h = project_feasible(res.x, n)
        c5 = compute_c5(h)
        h0p = project_feasible(h0, n)
        c5_0 = compute_c5(h0p)
        if c5 < c5_0:
            return h, c5
        return h0p, c5_0
    except Exception:
        h0p = project_feasible(h0, n)
        return h0p, compute_c5(h0p)

def refine_smoothed(h0, n):
    """Progressively sharpen log-sum-exp approximation."""
    h = project_feasible(h0, n)
    for beta in [10.0, 30.0, 100.0, 300.0]:
        bnds = Bounds(np.zeros(n), np.ones(n))

        try:
            res = minimize(lambda x: compute_c5_smooth(x, beta), h, method='SLSQP',
                           bounds=bnds, constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    return h, compute_c5(h)
