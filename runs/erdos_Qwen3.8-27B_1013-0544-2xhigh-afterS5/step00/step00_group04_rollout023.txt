import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def greedy_single(h, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_imp, best_move = 1e-12, None
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                c5 = compute_c5(h)
                if best_c5 - c5 > best_imp:
                    best_imp = best_c5 - c5
                    best_move = (i, j, c5)
                h[j] = 0
            h[i] = 1
        if best_move is None:
            break
        i, j, nc5 = best_move
        h[i] = 0
        h[j] = 1
        best_c5 = nc5
    return h, best_c5

def greedy_2swap(h, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0].tolist()
        zeros = np.where(h == 0)[0].tolist()
        best_imp, best_move = 1e-12, None
        for a in range(len(ones)):
            for b in range(a + 1, len(ones)):
                ia, ib = ones[a], ones[b]
                h[ia] = 0
                h[ib] = 0
                for c in range(len(zeros)):
                    for d in range(c + 1, len(zeros)):
                        ja, jb = zeros[c], zeros[d]
                        h[ja] = 1
                        h[jb] = 1
                        c5 = compute_c5(h)
                        if best_c5 - c5 > best_imp:
                            best_imp = best_c5 - c5
                            best_move = (ia, ib, ja, jb, c5)
                        h[ja] = 0
                        h[jb] = 0
                h[ia] = 1
                h[ib] = 1
        if best_move is None:

        ia, ib, ja, jb, nc5 = best_move

    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t0 = time.time()
    while time.time() - t0 < time_limit:
        frac = min(1.0, (time.time() - t0) / time_limit)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i] = 0
        h[j] = 1
        nc5 = compute_c5(h)
        d = nc5 - cur_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-15)):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5
