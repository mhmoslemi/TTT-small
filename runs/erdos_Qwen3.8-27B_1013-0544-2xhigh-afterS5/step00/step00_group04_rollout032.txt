        d = nm - mc
        if d <= 0 or rng.random() < np.exp(-d / max(T, 1e-9)):
            mc = nm
            if nm < bmc:
                bmc = nm
                bx = x.copy()
        else:
            x[i] = 1; x[j] = 0
    return bx, bmc

def _bin(h, n):
    x = (h > 0.5).astype(np.float64)
    t = n // 2
    if x.sum() > t:
        for i in np.where(x == 1)[0]:
            if x.sum() == t:
                break
            x[i] = 0
    elif x.sum() < t:
        for i in np.where(x == 0)[0]:
            if x.sum() == t:
                break
            x[i] = 1
    return x

def _perturb(h, n, k, rng):
    x = h.copy()
    for _ in range(k):
        oi = np.where(x == 1)[0]
        oz = np.where(x == 0)[0]
        x[oi[rng.integers(len(oi))]] = 0
        x[oz[rng.integers(len(oz))]] = 1
    return x

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    h0 = None
    if 'initial_h_values' in globals():
        try:
            h0 = np.asarray(initial_h_values, dtype=float)
        except Exception:
            pass
    if h0 is None:
        h0 = np.zeros(42)
        h0[::2] = 1.0
        h0[-1] = 0.0
    n0 = len(h0)
    h = _bin(h0, n0)

    bh = h.copy()
    bm = _mc(bh)
    bn = n0
    bc = bm * 2.0 / bn

    # Phase 1: Hill climb initial
    h, m = _hill(h, n0, 3000)
    if m < bm:
        bh, bm, bc = h.copy(), m, m * 2.0 / n0

    # Phase 2: ILS at n0
    max_restarts = 300
    for r in range(max_restarts):
        if rem() < budget_s * 0.30:
            break
        if r % 3 == 2:
            hp = _bin(rng.random(n0) - 0.2, n0)
        else:
            k = min(5, 2 + (r % 4))
            hp = _perturb(bh[:n0], n0, k, rng)
        hs, ms = _sa(hp, n0, 15000, 4.0, 0.15, rng)
        hh, mh = _hill(hs, n0, 800)
        if mh < bm:
            bh, bm, bc = hh.copy(), mh, mh * 2.0 / n0

    # Phase 3: Upscale to 2*n0
    if rem() > budget_s * 0.25:
        n2 = n0 * 2
        h2 = np.repeat(bh[:n0], 2)
        if h2.sum() == n2 // 2:
            h2, m2 = _hill(h2, n2, 2000)
            c2 = m2 * 2.0 / n2
            if c2 < bc:
                bh, bm, bn, bc = h2.copy(), m2, n2, c2
            for r in range(150):
                if rem() < budget_s * 0.10:
                    break
                if r % 4 == 3:
                    hp = _bin(rng.random(bn) - 0.2, bn)
                else:
                    hp = _perturb(bh, bn, 3, rng)
                hs, ms = _sa(hp, bn, 10000, 4.0, 0.2, rng)
                hh, mh = _hill(hs, bn, 500)
                cn = mh * 2.0 / bn
                if cn < bc:
                    bh, bm, bn, bc = hh.copy(), mh, bn, cn

    # Phase 4: Try 4*n0
    if rem() > budget_s * 0.08:
        n4 = n0 * 4
        h4 = np.repeat(bh[:n0], 4)
        if h4.sum() == n4 // 2:
            h4, m4 = _hill(h4, n4, 800)
            c4 = m4 * 2.0 / n4
            if c4 < bc:
                bh, bn, bc = h4.copy(), n4, c4

    # Phase 5: Final polish
    if rem() > 10:
        hh, mh = _hill(bh, bn, 1000)
        if mh * 2.0 / bn < bc:
            bh, bc = hh, mh * 2.0 / bn

    return bh, float(bc), bn
