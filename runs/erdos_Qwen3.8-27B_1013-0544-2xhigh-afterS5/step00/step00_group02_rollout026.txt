import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    for _ in range(200):
        rem = target - h.sum()
        if abs(rem) < 1e-14:
            break
        if rem > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if len(free) == 0:
            break
        idx = free[np.random.randint(len(free))]
        adj = min(rem, 1.0 - h[idx]) if rem > 0 else max(rem, -h[idx])
        h[idx] += adj
    return h

def slsqp_optimize(h_start, n, target_sum, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return compute_c5(x)
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            h_new = fix_sum(res.x, target_sum)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h_start, compute_c5(h_start)

def to_binary(h, target_sum):
    n = len(h)
    h_b = (h >= 0.5).astype(float)
    diff = target_sum - h_b.sum()
    if abs(diff) < 1e-10:
        return h_b
    if abs(diff - 0.5) < 1e-10:
        idx = np.where(h_b == 1)[0]
        if len(idx) > 0:
            h_b[np.random.choice(idx)] = 0.5
    elif abs(diff + 0.5) < 1e-10:
        idx = np.where(h_b == 0)[0]
        if len(idx) > 0:
            h_b[np.random.choice(idx)] = 0.5
    else:
        h_b = fix_sum(h_b, target_sum)
    return h_b

def swap_search(h, target_sum, n_iters, t_anneal=0.0, t_min=0.001):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = list(np.where(h > 0.75)[0])
    zeros = list(np.where(h < 0.25)[0])
    mids = list(np.where((h >= 0.25) & (h <= 0.75))[0])
    
    n_improve = 0
    for it in range(n_iters):
        t = max(t_min, t_anneal * np.exp(-it / max(1, n_iters // 10)))
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = 0.0, 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                ones.remove(i); ones.append(j)
                zeros.remove(j); zeros.append(i)
                n_improve = 0
            elif c5_new < best_c5 + t:
                ones.remove(i); ones.append(j)
                zeros.remove(j); zeros.append(i)
            else:
                h[i], h[j] = 1.0, 0.0
        elif mids:
            i = mids[np.random.randint(len(mids))]
            old = h[i]
            h[i] = 1.0 - old
            h = fix_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                n_improve = 0
            else:
                h[i] = old
                h = fix_sum(h, target_sum)
                ones = list(np.where(h > 0.75)[0])
                zeros = list(np.where(h < 0.25)[0])
                mids = list(np.where((h >= 0.25) & (h <= 0.75))[0])
        if it % 10000 == 0 and n_improve > 50000:
            break
        n_improve += 1
    return best_h, best_c5
