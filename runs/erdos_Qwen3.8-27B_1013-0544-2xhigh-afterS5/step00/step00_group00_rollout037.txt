import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sum_exps = np.sum(exps)
    obj = m + np.log(sum_exps) / beta
    weights = exps / sum_exps
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            n_valid = n - s
            grad[:n_valid] += w * dx * (1.0 - h[s:s+n_valid])
            grad[s:] -= w * dx * h[:n_valid]
        else:
            t = -s
            n_valid = n - t
            grad[t:] += w * dx * (1.0 - h[:n_valid])
            grad[:n_valid] -= w * dx * h[t:]
    return obj, grad

def project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] += diff
        else:
            h[mask] += diff / mask.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(globals()['initial_h_values'], dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def rem():
        return budget_s - (time.time() - start_time)
    
    def try_optimize(h_init, n, target, beta, max_iter=500):
        def f_obj(h):
            return smooth_obj_and_grad(h, beta)[0]
        def f_grad(h):
            return smooth_obj_and_grad(h, beta)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        result = minimize(f_obj, h_init, jac=f_grad, method='SLSQP',
                         bounds=bounds, constraints=cons, maxiter=max_iter,
                         ftol=1e-14)
        return result
    
    # Phase 1: Annealing beta on original grid
    for beta in [30.0, 100.0, 300.0, 1000.0]:
        if rem() < 20:
            break
        res = try_optimize(best_h, n0, target0, beta)
        if hasattr(res, 'x'):
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()
    
    # Phase 2: Random restarts
    for _ in range(8):
        if rem() < 30:
            break
        noise = rng.normal(0, 0.15, n0)
        h_p = project_feasible(best_h + noise, target0)
        res = try_optimize(h_p, n0, target0, 300.0, max_iter=300)
        if hasattr(res, 'x'):
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()
    
    # Phase 3: Grid refinement
    for mult in [2, 4, 8]:
        n_new = n0 * mult
        if rem() < 90:
            break
        target = n_new / 2.0
        old_x = (np.arange(n0) + 0.5) * (2.0 / n0)
        new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        h_new = np.interp(new_x, old_x, best_h)
        h_new = project_feasible(h_new, target)
        for beta in [100.0, 1000.0]:
            if rem() < 30:
                break
            res = try_optimize(h_new, n_new, target, beta, max_iter=300)
            if hasattr(res, 'x'):
                c = compute_c5(res.x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res.x.copy()
                    n0 = n_new
                    old_x = new_x
    
    # Final: local polish with subgradient
    n = len(best_h)
    target = n / 2.0
    h = best_h.copy()
    step = 0.02

        h = project_feasible(h - step * g, target)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        step *= 0.9999
    
    return best_h, best_c5, len(best_h)

            obj, _ = smooth_obj_and_grad(h, beta)
            return obj + lam * (np.sum(h) - target) ** 2
        def fg(h):
            _, g = smooth_obj_and_grad(h, beta)
            return g + 2.0 * lam * (np.sum(h) - target)
        bounds = [(0.0, 1.0)] * n
        return minimize(fo, h_init, jac=fg, method='L-BFGS-B',
                       bounds=bounds, maxiter=max_iter, ftol=1e-15)

    # Phase 1: Annealing beta with SLSQP on original grid
    for beta in [20.0, 50.0, 150.0, 500.0]:
        if rem() < 15:
            break
        res = optimize_slqp(best_h, n0, target0, beta, 400)
        if hasattr(res, 'x'):
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()

    # Phase 2: L-BFGS-B polish
    if rem() > 30:
        res = optimize_lbfgs(best_h, n0, target0, 500.0, 2000)
        if hasattr(res, 'x'):
            hx = project_feasible(res.x, target0)
            c = compute_c5(hx)
            if c < best_c5:
                best_c5 = c
                best_h = hx

    # Phase 3: Random restarts on original grid
    for i in range(10):
        if rem() < 20:
            break
        scale = 0.2 * (1.0 - 0.5 * i / 10.0)
        h_p = project_feasible(best_h + rng.normal(0, scale, n0), target0)
        res = optimize_slqp(h_p, n0, target0, 300.0, 200)
        if hasattr(res, 'x'):
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()

    # Phase 4: Try alternative starting structures
    if rem() > 60:
        # Alternating pattern
        h_alt = np.zeros(n0)
        h_alt[::2] = 1.0
        h_alt = project_feasible(h_alt, target0)
        res = optimize_slqp(h_alt, n0, target0, 200.0, 300)
        if hasattr(res, 'x'):
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()

    # Phase 5: Grid refinement
    n_cur = len(best_h)
    for mult in [2, 4]:
        if rem() < 90:
            break
        n_new = n_cur * mult
        target = n_new / 2.0
        old_x = (np.arange(n_cur) + 0.5) * (2.0 / n_cur)
        new_x = (np.arange(n_new) + 0.5) * (2.0 / n_new)
        h_new = np.interp(new_x, old_x, best_h)
        h_new = project_feasible(h_new, target)
        for beta in [50.0, 500.0]:
            if rem() < 30:
                break
            res = optimize_slqp(h_new, n_new, target, beta, 300)
            if hasattr(res, 'x'):
                c = compute_c5(res.x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res.x.copy()
                    n_cur = n_new

    # Phase 6: Subgradient descent polish
    n = len(best_h)
    target = n / 2.0
    h = best_h.copy()
    step = 0.05
    for it in range(30000):
        if rem() < 1:
            break
        _, g = smooth_obj_and_grad(h, 800.0)
        h = project_feasible(h - step * g, target)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
        step *= 0.9999
        if step < 1e-5:
            break

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
