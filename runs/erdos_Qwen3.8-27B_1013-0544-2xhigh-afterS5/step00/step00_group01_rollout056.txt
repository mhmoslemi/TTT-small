    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

def compute_smooth_gradient(h, n, T):
    """Compute gradient of smooth approximation of C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    # Softmax for numerical stability
    scaled_max = np.max(scaled)
    exp_scaled = np.exp(scaled - scaled_max)
    p = exp_scaled / np.sum(exp_scaled)
    
    # Compute gradient: sum_k p[k] * G[k, :]
    # G[k, j] = d(corr[k])/d(h[j]) where lag = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lag = k - (n - 1)
        # First term: (1-h[j-lag]) for valid j
        # j-lag must be in [0, n-1] and j in [0, n-1]
        j_start = max(0, lag)
        j_end = min(n, n + lag)  # j-lag < n => j < n+lag
        for j in range(j_start, j_end):
            if 0 <= j - lag < n:
                grad[j] += p[k] * (1.0 - h[j - lag])
        # Second term: -h[j+lag] for valid j
        # j+lag must be in [0, n-1] and j in [0, n-1]
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)  # j+lag < n => j < n-lag
        for j in range(j_start2, j_end2):
            if 0 <= j + lag < n:
                grad[j] += p[k] * (-h[j + lag])
    
    return dx * grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    smax = np.max(scaled)
    exp_s = np.exp(scaled - smax)
    p = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += pk * (1.0 - 2.0 * h)
            else:
                grad[lag:n] += pk * (1.0 - h[0:n-lag])
                grad[0:n-lag] += -pk * h[lag:n]
        else:
            if lag == 0:
                pass
            else:
                grad[0:n+lag] += pk * (1.0 - h[-lag:n])
                grad[-lag:n] += -pk * h[0:n+lag]
    
    return dx * grad
