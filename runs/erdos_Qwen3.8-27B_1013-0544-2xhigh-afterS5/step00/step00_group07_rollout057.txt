import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def obj_func(x):
    return float(np.max(np.correlate(x, 1.0 - x, mode="full")))

def sum_constraint(x):
    return x.sum() - len(x) * 0.5

def coord_desc_step(h, i, c0, c1):
    a = (c1 - c0).astype(np.float64)
    b = c0.astype(np.float64)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_desc(h, target_sum, n_iter=300):
    n = len(h)
    h = h.copy()
    best = compute_c5(h)
    for it in range(n_iter):
        improved = False
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = old
            x_opt = coord_desc_step(h, i, c0, c1)
            h[i] = x_opt
            h = project_feasible(h, target_sum)
            c5 = compute_c5(h)
            if c5 < best - 1e-13:
                best = c5
                improved = True
            else:
                h[i] = old
                h = project_feasible(h, target_sum)
        if not improved:
            break
    return h, best
