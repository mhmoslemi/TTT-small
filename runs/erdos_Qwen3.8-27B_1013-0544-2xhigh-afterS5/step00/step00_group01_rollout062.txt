```python
import time
from scipy.interpolate import interp1d
```

```python
        proj = np.clip(h - mid, 0.0, 1.0)

        g[:n-s] -= h[s:]
```

```python
def optimize_mw(h_init, time_limit, seed=42):
    """Multiplicative weights / mirror descent for minimax."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    
    # Distribution over shifts (2n-1 possible shifts)
    n_shifts = 2 * n - 1
    dist = np.ones(n_shifts) / n_shifts
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    t0 = time.time()
    alpha = 0.01
    eta = 0.5  # learning rate for distribution update
    
    while time.time() - t0 < time_limit:
        # Sample a shift
        k = rng.choice(n_shifts, p=dist)
        s = k - (n - 1)
        
        # Compute gradient for this shift
        g = grad_shift(h, s, n)
        
        # Gradient step
        h_new = project_to_feasible(h - alpha * g, n)
        
        # Evaluate
        corr = np.correlate(h_new, 1.0 - h_new, mode="full")
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Update h
        h = h_new
        
        # Update distribution (multiplicative weights)
        # Increase weight for shifts with high correlation
        corr_vals = np.max(corr)  # this is the max, not what we want
        # Actually, we want to update based on all correlation values
        corr_all = np.correlate(h, 1.0 - h, mode="full")
        # Normalize to get "losses"
        losses = corr_all * dx
        # Multiplicative weights update
        dist *= np.exp(eta * losses / np.max(losses))
        dist /= np.sum(dist)
        
        # Decay step size
        alpha *= 0.999
    
    return best_h, best_c5
```
