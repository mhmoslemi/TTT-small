import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def coordinate_sweep(h, n_candidates=200):
    """One full coordinate descent sweep."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    changed = False
    
    for j in range(n):
        # Precompute current correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        # Compute K for each shift
        # corr[m] corresponds to shift s = m - (n-1)
        K = np.zeros(2*n - 1)
        for m in range(2*n - 1):
            s = m - (n - 1)
            if s == 0:
                continue  # handle separately
            k = 0.0
            if 0 <= j + s < n:
                k += (1.0 - h[j + s])
            if 0 <= j - s < n:
                k -= h[j - s]
            K[m] = k
        
        # Candidate values
        v_batch = np.linspace(0.0, 1.0, n_candidates)
        
        # Compute new correlations for each candidate
        # For s≠0: new_corr[m] = corr[m] + (v - h[j]) * K[m]
        # For s=0: new_corr[n-1] = corr[n-1] + (v-h[j])*(1-v-h[j])
        
        delta = v_batch[:, None] - h[j]  # shape (B, 1)
        new_corr = corr[None, :] + delta * K[None, :]  # shape (B, 2n-1)
        # Fix s=0 (index n-1)
        new_corr[:, n-1] = corr[n-1] + (v_batch - h[j]) * (1.0 - v_batch - h[j])
        
        objs = new_corr.max(axis=1) * dx
        best_idx = np.argmin(objs)
        best_v = v_batch[best_idx]
        
        if abs(best_v - h[j]) > 1e-10:
            h[j] = best_v
            changed = True
    
    return h, changed

def coordinate_descent(h, max_sweeps=1000, seed=42, time_limit=None):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time_limit and time.time() - start > time_limit:
            break
        # Random coordinate order
        order = rng.permutation(n)
        h_sweep = h.copy()
        for j in order:
            # ... same as coordinate_sweep but for single coordinate
            corr = np.correlate(h_sweep, 1.0 - h_sweep, mode="full")
            K = np.zeros(2*n - 1)
            for m in range(2*n - 1):
                s = m - (n - 1)
                if s == 0: continue
                k = 0.0
                if 0 <= j + s < n: k += (1.0 - h_sweep[j + s])
                if 0 <= j - s < n: k -= h_sweep[j - s]
                K[m] = k
            
            v_batch = np.linspace(0.0, 1.0, 200)
            delta = v_batch[:, None] - h_sweep[j]
            new_corr = corr[None, :] + delta * K[None, :]
            new_corr[:, n-1] = corr[n-1] + (v_batch - h_sweep[j]) * (1.0 - v_batch - h_sweep[j])
            objs = new_corr.max(axis=1)
            best_v = v_batch[np.argmin(objs)]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sweep.copy()
            h = h_sweep.copy()
        else:
            # No improvement, maybe stop or try perturbation
            if c5 >= best_c5 - 1e-10:
                # Try a random perturbation
                pass
    
    return best_h, best_c5
