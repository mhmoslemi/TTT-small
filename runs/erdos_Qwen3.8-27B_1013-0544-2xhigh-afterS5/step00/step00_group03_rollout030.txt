import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def repair_binary(h, target_sum):
    # Round to binary while preserving sum
    h = np.round(h)
    h = h.astype(int)
    current_sum = h.sum()
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        excess = current_sum - target_sum
        ones = np.where(h == 1)[0]
        # Flip random ones
        idx = np.random.choice(ones, excess, replace=False)
        h[idx] = 0
    elif current_sum < target_sum:
        deficit = target_sum - current_sum
        zeros = np.where(h == 0)[0]
        idx = np.random.choice(zeros, deficit, replace=False)
        h[idx] = 1
    return h.astype(float)

def local_search_binary(h, dx, target_sum, max_iter=10000):
    # Hill climbing with swap moves
    n = len(h)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        i = h == 1
        j = h == 0
        if len(i) == 0 or len(j) == 0:
            break
        i_idx = np.random.choice(i)
        j_idx = np.random.choice(j)
        
        # Try swap
        h[i_idx] = 0
        h[j_idx] = 1
        c5 = compute_c5(h, dx)
        if c5 > best_c5:
            # Revert
            h[i_idx] = 1
            h[j_idx] = 0
        else:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def simulated_annealing(h, dx, target_sum, start_temp=0.01, end_temp=0.001, max_iter=50000):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    current_c5 = best_c5
    current_h = h.copy()
    
    for iteration in range(max_iter):
        temp = start_temp * (end_temp / start_temp) ** (iteration / max_iter)
        
        i = current_h == 1
        j = current_h == 0
        if len(i) == 0 or len(j) == 0:
            break
        i_idx = np.random.choice(i)
        j_idx = np.random.choice(j)
        
        new_h = current_h.copy()
        new_h[i_idx] = 0
        new_h[j_idx] = 1
        new_c5 = compute_c5(new_h, dx)
        
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def to_binary(h, target_sum):
    """Convert to binary vector with exact sum."""
    n = len(h)
    h_bin = np.zeros(n, dtype=float)
    # Take the top target_sum values
    indices = np.argsort(h)[::-1][:target_sum]
    h_bin[indices] = 1.0
    return h_bin

def random_binary(n, target_sum):
    h = np.zeros(n)
    indices = np.random.choice(n, target_sum, replace=False)
    h[indices] = 1.0
    return h

def swap_eval(h, i, j, dx):
    """Evaluate c5 after swapping h[i]=1,h[j]=0 to h[i]=0,h[j]=1."""
    h[i] = 0
    h[j] = 1
    c5 = compute_c5(h, dx)
    h[i] = 1
    h[j] = 0
    return c5

def local_search(h, dx, target_sum, max_iter=50000, time_limit=None, start_time=None):
    """Hill climbing with random swap moves."""
    n = len(h)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for _ in range(max_iter):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        if not ones or not zeros:
            break
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        new_c5 = swap_eval(h, i, j, dx)
        if new_c5 <= best_c5:
            # Accept
            h[i] = 0; h[j] = 1
            ones.discard(i); zeros.discard(j)
            ones.add(j); zeros.add(i)
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5

def simulated_annealing(h, dx, target_sum, max_iter=100000, t_start=0.05, t_end=0.001, time_limit=None, start_time=None):
    """Simulated annealing with swap moves."""
    n = len(h)
    cur_c5 = compute_c5(h, dx)
    cur_h = h.copy()
    best_c5 = cur_c5
    best_h = cur_h.copy()
    ones = set(np.where(cur_h == 1)[0].tolist())
    zeros = set(np.where(cur_h == 0)[0].tolist())

        if not ones or not zeros:
            break

        new_c5 = compute_c5(new_h, dx)

            cur_h = new_h
            cur_c5 = new_c5
            ones.discard(i); zeros.discard(j)
            ones.add(j); zeros.add(i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    
    return best_h, best_c5

            zeros_p = np.where(hp == 0)[0].tolist()
            for _ in range(max(1, bn // 15)):
                ip = int(rng.integers(len(ones_p)))
                jp = int(rng.integers(len(zeros_p)))
                hp[ones_p[ip]] = 0.0; hp[zeros_p[jp]] = 1.0
                ones_p[ip] = zeros_p[jp]; zeros_p[jp] = ones_p[ip] if False else zeros_p[jp]
                # fix the list swap properly
                tmp = ones_p[ip]; ones_p[ip] = zeros_p[jp]; zeros_p[jp] = tmp
            hs2, cs2 = sa(hp, dx, t0 + budget_s * 0.97, rng)
            hl2, cl2 = local_search(hs2.copy(), dx, t0 + budget_s * 0.99, rng)
            if cl2 < bc:
                bh, bc, bn = hl2, cl2, bn

    if bh is None:
        n_fb = 70
        bh = random_binary(n_fb, n_fb // 2, rng)
        bc = compute_c5(bh, 2.0 / n_fb)
        bn = n_fb

    return (bh, bc, bn)
