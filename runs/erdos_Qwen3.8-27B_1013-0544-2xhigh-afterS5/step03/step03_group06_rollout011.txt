import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_s = np.exp(x_shift)
    w = exp_s / np.sum(exp_s)
    val = m + np.log(np.sum(exp_s) / len(x)) / beta
    return val, w


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def full_gradient(h, beta):
    """Full gradient of smooth_max(corr, beta) * dx w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = smooth_max(corr, beta)
    grad = np.zeros(n)
    for s_idx in range(2 * n - 1):
        if w[s_idx] < 1e-14:
            continue
        s = s_idx - (n - 1)
        ws = w[s_idx]
        if s == 0:
            grad += ws * (1.0 - 2.0 * h)
        elif s > 0:
            nv = n - s
            grad[:nv] += ws * (1.0 - h[s:])
            grad[s:] -= ws * h[:nv]
        else:
            m = -s
            nv = n - m
            grad[:nv] -= ws * h[m:]
            grad[m:] += ws * (1.0 - h[:nv])
    return grad * dx


def smooth_obj_func(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max(corr, beta)
    return val * dx


def eq_constraint(h, target_sum):
    return h.sum() - target_sum


def eq_constraint_jac(h, target_sum):
    return np.ones(len(h))


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)

    # Multi-resolution SLSQP with full gradient
    if len(best_h) <= 250:
        resolutions = [80, 120, 160, 200, 240, 280, 320]
    else:
        resolutions = [200, 240, 280, 320, 360, 400]

    for n_pts in resolutions:
        if time_left() < 80:
            break
        target_sum = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        if len(best_h) != n_pts:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_pts)
            h_cur = np.interp(x_new, x_old, best_h)
        else:
            h_cur = best_h.copy()
        h_cur = project_h(h_cur, target_sum)
        cur_c5 = compute_c5(h_cur)
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

        betas = [20, 50, 100, 200, 400, 800, 1600, 3200]
        for beta in betas:
            if time_left() < 20:
                break
            obj_f = partial(smooth_obj_func, beta=beta)
            grad_f = partial(full_gradient, beta=beta)
            con_f = partial(eq_constraint, target_sum=target_sum)
            con_jac = partial(eq_constraint_jac, target_sum=target_sum)
            constraints = [{'type': 'eq', 'fun': con_f, 'jac': con_jac}]
            try:
                res = minimize(obj_f, h_cur, jac=grad_f, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 2000, 'ftol': 1e-18, 'iprint': 0})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < cur_c5 - 1e-15:
                    h_cur = h_new.copy()
                    cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = cur_c5
            except Exception:
                pass

        # Subgradient polish at this resolution
        tg = n_pts / 2.0
        for _it in range(3000):
            if time_left() < 5:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            top_k = min(5, max(3, n_pts // 50))
            top_idx = np.argsort(corr)[::-1][:top_k]
            grad = np.zeros(n_pts)
            for idx in top_idx:
                lag = int(idx) - (n_pts - 1)
                if lag == 0:
                    grad += (1.0 - 2.0 * h_cur)
                elif lag > 0:
                    nv = n_pts - lag
                    grad[:nv] += (1.0 - h_cur[lag:])
                    grad[lag:] -= h_cur[:nv]
                else:
                    m = -lag
                    nv = n_pts - m
                    grad[:nv] -= h_cur[m:]
                    grad[m:] += (1.0 - h_cur[:nv])
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            grad /= gnorm
            improved = False
            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
                h_trial = project_h(h_cur - step * grad, tg)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5:
                    h_cur = h_trial
                    cur_c5 = c5_trial
                    improved = True
                    break
            if not improved:
                break
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5

    # Random restarts from perturbed best
    n_r = len(best_h)
    target_r = n_r / 2.0
    if time_left() > 40:
        for attempt in range(12):
            if time_left() < 20:
                break
            scale = 0.04 * (1.0 - 0.07 * attempt)
            perturbed = project_h(best_h + np.random.randn(n_r) * scale, target_r)
            bounds_r = Bounds(np.zeros(n_r), np.ones(n_r))
            for beta in [80, 200, 500]:
                if time_left() < 10:
                    break
                obj_fr = partial(smooth_obj_func, beta=beta)
                grad_fr = partial(full_gradient, beta=beta)
                con_fr = partial(eq_constraint, target_sum=target_r)
                con_jr = partial(eq_constraint_jac, target_sum=target_r)

                                   bounds=bounds_r, constraints=cons_r,
                                   options={'maxiter': 1500, 'ftol': 1e-16, 'iprint': 0})
                    h_new = project_h(res.x, target_r)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        perturbed = project_h(best_h + np.random.randn(n_r) * scale, target_r)
                except Exception:
                    pass

    # Final high-beta polish
    if time_left() > 15:
        n_f = len(best_h)
        target_f = n_f / 2.0
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        for beta in [4000, 8000]:
            if time_left() < 8:
                break
            obj_ff = partial(smooth_obj_func, beta=beta)
            grad_ff = partial(full_gradient, beta=beta)
            con_ff = partial(eq_constraint, target_sum=target_f)
            con_jf = partial(eq_constraint_jac, target_sum=target_f)

            try:
                res = minimize(obj_ff, best_h, jac=grad_ff, method='SLSQP',
                               bounds=bounds_f, constraints=cons_f,
                               options={'maxiter': 4000, 'ftol': 1e-18, 'iprint': 0})
                h_new = project_h(res.x, target_f)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Final subgradient sweep
    n_s = len(best_h)
    target_s = n_s / 2.0
    for _it in range(8000):
        if time_left() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        grad = np.zeros(n_s)
        for idx in top_idx:

            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif lag > 0:
                nv = n_s - lag
                grad[:nv] += (1.0 - best_h[lag:])
                grad[lag:] -= best_h[:nv]
            else:
                m = -lag
                nv = n_s - m
                grad[:nv] -= best_h[m:]
                grad[m:] += (1.0 - best_h[:nv])
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        for step in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_trial = project_h(best_h - step * grad, target_s)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                break

    return best_h, best_c5, len(best_h)
