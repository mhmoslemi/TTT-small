def prox_h(y, target_sum):
    """Proj onto {0<=h<=1, sum(h)=target_sum}"""
    # Bisection on threshold t: h_i = clip(y_i - t, 0, 1)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        t = (lo + hi) / 2
        h = np.clip(y - t, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = t
        else:
            hi = t
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fix small numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-bound entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

    return float(np.max(corr) * dx)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    return m + np.log(e.sum()) / beta, e / e.sum()

    corr = np.correlate(h, 1.0 - h, mode='full')
    val, weights = smooth_max_vec(corr, beta)

    if top_k < len(weights):

        weights = np.where(np.isin(np.arange(len(weights)), top_idx), weights, 0.0)

        t = t_new
        h = h_new
        if abs(prev_val - val) < tol:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_vec(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(shifted)
    val = m + np.log(e.sum()) / beta
    return val, e / e.sum()

def prox_constraint(y, target_sum):
    """Project y onto {0<=h<=1, sum(h)=target_sum} using bisection on offset"""
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        t = (lo + hi) / 2
        h = np.clip(y - t, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = t
        else:
            hi = t
    h = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_obj(h, dx, beta):
    """Compute (objective value, gradient) of smooth max objective"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, weights = smooth_max_vec(corr, beta)
    grad = np.zeros(n)
    # Only process significant weights
    thresh = 1e-10
    active = np.where(weights > thresh)[0]
    for idx_k in active:
        w = weights[idx_k]
        k = idx_k - (n - 1)
        if k >= 0:
            m = n - 1 - k
            if m >= 0:
                grad[:m+1] += w * (1.0 - h[k:k+m+1])
                grad[k:] -= w * h[:m+1]
        else:
            ak = -k
            grad[ak:] += w * (1.0 - h[:n-ak])
            grad[:n-ak] -= w * h[ak:]
    return val * dx, grad * dx

def fista_solve(h0, dx, beta, target_sum, max_iter=2000, L_init=None):
    """FISTA for smooth objective with box+sum constraint"""
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    # Estimate L
    if L_init is None:
        _, g0 = grad_obj(y, dx, beta)
        L = max(np.linalg.norm(g0) * 2.0, 1.0)
        # Safe upper bound: L <= 2 * beta * n (from Hessian analysis)
        L = max(L, 2.0 * beta * n * 0.1)
    else:
        L = L_init
    
    for it in range(max_iter):
        val, g = grad_obj(y, dx, beta)
        # Backtracking to ensure L is valid
        # (skip for speed, use conservative L)
        z = y - g / L
        h_new = prox_constraint(z, target_sum)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + (t_k / t_next - 1.0) * (h_new - h)
        t_k = t_next
        h = h_new
    return h

def slsqp_solve(h0, dx, beta, target_sum, n, maxiter=5000):
    """SLSQP with smooth max objective"""
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum, 
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return smooth_max_vec(np.correlate(x, 1.0-x, mode='full'), beta)[0] * dx
    def obj_grad(x):
        return grad_obj(x, dx, beta)[1]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return prox_constraint(res.x, target_sum)
    except Exception:
        return h0

def smax(x, beta):

    s = np.clip(beta * (x - m), -500, 500)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta, e / e.sum()

def prox(y, ts):
    lo, hi = y.min() - 2, y.max() + 2
    for _ in range(80):
        t = (lo + hi) / 2
        h = np.clip(y - t, 0, 1)
        if h.sum() > ts: lo = t
        else: hi = t

            a = -kk
            g[a:] += w[i]*(1-h[:n-a])
            g[:n-a] -= w[i]*h[a:]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smax(x, beta):
    m = x.max()
    s = np.clip(beta * (x - m), -500.0, 500.0)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta, e / e.sum()

def prox(y, ts):
    lo, hi = y.min() - 2.0, y.max() + 2.0
    for _ in range(90):
        t = (lo + hi) / 2
        h = np.clip(y - t, 0.0, 1.0)
        if h.sum() > ts:
            lo = t
        else:
            hi = t
    h = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    d = ts - h.sum()
    if abs(d) > 1e-12:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def goh(h, dx, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    v, w = smax(c, beta)
    g = np.zeros(n)
    act = w > 1e-11
    for i in np.where(act)[0]:
        kk = i - (n - 1)
        if kk >= 0:
            m = n - 1 - kk
            if m >= 0:
                g[:m+1] += w[i] * (1.0 - h[kk:kk+m+1])
                g[kk:] -= w[i] * h[:m+1]
        else:
            a = -kk
            g[a:] += w[i] * (1.0 - h[:n-a])
            g[:n-a] -= w[i] * h[a:]
    return v * dx, g * dx

def fista(h0, dx, beta, ts, iters=2000):
    n = len(h0)
    h, y, tk = h0.copy(), h0.copy(), 1.0
    L = 4.0 * beta + 10.0
    for it in range(iters):
        v, g = goh(y, dx, beta)
        h_new = prox(y - g / L, ts)
        tk2 = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = h_new + (tk / tk2 - 1.0) * (h_new - h)
        tk = tk2
        h = h_new
    return h

def con_fun(x, ts):
    return x.sum() - ts

def con_jac(x, ts):
    return np.ones(len(x))

def obj_fn(x, dx, beta):
    return goh(x, dx, beta)[0]

def obj_grad(x, dx, beta):
    return goh(x, dx, beta)[1]

def slsqp(h0, dx, beta, ts, n, mit=3000):
    b = Bounds(np.zeros(n), np.ones(n))
    c = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    try:
        r = minimize(obj_fn, h0, args=(dx, beta), jac=obj_grad, method='SLSQP',
                     bounds=b, constraints=c, options={'maxiter': mit, 'ftol': 1e-16})
        return prox(r.x, ts)
    except Exception:
        return h0

def subgrad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        m = n - 1 - lag
        if m >= 0:
            g[:m+1] += (1.0 - h[lag:lag+m+1])
            g[lag:] -= h[:m+1]
    else:
        a = -lag
        g[a:] += (1.0 - h[:n-a])
        g[:n-a] -= h[a:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(210, 0.5)
    n = len(h0)
    dx = 2.0 / n
    ts = n / 2.0
    bh = prox(h0, ts)
    bc = compute_c5(bh)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: FISTA with increasing beta
    for beta in [10, 30, 100, 300, 1000, 3000]:
        if rem() < 120:
            break
        h = fista(bh, dx, beta, ts, iters=min(3000, max(500, int(rem() * 1.5))))
        c = compute_c5(h)
        if c < bc - 1e-8:
            bh, bc = h, c

    # Phase 2: SLSQP continuation
    for beta in [300, 1000, 3000, 10000, 30000]:
        if rem() < 60:
            break
        h = slsqp(bh, dx, beta, ts, n, mit=min(5000, max(500, int(rem() * 5))))
        c = compute_c5(h)
        if c < bc - 1e-9:
            bh, bc = h, c

    # Phase 3: Subgradient polish on worst lag
    if rem() > 80:
        h = bh.copy()
        for it in range(5000):
            if rem() < 60:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_w = np.argmax(corr)
            lag = k_w - (n - 1)
            g = subgrad_lag(h, lag, n)
            lr = 0.003 / (1.0 + 0.002 * it)
            h2 = prox(h - lr * g, ts)
            c2 = compute_c5(h2)
            if c2 < bc:
                bh, bc, h = h2, c2, h2
            else:
                lr *= 0.5
                h2 = prox(h - lr * g, ts)
                c2 = compute_c5(h2)
                if c2 < bc:
                    bh, bc, h = h2, c2, h2

    # Phase 4: Resolution exploration
    res_list = sorted(set([n // 2, n * 2 // 3, n - 20, n - 10, n + 20, n + 50,
                           n + 100, n * 3 // 2, n * 2, 100, 150, 200, 300, 400, 500]))
    res_list = [r for r in res_list if 60 <= r <= 600 and r != n]
    for n_t in res_list:
        if rem() < 70:
            break
        dx_t = 2.0 / n_t
        ts_t = n_t / 2.0
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, n_t, endpoint=False)
        ht = prox(np.interp(xn, xo, bh), ts_t)
        for beta in [100, 1000, 5000]:
            if rem() < 40:
                break
            h = fista(ht, dx_t, beta, ts_t, iters=min(2000, int(rem() * 2)))
            c = compute_c5(h)
            if c < bc - 1e-8:
                bh, bc, n, dx, ts = h, c, n_t, dx_t, ts_t
                ht = h
        if n != 210 and rem() < 50:
            break

    # Phase 5: Random restarts from diverse initializations
    inits = [
        prox(bh + np.random.randn(n) * 0.05, ts),
        prox(np.clip(np.sin(2 * np.pi * np.arange(n) * 2.0 / n) * 0.4 + 0.5, 0, 1), ts),
        prox(np.random.randint(0, 2, n).astype(float), ts),
    ]
    for hr in inits:
        if rem() < 50:
            break
        for beta in [100, 500, 2000]:
            if rem() < 30:
                break
            h = fista(hr, dx, beta, ts, iters=min(1500, int(rem() * 2)))
            c = compute_c5(h)
            if c < bc - 1e-8:
                bh, bc = h, c
                break

    # Phase 6: Final high-precision SLSQP
    if rem() > 25:
        for beta in [10000, 50000, 200000]:
            if rem() < 18:
                break
            h = slsqp(bh, dx, beta, ts, n, mit=min(8000, max(500, int(rem() * 8))))
            c = compute_c5(h)
            if c < bc - 1e-9:
                bh, bc = h, c

    # Phase 7: Micro subgradient polish
    if rem() > 10:
        h = bh.copy()
        for it in range(3000):
            if rem() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            top3 = np.argsort(corr)[-3:]
            g = np.zeros(n)
            for idx in top3:
                lag = idx - (n - 1)
                g += subgrad_lag(h, lag, n)
            g /= len(top3)
            lr = 0.001 / (1.0 + 0.01 * it)
            h2 = prox(h - lr * g, ts)
            c2 = compute_c5(h2)
            if c2 < bc:
                bh, bc, h = h2, c2, h2
            elif it > 500:
                break

    bh = prox(bh, ts)
    bc = compute_c5(bh)
    return bh, bc, len(bh)
