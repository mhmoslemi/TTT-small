def solve_1d_minimax(a_vals, c_vals):
    """Find t in [0,1] minimizing max_k(c_vals[k] + a_vals[k]*t)"""
    # Evaluate at boundaries and all pairwise intersections
    candidates = [0.0, 1.0]
    m = len(a_vals)
    for i in range(m):
        for j in range(i+1, m):
            if abs(a_vals[i] - a_vals[j]) > 1e-15:
                t = (c_vals[j] - c_vals[i]) / (a_vals[i] - a_vals[j])
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    
    best_t = 0.0
    best_val = float('inf')
    for t in candidates:
        val = max(c_vals[k] + a_vals[k]*t for k in range(m))
        if val < best_val:
            best_val = val
            best_t = t
    return best_t

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14: break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14: break
            h += (diff / total) * h

    dx = 2.0 / n

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent_sweep(h, n, active_m):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = np.argsort(c)[-active_m:]
    
    for i in range(n):
        a_arr = np.empty(active_m)
        c_arr = np.empty(active_m)
        for k, idx in enumerate(active):
            L = idx - (n - 1)
            ip = i + L
            im = i - L
            hp = h[ip] if 0 <= ip < n else 0.0
            hm = h[im] if 0 <= im < n else 0.0
            a_arr[k] = dx * (1.0 - hp - hm)
            c_arr[k] = c[idx] - a_arr[k] * h[i]
        
        # Solve 1D minimax: find t in [0,1] minimizing max(c_arr + a_arr * t)
        candidates = [0.0, 1.0]
        m = active_m
        for ii in range(m):
            for jj in range(ii + 1, m):
                da = a_arr[ii] - a_arr[jj]
                if abs(da) > 1e-14:
                    t = (c_arr[jj] - c_arr[ii]) / da
                    if 0.0 <= t <= 1.0:
                        candidates.append(t)
        
        best_t = h[i]
        best_val = np.max(c_arr + a_arr * h[i])
        for t in candidates:
            v = np.max(c_arr + a_arr * t)
            if v < best_val:
                best_val = v
                best_t = t
        h[i] = best_t
    
    return h, c

def run_coord_descent(h, n, deadline, active_m=30, max_sweeps=100000):

    best_c = compute_c5(h, n)
    
    for sweep in range(max_sweeps):

def smooth_obj_trunc(x, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    sorted_c = np.sort(c)[-top_m:]
    m = sorted_c.max()
    shifted = alpha * (sorted_c - m)
    if shifted.max() > 500:
        return float(m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqq_refine(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(smooth_obj_trunc, h, args=(n, alpha, 50),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_update(h, n, delta):
    """Try moving mass between pairs of coordinates."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    improved = False
    for i in range(n):
        for j in range(i + 1, n):
            for s in (1.0, -1.0):
                hi = h[i] + s * delta
                hj = h[j] - s * delta
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i] = hi
                    h[j] = hj
                    c = compute_c5(h, n)
                    if c < best_c:
                        best_c = c
                        improved = True
                    else:
                        h[i] = hi - s * delta
                        h[j] = hj + s * delta
    return h, best_c

def binary_projection(h, n):
    """Project h to a binary solution with exactly n/2 ones."""
    # Sort by value, keep top n/2 as 1s
    thresh = np.sort(h)[n//2]
    result = (h >= thresh).astype(float)
    # Fix the count
    ones = int(result.sum())
    if ones > n//2:
        # Set some 1s to 0s (those closest to threshold)
        idx_ones = np.where(result == 1.0)[0]
        order = np.argsort(h[idx_ones])
        for k in range(ones - n//2):
            result[idx_ones[order[k]]] = 0.0
    elif ones < n//2:
        idx_zeros = np.where(result == 0.0)[0]
        order = np.argsort(-h[idx_zeros])
        for k in range(n//2 - ones):
            result[idx_zeros[order[k]]] = 1.0
    return result

def cd_sweep(h, n, active_m):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    active = np.argsort(c)[-active_m:]
    lags = active - (n - 1)
    m = len(active)

    for i in range(n):
        ip = i + lags
        im = i - lags
        hp = np.where((ip >= 0) & (ip < n), h[np.clip(ip, 0, n - 1)], 0.0)
        hm = np.where((im >= 0) & (im < n), h[np.clip(im, 0, n - 1)], 0.0)
