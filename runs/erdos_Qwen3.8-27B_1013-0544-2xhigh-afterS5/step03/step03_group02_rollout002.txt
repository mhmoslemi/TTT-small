    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

        grad[la:] += (1.0 - h[:m])
        grad[:m] -= h[la:]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:])
            grad[lag:] -= h[:m]
    else:
        la = -lag
        m = n - la
        if m > 0:
            grad[la:] += (1.0 - h[:m])
            grad[:m] -= h[la:]
    return grad

def sml_obj(h_in, params):
    beta, dx, top_k = params
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    if top_k is not None and top_k < len(c):
        thresh = np.partition(c, len(c) - top_k)[len(c) - top_k]
        c = np.maximum(c, thresh)
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol, top_k=None):
    dx_val = 2.0 / n
    params = (beta, dx_val, top_k)
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(lambda x: sml_obj(x, params), h0, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def active_set_opt(cur, n, target, t0, budget, max_rounds=200):
    best_c5 = compute_c5(cur)
    for _ in range(max_rounds):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        improved = False
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g = lag_gradient(cur, lag, n)
            g -= np.mean(g)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                continue
            g /= gn
            for alpha in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
                trial = project_h(cur - alpha * g, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def coord_polish(cur, n, target, t0, budget, sweeps=20):
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002,
              0.001, -0.001, 0.0005, -0.0005]
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                if cur[j] + d < 0 or cur[j] + d > 1:
                    continue
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw >= 3:
            break
    return cur, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2.0, n_new), np.linspace(0, 2.0, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_parent = np.full(140, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    resolutions = [n_base]
    if n_base <= 320:
        resolutions.extend([360, 400])

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        target = n_pts / 2.0

        for beta in [60, 150, 400, 1000, 2500]:
            remaining = budget_s - (time.time() - t0)

            tk = 15 if beta < 500 else 10
            cur = do_slsqp(cur, n_pts, target, beta, bounds, 3000, 1e-14, tk)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        remaining = budget_s - (time.time() - t0)

    # Final intensive polish
    remaining = budget_s - (time.time() - t0)

        nf = len(best_h)

            best_h = do_slsqp(best_h, nf, tf, beta, bf, 8000, 1e-15, 6)
            best_c5 = compute_c5(best_h)

            best_h, best_c5 = active_set_opt(best_h, nf, tf, t0, budget_s, 500)

        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = coord_polish(best_h, nf, tf, t0, budget_s, 30)

        remaining = budget_s - (time.time() - t0)
