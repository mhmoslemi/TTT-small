def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)

    if abs(diff) > 1e-12:

        h = np.clip(h, 0.0, 1.0)

    """L-BFGS-B with penalty for sum constraint."""

    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)

    target = n / 2.0
    x = project_feasible(h.copy(), target)
    best_x = x.copy()
    best_v = compute_c5(x)
    for stage in range(n_stages):
        mu = 200.0 * (8.0 ** stage)
        def obj_pen(x):
            xc = np.clip(x, 0, 1)
            v, g = smoothmax_val_grad(xc, beta, top_k)
            d = xc.sum() - target
            return v + mu * d * d, g + 2.0 * mu * d
        try:
            res = minimize(obj_pen, x, jac=True, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'gtol': 1e-12})
            x = project_feasible(res.x, target)
            v = compute_c5(x)
            if v < best_v:
                best_v = v
                best_x = x.copy()
        except Exception:
            pass
    return best_x, best_v

def coord_descent(h, n_sweeps=3, n_candidates=12):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_v = compute_c5(x)
    grid = np.linspace(0, 1, n_candidates)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old = x[j]
            best_obj = best_v
            best_c = old
            for c in grid:
                if abs(c - old) < 1e-6:
                    continue
                x[j] = c
                v = compute_c5(x)
                if v < best_obj:
                    best_obj = v
                    best_c = c
            x[j] = best_c
            if best_c != old:
                improved = True
        x = project_feasible(x, target)
        v = compute_c5(x)
        if v < best_v:
            best_v = v
        if not improved:
            break
    return x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP progressive beta
    for beta in [50, 100, 200, 500, 1000, 2000]:
        if tl() < 35:
            break
        top_k = n if beta < 200 else max(10, n // 3)
        h_new = slsqp_optimize(h_best, maxiter=500, beta=beta, top_k=top_k)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: L-BFGS-B complement
    if tl() > 50:
        h_new, c = lbfgsb_optimize(h_best, 300, 800, max(10, n // 4))
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Multi-start diverse perturbations
    for r in range(20):
        if tl() < 40:
            break
        scale = 0.008 * (1.0 + 0.4 * (r % 5))
        kind = r % 5
        if kind == 0:
            freqs = np.fft.rfftfreq(n)
            spec = np.zeros(len(freqs))
            for k in range(1, min(6, len(freqs))):
                spec[k] = np.random.randn()
            pert = np.fft.irfft(spec * 2 * n) * scale
        elif kind == 1:
            pert = np.random.randn(n) * scale
        elif kind == 2:
            bs = max(2, n // 20)
            pert = np.repeat(np.random.randn(n // bs + 1) * scale, bs)[:n]
        elif kind == 3:
            freq = np.random.uniform(3, 20)
            phase = np.random.uniform(0, 2 * np.pi)
            pert = scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        else:
            # Asymmetric perturbation
            ramp = np.linspace(-1, 1, n)
            pert = scale * ramp * np.random.randn(n)
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_s = slsqp_optimize(h_pert, maxiter=300, beta=1000, top_k=max(8, n // 4))
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            h_best, best_c5 = h_s, c_s
        if tl() > 20 and c_s < best_c5 + 0.001:
            h_l, c_l = lbfgsb_optimize(h_pert, 200, 1000, max(8, n // 4))
            if c_l < best_c5:
                h_best, best_c5 = h_l, c_l

    # Phase 4: Coordinate descent polish
    if tl() > 25:
        h_cd, c_cd = coord_descent(h_best, 3, 15)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd
    if tl() > 20:
        h_cd2, c_cd2 = coord_descent(h_best, 2, 20)
        if c_cd2 < best_c5:
            h_best, best_c5 = h_cd2, c_cd2

    # Phase 5: Higher resolution
    for n_try in [600, 800, 1000]:
        if tl() < 140:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [200, 500, 1200, 2500]:
            if tl() < 45:
                break
            top_k = max(12, n_try // 3)
            h_new = slsqp_optimize(h_interp, maxiter=400, beta=beta, top_k=top_k)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        if tl() > 25:
            h_cd, c_cd = coord_descent(h_best, 2, 12)
            if c_cd < best_c5:
                h_best, best_c5 = h_cd, c_cd

    # Phase 6: Final high-beta SLSQP
    if tl() > 30:
        h_new = slsqp_optimize(h_best, 600, 3000, max(6, len(h_best) // 5))
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 7: Very fine coordinate descent
    if tl() > 15:
        h_cd, c_cd = coord_descent(h_best, 2, 25)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
