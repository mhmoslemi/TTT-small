    corr = np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    shifted = beta * (corr - np.max(corr))
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    weights = e / e.sum()
    value = np.max(corr) + np.log(e.sum()) / beta
