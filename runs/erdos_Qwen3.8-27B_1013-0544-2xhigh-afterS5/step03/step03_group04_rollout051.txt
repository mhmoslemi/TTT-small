import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_convex(c, b):
    """Minimize max(c*x + b) over x in [0,1]. c,b are 1D arrays."""
    vals_lo = np.max(b)
    active_lo = np.where(b >= vals_lo - 1e-13)[0]
    if c[active_lo].min() >= -1e-13:
        return 0.0
    vals_hi = np.max(c + b)
    active_hi = np.where(c + b >= vals_hi - 1e-13)[0]
    if c[active_hi].max() <= 1e-13:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        vals = c * mid + b
        mx = np.max(vals)
        active = np.where(vals >= mx - 1e-13)[0]
        if c[active].min() >= 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent(h, lam, n_sweeps=300):
    n = len(h)
    h = h.copy()
    s = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_change = 0.0
        for i in range(n):
            T1 = (i - s >= 0) & (i - s < n)
            T2 = (i + s >= 0) & (i + s < n)
            a = np.zeros(2 * n - 1)
            if T1.any():
                a[T1] = 1.0 - h[i - s[T1]]
            if T2.any():
                a[T2] -= h[i + s[T2]]
            b = corr - a * h[i]
            c_arr = a + lam
            new_val = solve_1d_convex(c_arr, b)
            ch = abs(new_val - h[i])
            if ch > max_change:
                max_change = ch
            h[i] = new_val
        if max_change < 1e-13:
            break
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def slsqq_refine(h, target, n_iter=200):
    n = len(h)
    dx = 2.0 / n
    beta_seq = [50, 100, 200, 500, 1000, 2000, 4000, 8000]
    h = h.copy()
    for beta in beta_seq:
        def obj(x):
            h2 = np.clip(x, 0, 1)
            corr = np.correlate(h2, 1.0 - h2, mode="full")
            return smooth_max(corr, beta) * dx + 100.0 * (h2.sum() - target)**2
        def grad(x):
            h2 = np.clip(x, 0, 1)
            corr = np.correlate(h2, 1.0 - h2, mode="full")
            xs = np.clip(beta * (corr - np.max(corr)), -500, 500)
            w = np.exp(xs) / np.sum(np.exp(xs))
            s_arr = np.arange(2*n-1) - (n-1)
            g = np.zeros(n)
            for i in range(n):
                T1 = (i - s_arr >= 0) & (i - s_arr < n)
                T2 = (i + s_arr >= 0) & (i + s_arr < n)
                a = np.zeros(2*n-1)
                if T1.any():
                    a[T1] = 1.0 - h2[i - s_arr[T1]]
                if T2.any():
                    a[T2] -= h2[i + s_arr[T2]]
                g[i] = np.sum(w * a) * dx + 200.0 * (h2.sum() - target)
            return g
        for it in range(n_iter):
            g = grad(h)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            alpha = 0.05
            h_new = h - alpha * g / gn
            h_new = np.clip(h_new, 0, 1)
            h_new = project_h(h_new, target)
            if obj(h_new) < obj(h):
                h = h_new
            else:
                alpha *= 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)
    
    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    resolutions = [n_base]
    for mult in [2, 3, 4]:
        nr = n_base * mult
        if nr <= 1200:
            resolutions.append(nr)
    
    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 90:
            break
        
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_pts)
        h_cur = np.interp(x_new, x_old, best_h)
        h_cur = project_h(h_cur, n_pts / 2.0)
        cur_c5 = compute_c5(h_cur)
        
        # Coordinate descent with various lambda values
        for lam in [0.0, 0.002, -0.002, 0.01, -0.01, 0.005, -0.005]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 40:
                break
            h_new = coord_descent(h_cur, lam, n_sweeps=200)
            h_new = project_h(h_new, n_pts / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h_cur = h_new
                cur_c5 = c5_new
        
        # Gradient refinement
        remaining = budget_s - (time.time() - t0)
        if remaining > 60:
            h_grad = slsqq_refine(h_cur, n_pts / 2.0)
            c5_grad = compute_c5(h_grad)
            if c5_grad < cur_c5:
                h_cur = h_grad
                cur_c5 = c5_grad
        
        # Random perturbations
        remaining = budget_s - (time.time() - t0)
        n_perturb = max(3, int(min(20, remaining / 30)))
        for _ in range(n_perturb):
            remaining = budget_s - (time.time() - t0)
            if remaining < 30:
                break
            noise = np.random.uniform(-0.05, 0.05, n_pts)
            h_pert = project_h(h_cur + noise, n_pts / 2.0)
            h_pert = coord_descent(h_pert, 0.0, n_sweeps=100)
            h_pert = project_h(h_pert, n_pts / 2.0)
            c5_pert = compute_c5(h_pert)
            if c5_pert < cur_c5:
                h_cur = h_pert
                cur_c5 = c5_pert
        
        if cur_c5 < best_c5:
            best_h = h_cur.copy()
            best_c5 = cur_c5
    
    # Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_final = len(best_h)
        h_fin = coord_descent(best_h, 0.0, n_sweeps=200)
        h_fin = project_h(h_fin, n_final / 2.0)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin
    
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
