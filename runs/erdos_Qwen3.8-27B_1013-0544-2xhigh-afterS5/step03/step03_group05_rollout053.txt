import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def obj_smooth_max(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def con_sum(x, target):
    return np.sum(x) - target

def con_sum_jac(x, target):
    return np.ones(x.shape)

def get_subgradient(h, n, dx, top_k=2, temp=500.0):
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr_val = corr_raw * dx
    top_idx = np.argsort(corr_val)[::-1][:top_k]
    vals = corr_val[top_idx]
    weights = np.exp((vals - vals[0]) * temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    center = n - 1
    for t_idx, w in zip(top_idx, weights):
        lag = int(t_idx) - center
        grad += w * grad_for_lag(h, lag, n, dx)
    return grad, corr_val

def mom_descent(cur, n, target, dx, max_iters, time_fn, min_time, best_state,
                top_k=2, temp=500.0, step_init=0.03, momentum=0.3):

    vel = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        grad, _ = get_subgradient(cur, n, dx, top_k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        vel = momentum * vel + (1 - momentum) * grad
        vnorm = np.linalg.norm(vel)
        if vnorm < 1e-15:
            break
        vel /= vnorm
        step = step_init / (1.0 + 0.001 * it)
        improved = False
        for _ in range(30):
            h_trial = project_h(cur - step * vel, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                stall = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                improved = True
                break
            step *= 0.5
        if not improved:
            stall += 1
            if stall > 30:
                if stall > 200:
                    T = 0.008 * np.exp(-(stall - 30) / 300.0)
                    if T < 1e-5:
                        break
                    noise = np.random.normal(0, T, n)
                    h_trial = project_h(cur + noise, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < cur_c5 + T * 4:
                        cur = h_trial
                        cur_c5 = c5_trial
                        if cur_c5 < best_state[0]:
                            best_state[0] = cur_c5
                            best_state[1] = cur.copy()
                            best_state[2] = n
                    stall = 0
                else:
                    # Pairwise swap
                    i = np.random.randint(n)
                    j = np.random.randint(n)
                    if i != j:
                        tmp = cur[i]
                        cur[i] = cur[j]
                        cur[j] = tmp
                        cur = project_h(cur, target)
                        c5_trial = compute_c5(cur)
                        if c5_trial < cur_c5 + 0.0001:
                            cur_c5 = c5_trial
                            if cur_c5 < best_state[0]:
                                best_state[0] = cur_c5
                                best_state[1] = cur.copy()
                                best_state[2] = n
                        else:
                            cur[i] = tmp
                            cur[j] = tmp2 if False else cur[j]
                            # undo swap
                            cur[i], cur[j] = cur[j], cur[i]
                            cur = project_h(cur, target)
                        stall = 0
    return cur, cur_c5, n

def swap_search(cur, n, target, time_fn, min_time, best_state, n_iters=50000):
    cur_c5 = compute_c5(cur)
    T = 0.003
    for it in range(n_iters):
        if time_fn() < min_time:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Try small perturbation: shift value from i to j
        delta = 0.05 * np.random.choice([-1, 1])
        h_trial = cur.copy()
        h_trial[i] += delta
        h_trial[j] -= delta
        h_trial = project_h(h_trial, target)
        c5_trial = compute_c5(h_trial)
        d_c5 = c5_trial - cur_c5
        if d_c5 < 0 or np.random.rand() < np.exp(-d_c5 / T):
            cur = h_trial
            cur_c5 = c5_trial
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n
        T *= 0.9998
    return cur, cur_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    # Phase 1: SLSQP at parent resolution for quick improvement
    n = n_parent
    target = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cur = project_h(h_parent, target)

    for beta in [100, 300, 800]:
        if time_left() < 15:
            break
        obj = partial(obj_smooth_max, dx=dx, beta=beta)
        con = partial(con_sum, target=target)
        con_j = partial(con_sum_jac, target=target)
        cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
        try:
            res = minimize(obj, cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(cur):
                cur = h_new
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = cur.copy()
                best_state[2] = n
        except Exception:
            pass

    # Phase 2: Main descent with momentum
    cur, cur_c5, _ = mom_descent(
        cur, n, target, dx, max_iters=8000, time_fn=time_left,
        min_time=60, best_state=best_state, top_k=2, temp=400.0,
        step_init=0.025, momentum=0.25
    )

    # Phase 3: Swap-based local search
    if time_left() > 40:
        cur, cur_c5, _ = swap_search(
            best_state[1], best_state[2], best_state[2] / 2.0,
            time_fn=time_left, min_time=20, best_state=best_state, n_iters=30000
        )

    # Phase 4: Try other resolutions
    for n2 in [200, 300, 350]:
        if time_left() < 50:
            break
        if n2 == n_parent:
            continue
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        bounds2 = Bounds(np.zeros(n2), np.ones(n2))
        h_res = project_h(resample(best_state[1], n2), target2)
        cur2 = h_res.copy()

        if time_left() > 20:
            obj = partial(obj_smooth_max, dx=dx2, beta=300)
            con = partial(con_sum, target=target2)
            con_j = partial(con_sum_jac, target=target2)
            cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
            try:
                res = minimize(obj, cur2, method='SLSQP', bounds=bounds2,
                               constraints=cons, options={'maxiter': 500, 'ftol': 1e-16})
                h_new = project_h(res.x, target2)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(cur2):
                    cur2 = h_new
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = cur2.copy()
                    best_state[2] = n2
            except Exception:
                pass

        if time_left() > 20:
            cur2, _, _ = mom_descent(
                cur2, n2, target2, dx2, max_iters=3000, time_fn=time_left,
                min_time=15, best_state=best_state, top_k=2, temp=400.0,
                step_init=0.02, momentum=0.3
            )

    # Phase 5: Final fine polish
    if time_left() > 8:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target_f = best_n / 2.0
        dx_f = 2.0 / best_n
        best_c5 = best_state[0]

        # Very sharp subgradient descent
        sharp_descent = 0
        for it in range(50000):
            if time_left() < 2:
                break
            grad, corr_val = get_subgradient(best_h, best_n, dx_f, top_k=1, temp=1e9)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            grad /= gnorm
            step = 0.004 / (1.0 + 0.0005 * sharp_descent)
            improved = False
            for _ in range(20):
                h_trial = project_h(best_h - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
                step *= 0.5
            if improved:
                sharp_descent = 0
            else:
                sharp_descent += 1
                if sharp_descent > 200:
                    # Micro perturbation
                    idx = np.random.randint(best_n)
                    h_trial = best_h.copy()
                    h_trial[idx] += 0.02 * np.random.choice([-1, 1])
                    h_trial = project_h(h_trial, target_f)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial
                        best_c5 = c5_trial
                    sharp_descent = 0
                elif sharp_descent > 50:
                    break

        best_state[0] = best_c5
        best_state[1] = best_h
        best_state[2] = best_n

    return best_state[1], best_state[0], best_state[2]
