import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_fk(h, k, n):
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        kk = -k
        g[:n-kk] += (1.0 - h[kk:])
        g[kk:] -= h[:n-kk]
    return g


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta


def smooth_max_grad(h, dx, n, beta):
    c = all_corr(h, dx)
    m = np.max(c)
    w = np.exp(np.clip(beta * (c - m), -500, 500))
    w /= w.sum()
    mask = w > 1e-7
    g = np.zeros(n)
    center = n - 1
    for i in np.where(mask)[0]:
        k = int(i) - center
        g += w[i] * grad_fk(h, k, n)
    return g * dx


def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)


def subgrad_loop(h, target, dx, n, max_iters, tleft, best_state, step0=0.06, top_k=2, wb=150, decay=0.9992):
    step = step0
    cur_c5 = compute_c5(h)
    for it in range(max_iters):
        if tleft() < 10:
            break
        corr = all_corr(h, dx)
        top_idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[top_idx]
        w = np.exp((vals - vals[0]) * wb)
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for idx, wi in zip(top_idx, w):
            k = int(idx) - center
            g += wi * grad_fk(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_h(h - step * g / gn, target)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 - 1e-13:
            h = h_new
            cur_c5 = c5_new
        if c5_new < best_state[0]:
            best_state[0] = c5_new
            best_state[1] = h.copy()
            best_state[2] = n
        step *= decay
    return h


def slsqp_refine(h, target, dx, n, beta, maxiter, tleft, best_state):
    def obj(x):
        return smooth_max_val(all_corr(x, dx), beta)
    def obj_grad(x):
        return smooth_max_grad(x, dx, n, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        c5 = compute_c5(h_new)
        if c5 < best_state[0]:
            best_state[0] = c5
            best_state[1] = h_new.copy()
            best_state[2] = n
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(250, 0.5)

    n0 = len(h0)
    best_state = [compute_c5(h0), h0.copy(), n0]

    # Phase 1: Refine at original resolution
    n = n0
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()

    h = subgrad_loop(h, target, dx, n, 4000, tleft, best_state, 0.08, 2, 120, 0.999)

    for beta in [60, 120, 250, 500, 1000]:
        if tleft() < 80:
            break
        h = slsqp_refine(h, target, dx, n, beta, 500, tleft, best_state)

    h = subgrad_loop(h, target, dx, n, 4000, tleft, best_state, 0.03, 3, 300, 0.9995)

    # Try top_k=1 for aggressive max reduction
    h = subgrad_loop(h, target, dx, n, 3000, tleft, best_state, 0.04, 1, 1, 0.999)

    for beta in [800, 1500, 3000]:
        if tleft() < 50:
            break
        h = slsqp_refine(h, target, dx, n, beta, 800, tleft, best_state)

    h = subgrad_loop(h, target, dx, n, 3000, tleft, best_state, 0.015, 4, 400, 0.9997)

    # Phase 2: Higher resolution refinement
    for n2 in [400, 500, 700]:
        if tleft() < 90:
            break
        h2 = project_h(resample(best_state[1], n2), n2 / 2.0)
        target2 = n2 / 2.0
        dx2 = 2.0 / n2

        h2 = subgrad_loop(h2, target2, dx2, n2, 2500, tleft, best_state, 0.05, 2, 150, 0.999)

        for beta in [150, 400, 800]:
            if tleft() < 50:
                break
            h2 = slsqp_refine(h2, target2, dx2, n2, beta, 400, tleft, best_state)

        h2 = subgrad_loop(h2, target2, dx2, n2, 2500, tleft, best_state, 0.02, 3, 300, 0.9995)

        h2 = subgrad_loop(h2, target2, dx2, n2, 1500, tleft, best_state, 0.03, 1, 1, 0.999)

        if tleft() > 30:
            h2 = subgrad_loop(h2, target2, dx2, n2, 2000, tleft, best_state, 0.01, 4, 500, 0.9997)

    # Phase 3: Randomized restarts at best resolution
    n_rb = best_state[2]
    target_rb = n_rb / 2.0
    dx_rb = 2.0 / n_rb
    x_r = np.linspace(0, 2, n_rb)

    for r in range(12):
        if tleft() < 25:
            break
        h_r = best_state[1].copy()
        if r % 4 == 0:
            h_r += 0.03 * np.sin((r // 4 + 3) * np.pi * x_r)
        elif r % 4 == 1:
            h_r += 0.02 * np.cos((r // 4 + 5) * 2 * np.pi * x_r)
        elif r % 4 == 2:
            idx = np.random.choice(n_rb, size=max(8, n_rb // 15), replace=False)
            h_r[idx] += np.random.uniform(-0.08, 0.08, len(idx))
        else:
            bl = max(3, n_rb // 25)
            st = np.random.randint(0, n_rb - bl)
            h_r[st:st+bl] += 0.04 * np.random.choice([-1, 1])
        h_r = project_h(h_r, target_rb)
        c5_r = compute_c5(h_r)
        if c5_r < best_state[0]:
            best_state[0] = c5_r
            best_state[1] = h_r.copy()
            best_state[2] = n_rb

        h_r = subgrad_loop(h_r, target_rb, dx_rb, n_rb, 1500, tleft, best_state, 0.03, 3, 200, 0.999)
        if tleft() > 20:
            h_r = slsqp_refine(h_r, target_rb, dx_rb, n_rb, 500, 300, tleft, best_state)

    # Final polish
    if tleft() > 15:
        n_f = best_state[2]
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        h_f = best_state[1].copy()
        h_f = subgrad_loop(h_f, target_f, dx_f, n_f, 3000, tleft, best_state, 0.008, 3, 500, 0.9998)
        if tleft() > 8:
            h_f = subgrad_loop(h_f, target_f, dx_f, n_f, 2000, tleft, best_state, 0.005, 5, 600, 0.9999)

    return best_state[1], best_state[0], best_state[2]
