```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def topk_subgrad(h, n_iters, lr0, decay, top_k, binary_push=0.0, use_nesterov=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(n_iters):
        lr_t = lr0 * (decay ** t)
        
        if use_nesterov and t > 0:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
            y = project_feasible(y, target)
        else:
            y = x.copy()

        top_idx = np.argsort(corr)[-top_k:]

        for s in lags:

                ms = np.arange(s, n)
                grad[ms] += (1.0 - y[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            elif s < 0:

        x_prev = x
        x = x_new

def smoothmax_grad(h, beta):

    sw = np.sum(exps)
    w = exps / sw
    
    val = (m + np.log(sw) / beta) * dx

def slsqp_smoothmax(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_grad(x, beta)
        return v
    
    def grad(x):
        _, g = smoothmax_grad(x, beta)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    base = np.zeros(block_len)
    base[:n_ones_in_block] = 1.0
    n_periods = n // block_len
    h = np.tile(base, n_periods)
    if len(h) < n:
        h = np.concatenate([h, base[:n - len(h)]])
    # Normalize to have exactly n/2 ones (sum = n/2)
    # Current sum = n_periods * n_ones_in_block + max(0, n - n_periods*block_len) * (n_ones_in_block / block_len)
    # We need sum = n/2, so we may need to adjust
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Add or remove fractional ones
        if diff > 0:
            for i in range(len(h)):
                if h[i] < 1.0:
                    add = min(1.0 - h[i], diff)
                    h[i] += add
                    diff -= add

        else:

                if h[i] > 0:
                    rem = min(h[i], -diff)
                    h[i] -= rem
                    diff += rem
                    if abs(diff) < 1e-10:
                        break
    return project_feasible(h, target)
```

- Start with small top_k (e.g., 3) to focus on the worst lags
- Gradually increase top_k as we make progress
- Use momentum for stability
- Add a small binary push

And another idea: **alternating between reducing the current max and balancing the top lags**. If I always minimize the top-1 lag, other lags might increase. By using top-k, I balance them.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def topk_subgrad(h, n_iters, lr0, decay, top_k, binary_push=0.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    stagnant = 0
    for t in range(n_iters):
        lr_t = lr0 * (decay ** t)
        mu = (t - 1) / (t + 2.0) if t > 0 else 0.0
        y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                ms = np.arange(s, n)
                grad[ms] += (1.0 - y[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            elif s < 0:
                tt = -s
                ms = np.arange(0, n - tt)
                grad[ms] += (1.0 - y[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += (-y[mp - tt])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        if binary_push > 0:
            grad -= binary_push * (1.0 - 2.0 * y) * dx
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v - 1e-10:
            best_v = v
            best_x = x_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        x_prev = x
        x = x_new
    return best_x, best_v

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50.0, 0.0))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        elif s < 0:
            tt = -s
            ms = np.arange(0, n - tt)
            grad[ms] += wk * (1.0 - h[ms + tt])
            mp = np.arange(tt, n)
            grad[mp] += wk * (-h[mp - tt])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def slsqp_smoothmax(h, beta, maxiter=300):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_grad(x, beta)
        return v
    def gr(x):
        _, g = smoothmax_grad(x, beta)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def make_block_pattern(n, period, n_ones):
    base = np.zeros(period)
    base[:n_ones] = 1.0
    n_rep = n // period
    h = np.tile(base, n_rep)
    if len(h) < n:
        h = np.concatenate([h, base[:n - len(h)]])
    return project_feasible(h, n / 2.0)

def make_periodic_ones(n, period, ones_positions):
    base = np.zeros(period)
    for p in ones_positions:
        if 0 <= p < period:
            base[p] = 1.0
    n_rep = n // period
    h = np.tile(base, n_rep)
    if len(h) < n:
        h = np.concatenate([h, base[:n - len(h)]])
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Aggressive top-k subgradient with increasing k and binary push
    for top_k in [4, 6, 8, 10, 12]:
        if tl() < 25:
            break
        for bp in [0.01, 0.03]:
            if tl() < 20:
                break
            h_sg, c_sg = topk_subgrad(h_best, 6000, 0.025, 0.99997, top_k, bp)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg

    # Phase 2: SLSQP refinement
    for beta in [150, 400, 800]:
        if tl() < 35:
            break
        h_new = slsqp_smoothmax(h_best, beta, maxiter=250)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Diverse starting structures
    structures = []
    for period in [6, 8, 10, 12, 14, 16, 18, 20, 24, 28, 32]:
        structures.append(make_block_pattern(n, period, period // 2))
    # Symmetric patterns
    for period in [10, 14, 18, 22, 26, 30]:
        half = period // 2
        pos = list(range(0, half, max(1, half // (half // 2))))
        structures.append(make_periodic_ones(n, period, pos))
    # Random starts
    for _ in range(4):
        h_r = np.random.rand(n)
        h_r = project_feasible(h_r, n / 2.0)
        structures.append(h_r)

    for h_s in structures:
        if tl() < 45:
            break
        c0 = compute_c5(h_s)
        if c0 > best_c5 + 0.02:
            continue
        h_opt, c_opt = topk_subgrad(h_s, 4000, 0.025, 0.99997, 8, 0.02)
        if tl() > 25:
            h_opt = slsqp_smoothmax(h_opt, 400, maxiter=200)
        c = compute_c5(h_opt)
        if c < best_c5:
            h_best, best_c5 = h_opt, c

    # Phase 4: Longer subgradient with adapted schedule
    if tl() > 60:
        h_sg, c_sg = topk_subgrad(h_best, 15000, 0.015, 0.99999, 10, 0.015)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    # Phase 5: Shake and re-optimize
    for trial in range(8):
        if tl() < 35:
            break
        scale = 0.006 * (1.0 + 0.4 * trial)
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new, c_tmp = topk_subgrad(h_pert, 4000, 0.02, 0.99997, 8, 0.02)
        if tl() > 20:
            h_new = slsqp_smoothmax(h_new, 600, maxiter=200)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 6: Higher resolution
    for n_try in [500, 700]:
        if tl() < 130:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_opt, c_opt = topk_subgrad(h_interp, 10000, 0.015, 0.99998, 8, 0.01)
        if tl() > 40:
            h_opt = slsqp_smoothmax(h_opt, 500, maxiter=250)
        if tl() > 20:
            h_opt = topk_subgrad(h_opt, 5000, 0.01, 0.99999, 10, 0.01)
        c = compute_c5(h_opt)
        if c < best_c5:
            h_best, best_c5 = h_opt, c

    # Phase 7: Final high-beta polish
    if tl() > 30:
        h_new = slsqp_smoothmax(h_best, 1200, maxiter=350)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
    if tl() > 20:
        h_sg, c_sg = topk_subgrad(h_best, 5000, 0.01, 0.99999, 6, 0.01)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
```
