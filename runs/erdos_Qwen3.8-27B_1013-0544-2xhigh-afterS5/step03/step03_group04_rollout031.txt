    best_h = h.copy()
    best_c5 = cur_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            return h
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def _obj_slsxq(x, beta, dx, target):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, beta) * dx

def _con_eq(x, target):
    return np.sum(x) - target

def _con_jac(x):
    return np.ones(x.shape)

def compute_grad_smoothmax(y, beta, n):
    corr = np.correlate(y, 1.0 - y, mode="full")
    m_val = np.max(corr)
    xs = np.clip(beta * (corr - m_val), -500, 500)
    w = np.exp(xs)
    w /= w.sum()
    grad = np.zeros(n)
    k_range = np.arange(-(n - 1), n)
    valid_pos = (n - 1) >= 0  # always true, just for structure
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-12:
            continue
        k = mi - (n - 1)
        if k >= 0:
            grad[k:] += wm * (1.0 - y[:n - k])
            grad[:n - k] -= wm * y[k:]
        else:
            kk = -k
            grad[:n - kk] += wm * (1.0 - y[kk:])
            grad[kk:] -= wm * y[:n - kk]
    return grad

def nesterov_optimize(h0, target, dx, beta, max_iter, t0, time_limit):
    n = len(h0)
    x = h0.copy()
    x_prev = x.copy()
    t_nest = 1.0
    alpha = 0.03
    best_c5 = compute_c5(x)
    best_x = x.copy()
    stall = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        gamma = (t_nest - 1) / (t_nest + 1)
        y = np.clip(x + gamma * (x - x_prev), 0, 1)
        grad = compute_grad_smoothmax(y, beta, n) * dx
        x_new = project_h(y - alpha * grad, target)
        c5_new = compute_c5(x_new)
        if c5_new > best_c5 * 1.0005 + 1e-8:
            alpha *= 0.6
            x_new = project_h(y - alpha * grad, target)
            c5_new = compute_c5(x_new)
            stall += 1
        else:
            alpha = min(alpha * 1.03, 0.15)
            if c5_new < best_c5:
                stall = 0
        x_prev = x
        x = x_new
        t_nest = (1 + np.sqrt(1 + 4 * t_nest ** 2)) / 2.0
        if c5_new < best_c5:
            best_c5 = c5_new
            best_x = x.copy()
        if stall > 300:
            break
    return best_x, best_c5

def coord_descent_exact(h0, target, n_sweeps, t0, time_limit):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    k_range = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old = h[i]
            corr = np.correlate(h, 1.0 - h, mode="full")
            B = np.zeros(2 * n - 1)
            vp = (i + k_range >= 0) & (i + k_range < n)
            B[vp] += (1.0 - h[i + k_range[vp]])
            vn = (i - k_range >= 0) & (i - k_range < n)
            B[vn] -= h[i - k_range[vn]]
            A = corr - B * old
            # Find candidate t: breakpoints of top correlations + grid
            top_idx = np.argsort(A)[-25:]
            candidates = np.array([0.0, 1.0])
            for p in range(len(top_idx)):

                    db = B[top_idx[p]] - B[top_idx[q]]

                        if 0.0 < ts < 1.0:

            # Add fine grid

            candidates = np.unique(np.clip(candidates, 0, 1))
