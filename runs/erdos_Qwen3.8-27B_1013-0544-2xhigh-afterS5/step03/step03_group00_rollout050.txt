def smoothmax_eval(h, beta, top_k):
    """Compute smoothmax value and gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c = corr[top_idx]
    m = np.max(c)
    # Clip to avoid overflow
    c_clip = c - m
    exps = np.exp(beta * c_clip)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    # Gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = k - n + 1
        # h[i] appears in first factor when i+s is in [0,n)

        i_hi = min(n, n + s)

        # h[i] appears in second factor when i-s is in [0,n)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * (-h[j_lo + s:j_hi + s])
    grad *= dx
    return val, grad

    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_eval(x, beta, top_k)[0]

        return smoothmax_eval(x, beta, top_k)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_eval_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c = corr[top_idx]
    m = np.max(c)
    exps = np.exp(beta * (c - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wi * (1.0 - h[i_lo - s:i_hi - s])
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wi * (-h[j_lo + s:j_hi + s])
    grad *= dx
    return val, grad


def do_slsqp(h, beta, maxiter, top_k):
    n = len(h)
    target = n / 2.0

        return smoothmax_eval_grad(x, beta, top_k)[0]

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)
