def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def smooth_max_w(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -700.0, 700.0)
    e = np.exp(sh)
    return e / np.sum(e)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        d = target_sum - s
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_val(corr, beta) * dx
    w = smooth_max_w(corr, beta)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            N = n - lag
            if N <= 0:
                continue
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:

    return val, grad

    return h.sum() - ts

    return np.ones(len(h))

Another idea: **Use the Frank-Wolfe (conditional gradient) method**. This is natural for problems with simplex-like constraints. At each step, we find the vertex of the feasible polytope that most reduces the objective, then take a step towards it.

The feasible set is {h: 0 ≤ h ≤ 1, sum(h) = n/2}. The vertices of this polytope are the 0-1 vectors with exactly n/2 ones. So Frank-Wolfe would:
1. Compute gradient g of the smooth objective
2. Find the 0-1 vector v (with n/2 ones) that minimizes g·v
   - This is: set v[i] = 1 for the n/2 indices where g[i] is smallest
3. Update: h = (1-γ)h + γ*v for some step size γ

This is actually a very natural fit! And it's different from what the previous code did.

            else:
                velocity *= 0.4

    # Phase 5: Multi-resolution exploration
    if tl() > 50:
        n_cand = []
        for mult in [2, 3]:
            n_cand.append(n * mult)
        for off in [-40, -20, 20, 40, 60]:
            n_cand.append(n + off)
        n_cand = sorted(set([r for r in n_cand if 60 <= r <= 800 and r != n]))
        for n_try in n_cand:
            if tl() < 30:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t,)}]
            for beta_r in [200, 1000, 4000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_func, h_t, args=(dx_t, beta_r), jac=obj_jac,
                                  method='L-BFGS-B', bounds=bnds_t, constraints=cons_t,
                                  options={'maxiter': 2000, 'ftol': 1e-17})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        cons = cons_t
                        h_t = best_h.copy()
                except Exception:
                    pass

    # Phase 6: Final high-precision L-BFGS-B
    if tl() > 20:
        for beta in [10000, 50000, 200000]:
            if tl() < 12:
                break
            try:
                res = minimize(obj_func, best_h, args=(dx, beta), jac=obj_jac,
                              method='L-BFGS-B', bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Final micro subgradient polish
    if tl() > 8:
        h = best_h.copy()
        for it in range(3000):
            if tl() < 3:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = int(np.argmax(corr))
            grad = grad_corr(h, k_worst, n) * dx
            lr = 0.0015 / (1.0 + 0.005 * it)
            h_new = project_feasible(h - lr * grad, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
            elif it > 500:
                break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
