```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val(h, beta):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    m = np.max(corr)
    shifted = corr - m
    logsw = np.logaddexp.reduce(beta * shifted) / beta  # log(sum(exp(beta*shifted)))/beta
    # Actually: m + log(sum(exp(beta*shifted)))/beta
    logsw = np.log(np.sum(np.exp(beta * shifted))) / beta
    return (m + logsw) * dx

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    # Vectorized gradient
    s = np.arange(-(n-1), n)
    j_idx = np.arange(n)[:, None]
    js = j_idx + s[None, :]
    valid1 = (js >= 0) & (js < n)
    h_js = h[js.clip(0, n-1)]
    term1 = (w[None, :] * valid1 * (1.0 - h_js)).sum(axis=1)
    js2 = j_idx - s[None, :]
    valid2 = (js2 >= 0) & (js2 < n)
    h_js2 = h[js2.clip(0, n-1)]
    term2 = (w[None, :] * valid2 * h_js2).sum(axis=1)
    grad = dx * (term1 - term2)
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def con_f(x):
        return x.sum() - target
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def coord_descent(h, max_sweeps, mu=100.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Minimize over h[j] in [0,1] with penalty for constraint
            def sub_obj(v):
                x[j] = v
                val = smoothmax_val(x, 200)
                penalty = mu * (x.sum() - target) ** 2
                return val + penalty
            # Use scipy minimize_scalar
            from scipy.optimize import minimize_scalar
            res = minimize_scalar(sub_obj, bounds=(0, 1), method='bounded',
                                 options={'xatol': 1e-10})
            x[j] = res.x
            v = compute_c5(x)
            if v < best_v:
                best_v = v
                best_x = x.copy()
            improved = True
        x = project_feasible(x, target)
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def tl():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Progressive SLSQP
    for beta in [30, 60, 100, 200, 400, 800]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
    
    # Phase 2: Coordinate descent
    if tl() > 100:
        h_cd, c_cd = coord_descent(h_best, max_sweeps=20, mu=200.0)
        if c_cd < best_c5:
            h_best, best_c5 = h_cd, c_cd
    
    # Phase 3: SLSQP high beta
    for beta in [600, 1200, 2000]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=400, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
    
    # Phase 4: Shake and re-optimize
    for trial in range(10):
        if tl() < 45:
            break
        scale = 0.01 * (1.0 + 0.3 * trial)
        n = len(h_best)
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = slsqp_refine(h_pert, maxiter=400, beta=500)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
            h_pert = h_best.copy()  # warm start for next
    
    # Phase 5: Multi-resolution
    for n_try in [200, 300, 500, 600]:
        if tl() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [80, 300, 1000]:
            if tl() < 50:

            h_new = slsqp_refine(h_interp, maxiter=300, beta=beta)
            c = compute_c5(h_new)

                h_interp = h_best.copy()

        h_final = slsqp_refine(h_best, maxiter=600, beta=3000)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c
    
    # Final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    
    return h_best, best_c5, len(h_best)
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    s = np.arange(-(n - 1), n)
    j = np.arange(n)[:, None]
    js = j + s[None, :]
    v1 = (js >= 0) & (js < n)
    h1 = h[js.clip(0, n - 1)]
    t1 = (w[None, :] * v1 * (1.0 - h1)).sum(axis=1)
    js2 = j - s[None, :]
    v2 = (js2 >= 0) & (js2 < n)
    h2 = h[js2.clip(0, n - 1)]
    t2 = (w[None, :] * v2 * h2).sum(axis=1)
    return val, dx * (t1 - t2)

def lbfgsb_refine(h, maxiter, beta, mu):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    ones = np.ones(n)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v + mu * (x.sum() - target) ** 2
    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g + 2.0 * mu * (x.sum() - target) * ones
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    ones = np.ones(n)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: ones}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    return project_feasible(np.clip(f(x_new), 0.0, 1.0), n_new / 2.0)
```
