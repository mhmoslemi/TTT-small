    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sh_max = sh.max()
    return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha

def smooth_grad(h, n, alpha, eps=1e-6):
    """Numerical gradient of the smooth objective"""
    g = compute_all_g(h, n)
    m = g.max()
    sh = alpha * (g - m)
    sh_max = sh.max()
    w = np.exp(sh - sh_max)
    w /= w.sum()
    # Gradient = sum_k w_k * grad_k
    # grad_k[i] = dx * d/dh[i] [sum_j h[j]*(1-h[j+d(k)))]
    # For efficiency, compute via finite difference of the weighted sum
    val = m + (sh_max + np.log(w.sum())) / alpha  # wait, w already normalized

    grad = np.zeros(n)

        grad[i] = (smooth_obj(h, n, alpha) - base) / eps

        return m + (sh_max + np.log(np.exp(sh - sh_max).sum())) / alpha

        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return h

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    n = len(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m = g.max()

            grad[i] = (np.correlate(h, 1-h, 'full')[k_star] * (2.0/n) - g[k_star]) / eps
            h[i] -= eps

def pair_refine(h, n, deadline, deltas):
    """Pair local search"""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        progress = True
        while progress and time.time() < deadline:
            progress = False
            for i in range(n):
                if time.time() > deadline:

                for j in range(n-1, i, -1):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d

                                progress = True
                            else:

    return h, best_c
