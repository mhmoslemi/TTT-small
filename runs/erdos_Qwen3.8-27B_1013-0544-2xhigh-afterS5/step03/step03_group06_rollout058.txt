import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode="full")
    z = raw * dx
    m = np.max(z)
    sh = np.clip(beta * (z - m), -500, 500)
    es = np.exp(sh)
    se = np.sum(es)
    val = m + np.log(se) / beta
    sw = es / se
    grad = np.zeros(n)
    w0 = sw[n - 1] * dx
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    for a in range(1, n):
        wa = (sw[n - 1 + a] + sw[n - 1 - a]) * dx
        if abs(wa) < 1e-15:
            continue
        grad[a:] += wa * (1.0 - h[:n - a])
        grad[:n - a] += wa * (-h[a:])
    return val, grad


def make_obj(beta):
    return lambda h: obj_grad(h, beta)[0]


def make_jac(beta):
    return lambda h: obj_grad(h, beta)[1]


def make_con(target):
    return lambda h: h.sum() - target


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytical gradient + equality constraint
    n = len(best_h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    con = {'type': 'eq', 'fun': make_con(target), 'jac': lambda x: np.ones(x.shape)}

    for beta in [5, 15, 50, 150, 500, 1500, 5000, 15000]:
        if tl() < 30:
            break
        h_s = project_h(best_h, target)
        try:
            res = minimize(make_obj(beta), h_s, jac=make_jac(beta),
                           method='SLSQP', bounds=bounds,
                           constraints=[con],
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Multi-resolution with L-BFGS-B + projection
    for n_new in [250, 300, 350, 400]:
        if tl() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_new)

                found = True

        top_idx = np.argsort(corr)[::-1][:4]

            lag = int(idx) - (n_f - 1)
            if lag == 0:

                    h_t = best_h.copy()
                    h_t[i] += d

                    if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                        h_t = project_h(h_t, t_f)
                        c = compute_c5(h_t)
                        if c < best_c5:
                            best_h = h_t
                            best_c5 = c
                            imp = True

        if not imp and it % 3000 == 2999:

            c = compute_c5(h_t)

    # Phase 6: Final SLSQP polish
    if tl() > 5:
        n_p = len(best_h)
        t_p = n_p / 2.0
        bnd = Bounds(np.zeros(n_p), np.ones(n_p))

                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
