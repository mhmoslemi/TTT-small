def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def get_critical_lags(h, k=10):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.argsort(corr)[::-1][:k], corr

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    # Get lags within epsilon of max, top n_active_max
    thresh = max_val - epsilon
    active = np.where(corr >= thresh)[0]
    if len(active) > n_active_max:
        # Keep only the top n_active_max
        top_idx = np.argsort(corr[active])[::-1][:n_active_max]
        active = active[top_idx]
    
    grad = np.zeros(n)
    for m in active:
        lag = int(m) - (n - 1)
        # First term: (1-h[i-lag]) for valid i-lag
        if lag >= 0:
            # i ranges from lag to n-1
            grad[lag:] += (1.0 - h[:n-lag])
        else:
            # i ranges from 0 to n-1+lag
            grad[:n+lag] += (1.0 - h[-lag:])
        # Second term: -h[i+lag] for valid i+lag
        if lag >= 0:
            # i ranges from 0 to n-1-lag
            grad[:n-lag] -= h[lag:]
        else:
            # i ranges from -lag to n-1
            grad[-lag:] -= h[:n+lag]
    
    # Negative gradient (we want to decrease the max)
    return -grad / len(active)

            if -m > 0:

    grad /= len(active)
    
    # Update momentum
    momentum_vec = momentum_rate * momentum_vec + (1 - momentum_rate) * (-grad)
    
    return h, momentum_vec, max_val
