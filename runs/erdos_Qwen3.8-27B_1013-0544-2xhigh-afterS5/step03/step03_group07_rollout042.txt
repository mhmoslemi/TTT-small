import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_n = 0
_g_beta = 0.0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += d2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def _sm_compute(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = corr - m
    mk = np.max(sh)
    e = np.exp(_g_beta * (sh - mk))
    se = e.sum()
    w = e / se
    val = (m + mk + np.log(se)) / _g_beta * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * (1.0 - x[s:])
            grad[s:] += wk * (-x[:n-s])
        else:
            t = -s
            grad[t:] += wk * (1.0 - x[:n-t])
            grad[:n-t] += wk * (-x[t:])
    grad *= dx
    return val, grad

def _sm_f(x):
    return _sm_compute(x)[0]

def _sm_g(x):
    return _sm_compute(x)[1]

def _cf(x):
    return x.sum() - _g_target

def _cg(x):
    return np.ones(_g_n)

def slsqp(h, beta, maxiter=300):
    global _g_n, _g_beta, _g_target
    n = len(h)
    _g_n = n
    _g_beta = beta
    _g_target = n / 2.0
    try:
        res = minimize(_sm_f, h, jac=_sm_g, method='SLSQP',
                       bounds=Bounds(np.zeros(n), np.ones(n)),
                       constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cg}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, _g_target)
    except Exception:
        return project_feasible(h.copy(), _g_target)

def sg_step(h, lr, top_k, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    ti = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in ti:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            t = -s
            grad[t:] += (1.0 - h[:n-t])
            grad[:n-t] += (-h[t:])
    return project_feasible(h - lr * grad * dx / top_k, target)

def nesterov_loop(h, max_iters, lr0, decay, top_k, target):
    n = len(h)
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        ti = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in ti:
            s = idx - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - y[s:])
                grad[s:] += (-y[:n-s])
            else:
                t2 = -s
                grad[t2:] += (1.0 - y[:n-t2])
                grad[:n-t2] += (-y[t2:])
        grad *= (2.0 / n) / top_k
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

        h_best = np.array(initial_h_values, dtype=float).copy()
    else:

    for beta in [50, 100, 200, 500, 1000, 2000]:

        h = slsqp(h_best, beta, maxiter=300)
        c = compute_c5(h)
        if c < best_c5:

    # Phase 2: Nesterov subgradient with adaptive top-k
    if rem() > 200:

        if v < best_c5:

        h = slsqp(h_best, 1000, 250)
        c = compute_c5(h)
        if c < best_c5:
            h_best, best_c5 = h, c

    # Phase 3: Diversified restarts
    for trial in range(25):
        if rem() < 70:
            break
        ptype = trial % 5
        if ptype == 0:
            h_p = project_feasible(h_best + rng.randn(n) * 0.012, target)
        elif ptype == 1:
            h_p = h_best.copy()
            idx = rng.choice(n, size=n // 5, replace=False)
            h_p[idx] = (h_p[idx] + rng.randn(len(idx)) * 0.08).clip(0, 1)
            h_p = project_feasible(h_p, target)
        elif ptype == 2:
            h_p = h_best.copy()
            idx = rng.choice(n, size=n // 8, replace=False)
            h_p[idx] = (h_p[idx] > 0.5).astype(float)
            h_p = project_feasible(h_p, target)
        elif ptype == 3:
            shift = rng.randint(1, n // 10)
            h_p = np.roll(h_best, shift)
            h_p = project_feasible(h_p, target)
        else:
            h_p = project_feasible(h_best + rng.randn(n) * 0.025, target)

        h_new = slsqp(h_p, 800, 250)
