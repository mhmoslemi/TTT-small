from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, penalty=1000.0):
    """Smoothmax of correlations with sum constraint penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # smoothmax with subtracted max for stability
    m = np.max(corr)
    scaled = beta * (corr - m)
    # clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    exps = np.exp(scaled)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    # Add penalty for sum constraint
    s = h.sum() - n / 2.0
    val += penalty * s * s
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s_lag = k - n + 1
        # First term: (1 - h[i - s_lag]) for valid i
        i_lo = max(0, s_lag)
        i_hi = min(n, n + s_lag)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s_lag:i_hi - s_lag])
        # Second term: -h[i + s_lag] for valid i
        i_lo2 = max(0, -s_lag)
        i_hi2 = min(n, n - s_lag)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s_lag:i_hi2 + s_lag])
    
    # Penalty gradient
    grad += 2 * penalty * s
    
    return val, grad

def lbfgsb_solve(h, beta, maxiter, penalty=1000.0):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta, penalty)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta, penalty)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lp_solve(h_cur):
    n = len(h_cur)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_rows = np.empty((n_corr, n + 1))
    b_rows = np.empty(n_corr)
    for k in range(n_corr):
        s_lag = k - (n - 1)
        g = np.zeros(n)
        i_lo = max(0, s_lag)
        i_hi = min(n, n + s_lag)
        if i_hi > i_lo:
            g[i_lo:i_hi] += (1.0 - h_cur[i_lo - s_lag:i_hi - s_lag])
        i_lo2 = max(0, -s_lag)
        i_hi2 = min(n, n - s_lag)
        if i_hi2 > i_lo2:
            g[i_lo2:i_hi2] += (-h_cur[i_lo2 + s_lag:i_hi2 + s_lag])
        A_rows[k, :n] = dx * g
        A_rows[k, n] = -1.0
        b_rows[k] = -corr[k] * dx + dx * np.dot(g, h_cur)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_rows, b_ub=b_rows, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_feasible(res.x[:n], n / 2.0)
    except Exception:
        pass
    return project_feasible(h_cur, n / 2.0)

def lbfgsb_optimize(h, beta, maxiter, penalty):
    n = len(h)
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        scaled = np.clip(beta * (corr - m), -100, 100)
        exps = np.exp(scaled)
        sw = exps.sum()
        w = exps / sw
        val = (m + np.log(sw) / beta) * dx
        s = x.sum() - n / 2.0
        val += 0.5 * penalty * s * s
        return val

    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        scaled = np.clip(beta * (corr - m), -100, 100)
        exps = np.exp(scaled)
        sw = exps.sum()
        w = exps / sw
        val = (m + np.log(sw) / beta) * dx
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k] * dx
            if wk < 1e-16:
                continue
            s_lag = k - n + 1
            i_lo = max(0, s_lag)
            i_hi = min(n, n + s_lag)
            if i_hi > i_lo:
                grad[i_lo:i_hi] += wk * (1.0 - x[i_lo - s_lag:i_hi - s_lag])
            i_lo2 = max(0, -s_lag)
            i_hi2 = min(n, n - s_lag)
            if i_hi2 > i_lo2:
                grad[i_lo2:i_hi2] += wk * (-x[i_lo2 + s_lag:i_hi2 + s_lag])
        s = x.sum() - n / 2.0
        val += 0.5 * penalty * s * s
        grad += penalty * s
        return val

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def basin_hop_lp(h_best, best_c5, n_solves, noises, seed, t0, budget):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    for noise in noises:
        if budget - (time.time() - t0) < 40:
            break
        for _ in range(n_solves):
            if budget - (time.time() - t0) < 30:
                break
            h_p = h_best + rng.randn(n) * noise
            h_p = project_feasible(h_p, n / 2.0)
            for _ in range(30):
                h_p = lp_solve(h_p)
            c = compute_c5(h_p)
            if c < best_c5:
                best_c5 = c
                h_best = h_p.copy()
    return h_best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: Successive LP iterations
    for it in range(2000):
        if rem() < 30:
            break
        h_new = lp_solve(h_cur)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            h_best = h_new.copy()
        if abs(compute_c5(h_cur) - c) < 1e-12:
            h_cur = h_new
            break
        h_cur = h_new

    # Phase 2: Basin hopping with LP at current resolution
    h_best, best_c5 = basin_hop_lp(
        h_best, best_c5, n_solves=4,
        noises=[0.03, 0.02, 0.012, 0.008, 0.005, 0.003],
        seed=seed + 1, t0=t0, budget=budget_s)

    # Phase 3: L-BFGS-B refinement (different optimizer for diversity)
    n_cur = len(h_best)
    for beta in [50, 200, 500]:
        if rem() < 50:
            break
        h_lb = lbfgsb_optimize(h_best, beta, maxiter=400, penalty=5000.0)
        for _ in range(50):
            h_lb = lp_solve(h_lb)
        c = compute_c5(h_lb)
        if c < best_c5:
            best_c5 = c
            h_best = h_lb.copy()

    # Phase 4: Multi-resolution
    for n_try in [300, 400, 500, 600]:
        if rem() < 100:
            break
        h_t = interp_to_n(h_best, n_try)
        for _ in range(500):
            if rem() < 40:
                break
            h_new_t = lp_solve(h_t)
            if abs(compute_c5(h_t) - compute_c5(h_new_t)) < 1e-11:
                h_t = h_new_t
                break
            h_t = h_new_t
        c = compute_c5(h_t)
        if c < best_c5:
            best_c5 = c
            h_best = h_t.copy()

    # Phase 5: Basin hopping at best resolution (fine)
    h_best, best_c5 = basin_hop_lp(
        h_best, best_c5, n_solves=3,
        noises=[0.01, 0.006, 0.004, 0.002],
        seed=seed + 100, t0=t0, budget=budget_s)

    # Phase 6: L-BFGS-B + LP alternating at best resolution
    if rem() > 60:
        for beta in [200, 500, 1000]:
            if rem() < 40:
                break
            h_alt = lbfgsb_optimize(h_best, beta, maxiter=300, penalty=5000.0)
            for _ in range(100):
                if rem() < 30:
                    break
                h_alt = lp_solve(h_alt)
            c = compute_c5(h_alt)
            if c < best_c5:
                best_c5 = c
                h_best = h_alt.copy()

    # Phase 7: Final LP polish
    if rem() > 25:
        h_fin = h_best.copy()
        for _ in range(500):
            if rem() < 15:
                break
            h_new = lp_solve(h_fin)
            if abs(compute_c5(h_fin) - compute_c5(h_new)) < 1e-12:
                break
            h_fin = h_new
        c = compute_c5(h_fin)
        if c < best_c5:
            best_c5 = c
            h_best = h_fin

    # Final projection
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
