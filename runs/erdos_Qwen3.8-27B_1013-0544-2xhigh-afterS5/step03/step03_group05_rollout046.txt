def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        n_free = free.sum()
        if n_free == 0:
            # All at bounds, need to adjust
            if diff > 0:
                # Need to increase, pick those closest to 1
                idx = np.argsort(h)[-min(int(abs(diff)/0.5)+1, len(h)):]
                h[idx] += diff / len(idx)
            else:
                idx = np.argsort(h)[:min(int(abs(diff)/0.5)+1, len(h)):]
                h[idx] += diff / len(idx)
        else:
            h[free] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx

    return grad

    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

    """Momentum descent with smooth max gradient"""
    cur = h.copy()

    momentum = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nf = free.sum()
        if nf == 0:
            if diff > 0:
                idx = np.argsort(h)[-max(1, int(abs(diff) * 2)):]
            else:
                idx = np.argsort(h)[:max(1, int(abs(diff) * 2)):]
            h[idx] += diff / len(idx)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad


def full_smooth_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-9:
            continue
        lag = k - center
        grad += w * grad_for_lag(h, lag, n, dx)
    return grad


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)


def momentum_descent(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    mom = np.zeros(n)
    mu = 0.65
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        beta = 20.0 + 80.0 * (it / max(1, max_iters))
        beta = min(beta, 400.0)
        grad = full_smooth_grad(cur, n, dx, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        mom = mu * mom + (1.0 - mu) * grad
        mnorm = np.linalg.norm(mom)
        if mnorm > 1e-15:
            mom /= mnorm
        step = 0.04 / (1.0 + 0.005 * it)
        improved = False
        for _ in range(35):
            h_t = project_h(cur - step * mom, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5 - 1e-15:
                cur = h_t
                cur_c5 = c5_t
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                improved = True
                break
            step *= 0.5
        if not improved:
            step = 0.0005
            h_t = project_h(cur - step * mom, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5:
                cur = h_t
                cur_c5 = c5_t
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
            else:
                T = 0.002 * np.exp(-it / 1500.0)
                if T > 1e-6:
                    noise = np.random.normal(0, T, n)
                    h_t = project_h(cur + noise, target)
                    c5_t = compute_c5(h_t)
                    if c5_t < cur_c5 + T:
                        cur = h_t
                        cur_c5 = c5_t
    return cur, cur_c5, n


def slsqp_smooth_obj(h_in, dx, beta):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(c)
    s = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(s)) / len(s)) / beta) * dx


def con_sum_eq(x, target):
    return np.sum(x) - target


def con_sum_eq_jac(x, target):
    return np.ones(x.shape)


def topk_polish(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_idx = np.argsort(corr_val)[::-1][:3]
        vals = corr_val[top_idx]
        wts = np.exp((vals - vals[0]) * 400.0)
        wts /= wts.sum()
        grad = np.zeros(n)
        for ti, w in zip(top_idx, wts):
            lag = int(ti) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        step = 0.003 / (1.0 + 0.001 * it)
        for _ in range(25):
            h_t = project_h(cur - step * grad, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5 - 1e-16:
                cur = h_t
                cur_c5 = c5_t
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
    return cur, cur_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 200
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    # Phase 1: Multi-resolution momentum descent
    n_candidates = [120, 160, 200, 250, 300]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:

            break
        target = n / 2.0
        dx = 2.0 / n

        # Start from resampled parent

        # Also try from current best if different resolution

            if compute_c5(h_alt) < compute_c5(h_init):
                h_init = h_alt

        # Add diverse initializations
        x = np.linspace(0, 2, n)
        diverse_inits = [h_init]
        # Cosine wave init
        for mode in [1, 2, 3]:
            h_c = 0.5 + 0.3 * np.cos(mode * np.pi * x)
            diverse_inits.append(project_h(h_c, target))
        # Two-bump init
        h_b = 0.5 + 0.3 * (np.exp(-((x - 0.4) / 0.25) ** 2) + np.exp(-((x - 1.6) / 0.25) ** 2))
        diverse_inits.append(project_h(h_b, target))

        for h_d in diverse_inits:
            if time_left() < 50:
                break
            cur_c5 = compute_c5(h_d)
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = h_d.copy()
                best_state[2] = n
            # Run momentum descent
            cur, cur_c5, _ = momentum_descent(
                h_d, n, target, dx,
                max_iters=4000, time_fn=time_left, min_time=15,
                best_state=best_state
            )

    # Phase 2: SLSQP refinement at best resolution
    if time_left() > 40:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target = best_n / 2.0
        dx = 2.0 / best_n
        bounds = Bounds(np.zeros(best_n), np.ones(best_n))

            if time_left() < 15:

            obj = partial(slsqp_smooth_obj, dx=dx, beta=beta)

            cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
            try:

                               constraints=cons, options={'maxiter': 4000, 'ftol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_state[0]:
                    best_state[0] = c5_new
                    best_state[1] = h_new
                    best_state[2] = best_n
            except Exception:
                pass

    # Phase 3: Random perturbation restarts
    if time_left() > 60:
        best_h = best_state[1].copy()
        best_n = best_state[2]
        target = best_n / 2.0
        dx = 2.0 / best_n
        for restart in range(8):
            if time_left() < 20:
                break
            scale = 0.05 * (1.0 - restart / 10.0)
            h_pert = project_h(best_h + np.random.normal(0, scale, best_n), target)

                h_pert, best_n, target, dx,
                max_iters=2000, time_fn=time_left, min_time=5,

            )

    # Phase 4: Final top-k subgradient polish
    if time_left() > 10:
        best_state[1], best_state[0], _ = topk_polish(
            best_state[1], best_state[2],
            best_state[2] / 2.0, 2.0 / best_state[2],
            max_iters=30000, time_fn=time_left, min_time=2,
            best_state=best_state
        )

    # Ensure valid output
    h_out = best_state[1]
    h_out = np.clip(h_out, 0.0, 1.0)
    c5_out = compute_c5(h_out)
    return h_out, c5_out, len(h_out)
