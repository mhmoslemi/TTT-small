import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_feasible(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = tau
        else:
            hi = tau
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lags(h, lags, n):
    g = np.zeros(n)
    for lag in lags:
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m = -lag
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g

def smooth_grad(h, corr, beta, dx, top_k=20):
    n = len(h)
    xs = np.clip(beta * (corr - corr.max()), -500, 500)
    w = np.exp(xs)
    w /= w.sum()
    si = np.argsort(w)[::-1]
    cw = np.cumsum(w[si])
    na = min(max(int(np.searchsorted(cw, 0.9999)) + 1, 3), top_k)
    ai = si[:na]
    lags = [int(idx) - (n - 1) for idx in ai]
    g = np.zeros(n)
    for idx, lag in zip(ai, lags):
        wt = w[idx]
        if lag == 0:
            g += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += wt * (1.0 - h[:-lag])
            g[:n - lag] -= wt * h[lag:]
        else:
            m = -lag
            g[:n - m] += wt * (1.0 - h[m:])
            g[m:] -= wt * h[:n - m]
    return g * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-resolution projected gradient with smooth max
    cur_n = len(best_h)
    res_list = []
    nn = cur_n
    for _ in range(7):
        nn = int(nn * 1.25)
        if nn > 700:
            break
        res_list.append(nn)
    if not res_list:
        res_list = [300, 400, 500]

    beta_schedule = [0.005, 0.02, 0.05, 0.1, 0.3, 1.0, 3.0, 10.0, 50.0]

    for n_p in res_list:
        if tl() < 50:
            break
        ts = n_p / 2.0
        dx = 2.0 / n_p
        if n_p == cur_n:
            hc = best_h.copy()
        else:
            xo = np.linspace(0, 2, cur_n)
            xn = np.linspace(0, 2, n_p)
            hc = np.interp(xn, xo, best_h)
            hc = project_feasible(hc, ts)
        cur_n = n_p
        hc5 = compute_c5(hc)

        for bi, beta in enumerate(beta_schedule):
            if tl() < 5:
                break
            remaining_betas = len(beta_schedule) - bi
            n_iter = int(min(2000, max(50, tl() * 40.0 / max(remaining_betas, 1))))
            for it in range(n_iter):
                if tl() < 3:
                    break
                corr = np.correlate(hc, 1.0 - hc, mode="full")
                g = smooth_grad(hc, corr, beta, dx, top_k=25)
                gn = np.linalg.norm(g)
                if gn < 1e-14:
                    break
                step = 1.0 / (gn + 1e-10)
                step = min(step, 2.0)
                ok = False
                for _ in range(30):
                    ht = project_feasible(hc - step * g, ts)
                    c5t = compute_c5(ht)
                    if c5t < hc5 - 1e-15:
                        hc, hc5 = ht, c5t
                        ok = True
                        break
                    step *= 0.4
                if not ok and it > 100:
                    break
        if hc5 < best_c5:
            best_h, best_c5 = hc.copy(), hc5

    # Phase 2: Subgradient descent with top lags
    nf = len(best_h)
    tf = nf / 2.0
    no_imp = 0
    for it in range(100000):
        if tl() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        ti = np.argsort(corr)[::-1][:7]
        tlags = [int(i) - (nf - 1) for i in ti]
        g = subgrad_for_lags(best_h, tlags, nf)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        ok = False
        for step in [0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.001, 0.0005]:
            ht = project_feasible(best_h - step * g, tf)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_h, best_c5, no_imp = ht, c5t, 0
                ok = True
                break
        if not ok:
            no_imp += 1
            if no_imp > 1000 and it % 500 == 499:
                sig = 0.005 + 0.01 * (it % 3)
                ht = project_feasible(best_h + np.random.randn(nf) * sig, tf)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5, no_imp = ht, c5t, 0
            if no_imp > 15000:
                break

    # Phase 3: Pairwise coordinate moves on critical lags
    for it in range(200000):
        if tl() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        ti = np.argsort(corr)[::-1][:5]
        imp = False
        for idx in ti:
            if tl() < 1:
                break
            lag = int(idx) - (nf - 1)
            if lag == 0:
                cands = np.argsort(np.abs(best_h - 0.5))[-10:]
                for j in cands:
                    for d in [0.012, -0.012, 0.006, -0.006, 0.003, -0.003]:
                        ht = best_h.copy()
                        ht[j] += d
                        if 0 <= ht[j] <= 1:
                            ht = project_feasible(ht, tf)
                            c5t = compute_c5(ht)
                            if c5t < best_c5 - 1e-15:
                                best_h, best_c5 = ht, c5t
                                imp = True
                                break
                    if imp:
                        break
            else:
                al = abs(lag)
                if al >= nf:
                    continue
                nv = nf - al
                for _ in range(12):
                    i = np.random.randint(nv)
                    j = i + al if lag > 0 else i - lag
                    for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                        ht = best_h.copy()
                        ht[i] += d
                        ht[j] -= d
                        if 0 <= ht[i] <= 1 and 0 <= ht[j] <= 1:
                            ht = project_feasible(ht, tf)
                            c5t = compute_c5(ht)
                            if c5t < best_c5 - 1e-15:
                                best_h, best_c5 = ht, c5t
                                imp = True
                                break
                    if imp:
                        break
            if imp:
                break
        if not imp and it % 3000 == 2999:
            ht = project_feasible(best_h + np.random.randn(nf) * 0.004, tf)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 4: SLSQP polish with smooth objective
    if tl() > 20:
        np_ = len(best_h)
        tp = np_ / 2.0
        dxp = 2.0 / np_
        bp = Bounds(np.zeros(np_), np.ones(np_))

        def _obj(hi):
            return smooth_max_val(np.correlate(hi, 1.0 - hi, mode="full"), 3.0) * dxp

        def _jac(hi):
            return smooth_grad(hi, np.correlate(hi, 1.0 - hi, mode="full"), 3.0, dxp, top_k=40)

        def _con(xi):
            return np.sum(xi) - tp

        def _conjac(xi):
            return np.ones(xi.shape)

        try:
            r = minimize(_obj, best_h, method='SLSQP', bounds=bp,
                         constraints=[{'type': 'eq', 'fun': _con, 'jac': _conjac}],
                         jac=_jac, options={'maxiter': 5000, 'ftol': 1e-17})
            hn = project_feasible(r.x, tp)
            c5n = compute_c5(hn)
            if c5n < best_c5:
                best_h, best_c5 = hn, c5n
        except Exception:
            pass

    # Phase 5: Final intensive subgradient sweep
    for it in range(30000):
        if tl() < 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        ti = np.argsort(corr)[::-1][:10]
        tlags = [int(i) - (nf - 1) for i in ti]
        g = subgrad_for_lags(best_h, tlags, nf)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        for step in [0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]:
            ht = project_feasible(best_h - step * g, tf)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_h, best_c5 = ht, c5t
                break

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
