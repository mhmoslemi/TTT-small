    sig = np.where(weights > 1e-8)[0]
    grad = np.zeros(n)
    for k in sig:
        lag = k - (n - 1)
        if lag >= 0:
            m_idx = n - lag
            idx = np.arange(m_idx)
            grad[idx] += weights[k] * (1.0 - h[idx + lag])
            grad[idx + lag] -= weights[k] * h[idx]
        else:
            la = -lag
            m_idx = n - la
            idx = np.arange(m_idx)
            grad[idx + la] += weights[k] * (1.0 - h[idx])
            grad[idx] -= weights[k] * h[idx + la]
    return grad

Now let me think about the overall algorithm more carefully. The key question is: what's the actual bottleneck preventing us from getting below 0.3808?

I think the issue is:
1. The landscape is very flat near the optimum (many lags are close to the max)
2. Standard optimization methods get stuck in slightly suboptimal local minima
3. We need either better global exploration or a more targeted local optimization

**New idea: Use "active set" method**

At the optimum, suppose the top m lags are active (all equal to the max). We can:
1. Identify these m lags
2. Set up the system: for each active lag k_j, the gradient of Q_{k_j} should be proportional to the constraint gradient (all ones)
3. This gives us m equations in n unknowns (with n/2 - 1 free variables after the sum constraint)

But this is over-determined unless m ≤ n/2 - 1.

Actually, the KKT conditions are:
∇f(h*) + λ * 1 + μ = 0

where f is the max (smoothed), λ is the Lagrange multiplier for the sum constraint, and μ is for the box constraints (non-zero only at active bounds).

At the optimum, the subdifferential of the max contains the weighted average of gradients of active lags. So:
Σ_j w_j * ∇Q_{k_j}(h*) = -λ * 1 - μ

This is hard to solve directly, but it gives us a direction for improvement.

**Practical approach**: I'll use a hybrid method:
1. Softmax subgradient for descent (with adaptive temperature)
2. Periodic SLSQP refinement
3. Multi-resolution search
4. Random restarts from perturbed solutions

Let me also consider using **cvxpy** for a convex relaxation. For fixed active lags, the problem:
minimize t
s.t. Q_k(h) ≤ t for k in K
sum(h) = n/2
0 ≤ h ≤ 1

is non-convex (quadratic constraints). But if we approximate Q_k(h) ≤ t by a linear constraint (first-order Taylor at current iterate), we get a cutting plane / SPO+ approach.

Actually, for a specific lag k, Q_k(h) = Σᵢ h[i]*(1-h[i+k]) is a bilinear form. The set {h : Q_k(h) ≤ t} is non-convex. So CVXPY won't directly help.

For the L-BFGS-B approach:

This should be faster than SLSQP because L-BFGS-B uses a quasi-Newton method with finite memory, while SLSQP solves a QP subproblem at each iteration.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_subgrad(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m <= 0:
            return grad
        idx = np.arange(m)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad

def softmax_subgrad(h, corr, n, beta_sm):
    m = np.max(corr)
    shifted = np.clip(beta_sm * (corr - m), -700.0, 700.0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return lag_subgrad(h, k_max - (n - 1), n)
    weights /= total
    sig = np.where(weights > 1e-8)[0]
    grad = np.zeros(n)
    for k in sig:
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += weights[k] * (1.0 - h[idx + lag])
            grad[idx + lag] -= weights[k] * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += weights[k] * (1.0 - h[idx])
            grad[idx] -= weights[k] * h[idx + la]
    return grad

def obj_smooth(h_x, beta, mu, ts):
    corr = np.correlate(h_x, 1.0 - h_x, mode='full')
    sm = smooth_max_val(corr, beta) * (2.0 / len(h_x))
    pen = mu * (h_x.sum() - ts) ** 2
    return sm + pen

def obj_smooth_grad(h_x, beta, mu, ts):
    n = len(h_x)
    dx = 2.0 / n
    corr = np.correlate(h_x, 1.0 - h_x, mode='full')
    grad = softmax_subgrad(h_x, corr, n, beta)
    grad *= dx
    grad += 2.0 * mu * (h_x.sum() - ts) * np.ones(n)
    return grad

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

        return budget_s - (time.time() - t0)

        h_init = np.array(initial_h_values, dtype=float).copy()
    else:

    best_h = project_feasible(h_init.copy(), len(h_init) / 2.0)
    best_c5 = compute_c5(best_h)

    if tl() > 200:
        h = best_h.copy()
        n = len(h)
        ts = n / 2.0
        velocity = np.zeros(n)
        beta_mom = 0.88
        for it in range(30000):
            if tl() < 180:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            beta_sm = 30.0 + it * 0.05
            grad = softmax_subgrad(h, corr, n, beta_sm)
            lr = 0.006 / (1.0 + 0.0002 * it)
            velocity = beta_mom * velocity + (1.0 - beta_mom) * (-lr * grad)
            h_new = project_feasible(h + velocity, ts)
            c_new = compute_c5(h_new)
            if c_new < best_c5 + 0.0015:
                h = h_new
                if c_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c_new
            else:
                velocity *= 0.4
                lr *= 0.7

    # Phase 2: L-BFGS-B with penalty (faster than SLSQP)
    if tl() > 150:
        n = len(best_h)
        ts = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        for beta in [100, 500, 2000]:
            if tl() < 40:
                break
            for mu in [100, 1000, 10000]:
                if tl() < 25:
                    break
                x0 = best_h.copy()
                try:
                    res = minimize(obj_smooth, x0, args=(beta, mu, ts),
                                   jac=lambda x, b, m, t: obj_smooth_grad(x, b, m, t),
                                   args=(beta, mu, ts), method='L-BFGS-B',
                                   bounds=bnds,
                                   options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:

                        best_c5 = c5
                except Exception:
                    pass

                                   x0, method='SLSQP', bounds=bnds, constraints=cons,

                    h_new = project_feasible(res.x, ts)
