    H = np.fft.fft(h_pad)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lags(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        elif -n < lag < 0:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def spectral_perturb(h, scale):
    n = len(h)
    h_pad = np.zeros(2 * n)
    h_pad[:n] = h
    H = np.fft.rfft(h_pad)
    phase = np.angle(H)
    mag = np.abs(H).copy()
    mag += np.random.randn(len(mag)) * scale
    mag = np.maximum(0, mag)
    H_new = mag * np.exp(1j * phase)
    h_new = np.fft.irfft(H_new, 2 * n)[:n]
    return h_new

def slsqp_pass(h, n, target_sum, dx, beta, maxiter=5000, time_limit_s=20):
    t0 = time.time()
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]

        if time.time() - t0 > time_limit_s:
            return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
