```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_target = 0.0
_g_dx = 0.0
_g_beta = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -50, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgrad_h(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for li in top_idx:
        d = int(li) - (n - 1)
        if d == 0:
            grad += 1.0 - 2.0 * h
        elif d > 0:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
        else:
            dd = -d
            grad[:n - dd] += (1.0 - h[dd:])
            grad[dd:] -= h[:n - dd]
    return grad / top_k

def _obj_smooth(x):
    return smooth_max(np.correlate(x, 1.0 - x, mode="full"), _g_beta) * _g_dx

def _con_eq(x):
    return np.sum(x) - _g_target

def _con_eq_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, target, dx, time_limit, beta_start=500):
    n = len(h)
    global _g_target, _g_dx, _g_beta
    _g_target = target
    _g_dx = dx
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    t0 = time.time()
    cur = project_h(h, target)
    cur_c5 = compute_c5(cur)
    betas = [beta_start, beta_start * 5, beta_start * 20, beta_start * 80, beta_start * 200]
    for b in betas:
        if time.time() - t0 > time_limit:
            break
        _g_beta = float(b)
        try:
            res = minimize(_obj_smooth, cur, method='SLSQP', bounds=bounds,
                          constraints=constr, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-15:
                cur = h_new
                cur_c5 = c5_new
        except Exception:
            pass
    return cur, cur_c5

def subgrad_optimize(h, target, time_limit, top_k=5, max_iters=8000):
    t0 = time.time()
    n = len(h)
    cur = project_h(h, target).copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    step = 0.05
    momentum = np.zeros(n)
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        g = subgrad_h(cur, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        direction = -g / gn
        momentum = 0.7 * momentum + 0.3 * direction
        mn = np.linalg.norm(momentum)
        if mn > 1e-14:
            momentum = momentum / mn
        improved = False
        for s in [step, step * 0.5, step * 0.2, step * 2.0, step * 0.8]:
            h_trial = project_h(cur + s * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                step = s
                improved = True
                if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
                break
        if not improved:
            step *= 0.5
            momentum *= 0.5
            if step < 1e-7:
                break
    return best, best_c5

def perturb_h(h, target, rng, scale=0.02):
    noise = rng.normal(0, scale, len(h))
    return project_h(h + noise, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_parent)
    else:
        n_base = 280
        h_parent = np.full(n_base, 0.5)

    target = n_base / 2.0
    dx = 2.0 / n_base
    best_h = project_h(h_parent, target)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from parent (150s)
    if budget_s - elapsed() > 160:
        h1, c1 = slsqp_refine(best_h, target, dx, time_limit=100, beta_start=500)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Subgradient (120s)
    if budget_s - elapsed() > 130:
        h2, c2 = subgrad_optimize(best_h, target, time_limit=80, top_k=3)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP restart from perturbed points (120s)
    if budget_s - elapsed() > 130:
        for pert in range(3):
            if budget_s - elapsed() < 40:
                break
            h_pert = perturb_h(best_h, target, rng, scale=0.015)
            h3, c3 = slsqp_refine(h_pert, target, dx, time_limit=30, beta_start=1000)
            if c3 < best_c5:
                best_h, best_c5 = h3, c3

    # Phase 4: Multiple resolutions (350s)
    n_candidates = [int(n_base * m) for m in [0.65, 0.75, 0.85, 0.9, 1.1, 1.25, 1.4]]
    n_candidates = sorted(set(n for n in n_candidates if 60 <= n <= 500 and n != n_base))

    for nc in n_candidates:
        if budget_s - elapsed() < 40:
            break
        tgt = nc / 2.0
        dxn = 2.0 / nc
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, nc)
        h_cur = project_h(np.interp(x_new, x_old, best_h), tgt)
        t_alloc = min(45.0, (budget_s - elapsed()) / 2.0)
        h_cur, c_cur = slsqp_refine(h_cur, tgt, dxn, time_limit=t_alloc * 0.6, beta_start=1000)
        if budget_s - elapsed() > 20:
            h_cur, c_cur = subgrad_optimize(h_cur, tgt, time_limit=min(15, budget_s - elapsed() - 5), top_k=3)
        if c_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c_cur

    # Phase 5: Final intensive polish (remaining time)
    n_fin = len(best_h)
    tgt_fin = n_fin / 2.0
    dx_fin = 2.0 / n_fin
    remaining = budget_s - elapsed()

    if remaining > 60:
        h5, c5 = slsqp_refine(best_h, tgt_fin, dx_fin, time_limit=remaining * 0.4, beta_start=3000)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    remaining = budget_s - elapsed()
    if remaining > 30:
        h6, c6 = subgrad_optimize(best_h, tgt_fin, time_limit=min(remaining * 0.4, 50), top_k=3)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    remaining = budget_s - elapsed()
    if remaining > 20:
        h7, c7 = slsqp_refine(best_h, tgt_fin, dx_fin, time_limit=min(remaining * 0.5, 40), beta_start=10000)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    # Ensure feasibility
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(len(best_h))
```
