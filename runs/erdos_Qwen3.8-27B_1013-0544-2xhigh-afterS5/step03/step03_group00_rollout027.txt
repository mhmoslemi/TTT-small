import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + d / len(h), 0, 1)
        else:
            h[free] += d / free.sum()
            h = np.clip(h, 0, 1)
    return h

def to_binary(h, seed=0):
    n = len(h)
    target = n // 2
    rng = np.random.RandomState(seed)
    order = np.argsort(-(h + rng.rand(n) * 1e-9))
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb

def sa_binary(h0, t_start, t_end, n_iter, seed=0, t_check=None, t_budget=None):
    rng = np.random.RandomState(seed)
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    c_cur = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    c_best = c_cur
    h_best = h.copy()
    for it in range(n_iter):
        if t_check is not None and (it % 5000 == 0) and t_budget - (time.time() - t_check) < 10:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c_new = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
        if c_new < c_cur or rng.rand() < np.exp(-(c_new - c_cur) / max(t, 1e-12)):
            c_cur = c_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c_cur < c_best:
                c_best = c_cur
                h_best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return h_best, c_best

def greedy_binary(h0, n_trials=50000, batch=25, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    c_cur = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    c_best = c_cur
    h_best = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_trials):
        found = False
        for _ in range(batch):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            c_new = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
            if c_new < c_cur:
                c_cur = c_new
                found = True
                if c_cur < c_best:
                    c_best = c_cur
                    h_best = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break
            h[i], h[j] = 1.0, 0.0
        if not found:
            break
    return h_best, c_best

def smoothmax_c5_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(len(corr))
    cs = corr[idx]
    m = cs.max()
    e = np.exp(beta * (cs - m))
    es = e.sum()
    w = e / es
    val = (m + np.log(es) / beta) * dx
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-14:
            continue
        k = idx[ii]
        s = k - n + 1
        lo1, hi1 = max(0, s), min(n, n + s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - s:hi1 - s])
        lo2, hi2 = max(0, -s), min(n, n - s)
        if hi2 > lo2:
            grad[lo2:hi2] += wk * (-h[lo2 + s:hi2 + s])
    grad *= dx
    return val, grad

def slsqp_step(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0
    def f(x):
        return smoothmax_c5_grad(x, beta, top_k)[0]
    def g(x):
        return smoothmax_c5_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)})
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(200, 0.5)
    n = len(h0)
    h0 = project_feasible(h0, n / 2.0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Binary conversion and greedy search
    h_bin = to_binary(h0, seed)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()

    h_g, c_g = greedy_binary(h_bin, 30000, 25, seed + 1)
    if c_g < best_c5:
        best_c5 = c_g
        best_h = h_g.copy()

    # Phase 2: Simulated annealing (main search)
    if rem() > 250:
        n_iter = int(min(rem() * 0.25, 400)) * 5000
        n_iter = max(n_iter, 200000)
        h_sa, c_sa = sa_binary(h_g, 0.04, 0.0005, n_iter, seed + 2,
                               t_check=t0, t_budget=time.time() + rem() * 0.25)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa.copy()
        h_g2, c_g2 = greedy_binary(h_sa, 30000, 25, seed + 3)
        if c_g2 < best_c5:
            best_c5 = c_g2
            best_h = h_g2.copy()

    # Phase 3: Second SA round with different parameters
    if rem() > 150:
        n_iter = int(min(rem() * 0.2, 300)) * 5000
        n_iter = max(n_iter, 100000)
        h_sa2, c_sa2 = sa_binary(best_h, 0.02, 0.0003, n_iter, seed + 10,
                                 t_check=t0, t_budget=time.time() + rem() * 0.2)
        h_g3, c_g3 = greedy_binary(h_sa2, 30000, 25, seed + 11)
        c_try = min(c_sa2, c_g3)
        h_try = h_sa2 if c_sa2 <= c_g3 else h_g3
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 4: Multi-start SA with random binary initializations
    if rem() > 120:
        for s in range(4):
            if rem() < 40:
                break
            idx = rng.choice(n, size=n // 2, replace=False)
            h_rand = np.zeros(n)
            h_rand[idx] = 1.0
            h_rs, c_rs = sa_binary(h_rand, 0.03, 0.001, 150000, seed + 100 + s)
            h_rg, c_rg = greedy_binary(h_rs, 20000, 20, seed + 200 + s)
            c_try = min(c_rs, c_rg)
            h_try = h_rs if c_rs <= c_rg else h_rg
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()

    # Phase 5: Fractional SLSQP polish from best binary
    if rem() > 80:
        h_frac = best_h.copy()
        for beta in [15, 40, 100, 250]:
            if rem() < 35:
                break
            h_frac = slsqp_step(h_frac, beta, 350, top_k=15)
        c_frac = compute_c5(h_frac)
        if c_frac < best_c5:
            best_c5 = c_frac
            best_h = h_frac.copy()

    # Phase 6: Final high-beta polish
    if rem() > 35:
        h_fin = slsqp_step(best_h, 500, 500, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin

    return best_h, best_c5, len(best_h)
