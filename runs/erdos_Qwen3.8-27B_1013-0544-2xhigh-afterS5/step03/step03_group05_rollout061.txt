def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def compute_C_mat(cur, n):
    center = n - 1
    C_mat = np.zeros((n, 2*n-1))
    for k in range(1, n):
        C_mat[:n-k, center+k] += 1.0 - cur[k:n]
        C_mat[k:, center+k] -= cur[:n-k]
        C_mat[k:, center-k] += 1.0 - cur[:n-k]
        C_mat[:n-k, center-k] -= cur[k:n]
    return C_mat

def coord_descent(h_init, n, n_passes, time_fn, min_time, t_npts=20):
    target = n / 2.0
    center = n - 1
    cur = project_h(h_init.copy(), target)
    cur_c5 = compute_c5(cur)
    best_c5 = cur_c5
    best_h = cur.copy()
    t_grid = np.linspace(0.0, 1.0, t_npts)
    
    for p in range(n_passes):
        if time_fn() < min_time:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full").astype(float)
        C_mat = compute_C_mat(cur, n)

            C_i = C_mat[i]
            best_t = old
            best_obj = np.max(corr)
