    n = len(h)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(y, target_sum):
    """Project y onto {x in [0,1]^n : sum(x) = target_sum} via bisection."""
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n

def smoothmax_grad_fast(h, beta, top_k=None):
    """Fast smoothmax gradient using only top_k lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):
        top_idx = np.argpartition(-corr, top_k)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    w_sel = exps / exps.sum()
    
    val = (m + np.log(exps.sum()) / beta) * dx
    
    grad = np.zeros(n)
    for idx, mi in enumerate(top_idx):
        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        s = mi - (n - 1)  # lag
        # d/dh[j] c(s) = (1-h[j+s])*1_{valid} - h[j-s]*1_{valid}
        j_lo = max(0, -s)
        j_hi = min(n, n - s)

        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2-s:j_hi2-s]
    
    grad *= dx
    return val, grad

from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_polytope(y, target_sum):
    n = len(y)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def smoothmax_grad_fast(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argpartition(-corr, top_k - 1)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    w_sel = exps / exps.sum()
    val = (m + np.log(exps.sum()) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        s = top_idx[idx] - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wk * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wk * h[j_lo2 - s:j_hi2 - s]
    grad *= dx
    return val, grad

        h_new = project_polytope(h - lr * grad, target_sum)
        val_new, _ = smoothmax_grad_fast(h_new, beta, top_k)
        if val_new < val:
            h = h_new
            lr = min(lr * 1.02, 20.0)
        else:
            lr *= 0.6
            if lr < 1e-8:
                break
    return h

    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_polytope(h_new, n_new / 2.0)
