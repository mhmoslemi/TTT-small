        if penalty > 0:
            v += penalty * (np.sum(x) - target)**2
        return v

        if penalty > 0:
            g += 2 * penalty * (np.sum(x) - target)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_corr)
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo+s:i_hi+s])
            grad[i_lo+s:i_hi+s] += wk * (-h[i_lo:i_hi])
    grad *= dx
    return val, grad


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def lbfgs_solve(h, beta, maxiter, top_k=None, penalty=100.0):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        v += penalty * (np.sum(x) - target) ** 2
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        g += 2.0 * penalty * (np.sum(x) - target)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def targeted_descent(h, n_rounds, t0, budget_s):
    """Directly reduce max correlation via coordinate-level pump operations."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    curr = compute_c5(h)
    for _ in range(n_rounds):
        if budget_s - (time.time() - t0) < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_lags = np.argsort(-corr)[:8]
        total_grad = np.zeros(n)
        for k in top_lags:
            s = k - n + 1
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            if i_hi > i_lo:
                total_grad[i_lo:i_hi] += (1.0 - h[i_lo+s:i_hi+s])
                total_grad[i_lo+s:i_hi+s] -= h[i_lo:i_hi]
        total_grad *= dx
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001]:
            h_new = h - alpha * total_grad
            h_new = project_feasible(h_new, target)
            c_new = compute_c5(h_new)
            if c_new < curr:
                h = h_new
                curr = c_new
                break
    return h


def binary_push(h, strength):
    n = len(h)
    target = n / 2.0
    shifted = h - 0.5
    h_new = 0.5 + np.tanh(strength * shifted)
    return project_feasible(h_new, target)


def basin_hop(h_best, best_c5, n_hops, beta_list, maxiter, top_k,
              noise_base, noise_decay, seed, t0, budget_s):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 40:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_local = h_pert
        for beta in beta_list:
            if budget_s - (time.time() - t0) < 25:
                break
            h_local = slsqp_solve(h_local, beta, maxiter, top_k=top_k)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5
