import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def slsqp_softmax(h, n, beta, maxiter=2000):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        bc = np.clip(beta * (corr - m), -400.0, 400.0)
        e = np.exp(bc)
        return (m + np.log(e.sum() / len(corr)) / beta) * dx
    def grad(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        bc = np.clip(beta * (corr - m), -400.0, 400.0)
        e = np.exp(bc)
        w = e / e.sum()
        nc = len(corr)
        g = np.zeros(n)
        for j in range(nc):
            if w[j] < 1e-12:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - x[:-lag])
                g[:n-lag] -= w[j] * x[lag:]
            else:
                m2 = -lag
                g[:n-m2] += w[j] * (1.0 - x[m2:])
                g[m2:] -= w[j] * x[:n-m2]
        return g * dx
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    res = minimize(obj, np.clip(h, 0.0, 1.0), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=[{'type':'eq','fun':eq,'jac':eqj}],
                   options={'maxiter':maxiter,'ftol':1e-14})
    return project_sum(res.x, target)

def subgrad_polish(h, t0, budget, max_iter=10000, tol_frac=0.02, use_momentum=True):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    stall = 0
    g_prev = np.zeros(n)
    for it in range(max_iter):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m = corr.max()
        thresh = m * max(1.0 - tol_frac, 0.001)
        active = np.where(corr >= thresh)[0]
        lags = active - (n - 1)
        g = np.zeros(n)
        for lag in lags:
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n-lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n-m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n-m2]
        if use_momentum and it > 0:
            g = 0.7 * g + 0.3 * g_prev
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        g_prev = g.copy()
        improved = False
        step = 0.06
        for _ in range(10):
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                improved = True
                break
            step *= 0.5
        if not improved:
            stall += 1
            if stall > 80 and it % 40 == 39:
                scale = 0.003 + 0.01 * min(1.0, stall / 500.0)
                pert = np.random.randn(n) * scale
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    stall = 0
            if stall > 3000:
                break
        else:
            if stall > 10:
                tol_frac = max(0.003, tol_frac * 0.95)
            stall = 0
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    # Phase 1: SLSQP progressive beta
    for beta in [80, 400, 2000, 10000, 40000]:
        if tl() < 50:
            break
        hnew = slsqp_softmax(best_h, n0, beta, maxiter=1500)
        c5new = compute_c5(hnew)
        if c5new < best_c5:
            best_h, best_c5 = hnew, c5new
        best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, max_iter=400, tol_frac=0.04)
    # Phase 2: Resolution upgrade
    for ntry in [360, 480, 600]:
        if tl() < 100:
            break
        if ntry <= n0:
            continue
        xo = np.linspace(0.0, 2.0, n0, endpoint=False)
        xn = np.linspace(0.0, 2.0, ntry, endpoint=False)
        hi = project_sum(np.interp(xn, xo, best_h), ntry / 2.0)
        for beta in [100, 600, 3000, 15000]:
            if tl() < 45:
                break
            hnew = slsqp_softmax(hi, ntry, beta, maxiter=1200)
            if compute_c5(hnew) < best_c5:

            hi = hnew
        best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, max_iter=600, tol_frac=0.025)
    # Phase 3: Stochastic restarts
    nb = len(best_h)
    for r in range(5):
        if tl() < 70:
            break
        if r == 0:
            ht = project_sum(best_h + np.random.randn(nb) * 0.015, nb / 2.0)
        elif r == 1:

            ht = hnew
        best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, max_iter=400, tol_frac=0.02)

    best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, max_iter=20000, tol_frac=0.001)
    return best_h, best_c5, len(best_h)
