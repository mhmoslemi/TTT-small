            # Where h[idx+lag] is large, h[idx] contributes more to overlap
            contrib = h[idx] * (1.0 - h[idx + lag])
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            contrib = h[idx + la] * (1.0 - h[idx])
        # Find top contributors and reduce h there slightly
        top_count = max(5, m // 10)
        top_idx = np.argsort(contrib)[-top_count:]
        # Reduce h at those positions, increase at partners
        step = 0.001
        if lag >= 0:
            for i in top_idx:
                partner = i + lag
                if h_mod[i] > step and h_mod[partner] < 1.0 - step:
                    h_mod[i] -= step
                    h_mod[partner] += step
        else:
            la = -lag
            for i in top_idx:
                partner = i + la
                if h_mod[partner] > step and h_mod[i] < 1.0 - step:
                    h_mod[partner] -= step
                    h_mod[i] += step
        h_mod = project_feasible(h_mod, ts)
        c_new = compute_c5(h_mod)
        if c_new < c5 - 1e-14:
            h = h_mod
            c5 = c_new
        else:
            step *= 0.7
            if step < 1e-7:
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Optimize at parent resolution (45% budget)
    d1 = t0 + budget_s * 0.45

    h1 = project_feasible(parent.copy(), pn / 2.0)
    h1, c1 = slsqp_optimize(h1, pn, d1, seed)
    h1, c1 = pg_polish(h1, pn, d1, seed + 1)
    h1, c1 = targeted_reduction(h1, pn, d1, seed + 2)
    update_best(h1, c1, pn)

    # Reversed
    if time.time() < d1:
        hr = project_feasible(parent[::-1].copy(), pn / 2.0)
        hr, cr = slsqp_optimize(hr, pn, d1, seed + 50)
        hr, cr = pg_polish(hr, pn, d1, seed + 51)
        update_best(hr, cr, pn)

    # Small shifts
    for sh in [1, -1, 2]:
        if time.time() > d1:
            break
        hs = np.roll(parent, sh)
        hs = project_feasible(hs, pn / 2.0)
        hs, cs = slsqp_optimize(hs, pn, d1, seed + 100 + sh, maxiter_base=2000)
        hs, cs = pg_polish(hs, pn, d1, seed + 101 + sh)
        update_best(hs, cs, pn)

    # Phase 2: Other resolutions (30% budget)
    d2 = t0 + budget_s * 0.78
    x_old = np.linspace(0, 2, best_n, endpoint=False)

    for n_try in [280, 320, 350, 400, 450, 500, 560]:
        if time.time() > d2:
            break
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_try / 2.0)
        h0, c0 = slsqp_optimize(h0, n_try, d2, seed + n_try, maxiter_base=2500)
        h0, c0 = pg_polish(h0, n_try, d2, seed + n_try + 1)
        h0, c0 = targeted_reduction(h0, n_try, d2, seed + n_try + 2)
        update_best(h0, c0, n_try)

    # Phase 3: High-precision SLSQP on best (12%)
    d3 = t0 + budget_s * 0.90
    if best_h is not None and time.time() < d3:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
        for beta in [100000, 300000, 800000]:
            if time.time() > d3:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 4: Final polish (10%)
    d4 = t0 + budget_s * 0.97
    if best_h is not None and time.time() < d4:
        best_h, best_c5 = pg_polish(best_h, best_n, d4, seed + 999, temp0=0.005)
        best_h, best_c5 = targeted_reduction(best_h, best_n, d4, seed + 998)

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
