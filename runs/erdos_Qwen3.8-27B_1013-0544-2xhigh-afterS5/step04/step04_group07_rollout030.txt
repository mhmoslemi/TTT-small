import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_top_lags(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.01 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > top_k:
        idx = idx[np.argsort(corr[idx])[::-1][:top_k]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    g /= max(len(idx), 1)
    return g, mc, corr

def nesterov_opt(h0, n, target, t0, budget, n_iter=100000):
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = 0.03
    mu = 0.9  # momentum
    stall = 0
    
    for it in range(n_iter):
        if time.time() - t0 > budget - 2:
            break
        top_k = max(10, 50 - it // 500)
        g, mc, _ = grad_top_lags(y, n, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        v = mu * v - step * g
        y_new = y + v
        h_new = project_h(y_new, target)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
            if stall % 500 == 0:
                step *= 0.85
                mu *= 0.95
        y = h_new
        h = h_new
        step *= 0.9998
    
    return best_h, best_c5

    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for start in range(0, n - bs + 1, bs):
            if time.time() - t0 > budget - 1:
                break
            end = start + bs
            # Try all 2^bs binary assignments
            for mask in range(2 ** bs):
                trial = cur.copy()
                for j in range(bs):
                    trial[start + j] = (mask >> j) & 1
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def pair_opt(cur, n, target, n_iters, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(789)
    steps = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]

            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                cur, best_c5 = trial, c
                break
    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        z[zeros[order[:int(diff)]]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:int(-diff)]]] = 0.0
    return z

    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

        return np.sum(x) - target

        return np.ones(n)

                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except:
        return h0

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(150):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lags(h, n, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.012 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > top_k:
        idx = idx[np.argsort(corr[idx])[::-1][:top_k]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    g /= max(len(idx), 1)
    return g, mc

def nesterov_run(h0, n, target, t0, deadline, n_iter=200000, step0=0.035):
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    mu = 0.92
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        tk = max(8, 40 - it // 2000)
        g, _ = grad_lags(y, n, tk)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        v = mu * v - step * g
        h_new = project_h(y + v, target)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5, stall = h_new.copy(), c, 0
        else:
            stall += 1
            if stall % 300 == 0:
                step *= 0.82
                mu = max(0.8, mu * 0.97)
        y = h_new
        step *= 0.9999
    return best_h, best_c5

def block_descent(cur, n, target, bs, sweeps, t0, deadline):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n - bs + 1, bs):
            if time.time() > deadline:
                break
            for mask in range(2 ** bs):
                trial = cur.copy()
                for j in range(bs):
                    trial[start + j] = (mask >> j) & 1
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 4:
            break
    return cur, best_c5

def pair_search(cur, n, target, n_iters, t0, deadline, seed=99):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(seed)
    steps = [0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.015, -0.015]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        j, k = rng.choice(n, 2, replace=False)
        for d in steps:
            if cur[j] + d < -0.05 or cur[j] + d > 1.05 or cur[k] - d < -0.05 or cur[k] - d > 1.05:
                continue
            trial = cur.copy()
            trial[j] += d
            trial[k] -= d
            trial = project_h(trial, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                cur, best_c5 = trial, c
                break
    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 1e-10:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        z[zeros[order[:int(diff)]]] = 1.0
    elif diff < -1e-10:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:int(-diff)]]] = 0.0
    return z

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def slsqp_solve(h0, n, target, beta, maxiter):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        xc = np.clip(beta * (corr - m), -300, 300)
        e = np.exp(xc)
        return (m + np.log(np.sum(e) / len(corr)) / beta) * dx
    def ceq(x):
        return np.sum(x) - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except Exception:
        return h0

def coordinate_refine(cur, n, target, t0, deadline):
    best_c5 = compute_c5(cur)
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()

    best_h = h_parent.copy()

    # Phase 1: Nesterov subgradient (main driver)
    dl = t0 + budget_s * 0.30
    cur, c5 = nesterov_run(best_h, best_n, target, t0, dl, n_iter=200000, step0=0.04)
    if c5 < best_c5:
        best_h, best_c5 = cur, c5

    # Phase 2: Block descent (binary blocks of size 4)
    rem = budget_s - (time.time() - t0)

        dl = t0 + t0 + rem * 0.25

        if c5 < best_c5:

    # Phase 3: SLSQP with moderate smoothing
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur = slsqp_solve(best_h, best_n, target, 800, 2000)
        c5 = compute_c5(cur)
        if c5 < best_c5:

    # Phase 4: Binary snap then re-optimize
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        hb = snap_binary(best_h, target)
        cb = compute_c5(hb)
        if cb < best_c5:
            best_h, best_c5 = hb, cb
        cur = slsqp_solve(best_h, best_n, target, 400, 1500)
        c5 = compute_c5(cur)
        if c5 < best_c5:

    # Phase 5: Pair search (fine-grained)
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        cur, c5 = pair_search(best_h, best_n, target, 100000, t0, t0 + rem * 0.5, seed=789)
        if c5 < best_c5:
            best_h, best_c5 = cur, c5

    # Phase 6: Try higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        for n_hi in [448, 512]:
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            th = n_hi / 2.0
            cur = project_h(resample_h(best_h, n_hi), th)
            c5 = compute_c5(cur)
            if c5 < best_c5:
                best_h, best_c5, best_n = cur, c5, n_hi
            rem = budget_s - (time.time() - t0)
            if rem > 20:
                cur, c5 = nesterov_run(best_h, n_hi, th, t0, t0 + rem * 0.5, n_iter=50000, step0=0.03)
                if c5 < best_c5:
                    best_h, best_c5, best_n = cur, c5, n_hi

    # Phase 7: Final polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        nf = len(best_h)
        tf = nf / 2.0
        best_h = slsqp_solve(best_h, nf, tf, 2000, 3000)

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        best_h, best_c5 = pair_search(best_h, nf, nf / 2.0, 50000, t0, t0 + rem * 0.6, seed=321)

    rem = budget_s - (time.time() - t0)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
