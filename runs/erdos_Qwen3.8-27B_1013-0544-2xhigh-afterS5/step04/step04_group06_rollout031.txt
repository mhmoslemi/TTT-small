    hi_a = min(2*n-2, offset - i + n - 1)
    if lo_a <= hi_a:

        idx = i + np.arange(lo_a, hi_a+1) - offset
        coeff[lo_a:hi_a+1] = 1.0 - h[idx]
    
    # b_m contribution: -h[i-m+offset] where valid and i-m+offset != i (m != offset)
    lo_b = max(0, offset - i - n + 1)
    hi_b = min(2*n-2, offset - i)
    if lo_b <= hi_b:
        idx = i - np.arange(lo_b, hi_b+1) + offset
        coeff[lo_b:hi_b+1] -= h[idx]
    
    # For m == offset, the formula is different (quadratic term)
    # We'll handle that separately
    return coeff

            # For m == offset: new_corr[offset] = corr[offset] + (t-old)*(1-t-old)

                # Fix m = offset
                new_corr[offset] = corr[offset] + (t - old) * (1.0 - t - old)
                return float(new_corr.max() * dx)
            
            # Grid search
            if old == 0.0:

            elif old == 1.0:
                candidates = np.array([0.0, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 0.99, 1.0])
            else:
                candidates = np.clip(np.linspace(0, 1, grid_size), 0, 1)

                    best_c5 = best_val
                    any_improved = True

            break

def lbfgsb_optimize(h, n, beta, lam, maxiter=200):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        scaled = np.clip(beta * (corr - m), -500, 0)
        e = np.exp(scaled)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_correlate(x, n):
    """Compute gradient of correlate(x, 1-x, 'full') w.r.t. x."""
    g = np.zeros(n)
    offset = n - 1
    corr = np.correlate(x, 1.0 - x, mode="full")
    return corr, g, offset

def softmax_obj_grad(x, beta, lam, target, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    scaled = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(scaled)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx + lam * (x.sum() - target) ** 2
    w = e / se
    g = np.zeros(n)
    offset = n - 1
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = j - offset
        if lag == 0:
            g += wj * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += wj * (1.0 - x[:-lag])
            g[:n - lag] -= wj * x[lag:]
        else:
            m2 = -lag
            g[:n - m2] += wj * (1.0 - x[m2:])
            g[m2:] -= wj * x[:n - m2]
    g *= dx
    g += 2.0 * lam * (x.sum() - target)
    return val, g

def lbfgsb_optimize(h, n, beta, lam, maxiter=300):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return softmax_obj_grad(x, beta, lam, target, n)[0]

    def grad(x):
        return softmax_obj_grad(x, beta, lam, target, n)[1]

    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'maxls': 50})
    return project_sum(res.x, target)

def coord_descent(h, max_sweeps=8, grid_n=40):
    n = len(h)
    dx = 2.0 / n
    offset = n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(corr.max() * dx)

    for sweep in range(max_sweeps):
        any_imp = False
        for i in range(n):
            old = h[i]
            # Compute coeff[m] = (1 - h[i+m-offset]) - h[i-m+offset] (OOB=0)
            coeff = np.zeros(2 * n - 1)
            lo_a = max(0, offset - i)
            hi_a = min(2 * n - 2, 2 * offset - i)
            if lo_a <= hi_a:
                idx = i + np.arange(lo_a, hi_a + 1) - offset
                coeff[lo_a:hi_a + 1] = 1.0 - h[idx]
            lo_b = max(0, offset - i - n + 1)
            hi_b = min(2 * n - 2, offset - i)
            if lo_b <= hi_b:
                idx = i - np.arange(lo_b, hi_b + 1) + offset
                coeff[lo_b:hi_b + 1] -= h[idx]

            # Evaluate f(t) = max(corr + (t-old)*coeff) * dx with offset fix
            def fc5(t):
                delta = t - old
                new_corr = corr + delta * coeff
                new_corr[offset] -= delta * coeff[offset]
                new_corr[offset] += delta * (1.0 - t - old)
                return float(new_corr.max() * dx)

            if old <= 0.01:

                cands = np.array([0.0, 0.4, 0.6, 0.8, 0.9, 0.95, 0.98, 1.0])
            else:
                cands = np.linspace(0.0, 1.0, grid_n)

            best_t = old
            best_v = best_c5
            for t in cands:

                    continue

                if v < best_v:

    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode="full")
    no_imp = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        offset = n - 1
        for idx in top_idx:
            lag = int(idx) - offset
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.05, 0.02, 0.01, 0.005, 0.002, 0.0008, 0.0003]:

                best = ht
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode="full")
                improved = True
                break
        if not improved:

            if no_imp > 500 and it % 200 == 199:

                if c5t < best_c5:

                    best_c5 = c5t
                    corr = np.correlate(best, 1.0 - best, mode="full")
                    no_imp = 0
            if no_imp > 5000:
                break
        else:
            no_imp = 0
    return best, best_c5
