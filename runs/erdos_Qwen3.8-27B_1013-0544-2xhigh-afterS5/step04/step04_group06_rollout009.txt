from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def all_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n-lag] -= h[lag:]
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = all_corrs(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    nh = n - 1
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - nh
        g += w[j] * lag_gradient(h, lag)
    return val, g * dx

def slsqp_smooth(h_init, n, beta, maxiter=3000, ftol=1e-16):
    target = n / 2.0
    bounds = [(0, 1)] * n
    
    def obj(x):
        return smooth_obj_grad(np.clip(x, 0, 1), beta)[0]
    def grad(x):
        return smooth_obj_grad(np.clip(x, 0, 1), beta)[1]
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

_g_beta = 100.0
_g_n = 0

def _smooth_obj(x):
    return _smooth_obj_grad(np.clip(x, 0, 1), _g_beta)[0]

def _smooth_grad(x):
    return _smooth_obj_grad(np.clip(x, 0, 1), _g_beta)[1]

def _eq_con(x):
    return np.sum(x) - _g_n / 2.0

def _eq_jac(x):
    return np.ones(_g_n)

def _smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nh = n - 1
    for j in range(len(corr)):
        if w[j] < 1e-12:
            continue
        g += w[j] * lag_gradient(h, j - nh)
    return val, g * dx

def _penalty_obj(x):
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    active = corr > m * 0.99
    pen = np.sum(np.maximum(corr[active] - m * 0.995, 0) ** 2)
    return (m + 0.01 * pen) * (2.0 / len(h))

def _penalty_grad(x):
    h = np.clip(x, 0, 1)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    thresh = m * 0.99
    g = np.zeros(n)
    nh = n - 1
    for j in range(len(corr)):
        if corr[j] < thresh:
            continue
        lag = j - nh
        gd = lag_gradient(h, lag)
        if corr[j] > m * 0.995:
            g += gd + 2.0 * 0.01 * (corr[j] - m * 0.995) * gd
        else:
            g += gd
    return g * dx

def slsqp_run(h_init, n, beta, maxiter=3000, use_penalty=False, ftol=1e-15):
    global _g_beta, _g_n
    _g_n = n
    bounds = [(0, 1)] * n
    obj = _penalty_obj if use_penalty else _smooth_obj
    grad = _penalty_grad if use_penalty else _smooth_grad
    if not use_penalty:
        _g_beta = beta
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, n / 2.0)

            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, target, 6, 500, t0, budget_s)

    # Phase 2: Penalty-based SLSQP
    for rnd in range(3):
        if rem() < 80:
            break
        h_new = slsqp_run(best_h, n, 1, maxiter=2000, use_penalty=True)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, target, 8, 600, t0, budget_s)

    # Phase 3: Resolution exploration
    for n_try in [400, 560, 640]:
        if rem() < 200:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        t_best = best_c5
        h_best_local = best_h.copy()
        for beta in [500, 5000, 50000]:
            if rem() < 80:
                break
            h_new = slsqp_run(h_interp, n_try, beta, maxiter=2500)
            c5_new = compute_c5(h_new)
            if c5_new < t_best:
                t_best = c5_new
                h_best_local = h_new.copy()
            h_interp = h_new
        h_interp, t_test = subgrad_polish(h_interp, n_try / 2.0, 8, 1500, t0, budget_s)
        if t_test < best_c5:
            best_h = h_interp
            best_c5 = t_test
        elif t_best < best_c5:
            best_h = h_best_local
            best_c5 = t_best

    # Phase 4: Multi-start diversity
    n_best = len(best_h)
    target_best = n_best / 2.0
    for restart in range(4):
        if rem() < 150:
            break
        if restart == 0:
            h_try = best_h.copy()
        elif restart == 1:
            h_try = project_sum(best_h + np.random.randn(n_best) * 0.04, target_best)
        elif restart == 2:
            bl = n_best // 8
            start = np.random.randint(0, n_best - bl)
            h_try = best_h.copy()
            h_try[start:start + bl] = 1.0 - h_try[start:start + bl]
            h_try = project_sum(h_try, target_best)
        else:
            h_try = np.zeros(n_best)
            h_try[:n_best // 2] = best_h[:n_best // 2]
            h_try[n_best // 2:] = 1.0 - best_h[:n_best // 2][::-1]
            h_try = project_sum(h_try, target_best)
        for beta in [1000, 10000]:
            if rem() < 60:
                break
            h_new = slsqp_run(h_try, n_best, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new

    # Phase 5: Intensive subgradient polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, best_c5 = subgrad_polish(best_h, target_final, 10, 50000, t0, budget_s)

    # Phase 6: Coordinate descent polish
    if rem() > 60:
        best_h, best_c5 = coordinate_polish(best_h, target_final, 500, t0, budget_s)

    # Phase 7: Final SLSQP push
    if rem() > 40:
        h_new = slsqp_run(best_h, n_final, 200000, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, target_final, 12, 10000, t0, budget_s)

    # Phase 8: Final micro-polish
    if rem() > 5:
        best_h, best_c5 = subgrad_polish(best_h, target_final, 15, 20000, t0, budget_s)

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
