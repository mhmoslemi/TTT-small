from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def corr_gradients(h, lags):
    """Compute gradient of corr[k] w.r.t. h for given lags (indices into full corr)."""
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        lag = int(k) - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g = 1.0 - 2.0 * h
        elif 0 < lag < n:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
        grads[idx] = g * dx
    return grads

def linprog_descent(h, M=20, max_iter=100, time_fn=None):
    """Frank-Wolfe style: linearize top-M correlations, solve LP for step."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(max_iter):
        if time_fn is not None and time_fn() < 5:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full") * dx
        top_idx = np.argsort(corr)[::-1][:M]
        t_cur = corr[top_idx[0]]
        
        # Get gradients for top M
        grads = corr_gradients(h_cur, top_idx)  # (M, n), already scaled by dx

        # vars: d[0..n-1], t  (total n+1 vars)

        #   sum(d) = 0

                if c5_nr2 < best_c5:
                    best_h, best_c5 = h_nr2, c5_nr2

    # Phase 7: Final SLSQP polish
    if time_left() > 20:
        h_new, c5_new = do_slsqp(best_h, [50000, 100000, 200000],
                                  maxiter_each=6000, time_fn=time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 8: Final LP polish
    if time_left() > 15:
        h_new, c5_new = do_lp_descent(best_h, M=100, max_iter=100, time_fn=time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    return best_h, best_c5, len(best_h)
