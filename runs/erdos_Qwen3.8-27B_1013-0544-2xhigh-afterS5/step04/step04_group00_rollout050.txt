```python
    # Bisection on lambda

        else:
            hi = mid

    dx = 2.0 / n
    grad = np.zeros(n)
    # First term: -h[j+m-n+1] for max(0,n-1-m) <= j <= n-1

        bidx = idx + m_star - n + 1
        grad[idx] -= h[bidx]

    j_start2 = max(0, m_star - n + 1)
    j_end2 = min(n - 1, m_star)
    if j_start2 <= j_end2:

        bidx = idx - m_star + n - 1
        grad[idx] += (1 - h[bidx])
    return grad * dx
```

```python
import time

    return np.correlate(h, 1 - h, mode='full') * dx

    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):

        if np.clip(x - mid, 0, 1).sum() > target:

        else:
            hi = mid

def gradient_c(h, n, m):
    """Analytic gradient of c[m] w.r.t. h, multiplied by dx"""
    dx = 2.0 / n
    grad = np.zeros(n)
    # Term 1: -h[j+m-n+1] for max(0,n-1-m) <= j <= n-1
    lo1 = max(0, n - 1 - m)
    hi1 = n - 1
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        grad[idx] -= h[idx + m - n + 1]
    # Term 2: (1-h)[j-m+n-1] for max(0,m-n+1) <= j <= min(n-1,m)
    lo2 = max(0, m - n + 1)
    hi2 = min(n - 1, m)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += 1.0 - h[idx - m + n - 1]
    return grad * dx

def soft_max_gradient(h, n, alpha, top_k=3):
    """Compute gradient of smoothed max of c values"""
    c_vals = compute_all_c(h, n)
    # Get top-k indices
    top_idx = np.argsort(c_vals)[-top_k:]
    c_top = c_vals[top_idx]
    w = np.exp(alpha * (c_top - c_top.max()))
    w /= w.sum()
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        grad += w[i] * gradient_c(h, n, int(m))
    # Soft max value
    sm = c_top.max()
    soft_val = sm + np.log(np.exp(alpha * (c_top - sm)).sum()) / alpha
    return grad, soft_val, c_vals

def optimize_pgd(h, n, deadline, lr=0.5, alpha_sm=50, top_k=3, max_iter=20000):

        else:

    return best_h, best_c
```

For a binary function with n/2 ones and n/2 zeros, the overlap at shift k is the number of 1s in positions where the shifted function has 0s. If we think of it as: for shift k, overlap = |A ∩ (A+k)̄| where A is the set of 1-positions.

```python
    n = len(a)
    N = 2 * n  # power of 2 for efficiency
    A = np.fft.rfft(a, N)
```

```python
import numpy as np
import time

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_c(h, n)))

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c_m(h, n, m):
    dx = 2.0 / n
    grad = np.zeros(n)
    lo1 = max(0, n - 1 - m)
    if lo1 < n:
        idx = np.arange(lo1, n)
        grad[idx] -= h[idx + m - n + 1]
    lo2 = max(0, m - n + 1)
    hi2 = min(n - 1, m)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += 1.0 - h[idx - m + n - 1]
    return grad * dx

def pgd_optimize(h, n, deadline, lr=0.5, alpha_sm=50.0, top_k=2, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr_c = lr
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = compute_all_c(h, n)
        max_c = c_vals.max()
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        
        # Soft-max weights for top-k
        top_idx = np.argsort(c_vals)[-top_k:]
        c_top = c_vals[top_idx]
        w = np.exp(alpha_sm * (c_top - c_top.max()))
        w /= w.sum()
        
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            grad += w[i] * grad_c_m(h, n, int(top_idx[i]))
        
        # Project gradient
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Backtracking
        alpha = lr_c
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - alpha * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < max_c - 1e-14:
                h = h_new
                improved = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
            alpha *= 0.5
        if not improved:
            lr_c *= 0.85
            if lr_c < 1e-10:
                break
        else:
            lr_c = min(lr_c * 1.1, 3.0)
    
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=500):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    c_vals = compute_all_c(h_bin, n)
    best_c = c_vals.max()
    ones = np.where(h_bin == 1.0)[0].tolist()
    zeros = np.where(h_bin == 0.0)[0].tolist()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if improved:
                break
            for j in ones:
                # Try swap
                h_bin[i], h_bin[j] = 1.0, 0.0
                c_new = compute_all_c(h_bin, n)
                max_new = c_new.max()
                if max_new < best_c - 1e-15:
                    best_c = max_new
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
        if not improved:
            break
    return h_bin, best_c
```

```python
    # +1 at positions where i appears as first index: m such that i+m-n+1 is in [0,n-1] and h[i+m-n+1]=0
    #   m range: n-1-i <= m <= n-1 (since i+m-n+1 >= 0 for m >= n-1-i, and i+m-n+1 <= n-1 for m <= n-1)
    #   Wait, we need h[p]=0 where p = i+m-n+1, i.e., the (1-h) factor is 1
    #   So we add 1 for m in [n-1-i, n-1] where h[i+m-n+1] = 0
    #   This corresponds to: for p in [0,n-1] where p = i+m-n+1, m = p-n+1+i
    #   i.e., m ranges from i-n+1 to i (when p ranges from 0 to n-1)
    #   But m must be in [0, 2n-2]
    #   So m in [max(0, i-n+1), min(2n-2, i)] and we add (1-h[p]) = 1-h[i+m-n+1]

    #   m in [n-1-i, n-1+i] (but since m in [0,2n-2], m in [max(0,n-1-i), min(2n-2, n-1+i)])
    #   Actually i+m-n+1 in [0,n-1] means m in [n-1-i, n-1+i]
    #   For m in this range: delta[m] -= h[i+m-n+1]

        delta[lo1:hi1+1] -= h[p]
    
    # Term 2: +(1-h[i-m+n-1]) for valid m (i-m+n-1 in [0,n-1])
    #   i-m+n-1 in [0,n-1] means m in [i-n+1, i+n-1]
    #   For m in [0, 2n-2]: m in [max(0, i-n+1), min(2n-2, i+n-1)]
    lo2 = max(0, i - n + 1)
    hi2 = min(2*n-2, i + n - 1)
    if lo2 <= hi2:
        p = i - np.arange(lo2, hi2+1) + n - 1  # = i-m+n-1
        delta[lo2:hi2+1] += 1.0 - h[p]
    
    # Term 3: +h[j+m-n+1] for valid m (j+m-n+1 in [0,n-1])
    lo3 = max(0, n - 1 - j)
    hi3 = min(2*n-2, n - 1 + j)
    if lo3 <= hi3:
        p = np.arange(lo3, hi3+1) + n - 1 - j
        delta[lo3:hi3+1] += h[p]
    
    # Term 4: -(1-h[j-m+n-1]) for valid m (j-m+n-1 in [0,n-1])
    lo4 = max(0, j - n + 1)
    hi4 = min(2*n-2, j + n - 1)
    if lo4 <= hi4:
        p = j - np.arange(lo4, hi4+1) + n - 1
        delta[lo4:hi4+1] -= 1.0 - h[p]
    
    return delta
```

Hmm, this is getting complex. Let me simplify: since h is binary, h[p] is 0 or 1, and 1-h[p] is 1-h[p]. The delta computation is O(n) per swap.

Actually, let me just use the direct computation (O(n²) per swap via np.correlate) and accept that we get fewer sweeps. The key improvement is in the continuous optimization, not the binary search.

Let me reconsider the overall strategy. The previous solution achieved 0.38087. To beat this, I need:
1. A faster gradient evaluation (O(n) vs O(n²) for numerical gradient)
2. More effective optimization (more iterations, better restarts)

Let me estimate: with O(n) gradient and O(n²) correlation, each PGD iteration costs about O(n²) for the correlation + O(n) for the gradient ≈ O(n²). For n=250, that's about 62500 operations per iteration (dominated by the correlation). With backtracking (say 5 evaluations), that's ~300K per iteration.

In numpy, a correlate of 250 elements takes maybe 5-20μs. So each iteration with 5 line search evaluations is about 50-100μs. In 600 seconds, we can do 6-12 million iterations. That's way more than enough.

The real question is: will PGD converge to the global optimum? Probably not, but with enough restarts and different hyperparameters, we should find a better local optimum.

Let me also consider: what if I use L-BFGS-B with the analytic gradient? L-BFGS-B is much more efficient than PGD for smooth problems. The issue is the non-smoothness of the max, but the soft-max smoothing addresses that.

L-BFGS-B with bounds [0,1] and a penalty for the sum constraint:
- Objective: soft_max(c_vals) + λ*(sum(h)-n/2)²
- Gradient: gradient of soft_max + 2λ*(sum(h)-n/2) * ones

This should work well. L-BFGS-B maintains a limited-memory BFGS approximation and takes larger steps than PGD.

Let me combine both approaches: L-BFGS-B for the main optimization, PGD for polishing.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_all_c(h, n)))

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c_m(h, n, m):
    dx = 2.0 / n
    grad = np.zeros(n)
    lo1 = max(0, n - 1 - m)
    if lo1 < n:
        idx = np.arange(lo1, n)
        grad[idx] -= h[idx + m - n + 1]
    lo2 = max(0, m - n + 1)
    hi2 = min(n - 1, m)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += 1.0 - h[idx - m + n - 1]
    return grad * dx

def soft_max_and_grad(h, n, alpha_sm, top_k):
    c_vals = compute_all_c(h, n)
    top_idx = np.argsort(c_vals)[-top_k:]
    c_top = c_vals[top_idx]
    sm = c_top.max()
    w = np.exp(alpha_sm * (c_top - sm))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        grad += w[i] * grad_c_m(h, n, int(top_idx[i]))
    soft_val = sm + np.log(np.exp(alpha_sm * (c_top - sm)).sum()) / alpha_sm
    return grad, soft_val, c_vals

def pgd_optimize(h, n, deadline, lr=0.5, alpha_sm=50.0, top_k=2, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr_c = lr
    stagn = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_vals = compute_all_c(h, n)
        max_c = c_vals.max()
        if max_c < best_c - 1e-14:
            best_c = max_c
            best_h = h.copy()
            stagn = 0
        else:
            stagn += 1
        if stagn > 2000:
            break
        top_idx = np.argsort(c_vals)[-top_k:]
        c_top = c_vals[top_idx]
        sm = c_top.max()
        w = np.exp(alpha_sm * (c_top - sm))
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            grad += w[i] * grad_c_m(h, n, int(top_idx[i]))
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        alpha = lr_c
        improved = False
        for _ in range(30):
            h_new = project_feasible(h - alpha * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < max_c - 1e-14:
                h = h_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            lr_c *= 0.8
            if lr_c < 1e-12:
                break
        else:
            lr_c = min(lr_c * 1.15, 3.0)
    return best_h, best_c

def lbfgs_optimize(h, n, deadline, alpha_sm=100.0, top_k=2, max_iter=3000):
    target = n / 2.0
    pen_w = 500.0
    def obj(x):
        hi = np.clip(x, 0, 0) + np.clip(x, 0, 1)
        hi = np.clip(x, 0.0, 1.0)
        c_vals = compute_all_c(hi, n)
        top = np.argsort(c_vals)[-top_k:]
        ct = c_vals[top]
        sm = ct.max()
        sv = sm + np.log(np.exp(alpha_sm * (ct - sm)).sum()) / alpha_sm
        return sv + pen_w * (hi.sum() - target) ** 2
    def jac(x):
        hi = np.clip(x, 0.0, 1.0)
        c_vals = compute_all_c(hi, n)
        top = np.argsort(c_vals)[-top_k:]
        ct = c_vals[top]
        sm = ct.max()
        w = np.exp(alpha_sm * (ct - sm))
        w /= w.sum()
        g = np.zeros(n)
        for i in range(len(top)):
            g += w[i] * grad_c_m(hi, n, int(top[i]))
        g += 2.0 * pen_w * (hi.sum() - target)
        return g
    res = minimize(obj, h, method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out, n)

def binary_local_search(h, n, deadline, max_sweeps=200):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    c_vals = compute_all_c(h_bin, n)
    best_c = c_vals.max()
    ones = np.where(h_bin == 1.0)[0].tolist()
    zeros = np.where(h_bin == 0.0)[0].tolist()
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if improved:
                break
            for j in ones:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c_new = compute_all_c(h_bin, n).max()
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    ones.remove(j)
                    ones.append(i)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
        if not improved:
            break
    return h_bin, best_c

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 100
        h_init = np.full(n_init, 0.5)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)
    n_best = n_init

    # Phase 1: PGD on initial (30%)
    dl1 = t0 + 0.30 * (budget_s - 5)
    h = project_feasible(h_init, n_init)
    h, c = pgd_optimize(h, n_init, dl1, lr=0.5, alpha_sm=50, top_k=2)
    if c < best_c:
        best_c, best_h, n_best = c, h.copy(), n_init

    # Phase 2: L-BFGS-B with different params (20%)
    dl2 = t0 + 0.50 * (budget_s - 5)
    for a, k in [(50, 1), (100, 2), (200, 1), (30, 3)]:
        if time.time() > dl2:
            break
        h = project_feasible(best_h, n_best)
        h, c = lbfgs_optimize(h, n_best, dl2, alpha_sm=a, top_k=k, max_iter=1500)
        if c < best_c:
            best_c, best_h, n_best = c, h.copy(), n_init

    # Phase 3: Random restarts with PGD (25%)
    dl3 = t0 + 0.75 * (budget_s - 5)
    for r in range(40):
        if time.time() > dl3 - 5:
            break
        sc = 0.05 + 0.4 * (r / 40.0)
        h_r = project_feasible(best_h + sc * rng.randn(n_best), n_best)
        a_sm = [20, 50, 100, 200][r % 4]
        tk = [1, 2, 3][r % 3]
        h_r, c_r = pgd_optimize(h_r, n_best, dl3 - 5, lr=0.8, alpha_sm=a_sm, top_k=tk, max_iter=20000)
        if c_r < best_c:
            best_c, best_h, n_best = c_r, h_r.copy(), n_init
            # Polish the new best
            h_p = project_feasible(best_h, n_best)
            h_p, c_p = pgd_optimize(h_p, n_best, min(dl3 - 5, time.time() + 10), lr=0.3, alpha_sm=100, top_k=1, max_iter=5000)
            if c_p < best_c:
                best_c, best_h = c_p, h_p.copy()

    # Phase 4: Binary local search (15%)
    dl4 = t0 + 0.90 * (budget_s - 5)
    if time.time() < dl4 - 10:
        hb = to_binary(best_h, n_best)
        hb, cb = binary_local_search(hb, n_best, dl4 - 10, max_sweeps=300)
        if cb < best_c:
            best_c, best_h, n_best = cb, hb.copy(), n_init
        # Also try binary from perturbed
        hp = project_feasible(best_h + 0.2 * rng.randn(n_best), n_best)
        hb2 = to_binary(hp, n_best)
        hb2, cb2 = binary_local_search(hb2, n_best, dl4 - 10, max_sweeps=100)
        if cb2 < best_c:
            best_c, best_h, n_best = cb2, hb2.copy(), n_init

    # Phase 5: Try different resolutions (5%)
    if time.time() < deadline - 40:
        for n_try in [150, 200, 300, 400]:
            if time.time() > deadline - 40:
                break
            xo = np.linspace(0, 2, n_best)
            xn = np.linspace(0, 2, n_try)
            ht = project_feasible(np.interp(xn, xo, best_h), n_try)
            ht, ct = pgd_optimize(ht, n_try, deadline - 40, lr=0.5, alpha_sm=50, top_k=2, max_iter=15000)
            if ct < best_c:
                best_c, best_h, n_best = ct, ht.copy(), n_try

    # Final polish
    if time.time() < deadline - 5:
        h_f = project_feasible(best_h, n_best)
        h_f, cf = pgd_optimize(h_f, n_best, deadline - 5, lr=0.3, alpha_sm=200, top_k=1, max_iter=20000)
        if cf < best_c:
            best_c, best_h = cf, h_f.copy()

    best_h = project_feasible(best_h, n_best)
    best_c = compute_c5(best_h, n_best)
    return best_h, best_c, n_best
```
