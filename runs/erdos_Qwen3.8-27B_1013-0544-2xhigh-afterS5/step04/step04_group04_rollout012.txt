                momentum = max(0.3, momentum * 0.95)
                # Perturbation
                scale = 0.005 * min(10, no_improve // 500)
                h_pert = project_h(h_best + np.random.randn(n) * scale, target)
                c5_pert = compute_c5(h_pert)
                if c5_pert < c5_best:
                    h = h_pert
                    c5_best = c5_pert
                    h_best = h_pert.copy()
                    vel = np.zeros(n)
                    no_improve = 0
                else:
                    h = h_best.copy()
                if no_improve > 5000:
                    break
    return h_best, c5_best

def parametric_h(n, m, alpha, beta):
    x = np.linspace(0, 2, n)
    h = 1.0 / (1.0 + np.exp(-alpha * (np.cos(np.pi * m * x) - beta)))
    return project_h(h, n / 2.0)

def do_parametric_search(n, time_fn):
    best = None
    best_c5 = 1.0
    for m in range(1, 15):
        if time_fn() < 10:
            break
        # Coarse grid for alpha, beta
        for alpha in [2, 5, 10, 20, 40, 80]:
            for beta in np.linspace(-1, 1, 9):
                if time_fn() < 5:
                    break
                h = parametric_h(n, m, float(alpha), float(beta))
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best = h.copy()
        # Refine best for this m
        if best is not None and best_c5 < 0.5:
            h0 = best.copy()
            try:
                res = minimize(compute_c5, h0[:n-1], method='Nelder-Mead',
                              options={'maxiter': 200, 'xatol': 1e-6})
            except:
                pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(300, 0.5)
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Try parametric initializations for diversity
    for n_try in [n_best]:
        if time_left() < 60:
            break
        ph, pc5 = do_parametric_search(n_try, time_left)
        if ph is not None and pc5 < best_c5:
            best_h = ph
            best_c5 = pc5

    # Phase 2: SLSQP optimization with beta schedule
    target = n_best / 2.0
    h_new, c5_new = do_slsqp(best_h, [20, 50, 200, 1000, 5000, 20000],
                              maxiter_each=6000, time_fn=time_left, target=target)
    if c5_new < best_c5:
        best_h, best_c5 = h_new, c5_new
        n_best = len(best_h)
        target = n_best / 2.0

    # Phase 3: Nesterov-accelerated subgradient
    n_f = len(best_h)
    tgt = n_f / 2.0
    lr_init = 0.03
    best_h, best_c5 = do_nesterov(best_h, tgt, 60000, time_left,
                                   top_k=20, base_lr=lr_init, momentum=0.8)

    # Phase 4: Multi-resolution exploration
    for n_res in [n_best - 50, n_best + 50, n_best + 100, n_best // 2]:
        if time_left() < 80:
            break
        if n_res < 50:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        tgt_res = n_res / 2.0
        h_nr, c5_nr = do_slsqp(h_res, [50, 500, 5000, 20000],
                                maxiter_each=4000, time_fn=time_left, target=tgt_res)
        if time_left() > 30:
            h_nr, c5_nr = do_nesterov(h_nr, tgt_res, 20000, time_left,
                                       top_k=15, base_lr=0.02, momentum=0.75)
        if c5_nr < best_c5:
            best_h, best_c5 = h_nr, c5_nr
            n_best = n_res
            tgt = n_res / 2.0

    # Phase 5: Second SLSQP round with high beta
    if time_left() > 30:
        tgt = len(best_h) / 2.0
        h_new, c5_new = do_slsqp(best_h, [10000, 30000, 50000],
                                  maxiter_each=10000, time_fn=time_left, target=tgt)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            tgt = len(best_h) / 2.0

    # Phase 6: Final Nesterov polish with small steps
    if time_left() > 20:
        tgt = len(best_h) / 2.0
        best_h, best_c5 = do_nesterov(best_h, tgt, 30000, time_left,
                                       top_k=30, base_lr=0.005, momentum=0.7)

    # Phase 7: Final SLSQP if time permits
    if time_left() > 20:
        tgt = len(best_h) / 2.0
        h_new, c5_new = do_slsqp(best_h, [50000, 100000],
                                  maxiter_each=15000, time_fn=time_left, target=tgt)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final projection and validation
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
