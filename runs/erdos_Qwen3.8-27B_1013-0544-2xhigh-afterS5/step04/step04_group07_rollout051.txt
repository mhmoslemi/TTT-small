def project_h(h, target_sum):

        h = np.clip(h, 0.0, 1.0)

def smax_obj_grad(h, beta, lam, target, dx):

    if budget_s - (time.time() - t0) > 10:
        best_h = runSLSQP(best_h, nf, tf, dfx, 2000, 3000, t0, budget_s)
        best_c5 = compute_c5(best_h)

    if budget_s - (time.time() - t0) > 5:
        best_h, best_c5 = coord_descent(best_h, tf, nf, 30, t0, budget_s)

    best_h = project_h(best_h, tf)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
