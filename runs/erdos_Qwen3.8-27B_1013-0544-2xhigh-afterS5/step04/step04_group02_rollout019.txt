    n = len(h)

    sm = smooth_max(vals, beta) * dx
    s = h.sum() - ts

    return float(np.max(corr) * dx)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    cv = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    lags_idx = np.argsort(cv)[-k:]
    return lags_idx, cv

    n = len(h)
    cv = np.correlate(h, 1.0 - h, mode='full')
    vals = cv[lags]
    sm = smooth_max(vals, beta) * dx
    s = h.sum() - ts
    return sm + pen * s * s

def active_grad(h, dx, beta, lags, ts, pen):
    n = len(h)
    cv = np.correlate(h, 1.0 - h, mode='full')
    vals = cv[lags]
    m = np.max(vals)

        if w[j] < 1e-14:
            continue
        lag = li - (n - 1)
        if lag >= 0:

            grad[idx + lag] -= w[j] * h[idx]

            idx = np.arange(n - la)
            grad[idx + la] += w[j] * (1.0 - h[idx])
            grad[idx] -= w[j] * h[idx + la]
    grad *= dx
    s = h.sum() - ts

    return grad

        best_h, best_c5, best_n = h4.copy(), c4, best_n
    h4, c4 = momentum_opt(best_h.copy(), best_n, d4, seed + 6000, n_active=5)
    if c4 < best_c5:
        best_h, best_c5, best_n = h4.copy(), c4, best_n
    h4, c4 = coord_refine(best_h.copy(), best_n, d4, seed + 7000)
    if c4 < best_c5:
        best_h, best_c5, best_n = h4.copy(), c4, best_n

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
