import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def solve_linearized_lp(h0, target):
    n = len(h0)
    dx = 2.0 / n
    corr_vals = np.correlate(h0, 1.0 - h0, mode="full")
    
    # Build gradient matrix G (2n-1) x n
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag == 0:
            G[k] = 1.0 - 2.0 * h0
        elif lag > 0:
            G[k, lag:] += 1.0 - h0[:n - lag]
            G[k, :n - lag] -= h0[lag:]
        else:
            m = -lag
            G[k, :n - m] += 1.0 - h0[m:]
            G[k, m:] -= h0[:n - m]
    
    # LP: min t s.t. G[k,:]·h - t ≤ G[k,:]·h0 - corr_vals[k], 0≤h≤1, sum(h)=target
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = G @ h0 - corr_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return h0.copy()

def do_lp_optimize(h0, time_fn, max_iters=200):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    c5_cur = compute_c5(h)
    alphas = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
    
    for it in range(max_iters):
        if time_fn() < 15:
            break
        h_lp = solve_linearized_lp(h, target)
        improved = False
        for alpha in alphas:
            h_try = project_h((1 - alpha) * h + alpha * h_lp, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-14:
                h = h_try
                c5_cur = c5_try
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def do_sa_optimize(h0, time_fn, n_rounds=200):
    n = len(h0)
    target = n / 2.0
    h_best = h0.copy()
    c5_best = compute_c5(h_best)
    T = 0.05
    
    for rnd in range(n_rounds):
        if time_fn() < 30:
            break

def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x))) / beta) * dx

def slsqp_obj(x, beta, target):
    return smooth_max_c5(x, beta)

def slsqp_grad(x, beta, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - x[:-lag])
            grad[:n - lag] -= w * x[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - x[m2:])
            grad[m2:] -= w * x[:n - m2]
    return grad

def do_slsqp_polish(h0, time_fn, betas, maxiter_each):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    h = h0.copy()
    c5 = compute_c5(h)
    for beta in betas:
        if time_fn() < 20:
            break
        try:
            res = minimize(slsqp_obj, h, method='SLSQP', bounds=bounds,
                          jac=lambda x: slsqp_grad(x, beta, target),
                          constraints=constraints,
                          args=(beta, target),
                          options={'maxiter': maxiter_each, 'ftol': 1e-12})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                c5 = c5_new
        except Exception:
            pass
    return h, c5

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2.0, n_old)
    x_new = np.linspace(0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def build_G(h0):
    n = len(h0)
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag == 0:
            G[k] = 1.0 - 2.0 * h0
        elif lag > 0:
            G[k, lag:] += 1.0 - h0[:n - lag]
            G[k, :n - lag] -= h0[lag:]
        else:
            m = -lag
            G[k, :n - m] += 1.0 - h0[m:]
            G[k, m:] -= h0[:n - m]
    return G

def solve_lp(h0, target):
    n = len(h0)
    corr_vals = np.correlate(h0, 1.0 - h0, mode="full")
    G = build_G(h0)
    A_ub = np.hstack([G, -np.ones((2 * n - 1, 1))])
    b_ub = G @ h0 - corr_vals
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h0.copy()
