import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    if len(h) > 1500:
        return fftconvolve(h, (1.0 - h)[::-1], mode='full')
    return np.correlate(h, 1.0 - h, mode="full")

def prox_constraint(h, target_sum):
    lo = float(h.min()) - 2.0
    hi = float(h.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_grad(h, beta_w, corr=None):
    if corr is None:
        corr = get_corr(h)
    m = np.max(corr)
    shift = np.clip(beta_w * (corr - m), -200, 200)
    ev = np.exp(shift)
    se = ev.sum()
    w = ev / se
    n = len(h)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        wi = w[idx]
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        elif -n < lag < 0:
            m2 = -lag
            grad[m2:] += wi * (1.0 - h[:n - m2])
            grad[:n - m2] -= wi * h[m2:]
    val = (m + np.log(se) / beta_w)
    return grad, val

    n = len(h0)
    x = h0.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        x = x_new
        t = t_new
        c5 = compute_c5(x)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            stall = 0
        else:
            stall += 1
        if stall > 200:
            step *= 0.7
            stall = 0
    return x, best_c5

def subgrad_refine(h0, target, n_iter, time_fn, top_k=12):

    h = h0.copy()
    best = compute_c5(h)
    no_imp = 0

def slsqp_polish(h0, target, beta, maxiter, time_fn):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        g, v = soft_grad(x, beta)
        return v * (2.0 / n)

    def jac(x):
        g, _ = soft_grad(x, beta)
        return g * (2.0 / n)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, jac=jac,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = prox_constraint(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)
