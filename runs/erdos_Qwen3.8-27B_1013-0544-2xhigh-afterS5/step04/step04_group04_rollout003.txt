Actually, there's a simpler observation: the function f(t) = max_r (c_r + t*dcorr_r) is the maximum of n linear functions. Its minimum over [0,1] is at:
- t=0 if the right derivative at 0 is >= 0
- t=1 if the left derivative at 1 is <= 0
- Otherwise, at an intersection of two lines

The right derivative at t=0 is max r { dcorr_r : c_r = max(c) } (among the lines achieving the max at t=0).
The left derivative at t=1 is min r { dcorr_r : c_r + dcorr_r = max(c+dcorr) } (among the lines achieving the max at t=1).

If neither endpoint is optimal, the minimum is at a vertex of the upper envelope. A vertex is where two lines cross: c_a + t*d_a = c_b + t*d_b, i.e., t = (c_b - c_a) / (d_a - d_b).

To find the optimal vertex, we can:
1. Compute the value at t=0 and t=1
2. Find all intersection points of pairs of lines that lie in [0,1]
3. Evaluate f at each intersection point
4. Take the minimum

This is O(n²) in the worst case, which is the same as the binary search approach. So no improvement.

def do_slsqp(h0, betas, maxiter_each, time_fn):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    con_fun = lambda x: x.sum() - target
    con_jac = lambda x: np.ones(len(x))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    for beta in betas:
        if time_fn() < 10:
            break
        def obj(x):
            v, _ = smooth_max_val_grad(x, beta)
            return v
        def grd(x):
            _, g = smooth_max_val_grad(x, beta)
            return g
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          jac=grd, constraints=constraints,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def gradient_descent(h, n_steps, beta, time_fn):
    n = len(h)
    target = n / 2.0
    for _ in range(n_steps):
        if time_fn() < 1:
            break
        val, grad = smooth_max_val_grad(h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        c5_cur = compute_c5(h)
        found = False
        for ss in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = project_h(h - ss * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h = h_new
                found = True
                break
        if not found:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(500, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)
    
    # Phase 1: Coordinate descent (fast local convergence)
    h1, _ = coord_descent(best_h.copy(), 60, time_left)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    n_best = len(best_h)
    
    # Phase 2: SLSQP refinement
    if time_left() > 60:
        h2, c5_2 = do_slsqp(best_h, [100, 500, 2000, 8000, 20000, 50000],
                           maxiter_each=3000, time_fn=time_left)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        n_best = len(best_h)
    
    # Phase 3: More coordinate descent after SLSQP
    h3, _ = coord_descent(best_h.copy(), 80, time_left)
    c5_3 = compute_c5(h3)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3
    n_best = len(best_h)
    
    # Phase 4: Multi-resolution search
    for n_res in [200, 300, 400, 600, 800]:
        if time_left() < 50:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        h_res, _ = coord_descent(h_res, 60, time_left)
        c5_r = compute_c5(h_res)
        if c5_r < best_c5:
            best_h, best_c5 = h_res, c5_r
            n_best = n_res
            # Polish at this resolution
            h_res, _ = coord_descent(best_h.copy(), 60, time_left)
            c5_r = compute_c5(h_res)
            if c5_r < best_c5:
                best_h, best_c5 = h_res, c5_r
    
    # Phase 5: Random restarts with perturbation
    for restart in range(30):
        if time_left() < 20:
            break
        scale = 0.015 + 0.003 * restart
        h_p = project_h(best_h + np.random.randn(len(best_h)) * scale, len(best_h) / 2.0)
        h_p, _ = coord_descent(h_p, 40, time_left)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p
    
    # Phase 6: Subgradient escape + coord descent
    if time_left() > 30:
        h_g = gradient_descent(best_h.copy(), 150, beta=300, time_fn=time_left)
        c5_g = compute_c5(h_g)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        h_cd, _ = coord_descent(best_h.copy(), 60, time_left)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    
    # Phase 7: Final SLSQP polish
    if time_left() > 20:
        h_f, c5_f = do_slsqp(best_h, [20000, 50000, 100000],
                            maxiter_each=5000, time_fn=time_left)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    
    # Phase 8: Final coordinate descent polish
    if time_left() > 5:
        h_f2, _ = coord_descent(best_h.copy(), 80, time_left)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_h, best_c5 = h_f2, c5_f2
    
    # Ensure exact constraint satisfaction
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
