import numpy as np
from scipy.optimize import minimize
import time

_beta_g = [5000.0]
_tgt_g = [100.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mu = 0.5 * (lo + hi)
        if np.clip(h - mu, 0.0, 1.0).sum() > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    sw = np.clip(beta * (corr - m), -50, 50)
    ew = np.exp(sw)
    sws = ew.sum()
    ws = ew / sws
    aidx = np.where(ws > 1e-8)[0]
    val = (m + np.log(sws) / beta) * dx
    grad = np.zeros(n)
    for i in aidx:
        w = ws[i] * dx
        lag = int(i) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        elif -n < lag < 0:
            m2 = -lag
            grad[m2:] += w * (1.0 - h[:n - m2])
            grad[:n - m2] -= w * h[m2:]
    return val, grad

def _sobj(x):
    return c5_grad(x, _beta_g[0])[0]

def _sgrad(x):
    return c5_grad(x, _beta_g[0])[1]

def _scon(x):
    return x.sum() - _tgt_g[0]

def _scon_jac(x):
    return np.ones(len(x))

def slsqp_round(h, betas, maxit, t0, budget_s):
    n = len(h)
    tgt = n / 2.0
    _tgt_g[0] = tgt
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _scon_jac}]
    for b in betas:
        if budget_s - (time.time() - t0) < 15:
            break
        _beta_g[0] = b
        try:
            res = minimize(_sobj, h, method='SLSQP', jac=_sgrad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxit, 'ftol': 1e-15})
            h_new = project_h(res.x, tgt)
            if compute_c5(h_new) < compute_c5(h) - 1e-12:
                h = h_new
        except Exception:
            pass
    return h

def subgrad_momentum(h, t0, budget_s, n_iter, top_k):
    n = len(h)
    tgt = n / 2.0
    dx = 2.0 / n
    v = np.zeros(n)
    lr = 0.01
    mom = 0.9
    best_local = compute_c5(h)
    stale = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
        mv = vals.max()
        sw = np.exp(np.clip(50.0 * (vals - mv), -50, 50))
        wts = sw / sw.sum()
        grad = np.zeros(n)
        for i in range(len(idx)):
            w = wts[i] * dx
            lag = int(idx[i]) - (n - 1)
            if lag == 0:

            elif 0 < lag < n:

            elif -n < lag < 0:

                grad[m2:] += w * (1.0 - h[:n - m2])
                grad[:n - m2] -= w * h[m2:]
        v = mom * v - lr * grad
        h_new = project_h(h + v, tgt)
        c5_new = compute_c5(h_new)
        if c5_new < best_local - 1e-10:

                v *= 0.5
            if stale > 400:
                break
        if it % 2000 == 0 and it > 0:
            lr *= 0.93
    return h, compute_c5(h)

    h5 = slsqp_round(best_h, [20000, 80000, 200000], 5000, t0, budget_s)
    c5_5 = compute_c5(h5)
    if c5_5 < best_c5:
        best_h, best_c5 = h5, c5_5

    # Phase 6: Upsample to higher resolution for tighter bound
    if budget_s - (time.time() - t0) > 80:
        for n_up in [1000, 2000]:
            if budget_s - (time.time() - t0) < 50:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_up)
            h_up = project_h(np.interp(x_new, x_old, best_h), n_up / 2.0)
            h_up = slsqp_round(h_up, [5000, 30000, 100000], 4000, t0, budget_s)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_h, best_c5 = h_up, c5_up

    # Phase 7: Final polish
    if budget_s - (time.time() - t0) > 20:
        h7 = slsqp_round(best_h, [50000, 200000, 500000], 5000, t0, budget_s)
        c5_7 = compute_c5(h7)
        if c5_7 < best_c5:
            best_h, best_c5 = h7, c5_7
        # Momentum subgradient final
        h7b, c5_7b = subgrad_momentum(best_h, t0, budget_s, 10000, 8)
        if c5_7b < best_c5:
            best_h, best_c5 = h7b, c5_7b

    # Ensure exact feasibility
    n_final = len(best_h)
    best_h = project_h(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
