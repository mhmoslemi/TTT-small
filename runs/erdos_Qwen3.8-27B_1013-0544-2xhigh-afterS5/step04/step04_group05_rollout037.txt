            # d/dh[j] = (1-2*h[j])

            g = np.zeros(n)
            g[:n-m] += (1.0 - h[m:])  # (1-h[j+m]) for j = 0..n-1-m
            g[m:] -= h[:n-m]           # -h[j-m] for j = m..n-1
            grad += g
        else:
            jm = -m
            # corr = sum_{i=jm}^{n-1} h[i]*(1-h[i-jm])
            # d/dh[j] = (1-h[j-jm]) if j >= jm, minus h[j+jm] if j+jm < n
            g = np.zeros(n)
            g[jm:] += (1.0 - h[:n-jm])   # (1-h[j-jm]) for j = jm..n-1
            g[:n-jm] -= h[jm:]           # -h[j+jm] for j = 0..n-1-jm
            grad += g
    return grad / np.sum(active)  # average over active lags

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def subgrad_active(h, tol=0.5):
    """Subgradient using active set (lags within tol of max)."""
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            m = -lag
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] -= h[m:]
    return grad / len(active), len(active)

def slsqp_smooth(h, target, dx, beta, maxiter, time_fn, min_time=5):

                      options={'maxiter': maxiter, 'ftol': 1e-18})

        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            return h_new
    except Exception:
        pass
    return h

                    tol=0.5, momentum=0.85, base_step=0.01):
