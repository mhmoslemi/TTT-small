import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


class SmoothMaxObj:
    def __init__(self, beta, dx):
        self.beta = beta
        self.dx = dx

    def __call__(self, h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, self.beta) * self.dx


class SumCon:
    def __init__(self, target):
        self.target = target

    def __call__(self, h):
        return h.sum() - self.target

    def jac(self, h):
        return np.ones(len(h))


def slsqp_opt(h, target, dx, beta, max_iter, rem_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    obj = SmoothMaxObj(beta, dx)
    con = SumCon(target)
    cons = [{'type': 'eq', 'fun': con, 'jac': con.jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def compute_subgrad_vector(y, lags, n):
    grad = np.zeros(n)
    for li in lags:
        lag = int(li) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * y)
        elif lag > 0:
            grad[lag:] += (1.0 - y[:-lag])
            grad[:n - lag] -= y[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - y[m:])
            grad[m:] -= y[:n - m]
    return grad


def nesterov_sg(h, target, rem_fn, max_iters, m_act, step_init,
                decay, mom_init, mom_end, perturb_every, perturb_scale):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = step_init
    no_imp = 0
    rng = np.random.RandomState(7)

    for it in range(max_iters):
        if rem_fn() < 3:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        top = np.argsort(corr)[::-1][:m_act]
        g = compute_subgrad_vector(y, top, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        x_new = project_h(y - step * g, target)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best = x_new.copy()
            best_c5 = c5_new
            no_imp = 0
        else:
            no_imp += 1
            step *= decay
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (1.0 - t) / t_next
        mom = mom_init + (mom_end - mom_init) * min(1.0, it / (max_iters * 0.6))
        y = (1.0 - alpha * (1.0 - mom)) * x_new + alpha * (1.0 - mom) * (x_new - x)
        y = project_h(y, target)
        x = x_new
        if no_imp > perturb_every and it % perturb_every == perturb_every - 1:
            p = rng.randn(n) * perturb_scale
            hp = project_h(best + p, target)
            if compute_c5(hp) < best_c5:
                best = hp
                best_c5 = compute_c5(best)
                x = best.copy()
                y = best.copy()
                step = step_init * 0.5
                no_imp = 0
        if no_imp > perturb_every * 8:
            break
    return best, best_c5


def perturbation(h, n, mode, scale, rng):
    if mode == 0:
        return rng.randn(n) * scale
    elif mode == 1:
        x = np.linspace(0, 2, n)
        freq = rng.uniform(3, 25)
        ph = rng.uniform(0, 2 * np.pi)
        return scale * np.sin(2 * np.pi * freq * x + ph)
    elif mode == 2:
        p = np.zeros(n)
        bs = rng.randint(n // 8, n // 3)
        st = rng.randint(0, max(1, n - bs))
        p[st:st + bs] = rng.choice([-1, 1]) * scale * rng.rand(bs)
        return p
    elif mode == 3:
        corr = np.correlate(h, 1.0 - h, mode="full")
        top = np.argsort(corr)[::-1][:8]
        g = compute_subgrad_vector(h, top, n)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            return rng.randn(n) * scale
        return -scale * g / gn
    elif mode == 4:
        x = np.linspace(0, 2, n)
        p = np.zeros(n)
        for _ in range(3):
            freq = rng.uniform(1, 15)
            p += scale * 0.5 * np.cos(2 * np.pi * freq * x + rng.uniform(0, 6.28))
        return p
    else:
        return rng.randn(n) * scale


def resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)


def optimize_at_resolution(h, rem_fn, slsqp_betas, sg_iters, sg_params):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    for beta in slsqp_betas:
        if rem_fn() < 15:
            break
        h, c5 = slsqp_opt(h, target, dx, beta, 4000, rem_fn)
    if rem_fn() > 10:
        h, c5 = nesterov_sg(h, target, rem_fn, sg_iters, sg_params,
                            0.01, 0.97, 0.87, 0.92, 400, 0.01)
    return h, c5
