from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = beta * (corr - m)
    x_shift = np.clip(x_shift, -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    # Only iterate over lags with significant weight
    threshold = 1e-8 / n
    active = np.where(weights > threshold)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    
    return val, grad

    h_curr = h.copy()
    h_prev = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    c5_curr = best_c5
    
    alpha = 0.9  # momentum
    lr = 0.05    # initial learning rate
    
    for it in range(n_iter):

            break

        h_nest = h_curr + alpha * (h_curr - h_prev)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(xs)
    se = ev.sum()
    w = ev / se
    val = (m + np.log(se) / beta) * dx
    grad = np.zeros(n)
    thresh = max(1e-10, 1e-12 * n)
    active = np.where(w > thresh)[0]
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:-lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - h[m2:])
            grad[m2:] -= wk * h[:n - m2]
    return val, grad

def exact_subgrad_direction(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    active = np.where(corr > m - 1e-6)[0]
    nk = min(top_k, max(3, len(active)))
    top_idx = np.argsort(corr)[::-1][:nk]
    grad = np.zeros(n)
    for k in top_idx:
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += (1.0 - h[m2:])
            grad[m2:] -= h[:n - m2]
    grad /= nk
    return grad

def slsqp_round(h, target, betas, maxiter, time_fn):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    c5_cur = compute_c5(h)
    h_cur = h.copy()
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    for beta in betas:
        if time_fn() < 5:
            break
        def obj(x):
            v, _ = smooth_max_val_grad(x, beta)
            return v
        def grad_fn(x):
            _, g = smooth_max_val_grad(x, beta)
            return g
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           jac=grad_fn, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def projected_subgrad(h, target, n_iter, time_fn, top_k=10,
                      steps=None, perturb_every=150, perturb_scale=0.003):
    n = len(h)
    if steps is None:
        steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    h = h.copy()
    c5_cur = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 2:
            break
        d = exact_subgrad_direction(h, top_k)
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-14:
            break
        d /= dnorm
        found = False
        for step in steps:
            ht = project_h(h - step * d, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-16:
                h = ht
                c5_cur = c5t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % perturb_every == perturb_every - 1:
                sc = perturb_scale * min(8, no_imp // perturb_every + 1)
                noise = np.random.randn(n) * sc
                ht = project_h(h + noise, target)
                c5t = compute_c5(ht)
                if c5t < c5_cur:
                    h = ht
                    c5_cur = c5t
                    no_imp = 0
            if no_imp > max(50, n_iter // 15):
                break
    return h, c5_cur

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_h(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(300, 0.5)
        # Add structure: alternating blocks
        nblk = best_h.size
        for i in range(nblk):
            if i % 4 < 2:
                best_h[i] = 0.7
            else:
                best_h[i] = 0.3
        best_h = project_h(best_h, nblk / 2.0)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Optimize at current resolution (n=500)
    target = n_best / 2.0
    h1, c51 = slsqp_round(best_h, target,
                          [50, 200, 1000, 5000, 20000], 3000, time_left)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51
    h2, c52 = projected_subgrad(best_h, target, 20000, time_left,
                                 top_k=8, perturb_every=120, perturb_scale=0.004)
    if c52 < best_c5:
        best_h, best_c5 = h2, c52
    if time_left() > 20:
        h3, c53 = slsqp_round(best_h, target,
                              [20000, 50000, 100000], 5000, time_left)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 2: Try higher resolutions
    for n_res in [700, 800, 600]:
        if time_left() < 120:
            break
        h_res = interpolate_h(best_h, n_res)
        tgt = n_res / 2.0
        c5_res = compute_c5(h_res)
        # Optimize at new resolution
        h_a, c5_a = slsqp_round(h_res, tgt,
                                [30, 100, 500, 2000, 10000, 30000],
                                min(3000, max(500, int(time_left() * 0.3))),
                                time_left)
        if c5_a < c5_res:
            c5_res = c5_a
            h_res = h_a
        h_b, c5_b = projected_subgrad(h_res, tgt, int(time_left() * 0.4),
                                       time_left, top_k=10,
                                       perturb_every=100, perturb_scale=0.003)
        if c5_b < c5_res:
            c5_res = c5_b
            h_res = h_b
        if time_left() > 60:
            h_c, c5_c = slsqp_round(h_res, tgt,
                                    [10000, 30000, 80000],
                                    min(5000, int(time_left() * 0.4)), time_left)
            if c5_c < c5_res:
                c5_res = c5_c
                h_res = h_c
        if c5_res < best_c5:
            best_h, best_c5, n_best = h_res, c5_res, n_res

    # Phase 3: Final polish at best resolution
    tgt = n_best / 2.0
    if time_left() > 30:
        h_f, c5_f = projected_subgrad(best_h, tgt,
                                       int(time_left() * 0.6), time_left,
                                       top_k=6,
                                       steps=[0.01, 0.005, 0.002, 0.001,
                                              0.0005, 0.0002],
                                       perturb_every=200, perturb_scale=0.002)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    if time_left() > 15:
        h_g, c5_g = slsqp_round(best_h, tgt,
                                [50000, 100000, 200000],
                                min(5000, int(time_left() * 0.8)), time_left)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g

    # Final projection to ensure validity
    best_h = project_h(best_h, n_best / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_best
