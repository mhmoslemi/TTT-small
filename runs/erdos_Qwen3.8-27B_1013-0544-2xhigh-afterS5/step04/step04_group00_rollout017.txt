import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    return np.correlate(h, 1 - h, mode='full')

def c5_val(h, n):
    return float(np.max(compute_g(h, n)) * 2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj_and_grad(x, n, alpha):
    g = compute_g(x, n)
    gm = g.max()
    dg = alpha * (g - gm)
    exp_g = np.exp(dg)
    se = exp_g.sum()
    p = exp_g / se
    smax = gm + np.log(se) / alpha
    
    # Gradient w.r.t. x (h)
    # grad[i] = 1 - sum_t p[t+n-1] * h[i-t] - sum_t p[t+n-1] * h[i+t]
    # Vectorized:
    ii = np.arange(n)[:, None]
    tt = np.arange(-(n-1), n)[None, :]
    it_plus = ii + tt
    it_minus = ii - tt
    
    # h[i+t] with out-of-range = 0
    mask_p = (it_plus >= 0) & (it_plus < n)
    h_it = np.where(mask_p, x[np.clip(it_plus, 0, n-1)], 0.0)
    
    # h[i-t] with out-of-range = 0
    mask_m = (it_minus >= 0) & (it_minus < n)
    h_itm = np.where(mask_m, x[np.clip(it_minus, 0, n-1)], 0.0)
    
    grad = 1.0 - h_itm.dot(p) - h_it.dot(p)
    
    return smax, grad

def lbfgs_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        v, _ = smooth_max_obj_and_grad(x, n, alpha)
        return v
    
    def jac(x):
        _, g = smooth_max_obj_and_grad(x, n, alpha)
        return g
    
    res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                            'maxls': 50})
    return project_feasible(res.x, n)

def binary_sa(h_init, n, deadline, n_iter=200000):
    """Simulated annealing on binary solutions using 2-swap moves"""
    # Initialize binary
    h = h_init.copy()
    h_bin = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (smallest h values among ones)
        one_idx = np.where(h_bin == 1)[0]
        remove = one_idx[np.argsort(h[one_idx])[:n_ones - target]]
        h_bin[remove] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        add = zero_idx[np.argsort(-h[zero_idx])[:target - n_ones]]
        h_bin[add] = 1
    
    g = compute_g(h_bin, n)
    best_c = float(g.max() * 2.0 / n)
    best_g_max = g.max()
    best_h = h_bin.copy()
    
    rng = np.random.RandomState(12345)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    T = 0.5 * best_g_max  # initial temperature
    T_min = 1e-8
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Random 2-swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        if i == j:
            continue
        
        # Compute delta_g
        # delta_g[t] = ((1-h[i-t]) - h[i+t]) - ((1-h[j-t]) - h[j+t])
        # where h is h_bin and out-of-range = 0
        t = np.arange(-(n-1), n)
        it_p = i + t
        it_m = i - t
        jt_p = j + t
        jt_m = j - t
        
        h_it_p = np.where((it_p >= 0) & (it_p < n), h_bin[np.clip(it_p, 0, n-1)], 0.0)
        h_it_m = np.where((it_m >= 0) & (it_m < n), h_bin[np.clip(it_m, 0, n-1)], 0.0)
        h_jt_p = np.where((jt_p >= 0) & (jt_p < n), h_bin[np.clip(jt_p, 0, n-1)], 0.0)
        h_jt_m = np.where((jt_m >= 0) & (jt_m < n), h_bin[np.clip(jt_m, 0, n-1)], 0.0)
        
        delta_g = (1.0 - h_it_m) - h_it_p - (1.0 - h_jt_m) + h_jt_p
        # = (h_jt_m - h_it_m) + (h_jt_p - h_it_p)
        
        new_g = g + delta_g
        new_max = new_g.max()
        old_max = g.max()
        delta = new_max - old_max
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            g = new_g
            h_bin[i], h_bin[j] = 1.0, 0.0
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if new_max < best_g_max:
                best_g_max = new_max
                best_c = best_g_max * 2.0 / n
                best_h = h_bin.copy()
        
        # Cool
        T *= 0.9999  # very slow cooling
        if T < T_min:
            T = T_min
    
    return best_h, best_c

            if s < 1e-14:

    exp_g = np.exp(dg)
    se = exp_g.sum()
    p = exp_g / se
    smax = gm + np.log(se) / alpha
    ii = np.arange(n)[:, None]

    stack_p = ii + tt
    stack_m = ii - tt
    mp = (stack_p >= 0) & (stack_p < n)

    hm = np.where(mm, x[np.clip(stack_m, 0, n-1)], 0.0)

        return smooth_max_obj_grad(x, n, alpha)[1]

                       bounds=[(0.0, 1.0)]*n,
