import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    # Project onto {0<=h<=1, sum(h)=n/2}
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h *= (1.0 + diff / tot)
    return np.clip(h, 0.0, 1.0)

def softmax_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, 'full') * dx
    c_shifted = c - c.max()
    w = np.exp(alpha * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(c)) / len(c)
    else:
        w /= w_sum
    g = np.zeros(n)
    # O(n^2) computation
    for j in range(n):
        s = 0.0
        for k in range(len(c)):
            d = k - (n - 1)
            if 0 <= j + d < n:
                s += w[k] * (1.0 - h[j + d])
            if 0 <= j - d < n:
                s -= w[k] * h[j - d]
        g[j] = s
    return g

def subgradient_descent(h, n, alpha, lr, max_iter, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = softmax_gradient(h, n, alpha)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

    mask1 = (d1 >= 0) & (d1 < n)
    mask2 = (d2 >= 0) & (d2 < n)
    # Term 1: for each j, sum_k w[k] * (1-h[d1[j,k]]) where mask1[j,k]
    # We can do: for each j, take w * (1-h[d1[j,:]]) and sum where mask
    one_minus_h_d1 = one_minus_h[np.clip(d1, 0, n-1)]  # (n, 2n-1)
    h_d2 = h[np.clip(d2, 0, n-1)]  # (n, 2n-1)
    g = (w[None, :] * one_minus_h_d1 * mask1).sum(axis=1) - (w[None, :] * h_d2 * mask2).sum(axis=1)
    return g

    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Main subgradient descent (50%)
    alloc = (deadline - t0) * 0.45
    h_opt, c_opt = subgrad_descent(h_init, n_init, t0 + alloc,
                                    alpha_lo=100, alpha_hi=20000,
                                    lr_lo=0.0005, lr_hi=0.1,
                                    momentum=0.85, max_steps=50000)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # Phase 2: SLSQP polish (15%)
    if time.time() < deadline - (deadline - t0) * 0.20:
        hp = slsqp_polish(best_h, n_init, alpha=5000, maxiter=2000,
                          deadline=deadline - 30)
        cp = compute_c5(hp, n_init)
        if cp < best_c:
            best_h, best_c = hp, cp

    # Phase 3: Restart with perturbation (15%)
    if time.time() < deadline - (deadline - t0) * 0.12:
        alloc3 = (deadline - time.time()) * 0.35
        for scale in [0.03, 0.08, 0.15]:
            if time.time() > deadline - (deadline - t0) * 0.12:
                break
            hp = best_h + rng.normal(0, scale, n_init)
            hp = project_feasible(hp, n_init)
            h_opt, c_opt = subgrad_descent(hp, n_init, time.time() + alloc3,
                                            alpha_lo=200, alpha_hi=30000,
                                            lr_lo=0.001, lr_hi=0.05,
                                            momentum=0.8, max_steps=20000)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                alloc3 = max(alloc3 * 0.5, 10)

    # Phase 4: SLSQP with higher alpha (10%)
    if time.time() < deadline - (deadline - t0) * 0.08:
        hp = slsqp_polish(best_h, n_init, alpha=20000, maxiter=3000,
                          deadline=deadline - 15)
        cp = compute_c5(hp, n_init)
        if cp < best_c:
            best_h, best_c = hp, cp

    # Phase 5: Fine pair search (15%)
    if time.time() < deadline - 5:
        best_h, best_c = near_pair_search(best_h, n_init, deadline - 2,
                                           [0.01, 0.005, 0.002, 0.001, 0.0005],
                                           radius=30)

    # Final polish
    if time.time() < deadline - 2:
        hp = slsqp_polish(best_h, n_init, alpha=50000, maxiter=1000)
        cp = compute_c5(hp, n_init)
        if cp < best_c:
            best_h, best_c = hp, cp

    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
