import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, beta):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))

    # C(L) = sum_i h[i]*(1-h[i+L]) for valid i

def smooth_max_val(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def pg_descent(h0, n, target, n_iter, beta, t0, budget):
    """Projected gradient descent with backtracking line search on smooth max."""
    h = h0.copy()
    best_h = h.copy()
    best_f = smooth_max_val(h, beta) * (2.0/n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        g, corr = smooth_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g_dir = -g / gn  # descent direction
        
        # Backtracking line search
        alpha = 0.1
        f_cur = smooth_max_val(h, beta)
        best_alpha = 0
        best_f_trial = f_cur
        for ls in range(30):
            h_trial = project_h(h + alpha * g_dir * gn, target)
            f_trial = smooth_max_val(h_trial, beta)
            if f_trial < best_f_trial:
                best_f_trial = f_trial
                best_alpha = alpha
            if f_trial < f_cur - 1e-4 * alpha * gn * gn:
                break
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        if best_alpha > 0:
            h = project_h(h + best_alpha * g_dir * gn, target)
        else:
            h = project_h(h + 1e-6 * g_dir * gn, target)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        
        if it % 500 == 0 and it > 0:
            # Re-evaluate
            pass
    
    return best_h, best_c5

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    def obj(x):
        return smooth_max_val(x, beta) * (2.0/n)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, jac=None, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if r.success or r.fun < smooth_max_val(h0, beta) * (2.0/n):
            return project_h(r.x, target)
        return h0
    except Exception:
        return h0

def coord_descent(cur, target, n, n_sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        # Try small perturbations at random positions
        rng_local = np.random.RandomState(sw + 100)
        n_trials = n * 2
        for _ in range(n_trials):
            if time.time() - t0 > budget - 2:
                break
            j = rng_local.randint(n)
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005):
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

grad[M:n] += wk * (1.0 - h[0:n-M])  → (1-h[0], 1-h[1], ..., 1-h[n-1-M]) added to positions M, M+1, ..., n-1
grad[0:n-M] -= wk * h[M:n]  → h[M], h[M+1], ..., h[n-1] subtracted from positions 0, 1, ..., n-1-M

Let me verify with n=5, M=1, h=[a,b,c,d,e]:
C(-1) = b(1-a) + c(1-b) + d(1-c) + e(1-d)
= b - ab + c - bc + d - cd + e - de

dC(-1)/da = -b (j=0, first term: j=0 < M=1, no; second term: j=0 <= n-1-M=3, subtract h[1]=b) ✓
dC(-1)/db = (1-a) - c (first: j=1>=1, add 1-h[0]=1-a; second: j=1<=3, subtract h[2]=c) ✓
dC(-1)/dc = (1-b) - d ✓
dC(-1)/dd = (1-c) - e ✓
dC(-1)/de = (1-d) - 0 (first: j=4>=1, add 1-h[3]=1-d; second: j=4 > n-1-M=3, no) ✓

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            return h
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    s = e.sum()
    w = e / s
    smv = m + np.log(s) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            if n > L:
                grad[:n - L] += wk * (1.0 - h[L:])
                if L > 0:
                    grad[L:] -= wk * h[:n - L]
        else:
            M = -L
            if n > M:
                grad[M:] += wk * (1.0 - h[:n - M])
                grad[:n - M] -= wk * h[M:]
    return smv, grad

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter, ftol):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, beta, dx)
    def grad(x):
        _, g = smooth_max_grad(x, beta)
        return g * dx
    try:
        r = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if r.fun <= obj(h0) + 1e-12:
            return project_h(r.x, target)
    except Exception:
        pass
    return h0

def pg_line_search(h0, n, target, beta, n_iter, t0, budget):
    h = h0.copy()
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        f, g = smooth_max_grad(h, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        alpha = 0.5
        h_trial = h
        for ls in range(35):
            h_trial = project_h(h - alpha * g, target)
            f_trial, _ = smooth_max_grad(h_trial, beta)
            if f_trial < f - 1e-5 * alpha * gn * gn:
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        h = h_trial
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
    return best_h, best_c5

def binary_hill_climb(h0, n, n_iters, t0, budget):
    target = n / 2.0
    h = (h0 > 0.5).astype(float)
    diff = target - h.sum()
    if diff > 0:
        order = np.argsort(-h0)
        h[order[:int(diff)]] = 1.0
    elif diff < 0:
        order = np.argsort(h0)
        h[order[:int(-diff)]] = 0.0
    h = project_h(h, target)
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(7)
    for it in range(n_iters):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        for _ in range(n):
            i, j = rng.choice(n, 2, replace=False)
            if h[i] > 0.5 and h[j] < 0.5:
                trial = h.copy()
                trial[i], trial[j] = 0.0, 1.0
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h, best_c5, improved = trial, c, True
            elif h[i] < 0.5 and h[j] > 0.5:
                trial = h.copy()
                trial[i], trial[j] = 1.0, 0.0
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h, best_c5, improved = trial, c, True
        if not improved:
            break
    return h, best_c5

def pair_descent(cur, target, n, n_sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(123)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        n_trials = n * 3
        for _ in range(n_trials):
            if time.time() - t0 > budget - 2:
                break
            i, j = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.015, -0.015, 0.007, -0.007):
                trial = cur.copy()
                trial[i] = np.clip(trial[i] + d, 0, 1)
                trial[j] = np.clip(trial[j] - d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def coord_descent_fast(cur, target, n, n_sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(456)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for _ in range(n * 2):
            if time.time() - t0 > budget - 2:
                break
            j = rng.randint(n)
            for d in (0.02, -0.02, 0.01, -0.01, 0.005, -0.005):
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    new = np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)
    return project_h(new, n_new / 2.0)
