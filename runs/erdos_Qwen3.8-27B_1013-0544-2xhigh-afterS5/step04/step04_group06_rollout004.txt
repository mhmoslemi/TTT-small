import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -600, 600)
    e = np.exp(bc)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_opt(h_init, n, beta, maxiter=2000, ftol=1e-16):
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[0]
    def grad(x):
        return softmax_obj_grad(np.clip(x, 0, 1), beta)[1]
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def subgrad_dir(h, n_active):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    n = len(h)
    g = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n - lag] -= h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += (1.0 - h[m2:])
            g[m2:] -= h[:n - m2]
    return g

def nesterov_subgrad(h_init, target, n_active, max_iter, t0, budget, momentum=0.9, base_step=0.04):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        step = base_step / (1.0 + it * 0.01)
        # Nesterov lookahead
        h_look = project_sum(h_cur + momentum * vel, target)
        g = subgrad_dir(h_look, n_active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        # Update

        vel = h_new - h_cur
        h_cur = h_new
        c5_cur = compute_c5(h_cur)
