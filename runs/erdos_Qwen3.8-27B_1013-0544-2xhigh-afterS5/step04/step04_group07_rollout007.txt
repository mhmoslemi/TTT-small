```python
def active_gradient(h, n, K):
    """Compute gradient as average of top-K lag gradients."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top-K lags
    top_k = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for k in top_k:
        d = int(k) - (n - 1)  # lag
        # d > 0: h[j+d] and h[j-d]
        # Gradient: (1-h[j+d]) for j+d < n, minus h[j-d] for j-d >= 0
        if d >= 0:
            g[:n-d] += (1.0 - h[d:])
            g[d:] -= h[:n-d]
        else:
            la = -d
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    return g / K, np.max(corr)
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def active_grad(h, n, K):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    thr = np.max(corr)
    # Get lags within a fraction of max
    tol = 0.005 * (thr + 1e-10)
    idx = np.where(corr >= thr - tol)[0]
    # If too many, take top K
    if len(idx) > K:
        idx = idx[np.argsort(corr[idx])[::-1][:K]]
    g = np.zeros(n)
    for k in idx:
        d = int(k) - (n - 1)
        if d >= 0:
            g[:n-d] += (1.0 - h[d:])
            g[d:] -= h[:n-d]
        else:
            la = -d
            g[la:] += (1.0 - h[:n-la])
            g[:n-la] -= h[la:]
    return g / max(len(idx), 1), thr

def grad_descent_ls(h0, n, target, n_iter, t0, budget, base_alpha=0.02):
    """Gradient descent with backtracking line search and momentum."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum
    alpha = base_alpha
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        g, _ = active_grad(h, n, K=50)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        # Momentum
        v = 0.9 * v - alpha * g
        h_trial = project_h(h + v, target)
        c_trial = compute_c5(h_trial)
        # Backtracking if no improvement
        if c_trial > best_c5:
            # Try smaller step
            found = False
            for shrink in [0.5, 0.25, 0.1]:
                h_trial = project_h(h - alpha * shrink * g, target)
                c_trial = compute_c5(h_trial)
                if c_trial < best_c5:
                    v = -alpha * shrink * g
                    found = True
                    break
            if not found:
                alpha *= 0.5
                v = -alpha * g
                h_trial = project_h(h + v, target)
                c_trial = compute_c5(h_trial)
        h = h_trial
        if c_trial < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = c_trial
    return best_h, best_c5

                if stall > 300:
                    break
            elif stall > 80:
                alpha *= 0.9
                stall = 0
    return best_h, best_c5

def pair_local_search(h0, n, target, n_rounds, t0, deadline):
    h = h0.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n):
            if time.time() > deadline:
                break
            i, j = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.01, -0.01, 0.005, -0.005):
                trial = h.copy()
                trial[i] += d
                trial[j] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    h, best_c5 = trial, c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def resample_to(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(384, 0.5)

    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP beta=300
    rem = budget_s - (time.time() - t0)
    if rem > 150:
        cur = slsqp_opt(best_h, n, target, 300, 4000, 1e-14)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 2: Active-set gradient
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        dl = t0 + min(rem - 10, 250)
        cur, c5 = active_grad_method(best_h, n, target, 8000, t0, dl)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 3: SLSQP beta=1000
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        cur = slsqp_opt(best_h, n, target, 1000, 5000, 1e-15)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 4: Active-set gradient again
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        dl = t0 + min(rem - 5, 150)
        cur, c5 = active_grad_method(best_h, n, target, 6000, t0, dl)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 5: Multi-start perturbations
    rem = budget_s - (time.time() - t0)
    for trial in range(6):
        if time.time() - t0 > budget_s - 50:
            break
        noise = np.random.randn(n) * 0.008
        perturbed = project_h(best_h + noise, target)
        cur = slsqp_opt(perturbed, n, target, 500, 2000, 1e-14)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 6: Final SLSQP high beta
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur = slsqp_opt(best_h, n, target, 3000, 4000, 1e-15)
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    # Phase 7: Final active-set + local search
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        dl = t0 + min(rem - 5, 80)
        cur, c5 = active_grad_method(best_h, n, target, 4000, t0, dl)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        dl = t0 + min(rem - 3, 50)
        cur, c5 = pair_local_search(best_h, n, target, 50, t0, dl)
        if c5 < best_c5:
            best_h, best_c5 = cur.copy(), c5

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
