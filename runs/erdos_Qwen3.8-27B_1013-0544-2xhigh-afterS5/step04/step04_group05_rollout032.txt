from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0) & (h < 1.0)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h, target, betas, max_iter, time_fn):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for beta in betas:
        if time_fn() < 10:
            break
        
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx

            return h_in.sum() - target
        
        def jac_fn(h_in):
            return np.ones(n)

                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:

                best_c5 = c5_new

            else:

                no_improve = 0
        if no_improve > perturb_every * 8:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    # Phase 1: SLSQP at initial resolution
    best_h, best_c5 = slsqp_phase(best_h, target0,
        [30, 80, 200, 500, 1500, 4000, 10000], 5000, t0, budget_s)

    # Phase 2: Subgradient at initial resolution
    best_h, best_c5 = subgrad_phase(best_h, target0, t0, budget_s,
        max_iters=50000, m_factor=20,
        perturb_every=200, perturb_scale=0.004, momentum=0.85)

    # Phase 3: Upscale to 2x resolution
    if budget_s - (time.time() - t0) > 80:
        n2 = n0 * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        if budget_s - (time.time() - t0) > 30:
            h2, c5_2 = slsqp_phase(h2, target2,
                [100, 400, 1500, 5000, 15000], 4000, t0, budget_s)
        if budget_s - (time.time() - t0) > 15:
            h2, c5_2 = subgrad_phase(h2, target2, t0, budget_s,
                max_iters=30000, m_factor=25,
                perturb_every=300, perturb_scale=0.003, momentum=0.88)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 4: Final SLSQP polish at high beta
    n_f = len(best_h)
    target_f = n_f / 2.0
    if budget_s - (time.time() - t0) > 10:
        best_h, best_c5 = slsqp_phase(best_h, target_f,
            [15000, 40000, 80000], 15000, t0, budget_s)

    # Phase 5: Fine subgradient
    best_h, best_c5 = subgrad_phase(best_h, target_f, t0, budget_s,
        max_iters=20000, m_factor=25,
        perturb_every=400, perturb_scale=0.002, momentum=0.92)

    # Phase 6: One more SLSQP round if time remains
    if budget_s - (time.time() - t0) > 8:
        best_h, best_c5 = slsqp_phase(best_h, target_f,
            [50000, 100000], 10000, t0, budget_s)

    return best_h, best_c5, len(best_h)
