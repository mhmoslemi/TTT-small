def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = x.max()
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def subgrad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return grad
        i = np.arange(m)
        grad[i] += (1.0 - h[i + lag])
        grad[i + lag] -= h[i]
    else:
        la = -lag
        m = n - la
        if m <= 0:
            return grad
        i = np.arange(m)
        grad[i + la] += (1.0 - h[i])
        grad[i] -= h[i + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqp_phased(h, n, deadline, target_sum, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    c5 = compute_c5(h)
    betas = [15, 40, 100, 400, 1500, 6000, 25000, 100000]
    for beta in betas:
        if time.time() > deadline:
            break
        perts = [0.0, 0.001, 0.003, 0.008, 0.02]
        for p in perts:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                noise = np.random.randn(n) * p
                x0 = project_feasible(x0 + noise, target_sum)
            maxiter = 5000 if beta < 1000 else 12000
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-13:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def active_polish(h, n, deadline, target_sum, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    c5 = compute_c5(h)
    best_c5 = c5
    stagnant = 0
    lr = 0.002
    momentum = np.zeros(n)
    
    for it in range(80000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        
        # Adaptive threshold: start wide, tighten
        frac = min(2.0 + it * 0.001, 8.0)
        thresh = max_corr - frac
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            top_k = min(5, len(corr))
            active = np.argsort(corr)[-top_k:]
        
        w = np.exp(30.0 * (corr[active] - max_corr))
        w /= w.sum()
        
        grad = np.zeros(n)
        for j, k in enumerate(active):
            lag = k - (n - 1)
            grad += w[j] * subgrad_lag(h, lag, n)
        grad *= dx
        
        # Momentum update
        momentum = 0.8 * momentum - lr * grad
        improved = False
        for alpha_m in [1.0, 0.6, 0.3, 0.1]:
            h_new = project_feasible(h + alpha_m * momentum, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c_new
                improved = True
                stagnant = 0
                lr = min(lr * 1.003, 0.008)
                break
        if not improved:
            stagnant += 1
            momentum *= 0.5
            lr *= 0.92
            if stagnant > 400 and lr < 5e-8:
                break
    return h, best_c5

def restart_polish(h, n, deadline, target_sum, seed):
    """Try small random restarts around current best."""
    np.random.seed(seed)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for trial in range(50):
        if time.time() > deadline:
            break
        scale = 0.0005 * (1.0 + trial * 0.3)
        noise = np.random.randn(n) * scale
        x0 = project_feasible(best_h + noise, target_sum)
        try:
            res = minimize(obj_smooth, x0, args=(dx, 50000), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c_new
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(200, 0.5)
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_feasible(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n
    
    # Phase 1: Main SLSQP optimization (45%)
    d1 = t0 + budget_s * 0.45
    h1, c1 = slsqp_phased(h0, n, d1, target_sum, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, n
    
    # Phase 2: Active-set polishing (30%)
    d2 = t0 + budget_s * 0.75
    h2, c2 = active_polish(best_h, best_n, d2, best_n / 2.0, seed + 1)
    if c2 < best_c5:
        best_h, best_c5 = h2.copy(), c2
    
    # Phase 3: Restart + high-beta SLSQP (15%)
    d3 = t0 + budget_s * 0.90
    h3, c3 = restart_polish(best_h, best_n, d3, best_n / 2.0, seed + 100)
    if c3 < best_c5:
        best_h, best_c5 = h3.copy(), c3
    
    # Phase 4: Final polishing (8%)
    d4 = t0 + budget_s * 0.98
    h4, c4 = active_polish(best_h, best_n, d4, best_n / 2.0, seed + 500)
    if c4 < best_c5:
        best_h, best_c5 = h4.copy(), c4
    
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
