import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(x)
    se = np.sum(ev)
    val = (m + np.log(se) / beta) * dx
    sd = h.sum() - target
    val += lam * sd * sd
    return val

def smooth_max_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(x)
    se = np.sum(ev)
    w = ev / se
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wi = w[k]
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n-lag])
            grad[:n-lag] -= wi * h[lag:]
        else:
            m2 = -lag
            grad[:n-m2] += wi * (1.0 - h[m2:])
            grad[m2:] -= wi * h[:n-m2]
    grad *= dx
    
    sd = h.sum() - target
    grad += 2.0 * lam * sd
    return grad

def do_lbfgsb(h0, betas, lam, maxiter_each, time_fn):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 5:
            break
        res = minimize(smooth_max_obj, h_cur, method='L-BFGS-B',
                      bounds=bounds, jac=smooth_max_grad,
                      args=(beta, lam),
                      options={'maxiter': maxiter_each, 'ftol': 1e-16, 'gtol': 1e-14})
        h_new = project_h(res.x, n / 2.0)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur

def do_subgrad_momentum(h, n_iter, time_fn, top_k, beta_sg, lr_init, lr_decay, momentum):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    v = np.zeros(n)
    lr = lr_init
    c5_cur = compute_c5(h)
    for it in range(n_iter):
        if time_fn() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        # Softmax weights
        x = np.clip(beta_sg * (corr - m), -50, 50)
        ev = np.exp(x)
        w = ev / np.sum(ev)
        
        # Gradient
        grad = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for k in active:
            wi = w[k]
            lag = int(k) - (n - 1)
            if lag == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:] += wi * (1.0 - h[:n-lag])
                grad[:n-lag] -= wi * h[lag:]
            else:
                m2 = -lag
                grad[:n-m2] += wi * (1.0 - h[m2:])
                grad[m2:] -= wi * h[:n-m2]
        grad *= dx
        
        # Remove component along constraint normal
        grad -= np.mean(grad)
        
        # Momentum update
        v = momentum * v - lr * grad
        h_new = project_h(h + v, target)
        
        # Backtracking
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            lr = lr * 1.05  # increase step slightly
        else:
            lr *= 0.7
            v *= 0.5
        lr = min(lr, lr_init * 2)
    
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):

    start = time.time()

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _corr_grad_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += 1.0 - h[m2:]
        g[m2:] -= h[:n - m2]
    return g

def smooth_obj(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(x)
    se = np.sum(ev)
    val = (m + np.log(se) / beta) * dx
    sd = h.sum() - target
    val += lam * sd * sd
    return val

def smooth_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(x)
    se = np.sum(ev)
    w = ev / se
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    n1 = n - 1
    for k in active:
        wi = w[k]
        lag = int(k) - n1
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n - lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            m2 = -lag
            grad[:n - m2] += wi * (1.0 - h[m2:])
            grad[m2:] -= wi * h[:n - m2]
    grad *= dx
    sd = h.sum() - target
    grad += 2.0 * lam * sd
    return grad

def do_lbfgsb(h0, betas, lam, maxiter_each, time_fn):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 5:
            break
        try:
            res = minimize(smooth_obj, h_cur, method='L-BFGS-B',
                          bounds=bounds, jac=smooth_grad,
                          args=(beta, lam),
                          options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-16, 'maxls': 50})
            h_new = project_h(res.x, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h, n_iter, time_fn, beta_sg, lr_init, momentum):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    n1 = n - 1
    v = np.zeros(n)
    lr = lr_init
    c5_cur = compute_c5(h)
    no_improve = 0
    for it in range(n_iter):
        if time_fn() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        x = np.clip(beta_sg * (corr - m), -60, 60)
        ev = np.exp(x)
        w = ev / np.sum(ev)
        grad = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k in active:
            wi = w[k]
            lag = int(k) - n1
            if lag == 0:
                grad += wi * (1.0 - 2.0 * h)
            elif lag > 0:
                grad[lag:] += wi * (1.0 - h[:n - lag])
                grad[:n - lag] -= wi * h[lag:]
            else:
                m2 = -lag
                grad[:n - m2] += wi * (1.0 - h[m2:])
                grad[m2:] -= wi * h[:n - m2]
        grad *= dx
        grad -= np.mean(grad)
        v = momentum * v - lr * grad
        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            no_improve = 0
            lr = min(lr * 1.02, lr_init * 3)
        else:
            lr *= 0.8
            v *= 0.3
            no_improve += 1
            if no_improve > 500:
                break
            if no_improve % 200 == 0:
                sc = 0.002 * (no_improve // 200 + 1)
                hp = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(hp) < c5_cur:
                    h = hp
                    c5_cur = compute_c5(h)
                    no_improve = 0
                    v = np.zeros(n)
    return h, c5_cur

def do_multiresolve(h_best, c5_best, time_fn):
    n0 = len(h_best)
    best_h = h_best.copy()
    best_c5 = c5_best
    for n_res in [150, 200, 300, 400]:
        if time_fn() < 60:
            break
        if n_res >= n0:
            continue
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_res)
        h_low = np.interp(x_new, x_old, best_h)
        h_low = project_h(h_low, n_res / 2.0)
        h_opt, c5_opt = do_lbfgsb(h_low, [50, 200, 1000, 5000, 20000], 100.0, 400, time_fn)
        if time_fn() < 30:
            break
        x_up = np.linspace(0, 2, n0)
        h_up = np.interp(x_up, np.linspace(0, 2, len(h_opt)), h_opt)
        h_up = project_h(h_up, n0 / 2.0)
        h_pol, c5_pol = do_lbfgsb(h_up, [2000, 10000, 30000], 100.0, 200, time_fn)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    def tl():
        return budget_s - (time.time() - start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: L-BFGS-B with beta annealing
    h1, c5_1 = do_lbfgsb(best_h, [30, 80, 200, 500, 1500, 5000, 15000, 40000], 200.0, 400, tl)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Subgradient with momentum
    h2, c5_2 = do_subgrad(best_h, 60000, tl, beta_sg=80.0, lr_init=0.008, momentum=0.4)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Another L-BFGS-B round at high beta
    if tl() > 40:
        h3, c5_3 = do_lbfgsb(best_h, [10000, 30000, 80000, 150000], 500.0, 600, tl)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Multi-resolution escape
    if tl() > 120:
        h4, c5_4 = do_multiresolve(best_h, best_c5, tl)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Subgradient polish with finer parameters
    if tl() > 20:
        h5, c5_5 = do_subgrad(best_h, 40000, tl, beta_sg=200.0, lr_init=0.003, momentum=0.2)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Phase 6: Final L-BFGS-B
    if tl() > 25:
        h6, c5_6 = do_lbfgsb(best_h, [50000, 100000, 200000, 400000], 1000.0, 800, tl)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    # Phase 7: Short subgradient final polish
    if tl() > 5:
        h7, c5_7 = do_subgrad(best_h, 20000, tl, beta_sg=500.0, lr_init=0.001, momentum=0.1)
        if c5_7 < best_c5:
            best_h, best_c5 = h7, c5_7

    return best_h, best_c5, len(best_h)
