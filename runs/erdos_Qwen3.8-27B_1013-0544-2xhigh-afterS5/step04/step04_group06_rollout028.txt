        else:

        return None
    return res.x[:n]

def slsqp_optimize(h_init, n, beta, maxiter=2000, ftol=1e-16):
    target = n / 2.0
    bounds = [(0, 1)] * n
    dx = 2.0 / n
    nc = 2 * n - 1
    def obj(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        return (m + np.log(se / nc) / beta) * dx
    def grad(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        bc = np.clip(beta * (corr - m), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        for j in range(nc):
            if w[j] < 1e-11:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += w[j] * (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += w[j] * (1.0 - h[:-lag])
                g[:n-lag] -= w[j] * h[lag:]
            else:
                m2 = -lag
                g[:n-m2] += w[j] * (1.0 - h[m2:])
                g[m2:] -= w[j] * h[:n-m2]
        return g * dx
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eq, 'jac': eqj}]
    res = minimize(obj, h_init, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n_active, max_iter, t0, budget_s):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    momentum = np.zeros(n)
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 3:
            break
        corr = compute_all_corr(best)
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n-lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n-m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n-m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        momentum = 0.7 * momentum + 0.3 * g
        mn = np.linalg.norm(momentum)
        if mn > 0:
            momentum /= mn
        improved = False
        for use_dir in [momentum, g]:
            for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]:
                ht = project_sum(best - step * use_dir, target)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-15:
                    best = ht
                    best_c5 = c5t
                    improved = True
                    break
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 150 and it % 80 == 79:
                scale = 0.01 + 0.02 * (no_improve // 400)
                pert = np.random.randn(n) * scale
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                    momentum = np.zeros(n)
            if no_improve > 1500:
                break
        else:
            no_improve = 0
    return best, best_c5

def gen_thue_morse(n):
    seq = np.array([0])
    while len(seq) < n:
        seq = np.concatenate([seq, 1 - seq])
    return seq[:n].astype(float)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    # Phase 1: LP-based optimization
    for it in range(40):
        if tl() < 150:
            break
        direction = lp_direction(best_h, top_k=50)
        if direction is None:
            break
        improved = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            h_new = project_sum(best_h + alpha * direction, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break
        if it % 5 == 4:
            best_h, best_c5 = subgrad_polish(best_h, target, 5, 100, t0, budget_s)
    # Phase 2: SLSQP with increasing beta
    for beta in [100, 500, 2000, 10000, 40000, 150000]:
        if tl() < 100:
            break
        h_new = slsqp_optimize(best_h, n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, target, 5, 200, t0, budget_s)
    # Phase 3: More LP rounds
    for it in range(40):
        if tl() < 100:
            break
        direction = lp_direction(best_h, top_k=80)
        if direction is None:
            break
        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_new = project_sum(best_h + alpha * direction, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break
        best_h, best_c5 = subgrad_polish(best_h, target, 6, 150, t0, budget_s)
    # Phase 4: Try Thue-Morse restart if time permits
    if tl() > 200:
        h_tm = gen_thue_morse(n)
        h_tm = project_sum(h_tm, target)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.05:
            h_try = h_tm.copy()
            for beta in [200, 2000, 10000]:
                if tl() < 60:
                    break
                h_try = slsqp_optimize(h_try, n, beta, maxiter=1500)
                h_try, c5_try = subgrad_polish(h_try, target, 5, 100, t0, budget_s)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    # Phase 5: Final intensive polishing
    best_h, best_c5 = subgrad_polish(best_h, target, 8, 8000, t0, budget_s)
    # Phase 6: Final LP + SLSQP
    if tl() > 60:
        for it in range(20):
            if tl() < 30:
                break
            direction = lp_direction(best_h, top_k=100)
            if direction is None:
                break
            improved = False
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                h_new = project_sum(best_h + alpha * direction, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
            if not improved:
                break
            best_h, best_c5 = subgrad_polish(best_h, target, 5, 80, t0, budget_s)
    if tl() > 40:
        h_new = slsqp_optimize(best_h, n, 80000, maxiter=4000, ftol=1e-18)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    # Phase 7: Micro-polish
    for it in range(5000):
        if tl() < 2:
            break
        corr = compute_all_corr(best_h)
        top_idx = np.argsort(corr)[::-1][:4]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[lag:] += (1.0 - best_h[:-lag])
                g[:n-lag] -= best_h[lag:]
            else:
                m2 = -lag
                g[:n-m2] += (1.0 - best_h[m2:])
                g[m2:] -= best_h[:n-m2]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        for step in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            ht = project_sum(best_h - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                break
    return best_h, best_c5, n
