from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def all_active_subgrad(h, n, eps_frac=0.001):
    """Subgradient using all lags within eps_frac of the max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc * (1 - eps_frac)
    idx = np.where(corr >= thr)[0]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / max(len(idx), 1), corr, mc

```python
def min_norm_subgrad(h, n, num_active=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    # Get the top num_active lags
    idx = np.argsort(corr)[::-1][:num_active]
    # Only keep those at (or very near) the max
    thr = mc - 1e-6 * max(mc, 1)
    idx = idx[corr[idx] >= thr]
    if len(idx) == 0:
        idx = np.argsort(corr)[::-1][:5]
    
    # Compute gradients
    grads = np.zeros((len(idx), n))
    for a, i in enumerate(idx):
        lag = int(i) - (n - 1)
        if lag >= 0:
            grads[a, :n-lag] += (1.0 - h[lag:])
            grads[a, lag:] -= h[:n-lag]
        else:
            la = -lag
            grads[a, la:] += (1.0 - h[:n-la])
            grads[a, :n-la] -= h[la:]
    
    # Find minimum norm in convex hull
    m = len(idx)
    if m == 1:
        return grads[0]
    
    # Solve: min ||G^T alpha||^2 s.t. sum alpha = 1, alpha >= 0
    # Using a simple iterative method (projected gradient on the simplex)
    alpha = np.ones(m) / m
    for _ in range(100):
        g_avg = grads.T @ alpha
        # Gradient of (1/2)||G^T alpha||^2 w.r.t. alpha: G * (G^T alpha)
        grad_alpha = grads @ g_avg
        # Projected gradient step on simplex
        alpha = alpha - 0.1 * grad_alpha
        alpha = np.maximum(alpha, 0)
        s = alpha.sum()
        if s > 0:
            alpha /= s
        else:
            alpha = np.ones(m) / m
    return grads.T @ alpha
```

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        la = -lag
        g[la:] += (1.0 - h[:n - la])
        g[:n - la] -= h[la:]
    return g

def all_active_subgrad(h, n, eps_frac=0.001):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc * (1.0 - eps_frac)
    idx = np.where(corr >= thr)[0]
    if len(idx) > 80:
        idx = idx[np.argsort(corr[idx])[::-1][:80]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        g += grad_at_lag(h, lag, n)
    return g / max(len(idx), 1), corr, mc

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def c_fun(x, target):
    return np.sum(x) - target

def c_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter=3000, ftol=1e-14):

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h0

def pair_descent(cur, target, n, sweeps, t0, deadline):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(77)
    deltas = (0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002)
    for sw in range(sweeps):
        if time.time() > deadline:
            break
        improved = False

                break

                    cur, best_c5, improved = trial, c, True

    return cur, best_c5
