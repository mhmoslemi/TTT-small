import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_at_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def greedy_paired(h, n, deadline, seed):

    dx = 2.0 / n
    ts = n / 2.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for outer in range(5000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        K = min(15, len(corr))
        top_idx = np.argpartition(corr, -K)[-K:]
        improved = False
        for k in np.sort(top_idx):
            if time.time() > deadline:
                break
            lag = k - (n - 1)
            if lag <= 0:
                continue
            m = n - lag
            pairs_val = h[:m] * (1.0 - h[lag:lag + m])
            bad = np.argsort(pairs_val)[-5:]
            for worst in bad:
                pv = pairs_val[worst]
                if pv < 0.05:
                    continue
                step = min(h[worst], 1.0 - h[worst + lag]) * 0.15
                if step < 1e-6:
                    continue
                h[worst] -= step
                h[worst + lag] += step
                h = np.clip(h, 0.0, 1.0)
                h = project_feasible(h, ts)
                c_new = compute_c5(h)
                if c_new < c5 - 1e-13:
                    c5 = c_new
                    improved = True
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                else:
                    h[worst] += step
                    h[worst + lag] -= step
                    h = project_feasible(h, ts)
        if not improved:
            break
    return best_h, best_c5

    c5 = compute_c5(h)

    lr = base_lr
    momentum = np.zeros(n)

        K = min(20, len(corr))

def slsqp_refine(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x, beta):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx

    def cfun(x):
        return x.sum() - ts

    def cjac(x):
        return np.ones_like(x)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    c5 = compute_c5(h)
    for beta in [50, 200, 1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline:
            break
        mi = 8000 if beta < 5000 else 4000
        try:
            res = minimize(lambda x: obj(x, beta), h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mi, 'ftol': 1e-15})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                c5 = c5_new
        except Exception:
            pass
    return h, c5
