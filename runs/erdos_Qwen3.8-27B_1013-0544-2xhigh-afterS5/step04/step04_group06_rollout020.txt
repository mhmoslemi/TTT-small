    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid

def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    return (m + np.log(se / len(corr)) / beta) * dx

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(shifted)
    se = e.sum()
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]

        return np.sum(x) - target

        return np.ones(n)

    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g += (1.0 - 2.0 * h)
    elif lag > 0:

    else:

        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += gradient_for_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            s = step * mult
            ht = project_sum(best - s * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            step *= 0.997
            if no_improve > 300 and no_improve % 80 == 79:
                p = np.random.randn(n) * 0.02
                ht = project_sum(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
                    step = 0.04
            if no_improve > 2000:
                break
        else:
            no_improve = 0
    return best, best_c5

def fine_polish(h, t0, budget):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(30000):
        if budget - (time.time() - t0) < 1:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += gradient_for_lag(best, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        for step in [0.003, 0.001, 0.0004, 0.0001, 0.00003, 0.00001]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best = ht
                best_c5 = c5t
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(480, 0.5)
    best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining < 10:
        return best_h, best_c5, len(best_h)

    # Phase 1: SLSQP with increasing beta
    n0 = len(best_h)
    phase1_betas = [200, 1000, 5000, 20000, 80000]
    for beta in phase1_betas:
        if budget_s - (time.time() - t0) < 60:
            break
        h_new = slsqp_optimize(best_h, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = subgrad_polish(best_h, 8, 500, t0, budget_s)

    # Phase 2: Resolution exploration
    for n_try in [320, 400, 560]:
        if budget_s - (time.time() - t0) < 180:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        for beta in [500, 5000, 50000]:
            if budget_s - (time.time() - t0) < 60:
                break
            h_new = slsqp_optimize(h_interp, beta, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new
        best_h, best_c5 = subgrad_polish(best_h, 10, 500, t0, budget_s)

    # Phase 3: Structured perturbations
    n_best = len(best_h)
    candidates = []
    for s in [3, 7, 13]:
        if s < n_best // 2:
            candidates.append(np.roll(best_h, s))
    for frac in [0.15, 0.25]:
        blk = int(n_best * frac)
        h_mod = best_h.copy()
        start = n_best // 3
        h_mod[start:start + blk] = 1.0 - h_mod[start:start + blk]
        candidates.append(h_mod)
    candidates.append(best_h + np.random.randn(n_best) * 0.08)
    h_as = np.zeros(n_best)
    half = n_best // 2
    h_as[:half] = best_h[:half]
    h_as[half:] = 1.0 - best_h[:half][::-1]
    candidates.append(h_as)

    for h_mod in candidates:
        if budget_s - (time.time() - t0) < 80:
            break
        h_try = project_sum(np.clip(h_mod, 0, 1), n_best / 2.0)
        for beta in [1000, 10000]:
            if budget_s - (time.time() - t0) < 40:
                break
            h_new = slsqp_optimize(h_try, beta, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new

    # Phase 4: Final intensive SLSQP + subgradient
    best_h, best_c5 = subgrad_polish(best_h, 12, 5000, t0, budget_s)
    if budget_s - (time.time() - t0) > 40:
        h_new = slsqp_optimize(best_h, 200000, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Ultra-fine subgradient polish
    best_h, best_c5 = fine_polish(best_h, t0, budget_s)

    return best_h, best_c5, len(best_h)
