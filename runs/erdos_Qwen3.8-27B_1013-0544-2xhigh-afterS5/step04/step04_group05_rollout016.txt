    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, h, beta):
    """Compute gradient of smooth_max(x, beta) w.r.t. h where x = corr(h)"""
    n = len(h)
    m = np.max(x)
    w = np.exp(np.clip(beta * (x - m), -500, 500))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    ncorr = len(x)

        wk = w[k_idx]
        if wk < 1e-15:
            continue

            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:-lag])

        else:
            mm = -lag
            grad[:n - mm] += wk * (1.0 - h[mm:])
