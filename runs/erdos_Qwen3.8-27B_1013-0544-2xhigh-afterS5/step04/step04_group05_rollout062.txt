6. If yes, t* is the answer. If no, we need to find the actual minimum.

```python
    dx = 2.0 / n

        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    f0 = np.max(b)
    f1 = np.max(a + b)
    
    active0 = (b >= f0 - 1e-14)
    if np.all(a[active0] >= -1e-14):
        return 0.0
    active1 = (a + b >= f1 - 1e-14)
    if np.all(a[active1] <= 1e-14):
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        fmid = np.max(vals)
        active = (vals >= fmid - 1e-14)

        if smin > 1e-12:
            hi = mid
        elif smax < -1e-12:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5
```

```python
    n = len(h)
    h = h.copy()
    inv_h = 1.0 - h
    inv_h_rev = inv_h[::-1]
    corr = np.correlate(h, inv_h, mode="full")
    current_max = np.max(corr)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time_fn() < 10:
            break
        order = rng.permutation(n)
        
        for j in order:
            a = np.zeros(2 * n - 1)
            a[j : j + n] += inv_h_rev
            a[n - 1 - j : 2 * n - j - 1] -= h

def slsqp_polish(h, target_sum, time_fn, max_iter=10000, betas=None):
    n = len(h)
    dx = 2.0 / n
    if betas is None:
        betas = [500, 2000, 8000]
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    
    def smooth_max(x, beta):
        m = np.max(x)
        x_shift = np.clip(beta * (x - m), -400, 400)
        return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta
    
    for beta in betas:
        if time_fn() < 15:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target_sum
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def minimize_max_line(a, b):
    """Find t in [0,1] minimizing max_k(a[k]*t + b[k])"""
    f0 = np.max(b)
    f1 = np.max(a + b)
    if np.max(np.abs(a)) < 1e-15:
        return 0.0
    scale = max(1.0, abs(f0), abs(f1))
    active0 = (b >= f0 - 1e-12 * scale)
    if np.all(a[active0] >= -1e-12):
        return 0.0
    active1 = (a + b >= f1 - 1e-12 * scale)
    if np.all(a[active1] <= 1e-12):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        vals = a * mid + b
        fmid = np.max(vals)
        thr = 1e-12 * max(1.0, abs(fmid))
        active = (vals >= fmid - thr)
        if active.sum() == 0:
            return mid
        smin = np.min(a[active])
        smax = np.max(a[active])
        if smin > 1e-12:
            hi = mid
        elif smax < -1e-12:
            lo = mid
        else:
            return mid
    f_lo = np.max(a * lo + b)
    f_hi = np.max(a * hi + b)
    return lo if f_lo <= f_hi else hi

def coord_descent(h, target_sum, max_sweeps, rng, time_fn, perturb_interval=8, base_perturb=0.008):
    n = len(h)
    h = h.copy()
    inv_h = 1.0 - h
    corr = np.correlate(h, inv_h, mode="full")
    current_max = np.max(corr)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time_fn() < 8:
            break
        order = rng.permutation(n)
        for j in order:
            a = np.zeros(2 * n - 1)
            a[j:j+n] += (1.0 - h)[::-1]
            a[n-1-j:2*n-j-1] -= h
            if np.max(np.abs(a)) < 1e-15:
                continue
            old_val = h[j]
            b = corr - a * old_val
            t_opt = minimize_max_line(a, b)
            if abs(t_opt - old_val) > 1e-13:
                delta = t_opt - old_val
                corr += delta * a
                h[j] = t_opt
        h = project_h(h, target_sum)
        inv_h = 1.0 - h
        corr = np.correlate(h, inv_h, mode="full")
        new_max = np.max(corr)
        if new_max < current_max - 1e-13:
            current_max = new_max
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= perturb_interval:
            scale = min(0.08, base_perturb * (1.0 + no_improve // perturb_interval * 0.5))
            h = project_h(h + rng.standard_normal(n) * scale, target_sum)
            inv_h = 1.0 - h
            corr = np.correlate(h, inv_h, mode="full")
            current_max = np.max(corr)
            no_improve = 0
    return h, compute_c5(h)

def slsqp_polish(h, target_sum, time_fn, max_iter=6000, betas=None):
    n = len(h)
    dx = 2.0 / n
    if betas is None:
        betas = [300, 1000, 4000, 10000]
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 12:
            break
        def obj_fn(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            m = np.max(c)
            xs = np.clip(beta * (c - m), -300, 300)
            return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta) * dx
        def con_fn(h_in):
            return h_in.sum() - target_sum
        def jac_fn(h_in):
            return np.ones(n)
        try:
            res = minimize(obj_fn, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    # Phase 1: Coordinate descent from initial
    h1, c5_1 = coord_descent(best_h, target, 5000, rng, rem,
                              perturb_interval=6, base_perturb=0.006)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1
    # Phase 2: SLSQP polish
    if rem() > 25:
        h2, c5_2 = slsqp_polish(best_h, target, rem, max_iter=5000)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    # Phase 3: Extended coordinate descent
    if rem() > 20:
        h3, c5_3 = coord_descent(best_h, target, int(rem() * 4), rng, rem,
                                  perturb_interval=5, base_perturb=0.004)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3
    # Phase 4: Random restarts
    if rem() > 50:
        for _ in range(4):
            if rem() < 25:
                break
            h_r = project_h(best_h + rng.standard_normal(n) * 0.02, target)
            h_r, c5_r = coord_descent(h_r, target, min(800, int(rem() * 2)),
                                       rng, rem, perturb_interval=5, base_perturb=0.004)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r
    # Phase 5: Final SLSQP + coordinate descent
    if rem() > 15:
        h5, c5_5 = slsqp_polish(best_h, target, rem, max_iter=4000,
                                 betas=[3000, 10000, 30000])
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
    if rem() > 10:
        h6, c5_6 = coord_descent(best_h, target, int(rem() * 3), rng, rem,
                                  perturb_interval=20, base_perturb=0.002)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
