        sh = alpha * (g - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def grad_g_k(h, n, k):
    dx = 2.0 / n
    r = k - (n - 1)
    grad = np.zeros(n)
    if r >= 0:
        grad[r:] += dx * (1.0 - h[:n - r])
        grad[:n - r] -= dx * h[r:]
    else:
        rr = -r
        grad[:n - rr] += dx * (1.0 - h[rr:])
        grad[rr:] -= dx * h[:n - rr]
    return grad

def grad_smoothed(h, n, alpha):

        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        return grad_smoothed(x, n, alpha)

def step_func_to_array(breakpoints, values, n):
    """Convert step function representation to array"""
    h = np.zeros(n)
    edges = [0] + list(breakpoints) + [2.0]
    for i in range(len(values)):
        left = edges[i]
        right = edges[i+1]
        idx = (np.arange(n) * (2.0/n)).clip(0, 2)
        mask = (idx >= left) & (idx < right)
        h[mask] = values[i]
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, 'full')) * dx)

def compute_all_g(h, n):
    return np.correlate(h, 1 - h, 'full') * (2.0 / n)

def project_feasible(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_g_k(h, n, k):
    dx = 2.0 / n
    r = k - (n - 1)
    g = np.zeros(n)
    if r >= 0:
        g[r:] += dx * (1.0 - h[:n-r])
        g[:n-r] -= dx * h[r:]
    else:
        rr = -r
        g[:n-rr] += dx * (1.0 - h[rr:])
        g[rr:] -= dx * h[:n-rr]
    return g

def gd_backtrack(h, n, deadline, max_it=300000):
    best_h, best_c = h.copy(), compute_c5(h, n)
    h = h.copy()
    stall, alpha = 0, 1.0
    for _ in range(max_it):
        if time.time() > deadline:
            break
        g = compute_all_g(h, n)
        m, ks = g.max(), int(np.argmax(g))
        if m < best_c - 1e-13:
            best_c, best_h, stall = m, h.copy(), 0
        else:
            stall += 1
        if stall > 5000:
            h = project_feasible(h + 0.03 * np.random.randn(n), n)
            stall, alpha = 0, 0.5
            continue
        grad = grad_g_k(h, n, ks)
        if np.linalg.norm(grad) < 1e-15:
            break
        a, ok = alpha, False
        for _ in range(35):
            hn = project_feasible(h - a * grad, n)
            if np.correlate(hn, 1-hn, 'full')[ks] * (2.0/n) < m - 1e-14:
                h, ok = hn, True
                break
            a *= 0.5
        if not ok:
            alpha *= 0.9
            if alpha < 1e-12:
                break
    return best_h, best_c

def slsqp_run(h, n, alpha, deadline, mi=400):
    dx = 2.0 / n
    def f(x):
        g = np.correlate(x, 1-x, 'full') * dx
        m = g.max()
        s = alpha * (g - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    def fg(x):
        g = np.correlate(x, 1-x, 'full') * dx
        m = g.max()
        grad = grad_g_k(x, n, int(np.argmax(g)))
        s = alpha * (g - m)
        sm = s.max()
        w = np.exp(s - sm)
        w /= w.sum()
        for kk in np.argsort(w)[::-1][:6]:
            grad += (w[int(kk)] / alpha) * grad_g_k(x, n, int(kk))
        return f(x), grad
    cns = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
            'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(fg, h, method='SLSQP', bounds=[(0,1)]*n,
                     constraints=cns, options={'maxiter': mi, 'ftol': 1e-15})
        return project_feasible(r.x, n)
    except Exception:
        return h
