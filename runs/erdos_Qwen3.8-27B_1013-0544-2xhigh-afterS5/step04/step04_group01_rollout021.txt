def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj

def smooth_obj_grad(x, n, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    e = np.exp(alpha * (c - c_max))
    se = e.sum()
    w = e / se
    obj = (c_max + np.log(se)) / alpha
    N = 1
    while N < 4 * n:
        N <<= 1
    H = np.fft.rfft(h, N)
    wc = np.zeros(N)
    wc[:n] = w[n-1:]
    wc[N-(n-1):] = w[:n-1]
    Wc = np.fft.rfft(wc, N)
    A = np.fft.irfft(Wc * H, N).real[:n]
    B = np.fft.irfft(np.conj(Wc) * H, N).real[:n]
    grad = 1.0 - A - B
    return obj, grad

def proj_grad_step(h, n, alpha, lr, deadline, max_steps=5000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(max_steps):
        if time.time() > deadline:

        _, grad = smooth_obj_grad(h, n, alpha)
        h = h - lr * grad
        h = project_feasible(h, n)

            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def lbb_optimize(h, n, alpha, deadline, maxiter=2000):
    def obj_pen(x):
        o, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return o + 1e4 * s * s
    
    def obj_pen_grad(x):
        o, g = smooth_obj_grad(x, n, alpha)
        s = x.sum() - n / 2.0
        return g + 2e4 * s * np.ones(n)
    
    try:
        res = minimize(obj_pen, h, jac=obj_pen_grad, method='L-BFGS-B',
                       bounds=[(0, 1)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = project_feasible(res.x, n)
        return h_new
    except:
        return h.copy()

def pair_search(h, n, deadline, deltas):

    best_c = compute_c5(h, n)
