```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_grad(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    values = m + np.log(sum_exp) / beta
    # gradient wrt x: beta * exp_shifted / sum_exp = exp_shifted / sum_exp
    grad = exp_shifted / sum_exp
    return values, grad
```

```python
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(c, beta):
    """Compute smooth max and its gradient w.r.t. c."""
    m = np.max(c)
    shifted = beta * (c - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = m + np.log(sum_exp) / beta
    grad = exp_s / sum_exp  # d(smooth_max)/dc
    return val, grad

def compute_gradient(h, beta):
    """Compute gradient of smooth_max(correlation * dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    val, p = smooth_max_and_grad(c, beta)
    # p[k] = d(smooth_max)/dc[k]
    # Now compute dObj/dh[j] = dx * sum_k p[k] * dc[k]/dh[j]
    # dc[k]/dh[j] = (1-h[j+s]) * [0<=j+s<n] - h[j-s] * [0<=j-s<n]
    # where s = (n-1) - k
    
    grad = np.zeros(n)
    for k in range(len(c)):
        if p[k] < 1e-10:
            continue
        s = (n - 1) - k
        # j appears as first factor when j+s is valid: j + s in [0, n-1], j in [0, n-1]
        # j appears as second factor when j-s is valid: j - s in [0, n-1], j in [0, n-1]
        
        # First factor: j+s in [0, n-1] => j in [-s, n-1-s] ∩ [0, n-1]
        j_min1 = max(0, -s)
        j_max1 = min(n-1, n-1-s)
        if j_min1 <= j_max1:
            for j in range(j_min1, j_max1+1):
                grad[j] += p[k] * (1.0 - h[j+s])
        
        # Second factor: j-s in [0, n-1] => j in [s, s+n-1] ∩ [0, n-1]
        j_min2 = max(0, s)
        j_max2 = min(n-1, s+n-1)
        if j_min2 <= j_max2:
            for j in range(j_min2, j_max2+1):
                grad[j] -= p[k] * h[j-s]
    
    return val * dx, grad * dx
```

```python
def project_h(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}."""
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    # clip(h - lambda, 0, 1) is decreasing in lambda
    # At lambda -> -inf: sum = n (all 1s)
    # At lambda -> +inf: sum = 0 (all 0s)
    
    lo, hi = -2.0, 2.0  # wide bounds for lambda
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid  # need to increase lambda (decrease values)
        else:
            hi = mid
        if abs(s - target_sum) < 1e-12:
            break
    
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def gradient_c5_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    exp_s = np.exp(beta * (c - m))
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        pk = p[(n-1) - s]
        if pk < 1e-14:
            continue
        a = max(0, -s)
        b = min(n, n - s)
        if a < b:
            grad[a:b] += pk * (1.0 - h[a+s:b+s])
        a = max(0, s)
        b = min(n, n + s)
        if a < b:
            grad[a:b] -= pk * h[a-s:b-s]
    return grad * dx

def project_h(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def adam_optimize(h, target_sum, beta, max_iter, lr, time_limit, t0):

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
```
