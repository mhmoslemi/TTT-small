def fix_sum(h, target):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:

    if diff > 0:

    else:

    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def sxp_one_step(h_cur, n):

    dx = 2.0 / n
    target = n / 2.0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for k in range(2 * n - 1):
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        j_idx = idx + k
        
        # overlap = sum h[j]*(1-h[i])*dx = sum(h[j] - h[j]*h[i])*dx
        # Linearize h[j]*h[i] at h_cur:
        # h_cur[j]*h[i] + h[j]*h_cur[i] - h_cur[j]*h_cur[i]
        # So h[j] - h[j]*h[i] ≈ h[j] - h_cur[j]*h[i] - h[j]*h_cur[i] + h_cur[j]*h_cur[i]
        
        # Linear coefficients for each h[m]:
        # For h[j]: coefficient is 1 - h_cur[i] (from h[j] - h[j]*h_cur[i])
        # For h[i]: coefficient is -h_cur[j] (from -h_cur[j]*h[i])
        # Constant: h_cur[j]*h_cur[i]
        
        lin = h_var[j_idx] - h_var[j_idx] * h_cur[idx] - h_var[idx] * h_cur[j_idx]
        const = np.sum(h_cur[j_idx] * h_cur[idx])
        
        if len(idx) == 1:
            constraints.append(lin * dx + const * dx <= t_var)
        else:
            constraints.append(cp.sum(lin) * dx + const * dx <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.GLPK, verbose=False)
    
    if h_var.value is not None:
        return np.array(h_var.value).flatten()
    return h_cur.copy()

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def sxp_step(h_cur, n, solver=None):
    """One SXIP step: solve LP linearization at h_cur."""
    dx = 2.0 / n
    target = n / 2.0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    
    for k in range(2 * n - 1):
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        j_idx = idx + k
        # h[j]*(1-h[i]) linearized at h_cur
        # = h[j]*(1-h_cur[i]) - h_cur[j]*h[i] + h_cur[j]*h_cur[i]
        lin = h_var[j_idx] * (1.0 - h_cur[idx]) - h_var[idx] * h_cur[j_idx]
        const = np.sum(h_cur[j_idx] * h_cur[idx])
        constraints.append(cp.sum(lin) * dx + const * dx <= t_var)
    
    solver = solver or cp.GLPK
    try:
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=solver, verbose=False)
        if h_var.value is not None:
            return np.array(h_var.value).flatten()
    except Exception:
        pass
    return h_cur.copy()

def sxp_refine(h_start, max_iter=40, damped=True):
    """Iterative SXIP refinement."""
    n = len(h_start)
    target = n / 2.0
    h_cur = h_start.copy()
    c5_cur = evaluate_c5(h_cur)
    h_prev = h_cur.copy()
    
    for it in range(max_iter):
        h_lp = sxp_step(h_cur, n)
        h_lp = fix_sum(h_lp, target)
        
        if damped:
            alpha = 0.8
            h_new = alpha * h_lp + (1 - alpha) * h_cur
            h_new = fix_sum(h_new, target)
        else:
            h_new = h_lp
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_cur - 1e-12:
            h_prev = h_cur.copy()
            h_cur = h_new
            c5_cur = c5_new
        else:
            # Try without damping
            if damped:
                c5_undamped = evaluate_c5(h_lp)
                if c5_undamped < c5_cur:
                    h_cur = h_lp
                    c5_cur = c5_undamped
                else:
                    break
            else:
                break
    
    return h_cur, c5_cur

def slsqp_optimize(h_start, tau, maxiter=300):
    """SLSQP with smooth max."""
    n = len(h_start)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def make_binary_pattern(n, pattern_type=0, rng=None):
    """Create a binary-like starting pattern."""
    if rng is None:
        rng = np.random
    if pattern_type == 0:  # random
        h = (rng.random(n) < 0.5).astype(float)
    elif pattern_type == 1:  # alternating
        h = np.zeros(n)
        h[1::2] = 1.0
    elif pattern_type == 2:  # block
        h = np.zeros(n)
        h[n//2:] = 1.0
    elif pattern_type == 3:  # evenly spaced
        step = n // (n // 2)
        positions = np.arange(0, n, step)
        h = np.zeros(n)
        if len(positions) > n // 2:
            positions = positions[:n//2]
        h[positions] = 1.0
    else:  # golden ratio spacing
        positions = np.unique((np.arange(n) * 0.618).astype(int) % n)
        positions = positions[:n//2]
        h = np.zeros(n)
        h[positions] = 1.0
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(80, 0.5)
    
    best_c5 = evaluate_c5(best_h)
    
    def rem():

        if c5_sxip < best_c5:

    # Phase 2: SLSQP polish

            h_new, c5_new = slsqp_optimize(best_h, tau, 300)
            if c5_new is not None and c5_new < best_c5:

    for tn in [80, 100, 140, 160]:

        target = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, target)
        
        # SXIP on this resolution
        h_ref, c5_ref = sxp_refine(h_t, max_iter=20, damped=True)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        else:

                if c5_new is not None and c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                h_ref = h_new

    best_c5 = evaluate_c5(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import cvxpy as cp
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def sxp_step(h_cur, n):
    """One SXIP step: solve LP linearization of the max-overlap at h_cur."""
    dx = 2.0 / n
    target = n / 2.0
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    for out_idx in range(2 * n - 1):
        k = out_idx - (n - 1)
        i_min = max(0, -k)
        i_max = min(n - 1, n - 1 - k)
        if i_min > i_max:
            continue
        idx = np.arange(i_min, i_max + 1)
        j_idx = idx + k
        # Linearize h[j]*(1-h[i]) at h_cur:
        # (1-h_cur[i])*h[j] - h_cur[j]*h[i] + h_cur[i]*h_cur[j]
        lin = h_var[j_idx] * (1.0 - h_cur[idx]) - h_var[idx] * h_cur[j_idx]
        const = np.dot(h_cur[idx], h_cur[j_idx])
        constraints.append(cp.sum(lin) * dx + const * dx <= t_var)
    try:
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        if h_var.value is not None:
            return np.array(h_var.value).flatten()
    except Exception:
        pass
    return h_cur.copy()

def sxp_refine(h_start, max_iter=40, damped=True):
    n = len(h_start)
    target = n / 2.0
    h_cur = h_start.copy()
    c5_cur = evaluate_c5(h_cur)
    for it in range(max_iter):
        h_lp = fix_sum(sxp_step(h_cur, n), target)
        if damped:
            h_new = 0.7 * h_lp + 0.3 * h_cur
            h_new = fix_sum(h_new, target)
        else:
            h_new = h_lp
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_cur - 1e-13:
            h_cur = h_new
            c5_cur = c5_new
        else:
            if damped and it >= 3:
                c5_und = evaluate_c5(h_lp)
                if c5_und < c5_cur:
                    h_cur = h_lp
                    c5_cur = c5_und
                    continue
            break
    return h_cur, c5_cur

def slsqp_opt(h_start, tau, maxiter=300):
    n = len(h_start)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def make_binary(n, ptype, rng):
    h = np.zeros(n)
    if ptype == 0:
        h = (rng.random(n) < 0.5).astype(float)
    elif ptype == 1:
        h[1::2] = 1.0
    elif ptype == 2:
        h[n // 2:] = 1.0
    elif ptype == 3:
        step = n // (n // 2)
        pos = np.arange(0, n, step)[:n // 2]
        h[pos] = 1.0
    else:
        pos = np.unique((np.arange(n) * 0.6180339887).astype(int) % n)[:n // 2]
        h[pos] = 1.0
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(80, 0.5)
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SXIP from parent
    if rem() > 80:
        h_s, c5_s = sxp_refine(best_h, max_iter=35, damped=True)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 2: SLSQP polish
    if rem() > 30:
        for tau in [300, 800, 2000]:
            if rem() < 15:
                break
            h_new, c5_new = slsqp_opt(best_h, tau, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 3: Multi-resolution SXIP
    for tn in [60, 80, 100, 140, 160]:
        if rem() < 50:
            break
        if tn % 2 != 0:
            tn += 1
        target = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, target)
        h_ref, c5_ref = sxp_refine(h_t, max_iter=25, damped=True)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        else:
            h_ref = h_t
        if rem() > 20:
            for tau in [400, 1200]:
                if rem() < 10:
                    break
                h_new, c5_new = slsqp_opt(h_ref, tau, 200)
                if c5_new is not None and c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new
                h_ref = h_new

    # Phase 4: Binary restarts with SXIP
    hn = len(best_h)
    for pt in range(5):
        if rem() < 40:
            break
        h_bin = make_binary(hn, pt, rng)
        h_ref, c5_ref = sxp_refine(h_bin, max_iter=30, damped=True)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        if rem() > 15:
            h_new, c5_new = slsqp_opt(h_ref, 600, 250)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Interpolation + SXIP at finer grids
    for tn in [200, 240]:
        if rem() < 50:
            break
        if tn % 2 != 0:
            tn += 1
        target = tn / 2.0
        h_f = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_f = fix_sum(h_f, target)
        h_ref, c5_ref = sxp_refine(h_f, max_iter=20, damped=True)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        if rem() > 15:
            h_new, c5_new = slsqp_opt(h_ref, 1000, 200)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 6: Random perturbations
    hn2 = len(best_h)
    for _ in range(20):
        if rem() < 15:
            break
        scale = 0.03 + 0.15 * rng.random()
        hp = best_h + rng.normal(0, scale, hn2)
        hp = fix_sum(hp, hn2 / 2.0)
        h_new, c5_new = slsqp_opt(hp, 800, 200)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Final SXIP + high-tau polish
    if rem() > 30:
        h_s, c5_s = sxp_refine(best_h, max_iter=20, damped=False)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    if rem() > 15:
        h_new, c5_new = slsqp_opt(best_h, 5000, 400)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
