def project(h, target_sum):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

def grad_shift(h, k):
    """Gradient of corr_k = sum h[i](1-h[i+k]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            return 1 - 2 * h
        g[:n-k] += (1 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1 - h[:n-p])

import numpy as np
import time
from itertools import combinations

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            return 1.0 - 2.0 * h
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_shift(h, k, n)
    return grad * (2.0 / n)

def optimize(h0, target_sum, deadline, alpha=100, lr=0.008):
    n = len(h0)
    best = project(h0, target_sum).copy()
    best_c5 = compute_c5(best)
    cur_lr = lr
    cur_alpha = alpha
    patience = 0
    
    while time.time() < deadline:
        g = smooth_grad(best, cur_alpha)
        accepted = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = project(best - cur_lr * scale * g, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best = h_new
                best_c5 = c5_new
                accepted = True
                break
        if not accepted:
            patience += 1
            cur_lr *= 0.93
            if patience % 40 == 0:
                cur_alpha = min(cur_alpha * 1.4, 600.0)
            if patience > 300 or cur_lr < 1e-6:
                break
        else:
            patience = 0
            cur_lr = min(cur_lr * 1.01, 0.05)
    
    return best, best_c5

def exhaustive_binary(m, deadline):
    """Try all binary patterns with exactly m/2 ones"""
    half = m // 2
    best_c5 = 1.0
    best_bits = None
    indices = range(m)
    for bits in combinations(indices, half):
        if time.time() > deadline:
            break
        h = np.zeros(m)
        h[list(bits)] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_bits = bits
    if best_bits is not None:
        h = np.zeros(m)
        h[list(best_bits)] = 1.0
        return h, best_c5
    return None, 1.0

def interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Check for initial_h_values

    try:
        if initial_h_values is not None:
            h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        pass
    
    # === Phase 1: Exhaustive binary search ===
    p1_deadline = t0 + min(budget_s * 0.2, 200.0)
    
    for m in [12, 14, 16, 18, 20]:
        if time.time() > p1_deadline:

        m_deadline = min(time.time() + (p1_deadline - time.time()) * 0.5, p1_deadline)
        h_bin, c5_bin = exhaustive_binary(m, m_deadline)
        if h_bin is not None and c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            best_n = m
    
    # Also try structured patterns at low res
    for m in [20, 30, 40]:
        if time.time() > p1_deadline:

        # Alternating
        h_alt = np.zeros(m)
        h_alt[0::2] = 1.0 if m % 2 == 0 else 0.5
        # Adjust sum

        s = h_alt.sum()
        if s != target:

        c5 = compute_c5(h_alt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_alt.copy()
            best_n = m
        
        # Sinusoidal
        x = np.arange(m)
        h_sin = (np.sin(2 * np.pi * x / m) > 0).astype(float)
        if h_sin.sum() != target:
            # Adjust
            h_sin = project(h_sin, target)
        c5 = compute_c5(h_sin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sin.copy()
            best_n = m
    
    # === Phase 2: Gradient optimization from multiple starts ===
    p2_deadline = t0 + budget_s * 0.7
    
    starts = []
    # From best binary
    if best_h is not None:
        for n in [80, 120, 200]:
            h_interp = interpolate(best_h, n)
            h_interp = project(h_interp, n / 2.0)
            starts.append(h_interp)
    # From initial
    if h_init is not None:

    # Random starts

        h_r = 0.5 + 0.4 * rng.randn(n)
        starts.append(project(h_r, n / 2.0))
    
    # Remove duplicates (same length and similar)
    unique_starts = []
    for s in starts:
        is_dup = False
        for u in unique_starts:
            if len(s) == len(u) and np.max(np.abs(s - u)) < 0.01:
                is_dup = True
                break
        if not is_dup:
            unique_starts.append(s)
    starts = unique_starts
    
    n_starts = len(starts)
    for i, h_start in enumerate(starts):
        if time.time() > p2_deadline:
            break
        n = len(h_start)
        remaining = p2_deadline - time.time()

        opt_deadline = time.time() + max(10, per_start)

        n_restarts = 8
        for r in range(n_restarts):

                break

            h_pert = best_h + pert_size * rng.randn(n_best)

            h_opt, c5_opt = optimize(h_pert, n_best / 2.0, opt_deadline, alpha=150, lr=0.005)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # === Phase 4: Final polish ===
    p4_deadline = t0 + budget_s - 3
    if best_h is not None and time.time() < p4_deadline:
        best_h, best_c5 = optimize(best_h, len(best_h) / 2.0, p4_deadline, alpha=300, lr=0.003)
    
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)

def optimize_nesterov(h0, target_sum, deadline, alpha_init=80, lr_init=0.01):
    n = len(h0)
    h = project(h0, target_sum).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    beta = 0.9
    alpha = alpha_init
    lr = lr_init
    patience = 0
    
    while time.time() < deadline:
        g = smooth_grad(h, alpha)
        v = beta * v - lr * g
        h_new = project(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            lr = min(lr * 1.01, 0.08)
        else:
            h = h_new  # still update h for momentum
            patience += 1
            lr *= 0.95
            if patience % 50 == 0:
                alpha = min(alpha * 1.3, 500.0)
                v *= 0.5  # dampen momentum
            if patience > 500 or lr < 1e-6:
                break
    
    return best_h, best_c5
