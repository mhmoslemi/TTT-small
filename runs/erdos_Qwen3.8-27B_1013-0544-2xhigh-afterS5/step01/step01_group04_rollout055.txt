import time

def compute_c5(h, dx):

    return np.max(corr) * dx

    if tau > 1e-10:

        return max_val + np.log(np.sum(np.exp(tau * shifted))) / tau
    return max_val

    """Differential Evolution for global exploration"""
    target = n / 2.0
    rng = np.random.RandomState(42)

    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    for i in range(1, pop_size):
        pop[i] = rng.uniform(0, 1, n)
        pop[i] = project_feasible(pop[i], n)
    
    fitness = np.array([compute_c5(pop[i], dx) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start = time.time()
    gen = 0
    
    while time.time() - start < budget:
        gen += 1
        # Decay F
        F_current = F * (1.0 - 0.5 * gen / max(1, gen + 50))  # gentle decay
        
        for i in range(pop_size):
            if time.time() - start > budget:
                break
            
            # Select 3 distinct parents
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            tri = rng.choice(np.where(mask)[0], 3, replace=False)
            a, b, c = pop[tri[0]], pop[tri[1]], pop[tri[2]]

                pop[i] = trial
                fitness[i] = fit_trial
                if fit_trial < best_c5:
                    best_c5 = fit_trial
                    best_h = trial.copy()

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

        c5 = smooth_max_c5(h_proj, dx, tau)
        penalty = 1000.0 * (np.sum(x) - target) ** 2

        eps = 1e-7
        grad = np.zeros(n)
        for i in range(n):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad
    
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_obj(x, dx, n, tau, penalty_w):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx

    smooth = mx + np.log(np.sum(np.exp(tau * shifted))) / tau
    pen = penalty_w * (np.sum(x) - n / 2.0) ** 2
    return smooth + pen
