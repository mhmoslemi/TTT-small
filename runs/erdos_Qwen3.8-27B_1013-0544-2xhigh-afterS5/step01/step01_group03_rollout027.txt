        wm = w[m]
        if wm < 1e-15:
            continue
        # First term: (1-h[j-km]) for j-km in [0,n-1]
        # j ranges from km to n-1+km (if km >= 0) or 0 to n-1-km (if km < 0)
        if km >= 0:
            j_start = km
            j_end = n
            # j-km ranges from 0 to n-1-km
            grad[j_start:j_end] += wm * (1.0 - h[:n-km])
            # Second term: -h[j+km] for j+km in [0,n-1]
            # j ranges from 0 to n-1-km
            if km < n:
                grad[:n-km] -= wm * h[km:]
        else:
            # km < 0, let km_abs = -km
            km_abs = -km
            # First term: j - km = j + km_abs, valid when 0 <= j+km_abs < n, i.e., j in [0, n-1-km_abs]
            j_end = n - km_abs
            grad[:j_end] += wm * (1.0 - h[km_abs:])
            # Second term: j + km = j - km_abs, valid when 0 <= j-km_abs < n, i.e., j in [km_abs, n-1]
            grad[km_abs:] -= wm * h[:j_end]
    
    grad *= dx
    return f, grad

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()

    if abs(diff) < 1e-14:

    for i in range(1, n):

    h_new = np.clip(h_new, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    y = corr * dx
    y_max = np.max(y)
    exp_y = np.exp((y - y_max) * tau)
    sum_exp = np.sum(exp_y)
    w = exp_y / sum_exp
    f = y_max + np.log(sum_exp) / tau
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        km = m - (n - 1)
        wm = w[m]
        if wm < 1e-15:
            continue
        if km >= 0:
            if n - km > 0:
                grad[km:n] += wm * (1.0 - h[:n - km])
                grad[:n - km] -= wm * h[km:]
        else:
            ka = -km
            if n - ka > 0:
                grad[:n - ka] += wm * (1.0 - h[ka:])
                grad[ka:] -= wm * h[:n - ka]
    grad *= dx
    return f, grad


def thue_morse(n):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def remaining_time(t0, budget_s):
    return budget_s - (time.time() - t0)


def bb_descent(h_start, n, t0, budget_s, tau_start=100, max_iter=400, min_remain=25):
    target = n / 2.0
    h_curr = h_start.copy()
    h_prev = h_curr + np.random.normal(0, 0.01, n)
    h_prev = fix_sum(h_prev, target)
    _, grad_prev = smooth_obj_and_grad(h_prev, tau_start)
    _, grad_curr = smooth_obj_and_grad(h_curr, tau_start)
    tau = tau_start
    best_local = h_curr.copy()
    best_c5_local = evaluate_c5(best_local)
    f_curr = smooth_obj_and_grad(h_curr, tau)[0]

    for it in range(max_iter):
        if remaining_time(t0, budget_s) < min_remain:
            break
        s = h_curr - h_prev
        y_vec = grad_curr - grad_prev
        s_dot_s = np.dot(s, s)
        s_dot_y = np.dot(s, y_vec)
        if s_dot_s > 1e-15 and abs(s_dot_y) > 1e-15:
            gamma = np.clip(s_dot_s / abs(s_dot_y), 1e-6, 0.5)
        else:
            gamma = 0.01

        # Backtracking line search
        alpha = gamma
        f_new = f_curr
        h_new = h_curr
        for bt in range(8):
            h_try = h_curr - alpha * grad_curr
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = fix_sum(h_try, target)
            f_try, _ = smooth_obj_and_grad(h_try, tau)
            if f_try < f_curr - 1e-12 or bt == 7:
                f_new = f_try
                h_new = h_try
                break
            alpha *= 0.5

        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5_local:
            best_local = h_new.copy()
            best_c5_local = c5_new

        h_prev = h_curr
        grad_prev = grad_curr
        h_curr = h_new
        f_curr = f_new
        _, grad_curr = smooth_obj_and_grad(h_curr, tau)
        if it % 40 == 39:
            tau = min(int(tau * 1.8), 8000)

    return best_local, best_c5_local


def slsqp_objective(x, n, tau):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    y = corr * dx
    y_max = np.max(y)
    return float(y_max + np.log(np.sum(np.exp((y - y_max) * tau))) / tau)


def slsqp_constraint(x, n):
    return np.sum(x) - n / 2.0


def slsqp_constraint_jac(x, n):
    return np.ones(n)


def slsqp_polish(h_start, n, t0, budget_s, tau=300):
    if remaining_time(t0, budget_s) < 15:
        return h_start, evaluate_c5(h_start)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'jac': slsqp_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_objective, h_start, args=(n, tau), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': 250, 'ftol': 1e-14})
        h_new = fix_sum(res.x, n / 2.0)
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_start, evaluate_c5(h_start)


def lbfgs_penalized(h_start, n, t0, budget_s, tau=200, penalty=50.0):
    if remaining_time(t0, budget_s) < 15:
        return h_start, evaluate_c5(h_start)
    target = n / 2.0

    def obj_pen(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        y = corr * dx
        y_max = np.max(y)
        sm = y_max + np.log(np.sum(np.exp((y - y_max) * tau))) / tau
        pen = penalty * (np.sum(x) - target) ** 2
        return float(sm + pen)

    def obj_pen_grad(x):
        f, g = smooth_obj_and_grad(x, tau)
        g = g + 2.0 * penalty * (np.sum(x) - target) * np.ones(n)
        return f + penalty * (np.sum(x) - target) ** 2, g

    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h_start, jac=obj_pen_grad, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8})
        h_new = fix_sum(res.x, target)
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h_start, evaluate_c5(h_start)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: BB gradient descent, low tau
    h1, c5_1 = bb_descent(best_h, n, t0, budget_s, tau_start=50, max_iter=300, min_remain=80)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: BB with higher tau
    h2, c5_2 = bb_descent(best_h, n, t0, budget_s, tau_start=300, max_iter=300, min_remain=50)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: L-BFGS-B with penalty
    h3, c5_3 = lbfgs_penalized(best_h, n, t0, budget_s, tau=200, penalty=30.0)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: SLSQP polish
    h4, c5_4 = slsqp_polish(best_h, n, t0, budget_s, tau=400)
    if c5_4 < best_c5:
        best_h, best_c5 = h4, c5_4

        tm = thue_morse(n).astype(float)
        tm_start = 0.25 + 0.5 * tm
        tm_start = fix_sum(tm_start, n / 2.0)
        h5, c5_5 = bb_descent(tm_start, n, t0, budget_s, tau_start=100, max_iter=200, min_remain=40)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
        h5b, c5_5b = lbfgs_penalized(best_h, n, t0, budget_s, tau=500, penalty=50.0)

    # Phase 6: Binary-inspired restart

        h_bin[idx[:n // 2]] = 1.0
        h6, c5_6 = bb_descent(h_bin, n, t0, budget_s, tau_start=80, max_iter=250, min_remain=35)
        if c5_6 < best_c5:

        h6b, c5_6b = slsqp_polish(best_h, n, t0, budget_s, tau=500)

    # Phase 7: Random restarts

            h_rand = np.random.random(n) * 0.5 + 0.25
            h_rand = fix_sum(h_rand, n / 2.0)

            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_2 if c5_2 < best_c5 else h_r

            h_rb, c5_rb = lbfgs_penalized(h_r, n, t0, budget_s, tau=400, penalty=40.0)
            if c5_rb < best_c5:
                best_h, best_c5 = h_rb, c5_rb

    # Phase 8: Final high-tau polish
    if remaining_time(t0, budget_s) > 15:
        h_f, c5_f = bb_descent(best_h, n, t0, budget_s, tau_start=2000, max_iter=150, min_remain=5)

    if remaining_time(t0, budget_s) > 12:

    # Final constraint fix

    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
