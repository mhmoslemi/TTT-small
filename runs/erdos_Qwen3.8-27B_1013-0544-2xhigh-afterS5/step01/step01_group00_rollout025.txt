from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    """Exact Euclidean projection onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Binary search for the shift λ
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_active(h):
    """Subgradient of max_k corr_k(h) at h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.argmax(corr)
    s = m - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    # Contribution from h appearing as first factor h[i+s]:
    # h[j] = h[i+s] when i = j-s, contributes (1-h[i]) = (1-h[j-s])
    idx1 = j - s
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]]) * dx
    # Contribution from h appearing in (1-h[i]):
    # h[j] = h[i] when i = j, and the term is h[i+s]*(1-h[i]),
    # d/dh[j] = -h[j+s]
    idx2 = j + s
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += (-h[idx2[mask2]]) * dx
    return grad

def slsqp_minmax(h0, maxiter=1000, ftol=1e-13):
    """SLSQP with t-variable for min-max problem"""
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.concatenate([h0, [c5 + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def obj(v):
        return v[-1]
    
    def cons_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        return t - np.correlate(h, 1.0 - h, mode='full') * dx
    
    def cons_eq(v):
        return np.sum(v[:n]) - n * 0.5
    
    def cons_eq_jac(v):
        jj = np.zeros(len(v))
        jj[:n] = 1.0
        return jj
    
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': cons_ineq},
                                {'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)

def lse_optimize(h0, temp, maxiter=500, ftol=1e-13):
    """Optimize log-sum-exp approximation"""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    
    def con_eq(h):
        return np.sum(h) - n * 0.5
    
    def con_jac(h):
        return np.ones(n)
    
    res = minimize(obj, h0.copy(), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

    """Adaptive subgradient method"""

    target = n * 0.5
    h = h0.copy()
    best_h = h.copy()

        if c5 < best_c5:

            best_c5 = c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_active(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.argmax(corr)
    s = m - (n - 1)
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j - s
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]]) * dx
    idx2 = j + s
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] += (-h[idx2[mask2]]) * dx
    return grad

def slsqp_minmax(h0, maxiter=1000, ftol=1e-13):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.concatenate([h0, [c5 + 0.005]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]

    def obj(v):
        return v[-1]

    def cons_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        return t - np.correlate(h, 1.0 - h, mode='full') * dx

    def cons_eq(v):
        return np.sum(v[:n]) - n * 0.5

    def cons_eq_jac(v):
        jj = np.zeros(len(v))
        jj[:n] = 1.0
        return jj

    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': cons_ineq},
                                {'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)

def lse_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def lse_optimize(h0, temp, maxiter=400, ftol=1e-13):
    n = len(h0)

    def con_eq(h):
        return np.sum(h) - n * 0.5

    def con_jac(h):
        return np.ones(n)

    res = minimize(lse_objective, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

def subgrad_method(h0, time_limit=30, alpha0=0.003, decay=0.9992):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    alpha = alpha0
    for _ in range(80000):
        if time.time() - t0 > time_limit:
            break
        g = grad_active(h)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        h = h - alpha * g
        h = project_h(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        alpha *= decay
    return best_h, best_c5

def interp_resize(h, n_new):
    x_old = np.arange(len(h)) / len(h)
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new * 0.5)

def multi_start_slsqp(h_best, rng, n_starts=6, time_limit=200):
    n = len(h_best)
    t0 = time.time()
    best_h, best_c5 = h_best, compute_c5(h_best)
    for i in range(n_starts):
        if time.time() - t0 > time_limit:
            break
        scale = 0.015 + 0.008 * i
        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h, c5 = slsqp_minmax(h_p, maxiter=500, ftol=1e-12)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    return best_h, best_c5
