def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_capped_simplex(x, s):
    n = len(x)
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_at_lag(h, k):
    n = len(h)
    idx = np.arange(n)
    g = np.zeros(n)
    if k >= 0:
        m1 = idx >= k
        m2 = idx + k <= n - 1
        g[m1] += 1.0 - h[idx[m1] - k]
        g[m2] -= h[idx[m2] + k]
    else:
        kp = -k
        m1 = idx <= n - 1 - kp
        m2 = idx >= kp
        g[m1] += 1.0 - h[idx[m1] + kp]
        g[m2] -= h[idx[m2] - kp]
    return g

def weighted_grad(h, corr, top_m):
    n = len(h)
    sort_idx = np.argsort(corr)[::-1]
    top = sort_idx[:min(top_m, n * 2 - 1)]
    c_vals = corr[top]
    w = np.exp(20.0 * (c_vals - c_vals.max()))
    w /= w.sum()
    g = np.zeros(len(h))
    for i in range(len(top)):
        if w[i] > 1e-8:
            k = top[i] - (n - 1)
            g += w[i] * grad_at_lag(h, k)
    return g

            for top_m, n_it, lr in [(10, 2000, 0.08), (2, 3000, 0.03)]:
                if elapsed() > budget_s - 15:
                    break
                h_r, c5_cur = subgrad_loop(h_r, dx, target, top_m, n_it, lr, elapsed, budget_s - 15)
                if c5_cur < best_c5:
                    best_h = h_r.copy()
                    best_c5 = c5_cur
                    best_n = n

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_capped_simplex(best_h, best_n / 2.0)
    dx_f = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, best_n
