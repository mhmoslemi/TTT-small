                probs = contrib / tot
                j = rng.choice(n - k, p=probs)
                delta = rng.uniform(0.01, 0.06)
                if rng.rand() < 0.6:
                    h[j + k] -= delta
                    h[j] += delta
                else:
                    j2 = rng.randint(0, n)
                    h[j + k] -= delta
                    h[j2] += delta
        elif k < 0:
            p = -k
            contrib = h[:n - p] * (1.0 - h[p:])
            tot = np.sum(contrib)
            if tot > 1e-10:
                probs = contrib / tot
                j = rng.choice(n - p, p=probs)
                delta = rng.uniform(0.01, 0.06)
                if rng.rand() < 0.6:
                    h[j] -= delta
                    h[j + p] += delta
                else:
                    j2 = rng.randint(0, n)
                    h[j] -= delta
                    h[j2] += delta
        else:
            j1 = rng.randint(0, n)
            j2 = rng.randint(0, n)
            if j1 == j2:
                continue
            delta = rng.uniform(0.01, 0.06)
            if h[j1] < 0.5:
                h[j1] += delta
                h[j2] -= delta
            else:
                h[j1] -= delta
                h[j2] += delta

        h = project_onto_simplex_box(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elif rng.rand() < np.exp(-(c5 - best_c5) / T):
            pass
        else:
            h = best_h.copy()
        T = max(0.001, T * 0.9995)
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.clip(0.5 + 0.3 * rng.randn(100), 0, 1)
        best_h = project_onto_simplex_box(best_h, 50.0)

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: Levelling optimization at current resolution
    rem = budget_s - (time.time() - t0)
    dl = t0 + rem * 0.25
    h1, c5_1 = optimize_levelling(best_h, best_n / 2.0, dl, rng, margin_init=0.025)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: SA at lower resolution for exploration
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        n_sa = min(best_n, 60)
        h_sa = interpolate_to_n(best_h, best_n, n_sa)
        h_sa = project_onto_simplex_box(h_sa, n_sa / 2.0)
        dl = t0 + budget_s * 0.45
        h_sa, c5_sa = targeted_sa(h_sa, n_sa / 2.0, dl, rng, T_init=0.05)
        # Upsample and refine
        h_up = interpolate_to_n(h_sa, n_sa, best_n)
        h_up = project_onto_simplex_box(h_up, best_n / 2.0)
        dl2 = t0 + budget_s * 0.6
        h_up, c5_up = optimize_levelling(h_up, best_n / 2.0, dl2, rng, margin_init=0.02)
        if c5_up < best_c5:
            best_h, best_c5 = h_up, c5_up

    # Phase 3: Try smoothed gradient at different alpha for diversification
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        # Perturb current best
        h_pert = best_h + rng.randn(best_n) * 0.03
        h_pert = project_onto_simplex_box(h_pert, best_n / 2.0)
        dl = t0 + budget_s * 0.75
        h_s, c5_s = optimize_smoothed(h_pert, best_n / 2.0, dl, rng,
                                       alpha_init=40, alpha_end=200)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 4: Try higher resolution if we haven't
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n_try = max(best_n, 200)
        if n_try != best_n:
            h_hi = interpolate_to_n(best_h, best_n, n_try)
            h_hi = project_onto_simplex_box(h_hi, n_try / 2.0)
        else:
            h_hi = best_h.copy()
        dl = t0 + budget_s * 0.88
        h_hi, c5_hi = optimize_levelling(h_hi, n_try / 2.0, dl, rng, margin_init=0.015)
        if c5_hi < best_c5:
            best_h, best_c5, best_n = h_hi, c5_hi, n_try

    # Phase 5: Final polish with high alpha
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        dl = t0 + budget_s - 3
        best_h, best_c5 = optimize_smoothed(best_h, best_n / 2.0, dl, rng,
                                             alpha_init=200, alpha_end=500)

    best_h = project_onto_simplex_box(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
