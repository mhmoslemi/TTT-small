    """Flip pairs of bits to maintain sum constraint."""
    h = h.copy()
    ones = np.where(h == 1.0)[0]

    if len(ones) == 0 or len(zeros) == 0:
        return h

        h[i] = 0.0
        h[j] = 1.0
        # Update lists

        zeros = np.where(h == 0.0)[0]

                    fits[i] = compute_c5(pop[i])
                best_i = np.argmin(fits)
                best_h = pop[best_i].copy()
                best_f = fits[best_i]
                stall = 0
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = None

    # Phase 1: Refine parent
    if h_parent is not None:
        n = len(h_parent)
        h_s = project_simplex_box(h_parent.copy(), n / 2.0)
        dl = t0 + budget_s * 0.28
        h, c5 = local_search(h_s, n / 2.0, dl, rng, 25, 280)
        best_h, best_c5, best_n = h, c5, n

    # Phase 2: GA at n=200
    n_ga = 200
    dl_ga = t0 + budget_s * 0.52
    h_ga, c5_ga = ga_search(n_ga, dl_ga, rng, pop_size=28)
    dl_ref = min(t0 + budget_s * 0.66, time.time() + budget_s * 0.13)
    h_r, c5_r = local_search(h_ga, n_ga / 2.0, dl_ref, rng, 40, 350)
    if c5_r < best_c5:
        best_h, best_c5, best_n = h_r, c5_r, n_ga
    elif c5_ga < best_c5:
        best_h, best_c5, best_n = h_ga, c5_ga, n_ga

    # Phase 3: Diverse restarts
    inits = [200, 160, 240, 140]
    for i, n_try in enumerate(inits):
        if time.time() > t0 + budget_s - 50:
            break
        # Structured + noise initialization
        h_init = np.zeros(n_try)
        blk = max(n_try // 10, 1)
        for b in range(n_try // blk + 1):
            if b % 2 == 0:
                h_init[b*blk:min((b+1)*blk, n_try)] = 1.0
        h_init += rng.randn(n_try) * 0.15
        h_init = project_simplex_box(h_init, n_try / 2.0)
        dl = min(t0 + budget_s - 30, time.time() + budget_s * 0.09)
        h, c5 = local_search(h_init, n_try / 2.0, dl, rng, 20, 250)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n_try

    # Phase 4: Final polish with high alpha
    if best_h is not None and time.time() < t0 + budget_s - 12:
        dl = t0 + budget_s - 4
        h, c5 = local_search(best_h, len(best_h) / 2.0, dl, rng, 120, 600)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = project_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
