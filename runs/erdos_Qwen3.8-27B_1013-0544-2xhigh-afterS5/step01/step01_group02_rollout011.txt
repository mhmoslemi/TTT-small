def project_feasible(h, n):

            lo = mid
        else:
            hi = mid

For np.correlate(a, v, mode='full'), the result has length len(a) + len(v) - 1 = 2n - 1.
The result at index m is: ∑_i a[i] * v[m-i]

where the sum is over all i such that 0 ≤ i < len(a) and 0 ≤ m-i < len(v).

So for a = h, v = 1-h:
result[m] = ∑_i h[i] * (1-h[m-i])

The "lag" in the continuous sense corresponds to:
- The zero lag (k=0) is at index m = n-1 (where the two sequences fully overlap)
- Positive lag k (h shifted right) corresponds to index m = n-1+k
- Negative lag k corresponds to index m = n-1+k

So the C₅ value is max_m result[m] * dx.

Now, the subgradient of result[m] with respect to h[j]:
result[m] = ∑_i h[i] * (1 - h[m-i])

h[j] appears:
1. As h[i] when i = j: contribution is h[j] * (1-h[m-j]), derivative w.r.t. h[j] is (1-h[m-j])
2. As h[m-i] when m-i = j, i.e., i = m-j: contribution is h[m-j] * (1-h[j]), derivative w.r.t. h[j] is -h[m-j]

So ∂result[m]/∂h[j] = (1 - h[m-j]) - h[m-j] = 1 - 2*h[m-j]

where h[m-j] is 0 if m-j < 0 or m-j ≥ n.

    # Phase 1: Subgradient on initial
    if remaining() > 120:
        h_s, c5_s = subgrad_optimize(h0, n0, seed, remaining() - 120, top_m=10, beta=0.9)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    # Phase 2: Smooth L-BFGS-B
    if remaining() > 60:
        h_sm, c5_sm = smooth_optimize(best_h, n0, alpha=3000, time_limit=min(remaining() - 30, 80))
        if c5_sm < best_c5:
            best_h, best_c5 = h_sm, c5_sm
    # Phase 3: Subgradient with varied parameters
    if remaining() > 80:
        for tm, bt in [(5, 0.85), (20, 0.9), (35, 0.8)]:
            if remaining() < 30:
                break
            h_s, c5_s = subgrad_optimize(best_h, n0, seed + tm * 7, min(remaining() - 25, 35), top_m=tm, beta=bt)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s
    # Phase 4: Different n values
    n_candidates = sorted(set([n0 - 20, n0 + 20, n0 + 40, n0 - 40, n0 + 60]), reverse=True)
    for n_try in n_candidates:
        if remaining() < 90 or n_try < 24:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False) + 1.0 / best_n
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_interp = np.interp(x_new, x_old, np.concatenate([best_h, best_h[:1]]))
        h_interp = project_feasible(h_interp, n_try)
        h_s, c5_s = subgrad_optimize(h_interp, n_try, seed + n_try, min(remaining() - 70, 35), top_m=10, beta=0.9)
        if remaining() > 40:
            h_sm2, c5_sm2 = smooth_optimize(h_s, n_try, alpha=5000, time_limit=min(remaining() - 30, 25))
            if c5_sm2 < c5_s:
                h_s, c5_s = h_sm2, c5_sm2
        if c5_s < best_c5:
            best_h, best_c5, best_n = h_s, c5_s, n_try
    # Phase 5: Final polish
    if remaining() > 40:
        h_sm, c5_sm = smooth_optimize(best_h, best_n, alpha=10000, time_limit=min(remaining() - 15, 25))
        if c5_sm < best_c5:
            best_h, best_c5 = h_sm, c5_sm
    if remaining() > 20:
        h_cd, c5_cd = coord_descent(best_h, best_n, min(remaining() - 5, 15), seed + 999)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    # Final subgradient burst
    if remaining() > 15:
        h_s, c5_s = subgrad_optimize(best_h, best_n, seed + 7777, remaining() - 3, top_m=8, beta=0.92)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
