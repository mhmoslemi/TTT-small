    best_local_val = evaluate_c5(h)
    for it in range(n_iter):
        if (it % 50 == 0) and (budget_s - (time.time() - t0) < 5):
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        x = tau * corr
        x_max = np.max(x)
        w = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
        grad = compute_grad(h, w)
        h = h - lr * grad
        h = fix_sum(h, target)
        if kick_every > 0 and it % kick_every == 0 and it > 0:
            h = h + np.random.normal(0, kick_scale, n)
            h = fix_sum(h, target)
        if it % 20 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_local_val:
                best_local = h.copy()
                best_local_val = c5
    h = fix_sum(best_local, target)
    return h, evaluate_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SLSQP with increasing tau from initial
    for tau in [300, 800, 2000, 5000]:
        if budget_s - (time.time() - t0) < 20:
            break
        h_new, c5_new = _slsqp(best_h, tau, 600, t0, budget_s)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Projected gradient with analytical gradients
    if budget_s - (time.time() - t0) > 200:
        for tau in [50, 100, 200, 500]:
            if budget_s - (time.time() - t0) < 40:
                break
            h_new, c5_new = _proj_grad(best_h, tau, 0.002, 5000, 500, 0.03, t0, budget_s)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: SLSQP fine-tune at high tau
    for tau in [5000, 10000, 20000]:
        if budget_s - (time.time() - t0) < 20:
            break
        h_new, c5_new = _slsqp(best_h, tau, 800, t0, budget_s)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Random restarts from perturbed best
    n = len(best_h)
    target = n / 2.0
    for trial in range(30):
        if budget_s - (time.time() - t0) < 30:
            break
        for scale in [0.03, 0.08, 0.15]:
            h_try = best_h + np.random.normal(0, scale, n)
            h_try = fix_sum(h_try, target)
            h_new, c5_new = _slsqp(h_try, 2000, 400, t0, budget_s)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                break

    # Phase 5: Try different resolutions
    for tn in [80, 100, 140, 160, 180, 200]:
        if budget_s - (time.time() - t0) < 50:
            break
        if tn % 2 != 0:
            continue
        x_old = np.linspace(0, 1, len(best_h))
        x_new = np.linspace(0, 1, tn)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = fix_sum(h_interp, tn / 2.0)
        for tau in [1000, 3000]:
            if budget_s - (time.time() - t0) < 25:
                break
            h_new, c5_new = _slsqp(h_interp, tau, 500, t0, budget_s)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new

    # Phase 6: Final projected gradient polish
    if budget_s - (time.time() - t0) > 30:
        n = len(best_h)
        target = n / 2.0
        for tau in [300, 1000]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_new, c5_new = _proj_grad(best_h, tau, 0.001, 3000, 0, 0.0, t0, budget_s)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Final SLSQP
    if budget_s - (time.time() - t0) > 15:
        h_new, c5_new = _slsqp(best_h, 20000, 500, t0, budget_s)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
