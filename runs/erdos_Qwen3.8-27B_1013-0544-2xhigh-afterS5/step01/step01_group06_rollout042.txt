    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    w = exp_shifted / total

This is a sequence of linear programs (solved by CVXPY/SCIP) that converges to a local optimum. The advantage is that each subproblem is convex and can be solved to optimality.

The LP at each iteration:
minimize t
subject to: t ≥ c_k(h_old) + grad_k(h_old)^T (h - h_old) for k = -(n-1), ..., n-1
           0 ≤ h[i] ≤ 1 for all i
           sum(h) = n/2

This LP has n+1 variables (h[0],...,h[n-1], t) and approximately 2n + (2n-1) + 1 = 4n constraints. For n=300, that's ~1200 constraints. A modern LP solver can handle this in milliseconds.

The gradient grad_k is:
grad_k = d/dh c_k = vector where component j is (1-2*h_old[k_full-k... ])... 

Actually, from my earlier derivation:
d/dh[j] c_{k_full} = 1-2*h[k_full-j] for k_full ∈ [j, j+n-1], else 0.

So the LP constraint for lag k_full is:
t ≥ c_{k_full}(h_old) + sum_{j: k_full∈[j,j+n-1]} (1-2*h_old[k_full-j]) * (h[j] - h_old[j])

This is linear in h! Great.

import cvxpy as cp

def sca_iteration(h_old, n, target_sum):
    """One SCA iteration: solve the linearized subproblem."""
    h = cp.Variable(n)
    t = cp.Variable()
    
    corr_old = np.correlate(h_old, 1 - h_old, mode='full')
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
    
    two_h_old = 1.0 - 2.0 * h_old
    for k_full in range(2 * n - 1):
        # c_{k_full} ≈ corr_old[k_full] + sum_j (1-2*h_old[k_full-j]) * (h[j] - h_old[j])
        #             = corr_old[k_full] - sum_j (1-2*h_old[k_full-j]) * h_old[j] + sum_j (1-2*h_old[k_full-j]) * h[j]
        # The gradient terms: for j in [max(0, k_full-n+1), min(n-1, k_full)]
        j_lo = max(0, k_full - (n - 1))
        j_hi = min(n - 1, k_full)
        m_vals = k_full - np.arange(j_lo, j_hi + 1)
        grad_vec = np.zeros(n)
        grad_vec[j_lo:j_hi+1] = two_h_old[m_vals]
        
        # Linearization: corr_old[k_full] + grad_vec @ (h - h_old)
        # = corr_old[k_full] - grad_vec @ h_old + grad_vec @ h
        linearized = corr_old[k_full] - np.dot(grad_vec, h_old) + grad_vec @ h
        constraints.append(t >= linearized)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS)  # or another solver
    
    if h.value is not None:
        h_new = h.value
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = project_simplex_box(h_new, target_sum)
        return h_new
    return h_old

import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n))

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    two_h = 1.0 - 2.0 * h
    grad = np.correlate(w, two_h, mode='full')[n-1:2*n-1] * (2.0 / n)
    return grad

def sca_one_step(h_old, n, target_sum):
    corr_old = np.correlate(h_old, 1 - h_old, mode='full')
    two_h = 1.0 - 2.0 * h_old
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for kf in range(2 * n - 1):
        j_lo = max(0, kf - (n - 1))
        j_hi = min(n - 1, kf)
        m_vals = kf - np.arange(j_lo, j_hi + 1)
        grad_vec = np.zeros(n)
        grad_vec[j_lo:j_hi+1] = two_h[m_vals]
        const = corr_old[kf] - np.dot(grad_vec, h_old)
        constraints.append(t_var >= const + grad_vec @ h_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=50000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if h_var.value is not None:
        return np.clip(h_var.value, 0.0, 1.0)
    return None

def sca_optimize(h0, n, target_sum, max_iters=40, tol=1e-7):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        h_new = sca_one_step(h, n, target_sum)
        if h_new is None:
            break
        h_new = project_simplex_box(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h)) < tol:
            h = h_new
            break
        h = h_new
    return best_h, best_c5

def nesterov_optimize(h0, target_sum, deadline, rng, alpha_init=50, alpha_max=600):
    n = len(h0)
    h = project_simplex_box(h0.astype(float).copy(), target_sum)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.003
    beta = 0.9
    alpha = alpha_init
    no_improve = 0
    stall = 0
    while time.time() < deadline:
        h_look = h + beta * v
        h_look = project_simplex_box(h_look, target_sum)
        grad = compute_gradient(h_look, alpha)
        v = beta * v - lr * grad
        h_new = h + v
        h_new = project_simplex_box(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve = 0
            if lr < 0.02:
                lr = min(lr * 1.005, 0.02)
        else:
            no_improve += 1
            lr *= 0.98
            if no_improve > 80:
                no_improve = 0
                bs = max(3, n // 15)
                start = rng.randint(0, n - bs)
                h_flip = best_h.copy()
                h_flip[start:start+bs] = 1.0 - h_flip[start:start+bs]
                h_flip = project_simplex_box(h_flip, target_sum)
                fc = compute_c5(h_flip)
                if fc < best_c5:
                    best_h, best_c5 = h_flip, fc
                h = best_h + rng.randn(n) * 0.015
                h = project_simplex_box(h, target_sum)
                v = np.zeros(n)
                alpha = min(alpha * 1.4, alpha_max)
                lr = 0.003
                stall += 1
                if stall > 20 and alpha >= alpha_max * 0.8:
                    break
        h = h_new
        if no_improve % 100 == 0 and no_improve > 0:
            alpha = min(alpha * 1.08, alpha_max)
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0/n_old)
    return np.interp(x_new, x_ext, h_ext)
