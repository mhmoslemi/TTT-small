```python
def project_box_sum(h, target_sum):
    """Project h onto {x in [0,1]^n : sum(x) = target_sum}"""
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
```

```python
def frank_wolfe(h0, n_iter, top_m=10):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    for i in range(n_iter):
        # Compute subgradient: gradient of the worst correlation
        raw = np.correlate(h, 1.0 - h, mode='full')
        scaled = raw * dx
        # Use top-m for averaging (subgradient of smoothed max)
        top_idx = np.argpartition(scaled, -top_m)[-top_m:]
        # Gradient of C_k for each top shift
        g = np.zeros(n)
        m = len(top_idx)
        for k in top_idx:
            s = k - (n - 1)
            lo1, hi1 = max(0, s), min(n, s + n)
            if lo1 < hi1:
                g[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
            lo2, hi2 = max(0, -s), min(n, n - s)
            if lo2 < hi2:
                g[lo2:hi2] -= h[lo2 + s:hi2 + s]
        g *= dx / m
        
        # Frank-Wolfe step: find s = argmin g^T h' s.t. h' in feasible set
        # Sort g, set h' = 1 where g is smallest (n/2 indices), 0 elsewhere
        order = np.argsort(g)
        s_point = np.zeros(n)
        s_point[order[:n//2]] = 1.0
        
        # Step size: γ = 2/(i+2) for FW, or line search
        gamma = 2.0 / (i + 2)
        h_new = (1 - gamma) * h + gamma * s_point
        
        # Check improvement
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h)
        if c5_new < c5_cur:
            h = h_new
        else:
            # Try smaller step
            gamma = gamma * 0.5
            h_new = (1 - gamma) * h + gamma * s_point
            if compute_c5(h_new) < c5_cur:
                h = h_new
    
    return h, compute_c5(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def grad_of_corr_k(h, k):
    """Gradient of C(k) w.r.t. h"""
    n = len(h)
    s = k - (n - 1)
    g = np.zeros(n)
    lo1, hi1 = max(0, s), min(n, s + n)
    if lo1 < hi1:
        g[lo1:hi1] = 1.0 - h[lo1 - s:hi1 - s]
    lo2, hi2 = max(0, -s), min(n, n - s)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return g * (2.0 / n)

def project_box_sum(h, target):
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fw_optimize(h0, n_iter, top_m=10):
    """Frank-Wolfe with top-m subgradient averaging"""
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for i in range(n_iter):
        scaled = corr_raw(h)
        top_idx = np.argpartition(scaled, -top_m)[-top_m:]
        g = np.zeros(n)
        for k in top_idx:
            g += grad_of_corr_k(h, k)
        g /= top_m
        # FW point
        order = np.argsort(g)
        s_point = np.zeros(n)
        s_point[order[:n // 2]] = 1.0
        # Step size
        gamma = 2.0 / (i + 2)
        h_new = (1.0 - gamma) * h + gamma * s_point
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5

def lse_optimize(h0, tau, n_iter, top_m=50):
    """Nesterov-accelerated gradient on LSE-smoothed objective"""
    n = len(h0)
    dx = 2.0 / n
    xk = h0.copy()
    xk_prev = h0.copy()
    for i in range(n_iter):
        t_rat = (i + 1.0) / (i + 3.0)
        y = xk + t_rat * (xk - xk_prev)
        y = np.clip(y, 0.0, 1.0)
        # Compute LSE gradient
        scaled = corr_raw(y)
        m = np.max(scaled)
        shifted = scaled - m
        exp_v = np.exp(shifted / tau)
        if top_m is not None and top_m < len(scaled):
            idx = np.argpartition(exp_v, -top_m)[-top_m:]
            mask = np.zeros(len(scaled), dtype=bool)
            mask[idx] = True
            exp_v[~mask] = 0.0
        total = np.sum(exp_v)
        p = exp_v / total
        g = np.zeros(n)
        for k in range(len(p)):
            w = p[k]
            if w < 1e-12:
                continue
            s = k - (n - 1)
            lo1, hi1 = max(0, s), min(n, s + n)
            if lo1 < hi1:
                g[lo1:hi1] += w * (1.0 - y[lo1 - s:hi1 - s])
            lo2, hi2 = max(0, -s), min(n, n - s)
            if lo2 < hi2:
                g[lo2:hi2] -= w * y[lo2 + s:hi2 + s]
        g *= dx
        # Step with backtracking
        step = 0.01
        f_curr = m + tau * np.log(total)
        for bt in range(50):
            xn = y - step * g
            xn = project_box_sum(xn, n * 0.5)
            scaled_n = corr_raw(xn)
            m_n = np.max(scaled_n)
            f_n = m_n + tau * np.log(np.sum(np.exp((scaled_n - m_n) / tau)))
            if f_n <= f_curr + 1e-14:
                break
            step *= 0.5
        xk_prev = xk
        xk = xn
    return xk, compute_c5(xk)

def slsqp_tvar(h0, max_iter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.concatenate([h0, [c5_cur + 0.003]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def obj_fn(v):
        return v[-1]
    def ineq_fn(v):
        return v[-1] - np.correlate(v[:n], 1.0 - v[:n], mode='full') * (2.0 / n)
    def eq_fn(v):
        return np.sum(v[:n]) - n * 0.5
    def eq_jac_fn(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    res = minimize(obj_fn, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': ineq_fn},
                                {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}],
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_box_sum(h_new, n * 0.5)
    return h_new, compute_c5(h_new)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_scaled(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_corr_k(h, k):
    n = len(h)
    s = k - (n - 1)
    dx = 2.0 / n
    g = np.zeros(n)
    lo1, hi1 = max(0, s), min(n, s + n)
    if lo1 < hi1:
        g[lo1:hi1] = 1.0 - h[lo1 - s:hi1 - s]
    lo2, hi2 = max(0, -s), min(n, n - s)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return g * dx

def project_box_sum(h, target):
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def fw_optimize(h0, n_iter, top_m=10, use_ls=True):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for i in range(n_iter):
        scaled = corr_scaled(h)
        if top_m < len(scaled):
            top_idx = np.argpartition(scaled, -top_m)[-top_m:]
        else:
            top_idx = np.arange(len(scaled))
        g = np.zeros(n)
        for k in top_idx:
            g += grad_corr_k(h, k)
        g /= len(top_idx)
        order = np.argsort(g)
        s_point = np.zeros(n)
        s_point[order[:n // 2]] = 1.0
        if use_ls:
            best_gamma, best_c5 = 0.0, c5
            for gamma in [1.0, 0.7, 0.4, 0.2, 0.1]:
                h_new = (1.0 - gamma) * h + gamma * s_point
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_gamma = gamma
            if best_gamma > 0:
                h = (1.0 - best_gamma) * h + best_gamma * s_point
                c5 = best_c5
        else:
            gamma = 2.0 / (i + 2)
            h_new = (1.0 - gamma) * h + gamma * s_point
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
    return h, c5

def lse_optimize(h0, tau, n_iter, top_m=50):
    n = len(h0)
    dx = 2.0 / n
    xk = h0.copy()
    xk_prev = h0.copy()
    for i in range(n_iter):
        t_rat = (i + 1.0) / (i + 3.0)
        y = xk + t_rat * (xk - xk_prev)
        y = np.clip(y, 0.0, 1.0)
        scaled = corr_scaled(y)
        m = np.max(scaled)
        shifted = scaled - m
        exp_v = np.exp(shifted / tau)
        if top_m is not None and top_m < len(scaled):
            idx = np.argpartition(exp_v, -top_m)[-top_m:]
            mask = np.zeros(len(scaled), dtype=bool)
            mask[idx] = True
            exp_v[~mask] = 0.0
        total = np.sum(exp_v)
        if total < 1e-300:
            xk_prev = xk
            continue
        p = exp_v / total
        g = np.zeros(n)
        for k in range(len(p)):
            w = p[k]
            if w < 1e-12:
                continue
            s = k - (n - 1)
            lo1, hi1 = max(0, s), min(n, s + n)
            if lo1 < hi1:
                g[lo1:hi1] += w * (1.0 - y[lo1 - s:hi1 - s])
            lo2, hi2 = max(0, -s), min(n, n - s)
            if lo2 < hi2:
                g[lo2:hi2] -= w * y[lo2 + s:hi2 + s]
        g *= dx
        step = 0.005
        f_curr = m + tau * np.log(total)
        for bt in range(50):
            xn = y - step * g
            xn = project_box_sum(xn, n * 0.5)
            scaled_n = corr_scaled(xn)
            m_n = np.max(scaled_n)
            f_n = m_n + tau * np.log(np.sum(np.exp((scaled_n - m_n) / tau)))
            if f_n <= f_curr + 1e-14:
                break
            step *= 0.5
        xk_prev = xk
        xk = xn
    return xk, compute_c5(xk)

def slsqp_tvar(h0, max_iter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.concatenate([h0, [c5_cur + 0.003]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    def obj_fn(v):
        return v[-1]
    def ineq_fn(v):
        return v[-1] - np.correlate(v[:n], 1.0 - v[:n], mode='full') * (2.0 / n)
    def eq_fn(v):
        return np.sum(v[:n]) - n * 0.5
    def eq_jac_fn(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    res = minimize(obj_fn, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': ineq_fn},
                                {'type': 'eq', 'fun': eq_fn, 'jac': eq_jac_fn}],
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_box_sum(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

    return h + amp * np.sin(freq * x + phase)
```
