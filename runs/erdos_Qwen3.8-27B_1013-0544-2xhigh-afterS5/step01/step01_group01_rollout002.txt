    n_candidates = [60, 70, 80]
    time_per_n = (budget_s - 60) / len(n_candidates)

    for n_points in n_candidates:
        t0 = time.time()
        if time.time() - start_time > budget_s - 50:
            break

        target_sum = n_points / 2.0
        dx = 2.0 / n_points
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))

        def con_fun(x):
            return np.sum(x) - target_sum
        def con_jac(x):
            return np.ones(n_points)
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        # Build starting points
        starts = []
        if h_init is not None and len(h_init) == n_points:
            starts.append(h_init.copy())
        if h_init is not None and len(h_init) != n_points:
            xo = np.linspace(0, 2, len(h_init), endpoint=False)
            xn = np.linspace(0, 2, n_points, endpoint=False)
            h_interp = np.interp(xn, xo, h_init, period=2.0)
            starts.append(project_h(h_interp, target_sum))
        starts.append(np.full(n_points, 0.5))
        starts.append(project_h(np.random.random(n_points), target_sum))

        curr_h = None
        curr_c5 = float('inf')

        for h0 in starts:
            h0 = project_h(h0, target_sum)

            # Phase 1: Adaptive smooth_max SLSQP
            for beta in [40, 80, 150, 250]:
                if time.time() - t0 > time_per_n * 0.5:
                    break

                def obj_sm(h):
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    return smooth_max(corr, float(beta)) * dx

                try:
                    res = minimize(obj_sm, h0, method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 300, 'ftol': 1e-12})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5
                        h0 = h_new.copy()
                except Exception:
                    pass

            # Phase 2: Active-lag focused optimization
            if curr_h is not None:
                for top_k in [3, 5, 8]:
                    if time.time() - t0 > time_per_n * 0.7:
                        break
                    for pb in [80, 150]:
                        def obj_al(h):
                            return active_lag_obj(h, top_k, pb, dx)
                        try:
                            res = minimize(obj_al, h0, method='SLSQP', bounds=bounds,
                                           constraints=constraints,
                                           options={'maxiter': 200, 'ftol': 1e-13})
                            h_new = project_h(res.x, target_sum)
                            c5 = compute_c5(h_new)
                            if c5 < curr_c5:
                                curr_h, curr_c5 = h_new.copy(), c5
                                h0 = h_new.copy()
                        except Exception:
                            pass

            if curr_c5 < best_c5:
                best_h, best_c5, best_n = curr_h.copy(), curr_c5, n_points

        # Phase 3: Stochastic local search around best
        if time.time() - t0 < time_per_n * 0.85 and curr_h is not None:
            base = curr_h.copy()
            no_improve = 0
            for iteration in range(30):
                if time.time() - t0 > time_per_n * 0.9 or time.time() - start_time > budget_s - 40:
                    break

                # Diversification: random perturbation with adaptive scale
                scale = 0.03 * (1.0 + 0.5 * np.sin(iteration * 0.3))
                h0 = base + np.random.randn(n_points) * scale
                h0 = project_h(h0, target_sum)

                def obj_search(h):
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    return smooth_max(corr, 120.0) * dx

                try:
                    res = minimize(obj_search, h0, method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 150, 'ftol': 1e-11})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_h, curr_c5 = h_new.copy(), c5
                        base = h_new.copy()
                        no_improve = 0
                        if curr_c5 < best_c5:
                            best_h, best_c5, best_n = h_new.copy(), c5, n_points
                    else:
                        no_improve += 1
                except Exception:
                    no_improve += 1

                if no_improve > 15:
                    break

        # Phase 4: Final high-precision polish
        if time.time() - start_time < budget_s - 15 and best_n == n_points and best_c5 < 0.381:
            for beta in [400, 600]:
                if time.time() - start_time > budget_s - 10:
                    break
                def obj_fin(h):
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    return smooth_max(corr, float(beta)) * dx
                try:
                    res = minimize(obj_fin, best_h, method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 1000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

            # Nelder-Mead on a small subspace of most important coordinates
            if time.time() - start_time < budget_s - 8:
                # Identify coordinates with highest sensitivity
                h_test = best_h + np.random.randn(n_points) * 1e-4
                c5_test = compute_c5(h_test)
                # Just do a small Nelder-Mead polish
                def obj_nm(h):
                    return compute_c5(h)
                try:
                    res = minimize(obj_nm, best_h, method='Nelder-Mead',
                                   options={'maxiter': 500, 'xatol': 1e-12, 'fatol': 1e-14})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    # Ultimate final check
    if best_h is not None:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        def con_f(x):
            return np.sum(x) - ts
        def jac_f(x):
            return np.ones(n_f)
        def obj_ult(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, 500.0) * dx_f
        try:
            res = minimize(obj_ult, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=[{'type': 'eq', 'fun': con_f, 'jac': jac_f}],
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_h(res.x, ts)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    return best_h, best_c5, best_n
