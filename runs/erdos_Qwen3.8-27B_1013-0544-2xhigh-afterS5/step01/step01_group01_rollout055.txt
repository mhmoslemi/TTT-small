import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def coord_descent(h, target_sum, n_sweeps=200, n_grid=101):
    n = len(h)
    h = h.copy()
    h[0] = target_sum - np.sum(h[1:])
    if h[0] < 0 or h[0] > 1:
        h = project_feasible(h, target_sum)
        return h, compute_c5(h)
    
    best_c5 = compute_c5(h)
    t_grid = np.linspace(0.0, 1.0, n_grid)
    
    for sweep in range(n_sweeps):
        improved = False
        for j in range(1, n):
            old_val = h[j]
            h0_old = h[0]
            best_t = old_val
            best_trial = best_c5 + 1.0  # must improve
            
            for t in t_grid:
                h0_new = h0_old + (old_val - t)
                if h0_new < -1e-9 or h0_new > 1.0 + 1e-9:
                    continue
                h[j] = t
                h[0] = h0_new
                c5_t = compute_c5(h)

                    best_trial = c5_t
                    best_t = t

                h[j] = best_t
                h[0] = h0_old + (old_val - best_t)
                best_c5 = best_trial
                improved = True
            else:
                h[j] = old_val
                h[0] = h0_old
        if not improved:
            break
    return h, best_c5

def slsqp_polish(h, n_points, target_sum, max_iter=500):
    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, 200.0) * dx
    
    def con_fun(x):
        return np.sum(x) - target_sum
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n_points)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        if mask.any():
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def coord_descent(h, target_sum, n_sweeps=200, n_grid=151):
    n = len(h)
    h = h.copy()
    h[0] = target_sum - np.sum(h[1:])
    if h[0] < -1e-9 or h[0] > 1.0 + 1e-9:
        h = project_feasible(h, target_sum)
        return h, compute_c5(h)
    best_c5 = compute_c5(h)
    t_grid = np.linspace(0.0, 1.0, n_grid)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(1, n):
            old_val = h[j]
            h0_old = h[0]
            best_t = old_val
            best_trial = best_c5
            for t in t_grid:
                h0_new = h0_old + (old_val - t)
                if h0_new < -1e-9 or h0_new > 1.0 + 1e-9:
                    continue
                h[j] = t
                h[0] = h0_new
                c5_t = compute_c5(h)
                if c5_t < best_trial:
                    best_trial = c5_t
                    best_t = t
            if best_t != old_val and best_trial < best_c5:
                h[j] = best_t
                h[0] = h0_old + (old_val - best_t)
                best_c5 = best_trial
                improved = True
            else:
                h[j] = old_val
                h[0] = h0_old
        if not improved:
            break
    return h, best_c5


def coord_descent_fine(h, target_sum, n_sweeps=100):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(1, n):
            old_val = h[j]
            h0_old = h[0]
            lo = max(0.0, old_val - 0.15)
            hi = min(1.0, old_val + 0.15)

            best_t = old_val
            best_trial = best_c5

                h0_new = h0_old + (old_val - t)
                if h0_new < -1e-9 or h0_new > 1.0 + 1e-9:

                h[j] = t
                h[0] = h0_new
                c5_t = compute_c5(h)
                if c5_t < best_trial:
                    best_trial = c5_t
                    best_t = t

                h[0] = h0_old + (old_val - best_t)

                improved = True
            else:
                h[j] = old_val
                h[0] = h0_old
        if not improved:
            break
    return h, best_c5


def slsqp_polish(h, n_points, target_sum, max_iter=1000):
    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, 300.0) * dx

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n_points)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    parent = None
