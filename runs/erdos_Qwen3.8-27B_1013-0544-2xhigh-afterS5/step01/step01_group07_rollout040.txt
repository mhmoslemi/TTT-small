import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, t, dx):
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return float((max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx)

def _full_obj(x, t, dx):
    h_tmp = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    return smooth_max_val(corr, t, dx)

def _group_obj(x, t, dx, base_h, idx):
    h_tmp = base_h.copy()
    h_tmp[idx] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
    return smooth_max_val(corr, t, dx)

def _sum_con(x, target):
    return float(np.sum(x) - target)

def binary_sa(n, n_iter, T0, T1, seed=0, h_start=None):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    if h_start is not None:
        h = h_start.copy()
    else:
        h = np.zeros(n)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    e_curr = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    best_h = h.copy()
    best_e = e_curr
    no_improve = 0
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = T0 * (T1 / T0) ** frac
        if no_improve > 30000:
            T = max(T, T0 * 0.4)
            no_improve = 0
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(ones_idx[rng.randint(len(ones_idx))])
        j = int(zeros_idx[rng.randint(len(zeros_idx))])
        h[i], h[j] = h[j], h[i]
        e_new = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        if e_new <= e_curr:
            e_curr = e_new
            no_improve = 0
            if e_new < best_e:
                best_e = e_new
                best_h = h.copy()
        else:
            if rng.random() < np.exp(-(e_new - e_curr) / max(T, 1e-10)):
                e_curr = e_new
                no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
                no_improve += 1
    return best_h, best_e * dx

def continuous_sa(h_init, n_iter, T0, T1, dx, target_sum, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h_curr = h_init.copy()
    e_curr = compute_c5(h_curr, dx)
    best_h = h_curr.copy()
    best_e = e_curr
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = T0 * (T1 / T0) ** frac
        step = 0.2 * (1.0 - 0.8 * frac)
        i = int(rng.randint(n))
        delta = rng.uniform(-step, step)
        h_try = h_curr.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
        h_try = fix_sum(h_try, target_sum)
        e_new = compute_c5(h_try, dx)
        if e_new <= e_curr or rng.random() < np.exp(-(e_new - e_curr) / max(T, 1e-10)):
            h_curr = h_try
            e_curr = e_new
            if e_new < best_e:
                best_e = e_new
                best_h = h_curr.copy()
    return best_h, best_e

def group_slsqp(h, n_groups, t, maxiter, dx, target_sum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    gs = max(3, n // n_groups)
    for start in range(0, n, gs):
        idx = np.arange(start, min(start + gs, n))
        g_size = len(idx)
        g_sum = float(np.sum(best[idx]))
        try:
            res = minimize(_group_obj, best[idx].copy(),
                           args=(t, dx, best, idx),
                           method='SLSQP',
                           bounds=[(0.0, 1.0)] * g_size,
                           constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (g_sum,)}],
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = best.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def full_slsqp(h, t, maxiter, dx, target_sum):
    n = len(h)
    try:
        res = minimize(_full_obj, h.copy(), args=(t, dx),
                       method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target_sum,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = fix_sum(res.x, target_sum)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h0)
    dx_init = 2.0 / n_init
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx_init)
    best_n = n_init

    # Phase 1: Binary SA at n=160
    n160 = 160
    dx160 = 2.0 / n160
    for trial in range(10):
        if time.time() - t_start > budget_s - 300:
            break
        h_b, c5_b = binary_sa(n160, 300000, 5.0, 0.03, seed=seed + trial * 7919)
        if c5_b < best_c5:
            best_h = h_b
            best_c5 = c5_b
            best_n = n160

    # Phase 2: SA refinement from best solution
    if time.time() - t_start < budget_s - 200:
        for trial in range(6):
            if time.time() - t_start > budget_s - 190:
                break
            h_b, c5_b = binary_sa(n160, 350000, 2.5, 0.005, seed=seed + 5000 + trial, h_start=best_h)
            if c5_b < best_c5:
                best_h = h_b
                best_c5 = c5_b
                best_n = n160

    # Phase 3: Continuous SA from best
    if time.time() - t_start < budget_s - 120:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        h_c, c5_c = continuous_sa(best_h, 120000, 0.06, 0.001, dd, ts, seed=seed + 9000)
        if c5_c < best_c5:
            best_h = h_c
            best_c5 = c5_c

    # Phase 4: SLSQP group optimization
    if time.time() - t_start < budget_s - 80:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        for t_val in [40, 100, 300]:
            if time.time() - t_start > budget_s - 70:
                break
            h_g, c5_g = group_slsqp(best_h, 16, t_val, 250, dd, ts)
            if c5_g < best_c5:
                best_h = h_g
                best_c5 = c5_g

    # Phase 5: Multi-start from random binaries
    if time.time() - t_start < budget_s - 40:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        for trial in range(8):
            if time.time() - t_start > budget_s - 35:
                break
            rng_ts = np.random.RandomState(seed + 20000 + trial * 331)
            h_rand = np.zeros(best_n)
            ones = rng_ts.choice(best_n, best_n // 2, replace=False)
            h_rand[ones] = 1.0
            h_rand = continuous_sa(h_rand, 30000, 0.1, 0.01, dd, ts, seed=seed + 20000 + trial)
            h_opt, c5_opt = full_slsqp(h_rand, 80, 300, dd, ts)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 6: Try n=80
    if time.time() - t_start < budget_s - 25:
        n80 = 80
        dx80 = 2.0 / n80
        for trial in range(4):
            if time.time() - t_start > budget_s - 20:
                break
            h_b, c5_b = binary_sa(n80, 200000, 5.0, 0.03, seed=seed + 50000 + trial)
            if c5_b < best_c5:
                best_h = h_b
                best_c5 = c5_b
                best_n = n80
                dd80 = 2.0 / n80
                h_c, c5_c = continuous_sa(best_h, 50000, 0.06, 0.001, dd80, n80 // 2, seed=seed + 60000)
                if c5_c < best_c5:
                    best_h = h_c
                    best_c5 = c5_c

    # Final polish
    if time.time() - t_start < budget_s - 5:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        h_f, c5_f = full_slsqp(best_h, 400, 1000, dd, ts)
        if c5_f < best_c5:
            best_h = h_f
            best_c5 = c5_f

    best_h = np.clip(best_h, 0.0, 1.0)
    dd = 2.0 / best_n
    best_c5 = compute_c5(best_h, dd)
    return best_h, best_c5, best_n
