    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def gradient_of_smooth_max(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_overlap(h)  # shape (2n-1,)
    x = corr / tau
    x_shift = x - x.max()
    w = np.exp(x_shift)
    w /= w.sum()

    # d(corr[m])/dh[r] = dx * [(1-h[r-k]) if 0<=r-k<n else 0 - h[r+k] if 0<=r+k<n else 0]
    
    grad = np.zeros(n)

        g = 0.0
        for m in range(2*n-1):

            wk = w[m]
            if wk < 1e-15:

            term = 0.0

            if 0 <= r+k < n:

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    x = corr / tau
    x -= x.max()
    w = np.exp(x)
    w /= w.sum()
    
    u = 1.0 - h
    conv1 = np.convolve(u, w, mode="full")
    T1 = conv1[n-1:n-1+n]
    
    corr1 = np.correlate(w, h, mode="full")
    T2 = corr1[:n][::-1]
    
    return dx * (T1 - T2)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:

        else:

        if mask.sum() == 0:

        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.zeros(n)

    return h

def mmw_solve(h0, t_start, budget, n_inner=10, lr=0.02):
    """Min-max with multiplicative weights."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    nlags = 2 * n - 1
    lam = np.ones(nlags) / nlags
    eta = 2.0
    
    while time.time() - t_start < budget:
        # Inner optimization
        for _ in range(n_inner):
            if time.time() - t_start > budget:
                break
            grad = get_grad(h, lam)
            h = h - lr * grad
            h = project_feasible(h)
        
        # Update weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        c = corr * (2.0 / n)
        eta = max(eta * 0.995, 0.5)
        lam = np.exp(eta * c / max(np.max(c), 1e-10))
        lam /= lam.sum()
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def lion_solve(h0, tau, lr, t_start, budget):
    """Lion optimizer with smooth max."""
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    while time.time() - t_start < budget:
        # Smooth max weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        c = corr * (2.0 / n)
        x = c / tau
        x -= x.max()
        w = np.exp(x)
        w /= w.sum()
        
        grad = get_grad(h, w)
        m = beta * m + (1 - beta) * grad
        h = h - lr * np.sign(m)
        h = project_feasible(h)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

        if c5_new < best_c5:

    # Phase 3: Multi-resolution
    phase3_budget = t0 + budget_s * 0.80

        if time.time() > phase3_budget:

        h_t = np.interp(np.linspace(0, 2, tn, endpoint=False),

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    conv1 = np.convolve(u, weights, mode="full")
    T1 = conv1[n-1:n-1+n]
    corr1 = np.correlate(weights, h, mode="full")
    T2 = corr1[:n][::-1]
    return dx * (T1 - T2)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def mmw_solve(h0, t_start, t_end, n_inner=10, lr=0.02, eta0=3.0):

    while time.time() < t_end:
        for _ in range(n_inner):
            if time.time() > t_end:
                break
            grad = get_grad(h, lam)
            h = h - lr * grad
            h = project_feasible(h)
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        eta = max(eta * 0.997, 0.5)
        lam = np.exp(eta * corr / max(np.max(corr), 1e-10))

        c5 = evaluate_c5(h)
        if c5 < best_c5:

def adam_solve(h0, tau, lr, t_start, t_end, b1=0.9, b2=0.999, eps=1e-8):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    while time.time() < t_end:
        corr = np.correlate(h, 1.0 - h, mode="full")
        c = corr * (2.0 / n)
        x = c / tau
        x -= x.max()
        w = np.exp(x)
        w /= w.sum()
        grad = get_grad(h, w)
        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        h = h - lr * mhat / (np.sqrt(vhat) + eps)
        h = project_feasible(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
