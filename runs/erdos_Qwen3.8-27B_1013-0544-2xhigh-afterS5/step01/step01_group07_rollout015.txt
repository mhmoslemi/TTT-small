    corr = np.correlate(h, 1 - h, mode="full")

def smooth_max(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 50)
    val = max_c + (1.0/t) * np.log(1e-300 + np.sum(np.exp(shifted)))
    return val * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 50)

    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)

        w = weights[j]
        if w < 1e-12:
            continue
        k = j - (n - 1)

        lo1 = max(0, k)

            grad[lo2:hi2] += w * (-h[lo2+k:hi2+k])

def make_periodic_binary(n, n_periods):
    """Binary periodic pattern: n_periods groups of 1s and 0s alternating"""
    h = np.zeros(n)
    period = n // n_periods

        add = zero_idx[:n//2 - n_ones]

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_val(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 50)
    val = max_c + (1.0 / t) * np.log(1e-300 + np.sum(np.exp(shifted)))
    return val * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -50, 50)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-13:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] += w * (-h[lo2 + k:hi2 + k])
    return grad * dx

def fix_sum(h, target_sum):
    h = h.copy()
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-13:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def penalized_obj(h, t, dx, target_sum, lam):
    return smooth_max_val(h, t, dx) + lam * (np.sum(h) - target_sum) ** 2

def penalized_grad(h, t, dx, target_sum, lam):
    g = smooth_max_grad(h, t, dx)
    g += 2.0 * lam * (np.sum(h) - target_sum)
    return g

def lbfgsb_run(h, dx, target_sum, t, maxiter, lam=5000.0):
    n = len(h)
    try:
        res = minimize(
            lambda x: penalized_obj(x, t, dx, target_sum, lam),
            h.copy(),
            jac=lambda x: penalized_grad(x, t, dx, target_sum, lam),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9,
                     'iprint': -1, 'maxls': 40, 'maxcor': 25}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def binarize(h, n):
    order = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[order[:n // 2]] = 1.0
    return h_b

def binary_local_search(h, dx, target_sum, n_passes, rng, n_tries_per_pass=300):
    n = len(h)
    h_b = binarize(h, n)
    best_c5 = compute_c5(h_b, dx)
    h_best = h_b.copy()
    ones = np.where(h_b == 1.0)[0]
    zeros = np.where(h_b == 0.0)[0]
    for p in range(n_passes):
        improved = False
        for _ in range(n_tries_per_pass):
            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            h_b[a] = 1.0
            h_b[b] = 0.0
            c5_new = compute_c5(h_b, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_best = h_b.copy()
                improved = True
                ones = np.where(h_b == 1.0)[0]
                zeros = np.where(h_b == 0.0)[0]
            else:
                h_b[a] = 0.0
                h_b[b] = 1.0
        if not improved:
            break
    return h_best, best_c5

def make_periodic_binary(n, n_periods):
    h = np.zeros(n)
    period = n // n_periods
    for p in range(n_periods):
        start = p * period
        end = min(start + period // 2, n)
        h[start:end] = 1.0
    n_ones = int(np.sum(h == 1))
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        h[idx[-(n_ones - n // 2):]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        h[idx[:n // 2 - n_ones]] = 1
    return h
