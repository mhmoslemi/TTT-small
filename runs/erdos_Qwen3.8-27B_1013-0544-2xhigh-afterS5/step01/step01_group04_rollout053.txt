def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)

    best_c = obj[best_idx]
    
    gen = 0
    while gen < max_gens:
        if time_limit_func():
            break

                break

            mask = np.ones(pop_size, dtype=bool)

                obj[i] = u_obj

                    best_h = u.copy()
        
        gen += 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h, n, dx, target_sum, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(
            lambda x: compute_c5(x, dx),
            h, method='SLSQP', bounds=bounds, constraints=constraint,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgs_polish(h, n, dx, target_sum, max_iter=500):
    """L-BFGS-B with penalty for constraint."""
    def objective(x):
        x_clip = np.clip(x, 0, 1)
        penalty = 1000.0 * (np.sum(x_clip) - target_sum) ** 2
        return compute_c5(x_clip, dx) + penalty
    try:
        res = minimize(objective, h, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def de_run(h_init, n, dx, target_sum, pop_size, rng, time_check):
    """Custom DE with best/2/bin, adaptive params, periodic polish."""
    pop = np.zeros((pop_size, n))
    pop[0] = project_feasible(h_init.copy(), n)
    for i in range(1, pop_size):
        r = rng.rand()
        if r < 0.3:
            pop[i] = project_feasible(rng.rand(n), n)
        elif r < 0.6:
            pop[i] = project_feasible(h_init + rng.randn(n) * 0.15, n)
        else:
            pop[i] = project_feasible(1.0 - h_init + rng.randn(n) * 0.1, n)
    
    obj = np.array([compute_c5(pop[i], dx) for i in range(pop_size)])
    best_idx = int(np.argmin(obj))
    best_h = pop[best_idx].copy()
    best_c = obj[best_idx]
    gen = 0
    stagnant = 0
    threshold = best_c
    
    while not time_check():
        F = 0.8 * max(0.3, 1.0 - gen / 300.0)
        CR = 0.8 * max(0.3, 1.0 - gen / 400.0)
        
        for i in range(pop_size):
            if time_check():
                break
            # best/2/bin
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            mask[best_idx] = False
            idxs = np.where(mask)[0]
            p1, p2, p3 = rng.choice(idxs, 3, replace=False)
            p_best = best_idx
            
            if rng.rand() < 0.5:
                v = pop[p_best] + F * (pop[p1] - pop[p2])
            else:
                v = pop[p1] + F * (pop[p_best] - pop[p3])
            
            r = rng.randint(n)
            cross = rng.rand(n) <= CR
            cross[r] = True
            u = np.where(cross, v, pop[i])
            u = project_feasible(u, n)
            u_obj = compute_c5(u, dx)
            
            if u_obj <= obj[i]:
                pop[i] = u
                obj[i] = u_obj
                if u_obj < best_c:
                    best_c = u_obj
                    best_h = u.copy()
                    best_idx = i
                    stagnant = 0
                    threshold = best_c - 0.0001
        
        gen += 1
        
        # Periodic polish
        if gen % 50 == 0 and not time_check():
            polished = slsqp_polish(best_h, n, dx, target_sum, max_iter=200)
            pc = compute_c5(polished, dx)
            if pc < best_c:
                best_c = pc
                best_h = polished
                pop[0] = polished.copy()
                obj[0] = pc
                best_idx = 0
                threshold = best_c - 0.0001
            else:
                stagnant += 1
                if stagnant > 3:
                    # Immigrate: replace worst with random
                    worst = int(np.argmax(obj))
                    pop[worst] = project_feasible(rng.rand(n), n)
                    obj[worst] = compute_c5(pop[worst], dx)
                    stagnant = 0
    
    return best_h, best_c
