    h = h.copy()

    h = np.clip(h, 0.0, 1.0)
    # Then adjust sum
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h

        order = np.argsort(h)
        for i in order:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:

        order = np.argsort(-h)
        for i in order:

        # Now smooth the binary solution

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, tau):
    x_max = np.max(x)
    return float(x_max + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_box_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
        for i in order:
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(-h)
        for i in order:
            rem = min(h[i], -diff)
            h[i] -= rem
            diff += rem
            if diff >= -1e-14:
                break
    return h

def grad_smooth_corr(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    x_max = np.max(corr)
    e = np.exp(corr - x_max)
    w = e / np.sum(e)
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = shifts[k]
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] += wk * (-h[lo2 + s:hi2 + s])
    return grad

def pgd_loop(h0, n, tau_start, tau_end, lr, n_iter, target):
    h = h0.copy()
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        g = grad_smooth_corr(h, tau)
        h = project_box_sum(h - lr * g, target)
    return h

def binary_anneal(h0, n, n_iter, temp_start, temp_end, target):
    h = (h0 > 0.5).astype(float)
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    ts = temp_start
    te = max(temp_end, 1e-6)
    for it in range(n_iter):
        if it % 1000 == 0:
            frac = it / max(1, n_iter - 1)
            temp = ts * (te / ts) ** frac
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                delta = c5 - best_c5
                if np.random.random() > np.exp(-delta / temp):
                    h[i], h[j] = h[j], h[i]
    return best_h
