def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def grad_corr_single(h, k_offset):
    """Gradient of corr[k_offset] with respect to h.
    k_offset is the index into the full correlation array.
    k = k_offset - (n-1) is the actual lag."""
    n = len(h)
    k = k_offset - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[j] for j in [0, n-1-k]: (1 - h[j+k])
        # d/dh[j] for j in [k, n-1]: -h[j-k]
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        # corr = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
        # d/dh[j] for j in [p, n-1]: (1 - h[j-p])
        # d/dh[j] for j in [0, n-1-p]: -h[j+p]
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

def grad_corr_single(h, k_offset):

The idea:

- At each step, compute subgradient g of the max function
- Step size: γ = (f(x) - z_lb) / ||g||² where z_lb is a lower bound on the optimum
- Update: x_new = P_LP(x - γ*g)

The Polyak step size is designed to converge to the optimum without knowing the exact optimal value. We use z_lb as an estimate (e.g., 0.38).

The key advantage is that the step size automatically adapts: when we're far from the optimum, large steps; when close, small steps.

import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def grad_corr_single(h, k_offset):
    n = len(h)
    k = k_offset - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def weighted_grad(h, alpha, top_k):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_c = np.exp(alpha * (corr - m))
    w = exp_c / np.sum(exp_c)
    if top_k < len(w):
        idx = np.argsort(w)[-top_k:]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        w = w2 / np.sum(w2)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for i in active:
        grad += w[i] * grad_corr_single(h, i)
    return grad * (2.0 / n)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
