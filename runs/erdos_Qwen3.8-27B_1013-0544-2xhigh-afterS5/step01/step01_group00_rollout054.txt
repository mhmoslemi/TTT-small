def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Second pass
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        cnt2 = np.sum(mask2)
        if cnt2 > 0:
            h[mask2] += diff2 / cnt2
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(h, temp, pw):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    val = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    val += pw * (np.sum(h) - 0.5 * n) ** 2
    return val


def lse_grad(h, temp, pw):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp((corr - m) / temp)
    probs = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = n - 1 + k
        p = probs[idx]
        if p < 1e-14:
            continue
        g = np.zeros(n)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2-k:hi2-k]
        grad += p * dx * g
    
    grad += 2.0 * pw * (np.sum(h) - 0.5 * n)
    return grad

    return v[-1]

    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

4. Repeat

This is very simple! And Frank-Wolfe has a known convergence rate of O(1/t).

The step size γ can be:
- Constant: γ = 2/(t+2) (standard FW)
- Or: γ = argmin_{γ in [0,1]} f((1-γ)h + γv) (line search FW, better in practice)

Let me add Frank-Wolfe as an additional optimization method! It's simple, fast, and handles the constraints exactly (no penalty needed).

The objective for FW: I'll use the LSE approximation (smooth) to get a proper gradient.

Actually, even better: I can use the **exact max objective** with FW. The subgradient of max_k corr_k at h is the gradient of corr_{k*} where k* = argmax. Frank-Wolfe works with subgradients.

But for better convergence, the smooth LSE approximation is preferred.

Let me include FW as an additional phase:

```python
def fw_step(h, grad, n):
    """Frank-Wolfe step: find vertex with smallest g^T v"""
    n_half = n // 2  # number of ones in the vertex
    # Vertex: set the n_half smallest gradient components to 1
    idx = np.argpartition(grad, n_half)[:n_half]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def fw_line_search(h, v, f, n, max_iter=50):
    """Find best gamma in [0,1] for h_new = (1-gamma)*h + gamma*v"""
    # Golden section or ternary search
    a, b = 0.0, 1.0
    gr = (np.sqrt(5) - 1) / 2
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f((1-c)*h + c*v, n)
    fd = f((1-d)*h + d*v, n)
    for _ in range(max_iter):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr*(b-a)
            fc = f((1-c)*h + c*v, n)
        else:
            a = c; c = d; fc = fd
            d = a + gr*(b-a)
            fd = f((1-d)*h + d*v, n)
    gamma = (a + b) / 2
    return (1-gamma)*h + gamma*v
```

Wait, but the line search requires evaluating f multiple times, each of which is O(n²) due to the correlate call. That's fine for n up to 360.

Actually, for the line search, I can use a simpler approach: since f((1-γ)h + γv) is the LSE of a quadratic function of γ (each corr_k is quadratic in γ), I can find the minimum analytically or with a few evaluations.

But let's keep it simple: use a fixed step size γ = 2/(t+2) where t is the iteration number. This is the standard FW step size and avoids the line search cost.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = h < 1.0
        else:
            mask2 = h > 0.0
        cnt2 = np.sum(mask2)
        if cnt2 > 0:
            h[mask2] += diff2 / cnt2
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(h, temp):
    corr = compute_corr(h)
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    exps = np.exp((corr - m) / temp)
    probs = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = n - 1 + k
        p = probs[idx]
        if p < 1e-15:
            continue
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += p * (1.0 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, k + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= p * h[lo2-k:hi2-k]
    grad *= dx
    return grad


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    return t - compute_corr(h)


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - 0.5 * n


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n)
    return h_new, compute_c5(h_new)


def interp_to_res(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new)


def lse_lbfgs(h0, temp, n, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_obj, h0, args=(temp,), method='L-BFGS-B',
                   jac=lse_grad, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_h(res.x, n)


def fw_optimize(h0, temp, n, max_iter=500):
    """Frank-Wolfe with LSE smooth approximation"""
    h = h0.copy()
    n_half = n // 2
    best_h = h.copy()
    best_val = lse_obj(h, temp)
    for t in range(max_iter):
        grad = lse_grad(h, temp)
        # Find vertex: n_half smallest gradient components get value 1
        idx = np.argpartition(grad, n_half)[:n_half]
        v = np.zeros(n)
        v[idx] = 1.0
        gamma = 2.0 / (t + 2)
        h_new = (1 - gamma) * h + gamma * v
        h_new = np.clip(h_new, 0.0, 1.0)
        # Project to sum constraint
        h_new = project_h(h_new, n)
        val = lse_obj(h_new, temp)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    return best_h

        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin.copy(), c5_fin
    
    best_h = project_h(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
