import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smoothed_grad(h, alpha):
    """Return (smoothed_obj, gradient, corr_values)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    
    x = alpha * F
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    w = exp_x / s
    obj = (x_max + np.log(s)) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        tau = k - (n - 1)
        c = wk * dx
        if tau >= 0:
            # (1-h[j-tau]) for j in [tau, n)
            grad[tau:n] += c * (1.0 - h[0:n-tau])
            # -h[j+tau] for j in [0, n-tau)
            grad[0:n-tau] += -c * h[tau:n]
        else:
            t = -tau
            # (1-h[j+t]) for j in [0, n-t)
            grad[0:n-t] += c * (1.0 - h[t:n])
            # -h[j-t] for j in [t, n)
            grad[t:n] += -c * h[0:n-t]
    
    return obj, grad, F

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smoothed_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    F = corr * dx
    x = alpha * F
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    w = exp_x / s
    obj = (x_max + np.log(s)) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        tau = k - (n - 1)
        c = wk * dx
        if tau >= 0:
            grad[tau:n] += c * (1.0 - h[:n-tau])
            grad[:n-tau] += -c * h[tau:]
        else:
            t = -tau
            grad[:n-t] += c * (1.0 - h[t:])
            grad[t:] += -c * h[:n-t]
    return obj, grad
