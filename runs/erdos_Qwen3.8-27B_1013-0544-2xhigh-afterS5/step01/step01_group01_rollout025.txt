import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)


def obj_smooth(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def obj_de(h, target_sum, dx):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    pen = (hc.sum() - target_sum) ** 2 * 50.0
    return smooth_max(corr, 200.0) * dx + pen


def obj_nm(h, target_sum, dx):
    hc = np.clip(h, 0, 1)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    return smooth_max(corr, 200.0) * dx + 10.0 * (hc.sum() - target_sum) ** 2


def con_fun(x, target_sum):
    return x.sum() - target_sum


def con_jac(x, target_sum):
    return np.ones(len(x))


def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    bounds = list(zip(np.zeros(n_points), np.ones(n_points)))
    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}

        res = minimize(obj_smooth, best_h, args=(200.0, dx), method='SLSQP',
                       bounds=bounds, constraints=[con],
                       options={'maxiter': 3000, 'ftol': 1e-14})
        h_new = project_h(res.x, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
    except Exception:
        pass

    # Phase 2: Differential Evolution for global exploration
    elapsed = time.time() - start_time
    if budget_s - elapsed > 150:
        de_bounds = [(0.0, 1.0)] * n_points
        try:
            de_res = differential_evolution(
                obj_de, de_bounds, args=(target_sum, dx),
                seed=seed, maxiter=25, popsize=5, tol=1e-4,
                mut=(0.5, 1.0), reccombin=0.7, polish=False,
                updating='deferred', init='latinhypercube')
            h_de = project_h(de_res.x, target_sum)
            c5_de = compute_c5(h_de)

        except Exception:
            pass

            c5 = compute_c5(h_new)

        except Exception:

    return best_h, best_c5, n_points
