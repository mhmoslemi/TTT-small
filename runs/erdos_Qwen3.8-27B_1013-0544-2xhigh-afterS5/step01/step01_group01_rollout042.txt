```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_corr(h):
    """Compute gradient of smooth_max(corr) with respect to h."""
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr

def obj_and_grad_sym(u, n, beta):
    """Objective and gradient for symmetric parameterization."""
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    h[m:] = u[::-1]
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    exp_terms = np.exp(beta * (corr - cmax))
    obj = (cmax + np.log(exp_terms.sum()) / beta) * dx
    
    # Gradient of corr[l] w.r.t. h[k]
    weights = exp_terms / exp_terms.sum()
    grad_h = np.zeros(n)
    for l in range(len(corr)):
        if weights[l] < 1e-12:
            continue
        shift = l - (n - 1)
        lo = max(0, -shift)
        hi = min(n - 1, n - 1 - shift)
        w = weights[l]
        for k in range(lo, hi + 1):
            val = w
            kp = k + shift
            if 0 <= kp < n:
                val -= w * h[kp]
            km = k - shift
            if lo <= km <= hi:
                val -= w * h[km]
            grad_h[k] += val
    
    grad_h *= dx
    
    # Chain rule: grad_u[k] = grad_h[k] + grad_h[n-1-k]
    grad_u = np.empty(m)
    for k in range(m):
        grad_u[k] = grad_h[k] + grad_h[n - 1 - k]
    
    return obj, grad_u
```

```python
def obj_and_grad_sym_fast(u, n, beta):
    m = n // 2
    h = np.empty(n)
    h[:m] = u
    h[m:] = u[::-1]
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    exp_terms = np.exp(beta * (corr - cmax))
    obj = (cmax + np.log(exp_terms.sum()) / beta) * dx
    weights = exp_terms / exp_terms.sum()
    
    grad_h = np.zeros(n)
    for l in range(len(corr)):
        w = weights[l]
        if w < 1e-12:
            continue
        shift = l - (n - 1)
        lo = max(0, -shift)
        hi = min(n - 1, n - 1 - shift)
        # Main term
        grad_h[lo:hi+1] += w
        # -w*h[k+shift] for k in [lo, hi]
        kp_lo = max(0, lo)
        kp_hi = min(n-1, hi)  # k is already in [0, n-1]
        # We need k+shift to be in [0, n-1], and k in [lo, hi]
        # k+shift in [0, n-1] means k in [-shift, n-1-shift]
        kp_valid_lo = max(lo, -shift)
        kp_valid_hi = min(hi, n - 1 - shift)
        if kp_valid_lo <= kp_valid_hi:
            grad_h[kp_valid_lo:kp_valid_hi+1] -= w * h[kp_valid_lo+shift:kp_valid_hi+1+shift]
        # -w*h[k-shift] for k in [lo, hi] where k-shift in [lo, hi]
        # k-shift in [lo, hi] means k in [lo+shift, hi+shift]
        km_valid_lo = max(lo, lo + shift)
        km_valid_hi = min(hi, hi + shift)
        if km_valid_lo <= km_valid_hi:
            grad_h[km_valid_lo:km_valid_hi+1] -= w * h[km_valid_lo-shift:km_valid_hi+1-shift]
    
    grad_h *= dx
    grad_u = np.empty(m)
    for k in range(m):
        grad_u[k] = grad_h[k] + grad_h[n - 1 - k]
    return obj, grad_u
```

```python
    return gh[:m] + gh[m:][::-1]

def penalized_obj(u, n, beta, tgt, lam):
    return sym_obj(u, n, beta) + lam * (u.sum() - tgt) ** 2

def penalized_grad(u, n, beta, tgt, lam):
    g = sym_grad(u, n, beta)
    g = g + 2.0 * lam * (u.sum() - tgt)
    return g

def project_u(u, tgt):
    for _ in range(200):
        u = np.clip(u, 0.0, 1.0)
        d = tgt - u.sum()
        if abs(d) < 1e-13:
            break
        mask = (u > 1e-10) & (u < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        u[mask] += d / mask.sum()
    return np.clip(u, 0.0, 1.0)

def make_guesses(m, tgt, parent, n, rng):
    guesses = []
    guesses.append(np.full(m, 0.5))
    
    if parent is not None:
        if len(parent) == n:
            guesses.append(parent[:m].copy())
        else:
            xo = np.linspace(0, 2, len(parent))
            xn = np.linspace(0, 2, n)
            hi = np.interp(xn, xo, parent)
            guesses.append(project_u(hi[:m].copy(), tgt))
    
    for freq in [2, 3, 4, 5, 7]:
        for amp in [0.15, 0.25, 0.35]:
            x = np.linspace(0, 1, m)
            u = 0.5 + amp * np.sin(2 * np.pi * freq * x)
            u = project_u(u, tgt)
            guesses.append(u.copy())
    
    for _ in range(3):
        u = rng.random(m)
        u = project_u(u, tgt)
        guesses.append(u.copy())
    
    # Block structure
    for block_frac in [0.25, 0.3, 0.4]:
        u = np.zeros(m)
        blk = max(2, int(m * block_frac))
        for i in range(m):
            u[i] = 0.8 if (i // blk) % 2 == 0 else 0.2
        u = project_u(u, tgt)
        guesses.append(u.copy())
    
    return guesses
```
