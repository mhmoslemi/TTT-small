def compute_sensitivity(h, m, n):
    """Compute s[k] = d(corr[k])/dh[m] for full correlation vector."""
    # corr[j] = sum_i h[i] * (1-h[i+j-(n-1)]) for valid i, i+j-(n-1)
    s = np.zeros(2*n-1)
    k_offset_per_j = np.arange(2*n-1) - (n-1)
    # h[m] as first factor (h[i] where i=m): contributes (1-h[m+k]) where k = j-(n-1)
    #   valid when 0 <= m+k_offset_per_j < n
    mask1 = (m + k_offset_per_j >= 0) & (m + k_offset_per_j < n)
    s[mask1] += (1 - h[m + k_offset_per_j[mask1]])
    # h[m] as second factor (1-h[j2] where j2=m, i.e., i+k_offset=m, i=m-k_offset):
    #   contributes -h[i] where i = m - k_offset
    #   valid when 0 <= m-k_offset_per_j < n
    mask2 = (m - k_offset_per_j >= 0) & (m - k_offset_per_j < n)
    s[mask2] -= h[m - k_offset_per_j[mask2]]
    return s

def pair_coordinate_minimize(h, m, l, corr):
    """Find optimal delta for h[m] += delta, h[l] -= delta to minimize max corr."""
    n = len(h)
    s_m = compute_sensitivity(h, m, n)
    s_l = compute_sensitivity(h, l, n)
    d = s_m - s_l
    
    # Feasible range for delta
    lo = max(-h[m], h[l] - 1.0)
    hi = min(1.0 - h[m], h[l])
    if lo >= hi:
        return 0.0
    
    # f(delta) = max_k (corr[k] + d[k] * delta)
    # Find minimum using ternary search
    a, b = lo, hi
    for _ in range(60):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        f1 = np.max(corr + d * m1)
        f2 = np.max(corr + d * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    delta = (a + b) / 2
    return delta

def fast_coordinate_descent(h, dx, target_sum, n_passes=2000, seed=42):

        # Random pair
        m = rng.randint(n)
        l = rng.randint(n - 1)
        if l >= m:
            l += 1

        delta = pair_coordinate_minimize(h, m, l, corr)
        if abs(delta) > 1e-12:
            h[m] += delta
            h[l] -= delta
            # Update correlations
            corr = np.correlate(h, 1-h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
    
    return h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def get_sensitivity(h, m, n, k_off, corr_len):
    s = np.zeros(corr_len)
    idx1 = np.where((m + k_off >= 0) & (m + k_off < n))[0]
    if len(idx1) > 0:
        s[idx1] += (1.0 - h[m + k_off[idx1]])
    idx2 = np.where((m - k_off >= 0) & (m - k_off < n))[0]
    if len(idx2) > 0:
        s[idx2] -= h[m - k_off[idx2]]
    return s

def ternary_min_1d(corr, d, lo, hi, n_iter=45):
    a, b = lo, hi
    for _ in range(n_iter):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(corr + d * m1) < np.max(corr + d * m2):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def fast_cd(h, dx, target_sum, n_iters, seed=42, full_sweep_every=500):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    corr_len = 2 * n - 1
    k_off = np.arange(corr_len) - (n - 1)
    center = n - 1

    for it in range(n_iters):
        if it % full_sweep_every == 0 and it > 0:
            order = rng.permutation(n)
            for m in order:
                l = rng.randint(n - 1)
                if l >= m:
                    l += 1
                s_m = get_sensitivity(h, m, n, k_off, corr_len)
                s_l = get_sensitivity(h, l, n, k_off, corr_len)
                d = s_m - s_l
                lo = max(-h[m], h[l] - 1.0)
                hi = min(1.0 - h[m], h[l])
                if lo >= hi:
                    continue
                delta = ternary_min_1d(corr, d, lo, hi, 30)
                if abs(delta) > 1e-14:
                    h[m] += delta
                    h[l] -= delta
                    corr += d * delta
                    corr[center] -= 2.0 * delta * delta
        else:
            m = rng.randint(n)
            l = rng.randint(n - 1)
            if l >= m:
                l += 1
            s_m = get_sensitivity(h, m, n, k_off, corr_len)
            s_l = get_sensitivity(h, l, n, k_off, corr_len)
            d = s_m - s_l
            lo = max(-h[m], h[l] - 1.0)
            hi = min(1.0 - h[m], h[l])
            if lo >= hi:
                continue
            delta = ternary_min_1d(corr, d, lo, hi, 40)
            if abs(delta) > 1e-14:
                h[m] += delta
                h[l] -= delta
                corr += d * delta
                corr[center] -= 2.0 * delta * delta

        if it % 200 == 0:
            c5_val = float(np.max(corr) * dx)
            if c5_val < best_c5:
                best_c5 = c5_val

    c5_val = float(np.max(corr) * dx)
    if c5_val < best_c5:
        best_c5 = c5_val
    return h, best_c5

def slsqp_refine(h, dx, target_sum, t=100, maxiter=500):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1 - x, mode="full")
        mc = np.max(corr)
        shifted = t * (corr - mc)
        return (mc + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx
    def grad(x):
        corr = np.correlate(x, 1 - x, mode="full")
        mc = np.max(corr)
        shifted = t * (corr - mc)
        weights = np.exp(shifted)
        wsum = np.sum(weights)
        corr_len = 2 * n - 1
        k_off = np.arange(corr_len) - (n - 1)
        g = np.zeros(n)
        for j in range(corr_len):
            wj = weights[j] / wsum
            idx1 = np.where((np.arange(n) + k_off[j] >= 0) & (np.arange(n) + k_off[j] < n))[0]
            if len(idx1) > 0:
                g[idx1] += wj * (1.0 - x[idx1 + k_off[j]])
            idx2 = np.where((np.arange(n) - k_off[j] >= 0) & (np.arange(n) - k_off[j] < n))[0]
            if len(idx2) > 0:
                g[idx2] -= wj * x[idx2 - k_off[j]]
        return g * dx
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = fix_sum(res.x, target_sum)
        return h_new
    except Exception:
        return h

def make_binary(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

        h_new, c5_new = fast_cd(best_h, dx, target_sum, 15000, seed=seed + 2000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if elapsed() < budget_s - 80:
        for bs in range(5):
            if elapsed() > budget_s - 60:

            h_b = fix_sum(h_b, target_sum)

        h2 = upsample(best_h, n2)
        h2 = fix_sum(h2, ts2)
        h2, c5_2 = fast_cd(h2, dx2, ts2, 10000, seed=seed + 7000)

        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    if elapsed() < budget_s - 10:

        ts = best_n / 2.0
        h_final = slsqp_refine(best_h, dd, ts, t=300, maxiter=1000)

        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, best_n
