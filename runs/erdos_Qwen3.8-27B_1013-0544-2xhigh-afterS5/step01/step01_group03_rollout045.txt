from scipy.optimize import minimize, Bounds
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, target, tau, maxiter=300):
    """One SLSQP optimization step."""
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        x_scaled = tau * corr
        x_max = np.max(x_scaled)
        weights = np.exp(x_scaled - x_max) / np.sum(np.exp(x_scaled - x_max))
        # Gradient of smooth max is the weighted sum of gradients of each corr
        # For shift k (index n-1+k in full mode):
        # corr[n-1+k] = sum_{i=0}^{n-1-k} x[i+k]*(1-x[i])
        # d/dx[j] = sum over k where j appears
        # This is complex to compute analytically. Let's use numerical gradient or skip it.
        # Actually, let's just not provide gradient and let SLSQP use numerical.
        return None  # Don't provide gradient
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

                  constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return fix_sum(res.x, target)

                best_h = current_h.copy()
