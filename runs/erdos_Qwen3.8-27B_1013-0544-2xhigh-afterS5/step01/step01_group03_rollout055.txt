```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = corr / tau
    x_max = np.max(x)
    return tau * (x_max + np.log(np.sum(np.exp(x - x_max))))

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, tau):
    """Gradient of smooth_max(correlate(h,1-h,'full'), tau) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr / tau
    x_max = np.max(x)
    w = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    
    grad = np.zeros(n)
    half = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        offset = m - half
        i_lo = max(0, -offset)
        i_hi = min(n - 1, n - 1 - offset)
        for i in range(i_lo, i_hi + 1):
            grad[i] += wm * (1.0 - h[i + offset])
            j = i - offset
            if 0 <= j < n:
                grad[i] -= wm * h[j]
    return smooth_max(corr, tau), grad
