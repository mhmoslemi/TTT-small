        g[:n-k] -= h[k:]

        g[p:] -= h[:n-p]

    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(50):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))

        else:
            hi = mid

        grad += w[np.where(active == idx)[0][0]] * grad_corr_lag(h, k)

import cvxpy as cp

def solve_linearized_subproblem(h_current, target_sum, n_lags=None):
    n = len(h_current)
    corr = compute_corr(h_current)
    M = np.max(corr)
    
    # Select active lags (top ones)
    if n_lags is None:
        n_lags = 20
    top_lags = np.argsort(corr)[-n_lags:]
    
    # Get gradients for these lags
    active_grads = []
    for idx in top_lags:
        k = idx - (n-1)
        active_grads.append(grad_corr_lag(h_current, k))
    active_grads = np.array(active_grads)  # shape (n_lags, n)
    active_vals = corr[top_lags]
    
    # Solve: min_t t s.t. active_vals + active_grads @ (h - h_current) <= t
    #        0 <= h <= 1, sum(h) = target_sum
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target_sum,
    ]
    for i in range(n_lags):
        constraints.append(active_vals[i] + active_grads[i] @ (h_var - h_current) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve()
    
    if h_var.value is not None:
        return h_var.value, t.value
    return h_current, M

        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=solver or cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CLARKSON, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_box_sum(h_lp, target_sum)
        c5_lp = compute_c5(h_lp)

        if c5_lp < best_c5 - 1e-12:
            best_h = h_lp.copy()
            best_c5 = c5_lp
            h_curr = h_lp
        else:
            # Damped step toward LP solution
            for alpha in [0.5, 0.25, 0.1]:
                h_try = (1 - alpha) * h_curr + alpha * h_lp
                h_try = project_box_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    h_curr = h_try
                    break
            else:
                break

    return best_h, best_c5

def subgrad_optimize(h, target_sum, deadline, rng, tol_init=0.03, lr_init=0.008):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    lr = lr_init
    tol = tol_init
    stall = 0

    while time.time() < deadline:
        grad, M = subgradient(h_curr, tol)
        improved = False
        for bd in range(8):
            h_try = project_box_sum(h_curr - lr * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try.copy()
                best_c5 = c5_try
                h_curr = h_try
                improved = True
                break
            lr *= 0.5
        if improved:
            stall = 0
            lr = min(lr * 1.08, 0.06)
        else:
            stall += 1
            if stall > 40:
                h_curr = h_curr + rng.randn(n) * (0.03 + 0.02 * min(stall // 20, 3))
                h_curr = project_box_sum(h_curr, target_sum)
                c5_check = compute_c5(h_curr)
                if c5_check < best_c5:
                    best_h = h_curr.copy()
                    best_c5 = c5_check
                lr = lr_init
                tol = min(tol * 1.4, 0.15)
                stall = 0
                if tol > 0.12:
                    break

    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    # Use linear interpolation with wrap-around
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.concatenate([x_old, [2.0]])
    h_new = np.interp(x_new, x_ext, h_ext)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_init = 200
        h0 = 0.5 * np.ones(n_init)
        for i in range(n_init):
            h0[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i * 3 / n_init) + 0.1 * rng.randn()
        h0 = project_box_sum(h0, n_init / 2.0)

    n = len(h0)
    target_sum = n / 2.0
    best_h = project_box_sum(h0, target_sum)
    best_c5 = compute_c5(best_h)
    best_n = n

    deadline_main = t0 + budget_s - 30

    # Phase 1: Cutting plane with varying K
    for K in [15, 25, 40, 60]:
        if time.time() > deadline_main:
            break
        phase_dl = min(time.time() + 60, deadline_main)
        h_opt, c5_opt = cutting_plane(best_h, target_sum, K, max_iters=200, deadline=phase_dl)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Subgradient polish
    if time.time() < deadline_main:
        phase_dl = min(time.time() + 120, deadline_main)
        h_opt, c5_opt = subgrad_optimize(best_h, target_sum, phase_dl, rng)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Try different resolutions
    for n_try in [150, 120, 100, 250, 300]:
        if time.time() > deadline_main:
            break
        h_interp = interpolate_h(best_h, n_try)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        c5_interp = compute_c5(h_interp)

        phase_dl = min(time.time() + 80, deadline_main)
        h_opt, c5_opt = cutting_plane(h_interp, n_try / 2.0, 30, max_iters=150, deadline=phase_dl)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

        if time.time() < deadline_main:
            phase2_dl = min(time.time() + 40, deadline_main)
            h_opt2, c5_opt2 = subgrad_optimize(h_opt, n_try / 2.0, phase2_dl, rng)
            if c5_opt2 < best_c5:
                best_h, best_c5, best_n = h_opt2, c5_opt2, n_try

    # Phase 4: Multi-restart at best resolution
    remaining = deadline_main - time.time()
    if remaining > 30:
        n_use = best_n
        target = n_use / 2.0
        n_restarts = max(2, int(remaining // 60))
        for r in range(n_restarts):
            if time.time() > deadline_main - 5:
                break
            h_start = best_h + rng.randn(n_use) * 0.02 * (r + 1)
            h_start = project_box_sum(h_start, target)
            phase_dl = time.time() + 50
            h_opt, c5_opt = cutting_plane(h_start, target, 30, max_iters=100, deadline=phase_dl)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
            # Quick subgrad follow-up
            if time.time() < deadline_main - 10:
                h_opt2, c5_opt2 = subgrad_optimize(h_opt, target, time.time() + 20, rng)
                if c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2

    # Phase 5: Final polish
    final_dl = t0 + budget_s - 5
    if time.time() < final_dl:
        h_opt, c5_opt = cutting_plane(best_h, len(best_h) / 2.0, 50, max_iters=100, deadline=final_dl)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Ensure feasibility
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
