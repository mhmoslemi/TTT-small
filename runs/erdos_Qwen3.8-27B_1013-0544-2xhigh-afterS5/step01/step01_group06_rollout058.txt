def project_sum(h, target):
    """Project h onto {x: sum(x)=target} while keeping [0,1] constraints."""
    # Use bisection on the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        s = np.sum(np.clip(h-mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo+hi)/2
    return np.clip(h-lam, 0, 1)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr-m)))/alpha

def grad_smooth_max(h, alpha):
    """Gradient of smooth max of corr(h) = correlate(h, 1-h)."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha*(corr-m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:

    # corr[k] = sum_p h[p]*(1-h[p+k]) for appropriate shifts
    # d(corr[k])/d(h[i]) = (1-h[i+k]) - h[i-k] (with boundary handling)
    grad = np.zeros(n)

        if w[idx] < 1e-12:

        k = idx - (n-1)  # actual shift
        # For shift k: d/d(h[i]) [sum_p h[p]*(1-h[p+k])]
        # = (1-h[i+k]) if i+k in [0,n)
        # - h[i-k] if i-k in [0,n)
        # First term: i+k in [0,n)
        if k >= 0:
            grad[k:] += w[idx] * (1.0 - h[:n-k])
        else:
            grad[:-k] += w[idx] * (1.0 - h[-k:])
        # Second term: i-k in [0,n), i.e., i in [k, n+k)
        if k >= 0:
            grad[:n-k] -= w[idx] * h[k:]
        else:
            grad[-k:] -= w[idx] * h[:-k]

import time

def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0/n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        s = np.sum(np.clip(h-mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha*(corr-m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr); w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-10)[0]:
        k = idx - (n-1)
        grad += w[idx] * grad_at_shift(h, k)
    return grad * (2.0/n)

def push_binary(h, target):
    """Push h toward binary while maintaining sum."""
    # Sort and threshold
    sorted_h = np.sort(h)[::-1]
    threshold = sorted_h[int(target)]
    h_bin = (h >= threshold).astype(float)
    # Adjust to get exact sum
    deficit = target - np.sum(h_bin)
    if deficit > 0:
        # Need to add mass to some 0s
        zeros = np.where(h_bin == 0)[0]
        # Sort by original h value (descending) to fill the closest to threshold
        order = zeros[np.argsort(h[zeros])[::-1]]
        for idx in range(int(deficit)):
            h_bin[order[idx]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for idx in range(int(-deficit)):
            h_bin[order[idx]] = 0.0
    return h_bin

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0/n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        s = np.sum(np.clip(h-mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def grad_smooth(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha*(corr-m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr); w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-10)[0]:
        k = idx - (n-1)
        grad += w[idx] * grad_at_shift(h, k)
    return grad * (2.0/n)

def grad_topk(h, k_count=3):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Pick top k shifts
    top_indices = np.argsort(corr)[::-1][:k_count]
    w = np.exp(20*(corr[top_indices]-m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for idx, ti in zip(range(k_count), top_indices):
        k = ti - (n-1)
        grad += w[idx] * grad_at_shift(h, k)
    return grad * (2.0/n)

def optimize_phase(h, target, deadline, rng, method='smooth', alpha=150):
    n = len(h)
    best_h = project_sum(h.astype(float).copy(), target)
    best_c5 = compute_c5(best_h)
    lr = 0.008
    patience = 0
    stall = 0
    
    while time.time() < deadline:
        if method == 'smooth':
            grad = grad_smooth(best_h, alpha)
        elif method == 'topk':
            grad = grad_topk(best_h, 3)
        else:
            grad = grad_smooth(best_h, alpha)
        
        improved = False
        for scale in [1.0, 0.5, 2.0, 0.25]:
            h_try = project_sum(best_h - lr*scale*grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
        
        if improved:
            patience = 0
            lr = min(lr*1.03, 0.04)
        else:
            patience += 1
            lr *= 0.93
            if patience > 40:
                # Try different strategies
                strategies = ['smooth', 'topk', 'perturb']
                strat = strategies[stall % 3]
                if strat == 'perturb':
                    h_pert = best_h + rng.randn(n)*0.03
                    h_pert = project_sum(h_pert, target)
                    c5_pert = compute_c5(h_pert)
                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                else:
                    method = strat
                alpha = min(alpha*1.4, 800)
                lr = 0.008
                patience = 0
                stall += 1
                if stall > 15:
                    break
    return best_h, best_c5

def binary_local_search(h, target, rng, n_iters=3000):
    """Local search on binary solutions."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or best_h[i] == best_h[j]:
            continue
        h_test = best_h.copy()
        h_test[i], h_test[j] = 1.0-h_test[i], 1.0-h_test[j]
        c5 = compute_c5(h_test)
        if c5 < best_c5 - 1e-12:
            best_h = h_test
            best_c5 = c5
    return best_h, best_c5

def make_binary(h, target):
    """Convert to binary with correct sum."""
    n = len(h)
    # Sort descending, set top target to 1
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:int(target)]] = 1.0
    # Handle fractional part
    frac = target - int(target)
    if frac > 0 and int(target) < n:
        h_bin[idx[int(target)]] = frac
    return h_bin

def interpolate(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0/n_old]])
    return np.interp(x_new, x_ext, h_ext)

    return best_h, best_c5, len(best_h)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0/n)

def project_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        s = np.sum(np.clip(h-mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h-(lo+hi)/2, 0, 1)

def grad_at_shift(h, d):
    """Correct gradient of correlation at shift d."""
    n = len(h)
    g = np.zeros(n)
    if d >= 0:
        g[:n-d] += (1.0 - h[d:])
        if d > 0:
            g[d:] -= h[:n-d]
    else:
        p = -d
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def grad_smooth_max(h, alpha):
    """Gradient of smooth max of correlation."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha*(corr-m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-10)[0]:
        d = idx - (n-1)
        grad += w[idx] * grad_at_shift(h, d)
    return grad * (2.0/n)

def grad_topk(h, k_count=3, sharpness=20):
    """Gradient weighted by top-k shifts."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    top_idx = np.argsort(corr)[::-1][:k_count]
    w = np.exp(sharpness*(corr[top_idx]-m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for ii, ti in zip(range(k_count), top_idx):
        d = ti - (n-1)
        grad += w[ii] * grad_at_shift(h, d)
    return grad * (2.0/n)

def subgrad_random(h, rng, k_count=5):
    """Random subgradient: average of k random active shifts."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    # Pick from top shifts
    top = np.argsort(corr)[::-1][:max(k_count*3, 10)]
    chosen = rng.choice(top, size=min(k_count, len(top)), replace=False)
    grad = np.zeros(n)
    for ti in chosen:
        d = ti - (n-1)
        grad += grad_at_shift(h, d)
    grad /= len(chosen)
    return grad * (2.0/n)

def optimize(h0, target, deadline, rng, alpha=150):
    n = len(h0)
    best_h = project_sum(h0.astype(float).copy(), target)
    best_c5 = compute_c5(best_h)
    lr = 0.008
    patience = 0
    stall = 0
    method_cycle = ['smooth', 'topk', 'subgrad']
    
    while time.time() < deadline:
        method = method_cycle[stall % 3]
        if method == 'smooth':
            grad = grad_smooth_max(best_h, alpha)
        elif method == 'topk':
            grad = grad_topk(best_h, 3)
        else:
            grad = subgrad_random(best_h, rng, 5)
        
        improved = False
        for scale in [1.0, 0.5, 2.0, 0.25, 4.0]:
            h_try = project_sum(best_h - lr*scale*grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
        
        if improved:
            patience = 0
            lr = min(lr*1.02, 0.04)
        else:
            patience += 1
            lr *= 0.94
            if patience > 30:
                # Perturbation
                h_pert = best_h + rng.randn(n)*0.03
                h_pert = project_sum(h_pert, target)
                if compute_c5(h_pert) < best_c5:
                    best_h = h_pert
                    best_c5 = compute_c5(best_h)
                alpha = min(alpha*1.3, 800)
                lr = 0.008
                patience = 0
                stall += 1
                if stall > 12:
                    break
    return best_h, best_c5

def binary_local_search(h, target, rng, n_iters=3000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or best_h[i] == best_h[j]:
            continue
        h_test = best_h.copy()
        h_test[i], h_test[j] = 1.0-h_test[i], 1.0-h_test[j]
        c5 = compute_c5(h_test)
        if c5 < best_c5 - 1e-12:
            best_h = h_test
            best_c5 = c5
    return best_h, best_c5

def make_binary(h, target):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:int(target)]] = 1.0
    frac = target - int(target)
    if frac > 1e-9 and int(target) < n:
        h_bin[idx[int(target)]] = frac
    return h_bin

def interpolate_to(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0/n_old]])
    return np.interp(x_new, x_ext, h_ext)
