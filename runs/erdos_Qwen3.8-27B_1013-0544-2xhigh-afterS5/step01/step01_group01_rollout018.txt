def overlap_at_lag(h, k, n):
    """Compute overlap integral at a specific lag k"""
    dx = 2.0 / n
    if k >= 0:
        # sum h[i] * (1-h[i+k]) for i=0..n-1-k
        idx1 = np.arange(n - k)
        idx2 = idx1 + k
    else:
        # sum h[i] * (1-h[i-|k|]) for i=|k|..n-1
        idx2 = np.arange(n + k)
        idx1 = idx2 - k
    return np.sum(h[idx1] * (1 - h[idx2])) * dx

def max_overlap_and_lag(h):

    k_star = np.argmax(corr)

from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        # Iterative projection
        for _ in range(50):
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            if abs(s - target_sum) < 1e-10:
                break
            h += (target_sum - s) / len(h)
    return h

def overlap_at_lag(h, lag, n):
    dx = 2.0 / n
    if lag >= 0:
        m = n - lag
        return np.sum(h[:m] * (1.0 - h[lag:])) * dx
    else:
        m = n + lag
        return np.sum(h[-lag:] * (1.0 - h[:m])) * dx

def grad_overlap_at_lag(h, lag, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        g[:m] += (1.0 - h[lag:]) * dx
        g[lag:] -= h[:m] * dx
    else:
        m = n + lag
        g[-lag:] += (1.0 - h[:m]) * dx
        g[:m] -= h[-lag:] * dx
    return g

            res = minimize(obj_h, best_h, jac=jac_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Lag-focused optimization (minimize individual worst lags)
    for _ in range(8):
        if time_left() < 15:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_star = int(np.argmax(corr))
        lag_star = k_star - (n_points - 1)
        if lag_star == 0:
            continue

        def obj_lag(x):
            return overlap_value(x, lag_star, n_points)

        def jac_lag(x):
            return overlap_grad(x, lag_star, n_points) * dx

        try:
            res = minimize(obj_lag, best_h, jac=jac_lag, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

        # Also try minimizing top-2 lags jointly
        top2 = np.argsort(corr)[-2:]
        lags2 = top2 - (n_points - 1)
        lags2 = [l for l in lags2 if l != 0]
        if len(lags2) >= 2:
            def obj_2lag(x):
                return max(overlap_value(x, lags2[0], n_points),
                           overlap_value(x, lags2[1], n_points))

            try:
                res = minimize(obj_2lag, best_h, method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-8})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 3: Diversified perturbations with local optimization
    attempt = 0
    no_improve = 0
    best_perturb_scale = 0.03

    while time_left() > 12 and no_improve < 50:
        attempt += 1
        strategy = attempt % 8

        if strategy == 0:
            scale = best_perturb_scale * np.exp(-attempt / 50.0) + 0.005
            h0 = best_h + np.random.randn(n_points) * scale
        elif strategy == 1:
            # Target worst lag: push values to reduce that specific overlap
            h0 = best_h.copy()
            corr_tmp = np.correlate(h0, 1.0 - h0, mode="full")
            ks = int(np.argmax(corr_tmp))
            lag = ks - (n_points - 1)
            if lag > 0:
                m = n_points - lag
                contrib = h0[:m] * (1.0 - h0[lag:])
                n_top = max(2, m // 6)
                top_idx = np.argsort(contrib)[-n_top:]
                amount = 0.03 + 0.02 * np.random.random()
                h0[top_idx] -= amount
                h0[top_idx + lag] += amount
            elif lag < 0:
                m = n_points + lag
                contrib = h0[-lag:] * (1.0 - h0[:m])
                n_top = max(2, m // 6)
                top_idx = np.argsort(contrib)[-n_top:]
                amount = 0.03 + 0.02 * np.random.random()
                h0[top_idx - lag] -= amount
                h0[top_idx] += amount
        elif strategy == 2:
            # Antisymmetric perturbation around worst lag
            h0 = best_h.copy()
            corr_tmp = np.correlate(h0, 1.0 - h0, mode="full")
            ks = int(np.argmax(corr_tmp))
            lag = ks - (n_points - 1)
            if abs(lag) > 0:
                for j in range(n_points):
                    i1 = j + lag
                    if 0 <= i1 < n_points:
                        h0[j] += 0.02 * (h0[i1] - 0.5)
        elif strategy == 3:
            # Binary flip
            h0 = best_h.copy()
            n_flip = np.random.randint(2, 7)
            idx = np.random.choice(n_points, n_flip, replace=False)
            h0[idx] = 1.0 - h0[idx]
        elif strategy == 4:
            # Block swap
            h0 = best_h.copy()
            blen = np.random.randint(2, max(3, n_points // 5))
            b1 = np.random.randint(0, n_points - blen)
            b2 = np.random.randint(0, n_points - blen)
            if abs(b1 - b2) >= blen:
                tmp = h0[b1:b1 + blen].copy()
                h0[b1:b1 + blen] = h0[b2:b2 + blen]
                h0[b2:b2 + blen] = tmp
        elif strategy == 5:
            # Push toward binary (sharpen)
            h0 = best_h.copy()
            h0[h0 > 0.5] = np.clip(h0[h0 > 0.5] + 0.1, 0, 1)
            h0[h0 <= 0.5] = np.clip(h0[h0 <= 0.5] - 0.1, 0, 1)
        elif strategy == 6:
            # Partial random restart on subset
            h0 = best_h.copy()
            mask = np.random.random(n_points) < 0.25
            h0[mask] = 0.15 + 0.7 * np.random.random(int(mask.sum()))
        else:
            # Smooth structured perturbation
            h0 = best_h.copy()
            freq = np.random.uniform(1, 4)
            phase = np.random.uniform(0, 2 * np.pi)
            x = np.arange(n_points)
            h0 += 0.04 * np.sin(2 * np.pi * freq * x / n_points + phase)

        h0 = project_h(h0, target_sum)

        # Local optimization from perturbed point
        try:
            def obj_r(x):
                return smooth_max(np.correlate(x, 1.0 - x, mode="full"), 200.0) * dx

            res = minimize(obj_r, h0, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 250, 'ftol': 1e-13})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                no_improve = 0
                best_perturb_scale *= 0.85
            else:
                no_improve += 1
                best_perturb_scale *= 1.05
                best_perturb_scale = min(best_perturb_scale, 0.08)
        except Exception:
            no_improve += 1

    # Phase 4: Additional lag-focused passes
    for _ in range(5):
        if time_left() < 10:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        # Find top 3 lags
        top3 = np.argsort(corr)[-3:]
        lags3 = [int(l) - (n_points - 1) for l in top3 if int(l) - (n_points - 1) != 0]

        for lag in lags3:
            if time_left() < 8:
                break
            def obj_l(x):
                return overlap_value(x, lag, n_points)
            def jac_l(x):
                return overlap_grad(x, lag, n_points) * dx
            try:
                res = minimize(obj_l, best_h, jac=jac_l, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 200, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    if time_left() > 5:
        for beta in [800, 1500, 3000]:
            if time_left() < 3:
                break

            def obj_p(x):
                return smooth_max_obj_grad(x, n_points, beta)[0]

            def jac_p(x):
                return smooth_max_obj_grad(x, n_points, beta)[1]

            try:
                res = minimize(obj_p, best_h, jac=jac_p, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
