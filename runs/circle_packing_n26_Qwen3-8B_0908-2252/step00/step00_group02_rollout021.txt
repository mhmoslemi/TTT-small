import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based configuration with some spacing
    num_circles = 26
    grid_rows = 4
    grid_cols = 7
    spacing = 0.35
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place circles in a grid layout
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        x = col * spacing
        y = row * spacing
        centers[i] = [x, y]

    # Function to compute sum of radii
    def sum_radii(radii):
        return np.sum(radii)

    # Optimization function with constraints
    def objective(radii):
        return -sum_radii(radii)  # Minimize negative sum to maximize sum

    # Boundary constraints: each radius must be non-negative and fit within the square
    def constraint_boundary(radii):
        return np.array([r for r in radii])

    # Constraint that circles must not overlap
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx * dx + dy * dy)
                if distance < radii[i] + radii[j] - 1e-12:
                    return np.array([1.0])
        return np.array([0.0])

    # Optimization options
    bounds = [(0, 1) for _ in range(num_circles)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary, 'args': (radii,)},
        {'type': 'ineq', 'fun': constraint_overlap, 'args': (radii, centers)},
    ]

    # Initial guess for radii
    initial_guess = np.ones(num_circles) * 0.1

    # Perform optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Check if optimization was successful
    if not result.success:
        raise RuntimeError("Optimization failed")

    # Extract results
    optimized_radii = result.x
    optimized_centers = centers.copy()

    # Validate the result
    valid, message = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    sum_radii_value = np.sum(optimized_radii)
    return optimized_centers, optimized_radii, sum_radii_value

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
