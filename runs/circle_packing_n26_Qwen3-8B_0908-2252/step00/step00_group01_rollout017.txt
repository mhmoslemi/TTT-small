import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():

    n_circles = 26

    radii = np.full(n_circles, 0.03)  # Small initial radii

    # Objective function to maximize (sum of radii)
    def objective(x):
        # x is a flattened array of [centers..., radii...]
        centers_flat = x[:n_circles * 2].reshape(n_circles, 2)
        radii_flat = x[n_circles * 2:]
        return -np.sum(radii_flat)  # Minimize negative sum of radii

    # Constraint: circles within the square [0,1]x[0,1]
    def bounds_func(x):
        centers_flat = x[:n_circles * 2].reshape(n_circles, 2)
        radii_flat = x[n_circles * 2:]
        for i in range(n_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf
        return 0

    # Constraint: circles do not overlap
    def constraints_func(x):
        centers_flat = x[:n_circles * 2].reshape(n_circles, 2)
        radii_flat = x[n_circles * 2:]
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return np.inf
        return 0

    # Total number of variables: 2*n_circles + n_circles = 3*n_circles
    num_vars = 3 * n_circles
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds: 0 <= x <= 1 for centers, 0 <= r <= 1 for radii

        low=np.concatenate([np.zeros(num_vars - n_circles), np.zeros(n_circles)]),
        high=np.concatenate([np.ones(num_vars - n_circles), np.ones(n_circles)])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
