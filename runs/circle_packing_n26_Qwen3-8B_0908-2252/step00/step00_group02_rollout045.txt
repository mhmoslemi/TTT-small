import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    grid_size = 5
    initial_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            if i == 0 and j == 4:
                continue
            x = j / (grid_size - 1)
            y = i / (grid_size - 1)
            initial_centers.append([x, y])
    
    centers = np.array(initial_centers)
    radii = np.array([0.05] * len(centers))  # Initial guess for radii

    # Define the objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Define constraints
    def constraints(x):
        # x is a flat array of centers and radii
        n = len(centers)
        x_centers = x[:2 * n].reshape(n, 2)

        # Boundary constraints
        for i in range(n):
            x_i, y_i = x_centers[i]
            r_i = x_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf  # Violation of boundary constraint
        
        # Non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i][0] - x_centers[j][0]
                dy = x_centers[i][1] - x_centers[j][1]

        # All constraints satisfied
        return 0

    # Flatten centers and radii for optimization
