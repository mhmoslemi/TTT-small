        return -np.sum(radii_flat)  # Negative because we minimize

    def constraint_dist(x: np.ndarray) -> np.ndarray:
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:]
