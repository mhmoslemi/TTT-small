def run_packing():
    # Initial hexagonal grid setup
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))

    # Define hexagonal packing spacing
    spacing = 0.45  # Adjusted for better packing within unit square
    radii = np.full(num_circles, 0.02)  # Initial guess for radii

    # Initially place circles in hexagonal grid
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col * spacing
        y = row * (np.sqrt(3)/2) * spacing
        centers[i, 0] = x
        centers[i, 1] = y

    # Ensure circle positions are within [0, 1] by scaling
    centers /= max(np.max(centers, axis=0))
    centers = np.clip(centers, 0, 1)

    # Define objective function
    def objective(x):
        # x is a vector of radii: x[0], x[1], ..., x[25]
        return -np.sum(x)  # Maximize by minimizing -sum(x)

    # Define constraints for boundaries and non-overlapping circles
    def constraint_boundary(x, centers):
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return False
        return True

    def constraint_overlap(x, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-8:
                    return False
        return True

    bounds = Bounds(0, 1.0)  # Radii must be non-negative and fit in unit square

    # Constraints for boundaries and non-overlapping

        {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)},

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
