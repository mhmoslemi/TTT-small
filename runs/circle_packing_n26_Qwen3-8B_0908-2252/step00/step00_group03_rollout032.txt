import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():

    n = 26
    radius_start = 0.05
    centers = []
    radii = []

    # For hexagonal packing, place circles in rows with alternating offsets
    rows = 6
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            x_scaled = x * 0.9
            y_scaled = y * 0.9
            centers.append([x_scaled, y_scaled])
            radii.append(radius_start)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Step 2: Define the objective function to maximize sum of radii
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        x_flat = x.reshape(-1, 3)
        centers = x_flat[:, :2]
        radii = x_flat[:, 2]
        
        # Compute total sum of radii
        total_sum = np.sum(radii)

        # Add constraints as penalties
        # 1. Ensure circles are within the square
        penalty = 0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                penalty += 1e5
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    penalty += 1e5
        return -total_sum + penalty  # Minimize negative of total sum

    # Step 3: Add bounds and constraints to the optimization
    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 0.5))  # radius (since initial radius is 0.05)

    # Initial guess: centers and radii we generated
    x0 = np.reshape(np.vstack((centers, radii)), -1)

    # Run the optimization
    result = optimize.minimize(objective, x0, bounds=bounds, method='SLSQP', tol=1e-6)

    x_flat_result = result.x.reshape(-1, 3)
