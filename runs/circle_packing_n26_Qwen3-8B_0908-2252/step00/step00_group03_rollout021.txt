import numpy as np
import scipy.optimize as optimize

def run_packing():

    n = 26

    rows = 5
    cols = 6
    spacing = 0.25
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * spacing + 0.05
        y = row * spacing + 0.05
        centers[i] = [x, y]
        radii[i] = 0.05  # Initial radius

    # Define objective function (negative of sum of radii to be minimized)
    def objective(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)

    # Constraints: no circle outside the square
    def square_bounds(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return [
            centers_flat[i, 0] - radii_flat[i] + 1e-12 for i in range(n)
        ] + [
            centers_flat[i, 0] + radii_flat[i] - 1 + 1e-12 for i in range(n)
        ] + [
            centers_flat[i, 1] - radii_flat[i] + 1e-12 for i in range(n)
        ] + [
            centers_flat[i, 1] + radii_flat[i] - 1 + 1e-12 for i in range(n)
        ]

    # Constraints: no overlap between circles
    def circle_overlap(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist_sq = dx*dx + dy*dy
                dist = np.sqrt(dist_sq)
                overlaps.append(dist - radii_flat[i] - radii_flat[j] + 1e-12)
        return overlaps

    # Optimization problem
    x0 = np.concatenate([centers.flatten(), radii])
    bounds = [
        (0.0, 1.0) for _ in range(n*2)
    ] + [
        (0.0, 0.5) for _ in range(n)

    # Use SLSQP with bounds and constraints
    result = optimize.minimize(
