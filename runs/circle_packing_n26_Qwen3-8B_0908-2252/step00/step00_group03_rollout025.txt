import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        centers[i, 0] = col / (grid_cols - 1) * 1.0
        centers[i, 1] = row / (grid_rows - 1) * 1.0
    radii = np.full(num_circles, 0.05)  # Initial small radius

    # Function to compute the objective: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Function to compute the constraints
    def constraints(centers, radii):
        n = centers.shape[0]
        constraints = []
        for i in range(n):
            # Boundary constraints
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                constraints.append((centers[i, 0], -radii[i], 1 - radii[i]))
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                constraints.append((centers[i, 1], -radii[i], 1 - radii[i]))

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    constraints.append((dist, radii[i] + radii[j] - 1e-12))
        return constraints

    # Define optimization bounds
    bounds = Bounds([0, 0], [1, 1])

    # Run optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        jac=None,
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
