import numpy as np
import cvxpy as cp
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    radius = cp.Variable(n)
    centers = cp.Variable((n, 2))

    # Objective: Maximize sum of radii
    obj = cp.Maximize(cp.sum(radius))

    # Constraints
    constraints = []

    # Non-overlapping constraint
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radius[i] + radius[j] - 1e-8)

    # Boundary constraint
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= radius[i] - 1e-8)
        constraints.append(x <= 1 - radius[i] + 1e-8)
        constraints.append(y >= radius[i] - 1e-8)
        constraints.append(y <= 1 - radius[i] + 1e-8)

    # Non-negativity constraint
    constraints.append(radius >= 0)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    # Extract solution
    centers_np = centers.value
    radii_np = radius.value
    sum_radii = np.sum(radii_np)

    # Validate the packing
    valid, message = validate_packing(centers_np, radii_np)
    if not valid:
        raise ValueError(message)

    return (centers_np, radii_np, sum_radii)


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
