import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Total number of circles
    num_circles = 26

    # Initialize centers with a regular grid-like pattern
    # 5 rows with 5, 5, 5, 5, and 6 circles
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Rows
    rows = 5
    cols = 5
    spacing_x = 1.0 / (cols + 1)
    spacing_y = 1.0 / (rows + 1)

    # Grid-based initial placement
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col * spacing_x + spacing_x / 2
        y = row * spacing_y + spacing_y / 2
        centers[i] = [x, y]

    # Initial estimate for the maximum radius
    initial_radius = 0.12

    # Define optimization variables
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles)

    # Constraints: centers must be within [0,1] and radii non-negative
    constraints = []
    for i in range(num_circles):
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)
        constraints.append(r[i] >= 0)

    # Constraints: distance between centers >= sum of radii
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Define and solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    # Extract the optimized centers and radii
    centers_opt = np.array(x.value)
