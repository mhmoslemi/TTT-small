import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal arrangement with 5 rows and alternating row sizes
    num_circles = 26
    rows = 5
    cols_per_row = [5, 4, 5, 4, 5]
    
    # Generate hexagonal grid positions
    centers = []
    for i in range(rows):
        row_size = cols_per_row[i]
        for j in range(row_size):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers.append([x, y])
    
    centers = np.array(centers)
    
    # Scale the centers to fit in [0,1]x[1]
    centers /= max(centers.max(axis=0))
    
    # Initial radii: equally sized circles
    radii = np.full(num_circles, 0.05)
    
    # Optimization bounds for radii
    bounds = [(0, 1 - 1e-6) for _ in range(num_circles)]
    
    # Objective function to maximize sum_radii (minimize -sum_radii)
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints
    constraints = []
    
    # Distance to boundaries
    for i in range(num_circles):
        def dist_to_boundary(i):
            x, y = centers[i]
            r = radii[i]
            return [(x - r, 1 - x - r), (y - r, 1 - y - r)]
        
        # Ensure circle is within the square
        for dx, dy in dist_to_boundary(i):
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, i=i, dx=dx, dy=dy: dx - r[i] if dx > 0 else dy - r[i]
            })
    
    # Circle-circle distance constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):

                    return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (r[i] + r[j])
                return func
            
            constraints.append({
                'type': 'ineq',
                'fun': dist_between(i, j)
            })
    
    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints
    )
    
    # Validate packing

    return centers, result.x, np.sum(result.x)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
