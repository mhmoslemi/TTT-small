import scipy.optimize as opt
import random

    Returns the optimal centers and radii for 26 circles packed in a unit square
    to maximize the total sum of radii, along with the total sum.
    """

    centers = np.empty((26, 2))
    radii = np.empty(26)

    # Hexagonal placement for 16 circles
    hex_grid_centers = []
    for row in range(4):
        for col in range(4):
            if row % 2 == 0:
                x = col * np.sqrt(3)/2 + 0.5
                y = row * 0.5 + 0.5
            else:
                x = (col + 0.5) * np.sqrt(3)/2 + 0.5
                y = row * 0.5 + 0.5
            hex_grid_centers.append((x, y))
    
    # Adjust to fit in unit square
    for i, (x, y) in enumerate(hex_grid_centers):
        x = x * 0.8 + 0.1
        y = y * 0.8 + 0.1
        centers[i, 0] = x
        centers[i, 1] = y

    # Edge and corner circles
    for i in range(16, 26):
        angle = 2 * np.pi * i / 10
        radius = 0.2
        x = 0.5 + radius * np.cos(angle)
        y = 0.5 + radius * np.sin(angle)
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.2

    # Initial radii
    for i in range(16):
        radii[i] = 0.15

    # Optimization
    def objective(radii_opt):
        # Ensure non-negativity
        radii_opt = np.maximum(radii_opt, 0)
        # Check if within square
        in_square = np.all(
            (centers[:, 0] - radii_opt >= -1e-12) &
            (centers[:, 0] + radii_opt <= 1 + 1e-12) &
            (centers[:, 1] - radii_opt >= -1e-12) &

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
