def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    initial_radii = np.full(n, 1 / (np.sqrt(3) + 1))  # estimate based on hexagonal packing density
    # Initial centers from a hexagonal grid in a 1x1 square
    centers = np.zeros((n, 2))
    for i in range(n):
        col = i % 6
        row = i // 6
        # Hexagonal grid with spacing based on radius
        centers[i, 0] = col * (1 / (np.sqrt(3) * initial_radii[i] * 2))
        centers[i, 1] = row * (1 / (initial_radii[i] * 2)) + 0.5 * initial_radii[i]
    
    # Ensure centers are within the square

    # Optimization using CVXPY
    radii = cp.Variable(n, nonneg=True)
    centers_var = cp.Variable((n, 2))
    constraints = []

    # Boundary constraints
    for i in range(n):
        constraints.append(centers_var[i, 0] >= 0)
        constraints.append(centers_var[i, 0] <= 1)
        constraints.append(centers_var[i, 1] >= 0)
        constraints.append(centers_var[i, 1] <= 1)

    # Non-overlap constraints (distance between centers >= sum of radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
