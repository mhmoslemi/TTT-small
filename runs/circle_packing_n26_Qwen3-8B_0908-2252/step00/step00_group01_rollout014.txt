import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Hexagonal grid: 5 rows with 6 and 1 row with 4
    # Start with a hexagonal grid layout
    centers = []
    for row in range(5):
        for col in range(6 if row % 2 == 0 else 5):
            x = col * (1 / 3) + 0.5 * (1 / 6)
            y = row * (np.sqrt(3) / 3) + 0.5
            if x > 1 or y > 1 or x < 0 or y < 0:
                continue
            centers.append([x, y])
    # Adjust to fit remaining 4 circles in the last row
    for col in range(4):
        x = col * (1 / 3) + 0.5 * (1 / 6)
        y = 5 * (np.sqrt(3) / 3) + 0.5
        if x > 1 or y > 1 or x < 0 or y < 0:
            continue
        centers.append([x, y])
    centers = np.array(centers)
    # Initial radii (small)
    radii = np.full(n, 0.001)

    def objective(radii):
        return -np.sum(radii)

    def constraint_distance(i, j, x, y, radii):
        dx = x[i] - x[j]
        dy = y[i] - y[j]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - (radii[i] + radii[j])

    def constraint_bound(i, x, y, radii):
        return x[i] - radii[i] - 0.0
    def constraint_bound2(i, x, y, radii):
        return 1.0 - x[i] - radii[i]
    def constraint_bound3(i, x, y, radii):
        return y[i] - radii[i] - 0.0
    def constraint_bound4(i, x, y, radii):
        return 1.0 - y[i] - radii[i]

    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # radii[i] must be between 0 and 1

    # Set up constraints
    cons = []

    # Bounding box constraints
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_bound(i, centers, centers, r)})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_bound2(i, centers, centers, r)})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_bound3(i, centers, centers, r)})
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_bound4(i, centers, centers, r)})

    for i in range(n):

                return c
            cons.append({'type': 'ineq', 'fun': constraint})

    result = opt.minimize(

        x0=radii,
        bounds=bounds,
        constraints=cons,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
