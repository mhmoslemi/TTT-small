import cvxpy as cp

def run_packing():
    n = 26
    # Initial hexagonal grid: 5 rows with 5, 5, 6, 5, 5 circles
    rows = 5
    cols = np.array([5, 5, 6, 5, 5])
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols[i]):

            y = i * (np.sqrt(3)/2)
            centers[idx] = [x, y]
            idx += 1
    
    # Define variables
    radii = cp.Variable(n, nonneg=True)
    x = cp.Variable((n, 2), nonneg=True)
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    
    # Each circle must lie within the unit square
    for i in range(n):
        x_i, y_i = x[i]
        r_i = radii[i]
        constraints += [x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1]
    
    # Circles must not overlap

        for j in range(i + 1, n):
            dx = x[i, 0] - x[j, 0]
            dy = x[i, 1] - x[j, 1]
            dist = cp.sqrt(dx*dx + dy*dy)
            constraints += [dist >= radii[i] + radii[j]]
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS)
    
    # Extract the solution
    centers = x.value

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
