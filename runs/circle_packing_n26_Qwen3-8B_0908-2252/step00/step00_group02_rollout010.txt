from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    rows = 5
    cols = 5
    spacing = 0.2  # Adjusted spacing to avoid overlap
    radius_guess = spacing / 2

            y = i * spacing * math.sqrt(3) / 2 + spacing / 2

            radii[i * cols + j] = radius_guess

    radii[-1] = radius_guess

        # x is a vector of all variables (flattened centers and radii)
        # Unpack the variables
        x_flat = x.reshape(-1, 2)
        r_flat = x[-n:]  # Last n variables are the radii

        # Objective function
        return -np.sum(r_flat)  # Negative because we're minimizing

    # Define the constraints
    def constraints(x):
        x_flat = x.reshape(-1, 2)
        r_flat = x[-n:]  # Last n variables are the radii

        # Constraint 1: Each circle center must be inside [0,1]^2
        center_inside = np.all((x_flat[:, 0] >= 0) & (x_flat[:, 0] <= 1) &

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
