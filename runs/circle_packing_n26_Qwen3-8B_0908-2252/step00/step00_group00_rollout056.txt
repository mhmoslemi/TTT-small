import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial positions and radii for a hexagonal lattice
    def initial_hexagonal_positions(n):
        radius = 0.06  # Initial guess for the radius
        positions = []
        for i in range(n):
            row = i // 6
            col = i % 6
            x = col * (2 * radius)
            y = row * (np.sqrt(3) * radius)
            if row % 2 == 1:
                x += radius
            positions.append([x, y])
        return np.array(positions), np.full(n, radius)

    # Objective function to maximize sum of radii
    def objective(x, n, positions):
        # x contains all radii
        radii = x
        return -np.sum(radii)  # Negative for minimization

    # Constraints: circles inside the square
    def boundary_constraints(x, n, positions):
        radii = x
        return [positions[i][0] - radii[i] + 1e-12 for i in range(n)] + \
               [positions[i][1] - radii[i] + 1e-12 for i in range(n)] + \
               [1 - positions[i][0] - radii[i] + 1e-12 for i in range(n)] + \
               [1 - positions[i][1] - radii[i] + 1e-12 for i in range(n)]

    # Constraints: circles do not overlap
    def non_overlap_constraints(x, n, positions):
        radii = x
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = positions[i][0] - positions[j][0]
                dy = positions[i][1] - positions[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(radii[i] + radii[j] - dist + 1e-12)
        return constraints

    # Optimization setup
    n = 26
    positions, initial_radii = initial_hexagonal_positions(n)
    
    # Optimization problem
    bounds = [(0, 1.0) for _ in range(n)]
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda x: boundary_constraints(x, n, positions)})
    cons.append({'type': 'ineq', 'fun': lambda x: non_overlap_constraints(x, n, positions)})
    
    result = opt.minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'eps': 1e-6}
    )
    
    # Extract the solution
    centers = positions
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
