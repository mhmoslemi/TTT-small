import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal packing for 26 circles
    num_circles = 26
    num_rows = 5
    num_cols = 5
    # A hexagonal grid with 5 rows and 5 columns
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Initial spacing
    spacing = 1.0 / (num_rows + 1)
    radius_initial = spacing / 2.0  # Start with a small radius

    # Hexagonal grid positions
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * spacing + spacing / 2.0
            y = i * spacing * np.sqrt(3) / 2.0 + spacing / 2.0
            if i % 2 == 0:
                x += spacing / 2.0
            centers[i * num_cols + j] = [x, y]
            radii[i * num_cols + j] = radius_initial

    # Optimization
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum (maximize sum)

    def constraints(radii, centers):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    constraints.append((radii[i] + radii[j]) - dist)
        return constraints

    # Optimize
    initial_radii = radii.copy()
    bounds = Bounds([0.0] * num_circles, [1.0] * num_circles)

    result = minimize(
        fun=objective,
        x0=initial_radii,
        method="L-BFGS-B",
        bounds=bounds,
        constraints={"type": "ineq", "fun": lambda r: constraints(r, centers)}
    )

    radii_opt = result.x

    if not valid:
        raise ValueError(message)

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
