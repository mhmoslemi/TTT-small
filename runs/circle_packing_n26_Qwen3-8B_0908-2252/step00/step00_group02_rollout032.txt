import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Constants and initial parameters
    n = 26
    max_radius = 1.0
    margin = 1e-6
    initial_centers = np.zeros((n, 2))
    bounds = np.array([[0.0, 1.0], [0.0, 1.0]])

    # Hexagonal packing of 26 circles
    # Approximate hexagonal grid that fits within the unit square
    # Start with a hexagonal packing, which is approximately the most efficient packing
    # Number of rows: 5 (since sqrt(26) ≈ 5.1)
    num_rows = 5
    num_cols = 6
    min_row_spacing = 1.0 / (num_rows - 1)
    row_offset = np.sqrt(3) * min_row_spacing / 2
    col_spacing = 1.0 / (num_cols - 1)

    # Generate initial hexagonal grid
    initial_centers = np.zeros((n, 2))
    idx = 0
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * col_spacing
            y = row * min_row_spacing + (row % 2) * row_offset
            if idx < n:
                initial_centers[idx] = [x, y]
                idx += 1
            else:
                break

    # Initial radii: all the same, with a small margin
    initial_radii = np.ones(n) * (1.0 / (np.sqrt(2) * 2))  # Initial estimate for radii

    # Objective function to maximize sum of radii
    def objective(x):
        # x is a vector of (centers, radii): first 2*n elements are centers, then n radii
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Constraints
    def constraint_centers(x):
        centers = x[:2*n].reshape(n, 2)
        return centers

    def constraint_radii(x):
        radii = x[2*n:]
        return radii

    def constraint_bound(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return np.concatenate([
            (centers[:, 0] - radii) >= -margin,
            (centers[:, 0] + radii) <= 1 + margin,
            (centers[:, 1] - radii) >= -margin,
            (centers[:, 1] + radii) <= 1 + margin
        ])

    def constraint_overlap(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]) + margin)
        return constraints

    # Set up optimization problem
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])

    # Define bounds for centers and radii
    center_bounds = np.tile([[0.0, 1.0], [0.0, 1.0]], (n, 1)).flatten()
    radius_bounds = np.array([0.0, max_radius]) * np.ones(n)
    bounds = Bounds(center_bounds, center_bounds)
    radius_bounds = Bounds(radius_bounds, radius_bounds)

    # Solve the optimization problem
    res = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_bound},
            {'type': 'ineq', 'fun': constraint_overlap}
        ],
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the result
    centers = res.x[:2*n].reshape(n, 2)
    radii = res.x[2*n:]

    # Final validation to ensure everything is correctly constrained
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing invalid: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
