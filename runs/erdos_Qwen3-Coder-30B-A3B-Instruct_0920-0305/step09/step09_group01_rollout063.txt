TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_shifted_gradients(h, max_shift_idx):
    """Compute gradient of the log-sum-exp of shift values"""
    # Compute all shift values via FFT
    cc = compute_corr_fft(h)
    
    # Use log-sum-exp trick for numerical stability
    maxval = np.max(cc)
    exp_vals = np.exp((cc - maxval) / 0.05)  # μ = 0.05
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for k in range(-max_shift_idx + 1, max_shift_idx):
        if k >= 0:
            # Positive shift
            grad[:len(h)-k] += weights[k] * (1 - h[k:])
            grad[k:] -= weights[k] * h[:len(h)-k]
        else:
            # Negative shift
            k = -k
            grad[:len(h)-k] -= weights[-k] * h[k:]
            grad[k:] += weights[-k] * (1 - h[:len(h)-k])
    
    return grad
