def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_continuous(h_bin, noise_scale=0.01):
    """Convert binary vector to continuous with noise and projection"""
    h_continuous = h_bin.astype(float) + np.random.uniform(-noise_scale, noise_scale, len(h_bin))
    return project_to_sum_box(h_continuous, TARGET_SUM)

def compute_gradient_smoothed(h, c, alpha):

    exp_c = np.exp(alpha * c)
    w = exp_c / np.sum(exp_c)

    # Iterate over all shifts
    for k in range(-len(h) + 1, len(h)):
        if k >= 0:
            g[:len(h)-k] -= w[k] * h[k:]
            g[k:] -= w[k] * h[:len(h)-k]
        else:
            s = -k
            g[s:] -= w[k] * h[:len(h)-s]
            g[:len(h)-s] -= w[k] * h[s:]
    
    return g
