import numpy as np

from scipy.special import logsumexp
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smooth_max_grad(h, alpha=2.0):
    """Compute gradient of smooth max using log-sum-exp approximation"""
    c = compute_corr_fft(h)
    F_alpha = logsumexp(alpha * c) / alpha
    
    # Compute weights
    exp_alpha_c = np.exp(alpha * c - np.max(alpha * c))
    weights = exp_alpha_c / np.sum(exp_alpha_c)
    
    # Compute gradient
    g = np.ones_like(h)
    n = len(h)
    
    # Loop over all shifts
    for k in range(-n+1, n):
        if k >= 0:
            g[:n-k] -= weights[k] * h[k:]
            g[k:] -= weights[k] * h[:n-k]
        else:
            s = -k
            g[s:] -= weights[k] * h[:n-s]
            g[:n-s] -= weights[k] * h[s:]
    
    return g, F_alpha

def compute_true_max_grad(h):
    """Compute subgradient of true max"""
    c = compute_corr_fft(h)
    max_c = np.max(c)
    active_indices = np.where(np.abs(c - max_c) < 1e-10)[0]
    
    # Uniform weights
    weights = np.zeros_like(c)
    weights[active_indices] = 1.0 / len(active_indices)
    
    # Compute subgradient
    g = np.ones_like(h)
    n = len(h)
    
    for k in active_indices:
        if k >= 0:
            g[:n-k] -= weights[k] * h[k:]
            g[k:] -= weights[k] * h[:n-k]
        else:
            s = -k
            g[s:] -= weights[k] * h[:n-s]
            g[:n-s] -= weights[k] * h[s:]
    
    return g

            return evaluate_c5(h)

            return np.sum(h) - TARGET_SUM

            )

                final_c5 = evaluate_c5(h_final)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = h_final.copy()
        except:
            pass
    
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
