import numpy as np
from scipy.signal import correlate

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_gradient_f_epsilon(h, c_vals, eps):
    """Compute gradient of log-sum-exp of overlaps"""
    # Compute log-sum-exp for numerical stability
    max_c = np.max(c_vals)
    if max_c == -np.inf:
        weights = np.ones_like(c_vals)
    else:
        exp_vals = np.exp(eps * (c_vals - max_c))
        weights = exp_vals / np.sum(exp_vals)
        
    # Construct gradient vector (for indexing)
    grad = np.zeros_like(h)
    n = len(h)
    
    # Add contributions from all shifts
    for k in range(-n+1, n):
        # Compute the contribution of this shift
        if k >= 0:
            # k is non-negative
            shift_idx = k
            # h_{i+k} * (1 - h_i) for i = 0, ..., n-1-k
            # Grad w.r.t. h_i: -h_{i+k} if i+k < n
            # Grad w.r.t. h_{i+k}: (1 - h_i) if i+k < n
            if k == 0:
                contrib = -(h[:-shift_idx] + h[shift_idx:]) 
            else:
                contrib = np.zeros_like(h)
                contrib[:-shift_idx] = -(h[shift_idx:])
                contrib[shift_idx:] = (1 - h[:-shift_idx])
        else:
            # k is negative, let s = -k so that s > 0
            s = -k
            # h_{i+k} * (1 - h_i) = h_{i-s} * (1 - h_i) for i = s, ..., n-1
            # Grad w.r.t. h_i: -h_{i-s} if i >= s  
            # Grad w.r.t. h_{i-s}: (1 - h_i) if i-s >= 0, i < n
            if s == 0:
                contrib = -(h[:-shift_idx] + h[shift_idx:]) 
            else:
                contrib = np.zeros_like(h)
                contrib[s:] = -(h[:-s])   # h_{i-k} term
                contrib[:-s] = (1 - h[s:]) # (1-h_i) term
        grad += weights[np.abs(k)] * contrib
            
    return grad

def compute_all_correlations(h):
    """Compute all c_k values for the current h"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract only k from -(n-1) to n-1
    c_vals = corr[n-1:2*n-1]
    return c_vals

def adaptive_gradient_descent(seed, max_iter=4000, step_size=0.2, eps_start=1.0, eps_factor=1.5, eps_max=100.0):
    np.random.seed(seed)
    h = np.random.uniform(0, 1, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Optimisation parameters
    eps = eps_start
    alpha = step_size
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    
    # Momentum terms (like Adam)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    nan_count = 0
    iter_no_improvement = 0
    
    for t in range(max_iter):
        # Compute all overlap values
        c_vals = compute_all_correlations(h)
        
        # Check for nan in c_vals due to numerical issues
        if np.isnan(c_vals).any():
            nan_count += 1
            # Try to continue or restart
            if nan_count > 5:
                h = np.random.uniform(0, 1, n_points)
                h = project_to_sum_box(h, TARGET_SUM)
                continue
        else:
            nan_count = 0  

        # Compute gradient of smooth objective
        grad = compute_gradient_f_epsilon(h, c_vals, eps)
        
        # Update momentum
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**(t + 1))
        v_hat = v / (1 - beta2**(t + 1))
        
        # Compute new h (projected gradient step)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
        h = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate exact C5 every 100 iterations
        if t % 100 == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                iter_no_improvement = 0
            else:
                iter_no_improvement += 1
                
        # Anneal epsilon every 500 steps
        if t % 500 == 0 and t > 0:
            eps = min(eps * eps_factor, eps_max)
            alpha = alpha * 0.5  # Reduce step size
            
        # Early stopping
        if eps > 20 and iter_no_improvement > 200:
            break
            
    return best_h, best_c5
