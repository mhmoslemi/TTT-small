def evaluate_c5_exact(h):
    """Evaluate the exact C5 using FFT correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def round_solution_fractional(h_frac, target_sum, n_points):
    """Rounds a fractional solution to a binary vector with correct sum"""
    # Use conditional expectation (cond. prob) to ensure exact sum
    h_binary = np.zeros(n_points)
    remaining_ones = target_sum
    remaining_positions = n_points
    
    for i in range(n_points):
        if remaining_ones <= 0:
            break
        remaining_positions -= 1
        # If we still need ones, assign 1 if it's favorable (more likely)
        p = h_frac[i]
        
        # Check what happens if we set to 1
        if remaining_ones >= 1 and remaining_ones <= remaining_positions + 1:
            # In case of remaining ones, assign a random 1 or 0
            if np.random.rand() < p:
                h_binary[i] = 1
                remaining_ones -= 1
        else:
            if p >= (remaining_ones / remaining_positions):
                h_binary[i] = 1
                remaining_ones -= 1
            else:
                h_binary[i] = 0
    
    # Do any leftover assignments to meet sum exactly
    ones_count = np.sum(h_binary)
    if ones_count > target_sum:
        # Need to turn some 1's into 0's
        flip_positions = np.where(h_binary == 1)[0]
        for i in range(int(ones_count - target_sum)):
            pos_to_flip = np.random.choice(flip_positions)
            h_binary[pos_to_flip] = 0
            flip_positions = np.where(h_binary == 1)[0]
    elif ones_count < target_sum:
        # Need to turn some 0's into 1's
        flip_positions = np.where(h_binary == 0)[0]
        for i in range(int(target_sum - ones_count)):
            pos_to_flip = np.random.choice(flip_positions)
            h_binary[pos_to_flip] = 1
            flip_positions = np.where(h_binary == 0)[0]
    
    return h_binary

    for it in range(max_iters):

        max_shift = np.argmax(corr)
        max_c5 = corr[max_shift] * dx
        
        # Look for beneficial swaps
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        
        # Sample a few combinations
        for i in range(min(1000, len(ones) * len(zeros))):
            p1 = np.random.choice(ones)
            p2 = np.random.choice(zeros)
            
            # Make a swap
            temp_h = best_h.copy()
            temp_h[p1] = 0
            temp_h[p2] = 1
            
            # Evaluate new C5
            new_c5 = evaluate_c5_exact(temp_h)
            if new_c5 < best_c5:
                best_h = temp_h
                best_c5 = new_c5
                improved = True
                break
                
        if not improved:
            break
            
    return best_h, best_c5
