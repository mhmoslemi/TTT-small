import numpy as np
from scipy.signal import correlate
import cvxpy as cp
from cvxpy import Minimize, Variable, Problem, sum as cp_sum
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 150
    dx = 2.0 / n
    target_sum = n / 2
    
    # Generate initial binary solution as warm start
    h_init = np.zeros(n, dtype=int)
    h_init[:int(target_sum)] = 1
    
    # Build MILP model
    # Variables
    h = Variable(n, boolean=True)
    y = Variable((n, n), boolean=True)  # y[i,k] = h[i] AND h[i+k]
    t = Variable()  # epigraph variable
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp_sum(h) == target_sum)
    
    # Linearization constraints for y[i,k] = h[i] AND h[i+k]
    # We only consider k >= 0 to avoid duplication due to symmetry
    for k in range(n):
        for i in range(n - k):
            # y[i,k] <= h[i]
            constraints.append(y[i, k] <= h[i])
            # y[i,k] <= h[i+k]
            constraints.append(y[i, k] <= h[i + k])
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y[i, k] >= h[i] + h[i + k] - 1)
    
    # Compute correlation terms
    # For correlation at shift k, we have sum_{i=0}^{n-k-1} h[i]*(1 - h[i+k])
    # = sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} y[i,k]
    # The objective is to minimize the max of these terms over all k
    
    # Add t >= c_k for all k
    # c_k = sum(h_i) - sum(y_{i,k}) for i=0 to n-k-1
    for k in range(n):
        # Compute sum(h_i) over valid indices
        sum_h_k = cp_sum(h[:n-k])
        # Compute sum(y_{i,k}) over valid indices
        sum_y_k = cp_sum(y[:n-k, k])
        c_k = sum_h_k - sum_y_k
        constraints.append(t >= c_k)
    
    # Objective
    objective = Minimize(t)
    
    # Solve MILP with CBC solver
    prob = Problem(objective, constraints)
    prob.solve(solver=cp.CBC, time_limit=budget_s/2, log_freq=100)
    
    # Default to initial solution if no good solution found
    best_h = h_init
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_sol = h.value.astype(int)
        c5_val = np.max(np.correlate(h_sol, 1 - h_sol, mode="full")) * dx
        if c5_val < best_c5:
            best_h = h_sol
            best_c5 = c5_val
            
    # If time remains, also solve LP relaxation and round
    if time.time() - start_time < budget_s - 5 and prob.status not in ['optimal', 'optimal_inaccurate']:
        try:
            h_relax = Variable(n, boolean=False)
            h_relax.value = np.clip(np.random.rand(n), 0, 1)
            y_relax = Variable((n, n), boolean=False)
            
            relax_cons = []
            relax_cons.append(cp_sum(h_relax) == target_sum)
            # y[i,k] <= h[i]
            for k in range(n):
                for i in range(n - k):
                    relax_cons.append(y_relax[i, k] <= h_relax[i])
                    relax_cons.append(y_relax[i, k] <= h_relax[i + k])
                    relax_cons.append(y_relax[i, k] >= h_relax[i] + h_relax[i + k] - 1)
            
            relax_arr = []
            for k in range(n):
                sum_h_k = cp_sum(h_relax[:n-k])
                sum_y_k = cp_sum(y_relax[:n-k, k])
                c_k = sum_h_k - sum_y_k
                relax_arr.append(c_k)
            
            relax_objective = Minimize(cp_max(relax_arr))
            relax_prob = Problem(relax_objective, relax_cons)
            relax_prob.solve(solver=cp.GLPK, verbose=False)
            
            if relax_prob.status in ['optimal', 'optimal_inaccurate']:
                h_frac = h_relax.value
                sorted_indices = np.argsort(h_frac)[::-1]
                h_round = np.zeros(n)
                h_round[sorted_indices[:int(target_sum)]] = 1
                
                c5_round = np.max(np.correlate(h_round, 1 - h_round, mode="full")) * dx
                if c5_round < best_c5:
                    best_h = h_round
                    best_c5 = c5_round
        except:
            pass
    
    # Local polishing with pair-swaps
    def compute_correlations(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return corr * dx
    
    def update_correlations(old_corr, h_orig, i, j):
        """Update correlations to reflect a swap between indices i and j"""
        corr = old_corr.copy()
        # Create new corr based on swapping positions i and j
        new_h = h_orig.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return compute_correlations(new_h)
    
    if time.time() - start_time < budget_s - 1:
        h_local = best_h.copy()
        corr_old = compute_correlations(h_local)
        best_corr = np.max(corr_old)
        
        for _ in range(100000):
            if time.time() - start_time > budget_s - 1:
                break
                
            change = False
            # Try swapping 1 with 0
            ones = np.where(h_local == 1)[0]
            zeros = np.where(h_local == 0)[0]
            
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                h_new = h_local.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]
                corr_new = compute_correlations(h_new)
                max_new = np.max(corr_new)
                
                if max_new < best_corr:
                    h_local = h_new
                    best_corr = max_new
                    change = True
                    corr_old = corr_new
            
            if not change:
                break
        
        final_c5 = best_corr
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_local
    
    return (best_h.tolist(), float(best_c5), n)
