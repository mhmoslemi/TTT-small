import numpy as np
from scipy.optimize import minimize
import time

# Precomputed constants
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def create_shift_matrices(n):
    """Create all shift matrices for shifts 0 to n-1"""
    M = []
    for k in range(n):
        M_k = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                M_k[i, i + k] = 1
                M_k[i + k, i] = 1  # symmetric
        M.append(M_k)
    return M

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * dx

def project_to_sum_and_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
        
    # Use binary search to find correct lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    x = np.clip(x + (a + b)/2, 0, 1)
    return x

def solve_sdp_relaxation(M_matrices, target_sum, alpha_guess=None):
    """Solve SDP relaxation for given alpha"""
    n = len(M_matrices[0])
    # Variable for lifted matrix
    X = Semidefinite(n)
    h = Variable(n)
    
    # Constraints
    constraints = [
        trace(X) == target_sum,  # sum of diagonal entries
        h >= 0, h <= 1,          # box constraints
        X >> 0,                  # positive semidefinite
    ]
    
    # Add McCormick envelopes for bounds
    for i in range(n):
        constraints.append(X[i, i] <= h[i])
        
    # Add overlap inequalities for each shift
    if alpha_guess is not None:
        for k, M_k in enumerate(M_matrices):
            if k == 0:  # skip for symmetric case at k=0
                continue
            # Add constraint for shift k
            overlap_constraint = trace(M_k @ X) >= target_sum - alpha_guess
            constraints.append(overlap_constraint)
    
    # Minimize sum (for feasible point)
    objective = Minimize(sum(h))
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, verbose=False)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return h.value, X.value
        else:
            return None, None
    except:
        return None, None

def random_round_and_refine(h_sol, X_sol, target_sum, M_matrices):
    """Round SDP solution to a vector and refine"""
    # Get candidate from random rounding
    if X_sol is not None:
        # Perform spectral decomposition
        eigenvals, eigenvecs = np.linalg.eigh(X_sol)
        # Use only significant components
        idx = np.argsort(eigenvals)[::-1]
        eigenvals = eigenvals[idx]
        eigenvecs = eigenvecs[:, idx]
        
        # Use rank-1 approximation
        h_candidate = eigenvecs[:, 0] * np.sqrt(eigenvals[0])
        
        # Clip and project again
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = project_to_sum_and_box(h_candidate, target_sum)
        
        # Refine with gradient descent step
        def loss(h):
            return -np.min([h.T @ M_k @ h for M_k in M_matrices[1:]])
        
        def grad(h):
            g = np.zeros_like(h)
            for M_k in M_matrices[1:]:
                g += 2 * M_k @ h
            return g
        
        # Simple gradient descent
        step_size = 0.005
        for _ in range(20):
            g = grad(h_candidate)
            h_candidate = h_candidate - step_size * g
            h_candidate = project_to_sum_and_box(h_candidate, target_sum)
            
        return h_candidate
    else:
        # Fallback to uniform distribution
        return np.full(n_points, 0.5)

def run(seed=42, budget_s=600, **kwargs):

    lower_alpha = 0.0
    upper_alpha = 1.0
    best_alpha = 1.0
    
    # Collect results
    best_result = None
    best_c5_val = float('inf')
    
    # Bisection to find min alpha such that SDP is feasible
    for iteration in range(50):  # Limited iterations to keep budget
        if time.time() - start_time > budget_s - 5:
            break
            
        alpha_guess = (lower_alpha + upper_alpha) / 2
        
        # Try SDP relaxation
        try:
            h_sol, X_sol = solve_sdp_relaxation(M_matrices, TARGET_SUM, alpha_guess)
            if h_sol is not None:
                # Validate by rounding
                h_rounded = random_round_and_refine(h_sol, X_sol, TARGET_SUM, M_matrices)
                if h_rounded is not None:

                lower_alpha = alpha_guess
            else:
                # Infeasible, tighten upper bound
                upper_alpha = alpha_guess
