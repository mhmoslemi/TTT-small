import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_individual(individual):
    """Evaluate an individual using FFT-based correlation"""
    h = np.array(individual)
    return evaluate_c5(h)

def initialize_binary_individual():
    """Create a binary initial solution with exact half 1s"""
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, n_points//2, replace=False)
    h[ones_indices] = 1
    return h.tolist()

def crossover(parent1, parent2):
    """Perform uniform crossover and ensure sum consistency"""
    child = [1 if np.random.random() < 0.5 else 0 for _ in range(n_points)]
    # Adjust to preserve sum constraint
    current_sum = sum(child)
    target_sum = n_points // 2
    if current_sum > target_sum:
        # Remove excess 1s
        indices = [i for i in range(n_points) if child[i] == 1]
        remove_indices = np.random.choice(indices, current_sum - target_sum, replace=False)
        for i in remove_indices:
            child[i] = 0
    elif current_sum < target_sum:
        # Add missing 1s
        indices = [i for i in range(n_points) if child[i] == 0]
        add_indices = np.random.choice(indices, target_sum - current_sum, replace=False)
        for i in add_indices:
            child[i] = 1
    return child

def mutate(individual):
    """Swap a 0 and a 1 bit"""
    indices_of_0s = [i for i in range(n_points) if individual[i] == 0]
    indices_of_1s = [i for i in range(n_points) if individual[i] == 1]
    if indices_of_0s and indices_of_1s:
        i = np.random.choice(indices_of_0s)
        j = np.random.choice(indices_of_1s)
        child = individual[:]
        child[i], child[j] = child[j], child[i]
        return child
    return individual

def genetic_algorithm(budget_s, start_time):
    """Run a simple genetic algorithm to obtain a good binary solution"""
    population_size = 60
    generations = 0
    max_generations = 500
    
    # Initialize population with binary individuals
    population = [initialize_binary_individual() for _ in range(population_size)]
    
    best_c5 = float('inf')
    best_individual = None
    
    while time.time() - start_time < budget_s * 0.7 and generations < max_generations:
        # Evaluate population
        fitnesses = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_individual = population[best_idx][:]
            
        # Selection via tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, 3, replace=False)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner][:])
        
        # Create next generation
        new_population = [best_individual[:]]  # Elitism
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        generations += 1
    
    return best_individual, best_c5

def simulated_annealing(initial_solution, budget_s, start_time):
    """Run simulated annealing to refine continuous solution"""
    current_h = np.array(initial_solution)
    current_c5 = evaluate_c5(current_h)
    best_c5 = current_c5
    best_h = current_h.copy()
    
    T = 1.0
    Tmin = 1e-6
    alpha = 0.95
    
    while time.time() - start_time < budget_s * 0.9 and T > Tmin:
        # Generate a move: swap a 0 and a 1 bit
        indices_of_0s = [i for i in range(n_points) if current_h[i] < 0.5]
        indices_of_1s = [i for i in range(n_points) if current_h[i] > 0.5]
        if not indices_of_0s or not indices_of_1s:
            break
        i = np.random.choice(indices_of_0s)
        j = np.random.choice(indices_of_1s)
        
        # Try move
        h_test = current_h.copy()
        h_test[i], h_test[j] = h_test[j], h_test[i]
        
        # Project to satisfy sum constraint
        h_test = project_to_sum_box(h_test, n_points / 2)
        test_c5 = evaluate_c5(h_test)
        
        # Accept or reject the move
        delta = test_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_h = h_test
            current_c5 = test_c5
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = current_h.copy()
        
        T *= alpha
    
    return best_h, best_c5

def smoothed_gradient_descent(initial_h, budget_s, start_time):
    """Run gradient ascent with log-sum-exp smoothing to improve solution"""
    h = initial_h.copy()
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    # Smoothed objective using log-sum-exp
    def smoothed_objective(h, alpha):
        corr = compute_corr_fft(h)
        return -np.log(np.sum(np.exp(-alpha * corr))) / alpha * dx
    
    # Gradient calculation with log-sum-exp smoothing
    def gradient_logsumexp(h, alpha):
        corr = compute_corr_fft(h)
        exp_vals = np.exp(-alpha * corr)
        Z = np.sum(exp_vals)
        grad = np.zeros_like(h)
        # We compute gradient by convolution - simplify computation with finite differences for now
        step = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += step
            h_minus = h.copy()
            h_minus[i] -= step
            val_plus = smoothed_objective(h_plus, alpha)
            val_minus = smoothed_objective(h_minus, alpha)
            grad[i] = (val_plus - val_minus) / (2 * step)
        return grad
    
    # Use learning rate decay schedule
    lr = 1e-3
    decay_rate = 0.99
    max_iterations = 1000
    
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
            
        # Increase smoothing slowly
        alpha = 10.0 ** (iteration // 100)  # Starting small to fast converge, then increase
        
        try:
            grad = gradient_logsumexp(h, alpha)
            h += lr * grad
            h = np.clip(h, 0, 1)
            # Project to satisfy sum constraint
            h = project_to_sum_box(h, n_points / 2)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= decay_rate
        except:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
