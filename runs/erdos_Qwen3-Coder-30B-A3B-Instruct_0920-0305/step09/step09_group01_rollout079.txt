import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

# Global constants
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
n_shifts = 2 * n_points - 1  # from -n+1 to n-1

def build_shift_matrices(n, max_shift=None):
    """Build all shift matrices M_ell for ell = 0 to max_shift"""
    if max_shift is None:
        max_shift = n - 1
    
    M_list = []
    for ell in range(max_shift + 1):
        M = np.zeros((n, n))
        for i in range(n - ell):
            M[i, i + ell] = 1
            M[i + ell, i] = 1  # symmetric
        M_list.append(M)
    return M_list

def sdp_relaxation(h_guess, alpha_guess, M_list):
    """Solve SDP relaxation for given alpha to check feasibility"""
    n = len(h_guess)
    
    # Decision variables
    X = cvx.Variable((n, n), symmetric=True)
    h = cvx.Variable(n)
    
    # Constraints
    constraints = [
        X == h @ h.T,  # rank constraint
        X >> 0,        # positive semidefinite
        cvx.sum(h) == TARGET_SUM,
        h >= 0, h <= 1,  # box constraints
    ]
    
    # Add McCormick envelopes for the rank-1 constraint X = h h^T
    for i in range(n):
        constraints.append(X[i, i] == h[i])
    
    # Add overlap constraints - this is actually not needed in relaxation but use for verification
    for ell in range(len(M_list)):
        constraints.append(cvx.trace(M_list[ell] @ X) >= TARGET_SUM/2 - alpha_guess)
    
    # The SDP feasibility problem
    prob = cvx.Problem(cvx.Minimize(0), constraints)
    
    try:
        prob.solve(solver=SCS, eps=1e-4, max_iters=5000)
        return prob.status == "optimal"
    except Exception as e:
        return False

def compute_c5(h, M_list):
    """Compute C5 value given h and shift matrices"""
    values = []
    for M in M_list:
        val = TARGET_SUM/2 - h @ M @ h
        values.append(val)
    return dx * max(values)

def round_rank_1(X):
    """Randomized rounding from SDP solution"""
    try:
        U, s, V = np.linalg.svd(X)
        sqrt_s = np.sqrt(np.maximum(s, 0))
        v = U @ np.diag(sqrt_s)
        
        # Generate random sample from normal distribution
        z = np.random.randn(len(v))
        h = np.clip(v @ z, 0, 1)
        h = h / np.sum(h) * TARGET_SUM
        return h
    except:
        # Fallback to simple rounding
        h = np.diag(X)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * TARGET_SUM
        return h

    return np.max(np.correlate(h, 1-h, mode="full")) * dx
