import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - np.array(h)
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Use the initial construction as starting point
    initial_h = np.array(initial_h_values)
    
    # Project to feasible region
    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum

        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1

        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    current_h = project_to_sum_box(initial_h, TARGET_SUM)
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    
    # Begin optimization using a modified local search with large steps
    step_limit = int(2e5)
    for step in range(step_limit):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor: random perturbation of several elements
        neighbor_h = current_h.copy()
        num_changes = max(1, min(20, n_points // 10))
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            neighbor_h[idx] += delta
            
        # Project back to constraints
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
        else:
            # Metropolis acceptance with temperature that decays over time
            temp = max(1e-6, 0.1 * (1 - step / step_limit))
            delta_e = neighbor_c5 - current_c5
            if np.random.random() < np.exp(-delta_e / temp):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5

    # Perform final improvement with local descent
    if time.time() - start_time < budget_s - 5:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Use a few iterations of gradient-based optimization
            h_refine = best_h.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Very simple gradient approx
                grad = np.zeros_like(h_refine)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[i] += eps
                    h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                    h_minus = h_refine.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                    grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
                
                # Update step
                step_size = 0.01
                h_refine -= step_size * grad
                h_refine = project_to_sum_box(h_refine, TARGET_SUM)
                
                new_c5 = evaluate_c5(h_refine)
                if new_c5 < best_c5:
                    best_h = h_refine.copy()
                    best_c5 = new_c5
                    
        except Exception:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
