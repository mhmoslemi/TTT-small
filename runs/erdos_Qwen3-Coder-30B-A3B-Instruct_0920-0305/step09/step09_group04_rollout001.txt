def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, eps):
    """Compute smooth max correlation with epsilon regularization"""
    corr = compute_corr_fft(h)
    # Normalize to prevent overflow
    max_corr = np.max(corr)
    normalized_corr = eps * (corr - max_corr)
    exp_corr = np.exp(normalized_corr)
    w = exp_corr / np.sum(exp_corr)
    smooth_max = max_corr + np.log(np.sum(exp_corr)) / eps
    return smooth_max, w

    # Compute gradient of each correlation using convolution
    grad = np.zeros_like(h)
    
    # For each lag k (from -n+1 to n-1)

        # Compute gradient contribution for shift k
        if k >= 0:
            # Positive shift: corr[k] = sum h[i+k] * (1-h[i]) for i from 0 to n-1-k
            # grad_wrt_h[j] = (1 - h[j-k]) * (j >= k) - h[j+k] * (j <= n-1-k)
            grad_term = np.zeros_like(h)
            if k < len(h):
                grad_term[k:] = 1 - h[:-k] if k > 0 else 1 - h
            if k < len(h):
                grad_term[:-k] -= h[k:] if k > 0 else h
            # Note: the following assumes only positive shifts in w indices
        else:
            # Negative shift: corr[k] = sum h[i] * (1-h[i-k]) for i from 0 to n-1+k
            # grad_wrt_h[j] = -h[j-k] * (j >= k) + (1-h[j+k]) * (j <= n-1+k) 
            grad_term = np.zeros_like(h)
            if k < 0:
                # For negative shift (k < 0) gradient from h[i] with 1-h[i+k]
                grad_term[:-k] = -h[(-k):] if -k > 0 else -h
                if k < 0 and -k < len(h):
                    grad_term[(-k):] += 1 - h[:-k] if -k > 0 else 1 - h
                    
        grad += w[k + n_points - 1] * grad_term
    
    return grad
