import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_algorithm(seed, budget_s):
    """Run genetic algorithm for binary solution"""
    np.random.seed(seed)
    start_time = time.time()
    
    population_size = 60
    generations = 200
    elite_count = 5
    
    # Initialize population with binary vectors
    population = []
    for _ in range(population_size):
        # Create binary vector with exactly half ones
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, n_points//2, replace=False)
        h[ones_indices] = 1
        population.append(h)
    
    best_fitness = float('inf')
    best_individual = None
    
    # Evaluation of initial population
    fitnesses = []
    for individual in population:
        fitness = evaluate_c5(individual)
        fitnesses.append(fitness)
        if fitness < best_fitness:
            best_fitness = fitness
            best_individual = individual.copy()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Uniform crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.zeros(n_points)
            child[:crossover_point] = parent1[:crossover_point]
            child[crossover_point:] = parent2[crossover_point:]
            
            # Repair to ensure exactly half ones
            ones_count = np.sum(child)
            if ones_count > n_points // 2:
                # Turn some 1s into 0s
                ones_indices = np.where(child == 1)[0]
                to_flip = int(ones_count - n_points // 2)
                flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
                child[flip_indices] = 0
            elif ones_count < n_points // 2:
                # Turn some 0s into 1s
                zeros_indices = np.where(child == 0)[0]
                to_flip = int(n_points // 2 - ones_count)
                flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
                child[flip_indices] = 1
            
            # Mutation
            if np.random.random() < 0.1:
                idx1 = np.random.randint(0, n_points)
                idx2 = np.random.randint(0, n_points)
                child[idx1], child[idx2] = child[idx2], child[idx1]
            
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        fitnesses = []
        for individual in population:
            fitness = evaluate_c5(individual)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = individual.copy()
    
    return best_individual, best_fitness

def simulated_annealing(h_start, budget_s):
    """Run simulated annealing to refine continuous solution"""
    start_time = time.time()
    
    current_h = h_start.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Find initial temperature (based on typical change in objective)
    temp = 0.1
    max_temp = 1.0
    
    # Number of moves per temperature
    moves_per_temp = 100
    
    # Temperature decay schedule
    alpha = 0.95
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    while time.time() - start_time < budget_s:
        for _ in range(moves_per_temp):
            if time.time() - start_time > budget_s:
                break
                
            # Make a random move: swap two elements
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            
            # Create new solution with swapped indices
            h_test = current_h.copy()
            h_test[idx1], h_test[idx2] = h_test[idx2], h_test[idx1]
            
            # Project to valid region
            h_test = project_to_sum_box(h_test, n_points / 2)
            
            test_c5 = evaluate_c5(h_test)
            
            # Accept or reject
            delta = test_c5 - current_c5
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                current_h = h_test
                current_c5 = test_c5
                
                if test_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = test_c5
        
        # Decrease temperature
        temp *= alpha
        
        # Stop if temperature is very small
        if temp < 1e-10:
            break
    
    return best_h, best_c5

def smooth_gradient_descent(h_start, budget_s):

    start_time = time.time()
    
    # Start with the best value from previous steps
    h = h_start.copy()

        # We optimize -c5 to maximize the minimum correlation
        # Smooth version of min_k Σ_i h_i h_{i+k}
        # Using log-sum-exp to approximate min_k
        def smooth_min_over_k():
            def compute_corr_components(h, shift):
                sum_terms = 0
                for i in range(len(h)):
                    if i + shift < len(h):
                        sum_terms += h[i] * h[i + shift]
                return sum_terms
            
            # Compute the terms being minimized
            terms = []
            for k in range(1, len(h)):
                term = compute_corr_components(h, k)
                terms.append(term)

def run(seed=42, budget_s=600, **kwargs):
