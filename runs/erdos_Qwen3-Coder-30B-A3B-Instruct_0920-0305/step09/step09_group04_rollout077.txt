import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import ECOS, GLPK
import random

def evaluate_c5_exact(h):
    """Evaluate the exact C5 value using full correlation."""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
