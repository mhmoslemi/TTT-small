import numpy as np

TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):

    corr = np.correlate(h, 1 - h, mode="full")
    
    # Scale by dx to match the integral normalization
    corr = corr * dx
    
    # Compute weighted sum using log-sum-exp for smooth approximation
    # Numerically stable computation
    scaled_corr = epsilon * corr
    max_corr = np.max(scaled_corr)
    exp_corr = np.exp(scaled_corr - max_corr)
    sum_exp = np.sum(exp_corr)
    
    # Log-sum-exp approximation of max
    smooth_max = (max_corr + np.log(sum_exp)) / epsilon
    
    # Compute gradient of log-sum-exp with respect to h
    weights = exp_corr / sum_exp
    
    # The gradient computation for smooth max correlation
    grad = np.zeros_like(h)
    
    # For each shift k, compute partial gradient contribution
    n = len(h)
    # Create shifted versions for gradients
    for k in range(-n+1, n):
        # Compute contribution of shift k to gradient
        if k >= 0:
            # Forward shift
            shift_term = np.concatenate([np.zeros(k), h[:n-k]])
        else:
            # Backward shift  
            shift_term = np.concatenate([h[-k:], np.zeros(-k)])
        
        # Gradient contribution: (1 - h) shifted by k minus h shifted by -k
        # But we only consider valid indices for correlation computation 
        if k >= 0:
            # Correlation is sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i) 
            # So for gradient with respect to h_j we compute:
            # if j - k >= 0: term from h_{j-k} is (1 - h_j) -> contributes (1 - h_j) 
            # if j + k < n: term from h_j is -h_{j+k} -> contributes -h_{j+k}
            # Need to trace this carefully with indexing
            gradient_term_1 = np.zeros(n)
            gradient_term_2 = np.zeros(n)
            
            # Term from h_{i+k} times (1 - h_i) -> d/dh_j( h_{j-k} * (1 - h_j) ) = (1 - h_j) * δ_{j=k+i} = (1 - h_{j-k})
            # With proper slicing
            # Let's build df/dh_j = -h_{j+k} if j+k<n, 0 otherwise (d/dh_{j+k})  
            # plus (1 - h_{j-k}) if j-k >= 0, 0 otherwise (d/dh_{j-k})
            
            # Implement with explicit loops for clarity
            for i in range(n):
                if i + k < n:
                    if i+k >= 0:
                        gradient_term_1[i+k] += (1 - h[i])
                    if i < n:
                        gradient_term_2[i] -= h[i+k]
                else:
                    break
            
        else:
            # k < 0, we compute for shift -k
            k_abs = -k
            gradient_term_1 = np.zeros(n)
            gradient_term_2 = np.zeros(n)
            for i in range(n):
                if i < n-k_abs:
                    if i+k_abs >= 0: 
                        gradient_term_1[i] -= h[i+k_abs]
                    if i < n:
                        gradient_term_2[i+k_abs] += (1 - h[i])
            
        # Add weighted term to total gradient, but only the correlation range
        # This is approximate - for padding, align properly
        if k >= 0: 
            # Get only valid region: sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i) 
            # gradient: for j in 0..n-1:
            #     - Σ_{i=j-k}^{n-1-k}, if i=k+j then add -(1 - h_{j-k}) 
            #     - Σ_{i=0}^{n-1-k+j}, if i=j+k then add (1 - h_i)
            
            # Simpler approach: directly compute based on indices
            valid_indices = slice(0, n-k if k < n else 0) if k >= 0 else slice(-k, n)
            
            # We'll accumulate it in the hard-and-fast way
            # Can we do better with vectorization?
            for j in range(n):
                grad_j = 0
                # At shift k, we have i from 0 to n-1-k, term h_{i+k}(1-h_i)
                # For d/dh_j, if j = i then we get - h_{i+k} 
                # If j = i+k then we get (1 - h_i)  
                if k >= 0:
                    # i from 0 to n-1-k 
                    # Add (1 - h_j) for i = j-k such that j-k >= 0 and j-k < n-1-k, i.e., 0 <= j <= n-1
                    # That means j >= k and j <= n-1 So j from max(k, 0) to min(n-1, n-1-k)
                    # If j = i+k => i = j-k, so start at max(0, j-k), end at min(n-1-k, n-1).
                    # Actually, from 0 to n-1-k indices, if i = j-k, for i to be valid, 0 <= j-k <= n-1-k
                    # So j-k >= 0 and j-k <= n-1-k
                    # which means j >= k and j <= n-1
                    
                    # For j, we consider where i+j-k = j-k, so i = j-k, and if i \in [0, n-1-k]
                    # Then j must be in [k, n-1]
                    if 0 <= j <= n-1-k: # j can't exceed n-1-k for valid i = j-k    
                        grad_j += (1 - h[j-k])
                    
                    # Similarly for i+k = j, i = j-k and for i \in [0, n-1], i = j-k \in [0,n-1] 
                    # So j-k \in [0, n-1], so j \in [k, n]
                    if k <= j < n:
                        grad_j -= h[j+k] 
                else: # k negative
                    # i from 0 to n-1+k, term h_i * (1 - h_{i+k})
                    # Derivative wrt h_j: 
                    # if j = i then -h_{i+k} = -h_{j+k}
                    # if j = i+k then (1 - h_i) = (1 - h_{j-k}) 
                    # So j from 0 to n-1    
                    if 0 <= j+k < n:
                        grad_j -= h[j+k] 
                    if 0 <= j-k < n:
                        grad_j += (1 - h[j-k])
                    
                grad[j] += weights[k + (n-1)] * grad_j
                
    return smooth_max, grad
