import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def exact_C5(h):
    """Compute the exact C5 using full correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def smoothed_objective(h, beta):
    """Compute the smooth approximation Φ_β(h)"""
    f = np.correlate(h, 1 - h, mode='full') * dx
    f_max = np.max(f)
    if np.isinf(f_max) or np.isnan(f_max):
        return float('inf')
    wgt = np.exp(beta * (f - f_max))
    return (1.0 / beta) * np.log(np.sum(wgt))

def compute_weighted_gradient(h, wgt):
    """Compute gradient of Φ_β(h) with given weights"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, compute gradient contribution
    for k in range(-n + 1, n):
        if k >= 0:
            # h[i] contributes to f_k where i+k < n
            # gradient wrt h[i]: -h[i+k] - h[i-k] + 1
            left_slice = np.zeros(n)
            right_slice = np.zeros(n)
            if k < n:
                left_slice[k:] = h[:-k]  # h[i-k]
            if k >= 0:
                right_slice = h[k:]  # h[i+k]
        else:
            # k < 0
            # h[i] contributes to f_k where i-k < n, i+k >= 0
            left_slice = np.zeros(n)
            right_slice = np.zeros(n)
            if k >= -n + 1:
                left_slice[:k] = h[-k:]  # h[i-k]
            if k < 0:
                right_slice = h[:n+k]  # h[i+k]
        
        # contribution to gradient for each element
        grad_contrib = -right_slice - left_slice + 1.0
        grad += wgt[k + n - 1] * grad_contrib
    
    return grad

def project_onto_box_and_sum(v, w):
    """Project vector v onto [0,1]^n with sum = w"""
    v = np.clip(v, 0, 1)
    
    # Find lambda to make sum equal to w

        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - w
