def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def conv_concave_programming(h0, max_iter=50):
    """Use convex-concave programming to solve the optimization"""
    n = len(h0)
    dx = 2.0 / n
    
    # Initialize
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Convex-concave programming approach
    for iteration in range(max_iter):
        if iteration % 10 == 0:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Find the maximum correlation index
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        # For simplicity, we'll approximate this with local gradient-based step
        # with adjustment based on the maximum correlation constraint
        grad = np.zeros(n)
        for k in range(n):
            # Approximate gradient for correlation term
            if k < n:
                grad[k] += (h[k] - h[(k + max_idx) % n]) * dx
        
        # Gradient step with adaptive step size
        step_size = 0.01 / (1 + iteration / 10)
        h_new = h - step_size * grad
        
        # Project back to feasible region
        h_new = project_to_sum_box(h_new, n / 2)
        
        # Local search around the new point
        current_c5 = evaluate_c5(h_new)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
        h = h_new
    
    return best_h, best_c5
