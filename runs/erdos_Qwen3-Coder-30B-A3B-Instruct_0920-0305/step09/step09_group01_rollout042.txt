import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_all_shifts(h):
    """Compute all c_k(h) values using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_C5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def compute_gradient_for_shifts(h, shifts):
    """Compute gradient for the soft max of all shifts"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, compute its contribution to the gradient
    max_k = len(shifts) - 1
    for k in range(-n+1, n):
        if k >= 0:
            # Positive shift case
            if k < n:
                grad[:n-k] += 1 - h[k:]
                grad[k:] -= h[:n-k]
        else:
            # Negative shift case
            k_abs = -k
            if k_abs < n: 
                grad[k_abs:] += 1 - h[:n-k_abs]
                grad[:k_abs] -= h[k_abs:]
    
    return grad

def compute_soft_max_grad(h, mu):
    """Compute gradient of soft max approximation to C5"""
    shifts = compute_all_shifts(h)
    
    # Stable computation of softmax weights
    shifts_max = np.max(shifts)
    if shifts_max > 0:
        exp_shifts = np.exp((shifts - shifts_max) / mu)
    else:
        exp_shifts = np.exp(shifts / mu)
    weights = exp_shifts / np.sum(exp_shifts)
    
    # Compute gradient using weights
    grad = np.zeros(len(h))
    n = len(h)
    
    # For each shift k, add weighted contribution to gradient
    for k in range(-n+1, n):
        if k >= 0:
            # Positive shift case: indices i, i+k
            if k < n:
                grad[:n-k] += weights[k+n-1] * (1 - h[k:])
                grad[k:] -= weights[k+n-1] * h[:n-k]
        else:
            # Negative shift case
            k_abs = -k
            if k_abs < n:
                grad[k_abs:] += weights[k+n-1] * (1 - h[:n-k_abs])
                grad[:k_abs] -= weights[k+n-1] * h[k_abs:]
    
    return grad, shifts

def line_search_step(h, grad, mu, max_iter=5):
    """Perform line search to find a good step size"""
    alpha = 0.2 * dx
    h_current = h.copy()
    c5_current = compute_C5_from_corr(compute_all_shifts(h_current))
    
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = compute_C5_from_corr(compute_all_shifts(h_new))
        
        if c5_new < c5_current:
            return h_new, c5_new, alpha
        alpha *= 0.5
        
    return h_current, c5_current, alpha
