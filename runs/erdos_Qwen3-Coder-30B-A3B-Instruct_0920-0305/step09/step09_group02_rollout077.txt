import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_violations(h, T, n_points):
    """Compute violations of the constraint T >= w - sum h_i * h_{i+k}"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n_points - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    dx = 2.0 / n_points
    violations = []
    
    # Only consider relevant shifts
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            overlap = corr[n_points - 1 + k] * dx  # sum h_i * (1 - h_{i+k}) 
        else:
            overlap = corr[n_points - 1 + k] * dx
            
        # Check if constraint T >= w - overlap is violated
        # That is, if overlap > w - T where w = n/2
        w = n_points / 2.0
        if overlap > w - T:
            violations.append((k, overlap))
            
    return violations

    h = cp.Variable(n_points, boolean=True)
    T = cp.Variable()

    constraints = [cp.sum(h) == TARGET_SUM]

    y = {}

                constraints.append(y[(i,k)] <= h[i])
                constraints.append(y[(i,k)] <= h[i+k])
