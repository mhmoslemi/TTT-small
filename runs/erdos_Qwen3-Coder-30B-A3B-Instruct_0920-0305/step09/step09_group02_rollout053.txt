import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_sa(current_h, current_c5, start_time, budget_s):
    """Simulated Annealing optimization"""
    temp_start = 0.1
    temp_end = 0.0001
    max_steps = int(5e4)
    cooling_rate = (temp_end / temp_start) ** (1.0 / max_steps)
    
    temp = temp_start
    step = 0
    best_h = current_h.copy()
    best_c5 = current_c5
    
    while step < max_steps and time.time() - start_time < budget_s:
        neighbor_h = current_h.copy()
        num_changes = max(1, min(15, len(current_h) // 15))
        indices = np.random.choice(len(current_h), size=num_changes, replace=False)
        
        for idx in indices:
            delta = np.random.normal(0, 0.03)
            neighbor_h[idx] += delta
            
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
        else:
            delta_e = neighbor_c5 - current_c5
            if np.random.random() < np.exp(-delta_e / temp):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
                
        temp *= cooling_rate
        step += 1
    
    return best_h, best_c5

def optimize_ga(start_time, budget_s):

    population_size = 30
    generations = 50
    mutation_rate = 0.05

    population = []

            best_h = population[min_fitness_idx].copy()
            
        # Create new population
        new_population = []
        # Keep best
        top_indices = np.argsort(fitness)[:population_size//2]
        for idx in top_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(top_indices)
            parent2_idx = np.random.choice(top_indices)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.02)
                    
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
            
        population = new_population[:population_size]

    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}

            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}

def optimize_local_search(current_h, current_c5, start_time, budget_s):

    best_h = current_h.copy()

    # Try different step sizes for perturbation

    for step_size in step_sizes:

            break
            
        h_temp = best_h.copy()
        old_c5 = best_c5
