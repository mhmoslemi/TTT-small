import numpy as np

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_single_shift_gradient(h, k):
    """Compute gradient for a single shift k"""
    grad = np.zeros_like(h)

    else:
        k = -k
        if k < len(h):
            grad[k:] += 1 - h[:len(h)-k]
            grad[:len(h)-k] -= h[k:]
    return grad

def compute_all_shift_gradients(h):

    return gradients

def smooth_max_surrogate(h, mu=0.05):
    """Compute the smooth max approximation"""
    corr = compute_corr_fft(h)
    # Compute the weights
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_shifted = np.exp(shifted / mu)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Compute gradient of smooth max
    grad = np.zeros_like(h)
    for i, weight in enumerate(weights):
        k = i - len(h) + 1
        grad += weight * compute_single_shift_gradient(h, k)
        
    return grad

def line_search(h, grad, alpha0=0.2*dx):
    """Perform backtracking line search"""
    alpha = alpha0
    h_current = h.copy()
    for _ in range(5):
        h_new = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        c5_current = evaluate_c5(h_current)
        if c5_new < c5_current:
            return h_new, c5_new, alpha
        alpha *= 0.5
    return h_current, evaluate_c5(h_current), alpha
