import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_delta_corr(h, i, j):
    """Compute change in correlation when swapping h[i] and h[j]"""
    delta_corr = np.zeros(2 * n_points - 1)
    for k in range(2 * n_points - 1):
        if k < n_points:
            # Forward shifts
            if i + k < n_points:
                delta_corr[k] -= h[i] * h[i + k] + (1 - h[i]) * (1 - h[i + k]) - h[j] * h[j + k] - (1 - h[j]) * (1 - h[j + k])
            if j + k < n_points:
                delta_corr[k] -= h[j] * h[j + k] + (1 - h[j]) * (1 - h[j + k]) - h[i] * h[i + k] - (1 - h[i]) * (1 - h[i + k])
        else:
            # Backward shifts
            k_back = 2 * n_points - 2 - k
            if i - k_back >= 0:
                delta_corr[k] -= h[i] * h[i - k_back] + (1 - h[i]) * (1 - h[i - k_back]) - h[j] * h[j - k_back] - (1 - h[j]) * (1 - h[j - k_back])
            if j - k_back >= 0:
                delta_corr[k] -= h[j] * h[j - k_back] + (1 - h[j]) * (1 - h[j - k_back]) - h[i] * h[i - k_back] - (1 - h[i]) * (1 - h[i - k_back])
    return delta_corr

def initial_binary():
    """Create binary initial solution"""

    h[:n_points//2] = 1
    np.random.shuffle(h)

            # Selection (tournament)
            selected = []
            for _ in range(population_size):
                tournament_size = 3
                tournament_indices = random.sample(range(population_size), tournament_size)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[tournament_fitness.index(max(tournament_fitness))]

            new_population = []
            for i in range(0, population_size, 2):
                parent1 = selected[i]
                parent2 = selected[i+1] if i+1 < population_size else selected[0]
                
                # Uniform crossover
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)

                        child2[j] = parent2[j]

                        child2[j] = parent1[j]

                child1 = project_to_sum_box(child1, TARGET_SUM)
                child2 = project_to_sum_box(child2, TARGET_SUM)

                    new_population[i][flip_idx] = 1 - new_population[i][flip_idx]

            eps = 1e-6
