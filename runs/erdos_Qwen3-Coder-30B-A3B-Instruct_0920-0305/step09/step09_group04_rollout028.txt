    def evaluate_c5_ccp(h):
        # Each C_k = sum h_i - sum h_i * h_{i+k}

        corr_vals = np.correlate(h, 1-h, mode="full")
        return dx * np.max(corr_vals)

            return np.sum(projected) - target_sum

            b += 0.1

                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
