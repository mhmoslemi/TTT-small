import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    n = len(h)
    H = fft(h, 2*n-1)
    A = ifft(H * np.conj(H)).real[:n]
    return A

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient using two convolutions"""
    n = len(h)
    # Convolution of p and h (linear)
    r1 = np.convolve(p, h, mode='full')[:n]
    # Convolution of p_rev and h
    r2 = np.convolve(p[::-1], h, mode='full')[:n]
    return -(r1 + r2)

def project_onto_sum_box(x, target_sum):
    """Project onto [0,1] box constraints with fixed sum using water-filling"""
    x_clipped = np.clip(x, 0, 1)
    def sum_with_lambda(lam):
        return np.sum(np.clip(x_clipped + lam, 0, 1)) - target_sum
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(30):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < 1e-10:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x_clipped + (a + b)/2, 0, 1)
    return result

def c5_true(h):
    """Exact computation of C5 using numpy correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 128
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initialize h with a binary pattern (Thue-Morse-like)
    h = np.zeros(n)
    for i in range(n):
        h[i] = (bin(i).count('1') % 2)  # parity of bit count
    
    # Ensure it sums to target
    h = project_onto_sum_box(h, target_sum)
    
    # Multi-resolution approach:
    # Coarse resolution
    n_coarse = 64
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n), h)
    h_coarse = project_onto_sum_box(h_coarse, n_coarse / 2)
    
    def refine_and_optimize(h_prev, n_refine):
        h = h_prev.copy()
        h = project_onto_sum_box(h, target_sum)
        
        # Annealing parameters
        alpha = 1.0
        alpha_max = 2000.0
        gamma = 1.3
        eta = 0.01
        max_iter_per_alpha = 200
        
        best_h = h.copy()
        best_c5 = c5_true(h)
        
        while alpha <= alpha_max and time.time() - start_time < budget_s - 10:
            for it in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s - 10:
                    break
                    
                A = autocorr_fft(h)
                g = -A
                p = softmax_weights(g, alpha)
                grad = gradient(h, p)
                h_new = h - eta * grad
                h_new = project_onto_sum_box(h_new, target_sum)
                h = h_new
                
                if c5_true(h) < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_true(h)
            
            alpha *= gamma
            eta *= 0.9
            
            if c5_true(h) < best_c5:
                best_h = h.copy()
                best_c5 = c5_true(h)
                
        return best_h, best_c5
    
    # Coarse optimization
    h_coarse, c5_coarse = refine_and_optimize(h_coarse, n_coarse)
    
    # Interpolate to finer grid
    h_fine = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, n_coarse), h_coarse)
    h_fine = project_onto_sum_box(h_fine, target_sum)
    
    # Finer optimization
    h_fine, c5_fine = refine_and_optimize(h_fine, n)
    
    # Optional refinement to n=256
    if time.time() - start_time < budget_s - 30:
        n_fine = 256
        h_very_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n), h_fine)
        h_very_fine = project_onto_sum_box(h_very_fine, n_fine / 2)
        h_very_fine, c5_very_fine = refine_and_optimize(h_very_fine, n_fine)
        h_final = h_very_fine
        best_c5 = c5_very_fine
    else:
        h_final = h_fine
        best_c5 = c5_fine
    
    # Discrete hill climbing
    best_h = h_final.copy()
    best_c5 = c5_true(best_h)
    
    for _ in range(10000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Try a random swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap values
            trial_h = best_h.copy()
            trial_h[i], trial_h[j] = trial_h[j], trial_h[i]
            
            trial_c5 = c5_true(trial_h)
            if trial_c5 < best_c5:
                best_h = trial_h
                best_c5 = trial_c5
    
    # Final SLSQP optimization
    def objective(h):
        return c5_true(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            best_h = res.x
        best_h = project_onto_sum_box(best_h, target_sum)
        final_c5 = c5_true(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n)
