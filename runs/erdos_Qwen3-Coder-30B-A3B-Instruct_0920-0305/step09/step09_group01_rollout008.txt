```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT for evaluation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points

    # Binary matrix representation for shifted products
    def create_shift_matrix(n, shift):
        M = np.zeros((n, n))
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                M[i, j] = 1
        return M

    # SDP based approach
    def sdp_optimize():
        # Initialize variables
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), PSD=True)

        # Constraints
        constraints = [
            cp.diag(X) == h,  # diag(X) = h
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
        ]

        # Loop for all shifts
        for k in range(n_points):
            M_k = create_shift_matrix(n_points, k)
            constraints.append(cp.trace(M_k @ X) >= (n_points / 2 - 0.380))

        # Objective: minimize the largest possible correlation
        objective = cp.Minimize(cp.sum(h))

        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, abstol=1e-5, reltol=1e-5)
            if prob.status == 'optimal':
                h_result = h.value
                return h_result
        except Exception:
            pass
        
        return None

    # Start with Gaussian initialization
    h_init = np.random.normal(0.5, 0.1, n_points)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points / 2) / np.sum(h_init)

    best_c5 = evaluate_c5(h_init)
    best_h = h_init.copy()

    # Run SDP optimization
    h_opt = sdp_optimize()
    if h_opt is not None:
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final refinement with local optimization if time permits
    if time.time() - start_time < budget_s - 1:
        try:
            # Simple gradient descent approach
            h_curr = best_h.copy()
            for _ in range(1000):
                if time.time() - start_time > budget_s - 1:
                    break
                corr = compute_corr_fft(h_curr)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    # Compute approximate gradient
                    h_plus = h_curr.copy()
                    h_plus[i] += 1e-6
                    corr_plus = compute_corr_fft(h_plus)
                    grad[i] = (c5_from_corr(corr_plus, n_points) - best_c5) / 1e-6
                
                # Update 
                h_curr -= 1e-5 * grad
                h_curr = np.clip(h_curr, 0, 1)
                h_curr = h_curr * (n_points / 2) / np.sum(h_curr)
                c5_new = evaluate_c5(h_curr)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_curr.copy()
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
```
