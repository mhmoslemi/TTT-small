import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def evaluate_smooth_surrogate(h, epsilon):
    """Evaluate smooth approximation of max overlap"""
    # compute all overlaps using correlation (this is equivalent to evaluation)
    corr = np.correlate(h, 1-h, mode="full")
    # take max to avoid numerical issues
    corr = corr - np.max(corr)
    # apply log-sum-exp with smoothness parameter epsilon
    log_sum_exp = np.log(np.sum(np.exp(epsilon * corr))) - np.log(len(corr))
    return log_sum_exp / epsilon

def compute_gradient(h, epsilon, return_weights=False):
    """Compute gradient of smooth surrogate using shifted arrays"""
    # Compute all c_k values for k from -(n-1) to n-1
    # But we will compute as: c_k = sum_{i=0}^{n-1-|k|} h_{i+|k|}(1-h_i) 
    # which leads to a symmetric form: 
            
    # The amount of shift varies from 0 to n-1, so we need to compute:
    # c_0 = sum h_{i+0}(1-h_i) = sum h_i*(1-h_i)  
    # c_k for k>0: sum h_{i+k}(1-h_i)
    # c_{-k} for k>0: sum h_{i}(1-h_{i+k})
    # or equivalently:
    # c_k = sum_{i=0}^{n-1-|k|} h_{i+|k|} · (1 - h_i) 
    # let's compute all the sums for k from -n+1 to n-1
    overlaps = []
    shifted_h = [h]
    P = h.shape[0]
    for i in range(1, P):
        shifted_h.append(np.concatenate([np.zeros(i), h[:-i]]))  # shift left by i
    
    shifted_1_minus_h = [1 - h]
    for i in range(1, P):
        shifted_1_minus_h.append(np.concatenate([np.zeros(i), (1 - h)[:-i]]))
    
    for k in range(-P+1, P):
        abs_k = abs(k)
        if abs_k >= P:
            overlaps.append(0.0)
            continue
            
        if k >= 0:
            # shift has k positions to the right, so opposite case of below
            # c_k = sum_{i=0}^{n-1-k} h_{i+k}(1 - h_i)
            # same as c_k = sum_{i=k}^{n-1} h_i (1 - h_{i-k}) = sum_{i=0}^{n-k-1} h_{i+k} (1 - h_i) 
            sum_val = np.sum(shifted_h[abs_k][:(P-abs_k)] * shifted_1_minus_h[0][abs_k:])
        else:
            # shifted on the left, same pattern, our new implementation  
            # c_{-k} = sum_{i=0}^{n-1-k} h_{i}(1 - h_{i+k}) 
            # same as sum_{i=k}^{n-1} h_{i-k} (1 - h_i)
            # This is the reverse case of above:
            # To compute original c_k = sum_{i=0}^{n-1-k} h_{k+i}(1 - h_i)
            # but we want c_{-k} = sum_{i=0}^{n-1-k} h_i(1 - h_{i+k})
            # Case: k=0 gives no shift; k=-1 so k+i = i-1: h_{i-1} (1 - h_i)  
            # Special handling here:
            sum_val = np.sum(shifted_h[0][abs_k:] * shifted_1_minus_h[abs_k][:(P-abs_k)])
            
        overlaps.append(sum_val)
        
    overlaps = np.array(overlaps)
    # Now do log-sum-exp trick to compute gradient  
    # Use a numerically safe version
    max_val = np.max(overlaps)
    # For numerical stability
    exp_vals = np.exp(epsilon * (overlaps - max_val))
    weights = exp_vals / np.sum(exp_vals)
    if not return_weights:
        # gradient = sum over k weights_k * gradient c_k
        # gradient of c_k(w.r.t. h) does not depend on m
        # for the j-th component
        total_grad = np.zeros_like(h)
        
        # k=0 case: c_0 = sum(h_i (1-h_i)) -> grad at h_j is (1-h_j - h_j) = 1 - 2*h_j
        for i in range(P):
            total_grad[i] += weights[0] * (1 - 2 * h[i])  # first term contributes
            
        # k>0 case: c_k = sum_{i=0}^{n-1-k} h_{i+k} (1-h_i) 
        # gradient w.r.t. h_j: 
        # For j < k: 0
        # for k <= j <= n-1-k: term is 1 - h_{j-k} (coeff of h_{j-k} in sum) 
        # for j > n-1-k: 0 (other equation)
        for k in range(1, P):
            # contribute to k-indexed positions  
            for i in range(k, P - k):
                # This is: delta_{i,j+k} * (1 - h_{i-k})
                # which is: h_{i+k} * (1 - h_{i}) 
                # for gradient of cv = sum h_{i+|k|}(1-h_i) wrt h_j
                # we look for: for each i, how does that term affect j?  
                #      For i-k = j, it's (-h_{j+k}) 
                #      For i+k = j, it's (1 - h_j)
                #バレる
                if i+k < P:
                    # this contributes to h_j term; h_j += 1 - h_{i}
                    # gradient with respect to h_{i}  
                    total_grad[i] += weights[k + P - 1] * (1 - h[i+k]) 

                if i-k >= 0:
                    # this contributes to h_{j} term; h_{j} += -h_{j+k}  
                    # Depends on h_i+k 
                    total_grad[i] += weights[P - 1 - k] * -h[i-k] 

        return total_grad
    else:
        return weights

    # Accelerated gradient descent setup  
    epsilon_schedule = 1.5
    epsilon = 1.0
    max_epsilon = 100.0
    lr = 0.1  # this might go down when epsilon increases
    lr_decay = 0.5
    
    # Adam momentum
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n_points)
    v = np.zeros(n_points)  
    epsilon_adam = 1e-8
    
    def get_lr(eps):
        # decreasing learning rate with epsilon 
        # if eps gets big, become smaller step
        factor = min(1., 1.0 + 0.05 * (eps - 1.0)) 
        return lr * factor

    # Multiple random restarts
    num_restarts = 15  
    
    # Set up for warm start and confidence
    restart_h = [None for _ in range(num_restarts)]
    restart_c5 = [float('inf') for _ in range(num_restarts)]

    for restart in range(num_restarts):
        # Some restarts start with the known value, others with uniform noise
        if restart == 0:
            h = np.array(initial_h_values)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        
        best_in_loop = float('inf')
        best_h_in_loop = h.copy()
        
        # Inner loop
        for iter_num in range(4000):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Inner gradient update step
            if iter_num % 100 == 0:
                # Evaluate and update best if needed
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                if current_c5 < best_in_loop:
                    best_in_loop = current_c5
                    best_h_in_loop = h.copy()
            
            # Transition to exact evaluation every 100 steps  
            if iter_num % 100 == 0:
                # Get gradient
                g = compute_gradient(h, epsilon)
                
                # Adam updates  
                m = beta1 * m + (1 - beta1) * g  
                v = beta2 * v + (1 - beta2) * (g * g)
                m_hat = m / (1 - beta1**(iter_num + 1))
                v_hat = v / (1 - beta2**(iter_num + 1))
                
                # Update
                alpha = get_lr(epsilon)
                h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon_adam)
                
                # Project onto feasible set
                h = project_to_sum_box(h_new, TARGET_SUM)
            
            # Annealing parameter 
            if iter_num % 500 == 0 and epsilon < max_epsilon:  
                epsilon *= epsilon_schedule
                lr *= lr_decay
          
        # Store best for this restart
        current_c5 = evaluate_c5(best_h_in_loop)
        if current_c5 < restart_c5[restart]:
            restart_c5[restart] = current_c5
            restart_h[restart] = best_h_in_loop.copy()

    # Find in global best so far now
    for i in range(num_restarts):
        if restart_c5[i] < best_c5:
            best_c5 = restart_c5[i]
            best_h = restart_h[i].copy()

    # Final refinement by small binary swaps
    if time.time() - start_time < 0.9 * budget_s:
        h = best_h.copy()
        # Make discrete (threshold)
        threshold = np.median(h)
        h_binary = np.array([1 if x > threshold else 0 for x in h])
        # Adjust to correct sum
        ones_count = np.sum(h_binary)
        needed_ones = int(TARGET_SUM) 
        to_flip = ones_count - needed_ones
        
        if to_flip > 0: 
            indices_ones = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(indices_ones, to_flip, replace=False)
            h_binary[flip_indices] = 0
        elif to_flip < 0: 
            indices_zeros = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(indices_zeros, -to_flip, replace=False)
            h_binary[flip_indices] = 1
        
        # Check that sum of h_binary is correct
        if np.sum(h_binary) == TARGET_SUM:
            # Now local search on binary  
            h_orig = h_binary.copy()
            C_orig = evaluate_c5(h_binary)  
            
            for _ in range(3000):  
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                i = np.random.randint(0, n_points)
                j = np.random.randint(0, n_points)
                if i == j:
                    continue
                    
                if h_binary[i] == h_binary[j]:
                    continue
                
                # mimic swapped bits: flip i and j
                h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
                
                C_new = evaluate_c5(h_binary) 
                if C_new < C_orig:  # accept improvement
                    C_orig = C_new
                else:
                    # revert  
                    h_binary[i], h_binary[j] = h_binary[j], h_binary[i]
                    
            if C_orig < best_c5:
                best_c5 = C_orig
                best_h = h_binary.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
