import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def gradient_of_c_k(h, k):
    """Compute gradient of c_k(h) = sum_{i=0}^{n-1-|k|} h_{i+|k|}*(1 - h_i)"""
    n = len(h)
    if k >= 0:
        # Forward shift
        grad = np.zeros(n)
        for i in range(n - k):
            grad[i] -= h[i + k]
            grad[i + k] += (1 - h[i])
    else:
        # Backward shift
        k = -k
        grad = np.zeros(n)
        for i in range(n - k):
            grad[i] -= h[i]
            grad[i + k] += (1 - h[i])
    return grad

def smooth_max_gradient(h, epsilon):
    """Compute gradient of smooth max"""
    # Precompute all c_k
    c_vals = []
    grad_list = []
    for k in range(-n_points + 1, n_points):
        grad_k = gradient_of_c_k(h, k)
        c_k = np.sum(h[:-abs(k) if k > 0 else h[abs(k):]] * (1 - h[abs(k):] if k > 0 else h[:-abs(k)]))
        c_vals.append(c_k)
        grad_list.append(grad_k)
    
    # Compute soft max weights
    c_vals = np.array(c_vals)
    max_c = np.max(c_vals)
    c_shifted = c_vals - max_c
    exp_c = np.exp(epsilon * c_shifted)
    sum_exp_c = np.sum(exp_c)
    weights = exp_c / sum_exp_c
    
    # Compute gradient
    g = np.zeros_like(h)
    for i, w in enumerate(weights):
        g += w * grad_list[i]
    
    return g
