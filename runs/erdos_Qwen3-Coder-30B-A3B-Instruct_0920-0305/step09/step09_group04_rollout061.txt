import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, eps):
    """Compute smooth max and its gradient"""
    # Compute correlation values using FFT for efficiency
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Use numerically stable computation for softmax
    corr_shifted = eps * corr
    corr_shifted = corr_shifted - np.max(corr_shifted)
    weights = np.exp(corr_shifted)
    weights = weights / np.sum(weights)
    
    # Smooth max value
    smooth_max_val = np.log(np.sum(weights)) / eps
    
    # Compute gradient: weighted sum of individual gradients
    n = len(h)
    grad = np.zeros(n)
    
    # For each correlation lag k, compute contribution to gradient
    # k ranges from -(n-1) to (n-1), so indices from 0 to 2*n-2
    for k in range(2 * n - 1):
        if k < n - 1:  # Negative k
            shift_val = k - (n - 1)  # - (n-1) to 0
            # The source of c_k gradient: derivative of h_i * (1 - h_{i+k}) 
            # w.r.t. h_j = (1 - h_{j-k}) if j-k >= 0 and j <= n-1-k
            # minus h_{j+k} if j+k <= n-1 and j >= 0
            
            # Derivative of h_{j+k} * (1 - h_j) for j in [0, n-1-shift_val]
            for j in range(n):
                if j <= n - 1 + shift_val:
                    if j >= 0:
                        grad[j] += weights[k] * (-h[j + shift_val])
                if j - shift_val >= 0:
                    if j <= n - 1:
                        grad[j] += weights[k] * (1 - h[j - shift_val])
        else:  # Positive k
            shift_val = k - (n - 1)  # 0 to n-1
            # For non-negative shifts, we iterate over j 0 to n-1-shift_val
            # Correlation c_k = sum_{i=0}^{n-1-shift_val} h_{i+shift_val} * (1 - h_i)
            for j in range(n):
                if j >= shift_val:
                    if j <= n - 1:
                        grad[j] += weights[k] * (-h[j - shift_val])
                if j <= n - 1 - shift_val:
                    if j >= 0:
                        grad[j] += weights[k] * (1 - h[j + shift_val])
    
    return smooth_max_val, grad

def adamsolver(func_grad, h0, learning_rate=1e-3, max_iter=2000, eps_schedule=[1e-1, 1, 5, 10, 20, 50], tolerance=1e-6):
    """Adam optimizer with epsilon schedule"""
    state = {
        'm': np.zeros_like(h0),
        'v': np.zeros_like(h0),
        't': 0,
    }
    
    h = h0.copy()
    best_h = h.copy()
    best_value = float('inf')
    
    # For each epsilon value in schedule
    for eps in eps_schedule:
        # Inner loop for this schedule value
        for i in range(max_iter):
            if i % 500 == 0:
                current_val = func_grad(h)[0]
                if current_val < best_value:
                    best_h = h.copy()
                    best_value = current_val
            # Compute loss and gradient for current epsilon
            val, grad = func_grad(h, eps)
            
            # Adam update step
            state['t'] += 1
            state['m'] = 0.9 * state['m'] + 0.1 * grad
            state['v'] = 0.999 * state['v'] + 0.001 * grad**2
            m_hat = state['m'] / (1 - 0.9**state['t'])
            v_hat = state['v'] / (1 - 0.999**state['t'])
            h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Check convergence
            if np.linalg.norm(h_new - h) < tolerance:
                break
            h = h_new
    
    return best_h, best_value
