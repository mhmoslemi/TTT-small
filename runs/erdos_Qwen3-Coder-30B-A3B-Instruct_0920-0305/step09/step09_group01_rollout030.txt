import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_c5(h, shift_idx):
    """Compute gradient of C(shift_idx) with respect to h"""
    n = len(h)
    gradient = np.zeros(n)
    
    # Find the actual shift value
    shift = shift_idx - (n - 1)
    
    if shift >= 0:
        # For positive shift: h[i] contributes to sum where i+shift < n
        # Gradients: 
        # - i in [0, n-shift-1]: derivative includes +1 - h[i+shift]  
        # - i in [shift, n-1]: derivative includes -h[i-shift]
        gradient[:n-shift] += 1 - h[shift:]
        gradient[shift:] -= h[:n-shift]
    else:
        # For negative shift: h[i-shift] contributes to sum where i-shift >= 0 and i < n
        # Similar logic but with sign flipped
        shift_abs = -shift
        gradient[shift_abs:] += 1 - h[:n-shift_abs]
        gradient[:n-shift_abs] -= h[shift_abs:]
    
    return gradient
