```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse binary sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    
    # Try Thue-Morse sequence
    h_tm = thue_morse_sequence(n_points)
    c5_tm = evaluate_c5(h_tm)
    if c5_tm < best_c5:
        best_h = h_tm.copy()
        best_c5 = c5_tm
    
    # Random search phase: 20% of time
    random_search_time = 0.2 * budget_s
    while time.time() - start_time < random_search_time:
        if time.time() - start_time > budget_s:
            break
        # Generate random binary vector with exactly half ones
        indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Local search phase: hill climbing
    if best_h is not None:
        h_cur = best_h.copy()
        c5_cur = best_c5
        improved = True
        iterations_without_improvement = 0
        max_no_improvement = 5000
        
        while time.time() - start_time < budget_s and iterations_without_improvement < max_no_improvement:
            if time.time() - start_time > budget_s:
                break
            # Generate random swaps
            for _ in range(200):
                if time.time() - start_time > budget_s:
                    break
                # Get current set of 1's and 0's
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur < 0.5)[0]
                
                if len(ones) == 0 or len(zeros) == 0:
                    break
                    
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Make swap
                h_new = h_cur.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                
                c5_new = evaluate_c5(h_new)
                
                if c5_new < c5_cur - 1e-12:
                    h_cur = h_new
                    c5_cur = c5_new
                    improved = True
                    iterations_without_improvement = 0
                    if c5_new < best_c5:
                        best_h = h_cur.copy()
                        best_c5 = c5_new
                    break  # Break inner loop after improvement to restart scanning
                
            if not improved:
                iterations_without_improvement += 1
            else:
                improved = False
    
    # Final continuous optimization
    if best_h is not None:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
            
            res = minimize(
                lambda x: evaluate_c5(x),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                best_h = res.x
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
```
