```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Global constants
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_overlap_matrix(n):
    """Compute the overlap matrix M_l for a given shift l"""
    M = np.zeros((n, n))
    for i in range(n):
        if i + 1 < n:
            M[i, i+1] = 1
            M[i+1, i] = 1
    return M

def compute_all_overlap_matrices(n):
    """Precompute overlap matrices for all shifts 0 to n-1"""
    Ms = []
    for l in range(n):
        M = np.zeros((n, n))
        for i in range(n - l):
            M[i, i + l] = 1
            M[i + l, i] = 1
        Ms.append(M)
    return Ms

def evaluate_c5_sdp(h_vals):
    """Fast evaluation using SDP relaxation and rounding"""
    h = np.array(h_vals)
    
    # Compute overlap for each shift using correlation matrix
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    
    # Generate initial guess from initial_h_values
    try:
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        c5_initial = evaluate_c5_sdp(h0)
        if c5_initial < best_c5:
            best_h = h0.copy()
            best_c5 = c5_initial
    except Exception:
        pass

    # Generate multiple random initializations to ensure diversity
    for i in range(50):
        if time.time() - start_time > budget_s - 2:
            break
            
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        c5_try = evaluate_c5_sdp(h_start)
        if c5_try < best_c5:
            best_h = h_start.copy()
            best_c5 = c5_try
            
        if i % 10 == 0 and time.time() - start_time > budget_s - 3:
            break

    # Apply SDP-based refinement if time permits
    if best_h is not None and time.time() - start_time < budget_s - 5:
        try:
            # Use the standard gradient-based optimization method with tight constraints
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_c5_sdp(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            
            if res.success:
                h_refined = res.x
                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5_sdp(h_refined)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    # Final polish phase using a simple gradient ascent approach
    if best_h is not None and time.time() - start_time < budget_s - 2:
        h_final = best_h.copy()
        prev_c5 = best_c5
        
        # Simple local refinement using gradient descent (a few iterations)
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Basic gradient update using finite differences for the smooth approximation
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_final.copy()
                h_plus[i] = min(1.0, h_final[i] + eps)
                h_minus = h_final.copy()
                h_minus[i] = max(0.0, h_final[i] - eps)
                
                c5_plus = evaluate_c5_sdp(h_plus)
                c5_minus = evaluate_c5_sdp(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
            # Update with gradient step
            step_size = 0.01
            h_final -= step_size * grad
            h_final = project_to_sum_box(h_final, TARGET_SUM)
            
            new_c5 = evaluate_c5_sdp(h_final)
            if new_c5 < prev_c5:
                prev_c5 = new_c5
            else:
                break
                
        # Final evaluation and update
        final_c5 = evaluate_c5_sdp(h_final)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5

    return (best_h.tolist(), float(best_c5), n_points)
```
