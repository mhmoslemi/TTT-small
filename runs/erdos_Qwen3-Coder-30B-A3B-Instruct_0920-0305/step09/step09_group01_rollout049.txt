import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 150  # Sufficiently large to capture detail, yet manageable
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Indices for shifts
    shifts = list(range(n_points))  # Up to n-1 shifts (symmetric)
    
    # Define binary matrix M for each shift ℓ
    def build_shift_matrix(n, ell):
        M = np.zeros((n, n))
        for i in range(n):
            j = i + ell
            if 0 <= j < n:
                M[i, j] = 1
                M[j, i] = 1  # Symmetric
        return M
    
    # Build all M matrices (only up to n-1 due to symmetry)
    M_matrices = [build_shift_matrix(n_points, ell) for ell in shifts]
    
    # SDP relaxation helper functions
    def sdp_relaxation(h0, alpha):
        """Try to find a feasible h with C5 <= alpha using SDP relaxation."""
        # Input is already feasible w.r.t. box/sum constraints
        h = Variable(n_points)
        X = Variable((n_points, n_points), PSD=True)
        
        # Constraints
        constraints = [
            X >> 0,
            diag(X) == h,  # diag(X) = h (since h_i are in [0,1])
            sum(h) == target_sum,
            0 <= h, h <= 1
        ]
        
        # Add quadratic constraints for each shift (only upper triangular part)
        for ell, M in enumerate(M_matrices):
            # Offset to avoid double-counting (we'll only use ell = 0 to n-1)
            if ell == 0:
                constraints.append(trace(M @ X) <= target_sum / 2 - alpha)
            else:
                # For ell > 0, account for symmetry (overlap is same)
                constraints.append(trace(M @ X) <= target_sum / 2 - alpha)
                
        # Solve the SDP using SCS
        try:
            prob = Problem(Maximize(0), constraints)
            prob.solve(solver=SCS, verbose=False, eps=1e-4)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_val = h.value
                return h_val
        except:
            pass
        return None

    def project_h_to_sum(h, target_sum):
        """Project h onto sum = target_sum with box constraints."""
        def f(lam):
            ph = np.clip(h + lam, 0, 1)
            return np.sum(ph) - target_sum
            
        # Binary search for lambda
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
            
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        return np.clip(h + (a + b)/2, 0, 1)
        
    def evaluate_c5(h):
        """Fast evaluation of C5 via convolution."""
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_corr = np.max(corr)
        return max_corr

    def random_round_sdp(h_vec, n_points):
        """Randomized rounding of SDP solution."""
        # Prefer to use the diagonal elements of the SDP matrix for candidate
        h_rounded = h_vec.copy()
        h_rounded = np.clip(h_rounded, 0, 1)
        h_rounded = project_h_to_sum(h_rounded, target_sum)
        return h_rounded

    # Global optimization using binary search on alpha
    best_h = None
    best_c5 = float('inf')
    
    # Binary search interval for alpha (C5)
    lower_alpha = 0.0
    upper_alpha = 1.0
    tolerance = 1e-4
    max_iterations = 30

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
        
        alpha = (lower_alpha + upper_alpha) / 2
        
        # Try SDP relaxation
        h_candidate = sdp_relaxation(np.random.rand(n_points), alpha)
        
        if h_candidate is not None:
            h_projected = project_h_to_sum(h_candidate, target_sum)
            # Evaluate
            c5_candidate = evaluate_c5(h_projected)
            
            if c5_candidate < best_c5:
                best_h = h_projected.copy()
                best_c5 = c5_candidate
            
            # Feasible, so try smaller alpha
            upper_alpha = alpha
        else:
            # Infeasible, so need larger alpha
            lower_alpha = alpha
    
    # Final polishing with local search
    if best_h is not None:
        def obj_to_minimize(h):
            return evaluate_c5(h)
            
        # Simple local gradient-like update (gradient-free local search)
        h_refined = best_h.copy()
        best_val = evaluate_c5(h_refined)

        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = h_refined.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_new[idx] += delta
            h_new = project_h_to_sum(h_new, target_sum)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < best_val:
                h_refined = h_new
                best_val = c5_new
                
        best_h = h_refined.copy()
        best_c5 = best_val

    # Make sure final result is good
    # Convert to list for return format
    final_h_list = best_h.tolist() if best_h is not None else [0.5]*n_points
    final_c5 = float(best_c5) if best_h is not None else 1.0
    return (final_h_list, final_c5, n_points)
