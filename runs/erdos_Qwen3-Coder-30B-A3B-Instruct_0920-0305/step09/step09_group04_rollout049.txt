import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Create variables for convex optimization
    h = Variable(n_points)
    t = Variable(1)

    objective = Minimize(t)

        # Using quadratic identity: h_i * h_{i+k} = 1/2 * [(h_i+h_{i+k})^2 - h_i^2 - h_{i+k}^2]
        # So C_k = sum(h_i) - 1/2 * sum((h_i+h_{i+k})^2) + 1/2 * sum(h_i^2) + 1/2 * sum(h_{i+k}^2)
        # This is convex-concave since sum(h_i) is linear and sum((h_i+h_{i+k})^2) is convex.
        # We want to enforce C_k <= t
        # Since sum(h_i) - 1/2 * sum((h_i+h_{i+k})^2) + 1/2 * sum(h_i^2) + 1/2 * sum(h_{i+k}^2) <= t
        # Rearranging: sum(h_i) + 1/2 * sum(h_i^2) + 1/2 * sum(h_{i+k}^2) - 1/2 * sum((h_i+h_{i+k})^2) <= t
        # But sum(h_{i+k}^2) is same as sum(h_i^2) over the overlapping domain.
        # So let's compute actually using the definition of C_k directly:
        
        # We use simpler approach - project back to valid h values on each iteration
        # Thus we directly solve the problem via CCP using barrier or direct optimization
        
    # Guidance for better problem setup and iterative solving strategy:
    # Since it's numerically challenging to represent all constraints directly,
    # we'll procedurally solve:
    # Maximize t subject to:
    # C_k <= t for all k=1,..,n-1  (where C_k = sum(h_i) - sum(h_i * h_{i+k}))
    # sum(h_i) = 1 and 0 <= h_i <= 1
    
    # Init solution using initial construction
    h_init = np.array(initial_h_values)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (TARGET_SUM / np.sum(h_init))  # Normalize to correct sum
    
    # Binary search approach for tight solution
    def compute_c5(h_vec):
        # Compute the maximum overlap which is similar to correlation with 
        # reflected and complemented function
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Simplified strategy: Put constraints on each pair of indices defining overlap
    # We'll try solving a small-scale version directly with known structure
    
    # Better approach: Use a direct optimization over a small number of parameters:
    # Represents h as a piecewise constant function with few steps.
    
    # Use a hybrid convex+sequential method.
    
    best_c5 = float('inf')
    best_h = None
    
    # Attempt 1: Convex relaxation of the concern through sectioned approach
    # Start with multi-step piecewise constant solution   
    def piecewise_optimize(n_pieces=3, n_iter=1000):
        # Initialize with piecewise constant build
        steps = np.linspace(0, 2, n_pieces + 1)
        heights = np.random.rand(n_pieces)  # Initial random values
        heights = heights / np.sum(heights) * 2.0  # normalize based on desired sum
        heights = np.clip(heights, 0, 1)
        
        def eval_h(heights):
            h = np.zeros(n_points)
            dx_fixed = 2.0 / n_points
            for i in range(n_pieces):
                start_idx = int(steps[i] / 2.0 * n_points)
                end_idx = int(steps[i+1] / 2.0 * n_points)
                h[start_idx:end_idx] = heights[i]
            return h
            
        def eval_objective(p):
            heights = np.array(p)
            h = eval_h(heights)
            # Ensure sum constraint
            h = h * (TARGET_SUM / np.sum(h))
            return compute_c5(h)
            
        # Optimize using scipy minimize with basinhopping
        bounds = [(0, 1) for _ in range(n_pieces)]
        x0 = heights
        result = minimize(eval_objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter':500})
        
        best_vals = result.x
        final_h = eval_h(best_vals)
        final_h = final_h * (TARGET_SUM / np.sum(final_h))  # Renormalize
        c5_val = compute_c5(final_h)
        return final_h, c5_val
    
    # Run piecewise optimization multiple times with different seeds
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        try:
            h_result, c5_result = piecewise_optimize()
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = h_result
        except:
            continue
    
    # Final refinement with local search around best solution
    if best_h is not None:
        def local_refinement(h):
            # One possible method: modify in small batches
            # Return modified version of h that is still feasible
            h_new = h.copy()
            idx = np.random.randint(0, len(h))
            delta = np.random.uniform(-0.02, 0.02)
            h_new[idx] += delta
            # Project back
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (TARGET_SUM / np.sum(h_new))
            return h_new
            
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            try:
                h_refined = local_refinement(best_h)
                c5_refined = compute_c5(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
            except:
                continue
    
    # If we did not find something better:
    if best_h is None:
        best_h = h_init
        best_c5 = compute_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
