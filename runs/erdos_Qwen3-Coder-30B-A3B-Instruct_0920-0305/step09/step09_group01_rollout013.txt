import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, corr_values, mu):
    """Compute the gradient of the log-sum-exp surrogate"""
    # Get the indices of the maximum correlation
    k_max = np.argmax(corr_values)
    
    # Compute weights
    max_corr = np.max(corr_values)
    exp_corr = np.exp((corr_values - max_corr) / mu)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each shift k, compute gradient and accumulate
    for k in range(-n+1, n):
        if k >= 0:
            # h[i] contributes to c_k if i+k < n
            if k < n:
                grad[:n-k] += weights[k] * (1 - h[k:])
                grad[k:] -= weights[k] * h[:n-k]
        else:
            # h[i] contributes to c_k if i < n and i+k >= 0
            k_abs = -k
            if k_abs < n:
                grad[k_abs:] += weights[k+n-1] * (1 - h[:n-k_abs])
                grad[:n-k_abs] -= weights[k+n-1] * h[k_abs:]
    
    return grad

def line_search(h, grad, mu, max_iter=5):
    """Perform line search to find a good step size"""
    alpha = 0.2 * dx
    h_current = h.copy()
    current_c5 = evaluate_c5(h_current)
    
    for _ in range(max_iter):
        h_new = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            return h_new, alpha
        alpha /= 2.0
        
    return h_current, alpha

def continuation_gradient_descent(h_start, mu_schedule, max_iter_per_mu=250):
    """Continuation gradient descent method"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for mu in mu_schedule:
        for _ in range(max_iter_per_mu):
            # Compute correlations
            corr = compute_corr_fft(h)
            c5_current = c5_from_corr(corr, n_points)
            
            # Compute gradient
            grad = compute_gradient(h, corr, mu)
            
            # Line search
            h_new, alpha = line_search(h, grad, mu)
            
            # Update
            h = h_new
            new_c5 = evaluate_c5(h)
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
            # Random jitter
            if (_ + 1) % 500 == 0:
                noise = np.random.normal(0, 0.02, n_points)
                h_perturbed = project_to_sum_box(h + noise, TARGET_SUM)
                c5_perturbed = evaluate_c5(h_perturbed)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_perturbed
                h = h_perturbed
                
    return best_h, best_c5

def pairwise_exchange_refinement(h, max_iter=10000):
    """Refine the solution with deterministic pairwise swaps"""
    h_refined = h.copy()
    best_c5 = evaluate_c5(h_refined)
    
    for _ in range(max_iter):
        # Find indices that can potentially be improved
        # Find a point at higher value and point with lower value
        high_indices = np.where(h_refined > 1e-4)[0]
        low_indices = np.where(h_refined < 1 - 1e-4)[0]
        
        if len(high_indices) == 0 or len(low_indices) == 0:
            break
            
        i = np.random.choice(high_indices)
        j = np.random.choice(low_indices)

        delta = 0.01

        if c5_i < best_c5:

            best_c5 = c5_i
