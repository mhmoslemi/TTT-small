def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth max approximation of correlation values"""
    corr = compute_corr_fft(h)
    # Compute log-sum-exp with epsilon scaling
    c_scaled = epsilon * corr
    # Subtract max for numerical stability
    c_scaled_stable = c_scaled - np.max(c_scaled)
    log_sum_exp = np.log(np.sum(np.exp(c_scaled_stable))) - np.log(len(corr)) + np.max(c_scaled)
    return log_sum_exp

def grad_smooth_max_corr(h, epsilon):
    """Compute gradient of smooth max correlation"""
    corr = compute_corr_fft(h)
    
    # Compute softmax weights
    c_scaled = epsilon * corr
    c_scaled_stable = c_scaled - np.max(c_scaled)
    weights = np.exp(c_scaled_stable)
    weights = weights / np.sum(weights)
    
    # Compute gradient components for the smooth max
    grad = np.zeros_like(h)
    
    # For each possible shift
    n = len(h)
    for k in range(-n+1, n):
        # Index for shift k (correlation indices from 0..2n-2)
        shift_idx = k + n - 1
        if shift_idx < 0 or shift_idx >= 2*n - 1:
            continue
            
        # Compute contribution to gradient for this shift k
        # This is the derivative of `h_i * (1 - h_{i+k})` w.r.t. h_j
        # Only contributions for j such that the pair potentially contributes
        
        # Contributions from h_{i+k} terms
        if k >= 0:
            # h_i+k * (1 - h_i) contributions to both h_i and h_{i+k}
            for i in range(n - k):
                j1 = i  
                j2 = i + k
                if j1 < n:
                    grad[j1] -= weights[shift_idx] * (1 - h[j2])  # - (1 - h_{i+k})
                if j2 < n:
                    grad[j2] += weights[shift_idx] * h[j1]  # + h_i
        else:
            # For negative k, h_i * (1 - h_{i+k})
            for i in range(n + k):
                j1 = i + k  
                j2 = i
                if j1 < n:
                    grad[j1] -= weights[shift_idx] * (1 - h[j2])  # - (1 - h_{i+k}) 
                if j2 < n:
                    grad[j2] += weights[shift_idx] * h[j1]  # + h_i
    
    return grad
