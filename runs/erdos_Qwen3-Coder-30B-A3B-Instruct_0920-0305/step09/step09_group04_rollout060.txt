def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth max of correlation values"""
    # Compute correlation: h * (1 - h) with zero padding
    corr = np.correlate(h, 1 - h, mode="full")
    # Use log-sum-exp to approximate max
    corr_shift = corr + epsilon * np.max(corr)
    exp_corr = np.exp(corr_shift)
    exp_sum = np.sum(exp_corr)
    weights = exp_corr / exp_sum
    
    return np.log(exp_sum) / epsilon - np.max(corr) / epsilon, weights

def grad_smooth_max_corr(h, epsilon):
    """Compute gradient of smooth max correlation"""
    corr = np.correlate(h, 1 - h, mode="full")

    corr_shift = epsilon * corr
    # Subtract max to avoid overflow
    corr_shift = corr_shift - np.max(corr_shift)
    exp_corr = np.exp(corr_shift)
    exp_sum = np.sum(exp_corr)
    weights = exp_corr / exp_sum
    
    # Compute the gradient
    grad = np.zeros_like(h)
    
    # For each shift, compute the gradient contribution
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            # Forward shift
            if k < n:
                # Contributing terms: h[j-k] * (1-h[j]) for j >= k
                # For j from k to n-1
                # Gradient from h[j-k] term: (1-h[j]) 
                # Gradient from (1-h[j]) term: -h[j-k]
                for j in range(k, n):
                    grad[j] += weights[k+n-1] * (1 - h[j-k]) if j >= k else 0
                    grad[j-k] -= weights[k+n-1] * h[j] if j-k >= 0 else 0
        else:
            # Backward shift -k
            k_abs = -k
            if k_abs < n:
                # Contributing terms: h[j] * (1-h[j+k_abs]) for j <= n-1-k_abs
                # For j from 0 to n-1-k_abs
                # Gradient from h[j] term: -(1-h[j+k_abs])
                # Gradient from (1-h[j+k_abs]) term: h[j]
                for j in range(0, n-k_abs):
                    grad[j] -= weights[k+n-1] * (1 - h[j+k_abs]) if j+k_abs < n else 0
                    grad[j+k_abs] += weights[k+n-1] * h[j] if j+k_abs < n else 0
    
    return grad * 1.0  # Normalize to preserve relative scaling
