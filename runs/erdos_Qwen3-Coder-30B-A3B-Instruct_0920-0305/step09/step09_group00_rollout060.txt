import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with Thue-Morse sequence
    h0 = thue_morse_sequence(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Random search phase - 20% of budget
    random_search_time = budget_s * 0.2
    trials = 0
    while time.time() - start_time < start_time + random_search_time:
        # Generate random binary vector with exactly TARGET_SUM ones
        indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        trials += 1
    
    # Local search with hill climbing
    # Keep track of best during hill climbing
    local_search_time = budget_s * 0.7
    while time.time() - start_time < start_time + local_search_time:
        # Start from best_h and perform hill climbing
        current_h = best_h.copy()
        current_c5 = best_c5
        improved = True
        iterations = 0
        max_iterations = 5000
        
        while improved and iterations < max_iterations:
            improved = False
            # Perform multiple random swaps
            for _ in range(100):
                if time.time() - start_time > start_time + local_search_time:
                    break
                    
                # Find indices with 1s and 0s
                ones = np.where(current_h > 0.5)[0]
                zeros = np.where(current_h < 0.5)[0]
                
                if len(ones) == 0 or len(zeros) == 0:
                    break
                    
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                
                # Create new vector with swap
                h_new = current_h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                
                # Evaluate new solution
                c5_new = evaluate_c5(h_new)
                
                if c5_new < current_c5 - 1e-12:
                    current_h = h_new.copy()
                    current_c5 = c5_new
                    improved = True
                    
                    # Update lists
                    ones = np.where(current_h > 0.5)[0]
                    zeros = np.where(current_h < 0.5)[0]
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        break
            
            iterations += 1
    
    # Final refinement using SLSQP
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Clean up to ensure constraints
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            best_h = res.x
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
