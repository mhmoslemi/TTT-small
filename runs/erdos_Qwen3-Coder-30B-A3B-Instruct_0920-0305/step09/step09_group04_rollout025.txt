import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def grad_c_single(h, s):
    """Gradient of single overlap c_s(h)"""
    grad = np.zeros_like(h)
    if s >= 0:
        # c_s = sum_{i=0}^{n-1-s} h_{i+s} * (1 - h_i)
        # For i = 0...n-1-s: d/dh_i = -h_{i+s}
        # For i = s...n-1: d/dh_{i} = 1-h_{i-s}
        if s > 0:
            grad[s:] -= h[:-s]
        grad[:-s] += 1 - h[s:]
    return grad

def build_gradient_fepsilon(h, c_values, epsilon):
    """Build the gradient from the smooth max"""
    # Compute softmax weights
    c_values = np.array(c_values)
    c_max = np.max(c_values)
    exp_vals = np.exp(epsilon * (c_values - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient as weighted sum of Gradient_c_k
    grad = np.zeros_like(h)
    for k, weight in enumerate(weights):
        s = k - (n_points - 1)
        grad += weight * grad_c_single(h, s)
    return grad

                if new_c5 < old_c5:
                    current_c5 = new_c5
                    improved = True
                    iterations = 0
                else:
                    h_final[i] = 1
                    h_final[j] = 0
            iterations += 1
            
        if current_c5 < best_c5:
            best_h = h_final.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
