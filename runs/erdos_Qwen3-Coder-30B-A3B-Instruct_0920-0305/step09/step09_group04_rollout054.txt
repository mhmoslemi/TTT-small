import numpy as np

import time

# Global variables for convenience
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_surrogate(h, eps):
    """Compute smooth max and its gradient"""
    # Compute all c_k values via FFT
    corr = compute_corr_fft(h)
    c_vals = corr * dx  # c_k values scaled by dx
    
    # Log-sum-exp trick
    c_max = np.max(c_vals)
    exp_vals = np.exp(eps * (c_vals - c_max))
    Z = np.sum(exp_vals)
    log_Z = np.log(Z) + c_max
    
    # Softmax weights
    w_vals = exp_vals / Z
    
    # Compute gradient
    grad = np.zeros_like(h)
    # For each k (from -n+1 to n-1)
    n = len(h)
    for k_idx, w_k in enumerate(w_vals):
        # k is from -n+1 to n-1, map to proper indices
        k = k_idx - (n - 1)
        if k >= 0:
            # Shifted indices
            h_shifted = h[k:]
            h_unshifted = h[:-k]
            diff = h_shifted * (1 - h_unshifted)
            grad[:-k] += w_k * (1 - h[k:])
            grad[k:] += -w_k * h_unshifted
        else:
            k_abs = -k
            h_shifted = h[:k_abs]
            h_unshifted = h[k_abs:]
            diff = h_shifted * (1 - h_unshifted)
            grad[k_abs:] += w_k * (1 - h[:k_abs])
            grad[:k_abs] += -w_k * h_unshifted
    
    return log_Z, grad
