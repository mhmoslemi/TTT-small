import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def softmax_weights(f, alpha):
    """Compute softmax weights for smooth approximation of max"""
    exp_f = np.exp(alpha * (f - np.max(alpha * f)))
    return exp_f / np.sum(exp_f)

            if current_c5 < best_c5:

            # Compute softmax weights
            max_f = np.max(alpha * f)
