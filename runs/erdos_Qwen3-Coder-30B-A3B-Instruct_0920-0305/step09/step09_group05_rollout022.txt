# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_smoothed(h, corr, alpha):
    """Compute smoothed gradient ∇F_α(h)"""
    n = len(h)
    # Compute weights for each shift
    c = corr
    # Log-sum-exp trick for numerical stability
    max_c = np.max(alpha * c)
    exp_terms = np.exp(alpha * c - max_c)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient
    g = np.ones(n)
    for k in range(-n+1, n):
        if k >= 0:
            g[:n-k] -= weights[k+n-1] * h[k:]
            g[k:] -= weights[k+n-1] * h[:n-k]
        else:
            s = -k
            g[s:] -= weights[k+n-1] * h[:n-s]
            g[:n-s] -= weights[k+n-1] * h[s:]
    return g
