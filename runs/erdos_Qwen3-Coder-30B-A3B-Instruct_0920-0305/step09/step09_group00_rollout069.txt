import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full autocorrelation using FFT"""
    n = len(h)
    H = fft(h, 2*n-1)
    corr = ifft(H * np.conj(H)).real
    return corr[:n]

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    return exp_g / np.sum(exp_g)

def gradient(h, p):
    """Compute gradient of the smooth surrogate"""
    n = len(h)
    # Two convolutions: p * h and p_rev * h
    p_padded = np.concatenate([p, np.zeros(n-1)])
    h_padded = np.concatenate([h, np.zeros(n-1)])
    
    conv1 = np.real(ifft(fft(p_padded) * fft(h_padded)))[:n]
    conv2 = np.real(ifft(fft(p_padded[::-1]) * fft(h_padded[::-1])))[:n]
    
    return -(conv1 + conv2)

def project_onto_sum_box(x, target_sum):
    """Project onto [0,1] box constraints with fixed sum using binary search"""
    x = np.array(x)
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def c5_true(h):
    """Compute true C5 using correlation with complement"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * 2.0 / len(h)
