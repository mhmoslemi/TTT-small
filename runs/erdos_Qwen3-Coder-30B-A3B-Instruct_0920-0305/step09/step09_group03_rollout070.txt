import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def softmax_weights(f, alpha):
    """Compute softmax weights for gradient descent"""
    max_f = np.max(alpha * f)
    exp_weights = np.exp(alpha * f - max_f)
    return exp_weights / exp_weights.sum()

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize problem parameters
    TARGET_SUM = n_points / 2
    
    # Use reference solution as starting point

    # Interpolate if necessary
    if len(h0) != n_points:

        h0 = np.interp(x_new, x_ref, h0)
    
    # Refine with optimization
    # First, project it to correct sum
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Add structured perturbations
    h_perturbed = h0.copy()
    # Add some peaks in regions with low values
    for _ in range(5):
        # Find low regions
        low_indices = np.where(h_perturbed < 0.3)[0]
        if len(low_indices) > 0:
            # Pick a random low index
            idx = np.random.choice(low_indices)
            # Set it to a high value and reduce nearby values
            h_perturbed[idx] = 1.0
            for offset in range(-2, 3):
                if 0 <= idx + offset < n_points and offset != 0:
                    h_perturbed[idx + offset] = max(0, h_perturbed[idx + offset] - 0.1)
    
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    c5_perturbed = evaluate_c5(h_perturbed)
    if c5_perturbed < best_c5:
        best_h = h_perturbed.copy()
        best_c5 = c5_perturbed
    
    # Phase 2: Gradient-based optimization with softmax approximation
    # Parameters for gradient descent
    alpha = 1.0
    step_size = 0.05
    max_iter = 1000
    alpha_increase_factor = 1.2
    alpha_max = 200.0
    
    current_h = best_h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current state
        corr = correlate(current_h, 1 - current_h, mode="full")
        f = TARGET_SUM - corr  # f_k = sum_h - correlation_k
        
        # Compute softmax weights
        w = softmax_weights(f, alpha)
        
        # Compute gradient direction
        w_rev = w[::-1]
        conv1 = np.convolve(w_rev, current_h, mode='full')
        conv2 = np.convolve(w, current_h, mode='full')
        
        # Get center portions
        start_idx = n_points - 1
        direction = conv1[start_idx:start_idx+n_points] + conv2[start_idx:start_idx+n_points] - 1.0
        
        # Update: we want to move in direction of negative gradient
        new_h = current_h + step_size * direction
        
        # Project back to feasible space
        new_h = np.clip(new_h, 0, 1)
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        # Evaluate new state
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            current_h = new_h.copy()
        else:
            current_h = new_h.copy()
        
        # Adjust parameters
        if iteration % 100 == 0:
            alpha = min(alpha * alpha_increase_factor, alpha_max)
            step_size *= 0.95
    
    # Phase 3: Final refinement with local search
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Try small stochastic changes
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        # Small change
        h_new[idx] += np.random.uniform(-0.02, 0.02)
        # Project to valid range and sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
