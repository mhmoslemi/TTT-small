def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):

    return dx * np.max(corr)

def evaluate_c5(h):

    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_for_shift(h, shift_ell):
    """Compute gradient of C_ell = sum h_i (1 - h_{i+ell}) w.r.t. h"""
    n = len(h)

    if shift_ell >= 0:
        # Gradient for shift ell >= 0
        # For i in [0, n-ell-1]: h_i contributes 1 - h_{i+ell}
        # For i in [ell, n-1]: h_i contributes -h_{i-ell}
        if n - shift_ell > 0:
            grad[:n-shift_ell] += 1 - h[shift_ell:]
            grad[shift_ell:] -= h[:n-shift_ell]
    else:
        # Gradient for shift ell < 0 
        # Treat as shift -ell but symmetric:
        # C_{-ell} = sum_i h[i+ell] (1 - h[i]) 
        # So we consider i+ell and modify accordingly.
        sh = -shift_ell
        if n - sh > 0:
            grad[sh:] += 1 - h[:n-sh]
            grad[:n-sh] -= h[sh:]
    return grad
