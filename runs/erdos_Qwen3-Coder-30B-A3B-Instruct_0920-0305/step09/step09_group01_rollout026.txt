import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Use the same logic as the original code
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n / 2
    
    # Warm start - use the known construction as the starting point
    h0 = np.array(initial_h_values)
    # Ensure it has the right length
    if len(h0) < n:
        h0 = np.resize(h0, n)
    elif len(h0) > n:
        h0 = h0[:n]
        
    # Project it to sum=TARGET_SUM
    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum

        # Binary search for lambda
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        
        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
        return result
    
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Now try to do an integer programming formulation with CBC
    # We'll use cvxpy to create a feasible MILP and try to solve it

    # Direct local search & safeguard - binary search on values that don't 
    # directly involve integrality but maybe give us some insight into local structures.
    
    # Phase 1: Binary optimization through direct MIQP solve using cvxpy with CBC
    # Let's create binary variables and constraints for the exact formulation
    
    # Since we're dealing with binary fixings, we'll implement basic local search with swaps
    # and a greedy rounded approach if needed
    
    # Use the existing solution as a warm start
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Local search: try ups and downs with swaps on existing binary representation
    def local_search(h_start, max_iters=2000):
        h_current = h_start.copy()
        c5_current = evaluate_c5(h_current)
        best_h = h_current.copy()
        best_c5 = c5_current
        
        num_swaps = 0
        for i in range(max_iters):
            if time.time() - start_time > budget_s - 1:
                break
            # Create candidate swap by moving a '1' to a '0' position and vice versa
            ones = np.where(h_current > 0.9)[0]
            zeros = np.where(h_current < 0.1)[0]
            
            if len(ones) > 0 and len(zeros) > 0:
                # Pick a random 1 to switch to 0
                i_idx = np.random.choice(ones)
                # Pick a random 0 to switch to 1
                j_idx = np.random.choice(zeros)
                
                # Make a temporary version and evaluate C5
                h_trial = h_current.copy()
                h_trial[i_idx] = 0
                h_trial[j_idx] = 1
                
                # Project it back to ensure the sum is correct
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                if c5_trial < c5_current:
                    h_current = h_trial
                    c5_current = c5_trial
                    num_swaps += 1
                    
        return best_h, best_c5
    
    # Run local search with the warm start
    h_refined, c5_refined = local_search(best_h.copy(), max_iters=2000)
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
    
    # Try to improve by another optimization phase with SLSQP
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass

    # Final polishing with swaps
    h_final = best_h.copy()
    best_final_c5 = best_c5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Find 1s and 0s
        ones = np.where(h_final > 0.9)[0]
        zeros = np.where(h_final < 0.1)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            i_idx = np.random.choice(ones)
            j_idx = np.random.choice(zeros)
            
            # Try swap
            h_trial = h_final.copy()
            h_trial[i_idx] = 0
            h_trial[j_idx] = 1
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_final_c5:
                best_final_c5 = c5_trial
                h_final = h_trial
        else:
            break
    
    return (h_final.tolist(), float(best_final_c5), n)
