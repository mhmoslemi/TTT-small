import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_shifted_gradients(h, k):
    """Compute gradient for a specific shift k"""
    grad = np.zeros_like(h)
    if k >= 0:
        # grad[:n-k] += 1 - h[k:]  
        # grad[k:] -= h[:n-k] 
        if k < len(h):
            grad[:len(h)-k] += 1 - h[k:]
        if k < len(h):
            grad[k:] -= h[:len(h)-k]
    else:
        # For negative k, swap roles
        k = -k
        if k < len(h):
            grad[k:] += 1 - h[:len(h)-k]
        if k < len(h):
            grad[:len(h)-k] -= h[k:]
    return grad

def compute_weighted_gradient(h, shifts, weights):
    """Compute weighted gradient"""
    grad = np.zeros_like(h)
    for k, w in zip(shifts, weights):
        grad += w * compute_shifted_gradients(h, k)
    return grad

def compute_shift_values(h):
    """Compute all shift values c_k(h) efficiently using FFT"""
    corr = compute_corr_fft(h)
    # For a correlation of h with (1-h), we need to extract the shifts properly
    # Positions in the correlation array for shifts -n+1 to n-1
    # The correlation result's length is 2*n-1
    # Indices 0 to n-1 correspond to lags 0 to n-1
    # Indices n to 2*n-2 correspond to lags -n+1 to -1
    corr_vals = np.concatenate([corr[n-1:], corr[:n-1]])
    return corr_vals

def logsumexp_shifts(shift_values, mu):
    """Compute log-sum-exp of shift values with temperature mu"""
    max_val = np.max(shift_values)
    shifted = shift_values - max_val
    exp_vals = np.exp(shifted / mu)
    log_sum_exp = max_val + mu * np.log(np.sum(exp_vals))
    return log_sum_exp, exp_vals

def line_search_step(h, grad, c5_current, alpha_0=0.2*dx):
    """Perform line search for a good step size"""
    alpha = alpha_0
    max_halvings = 5
    for _ in range(max_halvings):
        h_new = project_to_sum_box(h - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_current:
            return h_new, c5_new, alpha
        alpha /= 2
    return h, c5_current, alpha

    best_h = h_current.copy()

    n = len(h_current)
    mu = mu_init

    shifts = list(range(-n+1, n))
    
    iteration_count = 0
    iterations_since_last_jitter = 0
    
    while mu >= mu_min and iteration_count < max_iter:
        # Compute all shift values
        shift_values = compute_shift_values(h_current)
        
        # Compute log-sum-exp and weights
        log_sum_exp, exp_vals = logsumexp_shifts(shift_values, mu)
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute weighted gradient
        grad = compute_weighted_gradient(h_current, shifts, weights)
        
        # Perform line search
        h_new, c5_new, _ = line_search_step(h_current, grad, c5_current)
        
        # Update variables
        h_current = h_new
        c5_current = c5_new
        
        # Update best if improved
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
        
        iteration_count += 1
        iterations_since_last_jitter += 1
        
        # Random jitter to escape flat regions
        if iterations_since_last_jitter >= jitter_freq:
            indices_to_perturb = np.random.choice(n, size=min(5, n//10), replace=False)
            for idx in indices_to_perturb:
                h_current[idx] += np.random.normal(0, 0.02)
            h_current = project_to_sum_box(h_current, TARGET_SUM)
            c5_current = evaluate_c5(h_current)
            iterations_since_last_jitter = 0
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        # Decrease mu every iterations_per_mu iterations, or skip if no improvement
        if iteration_count % iterations_per_mu == 0:
            if iteration_count % (2 * iterations_per_mu) == 0:
                mu = max(mu_min, mu / 2)
    
    # Final pairwise swap refinement

        if iteration_count > max_iter:
            break

        for i in range(n):
            if h_current[i] > epsilon:
                neighbor_indices = np.random.choice(n, size=2, replace=False)
                k = neighbor_indices[0]
                if k == i:
                    k = neighbor_indices[1] 
                if h_current[k] < 1 - epsilon:
                    # Try moving a small amount from i to k
                    delta = min(h_current[i] * 0.01, (1 - h_current[k]) * 0.01)
                    h_candidate = h_current.copy()
                    h_candidate[i] -= delta
                    h_candidate[k] += delta
                    h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                    c5_candidate = evaluate_c5(h_candidate)
                    if c5_candidate < c5_current:
                        h_current = h_candidate
                        c5_current = c5_candidate
                        improved = True
                        if c5_current < best_c5:
                            best_h = h_current.copy()
                            best_c5 = c5_current
        if not improved:
            break
    
    return best_h, best_c5
