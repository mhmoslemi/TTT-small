import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid size - must stay at 150 to not exceed the task scope
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n // 2
    
    # Create decision variables
    h = Binary(f"h", shape=(n,))
    y = Variable(f"y", shape=(n, n))  # Will project these to binary later if needed
    t = Variable(name="t")
    
    # Constraints for y being valid binary variables
    constraints = [
        sum(h) == TARGET_SUM,
        t >= 0,
    ]
    
    # Add constrain for all shifts k from 0 to n - 1
    for k in range(n):
        # For shift k, define this as correlation term
        # c_k = \sum_{i=0}^{n-k-1} h_i - \sum_{i=0}^{n-k-1} h_i * h_{i+k}
        # Using auxiliary variables y[i,k] := h[i] * h[i+k]
        left_sum = sum(h[i] for i in range(n - k))
        right_sum = sum(y[i,k] for i in range(n - k))
        c_k = left_sum - right_sum
        # Use t as the upper bound on c_k
        constraints += [t >= c_k]
    
    # Add linearization of product constraint: y[i,k] <= h[i], y[i,k] <= h[i+k], y[i,k] >= h[i]+h[i+k]-1
    for k in range(n):
        for i in range(n - k):
            constraints += [
                y[i,k] <= h[i],
                y[i,k] <= h[i+k],
                y[i,k] >= h[i] + h[i+k] - 1,
            ]
    
    # Optimize 
    prob = Problem(Minimize(t), constraints)
    
    # Solve with time limit
