def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_vector(h):
    """Compute the correlation vector for all shifts"""
    return np.correlate(h, 1 - h, mode='full')

def exact_C5(h):
    """Compute exact C5 using full correlation"""
    corr = correlation_vector(h)
    return dx * np.max(corr)

def compute_weighted_gradient(h, wgt):
    """Compute weighted gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k (represented as index in correlation vector)
    # Note: correlation vector has 2*n-1 elements
    # indices 0 to n-1 represent shifts -n+1 to 0
    # indices n to 2*n-2 represent shifts 1 to n-1  
    
    for k in range(1, n):  # shifts 1 to n-1
        # For shift k, h[i] contributes to h[i-k] in the correlation
        # This represents h[i] * h[i-k] in the original problem
        shifted_right = np.roll(h, k)
        shifted_left = np.roll(h, -k)
        grad += wgt[n-1+k] * (1 - shifted_right - shifted_left)
    
    # Add contribution for k=0 (self-correlation)
    grad += wgt[n-1] * (1 - 2 * h)  # No shift: contribution from h[i]^2
    
    return grad

def project_onto_box_and_sum(v, w):

    v = np.clip(v, 0, 1)
    
    # Sum correction via binary search
    def sum_diff(lam):
        return np.sum(np.clip(v + lam, 0, 1)) - w
    
    a, b = -1.0, 1.0
    while sum_diff(a) * sum_diff(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search for lambda
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        diff = sum_diff(m)
        if abs(diff) < tol:
            break
        if diff * sum_diff(a) < 0:
            b = m
        else:
            a = m
    
    # Apply the correction
    lam = (a + b) / 2
    v = np.clip(v + lam, 0, 1)
    return v
