def compute_all_shifts(h):
    """Compute all shift values using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_all_shifts(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, c_vals, mu):
    """Compute gradient using soft-max approximation"""
    n = len(h)
    # Compute weights
    max_c = np.max(c_vals)
    exp_vals = np.exp((c_vals - max_c) / mu)
    weights = exp_vals / np.sum(exp_vals)
    
    # Build gradient vector
    grad = np.zeros(n)
    for k in range(-n+1, n):
        if k >= 0:
            # k >= 0 case: h[i] * (1 - h[i+k])
            if k < n:
                grad[:n-k] += weights[k] * (1 - h[k:])
                grad[k:] -= weights[k] * h[:n-k]
        else:
            # k < 0 case: h[i] * (1 - h[i+k]) = h[i+k] * (1 - h[i])
            k_abs = -k
            if k_abs < n:
                grad[k_abs:] += weights[n-1+k] * (1 - h[:n-k_abs])
                grad[:n-k_abs] -= weights[n-1+k] * h[k_abs:]
    
    return grad

def line_search(h, grad, mu, max_trials=5):
    """Perform line search with adaptive step size"""
    alpha = 0.2 * dx
    h_current = h.copy()
    
    for i in range(max_trials):
        h_new = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h_current) or i == 0:
            return h_new, c5_new
        alpha *= 0.5
    
    return h_current, evaluate_c5(h_current)
