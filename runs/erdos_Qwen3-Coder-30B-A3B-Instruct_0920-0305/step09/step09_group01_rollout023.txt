import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_all_shifts(h):
    """Compute all shift values c_k(h) via FFT"""
    corr = compute_corr_fft(h)
    # The full correlation result has length 2*n-1, but we want only the valid shifts
    # which are from -n+1 to n-1. The middle n elements of the full correlation
    # give us the proper shift values.
    c_vals = corr[(n_points-1):(3*n_points-2)]
    return c_vals

def build_gradient_from_weights(h, weights):
    """Build gradient using weights for each shift"""
    n = len(h)
    grad = np.zeros(n)
    for k in range(-n+1, n):
        if k >= 0:
            grad[:n-k] += weights[k+n-1] * (1 - h[k:])
            grad[k:] -= weights[k+n-1] * h[:n-k]
        else:
            # For k < 0, swap roles
            grad[:n+k] -= weights[-k+n-1] * h[-k:]
            grad[-k:] += weights[-k+n-1] * (1 - h[:n+k])
    return grad

def line_search(h, grad, weights, max_trials=5, alpha0=0.2*dx):
    """Perform line search with backtracking"""
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    alpha = alpha0
    for _ in range(max_trials):
        h_proposed = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_proposed = evaluate_c5(h_proposed)
        if c5_proposed < c5_current:
            return h_proposed, c5_proposed, alpha
        alpha *= 0.5
    
    return h_current, c5_current, alpha

    best_h = h_start_best.copy()
