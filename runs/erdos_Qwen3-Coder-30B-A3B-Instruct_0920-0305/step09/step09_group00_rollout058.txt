import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, 2*len(h)-1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights from g with temperature alpha"""
    g_shifted = g - np.max(g)
    exp_g = np.exp(alpha * g_shifted)
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of the surrogate objective"""
    # Two convolutions: p * h and p_rev * h
    n = len(h)
    
    # For p * h
    p_extended = np.concatenate([p, np.zeros(n-1)])
    h_extended = np.concatenate([h, np.zeros(n-1)])
    conv1 = ifft(fft(p_extended) * fft(h_extended)).real[:n]
    
    # For p_rev * h (reverse p)
    p_rev = p[::-1]
    p_rev_extended = np.concatenate([p_rev, np.zeros(n-1)])
    conv2 = ifft(fft(p_rev_extended) * fft(h_extended)).real[:n]
    
    return -(conv1 + conv2)

def project_onto_sum_box(x, target):
    """Euclidean projection onto box constraints with sum constraint"""
    # Clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    
    # Binary search for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(x_clipped + lam, 0, 1)) - target
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < tol:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x_clipped + (a+b)/2, 0, 1)
    return result

def c5_true(h):
    """Exact computation of C5 using np.correlate"""
    n = len(h)
    dx = 2.0 / n
    # Construct 1-h
    h_complement = 1 - h
    # Compute correlations
    corr_full = np.correlate(h, h_complement, mode="full")
    # Extract valid correlations (excluding padding)
    mid = len(corr_full) // 2
    corr_valid = corr_full[mid - (n-1):mid + n]
    return dx * np.max(corr_valid)
