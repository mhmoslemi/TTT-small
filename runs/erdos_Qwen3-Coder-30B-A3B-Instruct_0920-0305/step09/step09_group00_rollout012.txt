import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a better initial construction
    # Using a piecewise constant function that tries to avoid overlap
    h0 = np.zeros(n_points)
    
    # Create a better initial solution - piecewise constant with varying heights
    # Try to have a construction with low overlap
    for i in range(n_points//3):
        h0[i] = 0.5
    for i in range(n_points//3, 2*n_points//3):
        h0[i] = 0.0
    for i in range(2*n_points//3, n_points):
        h0[i] = 0.5
    
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try multiple optimization approaches
    # Approach 1: Gradient-based with restarts
    for restart in range(5):
        if time.time() - start_time > budget_s:
            break
            
        # Random perturbation of initial solution
        h_perturbed = h0.copy() + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_opt = project_to_sum_box(res.x, TARGET_SUM)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
    
    # Approach 2: Evolutionary strategy with mutation
    if time.time() - start_time < budget_s:
        # Create a population of solutions
        population_size = 20
        population = []
        
        for i in range(population_size):
            h = h0.copy()
            # Add noise
            h += np.random.normal(0, 0.02, n_points)
            h = project_to_sum_box(h, TARGET_SUM)
            population.append(h)
        
        # Evaluate population
        fitness = [evaluate_c5(h) for h in population]
        
        # Evolution loop
        for gen in range(100):
            if time.time() - start_time > budget_s:
                break
                
            # Select best
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = fitness[best_idx]
            
            # Create new population
            new_population = [population[best_idx]]  # Keep best
            
            # Add mutated versions
            for i in range(population_size - 1):
                parent = population[np.random.randint(0, population_size)]
                child = parent.copy()
                # Mutation: add random noise
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
                child = project_to_sum_box(child, TARGET_SUM)
                new_population.append(child)
            
            population = new_population
            fitness = [evaluate_c5(h) for h in population]
    
    # Final polish with local optimization
    if time.time() - start_time < budget_s:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = project_to_sum_box(res.x, TARGET_SUM)
                best_c5 = evaluate_c5(best_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
