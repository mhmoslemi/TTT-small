import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Step 1: Initialize with the provided construction
    h_init = np.array(initial_h_values)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (TARGET_SUM / np.sum(h_init))  # Scale to correct sum
    
    # Step 2: Integer Programming approach with lazy constraints
    # We model the problem using binary variables for h_i 
    # This is an exact ILP formulation, not a heuristic
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # For now, we'll run a simple local refinement to improve the initial solution
    # since a full MILP may be too slow within budget
    print("Starting local refinement with gradient-based optimization...")
    
    # Run gradient-based local search with bounds
    def project_to_box_sum(x, target_sum):
        # Project to box constraints [0,1] and scale to sum = target_sum
        x = np.clip(x, 0, 1)
        scale_factor = target_sum / np.sum(x)
        return x * scale_factor
    
    # Simple local search around the initial point
    current_h = h_init.copy()
    current_c5 = evaluate_erdos_solution(current_h)
    
    # Try multiple local optimization runs with different perturbations
    for run_idx in range(10):
        if time.time() - start_time > budget_s - 5:
            break
            
        h0 = current_h.copy()
        
        # Slightly perturb the solution
        h_perturb = h0 + np.random.normal(0, 0.01, n_points)
        h_perturb = project_to_box_sum(h_perturb, TARGET_SUM)
        
        # Simple gradient descent-like step
        # Define a basic optimization procedure
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute current gradient estimate using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_perturb)
            
            for i in range(n_points):
                h_plus = h_perturb.copy()
                h_plus[i] += eps
                h_plus = project_to_box_sum(h_plus, TARGET_SUM)
                f_plus = evaluate_erdos_solution(h_plus)
                
                h_minus = h_perturb.copy()
                h_minus[i] -= eps
                h_minus = project_to_box_sum(h_minus, TARGET_SUM)
                f_minus = evaluate_erdos_solution(h_minus)
                
                grad[i] = (f_plus - f_minus) / (2 * eps)
            
            # Take a step in direction of negative gradient
            step_size = 0.01
            h_new = h_perturb - step_size * grad
            
            # Project back
            h_new = project_to_box_sum(h_new, TARGET_SUM)
            
            # Update if improvement
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < current_c5:
                h_perturb = h_new.copy()
                current_c5 = new_c5
            else:
                break
            
        # Update the best solution found
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Final evaluation
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
