import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def mutate_large_changes(h, max_change, num_changes):
    h_try = h.copy()
    indices = np.random.choice(n_points, size=num_changes, replace=False)
    for idx in indices:
        change = np.random.uniform(-max_change, max_change)
        h_try[idx] += change
    h_try = project_to_sum_box(h_try, TARGET_SUM)
    c5_try = evaluate_c5(h_try)
    return h_try, c5_try

def mutate_stochastic_hill_climb(h, step_size, iterations):
    h_try = h.copy()
    for _ in range(iterations):
        idx = np.random.randint(0, n_points)
        original = h_try[idx]
        h_try[idx] += np.random.uniform(-step_size, step_size)
        h_try = project_to_sum_box(h_try, TARGET_SUM)
        c5_try = evaluate_c5(h_try)
        if c5_try >= evaluate_c5(h):
            h_try[idx] = original
    return h_try, evaluate_c5(h_try)

def mutate_peaks_and_valleys(h, strength, num_points):
    h_try = h.copy()
    # Find peaks and valleys
    for _ in range(num_points):
        idx = np.random.randint(0, n_points)
        # Determine if this is a peak or valley by neighborhood
        neighbors = []
        for i in range(-3, 4):
            if 0 <= idx + i < n_points and i != 0:
                neighbors.append(h_try[idx + i])
        if len(neighbors) > 0:
            neighbor_avg = np.mean(neighbors)
            if h_try[idx] > neighbor_avg:
                # Peak, try to reduce
                h_try[idx] = max(0, h_try[idx] - np.random.uniform(0, strength/2))
            else:
                # Valley, try to increase
                h_try[idx] = min(1, h_try[idx] + np.random.uniform(0, strength/2))
    h_try = project_to_sum_box(h_try, TARGET_SUM)
    c5_try = evaluate_c5(h_try)
    return h_try, c5_try
