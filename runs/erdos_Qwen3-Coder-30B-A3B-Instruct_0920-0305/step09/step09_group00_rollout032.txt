from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sim_anneal_hybrid(h0, c5_0, start_time, budget_s):
    """Hybrid Simulated Annealing with local refinement"""
    current_h = h0.copy()
    current_c5 = c5_0
    best_h = h0.copy()
    best_c5 = c5_0
    
    # Variable temperature schedule
    temp_start = 0.1
    temp_end = 0.0001
    cooling_rate = 0.9995
    max_iter = 20000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        temp = temp_start * (cooling_rate ** i)
        
        # Perturb solution
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        # More aggressive perturbation occasionally
        if np.random.random() < 0.3:
            delta = np.random.normal(0, 0.05)
        else:
            delta = np.random.normal(0, 0.01)
        h_new[idx] += delta
        
        # Project to valid range
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5

def gradient_descent_refinement(h0, c5_0, start_time, budget_s):
    """Refine with gradient-based optimization"""
    current_h = h0.copy()
    current_c5 = c5_0
    
    # Try multiple restarts with different random starts
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for restart in range(5):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Small random perturbation
        h_perturbed = current_h.copy() + np.random.normal(0, 0.005, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-13, 'gtol': 1e-13}
            )
            if res.success:
                h_result = res.x
            h_result = project_to_sum_box(h_result, TARGET_SUM)
            c5_result = evaluate_c5(h_result)
            
            if c5_result < best_c5:
                best_h = h_result
                best_c5 = c5_result
        except:
            continue
    
    return best_h, best_c5

def simple_ga(h0, c5_0, start_time, budget_s):
    """Simple Genetic Algorithm approach"""
    # Parameters
    pop_size = 30
    generations = 300
    mutation_rate = 0.1
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        h = h0 + np.random.normal(0, 0.02, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    best_h = h0.copy()
    best_c5 = c5_0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Find best in current generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            h1, h2 = selected[i], selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
            child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            
            # Mutation
            for h in [child1, child2]:
                if np.random.random() < mutation_rate:
                    mut_idx = np.random.randint(0, n_points)
                    h[mut_idx] += np.random.normal(0, 0.01)
                    h[mut_idx] = max(0, min(1, h[mut_idx]))
                new_population.append(project_to_sum_box(h, TARGET_SUM))
        
        population = new_population
    
    return best_h, best_c5
