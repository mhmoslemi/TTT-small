def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth max approximation with epsilon regularization"""
    # Compute full correlation
    corr = compute_corr_fft(h)
    
    # Numerically stable softmax of epsilon * corr
    corr_scaled = epsilon * corr
    max_corr = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - max_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute smooth max value
    smooth_max = (max_corr + np.log(np.sum(exp_corr))) / epsilon
    
    return smooth_max, weights

def grad_smooth_max_corr(h, weights):
    """Compute gradient of smooth max w.r.t h"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift, compute gradient contribution using correlation-based dense matrix
    # This makes derivation a bit more explicit than Direct computation
    for k in range(2*n - 1):
        shift_idx = k - (n - 1)
        if shift_idx >= 0:
            # Shift positive
            # The gradient contribution from shift k:
            # c_k = sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i)
            # Grad wrt h_j = (1 - h_{j-k}) if j >= k else 0 - h_{j+k} if j <= n-1-k else 0
            grad_component = np.zeros(n)
            if shift_idx > 0:
                grad_component[shift_idx:] += (1 - h[:-shift_idx])  # components from 1-h_{j-k}
                grad_component[:-shift_idx] -= h[shift_idx:]        # components from -h_{j+k}
            else:
                grad_component = 1 - 2*h  # shift == 0 case
            grad += weights[k] * grad_component
        else:
            # Shift negative: symmetric
            # For shift k < 0, let s = -k > 0, then c_{-s} = sum_i h_i * (1 - h_{i+s})
            # Grad wrt h_j = (1 - h_{j+s}) if j <= n-1-s else 0 - h_{j-s} if j >= s else 0
            abs_k = -shift_idx
            grad_component = np.zeros(n)
            if abs_k > 0:
                grad_component[:n-abs_k] += (1 - h[abs_k:])  # from (1-h_{j+s})
                grad_component[abs_k:] -= h[:-abs_k]         # from -h_{j-s}
            else:
                grad_component = 1 - 2*h  # shift == 0, same as above
            grad += weights[k] * grad_component
    
    return grad
