import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
beta_schedule = [0.5, 1, 2, 4, 8, 16, 32]

def compute_correlation(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def exact_c5(h):
    """Compute exact C5 value using correlation"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, weights):
    """Compute weighted gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # Create shifted versions
    for k in range(-n+1, n):
        if k >= 0:
            left = np.concatenate([np.zeros(k), h[:n-k]])
            right = np.concatenate([h[k:], np.zeros(k)])
        else:
            left = np.concatenate([h[-k:], np.zeros(-k)])
            right = np.concatenate([np.zeros(-k), h[:-k]])
        
        grad += weights[k+n-1] * (1 - left - right)
    
    return grad

def smoothed_objective(h, beta):
    """Compute the smoothed objective Φ_β(h)"""
    corr = compute_correlation(h)
    corr = corr * dx
    max_corr = np.max(corr)
    exp_terms = np.exp(beta * (corr - max_corr))
    return (1.0 / beta) * np.log(np.sum(exp_terms)) + max_corr / beta
