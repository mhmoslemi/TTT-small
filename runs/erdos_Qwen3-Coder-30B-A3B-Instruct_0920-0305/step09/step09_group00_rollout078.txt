import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    n = len(h)
    H = fft(h, 2*n)
    A = ifft(H * np.conj(H)).real[:n]
    return A

def softmax_weights(g, alpha):
    """Compute softmax weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of the surrogate using convolutions"""
    n = len(h)
    # Convolve with h (linear convolution)
    h_pad = np.pad(h, (0, n), mode='constant')
    p_pad = np.pad(p, (0, n), mode='constant')
    r1 = np.convolve(h_pad, p_pad, mode='valid')[:n]
    
    # Convolve with reversed h
    p_rev = p[::-1]
    r2 = np.convolve(h_pad, p_rev, mode='valid')[:n]
    
    return -(r1 + r2)

def project_onto_sum_box(x, target):
    """Project onto box constraints [0,1] with sum = target using water-filling"""
    # Clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    
    if abs(current_sum - target) < 1e-12:
        return x_clipped
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x_clipped + lam, 0, 1)
        return np.sum(projected) - target
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search (about 30 iterations)
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x_clipped + (a + b)/2, 0, 1)

def c5_true(h):
    """Compute exact C5 using correlation"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Optimisation parameters
    n_list = [64, 128, 256]
    alpha_start = 1.0
    alpha_max = 2000.0
    gamma = 1.3
    max_iter_per_alpha = 200
    eta_start = 0.01
    n_points = 128
    
    # Initial construction using Thue-Morse pattern
    h_tm = np.array([bin(i).count('1') % 2 for i in range(n_points)])
    target_sum = n_points / 2
    h_tm = project_onto_sum_box(h_tm, target_sum)
    
    # Perform coarse-to-fine optimization
    best_h = h_tm.copy()
    best_c5 = c5_true(h_tm)
    
    # Multi-resolution approach
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Set up for this resolution
        dx = 2.0 / n
        target_sum = n / 2
        
        # Interpolate from previous result if available
        if n > n_list[0]:
            h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(best_h)), best_h)
            h_interp = project_onto_sum_box(h_interp, target_sum)
        else:
            h_interp = best_h.copy()
            
        # Annealing with soft maximum surrogate
        h = h_interp.copy()
        alpha = alpha_start
        eta = eta_start
        
        while alpha <= alpha_max and time.time() - start_time < budget_s * 0.95:
            for it in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute autocorrelations
                A = autocorr_fft(h)
                g = -A  # negative autocorrelations
                
                # Compute softmax weights
                p = softmax_weights(g, alpha)
                
                # Compute gradient
                grad = gradient(h, p)
                
                # Update
                h_new = h - eta * grad
                h_new = project_onto_sum_box(h_new, target_sum)
                
                # Evaluate surrogate and update
                A_new = autocorr_fft(h_new)
                g_new = -A_new
                
                # Current surrogate value
                surrogate = np.log(np.sum(np.exp(alpha * g))) / alpha
                surrogate_new = np.log(np.sum(np.exp(alpha * g_new))) / alpha
                
                if surrogate_new < surrogate:
                    h = h_new.copy()
                    
                    # Check true C5
                    current_c5 = c5_true(h)
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        
            # Increase temperature
            alpha *= gamma
            # Reduce learning rate
            eta *= 0.9
            
        # Store best for next resolution
        best_h = h.copy()
        best_c5 = c5_true(h)
        
    # Discrete hill-climb improvement
    for _ in range(5000):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Random swap
        i, j = np.random.randint(0, len(best_h), 2)
        if i == j:
            continue
            
        # Swap values
        h_trial = best_h.copy()
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        
        # Project to sum constraint
        h_trial = project_onto_sum_box(h_trial, target_sum)
        
        # Evaluate
        c5_trial = c5_true(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final polishing with SLSQP
    if time.time() - start_time < budget_s * 0.995:
        def objective(h):
            return c5_true(h)
            
        def constraint_eq(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(len(best_h))]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            best_h = project_onto_sum_box(best_h, target_sum)
            best_c5 = c5_true(best_h)
        except:
            pass
            
    return (best_h.tolist(), float(best_c5), n_points)
