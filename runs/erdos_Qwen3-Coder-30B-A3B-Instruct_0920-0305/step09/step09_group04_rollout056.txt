import numpy as np

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def create_lp_model(n_points):
    """Create the LP model for the problem"""
    dx = 2.0 / n_points
    
    # Decision variables
    h = cp.Variable(n_points, bounds=(0, 1))
    t = cp.Variable()
    
    # Create z variables for products h_i * h_j
    # We only need upper triangular matrix, so z[i,j] = z[j,i] for i<j
    # But cvxpy handles symmetric matrices well, so just create full matrix
    z = cp.Variable((n_points, n_points), symmetric=True)
    
    # Constraints - McCormick envelopes for z[i,j] = h[i]*h[j]
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(i + 1, n_points):  # Only upper triangular
            constraints.append(z[i, j] <= h[i])
            constraints.append(z[i, j] <= h[j])
            constraints.append(z[i, j] >= h[i] + h[j] - 1)
            constraints.append(z[i, j] >= 0)
            # Also ensure z[i,j] = z[j,i] but cvxpy will handle by symmetry
    
    # Restore diagonal symmetry in z
    for i in range(n_points):
        constraints.append(z[i, i] == h[i])
    
    # Overlap constraints
    # For each shift k, compute sum_{j=0}^{n-1-k} (h_{j+k} - z_{j,j+k})
    for k in range(1, n_points):
        overlap_expr = 0
        for j in range(n_points - k):
            overlap_expr += h[j + k] - z[j, j + k]
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    return prob, h, t, z

def round_to_binary(h_frac, n_points):
    """Round fractional solution to binary while maintaining sum"""
    # Use a relaxed greedy approach to round
    # Order indices by h_frac values
    sorted_indices = np.argsort(-h_frac)
    
    # Initialize binary vector
    h_bin = np.zeros(n_points)
    ones_count = 0
    
    # Assign each index a value according to its fractional value
    # Process in order of decreasing h_frac
    for i in sorted_indices:
        if ones_count < n_points / 2:
            # Decide to assign 1 if we can do so to stay under target
            # (The fractional assignment is closer to a probability)
            h_bin[i] = 1
            ones_count += 1
        else:
            # Should not assign 1 anymore, do 0
            h_bin[i] = 0
    
    # Adjusting if we exceeded or fell short
    # If too many 1s, correct by setting some 1s to 0
    while ones_count > n_points / 2:
        # Find a 1 that can be set to 0
        for i in range(n_points):
            if h_bin[i] == 1:
                h_bin[i] = 0
                ones_count -= 1
                break
    
    # If too few 1s, correct by setting some 0s to 1
    while ones_count < n_points / 2:
        # Find a 0 that can be set to 1
        for i in range(n_points):
            if h_bin[i] == 0:
                h_bin[i] = 1
                ones_count += 1
                break
                
    return h_bin

        np.random.shuffle(ones)

        corr = np.correlate(best_h, 1 - best_h, mode='full')

                # Try the swap

                new_h[i] = 0
                new_h[j] = 1
                
                # Test the new overlap after swap (this is expensive)

                    new_c5 = evaluate_c5(new_h)
