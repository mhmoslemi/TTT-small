# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    ]

        strategy = np.random.choice(strategies)
        indices = strategy()
        
        # Make large perturbations
        neighbor_h = current_h.copy()
        for idx in indices:
            # Try different types of changes
            change_type = np.random.choice(['small', 'medium', 'large'])
            if change_type == 'small':
                delta = np.random.uniform(-0.05, 0.05)
            elif change_type == 'medium':
                delta = np.random.uniform(-0.1, 0.1)
            else:  # large
                delta = np.random.uniform(-0.2, 0.2)
            
            neighbor_h[idx] += delta
            
        # Project back to constraints
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        # Accept the move if it improves the solution
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
        elif np.random.random() < 0.05:  # Occasionally accept worse solutions
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            
        step += 1
        
        # Periodic refinement
        if step % 5000 == 0 and time.time() - start_time < budget_s:
            # Try to refine with local optimization
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - TARGET_SUM}
                res = minimize(
                    evaluate_c5, 
                    current_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
                )
                if res.success:
                    refined_h = res.x
                    refined_h = project_to_sum_box(refined_h, TARGET_SUM)
                    refined_c5 = evaluate_c5(refined_h)
                    if refined_c5 < current_c5:
                        current_h = refined_h
                        current_c5 = refined_c5
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
            except:
                pass
    
    return best_h, best_c5
