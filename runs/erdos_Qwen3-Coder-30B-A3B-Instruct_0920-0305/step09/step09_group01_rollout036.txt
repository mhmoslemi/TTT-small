import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def evaluate_c5_exact(h):
    """Evaluate C5 using the exact formula"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n // 2
    
    # Build the MILP model
    # Decision variables
    h = Variable(n, boolean=True)
    y = Variable((n, n), boolean=True)
    t = Variable()
    
    # Constraints
    constraints = [
        sum(h) == TARGET_SUM,  # Exactly half ones
    ]
    
    # Linearization constraints for y[i,k] = h[i] AND h[i+k]
    for k in range(n):
        for i in range(n - k):
            constraints.extend([
                y[i, k] <= h[i],
                y[i, k] <= h[i + k],
                y[i, k] >= h[i] + h[i + k] - 1
            ])
    
    # Epigraph constraints
    for k in range(n):
        # Correlation for shift k
        c_k = sum(h[i] for i in range(n - k)) - sum(y[i, k] for i in range(n - k))
        constraints.append(t >= c_k)
    
    # Objective
    objective = Minimize(t)
    
    # Create the problem
    prob = Problem(objective, constraints)
    
    # Solve the MILP
    try:
        prob.solve(solver=CPLEX, time_limit=budget_s/2, mipgap=0.001)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            best_h = h.value.astype(int)
            best_c5 = evaluate_c5_exact(best_h)
        else:
            # Fallback: Solve the LP relaxation
            h_relax = Variable(n, boolean=False)
            y_relax = Variable((n, n), boolean=False)
            t_relax = Variable()
            
            # Relax integrality constraints
            constraints_relax = [
                sum(h_relax) == TARGET_SUM,
            ]
            
            # Linearization constraints for relaxed variables
            for k in range(n):
                for i in range(n - k):
                    constraints_relax.extend([
                        y_relax[i, k] <= h_relax[i],
                        y_relax[i, k] <= h_relax[i + k],
                        y_relax[i, k] >= h_relax[i] + h_relax[i + k] - 1
                    ])
            
            # Epigraph constraints
            for k in range(n):
                c_k = sum(h_relax[i] for i in range(n - k)) - sum(y_relax[i, k] for i in range(n - k))
                constraints_relax.append(t_relax >= c_k)
            
            objective_relax = Minimize(t_relax)
            prob_relax = Problem(objective_relax, constraints_relax)
            
            # This is just a fast check - we keep the integer solution from above
            best_h = h.value.astype(int) if h.value is not None else np.zeros(n, dtype=int)
            best_c5 = evaluate_c5_exact(best_h)
    except:
        # Fallback: Use a random construction
        best_h = np.zeros(n, dtype=int)
        ones_indices = np.random.choice(n, TARGET_SUM, replace=False)
        best_h[ones_indices] = 1
        best_c5 = evaluate_c5_exact(best_h)
    
    # Local optimization with pair-swapping
    def compute_correlations(hv):
        corr = correlate(hv, 1 - hv, mode='full')
        return corr
    
    def evaluate_current_c5(hv):
        corr = compute_correlations(hv)
        return np.max(corr) * dx
    
    current_h = best_h.copy()
    current_c5 = evaluate_current_c5(current_h)
    
    # Pair swap local search
    num_swaps = 0
    while time.time() - start_time < budget_s - 5:
        improved = False
        # Find 1s and 0s
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
        
        for i in ones_indices:
            for j in zeros_indices:
                if i >= n or j >= n: continue
                
                # Try the swap
                test_h = current_h.copy()
                test_h[i] = 0
                test_h[j] = 1
                
                new_c5 = evaluate_current_c5(test_h)
                if new_c5 < current_c5:
                    current_h = test_h
                    current_c5 = new_c5
                    improved = True
                    num_swaps += 1
                    break
            if improved:
                break
                
        if not improved:
            break
            
    return (current_h.tolist(), float(current_c5), n)
