import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n // 2
    
    # Build the MILP model
    # Variables
    h = Binary(n, name='h')
    y = Variable((n, n), name='y', boolean=True)
    t = Variable(name='t')
    
    # Constraints for y (binary AND constraints)
    constraints = []
    for k in range(n):
        for i in range(n - k):
            # y[i,k] = h[i] AND h[i+k]
            constraints.append(y[i, k] <= h[i])
            constraints.append(y[i, k] <= h[i + k])
            constraints.append(y[i, k] >= h[i] + h[i + k] - 1)
    
    # Constraint that exactly half entries are 1
    constraints.append(sum(h) == TARGET_SUM)
    
    # Define correlation values
    # We only consider non-negative shifts (since correlation is symmetric)
    for k in range(n):
        # c_k = sum(h_i) - sum(y_i,k) for i from 0 to n-k-1
        # But sum(h_i) is a constant for fixed k, and sum(y_i,k) is a linear function
        sum_h_k = sum(h[i] for i in range(n - k))
        sum_y_k = sum(y[i, k] for i in range(n - k))
        c_k = sum_h_k - sum_y_k
        constraints.append(t >= c_k)
    
    # Objective
    objective = Minimize(t)
    
    # Solve the MILP model
    prob = Problem(objective, constraints)
    prob.solve(solver=CPLEX, time_limit=budget_s/2, verbose=False)
    
    # Check solution status
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        # Extract binary solution
        h_sol = np.array([h[i].value for i in range(n)])
        # Check if solution is valid (should be binary)
        if np.allclose(h_sol, np.round(h_sol)):
            h_sol = np.round(h_sol).astype(int)
        else:
            # Fall back to LP relaxation  
            h_sol = np.array([h[i].value for i in range(n)])
            # Greedy rounding
            sort_indices = np.argsort(h_sol)[::-1]
            h_sol = np.zeros(n)
            h_sol[sort_indices[:TARGET_SUM]] = 1
        h_sol = h_sol.astype(int)
    else:
        # Fallback: start with simple construction and improve
        h_sol = np.zeros(n)
        h_sol[:TARGET_SUM] = 1
    # Evaluate the final solution
    best_h = h_sol
    best_c5 = max(correlate(h_sol, 1 - h_sol, mode="full")) * dx
    
    # Optional polishing: pair swap heuristic
    def compute_corr(h):
        corr = correlate(h, 1 - h, mode="full")
        return corr * dx
    
    corr_values = compute_corr(h_sol)
    best_c5 = np.max(corr_values)
    h_current = h_sol.copy()
    improved = True
    while improved and (time.time() - start_time < budget_s - 1):
        improved = False
        # For each 1-bit, try to move it to a better position (0-bit)
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in range(len(ones)):
            if time.time() - start_time > budget_s - 1:
                break
            idx1 = ones[i]
            # Try moving it to each 0 position
            corr_orig = compute_corr(h_current)
            orig_max = np.max(corr_orig)
            for j in range(len(zeros)):
                if time.time() - start_time > budget_s - 1:
                    break
                idx0 = zeros[j]
                # Clone and swap
                h_test = h_current.copy()
                h_test[idx1] = 0
                h_test[idx0] = 1
                corr_test = compute_corr(h_test)
                test_max = np.max(corr_test)
                if test_max < orig_max:
                    h_current = h_test
                    improved = True
                if improved:
                    break
            if improved:
                break
        # Recompute max correlation for current solution
        corr_values = compute_corr(h_current)
        best_c5 = np.max(corr_values)
    
    return (h_current.tolist(), float(best_c5), n)
