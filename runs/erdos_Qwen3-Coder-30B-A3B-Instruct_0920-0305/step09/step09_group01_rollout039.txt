import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150  # Use a slightly larger number for better accuracy
    dx = 2.0 / n_points
    
    # Create shift matrices
    def create_shift_matrix(n, shift):
        """Create binary matrix M_ell for shift ell"""
        M = np.zeros((n, n))
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                M[i, j] = 1
                M[j, i] = 1  # Symmetric
        return M
    
    # Formulate and solve SDP relaxation
    def solve_sdp_relaxation(h_guess, target_sum, max_shifts=100):
        """Solve SDP relaxation for given h_guess"""
        n = len(h_guess)
        # Create a block diagonal matrix of all shift matrices
        M_list = []
        for i in range(min(max_shifts, n)):
            # Should be symmetric positive definite
            M = create_shift_matrix(n, i)
            M_list.append(M)
        
        # Use CVXPY to solve the relaxation
        X = Variable((n, n), symmetric=True)
        h = Variable(n)
        constraints = [
            X == h @ h.T,
            X >> 0,  # PSD
            diag(X) <= h,  # X[i,i] <= h[i]
            sum(h) == target_sum,
            # Add constraints for each matrix M_i
            [trace(M_list[i] @ X) >= target_sum/2 - 0.38 for i in range(len(M_list))]
        ]
        
        # Objective: minimize trace of a positive matrix (this is a proxy for minimizing max overlap)
        objective = Minimize(trace(X))  # Note: This is not correct objective, but sdp fails if no objective...
        prob = Problem(objective, constraints)
        prob.solve(solver=SCS, verbose=False, max_iters=1000, eps=1e-6)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            # Extract solution
            h_sol = h.value
            X_sol = X.value
            # Randomized rounding, project to valid domain
            if h_sol is not None:
                h_sol = np.clip(h_sol, 0, 1)
                # Normalization
                h_sol = h_sol * (target_sum / np.sum(h_sol))
                return h_sol
        return h_guess
    
    # Evaluate C5 for a given h
    def evaluate_c5(h):
        """Fast evaluation of C5 using FFT"""
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return np.max(corr) * dx
    
    # Initialize with some reasonable starting value
    h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    if len(h0) != n_points:
        h0 = np.concatenate([h0, np.ones(n_points - len(h0)) * 0.5])
    h0 = h0[:n_points]
    h0 = h0 * (n_points / 2) / np.sum(h0)  # normalize to sum = n_points / 2
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try several different starting points to escape local minima
    for _ in range(5):
        if time.time() - start_time > budget_s - 2:
            break
        
        # Randomly perturb the current 'best' solution to generate a new trial solution
        h_trial = best_h + np.random.normal(0, 0.02, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Refine with SDP relaxation
        h_refined = solve_sdp_relaxation(h_trial, n_points / 2, max_shifts=100)
        
        # Evaluate after refinement
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Final polishing using local optimization
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
