from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_simulated_annealing(h0, c5_0, start_time, budget_s):
    """Adaptive simulated annealing with multiple temperature schedules"""
    current_h = h0.copy()
    current_c5 = c5_0
    best_h = h0.copy()
    best_c5 = c5_0
    
    # Adaptive cooling schedule
    temp = 0.1
    decay_factor = 0.9995
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb solution
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.03)
        h_new[idx] += delta
        
        # Project to valid range
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Adaptive cooling
        temp *= decay_factor
        
        # Occasionally reset temperature for escape from local minima
        if i % 1000 == 0 and temp < 0.01:
            temp = 0.1
            
    return best_h, best_c5

def gradient_restarts(h0, c5_0, start_time, budget_s):
    """Gradient-based optimization with random restarts"""
    best_h = h0.copy()
    best_c5 = c5_0
    
    # Random restarts
    for restart in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Start with perturbed initial solution
        h_perturbed = h0 + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                h_result = res.x
                h_result = project_to_sum_box(h_result, TARGET_SUM)
                c5_result = evaluate_c5(h_result)
                if c5_result < best_c5:
                    best_h = h_result
                    best_c5 = c5_result
        except:
            continue
    
    return best_h, best_c5

    best_h = h0.copy()
    best_c5 = c5_0
    
    max_iter = 3000
    
    for i in range(max_iter):

        # Try several types of perturbations
        perturbation_type = np.random.choice(['single', 'group', 'wave'])
        
        h_new = current_h.copy()
        
        if perturbation_type == 'single':
            # Single point perturbation
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
            
        elif perturbation_type == 'group':
            # Group perturbation
            start_idx = np.random.randint(0, n_points - 5)
            for j in range(5):
                idx = start_idx + j
                delta = np.random.normal(0, 0.02)
                h_new[idx] += delta
                
        else:  # wave
            # Wave-like perturbation
            indices = np.random.choice(n_points, size=min(20, n_points//4), replace=False)
            wave_amp = np.random.uniform(0.01, 0.05)
            for idx in indices:
                phase = np.random.uniform(0, 2*np.pi)
                h_new[idx] += wave_amp * np.sin(2*np.pi*idx/n_points + phase)

        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new)

            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5
