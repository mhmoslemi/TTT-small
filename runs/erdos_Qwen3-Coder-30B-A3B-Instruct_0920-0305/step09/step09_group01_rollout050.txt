# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_c5(h, shift_ℓ):
    """Compute gradient of C_ℓ w.r.t h"""
    n = len(h)
    grad = np.zeros(n)
    
    # For shift ℓ:
    # C_ℓ = sum_{i: i+ℓ in [0,n-1]} h_i (1 - h_{i+ℓ})
    # = sum_{i: i+ℓ in [0,n-1]} h_i - sum_{i: i+ℓ in [0,n-1]} h_i h_{i+ℓ}
    
    # Gradient with respect to h_i:
    # ∂C_ℓ/∂h_i = 1 - h_{i+ℓ} (if i in valid range for first term)
    #            - h_{i-ℓ} (if (i-ℓ) in valid range for second term)
    
    if shift_ℓ >= 0:
        # First range: i in [0, n-shift_ℓ-1]
        i_valid = np.arange(0, n - shift_ℓ)
        grad[i_valid] += 1 - h[shift_ℓ:]
        # Second range: i in [shift_ℓ, n-1] 
        i_valid2 = np.arange(shift_ℓ, n)
        grad[i_valid2] -= h[:n-shift_ℓ]
    else:
        # shift_ℓ < 0
        shift_ℓ_abs = -shift_ℓ
        # First range: i in [0, n-shift_ℓ_abs-1] 
        i_valid = np.arange(0, n - shift_ℓ_abs)
        grad[i_valid] += 1 - h[shift_ℓ_abs:]
        # Second range: i in [shift_ℓ_abs, n-1]
        i_valid2 = np.arange(shift_ℓ_abs, n)
        grad[i_valid2] -= h[:n-shift_ℓ_abs]
    
    return grad
