def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_step(h, ε, α, m, v, β1, β2, adam_eps):
    """Compute one gradient step with momentum"""
    # Compute all c_k values efficiently via FFT
    corr = compute_corr_fft(h)
    # Convert to vector of all c_k values (for k = -(n-1) to n-1)
    c_vals = corr[0:2*n_points-1] * dx
    
    # Softmax weights
    c_vals_scaled = c_vals / ε
    c_vals_scaled -= np.max(c_vals_scaled)  # For numerical stability
    exp_c = np.exp(c_vals_scaled)
    w = exp_c / np.sum(exp_c)

    # Compute ∇c_k for each possible shift k
    grad = np.zeros_like(h)
    for i in range(n_points):
        for k in range(-(n_points-1), n_points):
            if k >= 0:
                if i + k < n_points:
                    grad[i] += w[k + (n_points-1)] * (1 - h[i+k])
                if i - k >= 0:
                    grad[i] -= w[-k + (n_points-1)] * h[i-k]
            else:
                # k < 0
                if i - k < n_points:
                    grad[i] -= w[-k + (n_points-1)] * h[i-k]
                if i + k >= 0:
                    grad[i] += w[k + (n_points-1)] * (1 - h[i+k])
    
    # Update momentum
    m = β1 * m + (1-β1) * grad
    v = β2 * v + (1-β2) * grad**2
    m_hat = m / (1 - β1**(len(h) + 1))
    v_hat = v / (1 - β2**(len(h) + 1))
    
    # Update h
    h_new = h - α * m_hat / (np.sqrt(v_hat) + adam_eps)
    
    return h_new, m, v
