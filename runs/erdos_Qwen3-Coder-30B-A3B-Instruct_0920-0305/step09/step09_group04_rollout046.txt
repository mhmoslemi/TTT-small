```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 120
    dx = 2.0 / n_points
    
    # Solve LP relaxation
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Create z_{ij} variables for McCormick relaxation
    # Use upper triangular part only for efficiency
    z_vars = {}
    for i in range(n_points):
        for j in range(i+1, n_points):
            z_vars[(i,j)] = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(i+1, n_points):
            z = z_vars[(i,j)]
            constraints.append(z <= h[i])
            constraints.append(z <= h[j])
            constraints.append(z >= h[i] + h[j] - 1)
            constraints.append(z >= 0)
    
    # Overlap constraints for each shift k
    for k in range(1, n_points):
        # c_k = sum_{i=0}^{n-1-k} (h_{i+k} - z_{i,i+k})
        expr = 0
        for i in range(n_points - k):
            expr += h[i + k] - z_vars[(i, i + k)]
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver options for faster solution
    prob.solve(solver=ECOS, verbose=False, 
               feastol=1e-7, abstol=1e-7, reltol=1e-7)
    
    if prob.status != cp.OPTIMAL:
        raise ValueError("LP solver did not converge to optimal solution")
    
    h_lp = h.value
    t_lp = t.value
    
    # Rounding and refinement
    best_h = None
    best_c5 = float('inf')
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time < 30:  # Not enough time for meaningful work
        # Return rough approximation from LP
        h_final = np.clip(h_lp, 0, 1)
        h_final = h_final / np.sum(h_final) * (n_points / 2)
        c5_final = np.max(np.correlate(np.round(h_final), 1 - np.round(h_final), mode="full")) * dx
        return (h_final.tolist(), float(c5_final), n_points)
    
    # Balanced rounding using conditional expectation
    def round_h(h_vec, target_sum):
        h_vec = np.clip(h_vec, 0, 1)
        result = np.zeros_like(h_vec)
        current_sum = 0
        remaining = target_sum
        remaining_indices = list(range(len(h_vec)))
        np.random.shuffle(remaining_indices)
        for i in remaining_indices:
            if remaining <= 0:
                break
            p = h_vec[i]
            if np.random.random() < p and current_sum < target_sum:
                result[i] = 1
                current_sum += 1
                remaining -= 1
        # Ensure exact sum by adjusting
        one_positions = np.where(result == 1)[0]
        if len(one_positions) > target_sum:
            # Remove some 1s
            to_remove = len(one_positions) - int(target_sum)
            remove_indices = np.random.choice(one_positions, to_remove, replace=False)
            result[remove_indices] = 0
        elif len(one_positions) < target_sum:
            # Add some 1s
            zero_positions = np.where(result == 0)[0]
            to_add = int(target_sum) - len(one_positions)
            add_indices = np.random.choice(zero_positions, to_add, replace=False)
            result[add_indices] = 1
        return result
    
    # Multiple rounds of rounding and evaluation
    n_rounds = min(2000, int(remaining_time * 50))  # Distribute with generous allocation
    for _ in range(n_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        h_rounded = round_h(h_lp, n_points / 2)
        if np.sum(h_rounded) != n_points / 2:
            # Ensuring exact mass via binary swap (corrective)
            one_indices = np.where(h_rounded == 1)[0]
            zero_indices = np.where(h_rounded == 0)[0]
            target_ones = n_points // 2
            if len(one_indices) > target_ones:
                num_to_flip = len(one_indices) - target_ones
                flip_indices = np.random.choice(one_indices, num_to_flip, replace=False)
                for idx in flip_indices:
                    h_rounded[idx] = 0
            elif len(one_indices) < target_ones:
                num_to_flip = target_ones - len(one_indices)
                flip_indices = np.random.choice(zero_indices, num_to_flip, replace=False)
                for idx in flip_indices:
                    h_rounded[idx] = 1
        
        # Evaluate exact C5
        corr = np.correlate(h_rounded, 1 - h_rounded, mode="full")
        c5_rounded = dx * np.max(corr)
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = h_rounded.copy()
    
    if best_h is None:
        # Fallback to LP solution
        best_h = np.clip(h_lp, 0, 1)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_c5 = dx * np.max(corr)
    
    # Local deterministic refinement with swapping
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        # Simple local improvement: perform swaps that reduce C5
        h_refined = best_h.copy()
        old_c5 = best_c5
        swapped = True
        max_swaps = min(5000, int(remaining_time * 200))
        num_swaps = 0
        while swapped and num_swaps < max_swaps:
            swapped = False
            if time.time() - start_time > budget_s - 2:
                break
            # Find indices of 1s and 0s
            ones = np.where(h_refined == 1)[0]
            zeros = np.where(h_refined == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            # Try all possible swaps
            # Try swapping one 1 with zero for one of the worst overlap shifts
            corr = np.correlate(h_refined, 1 - h_refined, mode="full")
            worst_k = np.argmax(corr)
            worst_val = corr[worst_k]
            for i in ones:
                for j in zeros:
                    # Tentative swap
                    h_temp = h_refined.copy()
                    h_temp[i] = 0
                    h_temp[j] = 1
                    # Recompute overlaps around worst shift
                    corr_temp = np.correlate(h_temp, 1 - h_temp, mode="full")
                    new_worst_val = corr_temp[worst_k]
                    if new_worst_val < worst_val:
                        h_refined = h_temp.copy()
                        worst_val = new_worst_val
                        swapped = True
                        num_swaps += 1
                        break
                if swapped:
                    break
            if not swapped:
                # Linear exploration of pairs if no better swap found
                # Always try couple swaps
                for i in range(min(50, len(ones))):
                    if time.time() - start_time > budget_s - 2:
                        break
                    i_idx = ones[i]
                    for j in range(min(50, len(zeros))):
                        if time.time() - start_time > budget_s - 2:
                            break
                        j_idx = zeros[j]
                        h_temp = h_refined.copy()
                        h_temp[i_idx] = 0
                        h_temp[j_idx] = 1
                        corr_temp = np.correlate(h_temp, 1 - h_temp, mode="full")
                        new_worst = dx * np.max(corr_temp)
                        if new_worst < old_c5:
                            h_refined = h_temp.copy()
                            old_c5 = new_worst
                            swapped = True
                            num_swaps += 1
                            break
                    if swapped:
                        break
        # Final verification
        corr_final = np.correlate(h_refined, 1 - h_refined, mode="full")
        final_c5 = dx * np.max(corr_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
```
