import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth approximation to max correlation with epsilon parameter"""
    # Compute correlations for all shifts
    corr = compute_corr_fft(h)
    # Use log-sum-exp approximation to max
    # Adjust for numerical stability
    shifted_corr = epsilon * corr
    max_val = np.max(shifted_corr)
    exp_corr = np.exp(shifted_corr - max_val)
    log_sum_exp = max_val + np.log(np.sum(exp_corr))
    return log_sum_exp / epsilon

def grad_smooth_max_corr(h, epsilon):
    """Compute gradient of smooth max correlation"""
    corr = compute_corr_fft(h)
    # Compute softmax weights
    shifted_corr = epsilon * corr
    max_val = np.max(shifted_corr)
    exp_corr = np.exp(shifted_corr - max_val)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient of each correlation term
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k (represented as index)
    for k in range(2*n-1):
        shift = k - (n-1)  # Convert index to shift value
        if shift >= 0:
            # Forward shift contribution
            # For h[i], we compute derivative of correlation term
            # Contribution to gradient from h[i] in correlation k
            # corr_k = sum_{i=0}^{n-1-shift} h_{i+shift} * (1 - h_i)
            # Gradient contributes to h_i: -h_{i+shift}, and to h_{i+shift}: (1 - h_i)
            # But needs to be handled by index mapping properly
            
            # Simple iterative method
            if shift == 0:
                # This is a special case for direct sums of h_i * (1 - h_i)
                # Gradient is 1 - 2*h_i with index i from 0 to n-1
                # So gradient contribution is (1 - 2*h_i) 
                grad += weights[k] * (1 - 2*h)
            else:
                # Forward shift derivative
                # For each i we move forward by shift 
                # Fredivative wrt h_i is -h_{i+shift} if i+shift < n
                # Forderivative wrt h_{i+shift} is (1 - h_i) if i < n-shift
                # Effectively: derivative of h_{i+shift} * (1-h_i) wrt h_i is -h_{i+shift}
                # derivative wrt h_{i+shift} is (1-h_i)
                # Triggering contribution using shifting arrays
                contrib1 = np.zeros(n)
                contrib2 = np.zeros(n)
                # When i goes from 0 to n-1-shift, h_i contributes to the gradient of h_{i+shift} (if i+shift <n)
                # and also h_{i+shift} contributes -h_i to gradient of h_i (if i<n-shift)
                # Assigning: k positive shift
                if shift < n:
                    contrib1[shift:] = -(1-h[:-shift]) if shift < n else np.zeros(n)
                    contrib2[:-shift] = (1-h[shift:]) if shift < n else np.zeros(n) 
                else:
                    contrib1.fill(0)
                    contrib2.fill(0)
                grad += weights[k] * (contrib1 + contrib2)
        # Handle negative shifts symmetrically  
        else:
            # Shift is negative
            abs_shift = -shift
            if abs_shift == 0:
                grad += weights[k] * (1 - 2*h)
            else:
                # Reverse shift
                contrib1 = np.zeros(n)
                contrib2 = np.zeros(n)
                if abs_shift < n:
                    contrib1[abs_shift:] = -(1-h[:-abs_shift]) 
                    contrib2[:-abs_shift] = 1-h[abs_shift:]
                grad += weights[k] * (contrib1 + contrib2)
    
    return grad
