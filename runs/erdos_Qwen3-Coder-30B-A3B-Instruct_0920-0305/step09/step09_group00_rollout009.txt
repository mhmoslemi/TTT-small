```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, 2 * len(h) - 1)
    corr = ifft(H * np.conj(H)).real
    return corr[:len(h)]

def softmax_weights(g, alpha):
    """Compute softmax weights for given g and alpha"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    weights = exp_g / np.sum(exp_g)
    return weights

def gradient(h, p):
    """Compute gradient of surrogate objective using FFT"""
    # Two convolutions: p * h and p_rev * h
    p_rev = p[::-1]
    
    # Convolution p * h
    conv1 = np.real(ifft(fft(p, 2 * len(h) - 1) * fft(h, 2 * len(h) - 1)))
    conv1 = conv1[:len(h)]
    
    # Convolution p_rev * h (should be same as p * h with reversed indices)
    conv2 = np.real(ifft(fft(p_rev, 2 * len(h) - 1) * fft(h, 2 * len(h) - 1)))
    conv2 = conv2[:len(h)]
    
    return -(conv1 + conv2)

def project_onto_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def c5_true(h):
    """Compute exact C5 value using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def generate_binary_pattern(n):
    """Generate Thue-Morse binary pattern"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Multi-resolution approach: coarse to fine
    resolutions = [64, 128, 256] 
    
    for resolution in resolutions:
        n_points = resolution
        dx = 2.0 / n_points
        TARGET_SUM = n_points / 2
        
        # Generate initial binary patterns
        initial_pattern = generate_binary_pattern(n_points)
        random_patterns = [np.random.rand(n_points) for _ in range(5)]
        patterns = [initial_pattern] + random_patterns
        
        for h_start in patterns:
            h_start = project_onto_sum_box(h_start, TARGET_SUM)
            h = h_start.copy()
            
            # Annealing loop
            alpha = 1.0
            alpha_max = 2000.0
            gamma = 1.3
            max_iter_per_alpha = 200
            eta = 0.01
            
            for alpha_it in range(2000):  # approximately 2000 iterations
                if time.time() - start_time > budget_s - 5:  # Leave 5s for final polish
                    break
                
                # Compute auto-correlations (negative for g)
                A = autocorr_fft(h)
                g = -A
                
                # Softmax weights
                p = softmax_weights(g, alpha)
                
                # Gradient computation
                grad = gradient(h, p)
                
                # Gradient step
                h_new = h - eta * grad
                
                # Projection
                h_new = project_onto_sum_box(h_new, TARGET_SUM)
                
                # Accept step
                h = h_new.copy()
                
                # Check if we found a better solution
                c5_new = c5_true(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    
                # Update alpha (deterministic annealing)
                if alpha < alpha_max:
                    alpha = min(alpha * gamma, alpha_max)
                
                # Adjust step size
                if alpha_it > 0 and alpha_it % 50 == 0:
                    eta *= 0.95

        # After each resolution level, try discrete improvement
        if best_h is not None and time.time() - start_time < budget_s - 10:
            # Binary swap improvement
            h_best = best_h.copy()
            for _ in range(1000):  # Limited iterations
                if time.time() - start_time > budget_s - 5:
                    break
                
                # Pick random pair (1,0)
                ones = np.where(h_best > 0.5)[0]
                zeros = np.where(h_best < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    # Swap values
                    h_test = h_best.copy()
                    h_test[i], h_test[j] = h_test[j], h_test[i]
                    c5_test = c5_true(h_test)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h_test.copy()
                        h_best = h_test.copy()
    
    # Final polishing with SLSQP if we have time
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def objective(h):
            return c5_true(h)
            
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                final_c5 = c5_true(res.x)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = res.x
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
