import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_correlation(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def exact_c5(h):
    """Compute exact C5 value"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, weights):
    """Compute weighted gradient ∇Φ_β(h)"""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute contributions from all shifts
    for k in range(-n+1, n):
        if k >= 0:
            left = h[:n-k]
            right = h[k:]
        else:
            left = h[-k:]
            right = h[:n+k]
        
        # pad to same length
        if len(left) != len(right):
            left = np.pad(left, (0, len(right)-len(left)), 'constant')
            right = np.pad(right, (len(left)-len(right), 0), 'constant')
            
        grad += weights[k+n-1] * (1 - left - right)
    
    return grad
