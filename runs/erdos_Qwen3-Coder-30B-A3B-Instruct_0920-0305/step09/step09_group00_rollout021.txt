import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n"""
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') % 2
    return seq.astype(float)

def hillclimb(h, max_iters=5000):
    """Perform hill climbing with greedy swaps"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    ones = np.where(current_h > 0.5)[0]
    zeros = np.where(current_h < 0.5)[0]
    
    improved = True
    iterations_without_improvement = 0
    for _ in range(max_iters):
        if improved:
            iterations_without_improvement = 0
            improved = False
        else:
            iterations_without_improvement += 1
            if iterations_without_improvement > 100:
                break
                
        # Make a random swap
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            h_new = current_h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5 - 1e-12:
                current_h = h_new
                current_c5 = c5_new
                improved = True
                
                # Update indices
                ones = np.where(current_h > 0.5)[0]
                zeros = np.where(current_h < 0.5)[0]
                
                if c5_new < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Initialize with Thue-Morse sequence
    h0 = thue_morse_sequence(n_points)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Random search to find good starting points
    random_search_time = budget_s * 0.2
    search_start = time.time()
    
    while time.time() - search_start < random_search_time:
        if time.time() - start_time > budget_s:
            break
        # Generate random binary vector with exactly half ones
        indices = np.random.permutation(n_points)[:int(TARGET_SUM)]
        h = np.zeros(n_points)
        h[indices] = 1.0
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Local hill climbing from best
    hillclimb_start = time.time()
    
    while time.time() - hillclimb_start < (budget_s - random_search_time) * 0.7:
        if time.time() - start_time > budget_s:
            break
            
        # Run multiple hillclimb restarts
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            h_start = best_h.copy()
            # Random perturbation
            for _ in range(5):
                ones = np.where(h_start > 0.5)[0]
                zeros = np.where(h_start < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    h_start[i] = 0.0
                    h_start[j] = 1.0
            
            h_improved, c5_improved = hillclimb(h_start, max_iters=2000)
            if c5_improved < best_c5:
                best_h = h_improved.copy()
                best_c5 = c5_improved
    
    # Final continuous refinement
    final_refinement_start = time.time()
    
    if time.time() - start_time < budget_s - 10:
        # Convert to continuous
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        try:
            res = minimize(
                lambda x: evaluate_c5(x), 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
