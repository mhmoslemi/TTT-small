import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Helper function to evaluate C5
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return dx * np.max(corr)
    
    # Phase 1: Solve LP relaxation
    # Variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(n_points - 1, name="t")  # for each shift
    z = cp.Variable((n_points, n_points), name="z", symmetric=True)
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # McCormick envelopes for z_{i,j} = h_i * h_j
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(z[i, j] <= h[i])
            constraints.append(z[i, j] <= h[j])
            constraints.append(z[i, j] >= h[i] + h[j] - 1)
            constraints.append(z[i, j] >= 0)
            # Symmetric constraint
            constraints.append(z[j, i] == z[i, j])
    
    # Diagonal constraints
    for i in range(n_points):
        constraints.append(z[i, i] == h[i])
    
    # Overlap constraints
    for k in range(1, n_points):
        # c_k = sum_{i=0}^{n-1-k} (h_{i+k} - z_{i, i+k})
        # We impose c_k <= t[k-1]
        c_k = cp.sum(h[k:] - cp.hstack([z[i, i+k] for i in range(n_points-k)]))
        constraints.append(c_k <= t[k-1])
    
    # Objective: minimize max of t
    objective = cp.Minimize(cp.max(t))
    
    # Formulate and solve LP
    prob = cp.Problem(objective, constraints)
    
    # Time monitoring
    elapsed = time.time() - start_time
    if elapsed > budget_s * 0.5:
        return ([0.5]*n_points, 0.5, n_points)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
        if prob.status != cp.OPTIMAL:
            # If not optimal, solve with GLPK which might work
            prob.solve(solver=cp.GLPK, verbose=False)
    except:
        pass
    
    # Extract solution if possible
    h_sol = h.value
    if h_sol is None:
        h_sol = np.ones(n_points) * 0.5  # Default fallback
    
    # Phase 2: Randomized rounding + deterministic swap refinement
    
    best_h = None
    best_c5 = float('inf')
    
    # Rounding using conditional expectation
    def round_conditional(h_vec, n_points, target_sum):
        h_bin = np.zeros(n_points)
        remaining_sum = target_sum
        
        for i in range(n_points):
            p = h_vec[i]
            if remaining_sum <= 0:
                h_bin[i] = 0
            elif remaining_sum >= n_points - i:
                h_bin[i] = 1
            else:
                # Probability of setting to 1
                if np.random.random() < p:
                    h_bin[i] = 1
                    remaining_sum -= 1
                else:
                    h_bin[i] = 0
        return h_bin
    
    # Round multiple times
    num_rounds = min(2000, max(100, int((budget_s - elapsed) / 0.15)))  # ~150ms per round
    
    for _ in range(num_rounds):
        h_rounded = round_conditional(h_sol, n_points, target_sum)
        if abs(np.sum(h_rounded) - target_sum) > 1e-10:
            # Correction to match sum exactly
            ones_indices = np.where(h_rounded == 1)[0]
            zeros_indices = np.where(h_rounded == 0)[0]
            if len(ones_indices) > target_sum:
                # Need to flip some 1s to 0s
                to_flip = len(ones_indices) - int(target_sum)
                flip_indices = np.random.choice(ones_indices, to_flip, replace=False)
                for idx in flip_indices:
                    h_rounded[idx] = 0
            elif len(ones_indices) < target_sum:
                # Need to flip some 0s to 1s
                to_flip = int(target_sum) - len(ones_indices)
                flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
                for idx in flip_indices:
                    h_rounded[idx] = 1
        
        c5 = evaluate_c5(h_rounded)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
    
    if best_h is None:
        # Fallback to the LP solution
        best_h = h_sol.copy()
        best_c5 = evaluate_c5(best_h)
    
    # Phase 3: Local deterministic swap refinement (simple hill climb)
    elapsed = time.time() - start_time
    swap_budget = max(500, int((budget_s - elapsed) * 100))
    
    # Efficiently update overlaps when swapping
    def update_correlations(h, i, j, corr_cache=None):
        """Update the correlation vector when swapping h[i] and h[j]."""
        # For simplicity and efficiency, recompute C5 fully
        return evaluate_c5(h)
        
    def hill_climb(h, max_swaps):
        h = h.copy()
        c5 = evaluate_c5(h)
        cnt = 0
        
        while cnt < max_swaps:
            best_swap = None
            best_improvement = 0
            
            # Try swapping pairs that can improve
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            # Sample pairs to test
            num_pairs = min(100, len(ones) * len(zeros))
            pairs = []
            for _ in range(num_pairs):
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                if i != j:
                    pairs.append((i, j))
            
            for i, j in pairs:
                h_swapped = h.copy()
                h_swapped[i] = 0
                h_swapped[j] = 1
                
                new_c5 = evaluate_c5(h_swapped)
                improvement = c5 - new_c5
                
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
            
            if best_swap is not None:
                i, j = best_swap
                h[i] = 0
                h[j] = 1
                c5 = evaluate_c5(h)
                cnt += 1
            else:
                break
                
        return h, c5
    
    # Do a final local search
    if time.time() - start_time < budget_s - 0.5:
        refined_h, refined_c5 = hill_climb(best_h, swap_budget)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
