import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def exact_C5(h):
    """Compute exact C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def compute_weighted_gradient(h, wgt):
    """Compute gradient of smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, compute gradient contribution
    for k in range(-n+1, n):
        if k >= 0:
            left = np.concatenate([np.zeros(k), h[:-k]])
            right = h[k:]
        else:
            left = h[-k:]
            right = np.concatenate([h[:k], np.zeros(-k)])
        
        grad -= wgt[k + n - 1] * (left + right)
    
    # Add the constant term
    grad += np.ones(n)
    
    return grad

def project_onto_box_and_sum(v, target_sum):
    """Project vector onto box [0,1] with sum = target_sum"""
    # Clip to box constraints
    v = np.clip(v, 0, 1)
    
    # Adjust to correct sum using binary search for lambda
    def sum_with_lambda(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    fa, fb = sum_with_lambda(a), sum_with_lambda(b)
    
    # Find initial bracket
    while fa * fb > 0:
        a -= 0.1
        b += 0.1
        fa, fb = sum_with_lambda(a), sum_with_lambda(b)
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < tol:
            break
        if fm * fa < 0:
            b, fb = m, fm
        else:
            a, fa = m, fm
    
    lam = (a + b) / 2
    result = np.clip(v + lam, 0, 1)
    return result
