def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_gradient(h, alpha):
    """Compute gradient of log-sum-exp of correlations"""
    corr = compute_corr_fft(h)
    
    # Compute weights using log-sum-exp trick
    max_corr = np.max(alpha * corr)
    exp_weights = np.exp(alpha * corr - max_corr)
    weights = exp_weights / np.sum(exp_weights)
    
    # Compute gradient using slice operations
    grad = np.ones_like(h)
    for k in range(-len(h)+1, len(h)):
        if k >= 0:
            grad[:len(h)-k] -= weights[k] * h[k:]
            grad[k:] -= weights[k] * h[:len(h)-k]
        else:
            s = -k
            grad[s:] -= weights[k] * h[:len(h)-s]
            grad[:len(h)-s] -= weights[k] * h[s:]
    
    return grad
