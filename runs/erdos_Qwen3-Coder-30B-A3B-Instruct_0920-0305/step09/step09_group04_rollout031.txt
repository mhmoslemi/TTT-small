import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth max and its gradient"""
    # Compute correlation: h * (1-h) with full mode (zero padding)
    corr = np.correlate(h, 1 - h, mode='full')
    # corr includes indices from -(n-1) to (n-1), so we transform it to match:
    # corr[i] represents correlation for shift i - (n-1) = i - (n-1) where i ranges from 0 to 2*(n-1)
    # To access the compact indices: shift s = i - (n-1), i = s + (n-1)
    # For s in [-n+1, n-1] or i from 0 to 2*(n-1), range becomes [-n+1, n-1]
    
    # Transform to proper shift indexing + use shift n-1 as offset:
    # corr has length 2*n-1
    shift_offset = n_points - 1
    corr_vals = corr  # shift indices go from -(n-1) to (n-1) at corr indices 0..2*(n-1)
    
    # Compute log-sum-exp with temperature epsilon
    # Scale to prevent overflow
    scaled_corr = epsilon * corr_vals
    max_val = np.max(scaled_corr)
    exp_corr = np.exp(scaled_corr - max_val)
    sum_exp = np.sum(exp_corr)
    
    # Log-sum-exp
    log_sum_exp = max_val + np.log(sum_exp)
    f_epsilon = log_sum_exp / epsilon
    
    # Compute gradient: d/dh_i of log-sum-exp of c_k
    # Each c_k = sum_{j = max(0, k) to min(n-1, n-1+k)} h_{j+k} * (1 - h_j)
    # This is complex to derive analytically, use finite differences or vectorized method:
    # We approximate gradient using identity that for smooth log-sum-exp, 
    # gradient = sum_k w_k * grad_k where w_k = exp(ε * c_k) / sum_i exp(ε * c_i)
    # So we need to compute each gradient grad_k = d/dh_i (c_k) = -h_{i+k} + (1-h_{i-k}) * I_{i >= k}
    #  where I is indicator.
    
    # Let's compute weighted gradient vector
    weights = exp_corr / sum_exp  # softmax weights
    grad = np.zeros(n_points)
    
    # Compute contribution from each correlation (shift)
    for k in range(-n_points+1, n_points):
        idx_k = k + shift_offset  # corresponds to consistent correlation index 
        if idx_k < 0 or idx_k >= len(corr_vals):
            continue
            
        # Compute d/dh_i of c_k with shift k
        # c_k = sum_i h_{i+k} * (1 - h_i) for valid i such that i+k < n
        # and i >= 0. So valid i: from max(0, k) to min(n-1, n-1-k) = n-1-k
        # But we must also consider that i+k is valid, so i is from max(0, -k) to min(n-1, n-1-k)
        # also must have i+k >= 0, i+k < n => i >= -k, i < n-k
        # Effectively i: from max(0, k, -k) to min(n-1, n-1-k) 
        # Actually i range: max(0, -k) to min(n-1, n-1-k)
        # Not right, i must have to valid both h_i and h_{i+k}
        # 0 <= i < n and 0 <= i+k < n => 0 <= i < n, -k <= i < n-k.
        # So i from max(0, -k) to min(n-1, n-1-k).
        # i >= max(0, -k) and i < min(n, n-k) = assume no overflow violation.
        # implies inequality valid: i >= 0, k >= -i
        # i <= n-1, k <= -(n-1-i) = n-1-i.
        # Specifically, c_k = sum_{i=max(0, -k)}^{min(n-1, n-1-k)} h_{i+k} (1 - h_i).
        # Gradient of c_k w.r.t. h_i:
        # if i is in summation range, we get:
        # ambient: h_{i+k} * (1 - h_i) => grad = -h_{i+k} (from 1-h_i) plus (1-h_i) (from h_{i+k})
        # But there is double-counting: terms where h_i or h_{i+k} appear in multiple sums...
        # Actually, simple way: for fixed k>=0:
        # c_k = Σ_{i=0}^{n-1-k} h_{i+k} * (1 - h_i)  
        # Gradient wrt h_{i}: 
        # for i in range 0 to n-1-k: grad = -h_{i+k} if i appears in that later term
        # and + (1 - h_{i-0}) if i appears in former: but former its h_{i+k}, k fixed. So d/dh_i(h_{i+k}) = 0 unless i+k=i => k=0.
        # So just derivatives term by term of c_k = sum_{i = 0}^{n-1-k} h_{i+k}(1 - h_i)  
        # d/dh_i c_k = -h_{i+k} if i+k (nexter) is in range [0, n-1], i.e., i+k <= n-1 (i <= n-1-k)  
        # and -(1-h_{i-k}) if i-k is in range 0<=i-k <= n-1, i.e., k<=i<=n-1+k  
        # For shift k, the terms are h_{i+k} (1 - h_i) for i from 0 to n-1-k. So we distinguish the(i) contribution:
        # For shift k negative: h_i * (1 - h_{i-k}) for i from -k to n-1 if -k >= 0  
        # Finishgrad computation:
        
        # Back to the core method: many terms in correlation have same basic form:
        # For shift k, we get terms h_{i+k} * (1 - h_i) = h_{i+k} - h_{i+k} * h_i
        # Note: consider derivative of the full expression
        # But it's easier to write finite difference approach that tracks all active indices in sum for a fixed k.
        # But for implementation precision, rebuild structured as better analytic form.
        
        # Instead use consistent computation:
        # Effective backward and forward convolution steps, compute each valid cross product term.
        
        # Easier: Add all indices represented by respect to weight of corr_k
        # Compute weight * gradient for each shift k efficiently using:expr sed from convolutions?
        
        # Sharper method: 
        # Compute for each i the contribution to all c_k's that contain it.
        
        # Just simulate explicit dense gradients:
        # Let's write it: compute contribution to each h_i from c_k term
        # For shift k from -n+1 to n-1:
        # In terms of array indices:
        # corr_k             = sum_{i} h_{i+k} * (1 - h_i) 
        # where valid range is 0 <= i < n and 0 <= i+k < n => 0 <= i < n, -k <= i < n-k.
        # So effective i from max(0, -k) to min(n-1, n-1-k). 
        
        # Same during weight gradient ascertainment:
        # Given k [0] 
        # d/dh_i c_k = ...
        # Each c_k=(sum_{i=max(0,-k)}^{min(n-1,n-1-k)}) h_{i+k}(1-h_i) 
        # For given term h_{i+k}, derivative w.r.t h_i: -h_{i+k} (from 1-h_i).
        # For given term h_{i+k}, derivative w.r.t h_{i+k}: 1-h_i (from h_{i+k})
        # But we have one term in the index summation.
        # For d/c_d h_i for fixed i (with fixed k), i appears in one term where i+k is increasing index into array  
        # So ∂c_k/∂h_i = -(correct index) if (i+k) exists in [0,n-1] and i is in [0, n-1-k] 
        #            or + (1-h_{i-k}) if {i-k} exists and i-k is in [0,n-1], and i is in [k,n-1]

        # Write out final derived formulas:
        # Given shift k,  
        # term h_{i+k} * (1 - h_i) in the sum. The influence on gradient ∂/∂h_i(c_k)
        # This is the way to look:

        # For fixed shift k, compute vector ∂c_k/∂h = partial gradient vector of correlation k w.r.t. h
        
        # This must handle k positive and negative and span valid indices.
        # k shift can be interpreted, assuming in range [-n+1, n-1] via full correlate
        
        # Let's compute a vectorized method:
        # Consider shift k (varies from -n+1 to n-1, mapped to full corr)  
        # The value is correlation c_k = Σ_{i=max(0, -k)}^{min(n-1, n-1-k)} h_{i+k}(1-h_i)  
        # So we get contribution to gradient for indices like this:
        # Gradients:
        # For each i such that it's in range [max(0, k), min(n-1, n-1-k)] for positive k  
        # and for negative k, add contributions to h_i as follows:
        # The contribution= -(h_{i+k}) when derivative from h_{i+k}*(1-h_i)  
        # plus + (1-h_{i-k}) when derivative from h_{i+k}*(1-h_i)
        # But we should only include one of these, based on type.
        # Actually clarity is given when expanding correlated terms together:
        # rgba_hc = (1-h) 
        # c_k = Σ_i h[i] * rgba_hc[i-k]  (note k is shifted right)  
        # Then ∂c_k/∂h_i = rgba_hc[i-k]  (so we get from inverse sync) 
        # and ◊(c_k/∂h_{i-k}) = h[i] 
        # So we start:
        # For shift k:
        
        # IndexError risk: k can be negative (denoting right shift) or positive (left-shift):  
        
        w_k = weights[idx_k]
        # Build contribution vector for shift k:
        grad_k = np.zeros(n_points) 
        
        # Determine valid range of i for a specific k: 0 <= i & 0 <= (i+k) < n  
        # i can vary from max(0, -k) to min(n-1, n-1-k)  
        # This determines valid i for sum = 0 to n-1-k but exit if i+k < 0 or i+k >= n.
        # Thus i in [max(0, -k), min(n-1, n-1-k)]
        
        if k >= 0: 
            # For shift k: i from 0 to n-1-k
            # Term h_{i+k} * (1 - h_i)
            # So we get chunk contribution  
            for i in range(0, n_points-k):
                grad_k[i] -= w_k * (1 - h[i+k])
                grad_k[i+k] += w_k * (1 - h[i])
        else:
            # For negative shift k: i from -k to n-1 because i+k >= 0, i < n  
            # Range (k < 0): valid i from -k to n-1: but also i such that i+k < n 
            # So i in [max(0, -k), min(n-1, n-1-k)] and we have i+k < n  
            # Let we take i from max(0, -k) etc 
            for i in range(max(0, -k), n_points):  
                if i+k < 0 or i+k >= n_points: continue
                grad_k[i] -= w_k * (1 - h[i+k])
                grad_k[i+k] += w_k * (1 - h[i])
    
    return f_epsilon, grad
