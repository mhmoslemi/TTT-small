    h = cp.Variable(n_points)
    t = cp.Variable()

    # We'll do it in a more direct way: add constraints for each product
    # For simplicity and efficiency, we'll compute overlap constraints directly
    # We'll need to define z variables matrix
    z = cp.Variable((n_points, n_points))
    
    # McCormick constraints: z[i,j] >= h[i] + h[j] - 1, etc.
    constraints = []
    
    # Add McCormick envelopes
    for i in range(n_points):
        for j in range(i+1, n_points):
            # z[i,j] = h[i] * h[j] in binary case
            constraints.append(z[i,j] <= h[i])
            constraints.append(z[i,j] <= h[j])
            constraints.append(z[i,j] >= h[i] + h[j] - 1)
            constraints.append(z[i,j] >= 0)
            # Symmetry: z[i,j] = z[j,i]
            z[j,i] = z[i,j]
    
    # Sum constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Non-negativity
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Overlap constraints
    # For each shift k = 1..n-1
    for k in range(1, n_points):
        # This would be sum_{i=0}^{n-1-k} (h_{i+k} - z[i,i+k]) <= t
        expr = 0
        for i in range(n_points - k):
            expr += h[i+k] - z[i, i+k]
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve the program
    prob = cp.Problem(objective, constraints)
    
    # Set solver options for speed
    solver_opts = {"max_iters": 5000, "abstol": 1e-6, "reltol": 1e-6}

    def project_to_sum_box(x, target_sum):

        # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
