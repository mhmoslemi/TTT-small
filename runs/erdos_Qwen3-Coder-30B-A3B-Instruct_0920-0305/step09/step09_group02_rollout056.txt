import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Setup variables
    h = Variable(n_points, integer=True)
    y = Variable((n_points, n_points), integer=True)  # y[i,k] represents h[i]*h[i+k]
    T = Variable(integer=True)  # Target upper bound
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(sum(h) == target_sum)
    
    # Binary constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Product constraints (McCormick envelopes)
    for i in range(n_points):
        for k in range(-n_points+1, n_points):
            if 0 <= i+k < n_points:
                j = i + k
                # y[i,k] <= h[i], y[i,k] <= h[j]
                constraints.append(y[i, k] <= h[i])
                constraints.append(y[i, k] <= h[j])
                # y[i,k] >= h[i] + h[j] - 1
                constraints.append(y[i, k] >= h[i] + h[j] - 1)
    
    # Overlap constraints
    for k in range(-n_points+1, n_points):
        if k >= 0:
            shifted_indices = range(n_points-k)
        else:
            shifted_indices = range(-k, n_points)
        overlap_sum = sum(y[i, k] for i in shifted_indices)
        constraints.append(overlap_sum >= target_sum - T)
    
    # Objective
    objective = Minimize(T)
    
    # Create the problem
    prob = Problem(objective, constraints)
    
    # Solve using CBC solver (limiting to 540 seconds)
    prob.solve(solver=CPLEX, time_limit=540)
    
    # Extract solution
    h_sol = h.value
    T_sol = T.value
    
    # Convert to floating point values
    h_final = np.array(h_sol).clip(0, 1)
    
    # Evaluate final C5
    corr_full = correlate(h_final, 1 - h_final, mode="full")
    C5 = dx * np.max(corr_full)
    
    # Return solution
    return (h_final.tolist(), float(C5), n_points)
