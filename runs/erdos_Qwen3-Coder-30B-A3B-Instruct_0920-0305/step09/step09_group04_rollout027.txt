import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5_exact(h):
    """Evaluate the exact C5 value using correlation"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

    h = cp.Variable(n_points)
    t = cp.Variable()
    z = cp.Variable((n_points, n_points), symmetric=True)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1,
    ]
    
    # McCormick envelopes for z[i,j] ≈ h[i]*h[j]
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.extend([
                z[i,j] <= h[i],
                z[i,j] <= h[j],
                z[i,j] >= h[i] + h[j] - 1,
                z[i,j] >= 0,
                z[j,i] == z[i,j]
            ])
    
    # Overlap constraints for each shift k
    for k in range(1, n_points):
        # Compute sum over valid i pairs with distance k
        shift_sum = 0
        for i in range(n_points - k):
            shift_sum += (h[i+k] - z[i, i+k])
        constraints.append(shift_sum <= t)

    prob = cp.Problem(objective, constraints)
