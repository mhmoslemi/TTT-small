import numpy as np
from scipy.optimize import minimize

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def ccqp_optimize(h_initial, max_iter=100):
    """Convex-concave procedure optimization"""
    h = h_initial.copy()
    n = len(h)
    dx = 2.0 / n
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n/2
    target_sum = n / 2
    
    # Initialize best solution
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        # Evaluate current correlation
        corr = compute_corr_fft(h)
        current_c5 = c5_from_corr(corr, n)
        
        # Update best solution if needed
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Update h using local optimization with fixed constraints
        # FIT a smooth function to the correlation values and adjust h
        # Construct a quadratic approximation of the objective
        # We're essentially doing a local Newton-like step.
        
        # Derivative approximation for gradient
        gradient = np.zeros_like(h)
        eps = 1e-5
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, target_sum)
            c_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, target_sum)
            c_minus = evaluate_c5(h_minus)
            
            gradient[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update step with small learning rate
        step_size = 0.01
        h_new = h - step_size * gradient
        
        h_new = project_to_sum_box(h_new, target_sum)
        h = h_new
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Initialize with the initial construction
    h = np.array(initial_h_values)
    h = project_to_sum_box(h, TARGET_SUM)
    best_c5 = evaluate_c5(h)

    # Phase 1: CCQP optimization
    h_opt, c5_opt = ccqp_optimize(h, max_iter=500)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
    
    # Phase 2: Local search with aggressive mutation
    def aggressive_local_search(h, max_iters=1000):
        h = h.copy()
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        for i in range(max_iters):
            # Sample random indices
            indices = np.random.choice(n_points, size=min(10, n_points//4), replace=False)
            
            # Try mutations
            for idx in indices:
                old_val = h[idx]
                # Try multiple steps
                steps = [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]
                for delta in steps:
                    h[idx] = old_val + delta
                    if h[idx] < 0 or h[idx] > 1:
                        continue
                    # Project back
                    h = project_to_sum_box(h, TARGET_SUM)
                    new_c5 = evaluate_c5(h)
                    
                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                    else:
                        h[idx] = old_val
            
            # Occasionally, do full perturbation
            if np.random.random() < 0.05:
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                h = np.clip(h + perturbation, 0, 1)
                h = project_to_sum_box(h, TARGET_SUM)
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            
            # Break if time exceeded
            if time.time() - start_time > 0.8 * budget_s:
                break
                
        return best_h, best_c5
    
    # Run local search
    total_time = time.time() - start_time
    while total_time < 0.8 * budget_s:
        h_ls, c5_ls = aggressive_local_search(best_h, 300)
        if c5_ls < best_c5:
            best_h = h_ls.copy()
            best_c5 = c5_ls
        total_time = time.time() - start_time
    
    # Phase 3: Sampling from different strategies
    strategies = [
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.random.choice([0, 1], size=n_points, p=[0.3, 0.7]),
        lambda: np.random.normal(0.5, 0.15, n_points),
        lambda: np.random.exponential(0.5, n_points)
    ]
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        strategy = np.random.choice(strategies)
        h_init = strategy()
        h_init = np.clip(h_init, 0, 1)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        h_trial, c5_trial = aggressive_local_search(h_init, 200)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
    
    # Final evaluation
    c5_final = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(c5_final), n_points)
