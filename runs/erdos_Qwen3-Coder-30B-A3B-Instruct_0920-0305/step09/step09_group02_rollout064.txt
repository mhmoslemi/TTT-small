import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max(x, beta):
    """Compute log-sum-exp with parameter beta"""
    x_max = np.max(x)
    exp_x = np.exp(beta * (x - x_max))
    return x_max + np.log(np.sum(exp_x)) / beta

def compute_smoothed_objective(h, beta):
    """Compute the smoothed objective Φ_β(h)"""
    corr = compute_corr_fft(h)
    corr = corr * dx
    return soft_max(corr, beta)

def compute_weighted_gradient(h, weights):
    """Compute weighted gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # Build the weighted sum of gradients for all shifts
    for k in range(-n+1, n):
        if k >= 0:
            # h[i] * (1 - h[i+k]) where i+k < n and i < n
            shifted = np.zeros(n)
            shifted[k:] = h[:-k] if k > 0 else h[:]
            right = shifted
            
            shifted = np.zeros(n)
            shifted[:-k] = h[k:] if k > 0 else h[:]
            left = shifted
            
        else:
            # k < 0: [0:-k] of h, and h[-k:] of h
            shifted = np.zeros(n)
            shifted[:k] = h[-k:] if k < 0 else h[:]
            left = shifted
            
            shifted = np.zeros(n)
            shifted[-k:] = h[:k] if k < 0 else h[:]
            right = shifted
            
        grad += weights[k + n - 1] * (1 - left - right)
    
    # Add gradient of constant term
    grad += np.ones(n) * weights.sum()
    
    return grad
