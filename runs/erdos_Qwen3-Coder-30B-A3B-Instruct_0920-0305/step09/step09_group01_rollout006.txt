import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a moderate grid size
    n = 150
    dx = 2.0 / n
    
    # Create the MILP model
    # Decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n // 2,  # Exactly half the entries are 1
    ]
    
    # Linearization constraints for y[i,k] = h[i] AND h[i+k]
    for k in range(n):
        for i in range(n - k):
            constraints.append(y[i, k] <= h[i])
            constraints.append(y[i, k] <= h[i + k])
            constraints.append(y[i, k] >= h[i] + h[i + k] - 1)
    
    # Epigraph constraints: t >= c_k for all k
    # c_k = sum(h_i) - sum(y_i,k) for shift k
    # We compute the correlation manually to ensure correctness
    for k in range(n):
        if k == 0:
            corr_sum = cp.sum(h)
        else:
            corr_sum = cp.sum(h[:-k]) - cp.sum(y[:-k, k])
        constraints.append(t >= corr_sum)  # Use same direction as original
      
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem and solve
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    prob.solve(solver=cp.CBC, timeout=budget_s/2, log=False)
    
    # Retrieve the solution and round to binary if needed
    h_sol = np.round(h.value).astype(int)
    
    # If not the exact solution, fallback to LP relaxation + greedy rounding
    if prob.status != cp.OPTIMAL:
        # Solve the LP relaxation
        h_relax = cp.Variable(n)
        y_relax = cp.Variable((n, n))
        
        # Relax binary constraints to [0,1]
        constraints_relax = [
            cp.sum(h_relax) == n // 2,
        ]
        
        for k in range(n):
            for i in range(n - k):
                constraints_relax.append(y_relax[i, k] <= h_relax[i])
                constraints_relax.append(y_relax[i, k] <= h_relax[i + k])
                constraints_relax.append(y_relax[i, k] >= h_relax[i] + h_relax[i + k] - 1)
        
        # Epigraph constraints
        for k in range(n):
            if k == 0:
                corr_sum = cp.sum(h_relax)
            else:
                corr_sum = cp.sum(h_relax[:-k]) - cp.sum(y_relax[:-k, k])
            constraints_relax.append(t >= corr_sum)
            
        objective_relax = cp.Minimize(t)
        prob_relax = cp.Problem(objective_relax, constraints_relax)
        
        try:
            prob_relax.solve(solver=cp.CBC, timeout=50, log=False)
            
            # Greedy rounding
            h_sol = np.zeros(n)
            sorted_indices = np.argsort(h_relax.value)[::-1]
            for i in range(n // 2):
                h_sol[sorted_indices[i]] = 1
        except:
            # Fallback to a good approximation
            h_sol = h.value >= 0.5
            h_sol = h_sol.astype(int)
            
    h_sol = h_sol.astype(float)
    
    # Local polishing with pair-swap
    best_h = h_sol.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Precompute shifts for efficiency
    shifts = list(range(n))
    
    # pair swapping
    import itertools
    exchange_count = 0
    
    for i in range(100000):
        # Ensure we don't run too long
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try swaps between 1s and 0s
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick random 1 and random 0
        i_idx = np.random.choice(ones)
        j_idx = np.random.choice(zeros)
        
        # Try the swap
        test_h = best_h.copy()
        test_h[i_idx] = 0
        test_h[j_idx] = 1
        
        test_c5 = evaluate_c5(test_h)
        if test_c5 < best_c5:
            best_h = test_h
            best_c5 = test_c5
        
        exchange_count += 1
    
    return (best_h.tolist(), float(best_c5), n)
