import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation via FFT"""
    H = fft(h, 2*len(h)-1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    return exp_g / np.sum(exp_g)

def gradient(h, p):
    """Compute gradient using convolutions"""
    # p_rev[k] = p[-k] with wraparound
    p_rev = np.concatenate([p[len(p)-1:0:-1], p[:1]])
    
    # Linear convolutions
    r1 = np.convolve(p, h, mode='full')[:len(h)]
    r2 = np.convolve(p_rev, h, mode='full')[:len(h)]
    
    return -(r1 + r2)

def project_onto_sum_box(x, target):
    """Project onto [0,1]^n with sum = target using binary search"""
    clipped = np.clip(x, 0, 1)
    sum_clipped = np.sum(clipped)
    
    if abs(sum_clipped - target) < 1e-10:
        return clipped
    
    def objective(lam):
        proj = np.clip(clipped + lam, 0, 1)
        return np.sum(proj) - target
    
    a, b = -1.0, 1.0
    while objective(a) * objective(b) > 0:
        a -= 0.1
        b += 0.1
        
    # Binary search
    for _ in range(30):
        m = (a + b) / 2
        if abs(objective(m)) < 1e-10:
            break
        if objective(m) * objective(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(clipped + (a + b)/2, 0, 1)

def c5_true(h):
    """Compute true C5 bound using np.correlate"""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / len(h))
