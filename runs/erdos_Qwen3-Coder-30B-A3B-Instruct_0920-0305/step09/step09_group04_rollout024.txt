def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_c_k_derivatives(h, k):
    """Compute derivative of c_k with respect to each h_i"""
    # Return gradient vector (n,) for shift k
    grad = np.zeros_like(h)
    if k >= 0:
        # h[i+k] * (1 - h[i]) for i = 0,...,n-1-k
        for i in range(n - k):
            grad[i] -= h[i + k]
            grad[i + k] += (1 - h[i])
    else:
        # h[i] * (1 - h[i-k]) for i = -k,...,n-1
        for i in range(-k, n):
            grad[i] -= h[i + k]
            grad[i + k] += (1 - h[i])
    return grad

def compute_all_gradients(h):
    """Compute all ∇c_k for k = -(n-1) to n-1"""
    grads = []
    for k in range(1 - n, n):
        g = compute_c_k_derivatives(h, k)
        grads.append(g)
    return np.array(grads)

def compute_smooth_max_grad(h, eps=1.0):
    """Compute gradient of smooth max approximation with regularizer epsilon"""
    # Compute all c_k values for current h
    c_vals = []
    for k in range(1 - n, n):
        if k >= 0:
            # For k >= 0: sum over i from 0 to n-1-k of h[i+k] * (1 - h[i])
            s = 0
            for i in range(n - k):
                s += h[i + k] * (1 - h[i])
            c_vals.append(s)
        else:
            # For k < 0: sum over i from 0 to n-1+|k| of h[i] * (1 - h[i+|k|])
            s = 0
            for i in range(n + k):
                s += h[i] * (1 - h[i - k])
            c_vals.append(s)
    
    c_vals = np.array(c_vals)
    
    # Compute soft-weights
    c_max = np.max(c_vals)
    c_shifted = c_vals - c_max
    exp_c = np.exp(eps * c_shifted)
    weights = exp_c / np.sum(exp_c)
    
    # Compute the gradient: sum of weighted gradients
    all_grads = compute_all_gradients(h)
    grad = np.sum(weights[:, np.newaxis] * all_grads, axis=0)
    
    return grad
