def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, eps):
    """Compute smooth max of correlation values with given epsilon"""
    corr = compute_corr_fft(h)
    # Use numerically stable version of softmax
    shifted_corr = eps * corr
    shifted_corr -= np.max(shifted_corr)
    exp_corr = np.exp(shifted_corr)
    weights = exp_corr / np.sum(exp_corr)
    smooth_max = np.sum(weights * corr) / eps
    return smooth_max, weights

def grad_smooth_max_corr(h, eps):
    """Compute gradient of smooth max correlation"""
    corr = compute_corr_fft(h)
    shifted_corr = eps * corr
    shifted_corr -= np.max(shifted_corr)
    exp_corr = np.exp(shifted_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient contributions
    grad = np.zeros_like(h)
    
    # For each possible shift, compute contribution
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            # Forward shift case
            # c_k = sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i)
            # Gradient: -h_{j+k} if j <= n-1-k and j >= 0; (1-h_{j-k}) if j >= k and j <= n-1
            a_s = np.concatenate([np.zeros(k), (1 - h[:n-k])])
            b_s = np.concatenate([h[k:], np.zeros(k)])
            grad_contrib = a_s - b_s
        else:
            # Backward shift case
            k_abs = -k
            a_s = np.concatenate([(1 - h)[k_abs:], np.zeros(k_abs)])
            b_s = np.concatenate([np.zeros(k_abs), h[:n-k_abs]])
            grad_contrib = a_s - b_s
            
        # Apply weight
        grad += weights[k+n-1] * grad_contrib
        
    return grad / eps
