import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def create_gradient_matrix(h):
    """Create h and 1-h shifted matrices for gradient computation"""
    n = len(h)
    grad_matrix = np.zeros((n, n))  # gradient contribution for each position
    
    # For each shift from -n+1 to n-1
    for k in range(1-n, n):
        if k >= 0:
            # k >= 0, so h_{i+k} * (1 - h_i)
            # columns i = 0 to n-k-1
            for i in range(n-k):
                grad_matrix[i, i] -= h[i+k]  # -h_{i+k} 
                grad_matrix[i+k, i] += (1 - h[i])  # (1 - h_i)
        else:
            # k < 0, so h_{i+|k|} * (1 - h_i) = h_{i-k} * (1 - h_i)
            # columns i = -k to n-1
            for i in range(-k, n):
                grad_matrix[i, i] -= h[i+k]  # -h_{i+k} 
                grad_matrix[i+k, i] += (1 - h[i])  # (1 - h_i)
    
    return grad_matrix

def compute_smooth_max_gradient(h, epsilon=1.0):
    """Compute gradient of smooth max approximation to overlap function"""
    # Compute all shifts c_k
    n = len(h)
    c_values = np.zeros(2*n - 1)
    
    # Compute correlations via FFT
    corr = compute_corr_fft(h)
    # The correct correlation is between h and (1-h) with full overlap
    # Reindex to get shifts from -(n-1) to n-1 
    # corr is correlation with shift 0..2n-2, where shift k corresponds to k-n+1 in our indexing
    # Assuming we compute (corr_reverse_convolved) correctly
    # Let's just make a straightforward way:
    
    # Manual computation for c_k:
    specific_c = np.zeros(2*n - 1)
    for k in range(1-n, n):
        # Example: k=0 -> h_0*(1-h_0) + h_1*(1-h_1) + ...
        # k=1 -> h_1*(1-h_0) + h_2*(1-h_1) + ... till h_{n-2}*(1-h_{n-3}) 
        total = 0
        if k >= 0:
            for i in range(n - k):
                total += h[i+k] * (1 - h[i])
        else:
            for i in range(-k, n): 
                total += h[i] * (1 - h[i+k])
        specific_c[k + n - 1] = total
    
    # Softmax weights
    c_values = specific_c  
    normalized_exps = np.exp(epsilon * (c_values - np.max(c_values)))
    weights = normalized_exps / np.sum(normalized_exps)
    
    # Now compute gradient w.r.t. h using the weighted sum of gradients
    grad = np.zeros(n)
    for k in range(1-n, n):
        # approximating grad_c_k
        shift = k + n - 1
        weight = weights[shift]  # wt for shift k
        
        # We will manually compute ∇c_k
        # Let's just approximate via finite differences for now
        # Better: use analytical discrete gradients
        if k >= 0:
            # c_k = sum h_{i+k} * (1 - h_i) for i=0 to n-k-1  
            # d/d h_j c_k = (1-h_{j-k}) if j >= k else 0 for the h_{j+k}(1-h_j) term
            # minus h_{j+k} if j <= n-k-1 for the h_{j+k}(1-h_j) term 
            for j in range(n):
                if j + k < n and j >= k:
                    grad[j] += weight * (-h[j + k])
                if j < n - k and j >= 0:
                    grad[j] -= weight * h[j + k] 
        else: # k < 0 means h_{i - k} (which is i |k|) * (1-h_i)
            # Reverse the indices for negative k
            for j in range(n):
                if j - k >= 0 and j < n:
                    # k=-1, j=1 -> same as h_1*(1-h_0) coefficients
                    # d/d h_j c_{-1}  = -(1-h_{j+1}) if j+1<n and j>=0
                    #                   -h_{j+1} if j>=0 and j+1<n
                    if j >= 0 and j+(-k) < n:
                        grad[j] -= weight * (1 - h[j - k])
                        grad[j - k] -= weight * h[j] 
        
    return grad

def update_step(h, grad, lr, momentum_terms, beta1=0.9, beta2=0.999, eps=1e-8):
    # Momentum update
    m_t = beta1 * momentum_terms[0] + (1 - beta1) * grad
    v_t = beta2 * momentum_terms[1] + (1 - beta2) * grad**2
    m_hat = m_t / (1 - beta1**(len(m_t) if hasattr(len(m_t), '__len__') else 1)) 
    v_hat = v_t / (1 - beta2**(len(v_t) if hasattr(len(v_t), '__len__') else 1))
    
    h_new = h - lr * (m_hat / (np.sqrt(v_hat) + eps))
    return h_new, (m_t, v_t)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    epsilon = 1.0  
    epsilon_max = 100.0
    epsilon_schedule_freq = 500
    epsilon_factor = 1.5
    lr = 0.2
    max_inner_iters = 4000
    num_restarts = 20
    early_stop_patience = 200
    
    # Keep track
    best_c5 = float('inf')
    best_h = None
    
    # Multi restart strategy
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate initial h
        if restart == 0:
            # Use the provided initial values for the first restart
            h = np.array(initial_h_values)
        else:
            h = np.random.rand(n_points)
            
        h = project_to_sum_box(h, TARGET_SUM)
        current_c5 = evaluate_c5(h)
        best_h_held = h.copy()
        best_c5_held = current_c5
        
        # Adam momentum terms
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        
        # Inner iteration loop
        patience_counter = 0
        for iter_count in range(max_inner_iters):
            if time.time() - start_time > budget_s - 2:
                break
            if patience_counter > early_stop_patience:
                break
                
            # Compute weights and gradients for smooth max
            # c_k = sum h_{i+k} * (1 - h_i) for i = 0 to n-1-|k|
            # we compute exact c_k for all k by correlation
                
            # compute all overlaps c_k manually now with fast knn-like way via correlation
            # This is equivalent to the existing code but critically we need the precise stable value for c_k
            # So let's do it analogously to the evaluation:
            corr = compute_corr_fft(h)
            # corr[0] to corr[n-1] has the shift -n+1 to 0
            # corr[n] to corr[2n-2] has the shift 1 to n-1
            # So for the actual overlaps c_k from k=-n+1 to n-1:
            c_vals = np.zeros(2 * n_points - 1)
            c_vals[n_points-1:n_points] = corr[:n_points]  # shift -n+1,...,0
            c_vals[n_points:] = corr[n_points-1:]  # shift 1,...,n-1
            
            # Use softmax approximation to gradient
            c_vals_np = np.array(c_vals)
            max_c = np.max(c_vals_np)
            c_shifted = c_vals_np - max_c
            exps = np.exp(epsilon * c_shifted)
            sum_exps = np.sum(exps)
            if sum_exps > 0:
                weights = exps / sum_exps
            else:
                weights = np.ones_like(exps) / len(exps)
            
            # Geometric way to estimate: use FCN forward pass instead
            # get gradient through finite differences?
            # Close enough with direct approach: 
            # Manual computation of gradient from the actual c
            # Manually compute ∂c_k/∂h_j as we analytically derived
            # Slower direct method in the spirit:
            grad = np.zeros(n_points)
            for k in range(1-n_points, n_points):
                # Weighted sum of each ∇c_k for all k, 
                # but actually correct way to assign weight is to normalize c_k values  
                if k >= 0:
                    # h_{i+k} * (1 - h_i) for i = 0 to n-1-k
                    for i in range(n_points - k):
                        grad[i] -= weights[k + n_points - 1] * h[i + k]
                        grad[i + k] += weights[k + n_points - 1] * (1 - h[i])
                else:
                    # Negative shift k
                    k_abs = -k
                    # h_{i} * (1 - h_{i+k}) for i = -k to n-1 = i = |k| to n-1
                    for i in range(k_abs, n_points):
                        grad[i] -= weights[-k + n_points - 1] * (1 - h[i + k])
                        grad[i + k] -= weights[-k + n_points - 1] * h[i]
                        
            # Update with momentum  
            new_h, (m, v) = update_step(h, grad, lr, (m, v))
            # Project to feasible set
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            # Evaluate true C5
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5_held:
                best_c5_held = new_c5
                best_h_held = new_h.copy()
                patience_counter = 0  # reset patience
            else:
                patience_counter += 1
                
            # Use torch-style schedule: increase epsilon every N iterations
            if iter_count % epsilon_schedule_freq == epsilon_schedule_freq - 1:
                epsilon = min(epsilon * epsilon_factor, epsilon_max)
                if epsilon >= epsilon_max:
                    lr = lr * 0.5  # Reduce learning rate when epsilon saturates

            h = new_h

        # Update global best
        if best_c5_held < best_c5:
            best_c5 = best_c5_held
            best_h = best_h_held.copy()
            
    if best_h is None:
        # Fallback
        best_h = np.array(initial_h_values) 
        best_c5 = evaluate_c5(best_h)
    
    # Postprocessing: binary refinement
    # 1. Threshold to binary
    h_median = np.median(best_h)
    h_binary = (best_h > h_median).astype(float)
    
    # Adjust to sum = n/2
    current_sum = np.sum(h_binary)
    if current_sum < TARGET_SUM:
        diff = int(TARGET_SUM - current_sum)
        indices = np.argsort(best_h)[:diff]
        h_binary[indices] = 1
    elif current_sum > TARGET_SUM:
        diff = int(current_sum - TARGET_SUM)
        indices = np.argsort(best_h)[-diff:]
        h_binary[indices] = 0
        
    # Now improve binary via small swaps
    best_c5 = evaluate_c5(h_binary)
    old_c5 = best_c5 + 1  # ensure we do swap loops
    
    while (old_c5 - best_c5) > 1e-6 and time.time() - start_time < budget_s - 3:
        old_c5 = best_c5
        
        stable_swap_count = 0
        
        while stable_swap_count < 200 and time.time() - start_time < budget_s - 3:
            idx_1 = np.random.randint(0, n_points)
            idx_2 = np.random.randint(0, n_points)
            
            if h_binary[idx_1] == h_binary[idx_2]:
                continue
                
            # Rotate values
            old_val_1, old_val_2 = h_binary[idx_1], h_binary[idx_2]
            h_binary[idx_1], h_binary[idx_2] = old_val_2, old_val_1
            
            # Evaluate new C5
            new_c5 = evaluate_c5(h_binary)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                # Update h from binary (approximately) 
                if best_c5 < old_c5 - 1e-6: 
                    stable_swap_count = 0
                else:
                    stable_swap_count += 1
            else:
                # Revert
                h_binary[idx_1], h_binary[idx_2] = old_val_1, old_val_2
                stable_swap_count += 1
                
        if stable_swap_count >= 200:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
