import time

    # Variables
    h = cp.Variable(n_points)
    t = cp.Variable()

    # z[i,j] = h[i] * h[j], but stored only for i < j

    z_vars = []
    z_indices = []
    for i in range(n_points):
        for j in range(i+1, n_points):
            z = cp.Variable()
            z_vars.append(z)
            z_indices.append((i,j))
    
    # Constraints for the z variables (McCormick envelopes)
    mccormick_constraints = []
    for idx, (i, j) in enumerate(z_indices):
        z = z_vars[idx]
        mccormick_constraints.extend([
            z <= h[i],
            z <= h[j],
            z >= h[i] + h[j] - 1,
            z >= 0
        ])
    
    # Mass constraint
    mass_constraint = cp.sum(h) == target_sum
    
    # Overlap constraints
    overlap_constraints = []
    for k in range(1, n_points):
        # c_k = sum_{i=0}^{n-1-k} ( h_{i+k} - z_{i, i+k} )
        c_k_expr = 0
        z_indices_k = []
        for i in range(n_points - k):
            c_k_expr += h[i + k] - z_vars[z_indices.index((i, i+k))] if (i, i+k) in z_indices else 0
            z_indices_k.append(z_vars[z_indices.index((i, i+k))] if (i, i+k) in z_indices else None)
        
        overlap_constraints.append(c_k_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create the problem and solve
    problem = cp.Problem(objective, 
                        [mass_constraint] + 
                        mccormick_constraints + 
                        overlap_constraints)

        problem.solve(solver=ECOS,

        best_h = None
        best_c5 = float('inf')
        
        for _ in range(n_samples):
            # Conditional expectation rounding
            h_binary = np.zeros(n_points)
            ones_remaining = n_half
            zeros_remaining = n_points - n_half
            
            for i in range(n_points):
                prob = h_float[i]
                if ones_remaining == zeros_remaining:
                    h_binary[i] = 1 if np.random.rand() < prob else 0
                elif ones_remaining == 0:
                    h_binary[i] = 0
                elif zeros_remaining == 0:
                    h_binary[i] = 1
                else:
                    # Decide the value: determine if setting it to 1 affects the future possibility
                    # Compute the ones_remaining needed after this position
                    proposed_1 = ones_remaining - 1 if h_binary[i] == 0 else ones_remaining
                    proposed_0 = ones_remaining if h_binary[i] == 1 else ones_remaining
                    
                    if proposed_1 >= 0 and proposed_0 >= 0:
                        # The flip is legal, proceed with probability
                        h_binary[i] = 1 if np.random.rand() < prob else 0
                    else:
                        h_binary[i] = 1 if ones_remaining > 0 else 0
                
                # Correct for written number of ones
                if h_binary[i] == 1:
                    ones_remaining -= 1
                else:
                    zeros_remaining -= 1

            c_k_vec = compute_c_k_vector(h_best)

            test_c5 = np.max(c_k_test)
