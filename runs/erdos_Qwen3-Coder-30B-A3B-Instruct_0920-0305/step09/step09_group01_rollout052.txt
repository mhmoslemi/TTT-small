import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_gradient(h, c_vals, mu):
    """Build gradient using soft-max weights"""
    # Center the c_vals for numerical stability
    c_max = np.max(c_vals)
    exp_vals = np.exp((c_vals - c_max) / mu)
    weights = exp_vals / np.sum(exp_vals)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # For each shift k, compute the gradient contribution
    # Note: we need to be careful with indexing for wraparound
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # h_i * (1 - h_{i+k})
            grad[:n_points-k] += weights[k+n_points-1] * (1 - h[k:])
            grad[k:] -= weights[k+n_points-1] * h[:n_points-k]
        else:
            # h_i * (1 - h_{i+k}) = h_i * (1 - h_{i-k})  
            k_abs = -k  # k is negative, so we look k positions backward
            grad[k_abs:] += weights[-k+n_points-1] * (1 - h[:n_points-k_abs])
            grad[:n_points-k_abs] -= weights[-k+n_points-1] * h[k_abs:]
    
    return grad

def line_search(h, g, mu, max_iter=5):
    """Perform line search to find good step size"""
    alpha = 0.2 * dx
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    for _ in range(max_iter):
        h_proposed = project_to_sum_box(h_current - alpha * g, TARGET_SUM)
        c5_proposed = evaluate_c5(h_proposed)
        if c5_proposed <= c5_current:
            return h_proposed, c5_proposed
        alpha /= 2
    return h_current, c5_current

def continuation_gradient_descent(h_start, mu0=0.05, mu_min=5e-4, steps_per_mu=250, 
                                jitter_period=500, max_time=600):
    """Main continuation gradient descent algorithm"""
    h_current = h_start.copy()
    c5_current = evaluate_c5(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    mu = mu0
    step_count = 0
    last_improvement = 0
    
    start_time = time.time()
    
    while mu >= mu_min and time.time() - start_time < max_time - 2:
        # Compute all correlation values
        corr = compute_corr_fft(h_current)
        c_vals = corr * dx
        
        # Outer loop for steps at current mu
        for _ in range(steps_per_mu):
            if time.time() - start_time > max_time - 2:
                return best_h, best_c5
                
            # Build gradient with current mu
            g = build_gradient(h_current, c_vals, mu)
            
            # Line search
            h_new, c5_new = line_search(h_current, g, mu)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
                last_improvement = step_count
            else:
                # If no progress, reduce step size 
                pass
            
            step_count += 1
            
            # Random jitter every few iterations
            if step_count % jitter_period == 0:
                noise = np.random.normal(0, 0.02, n_points)
                h_current = project_to_sum_box(h_current + noise, TARGET_SUM)
                c5_current = evaluate_c5(h_current)
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            # Periodic check for improvement
            if step_count - last_improvement > 1000:
                # If no progress, check if we can escape local minimum
                h_current = project_to_sum_box(h_current + np.random.normal(0, 0.001, n_points), TARGET_SUM)
                c5_current = evaluate_c5(h_current)
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
                last_improvement = step_count 

        # Decrease μ
        mu /= 2
    
    return best_h, best_c5
