import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5_exact(h):
    """Evaluate the exact C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    
    # Phase 1: Linear Program Relaxation
    print("Solving LP relaxation...")
    
    # Define variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(1, name="t")
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Bounds on h
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Product relaxation variables
    z = cp.Variable((n_points, n_points), name="z", symmetric=True)
    
    # McCormick envelopes for z[i,j] ≈ h[i]*h[j]
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(z[i,j] <= h[i])
            constraints.append(z[i,j] <= h[j])
            constraints.append(z[i,j] >= h[i] + h[j] - 1)
            constraints.append(z[i,j] >= 0)
    
    # Overlap constraints for each shift k
    for k in range(1, n_points):
        # c_k = sum_{i=0}^{n-1-k} (h_{i+k} - z_{i, i+k})
        expr = 0
        for i in range(n_points - k):
            expr += h[i + k] - z[i, i + k]
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve the LP
    prob = cp.Problem(objective, constraints)
    
    # Solve with a tighter tolerance
    try:
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-7, reltol=1e-7)
        if prob.status != cp.OPTIMAL:
            print("LP not solved to optimality:", prob.status)
            return None
    except Exception as e:
        print("LP solver error:", str(e))
        return None
    
    t_opt = t.value
    h_frac = h.value
    
    print(f"LP bound: {t_opt}")
    
    # Phase 2: Rounding
    print("Performing rounding...")
    
    best_h = None
    best_c5 = float('inf')
    
    # Function to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Multiple rounds of rounding
    n_rounds = 2000
    for r in range(n_rounds):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Balanced rounding
        h_rounded = np.zeros(n_points)
        remaining_ones = n_points // 2
        
        # Process indices in order
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        # Conditional expectation rounding
        for i in indices:
            if remaining_ones == 0:
                h_rounded[i] = 0
            elif remaining_ones == n_points - i:
                h_rounded[i] = 1
            else:
                p = h_frac[i]
                # If setting to 1 is not possible, set to 0
                if p > remaining_ones / (n_points - i):
                    h_rounded[i] = 0
                else:
                    # Sample
                    h_rounded[i] = 1 if np.random.random() < p else 0
                    
                if h_rounded[i] == 1:
                    remaining_ones -= 1
        
        # Ensure exact sum
        actual_sum = np.sum(h_rounded)
        if actual_sum != n_points / 2:
            # Adjust to get exactly n_points/2
            diff = int(n_points / 2 - actual_sum)
            if diff > 0:
                # Need to set some zeros to ones
                zero_indices = np.where(h_rounded == 0)[0]
                for i in range(-diff):
                    if i < len(zero_indices):
                        h_rounded[zero_indices[i]] = 1
            elif diff < 0:
                # Need to set some ones to zeros
                one_indices = np.where(h_rounded == 1)[0]
                for i in range(-diff):
                    if i < len(one_indices):
                        h_rounded[one_indices[i]] = 0
        
        # Evaluate
        c5 = compute_c5(h_rounded)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
    
    if best_h is None:
        return None
    
    # Phase 3: Local refinement with swaps
    print("Performing local refinement...")
    
    def update_overlap(h_vec, i, j, corr_cache):
        # Update the full correlation for this specific swap (i,j) where h[i]=1, h[j]=0
        # Using efficient incremental update
        old_c5 = np.max(corr_cache['corr'])
        corr = corr_cache['corr'].copy()
        
        # For each shift k, update the affected overlap values
        for k in range(1, n_points):
            shift_indices = []
            for idx in range(n_points - k):
                if idx == i or idx == j:
                    shift_indices.append(idx)
                if idx + k == i or idx + k == j:
                    shift_indices.append(idx)
                
            if len(shift_indices) > 0:
                # Recompute the contribution for this shift
                old_val = 0
                new_val = 0
                for idx in shift_indices:
                    if idx == i and (idx + k) == j:  # special case: swap (i,j) with overlap
                        old_val += h_vec[idx] * (1 - h_vec[idx + k])
                        new_val += h_vec[j] * (1 - h_vec[i])
                    elif idx == i:
                        old_val += h_vec[idx] * (1 - h_vec[idx + k])
                        new_val += h_vec[j] * (1 - h_vec[idx + k])
                    elif idx + k == j:
                        old_val += h_vec[i] * (1 - h_vec[idx])
                        new_val += h_vec[i] * (1 - h_vec[j])
                corr[len(corr) // 2 + k] = corr[len(corr) // 2 + k] + (new_val - old_val)
                corr[len(corr) // 2 - k] = corr[len(corr) // 2 - k] + (new_val - old_val)
                
        return corr
    
    # Initial correlation
    def get_full_corr(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return corr
    
    corr = get_full_corr(best_h)
    best_c5 = np.max(corr) * dx
    
    max_swaps = min(2000, 10 * n_points)
    swaps = 0
    while swaps < max_swaps:
        if time.time() - start_time > budget_s - 5:
            break
            
        improved = False
        # Indices of 1s and 0s
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Try a few random swaps
        for _ in range(5):
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Make temporary copy
            h_try = best_h.copy()
            h_try[i] = 0
            h_try[j] = 1
            
            # Recompute correlation
            corr_try = get_full_corr(h_try)
            c5_try = np.max(corr_try) * dx
            
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                corr = corr_try
                break
            
        if not improved:
            break
        swaps += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
