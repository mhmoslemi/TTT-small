import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_correlation(h):
    """Compute correlation using scipy.signal.correlate"""
    return correlate(h, 1 - h, mode='full')

def exact_c5(h):
    """Compute exact C5 value"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, weights):
    """Compute weighted gradient for the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, add contribution to gradient
    # Since we're doing correlation, the shift indices vary from -(n-1) to n-1
    # We're computing: Σ_k w_k * (1 - shift_{+k}(h) - shift_{-k}(h)) 
    
    # We'll use slicing for efficient computation of shifted arrays
    for k in range(-n+1, n):  # shift indices from -(n-1) to n-1
        if k > 0:
            # positive shift: h[i] contributes to f_{k}(h) at position i+k
            left_slice = np.zeros(n)
            left_slice[k:] = h[:-k] if k < n else np.array([])
            right_slice = np.zeros(n)
            right_slice[k:] = h[:-k] if k < n else np.array([])
        elif k < 0:
            # negative shift: h[i] contributes to f_{k}(h) at position i+k  
            left_slice = np.zeros(n)
            left_slice[:k] = h[-k:] if -k < n else np.array([])
            right_slice = np.zeros(n)
            right_slice[:k] = h[-k:] if -k < n else np.array([])
        else:  # k == 0
            left_slice = np.zeros(n)
            right_slice = np.zeros(n)
            # Contribution from h[i]*1 = 1-h[i] which results in -h[i] in gradient
            grad += np.ones(n)
            continue
            
        # The gradient contribution from this shift
        # The contributions cancel out when we compute over all shifts
        # This is just a 22-line algorithm to compute: -weight * (left + right)
        for i in range(n):
            if i+k >= 0 and i+k < n:
                grad[i] -= weights[k+n-1]  # which k in f_k? We map k to [0, 2n-2]

    # Apply weights properly
    # For each shift element k (stored at index k+n-1 in f array)
    # f_k = sum_i h[i] * (1 - h[i+k]) = 1 * sum_i h[i] - sum_i h[i] * h[i+k]
    # = sum(h) - sum(h[i] * h[i+k])
    # So we have f_k = w - sum(h * shift_k(h))
    
    # The best way is direct computation using slicing
    
    weights_expanded = np.zeros(2*n - 1)
    for k in range(-n+1, n):
        weights_expanded[k+n-1] = weights[k+n-1]
        
    # Efficient gradient computation using np.correlate with the shifted partials
    grad = np.zeros(n)
    
    for k in range(-n+1, n):
        if k == 0:
            # This is for the shift of h by 0, i.e. h itself
            continue
        else:
            # Compute shifted version of h
            shifted_h = np.zeros(n)
            if k > 0:
                shifted_h[k:] = h[:-k]
            else:
                shifted_h[:k] = h[-k:]
            # Add -weight * shifted_h 
            grad -= weights_expanded[k+n-1] * shifted_h
                
    # Add 1 weight
    grad += np.sum(weights_expanded)  # weighted sum of ones
    return grad

def smoothed_objective(h, beta):
    """Compute smoothed objective Φ_β(h)"""
    corr = compute_correlation(h)
    corr = corr * dx
    
    # Numerically stable: subtract max before exponentiation
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    return (1 / beta) * (np.log(np.sum(exp_corr)) + max_corr)
