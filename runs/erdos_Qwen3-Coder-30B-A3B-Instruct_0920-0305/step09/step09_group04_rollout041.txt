import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max_corr(h, eps):
    """Computes smooth approximation to max correlation with epsilon regularization"""
    corr = compute_corr_fft(h)
    # Subtract max to prevent overflow
    corr_centered = eps * corr - np.max(eps * corr)
    exp_corr = np.exp(corr_centered)
    weights = exp_corr / np.sum(exp_corr)
    return weights

    # Compute partial derivatives for each shift
    grad = np.zeros_like(h)

        # For shift k get the corresponding correlation value
        if k >= 0:
            # Sum_{i=0}^{n-1-k} h[i+k] * (1 - h[i])
            # Gradient w.r.t. h[j]:
            # For terms where j = i or j = i+k
            # If i = j-k, get (1-h[j])
            # If i+k = j, get -h[j-k]
            contrib = np.zeros_like(h)
            if k == 0:
                contrib = 1 - 2 * h
            else:
                # Here we manually construct the gradient contribution of shift k
                # Gradient of Σ h_{i+k} * (1 - h_i) w.r.t h_j 
                if k >= 0:
                    # contribution from h_{i+k} terms
                    contrib[k:] += -(1 - h[:-k]) if k > 0 else -(1 - h)
                    # contribution from (1-h_i) terms
                    contrib[:-k] += h[k:] if k > 0 else h
                else:
                    # This is same logic but handling negative k correctly
                    contrib[:k] -= (1 - h[-k:]) if k < 0 else (1 - h)
                    contrib[-k:] += h[:k] if k < 0 else h
            grad += weights[k + n - 1] * contrib
        else:
            # For negative k: sum_{i=0}^{n-1-(-k)} h[i] * (1 - h[i-k])
            # Gradient w.r.t. h[j] = -(1-h_{j-k}) for j-k >= 0, +h_{j+k} for j+k < n
            # More direct approach: swap signs based on roles
            contrib = np.zeros_like(h)
            if k == 0:
                contrib = 1 - 2 * h
            else:
                # At i + |k| = j, we have contribution -(1-h_j) if j + |k| < n
                # At i = j, we have contribution h_{j+|k|} if j + |k| < n
                k_magnitude = -k  # Since k < 0, make |k|
                if k_magnitude <= n - 1:
                    # Contribution to h[j] from terms with i = j in h[i]* (1 - h[i-k])
                    # i = j => h[i-k] = h[j-k]
                    # And the term is h[j] * (1 - h[j-k]) its derivative wrt h[j] = (1 - h[j-k])
                    # And derivative wrt h[j-k] = -h[j] 
                    # For contribution, at i j-k occur in the sum wrt to derivative wrt h_j
                    # The term in correlation, sub k in i-k in expression:
                    # We were shifting form h[i]* (1-h[i+k]): it became h[i]* (1-h[i-k])
                    # Now we do conv(k) on coefficients i + k when index shifted by left by k?
                    contrib[k_magnitude:] += -(1 - h[:-k_magnitude]) # if |k| given, we past from 
                    contrib[:-k_magnitude] += h[k_magnitude:]
            grad += weights[k + n - 1] * contrib
    
    return grad
