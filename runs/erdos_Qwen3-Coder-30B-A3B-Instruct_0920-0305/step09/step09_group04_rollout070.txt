def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth approximation of max correlation with epsilon regularization"""
    corr = compute_corr_fft(h)
    # Numerically stable softmax
    shift_idx = n_points - 1
    shifted_corr = corr[shift_idx - (n_points - 1):shift_idx + n_points]
    # Compute log-sum-exp with numerical stability
    stable_corr = epsilon * shifted_corr
    max_corr = np.max(stable_corr)
    log_sum_exp = max_corr + np.log(np.sum(np.exp(stable_corr - max_corr)))
    return log_sum_exp / epsilon
