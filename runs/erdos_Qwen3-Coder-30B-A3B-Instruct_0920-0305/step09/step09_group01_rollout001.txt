import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_c5(h, shift_idx):
    """Compute gradient of C_shift_idx with respect to h"""
    n = len(h)
    # This computes gradient of sum h_i * (1 - h_{i+shift_idx}) 
    # with respect to h
    grad = np.zeros(n)
    if shift_idx >= 0:
        # gradient for i=0..n-shift_idx-1 
        # +1 for first term, -h[i+shift_idx] for second term
        grad[:n-shift_idx] += 1.0
        grad[shift_idx:] -= h[:n-shift_idx]
    else:
        # shift_idx < 0
        # So we compute correlation (1-h) with h, so shift is -shift_idx
        shift_idx_pos = -shift_idx
        grad[shift_idx_pos:] += 1.0
        grad[:n-shift_idx_pos] -= h[shift_idx_pos:]
    return grad
