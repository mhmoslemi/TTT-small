```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from cvxpy import Problem, Minimize, Variable, Constraint

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n / 2
    
    # Build MILP model for binary case
    # Variables
    h = Variable(n, boolean=True)
    
    # Auxiliary variables for pairwise products
    y = {}
    constraints = []
    
    # Constraint 1: sum(h) = n/2
    constraints.append(cp.sum(h) == TARGET_SUM)
    
    # Define y_{i,k} = h_i AND h_{i+k} for valid shifts
    for k in range(n):  # k = 0, ..., n-1  
        for i in range(n - k):
            y[(i,k)] = Variable(boolean=True)
            constraints.extend([
                y[(i,k)] <= h[i],
                y[(i,k)] <= h[i + k],
                y[(i,k)] >= h[i] + h[i + k] - 1
            ])
    
    # Define correlation terms using auxiliary variables
    # c_k = sum_{i=0}^{n-k-1} (h_i - y_{i,k})
    # We'll use an epigraph to minimize max_k c_k
    
    # Epigraph: t >= c_k for all k
    t = Variable()
    for k in range(n):
        if k == 0:
            # c_0 = sum(h) - sum(y_{i,0}) = TARGET_SUM - sum(y_{i,0})
            # since h_i = 1 for all i, it's just: 0 - sum(y_{i,0})
            sum_y = cp.sum([y[(i, k)] for i in range(n - k)])
            constraints.append(t >= TARGET_SUM - sum_y)
        else:
            # c_k = sum(h[:n-k]) - sum(y_{i,k})
            sum_h = cp.sum(h[:n-k])
            sum_y = cp.sum([y[(i, k)] for i in range(n - k)])
            constraints.append(t >= sum_h - sum_y)
    
    # Objective
    objective = Minimize(t)
    
    # Formulate and solve
    prob = Problem(objective, constraints)
    
    # Solve with timeout
    time_left = budget_s / 2
    try:
        prob.solve(solver=cp.CBC, time_limit=time_left, verbose=False)
        h_opt = h.value
        c5_opt = t.value
    except:
        # Fallback to random initial solution
        h_opt = np.zeros(n, dtype=int)
        h_opt[:int(TARGET_SUM)] = 1
        np.random.shuffle(h_opt)
        c5_opt = evaluate_c5(h_opt)
    
    if not isinstance(h_opt, np.ndarray) or np.isnan(h_opt).any():
        # Fallback on local search
        current_h = h_opt
        current_c5 = evaluate_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        best_time = time.time()
        
        # Local search: simple swap
        for attempt in range(10000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Grey box: try pairs of indices to swap
            i_swap = np.random.choice(n)
            j_swap = np.random.choice(n)
            if i_swap == j_swap or (current_h[i_swap] == current_h[j_swap]):
                continue
                
            # Try the swap
            new_h = current_h.copy()
            new_h[i_swap], new_h[j_swap] = new_h[j_swap], new_h[i_swap]
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                    
        h_opt = best_h
        c5_opt = best_c5
    
    # If less than 0.3807, let's refine more
    if c5_opt > 0.3807:
        max_iter = 5000
        i = 0
        while i < max_iter and time.time() - start_time < budget_s - 1:
            # Try random discrete perturbations
            try:
                new_h = h_opt.copy()
                # Flip 2 positions
                pos1 = np.random.choice(n)
                pos2 = np.random.choice(n)
                new_h[pos1] = 1 - new_h[pos1]
                new_h[pos2] = 1 - new_h[pos2]
                # Normalize sum
                current_sum = np.sum(new_h)
                if current_sum > TARGET_SUM:
                    # Turn some 1s to 0s
                    ones = np.where(new_h == 1)[0]
                    np.random.shuffle(ones)
                    for idx in ones:
                        if current_sum <= TARGET_SUM:
                            break
                        new_h[idx] = 0
                        current_sum -= 1
                elif current_sum < TARGET_SUM:
                    # Turn some 0s to 1s
                    zeros = np.where(new_h == 0)[0]
                    np.random.shuffle(zeros)
                    for idx in zeros:
                        if current_sum >= TARGET_SUM:
                            break
                        new_h[idx] = 1
                        current_sum += 1
                # Recompute
                new_c5 = evaluate_c5(new_h)
                if new_c5 < c5_opt:
                    h_opt = new_h
                    c5_opt = new_c5
            except Exception:
                pass
            i += 1
    
    return (h_opt.tolist(), float(c5_opt), n)
```
