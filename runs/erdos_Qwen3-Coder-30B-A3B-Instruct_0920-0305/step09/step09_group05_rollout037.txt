1. **Discrete Combinatorial Search (Genetic Algorithm)**:
   - Represent h as a binary vector of length n with exactly n/2 ones (to satisfy integral constraint).
   - Use a Genetic Algorithm (GA) with population size 60, tournament selection, uniform crossover, and bit-flip mutation.
   - Fitness is the negative of the maximum overlap C₅ computed using FFT correlation.
   - Apply balanced crossover to maintain exact sum constraint.
   - Run for 200-500 generations based on remaining time budget.

2. **Stochastic Local Search (Simulated Annealing)**:
   - Start from the best binary individual found by GA.
   - Use Simulated Annealing (SA) to refine the binary solution.
   - Moves are swaps of 0 and 1 at random indices.
   - Use exponential cooling schedule: T(t) = T₀ * α^t, where α = 0.995.
   - Accept worse solutions with probability exp(-ΔC₅ / T).
   - Run until time budget is nearly exhausted.

3. **Continuous Gradient-based Refinement**:
   - Initialize from the binary solution refined by SA.
   - Treat the problem as a smooth minimization using log-sum-exp smoothing to approximate max.
   - Objective: minimize t such that all correlations c_k ≤ t.
   - Apply projected gradient descent using the smooth surrogate.
   - Use Nesterov momentum with adaptive step size.
   - Project back to feasible region after each update.

4. **Final Solution Validation**:
   - Evaluate final C₅ using FFT for correlation.

import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_binary_individual(n, target_sum):
    """Create a binary individual with exactly target_sum ones"""
    ind = np.zeros(n)
    indices = np.random.choice(n, target_sum, replace=False)
    ind[indices] = 1
    return ind

def evaluate_fitness_binary(individual):
    """Evaluate fitness of a binary individual"""
    # Convert binary to continuous with values 0 or 1
    h = individual.astype(float)
    return -evaluate_c5(h)  # Negative for minimization

def genetic_algorithm(seed, budget_s):
    """Perform genetic algorithm optimization on binary space"""
    np.random.seed(seed)
    start_time = time.time()
    
    population_size = 60
    generations = 200
    elite_size = 5
    
    # Initialize population
    population = [create_binary_individual(n_points, int(n_points/2)) for _ in range(population_size)]
    fitness_scores = [evaluate_fitness_binary(ind) for ind in population]
    
    best_individual = population[np.argmax(fitness_scores)].copy()
    best_fitness = np.max(fitness_scores) 
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Selection: tournament selection
        selected_indices = []
        for _ in range(population_size):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            selected_indices.append(winner_index)
        
        # Crossover and Mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness_scores)[-elite_size:]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_indices, 2, replace=False)
            # Uniform crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([population[parent1][:crossover_point], 
                                  population[parent2][crossover_point:]])
            
            # Repair to maintain exact sum
            ones_count = np.sum(child)
            if ones_count > n_points // 2:
                # Find zeros to convert to ones
                zero_indices = np.where(child == 0)[0]
                if len(zero_indices) >= ones_count - n_points // 2:
                    zeros_to_flip = np.random.choice(zero_indices, ones_count - n_points // 2, replace=False)
                    child[zeros_to_flip] = 1
                else:
                    child[:] = 1
                    child[:ones_count - n_points // 2] = 0
            elif ones_count < n_points // 2:
                # Find ones to convert to zeros
                one_indices = np.where(child == 1)[0]
                if len(one_indices) >= n_points // 2 - ones_count:
                    ones_to_flip = np.random.choice(one_indices, n_points // 2 - ones_count, replace=False)
                    child[ones_to_flip] = 0
                else:
                    child[:] = 0
                    child[:n_points // 2 - ones_count] = 1
            
            # Mutation: Flip random bit
            if np.random.rand() < 0.1:
                idx = np.random.randint(0, n_points)
                child[idx] = 1 - child[idx]
            
            new_population.append(child)
    
        # Update population
        population = new_population[:population_size]
        fitness_scores = [evaluate_fitness_binary(ind) for ind in population]
        
        current_best_idx = np.argmax(fitness_scores)
        if fitness_scores[current_best_idx] > best_fitness:
            best_fitness = fitness_scores[current_best_idx]
            best_individual = population[current_best_idx].copy()
    
    return best_individual, -best_fitness

def simulated_annealing_binary(initial_h, budget_s):
    """Perform simulated annealing on binary individuals"""
    start_time = time.time()
    current_h = initial_h.copy()
    current_c5 = evaluate_c5(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initial temperature
    T0 = 0.1
    alpha = 0.995
    T = T0
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Make a move: swap two random positions
        pos1, pos2 = np.random.choice(n_points, 2, replace=False)
        new_h = current_h.copy()
        new_h[pos1], new_h[pos2] = new_h[pos2], new_h[pos1]
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h)
        delta = new_c5 - current_c5
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cooling
        T *= alpha
    
    return best_h, best_c5

def smooth_max_objective(h, alpha=10.0):
    """Smooth approximation of max using log-sum-exp"""
    corr = compute_corr_fft(h)
    c5_values = c5_from_corr(corr, n_points)
    return -np.log(np.sum(np.exp(-alpha * c5_values))) / alpha

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Stage 1: Genetic Algorithm
    ga_h, ga_c5 = genetic_algorithm(seed, budget_s)
    
    if ga_c5 < best_c5:
        best_c5 = ga_c5
        best_h = ga_h.copy()
    
    # Stage 2: Simulated Annealing on binary
    sa_h, sa_c5 = simulated_annealing_binary(ga_h, budget_s)
    
    if sa_c5 < best_c5:
        best_c5 = sa_c5
        best_h = sa_h.copy()
    
    # Stage 3: Continuous refinement using projected gradient
    h_continuous = sa_h.copy()
    current_c5 = sa_c5
    
    # Use projected gradient descent on smoothed objective
    max_iter = 500
    learning_rate = 0.01
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient numerically (since we won't compute analytically)
        h_current = h_continuous.copy()
        corr = compute_corr_fft(h_current)
        c5_values = c5_from_corr(corr, n_points)
        
        # Objective: smooth max
        objective_scale = np.sum(np.exp(-10.0 * c5_values))
        gradient = np.zeros_like(h_current)
        
        # Simplified gradient by finite differences (momentum-style)
        eps = 1e-5
        for j in range(n_points):
            h_perturbed = h_current.copy()
            h_perturbed[j] += eps
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            corr_perturbed = compute_corr_fft(h_perturbed)
            c5_perturbed = c5_from_corr(corr_perturbed, n_points)
            
            diff = np.sum(np.exp(-10.0 * c5_perturbed)) - objective_scale
            gradient[j] = diff / eps
        
        # Update step with momentum and projection
        h_continuous = h_continuous - learning_rate * gradient
        h_continuous = project_to_sum_box(h_continuous, TARGET_SUM)
        
        current_c5 = evaluate_c5(h_continuous)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_continuous.copy()
            
    if best_h is None:
        best_h = ga_h.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
