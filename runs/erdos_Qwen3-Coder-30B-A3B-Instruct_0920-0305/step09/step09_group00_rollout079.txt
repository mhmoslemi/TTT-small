```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, 2*len(h)-1)
    ac = ifft(H * np.conj(H)).real
    return ac[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights for temperature alpha"""
    g_shifted = g - np.max(g)
    exp_g = np.exp(alpha * g_shifted)
    total = np.sum(exp_g)
    return exp_g / total

def gradient(h, p):
    """Compute gradient of smooth surrogate using convolutions"""
    # p * h
    p_h = np.convolve(p, h, mode='full')
    # p_rev * h
    p_rev_h = np.convolve(p[::-1], h, mode='full')
    
    # Only return the central part
    result = -(p_h[len(h)-1:] + p_rev_h[:len(h)])
    return result

def project_onto_sum_box(x, target_sum):
    """Project onto box constraints [0,1] with sum = target_sum using water filling"""
    def sum_with_lambda(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < tol:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def c5_true(h):
    """Compute true C5 using correlate"""
    h_complement = 1 - h
    corr = np.correlate(h, h_complement, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Build initial pattern: Thue-Morse sequence
    h_tm = np.array([(bin(i).count('1') % 2) for i in range(n_points)])
    h_tm = h_tm.astype(float)
    
    # Adjust sum to target
    h = project_onto_sum_box(h_tm, TARGET_SUM)
    
    best_h = h.copy()
    best_c5 = c5_true(h)
    
    # Multi-resolution refinement
    resolutions = [64, 128, 256]
    
    for n in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        if n == 64:
            h_interpolated = h
        elif n == 128:
            # Linear interpolation from 64 to 128
            h_upsampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h)), h)
            h_interpolated = project_onto_sum_box(h_upsampled, TARGET_SUM)
        else:  # n == 256
            # Linear interpolation from 128 to 256
            h_upsampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h)), h)
            h_interpolated = project_onto_sum_box(h_upsampled, TARGET_SUM)
        
        h = h_interpolated
        n_points = n
        dx = 2.0 / n_points
        TARGET_SUM = n_points / 2

        # Deterministic annealing with increasing alpha
        alpha = 1.0
        alpha_max = 2000 
        gamma = 1.3
        max_iter = 200
        step_size = 0.001
        
        for _ in range(100):  # Annealing iterations
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute auto-correlations  
            A = autocorr_fft(h)
            g = -A
            
            # Softmax weights
            p = softmax_weights(g, alpha)
            
            # Gradient
            grad = gradient(h, p)
            
            # Update
            h_new = h - step_size * grad
            h_new = project_onto_sum_box(h_new, TARGET_SUM)
            
            # Accept if improvement
            c5_new = c5_true(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            h = h_new
            alpha *= gamma
            step_size *= 0.995  # Decreasing learning rate
    
    # Discrete refinement with swaps
    h_discrete = (best_h > 0.5).astype(float)
    h_discrete = project_onto_sum_box(h_discrete, TARGET_SUM)
    c5_discrete = c5_true(h_discrete)
    
    if c5_discrete < best_c5:
        best_h = h_discrete
        best_c5 = c5_discrete
    
    # Final polish with SLSQP if time permits
    if time.time() - start_time < budget_s - 10:
        def objective(h_vec):
            return c5_true(h_vec)
        
        def constraint_eq(h_vec):
            return np.sum(h_vec) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            best_h = project_onto_sum_box(best_h, TARGET_SUM)
            best_c5 = c5_true(best_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
