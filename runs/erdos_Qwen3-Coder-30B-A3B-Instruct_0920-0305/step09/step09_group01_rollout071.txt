def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_c5(h, shift_idx):
    """Compute gradient of correlation at specific shift with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    
    # Convert shift index to actual shift value (shift = shift_idx - (n-1))
    shift = shift_idx - (n - 1)
    
    if shift >= 0:
        # For shift >= 0, we need to handle indices carefully
        # Correlation at shift: sum_{i=0}^{n-shift-1} h[i] * (1 - h[i+shift])
        # The gradient w.r.t. h[j] is:
        # 1. If j in [0, n-shift-1]: +1 - h[j+shift] 
        # 2. If j in [shift, n-1]: - h[j-shift]
        # But area is only for indices where both terms valid
        for i in range(n - shift):
            grad[i] += 1.0
            if i + shift < n:
                grad[i] -= h[i + shift]
        for i in range(shift, n):
            if i - shift >= 0:
                grad[i] -= h[i - shift]
    else:
        # For shift < 0, the correlation is shifted differently
        # Using symmetry, compute gradient for equivalent positive shift case
        # correlation at negative shift is computed identically to positive shift with sign reversal
        shift = -shift
        for i in range(n - shift):
            grad[i] += 1.0
            if i + shift < n:
                grad[i] -= h[i + shift]
        for i in range(shift, n):
            if i - shift >= 0:
                grad[i] -= h[i - shift]
    
    return grad
