import numpy as np
from scipy.signal import correlate
from scipy.special import logsumexp
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        best_h = h0.copy()

    # Stage A: Simulated Annealing on Binary Space
    def binary_to_continuous(h_bin, noise_level=0.01):
        h_cont = h_bin.astype(float) + np.random.uniform(-noise_level, noise_level, n_points)
        return project_to_sum_box(h_cont, TARGET_SUM)

    def compute_binary_corr(h_bin):
        h = h_bin.astype(float)
        corr = compute_corr_fft(h)
        return c5_from_corr(corr, n_points)

    # Simulated Annealing
    T0 = 0.5
    gamma = 0.995
    current_h_bin = np.random.permutation([1] * int(TARGET_SUM) + [0] * (n_points - int(TARGET_SUM)))
    current_c5 = compute_binary_corr(current_h_bin)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = binary_to_continuous(current_h_bin).copy()
    
    T = T0
    iterations = 0
    max_iterations = 100000
    
    while time.time() - start_time < budget_s * 0.2 and iterations < max_iterations:
        # Pick two random indices with different values
        ones = np.where(current_h_bin == 1)[0]
        zeros = np.where(current_h_bin == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Propose swap
        h_new = current_h_bin.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        new_c5 = compute_binary_corr(h_new)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h_bin = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = binary_to_continuous(current_h_bin).copy()
        
        T *= gamma
        iterations += 1

    # Stage B: Projected Gradient Descent on Smoothed Max
    def compute_gradient(h, c, alpha):
        # Compute weights
        c_alpha = alpha * c
        log_sum_exp = logsumexp(c_alpha)
        w = np.exp(c_alpha - log_sum_exp)
        
        # Compute gradient
        g = np.ones_like(h)
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                g[:n_points-k] -= w[k] * h[k:]
                g[k:] -= w[k] * h[:n_points-k]
            else:
                s = -k
                g[s:] -= w[k] * h[:n_points-s]
                g[:n_points-s] -= w[k] * h[s:]
        
        return g

    eta = 0.25

        # Trial step
        h_trial = current_h - eta * momentum

        current_h = project_to_sum_box(h_trial, TARGET_SUM)

        eta *= decay_rate
        
        # Update alpha
        if iter_b % 200 == 0:
            alpha = min(alpha * rho, alpha_max)
        
        # Early stopping based on improvement

            if eta < 1e-5:
                break
            eta *= 0.5
            last_improvement = iter_b
            
        iter_b += 1

    # Stage C: Sub-gradient Descent on True Max
    if best_h is None:
        current_h = h0.copy()
    else:
        current_h = best_h.copy()
    
    alpha = alpha_max
    eta_small = 0.02
    max_iter_c = 50
    
    # Use sub-gradient descent with true max
    for _ in range(max_iter_c):
        if time.time() - start_time >= budget_s * 0.95:
            break
            
        c = compute_corr_fft(current_h)
        max_val = np.max(c)
        active_indices = np.where(np.abs(c - max_val) < 1e-10)[0]
        
        if len(active_indices) == 0:
            break
            
        # Compute sub-gradient with uniform weights
        g = np.zeros_like(current_h)
        for k in active_indices:
            if k >= 0:
                g[:n_points-k] -= 1.0/len(active_indices) * current_h[k:]
                g[k:] -= 1.0/len(active_indices) * current_h[:n_points-k]
            else:
                s = -k
                g[s:] -= 1.0/len(active_indices) * current_h[:n_points-s]
                g[:n_points-s] -= 1.0/len(active_indices) * current_h[s:]
        
        # Trial step
        h_trial = current_h - eta_small * g
        
        # Project
        current_h = project_to_sum_box(h_trial, TARGET_SUM)
        
        # Check for improvement
        c_new = compute_corr_fft(current_h)
        new_c5 = c5_from_corr(c_new, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()

    # Final polishing with SLSQP
    try:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_erdos_solution(h)  # This will compute the exact value
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if res.success:
            polished_h = project_to_sum_box(res.x, TARGET_SUM)
            polished_c5 = evaluate_erdos_solution(polished_h)
            if polished_c5 < best_c5:
                best_c5 = polished_c5
                best_h = polished_h.copy()
                
    except:
        pass

    # Return final results
    return (best_h.tolist(), float(best_c5), n_points)
