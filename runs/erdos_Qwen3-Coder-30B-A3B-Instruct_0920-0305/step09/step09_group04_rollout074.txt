import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Smooth max approximation
    def smooth_max(h, eps=1.0):
        """Computes smooth max of correlations."""
        corr = compute_corr_fft(h)
        corr = corr * dx  # scale by dx
        max_val = np.max(corr)
        exp_corr = np.exp(eps * (corr - max_val))
        return (1/eps) * (np.log(np.sum(exp_corr)) + max_val)
    
    # Gradient of smooth max
    def smooth_max_grad(h, eps=1.0):
        corr = compute_corr_fft(h)
        corr = corr * dx
        max_val = np.max(corr)
        exp_corr = np.exp(eps * (corr - max_val))
        weights = exp_corr / np.sum(exp_corr)
        # weights times gradient contributions
        # This loop is roughly O(n^2) but could be optimized via FFT as well
        grad = np.zeros_like(h)
        for k in range(len(corr)):
            # indices in the overlap sum
            if k < len(h):
                for i in range(k, len(h)):
                    if i - k >= 0:
                        grad[i] += weights[k] * (1 - h[i-k])
                        grad[i-k] -= weights[k] * h[i]
        return grad
    
    # Adam optimizer
    def adam_update(h, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps_adam=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        return h_new, m, v
    
    # Multi-start optimization with warm restarts using Adam
    n_restarts = 20
    best_so_far = float('inf')
    best_result = None
    
    for restart in range(n_restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        # Generate initial guess
        h = np.random.random(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Optional: seed one restart with initial_h_values if available and it's a similar size
        if restart == 0:
            try:
                if len(initial_h_values) == n_points:
                    h = np.array(initial_h_values)
                    h = project_to_sum_box(h, TARGET_SUM)
            except:
                pass
        
        # Adam state
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        t = 1
        
        # Annealing schedule
        eps = 1.0
        eps_schedule = 1.5
        max_eps = 100.0
        lr = 0.2
        lr_schedule = 0.5
        
        # Inner loop
        patience = 0
        last_best = float('inf')
        
        max_iter = 4000
        for i in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            if i % 500 == 0 and eps < max_eps:
                eps = min(eps * eps_schedule, max_eps)
                lr *= lr_schedule
            
            # Compute smooth loss and grad
            loss = smooth_max(h, eps)
            grad = smooth_max_grad(h, eps)
            
            # Adam update
            h, m, v = adam_update(h, grad, m, v, t, lr=lr)
            t += 1
            
            # Project to feasible set
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Evaluation
            if i % 100 == 0:
                exact_c5 = evaluate_c5(h)
                if exact_c5 < best_so_far:
                    best_so_far = exact_c5
                    best_result = h.copy()
                
                if exact_c5 < last_best:
                    last_best = exact_c5
                    patience = 0
                else:
                    patience += 1
                
                if patience > 200:
                    break  # Early stopping on stagnation
    
    # Final polishing stage - binary rounding and local swaps
    if best_result is not None:
        # Thresholding
        sorted_vals = np.sort(best_result)
        median_val = sorted_vals[len(sorted_vals)//2]
        h_binary = (best_result > median_val).astype(float)
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        
        # Local swaps
        for _ in range(10000):
            if time.time() - start_time > 0.95 * budget_s:
                break
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if h_binary[i] == 1 and h_binary[j] == 0:
                # Temporarily swap
                old_c5 = evaluate_c5(h_binary)
                h_binary[i], h_binary[j] = 0, 1
                new_c5 = evaluate_c5(h_binary)
                if new_c5 >= old_c5:
                    h_binary[i], h_binary[j] = 1, 0  # Revert if worse
        
        final_c5 = evaluate_c5(h_binary)
        if final_c5 < best_so_far:
            best_so_far = final_c5
            best_result = h_binary.copy()
    
    if best_result is None:
        # Fallback
        h = np.array(initial_h_values)
        h = project_to_sum_box(h, TARGET_SUM)
        best_so_far = evaluate_c5(h)
        best_result = h.copy()
        
    return (best_result.tolist(), float(best_so_far), n_points)
