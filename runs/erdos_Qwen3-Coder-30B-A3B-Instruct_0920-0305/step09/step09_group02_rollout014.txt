import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT for efficiency"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, wgt):
    """Compute weighted gradient of smoothed max"""
    n = len(h)
    grad = np.zeros(n)
    # Compute weighted sum of gradients for each shift
    for k in range(1 - n, n):  # k ranges from -(n-1) to n-1
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i = 0 to n-k-1
            left = h[:n-k]
            right = h[k:]
            grad[:n-k] -= wgt[k+n-1] * right
            grad[k:] -= wgt[k+n-1] * left
        else:
            # k < 0
            # h[i] * (1 - h[i+k]) for i = -k to n-1
            left = h[-k:]
            right = h[:n+k]
            grad[-k:] -= wgt[k+n-1] * right
            grad[:n+k] -= wgt[k+n-1] * left
    grad += wgt.sum()  # Add 1 for all terms
    return grad
