# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_and_grad(h, alpha):
    """Compute smooth max and its gradient"""
    c = compute_corr_fft(h)
    max_alpha_c = np.max(alpha * c)
    exp_alpha_c = np.exp(alpha * c - max_alpha_c)
    sum_exp = exp_alpha_c.sum()
    w = exp_alpha_c / sum_exp
    smooth_val = (np.log(sum_exp) + max_alpha_c) / alpha
    n = len(h)
    g = np.ones(n)
    for idx, weight in enumerate(w):
        shift = idx - (n-1)
        if shift >= 0:
            g[:n-shift] -= weight * h[shift:]
            g[shift:]   -= weight * h[:n-shift]
        else:
            s = -shift
            g[s:]   -= weight * h[:n-s]
            g[:n-s] -= weight * h[s:]
    return smooth_val, g
