def compute_overlap(h, k, n):
    """Compute overlap for shift k"""
    # h is a list of length n
    # We compute sum_i h_i * (1 - h_{i+k})
    if k >= 0:
        # h_i * (1 - h_{i+k}) for i = 0..(n-k-1)
        # h_0 * (1 - h_k), h_1 * (1 - h_{k+1}), ..., h_{n-k-1} * (1 - h_{n-1})
        overlap = sum(h[i] * (1 - h[i+k]) for i in range(n-k))
    else:
        # h_{i+k} * (1 - h_i) for i = -k..(n-1)
        # h_{-k} * (1 - h_0), h_{1-k} * (1 - h_1), ..., h_{n-1} * (1 - h_{n+k-1})
        overlap = sum(h[i+k] * (1 - h[i]) for i in range(-k, n))
    return overlap

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Using correlate to compute all shifts
    corr = correlate(h, [1 - x for x in h], mode='full')
    max_corr = np.max(corr)
    return dx * max_corr
