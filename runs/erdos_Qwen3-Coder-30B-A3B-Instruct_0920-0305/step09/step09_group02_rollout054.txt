import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using numpy correlate"""
    return correlate(h, 1-h, mode="full")

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_box_and_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, wgt):
    """Compute gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, compute contribution to gradient
    for k in range(-n+1, n):
        # Calculate indices for the shifted versions
        if k >= 0:
            left = np.concatenate([np.zeros(k), h[:-k]])
            right = np.concatenate([h[k:], np.zeros(k)])
        else:
            left = np.concatenate([np.zeros(-k), h[-k:]])
            right = np.concatenate([h[:-k], np.zeros(-k)])
        
        # Add contribution: -wgt[k+n-1] * (left + right)
        grad += -wgt[k + n - 1] * (left + right)
    
    # Add contribution from constant term 1 for each position
    grad += np.ones(n)
    
    return grad

def smoothed_objective(h, beta):
    """Compute the smoothed objective with parameter beta"""
    corr = compute_corr(h)
    # Compute logsumexp of beta * corr
    max_val = np.max(beta * corr)
    exp_vals = np.exp(beta * (corr - max_val))
    return (1.0 / beta) * np.log(np.sum(exp_vals)) + max_val
