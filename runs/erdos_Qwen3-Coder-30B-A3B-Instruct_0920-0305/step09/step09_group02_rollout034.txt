import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, weights):
    """Compute gradient of smoothed max using weighted shifts"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # For each shift k (from -n+1 to n-1)
    for k in range(-(n-1), n):
        # Compute shifts
        if k >= 0:
            left = np.pad(h[:n-k], (k, 0), mode='constant', constant_values=0)
            right = np.pad(h[k:], (0, k), mode='constant', constant_values=0)
        else:
            left = np.pad(h[-k:], (0, -k), mode='constant', constant_values=0)
            right = np.pad(h[:n+k], (-k, 0), mode='constant', constant_values=0)
        
        # Weighted sum of gradients for this shift
        grad += weights[k + (n-1)] * (1 - left - right)
    
    # Add constant term from the 1 in the original f_k
    grad += np.ones_like(h)
    
    return grad

            exp_f = np.exp(f_shifted)
            wgt = exp_f / np.sum(exp_f)
