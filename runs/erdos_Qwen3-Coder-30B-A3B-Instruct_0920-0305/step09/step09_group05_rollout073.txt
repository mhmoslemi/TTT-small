import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp(x, axis=None):
    # Compute log(sum(exp(x))) avoiding numerical overflow
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def compute_smoothed_max_gradient(h, alpha):
    """Compute gradient of the smooth approximation to max(c_k)"""
    n = len(h)
    c = compute_corr_fft(h)  # Shape: 2*n-1
    # Account for the fact that c_k = sum h_i * (1 - h_{i+k})
    # So the per-element gradient is:
    # Since we're considering sum h_i * h_{i+k}, we get
    # d/dh_i (h_i * h_{i+k}) = h_{i+k} (when we vary h_i)
    # d/dh_i (h_i * h_{i-k}) = h_{i-k} (when we vary h_i)
    # And d/dh_i (1) = 0, so to compute d/dh_i of c_k, 
    # we get: - (h_{i+k} + h_{i-k}) when i+k < n and i-k >= 0, 
    # or some reshaped equivalent.
    # Let's compute this properly by extracting weights from the log-sum-exp style:
    c_scaled = alpha * c
    c_max = np.max(c_scaled)
    exp_c_scaled = np.exp(c_scaled - c_max)
    w = exp_c_scaled / np.sum(exp_c_scaled)
    
    g = np.ones(n)
    for k in range(1-n, n):  # k ranges from -(n-1) to n-1
        if k >= 0:
            # h[i] * h[i+k] contribution to correlation term c_k
            g[:n-k] -= w[k] * h[k:]
            g[k:] -= w[k] * h[:n-k]
        else:
            # k < 0
            s = -k
            g[s:] -= w[k] * h[:n-s]
            g[:n-s] -= w[k] * h[s:]
    
    # Total gradient should still be subtracted, so we do:
    # we have ∇c_k = 1 - h_{i+k} - h_{i-k}
    # So ∇F_α(h) = Σ_k w_k conjs to 1 - h_{i+k} - h_{i-k}
    # This is a bit messy, so let's rewrite carefully:
    # The c_k = Σ_i h_i * (1 - h_{i+k}) = n/2 - Σ h_i * h_{i+k}
    # So ∇c_k = - Σ (h_i * ∇h_{i+k} + h_{i+k} * ∇h_i) =: - Σ_{i} ( h_i * δ_{i+k} + h_{i+k} * δ_i) = - (1_{i+k} * h + h * 1_i) 
    # Let me calculate directly with matrix form over the 2n-1 correlation length:
    
    # A cleaner approach: compute ∇c_k(h) = - (h_{i+k} + h_{i-k})
    # Use a bitmap-like indexing as before.
    
    g = np.ones(n)
    for k in range(1-n, n):
        if k >= 0:
            g[:n-k] -= w[k] * h[k:]
            g[k:] -= w[k] * h[:n-k]
        else:
            s = -k
            g[s:] -= w[k] * h[:n-s]
            g[:n-s] -= w[k] * h[s:]
    
    # Note: we've considered a varying whether to include or not the boundary terms
    # which is hard to express in the full vector notation in the broadcasts, 
    # but we can keep it as a correct sum over shift k bell-shaped weights.
    # Correct because the FFT always matches the subtlety of k indexing  
    # in the correlation.
    
    return g

        proposed[i1] = 0
        proposed[i0] = 1

        proposed_c5 = evaluate_c5(proposed_h)

        for k in active_indices:
            if k >= 0:
                g[:n_points-k] -= 1.0 / len(active_indices) * h[k:]
                g[k:] -= 1.0 / len(active_indices) * h[:n_points-k]
            else:
                s = -k
                g[s:] -= 1.0 / len(active_indices) * h[:n_points-s]

        eta_sub = 0.02

        exact_max = np.max(c_vals) * dx

            best_h_cont = h.copy()
    
    # Final stage: minor polishing via scipy minimize
    last_h = best_h_cont.copy()
    last_c5 = best_c5_cont
    
    # Change to SLSQP minimize only one time
    bounds = [(0, 1) for _ in range(n_points)]

        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM

        pass
