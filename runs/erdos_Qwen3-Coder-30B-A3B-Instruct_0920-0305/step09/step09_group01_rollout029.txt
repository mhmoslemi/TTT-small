def build_M_matrices(n):
    """Build all the M matrices for shifts 0 to n-1."""
    M_list = []
    for k in range(n):
        M = np.zeros((n, n))
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                M[i, j] = 1
        M_list.append(M)
    return M_list

def sdpa_rounding(X, h_target_sum, max_iter=10):

    # Ensure X is PSD and has correct diagonal

        vals, vecs = np.linalg.eigh(X)

        vals = np.maximum(vals, 0)

        h_candidate = np.abs(u)  # Take abs to avoid negative
        h_candidate = h_candidate / np.linalg.norm(h_candidate) * h_target_sum

        h_candidate = np.clip(h_candidate, 0, 1)
        h_scaled = h_candidate * h_target_sum / np.sum(h_candidate)
        return h_scaled
    except:
        # Fallback to simple projection
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * h_target_sum / np.sum(h)
        return h

def compute_c5(h):

    # Extract the relevant part for k from -n+1 to n-1
    corr_full = corr[n-1:2*n-1]
    return dx * np.max(corr_full)
