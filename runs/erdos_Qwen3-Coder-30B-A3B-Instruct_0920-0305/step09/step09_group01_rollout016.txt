```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=150 as it's a good balance between performance and accuracy
    n = 150
    dx = 2.0 / n
    
    # Build the MILP model
    # Variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)  # y[i,k] = h[i] * h[i+k]
    t = cp.Variable()  # Epigraph variable
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n // 2)
    
    # Linearization of y[i,k] = h[i] * h[i+k]
    for i in range(n):
        for k in range(n - i):
            # y[i,k] <= h[i]
            constraints.append(y[i, k] <= h[i])
            # y[i,k] <= h[i+k]  
            constraints.append(y[i, k] <= h[i + k])
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y[i, k] >= h[i] + h[i + k] - 1)
    
    # Compute c_k = sum(h_i) - sum(y_{i,k})
    # We only consider k = 0 to n-1 (positive shifts)
    # Construct the constraints for epigraph
    for k in range(n):
        # sumh_k is sum of h from i=0 to n-k-1
        sumh_k = cp.sum(h[:n-k]) if k < n else 0
        # overlap_k is sum of y_{i,k} from i=0 to n-k-1
        overlap_k = cp.sum(y[:n-k, k]) if k < n else 0
        # c_k = sumh_k - overlap_k
        # We want t >= c_k for all k
        c_k = sumh_k - overlap_k 
        constraints.append(t >= c_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC (Time limit = budget_s / 2)
    prob.solve(solver=cp.CBC, timeout=budget_s / 2, log=True)
    
    best_h = h.value
    if best_h is None:  # Failed to solve fully
        # Fall back to LP relaxation
        h_relax = cp.Variable(n, boolean=False)
        y_relax = cp.Variable((n, n), boolean=False)
        t_relax = cp.Variable()
        
        # Same constraints but relaxed
        rel_constraints = []
        rel_constraints.append(cp.sum(h_relax) == n // 2)
        for i in range(n):
            for k in range(n - i):
                rel_constraints.append(y_relax[i, k] <= h_relax[i])
                rel_constraints.append(y_relax[i, k] <= h_relax[i + k])
                rel_constraints.append(y_relax[i, k] >= h_relax[i] + h_relax[i + k] - 1)
        
        for k in range(n):
            sumh_k = cp.sum(h_relax[:n-k]) if k < n else 0
            overlap_k = cp.sum(y_relax[:n-k, k]) if k < n else 0
            c_k = sumh_k - overlap_k 
            rel_constraints.append(t_relax >= c_k)
            
        rel_objective = cp.Minimize(t_relax)
        rel_prob = cp.Problem(rel_objective, rel_constraints)
        rel_prob.solve(solver=cp.CBC, timeout=budget_s / 4)
        
        if rel_prob.status in ["optimal", "optimal_inaccurate"]:
            # Apply rounding by sorting and selecting
            h_frac = h_relax.value
            ordered_indices = np.argsort(h_frac)[::-1]
            best_h = np.zeros(n)
            for i in range(n // 2):
                best_h[ordered_indices[i]] = 1
            
    # Polish with local search
    best_c5 = evaluate_c5(best_h)
    if best_h is None:
        best_h = np.zeros(n)
        best_h[:n//2] = 1
        
    # Local search: high-quality swaps (only if enough time)
    max_swaps = 5000
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        improved = False
        for i in range(n):
            if best_h[i] == 1:
                for j in range(n):
                    if best_h[j] == 0:
                        # Compute delta change in worst correlation
                        # For simplicity, we compute correlation manually
                        # Since h is binary, correlation changes are predictable
                        h_candidate = best_h.copy()
                        h_candidate[i] = 0
                        h_candidate[j] = 1
                        
                        # Evaluate new C5
                        c5_candidate = evaluate_c5(h_candidate)
                        
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                            improved = True
                            break
            if improved:
                break
    
    # Ensure final result is valid
    if not np.isclose(np.sum(best_h), n / 2):
        # Re-project to constraint
        h_proj = np.clip(best_h, 0, 1)
        h_proj = h_proj / np.sum(h_proj) * (n / 2)
        # Clip and reproject to integer
        h_final = np.zeros(n)
        indices = np.argsort(-h_proj)
        h_final[indices[:n//2]] = 1
        best_h = h_final
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n)
```
