import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full autocorrelation using FFT"""
    H = fft(h, 2*len(h)-1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute softmax weights for given alpha"""
    g_shifted = g - np.max(g)  # For numerical stability
    exp_g = np.exp(alpha * g_shifted)
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of the smooth surrogate"""
    # Compute two convolutions using FFT
    n = len(h)
    # p_rev = p[::-1] (reverse)
    p_rev = p[::-1]
    # Linear convolutions (zero-padded)
    conv1 = np.convolve(h, p, mode='full')[:2*n-1]
    conv2 = np.convolve(h, p_rev, mode='full')[:2*n-1]
    # We need the first n elements of the result
    grad = -(conv1[:n] + conv2[:n])
    return grad

def project_onto_sum_box(x, target):
    """Water-filling projection onto box constraints with fixed sum"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    # Binary search for lambda
    def f(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.5
        b += 0.5
        
    # Binary search with 30 iterations (sufficient for precision)
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x + (a + b)/2, 0, 1)

def c5_true(h):
    """Exact evaluation of C5 using correlate"""
    n = len(h)
    dx = 2.0 / n
    h_complement = 1 - h
    # Compute all correlations with the complement
    corr = np.correlate(h, h_complement, mode="full")
    # Return the maximum overlap
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    alpha_start = 1.0
    alpha_max = 2000.0
    gamma = 1.3
    step_size = 0.01
    max_iter_per_alpha = 200
    n_list = [64, 128, 256]  # Different resolutions
    
    # Best solution so far
    best_h = None
    best_c5 = np.inf
    
    # Run optimization at different resolutions
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        
        # Generate initial vectors
        candidates = []
        
        # 1. Thue-Morse pattern
        h_tm = np.array([bin(i).count('1') % 2 for i in range(n)])
        h_tm = h_tm * (target_sum / np.sum(h_tm))  # Scale to target sum
        candidates.append(h_tm)
        
        # 2. Random binary
        for _ in range(3):
            h_rand = np.random.randint(0, 2, n).astype(float)
            h_rand = h_rand * (target_sum / np.sum(h_rand))
            candidates.append(h_rand)
        
        # Keep the best initial candidate
        best_candidate = min(candidates, key=c5_true)
        
        # Initialize h
        h = project_onto_sum_box(best_candidate, target_sum)
        
        # Annealing loop
        alpha = alpha_start
        while alpha <= alpha_max:
            if time.time() - start_time > budget_s * 0.90:
                break
                
            for _ in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute auto-correlations
                A = autocorr_fft(h)
                g = -A  # negative to match the definition
                
                # Compute weights
                p = softmax_weights(g, alpha)
                
                # Compute gradient
                grad = gradient(h, p)
                
                # Update
                h_new = h - step_size * grad
                h_new = project_onto_sum_box(h_new, target_sum)
                
                # Accept if improved
                c5_new = c5_true(h_new)
                h = h_new.copy()
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    
                if time.time() - start_time > budget_s:
                    break
                    
            # Increase temperature
            alpha *= gamma
            
            # Optionally reduce step size
            if alpha > 100:
                step_size *= 0.95
                
    # Post-process with discrete swaps
    if best_h is not None and time.time() - start_time < budget_s - 10:
        h_binary = (best_h > 0.5).astype(float)
        h_binary = h_binary * (target_sum / np.sum(h_binary))  
        
        # Discrete hill-climbing swaps
        for _ in range(5000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Pick one 1 and one 0 at random
            ones = np.where(h_binary > 0.5)[0]
            zeros = np.where(h_binary < 0.5)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Swap
            h_test = h_binary.copy()
            h_test[i], h_test[j] = h_test[j], h_test[i]
            c5_test = c5_true(h_test)
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                
    # Final SLSQP optimization
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def objective(h):
            return c5_true(h)
        
        def constraint_eq(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(len(best_h))]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_h = res.x
                best_c5 = c5_true(best_h)
        except:
            pass
            
    # Return final solution
    if best_h is not None:
        return (best_h.tolist(), float(best_c5), len(best_h))
    else:
        # Fallback to a simple construction
        n_final = 128
        h_fallback = np.ones(n_final) * 0.5
        c5_fallback = c5_true(h_fallback)
        return (h_fallback.tolist(), float(c5_fallback), n_final)
