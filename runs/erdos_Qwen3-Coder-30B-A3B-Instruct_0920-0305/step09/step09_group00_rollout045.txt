TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    def objective(h):

        return np.sum(h) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}
    
    res = differential_evolution(

        maxiter=int(1000 - (time.time() - start_time) * 5),
        popsize=15,

        workers=1

        c5_result = evaluate_c5(h_result)
        return h_result, c5_result
    
    return h0, evaluate_c5(h0)

def multi_restart_sa(h0, start_time, budget_s):

    best_c5 = evaluate_c5(h0)

    for restart in range(10):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random initial solution
        h_start = h0 + np.random.normal(0, 0.05, n_points)
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Simulated annealing
        current_h = h_start.copy()
        current_c5 = evaluate_c5(h_start)
        
        # Cooling schedule
        temp = 0.1
        cooling_rate = 0.999
        max_iter = 5000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution
            h_new = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            
            # Accept/reject
            if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
                current_h = h_new.copy()
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
            temp *= cooling_rate
            
    return best_h, best_c5
