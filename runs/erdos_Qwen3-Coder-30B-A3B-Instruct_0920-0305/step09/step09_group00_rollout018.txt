import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, n=2*len(h)-1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    return exp_g / np.sum(exp_g)

def gradient(h, p):
    """Compute gradient of the smooth surrogate"""
    # Compute convolutions
    n = len(h)
    p_rev = p[::-1]
    
    # Linear convolutions
    r1 = np.convolve(p, h, mode='full')[:2*n-1]
    r2 = np.convolve(p_rev, h, mode='full')[:2*n-1]
    
    # Extract the first n components
    grad = -(r1[:n] + r2[:n])
    return grad

def project_onto_sum_box(x, target_sum):
    """Euclidean projection onto box constraints with sum constraint (water-filling)"""
    # Clip to [0, 1]
    x_clipped = np.clip(x, 0, 1)
    
    # Binary search for lambda
    def eval_sum(lam):
        proj = np.clip(x_clipped + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(30):
        m = (a + b) / 2
        fm = eval_sum(m)
        if abs(fm) < 1e-10:
            break
        if fm * eval_sum(a) < 0:
            b = m
        else:
            a = m
    
    # Final projection
    lam = (a + b) / 2
    result = np.clip(x_clipped + lam, 0, 1)
    return result

def c5_true(h):
    """Exact evaluation of C5 using np.correlate"""
    n = len(h)
    dx = 2.0 / n
    h_comp = 1 - h
    
    # Compute correlation (full mode)
    corr = np.correlate(h, h_comp, mode='full')
    
    # Correct for the offset in correlation indexing
    # abs_corr = np.abs(corr)
    # For maximizing, we look at the positive correlations
    # But since h(x) * (1-h(x+k)) is what we want
    # Better to manually compute all terms
    
    # We want to compute max_k int h(x) * (1-h(x+k)) dx 
    # This equals max_k [ int h(x) dx - int h(x) * h(x+k) dx ] 
    # Since int h(x) dx = 1, this becomes min_k int h(x) * h(x+k) dx
    
    # So we want to minimize the autocorrelation: 
    # min_k sum_i h_i * h_{i+k}
    
    # For that, we look at the correlation values directly, and 
    # avoid polynomial conversions
    
    # Actually, just compute the maximum of correlation out of all valid shifts
    # Only the valid shifts in the range (0 to 2-k) which means valid for k=0 to n-1
    
    # Both h and h_comp are of same length
    # The full correlation of 1-h with h gives correct samples
    # But can take always the corr and find the max
    # This is better to just to full correlation and extract the properly sized
	
    overlap = np.zeros(n)
    for k in range(n):
        # Evaluate int h(x) * (1-h(x+k)) dx over the interval where both exist
        if k == 0:
            overlap[k] = np.sum(h * (1 - h)) * dx
        else:
            # Mask for valid x values
            valid_len = n - k
            if valid_len <= 0:
                continue
            h_valid = h[:valid_len]
            h_comp_valid = h_comp[k:k+valid_len]
            overlap[k] = np.sum(h_valid * h_comp_valid) * dx
    
    max_overlap = np.max(overlap)
    return max_overlap

def compute_c5_value(h):
    """Compute objective from the smooth surrogate approach"""
    n = len(h)
    dx = 2.0 / n
    
    # Auto-correlation (directly)
    A = autocorr_fft(h)
    g = -A  # Negative auto-correlations
    
    # Soft max approximation is unstable without good alpha
    # In primal setting, let's compute using real data
    return c5_true(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [64, 128, 256]
    best_h = None
    best_c5 = float('inf')
    final_n = 128
    
    for n in n_values:
        if time.time() - start_time > budget_s:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        
        # Initialize h - first try Thue-Morse pattern
        h_init = np.zeros(n)
        for i in range(n):
            h_init[i] = (bin(i).count('1') % 2)
        
        # Ensure sum matches
        h_init = project_onto_sum_box(h_init, target_sum)
        
        # Optionally try a few random binary patterns for better initialization
        for _ in range(min(10, 50000 // n)):
            if time.time() - start_time > budget_s:
                break
            trial = np.random.randint(0, 2, n).astype(float)
            trial = project_onto_sum_box(trial, target_sum)
            c5_trial = compute_c5_value(trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = trial.copy()
                final_n = n
                
        # Run deterministic annealing loop for current n
        # Start with best initialized solution
        h = h_init.copy()
        current_c5 = compute_c5_value(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            final_n = n
        
        # Optimality procedure: gradient-based method
        # The process: deterministic gradient and softmax surrogate optimization
        # Step size and number of iterations based on time budget considerations

        # Determine number of iterations based on resolution
        # Larger resolution gets more iterations
        max_iterations = max(200, 10000 // n)

        # Prepare annealing parameter schedule
        alpha_min, alpha_max, alpha_gamma = 1.0, 2000.0, 1.3
        alpha = alpha_min

        # Loop of gradient descent with increasing alpha (annealing)
        iteration = 0
        while alpha <= alpha_max and iteration < max_iterations:
            if time.time() - start_time > budget_s:
                break
                
            # Compute auto-correlations
            A = autocorr_fft(h)
            g = -A
            
            # Compute soft-max weights 
            p = softmax_weights(g, alpha)
            
            # Compute gradient
            grad = gradient(h, p)
            
            # Step size for updates
            step_size = 0.001 * (1.0 / (alpha + 1))  
            
            # Update step
            h_new = h - step_size * grad
            
            # Project back onto feasible set
            h_new = project_onto_sum_box(h_new, target_sum)
            
            # Check if the new value improves
            new_c5 = compute_c5_value(h_new)
            if new_c5 < current_c5:
                h = h_new.copy()
                current_c5 = new_c5
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    final_n = n
                    
            iteration += 1
            
            # Increase alpha (into higher penalty on violation)
            alpha *= alpha_gamma
            
        # Optimization rounds completed for current grid size
    
    # Post-process discrete improvement (swaps)
    if best_h is not None:
        h_current = best_h.copy()
        c5_current = compute_c5_value(h_current)
        
        # Do a few rounds of hill climbing with swaps
        for _ in range(5000):
            if time.time() - start_time > budget_s:
                break
                
            # Select two random indices - one in ones, one in zeros
            ones_idx = np.where(h_current > 0.5)[0]
            zeros_idx = np.where(h_current < 0.5)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                # If we have no ones or zero, skip
                break
                
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Swap values
            h_new = h_current.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            # Evaluate true C5
            c5_new = compute_c5_value(h_new)
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
    
    # Final polishing with scipy optimizer if time remains
    if best_h is not None and time.time() - start_time < budget_s - 5:
        def objective(h_vec):
            return compute_c5_value(h_vec)
        
        def constraint_eq(h_vec):
            return np.sum(h_vec) - target_sum  

        bounds = [(0, 1) for _ in range(final_n)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_finer = project_onto_sum_box(res.x, target_sum)
                c5_finer = compute_c5_value(h_finer)
                if c5_finer < best_c5:
                    best_c5 = c5_finer
                    best_h = h_finer
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), final_n)
