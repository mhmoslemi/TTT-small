import numpy as np
from scipy.signal import correlate
import time
from cvxpy import*

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n / 2
    
    # Binary integer programming approach
    # Variables: h[i] in {0,1}, y[i,k] = h[i] AND h[i+k] for k=0..n-1
    # The constraint is to minimize t such that t >= c_k for all k
    # where c_k = sum h_i - sum y_{i,k} over all valid i
    
    # Create variables
    h = Variable(n, integer=True)
    y = {}  # Dictionary of binary variables: y[i,k]  where k = 0..n-1 and i = 0..n-k-1
    t = Variable()
    
    # Constraints setup
    constraints = []
    
    # Binary constraints
    constraints += [h >= 0]
    constraints += [h <= 1]
    
    # Sum constraint
    constraints += [sum(h) == TARGET_SUM]
    
    # Make sure our y variables exist for all (i,k)
    # Typical y_{i,k} = h_i AND h_{i+k}
    # Each y_{i,k} has a constraint that derives from AND logic
    
    # Simple way to compute constraints for this:
    # For i=0..n-k-1, we have y_{i,k} <= h_i, y_{i,k} <= h_{i+k}, and y_{i,k} >= h_i + h_{i+k} - 1
    # Also, y_{i,k} in [0,1] is binary
    
    # Let's simplify the approach and directly implement a greedy optimization that preserves key structure
    
    # First define helper functions for computing the correlation more accurately
    def compute_corr(h_vals):
        # Use numpy correlate with full mode
        h_rev = h_vals[::-1]
        b = 1 - h_vals  
        corr = np.correlate(h_rev, b, mode="full")
        return corr
    
    def c5_from_corr(corr, n):
        return np.max(corr) * dx
    
    # We started with a known good solution
    h_start = np.array(initial_h_values)
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start / np.sum(h_start) * TARGET_SUM
    
    # Convert to binary using a heuristic rounding method
    # Start with greedy rounding: sort by h_vals and set the first half to 1
    
    ranks = np.argsort(h_start)[::-1]
    h_binary = np.zeros(n)
    h_binary[ranks[:int(TARGET_SUM)]] = 1
    
    # Initial approximation
    c5 = c5_from_corr(compute_corr(h_binary), n)
    
    # Now apply the pair swap heuristic (fast and effective)
    current_h = h_binary.copy()
    best_h = current_h.copy()
    best_c5 = c5
    
    # Pair neighborhood swap local search
    num_swaps = 0
    max_swaps = 10000  # limit swaps to prevent slow progress
    
    while num_swaps < max_swaps and (time.time() - start_time) < budget_s - 2:
        improved = False
        # For each 1 in the current solution, find a 0 that improves the max corr value
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        
        # Note: we're searching for swaps that improve the worst-case correlation
        # which is a delayed evaluation: we have to compute correlations for all k
        # and then find the max. We will do a local update of those K values as we swap
        
        # For simplicity, let's look for the most beneficial single swap
        best_swap_delta = 0
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                # Create a temporary vector with swapped h[i] and h[j]
                temp_h = current_h.copy()
                temp_h[i] = 0
                temp_h[j] = 1
                
                # We'll compute correlation for this new h vector
                corr = compute_corr(temp_h)
                c5_new = c5_from_corr(corr, n)
                
                if c5_new < best_c5:
                    # Track the most beneficial single change
                    delta = best_c5 - c5_new
                    if delta > best_swap_delta:
                        best_swap_delta = delta
                        best_i = i
                        best_j = j
                        
        if best_i != -1:
            current_h[best_i] = 0
            current_h[best_j] = 1
            best_c5 = c5_from_corr(compute_corr(current_h), n)
            improved = True
            num_swaps += 1
            
        if not improved:
            break
            
    # Final refinement: minimize using a gradient-free routine like Nelder-Mead
    # We use a numpy-based warm-start for the local optimization if remaining time allows
    
    if time.time() - start_time < budget_s - 2:
        try:
            # Let's try a few deterministic refinements by adjusting 2-3 points
            h_refined = best_h.copy()
            original_c5 = best_c5
            
            # Generate candidate solutions by performing small perturbations 
            # We'll hold the sum constraint to only change positions that are near the extremes
            
            # Perturb a small block and evaluate
            max_perturb = 5
            test_points = np.random.choice(np.arange(n), size=min(max_perturb, n//4), replace=False)
            best_perturbed_h = h_refined.copy()
            best_perturbed_c5 = original_c5
            
            for i, point in enumerate(test_points):
                if h_refined[point] == 1:
                    # Try to reduce it to 0 and see if we can two-hop improve
                    h_down = h_refined.copy()
                    h_down[point] = 0
                    corr_down = compute_corr(h_down)
                    c5_down = c5_from_corr(corr_down, n)
                    
                    if c5_down < best_perturbed_c5:
                        best_perturbed_h = h_down
                        best_perturbed_c5 = c5_down
                else:
                    # Try to increase to 1
                    h_up = h_refined.copy()
                    h_up[point] = 1
                    corr_up = compute_corr(h_up)
                    c5_up = c5_from_corr(corr_up, n)
                    
                    if c5_up < best_perturbed_c5:
                        best_perturbed_h = h_up
                        best_perturbed_c5 = c5_up
                        
            # If we improved, use the refinement
            if best_perturbed_c5 < best_c5:
                h_refined = best_perturbed_h
                best_c5 = best_perturbed_c5
                
        except:
            pass  # Silently ignore refinement failure
    
    # Final evaluation before returning
    final_h = h_refined if 'h_refined' in locals() else best_h
    final_c5 = c5_from_corr(compute_corr(final_h), n)
    
    return (final_h.tolist(), float(final_c5), n)
