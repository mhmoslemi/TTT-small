import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_vector(h):
    """Compute the correlation vector f_k(h) = sum_i h[i] * (1 - h[i+k])"""
    return np.correlate(h, 1-h, mode='full')

def compute_weighted_gradient(h, wgt):
    """Compute weighted gradient for the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    # Process all shifts for the weighted gradient
    for k in range(-n+1, n):
        if k >= 0:
            left = h[k:]
            right = h[:n-k]
        else:
            left = h[-k:]
            right = h[:k]
        # Pad with zeros to make the arrays the same length for addition
        left_padded = np.pad(left, (0, max(0, n - len(left)), 0, len(right) - n) if len(right) < n else (0, 0), constant_values=0)
        right_padded = np.pad(right, (max(0, n - len(right)), 0, 0, len(left) - n) if len(left) < n else (0, 0), constant_values=0)
        grad += wgt[k+n-1] * (1 - right_padded - left_padded)
    
    # Add the contribution from 1 (sum is 1 constraint)
    grad += np.ones(n)
    return grad

def project_onto_box_and_sum(v, w):
    """Project v onto box [0,1] and sum constraint w"""
    # First, clip to box constraints
    v = np.clip(v, 0, 1)
    
    # Then adjust to achieve the correct sum
    # Binary search for lambda
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - w

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12

        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result
