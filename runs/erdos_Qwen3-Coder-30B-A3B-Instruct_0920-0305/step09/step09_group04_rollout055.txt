def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_c_s(h, s):
    """Compute gradient of overlap c_s (for shift s >= 0)"""
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad = np.ones(n) - 2 * h
    else:
        # For s > 0
        # Contribution from (1-h_{i-s}) where i >= s
        temp = np.zeros(n)
        temp[s:] = 1 - h[:-s]
        # Contribution from -h_{i+s} where i <= n-1-s
        temp2 = np.zeros(n)
        temp2[:-s] = -h[s:]
        grad = temp + temp2
    return grad

def compute_gradients(h):
    """Compute gradient of smooth max"""
    n = len(h)
    # Just compute a few dominant shifts 
    grads = []
    for k in range(n):
        if k == 0:
            grads.append(gradient_of_c_s(h, 0))
        else:
            # For positive shifts
            grads.append(gradient_of_c_s(h, k))
            # For negative shifts
            grads.append(gradient_of_c_s(h, -k))
    return grads

def smooth_max_gradient(h, eps=1.0):
    """Compute gradient of smooth max"""
    n = len(h)
    # Compute all c_k values
    c_vals = []
    for k in range(-n+1, n):
        if k >= 0:
            # Forward shift
            if k == 0:
                # c_0 = sum h_i (1 - h_i)
                c_k = np.sum(h * (1 - h))
            else:
                # c_k = sum h_{i+k} (1-h_i) for i = 0 to n-1-k  
                c_k = np.sum(h[k:] * (1 - h[:-k]))
        else:
            # Backward shift
            k_abs = -k
            # c_k = sum h_{i+k} (1-h_i) for i = 0 to n-1+k
            c_k = np.sum(h[:n-k_abs] * (1 - h[k_abs:]))
        c_vals.append(c_k)
    
    # Softmax weights
    c_vals = np.array(c_vals)
    # Subtract max to prevent overflow
    c_vals_shifted = c_vals - np.max(c_vals)
    w = np.exp(eps * c_vals_shifted)
    w = w / np.sum(w)
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(c_vals)):
        k = i - (n - 1)
        # k ranges from -(n-1) to n-1
        # Extract gradient for shift k
        if k == 0:
            g = np.ones(n) - 2 * h
        elif k > 0:
            # Forward shift, shape = [1-h_0,...,1-h_{n-k-1}] + [-h_k,...,-h_{n-1}] 
            g = np.zeros(n)
            g[k:] = 1 - h[:-k]
            g[:-k] -= h[k:]
        else:
            # Backward shift
            g = np.zeros(n)
            k_abs = -k
            g[:-k_abs] = 1 - h[k_abs:]
            g[k_abs:] -= h[:-k_abs]
        grad += w[i] * g
    
    return grad
