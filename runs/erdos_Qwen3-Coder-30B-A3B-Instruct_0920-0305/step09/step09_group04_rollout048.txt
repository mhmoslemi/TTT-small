import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Convex-concave procedure (DC programming) approach
    def evaluate_c5_dc(h):
        """Evaluate C5 using direct computation for DC approach"""
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_to_sum_box(x, target_sum):
        """Project vector x onto box constraints [0,1] with sum = target_sum"""
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        
        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
        return result
    
    # Generate initial feasible solution
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # DC optimization approach
    best_h = h0.copy()
    best_c5 = evaluate_c5_dc(h0)
    
    max_iter = 100
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Construct convex relaxation using current point
        current_h = best_h.copy()
        current_c5 = evaluate_c5_dc(current_h)
        
        # Use a heuristic refinement approach
        # Random local search around current solution
        num_changes = min(8, n_points // 8)
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        improved = False
        for idx in indices:
            old_val = current_h[idx]
            # Try various small perturbations
            for delta in [-0.03, -0.02, -0.01, 0.01, 0.02, 0.03]:
                current_h[idx] = old_val + delta
                if current_h[idx] < 0 or current_h[idx] > 1:
                    current_h[idx] = old_val
                    continue
                
                current_h = project_to_sum_box(current_h, TARGET_SUM)
                new_c5 = evaluate_c5_dc(current_h)
                
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                    improved = True
                    break
                else:
                    current_h[idx] = old_val
            
            if improved:
                break
    
    # Final local search on best solution
    current_h = best_h.copy()
    current_c5 = evaluate_c5_dc(current_h)
    
    for _ in range(300):
        if time.time() - start_time > budget_s * 0.99:
            break
        indices = np.random.choice(n_points, 5, replace=False)
        for idx in indices:
            old_val = current_h[idx]
            # Try more fine-grained perturbations
            for delta in [-0.015, -0.005, 0.005, 0.015]:
                current_h[idx] = old_val + delta
                if 0 <= current_h[idx] <= 1:
                    current_h = project_to_sum_box(current_h, TARGET_SUM)
                    new_c5 = evaluate_c5_dc(current_h)
                    if new_c5 < current_c5:
                        current_c5 = new_c5
                        best_h = current_h.copy()
                        break
                    else:
                        current_h[idx] = old_val
                else:
                    current_h[idx] = old_val
    
    return (best_h.tolist(), float(best_c5), n_points)
