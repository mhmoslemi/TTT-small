import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 value using the provided function"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set problem size
    n = 150
    dx = 2.0 / n
    
    # Create a MILP model
    # Variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n // 2,  # Sum constraint
        t >= 0  # Epigraph variable constraint
    ]
    
    # Linearization constraints for y[i,k] = h[i] AND h[i+k]
    for k in range(n):
        for i in range(n - k):
            constraints.extend([
                y[i, k] <= h[i],
                y[i, k] <= h[i + k],
                y[i, k] >= h[i] + h[i + k] - 1
            ])
    
    # Correlation calculation: c_k = sum(h_i) - sum(y_{i,k})
    # We only need to consider one side of the correlation for symmetry
    for k in range(n):
        # Compute sum(h_i) for the k-shift
        sum_h = cp.sum(h[:n-k]) if k < n else 0
        # Compute sum(y_{i,k})
        sum_y = cp.sum(y[:n-k, k]) if k < n else 0
        # c_k = sum_h - sum_y
        c_k = sum_h - sum_y
        constraints.append(t >= c_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Build model
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=budget_s/2, log_likelihood=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
        else:
            # Fallback to LP relaxation
            h.value = None
            prob.solve(solver=cp.CBC, time_limit=budget_s/4, log_likelihood=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h.value
            else:
                # Fallback to simple heuristic
                h_opt = np.zeros(n)
                h_opt[:n//2] = 1
    except Exception:
        # Last resort - simple construction
        h_opt = np.zeros(n)
        h_opt[:n//2] = 1
    
    # Convert to binary if necessary
    h_opt = np.round(h_opt).astype(int)
    
    # Local improvement using pair swap heuristic
    def compute_correlation(h_vec):
        corr = np.correlate(h_vec, 1-h_vec, mode="full")
        return np.max(corr)
        
    best_h = h_opt.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simple local improvement: try to swap positions
    improvements = 0
    max_iter = 1000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        improved = False
        
        # Try all possible swaps
        for i in range(n):
            if h_opt[i] == 1:  # look for 1s to move
                for j in range(n):
                    if h_opt[j] == 0:  # look for 0s to fill
                        # Try a swap and see if it improves C5
                        h_test = h_opt.copy()
                        h_test[i] = 0
                        h_test[j] = 1
                        
                        test_c5 = evaluate_erdos_solution(h_test)
                        if test_c5 < best_c5:
                            best_h = h_test.copy()
                            best_c5 = test_c5
                            h_opt = h_test.copy()
                            improved = True
                            improvements += 1
                            break
            if improved:
                break
                
    # If still very close to previous best, use more refined refinement
    if best_c5 > 0.3809:  
        # Try a better refinement
        h_refined = np.zeros(n)
        # Place 1s in a more structured way
        idxs = np.arange(n)
        np.random.shuffle(idxs)
        h_refined[idxs[:n//2]] = 1
        refined_c5 = evaluate_erdos_solution(h_refined)
        if refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = refined_c5
            
    return (best_h.tolist(), float(best_c5), n)
