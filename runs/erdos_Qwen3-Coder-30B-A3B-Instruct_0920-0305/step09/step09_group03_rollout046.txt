import numpy as np
import time
from numpy.fft import fft, ifft

# ------------------------------------------------------------
# Helper: projection of a vector onto [0,1]^n with a prescribed sum
# ------------------------------------------------------------
def project_to_sum_box(x, target_sum, tol=1e-12, max_iter=100):
    """Project x onto the intersection of the box [0,1]^n and the hyperplane sum(x)=target_sum.
    Returns the projected vector."""
    # First clip to the box
    x = np.clip(x, 0.0, 1.0)
    # If the sum is already correct we are done
    cur = x.sum()
    if abs(cur - target_sum) <= tol:
        return x

    # Binary search for lambda such that sum(clip(x+λ,0,1)) = target_sum
    # The function f(λ) = sum(clip(x+λ,0,1)) is monotone increasing.
    lo, hi = -1.0, 1.0
    # Expand the interval until the signs differ
    def f(lam):
        return np.clip(x + lam, 0.0, 1.0).sum() - target_sum

    while f(lo) * f(hi) > 0.0:
        lo -= 1.0
        hi += 1.0

    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        fm = f(mid)
        if abs(fm) <= tol:
            break
        if fm > 0:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    return np.clip(x + lam, 0.0, 1.0)

# ------------------------------------------------------------
# Core evaluation: exact C5 for a discrete h
# ------------------------------------------------------------
def exact_c5(h, dx):
    """Exact C5 (max correlation) for a step function h."""
    r = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(r)

# ------------------------------------------------------------
# Soft‑max weights (numerically stable)
# ------------------------------------------------------------
def softmax_weights(f, alpha):
    """Return the soft‑max probabilities w_k proportional to exp(alpha * f_k)."""
    if alpha == 0.0:
        # uniform weights
        w = np.full_like(f, 1.0 / f.size)
        return w
    af = alpha * f
    m = af.max()
    exp_shifted = np.exp(af - m)
    w = exp_shifted / exp_shifted.sum()
    return w

# ------------------------------------------------------------
# One gradient step based on the smooth‑max of the overlaps
# ------------------------------------------------------------
def gradient_step(h, alpha, eta, target_sum):
    """
    Perform a single projected gradient descent step on the smoothed objective.
    Returns the updated h and the soft‑max weights (useful for debugging).
    """
    n = h.size
    # 1) compute the overlap vector f_k = Σ_i h_i (1 - h_{i+k})
    f = np.correlate(h, 1.0 - h, mode='full')
    # 2) soft‑max weights
    w = softmax_weights(f, alpha)
    # 3) convolutions needed for the gradient
    w_rev = w[::-1]
    # full convolutions (length 2n-1 + n-1 = 3n-2)
    conv_rev = np.convolve(w_rev, h, mode='full')
    conv_fwd = np.convolve(w, h, mode='full')
    # central part corresponds to indices n-1 … n-1+n-1
    start = n - 1
    conv_sum = conv_rev[start:start + n] + conv_fwd[start:start + n]
    # gradient of the smoothed max: ∇T = 1 - conv_sum
    # update direction = -∇T = conv_sum - 1
    direction = conv_sum - 1.0
    # step
    h_new = h + eta * direction
    # project back to feasible set
    h_new = project_to_sum_box(h_new, target_sum)
    return h_new, w, f

# ------------------------------------------------------------
# Generate a random feasible vector (binary or continuous)
# ------------------------------------------------------------
def random_feasible(n, target_sum):
    """Generate a random point in [0,1]^n with the correct sum."""
    x = np.random.rand(n)
    return project_to_sum_box(x, target_sum)

# ------------------------------------------------------------
# Structured cosine‑based initial guess
# ------------------------------------------------------------
def cosine_initial(n, target_sum):
    """Low‑frequency cosine ansatz, clipped and projected."""
    xs = np.linspace(0.0, 2.0, n, endpoint=False)
    a = np.random.uniform(-0.45, 0.45)
    b = np.random.uniform(-0.45, 0.45)
    h = 0.5 + a * np.cos(np.pi * xs) + b * np.cos(2.0 * np.pi * xs)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum_box(h, target_sum)
    return h

# ------------------------------------------------------------
# Main optimisation routine (gradient descent + annealing)
# ------------------------------------------------------------
def optimise_h(n, dx, target_sum, time_budget, start_vectors):
    """
    Run the optimisation for a collection of starting vectors.
    Returns the best (h, C5) encountered.
    """
    best_h = None
    best_c5 = np.inf
    start_time = time.time()

    # Global hyper‑parameters (feel free to tweak)
    eta0 = 0.15                # initial step size
    alpha0 = 1.0               # initial soft‑max temperature
    alpha_factor = 1.5         # how fast we sharpen
    eta_factor = 0.8           # step‑size decay with each alpha increase
    alpha_max = 200.0
    iter_per_alpha = 120       # number of gradient steps before increasing α

    for h0 in start_vectors:
        h = h0.copy()
        eta = eta0
        alpha = alpha0
        it = 0

        while time.time() - start_time < time_budget * 0.90:
            h, w, f = gradient_step(h, alpha, eta, target_sum)
            # evaluate exact C5 every few steps (cheap enough)
            if it % 20 == 0:
                c5_val = exact_c5(h, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h.copy()
            it += 1
            # increase smoothness periodically
            if it % iter_per_alpha == 0:
                alpha = min(alpha * alpha_factor, alpha_max)
                eta *= eta_factor

    return best_h, best_c5

# ------------------------------------------------------------
# Simulated‑annealing polish (tiny local moves)
# ------------------------------------------------------------
def fine_polish(h, dx, target_sum, time_budget):
    """Local stochastic refinement on the best vector."""
    best_h = h.copy()
    best_c5 = exact_c5(best_h, dx)
    start = time.time()
    n = h.size
    # temperature schedule (few hundred moves)
    T0 = 0.02
    T_min = 0.0005
    decay = 0.995

    T = T0
    while time.time() - start < time_budget * 0.10:
        # propose a transfer between two coordinates
        i, j = np.random.choice(n, size=2, replace=False)
        delta = np.random.uniform(-T, T)
        h_try = best_h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        # project (the sum is exactly preserved by construction, but we still
        # need to respect the box constraints)
        h_try = np.clip(h_try, 0.0, 1.0)
        # re‑project to enforce the exact sum (tiny adjustment may be needed)
        h_try = project_to_sum_box(h_try, target_sum)

        c5_try = exact_c5(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
        else:
            # Metropolis acceptance (very low temperature, mostly greedy)
            if np.random.rand() < np.exp(-(c5_try - best_c5) / max(T, 1e-12)):
                best_h = h_try
        T = max(T * decay, T_min)
    return best_h, best_c5

# ------------------------------------------------------------
# The required entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=600, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    `h_values` is a Python list of floats in [0,1] with sum = 1 (integral).
    """
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # Grid choice (feel free to adjust)
    # --------------------------------------------------------
    n = 200                     # number of discretisation points
    dx = 2.0 / n
    target_sum = n / 2.0

    # --------------------------------------------------------
    # Build a pool of starting vectors
    # --------------------------------------------------------
    starts = []

    # 1) interpolate the supplied reference solution
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values, dtype=float)
        if h_ref.size != n:
            x_old = np.linspace(0.0, 2.0, h_ref.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h_ref = np.interp(x_new, x_old, h_ref)
        starts.append(project_to_sum_box(h_ref, target_sum))

    # 2) a few random feasible vectors
    for _ in range(4):
        starts.append(random_feasible(n, target_sum))

    # 3) a few cosine‑based guesses
    for _ in range(2):
        starts.append(cosine_initial(n, target_sum))

    # --------------------------------------------------------
    # Main optimisation (gradient + continuation)
    # --------------------------------------------------------
    # Allocate ~80 % of the budget to the heavy phase
    heavy_budget = budget_s * 0.80
    best_h, best_c5 = optimise_h(
        n=n,
        dx=dx,
        target_sum=target_sum,
        time_budget=heavy_budget,
        start_vectors=starts,
    )

    # --------------------------------------------------------
    # Fine‑grained stochastic refinement (remaining time)
    # --------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0 and best_h is not None:
        best_h, best_c5 = fine_polish(best_h, dx, target_sum, remaining)

    # --------------------------------------------------------
    # Safety net: if nothing succeeded (should never happen), fall back
    # --------------------------------------------------------
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_c5 = exact_c5(best_h, dx)

    # Return as plain Python list for the grader
    return (best_h.tolist(), float(best_c5), n)
