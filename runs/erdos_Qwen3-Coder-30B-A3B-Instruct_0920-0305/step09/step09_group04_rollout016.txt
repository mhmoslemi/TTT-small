import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import ECOS

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def solve_lp_relaxation(n_points):
    """Solve the LP relaxation of the Erdos problem"""
    dx = 2.0 / n_points
    
    # Variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    z = cp.Variable((n_points, n_points), symmetric=True)  # Product variables

    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2.0)
    
    # Bound constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # McCormick envelopes for z[i,j] ≈ h[i] * h[j]
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(z[i, j] <= h[i])
            constraints.append(z[i, j] <= h[j])
            constraints.append(z[i, j] >= h[i] + h[j] - 1)
            constraints.append(z[i, j] >= 0)
            constraints.append(z[j, i] == z[i, j])  # Symmetry
    
    # Add diagonal entries for z (they should equal h_i * h_i = h_i)
    for i in range(n_points):
        constraints.append(z[i, i] == h[i])
    
    # Overlap constraints
    for k in range(1, n_points):
        # Sum over i from 0 to n-1-k
        overlap_expr = 0
        for i in range(n_points - k):
            overlap_expr += (h[i + k] - z[i, i + k])
        constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=10000)
        if prob.status == cp.OPTIMAL:
            return h.value, t.value
        else:
            return None, None
    except:
        return None, None

def round_solution(h_frac, n_points):
    """Round the fractional solution to a binary one"""
    # Use conditional expectation rounding for balanced sampling
    h_bin = np.zeros(n_points)
    remaining_ones = n_points // 2
    
    for i in range(n_points):
        p = h_frac[i]
        if remaining_ones == (n_points - i):
            h_bin[i] = 1
        elif remaining_ones <= 0:
            h_bin[i] = 0
        else:
            # Sample with probability p
            if np.random.rand() < p:
                h_bin[i] = 1
                remaining_ones -= 1
            else:
                h_bin[i] = 0
    
    # Adjust if needed to ensure exactly n/2 ones
    ones_count = np.sum(h_bin)
    if ones_count > n_points // 2:
        # Convert excess 1s to 0s
        ones_indices = np.where(h_bin == 1)[0]
        excess = ones_count - n_points // 2
        for i in range(excess):
            chosen_idx = np.random.choice(ones_indices)
            h_bin[chosen_idx] = 0
            ones_indices = np.where(h_bin == 1)[0]
    elif ones_count < n_points // 2:
        # Convert excess 0s to 1s
        zeros_indices = np.where(h_bin == 0)[0]
        deficit = n_points // 2 - ones_count
        for i in range(deficit):
            chosen_idx = np.random.choice(zeros_indices)
            h_bin[chosen_idx] = 1
            zeros_indices = np.where(h_bin == 0)[0]
    
    # Ensure exactly correct count
    actual_ones = np.sum(h_bin)
    while actual_ones > n_points // 2:
        idx = np.random.choice(np.where(h_bin == 1)[0])
        h_bin[idx] = 0
        actual_ones -= 1
    while actual_ones < n_points // 2:
        idx = np.random.choice(np.where(h_bin == 0)[0])
        h_bin[idx] = 1
        actual_ones += 1

    return h_bin

    bad_shift = np.argmax(np.correlate(h, 1-h, mode="full"))
    
    iter_count = 0
    while iter_count < max_iter:
        improved = False
        # Look for improving swaps
        for i in range(n_points):
            for j in range(n_points):
                if h[i] == 1 and h[j] == 0:
                    # Temporarily swap
                    h[i] = 0
                    h[j] = 1
                    
                    # Compute new C5
                    new_c5 = evaluate_c5(h)
                    
                    if new_c5 < best_c5:
                        # Accept improvement
                        best_h = h.copy()
                        best_c5 = new_c5
                        improved = True
                        bad_shift = np.argmax(np.correlate(h, 1-h, mode="full"))

                        # Revert swap
                        h[i] = 1
                        h[j] = 0
        
        if not improved:
            # No improvement found, break early
            break
            
        iter_count += 1
    
    return best_h, best_c5
