def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def generate_random_h(seed):

    h = np.random.uniform(0, 1, n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def gradient_of_c_s(h, s):
    """Compute gradient of c_s for shift s"""
    # For shift s >= 0
    n = len(h)
    grad = np.zeros(n)
    
    if s == 0:
        return np.zeros(n)
        
    # For i < s: (1 - h_{i-s}) contributes -1 to h_i
    # For i >= s: h_{i-s} contributes -1 to h_i
    # Also h_i contributes (1 - h_{i-s}) to h_{i+s} when i+s < n
    if s > 0:
        # Contribution of h_{i+s} to h_i when i < n - s
        for i in range(n - s):
            grad[i] -= h[i + s]
        # Contribution of h_{i-s} to h_i when i >= s
        for i in range(s, n):
            grad[i] += (1 - h[i - s])
    else:
        s = -s
        # Negative shifts are handled via symmetry
        # Contribution of h_{i} to h_{i+s}  when i < n - s
        for i in range(n - s):
            grad[i + s] -= h[i]
        # Contribution of h_{i-s} to h_i when i >= s
        for i in range(s, n):
            grad[i] += (1 - h[i - s])
    
    return grad

def compute_gradient(h, c_vals, epsilon):
    """Compute gradient of smooth max with log-sum-exp"""
    # Compute soft weights
    c_vals = np.array(c_vals)
    c_max = np.max(c_vals)
    c_exp = np.exp(epsilon * (c_vals - c_max))
    w = c_exp / np.sum(c_exp)
    
    
    grad = np.zeros(len(h))
    # For each shift k, compute weight * gradient of c_k
    for k, weight in enumerate(w):
        if k == 0:
            # Special case for k = 0 shift (no shift)
            grad += weight * np.zeros_like(h)
        else:
            # We'll compute the gradient for this specific shift directly using helper function
            # But this requires recalculation of contributions properly
            
            # This is simplified version: in practice for communication optimizing, we'd
            # pre-compute contributions. But here we'll approximate with direct method:
            # Take derivative of c_k(h) w.r.t components
            s = k - len(c_vals)//2  # Map index back to shift values
            grad += weight * gradient_of_c_s(h, s)
            
    return grad
