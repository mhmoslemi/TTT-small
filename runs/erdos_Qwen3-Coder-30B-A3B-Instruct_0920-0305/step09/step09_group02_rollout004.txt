import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using scipy.signal.correlate"""
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h_rev, b, mode="full")
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        if k < n:
            # Left shift
            shifted_left = np.zeros(n)
            shifted_left[k:] = h[:n-k]
            grad += wgt[k] * (1 - shifted_left)
        else:
            # Right shift
            shifted_right = np.zeros(n)
            shifted_right[:n-(k-n+1)] = h[(k-n+1):]

    return grad

    exp_values = np.exp(exp_args)
    return (1.0 / beta) * (np.log(np.sum(exp_values)) + max_corr)
