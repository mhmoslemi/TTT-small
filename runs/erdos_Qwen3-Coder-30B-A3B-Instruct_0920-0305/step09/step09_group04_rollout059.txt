def compute_c5(h_vals):
    """Compute C5 value from h values efficiently"""
    h = np.array(h_vals)
    n = len(h)
    dx = 2.0 / n
    # Compute correlations using FFT for efficiency
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr[:n]  # Take only the relevant part
    return dx * np.max(corr)
