def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def compute_shifted_overlap(h, k):
    """Compute overlap for shift k"""
    n = len(h)
    if k >= 0:
        return np.sum(h[:n-k] * (1 - h[k:]))
    else:
        return np.sum(h[-k:] * (1 - h[:-k]))
