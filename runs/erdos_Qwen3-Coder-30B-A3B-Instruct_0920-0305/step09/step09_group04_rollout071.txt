import numpy as np
from scipy.signal import correlate

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max_corr(h, epsilon):
    """Compute smooth max approximation and its gradient efficiently"""
    # Compute full correlation (same as np.correlate with 'full')
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Shift correlation indices to map from -n+1 to n-1
    c_vec = corr
    # At this point c_vec has length 2n-1 where indices correspond to shifts 
    # since we compute correlation between h and 1-h with full padding
    
    # Compute smoothed max log-sum-exp
    shifted_corr = epsilon * c_vec
    # Avoid overflow
    shifted_corr = shifted_corr - np.max(shifted_corr)
    exp_corr = np.exp(shifted_corr)
    sum_exp = np.sum(exp_corr)
    w = exp_corr / sum_exp
    
    # Now compute gradient with respect to h
    grad = np.zeros_like(h)
    
    # For each shift index (they are actually negative to positive)
    n = len(h)
    shift_indices = np.arange(-n+1, n)  # this corresponds to actual shifts [0, 2n-2] -> shift values [-n+1, n-1]

    # We compute gradient for each shift
    for i in range(len(shift_indices)):
        s = shift_indices[i]
        w_i = w[i]
        
        if s >= 0:
            # For shift s (positive): c_s = sum_{i=0}^{n-1-s} h_{i+s} * (1 - h_i)
            # Gradient: 
            # For jth component of h, contributions from derivatives:
            # d/dh_j(c_s) = -(1 - h_{j-s}) if j-s >= 0 else 0  - h_{j+s} if j+s < n else 0
            # We have contribution from terms h_{i+s} * (1 - h_i)
            # This means:
            # When i+j = s it contributes: -h_{i+j} = -h_j  (if j+s < n) 
            # When i+j = s also we have h_{i+s} = h_j, so contribution from 1-h_i is +1
            
            # Actual partial: if j consider i where i + s = j or i = j - s, 
            # contributes -h_{j+s} if j+s < n, or 0 if j+s >= n
            
            # We compute individual shift contribution and use convolution-like approach
            
            # Gradient contribution to grad[j]: 
            # for all i such that i <= n-1-s, (since index limits)
            # Some stabilization needed for boundary values

            # Carefully build h_{j+s} term in i==j-s, i + s == j => i = j - s
            if j-s >= 0 and j-s < n:
                # Negative contribution from h{i+s} when we have derivative of h_{i+s} * (1 - h_i)
                grad[j] -= w_i * h[j+s] if j+s < n else 0
            if j >= s and j < n:
                # Contributes (1 - h_{j-s}) from derivative w.r.t. h_{j-s} 
                # only when j-s >= 0, so j >= s, and the term h_{j-s} * (1 - h_j)  
                # derivative w.r.t h_j is -h_{j-s} (same as above)
                # however we compute gradient d/dh_j of c_s = h_{j+s} * (1 - h_j)
                # = -h_{j+s} for term h_{j+s} *(1 - h_j) 
                # This typically requires to avoid this and handle form correctly
                # Also for derivative w.r.t h_{j-s} it's (1 - h_j) if j-s < n
                # so sum for contributions over i where we have terms with h_{j+s} and h_{j-s}
                # Let's take a cleaner mathematical definition:
                
                # The gradient of c_s w.r.t. h_j is:
                # for term h_{i+s}*(1 - h_i), if j = i (so h_j = h_i) we get -(1 - h_j)
                # if j = i+s (so h_j = h_{i+s}) we get (1 - h_i)  
                
                # But i = j-s, so we want i = j-s to be in valid range:
                # i >= 0 => j >= s; i <= n-1 => j <= n-1+s
                
                # So when j >= s, derivative from term h_i * (1 - h_{i+s}) where i+s = j:
                # So i = j-s, that is valid if 0 <= j-s <= n-1 => s <= j <= s+n-1. 
                # so range of j for which derivative comes from j-s = 0 is s to n-1
                # one value at i+j-s = j, where i equals j-s
                if j-s >= 0 and j-s < n:
                    grad[j] += w_i * (1 - h[j-s])
                    
        else:  # s < 0
            # shift -s is positive
            # so backward shift: c_{-s} = sum_{i=0}^{n-1+s} h_i * (1 - h_{i+s})  
            # Gradient of term h_{i} * (1 - h_{i+s}) w.r.t h_j is:
            # if j = i: -h_{i+s} = -h_{j+s}  
            # if j = i+s: (1 - h_i) = (1 - h_{j-s}) 
            # Also we compute for sum from i = 0 to n-1+s 
            # so i must be in [0, n-1+s].  
            
            # For index value j:
            # - from h_i * (1 - h_{i+s}) term where i+j = s => i = j-s.
            # which gives contributors from:
            #    1) Term h_i (differentiated wrt h_j): 
            #       if i=j, then h_i = h_j, derivation = -h_{i+s} = -h_{j+s}
            #       needed: i=j, so j = i, but i must be in [0, n-1+s]  
            #    2) Term (1- h_{i+s}) (differentiated wrt h_j):
            #       if i+s = j, so i = j - s, derivative = (1 - h_i) = (1 - h_{j-s})
            #       needed: i = j-s, which implies j-s >= 0 => j >= s; and i = j - s <= 0 + (n-1+s) = n-1+s => j <= n-1+2s == n-1-s+2s = n-1+s (right, cancels)  
                
            # Want to compute like reflection: c_{-s} = sum_{i=0}^{n-1+s} h_i * (1 - h_{i+s})
            # derivative wrt h_j:
            # the terms contribute to i such that either i = j or i+s = j (i = j - s)
            # That is we need to check index values j-s, j (and the sum range)  
            # If the norm of shift s >= 0, same computation but reversed ordering
            
            # Say shift s, meaning reverse shift is -s, so identity for i in range [0, n-1+(-s)] = [0, n-1-s]
            # Correlation points used: h_i * (1 - h_{i+s})
            # To compute derivative wrt h_j we consider these:
            # case 1) i = j: contribution = -h_{j+s}
            # case 2) i+s = j (i = j-s): contribution = (1 - h_{j-s})
            
            # Domains: 0 <= i <= n-1+s and i = j or i = j-s        
            # For case 1 (i = j)
            # Get term h_{j} * (1 - h_{j+s}) => derivative wrt h_j = -(1 - h_{j+s}) 
            # valid if 0 <= j <= n-1+s (but also j in [0,n-1] bounds) so j in [0, min(n-1, n-1+s)]
            # which is j in [0, n-1]
            # Also need i = j = j >= 0, i = j <= n-1+s, so j <= n-1+S = n-1-S, so j <= n-1-s
            # but we have a problem we need i >= 0, i = j, so j >= 0, and j <= n-1+s to overlap
            
            # Let's build index correctly:
            # Overlap interval of [0,n-1] and [0,n-1+s]: [0, min(n-1,n-1+s)] = [0, n-1] 
            # derivative contribution from term where i = j = i: -h_{j+s}  
            if j+s < n:  # i = j can be in [0] to [n-1] but we also have index j+s, constraint on j+s < n  
                grad[j] -= w_i * h[j+s]
                
            # Contribution from  (1 - h_{i+s}) where i = j - s
            # So we set i = j - s, require 0 <= j-s <= n-1+s  
            # i = j-s: valid if i in [0, n-1+s] => 0 <= j-s, j-s <= n-1+s => j >= s, j <= n-1+2s
            # also j in [0, n-1] from original domain  
            # overall: j >= max(0, s) && j <= min(n-1, n-1+2s) = min(n-1, n-1+2s)
            # But we also have j-s >= 0 => j >= s, so over all j in [max(0,s), min(n-1, n-1+2s)]
            if j-s >= 0 and j-s < n:  # i = j-s is valid range [0, n-1+s], 
                # i = j-s validity: 0 <= j-s <= n-1+s (all these constraints) -> j-s <= n-1+s (tautology?), j-s >= 0 (so j>=s),            
                # and 0 <= j-s which is equivalent to s <= j
                #   j <= n-1+s -> j-s <= n-1 hence j <= n-1+s
                # so j in [s, n-1] 
                # wait we assumed valid range for n-1+s, but j must also be in [0, n-1]?  
                # so j in [max(0,s), min(n-1, n-1+s)] = [max(0,s), n-1], with max(0,s) straightforward  
                # well such that we have explicit i range for computation we need: j-s in [0, n-1+s]  
                # i = j-s (base requirement of what we want), so j >= s (0-right) and j-s <= n-1+s (reverse is unnecessary)
                # Or j-s >= 0 and j-s <= n-1+s -> j >= s and j <= n-1+2s  
                # Actually better recursion:
                # j in valid range where: j-s >= 0 (i=j-s >= 0) 
                # and one more constraint that s = j-i so i = j-s, and i in [0,n-1+s]
                # so 0 <= j-s <= n-1+s -> s <= j <= n-1+2s, constraint that n-1+s >= 0 -> no issue as n>=1, s> 0.
                # let’s reconstruct:
                # derivative from h_i * (1 - h_{i+s}) w.r.t h_j:  
                # term: h_i * (1-h_{i+s})
                # i = 0 to n-1+s  
                # We want computations such that d/dh_j = coefficient of h_{i+s} in term h_i * (1-h_{i+s})
                # from i = j-s: coefficient from (1-h_{i+s}) *(1) which gives  (1-h_{i+s}) where i = j-s -> (1-h_{j-s})
                # i = j-s must be in range 0 to n-1+s (for indices in correlation value computation)
                # so 0 <= j-s <= n-1+s -> s <= j <= n-1+2s  and from other constraint j in [0, n-1]
                # So valid j: s <= j <= n-1, but also 0 <= j <= n-1 (all).

                grad[j] += w_i * (1 - h[j-s])

    return np.max(c_vec) if len(c_vec) > 0 else 0.0, grad
