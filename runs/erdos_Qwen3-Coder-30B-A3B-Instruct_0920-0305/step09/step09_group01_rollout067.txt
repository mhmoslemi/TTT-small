def run(seed=42, budget_s=600, **kwargs):
    # Set up problem parameters
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a moderate grid size for better results with reasonable time
    n = 150
    dx = 2.0 / n
    TARGET_SUM = n / 2
    
    # Create model variables
    h = Variable(n, boolean=True)
    
    # Construct all the auxiliary variables for pairwise products
    # We'll have y[i,k] where k goes from 0 to n-1 and i from 0 to n-k-1
    y = {}
    for k in range(n):
        for i in range(n - k):
            y[(i, k)] = Variable(boolean=True)
    
    # Constraints
    constraints = [
        sum(h) == TARGET_SUM  # Sum constraint
    ]
    
    # Linearisation constraints for all y[i,k]
    for k in range(n):
        for i in range(n - k):
            # y[i,k] <= h[i]
            constraints.append(y[(i, k)] <= h[i])
            # y[i,k] <= h[i+k]  
            constraints.append(y[(i, k)] <= h[i + k])
            # y[i,k] >= h[i] + h[i+k] - 1
            constraints.append(y[(i, k)] >= h[i] + h[i + k] - 1)
    
    # Objective (minimize the maximum correlation)
    # c_k = sum of h_i - sum of y[i,k] for shift k
    c = []
    t = Variable()
    
    # Add epigraph constraints: t >= c_k for all k
    for k in range(n):
        # For k shift, we sum over i = 0 to n-k-1
        sum_h = sum(h[i] for i in range(n - k))
        sum_y = sum(y[(i, k)] for i in range(n - k))
        c_k = sum_h - sum_y
        constraints.append(t >= c_k)
    
    # Objective
    objective = Minimize(t)
    
    # Create problem and solve
    prob = Problem(objective, constraints)
    
    # Solve with time limit and small MIP gap 
    prob.solve(solver=ECOS_BB, verbose=False, max_iters=1000)
    
    # Extract solution if available
    h_sol = np.array([float(h[i].value) for i in range(n)])
    
    best_h = h_sol.copy()
    # Compute correlation value
    corr_full = correlate(best_h, 1 - best_h, mode="full") * dx
    best_c5 = np.max(corr_full)
    
    # Dropping to the exact problem (apply the solver to the full domain)
    # Even if we did not get an exact optimal solution, we use the binary solution found:
    # Try rounding the LP relaxation if needed or refine with local search
    
    # Attempt LP rounding  
    if hasattr(prob, 'status') and prob.status == 'optimal':
        # We have a binary solution (should be the case with CBC)
        pass
    else:
        # Fall back to LP relaxation rounding
        # Free the boolean constraint for gathering LP relaxation solution
        for var in h:
            var.boolean = False
        prob.solve(solver=ECOS, verbose=False)
        h_frac = np.array([float(h[i].value) for i in range(n)])
        h_sol = np.zeros(n)
        # Greedy rounding
        sorted_indices = np.argsort(h_frac)[::-1]
        for i in range(int(TARGET_SUM)):
            h_sol[sorted_indices[i]] = 1
        
        best_h = h_sol.copy()
    
    # Now perform pair-swap local search to refine the solution
    # Convert to binary if needed
    h_best = np.round(best_h).astype(int)
    temp_c5 = np.max(correlate(h_best, 1 - h_best, mode="full") * dx)
    iters = 0
    while time.time() - start_time < budget_s - 1:
        improved = False
        for i in range(n):
            if h_best[i] == 1:
                for j in range(n):
                    if h_best[j] == 0:
                        # Compute change in correlations if we switch
                        # Only consider changes for affected shifts
                        delta = np.zeros(n)
                        
                        # Old value: h[i] = 1, h[j] = 0 
                        # New value: h[i] = 0, h[j] = 1
                        # Change affects k such that i+k or j+k are in valid index ranges
                        for k in range(n):  # k from 0 to n-1
                            # For shift k: i with i+k, j with j+k
                            # Old: 1*(1-h[i+k]) + 0*(1-h[j+k]) = 1-h[i+k]  
                            # New: 0*(1-h[i+k]) + 1*(1-h[j+k]) = 1-h[j+k]  
                            if i + k < n and i + k >= 0:
                                delta[k] += (1 - h_best[i + k]) - (1 - h_best[j])  # corr contributes to k only if i+k, j+k are valid
                            if j + k < n and j + k >= 0:
                                delta[k] += (1 - h_best[j + k]) - (1 - h_best[i]) 
                        
                        # Edge cases: we do not consider operator overloads
                        # But let's recalculate the correlations properly for these k values only
                        old_corr = np.maximum.reduce([np.correlate(h_best, 1 - h_best, mode="full") * dx,
                                                     np.full(n, -float('inf'))]) 
                        new_h = h_best.copy()
                        new_h[i] = 0
                        new_h[j] = 1
                        # Recompute full correlations for the new h vector
                        new_corr = correlate(new_h, 1 - new_h, mode="full") * dx
                        new_max_corr = np.max(new_corr)
                        old_max_corr = np.max(old_corr)  # We use full dcorr as in last example
                        
                        if new_max_corr < old_max_corr:
                            h_best[i] = 0
                            h_best[j] = 1
                            improved = True
                            break
        if not improved:
            break
        iters += 1
        if iters > 1000 or time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation  
    final_corr = correlate(h_best, 1 - h_best, mode="full") * dx
    final_c5 = np.max(final_corr)
    
    return (h_best.tolist(), float(final_c5), n)
