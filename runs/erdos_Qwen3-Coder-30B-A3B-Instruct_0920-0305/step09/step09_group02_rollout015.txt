import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using scipy.signal.correlate"""
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h_rev, b, mode="full")
    return corr

def exact_C5(h):
    """Evaluate exact C5 using full correlation"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_onto_box_and_sum(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, wgt):
    """Compute weighted gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute the full correlation vector
    corr = compute_corr(h)
    
    # For each shift k, we need to compute the gradient contribution
    # This is the weighted sum of gradients across all shifts
    # Gradient for shift k is:
    # ∂f_k/∂h[p] = 1 - h[p+k] - h[p-k] (with zero padding)
    
    # Efficient implementation using NumPy slicing
    for k in range(-n+1, n):
        # Build shifted vectors for this k
        if k >= 0:
            left = np.concatenate([np.zeros(k), h[:-k]]) if k > 0 else h.copy()
            right = np.concatenate([h[k:], np.zeros(k)]) if k > 0 else h.copy()
        else:
            left = np.concatenate([np.zeros(-k), h[-k:]]) if -k > 0 else h.copy()
            right = np.concatenate([h[:-k], np.zeros(-k)]) if -k > 0 else h.copy()
        
        # The gradient contribution for shift k
        grad_contrib = (1 - left - right)
        grad += wgt[k + n - 1] * grad_contrib
    
    return grad
