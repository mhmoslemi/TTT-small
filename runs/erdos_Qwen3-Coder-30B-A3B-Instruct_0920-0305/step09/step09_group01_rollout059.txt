import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Precompute shift matrices
    def build_shift_matrix(ell, n):
        M = np.zeros((n, n))
        for i in range(n - ell):
            M[i, i + ell] = 1
            M[i + ell, i] = 1
        return M

    shift_matrices = [build_shift_matrix(ell, n_points) for ell in range(n_points)]
    
    # SDP formulation
    def solve_sdp_for_alpha(alpha):
        # Variables
        h = Variable(n_points)
        X = Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            X == h @ h.T,
            0 <= h, h <= 1,
            sum(h) == TARGET_SUM,
            X >> 0
        ]
        
        # Add McCormick envelopes
        for i in range(n_points):
            for j in range(n_points):
                constraints += [X[i, j] <= h[i], X[i, j] <= h[j]]
                constraints += [X[i, j] >= 0, X[i, j] >= h[i] + h[j] - 1]
        
        # Add quadratic constraints
        for ell in range(n_points):
            constraints += [trace(shift_matrices[ell] @ X) >= TARGET_SUM - alpha]
        
        # Solve
        problem = Problem(Minimize(0), constraints)
        
        try:
            problem.solve(solver=SCS, verbose=False, max_iters=5000)
            return h.value, X.value
        except:
            return None, None

    # Rounding and refinement
    def round_and_refine(h_val, X_val):
        if h_val is None:
            return None, np.inf
            
        # Randomized rounding
        # Sample v ~ N(0, X_val) and use sign
        # We'll use a simpler local version:
        h_rounded = np.clip(h_val, 0, 1)
        h_proj = project_to_sum_box(h_rounded, TARGET_SUM)
        return h_proj, evaluate_c5(h_proj)
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    # Binary search on alpha
    left, right = 0.0, 1.0
    best_h = None
    best_c5 = np.inf
    
    while left < right and (time.time() - start_time) < budget_s - 1:
        alpha = (left + right) / 2.0
        h_sol, X_sol = solve_sdp_for_alpha(alpha)
        h_rounded, c5 = round_and_refine(h_sol, X_sol)
        
        if c5 < best_c5:
            best_h = h_rounded
            best_c5 = c5
            if c5 < 0.379:
                break
        
        if h_rounded is not None:
            right = alpha
        else:
            left = alpha

    # Final polishing
    if best_h is not None:
        h_final = best_h.copy()
        def obj(h):
            return evaluate_c5(h)
        for _ in range(200):
            if (time.time() - start_time) > budget_s - 1:
                break
            # Small perturbation
            h_new = h_final.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            h_new[idx] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h_final = h_new
                best_c5 = c5_new
    
    return (best_h.tolist(), float(best_c5), n_points)
