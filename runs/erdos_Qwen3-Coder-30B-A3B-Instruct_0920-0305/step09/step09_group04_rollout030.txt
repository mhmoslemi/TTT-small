def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):

    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            if shift == 0:
                # Special case: g_0 = sum h_i * (1-h_i)
                grad += weights[k] * (1 - 2*h)
            else:
                # Forward shift case
                # p = h_{i+shift} (backward shift)
                # pull = np.zeros(n)
                # pull[shift:] = h[:-shift]
                # q = (1 - h_i) (forward shift)
                # push = np.zeros(n)
                # push[:n-shift] = (1 - h)[shift:]
                
                pull = np.concatenate([np.zeros(shift), h[:-shift]]) if shift < n else np.zeros(n)
                push = np.concatenate([(1 - h)[shift:], np.zeros(shift)]) if shift < n else np.zeros(n)
                grad += weights[k] * (push - pull)
        else: # shift < 0
            # reverse case
            abs_shift = -shift
            if abs_shift == 0:
                continue
            else:
                # backward shift
                pull = np.concatenate([np.zeros(abs_shift), h[:-abs_shift]]) if abs_shift < n else np.zeros(n)
                push = np.concatenate([(1 - h)[abs_shift:], np.zeros(abs_shift)]) if abs_shift < n else np.zeros(n)
                grad += weights[k] * (push - pull)
    
    return log_sum_exp, grad
