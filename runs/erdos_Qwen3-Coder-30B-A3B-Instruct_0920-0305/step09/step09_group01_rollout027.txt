import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    # This is a placeholder, actual implementation should match the provided function
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

def initial_h_values():
    # This is a placeholder, should return the initial construction
    n = 120
    h = np.zeros(n)
    h[:n//2] = 1.0
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 150  # grid size
    dx = 2.0 / n
    target_sum = n / 2
    
    # Model building
    # Variables
    h = Binary(n, name='h')
    y = Variable((n, n), boolean=True, name='y')  # y[i,k] = h[i] AND h[i+k]
    
    # Constraints
    constraints = [sum(h) == target_sum]
    
    # Linearization constraints for y[i,k] = h[i] AND h[i+k]
    for i in range(n):
        for k in range(n - i):
            constraints += [
                y[i, k] <= h[i],
                y[i, k] <= h[i + k],
                y[i, k] >= h[i] + h[i + k] - 1
            ]
    
    # Define the objective
    # For each shift k, compute the correlation value:
    # c_k = sum(h_i * (1 - h_{i+k})) for i from 0 to n-k-1
    # This equals sum(h_i) - sum(y_{i,k}) for i from 0 to n-k-1
    # To minimize max_k c_k, we introduce t where t >= c_k for all k
    
    # Create epigraph variable t
    t = Variable(name='t')
    
    # Define correlation values explicitly for each shift k
    c = []
    for k in range(n):
        if k == 0:
            # For k = 0, c_0 = sum(h_i) = target_sum, so c_0 = (n/2) = target_sum
            # But this should be computed for the actual shifts
            c.append(target_sum)
        else:
            # c_k = sum(h_i) - sum(y_i,k) for i from 0 to n-k-1
            # Sum over first n-k elements for h
            sum_h_k = sum(h[:n-k])
            # ... and subtract sum of y[:n-k, k]
            sum_y_k = sum(y[i, k] for i in range(n-k))
            c.append(sum_h_k - sum_y_k)
    
    # If all variables above were correctly interpreted, the main improvements are:
    # Introduce t such that t >= c_k for all k.
    # But since we want to minimize max_k c_k, we want to minimize t subject to:
    constraints += [t >= c_k for c_k in c]
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Build the problem
    prob = Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=CPLEX, time_limit=budget_s / 2)
    except:
        prob.solve(solver=GLPK, time_limit=budget_s / 2, msg=1)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        # Extract solution
        h_sol = h.value
        # Since we are using binary variables with integer coefficients in constraints,
        # the solution h_sol will be close to integers, we take the rounded version
        h_sol = np.array([1 if x > 0.5 else 0 for x in h_sol])
    else:
        # Fallback: solve the LP relaxation first
        h_relax = Variable(n, boolean=False)
        constraints_relax = [sum(h_relax) == target_sum,
                             h_relax >= 0,
                             h_relax <= 1]
        y_relax = Variable((n, n), boolean=False)
        
        # Constraint relaxation (not the real AND in LP, so just approximate)
        for i in range(n):
            for k in range(n - i):
                if i + k < n:
                    constraints_relax += [
                        y_relax[i, k] <= h_relax[i],
                        y_relax[i, k] <= h_relax[i + k],
                        y_relax[i, k] >= h_relax[i] + h_relax[i + k] - 1
                    ]
        
        objective_relax = Minimize(t)
        prob_relax = Problem(objective_relax, constraints_relax)
        
        try:
            prob_relax.solve(solver=GLPK, time_limit=budget_s / 5)
        except:
            # Fallback to random solution if nothing works
            h_sol = np.zeros(n)
            indices = np.random.choice(n, size=int(target_sum), replace=False)
            h_sol[indices] = 1
        else:
            # Approximate rounding heuristics: sort indices by decreasing density
            h_relax_sol = h_relax.value
            indices_sorted = np.argsort(h_relax_sol)[::-1]
            h_sol = np.zeros(n)
            h_sol[indices_sorted[:int(target_sum)]] = 1
    
    # Final polishing with local search (pair swap)
    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Apply polishing only if time remains
    current_c5 = compute_correlation(h_sol)
    current_h = h_sol.copy()
    best_h = h_sol.copy()
    best_c5 = current_c5
    
    # Pair swap loop with early termination
    swapped = True
    while swapped and (time.time() - start_time < budget_s - 2):
        swapped = False
        # Loop over each pair of points to swap
        for i in np.where(current_h == 1)[0]:
            for j in np.where(current_h == 0)[0]:
                # Temporarily swap
                current_h[i], current_h[j] = 0, 1
                new_c5 = compute_correlation(current_h)
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
                    swapped = True
                # Swap back
                current_h[i], current_h[j] = 1, 0
    
    return (best_h.tolist(), float(best_c5), n)
