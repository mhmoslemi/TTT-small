def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_shifts(h):
    """Compute all shift correlations at once using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_gradient(h, shifts, mu):
    """Compute gradient using soft-max weights"""
    n = len(h)
    # Soft max weights
    max_shift = np.max(shifts)
    exp_shifts = np.exp((shifts - max_shift) / mu)
    weights = exp_shifts / np.sum(exp_shifts)
    
    # Initialize gradient
    grad = np.zeros(n)
    
    # Apply gradient for each shift
    for k in range(-n+1, n):
        if k >= 0:
            grad[:n-k] += weights[k] * (1 - h[k:])
            grad[k:] -= weights[k] * h[:n-k]
        else:
            k_abs = -k
            grad[k_abs:] += weights[n+k] * (1 - h[:n-k_abs])
            grad[:n-k_abs] -= weights[n+k] * h[k_abs:]
    
    return grad

def line_search(h, grad, mu, alpha0=0.2*dx):
    """Perform line search for step size"""
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    alpha = alpha0
    for _ in range(5):  # Max 5 halvings
        h_new = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5_current:
            return h_new, c5_new, alpha
        alpha /= 2
    
    return h_current, c5_current, alpha
