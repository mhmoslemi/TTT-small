import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with a better initial guess - a simple piecewise constant function
    h0 = np.zeros(n_points)
    # Create a function that's 1 on the first half and 0 on the second half, then adjust
    half_idx = n_points // 2
    h0[:half_idx] = 1.0
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Try a few different constructions
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Try slightly different initial configurations
    initial_configs = []
    
    # Config 1: Smooth step function
    h1 = np.ones(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            h1[i] = 1.0
        elif i < 2 * n_points // 3:
            h1[i] = 0.5
        else:
            h1[i] = 0.0
    h1 = project_to_sum_box(h1, TARGET_SUM)  
    initial_configs.append(h1)
    
    # Config 2: Wave-like function
    x_vals = np.linspace(0, 2, n_points)
    h2 = 0.5 + 0.5 * np.sin(np.pi * x_vals)
    h2 = project_to_sum_box(h2, TARGET_SUM)
    initial_configs.append(h2)
    
    # Config 3: Uniform with variance
    h3 = np.random.uniform(0.3, 0.7, n_points)
    h3 = project_to_sum_box(h3, TARGET_SUM)
    initial_configs.append(h3)
    
    # Evaluate all initial configurations
    for h in initial_configs:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Use stochastic optimization with adaptive step sizes and restarts
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Parameters for adaptive cooling schedule
    temp_start = 0.1
    temp_end = 0.0001
    max_steps = int(2e5)
    
    step = 0
    temp = temp_start
    
    # Decreasing cooling schedule (exponential)
    cooling_rate = (temp_end / temp_start) ** (1.0 / max_steps)
    
    # Restart counter
    restart_count = 0
    max_restarts = 10
    last_improvement = 0
    
    while step < max_steps and time.time() - start_time < budget_s:
        # Generate neighbor solution with adaptive step size
        neighbor_h = current_h.copy()
        
        # Use different step sizes based on iteration
        if step < max_steps // 4:
            step_size = 0.05
        elif step < max_steps // 2:
            step_size = 0.02
        else:
            step_size = 0.01
            
        # Make random changes to multiple elements
        num_changes = max(1, min(25, n_points // 10))
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        for idx in indices:
            # Random small perturbation 
            delta = np.random.normal(0, step_size)
            neighbor_h[idx] += delta
            
        # Project back to constraints 
        neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
        neighbor_c5 = evaluate_c5(neighbor_h)
        
        # Accept or reject the move
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            if neighbor_c5 < best_c5:
                best_h = neighbor_h.copy()
                best_c5 = neighbor_c5
                last_improvement = step
        else:
            # Accept with probability based on temperature
            delta_e = neighbor_c5 - current_c5
            if np.random.random() < np.exp(-delta_e / temp):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
                
        # Cool down
        temp *= cooling_rate
        step += 1
        
        # Periodically check for stagnation and restart
        if step - last_improvement > max_steps // 50 and restart_count < max_restarts:
            current_h = best_h.copy()
            current_c5 = best_c5
            temp = temp_start
            restart_count += 1
            last_improvement = step
    
    # Final local refinement using sequential least squares programming
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        res = minimize(
            evaluate_c5, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-13, 'gtol': 1e-13}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Try one more round of fine-tuning with gradient-free method
    if time.time() - start_time < budget_s:
        # Try Nelder-Mead optimization with good starting point
        try:
            # Linear constraints: sum(h) = TARGET_SUM
            def obj(h):
                return evaluate_c5(h)
                
            def cons(h): 
                return np.sum(h) - TARGET_SUM
                
            # Initial point 
            h_start = best_h.copy()
            
            # Nelder-Mead optimization
            res = minimize(
                obj, 
                h_start, 
                method='Nelder-Mead',
                options={'maxiter': 500, 'xatol': 1e-12, 'fatol': 1e-12}
            )
            
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, TARGET_SUM)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_h = h_final 
                    best_c5 = c5_final
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
