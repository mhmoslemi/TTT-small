def compute_corr_fft(h):
    """Compute correlation using FFT efficiently"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5_exact(h):
    """Evaluate the exact C5 value"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using bisection"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, c_values, weights):
    """Compute gradient based on current c values and weights"""
    grad = np.zeros_like(h)
    # Efficiently compute gradients via shifted vectors
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i)
            # Gradient: d/dh_j c_k = (1 - h_{j-k}) if j >= k, -h_{j+k} if j <= n-1-k
            # This computes efficient gradient via outer products
            if k == 0:
                # Only the direct terms matter
                # Gradient w.r.t. h_i is (1 - h_{i}) - h_{i} = 1 - 2*h_i
                grad += weights[k + n - 1] * (1 - 2 * h)
            else:
                # Gradient contribution from h_{i+k} * (1 - h_i)
                if k < n:
                    # Shift h and 1-h for positive shifts
                    shifted_h = np.concatenate([h[k:], np.zeros(k)])
                    shifted_comp = np.concatenate([1 - h[:-k], np.zeros(k)]) if k < n else 0
                    grad += weights[k + n - 1] * shifted_comp  # i+k = i shifts 
                    # Add contribution from h_{i+k} term
                    right_term = np.concatenate([h, np.zeros(k)])[k:]  # physically right-shifted h
                    right_term = np.concatenate([right_term, np.zeros(k)]).ravel()
                    grad -= weights[k + n - 1] * right_term[:n]  # h_{i+k} loss from i
                if k >= 0:
                    shifted_h = np.concatenate([h[k:], np.zeros(k)])
                    grad += weights[k + n - 1] * shifted_h
        else:
            k_neg = -k 
            if k_neg < n:
                # Negative shift: h_i * (1 - h_{i+k})
                shifted_h = np.concatenate([np.zeros(k_neg), h[:-k_neg]])
                grad -= weights[k + n - 1] * shifted_h
                shifted_comp = np.concatenate([1 - h[k_neg:], np.zeros(k_neg)])
                grad += weights[k + n - 1] * shifted_comp
    return grad

def smooth_max_around_c(h, epsilon):
    """Return the smooth max value and soft max weights"""
    # Compute overlaps
    c_vals = []
    for k in range(-n_points + 1, n_points):
        # Compute c_k loss for all k segments
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h_{i+k} * (1 - h_i)
            part = h[k:] * (1 - h[:-k])
            c_vals.append(np.sum(part))
        else:
            # k is negative: c_k = sum_{i=0}^{n-1+k} h_{i} * (1 - h_{i-k})
            k_abs = -k
            part = h[:-k_abs] * (1 - h[k_abs:])
            c_vals.append(np.sum(part))
    
    c_vals = np.array(c_vals)
    c_max = np.max(c_vals)
    shift_vals = c_vals - c_max  # for numerical stability
    expvals = np.exp(epsilon * shift_vals)
    Z = np.sum(expvals)
    weights = expvals / Z
    f_epsilon = c_max + (1/epsilon) * np.log(Z)
    return f_epsilon, weights, c_vals
