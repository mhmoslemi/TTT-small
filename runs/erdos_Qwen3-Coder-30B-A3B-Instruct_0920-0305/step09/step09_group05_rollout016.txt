import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_algorithm(budget_s, seed=42):
    """Genetic Algorithm for initial optimization"""
    np.random.seed(seed)
    start_time = time.time()
    
    population_size = 60
    generations = 200
    mutation_rate = 0.05
    
    # Initialize population
    def create_individual():
        # Create binary vector with exactly half ones
        individual = np.zeros(n_points)
        ones_indices = np.random.permutation(n_points)[:n_points//2]
        individual[ones_indices] = 1
        return individual
    
    population = [create_individual() for _ in range(population_size)]
    
    def fitness(individual):
        # Return negative of C5 (since we want to minimize C5)
        return -evaluate_c5(individual)
    
    # Evaluation
    fitness_values = [fitness(ind) for ind in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.6:
            break
            
        # Selection: tournament selection
        new_population = []
        for _ in range(population_size):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_values[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
            
        # Crossover: uniform crossover with balanced repair
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Repair to ensure correct sum
                def repair(ind):
                    # Ensure exactly half ones
                    current_sum = np.sum(ind)
                    if current_sum > n_points/2:
                        # Remove extra ones
                        ones_idx = np.where(ind == 1)[0]
                        remove_indices = np.random.choice(ones_idx, int(current_sum - n_points/2), replace=False)
                        ind[remove_indices] = 0
                    elif current_sum < n_points/2:
                        # Add missing ones
                        zeros_idx = np.where(ind == 0)[0]
                        add_indices = np.random.choice(zeros_idx, int(n_points/2 - current_sum), replace=False)
                        ind[add_indices] = 1
                    return ind
                
                child1 = repair(child1)
                child2 = repair(child2)
                
                new_population[i] = child1
                new_population[i+1] = child2
        
        # Mutation
        for ind in new_population:
            if np.random.random() < mutation_rate:
                # Swap one 0 and one 1
                zeros = np.where(ind == 0)[0]
                ones = np.where(ind == 1)[0]
                if len(zeros) > 0 and len(ones) > 0:
                    idx0 = np.random.choice(zeros)
                    idx1 = np.random.choice(ones)
                    ind[idx0], ind[idx1] = ind[idx1], ind[idx0]
        
        # Evaluate and replace
        population = new_population
        fitness_values = [fitness(ind) for ind in population]
        
        # Elitism
        best_idx = np.argmax(fitness_values)
        if gen % 10 == 0 and gen > 0:
            print(f"Gen {gen}: Best fitness = {-fitness_values[best_idx]}")
    
    # Return best individual
    best_idx = np.argmax(fitness_values)
    return population[best_idx]

def simulated_annealing(h_start, budget_s, seed=42):
    """Simulated Annealing refinement"""
    np.random.seed(seed)
    start_time = time.time()
    
    current_h = h_start.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing schedule
    initial_temp = 0.1
    final_temp = 0.001
    max_iter = 10000
    
    temp = initial_temp
    alpha = (final_temp / initial_temp) ** (1.0 / (max_iter * 0.2))
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.3:
        # Generate neighbor by swapping a 0 and 1
        zeros = np.where(current_h == 0)[0]
        ones = np.where(current_h == 1)[0]
        if len(zeros) > 0 and len(ones) > 0:
            idx0 = np.random.choice(zeros)
            idx1 = np.random.choice(ones)
            
            # Create neighbor
            neighbor_h = current_h.copy()
            neighbor_h[idx0] = 1
            neighbor_h[idx1] = 0
            
            # Project to sum constraint if needed
            neighbor_h = project_to_sum_box(neighbor_h, TARGET_SUM)
            
            neighbor_c5 = evaluate_c5(neighbor_h)
            
            # Accept or reject
            delta = neighbor_c5 - current_c5
            
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
        temp *= alpha
        iter_count += 1
        
        if iter_count % 100 == 0:
            print(f"SA Iter {iter_count}: C5 = {current_c5}")
    
    return best_h, best_c5

def smooth_gradient_descent(h_start, budget_s, seed=42):

    np.random.seed(seed)
    start_time = time.time()
    
    h = h_start.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Use a simple gradient descent variant with momentum
    learning_rate = 0.01
    momentum = 0.9
    prev_update = np.zeros_like(h)
    
    iter_count = 0
    while time.time() - start_time < budget_s * 0.1:
        # For a smooth approximation, we'll compute a gradient estimate
        # We'll use finite differences to get the gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(len(h)):
            # Perturb dimension i
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Apply update with momentum
        update = momentum * prev_update - learning_rate * grad
        h = h + update
        
        # Project onto feasible set
        h = project_to_sum_box(h, TARGET_SUM)
        
        iter_count += 1
        
        if iter_count % 100 == 0:
            print(f"GD Iter {iter_count}: C5 = {evaluate_c5(h)}")
    
    return h
