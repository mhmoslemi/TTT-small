import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradients(h, eps):
    """Compute gradient of smooth approximation to max overlap"""
    n = len(h)
    # Get all overlaps for current h
    corr = compute_corr_fft(h)
    corr = corr[n-1:2*n-2]  # Full correlation of same length as h
    
    # Softmax weights - subtract max for numerical stability
    max_c = np.max(corr)
    exp_corr = np.exp(eps * (corr - max_c))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient for each component
    grad = np.zeros(n)
    
    # The gradient of the smooth max comes from contributions of each
    # overlap term
    # Hand-crafted efficient handling of left/right shifts in the derivative
    # For a shift s, if s >= 0:
    #   c_s = sum_{i=0}^{n-1-s} h_{i+s} * (1-h_i)
    #   So for h_i, contributions are:
    #   - if i>=s: -h_{i-s} (coefficient of 1-h_i)
    #   - if i<n-1-s: (1-h_{i+s}) (coefficient of h_{i+s})
    
    # We precompute contributions using sliding windows
    # Instead of explicit loops, leverage np.correlate for efficiency
    
    # For s = 1 to n-1 (negative shift), compute as for positive shift but reversed
    # Also handle s=0 separately
    
    # For each shift k=0..n-1, we have derivative contributions
    for k in range(n):
        if k == 0:
            # Gradient of h_i with respect to c_0 = sum h_i * (1 - h_i)
            # ∂c_0/∂h_i = 1 - 2*h_i
            grad += weights[k] * (1 - 2*h)
        else:
            # Compute the contribution of shift k using vector operations
            # Let h_shifted_left be [0,...,0, h_0,...,h_{n-k-1}] (k zeros on left)
            h_shifted_left = np.zeros(n)
            h_shifted_left[k:] = h[:n-k]
            # Let h_shifted_right be [h_k,...,h_{n-1}, 0,...,0] (k zeros on right)
            h_shifted_right = np.zeros(n)
            h_shifted_right[:n-k] = h[k:]
            
            # So c_k = sum_{i=0}^{n-1-k} h_{i+k} (1 - h_i)
            # ∂c_k/∂h_i = (1 - h_{i-k}) if i >= k (coefficient of h_{i+k})
            #             -h_{i+k} if i <= n-k-1 (coefficient of h_i)
            #           = 0 else
            
            # For i, we want contributions of shift k:
            der1 = np.zeros(n)  # contribution from h_{i-k} term (which contributes to 1-h_i)
            der2 = np.zeros(n)  # contribution from h_{i+k} term (which contributes to h_{i+k})
            
            # For terms that are shifted left (contribute as h_{k+i} in the sum)  
            der1[k:] = -(1-h[:n-k])  # coefficient of (1-h_{i-k})  

            # For terms that are shifted right (contribute as h_{i+k} in the sum)
            der2[:n-k] = -h[k:]   # coefficient of h_{i+k}  

            # Add their sum
            grad += weights[k] * (der1 + der2)
            
    return grad

        restarts.append(h_init)
    
    # Keep track of the best found solution
    global_best_h = None
    global_best_c5 = float('inf')
    
    for seed_idx, h in enumerate(restarts):
        # Local restart optimization
        h = h.copy()
        eps = 1.0
        alpha_local = alpha
        
        # Start momentum
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        
        last_improvement_iter = 0
        for i in range(max_iter_per_restart):
            # Check time limit
            if time.time() - start_time > 0.95 * budget_s:
                break
            
            # Store current value to compute difference later
            current_h = h.copy()
            current_c5 = evaluate_c5(h)
            
            # Update epsilon every 500 iterations
            if i > 0 and i % 500 == 0:
                eps = min(eps * eps_schedule, eps_max)
                # Reduce learning rate when ε increases
                alpha_local = max(0.01, alpha_local / 2.0)
            
            # Compute gradient of smooth surrogate
            if eps > 0.0:
                grad = compute_gradients(h, eps)
            else:
                grad = np.zeros_like(h)
            
            # Update momentum
            m = momentum_beta1 * m + (1 - momentum_beta1) * grad
            v = momentum_beta2 * v + (1 - momentum_beta2) * grad**2
            
            # Compute biased estimates of moments
            m_hat = m / (1 - momentum_beta1**(i+1))
            v_hat = v / (1 - momentum_beta2**(i+1))
            
            # Update step
            new_h = h - alpha_local * m_hat / (np.sqrt(v_hat) + epsilon_adam)
            
            # Project back onto constraints
            h = project_to_sum_box(new_h, TARGET_SUM)
            
            # Periodically evaluate current and update best of this restart
            if i % 100 == 0:
                eval_c5 = evaluate_c5(h)
                if eval_c5 < current_c5:
                    current_c5 = eval_c5
                if eval_c5 < best_c5:
                    best_c5 = eval_c5
                    best_h = h.copy()
                if eval_c5 < global_best_c5:
                    global_best_c5 = eval_c5
                    global_best_h = h.copy()
                # Early stopping if no improvement in last 200 iterations
                if i - last_improvement_iter > 200:
                    break
                
                # Update last improvement counter when better than before
                if eval_c5 < global_best_c5 + 1e-10:
                    last_improvement_iter = i
        
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5

        elif current_sum < TARGET_SUM:

        best_h_refined = binary_h.copy()

            new_h = binary_h.copy()
            new_h[idx1] = 1 - new_h[idx1]
            new_h[idx2] = 1 - new_h[idx2]
            
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5_refined:
                best_h_refined = new_h.copy()
                best_c5_refined = new_c5
        
        return best_h_refined, best_c5_refined
    
    # Apply refinement if needed

        if c5_refined < global_best_c5:
            global_best_h = h_refined
            global_best_c5 = c5_refined
    
    # Return final result
    return (global_best_h.tolist(), float(global_best_c5), n_points)
