def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_gradients(h, c_k_vals, epsilon):
    """Compute gradient of smooth max function."""
    # Compute weights for each shift
    c_k_vals = np.array(c_k_vals)
    max_c = np.max(c_k_vals)
    exp_vals = np.exp(epsilon * (c_k_vals - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    for k in range(-n+1, n):
        # Determine indices for shift k
        if k >= 0:
            # Positive shift
            shifted_indices = np.arange(k, n)
            original_indices = np.arange(n - k)
            if len(shifted_indices) > 0 and len(original_indices) > 0:
                grad[original_indices] -= weights[k + n - 1] * h[shifted_indices]
                grad[shifted_indices] += weights[k + n - 1] * (1 - h[original_indices])
        else:
            # Negative shift
            shifted_indices = np.arange(n + k)
            original_indices = np.arange(-k, n)
            if len(shifted_indices) > 0 and len(original_indices) > 0:
                grad[original_indices] -= weights[-k + n - 2] * h[shifted_indices]
                grad[shifted_indices] += weights[-k + n - 2] * (1 - h[original_indices])
    
    return grad

def compute_c_k_values(h):
    """Compute all c_k values using FFT"""
    corr = compute_corr_fft(h)
    # c_k is value at index k+n-1 in the full correlation
    c_k_vals = []
    for k in range(-n_points+1, n_points):
        idx = k + n_points - 1
        c_k_vals.append(corr[idx])
    return c_k_vals

def gradient_descent_step(h, epsilon, alpha, m_prev, v_prev, beta1=0.9, beta2=0.999, eps_adam=1e-8):
    """Perform a single gradient descent step with Adam update"""
    c_k_vals = compute_c_k_values(h)
    grad = compute_gradients(h, c_k_vals, epsilon)
    
    # Adam update
    m = beta1 * m_prev + (1 - beta1) * grad
    v = beta2 * v_prev + (1 - beta2) * grad**2
    
    m_hat = m / (1 - beta1**1000)
    v_hat = v / (1 - beta2**1000)
    
    # Update h
    h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
    
    # Project onto feasible set
    h_proj = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_proj, m, v
