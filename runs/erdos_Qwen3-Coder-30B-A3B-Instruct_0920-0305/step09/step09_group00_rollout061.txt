import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse binary sequence of length n"""
    if n <= 0:
        return np.array([])
    # Generate the Thue-Morse sequence for 0 to n-1
    tm = np.array([bin(i).count('1') % 2 for i in range(n)])
    return tm.astype(float)

def hillclimb(h, max_iters):
    """Perform hill climbing with random swaps"""
    h_cur = h.copy()
    current_c5 = evaluate_c5(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur < 0.5)[0]
    
    improved = True
    iter_count = 0
    
    while improved and iter_count < max_iters:
        improved = False
        # Try several swaps
        for _ in range(100):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new = h_cur.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            
            c5_new = evaluate_c5(h_new)
            if c5_new < current_c5 - 1e-12:
                h_cur = h_new
                current_c5 = c5_new
                improved = True
                # Update sets
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur < 0.5)[0]
                break
                
        iter_count += 1
    return h_cur, current_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    remaining_time = lambda: budget_s - (time.time() - start_time)
    
    # Try Thue-Morse as initial solution
    h_tm = thue_morse_sequence(n_points)
    c5_tm = evaluate_c5(h_tm)
    if c5_tm < best_c5:
        best_h = h_tm.copy()
        best_c5 = c5_tm
    
    # Random search for 20% of time
    random_search_time = budget_s * 0.2
    random_trials = 0
    while remaining_time() > random_search_time * 0.5 and random_trials < 10000:
        # Generate random binary vector with exactly half ones
        indices = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        h = np.zeros(n_points)
        h[indices] = 1.0
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        random_trials += 1

    # Local improvement using hill climbing
    if remaining_time() > 10:
        h_improved, c5_improved = hillclimb(best_h, 5000)
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved
    
    # Simulated annealing with binary moves (if time allows)
    if remaining_time() > 30:
        current_h = best_h.copy()
        current_c5 = best_c5
        temp = 0.5
        end_temp = 0.001
        cooling_rate = 0.9995
        max_iter = 10000
        
        all_ones = np.where(current_h > 0.5)[0]
        all_zeros = np.where(current_h < 0.5)[0]
        
        for i in range(max_iter):
            if remaining_time() < 1:
                break
                
            # Temperature schedule
            temp = max(end_temp, temp * cooling_rate)
            
            # Perturb solution
            if len(all_ones) > 0 and len(all_zeros) > 0:
                idx_one = np.random.choice(all_ones) 
                idx_zero = np.random.choice(all_zeros)
                
                h_new = current_h.copy()
                h_new[idx_one] = 0.0
                h_new[idx_zero] = 1.0
                
                # Evaluate new solution
                c5_new = evaluate_c5(h_new)
                
                # Accept or reject based on Metropolis criterion
                if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
                    current_h = h_new.copy()
                    current_c5 = c5_new
                    
                    # Update sets
                    all_ones = np.where(current_h > 0.5)[0]
                    all_zeros = np.where(current_h < 0.5)[0]
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
    
    # Final refinement using continuous optimization
    if remaining_time() > 10:
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                best_h = res.x
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
        except:
            pass
            
    return (best_h.tolist(), float(best_c5), n_points)
