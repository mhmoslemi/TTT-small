def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_for_shift(h, shift_ell):

            grad[i] += 1 - h[i + shift_ell]
        # For shift ell >= 0, indices where second term contributes
        for i in range(shift_ell, n):
            grad[i] -= h[i - shift_ell]

        # For indices where second term contributes (h_{i-ell} contributes)
        for i in range(ell_abs, n):
            grad[i] -= h[i - ell_abs]
    
    return grad

        grad_norm = np.linalg.norm(grad)

            grad = grad / grad_norm

        h_new = project_to_sum_box(h_new, TARGET_SUM)

                best_c5 = c5_current

            # Reduce step size if no improvement

            if eta < 1e-8:
                break
    
    return best_h, best_c5
