def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_vector(h):
    """Compute correlation vector using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def exact_C5(h):
    """Compute the exact C5 value"""
    corr = correlation_vector(h)
    return dx * np.max(corr)

def compute_weighted_gradient(h, wgt):
    """Compute gradient of the smoothed objective"""
    n = len(h)
    grad = np.zeros(n)
    
    # Accumulate contributions for all shifts
    for k in range(-n+1, n):
        if k >= 0:
            left = np.concatenate([np.zeros(k), h[:-k]])
            right = h[k:]
        else:
            left = h[-k:]
            right = np.concatenate([h[:k], np.zeros(-k)])
        
        # Build gradient term
        grad += -wgt[k+n-1] * (left + right)
    
    # Add constant term (1 for each position)
    grad += np.ones(n)
    
    return grad

    v = np.clip(v, 0, 1)

    def sum_func(lam):
        return np.sum(np.clip(v + lam, 0, 1)) - w
        
    # Binary search for the right lambda
    a, b = -1.0, 1.0
    while sum_func(a) * sum_func(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_func(m)
        if abs(fm) < tol:
            break
        if fm * sum_func(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(v + (a + b)/2, 0, 1)
