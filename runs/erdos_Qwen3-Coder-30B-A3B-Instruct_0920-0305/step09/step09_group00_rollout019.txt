```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def autocorr_fft(h):
    """Compute full autocorrelation using FFT"""
    H = fft(h, 2 * len(h) - 1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights for given g and alpha"""
    g_shifted = g - np.max(g)  # For numerical stability
    exp_g = np.exp(alpha * g_shifted)
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of smooth surrogate"""
    # Convolution with h (linear)  
    r1 = np.convolve(p, h, mode='full')[:len(h)]
    # Convolution with reversed p
    r2 = np.convolve(p[::-1], h, mode='full')[:len(h)]
    return -(r1 + r2)

def project_onto_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Clip to box constraints
    x_clipped = np.clip(x, 0, 1)
    # Binary search for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(x_clipped + lam, 0, 1)) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < 1e-10:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x_clipped + (a + b) / 2, 0, 1)
    return result

def c5_true(h):
    """Exact computation of C5 using full correlation"""
    h_complement = 1 - h
    corr = np.correlate(h, h_complement, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Initial construction with Thue-Morse pattern
    h_tm = np.array([bin(i).count('1') % 2 for i in range(n_points)])
    # Scale to match target sum
    scale = TARGET_SUM / np.sum(h_tm)
    h_tm = np.clip(h_tm * scale, 0, 1)
    h = h_tm.copy()
    
    # Try a few random initializations
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.8:
            break
        h_rand = np.random.rand(n_points)
        h_rand = project_onto_sum_box(h_rand, TARGET_SUM)
        if c5_true(h_rand) < c5_true(h):
            h = h_rand.copy()
    
    # Coarse optimization (n=64)
    n_coarse = 64
    h_coarse = h[:n_coarse]
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    # Optimization loop with deterministic annealing
    alpha = 1.0
    alpha_max = 2000.0
    gamma = 1.3
    
    while alpha <= alpha_max and time.time() - start_time < budget_s * 0.6:
        # Inner loop for fixed alpha
        for it in range(200):
            if time.time() - start_time > budget_s:
                break
                
            # Compute auto-correlations
            A = autocorr_fft(h)
            g = -A  # Negative auto-correlations
            
            # Compute softmax weights
            p = softmax_weights(g, alpha)
            
            # Compute gradient
            grad = gradient(h, p)
            
            # Update step
            eta = 0.1 * np.exp(-alpha / 500)
            h_new = h - eta * grad
            
            # Project onto constraints
            h_new = project_onto_sum_box(h_new, TARGET_SUM)
            
            # Accept if improvement in surrogate
            A_new = autocorr_fft(h_new)
            g_new = -A_new
            F_new = np.log(np.sum(np.exp(alpha * g_new))) / alpha
            
            A_old = autocorr_fft(h)
            g_old = -A_old
            F_old = np.log(np.sum(np.exp(alpha * g_old))) / alpha
            
            if F_new < F_old:
                h = h_new.copy()
        
        # Increase alpha
        alpha *= gamma
        
        # Optionally reduce step size
        if time.time() - start_time > budget_s * 0.4:
            break
    
    # Refinement to finer grid
    if time.time() - start_time < budget_s * 0.8:
        # Interpolate coarse solution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_coarse), h_coarse)
        h = project_onto_sum_box(h_interp, TARGET_SUM)
        
        # Refine with same annealing procedure
        alpha = 1.0
        while alpha <= alpha_max and time.time() - start_time < budget_s * 0.9:
            for it in range(100):
                if time.time() - start_time > budget_s:
                    break
                    
                A = autocorr_fft(h)
                g = -A
                
                p = softmax_weights(g, alpha)
                grad = gradient(h, p)
                
                eta = 0.05 * np.exp(-alpha / 500)
                h_new = h - eta * grad
                h_new = project_onto_sum_box(h_new, TARGET_SUM)
                
                # Accept improvement
                A_new = autocorr_fft(h_new)
                g_new = -A_new
                F_new = np.log(np.sum(np.exp(alpha * g_new))) / alpha
                
                A_old = autocorr_fft(h)
                g_old = -A_old
                F_old = np.log(np.sum(np.exp(alpha * g_old))) / alpha
                
                if F_new < F_old:
                    h = h_new.copy()
            
            alpha *= gamma
            
            if time.time() - start_time > budget_s * 0.7:
                break
    
    # Discrete improvement via local search
    if time.time() - start_time < budget_s * 0.98:
        h_bin = (h > 0.5).astype(int)
        c5_old = c5_true(h)
        best_h = h.copy()
        best_c5 = c5_old
        
        for _ in range(2000):
            if time.time() - start_time > budget_s:
                break
                
            # Random swap
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Perform swap
            h_bin[i], h_bin[j] = 0, 1
            h_trial = h_bin.astype(float)
            h_trial = project_onto_sum_box(h_trial, TARGET_SUM)
            
            c5_new = c5_true(h_trial)
            if c5_new < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_new
                
            # Revert swap
            h_bin[i], h_bin[j] = 1, 0
            
        h = best_h.copy()
    
    # Final polishing with SLSQP
    if time.time() - start_time < budget_s * 0.99:
        def obj(h):
            return c5_true(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                h = res.x
            h = project_onto_sum_box(h, TARGET_SUM)
        except:
            pass
    
    return (h.tolist(), float(c5_true(h)), n_points)
```
