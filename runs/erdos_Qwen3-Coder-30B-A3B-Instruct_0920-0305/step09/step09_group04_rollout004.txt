import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, c_values, epsilon):
    """Compute gradient of smooth max: ∇f_ε(h) = Σ_k w_k · ∇c_k(h)""" 
    n = len(h)
    grad = np.zeros(n)
    c_max = np.max(c_values)
    
    # Compute softmax weights
    exp_c = np.exp(epsilon * (c_values - c_max))
    weights = exp_c / np.sum(exp_c)
    
    # For each k, compute ∇c_k(h) and accumulate
    # This is a simplified approach with precomputed shifts leveraging symmetry
    # We compute it efficiently by direct component-wise differentiation
    # But for speed, this function will be implemented using vectorized shifts
    
    # Precompute shifted versions of 1-h and h 
    # This creates a more efficient but slightly less straightforward gradient computation
    
    for k in range(n):
        if k == 0:
            # Original h derivative: -1 for first component, 0 otherwise (in case of straightforward negation)
            # This form reflects the derivative of c_k = sum over i h_{i+k} * (1 - h_i) 
            # Except at boundaries where shifts are not full
            
            # Full overlap case k=0: c_0 = sum_i h_i * (1 - h_i) = sum h_i - sum h_i^2
            # ∂c_0 / ∂h_j = 1 - 2 * h_j
            # This is not k=0 case in convolution but we're computing all k
            continue
        # For every non-zero shift k, shift h and 1-h accordingly and find that Δ = seg_diff
        # Actually for negative k, we'll compute both positive and negative versions
        # We include two parts: 1) positive shift contributions, 2) negative shift contributions
        if k > 0:
            # shift h right (first part) and 1-h left
            h1 = np.concatenate([np.zeros(k), h[:-k]])  # h_{i+k} 
            h2 = np.concatenate([h[k:], np.zeros(k)])   # h_i 
            grad += weights[k] * (h1 - h2)
        else:
            # k < 0, so we look at h_{i+|k|} (shift left)  
            k_abs = -k
            h1 = np.concatenate([np.zeros(k_abs), h[:-k_abs]])
            h2 = np.concatenate([h[k_abs:], np.zeros(k_abs)])
            grad += weights[-k] * (h1 - h2)
    
    return grad

    c_all[n_points:] = np.convolve(h, 1-h, mode='full')[n_points:]

    c_values = c_all[n_points-1:]
    
    # Add adjustment for symmetric shifts (negative shifts)
    # Some extra entries exist but we really need correlation to utilize symmetry
    
    # Proactive recalculation for k=0 to 2*(n-1) - 1
    corr_full = np.correlate(h, 1-h, mode='full')
    c_values = corr_full[n_points-1:]
    
    # Softmax estimation
    c_max = np.max(c_values)
    exp_c = np.exp(epsilon * (c_values - c_max))
    log_sum_exp = c_max + np.log(np.sum(exp_c))
    return log_sum_exp / epsilon
