import numpy as np
from scipy.signal import correlate

        corr = np.correlate(h, 1 - h, mode='full')
        return dx * np.max(corr)

    h = cp.Variable(n_points)
    t = cp.Variable(1)

    # Only upper triangular part is needed
    z = {}

        for j in range(i+1, n_points):
            z[(i,j)] = cp.Variable(1)
    
    # Constraints
    constraints = []
    
    # Bound constraints
    constraints += [0 <= h[i] for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # Mass constraint
    constraints += [cp.sum(h) == target_sum]
    
    # McCormick constraints
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints += [z[(i,j)] <= h[i]]
            constraints += [z[(i,j)] <= h[j]]
            constraints += [z[(i,j)] >= h[i] + h[j] - 1]
            constraints += [z[(i,j)] >= 0]
    
    # Overlap constraints
    for k in range(1, n_points):
        # c_k = sum_{i=0}^{n-1-k} ( h[i+k] - z[i,i+k] )

        for i in range(n_points - k):
            expr += h[i+k] - z[(i, i+k)]

    objective = cp.Minimize(t)

    prob = cp.Problem(objective, constraints)
