import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Construct initial feasible h using the given initial_h_values
    h_init = np.array(initial_h_values)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (TARGET_SUM / np.sum(h_init))
    
    # Convex-Concave Procedure (CCP) approach
    def evaluate_c5_ccp(h):  # evaluates max correlation
        h = np.array(h)
        # Compute all correlations using FFT for speed
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return np.max(corr) * dx  # Scale by dx to match continuous case
    
    # Initialize
    best_h = h_init.copy()
    best_c5 = evaluate_c5_ccp(best_h)
    
    # Record best solution found so far
    record_h = best_h.copy()
    record_c5 = best_c5
    
    # CCP iterations
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_c5 = evaluate_c5_ccp(best_h)
        if current_c5 < record_c5:
            record_h = best_h.copy()
            record_c5 = current_c5
            
        # Perform one step of CCP (linearization)
        # The objective is now to minimize the maximum correlation
        
        # At each step apply local modification to improve on current solution
        # We do so by gradient free local search with proposals
        # Try a few small changes to the step function
        
        # This being a discrete optimization problem, we do local neighborhood search
        # Create a perturbed version of current solution
        
        # Try a few random walks on 10-30 indices
        num_moves = min(30, n_points // 5)
        indices = np.random.choice(n_points, num_moves, replace=False)
        
        best_improved = False
        for _ in range(20):  # Take several tries per random indices set
            new_h = best_h.copy()
            dm = np.random.uniform(-0.05, 0.05, num_moves)
            for i, d in zip(indices, dm):
                new_h[i] += d
            # Ensure constraints
            new_h = np.clip(new_h, 0, 1)
            # Project to correct sum
            new_h = new_h * (TARGET_SUM / np.sum(new_h))
            
            new_c5 = evaluate_c5_ccp(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                best_improved = True
                break
                
        # If no improvement in this loop, continue
        if not best_improved:
            # Perform a global step with more exploration
            # Generate a candidate by sampling from a Gaussian
            candidate = np.random.normal(0.5, 0.15, n_points)
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (TARGET_SUM / np.sum(candidate))
            
            new_c5 = evaluate_c5_ccp(candidate)
            if new_c5 < best_c5:
                best_h = candidate
                best_c5 = new_c5
                
    return (record_h.tolist(), float(record_c5), n_points)
