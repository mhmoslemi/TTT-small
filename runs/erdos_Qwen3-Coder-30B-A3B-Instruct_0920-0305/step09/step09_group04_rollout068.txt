import numpy as np
from scipy.optimize import minimize
from cvxpy import*
import time

# Global constants
n_points = 101  # Prime number for better autocorrelation properties
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate improved initial solution using Legendre-like sequence
    # Using a binary sequence with good correlation properties
    # We'll use a variant that maps to [0,1] interval
    
    # For n=101, construct a binary sequence using a method that approximates 
    # low-correlation properties (similar to Legendre sequences)
    
    # Create pseudo-Legendre-like sequence with n=101
    # This is not a true Legendre sequence but mimics its low correlation properties
    def create_legendre_like(n):
        # Use a simple construction with good correlation properties
        # Start with an even distribution and make slight adjustments to reduce correlation
        h = np.ones(n) * 0.5
        # Add some small perturbations to break symmetry and reduce correlation peaks
        for i in range(n):
            if i % 7 == 0:
                h[i] += 0.05
            elif i % 11 == 0:
                h[i] -= 0.05
        return np.clip(h, 0, 1)
    
    h = create_legendre_like(n_points)
    h = project_to_sum_box(h, TARGET_SUM)
    current_c5 = evaluate_c5(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = current_c5
    
    # Phase 1: Local search with structured perturbations to reduce correlation peak
    def structured_local_search(h, max_iters=1000):
        h = h.copy()
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        for iteration in range(max_iters):
            # Large-scale modification strategy
            if np.random.random() < 0.3:
                # Perturb large blocks
                block_size = max(5, n_points // 20)
                start_idx = np.random.randint(0, n_points - block_size)
                for i in range(start_idx, start_idx + block_size):
                    h[i] += np.random.uniform(-0.02, 0.02)
                h = project_to_sum_box(h, TARGET_SUM)
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
            else:
                # Fine-grained perturbation
                idx = np.random.randint(0, n_points)
                old_val = h[idx]
                delta = np.random.uniform(-0.03, 0.03)
                h[idx] = old_val + delta
                if h[idx] < 0 or h[idx] > 1:
                    h[idx] = old_val
                    continue
                h = project_to_sum_box(h, TARGET_SUM)
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                else:
                    h[idx] = old_val
                    
            if time.time() - start_time > budget_s * 0.8:
                break
        
        return best_h, best_c5
    
    # Run first local search
    h, c5 = structured_local_search(best_h, 500)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    # Phase 2: Different local search using neighborhood exploration
    def neighborhood_search(h, max_iters=800):
        h = h.copy()
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        idx_set = set()  # Track indices that are currently modified
        
        for iteration in range(max_iters):
            # Choose which indices to perturb
            if np.random.random() < 0.5:
                # Perturb many indices
                indices = np.random.choice(n_points, min(10, n_points//5), replace=False)
                idx_set.update(indices)
            else:
                # Perturb single index
                indices = [np.random.randint(0, n_points)]
                idx_set.add(indices[0])
            
            # Store original values
            originals = h[indices].copy()
            
            # Try to improve
            improvement_found = False
            for idx in indices:
                original = h[idx]
                # Try different perturbations (including negative)
                for delta in np.array([-0.05, -0.03, -0.01, 0.01, 0.03, 0.05]):
                    h[idx] = original + delta
                    if h[idx] < 0 or h[idx] > 1:
                        continue
                    
                    h = project_to_sum_box(h, TARGET_SUM)
                    new_c5 = evaluate_c5(h)
                    
                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                        improvement_found = True
                        break
                    else:
                        h[idx] = original  # revert
                        
                if improvement_found:
                    break
            else:
                # No improvement found, restore indices
                h[indices] = originals
                
            if time.time() - start_time > budget_s * 0.9:
                break
        
        return best_h, best_c5
    
    # Run second search
    h, c5 = neighborhood_search(best_h, 500)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    # Phase 3: Final adaptive search to improve solution
    def adaptive_search(h, max_iters=1000):
        h = h.copy()
        current_c5 = evaluate_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        step_size = 0.05
        
        for iteration in range(max_iters):
            step_size = max(0.005, step_size * 0.995)  # Gradually decrease
            
            idx = np.random.randint(0, n_points)
            old_val = h[idx]
            delta = np.random.choice([-step_size, step_size])
            h[idx] = old_val + delta
            
            if h[idx] < 0 or h[idx] > 1:
                h[idx] = old_val
                continue
                
            h = project_to_sum_box(h, TARGET_SUM)
            new_c5 = evaluate_c5(h)
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                step_size = min(0.05, step_size * 1.1)  # Speed up if improving
            else:
                h[idx] = old_val
                
            if time.time() - start_time > budget_s:
                break
                
        return best_h, best_c5
    
    # Final search
    h, c5 = adaptive_search(best_h, 500)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5

    return (best_h.tolist(), float(best_c5), n_points)
