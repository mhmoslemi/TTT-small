import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 150
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_shifts = n_points  # Since we use symmetric shifts, we only consider up to n-1 shifts
    
    # The initial construction from `initial_h_values`
    h0 = np.array(initial_h_values)
    h0 = np.clip(h0, 0, 1)
    h0_sum = np.sum(h0)
    h0 = h0 * target_sum / h0_sum
    
    best_h = h0.copy()
    best_c5 = compute_c5_fast(best_h)
    
    # The main optimization loop
    # Function to compute overlap matrix
    def build_overlap_matrices(n):
        M_list = []
        for l in range(n):  # Shifts 0 to n-1
            M = np.zeros((n, n))
            for i in range(n):
                j = i + l
                if j < n:
                    M[i, j] = 1
                    M[j, i] = 1
            M_list.append(M)
        return M_list
    
    # Build matrices once
    M_matrices = build_overlap_matrices(n_points)
    
    def compute_c5_fast(h):
        c_vals = []
        for M in M_matrices:
            overlap_val = h.T @ M @ h
            c_val = 0.5 * n_points - overlap_val  # because sum(h) = n/2
            c_vals.append(c_val)
        return dx * max(c_vals)
    
    def solve_sdp_relaxation(alpha_bound, h_guess=None):
        # The SDP relaxation approach
        h = Variable(n_points)
        X = Variable((n_points, n_points), PSD=True)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == target_sum,
            trace(X) == target_sum,  # This ensures X is consistent with h
        ]
        
        # McCormick envelopes: X[i,j] <= min(h[i], h[j]), etc.
        for i in range(n_points):
            for j in range(i+1, n_points):
                constraints.append(X[i, j] <= h[i])
                constraints.append(X[i, j] <= h[j])
                constraints.append(X[i, j] >= 0)
                constraints.append(X[i, j] >= h[i] + h[j] - 1)
        
        # For each shift l, the constraint that trace(M_l X) >= (n/2 - alpha)
        for l, M in enumerate(M_matrices[:max_shifts]):
            overlap_l = trace(M @ X)
            constraints.append(overlap_l >= target_sum - alpha_bound)
            
        # Objective: minimize the objective (we're solving feasibility, not optimizing)
        # This is just to satisfy CVXPY syntax (using minimal objective)
        obj = Minimize(0)
        
        prob = Problem(obj, constraints)
        
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=5000)
            if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                h_sol = h.value
                if h_sol is not None:
                    # Now round to get a feasible solution
                    # We'll use randomized rounding
                    return h_sol
        except:
            pass
        return None
    
    # Bisection on alpha to find minimal valid bound
    def find_minimal_alpha():
        low, high = 0.0, 1.0
        best_alpha = 1.0
        
        max_iter = 25  # Limited iterations for time constraint
        for _ in range(max_iter):
            mid = (low + high) / 2.0
            h_result = solve_sdp_relaxation(mid)
            if h_result is not None and h_result.all() is not None:
                # Feasible solution found
                # Have to project to valid constraints
                h_proj = np.clip(h_result, 0, 1)
                h_proj_sum = np.sum(h_proj)
                h_proj = h_proj * target_sum / h_proj_sum
                c5_candidate = compute_c5_fast(h_proj)
                if c5_candidate < best_c5:
                    best_h[:] = h_proj
                    best_c5 = c5_candidate
                best_alpha = mid
                high = mid
            else:
                low = mid
                
            if time.time() - start_time > budget_s - 1:
                break
                
        return best_alpha

    # Start with the known starting point
    c5_init = compute_c5_fast(best_h)
    if c5_init < best_c5:
        best_c5 = c5_init

    # Run bisection search
    _ = find_minimal_alpha()

    # Final polishing step
    def polisher(h_start):
        h_current = h_start.copy()
        current_c5 = compute_c5_fast(h_current)
        
        for _ in range(2000):
            if time.time() - start_time > budget_s - 1:
                break
            # Random perturbation with projection
            i = np.random.randint(n_points)
            h_new = h_current.copy()
            h_new[i] += np.random.normal(0, 0.01)
            # Project to [0,1] and correct the sum
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * target_sum / h_new_sum
            new_c5 = compute_c5_fast(h_new)
            if new_c5 < current_c5:
                h_current = h_new
                current_c5 = new_c5
        return h_current, current_c5
        
    polished_h, polished_c5 = polisher(best_h)
    
    if polished_c5 < best_c5:
        best_h = polished_h
        best_c5 = polished_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
