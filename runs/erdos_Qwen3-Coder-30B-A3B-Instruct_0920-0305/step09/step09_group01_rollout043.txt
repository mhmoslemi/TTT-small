def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_shifted_gradients(h, k):
    """Compute gradient of c_k(h) with respect to h"""
    grad = np.zeros_like(h)

        # c_k = sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k})

        grad[k:] -= h[:n_points-k]
    else:
        # For negative k, swap roles  
        k = -k
        grad[k:] += 1 - h[:n_points-k]
        grad[:n_points-k] -= h[k:]
    return grad

def compute_soft_max_grad(h, mu):

    # Assemble gradient
    grad = np.zeros_like(h)

            grad[:n_points-k] += weights[k] * (1 - h[k:])

            grad[:n_points-k_neg] -= weights[n_points-1+k] * h[k_neg:]
    
    return grad

    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    for _ in range(max_trials):
        h_new = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_current:
            return h_new, c5_new, True
        alpha *= 0.5
    
    return h_current, c5_current, False
