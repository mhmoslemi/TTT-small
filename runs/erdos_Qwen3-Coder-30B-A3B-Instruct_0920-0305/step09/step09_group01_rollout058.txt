import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

# Global parameters
n_points = 150
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_M_matrices(n):
    """Compute the matrices M_l for l = 0 to n-1"""
    M_matrices = []
    for l in range(n):
        M = np.zeros((n, n))
        for i in range(n - l):
            M[i, i + l] = 1
            M[i + l, i] = 1
        M_matrices.append(M)
    return M_matrices

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Precompute M matrices
    M_matrices = compute_M_matrices(n_points)
    
    # Initialize with a simple construction
    h0 = np.array(initial_h_values)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # SDP Relaxation approach
    def sdp_solve_and_round(alpha_guess):
        # Create CVXPY variables
        h = Variable(n_points)
        X = Variable((n_points, n_points), symmetric=True)
        
        # Constraints for SDP relaxation
        constraints = [
            X >> 0,  # X is positive semidefinite
            diag(X) == h,  # Diagonal of X equals h
            sum(h) == TARGET_SUM,  # Sum constraint
        ]
        
        # Add McCormick envelope constraints
        for i in range(n_points):
            for j in range(n_points):
                if i != j:
                    constraints.append(X[i, j] <= h[i])
                    constraints.append(X[i, j] <= h[j])
                    constraints.append(X[i, j] >= h[i] + h[j] - 1)
        
        # Add quadratic constraints
        for l in range(n_points):
            constraints.append(trace(M_matrices[l] @ X) >= TARGET_SUM - alpha_guess)
        
        # Create objective (no objective, just feasibility check)
        prob = Problem(Maximize(0), constraints)
        try:
            prob.solve(solver=SCS, verbose=False, eps=1e-4)
        except:
            return None
            
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            X_sol = X.value
            h_sol = h.value
            
            # Random rounding
            U = np.random.randn(n_points)
            if X_sol is not None:
                try:
                    # Eigen-decomposition
                    V, D = np.linalg.eigh(X_sol)
                    D = np.maximum(D, 0)
                    sqrt_D = np.sqrt(D)
                    h_rounded = V @ np.diag(sqrt_D) @ U
                    h_rounded = np.clip(h_rounded, 0, 1)
                    h_rounded = project_to_sum_box(h_rounded, TARGET_SUM)
                    return h_rounded
                except:
                    return None
        return None

    # Bisection on alpha to find minimal feasible value
    left = 0.0
    right = 1.0
    best_alpha = right
    
    # Iterative refinement
    for iteration in range(20):
        if time.time() - start_time > budget_s - 2:
            break
            
        alpha = (left + right) / 2
        
        # Try to solve with this alpha
        h_trial = sdp_solve_and_round(alpha)
        if h_trial is not None:
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                if c5_trial < 0.379:
                    return (best_h.tolist(), float(best_c5), n_points)
            right = alpha
        else:
            left = alpha
    
    return (best_h.tolist(), float(best_c5), n_points)
