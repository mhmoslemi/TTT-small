import numpy as np
from scipy.signal import correlate

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_violations(h, t, n_points):
    """Check for violations of constraint t >= w - sum h_i * h_{i+k}"""
    dx = 2.0 / n_points
    # compute overlap for all shifts
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n_points - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr[:n_points] * dx
    # we have that corr[k] = sum over i of h[i] * (1 - h[i+k]) 
    # so overlap_k = sum h - corr[k] = w - corr[k] where w = n/2
    # so we want all overlaps <= t*dx
    # equivalently corr[k] >= w - t*dx
    violations = np.where(corr < (n_points/2 - t*dx))[0]
    return violations

    h_full = h_full[:n_points]  # ensure length

    h_relaxed = cp.Variable(n_points, name="h")

    # sum constraint
    sum_con = cp.sum(h_relaxed) == w
    constraints_relaxed = [sum_con]
    # add basic constraints on h
    constraints_relaxed += [h_relaxed >= 0, h_relaxed <= 1]
    
    # solve initial relaxed problem
    objective_relaxed = cp.Minimize(t_relaxed)
    prob_relaxed = cp.Problem(objective_relaxed, constraints_relaxed)
    try:
        prob_relaxed.solve(solver=CP_HOTUG, verbose=False)
        if prob_relaxed.status == 'optimal':
            h_relaxed_sol = h_relaxed.value
        else:
            print("LP Relaxed failed")
            h_relaxed_sol = h_initial
    except:
        h_relaxed_sol = h_initial

    h_vars = cp.Variable(n_points, boolean=True)

    while n_iter < max_iter and time.time() - start_time < budget_s * 0.95:

            # use guess or recompute
            h_opt = np.round(h_relaxed_sol).astype(int)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt[:n_points]
        else:
            # re-use last solution or provide better one
            h_opt = h_best.copy()
            # small repair if needed
            # project back to sum constraint
            sum_h = np.sum(h_opt)
            if sum_h < w:
                # fill positions
                indices = np.argsort(h_opt)
                for i in range(int(w - sum_h)):
                    if indices[i] < len(h_opt):
                        h_opt[indices[i]] = 1
            elif sum_h > w:
                # remove
                indices = np.argsort(h_opt)
                for i in range(int(sum_h - w)):
                    if indices[i] >= 0 and indices[i] < len(h_opt):

        if n_iter > 0:

            h_sample = np.random.binomial(1, 0.7, size=n_points)
            if np.sum(h_sample) > w:
                # need to randomly remove
                to_remove = int(np.sum(h_sample) - w)
                indices = np.random.permutation(np.where(h_sample > 0)[0])[:to_remove]
                h_sample[indices] = 0
            elif np.sum(h_sample) < w:
                # add randomly
                to_add = int(w - np.sum(h_sample))
                indices = np.random.permutation(np.where(h_sample == 0)[0])[:to_add]
                h_sample[indices] = 1
                
            new_c5 = evaluate_c5(h_sample)
            if new_c5 < min_c5:
                min_c5 = new_c5
                h_best = h_sample.copy()
        
        # Evaluate current solution once more for possible iteration improvement
        current_c5 = evaluate_c5(h_best)
        if current_c5 < min_c5:
            min_c5 = current_c5
            
        n_iter += 1
        
    # Final optimization via SLSQP run on best solution
    try:
        # optimize final form with continuous relaxation, then project
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_c5(h)
            
        def constraint_eq(h):
            return np.sum(h) - w
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # make final update starting at best solution
        res = minimize(
            objective, 
            h_best, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if res.success:
            final_h = res.x
            final_h = np.clip(final_h, 0, 1)
            final_sum = np.sum(final_h)
            final_h = final_h * w / final_sum  # normalize sum
            final_c5 = evaluate_c5(final_h)
            if final_c5 < min_c5:
                min_c5 = final_c5
                h_best = final_h.copy()
    except:
        pass
    
    # Final check with all shifts applied
    final_c5 = evaluate_c5(h_best)
    
    return (h_best.tolist(), float(final_c5), n_points)
