import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using numpy correlate"""
    return correlate(h, 1 - h, mode="full")

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weighted_gradient(h, wgt):
    """Compute weighted gradient ∇Φ_β(h) = Σ_k w_k · ( 1 – shift_{+k}(h) – shift_{‑k}(h) )"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift k, add contribution to the gradient
    for k in range(-n+1, n):  # k ranges from -(n-1) to n-1
        if k >= 0:
            # Left shift: h[i] contributes to h[i+k] (when i+k < n)
            right_slice = np.zeros(n)
            right_slice[k:] = h[:-k] if k > 0 else h[:]
            
            # Right shift: h[i] contributes to h[i-k] (when i-k >= 0)
            left_slice = np.zeros(n)
            left_slice[:-k] = h[k:] if k > 0 else h[:]
        else:
            # k < 0
            # Left shift: h[i] contributes to h[i+k] (when i+k >= 0)
            right_slice = np.zeros(n)
            right_slice[:k] = h[-k:] if k < 0 else h[:]
            
            # Right shift: h[i] contributes to h[i-k] (when i-k < n)
            left_slice = np.zeros(n)
            left_slice[-k:] = h[:k] if k < 0 else h[:]
        
        # Use the correct slice based on k
        if k >= 0:
            shifted_h = np.zeros(n)
            shifted_h[k:] = h[:-k] if k > 0 else h[:]
            grad += wgt[k + n - 1] * (1 - shifted_h - left_slice)
        else:
            shifted_h = np.zeros(n)
            shifted_h[:k] = h[-k:] if k < 0 else h[:]
            grad += wgt[k + n - 1] * (1 - shifted_h - right_slice)
    
    # Sum of ones term
    grad += np.ones(n) * wgt.sum()
    
    return grad
