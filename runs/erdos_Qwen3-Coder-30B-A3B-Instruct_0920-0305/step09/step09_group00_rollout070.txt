import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n (exactly half 1s)"""
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        seq[i] = bin(i).count('1') % 2
    return seq.astype(float)

def hillclimb(h, max_iters):
    """Perform greedy hill climbing on binary solution"""
    h_cur = h.copy()
    c5_cur = evaluate_c5(h_cur)
    ones = np.where(h_cur > 0.5)[0]
    zeros = np.where(h_cur < 0.5)[0]
    
    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        h_new = h_cur.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h_cur = h_new
            c5_cur = c5_new
            ones = np.where(h_cur > 0.5)[0]
            zeros = np.where(h_cur < 0.5)[0]
    
    return h_cur, c5_cur

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    
    # Start with Thue-Morse sequence
    h_tm = thue_morse_sequence(n_points)
    c5_tm = evaluate_c5(h_tm)
    if c5_tm < best_c5:
        best_h = h_tm.copy()
        best_c5 = c5_tm
    
    # Random search with binary sequences (30% of budget)
    search_time = budget_s * 0.3
    trials = 0
    while time.time() - start_time < start_time + search_time and trials < 10000:
        # Generate random binary vector with exactly half ones
        indices = np.random.permutation(n_points)[:int(TARGET_SUM)]
        h = np.zeros(n_points)
        h[indices] = 1.0
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        trials += 1
    
    # Local improvement using hill climbing
    if best_h is not None:
        h_improved, c5_improved = hillclimb(best_h, max_iters=10000)
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved
    
    # Simulated annealing improvement
    if best_h is not None:
        # Use a simpler simulated annealing with neighborhood moves
        current_h = best_h.copy()
        current_c5 = evaluate_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        temp = 0.5
        end_temp = 0.001
        cooling_rate = 0.9995
        max_iter = 20000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Temperature schedule
            temp = max(end_temp, temp * cooling_rate)
            
            # Perturb solution
            h_new = current_h.copy()
            # Move one 1 to a 0
            ones = np.where(h_new > 0.5)[0]
            zeros = np.where(h_new < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i_idx = np.random.choice(ones)
                j_idx = np.random.choice(zeros)
                h_new[i_idx] = 0.0
                h_new[j_idx] = 1.0
                
                # Evaluate new solution
                c5_new = evaluate_c5(h_new)
                
                # Accept or reject based on Metropolis criterion
                if c5_new < current_c5 or np.random.random() < np.exp(-(c5_new - current_c5) / temp):
                    current_h = h_new.copy()
                    current_c5 = c5_new
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
    
    # Final refinement using bounds-constrained optimization
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        # Use bounds to project the solution back into constraints
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
        
        # Try a final refinement with scipy
        res = minimize(
            lambda x: evaluate_c5(x),
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 1000}
        )
        if res.success:
            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception as e:
        # If optimization fails, keep the previous best
        pass
        
    return (best_h.tolist(), float(best_c5), n_points)
