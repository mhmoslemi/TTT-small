import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    # np.correlate with mode='full' computes the full correlation
    # We want to maximize the overlap over all shifts
    # C5 = max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx to convert to integral
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Target grid size
    n = 150
    dx = 2.0 / n
    
    # ========================
    # Phase 1: MILP Binary Optimization
    # ========================
    
    # Create problem variables
    h = cp.Variable(n, boolean=True)
    
    # Auxiliary variables for pairwise products
    # y[i,k] = h[i] * h[i+k], where k ranges from 0 to n-1
    # but only for valid indices
    y = {}
    
    # Build sumh_k for each shift k
    sumh = []
    for k in range(n):
        # Compute sum_{i=0}^{n-k-1} h[i] 
        # This is just sum of first n-k elements
        sumh.append(cp.sum(h[:n-k]))
    
    # Build the binary variables y[i,k]
    # Use a mapping to avoid dense matrix creation
    y_vars = []
    y_constraints = []
    
    # Define y[i,k] = h[i] * h[i+k] 
    # Constraints: 
    # y[i,k] <= h[i]
    # y[i,k] <= h[i+k]  
    # y[i,k] >= h[i] + h[i+k] - 1
    
    # Create auxiliary variables and constraints
    y_indices = []
    idx = 0
    for k in range(n):
        for i in range(n - k):
            y[i,k] = cp.Variable(boolean=True)
            y_vars.append(y[i,k])
            y_indices.append((i,k))
            y_constraints.append(y[i,k] <= h[i])
            y_constraints.append(y[i,k] <= h[i+k])
            y_constraints.append(y[i,k] >= h[i] + h[i+k] - 1)
    
    # Compute the overlaps efficiently 
    overlap = []
    for k in range(n):
        # sum of y[i,k] for i = 0,...,n-k-1
        if n - k == 0:
            overlap.append(0)
        else:
            overlap.append(cp.sum([y[i,k] for i in range(n-k)]))
    
    # Define c_k = sumh[k] - overlap[k]
    c = [sumh[k] - overlap[k] for k in range(n)]
    
    # Define t as the epigraph variable that bounds all c_k
    t = cp.Variable()
    
    # Add constraints t >= c_k for all k
    constraints = [t >= c[k] for k in range(n)]
    constraints += y_constraints
    constraints.append(cp.sum(h) == n // 2)  # fix sum to n/2
    
    # Create and solve the problem
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Use a solve time limit
    remaining_time = budget_s / 2
    try:
        prob.solve(solver=cp.CBC, time_limit=remaining_time)
        if prob.status == cp.OPTIMAL or prob.status == cp.TIME_LIMIT:
            h_final = h.value
            # Evaluate the actual C5 bound
            h_final = np.array([1 if val else 0 for val in h_final])
            c5_final = evaluate_erdos_solution(h_final)
        else:
            # Set to initial construction
            h_final = np.array(initial_h_values)  
            h_final = np.array([1 if val > 0.5 else 0 for val in h_final])
            c5_final = evaluate_erdos_solution(h_final)
    except:
        # Fall back to initial construction if anything goes wrong
        h_final = np.array(initial_h_values)  
        h_final = np.array([1 if val > 0.5 else 0 for val in h_final])
        c5_final = evaluate_erdos_solution(h_final)
    
    # ========================
    # Phase 2: LP Relaxation + Greedy Rounding (if time remains)
    # ========================
    if time.time() - start_time < budget_s - 10:
        try:
            # Relax to continuous variables
            h_cont = cp.Variable(n)
            constraints_cont = []
            # Same constraints
            for i in range(n):
                constraints_cont.append(h_cont[i] >= 0)
                constraints_cont.append(h_cont[i] <= 1)
                
            # Push these back into y vars using continuous constraints
            y_cont = {}
            idx = 0
            for k in range(n):
                for i in range(n - k):
                    y_cont[i,k] = cp.Variable()
                    y_cont[i,k] = h_cont[i] * h_cont[i+k]  # continuous product
                    # Insert a simpler approximation of the y cont.
                    # y <= h[i], y <= h[i+k], y >= h[i] + h[i+k] - 1
                    constraints_cont.append(y_cont[i,k] <= h_cont[i])
                    constraints_cont.append(y_cont[i,k] <= h_cont[i+k])
                    constraints_cont.append(y_cont[i,k] >= h_cont[i] + h_cont[i+k] - 1)
            
            # Overlap and c using continuous
            overlap_cont = []
            for k in range(n):
                if n - k == 0:
                    overlap_cont.append(0)
                else:
                    overlap_cont.append(cp.sum([y_cont[i,k] for i in range(n-k)]))
                
            c_cont = [sumh[k] - overlap_cont[k] for k in range(n)]
            t_cont = cp.Variable()
            constraints_cont += [t_cont >= c[k] for k in range(n)]
            constraints_cont.append(cp.sum(h_cont) == n // 2)
            
            obj = cp.Minimize(t_cont)
            prob_cont = cp.Problem(obj, constraints_cont)
            prob_cont.solve(solver=cp.CBC, time_limit=10)
            
            if prob_cont.status in [cp.OPTIMAL, cp.TIME_LIMIT]:
                h_lp = h_cont.value
                # Greedy rounding
                idxs = np.argsort(h_lp)[::-1]  # descending sort
                h_rounded = np.zeros(n)
                for i in range(n//2):
                    h_rounded[idxs[i]] = 1
                c5_rounded = evaluate_erdos_solution(h_rounded)
                
                if c5_rounded < c5_final:
                    h_final = h_rounded
                    c5_final = c5_rounded
                    
        except:
            pass
    
    # ========================
    # Phase 3: Local Pair-Swap Polish
    # ========================
    
    def compute_correlations(h_arr):
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        return corr * dx
    
    # Convert to array if needed
    h_current = h_final.copy()
    c5_current = c5_final
    
    # Simple efficient pair-swap local search
    max_swaps = 10000  # more than enough for good improvement
    swaps = 0
    
    while swaps < max_swaps and time.time() - start_time < budget_s - 2:
        improved = False
        
        # Find a 1 and a 0 to swap
        ones = np.where(h_current == 1)[0]
        zeros = np.where(h_current == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random one and zero to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute what would happen if we swap i and j
        old_corr = compute_correlations(h_current)
        old_max = np.max(old_corr)
        
        # Create candidate
        h_test = h_current.copy()
        h_test[i] = 0
        h_test[j] = 1
        
        new_corr = compute_correlations(h_test)
        new_max = np.max(new_corr)
        
        if new_max < old_max:
            h_current = h_test
            c5_current = new_max
            improved = True
            
        swaps += 1
        
        if not improved:
            continue
            
    return (h_current.tolist(), float(c5_current), n)
