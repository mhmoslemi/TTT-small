import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_solution():
    """Generate a better initial solution based on mathematical analysis"""
    h = np.zeros(n_points)
    
    # Create a piecewise constant function that's more balanced
    # Split into 3 regions: low, medium, high
    mid1 = n_points // 3
    mid2 = 2 * n_points // 3
    
    # Low values at start and end, high in middle
    h[:mid1] = 0.2
    h[mid1:mid2] = 0.8
    h[mid2:] = 0.2
    
    # Adjust to meet constraint
    current_sum = np.sum(h)
    scale_factor = TARGET_SUM / current_sum
    h *= scale_factor
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize problem parameters
    TARGET_SUM = n_points / 2
    
    # Use a better initial solution
    h0 = generate_initial_solution()
    
    # Refine with optimization
    # First, project it to correct sum
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Additional improvement: try different configurations
    # Try a more structured approach with peaks
    h_peaks = h0.copy()
    # Add peaks at specific locations
    peaks = [n_points // 4, n_points // 2, 3*n_points // 4]
    for peak in peaks:
        h_peaks[peak] = 1.0
        # Reduce neighbors
        for offset in [-2, -1, 1, 2]:
            if 0 <= peak + offset < n_points:
                h_peaks[peak + offset] = max(0, h_peaks[peak + offset] - 0.1)
    
    h_peaks = project_to_sum_box(h_peaks, TARGET_SUM)
    c5_peaks = evaluate_c5(h_peaks)
    if c5_peaks < best_c5:
        best_h = h_peaks.copy()
        best_c5 = c5_peaks
    
    # Phase 2: Local search with adaptive mutations
    remaining_time = budget_s
    # Use more iterations for early phases when timing allows
    iterations = int(5000 * (remaining_time / budget_s))
    
    for i in range(iterations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try mixed strategy: sometimes large changes, sometimes small
        h_try = best_h.copy()
        if i % 3 == 0:
            # Large changes
            num_changes = min(15, n_points // 6)
            indices = np.random.choice(n_points, size=num_changes, replace=False)
            for idx in indices:
                change = np.random.uniform(-0.15, 0.15)
                h_try[idx] += change
        else:
            # Small changes
            idx = np.random.randint(0, n_points)
            h_try[idx] += np.random.uniform(-0.02, 0.02)
        
        h_try = project_to_sum_box(h_try, TARGET_SUM)
        c5_try = evaluate_c5(h_try)
        
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            
    # Phase 3: Optimization using L-BFGS-B which can handle bounds better
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try a few different starting points for robustness
        for _ in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Start from current best with small random perturbation
            h_start = best_h + np.random.normal(0, 0.01, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            
            res = minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, 
                           options={'maxiter': 300, 'ftol': 1e-12})
            if res.success:
                h_opt = res.x
                h_opt = project_to_sum_box(h_opt, TARGET_SUM)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
    except:
        pass
        
    # Final local refinement with smaller steps
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        # Try small stochastic changes
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        # Small change
        h_new[idx] += np.random.uniform(-0.005, 0.005)
        # Project to valid range and sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
    return (best_h.tolist(), float(best_c5), n_points)
