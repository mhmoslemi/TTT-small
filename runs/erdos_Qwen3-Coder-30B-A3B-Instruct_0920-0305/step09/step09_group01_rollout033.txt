def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):

    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def analytic_gradient(h, c_values, mu):
    """Compute the gradient of log-sum-exp of c_values with temperature mu"""
    # Compute weights using log-sum-exp trick for numerical stability
    c_max = np.max(c_values)
    exp_c = np.exp((c_values - c_max) / mu)
    weights = exp_c / np.sum(exp_c)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient for each shift
    n = len(h)
    for k in range(-n+1, n):
        if k >= 0:
            # Positive shift
            if k < n:
                grad[:n-k] += weights[k] * (1 - h[k:])
                grad[k:] -= weights[k] * h[:n-k]
        else:
            # Negative shift
            k_abs = -k
            if k_abs < n:
                grad[k_abs:] += weights[n-1+k] * (1 - h[:n-k_abs])
                grad[:n-k_abs] -= weights[n-1+k] * h[k_abs:]
    
    return grad

        h_proj = project_to_sum_box(h_current - alpha * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_proj)
        trial += 1
