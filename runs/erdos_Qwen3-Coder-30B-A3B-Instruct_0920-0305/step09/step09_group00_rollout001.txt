import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr) * dx

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_sequence(n):
    """Generate Thue-Morse sequence of length n"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        # Count number of 1s in binary representation of i
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Start with Thue-Morse sequence
    h0 = thue_morse_sequence(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Random search phase (first 20% of time)
    remaining_time = budget_s * 0.2
    while time.time() - start_time < budget_s * 0.2:
        # Generate random binary solution
        indices = np.random.permutation(n_points)[:int(TARGET_SUM)]
        h = np.zeros(n_points)
        h[indices] = 1.0
        h = project_to_sum_box(h, TARGET_SUM)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Local improvement phase
    remaining_time = budget_s * 0.8
    max_iter = 50000
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Random swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i_idx = np.random.choice(ones)
            j_idx = np.random.choice(zeros)
            h_new = best_h.copy()
            h_new[i_idx] = 0.0
            h_new[j_idx] = 1.0
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Final continuous optimization
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        
        res = minimize(
            lambda x: evaluate_c5(x), 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            best_h = res.x
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    except:
        pass
        
    return (best_h.tolist(), float(best_c5), n_points)
