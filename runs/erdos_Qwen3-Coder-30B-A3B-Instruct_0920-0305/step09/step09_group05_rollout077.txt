def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_binary_individual():
    """Create a binary individual with exactly half ones"""
    h = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h[ones_indices] = 1.0
    return h

def crossover(parent1, parent2):
    """Uniform crossover that maintains sum constraint"""
    child = np.zeros(n_points)
    for i in range(n_points):
        if random.random() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    
    # Repair to maintain exactly half ones
    current_sum = np.sum(child)
    if current_sum > n_points / 2:
        # Find indices where value is 1, and set some to 0
        ones_idx = np.where(child == 1)[0]
        num_to_change = int(current_sum - n_points / 2)
        change_idx = np.random.choice(ones_idx, num_to_change, replace=False)
        child[change_idx] = 0
    elif current_sum < n_points / 2:
        # Find indices where value is 0, and set some to 1
        zeros_idx = np.where(child == 0)[0]
        num_to_change = int(n_points / 2 - current_sum)
        change_idx = np.random.choice(zeros_idx, num_to_change, replace=False)
        child[change_idx] = 1
    
    return child

def mutate(individual, mutation_rate=0.1):
    """Swap a 0 with a 1, maintaining the sum constraint"""
    if random.random() > mutation_rate:
        return individual
    
    # Find all 0s and 1s
    zeros = np.where(individual == 0)[0]
    ones = np.where(individual == 1)[0]
    
    if len(zeros) == 0 or len(ones) == 0:
        return individual
    
    # Pick a 0 and a 1 and swap
    i = random.choice(zeros)
    j = random.choice(ones)
    
    new_individual = individual.copy()
    new_individual[i] = 1
    new_individual[j] = 0
    
    return new_individual

def genetic_algorithm(budget_s):
    """Genetic algorithm for finding good binary solution"""
    start_time = time.time()
    
    population_size = 60
    num_generations = 200
    mutation_rate = 0.1
    
    # Initialize population
    population = [create_binary_individual() for _ in range(population_size)]
    
    # Evaluate fitness
    fitness = [evaluate_c5(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner].copy())
        
        # Create new population
        new_population = [best_individual.copy()]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitness = [evaluate_c5(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        current_best_fitness = fitness[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
    
    return best_individual, best_fitness

def simulated_annealing(h_start, budget_s):

    # Annealing parameters
    cooling_rate = 0.995
    min_temp = 1e-10
    
    move_count = 0
    while T > min_temp and time.time() - start_time < budget_s:
        # Perturb a random pair
        i = random.randint(0, n_points - 1)
        j = random.randint(0, n_points - 1)
        
        if current_h[i] == current_h[j]:
            # Find a 0 and 1 to swap
            zeros = np.where(current_h == 0)[0]
            ones = np.where(current_h == 1)[0]
            
            if len(zeros) > 0 and len(ones) > 0:
                i = random.choice(zeros)
                j = random.choice(ones)
            else:
                T *= cooling_rate
                continue
                
        # Create new solution
        new_h = current_h.copy()
        new_h[i] = current_h[j]
        new_h[j] = current_h[i]
        
        # Project to maintain constraint
        new_h = project_to_sum_box(new_h, n_points / 2)
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or random.random() < np.exp(-delta / T):
            current_h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        
        # Cool down
        if move_count % 10 == 0:
            T *= cooling_rate
        move_count += 1
    
    return best_h, best_c5
