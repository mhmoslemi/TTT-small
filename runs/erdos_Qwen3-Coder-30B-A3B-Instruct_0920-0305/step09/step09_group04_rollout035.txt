def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_grad(h, c_vals, eps):
    """Build gradient for smooth max approximation"""
    # Compute soft weights
    c_vals = np.array(c_vals)
    c_max = np.max(c_vals)
    exp_vals = np.exp(eps * (c_vals - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    # Build gradient of smooth max
    grad = np.zeros_like(h)
    n = len(h)
    
    for i in range(n):
        # For each k shift
        for k in range(n):
            if k == 0:
                if i < n-1:
                    grad[i] += weights[k] * (1 - h[i+1]) if i+1 < n else 0
                if i > 0:
                    grad[i] -= weights[k] * h[i-1] if i-1 >= 0 else 0
            else:
                # Positive shifts
                if i + k < n:
                    grad[i] += weights[k] * (1 - h[i+k])
                if i - k >= 0:
                    grad[i] -= weights[k] * h[i-k]
                    
    return grad

def smooth_max_objective(h, eps=1.0):

    # Compute all overlaps using FFT
    corr = compute_corr_fft(h)
    # Convert to overlaps (shifted correlation)
    c_vals = corr[len(corr)//2:]  # Take positive shifts
    # Apply smooth approximation
    c_vals = np.array(c_vals)
    c_max = np.max(c_vals)
    if c_max < -100:  # Avoid numerical issues
        return 0
    exp_vals = np.exp(eps * (c_vals - c_max))
    smooth_max = c_max + np.log(np.sum(exp_vals)) / eps
    return smooth_max
