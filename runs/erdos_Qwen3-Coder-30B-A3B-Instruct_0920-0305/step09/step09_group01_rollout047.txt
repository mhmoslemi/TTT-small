import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters 
    n_points = 150
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Phase A: Build model
    h = Binary(n_points, name='h')  # Binary variables for h_i
    
    # Helper function to get shift indices
    def get_shift_indices(shift):
        start = 0
        end = n_points - shift
        return list(range(start, end))
    
    # Create auxiliary variables y_{i,k}
    # Use dictionary to avoid storing all entries
    y = {}
    for k in range(n_points):
        shift_indices = get_shift_indices(k)
        for i in shift_indices:
            y[(i, k)] = Binary(name=f'y_{i}_{k}')
    
    # Constraints for y variables
    constraints = []
    for k in range(n_points):
        shift_indices = get_shift_indices(k)
        for i in shift_indices:
            # y_{i,k} <= h_i
            constraints.append(y[(i, k)] <= h[i])
            # y_{i,k} <= h_{i+k}
            constraints.append(y[(i, k)] <= h[i+k])
            # y_{i,k} >= h_i + h_{i+k} - 1
            constraints.append(y[(i, k)] >= h[i] + h[i+k] - 1)
    
    # Constraint for sum
    constraints.append(sum(h) == TARGET_SUM)
    
    # Create epigraph variable
    t = Variable(name='t')
    
    # Add epigraph constraints
    # For shifting k from 0 to n-1
    for k in range(n_points):
        shift_indices = get_shift_indices(k)
        sum_h_k = sum(h[i] for i in shift_indices)
        sum_y_k = sum(y[(i, k)] for i in shift_indices)
        c_k = sum_h_k - sum_y_k  # c_k = Σ h_i - Σ y_{i,k}
        constraints.append(t >= c_k)
    
    # Solve MILP
    objective = Minimize(t)
    prob = Problem(objective, constraints)
    
    # Set time limit
    prob.solve(solver=CPLEX, time_limit=budget_s/2, mip_tolerances_mipgap=0.001)
    
    # Get solution
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_solution = np.array([round(h[i].value) for i in range(n_points)])
        # Clean up
        h_solution = np.clip(h_solution, 0, 1)
    else:
        # Fallback to a random construction
        h_solution = np.zeros(n_points)
        h_solution[:int(TARGET_SUM)] = 1
        np.random.shuffle(h_solution)
    
    # Phase D: LP relaxation + rounding
    # Reset the problem for the LP relaxation
    h_lp = Variable(n_points, name='h')
    
    # Add the same constraints as before but with continuous h
    constraints_lp = []
    for k in range(n_points):
        shift_indices = get_shift_indices(k)
        for i in shift_indices:
            constraints_lp.append(y[(i, k)] <= h_lp[i])
            constraints_lp.append(y[(i, k)] <= h_lp[i+k])
            constraints_lp.append(y[(i, k)] >= h_lp[i] + h_lp[i+k] - 1)
        
    constraints_lp.append(sum(h_lp) == TARGET_SUM)
    t_lp = Variable(name='t')
    
    for k in range(n_points):
        shift_indices = get_shift_indices(k)
        sum_h_k = sum(h_lp[i] for i in shift_indices)
        sum_y_k = sum(y[(i, k)] for i in shift_indices)
        c_k = sum_h_k - sum_y_k
        constraints_lp.append(t_lp >= c_k)
    
    objective_lp = Minimize(t_lp)
    prob_lp = Problem(objective_lp, constraints_lp)
    prob_lp.solve(solver=CPLEX, time_limit=min(50, budget_s/4))
    
    # Rounding step
    if prob_lp.status in ['optimal', 'optimal_inaccurate']:
        h_frac = np.array([h_lp[i].value for i in range(n_points)])
        # Greedy rounding
        sorted_indices = np.argsort(h_frac)[::-1]
        h_round = np.zeros(n_points)
        h_round[sorted_indices[:int(TARGET_SUM)]] = 1
        h_solution = h_round
    
    # Phase E: Local search - pair swaps
    def compute_c5(h):
        return np.max(correlate(h, 1-h, mode="full")) * dx
    
    def get_shifts():
        shift_indices = []
        for k in range(n_points):
            shift_indices.append(get_shift_indices(k))
        return shift_indices
    
    # Precompute affected shifts for each index
    # Only for positive shifts since negative are symmetric
    def compute_shift_map():
        shift_map = {}
        for i in range(n_points):
            shift_map[i] = []
            for k in range(n_points):  # k = 0 to n_points-1
                if i + k < n_points:
                    shift_map[i].append(k)
        return shift_map
    
    shift_map = compute_shift_map()
    
    def apply_swap(h, i, j):
        # Apply the swap without re-computing everything
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        return h_new
    
    def calculate_c5_change(h, i, j):
        # This is a simplified version over actual full computation
        # Just estimate the reduction in C5
        current_c5 = compute_c5(h)
        h_new = apply_swap(h, i, j)
        new_c5 = compute_c5(h_new)
        return new_c5 - current_c5
    
    # Local search phase
    current_h = h_solution.copy()
    current_c5 = compute_c5(current_h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:  # Leave buffer
        improved = False
        for i in range(n_points):
            if current_h[i] != 1:
                continue
            for j in range(n_points):
                if current_h[j] != 0:
                    continue
                # Calculate change in C5 if we swap
                delta = calculate_c5_change(current_h, i, j)
                if delta < 0:  # Improves C5
                    current_h = apply_swap(current_h, i, j)
                    current_c5 = compute_c5(current_h)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return (current_h.tolist(), float(current_c5), n_points)
