import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_all_shifts(h):
    """Compute all shift values using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_all_shifts(h)
    return dx * np.max(corr)

def compute_gradient_at_shifts(h, shifts):
    """Compute gradient with respect to h at the worst shift"""
    # Find the index of maximum correlation
    max_k = np.argmax(shifts)
    
    # Create gradient vector
    grad = np.zeros_like(h)
    
    # Handle case k = max_k
    if max_k < len(h):
        # Shifts the positive direction
        shift_amount = max_k
        if shift_amount > 0:
            grad[:-shift_amount] += 1 - h[shift_amount:]
            grad[shift_amount:] -= h[:-shift_amount]
        else:
            grad[:] += 1 - h[:]
            
    # Handle negative shift case
    elif max_k >= len(h):
        # negative shift case, k = max_k - 2*(max_k - len(h)) 
        k = max_k - (2 * len(h) - 2)
        if k > 0:
            grad[:-k] += 1 - h[k:]
            grad[k:] -= h[:-k]
        else:
            grad[:] += 1 - h[:]
            
    return grad
