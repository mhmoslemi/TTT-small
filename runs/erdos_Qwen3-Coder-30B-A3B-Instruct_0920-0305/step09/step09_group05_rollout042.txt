import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp(x, axis=None):
    """Compute log(sum(exp(x))) in a numerically stable way"""
    x = np.asarray(x)
    max_x = np.max(x, axis=axis, keepdims=True)
    return max_x + np.log(np.sum(np.exp(x - max_x), axis=axis, keepdims=True))

def compute_gradient_F_alpha(h, c, alpha):
    n = len(h)
    weights = np.exp(alpha * c - np.max(alpha * c))
    weights = weights / np.sum(weights)
    
    # Compute gradient using the slicing technique
    g = np.ones(n)
    for k in range(-n+1, n):
        if k >= 0:
            g[:n-k] -= weights[k] * h[k:]
            g[k:] -= weights[k] * h[:n-k]
        else:
            s = -k
            g[s:] -= weights[k] * h[:n-s]
            g[:n-s] -= weights[k] * h[s:]
    
    return g

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    n_restarts = 5
    
    # Stage A - Simulated Annealing on binary space
    def binary_anneal():
        # Start with binary seed from initial_h_values
        h0 = np.array(initial_h_values)
        # Project to feasible set first
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        # Convert to binary
        h_bin = (h0 > 0.5).astype(int)
        
        # Ensure exactly TARGET_SUM ones
        ones_count = np.sum(h_bin)
        if ones_count > TARGET_SUM:
            # Need to flip some 1s to 0s
            flip_indices = np.where(h_bin == 1)[0]
            np.random.shuffle(flip_indices)
            for i in range(int(ones_count - TARGET_SUM)):
                h_bin[flip_indices[i]] = 0
        elif ones_count < TARGET_SUM:
            # Need to flip some 0s to 1s
            flip_indices = np.where(h_bin == 0)[0]
            np.random.shuffle(flip_indices)
            for i in range(int(TARGET_SUM - ones_count)):
                h_bin[flip_indices[i]] = 1
                
        # Evaluate initial correlation
        current_corr = compute_corr_fft(h_bin)
        current_c5 = c5_from_corr(current_corr, n_points)
        
        # Simulated annealing
        T = 0.5
        gamma = 0.995
        best_corr = current_corr.copy()
        best_c5_val = current_c5
        best_h_bin = h_bin.copy()
        
        # Allow more iterations for the binary search
        max_iter = int(0.1 * budget_s / 0.3)  # 1/3 of the budget to binary search
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Propose a swap: pick two random indices
            # One with 1, one with 0
            ones_idx = np.where(h_bin == 1)[0]
            zeros_idx = np.where(h_bin == 0)[0]
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                continue
                
            i_1 = np.random.choice(ones_idx)
            i_0 = np.random.choice(zeros_idx)
            
            # Make the swap
            h_trial = h_bin.copy()
            h_trial[i_1] = 0
            h_trial[i_0] = 1
            
            # Evaluate correlation
            trial_corr = compute_corr_fft(h_trial)
            trial_c5 = c5_from_corr(trial_corr, n_points)
            
            # Accept or reject
            delta = trial_c5 - current_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_c5 = trial_c5
                h_bin = h_trial.copy()
                current_corr = trial_corr.copy()
                
                if current_c5 < best_c5_val:
                    best_c5_val = current_c5
                    best_corr = current_corr.copy()
                    best_h_bin = h_bin.copy()
            
            # Cool down
            T *= gamma
            
        return best_h_bin, best_c5_val, best_corr
    
    # Run multiple binary annealing restarts
    for i in range(n_restarts):
        if time.time() - start_time > budget_s:
            break
        
        try:
            binary_h, binary_c5, binary_corr = binary_anneal()
            # Convert to continuous
            continuous_h = binary_h.astype(float) + np.random.uniform(-0.01, 0.01, n_points)
            continuous_h = project_to_sum_box(continuous_h, TARGET_SUM)
            
            if binary_c5 < best_c5:
                best_c5 = binary_c5
                best_h = binary_h.copy()
        except:
            continue
    
    # If we didn't get any good binary solution, use the initial one
    if best_h is None:
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        best_c5 = evaluate_c5(h0)
        best_h = h0.copy()
    
    # Stage B - Continuous Refinement with smoothed max
    h_continuous = best_h.copy()
    alpha = 2.0
    alpha_max = 30.0
    eta = 0.25
    decay_rate = 0.9995
    momentum = 0
    beta = 0.9
    iter_count = 0
    last_improvement = 0
    step_size = eta
    
    # Use Optimized Gradient Descent without Empirically Tunable Noise
    while time.time() - start_time < budget_s and iter_count < 10000:
        if time.time() - start_time > budget_s:
            break
            
        corr = compute_corr_fft(h_continuous)
        c = corr
        
        # Compute smooth approximation to max
        smoothed_max = logsumexp(alpha * c) / alpha
        
        # Compute gradient of smooth approximation
        gradient = compute_gradient_F_alpha(h_continuous, c, alpha)
        
        # Update momentum
        momentum = beta * momentum + (1 - beta) * gradient
        
        # Take a step
        trial_h = h_continuous - step_size * momentum
        
        # Project back onto feasible set
        new_h = project_to_sum_box(trial_h, TARGET_SUM)
        
        # Check if the exact value improved
        new_corr = compute_corr_fft(new_h)
        new_c5 = c5_from_corr(new_corr, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            last_improvement = iter_count
            
        # Accept step (projection guarantees feasibility)
        h_continuous = new_h.copy()
        
        # Update parameters
        iter_count += 1
        eta *= decay_rate
        alpha = min(alpha * 1.02, alpha_max)
        
        # Early stopping
        if iter_count - last_improvement > 500:
            if step_size < 1e-5:
                break
            step_size /= 2.0
            last_improvement = iter_count
        
    # Stage C - Sub-gradient polishing with true max
    while time.time() - start_time < budget_s and iter_count < 50000:
        if time.time() - start_time > budget_s:
            break
            
        corr = compute_corr_fft(best_h)
        c = corr
        max_val = np.max(c)
        active_indices = np.where(np.abs(c - max_val) < 1e-10)[0]
        
        # Uniform weights for active shifts
        weights = np.zeros_like(c)
        if len(active_indices) > 0:
            weights[active_indices] = 1.0 / len(active_indices)
        
        # Compute sub-gradient (simplified)
        sub_grad = np.ones(n_points)
        for k in range(-n_points+1, n_points):
            if k >= 0:
                sub_grad[:n_points-k] -= weights[k] * best_h[k:]
                sub_grad[k:] -= weights[k] * best_h[:n_points-k]
            else:
                s = -k
                sub_grad[s:] -= weights[k] * best_h[:n_points-s]
                sub_grad[:n_points-s] -= weights[k] * best_h[s:]
        
        # Take small steps
        small_eta = 0.02
        trial_h = best_h - small_eta * sub_grad
        new_h = project_to_sum_box(trial_h, TARGET_SUM)
        
        new_corr = compute_corr_fft(new_h)
        new_c5 = c5_from_corr(new_corr, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        iter_count += 1
    
    # Final polishing with scipy minimize 
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                final_h = project_to_sum_box(res.x, TARGET_SUM)
                final_c5 = evaluate_c5(final_h)
                
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h.copy()
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
