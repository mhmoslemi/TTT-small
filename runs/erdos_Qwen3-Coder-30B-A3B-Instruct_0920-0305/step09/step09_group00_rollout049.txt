import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full autocorrelation using FFT"""
    n = len(h)
    H = fft(h, 2*n - 1)
    ac = ifft(H * np.conj(H)).real[:n]
    return ac

def softmax_weights(g, alpha):
    """Compute soft-max weights for given values and temperature"""
    g_max = np.max(g)
    g_stable = g - g_max
    exp_g = np.exp(alpha * g_stable)
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of smooth surrogate using convolutions"""
    n = len(h)
    # Two convolutions: p * h and p_rev * h
    p_rev = p[::-1]
    
    # Pad for linear convolution
    pad_len = 2*n - 1
    h_padded = np.pad(h, (0, pad_len - n), mode='constant')
    p_padded = np.pad(p, (0, pad_len - len(p)), mode='constant')
    p_rev_padded = np.pad(p_rev, (0, pad_len - len(p_rev)), mode='constant')
    
    # Linear convolutions using FFT
    conv1 = ifft(fft(h_padded) * fft(p_padded)).real[:n]
    conv2 = ifft(fft(h_padded) * fft(p_rev_padded)).real[:n]
    
    return -(conv1 + conv2)

def project_onto_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    
    # Binary search for lambda
    def f(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search with tolerance
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def c5_true(h):
    """Compute true C5 using correlation"""
    n = len(h)
    dx = 2.0 / n
    # Compute with complement (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = dx * np.max(corr)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n = 128
    dx = 2.0 / n
    target_sum = n / 2
    
    # Create initial pattern
    # Try Thue-Morse pattern mixed with random binary patterns
    h_tm = np.array([bin(i).count('1') % 2 for i in range(n)])
    h_tm = h_tm.astype(float) * (n / (2 * np.sum(h_tm)) if np.sum(h_tm) != 0 else 1)
    
    # Evaluate all starts
    best_c5 = float('inf')
    best_h = None
    
    # Generate several random binary patterns and select the best
    for i in range(10):
        h_rand = np.random.binomial(1, 0.5, n).astype(float)
        h_rand = h_rand * (target_sum / np.sum(h_rand))
        h_rand = project_onto_sum_box(h_rand, target_sum)
        c5_val = c5_true(h_rand)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_rand.copy()
    
    # If Thue-Morse is better, use it
    c5_tm = c5_true(h_tm)
    if c5_tm < best_c5:
        best_c5 = c5_tm
        best_h = h_tm.copy()
    
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Multi-resolution coarse-to-fine optimization
    resolutions = [64, 128, 256]
    h_current = None
    
    for res in resolutions:
        if time.time() - start_time > budget_s * 0.8:
            break
            
        n_res = res
        dx_res = 2.0 / n_res
        target_sum_res = n_res / 2
        
        # Interpolate if necessary
        if res == 64:
            h_current = final_h.copy()
        elif res == 128:
            # Linear interpolation from 64 to 128
            h_current = np.interp(np.linspace(0, 2, n_res), 
                                 np.linspace(0, 2, len(final_h)), 
                                 final_h)
            h_current = h_current * (target_sum_res / np.sum(h_current))
            h_current = project_onto_sum_box(h_current, target_sum_res)
        elif res == 256:
            # Linear interpolation from 128 to 256
            h_current = np.interp(np.linspace(0, 2, n_res), 
                                 np.linspace(0, 2, len(final_h)), 
                                 final_h)
            h_current = h_current * (target_sum_res / np.sum(h_current))
            h_current = project_onto_sum_box(h_current, target_sum_res)
            
        # Deterministic annealing optimization
        max_iter_per_alpha = 200
        alpha = 1.0
        alpha_max = 2000.0
        gamma = 1.3
        eta = 0.001
        
        for alpha_iter in range(50):  # Limited number of alphas for budget
            if time.time() - start_time > budget_s * 0.95:
                break
                
            for it in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s:
                    break
                    
                # Compute autocorrelations
                auto_corr = autocorr_fft(h_current)
                g = -auto_corr  # negative because we want to maximize auto-correlation

                # Compute softmax weights
                p = softmax_weights(g, alpha)
                
                # Compute gradient
                grad = gradient(h_current, p)
                
                # Update h
                h_new = h_current - eta * grad
                
                # Project onto constraints
                h_new = project_onto_sum_box(h_new, target_sum_res)
                
                # Accept if better
                new_c5 = c5_true(h_new)
                old_c5 = c5_true(h_current)
                
                if new_c5 < old_c5 or np.random.random() < 0.1:
                    h_current = h_new.copy()
                    if new_c5 < final_c5:
                        final_c5 = new_c5
                        final_h = h_current.copy()
            
            # Increase temperature
            alpha *= gamma
            if alpha > alpha_max:
                alpha = alpha_max
            # Decrease step size slightly
            eta *= 0.99
        
        # Reset the final vectors for next resolution
        if res != 256:
            final_h = h_current.copy()
            final_c5 = c5_true(h_current)
    
    # Post-processing discrete hill climb
    best_h = final_h.copy()
    best_c5 = c5_true(best_h)
    
    # Make a few discrete swaps to slightly improve the solution
    swap_trials = 10000
    for _ in range(swap_trials):
        if time.time() - start_time > budget_s:
            break
            
        # Convert to binary
        binary_h = (best_h > 0.5).astype(int)
        ones = np.where(binary_h == 1)[0]
        zeros = np.where(binary_h == 0)[0]
        
        if len(ones) > 0 and len(zeros) > 0:
            # Pick two random positions to swap: one '1' and one '0'
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Make a copy and swap
            new_h = best_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
            
            # Project down to valid constraints
            new_h = project_onto_sum_box(new_h, target_sum)
            
            # Check new C5
            new_c5 = c5_true(new_h)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5

    # Final polishing with SLSQP
    try:
        # Define objective function that is differentiable
        def smooth_obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx  # This is similar to C5 but not exactly the same
        
        def constraint_eq(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Small perturbation to make it a good initial point
        x0 = best_h.copy() + np.random.normal(0, 0.001, n)
        x0 = project_onto_sum_box(x0, target_sum)
        
        res = minimize(
            lambda x: c5_true(x),
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if res.success:
            final_h = res.x
            final_c5 = c5_true(final_h)
    except:
        pass
    
    return (final_h.tolist(), float(final_c5), n)
