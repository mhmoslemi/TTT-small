import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5_exact(h, dx):
    """Compute exact C5 value using correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

    z_constraints = []
    
    # Add McCormick envelopes for z_{ij} = h_i * h_j
    for i in range(n_points):
        for j in range(i+1, n_points):
            z_name = f"z_{i}_{j}"
            z_vars[(i,j)] = cp.Variable(1, name=z_name)
            z_constraints.extend([
                z_vars[(i,j)] <= h[i],
                z_vars[(i,j)] <= h[j],
                z_vars[(i,j)] >= h[i] + h[j] - 1,
                z_vars[(i,j)] >= 0
            ])
    
    # Mass constraint
    mass_constraint = cp.Equality(cp.sum(h), TARGET_SUM)
    
    # Overlap constraints for each shift k = 1, ..., n-1
    overlap_constraints = []

        # Compute sum_{i=0}^{n-1-k} (h_{i+k} - z_{i,i+k})

        expr = 0

            expr += h[i+k] - z_vars[(i, i+k)]
        overlap_constraints.append(expr <= t)

    objective = cp.Minimize(t)
    
    # Constraints
    all_constraints = [h >= 0, h <= 1] + z_constraints + [mass_constraint] + overlap_constraints

    prob = cp.Problem(objective, all_constraints)

                sol_t = t.value
                sol_h = h.value
                if sol_t < best_t:
                    best_t = sol_t

    def round_h(h_vec):
        """Round h_vec to 0/1 vector maintaining sum constraint"""
        h_copy = h_vec.copy()
