def build_shift_matrices(n):
    """Build all shift matrices for the quadratic terms."""
    M_list = []
    for k in range(n):
        M = np.zeros((n, n))
        for i in range(n - k):
            M[i, i + k] = 1
            M[i + k, i] = 1
        M_list.append(M)
    return M_list

def evaluate_c5_sdp(h):
    """Compute C5 on a vector h using fast FFT."""
    corr = np.correlate(h, 1 - h, mode='full')
    result = np.max(corr) * dx
    return result

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
