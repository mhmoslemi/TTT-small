import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    n = len(h)
    H = fft(h, 2*n)
    A = ifft(H * np.conj(H)).real[:n]
    return A

def softmax_weights(g, alpha):
    """Compute softmax weights from g values with temperature alpha"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of smooth surrogate using convolutions"""
    n = len(h)
    # Two convolutions: p * h and p_rev * h
    p_extended = np.concatenate([p, np.zeros(n)])
    h_extended = np.concatenate([h, np.zeros(n)])
    conv1 = ifft(fft(p_extended) * fft(h_extended)).real[:n]
    conv2 = ifft(fft(p_extended[::-1]) * fft(h_extended[::-1])).real[:n]
    return -(conv1 + conv2)

def project_onto_sum_box(x, target):
    """Euclidean projection onto box constraints with sum constraint (water filling)"""
    x_clipped = np.clip(x, 0, 1)
    def sum_func(lam):
        return np.sum(np.clip(x_clipped + lam, 0, 1)) - target
    
    a, b = -1.0, 1.0
    while sum_func(a) * sum_func(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(30):
        m = (a + b) / 2
        fm = sum_func(m)
        if abs(fm) < 1e-10:
            break
        if fm * sum_func(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x_clipped + (a+b)/2, 0, 1)

def c5_true(h):
    """Compute exact C5 using full correlation"""
    n = len(h)
    dx = 2.0 / n
    h_comp = 1 - h
    corr = np.correlate(h, h_comp, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_list = [64, 128, 256]  # Multi-resolution approach
    max_iter_per_alpha = 200
    alpha_start = 1.0
    alpha_max = 2000.0
    gamma = 1.3
    eta = 0.1
    target_sum = None
    
    best_h = None
    best_c5 = float('inf')
    
    for n in n_list:
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        
        # Generate starting points
        candidates = []
        # Try Thue-Morse pattern
        h_tm = np.array([int(bin(i).count('1') % 2 == 0) for i in range(n)])
        h_tm = h_tm * (target_sum / np.sum(h_tm))
        candidates.append(h_tm)
        
        # Try random binary patterns
        for _ in range(5):
            h_rand = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
            h_rand = h_rand * (target_sum / np.sum(h_rand))
            candidates.append(h_rand)
            
        # Evaluate all candidates
        best_candidate = None
        min_surrogate = float('inf')
        for h in candidates:
            h = project_onto_sum_box(h, target_sum)
            A = autocorr_fft(h)
            g = -A
            surrogate_val = np.log(np.sum(np.exp(g))) / np.log(np.sum(np.exp(g))) if np.sum(np.exp(g)) > 0 else 0  # Simplified: only first term
            if surrogate_val < min_surrogate:
                min_surrogate = surrogate_val
                best_candidate = h.copy()
        
        h_start = best_candidate
        
        # Deterministic annealing
        alpha = alpha_start
        h = h_start.copy()
        
        while alpha <= alpha_max and time.time() - start_time < budget_s - 10:
            for _ in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s - 10:
                    break
                    
                A = autocorr_fft(h)
                g = -A
                p = softmax_weights(g, alpha)
                grad = gradient(h, p)
                h_new = h - eta * grad
                h_new = project_onto_sum_box(h_new, target_sum)
                
                # Update if improvement
                A_new = autocorr_fft(h_new)
                g_new = -A_new
                surrogate_new = np.log(np.sum(np.exp(alpha * g_new))) / alpha
                
                if surrogate_new < np.log(np.sum(np.exp(alpha * g))) / alpha:
                    h = h_new
                    if c5_true(h) < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_true(h)
            
            alpha *= gamma
            eta *= 0.9
        
        # Interpolation if needed
        if n == 64:
            h_interpolated = np.interp(np.linspace(0, 2, 128), np.linspace(0, 2, 64), h)
            h = project_onto_sum_box(h_interpolated, target_sum)
        elif n == 128:
            h_interpolated = np.interp(np.linspace(0, 2, 256), np.linspace(0, 2, 128), h)
            h = project_onto_sum_box(h_interpolated, target_sum)
    
    # Final discrete hill climbing
    if best_h is not None:
        h_final = best_h.copy()
        max_swaps = 10000
        for _ in range(max_swaps):
            if time.time() - start_time > budget_s - 10:
                break
                
            i = np.random.randint(0, len(h_final))
            j = np.random.randint(0, len(h_final))
            
            if h_final[i] > 0.5 and h_final[j] < 0.5:
                h_trial = h_final.copy()
                h_trial[i] = 1 - h_trial[i]
                h_trial[j] = 1 - h_trial[j]
                
                c5_trial = c5_true(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    h_final = h_trial
        # Final optimization with scipy
        def objective(h_array):
            return c5_true(h_array)
        
        def constraint_eq(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(len(best_h))]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                final_h = res.x
            else:
                final_h = best_h
        except:
            final_h = best_h
            
        final_h = project_onto_sum_box(final_h, target_sum)
        final_c5 = c5_true(final_h)
        
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    return (best_h.tolist(), float(best_c5), len(best_h))
