# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_all_shifts(h):
    """Compute all correlation values using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_all_shifts(h)
    return dx * np.max(corr)

def compute_gradient(h, shifts, mu):
    """Compute gradient with respect to current shifts"""
    n = len(h)
    grad = np.zeros(n)
    
    # Normalize shifts for numerical stability
    max_shift = np.max(shifts)
    if max_shift > -np.inf:
        exp_shifts = np.exp((shifts - max_shift) / mu)
    else:
        exp_shifts = np.ones_like(shifts)
    
    # Compute weights
    w = exp_shifts / np.sum(exp_shifts)
    
    # Compute weighted gradient
    for k in range(len(shifts)):
        if k < n:
            # Positive shift case
            if k == 0:
                grad[:n] += w[k] * (1 - h[k:])
            else:
                grad[:n-k] += w[k] * (1 - h[k:])
                grad[k:] -= w[k] * h[:n-k]
    
    return grad

def line_search(h, g, mu):
    """Perform line search with adaptive step size"""
    alpha0 = 0.2 * dx
    alpha = alpha0
    
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)

        h_proposed = project_to_sum_box(h_current - alpha * g, TARGET_SUM)
        c5_proposed = evaluate_c5(h_proposed)
        
        if c5_proposed < c5_current:
            return h_proposed, c5_proposed
        
        alpha /= 2.0
    
    return h_current, c5_current
