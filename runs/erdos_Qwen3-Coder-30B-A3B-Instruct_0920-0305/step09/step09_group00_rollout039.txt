def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    L = 2 * len(h) - 1
    H = fft(h, L)
    A = ifft(H * np.conj(H)).real
    return A

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    return exp_g / np.sum(exp_g)

def gradient(h, p):
    """Compute gradient of the surrogate objective"""
    # Compute convolutions via FFT
    L = 2 * len(h) - 1
    h_fft = fft(h, L)
    p_fft = fft(p, L)
    
    # Convolutions: p * h and p_rev * h
    r1 = ifft(p_fft * h_fft).real
    r2 = ifft(np.conj(p_fft) * h_fft).real
    
    # The gradient is -(r1 + r2)
    return -(r1 + r2)

def project_onto_sum_box(x, target_sum):
    """Project onto box constraints [0,1] with sum = target_sum using water-filling"""

    x_clipped = np.clip(x, 0, 1)
    
    # Binary search for lambda
    def sum_with_lambda(lam):
        proj = np.clip(x_clipped + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1

        if abs(fm) < tol:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    lambda_opt = (a + b) / 2
    result = np.clip(x_clipped + lambda_opt, 0, 1)
    return result

    corr = np.correlate(h, 1 - h, mode='full')
    # Max correlation
    max_corr = np.max(corr)
    return max_corr * dx
