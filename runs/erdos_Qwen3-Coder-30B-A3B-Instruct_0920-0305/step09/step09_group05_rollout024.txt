import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Initialize with a better construction
    alpha = 0.2
    beta = 0.8
    h0 = np.zeros(n_points)
    # Add a few smooth peaks and valleys
    for i in range(n_points):
        x = i * dx
        if 0.3 <= x <= 0.7:
            h0[i] = 1.0
        elif 0.1 <= x <= 0.3:
            h0[i] = 1.0 - (x - 0.1) / 0.2
        elif 0.7 <= x <= 0.9:
            h0[i] = 1.0 - (0.9 - x) / 0.2
        else:
            h0[i] = 0.0
    
    h0 = project_to_sum_box(h0, TARGET_SUM)
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # More aggressive optimization using L-BFGS-B
    max_iter = 0
    while time.time() - start_time < budget_s and max_iter < 10000:
        # Try optimization with L-BFGS-B for better convergence
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                current_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                optimized_h = res.x
                optimized_h = project_to_sum_box(optimized_h, TARGET_SUM)
                optimized_c5 = evaluate_c5(optimized_h)
                
                if optimized_c5 < best_c5:
                    best_c5 = optimized_c5
                    best_h = optimized_h.copy()
                    current_h = optimized_h.copy()
                    current_c5 = optimized_c5
        except:
            pass
        
        # Local search with larger steps
        step_size = 0.02
        for _ in range(20):
            if time.time() - start_time >= budget_s:
                break
                
            # Try perturbing multiple points at once
            indices = np.random.choice(n_points, size=min(5, n_points//10), replace=False)
            perturbation = np.random.normal(0, step_size, len(indices))
            
            h_test = current_h.copy()
            for i, delta in zip(indices, perturbation):
                h_test[i] += delta
                h_test[i] = np.clip(h_test[i], 0, 1)
                
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            test_c5 = evaluate_c5(h_test)
            
            if test_c5 < current_c5:
                current_h = h_test.copy()
                current_c5 = test_c5
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_h = current_h.copy()
        
        # Random jump to escape local minima
        if np.random.random() < 0.1:
            rand_h = np.random.uniform(0, 1, n_points)
            rand_h = project_to_sum_box(rand_h, TARGET_SUM)
            current_h = rand_h.copy()
            current_c5 = evaluate_c5(rand_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        max_iter += 1
    
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
