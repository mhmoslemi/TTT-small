import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp(x, axis=None):
    """Numerically stable logsumexp"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def compute_smoothed_max_gradient(h, alpha):
    """Compute gradient of the smoothed max"""
    corr = compute_corr_fft(h)
    c = corr
    # Weight computation
    c_scaled = alpha * c
    c_max = np.max(c_scaled)
    w_unnorm = np.exp(c_scaled - c_max)
    w = w_unnorm / np.sum(w_unnorm)
    
    # Gradient computation via sliding window
    g = np.ones_like(h)
    for k in range(len(h)):
        if k == 0:
            # No shift
            g -= w[k] * h
        else:
            if k > 0:
                # Positive shift h[i+k]
                g[:len(h)-k] -= w[k] * h[k:]
                # Negative shift h[i-k] 
                g[k:] -= w[k] * h[:len(h)-k]
            else:
                # Negative shift should not happen for our construction
                # But handle it just in case
                k_abs = -k
                g[k_abs:] -= w[k] * h[:len(h)-k_abs]
                g[:len(h)-k_abs] -= w[k] * h[k_abs:]
    
    return g, c

    sub_eta = 0.02

        if time.time() - start_time > budget_s:
