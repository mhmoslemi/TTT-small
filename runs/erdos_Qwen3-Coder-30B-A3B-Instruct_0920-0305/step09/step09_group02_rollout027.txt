```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * 2.0 / len(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * target_sum / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Apply symmetric and monotonicity constraints to reduce variable space
    # h_i = h_{n-1-i}, h_i >= h_{i+1} for i=0..n/2
    # This reduces full 120 variables to ~60 independent (symmetric pairs) + monotonicity
    
    # Use integer programming approach using cvxpy to model exact problem
    # Variables: binary h_i (0 or 1), auxiliary y_{i,k} representing h_i * h_{i+k}.
    # Minimize maximum overlap over all shifts.
    
    # Define problem using cvxpy with binary optimization
    # This is an alternative approach to search-based methods
    
    # Restrict to even simpler symmetric monotone structure
    # Here we enforce the structure: h is symmetric and decreasing
    h_sym = np.array([0.5] * (n_points // 2) + [0.5] * (n_points // 2))
    h_sym[:n_points//2] = np.linspace(1, 0, n_points//2)
    h_sym[n_points//2:] = h_sym[:n_points//2][::-1]
    h_sym = h_sym * target_sum / np.sum(h_sym)
    
    # Local refinement with gradient-based method:
    def gradient_step(h, step_size=0.01):
        # Evaluate current overlaps and their gradients
        corr = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        # Create small changes in direction of most constraining overlap
        h_new = h.copy()
        shift = max_idx - (len(h) - 1)
        if shift >= 0:
            # backward shift
            for i in range(shift, len(h)):
                h_new[i] = max(0, min(1, h_new[i] - step_size * (1 - h[i-shift])))
        else:
            # forward shift
            for i in range(len(h) + shift):
                h_new[i] = max(0, min(1, h_new[i] - step_size * (1 - h[i-shift])))
        h_new = h_new * target_sum / np.sum(h_new)
        return h_new
    
    h_refined = h_sym.copy()
    c5_refined = evaluate_c5(h_refined)
    
    max_iter = 500
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        h_next = gradient_step(h_refined)
        c5_next = evaluate_c5(h_next)
        if c5_next < c5_refined:
            h_refined = h_next
            c5_refined = c5_next
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    
    return (best_h.tolist(), float(best_c5), n_points)
```
