import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, shift_ℓ):
    """Compute gradient of correlation at given shift"""
    n = len(h)
    gradient = np.zeros(n)
    
    # Compute partial derivatives for C_ℓ = sum h_i (1 - h_{i+ℓ})
    if shift_ℓ >= 0:
        # For shift ℓ ≥ 0
        # First term: sum h_i, derivative is 1 for first region
        # Second term: -sum h_i h_{i+ℓ} 
        # Let's compute gradient for this shift
        # For i where i+ℓ < n (overlap indices for first term)
        if shift_ℓ < n:
            gradient[:n-shift_ℓ] += 1 - h[shift_ℓ:]
        # For i where i-ℓ ≥ 0 (second term)
        if shift_ℓ > 0:
            gradient[shift_ℓ:] -= h[:n-shift_ℓ]
    else:
        # For shift ℓ < 0
        shift_abs = -shift_ℓ
        # Equivalent to shift ℓ with positive sign but different indexing
        if shift_abs < n:
            gradient[shift_abs:] += 1 - h[:n-shift_abs]
        if shift_abs > 0:
            gradient[:n-shift_abs] -= h[shift_abs:]
    
    return gradient
