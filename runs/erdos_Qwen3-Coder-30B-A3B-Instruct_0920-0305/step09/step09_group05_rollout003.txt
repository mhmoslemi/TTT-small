import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp(x, axis=None):
    # Compute log(sum(exp(x))) in a numerically stable way
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def compute_gradient_smoothed(h, c, alpha):
    """Compute gradient of the smoothed max function F_alpha(h)"""
    n = len(h)
    # Compute weights
    c_alpha = alpha * c
    c_alpha -= np.max(c_alpha)  # For numerical stability
    w = np.exp(c_alpha)
    w /= np.sum(w)
    
    # Compute gradient
    g = np.ones(n)
    for k in range(-(n-1), n):
        if k >= 0:
            g[:n-k] -= w[k] * h[k:]
            g[k:] -= w[k] * h[:n-k]
        else:
            s = -k
            g[s:] -= w[k] * h[:n-s]
            g[:n-s] -= w[k] * h[s:]
    
    return g
