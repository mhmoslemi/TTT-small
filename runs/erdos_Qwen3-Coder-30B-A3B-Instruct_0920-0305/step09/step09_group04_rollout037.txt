import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5_exact(h):
    """Evaluate exact C5 using correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def solve_lp_relaxation(n_points):
    """Solve the linear programming relaxation"""
    # Create variables
    h = cp.Variable(n_points, name="h")
    t = cp.Variable(1, name="t")
    
    # Create z variables for McCormick envelopes (upper triangular part only)
    z = {}
    for i in range(n_points):
        for j in range(i+1, n_points):
            z[(i,j)] = cp.Variable(1, name=f"z_{i}_{j}")
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # McCormick envelopes for all z_ij
    for i in range(n_points):
        for j in range(i+1, n_points):
            constraints.append(z[(i,j)] <= h[i])
            constraints.append(z[(i,j)] <= h[j])
            constraints.append(z[(i,j)] >= h[i] + h[j] - 1)
            constraints.append(z[(i,j)] >= 0)
    
    # Overlap constraints
    dx = 2.0 / n_points
    for k in range(1, n_points):
        # Compute sum_{i=0}^{n-1-k} (h_{i+k} - z_{i,i+k})
        expr = 0
        for i in range(n_points - k):
            expr += h[i + k] - z[(i, i + k)]
        constraints.append(expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with appropriate solver
    prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-7, reltol=1e-7)
    
    return h.value, t.value

            prob = h_frac[i]

            if remaining_ones >= (n_points - i - 1) and remaining_ones > 0:

                remaining_ones -= 1
            elif remaining_ones == 0:
                pass
            else:
                # Draw based on probability
                if np.random.random() < prob:
                    if remaining_ones > 0:
                        h_bin[i] = 1
                        remaining_ones -= 1
                else:
                    h_bin[i] = 0
        
        # Ensure exact sum
        current_ones = np.sum(h_bin)
        diff = int(current_ones - n_points / 2)
        if diff > 0:
            # Remove extra 1s
            ones_indices = np.where(h_bin == 1)[0]
            to_remove = np.random.choice(ones_indices, diff, replace=False)
            for idx in to_remove:
                h_bin[idx] = 0
        elif diff < 0:
            # Add missing 1s
            zeros_indices = np.where(h_bin == 0)[0]
            to_add = np.random.choice(zeros_indices, -diff, replace=False)
            for idx in to_add:
                h_bin[idx] = 1
        
        # Compute exact C5
        c5 = evaluate_c5_exact(h_bin)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
    
    return best_h, best_c5
