def compute_overlap_matrix(n):
    """Compute the overlap matrices M_l for l = 0 to n-1."""
    M_list = []
    for l in range(n):
        M = np.zeros((n, n))
        for i in range(n):
            j = i + l
            if j < n:
                M[i, j] = 1
                M[j, i] = 1
        M_list.append(M)
    return M_list

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT, but we also implement the matrix version."""
    n = len(h)
    M_list = compute_overlap_matrix(n)
    max_overlap = 0
    for M in M_list:
        overlap = h.T @ M @ h
        max_overlap = max(max_overlap, overlap)
    return 0.5 - max_overlap / n  # Since c_l = 0.5 - overlap_l/n

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    """Solve SDP relaxation for a given feasible solution h0."""

    h = cp.Variable(n)

    constraints = [
        X == h @ h.T,
        cp.diag(X) == h,  # X_{ii} = h_i for 0 <= h_i <= 1

        for _ in range(20):

            constraints_copy = constraints[:]
            
            # Add overlap constraints
            for l in range(n):
                overlap_expr = cp.trace(M_list[l] @ X)
                constraints_copy.append(overlap_expr >= n/2 - alpha)
            
            # Solve relaxation
            objective = cp.Minimize(cp.sum(h))  # Dummy objective
            prob = cp.Problem(objective, constraints_copy)
            try:
                prob.solve(solver=SCS, verbose=False)
                if prob.status == cp.OPTIMAL:

                        if c5_val < best_c5:
                            best_h = h_sol.copy()
                            best_c5 = c5_val
                        alpha_max = alpha  # Tighten upper bound
                    else:
                        alpha_min = alpha  # No solution at this alpha, adjust
                else:
                    alpha_min = alpha  # Failed to solve, increase lower bound
            except:
                alpha_min = alpha  # Error in solving, increase lower bound

    return best_h, best_c5
