import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_grad_c_k(h, k):
    """Compute gradient of c_k with respect to h"""
    grad = np.zeros_like(h)
    if k >= 0:
        if k < len(h):
            grad[:len(h)-k] += 1 - h[k:]
            grad[k:] -= h[:len(h)-k]
    else:
        k = -k
        if k < len(h):
            grad[:len(h)-k] += 1 - h[k:]
            grad[k:] -= h[:len(h)-k]
    return grad

def compute_all_shifts_and_weights(h, mu):
    """Compute all c_k and associated weights for the soft-max"""
    corr = compute_corr_fft(h)
    c_vals = corr * dx
    
    # Numerical stability: subtract maximum
    c_vals_shifted = c_vals - np.max(c_vals)
    exp_vals = np.exp(c_vals_shifted / mu)
    weights = exp_vals / np.sum(exp_vals)
    
    return c_vals, weights

def get_worst_shift(h):
    """Get the index of the worst shift for current h"""
    corr = compute_corr_fft(h)
    max_idx = np.argmax(corr)
    # Convert index to actual shift value
    # corr has length 2*n - 1, centered at index n-1
    worst_shift = max_idx - (n_points - 1)
    return worst_shift

def projected_gradient_step(h, mu, alpha=0.2*dx, max_backtrack=5):
    """Perform one gradient descent step with line search"""
    # Compute all shift values and weights
    c_vals, weights = compute_all_shifts_and_weights(h, mu)
    
    # Assemble gradient
    grad = np.zeros_like(h)
    for k in range(-n_points+1, n_points):
        if k >= 0:
            if k < len(h):
                grad[:len(h)-k] += weights[k+n_points-1] * (1 - h[k:])
                grad[k:] -= weights[k+n_points-1] * h[:len(h)-k]
        else:
            k_neg = -k
            if k_neg < len(h):
                grad[:len(h)-k_neg] += weights[k+n_points-1] * (1 - h[k_neg:])
                grad[k_neg:] -= weights[k+n_points-1] * h[:len(h)-k_neg]
    
    # Line search
    for _ in range(max_backtrack+1):
        h_new = h - alpha * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old or _ == max_backtrack:
            return h_new, c5_new
        alpha *= 0.5
    
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize problem
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = float('inf')
    
    # Get initializations
    initial_h_list = get_initial_h()
    
    # Run optimization from each initialization
    for h0 in initial_h_list:
        h = h0.copy()
        h = project_to_sum_box(h, TARGET_SUM)
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Continue with projected gradient descent
        mu = 0.05  # Initial smoothing parameter
        mu_min = 5e-4
        inner_iter = 0
        restart_count = 0
        
        while mu >= mu_min and (time.time() - start_time) < budget_s - 2:
            # Inner loop
            iter_since_update = 0
            while iter_since_update < 250 and (time.time() - start_time) < budget_s - 2:
                inner_iter += 1
                iter_since_update += 1
                
                h_old = h.copy()
                c5_old = current_c5
                
                # Gradient step
                h, current_c5 = projected_gradient_step(h, mu)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    iter_since_update = 0  # Reset counter on improvement
                else:
                    # Check if improvement is merely numerical noise
                    if abs(current_c5 - c5_old) < 1e-10:
                        h = h_old  # Revert
                        current_c5 = c5_old
                
                # Occasionally add jitter
                if inner_iter % 500 == 0:

                    current_c5 = evaluate_c5(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                
                # Add escape from stagnation every 1000 steps
                if iter_since_update > 1000 and np.random.random() < 0.1:
                    # Add random jitter
                    indices = np.random.choice(n_points, size=5, replace=False)
                    for idx in indices:
                        h[idx] += np.random.normal(0, 0.02)
                    h = project_to_sum_box(h, TARGET_SUM)
                    current_c5 = evaluate_c5(h)

                        best_c5 = current_c5
                        best_h = h.copy()
                        iter_since_update = 0
            
            # Reduce mu
            mu /= 2
            
            # Check if we returned to the same mu in previous two iterations

    # Final polishing with pairwise swaps
    if best_h is not None:
        h = best_h.copy()
        current_c5 = evaluate_c5(h)
        
        # Pairwise swaps
        for _ in range(10000):
            if (time.time() - start_time) > budget_s - 2:
                break
                
            improvement = False
            
            # Try all pairs
            for i in range(n_points):
                if h[i] > 1e-4:  # Non-zero entry
                    for j in range(n_points):
                        if h[j] < 1 - 1e-4:  # Non-full entry
                            # Compute directional derivative
                            delta = min(h[i], 1 - h[j]) * 0.01
                            h_new = h.copy()
                            h_new[i] -= delta
                            h_new[j] += delta
                            h_new = project_to_sum_box(h_new, TARGET_SUM)
                            
                            c5_new = evaluate_c5(h_new)
                            if c5_new < current_c5:
                                h = h_new
                                current_c5 = c5_new
                                improvement = True
                                break
                    if improvement:
                        break
        
        best_c5 = current_c5
        best_h = h
    
    return (best_h.tolist(), float(best_c5), n_points)
