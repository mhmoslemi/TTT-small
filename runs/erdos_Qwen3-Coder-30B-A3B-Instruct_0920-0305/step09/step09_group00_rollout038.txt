```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, 2 * len(h) - 1)
    corr = ifft(H * np.conj(H)).real
    return corr[:len(h)]

def softmax_weights(g, alpha):
    """Compute softmax weights for given values and temperature alpha"""
    g_shifted = g - np.max(g)  # For numerical stability
    exp_g = np.exp(alpha * g_shifted)
    return exp_g / np.sum(exp_g)

def gradient(h, p):
    """Compute gradient of the surrogate objective"""
    # Compute two convolutions
    # p * h (linear convolution)
    p_padded = np.concatenate([p, np.zeros(len(h) - 1)])
    h_padded = np.concatenate([h, np.zeros(len(p) - 1)])
    conv1 = ifft(fft(p_padded) * fft(h_padded)).real[:len(h)]
    
    # p_rev * h (linear convolution)
    p_rev = p[::-1]
    p_rev_padded = np.concatenate([p_rev, np.zeros(len(h) - 1)])
    conv2 = ifft(fft(p_rev_padded) * fft(h_padded)).real[:len(h)]
    
    return -(conv1 + conv2)

def project_onto_sum_box(x, target_sum):
    """Euclidean projection onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm
    clipped = np.clip(x, 0, 1)
    current_sum = np.sum(clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return clipped
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    tolerance = 1e-12
    
    def compute_sum(lam):
        projected = np.clip(clipped + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    while abs(compute_sum(a) * compute_sum(b)) > tolerance:
        a -= 0.01
        b += 0.01
    
    for _ in range(50):
        m = (a + b) / 2
        fm = compute_sum(m)
        if abs(fm) < tolerance:
            break
        if fm * compute_sum(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(clipped + (a + b) / 2, 0, 1)
    return result

def c5_true(h):
    """Compute exact C5 using full correlation"""
    corr_full = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr_full)

def create_binary_pattern(n):
    """Create a Thue-Morse like binary pattern"""
    pattern = []
    for i in range(n):
        bits = bin(i)[2:].zfill(10)  # 10 bits
        ones = sum(int(b) for b in bits)
        pattern.append(1 if ones % 2 == 0 else 0)
    return np.array(pattern, dtype=float)

def swap_improvement(h, max_iterations=5000):
    """Local search improvement by swapping elements"""
    current_h = h.copy()
    current_c5 = c5_true(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for _ in range(max_iterations):
        # Create integer pattern
        int_pattern = (current_h > 0.5).astype(int)
        
        # Find ones and zeros
        ones_indices = np.where(int_pattern == 1)[0]
        zeros_indices = np.where(int_pattern == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        # Choose random swap candidate
        i = np.random.choice(ones_indices)
        j = np.random.choice(zeros_indices)
        
        # Perform swap
        new_h = current_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        
        new_c5 = c5_true(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Create multiple initializations
    initial_h_list = []
    
    # Thue-Morse pattern
    h_tm = create_binary_pattern(n_points)
    h_tm = project_onto_sum_box(h_tm, TARGET_SUM)
    initial_h_list.append(h_tm)
    
    # Random binary patterns
    for _ in range(3):
        h_rand = np.random.randint(0, 2, n_points).astype(float)
        h_rand = project_onto_sum_box(h_rand, TARGET_SUM)
        initial_h_list.append(h_rand)
    
    # Choose the best initial solution
    for h in initial_h_list:
        if c5_true(h) < best_c5:
            best_c5 = c5_true(h)
            best_h = h.copy()
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_h = project_onto_sum_box(best_h, TARGET_SUM)
        
    # Coarse resolution optimization (n=64)
    n_coarse = 64
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), best_h)
    h_coarse = project_onto_sum_box(h_coarse, n_coarse / 2)
    
    # Deterministic annealing for coarse resolution
    alpha_start = 0.1
    alpha_end = 2000.0
    gamma = 1.3
    max_iter_per_alpha = 500
    step_size = 0.001
    
    alpha = alpha_start
    
    while alpha <= alpha_end and time.time() - start_time < budget_s * 0.8:
        for _ in range(max_iter_per_alpha):
            if time.time() - start_time >= budget_s * 0.8:
                break
                
            # Compute auto-correlations
            A = autocorr_fft(h_coarse)
            g = -A
            
            # Compute softmax weights
            p = softmax_weights(g, alpha)
            
            # Compute gradient
            grad = gradient(h_coarse, p)
            
            # Update step
            h_new = h_coarse - step_size * grad
            
            # Project and update
            h_coarse = project_onto_sum_box(h_new, n_coarse / 2)
            
            # Evaluate if improved
            c5_new = c5_true(h_coarse)
            if c5_new < best_c5:
                best_c5 = c5_new
                # Interpolate back to fine resolution
                h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
                best_h = project_onto_sum_box(h_fine, TARGET_SUM)
        
        alpha *= gamma
        step_size *= 0.95
    
    # Finer resolution optimization (n=128)
    if time.time() - start_time < budget_s * 0.9:
        # Interpolate coarse to fine
        h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
        h_fine = project_onto_sum_box(h_fine, TARGET_SUM)
        
        # Continue optimization at fine resolution
        alpha = 1.0
        gamma = 1.2
        max_iter_per_alpha = 1000
        step_size = 0.0005
        
        while alpha <= alpha_end and time.time() - start_time < budget_s * 0.95:
            for _ in range(max_iter_per_alpha):
                if time.time() - start_time >= budget_s * 0.95:
                    break
                    
                # Compute auto-correlations
                A = autocorr_fft(h_fine)
                g = -A
                
                # Compute softmax weights
                p = softmax_weights(g, alpha)
                
                # Compute gradient
                grad = gradient(h_fine, p)
                
                # Update step
                h_new = h_fine - step_size * grad
                
                # Project and update
                h_fine = project_onto_sum_box(h_new, TARGET_SUM)
                
                # Evaluate if improved
                c5_new = c5_true(h_fine)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_fine.copy()
            
            alpha *= gamma
            step_size *= 0.95
    
    # Optional refinement at even higher resolution (n=256)
    if time.time() - start_time < budget_s * 0.99:
        n_fine = 256
        dx_fine = 2.0 / n_fine
        
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_points), best_h)
        h_fine = project_onto_sum_box(h_fine, n_fine / 2)
        
        # Optimization at this resolution
        alpha = 1.0
        gamma = 1.1
        max_iter_per_alpha = 500
        step_size = 0.0001
        
        while alpha <= 100.0 and time.time() - start_time < budget_s:
            for _ in range(max_iter_per_alpha):
                if time.time() - start_time >= budget_s:
                    break
                    
                # Compute auto-correlations
                A = autocorr_fft(h_fine)
                g = -A
                
                # Compute softmax weights
                p = softmax_weights(g, alpha)
                
                # Compute gradient
                grad = gradient(h_fine, p)
                
                # Update step
                h_new = h_fine - step_size * grad
                
                # Project and update
                h_fine = project_onto_sum_box(h_new, n_fine / 2)
                
                # Evaluate if improved
                c5_new = c5_true(h_fine)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_fine.copy()
            
            alpha *= gamma
            step_size *= 0.95
    
    # Discrete improvement
    if time.time() - start_time < budget_s:
        best_h, best_c5 = swap_improvement(best_h, max_iterations=2000)
    
    # Final polishing using scipy
    if time.time() - start_time < budget_s - 0.1:
        def objective(h):
            return c5_true(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_h = res.x
            best_h = project_onto_sum_box(best_h, TARGET_SUM)
            best_c5 = c5_true(best_h)
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
