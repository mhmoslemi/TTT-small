import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    # Binary MILP approach using cvxpy
    # Variables: h[i] binary for i=0..n-1, y[i,k] for overlapping pairs
    # Objective: minimize max_k (w - sum_i y[i,k]) = minimize T
    # Constraints: 
    # 1. sum h[i] = w
    # 2. for each overlapping pair (i,k), y[i,k] = h[i] * h[i+k] (using McCormick)
    # 3. for each shift k: sum_i y[i,k] >= w - T
    
    # For symmetry reasons and to reduce variables, we use a half-symmetric approach:
    # h[i] = h[n-1-i] for i < n/2
    # We represent only the first half and mirror it
    
    # Build the full problem
    # For binary case, n = 120, so we settle for computational tractability
    # Use only key shifts to speed up
    shifts = list(range(-n_points//2 + 1, n_points//2))
    h = cp.Variable(n_points, boolean=True)
    T = cp.Variable(integer=True)
    
    # All y[i,k] variables; only for overlapping pairs
    y_vars = {}
    # Precompute overlapping pairs
    y_indices = []
    for k in shifts:
        for i in range(n_points):
            if 0 <= i+k < n_points:
                y_indices.append((i,k))
                
    # Create binary variables y[i,k] for each overlap
    Y = {}
    for i,k in y_indices:
        Y[i,k] = cp.Variable(boolean=True)
    
    # Constraints: Model h_i * h_{i+k} = y_{i,k}
    # McCormick envelopes: y <= h_i, y <= h_{i+k}, y >= h_i + h_{i+k} - 1
    constraints = []
    # Sum constraint
    constraints.append(cp.sum(h) == TARGET_SUM)
    for i,k in y_indices:
        # Integer domain: it's optimal if h are binary, product should be respected
        # We force equality by bounding y to product level
        # y <= min(h_i, h_{i+k})
        constraints.append(Y[i,k] <= h[i])
        constraints.append(Y[i,k] <= h[i+k])
        # y >= h_i + h_{i+k} - 1
        constraints.append(Y[i,k] >= h[i] + h[i+k] - 1)
        # Also ensure maximum value constraint not violated 
        # by symbolic but redundant max use below
    
    # For objective formulation, aggregate overlaps
    # No explicit max, we linearize using T
    # Goal: minimize T as an upper bound on overlap >= all computed_o_k
    # Similar to DCP/convex formulation variant of the original
    # Point for each shift compute sum of y[i,k], constraint this is <= T
    o_k_sums = []
    for k in shifts:
        # Sum over all i such that i+k is valid
        total = 0
        for i in range(n_points):
            if 0 <= i+k < n_points:
                total += Y[i,k]
        # So we want sum_i y[i,k] >= w - T
        # Or rather we enforce w - sum_i y[i,k] <= T => sum_i y[i,k] >= w-T
        constraints.append(total >= TARGET_SUM - T)
    
    # Objective
    objective = cp.Minimize(T)
    
    # Solve using CBC solver
    prob = cp.Problem(objective, constraints)
    
    # Set time limit (use a guess as manipulating solver parameters is hard)
    # flush here to load solver
    # will compute best we get within time limit
    try:
        prob.solve(solver=CP.SCS, verbose=False, time_limit=540)
    except Exception as e:
        # If solver fails, try without solver-specific args
        try: 
            prob.solve(solver=GLPK_MI, verbose=False)
        except:
            pass

    if not prob.is_dcp():
        # Alternative: solve relaxation by allowing binary h to be continuous
        try:
            h_cont = cp.Variable(n_points, boolean=False)
            T = cp.Variable(boolean=False)
            # Apply same constraints with continuous variables
            constraints_l = []
            constraints_l.append(cp.sum(h_cont) == TARGET_SUM)
            for i,k in y_indices:
                # with continuous variables relax equality
                constraints_l.append(Y[i,k] <= h_cont[i])
                constraints_l.append(Y[i,k] <= h_cont[i+k])
                constraints_l.append(Y[i,k] >= h_cont[i] + h_cont[i+k] - 1)
                
            for k in shifts:
                # Sum over all valid indices
                total = 0
                for i in range(n_points):
                    if 0 <= i+k < n_points:
                        total += Y[i,k]
                constraints_l.append(total >= TARGET_SUM - T)
                
            objective = cp.Minimize(T)
            prob_l = cp.Problem(objective, constraints_l)
            prob_l.solve(solver=OSQP, verbose=False, time_limit=500)
            
            solution_h = h_cont.value
            if solution_h is not None:
                solution_h = np.clip(solution_h, 0, 1)
                final_h = solution_h
            else:
                final_h = np.ones(n_points) * 0.5
        except Exception as e:
            final_h = np.ones(n_points) * 0.5  # fallback    
    else:
        try:
            solution_h = h.value
            if solution_h is not None:
                final_h = np.array(solution_h).astype(float)
            else:
                final_h = np.ones(n_points) * 0.5
        except:
            final_h = np.ones(n_points) * 0.5

    # Final quality improvement: local search via gradient descent technique
    def compute_overlap_dense(h_vals):
        # Based on full correlation with 1-h
        h_unrolled = np.concatenate([h_vals, np.zeros(n_points-1)])
        one_minus_h = 1 - h_vals
        corr = correlate(h_unrolled, one_minus_h, mode='full')
        return np.max(corr[:len(corr)//2+1]) * dx
    
    def project_sum(h_vals, target_sum):
        # Ensure sum matches and values are between 0 and 1
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * (target_sum / current_sum)
        return np.clip(h_vals, 0, 1)
    
    best_h = final_h.copy()
    best_c5 = compute_overlap_dense(final_h)
    
    # Simple gradient ascent step: move h along gradient w.r.t. C5
    for iter_step in range(1000):
        if time.time() - start_time > budget_s:
            break
            
        # Since exact gradient is hard, just try random local move
        local_h = best_h.copy()
        # Small noisy move
        delta = np.random.normal(0, 0.001, n_points)
        local_h += delta
        local_h = project_sum(local_h, TARGET_SUM)
        local_c5 = compute_overlap_dense(local_h)
        if local_c5 < best_c5:
            best_h = local_h.copy()
            best_c5 = local_c5
    
    # Evaluate final C5 properly
    final_c5 = compute_overlap_dense(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
