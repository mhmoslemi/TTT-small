import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(f, alpha):
    """Compute softmax weights for gradient computation"""
    f_shifted = alpha * (f - np.max(f))
    exp_f = np.exp(f_shifted)
    return exp_f / np.sum(exp_f)

def compute_gradient(h, w):
    """Compute gradient using convolution approach"""
    w_rev = w[::-1]
    conv1 = np.convolve(w_rev, h, mode='full')
    conv2 = np.convolve(w, h, mode='full')
    
    # Extract center portion of appropriate size
    start_idx = n_points - 1
    grad = conv1[start_idx:start_idx + n_points] + conv2[start_idx:start_idx + n_points] - 1
    return grad
