```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def autocorr_fft(h):
    """Compute full auto-correlation using FFT"""
    H = fft(h, 2 * len(h))
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of smooth surrogate using convolutions"""
    # Two convolutions: p * h and p_rev * h
    # p_rev is reversed p
    p_rev = p[::-1]
    # Pad to avoid circular convolution effects
    n = len(h)
    padded_len = 2 * n
    h_padded = np.pad(h, (0, n), 'constant')
    p_padded = np.pad(p, (0, n), 'constant')
    p_rev_padded = np.pad(p_rev, (0, n), 'constant')
    
    # Linear convolutions
    r1 = np.convolve(p_padded, h_padded, mode='valid')[:n]
    r2 = np.convolve(p_rev_padded, h_padded, mode='valid')[:n]
    
    return -(r1 + r2)

def project_onto_sum_box(x, target_sum):
    """Project onto box constraints [0,1] with sum = target_sum using binary search"""
    x = np.array(x)
    def objective(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(100):
        if objective(a) * objective(b) > 0:
            a -= 0.1
            b += 0.1
        else:
            break
            
    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-10:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def c5_true(h):
    """Compute exact c5 using full correlation"""
    h_complement = 1 - h
    corr = np.correlate(h, h_complement, mode='full')
    return np.max(corr) * dx

def generate_binary_pattern():
    """Generate Thue-Morse binary pattern for initialization"""
    pattern = []
    for i in range(n_points):
        # Thue-Morse sequence: 0, 1, 1, 0, 1, 0, 0, 1, ...
        # This version computes the parity of the number of 1s in the binary representation of i
        count_ones = bin(i).count('1')
        pattern.append(1 if count_ones % 2 == 0 else 0)
    return np.array(pattern)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Coarse optimization first
    n_coarse = 64
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    # Initialize with binary pattern
    h_coarse = generate_binary_pattern()[:n_coarse]
    h_coarse = project_onto_sum_box(h_coarse, target_sum_coarse)
    
    # Annealing parameters
    alpha_start = 1.0
    alpha_max = 2000.0
    gamma = 1.3
    max_iter_per_alpha = 200
    eta = 0.2
    
    # Annealing loop over α
    alpha = alpha_start
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = c5_true(h_coarse)
    
    while alpha <= alpha_max and time.time() - start_time < budget_s * 0.8:
        for _ in range(max_iter_per_alpha):
            if time.time() - start_time > budget_s * 0.8:
                break
            # Compute auto-correlation
            A = autocorr_fft(h_coarse)
            g = -A
            # Compute soft max weights
            p = softmax_weights(g, alpha)
            # Compute gradient
            grad = gradient(h_coarse, p)
            # Update
            h_new = h_coarse - eta * grad
            # Project
            h_new = project_onto_sum_box(h_new, target_sum_coarse)
            # Evaluate
            c5_new = c5_true(h_new)
            if c5_new < c5_true(h_coarse):
                h_coarse = h_new.copy()
                if c5_new < best_c5_coarse:
                    best_h_coarse = h_coarse.copy()
                    best_c5_coarse = c5_new
                    
        alpha *= gamma
        eta *= 0.9
    
    # Interpolate to finer grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), 
                       best_h_coarse)
    h_fine = project_onto_sum_box(h_fine, TARGET_SUM)
    
    # Refine with full resolution
    h_opt = h_fine.copy()
    best_h = h_opt.copy()
    best_c5 = c5_true(h_opt)
    
    # Update annealing parameters for fine resolution
    alpha = alpha_start
    eta = 0.1
    
    while alpha <= alpha_max and time.time() - start_time < budget_s * 0.95:
        for _ in range(max_iter_per_alpha):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Compute auto-correlation
            A = autocorr_fft(h_opt)
            g = -A
            # Compute soft max weights
            p = softmax_weights(g, alpha)
            # Compute gradient
            grad = gradient(h_opt, p)
            # Update
            h_new = h_opt - eta * grad
            # Project
            h_new = project_onto_sum_box(h_new, TARGET_SUM)
            # Evaluate
            c5_new = c5_true(h_new)
            if c5_new < c5_true(h_opt):
                h_opt = h_new.copy()
                if c5_new < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_new
                    
        alpha *= gamma
        eta *= 0.95
    
    # Discrete hill-climbing post-processing
    h_best = best_h.copy()
    h_rounded = (h_best > 0.5).astype(float)
    
    # Evaluate current best
    c5_current = c5_true(h_best)
    
    # Try many swaps
    for _ in range(20000):
        if time.time() - start_time > budget_s:
            break
        i = np.random.choice(np.where(h_rounded > 0)[0])
        j = np.random.choice(np.where(h_rounded == 0)[0])
        # Swap values
        h_trial = h_best.copy()
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        c5_trial = c5_true(h_trial)
        if c5_trial < c5_current:
            h_best = h_trial
            c5_current = c5_trial
            h_rounded = (h_best > 0.5).astype(float)
    
    # Final polishing with SLSQP if time permits
    if time.time() - start_time < budget_s - 5:
        def objective(h_array):
            return c5_true(h_array)
        
        def constraint_eq(h_array):
            return np.sum(h_array) - TARGET_SUM
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective,
                h_best,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_best = res.x
            h_best = project_onto_sum_box(h_best, TARGET_SUM)
            best_c5 = c5_true(h_best)
        except:
            pass
    
    return (h_best.tolist(), float(best_c5), n_points)
```
