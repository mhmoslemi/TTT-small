import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_gradient(h, shift_idx):
    """Compute gradient of C_shift_idx with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    
    # Convert shift_idx to actual shift (negative means left shift)
    shift = shift_idx - (n - 1)
    
    if shift >= 0:
        # For shift >= 0, indices where h[i] contributes to summation
        # are i in [0, n-shift-1]
        grad[:n-shift] += 1
        # For h[i] * h[i+shift] terms, contribution to gradient
        # is -h[i+shift] (different for each i in [shift, n-1])
        grad[shift:] -= h[:n-shift]
    else:
        # For shift < 0, same pattern but adapted
        # shift = -abs_shift where abs_shift = -shift
        abs_shift = -shift
        # h[i] contribution for i in [abs_shift, n-1]
        grad[abs_shift:] += 1
        # h[i] * h[i-abs_shift] contribution
        grad[:n-abs_shift] -= h[abs_shift:]
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
