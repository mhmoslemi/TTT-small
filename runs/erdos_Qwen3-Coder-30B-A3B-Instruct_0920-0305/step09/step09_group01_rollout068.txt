import numpy as np
from scipy.linalg import eigh
from cvxpy import*
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem setup
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    # Store best result found
    best_h = None
    best_c5 = float('inf')
    
    # Helper function to compute overlap matrix
    def build_overlap_matrix(n):
        """Build a matrix M_l such that h'M_l h = sum of h_i * h_{i+l} over valid i"""
        M = []
        for l in range(n):
            M_l = np.zeros((n, n))
            # Add 1s where |i-j| = l and both indices valid
            for i in range(n - l):
                M_l[i, i + l] = 1
                M_l[i + l, i] = 1
            M.append(M_l)
        return M
    
    # Helper function to solve SDP relaxation
    def solve_sdp_relaxation(h0, M_list, alpha):
        n = len(h0)
        # Variables
        X = Variable((n, n), symmetric=True)
        h = Variable(n)
        # Constraints
        constraints = [
            X >> 0,  # X is psd
            diag(X) == h,  # diag(X) = h
            sum(h) == TARGET_SUM,
            0 <= h, h <= 1
        ]
        obj_values = []
        for M in M_list:
            objective = trace(M @ X)
            constraints.append(objective >= TARGET_SUM - alpha)
        # Solve SDP using SCS solver
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver='SCS', verbose=False, eps=1e-4)
            if prob.status == 'optimal':
                return h.value, X.value
        except:
            pass
        return None, None
    
    # Helper function to round SDP solution
    def round_sdp_solution(X_sol, h_sol, n):
        # Sample from the approximate solution
        try:
            # Cholesky decomposition of X
            U, s, Vt = np.linalg.svd(X_sol)
            s = np.maximum(s, 0)
            sqrt_s = np.sqrt(s)
            Y = U @ np.diag(sqrt_s)
            # Generate random vector from N(0, X)
            v = np.random.randn(n)
            h_rounded = Y @ v
            # Apply clipping and projection
            h_rounded = np.clip(h_rounded, 0, 1)
            # Project to correct sum
            h_rounded = h_rounded * TARGET_SUM / np.sum(h_rounded)
            return h_rounded
        except:
            # Fallback: use the rounded h_sol directly
            h_rounded = np.clip(h_sol, 0, 1)
            h_rounded = h_rounded * TARGET_SUM / np.sum(h_rounded)
            return h_rounded
    
    # Compute C5 for a given h
    def compute_c5(h):
        n = len(h)
        # Build overlap matrices for shifts 0 to n-1
        M = build_overlap_matrix(n)
        c5 = 0
        for k in range(n):
            overlap = np.dot(h, M[k] @ h)
            c5 = max(c5, 0.5 - overlap / 2)  # From c_k = 0.5 - overlap_k 
        return c5
    
    # Adaptive bisection on alpha
    def bisection_on_alpha():
        M_list = build_overlap_matrix(n_points)
        left, right = 0.0, 0.5
        best_alpha = right
        best_h = None
        
        # Start with initial feasible point
        h_start = np.ones(n_points) * 0.5  # Simple default
        h_start = h_start * TARGET_SUM / np.sum(h_start)
        c5_start = compute_c5(h_start)
        if c5_start < best_c5:
            best_h = h_start
            best_c5 = c5_start
        
        for _ in range(20):
            if time.time() - start_time > budget_s - 2:
                break
            
            mid = (left + right) / 2
            # Try to solve SDP with alpha = mid
            h_sdp, X_sdp = solve_sdp_relaxation(h_start, M_list, mid)
            if h_sdp is not None:
                # Round to get h
                h_rounded = round_sdp_solution(X_sdp, h_sdp, n_points)
                c5_test = compute_c5(h_rounded)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_rounded
                right = mid  # Feasible
                best_alpha = mid
            else:
                # Try rounding existing h_start, as fallback
                c5_test = compute_c5(h_start)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_start
                left = mid  # Not feasible
            h_start = np.random.rand(n_points)
            h_start = h_start * TARGET_SUM / np.sum(h_start)
            
        return best_h, best_c5

    # Run optimization
    h_opt, c5_opt = bisection_on_alpha()
    
    if h_opt is not None:
        # Final polishing
        h_final = h_opt.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Local improvement step: perturb and re-project
            h_new = h_final + np.random.normal(0, 1e-4, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * TARGET_SUM / np.sum(h_new)
            
            if compute_c5(h_new) < compute_c5(h_final):
                h_final = h_new

    c5_final = compute_c5(h_final)
    return (h_final.tolist(), float(c5_final), n_points)
