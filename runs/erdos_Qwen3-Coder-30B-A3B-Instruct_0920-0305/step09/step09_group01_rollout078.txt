import numpy as np
from scipy.linalg import block_diag
from cvxpy import *
import time

# Precomputed shift matrices and related constants
def build_shift_matrices(n_points):
    """Build all shift matrices M_ell for ell = 0..n-1"""
    matrices = []
    for ell in range(n_points):
        M = np.zeros((n_points, n_points))
        # Fill in the diagonal for shift ell
        for i in range(n_points - ell):
            M[i, i+ell] = 1
            M[i+ell, i] = 1
        matrices.append(M)
    return matrices

# SDP relaxation to solve feasibility for given alpha
def solve_sdp_feasibility(h0, shift_matrices, target_sum, alpha):
    n = len(h0)
    # Variables
    h = Variable(n)
    X = Variable((n, n), symmetric=True)
    
    # Constraints
    constraints = [
        X == h @ h.T,
        0 <= h, h <= 1,
        sum(h) == target_sum,
        X >> 0,  # Positive semidefinite
    ]
    
    # Add quadratic constraints
    for M in shift_matrices:
        # Tr(M @ X) >= n/2 - alpha
        constraints.append(trace(M @ X) >= target_sum/2 - alpha)
    
    # Objective (minimize a dummy function, just to solve feasibility)
    prob = Problem(Minimize(0), constraints)
    
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=5000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return h.value, X.value
        else:
            return None, None
    except:
        return None, None

# Randomized rounding from SDP solution
def round_sdp_solution(X, h_ref, target_sum, n_points):
    # Try randomized rounding with Gaussian
    u = np.random.randn(n_points)
    if X is not None:
        try:
            # Eigen decomposition of X
            eigvals, eigvecs = np.linalg.eigh(X)
            # Keep only positive eigenvalues
            eigvals = np.maximum(eigvals, 0)
            # Approximate matrix square root
            sqrt_X = eigvecs @ np.diag(np.sqrt(eigvals)) @ eigvecs.T
            v = sqrt_X @ u
        except:
            v = u
    else:
        v = u
        
    h_rounded = np.clip(v, 0, 1)
    # Project onto unit sum constraint
    h_rounded = h_rounded * (target_sum / np.sum(h_rounded))
    return h_rounded

# Evaluates C5 given h
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode="full")
    # The relevant correlations are those corresponding to shifts in [-n+1, n-1]
    # Only take relevant indices (center more than 0)  
    # Actually take max corr over valid shifts to get the C5
    max_corr = np.max(corr)
    return dx * max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Smaller for faster runs
    
    # Prepare shift matrices
    shift_matrices = build_shift_matrices(n_points)
    
    # Initialize baseline h
    h0 = np.array(initial_h_values)
    h0 = h0[:n_points]  # Keep it size n_points
    
    # Project onto sum constraint (exact)
    target_sum = n_points / 2
    h0 = h0 * (target_sum / np.sum(h0))
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Binary search on alpha
    alpha_low, alpha_high = 0.0, 1.0
    eps = 1e-5
    
    while alpha_high - alpha_low > eps:
        if time.time() - start_time > budget_s - 1:
            break
            
        alpha_mid = (alpha_low + alpha_high) / 2
        
        h_sol, X_sol = solve_sdp_feasibility(h0, shift_matrices, target_sum, alpha_mid)
        if h_sol is not None:
            # Try rounding
            h_rounded = round_sdp_solution(X_sol, h_sol, target_sum, n_points)
            c5_test = evaluate_c5(h_rounded)
            if c5_test < best_c5:
                best_h = h_rounded.copy()
                best_c5 = c5_test
            alpha_high = alpha_mid  # Feasible
        else:
            alpha_low = alpha_mid   # Infeasible
    
    # Post-processing: polishing via gradient-free method if time allows
    try:
        # Simple grid search near best solution to refine
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = best_h + np.random.normal(0, 0.001, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except:
        pass
        
    return (best_h.tolist(), float(best_c5), n_points)
