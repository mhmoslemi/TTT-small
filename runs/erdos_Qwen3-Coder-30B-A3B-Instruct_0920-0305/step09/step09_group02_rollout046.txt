def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_violations(h, T, n_points, dx):

    # shifts k = -n+1 to n-1
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # forward shift
            overlap = corr[n_points - 1 + k] * dx
        else:
            # backward shift
            overlap = corr[n_points - 1 + k] * dx
        if overlap > T:
            violations.append((k, overlap))
    return violations
