import numpy as np
from scipy.signal import correlate
from scipy.special import logsumexp
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_smoothed_max_grad(h, alpha):
    """Compute gradient of smooth max using log-sum-exp approximation"""
    corr = compute_corr_fft(h)
    c = corr
    
    # compute weights based on log-sum-exp
    c_scaled = alpha * c
    log_w = c_scaled - logsumexp(c_scaled)
    w = np.exp(log_w)
    
    # compute gradient 
    g = np.ones_like(h)
    
    # loop over shifts
    for k in range(-len(h) + 1, len(h)):
        if k >= 0:
            g[:len(h)-k] -= w[k] * h[k:]
            g[k:] -= w[k] * h[:len(h)-k]
        else:
            s = -k
            g[s:] -= w[k] * h[:len(h)-s]
            g[:len(h)-s] -= w[k] * h[s:]
    
    return g

def binary_search_projection(x, target_sum):
    """Projected point that satisfies sum constraint with box constraints [0,1]"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(x + (a + b)/2, 0, 1)

def simulated_annealing_binary(seed, budget_s):
    """Simulated Annealing on binary vectors"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize binary solution 
    h_bin = np.zeros(n_points, dtype=int)
    h_bin[:n_points//2] = 1
    np.random.shuffle(h_bin)
    
    current_c5 = evaluate_c5(h_bin.astype(float))
    best_c5 = current_c5
    best_h = h_bin.copy()
    
    T = 0.5
    gamma = 0.995
    
    iter_count = 0
    max_iter = 30000  # reasonable limit
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # find 1s and 0s
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
            
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        
        # propose swap
        h_new = h_bin.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # compute new correlation
        c5_new = evaluate_c5(h_new.astype(float))
        
        delta = c5_new - current_c5
        
        # acceptance probability
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h_bin = h_new
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_bin.copy()
        
        # cool down temperature
        T *= gamma
        iter_count += 1
    
    return best_h.astype(float), best_c5

def projected_gradient_descent(h_start, budget_s, alpha_max=30, alpha0=2, eta0=0.25, decay=0.9995,

    """Projected gradient descent on smoothed max"""
    start_time = time.time()
    
    h = h_start.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    alpha = alpha0
    eta = eta0
    momentum = 0
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        # compute correlation
        corr = compute_corr_fft(h)
        
        # evaluate smoothed maximum and gradient
        F = logsumexp(alpha * corr) / alpha
        grad = compute_smoothed_max_grad(h, alpha)
        
        # update momentum
        momentum = momentum_beta * momentum + (1 - momentum_beta) * grad
        
        # trial step
        h_trial = h - eta * momentum
        h_new = project_to_sum_box(h_trial, TARGET_SUM)
        
        # compute exact max to update best
        test_c5 = evaluate_c5(h_new)
        
        # accept update
        h = h_new
        
        # update best if necessary
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # linear decay of step size
        eta *= decay
        
        # increase smoothing  
        if iter_count % update_freq == 0:
            alpha = min(alpha * 1.02, alpha_max)
        
        # early stopping if no progress 
        if no_improvement_count > 500:
            if eta < min_eta:
                break
            eta /= 2
            no_improvement_count = 0
            
        iter_count += 1
    
    return best_h, best_c5

def subgradient_descent(h_start, budget_s, max_iter=50):
    """Subgradient descent on true max"""
    start_time = time.time()
    
    h = h_start.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # compute correlation
        corr = compute_corr_fft(h)
        max_idx = np.argmax(corr)
        
        # identify active shifts and build weights
        max_val = corr[max_idx]
        active = np.where(np.abs(corr - max_val) < 1e-10)[0]
        weights = np.zeros_like(corr)
        weights[active] = 1.0 / len(active)
        
        # compute subgradient
        g = np.ones_like(h)
        
        for k in active:
            if k >= 0:
                g[:len(h)-k] -= weights[k] * h[k:]
                g[k:] -= weights[k] * h[:len(h)-k] 
            else:
                s = -k
                g[s:] -= weights[k] * h[:len(h)-s]
                g[:len(h)-s] -= weights[k] * h[s:]
        
        # small step
        h_new = h - 0.02 * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # evaluate
        test_c5 = evaluate_c5(h_new)
        
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_h = h_new.copy()
            
        h = h_new
        iter_count += 1
    
    return best_h, best_c5
