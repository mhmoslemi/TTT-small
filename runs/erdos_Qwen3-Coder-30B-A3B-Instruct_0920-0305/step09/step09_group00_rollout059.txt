```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def autocorr_fft(h):
    """Compute full autocorrelation using FFT"""
    H = fft(h, 2*len(h)-1)
    A = ifft(H * np.conj(H)).real
    return A[:len(h)]

def softmax_weights(g, alpha):
    """Compute soft-max weights with numerical stability"""
    g_max = np.max(g)
    exp_g = np.exp(alpha * (g - g_max))
    p = exp_g / np.sum(exp_g)
    return p

def gradient(h, p):
    """Compute gradient of smooth surrogate using convolutions"""
    # Two convolutions with p and p_rev
    n = len(h)
    p_rev = p[::-1]
    
    # Linear convolution using FFT
    r1 = np.real(ifft(fft(h, 2*n-1) * fft(p, 2*n-1)))
    r2 = np.real(ifft(fft(h, 2*n-1) * fft(p_rev, 2*n-1)))
    
    # Take the first n components (valid convolution)
    r1 = r1[:n]
    r2 = r2[:n]
    
    return -(r1 + r2)

def project_onto_sum_box(x, target):
    """Euclidean projection onto [0,1]^n and sum constraint"""
    # Clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    
    def sum_with_lambda(lam):
        return np.sum(np.clip(x_clipped + lam, 0, 1)) - target
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        if sum_with_lambda(m) > 0:
            b = m
        else:
            a = m
    
    lambda_opt = (a + b) / 2
    result = np.clip(x_clipped + lambda_opt, 0, 1)
    return result

def c5_true(h):
    """Compute exact C5 using full correlation"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * (2.0 / len(h))

def build_thue_morse(n):
    """Build Thue-Morse binary pattern"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_list = [64, 128, 256]  # Multi-resolution
    alpha_max = 2000.0
    alpha_start = 1.0
    gamma = 1.3
    max_iter_per_alpha = 200
    eta = 0.01
    
    best_h = None
    best_C5 = float('inf')
    
    # Try multiple starting points
    start_h_list = []
    
    # Thue-Morse pattern
    h_tm = build_thue_morse(n_list[0])
    start_h_list.append(h_tm)
    
    # Random binary patterns
    for _ in range(3):
        h_rand = np.random.randint(0, 2, n_list[0]).astype(float)
        start_h_list.append(h_rand)
    
    # Project to sum and find best starting point
    best_start_h = None
    best_start_C5 = float('inf')
    for h_start in start_h_list:
        h_proj = project_onto_sum_box(h_start, n_list[0] / 2)
        c5 = c5_true(h_proj)
        if c5 < best_start_C5:
            best_start_C5 = c5
            best_start_h = h_proj
    
    h = best_start_h.copy()
    # coarse resolution optimization
    n = n_list[0]
    dx = 2.0 / n
    target_sum = n / 2
    
    h = project_onto_sum_box(h, target_sum)
    alpha = alpha_start
    
    while alpha <= alpha_max and time.time() - start_time < budget_s * 0.8:
        for it in range(max_iter_per_alpha):
            if time.time() - start_time > budget_s * 0.8:
                break
            A = autocorr_fft(h)
            g = -A
            p = softmax_weights(g, alpha)
            grad = gradient(h, p)
            h_new = h - eta * grad
            h_new = project_onto_sum_box(h_new, target_sum)
            
            # Accept if better surrogate
            if np.max(g) < alpha * np.log(np.sum(np.exp(alpha * g))):
                h = h_new.copy()
        
        alpha *= gamma
        # Slowly reduce step size
        eta *= 0.9
    
    # Interpolation to higher resolution
    if n_list[1] > n:
        h = np.interp(np.linspace(0, 2, n_list[1]), np.linspace(0, 2, n), h)
        h = project_onto_sum_box(h, n_list[1] / 2)
        n = n_list[1]
        dx = 2.0 / n
        target_sum = n / 2
        # Rerun optimization at higher resolution
        alpha = alpha_start
        while alpha <= alpha_max and time.time() - start_time < budget_s * 0.95:
            for it in range(max_iter_per_alpha):
                if time.time() - start_time > budget_s * 0.95:
                    break
                A = autocorr_fft(h)
                g = -A
                p = softmax_weights(g, alpha)
                grad = gradient(h, p)
                h_new = h - eta * grad
                h_new = project_onto_sum_box(h_new, target_sum)
                
                # Accept if better surrogate
                if np.max(g) < alpha * np.log(np.sum(np.exp(alpha * g))):
                    h = h_new.copy()
            alpha *= gamma
            eta *= 0.9
    
    # Continuous optimization with local search
    def objective(h):
        return c5_true(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            h = res.x
        h = project_onto_sum_box(h, target_sum)
        C5 = c5_true(h)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
    except:
        pass
    
    # Discrete refinement with swap operations
    if best_h is not None:
        h = best_h
        # Make binary pattern
        h_binary = (h > 0.5).astype(float)
        h_binary = project_onto_sum_box(h_binary, target_sum)
        C5_binary = c5_true(h_binary)
        if C5_binary < best_C5:
            best_C5 = C5_binary
            best_h = h_binary.copy()
        
        # Swap-based hill climb
        for _ in range(10000):
            if time.time() - start_time > budget_s:
                break
            # Random swap between 0 and 1
            ones = np.where(h_binary > 0.5)[0]
            zeros = np.where(h_binary < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Swap values
            h_new = h_binary.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            h_new = project_onto_sum_box(h_new, target_sum)
            C5_new = c5_true(h_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
            else:
                # Accept with small probability to escape local minima
                if np.random.random() < 0.001:
                    best_h = h_new.copy()
                    best_C5 = C5_new
        
        # Final SLSQP polish
        try:
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x
                h_final = project_onto_sum_box(h_final, target_sum)
                C5_final = c5_true(h_final)
                if C5_final < best_C5:
                    best_h = h_final
                    best_C5 = C5_final
        except:
            pass
    
    if best_h is None:
        # Fallback to simple method
        h_final = build_thue_morse(n_list[0])
        h_final = project_onto_sum_box(h_final, target_sum)
        best_C5 = c5_true(h_final)
        best_h = h_final
    
    return (best_h.tolist(), float(best_C5), len(best_h))
```
