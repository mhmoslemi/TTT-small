```python
import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 200
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def objective(h_values):
    """Objective function to minimize: C_5 bound."""
    return evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with a binary pattern that is known to be good
    # Use a pattern similar to quadratic residues for balance
    # Create a binary vector with exactly n_points/2 ones
    h = np.zeros(n_points)
    # Use a simple regular pattern for initialization
    # Pattern: alternating blocks of 2 ones and 2 zeros
    for i in range(n_points):
        h[i] = 1 if (i // 2) % 2 == 0 else 0
    # Adjust to get sum = n_points / 2
    current_sum = np.sum(h)
    need_ones = n_points // 2
    diff = need_ones - current_sum
    if diff > 0:
        # Set some zeros to ones
        zero_indices = np.where(h == 0)[0]
        for i in range(diff):
            h[zero_indices[i]] = 1
    elif diff < 0:
        # Set some ones to zeros
        one_indices = np.where(h == 1)[0]
        for i in range(-diff):
            h[one_indices[i]] = 0
    
    # Ensure integral constraint exactly (sum * dx = 1)
    h_sum = np.sum(h)
    if abs(h_sum * dx - 1.0) > 1e-10:
        scale = 1.0 / (h_sum * dx)
        h *= scale
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    
    # Use L-BFGS-B for improvement around current solution
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Limited optimization iterations allowed by time budget
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10.0:
        try:
            # SLSQP optimization
            result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraints, 
                              options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-6})
            if result.success:
                new_h = result.x
                new_c5 = evaluate_h(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except Exception as e:
            pass
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return list(best_h), final_c5, n_points
```
