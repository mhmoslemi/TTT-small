import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def solve_ccp_subproblem(h_prev, n, dx):
    """Solve the CCP subproblem using CVXPY and ECOS"""
    # CVXPY setup
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Precompute masks for shifts
    constraints = []
    
    # Build the convex quadratic terms for each shift
    for s in range(-(n-1), n):
        # forward indices
        mask_fwd = np.arange(max(0, -s), min(n, n-s))
        # backward indices  
        mask_bwd = np.arange(max(0, s), min(n, n+s))
        
        # Compute g_s^c(h) = sum(h) + 0.5*(sum(h_i^2) + sum(h_{i+s}^2))
        g_s_c = cp.sum(h) + 0.5 * cp.sum_squares(h[mask_fwd]) + 0.5 * cp.sum_squares(h[mask_bwd])
        
        # Compute g_s^d(h) = 0.5 * sum((h_i + h_{i+s})^2)
        g_s_d = 0.5 * cp.sum_squares(h[mask_fwd] + h[mask_bwd])
        
        # Linearise g_s^d at h_prev (gradient of quadratic)
        g_s_d_val = 0.5 * np.sum((h_prev[mask_fwd] + h_prev[mask_bwd])**2)
        grad_g_s_d = np.zeros(n)
        grad_g_s_d[mask_fwd] += h_prev[mask_fwd] + h_prev[mask_bwd]
        grad_g_s_d[mask_bwd] += h_prev[mask_bwd] + h_prev[mask_fwd]
        affine_term = g_s_d_val + np.dot(grad_g_s_d, h - h_prev)
        
        # Add constraint: tau >= g_s^c - affine_term
        constraints.append(tau >= g_s_c - affine_term)
    
    # Add bounding and integral constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1) 
    constraints.append(cp.sum(h) == n/2)
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve  
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            return h.value
        else:
            # Fallback to previous solution
            return h_prev
    except:
        return h_prev

def coordinate_descent_polish(h, n, dx):
    """Polish the solution using coordinate descent on the original max"""
    h = h.copy()
    total_sum = np.sum(h)
    
    # For faster computation, we'll do a few sweeps
    for sweep in range(5):
        # Shuffle indices
        indices = list(range(n))
        np.random.shuffle(indices)
        
        for i in indices:
            # Fixed all other h_j except h_i
            # Since we're changing h_i, we also need to adjust another component
            # to maintain the integral constraint
            
            # Set to extremes first to see effect on max overlap
            if i == 0:
                # For the first element, just try 0 and 1
                candidate_h = [0.0, 1.0]
            elif i == n-1:
                # For the last element, just try 0 and 1
                candidate_h = [0.0, 1.0]
            else:
                # Try 0, 1, and current value (if it's not at boundaries)
                current_val = h[i]
                if current_val > 0.001:
                    candidate_h = [0.0, current_val, 1.0]
                else:
                    candidate_h = [0.0, 1.0]
                
            # Compute current max overlap AND max overlap at feasible candidates
            best_h = h[i]
            best_c5 = evaluate_h(h)
            old_sum = total_sum
            total_sum -= h[i]
            
            for candidate_val in candidate_h:
                h[i] = candidate_val
                total_sum += candidate_val
                
                # Check constraints
                if 0 <= candidate_val <= 1 and abs(total_sum - n/2 * dx) < 1e-8:
                    new_c5 = evaluate_h(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = candidate_val

                total_sum -= candidate_val
                
            h[i] = best_h
            
    return h
