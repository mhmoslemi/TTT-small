import numpy as np

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum=1.0):
    """Project h onto the capped simplex: 0 <= h[i] <= 1, sum(h) = target_sum"""
    # Sort in descending order
    h_sorted = np.sort(h)[::-1]
    # Find the threshold
    cumsum = np.cumsum(h_sorted)
    threshold = (cumsum - target_sum) / np.arange(1, len(h_sorted) + 1)
    # Find the largest index where the threshold is less than or equal to the value
    idx = np.where(h_sorted > threshold)[0]
    if len(idx) == 0:
        tau = threshold[-1]
    else:
        tau = threshold[idx[-1]]
        
    # Apply projection
    h_proj = np.maximum(0, np.minimum(1, h - tau))
    # Adjust to exactly the right sum
    current_sum = np.sum(h_proj)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Increase some values, try to add to smallest ones
            indices = np.argsort(h_proj)
            for i in range(int(diff / 1e-10) + 1):
                if i >= len(indices):
                    break
                idx = indices[i]
                if h_proj[idx] < 1:
                    h_proj[idx] += 1e-10
        else:  # diff < 0
            # Decrease values
            indices = np.argsort(h_proj)[::-1]
            for i in range(int(-diff / 1e-10) + 1):
                if i >= len(indices):
                    break
                idx = indices[i]
                if h_proj[idx] > 0:
                    h_proj[idx] -= 1e-10
                    
    return h_proj

def compute_logsumexp_stable(log_vals):
    """Compute log(sum(exp(log_vals))) stably"""
    max_val = np.max(log_vals)
    if np.isinf(max_val):
        return max_val
    return max_val + np.log(np.sum(np.exp(log_vals - max_val)))

def compute_gradient(h, corr_vals, lambda_val):
    """Compute the gradient of log-sum-exp of correlations"""
    n = len(h)
    # Compute log weights
    max_corr = np.max(corr_vals)
    # Avoid numerical overflow by subtracting max
    log_weights = -lambda_val * (corr_vals - max_corr)
    # Compute log-sum-exp
    log_sum_exp = compute_logsumexp_stable(log_weights)
    # Get actual weights
    weights = np.exp(log_weights - log_sum_exp)
    
    # Compute gradient
    # g_j = sum_s w_s * (h_{j+s} + h_{j-s})
    # We'll compute this with iterative slicing to avoid memory complexities
    n_full = len(corr_vals)
    g = np.zeros_like(h)
    
    # Now we compute convolution of h with weights
    # weights has length = n_full = 2*n - 1
    # But it needs to be centered, so use full mode correlation to be safe
    # The idea is to compute for each valid index j:
    # g[j] = sum_s weights[s] * h[j + s] + h[j - s] 
    # But h is reflected
    # Let's think in terms of 1D convolution
    # weights * h for j where this is a valid convolution point
    
    # However, a simpler way to index correctly:
    # For convolution, full mode matches:
    # h and 1-h side-by-side is 2*n-1
    # corr_vals[0 : 2*n-1] or similar
    # The convolution used by np.correlate corresponds to h itself 
    # and h shifted by s where s = len(h) - 1 - i  
    # Let's do it directly for accuracy:
    
    # The weight vector corresponds to s = 0 to 2*n-2 
    # Actually it needs to be aligned right for follow-convolution
    # So for each j:
    # g[j] = sum_{s} weights[s] * (h[j + s - (n-1)] + h[j - s - (n-1)])
    # In the case where we shift h by s:
    # In corr it's s = 0 corresponds to h_0 * h_0 + h_1 * h_1 + ... (
    # But in the actual shifted case: h[j] shifts = s at j=s
    # h[i] * h[i+s] -> for s=0 -> h[i] * h[i]
    # So in the convention of full corr: 
    # h_i * h_{i+s}, result is {2*n-1} elements indexed from -(n-1) to (n-1)
    # The weight at index s changes with meaning: it corresponds to shift `s`
    # So index `i` of result: h[i] * h[i-s] where s = 0 to n-1, i = s, s+1, ..., n-1
    # i=0 to n-1, s = 0 to i, so shifts are from -(i) to (n-1-i)
    
    # Just simulate by full convolution of w + rev(w) with h piece by piece:
    
    # Implement convolution manually with such operations:
    weights_full = np.zeros(2*n - 1)
    # say indices i = 0 to 2*n-2 (shift from -(n-1) to n-1) where shift = i - (n-1)
    # weights[0] goes for shift -n+1 
    # weights[n-1] goes for shift 0
    # weights[-1] goes for shift n-1
    
    # Pragmatically: weight vector is aligned to shift index
    # s = 0 to 2*(n-1) = 2*n-2  
    # weight s corresponding to h_{i} * h_{i+s}, so for h_i with h_{i+shift}  
    # Titles the shift values from -s_max to s_max = -(n-1) to n-1, that means index 0 to 2*n-2
    # So if b_shift = 0 to 2*n-2, then actual shift = b_shift - (n-1)  
    # So w[i] corresponds to shift (i - (n-1))
    # The full correlation of h with h gives 0 to 2*n-2 indices for shift 0 to 2*(n-1) = 2*n-2
    # So we align them, it means:
    # In the convolution result: c[i] = sum over j h[j] * h[i-j]  
    # This corresponds to a convolution with h interchanged.  
    # That's unknown, in this case, the correlation of h with h very carefully
    # Let me derive position numbers from the actual usage
    shift_indices = np.arange(0, 2*n-1) - (n-1)
    # So index i of corr_vals corresponds to shift shift_indices[i] = i - (n-1)  
    # So if i from 0..2*n-2, then shift s = i - (n-1)  
    # So if weight i, it corresponds to shift = i - (n-1)  
    # We want to add weights[i] * h[j + (i - (n-1))]  
    # The vector weights has 2*n-1 elements  
    # The weight vectors [i] corresponds to shift (i - (n-1))
    
    # g_j = sum_{s=-n+1}^{n-1} weight[j+s+(n-1)] * (h_{j+s} + h_{j-s})
    # So here j = j, we shift by s, from s=-n+1 to n-1
    # weight's index p = s + n - 1, p from 0 to 2*n-2
    # When shift s = p - (n-1)  
    # For g_j:
    # You want sum_{p=0 to 2*n-2} weight[p] * (h_j + s) + h_j - s)  
    # where s = p - (n-1)
    # so s = p - (n-1) 
    # g_j = sum_{p=0}^{2*n-2} w[p] * h_{j + p - (n-1)} + h_{j - (p - (n-1))}
    # g_j = sum_{p=0}^{2*n-2} w[p] * h_{j + p - (n-1)} + h_{j + (n-1) - p} 
    # This has two components that can also be done by simple full convolution 
    # but direct indexing is easier for correctly handling edge conditions
    
    g = np.zeros_like(h) 
    for j in range(n):
        for p in range(2*n - 1):
            s = p - (n-1)
            if 0 <= j + s < n and 0 <= j - s < n:
                g[j] += weights[p] * (h[j + s] + h[j - s])
    
    return g

def run_gradient_search(h0, lambda_vals, step_sizes, iterations_per_phase, n, dx=0.01):

    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()

    # Get both correlation and reflection to construct full information
    for phase, (lambda_val, alpha, num_iter) in enumerate(zip(lambda_vals, step_sizes, iterations_per_phase)):
        for i in range(num_iter):
            # Compute local correlations
            corr_vals = np.correlate(h, h, mode="full") 
            # Compute gradient for this smooth approximation
            grad = compute_gradient(h, corr_vals, lambda_val)
            # Backtracking line search
            alpha_cur = alpha
            for linesearch_iter in range(5):
                h_new = h - alpha_cur * grad
                h_new = project_to_capped_simplex(h_new, target_sum=1.0)
                # Enforce symmetry after projection
                h_new = (h_new + h_new[::-1])/2
                h_new = project_to_capped_simplex(h_new, target_sum=1.0)

                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                if new_c5 < evaluate_erdos_solution(h, dx) or np.isnan(new_c5):
                    h = h_new
                    break
                else:
                    alpha_cur *= 0.5

    return best_h, best_c5
