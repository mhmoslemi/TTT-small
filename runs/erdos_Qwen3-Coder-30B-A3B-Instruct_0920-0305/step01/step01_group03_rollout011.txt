def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector onto box constraints [lower,upper] and sum=target_sum"""
    # Binary search for the right shift
    v = np.array(v)
    n = len(v)
    low, high = -10.0, 10.0
    for _ in range(50):
        theta = (low + high) / 2
        projected = np.clip(v + theta, lower, upper)
        sum_projected = np.sum(projected)
        if sum_projected < target_sum:
            low = theta
        else:
            high = theta
    # Final projection
    theta = (low + high) / 2
    result = np.clip(v + theta, lower, upper)
    # Adjust to exactly match target sum
    adjustment = (target_sum - np.sum(result)) / n
    result += adjustment
    return np.clip(result, lower, upper)

def smooth_max_objective(h, λ, dx=2.0/200):
    """Smooth approximation of max over correlations"""
    n = len(h)
    corr_hh = np.correlate(h, h, mode='full')
    max_corr = np.max(corr_hh)
    # Stable log-sum-exp computation
    shifted = corr_hh - max_corr
    exp_shifted = np.exp(-λ * shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = np.log(sum_exp) + max_corr
    return log_sum_exp / λ

    corr_hh = np.correlate(h, h, mode='full')

    shifted = corr_hh - max_corr
    exp_shifted = np.exp(-λ * shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    # Compute gradient
    grad = np.zeros_like(h)
    for s in range(-(n-1), n):
        idx = s + (n-1)
        w_s = weights[idx]
        if w_s == 0: continue
        if s >= 0:
            grad[:n-s] += w_s * h[s:]
            grad[s:] += w_s * h[:n-s]
        else:
            off = -s
            grad[off:] += w_s * h[:n-off]
            grad[:n-off] += w_s * h[off:]
    # Gradient is - sum_s w_s (h_{i+s} + h_{i-s}) 
    return -grad
