from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def proj_onto_simplex(v, target_sum, lower=0.0, upper=1.0):
    # v: array
    # return x solving min ||x - v||^2 subject to sum x = target_sum, lower <= x <= upper
    n = len(v)
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, lower, upper)
    return x
