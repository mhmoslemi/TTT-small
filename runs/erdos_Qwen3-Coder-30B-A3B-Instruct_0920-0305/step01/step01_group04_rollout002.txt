def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def projected_step(h, g, weight_sum=0.5 * n_points):
    """Project h onto the feasible space sum(h) = weight_sum and 0<=h<=1."""
    h_new = h - g

    h_new = np.clip(h_new, 0, 1)
    # Adjust to enforce sum constraint
    current_sum = np.sum(h_new)
    diff = weight_sum - current_sum
    if abs(diff) < 1e-8:
        return h_new
    # Find indices that can increase (not at upper bound) or decrease (not at lower bound)
    idxs = np.arange(len(h_new))
    idxs_lower = idxs[h_new == 0]
    idxs_upper = idxs[h_new == 1]
    idxs_free = idxs[(h_new > 0) & (h_new < 1)]
    # If we are short, help from free variables
    if diff > 0:
        # Add diff evenly to all free (not at upper limit) indices
        if len(idxs_free) > 0:
            delta = diff / len(idxs_free)
            h_new[idxs_free] += delta
            diff -= delta * len(idxs_free)
            if diff > 0:
                # Still have some diff, so set remaining to upper bound
                h_new[idxs_free] = 1.0
    else:
        # We need to decrease
        if len(idxs_free) > 0:
            delta = -diff / len(idxs_free)
            h_new[idxs_free] -= delta
            diff += delta * len(idxs_free)
            if diff < 0:
                h_new[idxs_free] = 0.0
    return h_new
