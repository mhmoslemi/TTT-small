```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_sum(h, target_sum):
    """Project h onto the feasible set {h ∈ [0,1]^n : sum(h) = target_sum}."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to sum to target_sum using water-filling approach
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum; sort ascending and increase from lowest
        idxs = np.argsort(h)
        remaining = diff
        for idx in idxs:
            if remaining <= 0:
                break
            increase = min(1 - h[idx], remaining)
            h[idx] += increase
            remaining -= increase
    else:
        # Need to decrease sum; sort descending and decrease from highest
        idxs = np.argsort(h)[::-1]
        remaining = -diff
        for idx in idxs:
            if remaining <= 0:
                break
            decrease = min(h[idx], remaining)
            h[idx] -= decrease
            remaining -= decrease
    return h

def compute_gradient(h, d_star):
    """Compute sub-gradient for shift d_star."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        s1 = h[(i + d_star) % n]
        s2 = h[(i - d_star) % n]
        grad[i] = 1 - s1 - s2
    return grad

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize constants
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Start with initial_h_values if available
    try:
        h = np.array(initial_h_values)
    except:
        h = np.random.rand(n_points)
        h = project_onto_box_and_sum(h, target_sum)
        
    h = h.copy()
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    print(f"Initial C5: {best_c5}")
    
    # Multi-start with different random seeds
    num_starts = 5
    for seed_start in range(num_starts):
        np.random.seed(seed_start * 100 + seed)
        if seed_start == 0:
            h0 = np.array(initial_h_values)
        elif seed_start == 1:
            h0 = np.random.rand(n_points)
            h0 = project_onto_box_and_sum(h0, target_sum)
        elif seed_start == 2:
            # Use Thue-Morse sequence
            thue_morse = [int(bin(i).count('1') % 2 == 0) for i in range(n_points)]
            h0 = np.array(thue_morse, dtype=float)
            h0 = project_onto_box_and_sum(h0, target_sum)
        else:
            h0 = np.random.rand(n_points)
            h0 = project_onto_box_and_sum(h0, target_sum)
        
        h_local = h0.copy()
        c5_local = evaluate_h(h_local)

        # Perform simulated annealing on this start
        T0 = 0.5
        gamma = 0.995
        T = T0
        alpha0 = 0.5
        max_iter = 30000
        iter_counter = 0
        while iter_counter < max_iter and time.time() - start_time < budget_s - 5:
            iter_counter += 1

            # Evaluate all correlations
            corr = np.correlate(h_local, 1 - h_local, mode="full")
            d_star = np.argmax(corr)
            max_corr = corr[d_star]
            
            # Compute sub-gradient
            grad = compute_gradient(h_local, d_star)
            
            # Step towards maximizer
            alpha = alpha0 / np.sqrt(iter_counter + 1)
            h_new = h_local - alpha * grad
            
            # Project back onto feasible region
            h_new = project_onto_box_and_sum(h_new, target_sum)
            
            # Compute new correlation
            corr_new = np.correlate(h_new, 1 - h_new, mode="full")
            max_corr_new = corr_new[d_star]
            
            # Accept or reject
            if max_corr_new < max_corr or np.random.rand() < np.exp(-(max_corr_new - max_corr) / T):
                h_local = h_new.copy()
                c5_local = evaluate_h(h_local)
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local
            
            # Cooling schedule
            T *= gamma

    # Final polishing via hill climbing
    h_hill = best_h.copy()
    c5_hill = evaluate_h(h_hill)
    improved = True
    while improved:
        improved = False
        for i in range(n_points):
            for j in range(n_points):
                if h_hill[i] == 1 and h_hill[j] == 0:
                    h_test = h_hill.copy()
                    h_test[i] = 0
                    h_test[j] = 1
                    c5_test = evaluate_h(h_test)
                    if c5_test < c5_hill:
                        h_hill = h_test
                        c5_hill = c5_test
                        improved = True
                        break
            if improved:
                break

    # Ensure final solution satisfies the integral constraint exactly
    sum_h = np.sum(h_hill)
    scale_factor = target_sum / sum_h
    h_hill = h_hill * scale_factor
    h_hill = project_onto_box_and_sum(h_hill, target_sum)
    final_c5 = evaluate_h(h_hill)

    return list(h_hill), final_c5, n_points
```
