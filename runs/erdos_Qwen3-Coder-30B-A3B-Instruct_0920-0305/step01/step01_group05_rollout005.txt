import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def dc_ccp_solve(h_init, n_points, max_iter=30, tol=1e-8):
    """Solve the DC program using convex-concave procedure"""
    dx = 2.0 / n_points
    h = h_init.copy()
    
    # Precompute masks for each shift
    shifts = range(-(n_points - 1), n_points)
    masks_fwd = {}
    masks_bwd = {}
    for s in shifts:
        if s >= 0:
            masks_fwd[s] = np.arange(s, n_points)
            masks_bwd[s] = np.arange(0, n_points - s)
        else:
            masks_fwd[s] = np.arange(0, n_points + s)
            masks_bwd[s] = np.arange(-s, n_points)
    
    # Build gradient information once
    def compute_gradient(gd_val, h_vals):
        grad = np.zeros_like(h_vals)
        for s in shifts:
            if s >= 0:
                grad[masks_bwd[s]] += (h_vals[masks_bwd[s]] + h_vals[masks_fwd[s]])
            else:
                grad[masks_fwd[s]] += (h_vals[masks_bwd[s]] + h_vals[masks_fwd[s]])
        return grad

    for _iter in range(max_iter):
        prev_h = h.copy()
        prev_tau = None
        
        # Solve convex subproblem with CVXPY
        h_var = cp.Variable(n_points)
        tau = cp.Variable()
        
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        for s in shifts:
            # Compute convex terms
            g_c_s = cp.sum(h_var) + 0.5 * cp.sum_squares(h_var[masks_fwd[s]]) + \
                    0.5 * cp.sum_squares(h_var[masks_bwd[s]])
            
            # Compute gradient of concave term
            g_d_grad_s = compute_gradient(0, prev_h)
            g_d_s = 0.5 * cp.sum_squares(h_var[masks_fwd[s]] + h_var[masks_bwd[s]])
            affine_s = g_d_s.value
            if s >= 0:
                affine_s = 0.5 * cp.sum_squares(prev_h[masks_bwd[s]] + prev_h[masks_fwd[s]]) + \
                           g_d_grad_s[masks_bwd[s]] @ (h_var[masks_bwd[s]] - prev_h[masks_bwd[s]]) + \
                           g_d_grad_s[masks_fwd[s]] @ (h_var[masks_fwd[s]] - prev_h[masks_fwd[s]])
            else:
                affine_s = 0.5 * cp.sum_squares(prev_h[masks_bwd[s]] + prev_h[masks_fwd[s]]) + \
                           g_d_grad_s[masks_bwd[s]] @ (h_var[masks_bwd[s]] - prev_h[masks_bwd[s]]) + \
                           g_d_grad_s[masks_fwd[s]] @ (h_var[masks_fwd[s]] - prev_h[masks_fwd[s]])
            
            constraints.append(tau >= g_c_s - affine_s)
        
        prob = cp.Problem(cp.Minimize(tau), constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
            if prob.status == cp.OPTIMAL:
                h = h_var.value
                if prev_tau is not None and abs(tau.value - prev_tau) < tol:
                    break
                prev_tau = tau.value
            else:
                break
        except Exception:
            break

    return h

def coordinate_descent_polish(h, n_points):
    """Polish solution with coordinate descent"""
    dx = 2.0 / n_points
    h = np.array(h)
    # For each variable
    indices = list(range(n_points))
    np.random.shuffle(indices)
    
    # Keeping integral constraint is maintained by pair changes
    for i in indices:
        old_val = h[i]
        best_score = np.inf
        best_h = h.copy()
        
        # Check at boundaries and when active shift changes
        # Since the function is piecewise quadratic (see note below)
        # But for simplicity, evaluate at 0, 1 and few interior points
        
        test_vals = [0.0, 1.0]
        for test_val in test_vals:
            # Change one variable
            temp_h = h.copy()
            temp_h[i] = test_val
            
            # Adjust another variable to maintain integral constraint
            # We pick j such that temp_h[j] will be adjusted to keep sum = n/2
            # Scale with reductions in total change
            for j in range(n_points):
                if j != i and temp_h[j] < 1.0:
                    diff = temp_h[i] - old_val
                    new_h_j = temp_h[j] - diff
                    if j == i or new_h_j < 0:
                        continue
                    else:
                        temp_h[j] = new_h_j
                        break
            else:
                # If no free variable found, abort or last resort
                new_h = temp_h
                corr = np.correlate(new_h, 1 - new_h, mode="full")
                score = np.max(corr) * dx
                if score < best_score:
                    best_score = score
                    best_h = new_h.copy()
                continue
                    
            # Evaluate
            corr = np.correlate(temp_h, 1 - temp_h, mode="full")
            score = np.max(corr) * dx
            if score < best_score:
                best_score = score
                best_h = temp_h
                
        h = best_h
        
    return h

def next_h_resolution(h_old, n_new):
    """Interpolate h_old to a new resolution n_new"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    
    # Normalize to integral = 1
    dx_new = 2.0 / n_new
    h_new = h_new * (1.0 / (np.sum(h_new) * dx_new))
    
    return h_new

        if i > 0:

        # CCP solve
        h_ccp = dc_ccp_solve(h_current, n_points)
        
        # Coordinate descent polish
        h_polished = coordinate_descent_polish(h_ccp, n_points)
        
        # Evaluate final result
        c5_bound = evaluate_h(h_polished)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_polished.copy()
            best_n = n_points
            
        h_current = h_polished
        
        if c5_bound < 0.3807:
            break
    
    # Try binary rounding if needed
    if best_c5 > 0.3807 and (time.time() - start_time) < budget_s - 5:
        # Round to binary
        h_binary = np.zeros_like(best_h)
        sorted_indices = np.argsort(best_h)[::-1]
        top_indices = sorted_indices[:best_n//2]
        h_binary[top_indices] = 1.0
        
        # Local binary hill-climb
        for _ in range(5000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute current overlap
            corr = np.correlate(h_binary, 1 - h_binary, mode="full")
            c5_current = np.max(corr) * (2.0 / best_n)
            
            # Try swapping
            i0 = np.random.choice(best_n)
            i1 = np.random.choice(best_n)
            if h_binary[i0] != h_binary[i1]:
                old0, old1 = h_binary[i0], h_binary[i1]
                h_binary[i0], h_binary[i1] = old1, old0
                corr = np.correlate(h_binary, 1 - h_binary, mode="full")
                c5_new = np.max(corr) * (2.0 / best_n)
                if c5_new < c5_current:
                    # Keep the swap
                    continue
                else:
                    # Rollback
                    h_binary[i0], h_binary[i1] = old0, old1
                    
        final_c5 = evaluate_h(h_binary)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_binary.copy()
            
    return (best_h.tolist(), best_c5, best_n)
