import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable(integer=True)
    
    # Objective: minimize max mismatch
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(x) == n_points // 2)
    
    # Binary product constraints for y[i,k] = x[i] * x[i+k] for all valid k
    # We will consider positive lags 1..n_points-1
    y_vars = {}
    
    # Add constraints for each positive lag k
    for k in range(1, n_points):
        # These are the overlapping indices for lag k
        # i from 0 to (n_points - 1 - k)
        for i in range(n_points - k):
            if (i, k) not in y_vars:
                y_vars[(i, k)] = cp.Variable(boolean=True)
                
            # Linearization of x[i] * x[i+k] = y[i,k]
            constraints.append(y_vars[(i, k)] <= x[i])
            constraints.append(y_vars[(i, k)] <= x[i+k])
            constraints.append(y_vars[(i, k)] >= x[i] + x[i+k] - 1)
    
    # Compute mismatches for each lag and define constraints
    # For lag k, we compute S_pos_k (sum of ones at i in [0, n_points-1-k])
    # and S_neg_k (sum of ones at i+k in [0, n_points-1-k] hence [k, n_points-1])
    # These must both be bounded by t
    for k in range(1, n_points):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Sum of x_i for i = 0 to n_points-1-k
        S_pos_k = cp.sum(x[:n_points-k])
        
        # Sum of y_{i,k} for i = 0 to n_points-1-k  
        sum_y_pos = cp.sum([y_vars[(i, k)] for i in range(n_points-k)])
        
        # Mismatch for positive lag k
        mismatch_pos_k = S_pos_k - sum_y_pos
        
        # Sum of x_{i+k} for i = 0 to n_points-1-k, which is sum of x_j for j = k to n_points-1
        S_neg_k = cp.sum(x[k:])
        
        # Sum of y_{i,k} for i = 0 to n_points-1-k (same subset we used above)
        # and compute mismatches for negative k as well
        sum_y_neg = cp.sum([y_vars[(i, k)] for i in range(n_points-k)])
        
        # Mismatch for negative lag -k
        mismatch_neg_k = S_neg_k - sum_y_neg
        
        # Add constraints: mismatch_pos_k <= t, mismatch_neg_k <= t
        constraints.append(mismatch_pos_k <= t)
        constraints.append(mismatch_neg_k <= t)
        
        if time.time() - start_time > budget_s * 0.95:
            break

    # Create and solve the problem
    problem = cp.Problem(objective, constraints)
    
    # Calculate solver time
    remaining_time = budget_s * 0.95 - (time.time() - start_time)

        solver_options = {'seconds': 0.1, 'msg': 0}
    else:

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:

    h_current = [h * scale for h in h_current]
    
    # Optimization parameters  
    best_h = h_current.copy()
    best_c5 = evaluate_h(h_current)

    max_iter = 10000
    current_iter = 0
    temp = 1.0
    cooling_rate = 0.9995

            # Subtract time from timeout
            break

        h_new = best_h.copy()

        idx = np.random.randint(len(h_new))
        factor = np.random.normal(0, 0.05)
        h_new[idx] = h_new[idx] + factor
        h_new[idx] = max(0, min(1, h_new[idx]))
        
        # Normalize again
        scale = 1.0 / (sum(h_new) * dx)

            best_h = h_new.copy()
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new.copy()
        
        temp *= cooling_rate
        current_iter += 1
    
    # Final check and return 
    final_h = best_h[:]
    final_c5 = evaluate_h(final_h)
    return (final_h, final_c5, n_points)
