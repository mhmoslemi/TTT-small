import numpy as np
from scipy.signal import correlate
import time

def project_simplex(y, target_sum):
    """Project y onto the simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Find the threshold
    tmp_sum = 0.0
    for i in range(len(sorted_y)):
        tmp_sum += sorted_y[i]
        if tmp_sum > target_sum - i * sorted_y[i]:
            break
    # Compute the threshold
    theta = (tmp_sum - target_sum) / (i + 1)
    # Compute projection
    x = np.maximum(y - theta, 0.0)
    return x

def compute_fs(h, s):
    """Compute f_s(h) = sum h_i (1 - h_{i+s}) for a fixed s"""
    n = len(h)
    if s >= 0:
        # Right shift h to get h[s:] + zeros on left
        shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        # Left shift h to get h[:s] + zeros on right
        shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.dot(h, 1 - shifted)

def compute_all_fs(h):
    """Compute all f_s(h) values for s from -(n-1) to (n-1)"""
    n = len(h)
    fs = np.zeros(2 * n - 1)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        fs[i] = compute_fs(h, s)
    return fs

def compute_grad(h, p, lambda_penalty):
    """Compute gradient of Phi_p + lambda_penalty * (sum(h) - n/2)^2"""
    n = len(h)
    fs = compute_all_fs(h)
    
    # Compute w_s = f_s^{p-1}
    # Add small epsilon to avoid 0^0
    eps = 1e-12
    w = np.abs(fs) ** (p - 1) + eps
    
    # Compute the full gradient
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        if s >= 0:
            shifted_fwd = np.concatenate([np.zeros(s), h[:-s]])
            shifted_bwd = np.zeros(n)
            shifted_bwd[s:] = h[:-s]
        else:
            shifted_bwd = np.concatenate([h[-s:], np.zeros(-s)])
            shifted_fwd = np.zeros(n)
            shifted_fwd[:-s] = h[-s:]
        
        # Gradient of f_s
        grad_term = 1.0 - shifted_bwd - shifted_fwd
        grad += w[i] * grad_term
    
    # Normalize by Phi_p^{p-1}
    Phi_p = np.sum(w * np.sign(fs) * fs) ** (1. / p)
    if Phi_p > eps:
        grad /= (Phi_p ** (p - 1))
    
    # Add penalty term
    grad += 2 * lambda_penalty * (np.sum(h) - n / 2)
    return grad

def optimize_h(h_init, n, p_vals, lambda_penalty, max_iter, dx):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = np.inf

    for p in p_vals:
        # Gradient descent with fixed step size
        step_size = 0.01
        for k in range(max_iter):
            grad = compute_grad(h, p, lambda_penalty)
            h = h - step_size * grad
            # Project to feasible set
            h = project_simplex(h, n / 2)
            # Clip to [0,1]
            h = np.clip(h, 0, 1)
            
            # Evaluate the objective
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Check convergence
            if k > 100 and abs(corr.max() * dx - best_c5) < 1e-10:
                break

    return best_h, best_c5

def interpolate_h(h_coarse, n_fine):
    """Linearly interpolate h_coarse to a finer grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
