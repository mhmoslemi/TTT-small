import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def project_to_simplex(y, target_sum):
    """Project y onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    y = np.asarray(y)
    n = len(y)
    # Ensure y is inside [0,1] by clipping
    y = np.clip(y, 0, 1)
    # Try bisection to find lambda
    lo, hi = y.min() - target_sum/n, y.max() - target_sum/n
    # Handle edge case where all values can't be clipped to feasible region
    # Check if we can even hit target_sum within bounds
    if target_sum > n:
        target_sum = n
    if target_sum < 0:
        target_sum = 0

    for _ in range(50):  # sufficient iterations for machine precision
        lam = (lo + hi) / 2
        x = np.clip(y - lam, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            lo = lam
        else:
            hi = lam

    lam = (lo + hi) / 2
    x = np.clip(y - lam, 0, 1)
    # Just in case, renormalize to exact sum
    scale = target_sum / np.sum(x)
    if scale > 0:
        x *= scale
    return x

def evaluate_c5(h):
    """Evaluate C5 using full correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, corr, alpha):
    """Given full correlation with shifts, compute gradient of smoothed max"""
    n = len(h)
    # remove zero padding in output from correlate: full mode produces 2*n-1 elements
    cvec = corr[n-1:] # indices 0 to 2*n-2 -> slice out middle n values
    # softmax weights
    cmax = np.max(cvec)
    exp_c = np.exp(alpha * (cvec - cmax))
    w = exp_c / np.sum(exp_c)
    # loop over indices to compute weighted sum of shifted copies  
    grad = np.ones(n)
    for s in range(n):
        w_s = w[s]
        # Add contribution from shift s (and symmetric negative shift)
        # shortcut: for s in [0,n-1], this is pc * h[start+s] + pc * h[start-s]
        # But more careful treatment needed
        tmp1 = np.zeros(n)
        tmp2 = np.zeros(n)
        if s < n-1:  # h[i+s] for valid indices only
            tmp1[:n-s] = h[s:]  
        if s != 0:   # h[i-s] for valid indices only
            tmp2[s:] = h[:n-s]
        #accumulate weighted contributions
        grad -= w_s * (tmp1 + tmp2)
    return grad

def compute_corr(h):
    """Compute the full correlation"""
    corr = correlate(h, 1-h, mode="full")
    return corr

def line_search(y, h, grad, alpha, step_size, target_sum, obj_fn):
    """Simple backtracking line search"""
    # try reduced step_size until objective decreases
    for _ in range(5):
        trial_h = project_to_simplex(y - step_size * grad, target_sum)
        trial_val = obj_fn(trial_h)
        if trial_val < obj_fn(h):
            return trial_h
        else:
            step_size /= 2
    # fallback if line search too aggressive  
    return project_to_simplex(y - step_size * grad, target_sum)

def projected_gradient_descent(start_h, target_sum, max_iter=5000, lr=0.5, alpha0=5.0, alpha_growth=1.03, alpha_max=200.0):
    h = start_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    alpha = alpha0
    for t in range(max_iter):
        corr = compute_corr(h)
        grad = compute_gradient(h, corr, alpha)

        # Backtracking line search
        eta_t = lr / np.sqrt(t+1)
        y = h - eta_t * grad
        h = line_search(y, h, grad, alpha, eta_t, target_sum, evaluate_c5)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Anneal alpha
        if (t+1) % 200 == 0:
            alpha = min(alpha * alpha_growth, alpha_max)
            
    return best_h, best_c5
