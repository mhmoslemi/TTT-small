import numpy as np

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

    constraints = [cp.sum(x) == n_points // 2]

    # We construct a_k = sum_{i} x_i * x_{i+k} via auxiliary variables
    
    # Linearize x[i] * x[i+k] = y[i,k]  
    y_vars = {}
    # For all valid shifts k (1 to N-1)
    for k in range(1, n_points):
        for i in range(n_points - k):
            # Create auxiliary variable for x[i] * x[i+k]
            y_var = cp.Variable(boolean=True)
            y_vars[(i, k)] = y_var
            # Linearize: y >= x[i] + x[i+k] - 1, y <= x[i], y <= x[i+k]
            constraints.extend([
                y_var >= x[i] + x[i+k] - 1,
                y_var <= x[i],
                y_var <= x[i+k]
            ])
    
    # Compute the mismatch functions for each lag k > 0  
    for k in range(1, n_points):

            sum_y = cp.sum([y_vars[(i,k)] for i in range(n_points - k)])

            sum_x_neg = cp.sum([x[i] for i in range(k, n_points)])

            constraints.append(sum_x_neg - sum_y <= t)
