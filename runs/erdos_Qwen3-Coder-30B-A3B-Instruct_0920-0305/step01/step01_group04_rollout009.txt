# Constants
n_points = 200  # Target size
dx = 2.0 / n_points

def build_shift_matrices(n):
    """Build Toeplitz shift matrices for each non-zero shift d."""
    C_d_list = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        C_d = np.zeros((n, n))
        for i in range(n):
            j1 = i + d
            j2 = i - d
            if 0 <= j1 < n:
                C_d[i, j1] = 1
            if 0 <= j2 < n:
                C_d[i, j2] = 1
        C_d_list.append((d, C_d))
    return C_d_list

def full_correlation(h):
    """Compute the full correlation vector and return max."""
    h = np.array(h)
    corr = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr

def round_vector(X, n):
    """Round a PSD matrix to a binary vector using hyperplane rounding."""
    # Eigen-decomposition
    try:
        U, V = eigh(X)
        V = V[:, U > 1e-8]
        U = U[U > 1e-8]
        if V.size == 0:
            raise ValueError("No significant eigenvectors")
        # Sample random Gaussian vector
        g = np.random.randn(V.shape[1])
        # Project
        v = V.T @ g
        # Get signs
        s = np.sign(v)
        # Make sure sum is even (or close)
        s_sum = np.sum(s)
        if s_sum % 2 != 0:
            # Flip the smallest one
            abs_v = np.abs(v)
            idx = np.argmin(abs_v)
            s[idx] = -s[idx]
        return s
    except Exception:
        # Fall back to random
        return np.random.choice([-1, 1], size=n)
