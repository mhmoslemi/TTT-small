import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """
    Project vector v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}
    """
    # Use binary search to find the right shift
    v_shift = v - lower  # shift so lower bound is 0
    upper_shifted = upper - lower  # new upper bound
    target_shifted = target_sum - len(v) * lower  # target sum shifted
    
    # Binary search for the shift theta
    left, right = -10.0, 10.0
    for _ in range(50):
        theta = (left + right) / 2.0
        x = np.clip(v_shift + theta, 0, upper_shifted)
        if np.sum(x) > target_shifted:
            right = theta
        else:
            left = theta
    
    # Calculate the final result
    theta = (left + right) / 2.0
    x = np.clip(v_shift + theta, 0, upper_shifted)
    return x + lower

                                    h_initial)
    h_initial_upsampled = project_onto_box_simplex(h_initial_upsampled, target_sum)
    
    # Generate more initial solutions
    initial_solutions = [h_initial_upsampled]
    for i in range(30):
        # Random uniform and project
        h_random = np.random.uniform(0, 1, n_points)
        h_random = project_onto_box_simplex(h_random, target_sum)
        initial_solutions.append(h_random)

    # Begin optimization
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Gradient-descent with smoothing routine
    def smooth_objective(h, lambda_val):
        corr = np.correlate(h, h, mode='full')
        max_corr = np.max(corr)
        # Stable logsumexp computation
        corr_shifted = corr - max_corr
        weights = np.exp(-lambda_val * corr_shifted)
        sum_weights = np.sum(weights)
        weights /= sum_weights
        # Gradient computation (convolution-like)
        grad = np.zeros_like(h)
        # Efficient implementation of gradient from weights and h
        for shift_idx in range(len(corr)):
            shift_val = shift_idx - (n_points - 1)
            w = weights[shift_idx]
            if w != 0:
                if shift_val >= 0:
                    grad[:n_points-shift_val] += w * h[shift_val:]
                    grad[shift_val:] += w * h[:n_points-shift_val]
                else:
                    shift_neg = -shift_val
                    grad[shift_neg:] += w * h[:n_points-shift_neg]
                    grad[:n_points-shift_neg] += w * h[shift_neg:]
        return np.sum(weights * corr), grad
    
    for restart in range(len(initial_solutions)):
        if time.time() - start_time > budget_s * 0.95:
            break

        h = initial_solutions[restart].copy()
        best_local_h = h.copy()
        best_local_c5 = evaluate_erdos_solution(h, dx)
        
        # Smoothed gradient descent
        schedule_lambdas = [1, 5, 10, 20, 40]
        iterations_per_lambda = 200
        alpha = 0.05
        
        for lambda_val in schedule_lambdas:
            if time.time() - start_time > budget_s * 0.95:
                break
            for _ in range(iterations_per_lambda):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute both objective and gradient
                obj_val, grad = smooth_objective(h, lambda_val)
                # Update with gradient step
                h -= alpha * grad
                # Project onto feasible region
                h = project_onto_box_simplex(h, target_sum)
                
                # Evaluate current solution
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_local_c5:
                    best_local_h = h.copy()
                    best_local_c5 = current_c5

        if time.time() - start_time <= budget_s * 0.95:

                return evaluate_erdos_solution(h_vals, dx)
            
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]

                objective_slsqp,
                best_local_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                final_c5 = evaluate_erdos_solution(result.x, dx)
                if final_c5 < best_local_c5:
                    best_local_c5 = final_c5
                    best_local_h = result.x.copy()
        
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
