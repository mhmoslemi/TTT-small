import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_simplex(y, target_sum):
    """Project y onto the simplex {x : sum(x) = target_sum, 0 <= x_i <= 1}"""
    n = len(y)
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the threshold index
    idx = np.arange(1, n + 1)
    condition = sorted_y - (target_sum - cumsum) / idx > 0
    if not np.any(condition):
        # All entries are too large, set to target_sum/n
        return np.full(n, target_sum / n)
    k = np.argmax(condition)
    # Compute threshold
    theta = (target_sum - cumsum[k]) / (k + 1)
    # Apply projection
    x = np.clip(y - theta, 0, 1)
    return x

def compute_correlation(h):
    """Compute full mode correlation of h with 1-h."""
    corr = correlate(h, 1 - h, mode="full")
    return corr

def compute_smooth_max_grad(h, alpha, corr):
    """Compute gradient of smooth max surrogate F_alpha(h)."""
    n = len(h)
    # Compute softmax weights
    c_vals = corr
    c_max = np.max(c_vals)
    # Subtract max for numerical stability
    exp_vals = np.exp(alpha * (c_vals - c_max))
    w = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros_like(h)
    # Iterate over all shifts - use the fact that corr is symmetric
    for s in range(-n+1, n):
        if s > 0:
            # h_{i-s} contributes forward shifts
            shifted = np.roll(h, s)
        else:
            # h_{i+s} contributes backward shifts
            shifted = np.roll(h, -s)
        if s == 0:
            # This is just h itself
            grad += (1 - 2 * h)
        else:
            shifted = np.roll(h, s)
            # Index offset in full mode correlation is s + n - 1
            # But ... we handle it directly.
            # For each shift, we want g_i = 1 - w_s * (h_{i+s} + h_{i-s}) 
            # where s is positive or negative shift amount
            # Let's build it carefully using loop
            w_s = w[n-1+s] 
            # so that we map s = - (n-1) to (n-1) to use w[0] to w[2*(n-1)]
            grad += -w_s * shifted
    
    return grad

def evaluate_c5(h):
    """Evaluate C5 for current h."""
    corr = compute_correlation(h)
    dx = 2.0 / len(h)
    return dx * np.max(corr)

    except:

        h0 = np.random.rand(n)
        h0 = project_to_simplex(h0, S)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Multi-start schedule
    max_restarts = 200
    restart_count = 0
    final_best_c5 = best_c5
    
    while (time.time() - start_time) < 0.7 * budget_s and restart_count < max_restarts:
        # Initialize new restart
        if restart_count == 0:
            h = h0.copy()
        else:
            # Random initial point
            h = np.random.rand(n)
            h = project_to_simplex(h, S)
        
        # Parameters for gradient descent
        alpha = 5.0
        alpha_incr = 1.02
        alpha_max = 200.0
        max_iter = 5000
        eta0 = 0.5  # Initial learning rate
        tol_improvement = 1e-6
        
        last_c5 = np.inf
        t = 0
        
        while t < max_iter and (time.time() - start_time) < budget_s - 5.0:
            t += 1
            
            # Compute full correlation (unscaled)
            corr = compute_correlation(h)
            
            # Compute smooth gradient
            grad = compute_smooth_max_grad(h, alpha, corr)
            
            # Gradient step with diminishing step size
            eta = eta0 / np.sqrt(t)
            h_new = h - eta * grad
            
            # Project back onto simplex
            h_new = project_to_simplex(h_new, S)
            
            h = h_new
            
            # Annealing of alpha
            if t % 200 == 0 and alpha < alpha_max:
                alpha = min(alpha_max, alpha * alpha_incr)
            
            # Evaluate current objective

            # Check for improvement

            # Check convergence
            if abs(last_c5 - c5) < tol_improvement:
                break
            last_c5 = c5
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break

    # Post-process with local optimizer on best solution
    # Apply slight refinement using scipy

        bounds = [(0, 1) for _ in range(n)]
