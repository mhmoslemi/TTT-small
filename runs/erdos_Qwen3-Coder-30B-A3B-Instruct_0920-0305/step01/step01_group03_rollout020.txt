import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}"""
    n = len(v)
    # Clip to bounds first
    v = np.clip(v, lower, upper)
    
    # Compute sum
    current_sum = np.sum(v)
    
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Binary search for the correct shift
    # We want to find theta such that sum_i clamp(v_i + theta, lower, upper) = target_sum
    # But since we are clamping to [0,1], it's simpler.
    # shift will be added to all values.
    # Sum after shift = sum(v) + n * shift = target_sum
    shift = (target_sum - np.sum(v)) / n
    
    # Apply shift and clamp
    v = v + shift
    v = np.clip(v, lower, upper)
    
    return v

            h_interp = np.interp(x_new, x_old, h0)
            # Normalize to sum = target_sum
            h_interp = h_interp * (target_sum / np.sum(h_interp))
            # Ensure within bounds
            h_interp = np.clip(h_interp, 0, 1)
            initial_solutions.append(h_interp)

    # 2. Random initializations

        h = h * (target_sum / np.sum(h))
        h = np.clip(h, 0, 1)

    # 3. Binary pattern
    binary_pattern = np.zeros(n_points)
    binary_pattern[:n_points//2] = 1
    np.random.shuffle(binary_pattern)
    binary_pattern = binary_pattern * (target_sum / np.sum(binary_pattern))
    binary_pattern = np.clip(binary_pattern, 0, 1)
    initial_solutions.append(binary_pattern)

    # Gradient-based optimization
    def smooth_objective(h, lambda_val):
        corr_hh = np.correlate(h, h, mode="full")
        # Avoid numerical overflow in exp
        max_corr = np.max(corr_hh)
        corr_shifted = corr_hh - max_corr
        exp_term = np.exp(-lambda_val * corr_shifted)
        sum_exp = np.sum(exp_term)
        log_sum_exp = np.log(sum_exp) + max_corr
        return log_sum_exp / lambda_val

    def smooth_gradient(h, lambda_val):
        corr_hh = np.correlate(h, h, mode="full")
        # Avoid numerical overflow in exp
        max_corr = np.max(corr_hh)
        corr_shifted = corr_hh - max_corr
        exp_term = np.exp(-lambda_val * corr_shifted)
        sum_exp = np.sum(exp_term)
        # Safe normalization
        weights = exp_term / sum_exp if sum_exp > 0 else np.ones_like(exp_term)
        
        # Compute gradient: - \sum_s weights_s * (h_{i+s} + h_{i-s})
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            for s in range(-(n-1), n):
                if 0 <= i + s < n:
                    grad[i] -= weights[s + (n-1)] * h[i+s]
                if 0 <= i - s < n and s != 0: # Avoid double count for s=0 
                    grad[i] -= weights[-s + (n-1)] * h[i-s]          
        return grad

        result = minimize(
            obj_func,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-12},
            callback=None,  # Can add callback to track progress if needed
        )
        if result.success:
            final_c5 = obj_func(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
        
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
    return (best_h.tolist(), best_c5, n_points)
