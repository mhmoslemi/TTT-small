import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def proj_onto_simplex(v, target_sum, lower=0.0, upper=1.0):
    # v: array
    # return x solving min ||x - v||^2 subject to sum x = target_sum, lower <= x <= upper
    n = len(v)
    # clamp to lower and upper? Actually we need to find lambda such that x = clamp(v - lambda, lower, upper) sums to target.
    # Define function f(lambda) = sum(clamp(v - lambda, lower, upper)) - target_sum.
    # f is decreasing in lambda (since clamp of v - lambda). So we can binary search.
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam  # need larger lambda to reduce sum
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, lower, upper)
    return x

    # Simulated Annealing parameters
    T = 0.1  
    cooling_rate = 0.99998
    iteration = 0
    max_iter = 1000000  # Safe upper bound
    
    # Main loop for Simulated Annealing
    while iteration < max_iter:
        if (time.time() - start_time) > (budget_s - 3):
            break
        
        # select two distinct indices at random
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_new = best_h.copy()

        # compute max delta
        delta_max_increase = min(1.0 - h_new[i], h_new[j])
        delta_max_decrease = min(h_new[i], 1.0 - h_new[j])

        # Ensure we can make non-zero move
        if delta_max_increase + delta_max_decrease < 1e-8:
            iteration += 1
            continue

        # sample random delta
        delta = np.random.uniform(-delta_max_decrease, delta_max_increase)
        if abs(delta) < 1e-10:
            iteration += 1
            continue

        # update values
        h_new[i] += delta
        h_new[j] -= delta
        
        # project back to [0,1] bounds and then to sum constraint
        h_new = proj_onto_simplex(h_new, target_sum)

            new_c5 = evaluate_c5(h_new)
