import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    
    # For MILP we'll solve for n_points = 48
    n_points = 48
    
    # Create binary variables
    x = cp.Variable(n_points, boolean=True)
    
    # Define the sum constraint
    sum_constraint = cp.sum(x) == n_points // 2
    
    # Create y variables for products (i, i+k)
    y = {}
    # For each k and i, we need i+k <= n_points-1, so i <= n_points-1-k
    for k in range(1, n_points):
        for i in range(n_points - k):
            y[(i, k)] = cp.Variable(boolean=True)
    
    # Product linearization constraints
    product_constraints = []
    for k in range(1, n_points):
        for i in range(n_points - k):
            # y[i,k] <= x[i]
            product_constraints.append(y[(i, k)] <= x[i])
            # y[i,k] <= x[i+k]
            product_constraints.append(y[(i, k)] <= x[i + k])
            # y[i,k] >= x[i] + x[i+k] - 1
            product_constraints.append(y[(i, k)] >= x[i] + x[i + k] - 1)
    
    # For each shift k, we need to compute the mismatch
    # m_k = sum_{i=0}^{n_points-1-k} (x_i - y_{i,k})
    # But we'll also consider negative shift
    # Objective is minimize max over all shifts of mismatch
    
    # Store mismatch for each symmetric shift pair (positive and negative)
    mismatch_pos = {}
    mismatch_neg = {}
    
    t = cp.Variable(integer=True)
    
    # Define the mismatch constraints
    constraints = [sum_constraint] + product_constraints
    
    # Each lag k needs its mismatch bounded by t
    for k in range(1, n_points):
        # Positive shift: sum_{i=0}^{n_points-1-k} x_i - sum_{i=0}^{n_points-1-k} y_{i,k}
        mismatch_pos[k] = cp.sum([x[i] for i in range(n_points - k)]) - cp.sum([y[(i, k)] for i in range(n_points - k)])
        # Negative shift: sum_{i=0}^{n_points-1-k} x_{i+k} - sum_{i=0}^{n_points-1-k} y_{i,k} 
        # But possibly can do by symmetric sum with i+k
        mismatch_neg[k] = cp.sum([x[i + k] for i in range(n_points - k)]) - cp.sum([y[(i, k)] for i in range(n_points - k)])
        
        # Both must be <= t
        constraints.append(mismatch_pos[k] <= t)
        constraints.append(mismatch_neg[k] <= t)
    
    # Objective: minimize the maximum mismatch
    objective = cp.Minimize(t)
    
    # Solve the MILP
    prob = cp.Problem(objective, constraints)
    start_time = time.time()
    
    try:
        # Use CBC solver for integer programming
        prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=budget_s*0.95)
        if prob.status == cp.OPTIMAL:
            final_h = [float(x[i].value) for i in range(n_points)]
            c5_bound = t.value * (2.0 / n_points)
            return (final_h, c5_bound, n_points)
        else:
            # Fallback to the original algorithm if MILP fails
            pass
    except:
        pass
    
    # Fallback if MILP solving failed or timed out
    # Start with initial construction for simplicity
    h0 = np.array(initial_h_values)
    # Adjust size to 48 if needed
    if len(h0) != n_points:
        if len(h0) > n_points:
            h0 = h0[:n_points]
        else:
            h0 = np.pad(h0, (0, n_points - len(h0)), 'constant', constant_values=0)
            h0 = h0[:-1]  # To make sure it's exactly n_points
    
    # Rescale to meet constraint
    dx = 2.0 / n_points
    scale_factor = 1.0 / (np.sum(h0) * dx)
    h0 = h0 * scale_factor
    h0 = np.clip(h0, 0, 1)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Try some local discretization tweaks
    max_iter = 5000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Perturb slightly
        h_new = best_h + np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale
        scale = 1.0 / (np.sum(h_new) * dx)
        h_new *= scale
        
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Ensure final constraint
    scale = 1.0 / (np.sum(best_h) * dx)
    best_h *= scale
    
    return (best_h.tolist(), best_c5, n_points)
