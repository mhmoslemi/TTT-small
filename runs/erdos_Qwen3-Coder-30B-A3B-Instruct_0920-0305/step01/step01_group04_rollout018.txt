def full_correlation(h):
    """Compute the full correlation vector (no dx scaling)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Trim to valid shifts only, max shift is n-1
    return corr[n-1:2*n-1]

def evaluate_h(h):
    """Evaluate C5 bound for given h."""
    corr = full_correlation(h)
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx
