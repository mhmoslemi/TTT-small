import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def interpolate_h(h_coarse, n_fine):
    """Interpolate coarse h to fine grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
    h_fine = f(x_fine)
    # Ensure it's within [0,1]
    h_fine = np.clip(h_fine, 0, 1)
    # Renormalize to integral = 1
    dx_fine = 2.0 / n_fine
    h_fine = h_fine * (1.0 / (np.sum(h_fine) * dx_fine))
    return h_fine

def evaluate_shift_term(h, s):
    """Evaluate f_s(h) = sum h_i * (1 - h_{i+s})"""
    n = len(h)
    if s >= 0:
        h_shifted = np.concatenate([h[s:], np.zeros(s)])
    else:
        h_shifted = np.concatenate([np.zeros(-s), h[:s]])
    return np.sum(h * (1 - h_shifted))

def compute_gradients(h):
    """Compute gradients of g_s^d(h) for all shifts s"""
    n = len(h)
    grad_dict = {}
    for s in range(-n+1, n):
        grad = np.zeros(n)
        if s >= 0:
            # grad[i] = (h[i] + h[i+s]) + (h[i-s] + h[i]) 
            # = 2*h[i] + h[i+s] + h[i-s] 
            # but avoid out-of-bound indices
            for i in range(n):
                grad[i] = 2 * h[i]
                if i + s < n:
                    grad[i] += h[i+s]
                if i - s >= 0:
                    grad[i] += h[i-s]
        else:
            s = -s # Make it positive for the loop below
            for i in range(n):
                grad[i] = 2 * h[i]
                if i + s < n:
                    grad[i] += h[i+s]
                if i - s >= 0:
                    grad[i] += h[i-s]
        grad_dict[s] = grad
    return grad_dict

def solve_ccp_subproblem(n, h_prev, dx, grad_dict, max_iter=30, tol=1e-6):
    """Solve the convex surrogate CCP subproblem using CVXPY"""
    h = cp.Variable(n)
    tau = cp.Variable()
    # Constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n/2
    ]
    
    # For each shift s, add the constraint
    for s in range(-n+1, n):
        # precompute terms needed for g_s^c(h)
        if s >= 0:
            mask_fwd = np.arange(s, n)
            mask_bwd = np.arange(0, n-s)
        else:
            s = -s
            mask_fwd = np.arange(s, n)
            mask_bwd = np.arange(0, n-s)
        
        # g_s^c(h) = \sum_i h_i + 0.5 * sum (h_i^2 + h_{i+s}^2) 
        # Our current h_prev was already normalized to integral=1, so dx is constant
        # The constraint to add is tau >= g_s^c(h) - linearization of g_s^d at h_prev
        g_c = cp.sum(h) + 0.5 * cp.sum_squares(h[mask_fwd]) + 0.5 * cp.sum_squares(h[mask_bwd])
        # linearization of g_s^d(h) at h_prev: 
        # g_s^d(h) ≈ g_s^d(h_prev) + grad_s.T * (h - h_prev)
        # So the linearized term is: g_s^d(h_prev) + grad_s.T * h - grad_s.T * h_prev
        linear_term = grad_dict[s].T @ h - grad_dict[s].T @ h_prev
        constraints.append(tau >= g_c - linear_term)
    
    # Solve
    prob = cp.Problem(cp.Minimize(tau), constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            return h.value, prob.value
        else:
            return None, None
    except:
        return None, None

def optimize_with_ccp(h0, dx, max_iter=30, tol=1e-6):
    """Optimize using convex-concave procedure"""
    h = h0.copy()
    n = len(h)
    # Precompute gradients
    grad_dict = compute_gradients(h)
    
    prev_value = float('inf')
    for i in range(max_iter):
        # Solve CC problem
        h_new, tau_val = solve_ccp_subproblem(n, h, dx, grad_dict, max_iter=10)
        if h_new is None:
            break
        h_new = np.array(h_new)
        # For numerical reasons: clamp and re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (1.0 / (np.sum(h_new) * dx))
        
        # Update gradients
        grad_dict = compute_gradients(h_new)
        
        # Check for convergence
        c5 = evaluate_h(h_new)
        if abs(c5 - prev_value) < tol:
            break
        h = h_new
        prev_value = c5
    
    return h, evaluate_h(h)

def coordinate_descent_polish(h, n_iter=3):
    """Polish the solution with coordinate descent"""
    dx = 2.0 / len(h)
    n = len(h)
    h_new = h.copy()
    
    for sweep in range(n_iter):
        indices = list(range(n))
        np.random.shuffle(indices)
        for i in indices:
            original = h_new[i]
            best_val = float('inf')
            best_h = original
            
            # Evaluate all breakpoints
            # But we'll do it in a coarse way first
            # Actually, let's just sample a few points
            tests = [0., 1.]
            if 0 < original < 1:
                tests.extend([max(0, original - 0.05), min(1, original + 0.05)])
                h_test = h_new.copy()
                for test_val in tests:
                    h_test[i] = test_val
                    # Make sure integral constraint is satisfied:
                    # If we change h[i], we must change another h[j]
                    # to maintain sum = n/2, i.e. adjust h[j] = h[j] - (test_val - h_new[i])
                    j = (i + 1) % n  # Not perfectly the way it should be, but acceptable
                    h_test[j] = h_test[j] - (test_val - h_new[i])
                    # Clamp values
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (1.0 / (np.sum(h_test) * dx))
                    c5 = evaluate_h(h_test)
                    if c5 < best_val:
                        best_val = c5
                        best_h = test_val
            
            h_new[i] = best_h
            # Maintain the constraint
            h_new = h_new * (1.0 / (np.sum(h_new) * dx))
    
    return h_new
