import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project vector y onto the capped simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Standard algorithm: sort y in descending order
    v = np.sort(y)[::-1]
    # Compute the cumulative sums
    sv = np.cumsum(v)
    # Find the threshold index
    i = np.arange(1, len(v) + 1)
    # We want the largest i such that (sv[i-1] - target_sum)/i <= v[i-1]
    # This is equivalent to: sv[i-1] - target_sum <= i * v[i-1]
    # Or: sv[i-1] <= i * v[i-1] + target_sum
    # Which implies: (sv[i-1] - target_sum)/i <= v[i-1]
    rho = np.where((sv - target_sum) / i <= v)[0]
    
    # If no such i, all are negative or zero
    if len(rho) == 0:
        lambda_val = (np.sum(y) - target_sum) / len(y)
        return np.clip(y - lambda_val, 0, 1)
    
    # Get the index of the last such i
    rho_max = rho[-1]
    # Compute lambda
    lambda_val = (sv[rho_max - 1] - target_sum) / rho_max
    # Project
    x = np.clip(y - lambda_val, 0, 1)
    return x

def compute_gradient(h, c, alpha):
    """Compute gradient of F_alpha with respect to h"""
    n = len(h)
    # Softmax weights
    c_max = np.max(c)
    eca = np.exp(alpha * (c - c_max))
    w = eca / np.sum(eca)
    
    # Build the gradient using shifts
    grad = np.ones(n)
    for i in range(n):
        # Forward shift (i -> i+1)
        grad[i] -= w[0] * h[i]
        # Backward shift (i -> i-1)
        grad[i] -= w[0] * h[i]

        # Compute shift index by offset: i goes to i+s, so shift s is i-s (for backward)
        # when we accumulate contribution from c_s, we're associating
        # [i] with [i+s] and also [i+s] with [i], for all s

        # Use scipy signal's correlation result and extract shifted sums
        # But for clarity of implementation:
        # Let's compute shifts of h properly and re-use full corr result
        
        # Collect all shift contributions
        grad_shift = np.zeros(n)
        # iterate over all shift indices appropriately
        # shift values are 0..2*(n-1) for a full crosscorrelation of size n with size n
        # Actual shifts +1 through +n-1 and -1 through -(n-1), plus shift 0 
        for s in range(2*n - 1):
            shift_idx = s - (n-1)
            c_sign = 1 if (s >= n) else -1
            # up to 2 * n - 1 terms for shifts in corr
            # Indices in full crosscorrelation
            if shift_idx >= 0:
                # shift forward: h[i] interacts with h[i + shift]
                # In full corr k=shift means h[i] * h[i + k] 
                # e.g. first element of corr: 0 shift; last element: 2n-2 shifts
                # So corr has 2n-1 elements, index k: 0 to 2n-2
                corr_idx = s # directly indexed by s
                # not quite right above, we may be using index s as offset into corr array
                # We use the convention that corr[s] corresponds to shift s - (n-1) positions
                
            # Ok simplify: compute full vectorized shift eval
            # shift by s: h[i]  * h[i + s - (n-1)] = h[i] * h[i + 1 - n + s]  
            # Another stance: in corr of a*b, shift k means [i] * [i + k]
            # shift indices: -n+1 to n-1
            # But k ranges from 0 to 2n-2
            # k for shift 0: Shift 0 = 0, 0 ranging from 0 to 2n-2 => k
            # corr[0] -> shift -n+1, corr[k] -> shift  k - (n-1)
            corrshift = s - (n-1)
            if -n+1 <= corrshift <= n-1:
                # Convert compact shift to actual found index in corr
                # Just assign contribution for the current s loop term
                if corrshift == 0:
                    grad[np.arange(n)] -= w[s] * 0  # or just skip
                else:
                    # Regular case where shift is non-zero
                    if corrshift > 0:
                        # forward shift: h[i] and h[i + corrshift]
                        # Can use extract-only trick
                        # weight the adjacent shifted elements via i,s
                        leading = -w[s] * h[:-corrshift]  # left end up to -corrshift
                        trailing = -w[s] * h[corrshift:]  # right part starting from corrshift
                        grad[:len(leading)] += leading
                        grad[corrshift:] += trailing
                    elif corrshift < 0:
                        # back shift: h[i] and h[i + corrshift]  
                        abs_corrshift = -corrshift
                        # shift backward: leading (index < abs_corrshift) is zero
                        # leading end: 
                        leading = -w[s] * h[abs_corrshift:]
                        trailing = -w[s] * h[:-abs_corrshift]  
                        # aggregate
                        grad[abs_corrshift:] += leading
                        grad[:len(trailing)] += trailing
            # we better do better and compute the gradient from scratch comprehensively
    
    # Re-do in robust, clear way
    # accumulates weights from convolution approach
    grad = np.ones(n)
    # for shifts in range(2* n - 1), process corr weights
    total_shifts = 2 * n - 1
    for k in range(total_shifts):
        # shift is k - (n-1)
        shift_k = k - (n-1)
        w_k = w[k]
        # Update grad
        if shift_k > 0:
            # h[i] contributes to h[i + shift_k] via correlation
            # Equivalent: correlation of h with itself shifted by k-n+1
            # We compute: grad[i] -= w_k * (h[i+shift_k] + h[i-shift_k])
            # so:
            # For positive shift_k: h[i] * h[i+shift_k] contributions to shift from the right to the left
            # Effect of positive shift k: pull from right
            if shift_k == 0:
                # contribute 0 directly
                pass
            else:
                # Use np.take with clip mode to wrap index
                # We compute when h[i] interacts with h[i + shift_k]
                shifted_h_right = np.take(h, np.arange(n) + shift_k, mode='clip') 
                # in actual expansion h[i] = h[i] * h[i + shift_k]
                shifted_h_left = np.take(h, np.arange(n) - shift_k, mode='clip')
                grad -= w_k * (shifted_h_left + shifted_h_right)
                
        elif shift_k < 0:
            shift_abs = -shift_k
            # Reduced to case above for positive and negative bash  
            # h[i] interacts with h[i + shift_k] and is same as h[i + shift_abs]
            shifted_h_left = np.take(h, np.arange(n) - shift_abs, mode='clip') 
            shifted_h_right = np.take(h, np.arange(n) + shift_abs, mode='clip')
            grad -= w_k * (shifted_h_left + shifted_h_right)
        else:
            # no shift
            pass # leave as +1
            
    return grad

def gradient_step(h, alpha, eta):
    """Perform one gradient ascent step on F_alpha and project back"""
    # Compute correlation values (this is the corr for c_s' directly).
    corr = correlate(h, 1 - h, mode="full")
    
    # F_alpha = (1/α) * log(sum(exp(α * c_s)))
    # For numerical stability shift by max(corr)
    # And then optimize using projected gradient step:
    # This derivation of the gradient is a simplified version assuming 1D        
    n = len(h)
    grad = compute_gradient(h, corr, alpha)
    
    # Compute the new point
    y = h - eta * grad
    
    # Project y back to capped simplex
    h_new = project_to_simplex(y, n/2)
    
    return h_new

    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Algorithm parameters
    max_iters = 5000
    alpha_start = 5.0
    alpha_factor = 1.03
    alpha_max = 200.0
    eta0 = 0.5
    keep_time = 0.5  # seconds to reserve for final polishing
    
    # Multi-start strategy – in practice, could restart many times
    num_restarts = 100
    restart_count = 0
    
    # Simple early termination check 
    done = False
    
    while (restart_count < num_restarts) and ((time.time() - start_time) < (budget_s - keep_time)) and not done:
        # Initialize
        h = h0.copy()
        alpha = alpha_start
        
        # Inner loop
        t = 0
        for t in range(max_iters):
            if (time.time() - start_time) > (budget_s - keep_time):
                break
                
            # Update learning rate
            eta = eta0 / np.sqrt(t + 1)
            
            # Gradient step
            h = gradient_step(h, alpha, eta)
            
            # Annealing
            if t % 200 == 199:
                alpha *= alpha_factor
                alpha = min(alpha, alpha_max)
                
            # Evaluate current candidate
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
            if best_c5 < 0.3807:
                done = True
                break
                
        restart_count += 1
        
        # Start next restart from a random point on simplex
        h0 = np.random.uniform(0, 1, n_points)
        h0 = project_to_simplex(h0, target_sum)
        
        # Manual break if done so far
        if done: break
        
    # Optional polish with SLSQP
    def obj_to_minimize(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(obj_to_minimize, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            best_h = res.x
            best_c5 = obj_to_minimize(best_h)
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
