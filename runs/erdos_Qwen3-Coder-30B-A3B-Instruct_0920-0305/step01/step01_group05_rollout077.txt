def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_sum_constraint(z, target_sum):
    """Project z onto the hyperplane sum = target_sum while respecting [0,1] bounds"""
    # Use bisection to find the right alpha
    low, high = -10.0, 10.0
    for _ in range(30):
        alpha = (low + high) / 2
        y = np.clip(z + alpha, 0, 1)
        s = np.sum(y)
        if s > target_sum:
            high = alpha
        else:
            low = alpha
    return np.clip(z + (low + high)/2, 0, 1)
