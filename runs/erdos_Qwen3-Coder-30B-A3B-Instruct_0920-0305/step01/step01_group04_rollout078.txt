def full_correlation(h):
    """Compute correlation vector and the maximum value."""
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)

    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def solve_sdp(n0):
    """Solve SDP relaxation for the autocorrelation problem on a ring of size n0."""
    # Build Toeplitz matrices for all non-zero shifts
    C_d = {}
    for d in range(-(n0-1), n0):
        if d == 0:
            continue
        C = np.zeros((n0, n0))
        for i in range(n0):
            j = (i + d) % n0
            C[i, j] = 1
            C[j, i] = 1
        C_d[d] = C

    # Define SDP variables
    X = cp.Variable((n0, n0), symmetric=True)
    tau = cp.Variable()

    # SDP constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == 1,  # Diagonal entries are 1
    ]

    # Autocorrelation constraints for all shifts d ≠ 0
    for d in C_d:
        constraints.append(cp.trace(C_d[d] @ X) >= tau)

    # Objective
    objective = cp.Maximize(tau)

    # Solve SDP
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4, max_iters=1000)

    if prob.status != cp.OPTIMAL:
        return None, None
    
    X_opt = X.value
    tau_opt = tau.value
    return X_opt, tau_opt

def round_sdp(X, n0):
    """Random hyperplane rounding of X to get a binary sequence."""

    V, w = eigh(X)
    V = V[:, w > 1e-8]
    w = w[w > 1e-8]
    if V.size == 0:
        # If no significant components, return random binary
        s = np.random.choice([-1, 1], size=n0)
    else:
        # Generate random vector in the span
        g = np.random.randn(V.shape[1])
        s = np.sign(V @ g)
    return s

def convert_to_h(s, n0):
    """Convert a ±1 sequence to 0/1, adjusting to sum = n0/2."""
    h = (s + 1) / 2
    sum_h = np.sum(h)
    # Make the sum exactly n0/2 by flipping bits
    if abs(sum_h - n0 / 2) < 1e-6:
        return h
    else:
        # Find indices where we can flip to reduce difference
        diff = int(sum_h - n0 / 2)
        if diff > 0:
            # Need to flip some 1s to 0s
            ones = np.where(h == 1)[0]
            flip_idx = ones[:diff]
        else:
            # Need to flip some 0s to 1s
            zeros = np.where(h == 0)[0]
            flip_idx = zeros[:abs(diff)]
        # Flip them
        h[flip_idx] = 1 - h[flip_idx]
        return h

def scale_to_target(h0, n_target):
    """Scale h0 to n_target by concatenation."""
    h = h0.copy()
    while len(h) < n_target:
        h = np.concatenate([h, 1 - h])
    if len(h) > n_target:
        h = h[:n_target]
    return h

def hill_climb(h_start, max_iter=10):
    """Simple hill climbing to improve h while enforcing sum constraint."""
    h = h_start.copy()
    prev_max_corr = full_correlation(h)
    for _ in range(max_iter):
        improved = False
        best_h = h.copy()
        best_corr = prev_max_corr
        # Scan all pairs (i, j) where h[i] = 1, h[j] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                # Try swapping h[i] and h[j]
                h_test = h.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                max_corr_test = full_correlation(h_test)
                if max_corr_test < best_corr:
                    best_corr = max_corr_test
                    best_h = h_test
                    improved = True
        if improved:
            h = best_h
            prev_max_corr = best_corr
        else:
            break
    return h
