def full_correlation(h):
    """Compute integer-valued correlation vector and its maximum."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return corr, max_corr

def build_shift_matrices(n):
    """Build Toeplitz shift matrices C_d for d in [-n+1, n-1] \\ {0}."""
    C_d_list = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        C_d = np.zeros((n, n))
        # Fill in the off-diagonals
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                C_d[i, j] = 1
                C_d[j, i] = 1
        C_d_list.append(C_d)
    return C_d_list
