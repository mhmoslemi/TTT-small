def evaluate_h(h_values):

    # Normalize h to have integral = 1
    n = len(h_values)

    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (1.0 / (current_sum * dx))
    
    # Clip to [0, 1] to respect constraints
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
