def project_onto_simplex(y, target_sum):
    """Project vector y onto the simplex with sum = target_sum"""
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the index where the projection should happen
    n = len(y)
    rho = np.argmax(sorted_y * np.arange(1, n+1) > (cumsum - target_sum))
    # Compute the threshold
    if rho == 0:
        theta = (cumsum[0] - target_sum) / 1
    else:
        theta = (cumsum[rho-1] - target_sum) / rho
    # Project
    projected = np.maximum(y - theta, 0)
    return projected

def compute_f_s(h, s):
    """Compute f_s(h) = sum h_i * (1 - h_{i+s}) for a given shift s"""
    if s >= 0:
        # Left shift h by s
        shifted_h = np.concatenate([np.zeros(s), h[:-s]])
    else:
        # Right shift h by |s|
        shifted_h = np.concatenate([h[-s:], np.zeros(-s)])
    return np.sum(h * (1 - shifted_h))

def compute_all_f(h):
    """Compute all f_s(h) values"""
    n = len(h)
    f_values = []
    for s in range(-n+1, n):
        f_values.append(compute_f_s(h, s))
    return np.array(f_values)

def smooth_max_p(h, p):
    """Compute smooth approximation of max in Lp norm"""
    f_vals = compute_all_f(h)
    return np.power(np.sum(np.power(np.abs(f_vals), p)), 1/p)

def smooth_max_gradient(h, p):
    """Compute gradient of smooth max with respect to h"""
    n = len(h)
    # Compute all f_s values
    f_vals = compute_all_f(h)
    # Compute weights
    w = np.power(np.abs(f_vals), p-1)
    # Compute the gradient for each component
    grad = np.zeros_like(h)
    for i in range(n):
        for s in range(-n+1, n):
            if s >= 0:
                if i+s < n:
                    grad[i] += w[s+n-1] * (1 - h[i+s])
                else:
                    grad[i] += w[s+n-1] * (1 - 0)  # out of bounds
            else:
                if i+s >= 0:
                    grad[i] += w[s+n-1] * (1 - h[i+s])
                else:
                    grad[i] += w[s+n-1] * (1 - 0)  # out of bounds
    
    # Divide by the norm to get smoother gradient
    norm_val = np.power(np.sum(np.power(np.abs(f_vals), p)), 1/p)
    if norm_val > 1e-12:
        grad /= np.power(norm_val, p-1)
    
    return grad

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def interpolate_h(h_coarse, n_fine):
    """Linearly interpolate coarse h to fine grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
