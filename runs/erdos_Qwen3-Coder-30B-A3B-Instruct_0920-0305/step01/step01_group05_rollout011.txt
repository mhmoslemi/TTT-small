import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    # Ensure sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        h = h * (1.0 / current_sum)
    
    # Clip to [0,1] to enforce constraints
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):
    """Perform local search on binary solutions"""
    dx = 2.0 / n
    half = n // 2
    best_sum = float('inf')
    best_bits = None
    
    start = time.time()
    
    # Precompute indices to avoid repeated scanning
    ones_idx = np.arange(half, dtype=int)
    zeros_idx = np.arange(half, n, dtype=int)
    
    while time.time() - start < time_limit:
        # Generate random binary vector with exactly half ones
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        # Evaluate initial solution
        curr_val = np.max(np.correlate(bits, 1 - bits, mode="full")) * dx
        
        # Local search loop
        improved = True
        while improved and time.time() - start < time_limit:
            improved = False
            
            # Try many random swaps
            for _ in range(n * 5):
                if time.time() - start >= time_limit:
                    break
                    
                # Pick random indices: one from ones, one from zeros
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Make swap
                bits[i], bits[j] = bits[j], bits[i]
                
                # Evaluate new solution
                new_val = np.max(np.correlate(bits, 1 - bits, mode="full")) * dx
                
                if new_val < curr_val - 1e-12:
                    curr_val = new_val
                    improved = True
                else:
                    # Revert swap
                    bits[i], bits[j] = bits[j], bits[i]
                    
                if time.time() - start >= time_limit:
                    break
                    
        # Check if this is our best solution so far
        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
            
        if best_sum < 0.379:
            break
    
    if best_bits is None:
        # Fallback to a working solution
        best_bits = np.zeros(n, dtype=int)
        best_bits[:half] = 1
        np.random.shuffle(best_bits)
        best_sum = np.max(np.correlate(best_bits, 1 - best_bits, mode="full")) * dx
    
    return best_bits, best_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points
    n_list = [42, 60, 80, 100]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n_points in n_list:
        if time.time() - start_time > budget_s - 10:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 5:
            break
            
        # Run binary local search only for selected n
        if n_points <= 100:
            try:
                # Binary local search
                local_search_time = min(remaining_time * 0.8, 300.0)  # Use 80% of remaining time
                h_binary, c5_binary = binary_local_search(n_points, local_search_time)
                
                if c5_binary < best_c5:
                    best_c5 = c5_binary
                    best_h = h_binary
                    best_n = n_points
                    
            except Exception as e:
                continue
            
    # Your preprocessing step to ensure best solution with best_n
    if best_h is None or best_c5 == float('inf'):
        # Fallback to some initial solution
        h_values = [0]*42
        h_values[:21] = [1]*21
        best_h = np.array(h_values)
        best_n = 42
        best_c5 = evaluate_h(best_h)
    
    # Final evaluation to make sure it's within constraints
    final_h = best_h.astype(float)
    final_c5 = evaluate_h(final_h)
    
    return (final_h.tolist(), final_c5, best_n)
