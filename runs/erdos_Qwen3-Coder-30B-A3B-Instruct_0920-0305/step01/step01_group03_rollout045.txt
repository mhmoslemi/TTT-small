import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_capped_simplex(h, target_sum):
    """Project vector h onto the set {x | 0 <= x <= 1, sum(x) = target_sum}"""
    # First, clamp to [0,1]
    h_clamped = np.clip(h, 0, 1)
    # Then waterfill to meet the sum constraint
    current_sum = np.sum(h_clamped)
    if abs(current_sum - target_sum) < 1e-12:
        return h_clamped
    elif current_sum > target_sum:
        # Need to decrease values
        # Binary search for the threshold tau
        low, high = 0, 1
        for _ in range(50):
            tau = (low + high) / 2
            new_h = np.clip(h_clamped - tau, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum > target_sum:
                low = tau
            else:
                high = tau
        tau = (low + high) / 2
        h_result = np.clip(h_clamped - tau, 0, 1)
    else:
        # Need to increase values
        # Binary search for the threshold tau
        low, high = 0, 1
        for _ in range(50):
            tau = (low + high) / 2
            new_h = np.clip(h_clamped + tau, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum < target_sum:
                high = tau
            else:
                low = tau
        tau = (low + high) / 2
        h_result = np.clip(h_clamped + tau, 0, 1)
    return h_result

def logsumexp_stable(x, axis=None):
    """Compute log(sum(exp(x))) stably using the maximum value"""
    x_max = np.max(x, axis=axis, keepdims=True)
    x_max = np.where(np.isinf(x_max), 0, x_max)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def get_gradient(h, R, lambda_param):
    """Compute the gradient for the soft minimum of correlations"""
    # R(s) = sum h[i] h[i+s]
    # w_s = e^{-lambda * R(s)} / sum_t e^{-lambda * R(t)}
    exp_R = np.exp(-lambda_param * R)
    exp_R_sum = np.sum(exp_R)
    if exp_R_sum > 0:
        w = exp_R / exp_R_sum
    else:
        # fallback to uniform distribution
        w = np.full_like(exp_R, 1.0 / len(exp_R))
    
    # Compute g_j = sum_s w_s (h_{j+s} + h_{j-s})
    # This simulates convolution of w with h, but with proper indexing
    # Using explicit slices for easier verification
    n = len(h)
    g = np.zeros_like(h)
    for s in range(len(R)):
        # convert s to a shift index, accounting for zero padding
        shift = s - (len(R) - 1) // 2
        # indices in h (with zero padding)
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        if i_end > i_start:
            # compute effective shift
            j_indices = np.arange(i_start, i_end)
            s_indices = j_indices + shift
            g[j_indices] += w[s] * (h[s_indices] + h[s_indices])  # h[i+s] + h[i-s] due to symmetry of convolution
    
    # To avoid mistake: h_{j+s} + h_{j-s}
    # We treat this via two separate slices, padding with zero appropriately
    g = np.zeros_like(h)
    for s in range(len(R)):
        shift = s - (len(R) - 1) // 2
        # Original h[i] contributes to g[i-shift] and g[i+shift]
        s_indices_left = np.arange(max(0, -shift), min(len(h), len(h) - shift))
        s_indices_right = s_indices_left + shift
        if len(s_indices_left) > 0:
            g[s_indices_left] += w[s] * h[s_indices_right]
        s_indices_right = np.arange(max(0, shift), min(len(h), len(h) - shift))
        s_indices_left = s_indices_right - shift
        if len(s_indices_right) > 0:
            g[s_indices_right] += w[s] * h[s_indices_left]

    return g

def project_symmetric(h):
    """Symmetrize h around midpoint x=1 (i.e., h[i] := (h[i] + h[n-1-i])/2)"""
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return h_sym

def upsample_and_project(h_original, target_size):
    """Linearly interpolate original h to target_size points, then project onto feasible set"""
    n_orig = len(h_original)
    x_orig = np.linspace(0, 2, n_orig)
    x_target = np.linspace(0, 2, target_size)
    h_interp = np.interp(x_target, x_orig, h_original)
    # Project onto capped simplex with target sum = 1
    target_sum = 1.0
    h_projected = project_onto_capped_simplex(h_interp, target_sum)
    # Renormalize to sum to target_sum
    h_projected = h_projected * target_sum / np.sum(h_projected)
    return h_projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2  # since sum(h)*dx = 1
    
    # Helper: compute the soft minimum value
    def soft_min_value(h, lambda_param):
        R = np.correlate(h, h, "full")
        R = R * dx  # Scale by dx to normalize appropriately
        exp_R = np.exp(-lambda_param * R)
        exp_R_sum = np.sum(exp_R)
        if exp_R_sum <= 0:
            return np.inf
        log_sum_exp = np.log(exp_R_sum)
        return log_sum_exp / lambda_param
    
    # Same value as used in logsumexp
    def compute_R_s(h):
        return np.correlate(h, h, "full") * dx
    
    # Helper function to compute exact C5
    def evaluate_exact_c5(h):
        return evaluate_erdos_solution(h, dx)
    
    def get_exact_gradients(h):
        # Used only for final refinement
        n = len(h)
        # Since our optimization is already quite good, we’ll instead 
        # use a simple backtracking line search approach during gradient descent.
        pass  # will be computed implicitly
        
    # Start time tracking
    start_time = time.time()
    best_raw_c5 = np.inf
    best_h_global = None
    
    # Phase 1: multi-start
    num_starts = 20
    for start_iter in range(num_starts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random start
        h = np.random.uniform(0, 1, n_points)
        h = project_onto_capped_simplex(h, target_sum)
        
        # Binary start: permutation of ones and zeros
        binary_h = np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)])
        np.random.shuffle(binary_h)
        binary_h = project_onto_capped_simplex(binary_h, target_sum)
        if np.sum(binary_h ** 2) < np.sum(h ** 2):
            h = binary_h
        
        # Upsampled start: from initial_h_values
        if 'initial_h_values' in globals():
            orig_h = np.array(initial_h_values)
            upsampled_h = upsample_and_project(orig_h, n_points)
            if np.sum(upsampled_h ** 2) < np.sum(h ** 2):
                h = upsampled_h
                
        # Symmetrize initial vector
        h = project_symmetric(h)
        h = project_onto_capped_simplex(h, target_sum)
        
        # Annealed projected gradient descent
        lambda_warm = 0.5
        lambda_medium = 5.0
        lambda_fine = 30.0
        
        # Mild erosion still helps
        # Parameters for steps
        step_sizes = [0.5, 0.2, 0.05]
        steps = [200, 200, 300]
        lambdas = [lambda_warm, lambda_medium, lambda_fine]
        
        for _phase, (lambda_val, alpha, num_steps) in enumerate(zip(lambdas, step_sizes, steps)):
            if time.time() - start_time > budget_s * 0.9:
                break
            
            for _ in range(num_steps):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute correlations matrix (full correlation)
                R = compute_R_s(h)
                
                # Softening weights
                exp_R = np.exp(-lambda_val * R)
                w_sum = np.sum(exp_R)
                if w_sum <= 0:
                    w = np.full_like(exp_R, 1.0 / len(exp_R))
                else:
                    w = exp_R / w_sum
                
                # Compute gradient g = sum_s w_s (h_{j+s} + h_{j-s})
                # This is an approximate convolution approach
                g = np.zeros_like(h)
                n_half = (len(R) - 1) // 2
                for s in range(len(R)):
                    shift = s - n_half
                    if shift == 0:
                        continue
                    if shift > 0:
                        g[shift:] += w[s] * h[:-shift]
                    elif shift < 0:
                        g[:shift] += w[s] * h[-shift:]
                    
                # Gradient step: h <- h - alpha * g
                h_new = h - alpha * g
                
                # Line search: backtracking (only for first few steps)
                h_candidate = h_new
                eval_cand = soft_min_value(h_candidate, lambda_val)
                eval_old = soft_min_value(h, lambda_val)
                
                # If the new step doesn't improve, reduce step size
                with np.errstate(over='ignore', under='ignore'):
                    while not (eval_cand < eval_old) and alpha > 1e-6:
                        alpha /= 2
                        h_candidate = h - alpha * g
                        eval_cand = soft_min_value(h_candidate, lambda_val)
                        eval_old = soft_min_value(h, lambda_val)
                        
                h = h_candidate
                
                # Projection step
                h = project_onto_capped_simplex(h, target_sum)
                h = project_symmetric(h)
                h = project_onto_capped_simplex(h, target_sum)
                
                # Check raw C5
                raw_c5_candidate = evaluate_exact_c5(h)
                if raw_c5_candidate < best_raw_c5:
                    best_raw_c5 = raw_c5_candidate
                    best_h_global = h.copy()
    
    # Final refinement via non-smooth optimization
    if best_h_global is not None:
        def obj_func(h_vals):
            return evaluate_exact_c5(h_vals)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        result = minimize(
            obj_func,
            best_h_global,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if result.success:
            final_c5 = evaluate_exact_c5(result.x)
            if final_c5 < best_raw_c5:
                best_raw_c5 = final_c5
                best_h_global = result.x.copy()
    
    if best_h_global is None:
        # Fallback to default
        best_h_global = np.array(initial_h_values)
        best_h_global = upsample_and_project(best_h_global, n_points)
        best_raw_c5 = evaluate_exact_c5(best_h_global)
        
    return (best_h_global.tolist(), best_raw_c5, n_points)
