import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: sum = target_sum, 0 <= h <= 1"""
    # Simple water-filling approach
    h = np.asarray(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Water-filling approach to adjust to target_sum
    # We'll iterate until we get the right sum via bisection search
    if current_sum > target_sum:
        # Need to decrease values
        tol = 1e-12
        lower = 0.0
        upper = 1.0
        while upper - lower > tol:
            mid = (lower + upper) / 2
            test_h = np.clip(h - mid, 0, 1)
            test_sum = np.sum(test_h)
            if test_sum >= target_sum:
                lower = mid
            else:
                upper = mid
        h = np.clip(h - lower, 0, 1)
    else:
        # Need to increase values
        tol = 1e-12
        lower = 0.0
        upper = 1.0
        while upper - lower > tol:
            mid = (lower + upper) / 2
            test_h = np.clip(h + mid, 0, 1)
            test_sum = np.sum(test_h)
            if test_sum <= target_sum:
                lower = mid
            else:
                upper = mid
        h = np.clip(h + lower, 0, 1)
    
    # Adjust to exact sum
    h = h * target_sum / np.sum(h)
    return h

def apply_symmetry(h):
    """Symmetrize h around x = 1 by averaging h[i] + h[n-1-i]"""
    n = len(h)
    h_sym = np.zeros_like(h)
    for i in range(n):
        j = n - 1 - i
        if i <= j:
            h_sym[i] = (h[i] + h[j]) / 2
        else:
            break
    return h_sym

def logsumexp_stable(log_weights):
    """Numerically stable log-sum-exp"""
    max_log = np.max(log_weights)
    return max_log + np.log(np.sum(np.exp(log_weights - max_log)))

def compute_gradient(h, log_R, lambda_param):
    """Compute gradient of surrogate function with soft minimum"""
    n = len(h)
    # Compute weights
    log_w = -lambda_param * log_R
    log_w = log_w - logsumexp_stable(log_w)  # Normalize to have max log_w = 0
    w = np.exp(log_w)
    
    # Compute g = \sum_s w_s (h_{j+s} + h_{j-s})
    g = np.zeros_like(h)
    
    # For each step s in correlation
    R_len = len(log_R)
    for s in range(R_len):
        # Convert full correlation index back to actual offset
        offset = s - (R_len - 1) // 2
        if offset == 0:
            g += 2 * w[s] * h  # Since h_{j+0} = h_{j-0} = h_j
        else:
            indices_forward = np.arange(n)
            indices_backward = np.arange(n)
            if offset > 0:
                indices_forward = indices_forward[offset:]
                indices_backward = indices_backward[:-offset]
                g[indices_forward] += w[s] * h[indices_backward]
                g[indices_backward] += w[s] * h[indices_forward]
            else:
                indices_forward = indices_forward[:offset]
                indices_backward = indices_backward[-offset:]
                g[indices_backward] -= w[s] * h[indices_forward]
                g[indices_forward] -= w[s] * h[indices_backward]
    
    return g

def backtracking_line_search(h, g, lambda_param, target_sum, dx):
    """Backtracking line search for gradient step"""
    alpha = 1.0
    max_alpha = 1.0
    initial_phi = phi_lambda(h, lambda_param)
    tolerance = 1e-8
    
    for _ in range(5):
        h_new = h + alpha * g
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        # Symmetrize after projection
        h_new = apply_symmetry(h_new)
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        phi_new = phi_lambda(h_new, lambda_param)
        if phi_new <= initial_phi - tolerance:
            return h_new, alpha
        alpha *= 0.5
    
    return h, max_alpha

def phi_lambda(h, lambda_param):
    """Surrogate function Phi_lambda"""
    R = np.correlate(h, h, "full")
    log_R = np.log(R + 1e-12)  # Add small to avoid log(0)
    # Using numerical stable logsumexp
    return (1.0 / lambda_param) * logsumexp_stable(-lambda_param * log_R)
