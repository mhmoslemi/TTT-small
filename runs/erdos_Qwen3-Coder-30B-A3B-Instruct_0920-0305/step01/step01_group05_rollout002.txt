import numpy as np

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h):
    """Project h onto the feasible set: [0,1]^n and sum = n/2"""
    # Sort h in descending order to perform the projection efficiently
    h_sorted = np.sort(h)[::-1]
    n = len(h_sorted)
    
    # Find the projection parameter 'alpha' using bisection
    # We want sum(max(0, h_sorted[i] - alpha)) = n/2
    # This function is decreasing in alpha
    
    def sum_after_projection(alpha):
        return np.sum(np.maximum(0, h_sorted - alpha))
    
    alpha_low = h_sorted[-1] - 1.0  # safe lower bound
    alpha_high = h_sorted[0] + 1.0  # safe upper bound
    tol = 1e-10
    
    # Bisection search
    while alpha_high - alpha_low > tol:
        alpha_mid = (alpha_low + alpha_high) / 2.0
        sum_val = sum_after_projection(alpha_mid)
        
        if sum_val > n / 2:
            alpha_low = alpha_mid
        else:
            alpha_high = alpha_mid
    
    alpha = (alpha_low + alpha_high) / 2.0
    
    # Apply projection
    h_proj = np.maximum(0, h - alpha)
    return h_proj

def shifted_sum(h, s):
    """Calculate f_s(h) = sum over i of h_i * (1 - h_{i+s}) for a given shift s"""
    n = len(h)
    if s >= n or s <= -n:
        return 0.0
    if s >= 0:
        return np.sum(h[:n-s] * (1 - h[s:]))
    else:
        return np.sum(h[-s:] * (1 - h[:-s]))

def compute_f(h):
    """Compute f_s(h) for all shifts s = -(n-1) to (n-1)"""
    n = len(h)
    fs = np.zeros(2 * n - 1)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        fs[i] = shifted_sum(h, s)
    return fs

def compute_phips(h, p):
    """Compute Phi_p(h) = (sum_s f_s^p)^(1/p)"""
    fs = compute_f(h)
    fs = np.abs(fs)
    if np.all(fs == 0):
        return 0.0
    return np.power(np.sum(np.power(fs, p)), 1.0 / p)

def compute_grad(h, p):

    # For each shift s in range(-n+1, n):
    #   h_shifted_s is h shifted by s (with zero padding)
    #   w_s = f_s^p-1
    
    # We'll use faster np.correlate-based computation:
    w_list = np.abs(fs) ** (p - 1)

    w_list = np.where(fs != 0, w_list, 0.0)
    
    # Compute all contributions to the gradient
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        s = i - (n - 1)
        if s == 0:
            continue  # already handled
            
        # This is an optimized way of computing derivative contributions:
        # but it's easier conceptually to just directly model it.
        
        # Use proper gradient computation based on the outer product like
        # grad_i += w_s * (1 - h_{i+s}) if s > 0, or (1 - h_{i+s}) if s < 0
        
        # Simplified and vectorized implementation:
        if s >= 0:

        # but we must be careful: h_shifted might be NaN or incorrect for edge cases

            s_grad[s:] = 1 - h[:-s]
        else:
            s_grad[:s] = 1 - h[-s:]

        # For simplicity:

        # Adjacent terms: for a specific s let's rebuild the contributions properly:
        # compute for all shifts how they contribute to the gradient of h_i
        # This grinds all f_s=h_i*(1-h_{i+s}) terms:
        # Let w_s = f_s^{p-1} 
        # dPhi/dh_i = sum_s w_s * (1 - h_{i+s} - h_{i-s})
        # We'll make a matrix-of-2-shifts k where 
        # k[i, :] gives the contributions to this i from all shifts
        # [1, 1, ..., 1], which is generic but leads to
        # complex indexing
        
        # A quick rework: straight scalar loop but using NumPy's vectorization efficiently
        # We simulate f_s = h_i * (1 - h_{i+s})  over i and all s
        # Use np.roll and complex indexing where necessary
        
        # easier:
        weights = w_list[i]
        if s >= 0:
            left_mask = slice(None, -s)
            right_mask = slice(s, None)
            h_right = h[right_mask]
            h_left = h[left_mask]
            one_h = np.ones_like(h)
            one_h[right_mask] = 0
            grad[left_mask] += weights * (one_h[left_mask] - h_right)
        else:  # s < 0
            s_abs = -s
            left_mask = slice(s_abs, None)
            right_mask = slice(None, s_abs)
            h_left = h[left_mask]
            h_right = h[right_mask]
            # The effect of (1-h_{i+s})
            grad[left_mask] += weights * (1 - h_left)

    # Add penalty term: lambda * 2 * (sum(h) - n/2)
    grad += 2 * lambda_penalty * (np.sum(h) - n / 2)
    return grad

def project_grad_h(h, grad, step_size):
    """Perform projected gradient update with projection to constraints."""
    # Perform gradient step
    h_new = h - step_size * grad
    # Project to feasible set
    h_proj = project_to_feasible(h_new)
    return h_proj

        if i % 500 == 0 and i > 0:

        # Compute Phi_p and its gradient
        Phi_p = compute_phips(h, p)
        if Phi_p == 0:
            grad = np.zeros_like(h)
        else:
            try:
                grad = compute_grad(h, p)
            except:
                # fallback gradient implementation
                grad = np.zeros_like(h)
                # integrate via basic loop here
                for k in range(n):
                    sum_grad = 0.0
                    for s in range(-n+1, n):
                        s_idx = k + s
                        if 0 <= s_idx < n:
                            sum_grad += (1 - h[s_idx]) * (abs(shifted_sum(h, s)) ** (p - 1))
                        else:
                            pass  # zero-padded case implicitly
                    grad[k] = 2 * lambda_penalty * (np.sum(h) - n / 2) + sum_grad / (Phi_p ** (p - 1))

        h = project_grad_h(h, grad, step_size)
        if i % 100 == 0:
            current_c5 = evaluate_h(h)
            if current_c5 < 0.380:
                break  # early stop if good enough

    return h

def interpolate_h(h_coarse, n_fine):
    """Linearly interpolate a coarse h to fine grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
