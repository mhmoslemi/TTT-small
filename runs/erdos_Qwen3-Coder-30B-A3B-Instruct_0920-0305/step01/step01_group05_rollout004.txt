import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def linear_interpolate(h, n_new):
    """Linearly interpolate h from old grid to new grid of size n_new"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    # Clamp to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Renormalize to sum = n_new/2
    h_new = h_new * (n_new / 2) / np.sum(h_new)
    return h_new

def build_shift_masks(n):
    """Build forward and backward masks for shifts"""
    masks_fwd = {}
    masks_bwd = {}
    for s in range(-(n-1), n):
        masks_fwd[s] = np.arange(max(0, -s), min(n, n-s))
        masks_bwd[s] = np.arange(max(0, s), min(n, n+s))
    return masks_fwd, masks_bwd

def compute_gradient_gd(h, s, masks_fwd, masks_bwd):
    """Compute gradient of g_s^d at h"""
    grad = np.zeros_like(h)
    # Forward indices
    if s >= 0:
        if len(masks_fwd[s]) > 0:
            grad[masks_fwd[s]] += h[masks_bwd[s]] + h[masks_fwd[s]]
    else:
        if len(masks_bwd[s]) > 0:
            grad[masks_bwd[s]] += h[masks_fwd[s]] + h[masks_bwd[s]]
    return grad

    dx = 2.0 / n
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Build the constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]
    g_c_s_vals = []
    g_d_s_vals = []
    
    for s in range(-(n-1), n):
        # Convex part of g_s^c(h) = sum(h) + 0.5*sum(h[i]^2) + 0.5*sum(h[i+s]^2)
        grad_gd = compute_gradient_gd(h_start, s, masks_fwd, masks_bwd)
        # Compute value of g_s^d at h_start
        g_d_at_h = 0.5 * np.sum((h_start + np.roll(h_start, s)) ** 2)
        # Linearized concave part -g_s^d(h_start) - grad_gd^T * (h - h_start)
        linearized_gd = g_d_at_h + grad_gd @ (h - h_start)
        # Convex quadratic g_s^c(h) - (linearized_gd)
        g_c_s = cp.sum(h) + 0.5*(cp.sum_squares(h[masks_fwd[s]]) + cp.sum_squares(h[masks_bwd[s]]))
        constraints.append(tau >= g_c_s - linearized_gd)
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Solve with ECOS
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver, abstol=1e-8, reltol=1e-8)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        return h_sol
    else:
        # Fallback to original if solve failed
        return h_start.copy()
