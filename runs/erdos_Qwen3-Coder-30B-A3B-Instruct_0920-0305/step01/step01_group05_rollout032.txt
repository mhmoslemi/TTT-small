def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_feasible(x, target_sum):
    """Project x onto the feasible set: [0,1]^n with sum = target_sum"""
    # This uses a simple bisection approach
    low, high = 0.0, 1.0
    for _ in range(100):
        alpha = (low + high) / 2.0
        candidate = np.clip(x - alpha, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate > target_sum:
            low = alpha
        else:
            high = alpha
    alpha = (low + high) / 2.0
    return np.clip(x - alpha, 0, 1)

def compute_fs(h, s):
    """Compute f_s(h) for shift s"""
    n = len(h)
    if s >= 0:
        # Right shift
        shifted = np.pad(h[s:], (0, s), mode='constant', constant_values=0)
    else:
        # Left shift
        shifted = np.pad(h[:s], (-s, 0), mode='constant', constant_values=0)
    return np.dot(h, 1 - shifted)

def compute_all_fs(h):
    """Compute all f_s(h) for s = -(n-1) to (n-1)"""
    n = len(h)
    result = np.empty(2 * n - 1)
    for s in range(1 - n, n):
        result[s + n - 1] = compute_fs(h, s)
    return result

def gradient_of_phi_p(h, p):

    all_fs_abs = np.where(all_fs_abs < 1e-12, 1, all_fs_abs)

    # Compute the full gradient analytically
    grad = np.zeros(n)
    for s in range(1 - n, n):
        if s >= 0:
            # Right shift
            shifted = np.pad(h[s:], (0, s), mode='constant', constant_values=0)
        else:
            # Left shift
            shifted = np.pad(h[:s], (-s, 0), mode='constant', constant_values=0)
        # Avoid out-of-bounds
        shifted = shifted[:n]
        # derivative of each term is (1 - h_{k+s} - h_{k-s})
        if s >= 0:
            grad[:n-s] += weights[s + n - 1] * (1 - shifted[:n-s])
            grad[s:] += weights[s + n - 1] * (-shifted[s:])
        else:
            grad[:n+s] += weights[s + n - 1] * (-shifted[:n+s])
            grad[-s:] += weights[s + n - 1] * (1 - shifted[-s:])
            
    return grad / (np.sum(weights ** (1 / p)) ** (p - 1))

def update_h(h, p, penalty, eta=0.01):
    """Update h using projected gradient step with smooth max p-norm"""
    grad = gradient_of_phi_p(h, p)
    # Add penalty term for the integral constraint
    grad += 2 * penalty * (np.sum(h) - len(h) / 2)
    
    # Take gradient step
    h_new = h - eta * grad
    # Project back onto feasible set
    h_new = project_feasible(h_new, len(h) / 2)
    return h_new
