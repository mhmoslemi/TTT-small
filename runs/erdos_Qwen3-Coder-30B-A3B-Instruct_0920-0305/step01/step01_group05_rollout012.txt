def project_onto_simplex(y, target_sum):
    """Project vector y onto the simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute the cumulative sum
    cumsum = np.cumsum(sorted_y)
    # Find the index where the projection happens
    n = len(y)
    tmp = (cumsum - target_sum) / np.arange(1, n + 1)
    rho = np.argmax(sorted_y > tmp)
    # Compute the threshold
    theta = (cumsum[rho] - target_sum) / (rho + 1)
    # Compute the projection
    x = np.maximum(y - theta, 0)
    return x

def compute_f_s(h, s):
    """Compute f_s(h) = sum_{i} h_i (1 - h_{i+s}) for a given shift s"""
    n = len(h)
    if s >= 0:
        # Left shift
        h_left = np.concatenate([h[s:], np.zeros(s)])
        h_right = h
    else:
        # Right shift
        h_right = np.concatenate([np.zeros(-s), h[:n+s]])
        h_left = h
    return np.sum(h_left * (1 - h_right))

def compute_all_fs(h):
    """Compute all f_s(h) for s in range(-n+1, n)"""
    n = len(h)
    fs = np.zeros(2*n - 1)
    for s in range(-n+1, n):
        fs[s + n - 1] = compute_f_s(h, s)
    return fs

def compute_gradient(h, p, lambda_sum):
    """Compute the gradient of J_p = Phi_p(h) + lambda_sum * (sum(h) - n/2)^2"""
    n = len(h)

    # Compute the p-norm terms
    abs_fs = np.abs(fs)
    # Avoid division by zero for very small values
    abs_fs_clean = np.where(abs_fs < 1e-10, 1e-10, abs_fs)
    # Compute weights for gradient computation
    weights = abs_fs_clean ** (p - 1)
    # Compute the component-wise gradient
    grad = np.zeros(n)
    # Vectorized gradient computation using shifts
    for s in range(-n+1, n):
        s_idx = s + n - 1
        w = weights[s_idx]
        # Compute shifted versions
        if s >= 0:
            h_shifted = np.concatenate([h[s:], np.zeros(s)])
            h_shifted_reverse = h
        else:
            h_shifted = np.concatenate([np.zeros(-s), h[:n+s]])
            h_shifted_reverse = h
        # The contribution from this shift
        grad_term = w * (1 - h_shifted_reverse - h_shifted)
        if s >= 0:
            grad[s:] += grad_term[:-s] if s > 0 else grad_term
        else:
            grad[:s] += grad_term[-s:] if s < 0 else grad_term
    # Add penalty term-gradient
    grad += 2 * lambda_sum * (np.sum(h) - n/2) * np.ones(n)
    return grad

def smooth_max(h, p):
    """Compute Phi_p(h) = (sum(f_s^p))^(1/p)"""
    fs = compute_all_fs(h)
    abs_fs = np.abs(fs)
    # Avoid numerical issues
    abs_fs = np.where(abs_fs < 1e-10, 1e-10, abs_fs)
    return np.power(np.sum(np.power(abs_fs, p)), 1/p)

def optimize_one_p(h_init, p, lambda_sum, n_iter, step_size=1e-3):
    """Optimize with fixed p"""
    h = h_init.copy()
    dx = 2.0 / len(h)
    last_obj_val = np.inf
    for i in range(n_iter):
        grad = compute_gradient(h, p, lambda_sum)
        # Update
        h = h - step_size * grad
        # Project
        h = project_onto_simplex(h, len(h) / 2)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Compute objective only every few steps for efficiency
        if i % 10 == 0:
            obj_val = smooth_max(h, p) + lambda_sum * (np.sum(h) - len(h)/2) ** 2
            if abs(obj_val - last_obj_val) < 1e-8:
                break
            last_obj_val = obj_val
    return h

def interpolate_h(h_coarse, n_fine):
    """Interpolate coarse h to fine grid using linear interpolation"""
    n_coarse = len(h_coarse)
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + dx_coarse / 2
    x_fine = np.linspace(0, 2, n_fine, endpoint=False) + dx_fine / 2
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
