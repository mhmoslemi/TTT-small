import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (1.0 / (h_sum * dx))
        h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_c5_binary(bits):
    """Evaluate C5 for binary vector"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
