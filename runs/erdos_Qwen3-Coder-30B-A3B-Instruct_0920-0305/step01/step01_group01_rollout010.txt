import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use n_points = 48 for faster resolution 
    n_points = 48
    dx = 2.0 / n_points
    
    # Define variables
    x = cp.Variable(n_points, boolean=True)
    y = cp.Variable((n_points, n_points), boolean=True)
    t = cp.Variable()  # Represents max mismatch
    
    # Constraints for x: sum = N/2
    constraints = [cp.sum(x) == n_points // 2]
    
    # Linearization constraints for y[i,k] = x[i] * x[i+k]
    # Note: k from 1 to n_points-1
    for k in range(1, n_points):
        for i in range(n_points - k):
            # y[i,k] <= x[i]
            constraints.append(y[i, k] <= x[i])
            # y[i,k] <= x[i+k]
            constraints.append(y[i, k] <= x[i + k])
            # y[i,k] >= x[i] + x[i+k] - 1
            constraints.append(y[i, k] >= x[i] + x[i + k] - 1)
    
    # Constraints for each positive shift k 
    for k in range(1, n_points):
        # sum_{i=0}^{N-1-k} x_i
        S_pos_k = cp.sum(x[:n_points - k])
        # sum_{i=0}^{N-1-k} y_{i,k}
        A_k = cp.sum(y[:n_points - k, k])
        mismatch_pos_k = S_pos_k - A_k
        constraints.append(mismatch_pos_k <= t)
        
        # For negative shift -k: sum_{i=0}^{N-1-k} x_{i+k}
        S_neg_k = cp.sum(x[k:])
        # sum_{i=0}^{N-1-k} y_{i,k} (same indices)
        A_neg_k = cp.sum(y[:n_points - k, k])
        mismatch_neg_k = S_neg_k - A_neg_k
        constraints.append(mismatch_neg_k <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Construct and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use CBC solver through cvxpy
    start_time = time.time()
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=budget_s*0.9)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_IIS]:
