def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_simplex(y, target_sum):
    """Project y onto the set {x : sum(x) = target_sum, 0 <= x <= 1}."""
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    
    # Find the index where the projection happens
    tmp = (1.0 / target_sum) * (target_sum - cumsum + sorted_y)
    idx = np.argmax(tmp >= sorted_y)
    if tmp[idx] < sorted_y[idx]:
        idx += 1

    # Compute the threshold
    theta = (cumsum[idx-1] - target_sum) / idx if idx > 0 else (sorted_y[0] - target_sum)

    # Project
    x = np.maximum(y - theta, 0)
    x = np.minimum(x, 1)  # Ensure constraints 

    return x

def compute_f(h, s):
    """Compute f_s(h) = sum h_i * (1 - h_{i+s}) where out-of-bounds indices are 0."""
    n = len(h)
    h_padded = np.pad(h, (max(0, s), max(0, -s)), mode='constant', constant_values=0)
    if s > 0:
        shifted = h_padded[s:]
    else:
        shifted = h_padded[:n]
    return np.sum(h * (1 - shifted))

def compute_fs_vec(h):
    """Compute all f_s(h) values for s = 1-n to n-1."""
    n = len(h)
    fs = np.empty(2*n - 1)
    for s in range(1-n, n):
        fs[s+n-1] = compute_f(h, s)
    return fs

def smooth_max(h, p=12):
    """Approximates max_s |f_s(h)| using p-norm."""
    fs = compute_fs_vec(h)
    return np.power(np.sum(np.power(np.abs(fs), p)), 1.0/p)

def grad_smooth_max(h, p=12):
    """Gradient of smooth_max, not the scaled version."""
    n = len(h)
    fs = compute_fs_vec(h)
    fs_abs = np.abs(fs)
    
    # Handle case where fs = 0
    safe_fs = np.where(fs_abs == 0, 1e-15, fs_abs)
    
    # Precompute w_s = sign(f_s) * abs(f_s)**(p-1) = sign(f_s) * (abs(f_s)**(p-1)) 
    w = np.sign(fs) * np.power(safe_fs, p-1)
    
    grad = np.zeros(n)
    for s in range(1-n, n):
        if s > 0:
            shifted_w = np.pad(w[n-1+s:], (0, s), mode='constant', constant_values=0)
        else:
            shifted_w = np.pad(w[:n-1+s], (-s, 0), mode='constant', constant_values=0)
        # gradient contribution of single f_s
        grad_i = 0
        for i in range(n):
            if i - s >= 0 and i - s < n:
                grad_i += shifted_w[i] * (1 - h[i-s])
            else:
                grad_i += shifted_w[i]
        grad += grad_i * w[n-1+s] / np.power(np.sum(np.power(fs_abs, p)), 1.0/p - 1)
    
    return grad

def projected_gradient_descent(h0, p_vals=[2, 4, 8, 12], max_iter=2000, tol=1e-8, penalty=1000.0):
    """Projected gradient descent with continuation on p."""
    h = h0.copy()
    h = project_to_simplex(h, len(h)/2)
    dx = 2/len(h)
    
    for p in p_vals:
        eta = 0.01
        prev_obj = float('inf')
        for _ in range(max_iter):
            smooth_val = smooth_max(h, p)
            fs = compute_fs_vec(h)
            grad = grad_smooth_max(h, p)
            grad += 2*penalty*(np.sum(h) - len(h)/2)
            
            # Step
            h_new = h - eta*grad
            h_new = project_to_simplex(h_new, len(h)/2)
            
            # Update h
            h = h_new
            obj = smooth_val + penalty*(np.sum(h) - len(h)/2)**2
            
            # Early stopping if no progress
            if abs(obj - prev_obj) < tol:
                break
            prev_obj = obj
            
            # Adaptive eta
            if obj > prev_obj:
                eta *= 0.5
            if eta < 1e-8:
                break
    
    # Final projection to be exact on sum constraint
    h = project_to_simplex(h, len(h)/2)
    return h
