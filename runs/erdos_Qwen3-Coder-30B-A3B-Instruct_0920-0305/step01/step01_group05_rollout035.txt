import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def build_shift_masks(n):
    """Build forward and backward masks for each shift"""
    masks_fwd = {}
    masks_bwd = {}
    for s in range(-(n-1), n):
        mask_fwd = []
        mask_bwd = []
        for i in range(n):
            if i + s < n and i + s >= 0:
                mask_fwd.append(i)
            if i - s >= 0 and i - s < n:
                mask_bwd.append(i)
        masks_fwd[s] = np.array(mask_fwd, dtype=int)
        masks_bwd[s] = np.array(mask_bwd, dtype=int)
    return masks_fwd, masks_bwd

def compute_gradient_gd(h, s, masks_fwd, masks_bwd):
    """Compute the gradient of g_s^d at h for shift s"""
    grad = np.zeros_like(h)
    if len(masks_fwd[s]) > 0:
        grad[masks_fwd[s]] += h[masks_fwd[s]] + h[masks_fwd[s] + s]
    if len(masks_bwd[s]) > 0:
        grad[masks_bwd[s]] += h[masks_bwd[s]] + h[masks_bwd[s] - s]
    return grad

def compute_gd_value(h, s, masks_fwd, masks_bwd, h_prev):
    """Compute value g_s^d(h_prev) + grad_s^T (h - h_prev)"""
    h_s = np.roll(h_prev, s)
    term = (h + h_s) ** 2
    return 0.5 * np.sum(term)

def solve_ccp_subproblem(h_prev, n, masks_fwd, masks_bwd, max_iter=30, tol=1e-6):
    """Solve the CCP subproblem using CVXPY"""
    h = cp.Variable(n)
    tau = cp.Variable()
    dx = 2.0 / n
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]
    
    # Add constraints for each shift s
    for s in range(-(n-1), n):
        # Compute g_s^c(h)
        g_s_c = cp.sum(h) + 0.5 * cp.sum_squares(h[masks_fwd[s]]) + 0.5 * cp.sum_squares(h[masks_bwd[s]])
        
        # Compute g_s^d(h_prev) + grad^T (h - h_prev)
        # Specifically compute the linearized term: g_s^d(h_prev) + sum_i (grad_i * (h_i - h_prev_i))
        grad = compute_gradient_gd(h_prev, s, masks_fwd, masks_bwd)
        g_s_d_linear = 0.5 * np.sum((h_prev + np.roll(h_prev, s)) ** 2) + np.sum(grad * (h - h_prev))
        
        constraints.append(tau >= g_s_c - g_s_d_linear)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        h_new = h.value
    except:
        # If solver fails, return previous solution
        h_new = h_prev
    return h_new

def coordinate_descent_polish(h, n, masks_fwd, masks_bwd, dx):
    """Polish solution using coordinate descent"""
    h = h.copy()
    old_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    h_sum = np.sum(h)
    for sweep in range(5):
        # Shuffle indices for random order
        indices = np.random.permutation(n)
        for i in indices:
            # Values to evaluate
            candidates = []
            current_val = h[i]
            
            # Temporarily set h[i] = 0
            h[i] = 0
            h_sum -= current_val
            # Update h_sum to maintain constraint
            c5_0 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
            candidates.append((0, c5_0, h_sum))
            
            # Temporarily set h[i] = 1
            h[i] = 1
            h_sum += (1 - current_val)
            c5_1 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
            candidates.append((1, c5_1, h_sum))
            
            # Check boundary values directly
            # For continuous range, find the optimum within interval where derivative is zero
            # Simple piecewise-linear approximation (acts on function of h[i])
            h[i] = current_val
            h_sum += current_val - 1
            
            if len(candidates) > 2:
                candidates.sort(key=lambda x: x[1])
                new_h_i = candidates[0][0]
                new_h_sum = candidates[0][2]
                h[i] = new_h_i
                h_sum = new_h_sum
            
            # Restore h[i]
            h[i] = current_val
            h_sum += current_val - 0
            
        new_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        if abs(new_c5 - old_c5) < 1e-6:
            break
        old_c5 = new_c5
    return h
