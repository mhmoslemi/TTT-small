import numpy as np
from scipy.optimize import minimize
import time

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x : lower <= x_i <= upper, sum x_i = target_sum}"""
    # Shift to handle lower bounds
    v_shift = v - lower
    target_shift = target_sum - lower * len(v)
    
    # Sort v_shift in descending order
    idx_sorted = np.argsort(v_shift)[::-1]
    v_sorted = v_shift[idx_sorted]
    
    # Find threshold
    cumsum = np.cumsum(v_sorted)
    threshold = (cumsum - target_shift) / np.arange(1, len(v_sorted) + 1)
    # find largest threshold such that threshold <= v_sorted[i] for all i
    # threshold should be smallest which satisfies the condition
    # because v_sorted is decreasing, we look at v_sorted[i] - threshold >= 0
    # which means threshold <= v_sorted[i]
    max_thres_idx = np.argmax(v_sorted >= threshold)
    if max_thres_idx == 0:
        theta = threshold[0] if threshold[0] <= v_sorted[0] else 0.0
    else:
        if threshold[max_thres_idx] <= v_sorted[max_thres_idx]:
            theta = threshold[max_thres_idx]
        else:
            theta = threshold[max_thres_idx - 1]
    
    # Apply projection
    x = np.maximum(0, v_shift - theta) + lower
    return x

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx
