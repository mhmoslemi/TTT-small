import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 48  # Even number, so N/2 = 24 ones
    dx = 2.0 / n_points
    S = n_points // 2  # Number of ones
    
    # Create variables
    x = cp.Variable(n_points, integer=True)
    y = cp.Variable((n_points, n_points), integer=True)
    t = cp.Variable(integer=True)
    
    # Constraints: x binary
    constraints = [x >= 0, x <= 1, cp.constraints.Zero(x - cp.floor(x))]
    
    # Constraints: sum of x equals S
    constraints.append(cp.sum(x) == S)
    
    # Constraints: y linearizes x[i] * x[i+k]
    # Only consider valid pairs where both indices are in range
    for k in range(1, n_points):
        for i in range(n_points - k):
            # y[i,k] represents x[i] * x[i+k]
            constraints.extend([
                y[i, k] <= x[i],
                y[i, k] <= x[i + k],
                y[i, k] >= x[i] + x[i + k] - 1
            ])
    
    # Constraints: for each shift k from 1 to n_points-1
    for k in range(1, n_points):
        # Positive shift: mismatch_pos_k = sum_{i=0}^{N-1-k} (x_i - y_{i,k})
        # Negative shift: mismatch_neg_k = sum_{i=0}^{N-1-k} (x_{i+k} - y_{i,k})
        pos_mismatch = cp.sum(x[:n_points-k]) - cp.sum(y[:n_points-k, k])
        neg_mismatch = cp.sum(x[k:]) - cp.sum(y[:n_points-k, k])
        
        # Both should be <= t
        constraints.append(pos_mismatch <= t)
        constraints.append(neg_mismatch <= t)
    
    # Objective: minimize t
    problem = cp.Problem(cp.Minimize(t), constraints)

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val
