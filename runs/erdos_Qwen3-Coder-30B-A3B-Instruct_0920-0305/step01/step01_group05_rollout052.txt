import numpy as np
from scipy import signal
import time

def project_to_feasible_set(x, target_sum):
    """Project x onto the set {y : 0 <= y_i <= 1, sum(y) = target_sum}"""
    # Sort x in descending order
    sorted_x = np.sort(x)[::-1]
    # Find the threshold index
    cumsum = np.cumsum(sorted_x)
    threshold_index = np.argmax(cumsum > target_sum)
    
    # Compute alpha
    if threshold_index == 0:
        alpha = sorted_x[0] - (cumsum[0] - target_sum) / len(sorted_x)
    else:
        alpha = sorted_x[threshold_index - 1] - (cumsum[threshold_index - 1] - target_sum) / threshold_index
    
    # Project and adjust
    y = np.clip(x - alpha, 0, 1)
    # Ensure exact sum
    adjustment = (target_sum - np.sum(y)) / len(y)
    return y + adjustment

def compute_f_s(h, s):
    """Compute f_s(h) for shift s"""
    n = len(h)
    if s >= 0:
        # Right shift h by s
        shifted_h = np.concatenate([np.zeros(s), h[:-s] if s < n else np.zeros(n)])
    else:
        # Left shift h by |s|
        shifted_h = np.concatenate([h[-s:], np.zeros(-s)])
    return np.dot(h, 1 - shifted_h)

def compute_all_f_s(h):
    """Compute all f_s values for all shifts"""
    n = len(h)
    f_s_vals = []
    for s in range(1 - n, n):
        f_s = compute_f_s(h, s)
        f_s_vals.append(f_s)
    return np.array(f_s_vals)

def gradient_of_phi_p(h, p):
    """Compute the gradient of Φ_p(h)"""
    n = len(h)
    f_s_vals = compute_all_f_s(h)
    # Compute weights w_s = f_s^{p-1}
    w_s = np.abs(f_s_vals) ** (p - 1)
    # Initialize gradient
    grad = np.zeros(n)
    # Compute gradient components using shifted vectors
    for s in range(1 - n, n):
        if s >= 0:
            # Right shift
            shifted_h = np.concatenate([np.zeros(s), h[:-s] if s < n else np.zeros(n)])
        else:
            # Left shift
            shifted_h = np.concatenate([h[-s:], np.zeros(-s)])
        # Gradient contribution: w_s * (1 - h_{k+s} - h_{k-s})
        if s >= 0:
            shifted_grad = np.concatenate([np.zeros(s), np.ones(n - s)])
        else:
            shifted_grad = np.concatenate([np.ones(-s), np.zeros(n + s)])
        grad += w_s * (1 - shifted_h - shifted_grad)
    # Handle numerical stability
    phi_p = np.power(np.sum(w_s), 1/p)
    if phi_p > 0:
        grad = grad / np.power(phi_p, p - 1)
    return grad

def optimize_p_continuation(h0, n_points, p_values, max_iter_per_p=2000, lambda_penalty=1000):
    best_h = h0.copy()
    best_c5 = np.inf
    for p in p_values:
        # Initialize
        h = best_h.copy()
        dt = 0.0005
        for iteration in range(max_iter_per_p):
            # Compute ϕ_p
            f_s_vals = compute_all_f_s(h)
            phi_p = np.power(np.sum(np.power(np.abs(f_s_vals), p)), 1.0 / p)
            # Compute gradient of Φ_p
            grad = gradient_of_phi_p(h, p)
            # Add penalty term
            grad += 2 * lambda_penalty * (np.sum(h) - n_points/2)
            # Update
            new_h = h - dt * grad
            # Project to feasible set
            new_h = project_to_feasible_set(new_h, n_points/2)
            # Clip to [0,1]
            new_h = np.clip(new_h, 0, 1)
            # Update
            h = new_h
            # Convert to actual C5 value
            c5 = np.max(np.abs(f_s_vals)) * 2.0 / n_points
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

    if current_c5 < best_c5:

        best_c5 = current_c5
    
    # Hill climbing swap
    for _ in range(max_iter):
        s_star = np.argmax(np.abs(f_s_vals))

        # Find a 1 and a 0 to swap

                    candidate_0s = indices_1[mask]

                        # Identify the actual 0 index
                        idx_0 = idx_1 + s_star
                        # Do swap 
                        h_binary[idx_1] = 0
                        h_binary[idx_0] = 1
                        # Recompute f_s_vals
                        f_s_vals = compute_all_f_s(h_binary)
                        new_c5 = np.max(np.abs(f_s_vals)) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_binary.copy()
                        else:
                            # Swap back
                            h_binary[idx_1] = 1
                            h_binary[idx_0] = 0
                        continue
            # Try flipping the most positive or negative value 
            indices_0 = np.where(h_binary == 0)[0]
            if len(indices_0) > 0:
                idx_0 = indices_0[0]
                idx_1 = np.argmax(h_binary)
                # Flip these
                h_binary[idx_0] = 1
                h_binary[idx_1] = 0
                # Recompute new f_s and c5
                f_s_vals = compute_all_f_s(h_binary)
                new_c5 = np.max(np.abs(f_s_vals)) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_binary.copy()
                else:
                    # Swap back
                    h_binary[idx_0] = 0
                    h_binary[idx_1] = 1
        except Exception:
            continue
    return best_h, best_c5
