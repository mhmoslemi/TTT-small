import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def soft_min_correlation(h, lambda_param):
    """Compute Φ_λ(h) = (1/λ)*log(sum(exp(-λ*R(s)))"""
    n = len(h)
    # Compute the full correlation R(s)
    R = np.correlate(h, h, mode="full")
    
    # Compute soft minimum via log-sum-exp
    R_max = np.max(R)
    exp_R = np.exp(-lambda_param * (R - R_max))
    sum_exp = np.sum(exp_R)
    log_sum_exp = R_max + np.log(sum_exp) / lambda_param
    
    return log_sum_exp

def compute_gradient(h, lambda_param):
    """Compute the gradient of Φ_λ(h) with respect to h"""
    n = len(h)
    R = np.correlate(h, h, mode="full")
    
    # Compute weights w_s for each shift
    R_max = np.max(R)
    exp_R = np.exp(-lambda_param * (R - R_max))
    w = exp_R / np.sum(exp_R)
    
    # Compute gradient: g_j = sum_s w_s (h_{j+s} + h_{j-s})
    g = np.zeros(n)
    for s in range(len(w)):
        shift = s - (len(w) // 2)
        if shift > 0:
            g[:n-shift] += w[s] * h[shift:]
        elif shift < 0:
            g[-shift:] += w[s] * h[:n+shift]
        else:
            g += w[s] * h
            
    return g

def project_capped_simplex(h, target_sum):
    """Project h onto capped simplex: h_i ∈ [0,1], sum h_i = target_sum"""
    # Sort h in decreasing order
    h_sorted = np.sort(h)[::-1]
    n = len(h)
    
    # Find the threshold tau using binary search
    diff = np.maximum(0, h_sorted - (np.cumsum(h_sorted) - target_sum) / np.arange(1, n+1))
    tau = np.max(diff)
    
    # Apply projection
    h_proj = np.clip(h - tau, 0, 1)
    
    # Renormalize
    h_proj = h_proj * (target_sum / np.sum(h_proj))
    
    return h_proj

def enforce_symmetry(h):
    """Enforce symmetry h[i] = (h[i] + h[n-1-i]) / 2"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
    return h_sym

def compute_best_c5(h):
    """Compute C5 with exact max objective"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(correlation)
    return max_corr * dx

        else:
            # Extrapolate initial
            h = np.array(initial_h_values)

        # Symmetrize starting guess

                for _ in range(10):  # Max line search steps
                    h_new = h - alpha * g

    return (best_h.tolist(), best_c5, n_points)
