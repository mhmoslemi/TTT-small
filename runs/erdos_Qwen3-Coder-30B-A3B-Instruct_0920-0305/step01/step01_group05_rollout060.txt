import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure sum equals n/2 (which makes integral = 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n / 2.0) / current_sum
        h = np.clip(h, 0, 1)  # Keep values in [0,1]
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_c5_binary(bits):
    """Evaluate C5 for binary vector"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def binary_local_search(n, time_limit):
    """Perform local search for binary solutions"""
    dx = 2.0 / n
    half = n // 2
    
    best_sum = float('inf')
    best_bits = None
    start = time.time()
    
    while time.time() - start < time_limit:
        # Generate random binary vector with exactly half ones
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        curr_val = evaluate_c5_binary(bits)
        
        # Iterative descent
        improved = True
        attempts = 0
        while improved and time.time() - start < time_limit and attempts < 1000:
            improved = False
            attempts += 1
            
            # Try some random swaps
            for _ in range(n):
                if time.time() - start > time_limit:
                    break
                ones_idx = np.flatnonzero(bits)
                zeros_idx = np.flatnonzero(1 - bits)
                
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                    
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Try the swap
                bits[i], bits[j] = bits[j], bits[i]
                new_val = evaluate_c5_binary(bits)
                
                if new_val < curr_val - 1e-12:
                    curr_val = new_val
                    improved = True
                else:
                    # Revert the swap
                    bits[i], bits[j] = bits[j], bits[i]
            
            # Early exit condition
            if curr_val < 0.379:
                break
        
        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
    
    if best_bits is None:
        # Fallback to random initialization
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        best_sum = evaluate_c5_binary(bits)
        best_bits = bits.copy()
        
    return best_bits, best_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different resolution values
    n_list = [42, 60, 80]
    
    # For each n, allocate time proportionally
    for i, n in enumerate(n_list):
        if time.time() - start_time > budget_s - 10:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
            
        # Assign time based on the number of n we're checking
        # Give more time to more promising candidate
        time_per_n = remaining_time * 0.7 / len(n_list)
        
        # Phase 1: Differential Evolution for continuous optimization
        dx = 2.0 / n
        bounds = [(0, 1) for _ in range(n)]
        
        def objective(h):
            h = np.array(h)
            current_sum = np.sum(h)
            if current_sum != 0:
                h = h * (n / 2.0) / current_sum
                h = np.clip(h, 0, 1)
            
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        try:
            # First try a simpler DE with fewer iterations
            res = differential_evolution(
                objective,
                bounds,
                maxiter=int(500),
                popsize=25,
                mutation=(0.5, 1),
                recombination=0.7,
                atol=1e-6,
                tol=1e-6,
                seed=seed + i,
                callback=lambda x, convergence: None
            )
            
            if hasattr(res, 'success') and res.success:
                continuous_h = res.x
                continuous_h = np.array(continuous_h)
                current_sum = np.sum(continuous_h)
                if current_sum != 0:
                    continuous_h = continuous_h * (n / 2.0) / current_sum
                    continuous_h = np.clip(continuous_h, 0, 1)
            else:
                continuous_h = np.random.rand(n)
                current_sum = np.sum(continuous_h)
                continuous_h = continuous_h * (n / 2.0) / current_sum
                continuous_h = np.clip(continuous_h, 0, 1)
                
        except Exception:
            # If DE fails, use random initialization
            continuous_h = np.random.rand(n)
            current_sum = np.sum(continuous_h)
            continuous_h = continuous_h * (n / 2.0) / current_sum
            continuous_h = np.clip(continuous_h, 0, 1)
        
        # Phase 2: Local Binary Search
        try:
            if time.time() - start_time < remaining_time * 0.9:
                bits, c5_binary = binary_local_search(n, min(remaining_time * 0.9, 300))
                if c5_binary < best_c5:
                    best_c5 = c5_binary
                    best_h = bits
                    best_n = n
        except Exception:
            pass
            
        # If no binary improvement or no binary was done, keep continuous solution
        c5_continuous = evaluate_h(continuous_h)
        if c5_continuous < best_c5:
            best_c5 = c5_continuous
            best_h = continuous_h
            best_n = n
            
        # Early exit if we've found a good enough bound
        if best_c5 < 0.379:
            break
    
    # Final evaluation of the best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    else:
        # Generate fallback solution
        n = 80  # Fallback size
        dx = 2.0 / n
        best_h = np.ones(n) * 0.5
        best_c5 = evaluate_h(best_h)
        best_n = n
    
    return (best_h.tolist(), best_c5, best_n)
