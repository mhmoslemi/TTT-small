import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h_values):
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_half):
    """Project h onto the simplex: sum(h) = n_half, 0 <= h_i <= 1"""
    # First shift each element by a scalar alpha to make sure sum = n_half,
    # then clip to [0,1] and adjust alpha as needed
    def sum_after_clip(alpha):
        h_proj = np.clip(h - alpha, 0, 1)
        return np.sum(h_proj) - n_half
    
    # bisection to find the right alpha
    a, b = -1.0, 1.0
    for _ in range(50):
        mid = (a + b) / 2.0
        s = sum_after_clip(mid)
        if s < 0:
            b = mid
        else:
            a = mid
    alpha = (a + b) / 2.0
    return np.clip(h - alpha, 0, 1)

def grad_f_s(h, s):
    """Compute gradient of f_s(h) w.r.t. each h_i"""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if 0 <= i - s < n:
            grad[i] -= h[i - s]
        if 0 <= i + s < n:
            grad[i] -= h[i + s]
        grad[i] += 1.0
    return grad

def compute_f_s(h, s):
    """Compute f_s(h) for a fixed shift s."""
    n = len(h)
    # Time shift by s (right shift)
    if s >= 0:
        h_shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.dot(h, 1 - h_shifted)

def compute_all_f_s(h):
    """Return all values f_s(h) for s = -(n-1) to n-1"""
    n = len(h)
    # Compute all shifts using np.correlate
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

        else:

            w_s = fs_abs ** (p - 1)
            # Weighted sum of gradients of f_s
            grad_Phi = np.zeros(n)
            for s_idx, w in enumerate(w_s):
                s = s_idx - (n - 1)
                # Gradient of f_s = h·(1 - h_{i+s})
                # grad_{i} f_s = 1 - h_{i+s} - h_{i-s} = 1 - h_{i+s} - h_{i-s}
                if s >= 0:  # Right shift case
                    h_shifted = np.concatenate([np.zeros(s), h[:-s]])
                else:  # Left shift case
                    h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
                grad_f = 1.0 - h_shifted  # Component-wise negation
                if s != 0:
                    # For i-s
                    if s > 0:
                        h_shifted_neg = np.concatenate([h[s:], np.zeros(s)])
                    else:
                        h_shifted_neg = np.concatenate([np.zeros(-s), h[:-s]])
                    grad_f -= h_shifted_neg
                # Need to weight by w_s for the gradient
                # Not vectorized but needed for accuracy
                g = w * grad_f
                # Accumulate gradient for each index
                if s >= 0:
                    grad_Phi[:n-s] += g[s:]
                elif s < 0:
                    grad_Phi[-s:] += g[:-s]
            grad_Phi /= (Phi ** (p - 1)) if Phi != 0 else 1.0
            
        # Add penalty term gradient
        penalty_grad = 2 * lambda_penalty * (np.sum(h) - n_half)
        grad_Phi += penalty_grad
        
        # Update with step size
        h -= step_size * grad_Phi
        
        # Project back to simplex
        h = project_to_feasible(h, n_half)
        
    return h
