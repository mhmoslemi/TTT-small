import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the simplex with sum = target_sum and bounds [lower, upper]"""
    # This is a simplified water-filling approach
    v = np.array(v)
    n = len(v)
    
    # First, shift to [0, upper-lower] space
    v_shifted = np.clip(v, lower, upper) - lower
    total = np.sum(v_shifted)
    
    # If already at target, return
    if abs(total - target_sum) < 1e-10:
        return np.clip(v, lower, upper)
        
    # Determine the shift needed
    if total > target_sum:
        # Need to reduce values
        # Apply iterative clipping with thresholding
        remain = target_sum
        # Sort in descending order
        idx = np.argsort(v_shifted)[::-1]
        sorted_v = v_shifted[idx]
        cumsum = np.cumsum(sorted_v)
        # Find cut point
        for i in range(len(sorted_v)):
            if i == 0:
                threshold = (cumsum[i] - remain) / (i + 1)
            else:
                threshold = (cumsum[i] - remain) / (i + 1)
            
            if threshold <= sorted_v[i]:
                break
        else:
            # all values should be reduced to min(upper-lower, 0)
            threshold = sorted_v[-1] - (cumsum[-1] - target_sum) / n
        
        result = np.zeros_like(v_shifted)
        # compute the new values
        for i in range(len(sorted_v)):
            if sorted_v[i] >= threshold:
                result[idx[i]] = threshold
            else:
                result[idx[i]] = sorted_v[i]
                
        # Finally, map back to [lower, upper]
        return result + lower
    else:
        # Need to increase values - fill by making them as large as possible
        # But we still clipped to [0, 1], so we can just return
        return np.clip(v, lower, upper)

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx
