import numpy as np
from scipy.optimize import minimize
import time

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x: lower <= x_i <= upper, sum x_i = target_sum}"""
    # Create a copy to avoid modifying the input
    v = np.array(v, dtype=float)
    
    # Sort in descending order
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_v)
    
    # Find the threshold
    # We need to find theta such that sum_i min(upper, max(lower, v_i + theta)) = target_sum
    # After sorting, we compute efficiently
    
    # We can optimize by considering only up to 1000 iterations or early termination
    # for small arrays
    n = len(v)
    
    # Try to compute threshold Theta
    # But for simplicity, use iterative method for now:
    # Binary search on threshold
    low, high = -10.0, 10.0
    
    for _ in range(100):  # limit iterations
        theta = (low + high) / 2.0
        proj = np.clip(v + theta, lower, upper)
        sum_proj = np.sum(proj)
        if sum_proj < target_sum:
            high = theta
        else:
            low = theta
    
    theta = (low + high) / 2.0
    result = np.clip(v + theta, lower, upper)
    return result

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    # Note: evaluate_erdos_solution is in global scope from import
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    return max_corr * dx

def evaluate_raw(h, dx=2.0/200):
    """Evaluate using raw correlation to avoid smoothing effects"""
    n = len(h)
    # Compute correlation of h with h
    corr_hh = np.correlate(h, h, mode='full')
    min_corr = np.min(corr_hh)
    # Total sum = n/2 (since dx = 2/n and integral = 1)
    c_sum = n / 2.0
    return c_sum - min_corr

def compute_gradient(h, corr, weights):
    # corr is correlation with h, weights are computed from negative corr
    # Compute gradient of logsumexp(-lambda * corr) / lambda
    # ∂L/∂h_i = Σ_s weights_s * (h_{i+s} + h_{i-s}) 
    grad = np.zeros_like(h)
    n = len(h)
    for s in range(-n+1, n):
        idx = s + n - 1 # index into weights
        w_s = weights[idx]
        if w_s == 0: continue
        if s >= 0:
            # h_{i+s} only valid for i=0..n-1-s
            grad[:n-s] += w_s * h[s:]
            # h_{i-s} only valid for i=s..n-1
            grad[s:] += w_s * h[:n-s]
        else:
            s_neg = -s
            grad[s_neg:] += w_s * h[:n-s_neg]
            grad[:n-s_neg] += w_s * h[s_neg:]
    return grad

    else:
        # Interpolate
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h_new = np.interp(x_new, x0, base_h)
        return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use 200 points for better resolution
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Generate some initial solutions:
    initial_solutions = []
    
    # 1. Use upscaled initial solution
    h_base = get_initial_h(n_points)
    h_base = project_onto_box_simplex(h_base, target_sum, 0, 1)
    initial_solutions.append(("upscale", h_base.copy()))
    
    # 2. Random uniform start
    h_random = np.random.uniform(0, 1, n_points)
    h_random = project_onto_box_simplex(h_random, target_sum, 0, 1)
    initial_solutions.append(("random", h_random.copy()))
    
    # 3. Binary indicator pattern: 1s in alternating positions
    h_binary = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_binary[i] = 1
    h_binary = project_onto_box_simplex(h_binary, target_sum, 0, 1)
    initial_solutions.append(("binary", h_binary.copy()))

    # Convergence tracking
    def perturb_h(h, perturb_rate=0.05):
        # Perturb h with small Gaussian noise, then project
        h_new = h + np.random.normal(0, 0.02, n_points)
        h_new = project_onto_box_simplex(h_new, target_sum, 0, 1)
        return h_new

    # Try multiple restarts
    num_starts = 0
    max_starts = 30
    
    for sol_type, h in initial_solutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        num_starts += 1
        current_h = h.copy()
        current_c5 = evaluate_raw(current_h, dx)
        
        # Gradient descent with smoothing + line search
        lambda_schedule = [1, 5, 10, 20]
        step_size = 0.1
        max_iters_per_lambda = 500
        
        for lam in lambda_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            for i in range(max_iters_per_lambda):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute full correlation with h
                corr = np.correlate(current_h, current_h, mode='full')  # c < n: corr of conv
                max_corr = np.max(corr)
                min_corr = np.min(corr)
                
                # Stable softmax weights
                corr_stable = corr - max_corr
                exp_values = np.exp(-lam * corr_stable)
                sum_exp = np.sum(exp_values)
                if sum_exp == 0:
                    weights = np.zeros_like(corr)
                else:
                    weights = exp_values / sum_exp
                    
                # Gradient step
                grad = compute_gradient(current_h, corr, weights)
                
                # Descent direction
                descent_dir = grad
                new_h = current_h - step_size * descent_dir
                
                # Project back
                new_h = project_onto_box_simplex(new_h, target_sum, 0, 1)
                
                # Quick line search or step acceptance
                test_c5 = evaluate_raw(new_h, dx)
                if test_c5 < current_c5:
                    current_h = new_h
                    current_c5 = test_c5
                else:
                    # Reduce step size
                    step_size *= 0.95
                    
                # A heuristic to decrease step size and early terminate
                if current_c5 < 0.365:
                    break
                    
                # Periodic check against time limit
                if time.time() - start_time > budget_s * 0.95:
                    break
                
                # print(f"Lambda {lam}, iter {i}: C5 = {current_c5:.6f}")
        
        # Final evaluation
        final_c5 = evaluate_raw(current_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = current_h.copy()
            
        if time.time() - start_time > budget_s * 0.95:
            break

    # SLSQP final refinement
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        try:
            # Rescale to 10000 function evaluations
            def obj(h_vals):
                return evaluate_erdos_solution(h_vals, dx)
            
            # Constraints
            def cons_sum(h_vals):
                return np.sum(h_vals) * dx - 1.0
            
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': cons_sum}
            
            result = minimize(
                obj,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-8}
            )
            
            if result.success:
                refined_c5 = obj(result.x)
                if refined_c5 < best_c5:
                    best_h = result.x.copy()
                    best_c5 = refined_c5
        except:
            pass
        
        # Evaluate final using raw function if needed
        if best_c5 > 0.4:
            # Re-evaluate with actual correlation
            best_c5 = evaluate_raw(best_h, dx)
            
    # Ensure feasibility
    if best_h is None:
        h = get_initial_h(n_points)
        h = project_onto_box_simplex(h, target_sum, 0, 1)
        best_h = h.copy()
        best_c5 = evaluate_raw(best_h, dx)
    else:
        best_h = project_onto_box_simplex(best_h, target_sum, 0, 1)
        best_c5 = evaluate_raw(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
