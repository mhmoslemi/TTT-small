def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project vector y onto the capped simplex: sum = target_sum, 0 <= x_i <= 1"""
    # Ensure values are within [0,1] and adjust sum
    y = np.clip(y, 0, 1)
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-10:
        return y
    if current_sum > target_sum:
        # Use binary search to find lambda for overshoot
        # y = clip(y - lambda, 0, 1) should sum to target_sum
        low, high = 0.0, np.max(y) + 1.0
        for _ in range(50):
            lam = (low + high) / 2.0
            projected = np.clip(y - lam, 0.0, 1.0)
            current_sum = np.sum(projected)
            if current_sum > target_sum:
                low = lam
            else:
                high = lam
        lam = (low + high) / 2.0
        projected = np.clip(y - lam, 0.0, 1.0)
        return projected
    else:
        # Need to increase values, but they are already at 1s in places
        # Adjust all to compensate
        scale = target_sum / current_sum
        projected = y * scale
        return np.clip(projected, 0.0, 1.0)

def compute_gradient(h, corr, alpha):
    """Compute gradient of the smooth surrogate F_alpha(h) = (1/alpha) log sum exp(alpha*C_s)"""
    # Compute softmax weights
    c_max = np.max(corr)
    # Ensure numerical stability
    weights = np.exp(alpha * (corr - c_max))
    weights /= np.sum(weights)
    
    # Initialize gradient
    grad = np.ones_like(h)
    # The gradient is 1 - sum_s w_s*(h_{i+s} + h_{i-s})
    # This can be done by shifting h forward and backward and taking weighted sum
    n = len(h)
    forward_shift = np.zeros_like(h)
    backward_shift = np.zeros_like(h)
    
    # Compute shifted versions (with zero-padding)
    h_ext = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    
    for s in range(len(corr)):
        if s < n:
            forward_shift[:n-s] += weights[s] * h_ext[s:n+s]
        else:
            backward_shift[s-n+1:] += weights[s] * h_ext[s-n+1:s]
    
    grad = 1 - forward_shift - backward_shift
    return grad

def safe_logsumexp(x, axis=None):
    """Numerically stable log-sum-exp with optional axis"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))
