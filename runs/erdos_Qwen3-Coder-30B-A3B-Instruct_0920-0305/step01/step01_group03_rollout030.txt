import numpy as np

import time

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the simplex with sum = target_sum and bounds [lower, upper]"""
    # We can solve this directly by finding theta such that
    # sum_i clamp(v_i + theta, lower, upper) = target_sum
    v = np.asarray(v)
    n = len(v)
    
    # Binary search for theta
    low, high = -10, 10
    
    def compute_sum(theta):
        return np.sum(np.clip(v + theta, lower, upper))
    
    # Find bounds for theta
    while compute_sum(low) > target_sum:
        low *= 2
    while compute_sum(high) < target_sum:
        high *= 2
    
    # Binary search
    for _ in range(50):
        mid = (low + high) / 2
        s = compute_sum(mid)
        if s < target_sum:
            low = mid
        else:
            high = mid
    
    theta = (low + high) / 2
    result = np.clip(v + theta, lower, upper)
    
    # Normalize to exact sum
    result = result * target_sum / np.sum(result)
    
    return result
