```python
import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return 1.0 - max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution schedule
    resolutions = [80, 100, 120]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        dx = 2.0 / n
        
        # Create variables
        x = Variable(n)
        t = Variable(1)
        
        # Constraint sum to 1
        constraints = [sum(x) == n/2.0]
        constraints += [0 <= x, x <= 1]
        
        # Build McCormick constraints and add to list
        w = {}
        # We'll collect all constraints at the end to reduce overhead
        mccormick_constraints = []
        
        # For each shift (from -(n-1) to n-1), we consider valid i values
        # It's safer to precompute indices
        shift_ranges = []
        for s in range(-(n-1), n):
            start_i = max(0, -s)
            end_i = min(n-1, n-1-s)
            if start_i <= end_i:
                shift_ranges.append((s, start_i, end_i))
        
        # Create all w_ij variables and constraints by shifting
        for s, start_i, end_i in shift_ranges:
            for i in range(start_i, end_i + 1):
                idx = (i, s)
                w[idx] = Variable(1)
                # McCormick inequalities:
                # w <= x_i
                mccormick_constraints.append(w[idx] <= x[i])
                # w <= x_{i+s}
                mccormick_constraints.append(w[idx] <= x[i+s])
                # w >= x_i + x_{i+s} - 1
                mccormick_constraints.append(w[idx] >= x[i] + x[i+s] - 1)
                # w >= 0
                mccormick_constraints.append(w[idx] >= 0)
                
                # Add sum constraint for shift s: sum(w) >= t
                # We'll collect all this in a formula directly below
                # So we need to keep w_ij for each (i,s) pair
        
        # Collect the total constraint per shift s:
        # sum_{i: valid} w_{i,s} >= t
        constraints_by_shift = {}
        for s, start_i, end_i in shift_ranges:
            temp_sum = 0.0
            for i in range(start_i, end_i + 1):
                temp_sum += w[(i, s)]
            constraints_by_shift[s] = temp_sum >= t

        # Add all shift sum constraints
        constraints += list(constraints_by_shift.values())
        # Add all McCormick constraints
        constraints += mccormick_constraints
        
        # Create problem and solve it
        objective = Maximize(t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, feastol=1e-8, abstol=1e-8, reltol=1e-6,verbose=False)
            x_opt = x.value
            t_opt = t.value
            if np.isnan(t_opt):
                continue
        except:
            continue
            
        # Rounding strategy: sort by x_opt descending, select top n/2
        order = np.argsort(x_opt)[::-1]
        b = np.zeros(n)
        b[order[:n//2]] = 1.0
        
        # Evaluate exact C5 of binary vector
        c5_exact = evaluate_erdos_solution(b, dx=dx)
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = b.copy()
            best_n = n
            
        # Local combinatorial search after rounding
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Run a local opt...
        # The cost of full correlation computation is manageable        
        # record as possible improvement
        old_c5 = c5_exact
        old_b = b.copy()
        
        # Mirror symmetrization (shouldn't affect the mass if n is even)
        if n % 2 == 0:
            b_symmetrized = (b + np.flip(b)) / 2.0
            b_rounded = np.round(b_symmetrized).astype(float)
            # Re-adjust to have exactly n/2 ones
            ones_count = np.sum(b_rounded)
            diff = n/2.0 - ones_count
            if diff != 0:
                # Simply modify some entries for balancing
                # This heuristic is very simple but robust
                indices = np.argsort(b_symmetrized)[::-1]
                for i in range(int(np.abs(diff))):
                    if diff > 0:
                        if b_rounded[indices[i]] == 0:
                            b_rounded[indices[i]] = 1.0
                    else:
                        if b_rounded[indices[i]] == 1:
                            b_rounded[indices[i]] = 0.0
            
            c5_sym = evaluate_erdos_solution(b_rounded, dx=dx)
            if c5_sym < best_c5:
                best_c5 = c5_sym
                best_h = b_rounded.copy()
                best_n = n
        
        # Simple swaps: a basic hill-climber
        swaps_allowed = min(20000, int(300000 * (budget_s - (time.time() - start_time)) / budget_s))
        swap_count = 0
        improved = True
        patience = 0
        max_patience = 1000

        while improved and swap_count < swaps_allowed and time.time() - start_time < budget_s * 0.95:
            improved = False
            for attempt in range(1000):
                # Get indices of 1s and 0s
                ones_idx = np.where(old_b == 1)[0]
                zeros_idx = np.where(old_b == 0)[0]
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Swap
                new_b = old_b.copy()
                new_b[i] = 0
                new_b[j] = 1
                c5_new = evaluate_erdos_solution(new_b, dx=dx)
                
                if c5_new < old_c5:
                    old_c5 = c5_new
                    old_b = new_b.copy()
                    improved = True
                    patience = 0
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = new_b.copy()
                        best_n = n
                    break
                else:
                    patience += 1
                    if patience > max_patience:
                        break

    # Final polish if time left
    if time.time() - start_time < budget_s - 10:
        # A scaling gradient-based update
        def smooth_min_corr(h_vals):
            corr = np.correlate(h_vals, 1-h_vals, mode="full")
            return -np.min(corr) * (2.0 / len(h_vals))

        def grad_smooth_corr(h_vals):
            corr = np.correlate(h_vals, 1-h_vals, mode="full")
            min_idx = np.argmin(corr)
            # compute grad at min shift
            # roughly the gradient of the min term
            # by finite differencing small eps in one direction:
            grad = np.zeros_like(h_vals)
            eps = 1e-6
            for i in range(len(h_vals)):
                hp = h_vals.copy()
                hp[i] += eps
                corr_p = np.correlate(hp, 1-hp, mode="full")
                grad[i] = (corr_p[min_idx] - corr[min_idx]) / eps
            return -grad * (2.0 / len(h_vals))
        
        l = 8.0  # lambda for log-sum-exp
        D = 100
        iter_count = 0
        h_smooth = best_h.copy()
        while iter_count < 20 and time.time() - start_time < budget_s - 5:
            iter_count += 1
            # approx log-sum-exp minimum
            corr = np.correlate(h_smooth, 1-h_smooth, mode="full")
            # ignore due to -logsumexp, add a small perturbation
            corrs_flip = np.array(corr).copy()
            if l > 0:
                linv = 1/l
                corrs_flip = corr*linv
                corrs_flip = np.exp(corrs_flip - np.max(corrs_flip))
                # weight by exp(-c/l) then scale
                total_weight = np.sum(corrs_flip)
                weighted = np.dot(corrs_flip, corr) / total_weight
                smooth_min = weighted
            else:
                smooth_min = np.min(corr)
            
            # Gradient approximation:
            g = grad_smooth_corr(h_smooth)
            
            # Projected step
            h_new = h_smooth + 0.001 * g
            # Project onto [0,1]
            h_new = np.clip(h_new, 0.0, 1.0)
            # Project on mass
            h_sum = np.sum(h_new)
            h_new = h_new * (n/2.0) / h_sum
            if np.sum(h_new) > 0:
                h_new = h_new * (n/2.0) / np.sum(h_new)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_new, dx=dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h_smooth = h_new
        
    return (best_h.tolist(), best_c5, best_n)
```
