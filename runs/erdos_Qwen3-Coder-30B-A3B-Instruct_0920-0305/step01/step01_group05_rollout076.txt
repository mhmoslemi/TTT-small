import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 80
    dx = 2.0 / n
    target = 0.3807
    
    # Setup optimization variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Build constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n/2,
        cp.diag(X) == h,
        X >> 0,  # PSD constraint
        t >= 0
    ]
    
    # Add McCormick constraints for X[i,j] = h[i] * h[j]
    for i in range(n):
        for j in range(i+1, n):
            constraints += [
                X[i, j] >= 0,
                X[i, j] <= h[i],
                X[i, j] <= h[j],
                X[i, j] >= h[i] + h[j] - 1
            ]
    
    # Overlap constraints for all non-zero shifts 
    # (using np.eye shifts for computing diagonal of shifted X)
    for s in range(-(n-1), n):
        if s == 0: continue
        # Extract diagonal with shift s
        shift_mat = np.eye(n, k=s)
        shift_sum = cp.sum(cp.multiply(shift_mat, X))
        constraints.append(t >= n/2 - shift_sum)
    
    # Minimize the epigraph variable
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve using SCS solver
    prob.solve(solver='SCS', eps=1e-5, max_iters=5000, verbose=False)

    if prob.status != cp.OPTIMAL:  # fallback if not optimal
        print("SDP did not solve to optimality")
        return (np.ones(n) * 0.5, 0.5, n)
    
    # Extract solutions
    h_sdp_val = h.value
    X_sdp_val = X.value 
    t_sdp_val = t.value
    
    # Bound on C5
    c5_lower = t_sdp_val * dx

    # Local function to enforce integral constraint by bisection
    def project_to_sum(h_vec, target_sum):
        # Clip to [0,1]
        h_clipped = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_clipped)
        if abs(current_sum - target_sum) < 1e-10:
            return h_clipped
        
        # Use bisection to find suitable α
        alpha_lo, alpha_hi = -10.0, 10.0
        for _ in range(30):
            alpha = (alpha_lo + alpha_hi) / 2
            h_trial = np.clip(h_clipped + alpha, 0, 1)
            trial_sum = np.sum(h_trial)
            if trial_sum < target_sum:
                alpha_lo = alpha
            else:
                alpha_hi = alpha
        alpha_final = (alpha_lo + alpha_hi) / 2
        return np.clip(h_clipped + alpha_final, 0, 1)
    
    # Sample from the covariance matrix using randomized rounding
    best_h = None
    best_c5 = np.inf
    n_samples = 2000  # Limit samples due to time constraint
    
    # Compute eigen-decomposition of X_sdp_val
    try:
        w, V = eigh(X_sdp_val)
        idx = w > 1e-8
        w = w[idx]
        V = V[:, idx]
        B = V @ np.diag(np.sqrt(w)) 
        
        for _ in range(n_samples):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Generate sample from normal
            g = np.random.randn(B.shape[1])
            z = B @ g  # Sample from multivariate normal with covariance X_sdp_val
            
            # Clip and project to correct sum
            h_candidate = project_to_sum(z, n/2)
            
            # Evaluate C5
            try:
                c5_val = evaluate_h(h_candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_candidate.copy()
            except:
                continue
            
    except:
        pass  # Fallback to deterministic rounding if decomposition fails

    # Fallback to deterministic rounding if randomized failed or not enough samples
    if best_h is None:
        # Deterministic rounding: threshold at 0.5
        rounded = np.zeros_like(h_sdp_val)
        sorted_indices = np.argsort(h_sdp_val)[::-1]
        top_k = n // 2
        rounded[sorted_indices[:top_k]] = 1.0
        
        # Ensure exactly sum = n/2
        actual_sum = np.sum(rounded)
        while actual_sum > n/2:
            # Find a 1 and flip it to 0
            flip_idx = np.where(rounded == 1.0)[0][0]
            rounded[flip_idx] = 0
            actual_sum -= 1
        while actual_sum < n/2:
            # Find a 0 and flip it to 1
            flip_idx = np.where(rounded == 0.0)[0][0]
            rounded[flip_idx] = 1
            actual_sum += 1
            
        try:
            best_c5 = evaluate_h(rounded)
            best_h = rounded
        except:
            return (np.ones(n) * 0.5, 0.5, n)

    # Local optimization starting from best rounded solution
    if time.time() - start_time < budget_s - 2:
        step_size = 0.01
        n_iter = 1000  # reduced iteration count for time limit
        h_current = best_h.copy()
        for i in range(n_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Try small perturbations
            for idx in range(n):
                h_trial = h_current.copy()
                delta = np.random.normal(0, step_size)
                h_trial[idx] += delta
                h_trial = np.clip(h_trial, 0, 1)
                # Re-scale to sum = n/2
                h_trial = h_trial * (n/2) / np.sum(h_trial)
                
                try:
                    c5_trial = evaluate_h(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        h_current = h_trial.copy()
                        best_h = h_current.copy()
                except:
                    continue

    return (best_h.tolist(), best_c5, n)
