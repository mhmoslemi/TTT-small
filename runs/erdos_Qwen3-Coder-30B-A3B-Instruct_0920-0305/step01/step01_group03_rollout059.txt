import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def build_and_solve_lp(n):
    """Build and solve the LP relaxation for the given n."""
    # Variables
    x = cp.Variable(n)
    t = cp.Variable()
    w = cp.Variable((n, 2*n - 1))  # w[i,s] for s = -(n-1) to (n-1)
    
    # Constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(x) == n / 2.0)
    
    # Box constraints
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # McCormick constraints for each valid (i, s) pair
    for s in range(-(n-1), n):
        for i in range(n):
            if 0 <= i+s < n:
                # Shift index
                w_idx = s + (n-1)  # Index into w
                # w[i,s] <= x[i]
                constraints.append(w[i, w_idx] <= x[i])
                # w[i,s] <= x[i+s]
                constraints.append(w[i, w_idx] <= x[i+s])
                # w[i,s] >= x[i] + x[i+s] - 1
                constraints.append(w[i, w_idx] >= x[i] + x[i+s] - 1)
                # w[i,s] >= 0
                constraints.append(w[i, w_idx] >= 0)
    
    # Constraint for each shift that sum of product is at least t
    for s in range(-(n-1), n):
        w_idx = s + (n-1)
        # Sum all valid w[i,s] for that shift
        valid_indices = [i for i in range(n) if 0 <= i+s < n]
        if valid_indices:
            constraints.append(cp.sum(w[valid_indices, w_idx]) >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
    
    if prob.status == cp.OPTIMAL:
        return x.value, t.value
    else:
        print("LP Solver did not converge optimally.")
        return x.value, t.value

def deterministic_rounding(x, n):
    """Round fractional x to binary vector with exact mass"""
    # Create indices sorted by decreasing x values
    sorted_indices = np.argsort(x)[::-1]
    # Create binary vector
    b = np.zeros(n)
    for i in range(int(n/2)):
        b[sorted_indices[i]] = 1
    return b

def swap_improvement(binary_h, dx, max_swaps=20000):
    """Perform swap-based improvements on binary solution"""
    h = binary_h.copy()
    n = len(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    no_improvement_count = 0
    swap_count = 0
    start_time = time.time()
    
    # Precompute the full correlation array to speed up updates
    def compute_full_correlation(curr_h):
        corr = np.correlate(curr_h, 1 - curr_h, mode='full')
        return corr
    
    full_corr = compute_full_correlation(h)
    max_corr = np.max(full_corr)
    
    while swap_count < max_swaps and (time.time() - start_time) < 580:
        # Find positions with 1s and 0s
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        swap_count += 1
        
        # Recompute full correlation
        new_corr = compute_full_correlation(h)
        new_max = np.max(new_corr)
        
        if new_max < max_corr:
            max_corr = new_max
            best_c5 = 1 - new_max * dx
            no_improvement_count = 0
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
            no_improvement_count += 1
            
        if no_improvement_count == 1000:
            break
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different resolutions to optimize within time
    resolutions = [80, 100, 120]
    best_h = None
    best_c5 = float('inf')
    n_best = 0
    
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Build and solve LP
        x_opt, t_reduced = build_and_solve_lp(n)
        
        # Do deterministic rounding
        binary_h = deterministic_rounding(x_opt, n)
        dx = 2.0 / n
        c5_raw = evaluate_erdos_solution(binary_h, dx)
        
        # Perform swap improvement
        improved_h, improved_c5 = swap_improvement(binary_h, dx, max_swaps=5000)
        
        if improved_c5 < best_c5:
            best_c5 = improved_c5
            best_h = improved_h
            n_best = n
            
    # If we have a good solution, run a final polish
    if best_h is not None:
        # Just in case, try one final round of swaps if time allows
        if time.time() - start_time < budget_s - 30:
            final_h, final_c5 = swap_improvement(best_h, 2.0/n_best, max_swaps=2000)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    
    return (best_h.tolist(), best_c5, n_best)
