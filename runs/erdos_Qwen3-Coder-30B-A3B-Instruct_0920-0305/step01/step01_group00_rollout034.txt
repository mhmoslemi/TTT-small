import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_simplex(y, target_sum):
    y = np.asarray(y)
    n = len(y)
    # Bisection to find lambda such that sum of clipped values equals target_sum
    # Clipped value is clip(y - lambda, 0, 1)
    low, high = np.min(y)-1, np.max(y)+1
    for _ in range(50):
        mid = (low+high)/2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = mid
        else:
            high = mid
    lambda_ = (low+high)/2
    x = np.clip(y - lambda_, 0, 1)
    # Small adjustment to ensure exactly target sum
    x += (target_sum - np.sum(x)) / n
    return x

def compute_grad(h, corr_vals, alpha):
    # Compute softmax weights from corr_vals
    c_max = np.max(corr_vals)
    exp_c = np.exp(alpha * (corr_vals - c_max))
    w = exp_c / np.sum(exp_c)
    
    # Compute gradient ∇h F_α = 1 - sum of (w_s * shifts)
    n = len(h)
    grad = np.ones(n)
    # For s from -n+1 to n-1
    for s in range(-n+1, n):
        if s >= 0:
            shifted = np.concatenate([np.zeros(s), h[:n-s]])
        else:
            shifted = np.concatenate([h[-s:], np.zeros(-s)])
        grad -= w[s + n - 1] * shifted  # Index mapping s -> s+n-1
        
    return grad

def evaluate_c5(h):
    # Compute correlation (full mode), the result of corr(h, 1-h), scaled
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
