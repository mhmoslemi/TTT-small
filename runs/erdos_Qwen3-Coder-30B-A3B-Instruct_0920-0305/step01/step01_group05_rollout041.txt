def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    # Ensure sum equals n/2
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / (2 * current_sum))
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):

    dx = 2.0 / n
    half = n // 2
    start = time.time()

        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        curr_val = evaluate_h(bits.astype(float))
        
        # Local descent
        improved = True
        attempts = 0
        while improved and time.time() - start < time_limit and attempts < 1000:
            improved = False
            attempts += 1
            # Try swaps
            ones_idx = np.flatnonzero(bits)
            zeros_idx = np.flatnonzero(1 - bits)
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Try the swap
            bits[i], bits[j] = bits[j], bits[i]
            new_val = evaluate_h(bits.astype(float))
            
            if new_val < curr_val - 1e-12:
                curr_val = new_val
                improved = True
            else:
                # Revert the swap
                bits[i], bits[j] = bits[j], bits[i]
                
        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
            
        # Early exit if acceptable bound reached
        if best_sum < 0.3795:

        best_bits[:half] = 1
        np.random.shuffle(best_bits)
        best_sum = evaluate_h(best_bits.astype(float))
        
    return best_bits, best_sum
