import numpy as np

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def build_and_solve_lp(n):
    """Build and solve the LP relaxation for the max-min problem"""
    # Decision variables
    x = Variable(n)  # h values at grid points
    w = Variable((n, 2*n-1))  # auxiliary variables for products
    t = Variable()  # minimal correlation
    
    # Build constraints: w_{i,s} = x_i * x_{i+s} where valid
    constraints = []
    
    # Mass constraint
    constraints.append(sum(x) == n/2)
    
    # Box constraints
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # McCormick constraints for each valid pair (i,s)
    for s in range(-(n-1), n):  # shift s from -(n-1) to n-1
        for i in range(n):
            if 0 <= i+s < n:
                w_idx = s + (n-1)
                # w[i,s] <= x[i]
                constraints.append(w[i, w_idx] <= x[i])
                # w[i,s] <= x[i+s]
                constraints.append(w[i, w_idx] <= x[i+s])
                # w[i,s] >= x[i] + x[i+s] - 1
                constraints.append(w[i, w_idx] >= x[i] + x[i+s] - 1)
                # w[i,s] >= 0
                constraints.append(w[i, w_idx] >= 0)
    
    # Constraints on t: sum of w_{i,s} >= t for all shifts
    # For each shift, there is a subset of valid indices
    for s in range(-(n-1), n):
        w_idx = s + (n-1)
        valid_indices = [i for i in range(n) if 0 <= i+s < n]
        #约束对应于 s 位移的 和  索引 i in [0,n) such that i+s in [0,n)
        sum_w = sum(w[i, w_idx] for i in valid_indices)
        constraints.append(sum_w >= t)
    
    # Objective: maximize t
    prob = Problem(Maximize(t), constraints)
    
    # Solve using ECOS or SCS (fast open-source solvers)
    prob.solve(solver=ECOS, verbose=False, max_iters=10000)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        return x.value, t.value
    else:
        return None, None

def deterministic_rounding(x_values, n):
    """Round fractional values to binary, preserving mass."""
    order = np.argsort(x_values)[::-1]  # descending
    b = np.zeros(n)
    for i in range(int(n/2)):
        b[order[i]] = 1.0
    return b

def swap_improvement(h, dx):
    """Perform a local search improving the binary vector with swaps."""
    n = len(h)
    h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    start_time = time.time()
    max_iters = 20000
    no_improvement_count = 0
    iteration = 0
    
    # Precompute the full correlation vector for efficient updates
    # For now, just do a simple hill climbing with full recomputation
    while iteration < max_iters and (time.time() - start_time < 540):
        # Find all 1 and 0 positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Pick random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap them
        h[i], h[j] = h[j], h[i]
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            # Revert swap
            h[i], h[j] = h[j], h[i]
            no_improvement_count += 1
            
        iteration += 1
        
        if no_improvement_count > 2000:
            break
    
    return h, best_c5
