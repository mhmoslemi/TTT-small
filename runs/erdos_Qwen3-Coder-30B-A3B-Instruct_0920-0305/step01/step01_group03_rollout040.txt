import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """
    Project vector v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}
    Uses water-filling algorithm for a capped simplex.
    """
    v = np.asarray(v)
    n = len(v)
    
    # Clip to bounds first
    v = np.clip(v, lower, upper)
    
    # If already at target sum, return
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Simple projection onto simplex: sort, adjust
    # Try water-filling approach
    # We want to find threshold t such that 
    # sum_i clamp(v_i + t, lower, upper) = target_sum
    # Which is the same as sum_i clamp(v_i + t, 0, upper-lower) = target_sum - n*lower
    
    # Start with rough bounds and binary search
    def projected_sum(t):
        return np.sum(np.clip(v + t, lower, upper))
        
    # Find bracketing values for t
    low_t = -100
    high_t = 100
    while projected_sum(low_t) > target_sum:
        low_t -= 10
    while projected_sum(high_t) < target_sum:
        high_t += 10
        
    # Binary search for correct t
    tol = 1e-8
    for _ in range(100):
        mid_t = (low_t + high_t) / 2
        mid_sum = projected_sum(mid_t)
        if abs(mid_sum - target_sum) < tol:
            break
        if mid_sum < target_sum:
            low_t = mid_t
        else:
            high_t = mid_t
    
    # Apply final projection
    result = np.clip(v + (low_t + high_t)/2, lower, upper)
    return result
