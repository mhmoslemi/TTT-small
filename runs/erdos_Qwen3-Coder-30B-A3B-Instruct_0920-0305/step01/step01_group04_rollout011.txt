import numpy as np
from scipy.optimize import dual_annealing
import time
import math

# Constants
n_points = 200
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def objective(h_values):
    """Objective function to minimize: C_5 bound."""
    return evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def optimize_step_function(seed=42):
    """Optimize a step function to minimize C_5."""
    np.random.seed(seed)
    
    # Initial solution: start with random binary pattern with exactly n/2 ones
    h0 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h0[ones_indices] = 1.0
    
    initial_c5 = evaluate_h(h0)
    best_h = h0.copy()
    best_c5 = initial_c5
    
    # Create bounds (each h[i] in [0, 1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add equality constraint by penalizing deviation from sum = 1
    def penalty_constraint(h_values):
        return np.sum(h_values) * dx - 1.0
    
    # Optimizer functions to test
    optimizers = []
    
    # Start with simpler approach if we're left with time
    start_time = time.time()
    budget_s = 600
    
    # Use differential evolution for global search
    try:
        result_de = dual_annealing(objective, bounds, maxiter=1000, 
                                   initial_temp=100, restart_temp_ratio=1e-4,
                                   maxfun=3000, seed=seed)
        if result_de.success:
            new_h = result_de.x
            new_c5 = evaluate_h(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    except Exception:
        pass
    
    # Better approach - heuristic local search, but not so coarse
    # Start with current best and do fast local search for better result
    current_h = best_h.copy()
    
    # Local hill climbing type search with small random increments
    for _ in range(20000):  # a significant random walk
        if time.time() - start_time > budget_s - 5:
            break
        
        # Make random small changes to h values
        idx = np.random.randint(0, n_points)  # pick index
        val = current_h[idx]
        # Slightly perturb
        delta = np.random.normal(0, 0.02)  # small change in range
        new_val = val + delta
        
        # Clip new value to [0,1]
        new_val = max(0, min(1, new_val))
        diff = new_val - val
        # Create candidate
        trial_h = current_h.copy()
        trial_h[idx] = new_val
        
        # Adjust other values to maintain sum constraint (approximation)
        # If we have too many or too few mass, we rebalance a little bit
        sum_h = np.sum(trial_h)
        if abs(sum_h * dx - 1) > 1e-5:
            # Adjust all to make integral exactly 1 (but since we expect small
            # changes we can just scale slightly) 
            scale = 1.0 / (sum_h * dx)
            trial_h *= scale
        
        # Evaluate
        new_c5 = evaluate_h(trial_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = trial_h.copy()
        elif np.random.rand() < 0.01:
            # Occasionally accept worse solutions to escape local minima
            current_h = trial_h.copy()
        else:
            current_h = trial_h.copy()
            
    # Refinement: use a local optimizer (SLSQP) starting from current best
    # We must ensure that the constraints are met
    try:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints,

            new_h = result.x
            if np.sum(new_h) * dx > 0.999 and np.sum(new_h) * dx < 1.001:
                new_c5 = evaluate_h(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
    except Exception:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    start_time = time.time()
    
    # Try to optimize starting from initial_h_values

    # Final validation
    h_final = np.array(h_values)
    sum_h = np.sum(h_final)
    if abs(sum_h * dx - 1.0) > 1e-6:
        # Rescale to enforce integral = 1
        scale = 1.0 / (sum_h * dx)
        h_final *= scale
    
    # Final evaluation
    final_c5 = evaluate_h(h_final)
    
    return list(h_final), final_c5, n_points
