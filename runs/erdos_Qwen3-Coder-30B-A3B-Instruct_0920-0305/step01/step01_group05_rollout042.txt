import numpy as np
from scipy.signal import correlate
import time

def project_onto_box_simplex(x, target_sum):
    """Project x onto the set {y : 0 <= y_i <= 1, sum(y) = target_sum}"""
    # Sort in descending order
    sorted_x = np.sort(x)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    
    # Find the threshold index
    n = len(x)
    i = 0
    while i < n and sorted_x[i] - (target_sum - cumsum[i]) / (i + 1) > 0:
        i += 1
    
    # Compute the threshold
    if i == 0:
        theta = sorted_x[0] - target_sum / n
    else:
        theta = (target_sum - cumsum[i - 1]) / i
    
    # Compute projection
    y = np.maximum(x - theta, 0)
    y = np.minimum(y, 1)
    
    # Adjust for numerical errors
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust by scaling
        y = y * target_sum / current_sum
        
    return y

def compute_fs(h, s):
    """Compute f_s for a given shift s"""
    n = len(h)
    if s >= 0:
        # Right shift
        shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        # Left shift
        shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.sum(h * (1 - shifted))

def compute_all_fs(h):
    """Compute all f_s values for all shifts"""
    n = len(h)
    fs = np.zeros(2 * n - 1)
    for s in range(-n + 1, n):
        fs[s + n - 1] = compute_fs(h, s)
    return fs

def gradient_p_norm(h, p, lambda_penalty=1e3):
    """Compute gradient of Phi_p(h) + lambda_penalty * (sum(h) - n/2)^2"""
    n = len(h)
    fs = compute_all_fs(h)
    abs_fs = np.abs(fs)
    
    # Compute Phi_p
    Phi_p = np.sum(abs_fs ** p) ** (1/p)
    
    # Avoid division by zero
    if Phi_p < 1e-15:
        Phi_p = 1e-15
        
    # Compute weights
    weights = abs_fs ** (p - 1)
    # To avoid infinities or NaNs, handle the case where f_s = 0
    weights = np.where(abs_fs == 0, 0, weights)
    
    # Compute the contribution from the p-norm
    grad = np.zeros(n)
    for s in range(-n + 1, n):
        if s >= 0:
            shifted_h = np.concatenate([np.zeros(s), h[:-s]])
        else:
            shifted_h = np.concatenate([h[-s:], np.zeros(-s)])
        this_weight = weights[s + n - 1]
        
        # Gradient contribution is this_weight * (1 - h_{k+s} - h_{k-s})
        # Shift the derivative appropriately.
        for i in range(n):
            # [-1] -> h_{i-1}, [0] -> h_i, [1] -> h_{i+1}
            term = this_weight * (1 - shifted_h[i])
            if i-1 >= 0:
                grad[i-1] += -term
            grad[i] += term
            if i+1 < n:
                grad[i+1] += -term
                
    # Correct the contribution from u_{i+s} (grad[i] from h_{i+s} and h_{i-s})
    # Compute additional terms
    term1 = np.zeros(n) # h_{i+s} 
    term2 = np.zeros(n) # h_{i-s}
    
    for s in range(1, n):
        shifted1 = np.concatenate([np.zeros(s), h[:-s]])
        shifted2 = np.concatenate([h[s:], np.zeros(s)])
        term1 += weights[n-1+s] * shifted1
        term2 += weights[n-1-s] * shifted2
        
    # Combine with the gradient from the p-norm:
    # The correct derivation
    grad = grad / (Phi_p ** (p - 1)) + 2 * lambda_penalty * (np.sum(h) - n/2)

    return grad

def smooth_max(h, p):
    """Return smooth approximation to max of f_s"""
    fs = compute_all_fs(h)
    abs_fs = np.abs(fs)
    return np.sum(abs_fs ** p) ** (1/p)
