```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Constants
n_points = 100
dx = 2.0 / n_points
initial_h = np.array(initial_h_values)

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_simplex(y, target_sum):
    """Project y onto the simplex {x : sum(x) = target_sum, x >= 0}"""
    # Clip to [0,1], then rescale
    y = np.clip(y, 0, 1)
    total = np.sum(y)
    if total > target_sum:
        # Need to decrease entries
        # Sort in descending order and shrink
        sorted_idx = np.argsort(-y)
        cumsum = np.cumsum(y[sorted_idx])
        # Find threshold index
        k = np.argmax(cumsum > target_sum)
        if cumsum[k] <= target_sum:
            # All entries can be reduced
            theta = 0.0
        else:
            theta = (cumsum[k] - target_sum) / (k+1)
        # Apply threshold
        y[sorted_idx] = np.maximum(y[sorted_idx] - theta, 0)
    else:
        # Need to increase entries
        delta = target_sum - total
        # Distribute delta among all positive entries
        # Make it proportional to 1/(n + 1) so that rounding is stable
        y += delta / len(y)
    # Ensure the sum is exactly target_sum
    y *= target_sum / np.sum(y)
    return y

def correlation(h):
    """Return full correlation vector (h with 1-h)"""
    return np.correlate(h, 1 - h, mode="full")

def gradient_mismatch(h, d_star):
    """Compute ∂m_d_star / ∂h_i"""
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        # Shifted indices with wrap-around
        i_plus = (i + d_star) % n
        i_minus = (i - d_star) % n
        grad[i] = 1.0 - h[i_plus] - h[i_minus]
    return grad

def subgradient_descent_step(h, d_star, alpha):
    """Apply one subgradient descent step"""
    g = gradient_mismatch(h, d_star)
    h_new = h - alpha * g
    # Project back onto feasible set
    h_new = project_onto_simplex(h_new, n_points / 2)
    return h_new

def run(seed=42, budget_s=600):
    """Main function to run optimization"""
    np.random.seed(seed)
    start_time = time.time()

    # Try exact solution first (MILP)
    best_h = None
    best_c5 = float('inf')
    try:
        # Skip MILP for very large n. Use the heuristic to solve faster
        if n_points > 100:
            raise RuntimeError("Skip MILP for larger n, use heuristic")
        
        # Binary variables h_i
        h = cp.Variable(n_points, boolean=True)
        # Auxiliary variables y_{i,d} = h_{i} * h_{i+d}
        y = cp.Variable((n_points, 2*n_points - 1))
        t = cp.Variable()
        
        # Sum constraints
        constraints = [cp.sum(h) == n_points / 2]

        # Add y_{i,d} constraints using big-M
        for d in range(-n_points+1, n_points):
            for i in range(n_points):
                # Ok, only if both indices within valid range
                if 0 <= i+d < n_points:
                    # y[i,d] = h[i]*h[i+d]
                    M = 1  # Since h_{i,d} are 0/1
                    # All constraints needed to define y correctly are:
                    constraints.append(y[i, d + (n_points - 1)] <= h[i])
                    constraints.append(y[i, d + (n_points - 1)] <= h[i+d])
                    constraints.append(y[i, d + (n_points - 1)] >= h[i] + h[i+d] - 1)
                    constraints.append(y[i, d + (n_points - 1)] >= 0)
                else:
                    # Outside valid range (already implicitly 0) so we don't add anything

        # Enforce M ≤ w_d - sum(y_{i,d})
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue  # Not needed
            # Only compute if both i and i+d lie in 0..n-1 for any i
            valid_indices = []
            for i in range(n_points):
                if 0 <= i+d < n_points:
                    valid_indices.append(i)
            if len(valid_indices) > 0:
                sum_y = cp.sum([y[i, d + (n_points - 1)] for i in valid_indices])
                w_d = cp.sum([h[i] for i in valid_indices])
                constraints.append(t >= w_d - sum_y)

        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=CPLEX, verbose=False, max_seconds=budget_s / 2)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value.astype(float)
            c5 = evaluate_h(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        else:
            # If MILP fails, proceed to heuristic
            pass
    except Exception:
        pass
    
    # Fallback to heuristic if no good solution from MILP
    if best_h is None:
        # Initialize from the previous construction
        h_initial = initial_h.copy()
        h_initial = h_initial * (n_points / 2.0) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = project_onto_simplex(h_initial, n_points / 2)

        max_iter = int(100000)
        current_h = h_initial.copy()
        current_c5 = evaluate_h(current_h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

        iter_count = 0
        alpha = 0.5
        T0 = 1.0
        gamma = 0.999
        t0 = time.time()
        while time.time() - t0 < budget_s - 5:
            # Get all correlations
            corr_vals = correlation(current_h)
            # Find index of max (all entries including full correlation)
            k = np.argmax(corr_vals)
            # Read d from k
            d_star = k - (2 * n_points - 2) // 2  # Midpoint index mapping back to shift offset
            # Make sure it's within bounds
            d_star = max(-n_points + 1, min(d_star, n_points - 1))
            
            # Update with gradient step
            h_next = subgradient_descent_step(current_h, d_star, alpha)
            
            # Annealing accept/reject based on change in mismatch
            old_corr = corr_vals[k]
            new_corr = np.correlate(h_next, 1 - h_next, mode="full")[k]
            delta_energy = new_corr - old_corr
            
            T = T0 * (gamma ** iter_count)
            if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / (T + 1e-8)):
                current_h = h_next
                current_c5 = evaluate_h(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    
            # Update step size
            alpha = 0.5 / np.sqrt(iter_count + 1)
            iter_count += 1

    # Final polishing to binary
    if best_h is not None:
        h_final = best_h.copy()
        # Sort values descending
        indices = np.argsort(-h_final)
        h_binary = np.zeros_like(h_final)
        half_count = n_points // 2
        for i in range(half_count):
            h_binary[indices[i]] = 1.0
        # If exact half isn't achievable, leave one fractional entry
        if n_points % 2 == 1:
            # The remaining slot
            idx = indices[half_count]
            h_binary[idx] = 0.5
        else:
            # Check if sum is still right
            pass

        # Local hill climbing
        while True:
            improved = False
            for i in range(n_points):
                for j in range(n_points):
                    if h_binary[i] == 1.0 and h_binary[j] == 0.0:
                        # Swap
                        h_test = h_binary.copy()
                        h_test[i] = 0.0
                        h_test[j] = 1.0
                        # Recompute C5
                        c5_new = evaluate_h(h_test)
                        if c5_new < best_c5:
                            h_binary = h_test.copy()
                            best_c5 = c5_new
                            improved = True
                            break
                if improved:
                    break
            if not improved:
                break

        return list(h_binary), best_c5, n_points

    # If all fails, return initial construction
    return list(initial_h), evaluate_h(initial_h), n_points
```
