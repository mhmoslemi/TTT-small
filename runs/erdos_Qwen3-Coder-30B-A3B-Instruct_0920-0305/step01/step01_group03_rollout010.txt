def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):

    v_shift = v - lower

    for i in range(len(sorted_values)):

        # The value where we would hit target with these values
        val = sorted_values[i]  
        # If value 0, we don't add anything new, so break early
        if val <= 0:
            break
        if sum_so_far >= target_shift:
            # The current var is our threshold
            threshold = val
            break
    
    # If not found, we might need to use a different approach: binary search
    if threshold == -1e10 and len(sorted_values) > 0:
        # Binary search directly on theta
        # Shifting by a constant theta: arr[i] = min(upper-lower, max(0, v[i] + theta))
        low = -10.0
        high = 10.0
        eps = 1e-10
        
        def compute_sum(theta):
            proj = np.clip(v_shift + theta, 0, upper - lower)
            return np.sum(proj)
            
        # Bracket
        while compute_sum(low) < target_shift - eps:
            low *= 2
        while compute_sum(high) > target_shift + eps:
            high *= 2
            
        # Binary search
        while high - low > eps:
            mid = (low + high) / 2
            if compute_sum(mid) > target_shift:
                high = mid
            else:
                low = mid
        theta = (low + high) / 2
    else:
        # Calculate the shift to make the sum match target
        if sum_so_far < target_shift:
            theta = 0
        else:
            # We need to adjust theta to bring sum closer to target
            # This is more complex so fallback to the iterative method
            # For now just calculate approximate, works in most cases
            # Use standard iterative projection  
            # compute sum of clipped values with a specific theta
            # an initial guess: let's just do occasional direct computation:
            sample_thetas = np.logspace(-2, 2, 20)
            sums = [compute_sum(t) for t in sample_thetas]
            i_min = np.argmin(np.abs(np.array(sums) - target_shift)) 
            theta = sample_thetas[i_min] 
            # verify effect:
            actual_sum = compute_sum(theta)
            if abs(actual_sum - target_shift) > 1e-6:
                # Refine with using bisection based on current
                low_theta = np.log10(max(1e-4, theta * 0.5)) 
                high_theta = np.log10(min(1e4, theta * 2.0))
                # Binary search in log space 
                for _ in range(20):
                    mid = (low_theta + high_theta) / 2
                    t = 10 ** mid
                    s = compute_sum(t)
                    if s > target_shift:
                        high_theta = mid
                    else:
                        low_theta = mid
                theta = 10 ** (low_theta + high_theta)/2
            
        # For final construction with best binary search
    final_probs = np.clip(v_shift + theta, 0, upper - lower) 
    final_result = final_probs + lower
    return final_result

            else:
                # s < 0, represent shift by -s
                off = -s
