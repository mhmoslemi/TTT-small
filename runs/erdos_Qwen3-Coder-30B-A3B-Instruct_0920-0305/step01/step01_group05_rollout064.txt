import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def ccp_solve(h0, n_points, start_time, budget_s):
    """Convex-Concave Procedure solver for minimizing C5"""
    dx = 2.0 / n_points
    max_iter = 30
    tol = 1e-6
    h_prev = h0.copy()
    
    # Precompute shift indices
    shift_indices = {}
    for s in range(-(n_points - 1), n_points):
        mask_fwd = []
        mask_bwd = []
        for i in range(n_points):
            if i + s >= 0 and i + s < n_points:
                mask_fwd.append(i)
            if i - s >= 0 and i - s < n_points:
                mask_bwd.append(i)
        shift_indices[s] = (np.array(mask_fwd), np.array(mask_bwd))
    
    # Clip initial h to [0,1]
    h = np.clip(h_prev, 0, 1)
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve convex subproblem using CVXPY
        h_new, tau = solve_convex_subproblem(h, shift_indices, n_points, dx)
        
        if h_new is None:
            # Fall back to previous iterate or clip
            h_new = np.clip(h, 0, 1)
            
        # Check convergence
        rel_diff = np.linalg.norm(h_new - h_prev) / (np.linalg.norm(h_prev) + 1e-12)
        if rel_diff < tol:
            break
            
        h_prev = h.copy()
        h = h_new.copy()
        
        # Ensure the integral constraint
        h = h * (1.0 / (np.sum(h) * dx))
    
    return h, evaluate_h(h)

def solve_convex_subproblem(h, shift_indices, n_points, dx):
    """Solve the convex surrogate CCP subproblem via CVXPY"""
    try:
        h_prev = h.copy()
        h_var = cp.Variable(n_points)
        tau = cp.Variable()
        
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2.0
        ]
        
        # Compute g_s^d(h_prev) and ∇g_s^d(h_prev) for all shifts
        for s in range(-(n_points - 1), n_points):
            if s == 0:
                continue
                
            mask_fwd, mask_bwd = shift_indices[s]
            # Compute ∇g_s^d(h_prev) = [2*(h[i] + h[i+s]) for i]
            # For i in mask_fwd (i+s < n)
            grad_small = np.zeros(n_points)
            if len(mask_fwd) > 0:
                grad_small[mask_fwd] = h_prev[mask_fwd] + h_prev[mask_fwd + s]
            # For i in mask_bwd (i-s >= 0)
            if len(mask_bwd) > 0:
                grad_small[mask_bwd] += h_prev[mask_bwd] + h_prev[mask_bwd - s]
            grad_small *= 2.0
            
            # Compute  g_s^d(h_prev)
            g_s_d_val = 0.5 * np.sum((h_prev + np.roll(h_prev, s)) ** 2)
            
            # Constraint: tau >= g_s^c(h) - (g_s^d(h_prev) + grad_s^T * (h - h_prev))
            # g_s^c(h) = 1 * h + 0.5 * sum(h[i]^2) + 0.5 * sum(h[i+s]^2)
            
            # For small indices only (efficiency)
            if len(mask_fwd) > 0:
                term1 = cp.sum(h_var[mask_fwd]) + 0.5 * cp.sum_squares(h_var[mask_fwd])
                term2 = 0.5 * cp.sum_squares(h_var[mask_fwd + s])
            else:
                term1 = cp.sum(h_var)  # All h_i
                term2 = 0.0
                
            if len(mask_bwd) > 0:
                term3 = cp.sum(h_var[mask_bwd]) + 0.5 * cp.sum_squares(h_var[mask_bwd])
                term4 = 0.5 * cp.sum_squares(h_var[mask_bwd - s])
            else:
                term3 = 0.0
                term4 = 0.0
                
            # Full term for current s
            g_s_c = term1 + term2 + term3 + term4
                
            # Affine term for linearization
            affine_term = g_s_d_val + np.dot(grad_small, (h_var - h_prev))
            
            constraints.append(tau >= g_s_c - affine_term)
        
        prob = cp.Problem(cp.Minimize(tau), constraints)
        
        # Only try to solve if not time running out
        if time.time() - start_time > budget_s - 10:
            return None, None
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_var.value
            tau_val = tau.value
            return h_new, tau_val
        else:
            return None, None
    except Exception as e:
        return None, None

def coordinate_descent_polish(h0, n_points, start_time, budget_s):
    """Apply coordinate-descent polishing to improve solution"""
    h = h0.copy()
    dx = 2.0 / n_points
    
    # Normalize initially
    h = h * (1.0 / (np.sum(h) * dx))
    current_c5 = evaluate_h(h)
    
    iterations = 3  # Number of sweeps
    
    for sweep in range(iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Shuffle indices randomly
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        # For each index
        for i in indices:
            if time.time() - start_time > budget_s - 5:
                break
            
            # Temporarily adjust h[i] to try different values
            original_h_i = h[i]
            best_h_i = original_h_i
            best_c5 = current_c5
            
            # Test boundary values
            if original_h_i != 0:
                h[i] = 0
                test_c5 = evaluate_h(h)
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_h_i = 0
                    
            if original_h_i != 1:
                h[i] = 1
                test_c5 = evaluate_h(h)
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_h_i = 1
                    
            # Try intermediate values
            step = 0.05
            for val in [original_h_i - step, original_h_i + step]:
                if 0 <= val <= 1:
                    h[i] = val
                    test_c5 = evaluate_h(h)
                    if test_c5 < best_c5:
                        best_c5 = test_c5
                        best_h_i = val
                        
            # Final update
            h[i] = best_h_i
            current_c5 = best_c5
            
    return h, current_c5
