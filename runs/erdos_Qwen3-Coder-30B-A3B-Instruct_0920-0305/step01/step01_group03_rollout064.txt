import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def logsumexp_stable(x, axis=None):
    """Compute log(sum(exp(x))) with numerical stability"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def project_capped_simplex(h, target_sum=1.0, bounds=(0, 1)):
    """Project h onto the capped simplex: sum(h) = target_sum, bounds[i] <= h[i] <= bounds[1]"""
    # Clip to bounds
    h = np.clip(h, bounds[0], bounds[1])
    # Adjust to sum to target_sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Use water filling
    # Sorted h values
    sorted_h = np.sort(h)[::-1]
    # Find where to cut
    cumsum = np.cumsum(sorted_h)
    n = len(sorted_h)
    # Find the index where we need to truncate
    # We solve: (target_sum - cumsum[i]) / (n - i) = sorted_h[i]
    # Or: target_sum = cumsum[i] + (n - i) * sorted_h[i]
    # We want the largest i such that sorted_h[i] >= (target_sum - cumsum[i]) / (n - i)
    for i in range(n):
        if sorted_h[i] >= (target_sum - cumsum[i]) / (n - i):
            continue
        # Check if we should take this point
        threshold = (target_sum - cumsum[i]) / (n - i)
        if i == 0:
            break
        if sorted_h[i-1] >= threshold:
            break
        # i is correct
        if i == n:
            # All values should be same
            return np.full_like(h, target_sum / n)
        threshold = (target_sum - cumsum[i-1]) / (n - (i-1))
        # Compute the adjusted values
        result = h.copy()
        result[result > threshold] = threshold
        result = np.clip(result, bounds[0], bounds[1])
        current_sum = np.sum(result)
        diff = target_sum - current_sum
        if abs(diff) < 1e-12:
            return result
        # Add diff uniformly
        diff_per = diff / np.sum(result < threshold)
        result[result < threshold] += diff_per
        return np.clip(result, bounds[0], bounds[1])
    # All values at the threshold
    threshold = target_sum / n
    result = np.full_like(h, threshold)
    return np.clip(result, bounds[0], bounds[1])

def symmetric_project(h, target_sum=1.0):
    """Symmetrize h with respect to midpoint and project to sum=target_sum"""
    n = len(h)
    # Symmetrize
    h_sym = np.zeros_like(h)
    for i in range((n + 1) // 2):
        j = n - 1 - i
        h_sym[i] = h_sym[j] = (h[i] + h[j]) / 2.0
    # Project to sum=target_sum
    return project_capped_simplex(h_sym, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    best_c5 = 1.0
    best_h = None
    start_time = time.time()
    
    # Helper to compute the soft minimum function
    def compute_surf(h, lmbda):
        correlation = np.correlate(h, h, mode="full")
        # Numerically stable log-sum-exp
        corr_scaled = -lmbda * correlation
        corr_scaled -= np.max(corr_scaled)
        log_exp_sum = np.log(np.sum(np.exp(corr_scaled)))
        phi = log_exp_sum / lmbda
        return correlation, phi
    
    # Helper to compute gradient
    def compute_gradient(h, lmbda):
        # Compute correlations
        correlation, _ = compute_surf(h, lmbda)
        # Compute weights
        corr_scaled = -lmbda * correlation
        corr_scaled -= np.max(corr_scaled)
        weights = np.exp(corr_scaled)
        weights /= np.sum(weights)
        # Compute gradient
        g = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            for s in range(-n+1, n):  # s from -n+1 to n-1
                j1 = i + s
                j2 = i - s
                if 0 <= j1 < n:
                    g[i] += weights[s + n - 1] * h[j1]
                if 0 <= j2 < n and j2 != j1:
                    g[i] += weights[s + n - 1] * h[j2]
        return g
    
    # Annealing phases
    phases = [
        {'lmbda': 0.5,   'alpha': 0.3, 'n_iter': 150},
        {'lmbda': 5.0,   'alpha': 0.1, 'n_iter': 200},
        {'lmbda': 30.0,  'alpha': 0.03, 'n_iter': 300}
    ]
    
    def backtracking_line_search(h, grad, lmbda, alpha, max_backtrack=5):
        h_original = h.copy()
        for _ in range(max_backtrack):
            h_new = h_original + alpha * grad
            h_new = project_capped_simplex(h_new, n_points/2)
            # Check surrogate decrease
            _, phi_old = compute_surf(h_original, lmbda)
            _, phi_new = compute_surf(h_new, lmbda)
            if phi_new <= phi_old * 0.999:
                return h_new, alpha
            alpha /= 2.0
        return h_original, alpha
    
    # Multi-start
    num_starts = 10
    for start_idx in range(num_starts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random start
        h = np.random.rand(n_points)
        h = project_capped_simplex(h, n_points/2)
        # Symmetrize
        h = symmetric_project(h, n_points/2)
        
        # Warm-up and annealing
        for phase in phases:
            lmbda = phase['lmbda']
            alpha = phase['alpha'] 
            n_iter = phase['n_iter']
            
            for _ in range(n_iter):
                if time.time() - start_time > budget_s * 0.9:
                    break
                g = compute_gradient(h, lmbda)
                h_new, alpha = backtracking_line_search(h, g, lmbda, alpha)
                h = h_new
                # Symmetrize after each update
                h = symmetric_project(h, n_points/2)
        
        # Evaluate current solution
        tmp_c5 = evaluate_erdos_solution(h, dx)
        if tmp_c5 < best_c5:
            best_c5 = tmp_c5
            best_h = h.copy()
    
    # Final polishing with SQP
    if best_h is not None:
        def objective(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
            
        # Constraints
        def constraint_sum(h_vals):
            return np.sum(h_vals) * dx - 1.0
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if result.success:
            final_c5 = objective(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
        else:
            # If SQP fails, use the best solution from gradient descent
            pass
    
    return (best_h.tolist(), best_c5, n_points)
