def full_correlation(h):
    """Compute integer-valued correlation vector and maximum value."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return corr, max_corr

def nearest_balanced_binary(s):
    """Convert signed vector to balanced binary with exact sum constraint."""
    n = len(s)
    s = np.array(s)
    # Decide what to set to 0.5
    target_sum = n / 2.0
    current_sum = np.sum(s)
    # Convert to binary + 0.5
    h = (s + 1) / 2
    # Adjust to get exact sum
    diff = int(current_sum - target_sum)
    if diff == 0:
        return h
    # Likely arises if we have a non-integer n or error
    if abs(diff) > 0:
        # Find indices to flip
        diff = int(diff)
        if diff > 0:
            # Need to reduce sum by diff; set diff 1s to 0s
            indices_to_flip = np.argsort(s)[::-1][:diff]  # largest values first
        else:
            # increase sum by |diff|; set |diff| 0s to 1s
            indices_to_flip = np.argsort(s)[:abs(diff)]  # smallest values first
        for i in indices_to_flip:
            h[i] = 1 - h[i]
    return h

    target_sum = n0 / 2.0  # We’ll handle exact sum later
    
    # Step 2 - Define Toeplitz matrices (sparse assembly)
    C_d = {}  # Store C_d for d = -n0+1,...,n0-1 \ {0}
    shift_range = list(range(1, n0)) + list(range(-n0+1, 0))

    def build_C(d):
        """Build sparse shift matrix"""
        rows, cols = [], []
        data = []
        for i in range(n0):
            j = i + d
            if 0 <= j < n0:
                rows.append(i)
                cols.append(j)
                data.append(1)
                # Symmetric:
                rows.append(j)
                cols.append(i)
                data.append(1)
        row = np.array(rows)
        col = np.array(cols)
        dat = np.array(data)
        C_d[d] = (row, col, dat)
    
    for d in shift_range:
        build_C(d)

    # Step 3 - Initialize SDP problem
    X = cp.Variable((n0, n0), symmetric=True)
    tau = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == 1  # Unit diagonal
    ]
    # All auto-correlations should be bounded below
    for d in shift_range:
        row, col, dat = C_d[d]
        # This only adds all the ones in C_d (note it has 2x) to compute the trace
        # So the correct formulation is:
        # trace(C_d @ X) >= tau equivalent to: sum_i sum_j C[i,j] * X[i,j] >= tau
        # But we only represent nonzeros: so k goes through nonzero entries of C_d
        term = cp.sum(cp.multiply(dat, X[row, col]))  # Efficient way to compute trace(C_d @ X)
        constraints.append(term >= tau)

    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)
    
    # Step 4 - Solve SDP
    try:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-4)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            print(f"SDP solving failed with status: {prob.status}")
            # Fall back to simpler case
            raise Exception("SDP failed to solve cleanly")
        
        tau_opt = prob.value
        X_opt = X.value
        print(f"Found SDP optimum τ = {tau_opt}")
    except Exception as e:

    n_upscaled = len(h_upscaled)
    desired_length = n_points
    if n_upscaled > desired_length:

                            improved = True

    integral = np.sum(h_final) * dx
    if abs(integral - 1.0) > 1e-5:
        # Rescale so that integral = 1
        scale = 1.0 / integral
        h_final *= scale

    final_c5 = eval_c5(h_final)

    return list(h_final), final_c5, len(h_final)
