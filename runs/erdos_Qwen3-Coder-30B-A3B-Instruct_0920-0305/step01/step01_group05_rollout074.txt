import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def solve_ccp_subproblem(h_prev, n, dx, max_iter=30, tol=1e-6):
    """Solve the convex-concave subproblem using CVXPY"""
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Precompute masks for forward and backward indices
    constraints = []
    
    # Build the constraints for each shift
    for s in range(-(n-1), n):
        mask_fwd = np.arange(n) + s < n
        mask_bwd = np.arange(n) - s >= 0
        
        # g_s^c(h) = sum(h) + 0.5 * sum(h_i^2 + h_{i+s}^2)
        g_s_c = cp.sum(h) + 0.5 * cp.sum_squares(h[mask_fwd]) + 0.5 * cp.sum_squares(h[mask_bwd])
        
        # g_s^d(h) = 0.5 * sum((h_i + h_{i+s})^2)
        g_s_d_val = 0.5 * cp.sum_squares(h[mask_fwd] + h[mask_bwd])
        
        # Compute gradient of g_s^d at h_prev
        g_s_d_grad = np.zeros(n)
        for i in range(n):
            if i + s < n:
                g_s_d_grad[i] += h_prev[i] + h_prev[i + s]
            if i - s >= 0:
                g_s_d_grad[i] += h_prev[i - s] + h_prev[i]
        
        # Affine linearization: g_s^d(h_prev) + grad^T * (h - h_prev)
        affine_term = g_s_d_val.value + np.dot(g_s_d_grad, h - h_prev)
        
        constraints.append(tau >= g_s_c - affine_term)
    
    # Add bounds and integral constraint
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == n/2)
    
    # Define and solve the problem
    objective = cp.Minimize(tau)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        h_new = h.value
        if h_new is None:
            h_new = h_prev
    except Exception:
        h_new = h_prev
    
    return h_new

def coordinate_descent_polish(h, n, dx, max_sweeps=5):
    """Perform coordinate descent polishing on the solution"""
    h = h.copy()
    total_sum = np.sum(h)
    
    for sweep in range(max_sweeps):
        # Shuffle indices to randomize order
        indices = list(range(n))
        np.random.shuffle(indices)
        
        improved = False
        for i in indices:
            # Compute all shift terms for current h values to find active shifts
            current_i = h[i]
            best_h_i = current_i
            min_max_val = float('inf')
            
            # Check all possible values of h_i between 0 and 1
            active_shifts = []
            for s in range(-(n-1), n):
                if 0 <= i+s < n:
                    active_shifts.append(s)
            
            # Evaluate each shift at boundary values and critical points
            values_to_check = [0.0, 1.0]
            
            # For each active shift, find critical points
            for s in active_shifts:
                if s > 0:
                    j = i + s
                    # f_s = h[i] + h[j] - h[i]*h[j] depends on h[i] only
                    # At critical point: df/dh[i] = 1 - h[j] = 0 => h[i] = 1
                    if 0 <= i+s < n:
                        values_to_check.append(1.0 if h[i+s] == 1 else 0.0)
                elif s < 0:
                    j = i + s
                    # At critical point: df/dh[i] = 1 - h[j] = 0 => h[i] = 1
                    if 0 <= i+s < n:
                        values_to_check.append(1.0 if h[i+s] == 1 else 0.0)
            
            # Evaluate objective at values
            values_to_check = list(set(values_to_check))
            for candidate_h in values_to_check:
                if candidate_h < 0 or candidate_h > 1:
                    continue
                    
                h_test = h.copy()
                h_test[i] = candidate_h
                # Adjust total_sum to maintain integral constraint
                diff = h_test[i] - current_i
                # Find a suitable j to compensate (find a non-saturated element)
                j = -1
                for k in range(n):
                    if h_test[k] < 1 and k != i:
                        j = k
                        break
                if j == -1:
                    for k in range(n):
                        if h_test[k] > 0 and k != i:
                            j = k
                            break
                if j != -1:
                    # We can adjust the complementing element so that total_sum remains unchanged
                    h_test[j] = max(0, min(1, h_test[j] - diff))
                
                corr = np.correlate(h_test, 1 - h_test, mode="full")
                max_val = np.max(corr) * dx
                if max_val < min_max_val:
                    min_max_val = max_val
                    best_h_i = candidate_h
            
            if best_h_i != current_i:
                improved = True
                h[i] = best_h_i
                diff = h[i] - current_i
                # Adjust another element
                j = -1
                for k in range(n):
                    if h[k] < 1 and k != i:
                        j = k
                        break
                if j == -1:
                    for k in range(n):
                        if h[k] > 0 and k != i:
                            j = k
                            break
                if j != -1:
                    h[j] = max(0, min(1, h[j] - diff))
        
        if not improved:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Resolution hierarchy
    resolutions = [60, 100, 150]
    
    h_current = None
    
    for n in resolutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n
        if h_current is None:
            h_current = initial_h(n)
        else:
            # Interpolate from previous resolution
            h_current = interp1d(np.linspace(0, 2, len(h_current)), h_current, kind='linear', fill_value="extrapolate")(np.linspace(0, 2, n))
            h_current = h_current * (1.0 / (np.sum(h_current) * dx))
        
        # CCP iterations
        for iter_count in range(30):
            if time.time() - start_time > budget_s - 5:
                break
                
            prev_h = h_current.copy()
            h_current = solve_ccp_subproblem(h_current, n, dx)
            
            # Compute current C5
            corr = np.correlate(h_current, 1 - h_current, mode="full")
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                best_n = n
                
            # Stopping condition
            if np.linalg.norm(h_current - prev_h, ord=1) < 1e-6:
                break
        
        # Coordinate-descent polishing
        h_current = coordinate_descent_polish(h_current, n, dx, 5)
        
        # evaluate the polished result
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            best_n = n
    
    if best_h is None:
        # Fallback to initial guess
        best_h = initial_h(42)
        best_n = 42
        dx = 2.0 / best_n
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_c5 = np.max(corr) * dx
