def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def solve_ccp(h_init, n, start_time, budget_s):
    # Build masks for shifts
    masks = {}
    for s in range(-(n-1), n):
        mask_fwd = []
        mask_bwd = []
        for i in range(n):
            if i + s < n and i + s >= 0:
                mask_fwd.append(i)
                mask_bwd.append(i + s)
        masks[s] = (np.array(mask_fwd), np.array(mask_bwd))
    
    # Initialize
    h_prev = h_init.copy()
    tau_prev = float('inf')
    
    # Specify solver options
    solver_opts = {'abstol': 1e-8, 'reltol': 1e-8}
    
    # CCP iterations
    max_iter = 30
    for t in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Build CVXPY problem
        h = cp.Variable(n)
        tau = cp.Variable()
        
        constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]
        
        for s in range(-(n-1), n):
            # Precomputed masks for fast indexing
            fwd_idx, bwd_idx = masks[s]
            # Compute terms
            if len(fwd_idx) > 0 and len(bwd_idx) > 0:
                # g_s^d(h) = 0.5 * sum (h_i + h_{i+s})^2
                # Gradient of g_s^d is (h_i + h_{i+s}) + (h_{i-s} + h_i)
                grad_term = 0.0
                if s > 0:
                    grad_term += 2 * (h_prev[:-s] + h_prev[s:])
                else:
                    grad_term += 2 * (h_prev[-s:] + h_prev[:-s])
                
                # Affine linearization at h_prev
                g_d_at_prev = 0.5 * np.sum((h_prev[:-s] + h_prev[s:]) ** 2) if s > 0 else 0.5 * np.sum((h_prev[-s:] + h_prev[:-s]) ** 2)
                
                proj = cp.sum(h) + 0.5 * cp.sum_squares(h[fwd_idx]) + 0.5 * cp.sum_squares(h[bwd_idx])
                affine = g_d_at_prev + cp.sum(grad_term * (h - h_prev)) if s > 0 else g_d_at_prev + cp.sum(grad_term * (h - h_prev))
                
                constraints.append(tau >= proj - affine)
            else:
                # If idx invalid or empty, just use convex quadratic version of g_s^c
                proj = cp.sum(h) + 0.5 * cp.sum_squares(h) + 0.5 * cp.sum_squares(h)
                affine = 0
                constraints.append(tau >= proj - affine)
        
        # Problem
        prob = cp.Problem(cp.Minimize(tau), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, **solver_opts)
        except Exception:
            # In case of solver issues, return previous solution
            return h_prev
            
        h_new = h.value
        if h_new is None:
            return h_prev
            
        # Update h_prev & tau
        h_prev = h_new
        tau_prev = tau.value
        
        # Check convergence
        if abs(tau_prev) < 1e-6:
            break
            
    return h_prev

def coordinate_descent_polishing(h_init, n, start_time, budget_s):
    h = h_init.copy()
    dx = 2.0 / n
    
    def shift_term(i, s, h_vec):
        # Compute f_s(h) = sum h_i * (1 - h_{i+s})
        total = 0
        for k in range(n):
            if k + s >= 0 and k + s < n:
                total += h_vec[k] * (1 - h_vec[k + s])
        return total
    
    def all_shifts(i, h_vec):
        # Compute f_s(h) for all s at fixed i
        values = []
        for s in range(-(n - 1), n):
            values.append(shift_term(i, s, h_vec))
        return np.array(values)
    
    # Do a few sweeps
    sweeps = 5
    for sweep in range(sweeps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random order
        indices = list(range(n))
        np.random.shuffle(indices)
        
        for i in indices:
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute current best value and all possible active shifts
            cur_val = all_shifts(i, h)
            best_s = np.argmax(cur_val)
            f_max_s = cur_val[best_s]
            
            # Try small adjustments from 0 to 1, and compare with boundary values
            best_val = float('inf')
            best_h_i = h[i]
            
            candidates = [0, 1, h[i]]
            for h_i_test in candidates:
                new_h = h.copy()
                new_h[i] = h_i_test
                # Maintain sum constraint by adjusting a complementary index
                # Pick a non-saturated at random
                j = np.random.choice(n)
                while new_h[j] <= 0 or j == i:
                    j = np.random.choice(n)
                delta = h_i_test - h[i]
                new_h[j] = np.clip(new_h[j] - delta, 0, 1)
                if abs(np.sum(new_h) * dx - 1.0) < 1e-8:
                    # Evaluate
                    h_vals = np.array([shift_term(k, s, new_h) for s in range(-(n - 1), n) for k in range(n)])
                    testval = np.max(h_vals) * dx
                    if testval < best_val:
                        best_val = testval
                        best_h_i = h_i_test
                        
            h[i] = best_h_i
            # Recompute sum constraint compliance
            # Adjust a complementary index
            j = np.random.choice(n)
            while h[j] <= 0 or j == i:
                j = np.random.choice(n)
            h[j] = np.clip(h[j] - (best_h_i - h[i]), 0, 1)
    
    return h
