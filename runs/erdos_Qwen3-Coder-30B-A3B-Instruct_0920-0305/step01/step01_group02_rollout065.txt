import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    # Compute correlations for all lags
    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx = 2/N
    dx = 2.0 / len(h)
    correlation *= dx
    # Return maximum correlation value
    return np.max(correlation)

def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    # Simple projection: clip and normalize
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale = target_sum / current_sum if current_sum > 0 else 1.0
    x = x * scale
    return x

def initial_h_values():
    # A better starting construction (based on the analysis of equivalent problems)
    # Using a piecewise constant function with specific values to reduce overlap
    n_points = 47
    h = np.zeros(n_points)
    
    # Divide the interval [0,2] into 4 segments
    # Make the function 0 for first 1/4, 1 for next 1/4, 0 for next 1/4, 1 for last 1/4
    # We need to modify this to make the integral equal to 1
    
    segment_length = 2.0 / 4.0
    dx = 2.0 / n_points
    
    # Set up four segments: zeros and ones  
    for i in range(n_points):
        x_pos = i * dx
        if (x_pos >= 0 and x_pos < segment_length) or (x_pos >= 1.5 * segment_length and x_pos < 2.0):
            h[i] = 0.0
        else:
            h[i] = 1.0
    
    # Scale appropriately so that sum * dx = 1
    current_sum = np.sum(h)
    scale = 1.0 / (current_sum * dx) if current_sum > 0 else 1.0
    h = h * scale
    
    # Ensure we project to strictly feasible region
    h = np.clip(h, 1e-6, 1.0 - 1e-6)
    h = project_to_feasible(h, n_points / 2)
    
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the better initial solution
    h_init = np.array(initial_h_values())
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Ensure feasible initialization
    h_current = project_to_feasible(h_init, target_sum)
    current_value = evaluate_erdos_solution(h_current)
    
    best_h = h_current.copy()
    best_value = current_value
    
    # Use differential evolution for global search
    # Define bounds [0,1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Make sure it satisfies constraints
        h_proj = project_to_feasible(h, target_sum)
        return evaluate_erdos_solution(h_proj)
    
    def constraint_func(x):
        return np.sum(x) * dx - 1.0  # Must equal 1.0
    
    # Use differential evolution with constraints
    try:
        # Good balance of convergence speed and solution quality
        de_result = differential_evolution(
            objective,
            bounds,
            maxiter=int(1000 + budget_s/4),
            popsize=15,  # Reasonable population size for faster convergence
            tol=1e-6,
            recombination=0.7,
            mutation=(0.5, 1.0),
            seed=seed,
            updating='deferred',
            workers=1,
            disp=False
        )
        
        if not de_result.success:
            print("DE failed to converge, using local search")
        
        # Final evaluation and check for improvement
        final_h = project_to_feasible(de_result.x, target_sum)
        final_value = evaluate_erdos_solution(final_h)
        
        if final_value < best_value:
            best_h = final_h
            best_value = final_value
            
    except Exception as e:
        pass  # Fall back to local optimization if DE fails
    
    # Local optimizer: use L-BFGS-B for refinement
    try:
        from scipy.optimize import minimize
        
        def objective_pure(h):
            return evaluate_erdos_solution(h)
            
        # Use L-BFGS-B for further refinement of current best
        result = minimize(

        )
        
        if result.success:
            refined_h = project_to_feasible(result.x, target_sum)

            if refined_value < best_value:
                best_h = refined_h
                best_value = refined_value
                
    except Exception as e:
        pass  # Continue with current best
    
    # Final check
    final_value = evaluate_erdos_solution(best_h)
