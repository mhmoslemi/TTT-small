import numpy as np

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum=1.0):
    """Project h onto the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Scale to target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    if current_sum == 0:
        h = np.full_like(h, target_sum / len(h))
    else:
        h = h * target_sum / current_sum
    return h

def symmetrize_and_project(h):
    """Symmetrize h around midpoint and project to capped simplex"""
    n = len(h)
    # Symmetrize
    h_sym = np.zeros_like(h)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2.0
    # Project to capped simplex
    return project_to_capped_simplex(h_sym, 1.0)

def logsumexp_stable(log_weights):
    """Numerically stable logsumexp"""
    max_log_w = np.max(log_weights)
    return max_log_w + np.log(np.sum(np.exp(log_weights - max_log_w)))

def compute_soft_weights(R, lambda_param):
    """Compute soft weights w_s from correlations R_s"""
    R_scaled = -lambda_param * R
    log_w = R_scaled - logsumexp_stable(R_scaled)
    return np.exp(log_w)

def compute_gradient(h, w):
    """Compute the gradient based on soft weights w"""
    n = len(h)
    g = np.zeros_like(h)
    # For each shift s, compute the contribution to gradient components
    # w[0] is for shift s=0 (no shift), w[1] for shift s=1, ..., w[2n-3] for shift s=2n-3 
    # We only use indices that fall within the bounds of h
    for s in range(2*n - 1):
        if s < n:
            # Slight overlap with right side
            g[:n-s] += w[s] * h[s:]
        else:
            # Shifts to the left side
            g[s-n+1:] += w[s] * h[:n-(s-n+1)]
    return g

def backtracking_line_search(h, g, lambda_param, alpha_init=0.2, max_iter=5):
    """Backtracking line search on the smooth surrogate"""
    # No need to renormalize h[:] = symmetrize_and_project(h) because we compute soft weights once here
    
    def surrogate_eval(h_candidate):
        """Evaluate the soft min surrogate function for a given h. """
        R = np.correlate(h_candidate, h_candidate, "full") 
        return (1.0/lambda_param) * logsumexp_stable(-lambda_param * R)
    
    alpha = alpha_init
    h_new = h + alpha * g  # Simple EULER step
    # Project before evaluation to ensure constraints 
    h_proj = project_to_capped_simplex(h_new, 1.0)
    
    best_eval = surrogate_eval(h_proj)
    best_h = h_proj.copy()
    
    while alpha > 1e-8:  # avoid too small steps
        h_test = h + alpha * g
        h_test = project_to_capped_simplex(h_test, 1.0)
        current_eval = surrogate_eval(h_test)
        if current_eval < best_eval:
            best_eval = current_eval
            best_h = h_test.copy()
        alpha *= 0.5
    return best_h, best_eval
