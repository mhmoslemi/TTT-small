def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    # Use iterative projection on capped simplex
    lo = 0.0
    hi = 1.0
    n = len(x)
    # initial bounds for lambda
    lam_low = np.min(x) - hi
    lam_high = np.max(x) - lo
    
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        x_proj = np.clip(x - lam, lo, hi)
        total = x_proj.sum()
        if total > target_sum:
            lam_low = lam
        else:
            lam_high = lam
    x_proj = np.clip(x - lam_high, lo, hi)
    return x_proj

def compute_overlaps(h):
    """Compute overlap integrals for all shifts"""
    n = len(h)
    overlaps = np.zeros(n)
    one_minus_h = 1.0 - h
    dx = 2.0 / n
    for k in range(n):
        if k == 0:
            overlaps[k] = np.dot(h, one_minus_h)
        else:
            overlaps[k] = np.dot(h[:n-k], one_minus_h[k:])
    overlaps *= dx
    return overlaps

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    overlaps = compute_overlaps(h_values)
    return np.max(overlaps)
