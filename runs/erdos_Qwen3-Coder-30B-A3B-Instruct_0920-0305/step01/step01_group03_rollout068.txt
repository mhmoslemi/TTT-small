def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    resolutions = [80, 100, 120]
    best_h = None
    best_c5 = float('inf')

        if time.time() - start_time > budget_s * 0.95:
            break

        x = Variable(n)

        # We'll generate constraints programmatically for all valid (i,s) pairs
        # First derive all valid pairs and store them
        w_vars = {}
        w_constraints = []
        
        # Generate constraint list
        # Need to track which w_{i,s} variables we define
        valid_pairs = []
        for s in range(-(n-1), n):
            for i in range(n):
                if 0 <= i+s < n:
                    valid_pairs.append((i, s))
                    w_vars[(i, s)] = Variable()
        
        # $ w_{i,s} <= x_i $
        for i, s in valid_pairs:
            w_constraints.append(w_vars[(i, s)] <= x[i])
            
        # $ w_{i,s} <= x_{i+s} $
        for i, s in valid_pairs:
            w_constraints.append(w_vars[(i, s)] <= x[i+s])
            
        # $ w_{i,s} >= x_i + x_{i+s} - 1 $
        for i, s in valid_pairs:
            w_constraints.append(w_vars[(i, s)] >= x[i] + x[i+s] - 1)
            
        # $ w_{i,s} >= 0 $
        for i, s in valid_pairs:
            w_constraints.append(w_vars[(i, s)] >= 0)
        
        # $ sum_i w_{i,s} >= t $
        for s in range(-(n-1), n):
            # Build sum of w_{i,s} for this shift
            w_terms = [w_vars[(i,s)] for i in range(n) if 0 <= i+s < n]
            w_constraints.append(sum(w_terms) >= t)
        
        # Mass constraint
        mass_constraint = [sum(x) == n/2]
        
        # Bound constraints
        bound_constraints = [x >= 0, x <= 1]
        
        # Formulate LP
        objective = Maximize(t)
        problem = Problem(objective, w_constraints + mass_constraint + bound_constraints)
        
        # Solve
        try:
            problem.solve(solver=ECOS, feastol=1e-8, reltol=1e-6, abstol=1e-8)
            if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                x_star = x.value
                t_star = t.value
                c5_bound_from_lp = 1 - t_star
            else:
                continue
        except Exception as e:
            continue

        best_rounded_c5 = float('inf')

                # Deterministic rounding based on descending values
                sorted_indices = np.argsort(-x_star)
                h_rounded = np.zeros(n)
                h_rounded[sorted_indices[:n//2]] = 1

                x_noisy = x_star + noise
                sorted_indices = np.argsort(-x_noisy)
                h_rounded = np.zeros(n)

            if c5_raw < best_rounded_c5:
                best_rounded_c5 = c5_raw
                best_rounded_h = h_rounded.copy()

        h_current = best_rounded_h.copy()
        current_c5 = best_rounded_c5

            """Given i=1, j=0. Flip one and zeros and return modified correlations"""

            correlation_full = np.correlate(h_current, 1 - h_current, mode="full")
            # We only care about max negative entry (1 - correlations)
            # so we compute C5 = 1 - min(new_correlations)
            # Correlation is now 1-sum_{k} h_{k} (1-h_{k±s})
            return np.max(correlation_full)

        stagnant = 0
        temp = 1.0
        temp_decay = 0.995

            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Try to swap
            h_trial = h_current.copy()
            h_trial[i] = 0
            h_trial[j] = 1
            
            # Compute new C5 by evaluating
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            num_swaps += 1
            
            # Accept or reject based on greedy or simulated annealing
            if c5_trial < current_c5:
                h_current = h_trial
                current_c5 = c5_trial
                stagnant = 0
            else:
                # Simulated Annealing
                if np.random.rand() < np.exp(-(c5_trial - current_c5) / temp):
                    h_current = h_trial
                    current_c5 = c5_trial
                    stagnant = 0
                else:
                    stagnant += 1
                    
            temp *= temp_decay
                
        # Final check

            best_c5 = current_c5
            best_h = h_current.copy()
