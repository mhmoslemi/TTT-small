def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def compute_soft_min_gradient(h, R, lambda_param):
    """Compute the gradient of the soft minimum surrogate"""
    n = len(h)
    # Compute soft weights using log-sum-exp trick for numerical stability
    R_max = np.max(R)
    exp_neg_lambda_R = np.exp(-lambda_param * (R - R_max))
    w = exp_neg_lambda_R / np.sum(exp_neg_lambda_R)
    
    # Compute the gradient direction g_j = sum_s w_s (h_{j+s} + h_{j-s})
    g = np.zeros(n)
    for s in range(len(R)):
        shift = s - (len(R) // 2)
        if shift >= 0:
            g += w[s] * np.concatenate([h[shift:], np.zeros(shift)])
        else:
            g += w[s] * np.concatenate([np.zeros(-shift), h[:shift]])
    
    return g

def project_onto_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: sum(h) = target_sum, 0 <= h <= 1 """
    n = len(h)
    
    # Start with clipped version
    h_proj = np.clip(h, 0, 1)
    
    # Adjust via water filling
    sum_h = np.sum(h_proj)
    if abs(sum_h - target_sum) < 1e-12:
        return h_proj
    
    # Binary search for the right tau
    low, high = -1.0, 1.0
    for _ in range(50):
        tau = (low + high) / 2.0
        temp = np.clip(h + tau, 0, 1)
        sum_temp = np.sum(temp)
        if sum_temp < target_sum:
            high = tau
        else:
            low = tau
            
    tau = (low + high) / 2.0
    h_proj = np.clip(h + tau, 0, 1)
    
    # Ensure exact sum
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Adjust by adding to lowest elements
        indices = np.argsort(h_proj)
        for i in range(len(indices)):
            if diff <= 0:
                break
            idx = indices[i]
            add = min(diff, 1.0 - h_proj[idx])
            h_proj[idx] += add
            diff -= add
    
    return h_proj

def symmetry_projection(h):
    """Symmetrize h around x=1, then reproject to maintain mass"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
    
    # Reproject to maintain mass constraint
    target_sum = n / 2
    h_sym = project_onto_capped_simplex(h_sym, target_sum)
    return h_sym
