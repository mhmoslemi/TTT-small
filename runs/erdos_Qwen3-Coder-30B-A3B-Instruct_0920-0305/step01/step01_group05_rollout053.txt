import numpy as np

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_simplex(y, target_sum):
    """Projects vector y onto the simplex with sum = target_sum"""
    # Shift and clip 
    def sum_after_projection(alpha):
        z = np.clip(y - alpha, 0, 1)
        return np.sum(z) - target_sum

    # Binary search for alpha
    a, b = np.min(y) - 1, np.max(y) + 1
    for _ in range(50):
        alpha = (a + b) / 2
        if sum_after_projection(alpha) > 0:
            a = alpha
        else:
            b = alpha
    alpha = (a + b) / 2
    return np.clip(y - alpha, 0, 1)

def compute_fs(h, s):
    """Compute one f_s(h) value"""
    n = len(h)
    if s >= 0:
        # Right shift
        h_shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        # Left shift  
        h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.dot(h, 1 - h_shifted)

def compute_all_fs(h):
    """Compute all f_s(h) values efficiently"""
    n = len(h)
    fs = []
    for s in range(-(n-1), n):
        fs.append(compute_fs(h, s))
    return np.array(fs)

def compute_gradient(h, p, fs, lambda_penalty):
    """Compute the gradient of smoothed max function and penalty"""
    n = len(h)
    grad = np.zeros(n)
    if p == 1:
        # Special case for p=1 (L1 norm): will not be used
        return grad
    
    # Compute weights for the gradient
    abs_fs = np.abs(fs)
    # Avoid dividing by zero
    weights = np.where(abs_fs > 1e-16, abs_fs**(p-1), 0)
    
    # Initialize the gradient
    grad = np.zeros(n)
    
    # For each shift s, compute contribution to gradient
    for s in range(-(n-1), n):
        w = weights[s + (n-1)]
        if w > 1e-16:
            # These are local contributions to gradient
            if s >= 0:
                # Right shift 
                h_shifted = np.concatenate([np.zeros(s), h[:-s]]) 
            else:
                # Left shift
                h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
                
            # Gradient contribution from this shift
            # df/dh_i = 1 - h_{i+s} - h_{i-s}
            # (sum over possible s for each i)
            for i in range(n):
                idx_l = i - s if i - s >= 0 else -1
                idx_r = i + s if i + s < n else -1
                
                if idx_l >= 0:
                    grad[i] += w * (-h[idx_l])
                if idx_r >= 0:
                    grad[i] += w * (-h[idx_r])
                grad[i] += w * 1.0  # from the '1' term

    # Add penalty term gradient  
    penalty_grad = 2 * lambda_penalty * (np.sum(h) - n/2.0)
    grad += penalty_grad
    
    return grad

def compute_smooth_max(h, p, fs):
    """Compute the smooth p-norm version of max of fs"""
    abs_fs = np.abs(fs)
    if p == 1:
        return np.sum(abs_fs)
    else:
        return np.power(np.sum(np.power(abs_fs, p)), 1.0 / p)

def projected_gradient_descent(h, n_points, p, lambda_penalty, max_iter, alpha=0.01):
    """Run projected gradient descent on the smooth approximation"""
    dx = 2.0 / n_points
    current_h = h.copy()
    
    # Precompute all fs (this could be cached if needs to be repeated)
    fs = compute_all_fs(current_h)
    
    # Possibly compute once here, and reuse for finite differences check
    prev_val = compute_smooth_max(current_h, p, fs) + lambda_penalty * (np.sum(current_h) - n_points/2.0)**2
    for iteration in range(max_iter):
        # Compute fs (not strictly necessary if using cached)
        fs = compute_all_fs(current_h)
        
        # Compute gradient
        grad = compute_gradient(current_h, p, fs, lambda_penalty)
        
        # Gradient step
        current_h = current_h - alpha * grad
        
        # Project onto feasible set
        current_h = project_to_simplex(current_h, n_points/2.0)
        
        # Compute current value 
        smooth_val = compute_smooth_max(current_h, p, fs) + lambda_penalty * (np.sum(current_h) - n_points/2.0)**2
        # Check if improvement is small
        if abs(prev_val - smooth_val) < 1e-9:
            break
        prev_val = smooth_val
        
    return current_h

def interpolate_to_new_size(h_old, n_new):
    """Interpolate from old size to new size"""
    old_size = len(h_old)
    x_old = np.linspace(0, 2, old_size)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return np.clip(h_new, 0, 1)
