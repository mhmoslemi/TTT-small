def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def clip_and_rescale(z, target_sum):
    """Clip vector to [0,1] and rescale to achieve target sum"""
    z = np.clip(z, 0, 1)
    current_sum = np.sum(z)
    if abs(current_sum - target_sum) < 1e-10:
        return z.copy()
    # Bisection to find scaling factor
    def sum_at_alpha(alpha):
        return np.sum(np.clip(z + alpha, 0, 1))
    
    def objective(alpha):
        return (sum_at_alpha(alpha) - target_sum) ** 2
    
    res = minimize_scalar(objective, bounds=(-10, 10), method='bounded')
    alpha_opt = res.x
    z_result = np.clip(z + alpha_opt, 0, 1)
    return z_result
