import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}"""
    # Find the proper shift θ that makes sum equal to target_sum
    # Use binary search on θ to find such that sum_i clamp(v_i + θ, lower, upper) = target_sum
    
    # We'll do a simple iterative algorithm with binary search
    v = np.array(v, dtype=float)
    n = len(v)
    v_shifted = v - lower  # shift to start from 0
    upper_shifted = upper - lower
    
    # Initial bounds for θ
    low = -10.0
    high = 10.0

    # Compute actual range
    max_v = np.max(v_shifted)
    min_v = np.min(v_shifted)
    
    low = max(low, -max_v)
    high = min(high, upper_shifted - min_v)

    # Binary search for θ
    for _ in range(50):
        theta = (low + high) / 2.0
        projected = np.clip(v_shifted + theta, 0, upper_shifted)
        projected_sum = np.sum(projected)
        
        if projected_sum < target_sum:
            low = theta
        else:
            high = theta
            
    theta = (low + high) / 2.0
    # Apply to original vector
    result = np.clip(v + theta, lower, upper)
    
    return result

def smooth_max_objective(h, lambda_factor=10.0):
    """Compute the smooth approximation to minimize max of h(x)(1-h(x+s))"""
    corr_hh = np.correlate(h, h, mode='full')  # correlation with itself
    # Softmin of correlation values to approximate min of correlation
    # negative because we want to maximize minimum correlation
    neg_logsumexp = -np.log(np.sum(np.exp(-lambda_factor * corr_hh)))
    return neg_logsumexp / lambda_factor

def compute_gradient(h, lambda_factor=10.0):
    """Compute the gradient of the smooth objective with respect to h"""
    n = len(h)
    corr_hh = np.correlate(h, h, mode='full') 
    max_corr = np.max(corr_hh)
    
    # Stable computation of weights
    shifted = corr_hh - max_corr
    exp_shifted = np.exp(-lambda_factor * shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # Initialize gradient
    grad = np.zeros(n)
    
    # compute weight contribution for each shift
    for s in range(-(n-1), n):
        if s >= 0:
            # For s >= 0, we have h[i] * h[i+s] contributing to shift s
            # So for each i, we increase grad[i] by w[s+n-1] * h[i+s]
            # and grad[i+s] by w[s+n-1] * h[i]
            if s < n:
                grad[:n-s] += w[s+n-1] * h[s:]
                grad[s:] += w[s+n-1] * h[:n-s]
        else:
            # s < 0, so shift left
            abs_s = -s
            if abs_s < n:
                grad[abs_s:] += w[s+n-1] * h[:n-abs_s]  
                grad[:n-abs_s] += w[s+n-1] * h[abs_s:]
                
    # gradient is negative because we maximize minimum correlation
    return -grad
