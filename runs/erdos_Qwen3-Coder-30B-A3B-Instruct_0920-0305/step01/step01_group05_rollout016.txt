import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
from scipy.sparse import diags
import time


def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5


def round_sdp_solution(X_sdp, h_sdp, n, num_samples=2000):
    """Extract a feasible h from SDP solution using randomized rounding"""
    # Decompose X into eigenvalues and vectors
    eigenvals, eigenvecs = eigh(X_sdp)
    positive_evals = eigenvals > 1e-8
    V = eigenvecs[:, positive_evals]
    λ = eigenvals[positive_evals]
    B = V @ np.diag(np.sqrt(λ))
    
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(num_samples):
        g = np.random.randn(B.shape[1])
        z = B @ g
        
        # Clamp to [0,1] and rescale to sum to n/2
        z = np.clip(z, 0, 1)
        scale_factor = (n / 2) / np.sum(z)
        h_candidate = z * scale_factor
        
        # Ensure integral constraint is respected
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n/2) / np.sum(h_candidate)
        
        # Evaluate C5
        c5_candidate = evaluate_h(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            
    return best_h, best_c5

    start_time = time.time()

    n_half = n // 2

    # Build SDP model
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_half,
        cp.diag(X) == h,
        t >= 0  # t non-negative
    ]
    
    # Add McCormick constraints for each i < j
    for i in range(n):
        for j in range(i + 1, n):
            constraints.extend([
                X[i, j] >= 0,
                X[i, j] <= h[i],
                X[i, j] <= h[j],
                X[i, j] >= h[i] + h[j] - 1
            ])
    
    # Add overlap constraints for each shift (excluding 0)
    for s in range(-(n-1), n):
        if s == 0:
            continue
            
        # Shift matrix - we'll build it using diagonal extraction
        shift_diag = np.zeros(n)
        if s > 0:
            shift_diag[s:] = 1
        else:
            shift_diag[:s] = 1
            
        # Get shifted indices
        I = np.eye(n)
        shift_mat = I * shift_diag.reshape(n, 1)  # broadcast to form a diagonal matrix
        shift_mat = shift_mat[np.arange(n), np.arange(n) + s]  # Selecting shifted entries
        shift_mat = shift_mat.reshape(n, 1) if shift_mat.shape[0] > 0 and shift_mat.ndim == 1 else shift_mat
        shift_mat = np.eye(n, k=s)  # Various implementations may be correct. Use      
        shift_matrix = shift_mat * shift_diag
        
        # Quick implementation: we extract indices through comprehension
        if s > 0:
            # Extract sum of X[i, i+s]
            sum_term = cp.sum(cp.multiply(shift_mat, X))
        else:
            # For negative s we do similar operation
            sum_term = cp.sum(cp.multiply(shift_mat, X))
            
        constraints += [t >= n_half - cp.sum(cp.multiply(shift_mat, X))]
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    status = None
    try:
        prob.solve(solver='SCS', eps=1e-5, max_iters=5000, verbose=False)
        status = prob.status
    except Exception as e:

    if status == cp.OPTIMAL:

        X_sdp = X.value
        t_sdp = t.value
        c5_lower = t_sdp * dx
    else:
        # Fallback
        h_sdp = initial_h
        X_sdp = np.outer(h_sdp, h_sdp)
        c5_lower = evaluate_h(h_sdp)
    
    # Randomized rounding to get feasible h
    best_h, c5_bound = round_sdp_solution(X_sdp, h_sdp, n, 2000)

    max_refinement_steps = 5000
    refinements_done = 0

        perturbed_h = best_h.copy() + np.random.normal(0, 0.001, n)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n / 2) / np.sum(perturbed_h)
        
        new_c5 = evaluate_h(perturbed_h)
        
        if new_c5 < current_c5:
            best_h = perturbed_h
            c5_bound = new_c5
            
        refinements_done += 1
        if refinements_done % 100 == 0:
            if (time.time() - start_time) > (budget_s - 5):
                break
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), final_c5, n)
