def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_onto_simplex(y, target_sum):
    """Project vector y onto the simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_y)
    # Find the largest k such that (cumsum[k] - (k+1)*target_sum) / (k+1) < sorted_y[k]
    # This is equivalent to finding the index where the decrease of the projection
    # meets the constraint
    n = len(sorted_y)
    temp = (cumsum - target_sum * np.arange(1, n + 1)) / np.arange(1, n + 1)
    # Find first index where value becomes non-positive
    index = 0
    for i in range(n):
        if temp[i] <= sorted_y[i]:
            index = i
        else:
            break
    # Compute the threshold
    theta = temp[index]
    # Compute the projection
    result = np.maximum(y - theta, 0)
    # Scale to meet sum constraint (change in the too small values)
    alpha = target_sum - np.sum(result)
    if alpha > 0:
        result += alpha / len(result)

    return result

def compute_f_s(h, s):
    """Compute f_s(h) = sum_i h_i * (1 - h_{i+s}) where index out of range is zero"""
    if len(h) <= abs(s):
        return 0.0
    if s >= 0:
        return np.sum(h[:len(h)-s] * (1.0 - h[s:]))
    else:
        return np.sum(h[-s:] * (1.0 - h[:-s]))

def compute_all_fs(h):
    """Compute all f_s(h) values for s from -(n-1) to n-1"""
    n = len(h)
    result = np.zeros(2*n-1)
    for i in range(2*n-1):
        s = i - (n-1)
        result[i] = compute_f_s(h, s)
    return result
