import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def compute_shifted_indices(n, s):
    """Compute forward and backward masks for shift s"""
    mask_fwd = np.arange(n) + s < n
    mask_bwd = np.arange(n) - s >= 0
    return mask_fwd, mask_bwd

def build_cvp_problem(h, n, s, prev_h, prev_grad):
    """Build a convex subproblem for shift s"""
    mask_fwd, mask_bwd = compute_shifted_indices(n, s)
    
    # g_s^c(h) = sum(h) + 0.5*sum(h_i^2) for i in mask_fwd + 0.5*sum(h_i^2) for i in mask_bwd
    g_c = cp.sum(h) + 0.5*cp.sum_squares(h[mask_fwd]) + 0.5*cp.sum_squares(h[mask_bwd])
    
    # g_s^d(h) = 0.5 * sum( (h_i + h_{i+s})^2 )
    grad_term = cp.sum(prev_grad[s] * (h - prev_h))
    return g_c - grad_term

def ccp_solver(h0, n, max_iter=30, tol=1e-6):
    """Convex-concave procedure solver"""
    h_prev = h0.copy()
    tau_prev = float('inf')

    for iter in range(max_iter):
        # Build gradient of g_s^d at h_prev for each shift s
        grad_s = []
        for s in range(-(n-1), n):
            mask_fwd, mask_bwd = compute_shifted_indices(n, s)
            grad = np.zeros(n)
            if s >= 0:
                grad[mask_fwd] = h_prev[mask_fwd] + h_prev[mask_fwd - s]
            else:
                grad[mask_bwd] = h_prev[mask_bwd] + h_prev[mask_bwd + (-s)]
            grad_s.append(grad)

        # Create CVXPY problem
        h = cp.Variable(n)
        tau = cp.Variable()

        # Constraints for all shifts
        constraints = []
        for s in range(-(n-1), n):
            if s >= 0:
                mask_fwd, mask_bwd = compute_shifted_indices(n, s)
                g_c = cp.sum(h) + 0.5*cp.sum_squares(h[mask_fwd]) + 0.5*cp.sum_squares(h[mask_bwd])
            else:
                mask_fwd, mask_bwd = compute_shifted_indices(n, s)
                g_c = cp.sum(h) + 0.5*cp.sum_squares(h[mask_fwd]) + 0.5*cp.sum_squares(h[mask_bwd])

            grad_term = cp.sum(grad_s[s] * (h - h_prev))
            constraints.append(tau >= g_c - grad_term)
        
        # Box constraints
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == n/2)
        
        # Objective
        objective = cp.Minimize(tau)
        prob = cp.Problem(objective, constraints)

        # Solve using ECOS
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
            if prob.status != cp.OPTIMAL:
                break
        except:
            break

        h_curr = h.value
        if h_curr is None:
            break
        h_curr = np.clip(h_curr, 0, 1)
        
        # Estimate integral violation and adjust if needed
        integral_violation = np.abs(np.sum(h_curr) - n/2)
        if integral_violation > 1e-8:
            # Correct the sum
            h_curr = h_curr * (n/2) / np.sum(h_curr)
            
        # Update h_prev
        h_prev = h_curr.copy()
        
        # Compute current tau and check for convergence
        tau_curr = tau.value
        if abs(tau_curr - tau_prev) < tol:
            break
        tau_prev = tau_curr

    return h_curr, tau_curr

            # Evaluate candidates and set to the best one that keeps the integral balanced

            for candidate in candidates:
                h[i] = candidate
                # Compute overlap without full correlation, but patches
                max_val = 0
                for s in active_s:
                    si = i + s
                    if 0 <= i < n and 0 <= si < n:
                        m = h[i] * (1 - h[si])
                        max_val = max(max_val, m)
                    if 0 <= si < n and 0 <= i < n:
                        m = h[si] * (1 - h[i])
                        max_val = max(max_val, m)
                if max_val < best_val:
                    best_val = max_val
                    best_candidate = candidate
            h[i] = best_candidate
            
        # After all coordinates, fix integral balance
        sum_diff = n/2 - np.sum(h)
        if abs(sum_diff) > 1e-3:
            # Adjust largest entry (if positive)
            idx = np.argmax(h)
            if h[idx] < 1.0 and h[idx] + sum_diff > 0:
                h[idx] += sum_diff
        h = np.clip(h, 0, 1)
    return h
