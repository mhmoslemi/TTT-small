def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible_set(x, target_sum):
    """Project vector x onto the set where sum = target_sum and all components in [0,1]"""
    # This implements a simple bisection method for the projection onto the simplex
    # with box constraints [0,1]
    x = np.copy(x)
    lower = np.min(x)
    upper = np.max(x)
    if upper <= 1.0 and lower >= 0.0:
        return np.clip(x, 0, 1)

    # Bisection to find the right alpha
    def sum_after_projection(alpha):
        x_proj = np.clip(x - alpha, 0, 1)
        return np.sum(x_proj)

    a, b = lower - 1.0, upper + 1.0
    for _ in range(100):
        alpha = (a + b) / 2
        s = sum_after_projection(alpha)
        if s > target_sum:
            a = alpha
        else:
            b = alpha
    # Use slightly more conservative alpha to avoid numerical issues
    alpha = (a + b) / 2
    x_proj = np.clip(x - alpha, 0, 1)
    return x_proj

def compute_shifted_vectors(h_padded, s):
    """Compute the shifted version of h_padded by s positions"""
    if s >= 0:
        return np.concatenate([h_padded[s:], np.zeros(s)])
    else:
        return np.concatenate([np.zeros(-s), h_padded[:s]])

def compute_f_s(h, s):
    """Compute f_s(h) for a given shift s"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    shifted_h = compute_shifted_vectors(h_padded, s)
    return np.dot(h, 1 - shifted_h)

def compute_all_shifts(h):
    """Compute all f_s(h) values efficiently"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    f_s = []
    for s in range(-(n-1), n):
        shifted_h = compute_shifted_vectors(h_padded, s)
        f = np.dot(h, 1 - shifted_h)
        f_s.append(f)
    return np.array(f_s)
