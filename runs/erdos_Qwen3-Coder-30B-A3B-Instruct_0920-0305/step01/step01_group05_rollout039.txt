import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

        corr = np.correlate(h_temp, 1 - h_temp, mode="full")
        return np.max(corr) * dx

    def constraint_func(x):
        h_temp = np.array(x)
        return np.sum(h_temp) * dx - 1.0
