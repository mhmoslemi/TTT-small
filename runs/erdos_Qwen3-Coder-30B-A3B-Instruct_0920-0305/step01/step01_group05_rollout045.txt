import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def compute_shifted_indices(n, s):
    """Compute forward and backward masks for shift s"""
    mask_fwd = np.arange(n) + s < n
    mask_bwd = np.arange(n) - s >= 0
    return mask_fwd, mask_bwd

def build_convex_quadratic(h, s, mask_fwd, mask_bwd):
    """Build g_s^c(h) = sum(h) + 0.5 * sum(h_i^2 + h_{i+s}^2)"""
    term1 = cp.sum(h)
    term2 = 0.5 * cp.sum_squares(h[mask_fwd]) + 0.5 * cp.sum_squares(h[mask_bwd])
    return term1 + term2

def build_affine_linearisation(h_prev, s, mask_fwd, mask_bwd):
    """Build affine approximation to -g_s^d(h) at h_prev"""
    # Compute gradient of g_s^d at h_prev
    grad = np.zeros_like(h_prev)
    grad[:-s] += h_prev[s:] + h_prev[:-s]
    grad[s:] += h_prev[s:] + h_prev[:-s]
    # Construct affine part
    grad_vector = cp.Parameter(n)
    grad_vector.value = grad
    
    # Affine approximation: g_s^d(h_prev) + grad.T @ (h - h_prev)
    # This is the linear part:
    affine_part = cp.sum(grad_vector * (h - h_prev))
    return affine_part

def solve_ccp_subproblem(h_prev, n, dx, max_iter=30, tol=1e-6):
    """Solve the CCP subproblem using CVXPY with SOCP"""
    # Precompute masks for all shifts
    masks = {}
    for s in range(-(n-1), n):
        masks[s] = compute_shifted_indices(n, s)

    # Create variables
    h = cp.Variable(n)
    tau = cp.Variable()

    # Create constraints
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == n/2
    ]

    # Add the constraint for each shift s:
    for s in range(-(n-1), n):
        mask_fwd, mask_bwd = masks[s]
        g_c = build_convex_quadratic(h, s, mask_fwd, mask_bwd)
        affine = build_affine_linearisation(h_prev, s, mask_fwd, mask_bwd)
        constraints.append(tau >= g_c - affine)

    # Set up the problem
    objective = cp.Minimize(tau)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            h_new = h.value
            tau_val = tau.value
            return h_new, tau_val
        else:
            # Fallback to previous point if not optimal
            return h_prev, evaluate_h(h_prev) * dx
    except Exception:
        # Fallback to previous point
        return h_prev, evaluate_h(h_prev) * dx

            # Collect possible breakpoints per shift
            max_shifts = []

            # is piecewise quadratic in h_i. We need the max across s
            
            # Let's emumrate difference terms we might change by moving h_i
            # Start with evaluating current values
            current_vals = []
            for s in range(-(n - 1), n):
                # compute derivative of h_i * (1 - h_{i+s}) w.r.t h_i, given h_i is constant
                val = 0
                # sum h_i + ...
                # derivative of sum h_{i+s} * h_i wrt h_i gives just h_{i+s}
                if i + s >= 0 and i + s < n:
                    val += h[i + s]
                # - sum h_i * h_{i+s} is most tricky:
                # derivative of - h_i * h_{i+s} is - h_{i+s} (constant with respect to h_i)
                # So total derivative wrt h_i is: 1 - h_{i+s} + Derivs from others
                
                derive = 1 - h[i+s] if 0 <= i+s < n else 0
                if 0 <= i+s < n:
                    current_vals.append((s, derive))
            
            # Now the main coordinate update
            # One could do an exhaustive evaluation of breakpoints (is computationally feasible for small any)  
            # For efficiency, we sample a few values around current h[i], edges, and interesting points
            
            candidates = []
            boundary_vals = [0, 1]
            
            # Sample values near current h[i]
            for cand in np.linspace(max(0, h[i] - 0.1), min(1, h[i] + 0.1), 5):
                if 0 <= cand <= 1:
                    candidates.append(cand)
            
            candidates = sorted(set(candidates + boundary_vals))
            
            # Evaluate current value and all neighbors
            for new_h_i in candidates:
                h_copy = h.copy()
                old_sum = np.sum(h_copy)
                h_copy[i] = new_h_i
                
                # Adjust another variable to preserve integral
                # Use a simple alternating fix, assuming other vars have same sum constraint
                # We preserve ∑ h_i = n/2 by adjusting i+1 or i-1
                fix_ix = (i + 1) % n
                if fix_ix == i:
                    fix_ix = (i + 2) % n
                    
                # change fix_ix to keep sum consistent
                # old_h = h[i] + h[fix_ix] - new_h_i - h[fix_ix] = old_sum = old_sum - h[i]
                diff = new_h_i - h[i]
                h_copy[fix_ix] = max(0, min(1, h_copy[fix_ix] - diff))
                
                if np.sum(h_copy) * dx == 1.0:
                    corr = np.correlate(h_copy, 1 - h_copy, mode="full")
                    c5 = np.max(corr) * dx
                    if c5 < best_val:
                        best_val = c5
                        best_h = new_h_i
                        
            if best_h != h[i]:
                h[i] = best_h
                # Fix other to maintain sum
                fix_ix = (i + 1) % n
                if fix_ix == i:
                    fix_ix = (i + 2) % n
                diff = new_h_i - h[i] 
                h[fix_ix] = max(0, min(1, h[fix_ix] - diff))
                improved = True
                
        if not improved:
            break
    return h
