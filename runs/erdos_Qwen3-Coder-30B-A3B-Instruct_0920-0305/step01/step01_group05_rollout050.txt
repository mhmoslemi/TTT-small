import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    
    # Renormalize to ensure integral = 1
    h_sum = np.sum(h) * dx
    if h_sum != 0:
        h = h * (1.0 / h_sum)
        h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):
    """Perform binary local search for improved bound"""
    dx = 2.0 / n
    half = n // 2
    
    best_sum = float('inf')
    best_bits = None
    start = time.time()
    
    # Initial random binary start
    bits = np.zeros(n, dtype=int)
    bits[:half] = 1
    np.random.shuffle(bits)
    
    while time.time() - start < time_limit:
        # Evaluate current
        curr_val = evaluate_h(bits.astype(float))
        
        # Local descent with random swaps
        improved = True
        attempts = 0
        max_attempts = 1000
        
        while improved and attempts < max_attempts and time.time() - start < time_limit:
            improved = False
            # Try several random swaps
            for _ in range(n * 2):
                # Find indices where bits are 1 and 0
                ones_idx = np.flatnonzero(bits)
                zeros_idx = np.flatnonzero(1 - bits)
                
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                    
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                # Swap
                bits[i], bits[j] = bits[j], bits[i]
                new_val = evaluate_h(bits.astype(float))
                if new_val < curr_val - 1e-12:
                    curr_val = new_val
                    improved = True
                    # slight break to try more variations
                    break
                else:
                    # Revert swap
                    bits[i], bits[j] = bits[j], bits[i]
                attempts += 1
        
        # Update best solution
        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
            
        # Random restart
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        # Early exit condition if we're close to target
        if best_sum < 0.3807:
            break
    
    if best_bits is None:
        # Fallback: return the initial random one
        best_bits = np.zeros(n, dtype=int)
        best_bits[:half] = 1
        np.random.shuffle(best_bits)
        best_sum = evaluate_h(best_bits.astype(float))
    
    return best_bits, best_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_list = [42, 60, 80, 100]
    
    best_h = None
    best_c5_bound = float('inf')
    best_n = None
    
    for n_points in n_list:
        if time.time() - start_time > budget_s - 10:
            break
            
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 30:
            break
            
        # First run binary search for this n
        binary_h, binary_c5 = binary_local_search(n_points, remaining_time * 0.7)
        
        # Evaluate final binary as continuous to compare to continuous
        eval_h = binary_h.astype(float)
        dx = 2.0 / n_points
        eval_h_sum = np.sum(eval_h) * dx
        if eval_h_sum != 0:
            eval_h = eval_h * (1.0 / eval_h_sum)
            eval_h = np.clip(eval_h, 0, 1)
        
        final_c5 = evaluate_h(eval_h)
        
        if final_c5 < best_c5_bound:
            best_c5_bound = final_c5
            best_h = eval_h
            best_n = n_points
            
        # If we're approaching the target, break early
        if best_c5_bound < 0.3807:
            break
    
    # Final evaluation to ensure normalization
    final_c5 = evaluate_h(best_h)
    
    return (best_h.tolist(), final_c5, best_n)
