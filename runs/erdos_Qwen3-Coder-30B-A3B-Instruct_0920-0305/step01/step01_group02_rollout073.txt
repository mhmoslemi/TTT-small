import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    # Simple projection: clip and normalize
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale = target_sum / current_sum if current_sum > 0 else 1.0
    x = x * scale
    return x

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    # Compute correlations for all lags
    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx = 2/N
    dx = 2.0 / len(h)
    correlation *= dx
    # Return maximum correlation value
    return np.max(correlation)

def local_search(h_current, target_sum, max_iter=1000):
    """Perform local search around current solution"""
    current_value = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    best_value = current_value
    
    for _ in range(max_iter):
        # Try small random perturbations
        h_proposed = h_current.copy()
        idx = np.random.randint(len(h_current))
        delta = np.random.normal(0, 0.01)
        
        # Ensure feasibility
        new_val = h_proposed[idx] + delta
        if 0 <= new_val <= 1:
            h_proposed[idx] = new_val
            
            # Project to feasible set
            h_proposed = project_to_feasible(h_proposed, target_sum)
            
            proposed_value = evaluate_erdos_solution(h_proposed)
            
            if proposed_value < current_value:
                h_current = h_proposed
                current_value = proposed_value
                if proposed_value < best_value:
                    best_h = h_proposed.copy()
                    best_value = proposed_value
                    
    return best_h, best_value
