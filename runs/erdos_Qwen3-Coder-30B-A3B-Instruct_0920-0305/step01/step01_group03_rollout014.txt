import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def soft_min(h, lambda_param):
    """Compute soft minimum of correlations"""
    n = len(h)
    # Compute all correlations
    R = np.correlate(h, h, mode="full")
    
    # Use log-sum-exp trick for numerical stability
    R_max = np.max(R)
    exp_R = np.exp(-lambda_param * (R - R_max))
    sum_exp = np.sum(exp_R)
    log_sum_exp = R_max - np.log(sum_exp) / lambda_param
    return log_sum_exp

def compute_gradient(h, lambda_param):
    """Compute gradient of soft minimum"""
    n = len(h)
    R = np.correlate(h, h, mode="full")
    
    # Log-sum-exp trick for stability
    R_max = np.max(R)
    exp_R = np.exp(-lambda_param * (R - R_max))
    sum_exp = np.sum(exp_R)
    w = exp_R / sum_exp
    
    # Compute gradient using slice operations
    g = np.zeros(n)
    for i in range(n):
        for s in range(-n+1, n):
            if 0 <= i+s < n:
                g[i] += w[s + n - 1] * h[i + s]
            if 0 <= i-s < n:
                g[i] += w[-s + n - 1] * h[i - s]
    return g

def project_to_capped_simplex(h, target_sum):
    """Project onto the capped simplex: h >= 0, sum(h) = target_sum"""
    # clip to [0,1]
    h = np.clip(h, 0, 1)
    # adjust to exact target sum
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # water-filling approach
    if current_sum > target_sum:
        # shrink
        sorted_h = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_h)
        n_excess = np.searchsorted(cumsum, target_sum, side='right')
        if n_excess == 0:
            # all in range already
            return h
        cutoff = (cumsum[n_excess-1] - target_sum) / n_excess
        h = np.clip(h - cutoff, 0, 1)
        return h
    else:  
        # expand
        sorted_h = np.sort(h)
        cumsum = np.cumsum(sorted_h)
        n_deficit = np.searchsorted(cumsum, target_sum, side='right')
        if n_deficit == len(sorted_h):
            # all out of range, compress to [0,1] 
            h = np.clip(h, 0, 1)
            return h
        cutoff = (target_sum - cumsum[n_deficit-1]) / (len(sorted_h) - n_deficit)
        h = np.clip(h + cutoff, 0, 1)
        return h

def symmetry_enforce(h):
    """Enforce symmetry h[i] = (h[i] + h[n-1-i]) / 2"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
    return h_sym

def backtracking_line_search(h, g, lambda_param, alpha_init=0.2, max_iter=5):
    """Backtracking line search"""
    h = np.array(h)
    alpha = alpha_init  
    for _ in range(max_iter):
        h_new = h - alpha * g
        h_proj = project_to_capped_simplex(h_new, len(h)/2)
        # Symmetry enforcement
        h_proj = symmetry_enforce(h_proj)
        new_phi = soft_min(h_proj, lambda_param)
        if new_phi < soft_min(h, lambda_param) - 1e-8:
            return h_proj, alpha
        else:
            alpha /= 2.0
    return h_proj, alpha
