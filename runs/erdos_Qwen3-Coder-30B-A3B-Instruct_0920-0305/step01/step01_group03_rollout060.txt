import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x: lower <= x_i <= upper, sum_i x_i = target_sum}"""
    # Simple iterative approach
    v = np.array(v)
    n = len(v)
    
    # Initial guess for shift
    theta = (target_sum - np.sum(np.clip(v, lower, upper))) / n
    
    # Adjust until both constraints satisfied
    max_iter = 1000
    for _ in range(max_iter):
        # Compute new values after applying shift
        new_v = np.clip(v + theta, lower, upper)
        diff = np.sum(new_v) - target_sum
        
        if abs(diff) < 1e-10:
            return new_v
            
        # Update shift theta
        theta = theta - diff / n
                
    # If didn't converge, return clipped version with sum adjusted
    final_v = np.clip(v + theta, lower, upper)
    return final_v

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum h * dx = 1 => sum h = n/2
    
    # Define initial h values
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)  # n=42
        x_base = np.linspace(0, 2, len(base_h), endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        initial_h = np.interp(x_new, x_base, base_h)
    else:
        # Fallback: random uniform
        initial_h = np.random.uniform(0, 1, n_points)
    
    # Project to sum constraint
    initial_h = project_onto_box_simplex(initial_h, target_sum)
    
    # Objective function to minimize with smoothing
    def smooth_objective(h_vals, lambda_val=10.0):
        corr_hh = np.correlate(h_vals, h_vals, mode="full")
        # Use log-sum-exp smoothing for max to make differentiable
        max_corr = np.max(corr_hh)
        min_corr = np.min(corr_hh)
        return -min_corr  # We want to maximize min_corr
    
    # Gradient of smooth objective
    def smooth_gradient(h_vals, lambda_val=10.0):
        corr_hh = np.correlate(h_vals, h_vals, mode="full")
        min_idx = np.argmin(corr_hh)
        # Construct corresponding weights for gradient
        w = np.zeros_like(corr_hh)
        w[min_idx] = 1.0  # Only the min shift matters for gradient
        
        # Compute gradient manually via loop for correctness:
        grad = np.zeros_like(h_vals)
        n = len(h_vals)
        for s in range(-(n-1), n):
            idx = s + (n-1)
            if w[idx] > 0:
                if s >= 0:
                    # h_{i+s}
                    grad[:n-s] += w[idx] * h_vals[s:]
                    # h_{i-s}
                    grad[s:] += w[idx] * h_vals[:n-s]
                else:
                    # h_{i+s}
                    grad[-s:] += w[idx] * h_vals[:n+s]
                    # h_{i-s}
                    grad[:n+s] += w[idx] * h_vals[-s:]
        return -grad
    
    # Gradient descent with smoothing
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Use multiple restarts with completely random initializations
    num_restarts = 30
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h = np.random.uniform(0, 1, n_points)
        h = project_onto_box_simplex(h, target_sum)
        
        # Run gradient descent with multiple λ values
        lambda_schedule = [1, 5, 10, 20, 40]
        for λ in lambda_schedule:

                break
            iter_count = 100

                    break
                    
                # Compute gradient using subgradient
                corr_hh = np.correlate(h, h, mode="full")
                min_idx = np.argmin(corr_hh)
                
                # Initialize gradient
                grad = np.zeros_like(h)
                n = len(h)
                w = np.zeros_like(corr_hh)

                # Compute gradient via directions
                for s in range(-(n-1), n):
                    idx = s + (n-1)
                    if w[idx] > 0:
                        if s >= 0:
                            grad[:n-s] += w[idx] * h_vals[s:]
                            grad[s:] += w[idx] * h_vals[:n-s]
                        else:
                            off = -s
                            grad[off:] += w[idx] * h_vals[:n-off]
                            grad[:n-off] += w[idx] * h_vals[off:]
                        
                # Minus because gradient of loss function
                grad = -grad
                
                # Step size
                alpha = 0.01
                
                # Update rule
                h_new = h - alpha * grad
                
                # Clip to [0,1] then project to sum constraint
                h_new = np.clip(h_new, 0, 1)
                h = project_onto_box_simplex(h_new, target_sum)
                
                # Update best if improvement
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
    # Do final refinement with scipy SLSQP
    def eval_constraint_sum(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    def eval_constraint_lower(h_vals):
        return h_vals - 0.0
    
    def eval_constraint_upper(h_vals):
        return 1.0 - h_vals
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': eval_constraint_sum},
        {'type': 'ineq', 'fun': eval_constraint_lower},
        {'type': 'ineq', 'fun': eval_constraint_upper}
    ]
    
    # Final optimization with pure objective
    result = minimize(
        lambda x: evaluate_erdos_solution(x, dx),
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-10}
    )
    
    if result.success:
        final_c5 = evaluate_erdos_solution(result.x, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = result.x.copy()
    
    return (best_h.tolist(), best_c5, n_points)
