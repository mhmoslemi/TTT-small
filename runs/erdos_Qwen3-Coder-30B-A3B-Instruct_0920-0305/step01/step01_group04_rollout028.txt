def full_correlation(h):
    """Compute full correlation (integer version), return (correlations, max_corr)."""
    h = np.array(h)
    n = len(h)
    correlations = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations)
    return correlations, max_corr

def build_shift_matrices(n):
    """Build Toeplitz matrices C_d for all non-zero shifts."""
    C_d_list = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        C = np.zeros((n, n))
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                C[i, j] = 1
                C[j, i] = 1
        C_d_list.append((d, C))
    return C_d_list

def sdp_rounding(X, n, R=200):
    """Perform random hyperplane rounding and return the best (h, M) found."""
    best_h = None
    best_M = float('inf')

    for _ in range(R):

        s = np.sign(V @ g)
        
        # Adjust for sum constraint (want sum h = n/2 hence sum s = 0) 
        S = np.sum(s)
        if abs(S) > 0.5:  # should be integer
            # Pick the element with smallest |s_i| to flip
            idx = np.argmin(np.abs(s))
            s[idx] *= -1

        _, M = full_correlation(h)
        
        if M < best_M:
            best_M = M
            best_h = h.copy()
    
    return best_h, best_M

def upscale_pattern(h, target_n):
    """Scale pattern by concatenating with its complement."""
    if len(h) == target_n:
        return h.copy()
    doubled = np.concatenate([h, 1 - h])
    if len(doubled) >= target_n:
        return doubled[:target_n]
    # If still too short, we might repeat or interpolate (not implemented, just a fallback)
    return doubled

    _, M = full_correlation(h)
    change = True
    iter_count = 0

                    if h[j] == 0:
                        # Try the swap
                        h[i], h[j] = 0, 1
                        _, new_M = full_correlation(h)
                        if new_M < M:
                            M = new_M
                            change = True
                            improve_count += 1
                        else:
                            h[i], h[j] = 1, 0  # undo
        iter_count += 1
    return h, M
