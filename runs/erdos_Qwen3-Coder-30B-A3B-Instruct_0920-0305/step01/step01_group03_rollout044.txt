import numpy as np

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_capped_simplex(h, target_sum=1.0):
    """Project h onto the capped simplex: 0 <= h_i <= 1 and sum(h) = target_sum"""
    # Sort h in descending order and keep track of indices
    sorted_indices = np.argsort(h)[::-1]
    sorted_h = h[sorted_indices]
    # Water-filling approach
    remaining_sum = target_sum
    n = len(sorted_h)
    # Find the threshold index
    for i in range(n):
        if sorted_h[i] * (n-i) < remaining_sum:
            remaining_sum -= sorted_h[i]
        else:
            # The threshold is between sorted_h[i-1] and sorted_h[i]
            threshold = remaining_sum / (n-i)
            # Reconstruct the project
            new_h = sorted_h.copy()
            for j in range(i, n):
                new_h[j] = threshold
            for j in range(i):
                new_h[j] = sorted_h[j]
            # Map back
            h_proj = h.copy()
            for j, idx in enumerate(sorted_indices):
                h_proj[idx] = new_h[j]
            return h_proj
    # All values are too small; set to target_sum/n
    h_proj = np.full_like(h, target_sum/n)
    return h_proj

def logsumexp_stable(log_weights, max_log_w):
    """Compute log(sum(exp(log_weights))) stably"""
    return max_log_w + np.log(np.sum(np.exp(log_weights - max_log_w)))

def softmin_gradient(h, log_R, lambda_param):
    """Compute gradient of the softmin surrogate with respect to h"""
    # compute exp(-lambda * R)
    log_probs = -lambda_param * log_R
    max_log_prob = np.max(log_probs)
    probs = np.exp(log_probs - max_log_prob)
    probs = probs / np.sum(probs)
    # gradient computation: sum of w_s * (h_{j+s} + h_{j-s})
    n = len(h)
    g = np.zeros_like(h)
    # for each shift s, add contribution from h_{j+s} and h_{j-s}
    # this is a bit tricky due to boundary conditions
    # We compute naively with slices; for h_{j+s} we shift the array by -s
    # and similarly for h_{j-s}. Let's use matrix or full mode convolution
    # More directly: compute self correlation for the gradient
    # But let's be efficient, and work with simple loops
    
    # forward correlation of convolved with mask
    # Note that we build the w vector correctly to represent full correlations warp of h
    R = np.correlate(h, h, 'full')
    log_R = np.log(R + 1e-20)  # added small value to avoid log(0)
    
    # compute the weighted sum of h_{j+s}+h_{j-s}
    n_full = len(R)
    n_h = len(h)
    g = np.zeros_like(h)
    
    # for large h, we can do this smartly:
    # use convolution of h with itself to compute correlations,
    # build gradient vector with convolution
    
    # Here one approach: for easier implementation, representing shifts as
    # full-correlations: h has indices from 0 to n_h-1. All the results of 
    # correlation are shifted by indices spanning from -(n_h-1) to + (n_h-1), 
    # so the total size is 2*n_h - 1. 
    # For gradient of Phi_lambda, for index j:
    # dPhi/dh_j = sum_s w_s * (h_{j+s} + h_{j-s})
    
    # Naively compute this contribution for each j:  
    for j in range(n_h):
        for s in range(n_full):
            idx_s = s - (n_h-1)  # current index of the shift
            # Check if indices are valid
            idx1 = j + idx_s
            idx2 = j - idx_s
            corr_idx = s
            if 0 <= idx1 < n_h and 0 <= idx2 < n_h:
                g[j] += probs[corr_idx] * (h[idx1] + h[idx2])
            elif 0 <= idx1 < n_h:
                g[j] += probs[corr_idx] * h[idx1]
            elif 0 <= idx2 < n_h:
                g[j] += probs[corr_idx] * h[idx2]
    
    # Return this component for each index j
    return g

def make_symmetric(h):
    """Symmetrize h around x = 1"""
    n = len(h)
    mid = n // 2
    for i in range(mid):
        avg = (h[i] + h[n - 1 - i]) / 2
        h[i] = h[n - 1 - i] = avg
    # Projection to maintain sum
    h = project_capped_simplex(h, target_sum=1.0)
    return h
