import numpy as np

def project_onto_capped_simplex(y, s):
    """Project vector y onto simplex with sum = s and bounds [0,1]"""
    lo = 0.0
    hi = 1.0
    # initial bounds for lambda
    lam_low = np.min(y) - hi
    lam_high = np.max(y) - lo
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(y - lam, lo, hi)
        total = x.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(y - lam_high, lo, hi)
    return x

def compute_overlaps(h):
    """Compute overlaps for shifts 0..N-1"""
    n = len(h)
    overlaps = np.empty(n)
    one_minus_h = 1.0 - h
    for k in range(n):
        overlaps[k] = np.dot(h[:n-k], one_minus_h[k:])
    overlaps *= (2.0 / n)
    return overlaps

    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)

    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx = 2/N
    dx = 2.0 / len(h)
    correlation *= dx
    # Return maximum correlation value
    return np.max(correlation)
