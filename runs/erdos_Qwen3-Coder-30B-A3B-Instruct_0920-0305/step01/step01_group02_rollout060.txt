import numpy as np
import time
from scipy.optimize import minimize

def project_onto_capped_simplex(y, s):
    lo = 0.0
    hi = 1.0
    # initial bounds for lambda
    lam_low = np.min(y) - hi
    lam_high = np.max(y) + 1.0
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(y - lam, lo, hi)
        total = x.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(y - lam_high, lo, hi)
    return x

def compute_overlaps(h):
    n = len(h)
    one_minus_h = 1.0 - h
    overlaps = np.empty(n)
    for k in range(n):
        overlaps[k] = np.dot(h[:n-k], one_minus_h[k:])
    overlaps *= (2.0 / n)
    return overlaps

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    correlation *= dx
    return np.max(correlation)
