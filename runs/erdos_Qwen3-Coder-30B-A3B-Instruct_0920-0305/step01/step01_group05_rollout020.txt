import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (1.0 / (h_sum * dx))
        h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_h_binary(bits):
    """Evaluate binary h values"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    
    # Ensure sum = n/2 exactly
    h_sum = np.sum(h)
    h = h * (n / 2.0) / h_sum
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):

    np.random.shuffle(bits)
    
    best_val = evaluate_h_binary(bits)

    start = time.time()
    idle_count = 0
    
    while time.time() - start < time_limit and idle_count < 50000:
        improved = False
        
        # Try multiple swaps
        for _ in range(50):
            if time.time() - start >= time_limit:
                break
                
            # Find positions where we can swap
            ones_idx = np.flatnonzero(bits)
            zeros_idx = np.flatnonzero(1 - bits)
            
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
                
            # Pick random indices to swap
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            
            # Try the swap
            bits[i], bits[j] = bits[j], bits[i]
            new_val = evaluate_h_binary(bits)
            
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_bits = bits.copy()
                improved = True
                idle_count = 0
            else:
                # Revert the swap
                bits[i], bits[j] = bits[j], bits[i]
                idle_count += 1
        
        if not improved:
            # Try restart with randomized binary
            bits = np.zeros(n, dtype=int)
            bits[:half] = 1
            np.random.shuffle(bits)
            idle_count = 0
    
    return best_bits, best_val
