import numpy as np

import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    # Ensure sum equals n/2 exactly (since integral should be 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2.0) / h_sum
    
    # Clamp to [0, 1] to respect constraints
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):
    """Run local search on binary solutions of length n"""
    dx = 2.0 / n
    half = n // 2
    start = time.time()
    
    best_sum = float('inf')

        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        curr_val = evaluate_h(bits.astype(float))
        
        # Local hill climbing
        improved = True
        attempts = 0
        while improved and attempts < 5000 and time.time() - start < time_limit:
            improved = False
            attempts += 1
            
            # Try several swaps
            for _ in range(100):
                ones_idx = np.flatnonzero(bits)
                zeros_idx = np.flatnonzero(1 - bits)
                
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                    
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Perform swap
                bits[i], bits[j] = bits[j], bits[i]
                new_val = evaluate_h(bits.astype(float))
                
                if new_val < curr_val - 1e-12:
                    curr_val = new_val
                    improved = True
                else:
                    # Revert swap
                    bits[i], bits[j] = bits[j], bits[i]
            
            if not improved:
                # Random restart: flip some bits
                flips = max(1, n // 20)  # Flip ~5% of bits

                curr_val = evaluate_h(bits.astype(float))
        
        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
    
    return best_bits, best_sum
