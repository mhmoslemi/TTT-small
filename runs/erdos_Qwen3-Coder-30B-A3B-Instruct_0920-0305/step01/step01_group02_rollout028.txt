import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    # Simple projection: clip and normalize
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale = target_sum / current_sum if current_sum > 0 else 1.0
    x = x * scale
    return x

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    # Compute correlations for all lags
    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx = 2/N
    dx = 2.0 / len(h)
    correlation *= dx
    # Return maximum correlation value
    return np.max(correlation)

    """Use differential evolution to optimize the function"""
    def objective(x):

        return evaluate_erdos_solution(h)
    
    # bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=1000, 
        popsize=15, 
        mutation=(0.5, 1),
        recombination=0.7,
        seed=42,
        disp=False
    )
    
    return project_to_feasible(result.x, target_sum)
