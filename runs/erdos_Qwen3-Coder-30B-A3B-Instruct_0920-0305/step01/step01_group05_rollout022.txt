def project_to_box_simplex(y, target_sum=0.5):
    """Project vector y onto the simplex {x : 0 <= x <= 1, sum(x) = target_sum}"""
    # Simple bisection on alpha
    low, high = np.min(y), np.max(y) + 1.0
    eps = 1e-12
    
    def sum_after_projection(alpha):
        z = np.clip(y - alpha, 0, 1)
        return np.sum(z)
    
    for _ in range(100):
        alpha = (low + high) / 2
        s = sum_after_projection(alpha)
        if abs(s - target_sum) < eps:
            break
        elif s > target_sum:
            low = alpha
        else:
            high = alpha
            
    return np.clip(y - alpha, 0, 1)

def compute_fs(h, s, dx):
    """Compute f_s(h) for shift s"""
    n = len(h)
    if s >= 0:
        # Right shift
        h_shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        # Left shift
        h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.sum(h * (1 - h_shifted)) * dx

def compute_all_fs(h, dx):
    """Compute all f_s(h) for s = -(n-1) to n-1"""
    n = len(h)
    fs = []
    for s in range(1 - n, n):
        fs.append(compute_fs(h, s, dx))
    return np.array(fs)

def compute_grad(h, fs, p):
    """Compute gradient of Phi_p(h)"""
    n = len(h)
    grad = np.zeros(n)
    if p == 1:
        # Special case for p=1 (use subgradient-like approach)
        # This is simplified; in reality, requires careful handling
        max_idx = np.argmax(np.abs(fs))
        s = max_idx - (n - 1)
        if s >= 0:
            h_shifted = np.concatenate([np.zeros(s), h[:-s]])
        else:
            h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
        grad = 1 - h_shifted - np.roll(h_shifted, -s)  # Just for direction
        return grad
    else:
        # Compute with proper weights
        weights = np.abs(fs)**(p-1)
        # Since the weights are already known, avoid the whole correlation again
        grad = np.zeros(n)
        for i in range(n):
            temp_grad = 0
            for j in range(-n+1, n):
                s = j
                if s >= 0:
                    h_shifted = np.concatenate([np.zeros(s), h[:-s]])
                else:
                    h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
                temp_grad += weights[j + n - 1] * (1 - h_shifted[i] - h_shifted[i])
            temp_grad /= np.sum(weights)**((p-1)/p)
            grad[i] = temp_grad
        return grad

def smooth_max_norm(h, p=2, dx=2.0/42):
    """compute the smooth p-norm max for given h and p"""
    fs = compute_all_fs(h, dx)
    return np.sum(np.abs(fs)**p)**(1./p)

def objective_p(h, p, lambda_penalty, dx=2./42):
    """Compute J_p(h) = Phi_p(h) + lambda_penalty * (sum(h) - n/2)^2"""
    n = len(h)
    fs = compute_all_fs(h, dx)
    Phi_p = np.sum(np.abs(fs)**p)**(1./p)
    penalty = lambda_penalty * (np.sum(h) - n/2)**2
    return Phi_p + penalty

def projected_gradient_descent(h0, p, lambda_penalty, max_iter=2000, step_size=1e-3, tol=1e-8, dx=2./42):
    """Perform projected gradient descent for smooth max norm"""
    h = h0.copy()
    n = len(h)
    prev_obj = objective_p(h, p, lambda_penalty, dx)
    
    for k in range(max_iter):
        fs = compute_all_fs(h, dx)
        grad = compute_grad(h, fs, p)
        grad += 2 * lambda_penalty * (np.sum(h) - n/2)  # Add penalty term 
        h = h - step_size * grad
        h = project_to_box_simplex(h, n/2)
        
        curr_obj = objective_p(h, p, lambda_penalty, dx)
        if abs(prev_obj - curr_obj) < tol:
            break
        prev_obj = curr_obj
        
    return h

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
