def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_onto_sum_constraint(z, target_sum):
    """Project z onto the set {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Find scalar alpha such that clip(z + alpha * ones, 0, 1) has sum = target_sum
    def sum_of_clipped(alpha):
        z_clipped = np.clip(z + alpha, 0, 1)
        return np.sum(z_clipped)
    
    # Ensure we can actually reach the target sum
    lower_bound = np.sum(np.clip(z - 1, 0, 1))
    upper_bound = np.sum(np.clip(z, 0, 1))
    
    if target_sum <= lower_bound:
        return np.clip(z - 1, 0, 1)
    elif target_sum >= upper_bound:
        return np.clip(z, 0, 1)
    
    # Bisection search
    a, b = -10000, 10000
    for _ in range(30):
        mid = (a + b) / 2
        s = sum_of_clipped(mid)
        if s < target_sum:
            a = mid
        else:
            b = mid
    
    alpha = (a + b) / 2
    return np.clip(z + alpha, 0, 1)

    w, V = eigh(X_hat)
    w = np.maximum(w, 1e-10)

    mask = w > 1e-8
    w = w[mask]
    V = V[:, mask]
    
    # Create square root of matrix
    B = V @ np.diag(np.sqrt(w))
    
    # Generate Gaussian sample
    r = np.random.randn(B.shape[1])
    z = B @ r
    
    # Clip and then project
    z_clipped = np.clip(z, 0, 1)
    h_candidate = project_onto_sum_constraint(z_clipped, target_sum)
    
    return h_candidate
