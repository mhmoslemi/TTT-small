import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project y onto the simplex {x | sum(x) = target_sum, 0 <= x_i <= 1}"""
    n = len(y)
    # Sort y in descending order
    sorted_y = np.sort(y)[::-1]
    # Compute partial sums
    cumsum = np.cumsum(sorted_y)
    # Find the index where the projection cuts the simplex
    idx = np.where(sorted_y > (target_sum - cumsum + sorted_y) / np.arange(1, n+1))[0]
    if len(idx) == 0:
        # All entries are too large, set to the same value
        lambda_ = target_sum / n
        x = np.clip(y - lambda_, 0, 1)
    else:
        # The last valid index
        k = idx[-1]
        lambda_ = (target_sum - cumsum[k]) / (k + 1)
        x = np.clip(y - lambda_, 0, 1)
    return x

def compute_grad(h, c_s, alpha):
    """Compute gradient of log-sum-exp approximation of the max correlation."""
    n = len(h)
    # Compute softmax weights with stability trick
    c_max = np.max(c_s)
    exp_c = np.exp(alpha * (c_s - c_max))
    w = exp_c / np.sum(exp_c)
    # Compute gradient for each index
    grad = np.zeros_like(h)
    for s in range(-n+1, n):
        # Compute the forward and backward shifts with zero-padding
        if s > 0:
            h_shift = np.concatenate([np.zeros(s), h[:-s]])
        elif s < 0:
            h_shift = np.concatenate([h[-s:], np.zeros(-s)])
        else:
            h_shift = h.copy()
        grad -= w[s+n-1] * h_shift
    grad += 1.0  # Add the constant term
    return grad
