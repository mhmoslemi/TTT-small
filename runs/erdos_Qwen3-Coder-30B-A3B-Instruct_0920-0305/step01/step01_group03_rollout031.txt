import numpy as np
from scipy.optimize import minimize
import time

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}"""
    # Simple method: sort and waterfilling approach
    v = v.copy()
    n = len(v)
    # Clip to bounds
    v = np.clip(v, lower, upper)
    total = np.sum(v)
    if abs(total - target_sum) < 1e-10:
        return v
    # Binary search to find the shift
    low, high = -10.0, 10.0
    for _ in range(50):
        theta = (low + high) / 2.0
        projected = np.clip(v + theta, lower, upper)
        if np.sum(projected) > target_sum:
            high = theta
        else:
            low = theta
    # Final projection
    theta = (low + high) / 2.0
    result = np.clip(v + theta, lower, upper)
    return result

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def smooth_objective(h, lambda_val, dx=2.0/200):
    """Smooth approximation of the objective using log-sum-exp"""
    n = len(h)
    
    # Compute correlation of h with h
    corr_hh = np.correlate(h, h, mode="full")
    # Compute -lambda * corr (log-sum-exp needs negative values)
    corr_scaled = -lambda_val * corr_hh
    
    # Numerically stable log-sum-exp
    max_corr = np.max(corr_scaled)
    log_sum_exp = max_corr + np.log(np.sum(np.exp(corr_scaled - max_corr)))
    # The smooth objective is: (1/lambda) * log_sum_exp
    return log_sum_exp / lambda_val

def gradient_smooth_objective(h, lambda_val, dx=2.0/200):
    """Compute gradient of the smooth objective"""
    n = len(h)
    
    # Compute correlation of h with h
    corr_hh = np.correlate(h, h, mode="full")
    # Compute weights for each shift
    corr_scaled = -lambda_val * corr_hh
    max_corr = np.max(corr_scaled)
    weights = np.exp(corr_scaled - max_corr)
    weights = weights / np.sum(weights)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for shift in range(-(n-1), n):
        idx = shift + (n-1)
        w = weights[idx]
        if w == 0:
            continue
        if shift >= 0:
            # h_{i+shift} contribution
            grad[:n-shift] += w * h[shift:]
            # h_{i-shift} contribution
            grad[shift:] += w * h[:n-shift]
        else:
            # for shift < 0, h_{i-shift} is h_{i+|shift|}
            shift_abs = -shift
            grad[shift_abs:] += w * h[:n-shift_abs]
            grad[:n-shift_abs] += w * h[shift_abs:]

    return -grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use fine grid with 200 points
    n_points = 200
    dx = 2.0 / n_points
    
    # Start from symmetrical pattern
    h = np.ones(n_points) * 0.5
    h = project_onto_box_simplex(h, n_points / 2, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Try multiple restarts
    for restart in range(30):  # Reduced number to save time
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Random restart with some perturbation
        h_random = np.random.uniform(0, 1, n_points)
        h_random = project_onto_box_simplex(h_random, n_points / 2, 0.0, 1.0)
        
        # Try a few random tweaks
        h_tweak = h_random.copy()
        indices = np.random.choice(n_points, size=int(n_points/10), replace=False)
        for idx in indices:
            h_tweak[idx] += np.random.uniform(-0.1, 0.1)
        h_tweak = np.clip(h_tweak, 0.0, 1.0)
        h_tweak = project_onto_box_simplex(h_tweak, n_points / 2, 0.0, 1.0)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(h_tweak, dx)
        if current_c5 < best_c5:
            best_h = h_tweak.copy()
            best_c5 = current_c5
    
    # Use gradient descent with smoothing
    # Schedule of lambda values from small to large
    lambda_schedule = [1, 5, 10, 20, 40]
    
    # Start with best solution so far
    h = best_h.copy()
    
    # Gradient descent with variable lambda
    for lambda_val in lambda_schedule:
        # Run for fewer iterations to save time
        iterations_per_lambda = 50
        
        # Adaptive learning rate
        alpha = 0.05 / lambda_val
        
        for _ in range(iterations_per_lambda):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            grad = gradient_smooth_objective(h, lambda_val, dx)
            h = h - alpha * grad
            h = project_onto_box_simplex(h, n_points / 2, 0.0, 1.0)
            
            # Evaluate and keep track of best
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final refinement using scipy minimize with raw objective
    def objective(h_vals):
        return evaluate_erdos_solution(h_vals, dx)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    if result.success:
        final_c5 = objective(result.x)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = result.x.copy()
    
    # Ensure final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = project_onto_box_simplex(best_h, n_points / 2, 0.0, 1.0)
    
    return (best_h.tolist(), best_c5, n_points)
