import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def proj_onto_simplex(v, target_sum, lower=0.0, upper=1.0):
    # v: array
    # return x solving min ||x - v||^2 subject to sum x = target_sum, lower <= x <= upper
    n = len(v)
    # clamp to lower and upper? Actually we need to find lambda such that x = clamp(v - lambda, lower, upper) sums to target.
    # Define function f(lambda) = sum(clamp(v - lambda, lower, upper)) - target_sum.
    # f is decreasing in lambda (since clamp of v - lambda). So we can binary search.
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam  # need larger lambda to reduce sum
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, lower, upper)
    return x

    iteration = 0
    
    while (time.time() - start_time) < budget_s * 0.95 and iteration < max_iter:
        iteration += 1
        
        # Propose new state by perturbing two random indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Compute max allowable delta
        max_up_i = 1 - current_h[i]
        max_down_i = current_h[i]
        max_up_j = 1 - current_h[j]
        max_down_j = current_h[j]
        
        delta_low = max(-max_down_i, -max_up_j)
        delta_high = min(max_up_i, max_down_j)
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        
        # Create new state
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Compute new C5
        new_c5 = evaluate_c5(new_h)
        
        # Accept with probability
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = new_h
                current_c5 = new_c5
                
        # Update best if needed
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Update temperature
        T = max(T_min, T * alpha)
    
    return (best_h.tolist(), best_c5, n_points)
