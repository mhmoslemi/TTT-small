import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (1.0 / (h_sum * dx))
    
    # Clip values to [0, 1] to maintain constraints
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_c5_binary(bits):
    """Evaluate C5 for binary vector"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def binary_local_search(n, time_limit):
    """Perform local search starting from binary vectors"""
    dx = 2.0 / n
    half = n // 2
    
    best_sum = float('inf')
    best_bits = None
    start = time.time()
    
    # Generate initial binary vector approximately
    bits = np.zeros(n, dtype=int)
    bits[:half] = 1
    np.random.shuffle(bits)
    
    # Local search loop
    while time.time() - start < time_limit:
        # Run local descent
        current_bits = bits.copy()
        current_c5 = evaluate_c5_binary(current_bits)
        improved = True
        
        # Try to improve by swapping
        attempts = 0
        while improved and time.time() - start < time_limit and attempts < 1000:
            improved = False
            attempts += 1
            
            # Generate new candidate by swapping
            ones_idx = np.flatnonzero(current_bits)
            zeros_idx = np.flatnonzero(1 - current_bits)
            
            if len(ones_idx) > 0 and len(zeros_idx) > 0:
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Create new candidate
                new_bits = current_bits.copy()
                new_bits[i], new_bits[j] = new_bits[j], new_bits[i]
                
                new_c5 = evaluate_c5_binary(new_bits)
                
                if new_c5 < current_c5 - 1e-12:
                    current_bits = new_bits
                    current_c5 = new_c5
                    improved = True
            else:
                break
    
        # Update global best
        if current_c5 < best_sum:
            best_sum = current_c5
            best_bits = current_bits.copy()
        
        # Restart with new random binary vector with exact half ones
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        # Break early if already found a good solution
        if best_sum < 0.3807:
            break
    
    if best_bits is None:
        # Fallback on random binary vector
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        best_sum = evaluate_c5_binary(bits)
        best_bits = bits
    
    return best_bits, best_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # List of n values to try
    n_list = [42, 60, 80]  # Start with these sizes, we already know 42 with C5=0.382878
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in n_list:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Run binary local search for this n
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break
            
        # Binary search for the current n
        binary_h, binary_c5 = binary_local_search(n, min(400.0, remaining_time * 0.9))
        
        # Final evaluation
        dx = 2.0 / n
        binary_h = np.array(binary_h, dtype=float)
        h_sum = np.sum(binary_h)
        if h_sum != 0:
            binary_h = binary_h * (1.0 / (h_sum * dx))
            binary_h = np.clip(binary_h, 0, 1)
        
        final_c5 = evaluate_h(binary_h)
        
        # Check if this is the best solution
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = binary_h
            best_n = n
            
    return (best_h.tolist() if best_h is not None else [0.0]*42, 
            best_c5, 
            best_n if best_n is not None else 42)
