import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

    # Use integer programming to solve exactly
    # Variables: x[i] ∈ {0,1}, binary; t the maximum mismatch
    x = cp.Variable(n_points, boolean=True)
    t = cp.Variable()

    # Constraints
    # Sum of x equals N/2 (half ones)
    constraints = [cp.sum(x) == n_points // 2]
    
    # Limit t from above by performing computation of max mismatch
    # For each shift k from 1 to N-1, consider the mismatch at that shift
    # Mismatch is sum_{i=0}^{N-1-k} (x_i - x_i * x_{i+k})
    
    # Compute mismatch for positive shifts k
    for k in range(1, n_points):
        # Only range where both x[i] and x[i+k] exist
        mismatch_pos = cp.sum(x[:n_points-k]) - cp.sum(cp.multiply(x[:n_points-k], x[k:]))
        # For negative shifts, we compute from the perspective of x[i+k] = x[i]*x[i+k]
        mismatch_neg = cp.sum(x[k:]) - cp.sum(cp.multiply(x[k:], x[:n_points-k]))
        
        # Both must be <= t
        constraints.append(mismatch_pos <= t)
        constraints.append(mismatch_neg <= t)

    problem = cp.Problem(cp.Minimize(t), constraints)
