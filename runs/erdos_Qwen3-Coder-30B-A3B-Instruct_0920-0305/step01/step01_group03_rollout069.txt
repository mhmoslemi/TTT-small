    resolutions = [80, 100, 120]
    resolutions = [r for r in resolutions if r <= max_resolution]

        if time.time() - start_time > budget_s * 0.95:
            break

        # Build LP variables
        x = Variable(n)
        t = Variable()
        w = {}
        
        # Build index list for all valid (i,s) pairs
        valid_pairs = []
        for s in range(-(n-1), n):
            for i in range(n):
                if 0 <= i+s < n:
                    valid_pairs.append((i,s))
                    w[(i,s)] = Variable()
        
        # Constraints
        constraints = []
        
        # McCormick inequalities
        for i, s in valid_pairs:
            constraints += [
                w[(i,s)] <= x[i],
                w[(i,s)] <= x[i+s],
                w[(i,s)] >= x[i] + x[i+s] - 1,
                w[(i,s)] >= 0
            ]

        for s in range(-(n-1), n):
            # Collect valid indices for this shift s
            valid_i = [i for i in range(n) if 0 <= i+s < n]
            if valid_i:
                # Sum of w_{i,s} >= t
                constraints.append(sum([w[(i,s)] for i in valid_i]) >= t)
        
        # Mass constraint
        constraints.append(sum(x) == n/2)
        
        # Box constraints
        constraints += [x[i] >= 0 for i in range(n)]
        constraints += [x[i] <= 1 for i in range(n)]
        
        # Objective
        objective = Maximize(t)
        
        # Solve the LP
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, abstol=1e-8, reltol=1e-6, max_iters=5000)
