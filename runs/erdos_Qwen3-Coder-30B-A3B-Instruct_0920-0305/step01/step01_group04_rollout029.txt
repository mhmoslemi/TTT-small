import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time

def full_correlation(h):
    """Compute the full correlation vector and its maximum value."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n0 = 61  # odd prime, "good" for SDP
    n_points = 200  # Target size
    dx = 2.0 / n_points
    w0 = n0 / 2  # Target weight for the base vector
    
    # Step 1: Build Toeplitz matrices C_d
    def build_toeplitz_matrix(n, shift):
        C = np.zeros((n, n))
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                C[i, j] = 1
                C[j, i] = 1
        return C

    # Build all Toeplitz matrices for shifts d = -n0+1 to n0-1, excluding 0
    C_matrices = {}
    for d in range(-n0+1, n0):
        if d != 0:
            C_matrices[d] = build_toeplitz_matrix(n0, d)

    # Step 2: SDP Model
    X = cp.Variable((n0, n0), symmetric=True)
    tau = cp.Variable()

    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == 1  # Unit diagonal
    ]

    # Add autocorrelation constraints
    for d in range(-n0+1, n0):
        if d != 0:
            constraints.append(cp.trace(C_matrices[d] @ X) >= tau)

    # Maximize tau
    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)

    # Solve SDP
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
        
        if prob.status != cp.OPTIMAL:
            raise Exception("SDP not solved to optimality")
    except Exception:

        c5_best = full_correlation(h0) * dx
        return list(h0), c5_best, len(h0)

    # Step 3: Random hyperplane rounding
    # Get the optimal X
    X_opt = X.value
    if X_opt is None:
        # Fallback
        h0 = np.array(initial_h_values)
        c5_best = full_correlation(h0) * dx
        return list(h0), c5_best, len(h0)

    # Factorize X = V V^T (rank-revealing)
    try:
        evals, evecs = eigh(X_opt)
        V = evecs[:, evals > 1e-8]
        # If rank is low, ensure V is full dimensional
        if V.shape[1] == 0:
            V = np.eye(n0)
    except Exception:
        V = np.eye(n0)
    
    R = 200  # Number of random roundings
    best_h = None
    best_M = float('inf')
    
    for r in range(R):
        if time.time() - start_time > budget_s - 2:
            break
            
        g = np.random.randn(V.shape[1])
        s = np.sign(V.T @ g)
        # Ensure perfect equilibrium by fixing sign of one element (if discrepancy)
        S = np.sum(s)
        if abs(S) > 1e-6:
            # Flip the element with smallest absolute dot product to balance
            if S > 0:
                # Flip most negative element
                idx = np.argmin(s)
                s[idx] = -s[idx]
            else:
                # Flip most positive element
                idx = np.argmax(s)
                s[idx] = -s[idx]
        
        h = (s + 1) / 2
        # Clamp to [0, 1]
        h = np.clip(h, 0, 1)
        
        M = full_correlation(h)
        if M < best_M:
            best_M = M
            best_h = h.copy()

    # Step 4: Upscale to target length
    if best_h is None:
        h0 = np.array(initial_h_values)
        c5_best = full_correlation(h0) * dx
        return list(h0), c5_best, len(h0)

    h_upscaled = np.concatenate([best_h, 1 - best_h])
    
    # If needed, replicate and/or adjust length
    while len(h_upscaled) < n_points:
        h_upscaled = np.concatenate([h_upscaled, 1 - best_h])

    # Trim or pad to proper length
    if len(h_upscaled) > n_points:
        h_upscaled = h_upscaled[:n_points]
    elif len(h_upscaled) < n_points:
        # Add complement to balance
        remainder = n_points - len(h_upscaled)
        h_upscaled = np.concatenate([h_upscaled, 1 - best_h[:remainder]])

    # Final local search (hill climbing)
    h_final = h_upscaled.copy()
    max_corr_prev = full_correlation(h_final)
    max_corr_current = max_corr_prev

    passes = 0
    MAX_PASSES = 10
    while passes < MAX_PASSES:
        improved = False
        # Try all possible swaps (i, j) where h[i]=1, h[j]=0
        for i in range(len(h_final)):
            if h_final[i] != 1:
                continue
            for j in range(len(h_final)):
                if h_final[j] != 0:
                    continue
                # Simulate the swap: h[i] = 0, h[j] = 1
                h_trial = h_final.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                
                max_corr_new = full_correlation(h_trial)
                if max_corr_new < max_corr_current:
                    max_corr_current = max_corr_new
                    h_final = h_trial
                    improved = True
                    
        if not improved:
            break
        passes += 1

    # Final evaluation
    final_c5 = full_correlation(h_final) * dx
    h_final = np.clip(h_final, 0, 1)
    return list(h_final), final_c5, n_points
