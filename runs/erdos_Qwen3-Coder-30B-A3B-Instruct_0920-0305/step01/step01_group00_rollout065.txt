import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.fft import ifft, fft

def project_simplex(y, s=0.5):
    """
    Project vector y onto the simplex {x: sum(x)=s, 0<=x<=1}
    Using algorithm from Duchi et al., "Efficient Projections onto the
    l1-ball for learning in high dimensions"
    """
    # Get the sorted values
    u = np.sort(y)[::-1]
    # Compute the cumulative sums
    su = np.cumsum(u)
    # Compute the threshold
    t = (su - s) / np.arange(1, len(u) + 1)
    # Find the largest k s.t. t_k <= u_k
    rho = np.sum(u > t)
    # The threshold lambda is t_rho
    if rho == 0:
        lambda_star = 0.0
    else:
        lambda_star = t[rho - 1]
    # Compute the projection
    x = np.clip(y - lambda_star, 0, 1)
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def smooth_max(objs, alpha):
    """Compute log-sum-exp of the objectives"""
    # Center to avoid overflow
    max_obj = np.max(objs)
    exps = np.exp(alpha * (objs - max_obj))
    log_sum = max_obj + np.log(np.sum(exps)) / alpha
    return log_sum

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation values (full mode)
    corr = correlate(h, 1-h, mode="full")
    # Max to avoid overflow in log-sum-exp
    maxcorr = np.max(corr)
    # Compute softmax weights
    exps = np.exp(alpha * (corr - maxcorr))
    w = exps / np.sum(exps)
    # Compute gradient using the formula for d/dh (sum w_s C_s)
    # C_s = 1 - sum h_i h_{i+s} or similar depending on shift
    # We calculate dC_s/dh_i = 0 if i not in support of shift s
    # or -h_{i+s} for forward shift, -h_{i-s} for backward
    grad = np.ones(n)
    # Accumulate contributions from each shift
    for s in range(2*n - 1):
        # This corresponds to shift -n+1 to n-1
        shift = s - (n - 1)
        if shift <= 0:
            # shift is negative or zero, so h_{i+s} = h_{i+shift}
            # Only valid indices
            if shift < 0:
                h_shifted = np.concatenate([h[-shift:], np.zeros(-shift)])
            else:
                h_shifted = h.copy()
        else:
            # Positive shift, so h_{i+s} = h_{i+shift} at end
            h_shifted = np.concatenate([np.zeros(shift), h[:-shift]])
        
        # Gradient contribution
        contrib = w[s] * h_shifted 
        grad -= contrib
    
    return grad

def projected_gradient_descent(h0, alpha_start, alpha_growth, max_iters, lr_start, time_budget, early_stop_c5):
    start_time = time.time()
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha_start
    
    for t in range(max_iters):
        if time.time() - start_time > time_budget:
            break
            
        # Compute the gradient
        grad = compute_gradient(h, alpha)
        # Compute learning rate
        eta = lr_start / np.sqrt(t + 1)
        # Update
        y = h - eta * grad
        # Project
        h = project_simplex(y, 0.5)
        
        # Compute true objective
        c5 = evaluate_c5(h)
        # Check if better
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early stop if good enough
        if best_c5 < early_stop_c5:
            break
            
        # Anneal alpha
        if t % 200 == 0 and t > 0:
            alpha *= alpha_growth
            if alpha > 200.0:
                alpha = 200.0
                
    return best_h, best_c5

    dx = 2.0 / n_points
    target_sum = 0.5  # since sum h_i = n/2 = 48
    
    # Create initial vector
    if 'initial_h_values' in globals():

        h0 = np.interp(x_new, x_orig, init_vals)
    else:
        # Draw from uniform and project to simplex
        h0 = np.random.rand(n_points)
        h0 = project_simplex(h0, target_sum)
    
    # Run multiple restarts
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Budget for global restarts (70%)
    restart_budget = budget_s * 0.7
    local_budget = budget_s - restart_budget
    
    # Perform multiple restarts
    max_restarts = 200
    restart_count = 0
    
    while (time.time() - start_time < restart_budget) and restart_count < max_restarts:
        # Start with current best or new random point
        if restart_count == 0:
            h_current = h0.copy()
        else:
            # Random initialization on simplex
            h_current = np.random.rand(n_points)
            h_current = project_simplex(h_current, target_sum)
        
        # Projected gradient descent
        h_refined, c5_refined = projected_gradient_descent(
            h_current, 
            alpha_start=5.0,
            alpha_growth=1.03,
            max_iters=2000,
            lr_start=0.5,
            time_budget=(restart_budget - (time.time() - start_time)) / (max_restarts - restart_count),
            early_stop_c5=best_c5
        )
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
        restart_count += 1
    
    # Local refinement with SLSQP (small scale)
    if time.time() - start_time < budget_s:
        # Define bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint function to enforce sum = n/2
        def constraint_eq(h):
            return np.sum(h) * dx - target_sum
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Refinement with SLSQP
        try:
            res = minimize(lambda h: evaluate_c5(h), best_h.copy(), method='SLSQP',

            if res.success and evaluate_c5(res.x) < best_c5:
                best_h = res.x
                best_c5 = evaluate_c5(res.x)
        except:
            pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
