import numpy as np

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def round_and_repair(h_vec, X, n):

        sqrt_s = np.sqrt(s)
        B = U * sqrt_s
        g = np.random.randn(B.shape[1])
        z = B @ g
        z = np.clip(z, 0, 1)
        
        # Project to correct sum
        def project_sum(alpha):
            y = z + alpha
            y = np.clip(y, 0, 1)
            return np.sum(y) - n/2
        
        # Bisection to find alpha
        a, b = -1.0, 1.0
        for _ in range(30):
            c = (a + b) / 2
            val = project_sum(c)
            if val < 0:
                a = c
            else:
                b = c
        
        final_h = np.clip(z + (a + b)/2, 0, 1)
        final_h = final_h * (n/2) / np.sum(final_h)
        return final_h
    except:
        # Fallback to simple rounding if SVD fails
        h_rounded = np.clip(h_vec, 0, 1)
