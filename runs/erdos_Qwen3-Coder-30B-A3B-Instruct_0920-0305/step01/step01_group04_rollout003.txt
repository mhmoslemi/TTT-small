def evaluate_h(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h_vals):
    """Project h_vals onto [0,1]^n with sum = n/2."""
    h = np.array(h_vals)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) < 1e-10:
        return h
    diff = (n_points / 2) - current_sum
    # Simple adjustment using constructive method: just increment/decrement values
    if diff > 0:
        # Add diff to the lowest values
        idxs = np.argsort(h)
        for i in range(int(np.ceil(diff))):
            if h[idxs[i]] < 1:
                h[idxs[i]] = min(1.0, h[idxs[i]] + (1.0 - h[idxs[i]]) * (diff / (n_points - np.sum(h >= 1.0))))
    else:
        # Subtract
        idxs = np.argsort(h)[::-1]
        for i in range(int(np.ceil(-diff))):
            if h[idxs[i]] > 0:
                h[idxs[i]] = max(0.0, h[idxs[i]] + diff / (n_points - np.sum(h <= 0.0)))
    return h

def round_binary(h_vals):
    """Round h_vals to 0,1,0.5 with sum exactly n_points/2."""
    h = np.array(h_vals)
    sorted_idx = np.argsort(h)[::-1]
    # Set top floor(n_points/2) values to 1
    goal_ones = n_points // 2
    result = np.zeros_like(h)
    if n_points % 2 == 0:
        result[sorted_idx[:goal_ones]] = 1.0
    else:
        # One 0.5
        result[sorted_idx[:goal_ones]] = 1.0
        # Put one 0.5 in a safe slot not used for 1's
        result[sorted_idx[goal_ones]] = 0.5
    return result

def hill_climb(h_vals):
    """Hill climb to improve binary 0/1 vector."""
    h = np.array(h_vals)
    h = np.clip(h, 0, 1)
    def compute_c5(h_arr):
        corr = np.correlate(h_arr, 1 - h_arr, mode="full")
        return np.max(corr)
        
    current_c5 = compute_c5(h)
    improved = True
    while improved:
        improved = False
        for i in range(len(h)):
            if h[i] != 0:
                # Try to decrease by changing h[i] to 0
                new = h.copy()
                new[i] = 0
                new = project_to_feasible(new)
                new_c5 = compute_c5(new)
                if new_c5 < current_c5:
                    h = new
                    current_c5 = new_c5
                    improved = True
                    break
        for i in range(len(h)):
            if h[i] != 1:
                # Try to increase by changing h[i] to 1
                new = h.copy()
                new[i] = 1
                new = project_to_feasible(new)
                new_c5 = compute_c5(new)
                if new_c5 < current_c5:
                    h = new
                    current_c5 = new_c5
                    improved = True
                    break
    # Ensure correct sum exactly
    h = project_to_feasible(h)
    return h

def sub_gradient_step(h, d_star, learning_rate=0.35):
    # Compute the gradient for the max shift d_star
    g = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        j_forward = (i + d_star) % n
        j_backward = (i - d_star) % n
        # Adjust for out of bounds
        g[i] = 1 - h[j_forward] - h[j_backward]
        
    # Compute update
    h_new = h - learning_rate * g
    # Project to feasible domain [0,1]
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_feasible(h_new)
    return h_new
