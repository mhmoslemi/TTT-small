import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project y onto the capped simplex with sum = target_sum"""
    n = len(y)
    # Binary search for the threshold lambda
    low, high = np.min(y) - 1.0, np.max(y) + 1.0
    for _ in range(50):
        lam = (low + high) / 2.0
        x = np.clip(y - lam, 0, 1)
        current_sum = np.sum(x)
        if current_sum > target_sum:
            low = lam
        else:
            high = lam
    x = np.clip(y - low, 0, 1)
    # Scale to enforce sum = target_sum
    scale = target_sum / np.sum(x)
    x = x * scale
    return x

def compute_gradient(h, correlation_values, alpha):
    """Compute gradient of the log-sum-exp surrogate"""
    n = len(h)
    # Compute softmax weights
    max_c = np.max(correlation_values)
    exp_c = np.exp(alpha * (correlation_values - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Initialize gradient
    grad = np.ones(n)
    
    # Compute weighted sum of shifted h values with appropriate indexing
    # Each shift contributes to the gradient according to the weight w_s
    for s in range(-n+1, n):
        if s >= 0:
            forward_shifted = np.concatenate([h[s:], np.zeros(s)])
            backward_shifted = np.concatenate([np.zeros(-s), h])
        else:
            forward_shifted = np.concatenate([np.zeros(-s), h])
            backward_shifted = np.concatenate([h[-s:], np.zeros(-s)])
        grad -= w[s + n - 1] * (forward_shifted + backward_shifted)
    
    return grad

def simple_line_search(y, h, grad, alpha, F_alpha, step_size, hitting_set, n):
    """Simple backtracking line search"""
    c1 = 1e-4
    rho = 0.5
    y_new = y - step_size * grad
    h_new = project_to_simplex(y_new, n / 2)
    # Compute new surrogate value
    corr_new = correlate(h_new, 1 - h_new, mode="full")
    max_new = np.max(corr_new)
    F_alpha_new = (1 / alpha) * np.log(np.sum(np.exp(alpha * (corr_new - max_new))))
    
    # Simple Armijo condition
    if F_alpha_new < F_alpha - c1 * step_size * np.dot(grad, grad):
        return h_new, step_size
    else:
        return simple_line_search(y, h, grad, alpha, F_alpha, step_size * rho, hitting_set, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set the grid size
    n_points = 96
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize the solution from the existing initial_h_values if possible
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Interpolate to required n_points

            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
    else:
        # Draw a random point on the simplex
        h0 = np.random.rand(n_points)
    
    # Scale to ensure sum = target_sum
    h0 = h0 * target_sum / np.sum(h0)
    
    # Initialize algorithm parameters
    alpha = 5.0
    r_alpha = 1.03
    max_alpha = 200.0
    step_size_init = 0.5
    max_iter = 5000
    min_improvement = 1e-6
    best_c5 = 1e10
    best_h = None
    
    iter_times = []  # For stats, not crucial
    
    def time_remaining():
        return budget_s - (time.time() - start_time)
    
    # Multi restart schedule
    restart_count = 0
    max_restarts = 300 if time_remaining() > 10 else 10  # Reduce if little time left
    
    while restart_count < max_restarts:
        if time_remaining() < 5:
            break
        start_restart = time.time()
        
        # Start with current h0
        h = h0.copy()
        
        # Annealing schedule
        alpha = 5.0  # Reset alpha for each restart
        r_alpha = 1.03
        max_alpha = 200.0
        max_iter_restart = 2000  # Trials per restart
        
        # For tracking improvement
        last_c5 = 1e10
        no_impr_count = 0
        
        for i in range(max_iter_restart):
            if time_remaining() < 5:
                break
                
            # If the difference in C5 is on the order of 1e-6 or less for 50 iterations, stop
            if i > 50 and abs(last_c5 - best_c5) < 1e-6:
                no_impr_count += 1
                if no_impr_count > 20:
                    break
            else:
                no_impr_count = 0
                
            # Compute unnormalized correlation values (full mode)
            corr = correlate(h, 1 - h, mode="full")
            # Compute surrogate value for the current alpha
            max_corr = np.max(corr)
            exp_corr = np.exp(alpha * (corr - max_corr))
            F_alpha = (1 / alpha) * np.log(np.sum(exp_corr))
            
            # Compute gradient (simplified version for linear shifts)
            grad = compute_gradient(h, corr, alpha)
            
            # Perform gradient step
            step_size = step_size_init / np.sqrt(i + 1)
            y = h - step_size * grad
            h_new = project_to_simplex(y, target_sum)
            
            # Update h
            h = h_new
            
            # Recompute true C5 for tracking purposes
            corr_true = correlate(h, 1 - h, mode="full")

                if best_c5 <= 0.3807:

            # Update alpha (annealing)
            if i % 200 == 0 and alpha < max_alpha:

            # Enforce timestopping
            if time.time() - start_time > budget_s - 2:
                break
                
        restart_count += 1
        iter_times.append(time.time() - start_restart)  # Not used for logic
    
    # Now refine with local optimizer to get better C5
    if best_h is not None:
        h_refined = best_h.copy()
        try:
            from scipy.optimize import minimize
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) * dx - 1.0
                
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Use a faster local optimizer with k=200 max iterations
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds,

                h_refined = res.x

            else:
                final_c5 = evaluate_c5(h_refined)
        except:
            final_c5 = evaluate_c5(h_refined)
        best_c5 = min(best_c5, final_c5)
        best_h = h_refined
    
    # Ensure exact constraint
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
