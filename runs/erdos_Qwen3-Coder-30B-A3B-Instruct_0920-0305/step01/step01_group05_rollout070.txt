import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        h = h * (1.0 / current_sum)
    
    # Clip to [0, 1] to enforce constraint
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def binary_local_search(n, time_limit):
    """Perform local search on binary vectors with exactly n/2 ones"""
    dx = 2.0 / n
    half = n // 2
    
    def evaluate_binary(bits):
        h = bits.astype(float)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    start = time.time()
    best_bits = None
    best_val = float('inf')
    
    while time.time() - start < time_limit:
        # Generate random binary vector with exactly half ones
        bits = np.zeros(n, dtype=int)
        bits[:half] = 1
        np.random.shuffle(bits)
        
        curr_val = evaluate_binary(bits)
        improved = True
        
        # Local descent by swapping
        while improved and time.time() - start < time_limit:
            improved = False
            for _ in range(2 * n):  # Limited attempts
                ones_idx = np.flatnonzero(bits)
                zeros_idx = np.flatnonzero(1 - bits)
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                    
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                
                # Try the swap
                bits[i], bits[j] = bits[j], bits[i]
                new_val = evaluate_binary(bits)
                
                if new_val < curr_val - 1e-12:
                    curr_val = new_val
                    improved = True
                else:
                    # Revert the swap
                    bits[i], bits[j] = bits[j], bits[i]
            
            # If no improvement after many attempts, restart
            if not improved:
                break
                
        if curr_val < best_val:
            best_val = curr_val
            best_bits = bits.copy()
            
        if best_val < 0.379:  # Early exit if target achieved
            break
            
    return best_bits, best_val

            break

            # Ensure sum constraint

                h = h * (1.0 / current_sum)
            
            h = np.clip(h, 0, 1)
            
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n)

        # Run DE with higher popsize and generation count
        try:
            res = differential_evolution(
                objective,
                bounds,
                maxiter=int(5000 / (n**0.5)),
                popsize=30,
                mutation=(0.5, 1),
                recombination=0.9,
                atol=1e-6,
                tol=1e-6,
                seed=seed,
                callback=lambda x, convergence: None
            )
            if res.success:

                    h = h * (1.0 / current_sum)
                
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h
                    best_n = n
        except Exception:

        # Phase 2: Binary search
        try:
            binary_bits, binary_c5 = binary_local_search(n, time_limit - 5)
            if binary_c5 < best_c5:
                best_c5 = binary_c5
                best_h = binary_bits.astype(float)
                best_n = n
        except Exception:
            pass # Continue

        if best_c5 < 0.379:
            break

    # Final verification and refinement with gradient method
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 20:
            try:
                # Refine using scipy.optimize.minimize
