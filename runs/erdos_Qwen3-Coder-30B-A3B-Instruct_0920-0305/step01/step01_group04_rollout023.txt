def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h_values)
    c5 = max_correlation * dx
    return c5

def project_onto_feasible(h):
    """Project h onto the set of vectors where sum = n/2 and 0 <= h[i] <= 1."""
    h = np.array(h)
    n = len(h)
    target_sum = n / 2.0
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - target_sum) < 1e-10:
        return h_clipped
    diff = target_sum - current_sum
    # Water-filling approach: adjust h by a uniform shift of diff/n
    if diff > 0:
        # Increase values
        free_indices = np.arange(n)[h_clipped < 1]
        if len(free_indices) == 0:
            return h_clipped
        # Calculate how much to increase
        # Sorted by value increasing for greedy pick
        h_sorted = h_clipped[free_indices]
        sorted_idx = np.argsort(h_sorted)
        accum_sum = 0.0
        for i in sorted_idx:
            val = h_sorted[i]
            needed = 1.0 - val
            if accum_sum + needed > diff:
                delta = diff - accum_sum
                h_clipped[free_indices[i]] += delta
                break
            h_clipped[free_indices[i]] = 1.0
            accum_sum += needed
    elif diff < 0:
        # Decrease values
        free_indices = np.arange(n)[h_clipped > 0]
        if len(free_indices) == 0:
            return h_clipped
        # Sorted by value decreasing for greedy pick
        h_sorted = h_clipped[free_indices]
        sorted_idx = np.argsort(h_sorted)[::-1]
        accum_sum = 0.0
        for i in sorted_idx:
            val = h_sorted[i]
            needed = val
            if accum_sum + needed > -diff:
                delta = -diff - accum_sum
                h_clipped[free_indices[i]] -= delta
                break
            h_clipped[free_indices[i]] = 0.0
            accum_sum += needed
    return np.clip(h_clipped, 0, 1)

def corr_eval(h_values):
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    return corr

def make_h_binary(h, n):
    """Convert float vector h to a binary vector closest to it, maintaining sum = n/2."""
    h = np.asarray(h)
    n_points = len(h)
    target = n_points / 2
    # Get sum of values
    current_sum = np.sum(h)
    # Find indices to change
    sorted_indices = np.argsort(h)[::-1]  # descending first
    # Try rounding to nearest integer with sum = n/2
    rounded = np.round(h)
    diff = target - np.sum(rounded)
    if diff > 0:
        # Need to add more 1s
        add_indices = sorted_indices[:int(diff)]
        for i in add_indices:
            rounded[i] = 1.0
    elif diff < 0:
        # Need to remove some 1s
        drop_indices = sorted_indices[int(-diff):]
        for i in drop_indices:
            rounded[i] = 0.0
    # Clip to 0-1 range and re-normalize if needed to maintain sum = n/2
    rounded = np.clip(rounded, 0, 1)
    if abs(np.sum(rounded) - target) > 1e-6:
        # Adjust
        current_sum = np.sum(rounded)
        delta = target - current_sum
        # Add delta to values that still have space
        if delta > 0:
            for i in range(n_points):
                if rounded[i] < 1.0 and delta > 0:
                    add = min(1.0 - rounded[i], delta)
                    rounded[i] += add
                    delta -= add
        else:
            for i in range(n_points):
                if rounded[i] > 0.0 and delta < 0:
                    sub = min(rounded[i], -delta)
                    rounded[i] -= sub
                    delta += sub
    return rounded.tolist()

def nearest_binary_h(h_vals, n):
    """Heuristic: round to binary while maintaining sum."""
    h = np.array(h_vals)
    # Ensure it sums to n/2
    h_scaled = h * n / np.sum(h)
    # Round to nearest ints near half, sum exact to n/2
    rounded = np.round(h_scaled)
    diff = n/2 - np.sum(rounded)
    if diff > 0:
        # Add 1s to some
        indices = np.argsort(h_scaled)
        for i in range(int(diff)):
            rounded[indices[-(i+1)]] = 1.0
    elif diff < 0:
        indices = np.argsort(h_scaled)[::-1]
        for i in range(int(-diff)):
            rounded[indices[i]] = 0.0
    return rounded.tolist()

def constrain_h(h, n_points=None):
    if n_points is None:
        n_points = len(h)
    h = np.clip(np.array(h), 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-6:
        h = h * (n_points/2) / h_sum
    return list(h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Set up problem
    n_points = 200  # Using larger n to possibly find better solutions
    dx = 2.0 / n_points

    # Try to solve exactly via MILP if feasible

    C5_final = float('inf')

    if abs(np.sum(h_start) - n_points/2) > 1e-6:
        h_start = h_start * (n_points/2) / np.sum(h_start)

    # Phase 1: Sub-gradient descent with simulated annealing
    best_h = h_start.copy()
    best_c5 = evaluate_h(best_h)
    best_h_list = list(best_h)

    # Multi-start
    seeds = [seed, seed+1, seed+2, seed+3]
    start_times = [time.time() for _ in range(len(seeds))]
    for idx, seed_val in enumerate(seeds):
        np.random.seed(seed_val)
        # Starting point: use nearest binary version of initial one
        h_init = np.asarray(initial_h_values[:n_points])
        # Ensure sum = n/2
        h_init = h_init * (n_points/2) / np.sum(h_init)

        # Random start for exploration
        if idx == 1:
            # Random feasible point
            h_init = np.random.rand(n_points)
            h_init = h_init * (n_points/2) / np.sum(h_init)
        elif idx == 2:
            # Thue-Morse-like pattern
            pattern = []
            for i in range(n_points):
                count = bin(i).count("1")
                pattern.append(1 if count % 2 == 0 else 0)
            h_init = np.array(pattern) * 1.0
            h_init = h_init * (n_points/2) / np.sum(h_init)

        # Start optimization
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        if abs(np.sum(h) - n_points/2) > 1e-6:
            h = h * (n_points/2) / np.sum(h)

        h_prev = h.copy()
        iter_count = 0
        max_iter = 30000
        temp = 1.0
        temp_decay = 0.9998
        gamma = 0.99  # Grad shrink rate - 0.995 was too slow

        total_elapsed = time.time() - start_time
        if total_elapsed > budget_s - 2:
            break

        while iter_count < max_iter:
            total_elapsed = time.time() - start_time
            if total_elapsed > budget_s - 2:
                break

            corr = corr_eval(h)
            max_shift = np.argmax(corr)
            max_val = corr[max_shift]

            # Compute gradient for shift
            # Needed: d = max_shift - (n-1)
            # Or more directly get shift from full correlation indices
            # We have shift from 0 up to 2*n-2 in full mode
            # shift == 0 -> not used, so we shift and add n-1
            shift = max_shift - (n_points - 1)

            grad = np.zeros_like(h)
            # Actual indices involved in shift
            for i in range(n_points):
                i_plus = (i + shift) % n_points
                i_minus = (i - shift) % n_points
                grad[i] = 1 - h[i_plus] - h[i_minus]

            # Perform gradient step
            alpha = 0.01 * gamma**(iter_count//100)
            h_new = h - alpha * grad

            # Project h_new back to feasible region
            h_new = project_onto_feasible(h_new)

            # Simple acceptance test (Simulated Annealing)
            old_val = corr_eval(h)[max_shift]
            new_val = corr_eval(h_new)[max_shift]
            delta_E = old_val - new_val  # Since we are minimizing

            # Metropolis criterion
            if delta_E > 0 or np.random.rand() < np.exp(-delta_E / temp):
                h = h_new
                if evaluate_h(h) < best_c5:
                    best_h = h.copy()
                    best_c5 = evaluate_h(h)
                    best_h_list = list(h)

            iter_count += 1
            temp *= temp_decay

            if iter_count % 1000 == 0:
                # Occasionally break if timeout is imminent
                total_elapsed = time.time() - start_time
                if total_elapsed > budget_s - 5:
                    break

    # Final polishing: hill climbing
    h_polished = np.array(best_h_list).copy()

    corr = corr_eval(h_polished)
    max_shift = np.argmax(corr)
    best_mismatch = corr[max_shift]
    corr_full = corr.copy()

    for iter_hill in range(10000):
        total_elapsed = time.time() - start_time
        if total_elapsed > budget_s - 5:
            break

        # Try all swaps (this can be rewritten in O(n) by computing diff)
        improvement = False
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    continue
                if h_polished[i] == 1.0 and h_polished[j] == 0.0:
                    # Try swap
                    h_temp = h_polished.copy()
                    h_temp[i] = 0
                    h_temp[j] = 1
                    c_temp = evaluate_h(h_temp)
                    if c_temp < best_c5:
                        h_polished = h_temp
                        best_c5 = c_temp
                        improvement = True
        if not improvement:
            break

    # Convert to binary approximation
    full_h = list(project_onto_feasible(h_polished))
    binary_h = make_h_binary(full_h, n_points)
    final_c5 = evaluate_h(binary_h)
    # Final check: if we improved even more, update
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h_list = binary_h

    answer = list(np.array(best_h_list).astype(float))
    return (answer, best_c5, n_points)
