import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def interpolate_h(h_old, n_new):
    """Interpolate h from old grid to new finer grid"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0, 1)
    # Adjust to maintain sum constraint
    h_new = h_new * (n_new / 2) / np.sum(h_new)
    return h_new

def build_shift_masks(n):
    """Precompute masks for shifts"""
    masks_fwd = {}
    masks_bwd = {}
    for s in range(-(n-1), n):
        masks_fwd[s] = np.arange(max(0, -s), min(n, n-s))
        masks_bwd[s] = np.arange(max(0, s), min(n, n+s))
    return masks_fwd, masks_bwd

def compute_gradient_gd(h, s, masks_fwd, masks_bwd):
    """Compute gradient of g_d^s at h"""
    grad = np.zeros_like(h)
    if s > 0:
        grad[masks_bwd[s]] += (h[masks_bwd[s]] + h[masks_bwd[s] + s])
    elif s < 0:
        grad[masks_fwd[s]] += (h[masks_fwd[s]] + h[masks_fwd[s] - s])
    else:
        grad += 2 * h
    return grad

def solve_ccp_subproblem(h_prev, n, masks_fwd, masks_bwd, dx):
    """Solve the CCP subproblem for given h_prev"""
    # Variables
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Constraints
    constraints = []
    
    # Supply h_prev and store for GD computation
    h_prev_np = np.array(h_prev)
    
    # Add constraints for each shift s
    for s in range(-(n-1), n):
        # Compute g_s^c(h) = sum(h) + 0.5 * (sum(h[mask_fwd[s]]^2) + sum(h[mask_bwd[s]]^2))
        g_s_c = cp.sum(h) + 0.5 * cp.sum_squares(h[masks_fwd[s]]) + 0.5 * cp.sum_squares(h[masks_bwd[s]])
                
        # Compute gradient of g_s^d at h_prev
        gd_grad = compute_gradient_gd(h_prev_np, s, masks_fwd, masks_bwd)
        
        # Compute g_s^d(h_prev)
        g_s_d_at_prev = 0.5 * np.sum((h_prev_np + np.roll(h_prev_np, s)) ** 2)
        
        # Compute affine linearization of g_s^d at h_prev
        affine = g_s_d_at_prev + gd_grad @ (h - h_prev_np)
        
        # Constraint: tau >= g_s^c - affine
        constraints.append(tau >= g_s_c - affine)
    
    # Box constraints
    constraints.extend([h >= 0, h <= 1])
    
    # Integral constraint
    constraints.append(cp.sum(h) == n / 2)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        if prob.status == cp.OPTIMAL:
            return h.value
        else:
            # print(f"Solver status: {prob.status}")
            return h_prev  # Return previous if infeasible
    except:
        return h_prev  # Return previous if solver error

def coordinate_descent(h, n, dx, max_iters=5):
    """Apply coordinate descent to refine h"""
    # For each coordinate, find optimal value that minimizes max overlap
    # We'll do a full sweep over all coordinates in a random order
    
    for sweep in range(max_iters):
        indices = list(range(n))
        np.random.shuffle(indices)
        
        improved = False
        for i in indices:
            h_old = h[i].copy()
            current_sum = np.sum(h)
            # We need to update the integral constraint when we change h[i]
            if not (0 <= h_old <= 1):
                continue
            
            # Reconstruct h from scratch to see what happens with candidate value
            # Candidate values: 0, 1, or nearby values from previous h values that are close to 0 or 1
            candidates = [0, 1]  # We sample just the boundary values
            
            # Add some nearby values to see if we get better
            for j in [max(0, i-1), min(n-1, i+1)]:
                if 0 <= h[j] <= 1:
                    val1 = max(0, h[j] - 0.1)
                    val2 = min(1, h[j] + 0.1)
                    candidates.extend([val1, val2])
                    
            candidates = set([max(0, min(1, c)) for c in candidates])
            
            best_cand = h_old
            best_val = np.inf
            for c in candidates:
                if c == h_old:
                    continue
                
                # Create new candidate
                h_new = h.copy()
                old_sum = np.sum(h_new)
                delta_i = c - h_old
                h_new[i] = c
                # Compensate by changing another coordinate which doesn't hit boundary
                idxs_to_modify = [j for j in range(n) if j != i and 0 < h_new[j] < 1]
                if len(idxs_to_modify) > 0:
                    j = idxs_to_modify[0]
                    h_new[j] = max(0, min(1, h_new[j] - delta_i))
                else:
                    # Fallback: adjust last one to keep integral constraint
                    # Just assume we keep the sum the same in a way that respects boundaries
                    # In a real refinement, we would do this much more carefully
                    h_new[-1] = max(0, min(1, h_new[-1] - delta_i))
                
                # Evaluate new overlaps
                corr = np.correlate(h_new, 1 - h_new, mode="full")
                test_c5 = np.max(corr) * dx;
                if test_c5 < best_val:
                    best_val = test_c5
                    best_cand = c
                    
            h[i] = best_cand
            improved = True
            
        if not improved:
            break
            
    return h
