import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project vector y onto the capped simplex with sum = target_sum"""
    # y is the input vector
    # target_sum is the desired sum (n/2 for our problem)
    # Use binary search to find the correct lambda
	
    # Get a rough estimate of the threshold
    l = np.min(y)
    u = np.max(y)
    
    # binary search for lambda
    for _ in range(50):
        lambda_ = (l + u) / 2.0
        # Compute projection with this lambda
        x_new = np.clip(y - lambda_, 0, 1)
        sum_new = np.sum(x_new)
        
        if sum_new > target_sum:
            l = lambda_
        else:
            u = lambda_
    
    lambda_ = (l + u) / 2.0
    x = np.clip(y - lambda_, 0, 1)
    # Rescale to match target_sum
    if np.sum(x) > 0:
        x = x * target_sum / np.sum(x)
    return x

def compute_gradient(h, alpha, corr_values):
    """
    Computes the gradient of the log-sum-exp surrogate at h:
    ∇F_α(h) = (1/n) * Σ_s w_s * ∇C_s
    where Σ_s w_s = 1, and ∇C_s = (1 - h_{i+s} - h_{i-s})
    """
    n = len(h)
    d = len(corr_values)
    
    # Make sure the corr_values match the shifts
    # corr_values has length 2*n - 1, corresponding to shifts from -(n-1) to (n-1)
    max_shift = n - 1
    # softmax weights
    c_shifts = corr_values
    c_shifts = np.array(c_shifts)
    max_c = np.max(c_shifts)
    exp_c = np.exp(alpha * (c_shifts - max_c))
    w = exp_c / np.sum(exp_c)
    
    # Initialize gradient
    grad = np.ones(n)
    
    # Compute the weighted sum: - Σ_s w_s (h_{i+s} + h_{i-s})
    for i in range(n):
        grad[i] -= 0.5 * (np.sum(w * np.roll(h, -i)) + np.sum(w * np.roll(h, i)))
    
    return grad

def line_search(h, grad, alpha, corr_values, eta, max_backtrack=5):
    """Perform line search to find suitable step size eta to decrease surrogate"""
    # Current surrogate
    current_surrogate = np.mean(np.exp(alpha * np.array(corr_values)))
    current_surrogate = np.log(current_surrogate) / alpha
    
    # Backtracking line search
    for _ in range(max_backtrack):
        h_new = h - eta * grad
        h_new = project_to_simplex(h_new, len(h)/2)
        
        # Evaluate surrogate at new point (just the correlation values)
        new_corr = correlate(h_new, 1 - h_new, mode="full")
        new_surrogate = np.log(np.sum(np.exp(alpha * (new_corr - np.max(new_corr))))) / alpha
        
        if new_surrogate <= current_surrogate:
            return h_new, eta
        eta /= 2.0
    return h, eta

def projected_gradient_descent(h0, n, alpha_start=5.0, alpha_factor=1.03, max_iter=5000, 
                              eta0=0.5, time_limit=None):
    
    # States
    h = h0.copy()
    alpha = alpha_start
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    for i in range(max_iter):
        if time_limit and (time.time() - time_limit) > 5:
            break
            
        # Compute correlation values
        corr = correlate(h, 1 - h, mode="full")
        corr_values = corr # full mode output
        # Compute gradient
        grad = compute_gradient(h, alpha, corr_values)
        # Update step size
        eta = eta0 / np.sqrt(i + 1)
        # Update h
        h = h - eta * grad
        h = project_to_simplex(h, n/2)
        
        # Evaluate objective
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Update alpha (annealing)
        if i % 200 == 0 and alpha < 200:
            alpha = alpha * alpha_factor
            
    return best_h, best_c5
