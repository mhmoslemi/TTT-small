import numpy as np
from scipy.optimize import minimize
from scipy import fftpack
import time
from cvxpy import *

def evaluate_h(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def compute_correlation(h):
    # Compute convolution h * (1 - h) using FFT for speed
    h = np.array(h)
    h_padded = np.pad(h, (0, len(h) - 1), mode='constant')
    one_h_padded = np.pad(1 - h, (0, len(h) - 1), mode='constant')
    corr = fftpack.fftconvolve(h_padded, one_h_padded, mode='full')
    corr = corr[len(h) - 1:]
    return corr

def project_to_feasible(h, target_sum):
    # Project vector h onto {x : sum(x) = target_sum, 0 <= x_i <= 1}
    # Using water-filling (sort, adjust, clip)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-6:
        return h
    # Bring to target_sum by uniform adjustment
    if s <= target_sum:
        # Need to add (target_sum - s) to underfilled elements
        add = (target_sum - s) / len(h)
        h = h + add
        return np.clip(h, 0, 1)
    else:
        # Need to subtract (s - target_sum) from overfilled elements
        diff = s - target_sum
        h = h - diff / len(h)
        return np.clip(h, 0, 1)

def generate_binary_heuristic(seed, n_points, target_sum):
    np.random.seed(seed)
    h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    h[indices] = 1.0
    return h

def get_thue_morse_vector(n):
    # Generate Thue-Morse sequence up to n
    tm = [0]
    while len(tm) < n:
        tm += [1 - x for x in tm]
    return np.array(tm[:n])

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 50000
    
    # Step 1: Try exact integer linear programming
    best_h = None
    best_c5 = float('inf')
    
    # Try MILP with a smaller n if the memory limits would be exceeded
    try:
        # Simplified call to MILP might be included here if necessary.
        # For now, we'll proceed with the heuristic phase.
        pass
    except:
        pass

    # Step 2: Heuristic phase
    best_h = None
    best_c5 = float('inf')

    # Multi-start strategy
    seeds = [seed + i for i in range(3)]
    for i, s in enumerate(seeds):
        if time.time() - start_time > budget_s - 5:
            break
        
        np.random.seed(s)
        h = None

        # Different initializations
        if i == 0:
            # Initialize from initial_h_values if available
            try:
                h = np.array(initial_h_values)
            except:
                h = generate_binary_heuristic(s, n_points, target_sum)
        elif i == 1:
            # Random binary initialization
            h = generate_binary_heuristic(s, n_points, target_sum)
        else:
            # Thue-Morse pattern (non-binary but good heuristic)
            tm = get_thue_morse_vector(n_points)
            tm = np.array(tm, dtype=float)
            # Normalize to sum=target_sum
            tm_scale = target_sum / np.sum(tm)
            h = tm * tm_scale

        # Ensure h meets constraints
        if np.sum(h) != target_sum:
            h = project_to_feasible(h, target_sum)

        # Iterative sub-gradient descent with cooling
        iter_count = 0
        temp = 10.0
        cooling_rate = 0.995
        alpha = 0.5
        
        while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
            iter_count += 1
            corr = compute_correlation(h)
            m_d = corr  # mismatches for all shifts (shifts 0 to 2n-2)
            d_star = np.argmax(m_d)
            
            # Compute subgradient
            g = np.zeros(n_points)
            for k in range(n_points):
                k_plus = (k + d_star) % n_points
                k_minus = (k - d_star) % n_points
                g[k] = 1 - h[k_plus] - h[k_minus]
            
            h_new = h - alpha * g
            h_new = project_to_feasible(h_new, target_sum)
            
            # Accept move w.p. exp(-(ΔM)/T)
            old_m = m_d[d_star]
            new_corr = compute_correlation(h_new)
            new_m = new_corr[d_star]
            
            if new_m < old_m or np.random.uniform() < np.exp(-(new_m - old_m) / (temp + 1e-8)):
                h = h_new
            
            # Update temp
            temp = max(0.001, temp * cooling_rate)
            
            # Update best solution
            if new_m < old_m:
                corr = compute_correlation(h)
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        
        # Only effect h from the current iteration if no better was found yet
        if best_h is None or best_c5 > evaluate_h(h):
            new_corr = compute_correlation(h)
            new_c5 = np.max(new_corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    # Final polishing
    if best_h is not None:
        # Round to binary and apply local search
        sorted_indices = np.argsort(best_h)[::-1]
        h_rounded = np.zeros_like(best_h)
        for i in range(int(target_sum)):
            h_rounded[sorted_indices[i]] = 1.0

        # Refine by swapping
        corr = compute_correlation(h_rounded)
        old_max_m = np.max(corr)
        improved = True
        while improved and (time.time() - start_time) < budget_s - 5:
            improved = False
            for i in range(n_points):
                if h_rounded[i] != 1:
                    continue
                for j in range(n_points):
                    if h_rounded[j] != 0:
                        continue
                    h_trial = h_rounded.copy()
                    h_trial[i] = 0.0
                    h_trial[j] = 1.0
                    corr_trial = compute_correlation(h_trial)
                    new_max_m = np.max(corr_trial)
                    if new_max_m < old_max_m:
                        old_max_m = new_max_m
                        h_rounded = h_trial
                        improved = True
                        break
                if improved:
                    break

        corr = compute_correlation(h_rounded)
        final_c5 = np.max(corr) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_rounded
    
    if best_h is None:
        best_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        best_c5 = evaluate_h(best_h)

    # Final evaluation
    final_h = best_h.tolist()
    final_c5 = evaluate_h(final_h)
    
    return final_h, final_c5, n_points
