def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure sum equals n/2 exactly for integral constraint
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n / 2.0) / h_sum
    
    # Ensure values within [0,1] after scaling
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
