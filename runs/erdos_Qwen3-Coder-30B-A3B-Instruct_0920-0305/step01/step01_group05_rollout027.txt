def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def clip_and_rescale(x, target_sum):
    """Clip x to [0,1] and rescale to sum to target_sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if current_sum > 0:
        x = x * (target_sum / current_sum)
    return x

def project_to_sum(x, target_sum):
    """Project x to the hyperplane where sum = target_sum using bisection"""
    x = np.array(x)
    x = np.clip(x, 0, 1)

    if abs(current_sum - target_sum) < 1e-10:
        return x
    
    # Use bisection to find alpha such that sum(clip(x + alpha * 1)) = target_sum
    low, high = -2.0, 2.0
    for _ in range(30):  # Bisection iterations
        mid = (low + high) / 2.0
        candidate = x + mid
        candidate = np.clip(candidate, 0, 1)
        new_sum = np.sum(candidate)
        if new_sum < target_sum:
            high = mid
        else:
            low = mid
    alpha = (low + high) / 2.0
    result = np.clip(x + alpha, 0, 1)
    return result
