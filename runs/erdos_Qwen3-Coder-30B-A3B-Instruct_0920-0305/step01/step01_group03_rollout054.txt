import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """
    Project h onto the capped simplex: sum(h) = target_sum, 0 <= h <= 1
    Using water-filling approach
    """
    h = np.array(h, dtype=float)
    n = len(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Find initial sum
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Water-filling approach
    # Sort in descending order to find the cutpoint
    sorted_indices = np.argsort(h)[::-1]
    h_sorted = h[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    prefix_sum = np.concatenate([[0], cumsum[:-1]])
    
    # Find the cutpoint
    # We want to find k such that:
    # sum of max(0, h_sorted[i] - t) = target_sum for some t
    # where t is the water level
    # This simplifies to: sum of h_sorted[0:k] - k*t = target_sum
    # If k is the last index for which h_sorted[k] > t, we have:
    # sum(h_sorted[0:k+1]) - (k+1)*t = target_sum
    # So t = [sum(h_sorted[0:k+1]) - target_sum] / (k+1)
    
    for i in range(n):
        if i == n - 1 or h_sorted[i] > h_sorted[i+1]:
            # Check the water level that makes this index active
            j = i + 1  # Number of indices to consider
            # To find how much can be filled to current level
            old_sum = prefix_sum[i]  # Sum from 0 to i-1
            temp_sum = old_sum + h_sorted[i]  # Sum up to i
            # If we try to fill up to level h_sorted[i]
            # remaining filling = target - old_sum
            # but try to spread across j elements
            if old_sum >= target_sum:
                level = 0.0
            else:
                level = (target_sum - old_sum) / j
                
            # We can satisfy the requirement if level <= h_sorted[i]
            if level <= h_sorted[i]:
                # Done: levels for indices 0..i-1 are at new_level
                # but check that other elements remain 1 or 0 if needed
                break
    
    # At this point we have found the level
    # Set all elements greater than or equal to level to level
    # (But only such that final sum is exactly target_sum)
    level = (target_sum - prefix_sum[i]) / (n - i)
    
    # Actually, let's do a cleaner approach
    # We run a binary search for the correct tau
    # h' = min(1, max(0, h + tau))
    # sum(h') = target_sum
    # Let's compute sum of clipped values.
    def evaluate_sum(tau):
        proj = np.clip(h + tau, 0, 1)
        return np.sum(proj)
    
    # Binary search for the right tau
    lower, upper = -h.min(), 1 - h.max()
    tol = 1e-12
    
    while upper - lower > tol:
        mid = (lower + upper) / 2
        actual_sum = evaluate_sum(mid)
        if actual_sum > target_sum:
            upper = mid
        else:
            lower = mid
    
    final_h = np.clip(h + (lower + upper) / 2, 0, 1)
    # Renormalize to exact target sum if needed
    final_h = final_h * target_sum / np.sum(final_h)
    
    # Return back in original order
    final_h_full = np.zeros_like(h)
    final_h_full[sorted_indices] = final_h
    
    return final_h_full

def symmetrize_and_project(h):
    """
    Symmetrize h with respect to midpoint x=1, then project to simplex.
    """
    n = len(h)
    h_sym = np.zeros_like(h)
    
    # Make symmetric
    for i in range(n):
        # i and n-1-i are symmetric indices around center
        j = n - 1 - i
        if i <= j:
            h_sym[i] = (h[i] + h[j]) / 2
        else:
            h_sym[i] = h_sym[j]
    
    # Project to capped simplex
    h_proj = project_to_capped_simplex(h_sym, 1.0)
    
    # Ensure mass constraint exactly
    h_proj = h_proj * (1.0 / np.sum(h_proj))
    
    return h_proj


def logsumexp_stable(x, axis=None):
    """Numerically stable logsumexp"""
    max_x = np.max(x, axis=axis, keepdims=True)
    return max_x + np.log(np.sum(np.exp(x - max_x), axis=axis, keepdims=True))


def compute_soft_minimum(h, lambda_val):
    """
    Compute the smooth approximation Φ_λ(h) = (1/λ) log Σ exp(-λ R(s))
    Returns the value and gradient of Φ_λ.
    """
    n = len(h)
    # Compute full correlations R(s) = h * h (full mode)
    R = correlate(h, h, mode="full")
    
    # Make sure the minimum value is not sum of all
    max_R = np.max(R)
    R_centered = R - max_R
    
    # Soft weights
    exp_neg_lambda_R = np.exp(-lambda_val * R_centered)
    Z = np.sum(exp_neg_lambda_R)
    w = exp_neg_lambda_R / Z
    
    # Gradient of Φ_λ w.r.t. h
    g = np.zeros_like(h)
    # For each shift s
    # w_s = exp(-lambda * R_s) / Z
    # For each index i:
    # ∂Φ_λ/∂h_i = -sum_s w_s * (h_{i+s} + h_{i-s})
    # But h_{i+s} and h_{i-s} are out of bounds - treat as 0
    
    # We compute this by convolution with the symmetric kernel:
    # K_s = w_s if s >= 0
    # K_s = w_{-s} if s < 0
    # But we want the discrete version: g_j = sum_s w_s * (h_{j+s} + h_{j-s})
    
    # Full kernel W = [w_{-n+1}, ..., w_{-1}, w_0, w_1, ..., w_{n-1}]
    # w is original size 2*n-1, with central index n-1 matching to s=0
    
    # Compute the gradient by slice as no efficient convolution here
    # To go through each element and accumulate over all shifts
    
    # Efficient slice way:
    # g = w0 * (h + h_{shifted}) = w0 * (h + roll(h, 1)) + w1 * (h + roll(h, -1)) + ...
    
    for s in range(-n+1, n):
        if s >= 0:
            # Shift h right by s
            h_shifted = np.roll(h, -s)
        else:
            # Shift h left by |s|
            h_shifted = np.roll(h, s)
        
        # Zero padding (no boundary handling since we add 0 for out-of-bounds)
        h_shifted = np.pad(h_shifted, (max(0, s), max(0, -s)), mode='constant')
        # For the midpoint, adjust the boundaries
        # Actually simpler to have extra padding:
        g += w[s + n - 1] * (h_shifted[:n] + h_shifted[n:])
        
    # Denormalize gradient: weights already normalized, but we want to mitigate the effect from
    # summing across the expanded signals. Let's divide by 2 to regularize a bit.
    # Simpler to just divide by the number of valid indices (n in any direction for n) 
    # Related to n not 2n (we're doing the full mode)
    # Or think of it as a convolution: 
    # g_j = sum_{s} w_s * (h_{j+s} + h_{j-s}) 
    # ...
    # Clean approach:
    # g = (w + w rev) * h  (convolution full mode)
    # w is size 2*n - 1 from -n+1 to n-1
    # Let's recompute:
    
    # Simpler construction
    # Now reverse shift for j-s
    g = np.zeros_like(h)
    for s in range(1 - n, n):
        if s >= 0:
            s_1 = s
            s_2 = 0
            h1 = np.roll(h, -s)
            h2 = h
        else:
            s_1 = 0
            s_2 = -s
            h1 = h
            h2 = np.roll(h, s)
        # We want: w_s * (h_{j+s_1} + h_{j+s_2})  
        # Simplify where valid:  
        shifted_h1 = np.roll(h, -s) 
        shifted_h2 = np.roll(h, s) 
        # Get the part of the full convolved signal that is the valid part
        # If full mode, we have 2n-1 output indices 
        s_idx = s + (n-1)  # mapping s to w[0,...,2n-2]
        w_s = w[s_idx]
        #g += w_s * (shifted_h1 + shifted_h2)
    
    # But using broadcasting again:
    # This way:
    # Create w vector properly expanded to the same length as convolution result (2*n - 1)
    w_full = np.zeros(2 * n - 1)
    w_full[...] = w  # should fit properly
    # Then, we make a convolution and accumulate
    
    # Recomputing properly with correct cross terms:
    # Two convolutions needed 
    # Constraint: convolution of vector of size n with another of size n gives output size 2*n - 1  
    # But since neutral element equivalent to 0, we should use the symmetry property
    shifted_h = h.copy()
    # Forms w*r, r being full (via matrix)
    # Use
    conv1 = correlate(w_full, shifted_h, "full") # Full mode (2*n - 1 outputs)  fullConv
    # Form zh is h 
    # But actually simpler to compute the shift manually
    # What we actually need:
    # ∂Φ_λ / ∂h_j = - \sum_s w_s (h_{j+s} + h_{j-s})
    # Reformulation:
    # h0 = (sum over s) w_s h_{j+s}
    # h1 = (sum over s) w_s h_{j-s} 
    # g = h0 + h1 (gradient vector)
    
    # Let's approach via a Numeric formulation for clarity.
    
    # Compute Φ_λ from the logsumexp and sum of W_i * R_i
    # Done via single pass:
    Phi_lambda = np.log(Z) / lambda_val
    
    # Compute gradient.
    # We will construct it via element-wise multiplication in a loop.
    g = np.zeros_like(h)
    # compute W * [ h_j+s + h j-s ]  
    # which is sum over s of w_s * (h_{s} + h_{-s}) at position j.
    
    # We will rebuild more slowly but honestly with understanding
    # i in h 
    for i in range(n):
        total = 0.0
        for s in range(1 - n + 1, n):  # full range E.g. [-n+1, ..., n-1]
            s_idx = i + s
            # h_{i+s}
            if 0 <= s_idx <= n-1:
                hsi = h[s_idx]
            else:
                hsi = 0.0
            # h_{i-s}
            s_idx2 = i - s
            if 0 <= s_idx2 <= n-1:
                hs2 = h[s_idx2]
            else:
                hs2 = 0.0
            total += w[s + n - 1] * (hsi + hs2)
        g[i] = -total
    
    return Phi_lambda, g


def backtracking_line_search(h, g, lambda_val, step_size, max_iter=5):
    """
    Perform backtracking line search for step size α
    """
    alpha = step_size
    phi_prev = compute_soft_minimum(h, lambda_val)[0]
    tol = 1e-12
    for _ in range(max_iter):
        # Compute new h'
        h_new = h - alpha * g
        
        # Project to capped simplex
        h_proj = project_to_capped_simplex(h_new, 1.0)
        
        # Compute Φ_λ(h')
        phi_new = compute_soft_minimum(h_proj, lambda_val)[0]
        
        # Accept if acceptable decrease
        if phi_new <= phi_prev + tol:  # Simple armijo line search for decrease
            return h_proj, alpha
        else:
            alpha *= 0.5  # Reduce by half
            
    return h_proj, alpha


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Rate of convergence estimate
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    time_budget = budget_s * 0.9  # Leave time for finish
    
    # Phase settings for annealing
    phases = [
        {'lambda': [0.5, 2.0], 'alpha': [0.2, 0.5], 'iterations': [150, 200]},
        {'lambda': [5.0, 15.0], 'alpha': [0.05, 0.2], 'iterations': [200, 300]},
        {'lambda': [30.0, 60.0], 'alpha': [0.01, 0.05], 'iterations': [300, 400]}
    ]
    
    total_iterations = 0
    max_iterations = 10000
    
    # Multi-start:
    start_types = ['binary', 'random', 'upsampled', 'symmetric']
    
    def sample_binary_initial():
        p = np.zeros(n_points)
        p[:n_points//2] = 1
        np.random.shuffle(p)
        return project_to_capped_simplex(p, 1.0)
        
    def sample_random_initial():
        r = np.random.rand(n_points)
        return project_to_capped_simplex(r, 1.0)
        
    def sample_upsampled_initial():
        # Interpolate initial_h_values (42 samples) to n_points
        old = np.array(initial_h_values)  
        # Use scipy/interp
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 2, 42)
        x_new = np.linspace(0, 2, n_points)
        f = interp1d(x_old, old, kind='linear')  
        h_new = f(x_new)
        return project_to_capped_simplex(h_new, 1.0)
        
    for start_type in start_types:
        if time.time() - start_time > time_budget:
            break
            
        # Generate initial h
        if start_type == 'binary':
            h = sample_binary_initial()
        elif start_type == 'random':
            h = sample_random_initial()
        elif start_type == 'upsampled':
            h = sample_upsampled_initial()
        elif start_type == 'symmetric':
            h = sample_upsampled_initial()
            h = symmetrize_and_project(h)
            
        # Annealing loop
        lambda_schedule = [0.5, 2.0, 5.0, 15.0, 30.0, 60.0]
        lambda_idx = 0
        
        for phase in phases:
            if time.time() - start_time > time_budget:
                break
                
            lambda_min, lambda_max = phase['lambda']
            alpha_min, alpha_max = phase['alpha']
            iterations_min, iterations_max = phase['iterations']
            
            for i in range(iterations_min):  # Start with min iterations
                if time.time() - start_time > time_budget:
                    break
                    
                # Anneal lambda if needed
                if lambda_idx < len(lambda_schedule) - 1 and i == iterations_min // 2:
                    lambda_min = lambda_schedule[lambda_idx]
                    lambda_max = lambda_schedule[lambda_idx + 1]
                    lambda_idx += 1
                    
                lambda_val = lambda_min + (lambda_max - lambda_min) * min(i / 50, 1.0)
                alpha = alpha_min + (alpha_max - alpha_min) * (i / iterations_min)
                
                # Compute Phi_lambda and gradient
                phi_lambda, g = compute_soft_minimum(h, lambda_val)
                
                # Backtracking line search
                h, _ = backtracking_line_search(h, g, lambda_val, alpha)
                
                # Apply symmetry constraint if needed
                if start_type == 'symmetric':
                    h = symmetrize_and_project(h)
                
                # Compute actual raw C5 for checking
                c5 = evaluate_erdos_solution(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                total_iterations += 1
                
    # Final refinement with SLSQP on exact function
    if best_h is not None:
        def objective(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
            
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
        
        # Final refinement
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        
        if result.success:
            final_c5 = objective(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()
        else:
            # Use best_h if optimization fails
            pass
            
    # Final ensure projection and return best found
    if best_h is None:
        # Fallback to initial construction
        best_h = np.array(initial_h_values)
        best_h = project_to_capped_simplex(best_h, 1.0) 
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final normalization check
    best_h = best_h * (1.0 / np.sum(best_h)*dx)
    return (best_h.tolist(), best_c5, n_points)
