import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def build_shift_masks(n):
    """Build masks for forward and backward shifts"""
    masks_fwd = {}
    masks_bwd = {}
    for s in range(-(n-1), n):
        fwd_indices = []
        bwd_indices = []
        for i in range(n):
            if i + s < n and i + s >= 0:
                fwd_indices.append(i)
            if i - s >= 0 and i - s < n:
                bwd_indices.append(i)
        masks_fwd[s] = np.array(fwd_indices, dtype=int)
        masks_bwd[s] = np.array(bwd_indices, dtype=int)
    return masks_fwd, masks_bwd

def compute_gradient_gd(h, s, masks_fwd, masks_bwd):
    """Compute the gradient of g_s^d at h"""
    grad = np.zeros_like(h)
    # For each i, gradient component is (h_i + h_{i+s}) + (h_{i-s} + h_i) = 2*h_i + h_{i+s} + h_{i-s}
    # But we need to carefully handle boundary conditions
    if s >= 0:
        # Forward shift: h_{i+s}
        if len(masks_fwd[s]) > 0:
            grad[masks_fwd[s]] += h[masks_fwd[s] + s]
        # Backward shift: h_{i-s}
        if len(masks_bwd[-s]) > 0:
            grad[masks_bwd[-s]] += h[masks_bwd[-s] - s]
    else:
        # Backward shift: h_{i+s}
        if len(masks_fwd[-s]) > 0:
            grad[masks_fwd[-s]] += h[masks_fwd[-s] - s]
        # Forward shift: h_{i-s}
        if len(masks_bwd[abs(s)]) > 0:
            grad[masks_bwd[abs(s)]] += h[masks_bwd[abs(s)] + s]
    
    # Add 2*h_i
    grad += 2 * h
    return grad

def solve_ccp_subproblem(h_prev, n, masks_fwd, masks_bwd, dx):
    """Solve the convex-subproblem via CCP using CVXPY"""
    # Variables
    h = cp.Variable(n)
    tau = cp.Variable()

    # Objective
    objective = cp.Minimize(tau)

    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]

    # Add all shift constraints
    shift_range = range(-(n-1), n)
    for s in shift_range:
        # Compute the gradient of g_s^d at h_prev
        grad = compute_gradient_gd(h_prev, s, masks_fwd, masks_bwd)
        # The h_prev term: g_s^d(h_prev)
        gd_prev = 0.5 * np.sum((h_prev + np.roll(h_prev, s))**2)
        # The linearization term: grad.T @ (h - h_prev)
        linear_term = grad.T @ (h - h_prev)
        # The convex part g_s^c(h)
        gsc = np.sum(h) + 0.5 * np.sum(h[masks_fwd[s]]**2) + 0.5 * np.sum(h[masks_bwd[s]]**2)
        
        # Reformulate g_s^c(h) - [g_s^d(h_prev) + linear_term] <= tau
        # Add constraint: tau >= gsc - (gd_prev + linear_term) 
        constraints.append(tau >= gsc - (gd_prev + linear_term))

    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
        h_new = h.value
        if h_new is None:
            # Fall back to previous h if infeasible
            h_new = h_prev.copy()
    except:
        # If solver fails, just return old point
        h_new = h_prev.copy()
        
    return h_new

def coordinate_descent_polish(h, n, dx):

            # Find all overlap terms where h_i contributes
            # For each shift s, compute f_s(h) as h_i*(1 - h_{i+s}) + (h_{i+s}*(1-h_i) if applicable)
            # Total: h_i + h_{i+s} - h_i*h_{i+s} - h_{i+s}*h_i = h_i + h_{i+s} - 2*h_i*h_{i+s}
