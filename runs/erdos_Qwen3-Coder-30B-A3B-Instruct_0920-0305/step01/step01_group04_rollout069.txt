def full_correlation(h):
    """Compute full correlation of h with (1-h), return correlation vector and max value."""
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the valid shifts, excluding the trivial zero shift
    # The full correlation is of length 2*n - 1
    mid = len(corr) // 2
    valid_corr = corr[mid + 1:]  # Exclude zero shift and the symmetric part
    return corr, np.max(valid_corr)

    """Perform random hyperplane rounding of PSD matrix X."""

    g = np.random.randn(V.shape[1])
    
    # Project g onto the subspace and compute signs
    v = V @ g
    s = np.sign(v)
    # Handle exact zero cases
    for i in range(len(s)):
        if s[i] == 0:
            s[i] = np.random.choice([-1, 1])
    
    return s

def fix_weight_constraint(s, target_sum):
    """Adjust the weight (sum) of the binary vector s to be exactly target_sum."""
    current_sum = np.sum(s)
    if current_sum == target_sum:
        return s
    
    diff = int(current_sum - target_sum)
    if diff == 0:
        return s
    elif diff > 0:
        # Need to flip some 1s to 0s
        indices = np.where(s == 1)[0]
        flips = np.random.choice(indices, size=diff, replace=False)
        s[flips] = -1
    else:
        # Need to flip some -1s to 1s
        indices = np.where(s == -1)[0]
        flips = np.random.choice(indices, size=-diff, replace=False)
        s[flips] = 1
    return s

def upscale_pattern(h, n_points):
    """Upscale the pattern to n_points by concatenating with complement."""
    h_big = np.concatenate([h, 1 - h])
    # Trim or pad to exactly n_points
    if len(h_big) > n_points:
        h_big = h_big[:n_points]
    elif len(h_big) < n_points:
        h_big = np.pad(h_big, (0, n_points - len(h_big)), 'constant')
    return h_big

    # Create an index map of 1s and 0s
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n = len(h)
    
    iter_count = 0
    while improved and iter_count < max_iter:
        improved = False
        old_max = best_max
        
        # Try all pairs of (i,j) where h[i] == 1, h[j] == 0
        for i in ones:
            for j in zeros:
                # Copy h and swap positions i and j
                h_temp = best_h.copy()
                h_temp[i] = 0
                h_temp[j] = 1
                
                new_max = full_correlation(h_temp)[1]
                if new_max < best_max:
                    best_h = h_temp
                    best_max = new_max
                    improved = True
                    break
            if improved:
                break
        iter_count += 1
    
    return best_h, best_max
