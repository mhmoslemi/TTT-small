def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_sum_constraint(z, target_sum, box_min=0, box_max=1):
    """Project z onto the hyperplane sum = target_sum while staying within [box_min, box_max]"""
    z_clipped = np.clip(z, box_min, box_max)
    current_sum = np.sum(z_clipped)
    if abs(current_sum - target_sum) < 1e-10:
        return z_clipped
    
    # Simple iterative scaling/shift procedure
    alpha = (target_sum - current_sum) / len(z)
    z = z_clipped + alpha
    z = np.clip(z, box_min, box_max)
    return z
