import numpy as np

def project_simplex(y, target_sum):
    """
    Project vector y onto the capped simplex: sum = target_sum, 0 <= x <= 1
    """
    lmin, lmax = 0.0, 1.0
    # Use bisection to find lambda
    for _ in range(100):
        lamb = (lmin + lmax) / 2.0
        x = np.clip(y - lamb, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lmin = lamb
        else:
            lmax = lamb
    lamb = (lmin + lmax) / 2.0
    x = np.clip(y - lamb, 0.0, 1.0)
    return x

def compute_gradient(h, corr, alpha):
    """
    Compute gradient of log-sum-exp surrogate of correlation with respect to h.
    corr is full correlation (no scaling) of (h, 1-h)
    Returns gradient in same shape as h
    """
    n = len(h)
    # Compute softmax weights
    cmax = np.max(corr)
    w = np.exp(alpha * (corr - cmax))
    w = w / np.sum(w)
    
    # Compute finite difference approximation (rearrange)
    grad = np.ones(n)
    for i in range(-n+1, n):
        # Forward shift if i >= 0, backward if i < 0
        if i >= 0:
            shift = np.concatenate([h[i:], np.zeros(i)])
        else:
            shift = np.concatenate([np.zeros(-i), h[:i]])
        grad -= w[i + (n - 1)] * shift
        
    return grad

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_objective_smooth(h, alpha, full_corr=None):
    """
    Evaluate the smooth log-sum-exp surrogate
    """
    n = len(h)
    if full_corr is None:
        corr = correlate(h, 1 - h, mode="full")
    else:
        corr = full_corr
    cmax = np.max(corr)
    exp_terms = np.exp(alpha * (corr - cmax))
    log_sum = cmax + np.log(np.sum(exp_terms)) / alpha
    return log_sum

def projected_gradient_descent(h0, n, dx, alpha_start=5.0, alpha_factor=1.03, n_iter=5000, eta0=0.5):
    """
    Runs projected gradient descent with annealed soft max surrogate
    Returns best h found and its C5 value
    """
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha_start
    for t in range(n_iter):
        # Compute full correlation
        corr = correlate(h, 1 - h, mode="full")
        # Compute gradient of surrogate
        grad = compute_gradient(h, corr, alpha)
        # Gradient step
        eta = eta0 / np.sqrt(t + 1)
        y = h - eta * grad
        # Project onto feasible set
        h = project_simplex(y, n/2)
        # Anneal alpha
        if t % 200 == 0 and t > 0:
            alpha = min(alpha * alpha_factor, 200.0)
        
        # Evaluate true objective
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    return best_h, best_c5
