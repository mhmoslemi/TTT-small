import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def build_and_solve_lp(n, dx):
    """Build and solve the LP relaxation for the Erdos problem"""
    # Variables
    x = cp.Variable(n)
    t = cp.Variable()
    w = cp.Variable((n, 2*n - 1))  # For each shift and each index
    
    # Constraints for McCormick envelopes
    constraints = []
    
    # x bounds
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Mass constraint
    constraints.append(cp.sum(x) == n / 2)
    
    # McCormick constraints for w_{i,s}
    # s ranges from -(n-1) to (n-1)
    for s in range(-(n-1), n):
        for i in range(n):
            if 0 <= i + s < n:
                # w_{i,s} \leq x_i
                constraints.append(w[i, s + n - 1] <= x[i])
                # w_{i,s} \leq x_{i+s}
                constraints.append(w[i, s + n - 1] <= x[i + s])
                # w_{i,s} \geq x_i + x_{i+s} - 1
                constraints.append(w[i, s + n - 1] >= x[i] + x[i + s] - 1)
                # w_{i,s} \geq 0
                constraints.append(w[i, s + n - 1] >= 0)
            else:
                # Out of bounds - set to 0
                constraints.append(w[i, s + n - 1] == 0)

    # Constraints for t (minimum correlation lower bound)
    for s in range(-(n-1), n):
        if s < 0:
            # s is negative, so we need to shift by -s positions
            # This means x[i + s] = x[i - |s|] but remember that i starts at 0
            # Calculate sum of w_{i, s} for valid indices i
            total_sum = 0
            for i in range(n):
                if 0 <= i - s < n:
                    total_sum += w[i, s + n - 1]
                else:
                    # Apply zero boundary, i.e., contribution zero to w if out of range
                    pass
            constraints.append(total_sum >= t)
        else:
            # s >= 0
            total_sum = 0
            for i in range(n):
                if 0 <= i + s < n:
                    total_sum += w[i, s + n - 1]
                else:
                    pass
            constraints.append(total_sum >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem with a validator or fallback for solver issues
    try:
        prob.solve(solver=cp.ECOS, feastol=1e-8, reltol=1e-6)
        return x.value, t.value
    except:
        # Fallback to a slightly менее precise solve if one fails
        try:
            prob.solve(solver=cp.SCS, feastol=1e-7, reltol=1e-5)
            return x.value, t.value
        except:
            # Last resort: use even more coarse settings
            try:
                prob.solve(solver=cp.ECOS, feastol=1e-6, reltol=1e-4)
                return x.value, t.value
            except:
                return None, None

def round_solution(x_fract, n):
    """Round fractional solution to binary vector"""
    # Create a copy to avoid modifying original
    x_rounded = np.zeros(n)
    # Assign indices based on sorted values of the fractional solution
    indices = np.argsort(x_fract)[::-1]
    ones_positions = indices[:n//2]
    x_rounded[ones_positions] = 1.0
    return x_rounded

def swap_based_local_search(h_initial, n_points, dx, max_time=300.0):
    # To store current best solution and best value
    current_h = h_initial.copy()

    best_c5 = evaluate_erdos_solution(current_h, dx)
    
    start_time = time.time()
    num_swaps = 0
    stagnation = 0  # Counter for stagnation
    max_stagnation = 5000
    
    # Precompute correlation of full vector for efficiency
    # Initialize full correlation vector
    def compute_full_correlation(h):
        correlation_full = np.correlate(h, 1-h, mode="full")
        return correlation_full
    
    full_corr = compute_full_correlation(current_h)
    min_corr_val = np.min(full_corr)
    
    while time.time() - start_time < max_time:
        
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i is a '1' and j is a '0'
        if current_h[i] != 1 or current_h[j] != 0:
            i, j = max(i, j) if current_h[i] != 1 else min(i, j), min(i, j) if current_h[j] != 0 else max(i, j)
            if current_h[i] != 1:
                if j == i: continue
                i, j = (j, i) if current_h[j] != 0 else (i, j)
            else:
                if j == i: continue
                if current_h[j] == 0:
                    pass
                else:
                    # Try different j until we find a 0
                    for new_j in range(n_points):
                        if current_h[new_j] == 0:
                            j = new_j
                            break
                    else:
                        continue  # No free j, skip this attempt
        
        # Modify current_h temporarily
        temp_h = current_h.copy()
        temp_h[i] = 0
        temp_h[j] = 1
        
        # Compute C5
        new_c5 = evaluate_erdos_solution(temp_h, dx)
        
        # Accept if it's better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = temp_h.copy()
            current_h = temp_h.copy()
            # Precompute updated full correlation vector
            full_corr = compute_full_correlation(current_h)
            min_corr_val = np.min(full_corr)
            stagnation = 0
        else:
            stagnation += 1
            
        num_swaps += 1
        
        if stagnation >= max_stagnation:
            # Complete restart to move out of local minimum
            restart_choice = np.where(current_h == 1)[0]
            if len(restart_choice) > 0:
                i = np.random.choice(restart_choice)
                # Pick a '0' to swap with
                zero_positions = np.where(current_h == 0)[0]
                if len(zero_positions) > 0:
                    j = np.random.choice(zero_positions)
                    current_h[i] = 0
                    current_h[j] = 1
                    stagnation = 0
                    # Recompute correlation if needed, though shouldn't be necessary
                    full_corr = compute_full_correlation(current_h)
                    min_corr_val = np.min(full_corr)
                    
    return best_h, best_c5

def symmetrize(h):
    """Symmetrize the h vector about center."""
    n = len(h)
    if n % 2 == 0:
        mid = n // 2
        h_sym = (h[:mid] + h[mid:][::-1]) / 2.0
        h_final = np.concatenate([h_sym, h_sym[::-1]])
    else:
        mid = n // 2
        h_sym = (h[:mid] + h[mid+1:][::-1]) / 2.0
        h_final = np.concatenate([h_sym, [h[mid]], h_sym[::-1]])
    # Project back onto capsimplex to maintain sum == 1
    h_final = np.clip(h_final, 0, 1)
    total = np.sum(h_final)
    h_final = h_final * (n / 2) / total
    return h_final

def project_onto_capsimplex(h, target_sum):
    """Project vector onto the set {x: 0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    # Make sure values are within bounds
    h = np.clip(h, 0, 1)
    # Adjust by linear projection
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.ones(n) * (target_sum / n)
    else:
        scale = target_sum / current_sum
        h_scaled = h * scale
        return h_scaled
