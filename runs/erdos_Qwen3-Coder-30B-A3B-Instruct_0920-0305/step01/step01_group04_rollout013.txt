from scipy import fftpack

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def compute_correlation(h):
    """Use FFT for faster correlation if n is large."""
    if len(h) > 100:
        # Use FFT for convolution (more efficient)
        h_padded = np.pad(h, (0, len(h)-1), mode='constant')
        h_flip = np.flip(h)
        corr = np.real(fftpack.ifft(fftpack.fft(h_padded) * np.conj(fftpack.fft(h_flip))))
        return corr[len(h)-1:]
    else:
        return np.correlate(h, 1 - h, mode="full")

def compute_max_mismatch(h_values):
    """Compute maximum mismatch (without dx)."""
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    corr = compute_correlation(h)
    max_mismatch = np.max(corr)
    return max_mismatch

def project_to_sum_constraint(h, target_sum):
    """Project h onto sum = target_sum and box [0,1]."""
    # Clip first
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Water-filling: sort h descending, and assign the remaining mass
    sorted_indices = np.argsort(h)[::-1]
    h_sorted = h[sorted_indices]
    remaining = target_sum - current_sum
    if remaining > 0:
        for i in range(len(h_sorted)):
            if remaining <= 0:
                break
            # How much more can we add to h_sorted[i]?
            incr = min(1 - h_sorted[i], remaining)
            remaining -= incr
            h_sorted[i] += incr
    elif remaining < 0:
        for i in range(len(h_sorted)):
            if remaining >= 0:
                break
            # How much can we subtract from h_sorted[i]?
            decr = min(h_sorted[i], -remaining)
            remaining += decr
            h_sorted[i] -= decr
    # Reindex to original positions
    h_out = h.copy()
    h_out[sorted_indices] = h_sorted
    return h_out

def get_grad(h, d):
    """Compute the gradient of m_d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 1 - h[(i+d) % n] - h[(i-d) % n]
    return grad

def hill_climb(h0):
    """Local improvement using swaps."""
    h = h0.copy()
    m_old = compute_max_mismatch(h)
    changed = True
    while changed:
        changed = False
        for i in range(len(h)):
            if h[i] <= 0:
                continue
            for j in range(len(h)):
                if h[j] >= 1:
                    continue
                # Try swapping h[i] and h[j] 
                h[i], h[j] = h[j], h[i]
                m_new = compute_max_mismatch(h)
                if m_new < m_old:
                    m_old = m_new
                    changed = True
                else:
                    h[i], h[j] = h[j], h[i]  # undo
    return h
