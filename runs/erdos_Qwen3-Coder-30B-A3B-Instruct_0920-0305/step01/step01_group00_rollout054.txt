import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from math import log

def project_capped_simplex(y, target_sum, lower=0, upper=1):
    """Project vector y onto the capped simplex: sum x_i = target_sum, lower <= x_i <= upper"""
    n = len(y)
    # The projection is of the form: x_i = clip(y_i - lambda, lower, upper)
    # We need to find lambda such that sum x_i = target_sum
    # Use bisection on the decreasing function sum(clip(y - lambda, l, u))
    def sum_proj(lambda_val):
        proj = np.clip(y - lambda_val, lower, upper)
        return np.sum(proj)
    
    # Bisection bounds
    y_min, y_max = np.min(y), np.max(y)
    low = y_min - upper
    high = y_max - lower
    # For safety
    low = min(low, -1.0)
    high = max(high, 1.0)
    eps = 1e-12
    
    for _ in range(100):  # Bisection steps
        mid = (low + high) / 2.0
        sum_mid = sum_proj(mid)
        if abs(sum_mid - target_sum) < eps:
            break
        if sum_mid > target_sum:
            low = mid
        else:
            high = mid
            
    lambda_opt = (low + high) / 2.0
    return np.clip(y - lambda_opt, lower, upper)

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def softmax_w(corr, alpha):
    """Compute softmax weights for a vector c"""
    c_max = np.max(corr)
    # Subtract max for numerical stability
    exp_c = np.exp(alpha * (corr - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def gradient_F_alpha(h, alpha):
    """Compute gradient of F_alpha(h) = (1/alpha)*log(sum exp(alpha*C_s))"""
    n = len(h)
    # Compute raw correlations
    corr = correlate(h, 1 - h, mode="full")
    # Get weights
    w = softmax_w(corr, alpha)
    # Now compute the gradient term: dF_alpha/dh_i = 1 - sum_s w_s *(h_{i+s} + h_{i-s})
    grad = np.ones(n)
    for s in range(-n+1, n):
        if s >= 0:
            shifted = np.concatenate([h[s:], np.zeros(s)])
        else:
            shifted = np.concatenate([np.zeros(-s), h[:s]])
        grad -= w[s + n - 1] * shifted
    return grad

def line_search(h_prev, grad, func, alpha, eta, max_backtracks=5):
    """Simple backtracking line search"""
    h_new = h_prev - eta * grad
    for _ in range(max_backtracks):
        h_proj = project_capped_simplex(h_new, n/2)
        if func(h_proj) < func(h_prev):
            return h_proj
        eta *= 0.5
        h_new = h_prev - eta * grad
        if eta < 1e-10:
            break
    return h_proj

    target_sum = n / 2.0
    max_iter_per_restart = 5000
    alpha_schedule = (5.0, 1.03, 200)
    eta0 = 0.5

    # Initial setup: use or interpolate initial_h_values

    if len(initial_h) != n:

        initial_h = np.interp(x_new, x_orig, initial_h)
    
    best_c5 = 1e10
    best_h = None

    # Reserve some time for final polishing
    remainder_time = 0.3 * budget_s  # 30% for refinement
    allocated_time = budget_s - remainder_time
    
    restart = 0
    time_allocated = time.time() - start_time
    
    # Multi-start repeat
    while time_allocated < allocated_time:
        # Start point
        if restart == 0:
            x = initial_h.copy()
        else:
            # Random start on simplex
            x = np.random.rand(n)
            # Project onto the simplex
            x = project_capped_simplex(x, target_sum)
        
        # Annealing schedule for alpha
        alpha_start, alpha_factor, alpha_max = alpha_schedule
        alpha = alpha_start
        alpha_step_interval = 200
        
        x_curr = x.copy()
        best_c5_so_far = 1e10
        best_h_so_far = None

        # Iterative gradient descent
        for i in range(max_iter_per_restart):
            # Check time for early exit
            if (time.time() - start_time) > allocated_time:
                break

            # Calculate current correlation vector
            corr = correlate(x_curr, 1 - x_curr, mode='full')
            cur_c5 = np.max(corr) * dx

            # Update best if improved
            if cur_c5 < best_c5_so_far:
                best_c5_so_far = cur_c5
                best_h_so_far = x_curr.copy()
            
            # Compute gradient
            try:
                grad = gradient_F_alpha(x_curr, alpha)
            except Exception as e:
                break # Escape if gradient fails

            # Update step size
            eta = eta0 / np.sqrt(i + 1)
            # Projected gradient step
            x_new = x_curr - eta * grad
            x_curr = project_capped_simplex(x_new, target_sum)

            # Anneal alpha
            if i % alpha_step_interval == 0 and i > 0:
                alpha = min(alpha * alpha_factor, alpha_max)

        # After loop, best_h_so_far from this restart
        if best_h_so_far is not None and best_c5_so_far < best_c5:
            best_c5 = best_c5_so_far
            best_h = best_h_so_far.copy()
            
        restart += 1
        time_allocated = time.time() - start_time
        
        # Early stopping if target achieved
        if best_c5 < 0.3807:
            break

    # Final polishing with local optimizer
    if best_h is not None:
        try:
            # Create bounds and constraint
            bounds = [(0, 1) for _ in range(n)]
            def objective(h):
                return evaluate_c5(h)

            def constraint_eq(h):
                return np.sum(h) * dx - 1.0

            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Refine the solution
            res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,

            if res.success:
                best_h = res.x
                # Ensure sum is correct
                scale = 1.0 / (np.sum(best_h) * dx)
                best_h = best_h * scale
                final_c5 = evaluate_c5(best_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
        except Exception as e:
            pass  # Fall back to previous best

    return (best_h.tolist(), best_c5, n)
