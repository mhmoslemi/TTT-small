import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from math import log, exp

def project_simplex(y, s):
    # Project vector y onto the simplex defined by sum(x) = s and 0 <= x <= 1
    # Uses the method of Dubois & Govaerts (1990) for the Euclidean projection
    n = len(y)
    # Sort in descending order
    u = np.sort(y)[::-1]
    # Compute cumulative sums
    cu = np.cumsum(u)
    # Find the threshold
    i = np.arange(1, n + 1)
    tmp = (u - (s - cu) / i)
    rho = np.max(i[tmp > 0])
    # Compute lambda
    if rho == 0:
        l = max(0, (s - np.sum(y)) / n)  # not necessary, but hedge against numerical issues
        lambda_val = 0
    else:
        lambda_val = (s - cu[rho - 1]) / rho
    
    # Clip and return result
    x = np.clip(y - lambda_val, 0, 1)
    return x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx

def get_grad_F_alpha(h, alpha, c):
    n = len(h)
    # Compute softmax weights: w_s = exp(alpha * c_s) / sum(exp(alpha * c_t))
    c_max = np.max(c)
    w = np.exp(alpha * (c - c_max))
    w = w / np.sum(w)

    # Gradient computation: (d_h F_alpha)_i = 1 - sum_s w_s (h_{i+s} + h_{i-s})
    # We compute the full convolution-like sum for each i
    grad = np.ones(n)
    for s in range(2*n - 1):
        # s is a shift offset in the correlation (full mode)
        shift = s - (n - 1)
        # Use circular indices to compute h_{i+s} and h_{i-s}
        # But here we use np.take with mode='clip' which handles boundary conditions
        # We compute both terms: h_{i+s} and h_{i-s} directly
        if shift > 0:
            h_left = np.append(np.zeros(shift), h[:-shift])
            h_right = np.append(h[shift:], np.zeros(shift))
        elif shift < 0:
            h_left = np.append(h[-shift:], np.zeros(-shift))
            h_right = np.append(np.zeros(-shift), h[:-shift])
        else:
            h_left = h.copy()
            h_right = h.copy()
        # Combine them
        shifted_sum = h_left + h_right
        grad -= w[s] * shifted_sum

    return grad

def line_search(h, grad, eta, alpha, c_old):
    # Simple backtracking line search
    for _ in range(10):
        h_new = h - eta * grad
        h_projected = project_simplex(h_new, np.sum(h) * 2.0 / len(h))
        # Get correlation values for new h
        n = len(h_projected)
        dx = 2.0 / n
        corr = correlate(h_projected, 1 - h_projected, mode="full")
        new_c = corr * dx
        F_new = (1/alpha) * log(np.sum(np.exp(alpha * new_c)))
        F_old = (1/alpha) * log(np.sum(np.exp(alpha * c_old)))
        if F_new <= F_old:
            return h_projected, F_new
        eta = eta * 0.5
    return h_projected, F_new

    start_time = time.time()
    n_points = 96  # Key performance tuning: balance grid size with speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # We want sum(h) = target_sum

    # Use initial_h_values for the first restart if possible

        # Interpolate to the new grid size

        h0_interp = np.interp(x_new, x_orig, h0)
        # Need to ensure the sum constraint approximately holds, realign it
        scale_factor = target_sum / np.sum(h0_interp)
        h0_interp = h0_interp * scale_factor
        initial_h = h0_interp.copy()
    except:
        # Fallback: random initial feasible vector, projected
        h_init = np.random.rand(n_points)
        initial_h = project_simplex(h_init, target_sum)

    # Main algorithm: Projected gradient descent with soft relaxation and restarts
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    best_f_alpha = np.inf

    # Optimization parameters
    max_restarts = 50  # Experiment with this for stability
    max_iters = 5000   # Higher for targeted improvement
    eta0 = 0.5
    alpha_schedule_factor = 1.03
    alpha_max = 200
    alpha_start = 5
    iteration_per_alpha = 200
    restart_iter_count = 0

    for restart in range(max_restarts):
        if (time.time() - start_time) > (budget_s - 30):
            break

        # Restart with different initializations
        if restart == 0:
            h_current = initial_h.copy()
        else:
            h_init = np.random.rand(n_points)
            h_current = project_simplex(h_init, target_sum)

        alpha = alpha_start
        prev_c = None  # Keep track of correlation values for grad computation

        for iter_num in range(max_iters):
            if (time.time() - start_time) > (budget_s - 10):
                break
                
            # Compute the correlation values (unscaled with dx, which is incorporated in objective)
            corr = correlate(h_current, 1 - h_current, mode="full")
            c = corr * dx  # Now c is the actual correlation values (with dx applied)
            if prev_c is None:
                prev_c = c
            # Compute surrogate value and gradient
            F_alpha = (1.0 / alpha) * log(np.sum(np.exp(alpha * c)))
            grad = get_grad_F_alpha(h_current, alpha, c)
            
            # Update with line search
            eta = eta0 / np.sqrt(iter_num + 1)
            h_projected, new_F_alpha = line_search(h_current, grad, eta, alpha, c)
            h_current = h_projected
            
            # Update alpha
            if iter_num % iteration_per_alpha == 0 and alpha < alpha_max:
                alpha *= alpha_schedule_factor
            # Update best if the true C5 has improved
            if iter_num % 10 == 0 or iter_num + 1 == max_iters:
                current_c5 = evaluate_c5(h_current)
                if current_c5 < best_c5:
                    best_h = h_current.copy()

        restart_iter_count += 1

    # Final polishing: use local optimization to refine the best solution
    try:
        # Set up bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_func(h):
            return evaluate_c5(h)
        
        # Constraints
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0  # This should be zero
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Try local refinement (few iterations)
        res = minimize(obj_func, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 200})
        
        if res.success:
            best_h_refined = res.x
            refined_c5 = evaluate_c5(best_h_refined)
            if refined_c5 < best_c5:
                best_h = best_h_refined.copy()
                best_c5 = refined_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
