import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def solve_lp_relaxation(n):
    """Solve the linear programming relaxation of the Erdos problem"""
    # Variables
    x = cp.Variable(n)
    w = cp.Variable((n, 2*n - 1))  # w[i,s] for shifts s in [-n+1, n-1]
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(x) == n/2,  # mass constraint
        0 <= x, x <= 1,    # box constraints
        t <= 0             # initial lower bound on t
    ]

    # Add McCormick constraints for each pair (i,s)
    for s in range(-(n-1), n):
        for i in range(n):
            if 0 <= i+s < n:
                # The four McCormick inequalities
                constraints.append(w[i, s + (n-1)] <= x[i])
                constraints.append(w[i, s + (n-1)] <= x[i+s])
                constraints.append(w[i, s + (n-1)] >= x[i] + x[i+s] - 1)
                constraints.append(w[i, s + (n-1)] >= 0)

    # Add t lower-bound constraints
    for s in range(-(n-1), n):
        col_sum = 0
        for i in range(n):
            if 0 <= i+s < n:
                col_sum += w[i, s + (n-1)]
        constraints.append(col_sum >= t)

    # Objective
    objective = cp.Maximize(t)

    # Solve the problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)

    return x.value, t.value

def deterministic_rounding(x, n):
    """Round the fractional solution to binary vector with exact mass"""
    indices = np.argsort(x)[::-1]  # sort by decreasing values
    b = np.zeros(n)
    b[indices[:n//2]] = 1
    return b.astype(int)

def swap_improve(h, n):
    """Improve h using deterministic swaps"""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Store the correlation vector for efficient updates
    def compute_full_correlation(h_val):
        corr = np.correlate(h_val, 1-h_val, mode="full")
        return corr
    
    best_corr = compute_full_correlation(h)
    last_improvement = 0
    iter_count = 0
    
    for _ in range(100000):
        if iter_count - last_improvement > 2000:
            break
            
        # Pick random 1-position and 0-position
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Try the swap
        h[i], h[j] = 0, 1
        new_c5 = evaluate_erdos_solution(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            last_improvement = iter_count
            best_corr = compute_full_correlation(h)
        else:
            # Revert the swap
            h[i], h[j] = 1, 0
            
        iter_count += 1
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_best = 100
    
    # Configuration: use several resolutions
    resolutions = [80, 100, 120]
    
    for n in resolutions:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        print(f"Solving for n={n}...")
        try:
            x_opt, t_opt = solve_lp_relaxation(n)
        
            # Rounding and improvement
            binary_h = deterministic_rounding(x_opt, n)
            
            print(f"LP bound t = {t_opt}, C5 raw = {1 - t_opt}")
            
            # Local improvement via swapping
            improved_h, improved_c5 = swap_improve(binary_h, n)
            
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
                n_best = n
            
            print(f"Improved to C5 = {best_c5}")
            
        except Exception as e:
            print(f"Error solving for n={n}: {e}")
            continue
            
    dx = 2.0 / n_best
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_best)
