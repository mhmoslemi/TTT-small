def project_to_box_simplex(y, target_sum):
    """Project vector y onto the box-simplex set {x : 0 <= x_i <= 1, sum x_i = target_sum}"""
    # Use bisection to find the right shift alpha
    low, high = np.min(y), np.max(y) + 1.0
    eps = 1e-12
    max_iter = 50
    for _ in range(max_iter):
        alpha = (low + high) / 2.0
        x_proj = np.clip(y - alpha, 0, 1)
        s = np.sum(x_proj)
        if s > target_sum:
            low = alpha
        else:
            high = alpha
        if abs(s - target_sum) < eps:
            break
    return np.clip(y - (low + high) / 2.0, 0, 1)

def compute_fs(h, s):
    """Compute f_s(h) for a fixed shift s"""
    n = len(h)
    if s >= 0:
        # Right shift
        c = h[s:]
        c_pad = np.concatenate([c, np.zeros(s)])
        c_bar = 1 - h
        c_bar_shift = np.concatenate([np.zeros(s), c_bar[:n-s]])
    else:
        # Left shift
        c = h[:s]
        c_pad = np.concatenate([np.zeros(-s), c])
        c_bar = 1 - h
        c_bar_shift = np.concatenate([c_bar[-s:], np.zeros(-s)])
    return np.dot(c, c_bar_shift)

def compute_all_fs(h):
    """Compute all f_s(h) for s = -(n-1) to (n-1)"""
    n = len(h)
    all_fs = []
    for s in range(-(n-1), n):
        all_fs.append(compute_fs(h, s))
    return np.array(all_fs)

def smooth_max(h, p=12):
    """Compute smooth p-norm approximation to max"""
    fs = compute_all_fs(h)
    fs_abs = np.abs(fs)
    if np.any(fs_abs == 0):
        return 0.0
    return (np.sum(fs_abs ** p))**(1.0/p)

def compute_grad(h, p=12):
    """Analytic gradient of smooth max, with penalty"""

    fs = compute_all_fs(h)
    fs_abs = np.abs(fs)

    # Add penalty for sum constraint

    # Compute gradient of smooth max
    eps = 1e-12
    if np.all(fs_abs < eps):
        return grad
    for s in range(-(n-1), n):
        if abs(fs[s + n-1]) < eps:
            continue
        w_s = fs_abs[s + n-1]**(p-1)
        if s >= 0:
            c_bar = 1 - h
            c_bar_shift = np.concatenate([np.zeros(s), c_bar[:n-s]])
            grad[:n-s] += w_s * (1 - c_bar_shift[s:])
        else:
            c_bar = 1 - h
            c_bar_shift = np.concatenate([c_bar[-s:], np.zeros(-s)])
            grad[-s:] += w_s * (1 - c_bar_shift[:n+s])
    return grad

def optimized_p_gradient_descent(h0, n, max_iter=2000, p=4, print_progress=False):
    """Projected gradient descent with smooth p-norm"""
    h = h0.copy()
    dx = 2.0 / n
    lambda_penalty = 1000.0
    
    for i in range(max_iter):
        # Compute gradient
        grad = compute_grad(h, p=p)
        
        # Gradient step
        eta = 0.1
        h_new = h - eta * grad
        
        # Project onto feasible set
        h = project_to_box_simplex(h_new, n/2)
        
        if print_progress and i % 500 == 0:
            c5 = smooth_max(h, p=p) * dx
            print(f"Iter: {i}, Max smooth approx C5: {c5:.6f}")
    
    return h
