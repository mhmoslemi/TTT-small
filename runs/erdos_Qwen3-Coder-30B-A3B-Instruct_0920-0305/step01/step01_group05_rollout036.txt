def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_sum_constraint(z, target_sum):
    """Project z onto the set {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Simple bisection to find alpha such that clip(z + alpha*1, 0, 1) has sum = target_sum
    low, high = -10, 10
    for _ in range(30):
        alpha = (low + high) / 2
        candidate = np.clip(z + alpha, 0, 1)
        sum_candidate = np.sum(candidate)
        if sum_candidate < target_sum:
            high = alpha
        else:
            low = alpha
    result = np.clip(z + (low + high)/2, 0, 1)
    return result
