import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def build_shift_masks(n):
    """Build forward and backward masks for each shift s"""
    mask_fwd = {}
    mask_bwd = {}
    for s in range(-(n-1), n):
        mask_fwd[s] = np.arange(n)[s < n - 1]
        mask_bwd[s] = np.arange(n)[s > 0]
    return mask_fwd, mask_bwd

def compute_gradient_gd(h, s, mask_fwd, mask_bwd):
    """Compute gradient of g_s^d at h"""
    grad = np.zeros_like(h)
    if s > 0:
        # forward shift
        grad[:-s] += (h[:-s] + h[s:])
        grad[s:] += (h[:-s] + h[s:])
    else:
        # backward shift
        s_abs = -s
        grad[s_abs:] += (h[s_abs:] + h[:-s_abs])
        grad[:-s_abs] += (h[s_abs:] + h[:-s_abs])
    return grad

def solve_ccp_subproblem(h_prev, n, mask_fwd, mask_bwd, dx):
    """Solve one CCP subproblem using CVXPY"""
    h = cp.Variable(n)
    tau = cp.Variable()
    
    constraints = [0 <= h, h <= 1, cp.sum(h) == n / 2]

    for s in range(-(n-1), n):
        # Compute g_s^c terms
        g_s_c = cp.sum(h) + 0.5 * cp.sum_squares(h[mask_fwd[s]]) + 0.5 * cp.sum_squares(h[mask_bwd[s]])
        # Compute g_s^d gradient at h_prev
        grad_s = compute_gradient_gd(h_prev, s, mask_fwd, mask_bwd)
        # Compute g_s^d at h_prev
        g_s_d_prev = 0.5 * np.sum((h_prev + np.roll(h_prev, s)) ** 2)
        # Compute affine approximation of -g_s^d at h_prev
        affine_approx = g_s_d_prev + grad_s.T @ (h - h_prev)
        constraints.append(tau >= g_s_c - affine_approx)
        
    prob = cp.Problem(cp.Minimize(tau), constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8)
    except:
        try:
            prob.solve(solver=cp.SCS, abstol=1e-8, reltol=1e-8)
        except:
            # fallback solution if solver fails
            return h_prev.copy()
    
    if prob.status == cp.OPTIMAL:
        return h.value
    else:
        return h_prev.copy()

def coordinate_descent_polish(h0, n, dx):
    """Perform coordinate descent polishing on the exact objective"""
    h = h0.copy()
    # Find active shift (for each index)
    active_shifts = {}
    for i in range(n):
        max_f = -1
        best_s = 0
        for s in range(-(n-1), n):
            f = h[i] * (1 - h[(i+s) % n])
            if f > max_f:
                max_f = f
                best_s = s
        active_shifts[i] = best_s

    iters = 5
    for _ in range(iters):
        indices = list(range(n))
        np.random.shuffle(indices)
        for i in indices:
            # Compute breakpoints for shift s where f_s maximizes
            # There can be at most 2n-1 breakpoints
            breakpoints = set()
            for s in range(-(n-1), n):
                if i+s >= 0 and i+s < n:
                    # only fixed-point shift terms affect h_i
                    breakpoints.add(0)
                    breakpoints.add(1)
            breakpoints = sorted(breakpoints)

            # Check active shift for h_i
            best_s = active_shifts[i]
            eff_s = best_s
            known_max = h[i] * (1 - h[(i+eff_s) % n])
            best_val = h[i]

            # Check boundaries and breakpoints
            for u in [0, 1] + breakpoints:
                if 0 <= u <= 1:
                    h[i] = u
                    h[(i+eff_s) % n] = max(h[(i+eff_s) % n], 0)
                    h[(i+eff_s) % n] = min(h[(i+eff_s) % n], 1)
                    # Need to recompute active shift
                    max_f = -1
                    best_s = None
                    for s in range(-(n-1), n):
                        f = h[i] * (1 - h[(i+s) % n])
                        if f > max_f:
                            max_f = f
                            best_s = s
                    if abs(max_f - known_max) < 1e-10:
                        # This point confirms current max
                        if 0 <= u <= 1:
                            best_val = u
                            break

            # Now try small adjustments to see if we improve
            # But simpler to just enforce a valid solution
            # Avoid advanced optimality checks for speed
            h[i] = max(0, min(1, best_val))

    return h
