def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

        corr = np.correlate(h_vals, h_vals, mode='full')
        corr = corr * dx  # Scale for integration over dx
        # The softmin to approximate min
        # logsumexp(-lambda * corr)/lambda
        max_val = np.max(corr)
        corr_scaled = corr - max_val
        exp_corr = np.exp(-lambda_param * corr_scaled)
        sum_exp = np.sum(exp_corr)
        # Softmin-like loss to be minimized
        loss = -np.log(sum_exp) / lambda_param  # -1/λ * log(sum(exp(-λ * f_s)))
        return loss
    
    def grad_smooth(h_vals, lambda_param):
        # Compute correlation w/o dx (or include it at the end via dot product)
        corr = np.correlate(h_vals, h_vals, mode='full')
        max_corr = np.max(corr)
        corr_shifted = corr - max_corr
        exp_corr = np.exp(-lambda_param * corr_shifted)
        sum_exp = np.sum(exp_corr)
        if sum_exp < 1e-100:
            weights = np.ones_like(exp_corr)
        else:
            weights = exp_corr / sum_exp

        # Now compute gradient
        grad = np.zeros_like(h_vals)
        n = len(h_vals)
        
        # For each shift
        for s in range(-(n-1), n):
            if s >= 0:
                grad[:n-s] += weights[s + (n-1)] * h_vals[s:]
                grad[s:] += weights[s + (n-1)] * h_vals[:n-s]
            else:
                s_abs = -s
                grad[s_abs:] += weights[s + (n-1)] * h_vals[:n-s_abs]
                grad[:n-s_abs] += weights[s + (n-1)] * h_vals[s_abs:]
        return grad  # Not multiplying by -gradient yet, will do with descent
