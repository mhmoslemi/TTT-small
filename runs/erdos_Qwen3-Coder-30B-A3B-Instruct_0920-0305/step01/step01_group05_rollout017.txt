def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_sum_constraint(z, target_sum, box_min=0, box_max=1):
    """Project z to lie in [box_min, box_max]^n and sum to target_sum"""
    # Ensure the vector is clipped to [0,1] and re-scaled appropriately
    z = np.clip(z, box_min, box_max)
    current_sum = np.sum(z)
    if abs(current_sum - target_sum) < 1e-8:
        return z
    if current_sum == 0:
        return np.full_like(z, target_sum / len(z))
    
    # Use bisection to find alpha such that clip(z + alpha*ones, 0, 1) has correct sum
    z_clipped = z.copy()
    alpha_min = -100
    alpha_max = 100
    for _ in range(50):
        alpha = (alpha_min + alpha_max) / 2
        z_test = np.clip(z_clipped + alpha, box_min, box_max)
        test_sum = np.sum(z_test)
        if test_sum < target_sum:
            alpha_min = alpha
        else:
            alpha_max = alpha
    alpha = (alpha_min + alpha_max) / 2
    z_out = np.clip(z_clipped + alpha, box_min, box_max)
    return z_out
