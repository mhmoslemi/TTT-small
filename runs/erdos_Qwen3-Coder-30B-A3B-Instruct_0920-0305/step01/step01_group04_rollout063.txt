def evaluate_h(h_values):
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h_values):

    diff = n_points/2 - total

                # Slightly complex, so do it carefully
                new_val = h_sorted[i] + needed / (i + 1)
                if new_val <= max_val:
                    h_sorted[i:] += needed / (i + 1)
                    diff = 0
                else:
                    # Set to 1 and move to next
                    diff -= (1.0 - h_sorted[i]) * (i+1)
