import numpy as np

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def clip_and_rescale(x, target_sum):
    """Clip array to [0,1] and rescale to match target sum"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if current_sum > 0:
        x = x * (target_sum / current_sum)
    return x

def solve_sdp(n, dx):
    """Solve the SDP relaxation"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n/2,
        cp.diag(X) == h,
        X >> 0,
        t >= 0
    ]

    # Add McCormick constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints += [
                X[i,j] >= 0,
                X[i,j] <= h[i],
                X[i,j] <= h[j],
                X[i,j] >= h[i] + h[j] - 1
            ]

    # Overlap constraints for all non-zero shifts
    for s in range(-(n-1), n):
        if s == 0:
            continue
        shift_mat = np.eye(n, k=s)
        shift_sum = cp.sum(cp.multiply(shift_mat, X))
        constraints += [t >= n/2 - shift_sum]

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    prob.solve(solver='SCS', eps=1e-5, max_iters=5000, verbose=False)
    
    if prob.status == cp.OPTIMAL and t.value is not None:
        return h.value, X.value, t.value
    else:
        return None, None, None

def deterministic_rounding(h_sdp, n):
    """Deterministic rounding - set top half to 1, rest to 0"""
    h = h_sdp.copy()
    # Order indices by value descending
    order = np.argsort(h)[::-1]
    new_h = np.zeros(n)
    new_h[order[:n//2]] = 1.0
    return new_h
