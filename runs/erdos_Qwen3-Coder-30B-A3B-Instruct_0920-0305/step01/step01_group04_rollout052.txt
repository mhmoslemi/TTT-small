import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# Constants
n_points = 100
dx = 2.0 / n_points

# Global for time tracking
start_time = None
budget_s = 600


def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def project_onto_feasible(h, target_sum):
    """Project h onto the feasible polytope {h ∈ [0,1]^n : sum h = target_sum}"""
    if np.abs(np.sum(h) - target_sum) < 1e-6:
        return h.copy()

    h_sorted = np.sort(h)[::-1]
    # Water-filling approach
    h_proj = h.copy()
    total = np.sum(h_proj)
    if total > target_sum:
        # We need to decrease
        delta = total - target_sum
        for i in reversed(range(len(h_sorted))):
            if h_proj[i] > 0:
                # Check if we can reduce
                cap = h_proj[i]
                decrease = min(delta, cap)
                h_proj[i] -= decrease
                delta -= decrease
                if delta <= 0:
                    break
    else:
        # We need to increase
        delta = target_sum - total
        for i in range(len(h_sorted)):
            if h_proj[i] < 1:
                cap = 1 - h_proj[i]
                increase = min(delta, cap)
                h_proj[i] += increase
                delta -= increase
                if delta <= 0:
                    break

    # Ensure bounds and grid sum
    h_proj = np.clip(h_proj, 0, 1)
    h_proj = np.minimum(h_proj, 1)
    h_proj = np.maximum(h_proj, 0)

    # Adjust slightly to enforce sum exactly
    current = np.sum(h_proj)
    diff = target_sum - current
    if abs(diff) > 1e-6:
        # Adjust elements towards center
        idxs = np.argsort(h_proj)
        for i in range(len(h_proj)):
            if diff < 0 and h_proj[idxs[i]] > 0:
                dec = min(-diff, h_proj[idxs[i]])
                h_proj[idxs[i]] -= dec
                diff += dec
                if abs(diff) < 1e-6:
                    break
            elif diff > 0 and h_proj[idxs[i]] < 1:
                inc = min(diff, 1 - h_proj[idxs[i]])
                h_proj[idxs[i]] += inc
                diff -= inc
                if abs(diff) < 1e-6:
                    break
    return h_proj


def generate_random_binary(seed, n):
    """Generate a random binary vector with exactly n//2 ones."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1
    return h


def get_shifted_neg_h(h, shift):
    """Return h shifted backward (i.e., -h_{i+shift}) using periodic indexing."""
    return np.roll(h, shift)

def get_gradient(h, worst_shift):
    """Return the gradient of the mismatch function at shift d."""
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = 1 - h[(i + worst_shift) % len(h)] - h[(i - worst_shift) % len(h)]
    return grad


def run_sub_gradient_annealing(h_start, max_iters, initial_lr, alpha_decay, temp_init, temp_decay, seed=42):
    """Run sub-gradient descent with simulated annealing."""
    rng = np.random.default_rng(seed)
    best_h = h_start.copy()
    best_c5 = evaluate_h(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    t = 0
    lr = initial_lr
    T = temp_init

    for it in range(max_iters):
        t += 1
        # Evaluate correlation for all shifts
        corr = correlate(current_h, 1 - current_h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        worst_shift = max_idx - len(corr)//2  # Convert full fft index to shift index

        # Compute gradient
        grad = get_gradient(current_h, worst_shift)

        # Step
        new_h = current_h - lr * grad
        new_h = project_onto_feasible(new_h, n_points/2)

        # Compute new C5
        new_c5 = evaluate_h(new_h)

        # Accept or reject
        delta = new_c5 - current_c5
        if T > 1e-10:
            if delta < 0 or rng.uniform(0, 1) < np.exp(-delta / T):
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()

        # Update parameters
        lr = initial_lr / (1 + alpha_decay * t)
        T = T * temp_decay

    return best_h, best_c5


def round_to_binary_with_fractional(h_values):
    """Round continuously valued h to binary with fractional entry if needed."""
    h = np.array(h_values)
    # Sort indices by value descending
    sorted_indices = np.argsort(h)[::-1]
    # Keep top ⌊n/2⌋ values as 1; rest as 0
    threshold = len(h) // 2
    binary = np.zeros(len(h))
    # Set top threshold entries to 1
    for i in range(threshold):
        binary[sorted_indices[i]] = 1
    # Check if the sum is correct
    sum_h = np.sum(binary)
    if abs(sum_h - n_points/2) > 1e-6 and n_points % 2 == 1:
        # Set one entry to 0.5 to balance sum to exactly n_points/2
        binary[sorted_indices[threshold]] = 0.5
        binary[sorted_indices[threshold-1]] = 1 - 0.5

    return binary


def local_improvement(h_initial):
    """Hill climb to improve solution by locally swapping a 1/0 pair."""
    h = h_initial.copy()
    # Iterate until no improving swap found
    improved = True
    while improved:
        improved = False
        orig_c5 = evaluate_h(h)
        for i in range(len(h)):
            for j in range(len(h)):
                if h[i] >= 1 and h[j] <= 0:
                    # Try swap
                    h[i] = 0
                    h[j] = 1
                    new_c5 = evaluate_h(h)
                    if new_c5 < orig_c5:
                        orig_c5 = new_c5
                        improved = True
                    else:
                        h[i] = 1
                        h[j] = 0
                elif h[i] <= 0 and h[j] >= 1:
                    # Try swap
                    h[i] = 1
                    h[j] = 0
                    new_c5 = evaluate_h(h)
                    if new_c5 < orig_c5:
                        orig_c5 = new_c5
                        improved = True
                    else:
                        h[i] = 0
                        h[j] = 1
    return h


def run(seed=42, budget_s=600):
    """Main run function."""
    global start_time
    start_time = time.time()

    # Tweakable parameters
    def burst_parameters(n):
        """Define algorithm parameters based on n."""
        return {
            'max_iters': 30000,
            'initial_lr': 0.5,
            'alpha_decay': 0.0005,
            'temp_init': 1.0,
            'temp_decay': 0.995
        }

    # Part 1: Try to find solution with exact program
    h_values = np.array(initial_h_values)
    best_h = h_values.copy()
    best_c5 = evaluate_h(best_h)

    # Part 2: Use sub-gradient method as fallback if needed
    attempts = 3

    for attempt in range(attempts):
        seed_val = seed + attempt + 1
        # Current h_start can be from initial or from previous round
        if attempt == 0:
            h_start = h_values.copy()  # From initial_h_values
        elif attempt == 1:
            h_start = generate_random_binary(seed_val, n_points)
        else:
            # Generate Thue-Morse based structure
            h_start = np.zeros(n_points)
            for i in range(n_points):
                h_start[i] = (bin(i).count('1')) % 2
            # Randomly shuffle it for variety
            indices = np.random.permutation(n_points)
            h_start = h_start[indices]

        h_rnd = round_to_binary_with_fractional(h_start)
        h_perturbed = local_improvement(h_rnd)
        c5_rnd = evaluate_h(h_perturbed)
        if c5_rnd < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_rnd

        # Sub gradient with annealing
        p = burst_parameters(n_points)
        h_iter, c5_iter = run_sub_gradient_annealing(
            h_start, p['max_iters'], p['initial_lr'], p['alpha_decay'],
            p['temp_init'], p['temp_decay'], seed_val)

        if c5_iter < best_c5:
            best_c5 = c5_iter
            best_h = h_iter.copy()

        # Early termination check
        if time.time() - start_time > budget_s - 5:
            break

    return list(best_h), best_c5, n_points
