import numpy as np
from cvxpy import *
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # Resolution schedule
    resolutions = [80, 100, 120]
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        # Create and solve the LP
        try:
            # Variables
            x = Variable(n)
            t = Variable(1)
            w = {}  # Dictionary of w_{i,s} variables indexed by (i,s)
            w_vars = []
            
            # Build constraints for McCormick relaxation and shift sums
            constraints = []
            
            # Add McCormick inequalities for each valid (i,s) pair
            for s in range(-(n-1), n):
                for i in range(n):
                    if 0 <= i+s < n:
                        # Create variable w_{i,s} only once per pair
                        name = f'w_{i}_{s}'
                        w[(i, s)] = Variable(1)
                        w_vars.append(w[(i, s)])
                        constraints.append(w[(i, s)] <= x[i])
                        constraints.append(w[(i, s)] <= x[i+s])
                        constraints.append(w[(i, s)] >= x[i] + x[i+s] - 1)
                        constraints.append(w[(i, s)] >= 0)
            
            # Add the constraint sum(w_{i,s}) >= t for all shifts s
            for s in range(-(n-1), n):
                ws_sum = 0
                valid_indices = []
                for i in range(n):
                    if 0 <= i+s < n:
                        valid_indices.append(i)
                        ws_sum += w[(i, s)]
                constraints.append(ws_sum >= t)
            
            # Add mass constraint
            constraints.append(sum(x) == n/2)
            
            # Add bounds
            for i in range(n):
                constraints.append(x[i] >= 0)
                constraints.append(x[i] <= 1)
            
            # Objective: maximize t
            objective = Maximize(t)
            
            # Solve the problem
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, abstol=1e-8, reltol=1e-6)
            
            if prob.status == 'optimal':
                x_opt = np.array([x.value[i] for i in range(n)])
                t_opt = t.value
                
                # Rounding
                x_rounded = np.zeros(n)
                sorted_indices = np.argsort(x_opt)[::-1]
                count = 0
                for i in sorted_indices:
                    if count < n // 2:
                        x_rounded[i] = 1
                        count += 1
                    else:
                        break
                
                # Evaluate the rounded solution
                c5 = evaluate_erdos_solution(x_rounded, dx)
                
                # Local search (swap)
                max_swaps = 20000
                swaps_done = 0
                stagnation = 0
                patience = 2000
                
                while swaps_done < max_swaps and time.time() - start_time < budget_s * 0.95:
                    # Random pick of 1 and 0 position
                    positions_1 = np.where(x_rounded == 1)[0]
                    positions_0 = np.where(x_rounded == 0)[0]
                    if len(positions_1) == 0 or len(positions_0) == 0:
                        break
                    i = np.random.choice(positions_1)
                    j = np.random.choice(positions_0)
                    
                    # Swap and evaluate
                    x_new = x_rounded.copy()
                    x_new[i] = 0
                    x_new[j] = 1
                    new_c5 = evaluate_erdos_solution(x_new, dx)
                    swaps_done += 1
                    
                    if new_c5 < c5:
                        x_rounded = x_new
                        c5 = new_c5
                        stagnation = 0
                    else:
                        stagnation += 1
                        
                    if stagnation >= patience:
                        break
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x_rounded.copy()
                    best_n = n
            else:
                print(f"LP solver failed for n={n}, status: {prob.status}")
        except Exception as e:
            print(f"Error solving LP for n={n}: {e}")
            continue
    
    if best_h is not None:
        return (best_h.tolist(), best_c5, best_n)
