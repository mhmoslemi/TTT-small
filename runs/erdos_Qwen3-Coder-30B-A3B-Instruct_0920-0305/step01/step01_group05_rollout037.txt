```python
import numpy as np
import cvxpy as cp
import time
from scipy.linalg import eigh

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=80 for SDP solve, due to time complexity.
    n_points = 80
    dx = 2.0 / n_points
    n_half = n_points // 2
    
    # Build shift extraction matrices S_s
    def build_shift_matrix(s):
        # Create matrix with ones on diagonal offset s (i, i+s)
        # For s >= 0, this is position (i, i+s) where i+s < n
        if s == 0:
            return np.eye(n_points)
        
        # Create sparse matrix with ones at (i, i+s)
        rows = np.arange(n_points - abs(s))
        cols = np.arange(abs(s), n_points)
        Sr = np.zeros((n_points, n_points))
        if s > 0:
            Sr[rows, cols] = 1
        else:  # s < 0
            Sr[cols, rows] = 1
        return Sr
    
    # Time remaining for all substeps
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # SDP Model (using CVXPY)
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_half,
        cp.diag(X) == h,
        X >> 0,  # PSD
        t >= 0
    ]
    
    # Add McCormick envelopes: for each i < j,
    #    0 <= X[i,j] <= h[i]
    #    0 <= X[i,j] <= h[j]
    #    X[i,j] >= h[i] + h[j] - 1
    for i in range(n_points):
        for j in range(i + 1, n_points):
            constraints.append(X[i, j] >= 0)
            constraints.append(X[i, j] <= h[i])
            constraints.append(X[i, j] <= h[j])
            constraints.append(X[i, j] >= h[i] + h[j] - 1)
    
    # Add overlap constraints
    for s in range(-(n_points - 1), n_points):
        if s == 0:
            continue
        S_s = build_shift_matrix(s)
        # This computes sum of X_{i,i+s} where valid
        # Extracting diagonals directly isn't easy; use trace of S_s @ X
        # Note: in non-symmetric X, we'd use both directions.
        constraints.append(t >= n_half - cp.trace(S_s @ X))
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve the problem
    solve_start = time.time()
    try:
        prob.solve(solver='SCS', eps=1e-5, max_iters=5000)
        if prob.status != cp.OPTIMAL:
            # fallback: try again with looser tolerance
            prob.solve(solver='SCS', eps=1e-4, max_iters=3000)
            if prob.status != cp.OPTIMAL:
                raise RuntimeError("SDP could not solve.")
    except:
        raise RuntimeError("SDP solver failed.")

    solve_time = time.time() - solve_start
    print("SDP solved in", solve_time, "seconds.")

    # Extract solutions
    h_sdp = h.value
    X_sdp = X.value
    t_sdp = t.value

    if h_sdp is None:
        raise RuntimeError("No SDP solution found.")
        
    c5_lb = t_sdp * dx

    # Rounding approach one: Random sampling
    best_h = None
    best_c5 = float('inf')

    # Eigen-decompose to obtain B = V*sqrt(Lambda) such that BB^T = X_sdp
    eigenvals, eigenvecs = eigh(X_sdp)
    pos_eigenvals = eigenvals[eigenvals > 1e-8]
    pos_indices = np.where(eigenvals > 1e-8)[0]
    if len(pos_indices) == 0:
        raise RuntimeError("No positive eigenvalues in rounding matrix")

    V = eigenvecs[:, pos_indices]
    Lambda = np.diag(pos_eigenvals)
    B = V @ np.sqrt(Lambda)

    # Sample multiple vectors from N(0, X_sdp)
    n_samples = min(2000, int(10000 / (n_points * dx)) if n_points * dx > 0 else 500)
    print(f"Sampling {n_samples} vectors for rounding.")

    for sample in range(n_samples):
        if time_left() < 30:
            break

        z = B @ np.random.randn(len(pos_indices))
        z = np.clip(z, 0, 1)
        
        # Normalize to sum = n_half
        ssum = np.sum(z)
        if ssum > 0:
            z = z * n_half / ssum
            z = np.clip(z, 0, 1)
            
            # Final adjustment to maintain exact sum
            ssum = np.sum(z)
            if abs(ssum - n_half) > 1e-6:
                diff = n_half - ssum
                # Adjust – add diff to smallest indices
                indices = np.argsort(z)
                i = 0
                while abs(diff) > 1e-10 and i < len(z):
                    idx = indices[i]
                    if z[idx] < 1:
                        to_change = min(abs(diff), 1 - z[idx])
                        z[idx] += to_change
                        diff += to_change
                    i += 1
                if abs(diff) > 1e-10:
                    # Resort to just sum correction
                    z = z * n_half / np.sum(z)
        else:
            z = np.ones_like(z) * 0.5 * n_half / n_points

        if np.any(z < 0) or np.any(z > 1):
            z = np.clip(z, 0, 1)
            z = z * n_half / np.sum(z)

        c5 = evaluate_h(z)
        if c5 < best_c5:
            best_c5 = c5
            best_h = z.copy()

    # Deterministic rounding as fallback (if need be)
    if best_h is None:
        # Sort indices by h_sdp value, set top n_half to 1, rest to 0
        indices_sorted = np.argsort(h_sdp)[::-1]
        h_det = np.zeros(n_points)
        for i in range(n_half):
            h_det[indices_sorted[i]] = 1.0
        # Enforce integral constraint
        ssum = np.sum(h_det)
        if abs(ssum - n_half) > 1e-6:
            # Try local swaps if balance off
            while abs(np.sum(h_det) - n_half) > 1e-8:
                # find a 1 and a 0 to swap
                ones = np.where(h_det == 1)[0]
                zeros = np.where(h_det == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i1 = ones[0]
                i0 = zeros[0]
                h_det[i1] = 0
                h_det[i0] = 1
        best_c5 = evaluate_h(h_det)
        best_h = h_det

    # Local refinement: hill climber
    h_refined = best_h.copy()
    local_start = time.time()
    max_iter = 5000
    for iter in range(max_iter):
        if time_left() < 10:
            break
        # Pick a 0 index and 1 index, swap
        ones = np.where(h_refined == 1)[0]
        zeros = np.where(h_refined == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i0 = np.random.choice(zeros)
        i1 = np.random.choice(ones)
        # Flip
        h_temp = h_refined.copy()
        h_temp[i0] = 1
        h_temp[i1] = 0
        c5_temp = evaluate_h(h_temp)
        if c5_temp < best_c5:
            best_c5 = c5_temp
            h_refined = h_temp

    # Final check
    final_c5 = evaluate_h(h_refined)
    
    return list(h_refined), final_c5, n_points
```
