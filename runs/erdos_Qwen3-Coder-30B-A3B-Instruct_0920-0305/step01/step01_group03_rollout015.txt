def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h <= 1, sum(h) = target_sum"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Compute current sum
    current_sum = np.sum(h)
    # If already correct, return
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Water-filling algorithm  
    # Sort in descending order
    h_sorted = np.sort(h)[::-1]
    # Find the breakpoint
    cumsum = np.cumsum(h_sorted)
    n_break = 0
    for i in range(len(h_sorted)):
        if h_sorted[i] * (i+1) - cumsum[i] < target_sum - cumsum[i]:
            n_break = i+1
        else:
            break
    # Compute threshold
    if n_break == 0:
        gamma = target_sum / len(h)
    else:
        gamma = (target_sum - cumsum[n_break-1]) / n_break
    # Apply threshold
    h_proj = np.clip(h - gamma, 0, 1)
    return h_proj

def symmetrize_and_project(h):
    """Symmetrize h around midpoint and reproject"""
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    h_proj = project_to_capped_simplex(h_sym, n/2)
    return h_proj
