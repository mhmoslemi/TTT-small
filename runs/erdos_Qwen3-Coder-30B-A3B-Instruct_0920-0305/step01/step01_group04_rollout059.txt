import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time

# Constants
n_points = 47  # Using the provided value from initial_h_values
dx = 2.0 / n_points
target_bound = 0.3807

def compute_correlations(h):
    """Compute full correlations of h with (1 - h)."""
    corr = np.correlate(h, 1 - h, mode="full")
    # Take only the valid range of correlations (excluding padding)
    mid_idx = len(corr) // 2
    corr = corr[mid_idx - (n_points - 1):mid_idx + n_points]
    return corr

def compute_C5(h_vals):
    """Compute C5 bound for given h values."""
    h = np.array(h_vals)
    corr = compute_correlations(h)
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_h_from_pattern(initial_h_values, n_points):
    """Initialize h to match the structure of initial_h_values."""
    h = np.array(initial_h_values)
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=0)
    elif len(h) > n_points:
        h = h[:n_points]
    # Scale to meet integral constraint
    integral = np.sum(h) * dx
    if abs(integral - 1.0) > 1e-6:
        scale = 1.0 / integral
        h *= scale
    return h

def solve_sdp(n0):
    """Solve SDP relaxation for binary sequence of length n0."""
    # Build Toeplitz matrices C_d for d in {1,2,...,n0-1} with circular wraparound
    C_matrices = []
    for d in range(1, n0):  # Skip d=0
        C = np.zeros((n0, n0))
        for i in range(n0):
            j = (i + d) % n0
            C[i, j] = 1
            C[j, i] = 1
        C_matrices.append(C)

    # SDP variables
    X = cp.Variable((n0, n0), symmetric=True)
    tau = cp.Variable()

    # SDP constraints
    constraints = [X >> 0]  # X is positive semidefinite
    constraints.append(cp.diag(X) == 1)  # Diagonal elements are 1
    for C in C_matrices:
        constraints.append(cp.trace(C @ X) >= tau)

    # Objective: maximize tau
    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)

    # Solve the SDP
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
        if prob.status in ["optimal", "approximal_optimal"]:
            return X.value, tau.value
        else:
            return None, None
    except:
        return None, None

def round_sdp_solution(X_sol, n0):
    """Use hyperplane rounding to get binary sequence from SDP solution."""
    if X_sol is None:
        return None, None
    # Perform eigenvalue decomposition
    try:
        vals, vecs = eigh(X_sol)
        # Keep components with sufficient eigenvalues
        threshold = 1e-8
        valid_idx = vals > threshold
        if not np.any(valid_idx):
            return None, None
        rank = np.sum(valid_idx)
        V = vecs[:, valid_idx]
        vals = vals[valid_idx]
        # Normalize columns
        V = V @ np.diag(np.sqrt(vals))
        # Generate random Gaussian vector
        g = np.random.randn(rank)
        # Compute s = sign(V^T g)
        s = np.sign(V.T @ g)
        # Fix sign flip to make sum close to zero
        total_s = np.sum(s)
        if abs(total_s) > 1e-3:
            # Find the entry with smallest absolute value to flip
            abs_s = np.abs(s)
            flip_idx = np.argmin(abs_s)
            s[flip_idx] *= -1
        # Convert to h
        h = (s + 1) / 2
        return h, s
    except Exception:
        return None, None

def scale_and_refine_pattern(h, n_points):
    """Scale up the pattern or do final refinement."""
    # Interleave with complement
    h_double = np.kron(h, [1, 0])
    h_double = np.concatenate([h_double, 1 - h_double[:len(h_double)]])
    
    # Trim to n_points
    h_final = h_double[:n_points]
    
    # Fix weight constraint
    actual_sum = np.sum(h_final)
    desired_sum = n_points / 2
    diff = actual_sum - desired_sum
    
    if abs(diff) > 1e-6:
        # Adjust slightly
        if diff > 0:
            # Need to reduce some 1s to 0s
            indices = np.where(h_final == 1)[0]
            if len(indices) > 0:
                # Flip smallest indices with high value
                idx = indices[0]
                h_final[idx] = 0
        else:
            # Need to increase some 0s to 1s
            indices = np.where(h_final == 0)[0]
            if len(indices) > 0:
                idx = indices[0]
                h_final[idx] = 1
                
    return h_final

def evaluate_pattern(h_vals):
    """Compute C5 from pattern."""
    return compute_C5(h_vals)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Step 1: Start from initial construction
    initial_h = initialize_h_from_pattern(initial_h_values, n_points)
    current_h = initial_h.copy()
    current_c5 = evaluate_pattern(current_h)
    
    # Initialize best solution
    best_h = current_h.copy()
    best_c5 = current_c5

    # Step 2: Try SDP approach
    n0 = 59  # Using a prime close to 61 for better structure
    try:
        X_sol, tau_opt = solve_sdp(n0)
        if X_sol is not None:
            for _ in range(200):  # 200 rounding attempts
                h_rounded, s = round_sdp_solution(X_sol, n0)
                if h_rounded is not None:
                    # Scale and adjust for the fixed weight
                    h_rounded = (h_rounded + 1)/2
                    integral = np.sum(h_rounded)
                    scale_factor = n0 / (2 * integral)
                    h_rounded *= scale_factor
                    # Convert to proper length (interleave, scale)
                    scaled_h = scale_and_refine_pattern(h_rounded, n_points)
                    # Evaluate
                    c5 = evaluate_pattern(scaled_h)
                    if c5 < best_c5:
                        best_h = scaled_h.copy()
                        best_c5 = c5
    except Exception as e:
        pass  # Continue with fallback

    # Final local refinement
    refined = best_h.copy()
    best_c5_refined = best_c5
    
    # Run basic hill climbing heuristic locally
    # Only swaps that strictly improve the bound
    for _ in range(1000):  # Limited number of swaps
        if time.time() - start_time > budget_s - 2:
            break
        improved = False
        for i in range(n_points):
            for j in range(i+1, n_points):
                # Only consider 0->1 and 1->0 swaps
                if (refined[i] == 0 and refined[j] == 1) or (refined[i] == 1 and refined[j] == 0):
                    # Try swapping
                    temp_h = refined.copy()
                    temp_h[i], temp_h[j] = temp_h[j], temp_h[i]
                    temp_c5 = compute_C5(temp_h)
                    if temp_c5 < best_c5_refined:
                        refined = temp_h
                        best_c5_refined = temp_c5
                        improved = True
                        break
            if improved:
                break
                
    # Ensure final constraint still met
    integral = np.sum(refined) * dx
    if abs(integral - 1.0) > 1e-6:
        corrected = refined * (1.0 / integral)
        corrected_c5 = compute_C5(corrected)
        if corrected_c5 < best_c5_refined:
            refined = corrected
            best_c5_refined = corrected_c5

    return list(refined), best_c5_refined, n_points
