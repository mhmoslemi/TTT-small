def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    # Sort values in descending order
    sorted_indices = np.argsort(h)[::-1]
    sorted_h = h[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_h)
    n = len(sorted_h)
    
    # Find threshold
    threshold_idx = -1
    for i in range(n):
        if sorted_h[i] - (cumsum[i] - sorted_h[i]) / (i + 1) > 0:
            threshold_idx = i
        else:
            break
    
    if threshold_idx == -1:
        # All values are too small
        tau = 0.0
    else:
        # Compute tau such that sum is target_sum
        tau = (cumsum[threshold_idx] - target_sum) / (threshold_idx + 1)
        # Clip values above 1
        tau = max(0, tau)
    
    # Apply projection
    projected = np.maximum(0, np.minimum(1, h - tau))
    
    # Ensure sum constraint
    actual_sum = np.sum(projected)
    if abs(actual_sum - target_sum) > 1e-10:
        # Adjust by linear interpolation
        adjustment = (target_sum - actual_sum) / n
        projected += adjustment
    
    return projected

def softmin_surrogate(h, R, lam):

    exp_vals = np.exp(-lam * (R - max_R))
    Z = np.sum(exp_vals)
    return (1/lam) * (np.log(Z) + max_R)
