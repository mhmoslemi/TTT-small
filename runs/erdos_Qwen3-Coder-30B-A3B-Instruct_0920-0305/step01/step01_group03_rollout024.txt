def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def logsumexp_stable(x, axis=None):
    """Numerically stable computation of log(sum(exp(x)))"""
    x_max = np.max(x, axis=axis, keepdims=True)
    return x_max + np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=True))

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex (box + mass constraint)"""
    h = np.array(h)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Use binary search to find the right tau
    def sum_clamped(tau):
        return np.sum(np.clip(h + tau, 0, 1))
    
    # Binary search for tau such that sum equals target_sum
    low, high = -10.0, 10.0
    tol = 1e-12
    for _ in range(50):
        tau = (low + high) / 2
        current_sum = sum_clamped(tau)
        if current_sum < target_sum:
            high = tau
        else:
            low = tau
        if abs(current_sum - target_sum) < tol:
            break
    
    h = np.clip(h + tau, 0, 1)
    return h

def symmetric_project(h):
    """Symmetrize and project to exact mass"""
    n = len(h)
    # Symmetrize around middle
    h_sym = np.zeros_like(h)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
    # Project to exact mass
    target_sum = n / 2
    h_sym = project_to_capped_simplex(h_sym, target_sum)
    return h_sym

def compute_gradient(h, weights):
    """Compute gradient g_j = sum_s w_s(h_{j+s} + h_{j-s})"""
    n = len(h)
    g = np.zeros_like(h)
    # Ensure zero-padding when indexes go out of bounds
    for s in range(-n+1, n):
        if s >= 0:
            g += weights[s] * h[s:]
        else:
            g += weights[s] * h[:s]
    return g

def backtracking_line_search(h, grad, alpha_max, surrogate_func, lambda_param):
    """Perform backtracking line search on step size"""
    c1 = 0.5  # Armijo condition parameter
    rho = 0.5  # Reduction factor
    
    alpha = alpha_max
    h_raw = h + alpha * grad
    
    # Project to feasible set
    h_proj = project_to_capped_simplex(h_raw, len(h) / 2)
    
    # Compute current surrogate
    phi_old = surrogate_func(h)
    phi_new = surrogate_func(h_proj)
    
    # Check Armijo condition: phi_new <= phi_old + c1 * alpha * grad.T * grad
    # But since this is not the actual gradient, use a simpler check
    if phi_new < phi_old:
        return h_proj, alpha
    
    # Backtrack
    for _ in range(5):
        alpha *= rho
        h_raw = h + alpha * grad
        h_proj = project_to_capped_simplex(h_raw, len(h) / 2)
        phi_new = surrogate_func(h_proj)
        if phi_new < phi_old:
            return h_proj, alpha
    
    return h_proj, alpha

def surrogate_objective(h, lambda_param):
    """Compute the surrogate log-sum-exp of correlations"""
    n = len(h)
    dx = 2.0 / n
    # Compute the correlation R(s)
    R = np.correlate(h, h, mode="full")  # This is R(s) for s = -n+1 to n-1
    # Log-sum-exp with log space computation
    log_weights = -lambda_param * R
    log_weights -= np.max(log_weights)  # Stability trick
    log_sum_exp = np.log(np.sum(np.exp(log_weights))) + np.max(log_weights)
    return log_sum_exp / lambda_param
