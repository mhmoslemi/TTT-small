import numpy as np
from scipy.signal import correlate
import time

def project_box_simplex(y, target_sum):
    """Project vector y onto the box-constrained simplex {x : 0 <= x <= 1, sum(x) = target_sum}"""
    # Sort y in descending order, compute cumulative sums
    sorted_y = np.sort(y)[::-1]
    cumsum = np.cumsum(sorted_y)
    n = len(y)
    
    # Find the point where the sum constraint is active
    # We want: sorted_y[i] * (i+1) - cumsum[i] <= target_sum - cumsum[i]
    # So: sorted_y[i] * (i+1) <= target_sum
    # And we want the largest i where this holds
    # i+1 is the number of elements [0..i] that should be fixed at their current value
    # The rest should be adjusted to equal alpha
    
    # Compute alpha in a way that the sum hits target_sum
    # If we have i+1 values equal to sorted_y[i], they sum to (i+1)*sorted_y[i]  
    # To make up the difference, [i+1, n) values are set to alpha to make the sum = target_sum
    # That gives: (i+1)*sorted_y[i] + (n-i-1)*alpha = target_sum
    # So alpha = (target_sum - (i+1)*sorted_y[i]) / (n-i-1)
    
    # If sorted_y[i] = 0, we're done with that part, so ignore if needed
    
    # Use bisection to find the correct alpha
    # Start with the fact that alpha decreases with increasing i 
    # Let's write a helper to avoid implementing the complex search manually
    # For simplicity: use projection by subgradient method with limited iterations
    
    # In practice, a binary search approach or root-finding is more robust.
    # Let's define a function that computes sum after clipping at some alpha.
    def compute_sum_after_clipping(alpha):
        clipped = np.clip(y - alpha, 0, 1)
        return np.sum(clipped)
    
    # Try using bracketing search for alpha
    max_val = np.max(y)
    min_val = np.min(y)
    # Iterate carefully using golden-section or binary search for finding the best alpha
    
    # Let's use bisection directly:
    low = min_val - 1
    high = max_val + 1
    
    # Simplify using a more direct yet robust method:
    # iterate to find the correct threshold alpha such that sum equals target
    # This part uses a Newton-like approach in a restricted problem.
    # However for readability we'll do a safe brute-force on a coarse grid in final fallback.
    
    def get_projection(y, target_sum):
        # Use root finding approach for simplicity and stability.
        # We solve for the scalar t such that sum(clamp(y - t, 0, 1)) = target_sum
        
        def f(alpha):
            return np.sum(np.clip(y - alpha, 0, 1)) - target_sum
            
        # Bracket with binary search
        lower = np.min(y) if np.min(y) > 0 else -100
        upper = np.max(y) if np.max(y) < 1 else 100
        
        for _ in range(50):  # Max iterations
            alpha = (lower + upper) / 2
            val = f(alpha)
            if abs(val) < 1e-10:
                break
            if val > 0:
                lower = alpha
            else:
                upper = alpha
        clipped = np.clip(y - alpha, 0, 1)
        return clipped
    
    return get_projection(y, target_sum)

def compute_fs(h, s):
    """Compute f_s(h) = sum_i h_i * (1 - h_{i+s}) with periodic padding renamed to avoid issues"""
    n = len(h)
    if s >= 0:
        # Right shift h by s
        shifted_h = np.pad(h[s:], (0, s), mode='constant', constant_values=0)
    else:
        # Left shift h by -s
        shifted_h = np.pad(h[:s], (-s, 0), mode='constant', constant_values=0)
    return np.dot(h, 1 - shifted_h)

def compute_all_fs(h):
    """Compute all f_s for s = -(n-1) to n-1."""
    n = len(h)
    f_vals = []
    # Shifts range from -(n-1) to n-1
    for s in range(-(n - 1), n):
        fs = compute_fs(h, s)
        f_vals.append(fs)
    return np.array(f_vals)

def compute_gradient(h, p, lambda_sum):
    """Compute approx gradient of Phi_p with penalty term."""
    n = len(h)
    f_vals = compute_all_fs(h)
    # Compute w_s = f_s^{p-1}
    w_vals = np.power(np.abs(f_vals), p - 1)
    
    # Compute gradient of smooth part
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        fs = f_vals[s + (n - 1)]
        w = w_vals[s + (n - 1)]
        # The derivative of f_s w.r.t. h_k is based on:
        # f_s = sum_k h_k (1 - h_{k+s})
        # d/dh_k f_s = 1 - h_{k+s} - h_{k-s} ?
        if s > 0:
            h_shifted = np.concatenate([h[s:], np.zeros(s)])
        elif s < 0:
            h_shifted = np.concatenate([np.zeros(-s), h[:s]])
        else:
            h_shifted = h
            
        # h_{k-s} is h_shifted[k]   
        # h_{k+s} is h_shifted[k]
        # This is:
        # d/dh_k sum_i h_i (1 - h_{i+s}) = 1 - h_{k+s} - h_{k-s} if both defined
        # if h_{k+s} is out of bounds or l.h.s., then it's just -h_{k-s}
        # if h_{k-s} is out of bounds or r.h.s., then it's just -h_{k+s}
        
        # For clarity, replace with proper use of indices
        # But, let's use the form from the math:
        # f_s(h) = h^T (c_bar shift_s) where c_bar = 1-h
        # derivative w.r.t. h_k is product rules:
        # 1 - h_{k+s} - h_{k-s}
        # But, when indices are out of range, we use 0 for h (which is equivalent to invalid)
        # This needs to be computed around the k-th entry
        
        # Compute derivative efficiently with array operations
        for i in range(n):
            val = 1.0
            # h_{i+s}, h_{i-s}
            if (i + s) < n and (i + s) >= 0:
                val -= h[i + s]
            if (i - s) < n and (i - s) >= 0:
                val -= h[i - s]
            grad[i] += w * val
        
    grad /= (np.sum(np.power(np.abs(f_vals), p)))**(1.0 - 1.0/p) if np.sum(np.power(np.abs(f_vals), p)) > 1e-10 else 1.0
    
    # Add penalty term (sum constraint)
    penalty = 2 * lambda_sum * (np.sum(h) - n/2)
    grad += penalty
    
    return grad

def smooth_p_norm_minimize(h0, max_iter=2000, p_vals=[2, 4, 8, 12], lambda_sum=1e3):
    """Minimise using projected gradient descent with a smooth approximation to max."""
    h = h0.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    smoothed_max = np.inf
    
    for p in p_vals:
        # Reset for new p
        for iteration in range(max_iter):
            if iteration % 500 == 0:
                f_vals = compute_all_fs(h)
                c5 = np.max(np.abs(f_vals)) * dx
                if c5 < smoothed_max:
                    smoothed_max = c5
            
            g = compute_gradient(h, p, lambda_sum)
            # Step size empirically tuned
            step_size = 1e-3  # Could be adaptive but let's stick to simple
            h_new = h - step_size * g
            # Project back onto feasible set
            h_new = project_box_simplex(h_new, n/2)
            # Ensure bounds to prevent numerical issues
            h_new = np.clip(h_new, 0, 1)
            
            # Check for convergence:
            diff = np.linalg.norm(h - h_new)
            if diff < 1e-8:
                break
            h = h_new
            
    return h

    # Normalize initial h by hand
    dx = 2.0 / len(initial_h_values)
    h_original = np.array(initial_h_values)
    scaled = h_original * (1.0 / (np.sum(h_original) * dx))
    h_init = scaled

    # Different resolutions to do coarse-to-fine iteration
    n_vals = [60, 100, 150]

    h_best = None
    c5_best = np.inf
    best_n = 42

    # Stage A, B, C with multi-resolution coarse to fine from 60 to 150
    current_h = h_init
    for i, n in enumerate(n_vals):
        if elapsed() > budget_s - 10:
            break

        # Interpolate from previous grid to next if needed
        if i > 0:
            old_vals = current_h
            x_old = np.linspace(0, 2, len(old_vals))
            x_new = np.linspace(0, 2, n)
            current_h = np.interp(x_new, x_old, old_vals)
            # Ensure sum constraint holds
            current_h *= (1.0 / (np.sum(current_h) * dx)) * (n/2)
        else:
            # Use existing initial
            current_h = h_init.copy()
            # Re-normalize for n != original n
            if n != len(h_init):
                current_h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_init)), h_init)
            
        # Normalize current_h to have integral = 1 (according to dx)
        dx_current = 2.0 / n
        current_h = current_h * (1.0 / (np.sum(current_h) * dx_current))
        current_h = np.clip(current_h, 0, 1)
        
        # Do projected gradient optimization on current resolution
        try:
            optimized_h = smooth_p_norm_minimize(current_h, max_iter=1500, p_vals=[2, 4, 8, 12], lambda_sum=1e3)
        except Exception:
            continue
            
        # Apply exact normalization (very important for integer shift to be accurate)
        norm_factor = (1.0 / (np.sum(optimized_h) * dx_current))
        optimized_h = optimized_h * norm_factor
        
        # Evaluate result
        try:
            corr = np.correlate(optimized_h, 1 - optimized_h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx_current
            
            if c5 < c5_best:
                c5_best = c5
                h_best = optimized_h.copy()
                best_n = n
        except Exception:
            continue
        
    # Check if we have time left for refinement.
    remaining_time = budget_s - elapsed()
    if remaining_time > 20:

        # Binary hill climbing
        h_refined = h_best.copy()
        dx_refined = 2.0 / best_n
        
        try:
            # Binary approach: round and refine a bit
            # Sort indices by value to identify "big" entries
            sorted_idx = np.argsort(h_refined)[::-1]
            h_bin = h_refined.copy()
            
            # move values to be 0 or 1 according to their magnitude and the constraint
            # Take first half to be 1 and rest to be 0 after sorting, but ensure sum = n/2
            # number of 1's needed is n/2
            
            n_half = int(best_n / 2)
            h_bin = np.zeros_like(h_refined)
            h_bin[sorted_idx[:n_half]] = 1.0
            
            # Scale to meet volume
            sum_bin = np.sum(h_bin)
            h_bin_scaled = h_bin * (n_half / sum_bin)
            
            # Now compute the overlap for this binary vector:
            corr_bin = np.correlate(h_bin_scaled, 1 - h_bin_scaled, mode="full")
            max_corr_bin = np.max(np.abs(corr_bin))
            c5_bin = max_corr_bin * dx_refined
            
            if c5_bin < c5_best:
                c5_best = c5_bin
                h_best = h_bin_scaled.copy()
        except Exception:
            pass

    # Final safeguard
    corr = np.correlate(h_best, 1 - h_best, mode="full")
    max_corr = np.max(np.abs(corr))
    c5_final = max_corr * (2.0 / best_n)

    return (h_best.tolist(), c5_final, best_n)
