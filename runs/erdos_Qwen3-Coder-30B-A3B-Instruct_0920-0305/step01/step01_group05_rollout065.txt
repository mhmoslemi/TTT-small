def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def solve_ccp_subproblem(h, shifts, shift_masks, n):

    h_prev = h.copy()

    tau_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # For each shift update the concave term linearization
    for s in shifts:
        mask_fwd = shift_masks[s][0]
        mask_bwd = shift_masks[s][1]
        
        # Evaluate the gradient at h_prev
        grad_d = np.zeros_like(h_prev)
        for i in range(n):
            total = 0
            if 0 <= i+s < n:
                total += h_prev[i] + h_prev[i+s]
            if 0 <= i-s < n:
                total += h_prev[i-s] + h_prev[i]
            grad_d[i] = total
            
        # Compute g^d(h_prev) and linearize the concave part
        g_d_prev = 0.5 * sum((h_prev[i] + h_prev[i+s])**2 for i in range(n) if 0 <= i+s < n)
        linear_term = g_d_prev + np.dot(grad_d, h_var - h_prev)
        
        # We have to evaluate g^c(h) and ensure that g^c(h) - linear_term <= tau_var
        # But both g^c and linear_term are convex functions

        g_c = cp.sum(h_var) + 0.5 * cp.sum_squares(h_var[mask_fwd]) + 0.5 * cp.sum_squares(h_var[mask_bwd])

    # Minimize tau var subject to constraints
    prob = cp.Problem(cp.Minimize(tau_var), constraints)
