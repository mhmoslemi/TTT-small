import numpy as np

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def solve_ccp(h_start, n, start_time, budget_s):
    dx = 2.0 / n
    # Prefetch masks for shifts
    shift_masks = {}
    for s in range(-(n-1), n):
        mask_fwd = np.arange(n) + s < n
        mask_bwd = np.arange(n) - s >= 0
        shift_masks[s] = [mask_fwd, mask_bwd]
    
    # Define CVXPY variables and constraints
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Build constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]
    
    # Evaluate initial value
    c5_initial = evaluate_h(h_start)
    
    for iter_count in range(30):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Evaluate at current point
        h_current = h_start.copy()
        g_d_vals = np.zeros(n)
        grad_g_d = np.zeros(n)
        for s in range(-(n-1), n):
            mask_fwd, mask_bwd = shift_masks[s]
            # Sum of non-zero terms in g_d
            idx_fwd = np.where(mask_fwd)[0]
            idx_bwd = np.where(mask_bwd)[0]
            if len(idx_fwd) > 0:
                g_d_vals[s+n-1] = np.sum((h_current[idx_fwd] + h_current[idx_fwd-s])**2) * 0.5
            if len(idx_bwd) > 0:
                g_d_vals[s+n-1] = np.sum((h_current[idx_bwd] + h_current[idx_bwd+s])**2) * 0.5
            
            # Compute gradient
            grad_val = np.zeros(n)
            for i in range(n):
                if i+s < n and i+s >= 0:
                    grad_val[i] += (h_current[i] + h_current[i+s])
                if i-s >= 0 and i-s < n:
                    grad_val[i] += (h_current[i-s] + h_current[i])
            
            grad_g_d += grad_val
            
        # Build constraints
        constraints = [0 <= h, h <= 1, cp.sum(h) == n/2]
        c5_vals = []
        for s in range(-(n-1), n):
            mask_fwd, mask_bwd = shift_masks[s]
            # g_s^c = sum(h) + 0.5 * sum(h_i^2 + h_{i+s}^2)
            g_c = cp.sum(h) + 0.5 * cp.sum_squares(h[mask_fwd]) + 0.5 * cp.sum_squares(h[mask_bwd])
            # linear approximation of g_s^d
            g_d_at_prev = g_d_vals[s+n-1]
            grad_at_prev = grad_g_d  # Not exact, but okay for local approx
            linearized = g_d_at_prev + grad_at_prev.T @ (h - h_start)
            constraints.append(tau >= g_c - linearized)
            c5_vals.append(g_c - linearized)
        
        # Solve the convex optimization problem
        objective = cp.Minimize(tau)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iters=1000)
        except:
            # Fallback to basic version if ECOS fails
            break
            
        if prob.status != 'optimal':
            break
            
        # Extract solution
        if np.isnan(h.value).any() or np.isinf(h.value).any() or h.value is None:
            break
            
        h_new = h.value.copy()
        # Adjust to keep integral constraint exactly
        h_new *= (n/2) / np.sum(h_new)
        h_start = h_new
        
        # Evaluate actual C5
        c5_new = evaluate_h(h_start)
        if abs(c5_new - c5_initial) < 1e-6:
            break
        c5_initial = c5_new
    
    return h_start, c5_initial

def coordinate_descent(h_start, n, start_time, budget_s):
    dx = 2.0 / n
    max_it = 5  # Number of sweeps
    h_new = h_start.copy()
    
    for sweep in range(max_it):
        if time.time() - start_time > budget_s - 2:
            break
            
        idx_order = np.arange(n)
        np.random.shuffle(idx_order)
        
        improve = True
        while improve:
            improve = False
            for i in idx_order:
                if time.time() - start_time > budget_s - 2:
                    break
                
                # Current h values
                h_current = h_new.copy()
                old_h_i = h_current[i]
                
                # Define shifting for i to find active max
                max_val = float('inf')
                best_h_i = old_h_i
                
                # Compute overlapping terms f_s(h) for all s for current h
                c5_vals = []
                for s in range(-(n-1), n):
                    val = 0.
                    for j in range(n):
                        j_shift = j + s
                        if 0 <= j_shift < n:
                            val += h_current[j] * (1 - h_current[j_shift])
                    c5_vals.append(val)
                
                # Considering only the active maximizers can reduce search space
                active_indices = []
                max_actual = max(c5_vals)
                for idx, val in enumerate(c5_vals):
                    if abs(val - max_actual) < 1e-8:
                        active_indices.append(idx)
                
                # For i, compute f_s(h) as a function of h_i only
                for s in active_indices:
                    # If we are shifting by s, the part that depends on h_i becomes: 
                    # h_j * (1 - h_{j+s}) where j corresponds to i (or differing field)
                    # This becomes a quadratic in h_i which can be minimized easily
                    base_idx = i
                    shifted_idx = i + s if 0 <= i + s < n else -1
                    
                    if 0 <= shifted_idx < n:
                        # h_i * (1 - h_{i+s})
                        # We are changing h_i, so h_{i+s} stays fixed
                        # Need to compute d / dh_i f_s
                        a = 0.0  # quadratic term
                        b = 1.0 - h_current[shifted_idx]  # linear term  
                        c = -h_current[shifted_idx]  # constant term
                        # f_s = h_i * (1 - h_{i+s}) = h_i - h_i * h_{i+s}
                        # minimizing w.r.t. h_i gives point at -b / 2a but a=0 => constant...
                        # The actual structure is we fix everything else, the term is:
                        # h_i * (1 - h_{i+s})
                        # So as quadratic in h_i: f_s = (1 - h_{i+s}) * h_i + 0 * h_i^2
                        # Minimizing is the same as maximizing when coefficient of h_i is negative (h_{i+s} > 1)  
                        
                        # We must evaluate all shift terms as quadratic expressions in h_i
                        # and get min/max; then choose the best one
                        c5c = c5_vals[s+n-1]  # This was the computation passed through for shift s
                        # Simplification: shift s contributes h_i * (1 - h_{i+s}) to f_s
                        # Therefore, the change of h_i affects term:  
                        # (1 - h_{i+s}) * h_i -> so derivative is (1 - h_{i+s})
                        # For fixed h_{i+s}, min/max over h_i
                        new_h = 1.0 - h_current[shifted_idx]
                        # This means f_s is linear in h_i  
                        # We want to minimize the overlap, so if new_h > 0, we minimize h_i to 0,
                        # if new_h < 0, we maximize h_i to 1
                        new_h_i = max(0.0, min(1.0, round(h_current[i] + 0.005 * new_h)))
                        if new_h_i != h_current[i]:
                            h_current[i] = new_h_i
                            new_c5 = evaluate_h(h_current)
                            if new_c5 < max_val:
                                max_val = new_c5
                                best_h_i = new_h_i
                            h_current[i] = old_h_i
    
    # Final evaluation
    final_h = h_new.copy()
    final_c5 = evaluate_h(final_h)
    return final_h, final_c5

def binary_hill_climb(h_start, n, start_time, budget_s):
    h = h_start.copy()
    # Find the n/2 indices with highest values
    idx = np.argsort(h)[::-1]
    selected = idx[:n//2]
    binary_h = np.zeros(n)
    binary_h[selected] = 1
    
    # Evaluate C5
    c5 = evaluate_h(binary_h)
    best_h = binary_h.copy()
    best_c5 = c5
    
    # Simulated annealing-like hill climbing on swaps
    max_iter = 10000
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Randomly choose two indices
        i1, i2 = np.random.choice(n, size=2, replace=False)
        if i1 in selected:
            new_selected = selected.copy()
            new_selected[new_selected == i1] = i2
        else:
            # i1 not in selected; swap with another not in selected
            new_selected = selected.copy()
            # Get index of i1 in current selected
            idx1 = np.where(new_selected == i1)[0][0]
            new_selected[idx1] = i2
            
        # Check integral constraint
        new_binary_h = np.zeros(n)
        new_binary_h[new_selected] = 1
        new_c5 = evaluate_h(new_binary_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_binary_h.copy()
            selected = new_selected.copy()
            
    return best_h, best_c5
