    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)

        h = h * (1.0 / (np.sum(h) * dx))

    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
