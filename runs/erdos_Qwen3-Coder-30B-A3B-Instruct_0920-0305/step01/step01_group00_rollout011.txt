import numpy as np
import time

def evaluate_c5(h):
    """Evaluates the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def propose_move(h, n):
    """Proposes a move modifying h at two random indices to maintain sum constraint"""
    i, j = np.random.choice(n, size=2, replace=False)
    # Compute allowable change to maintain bounds
    max_up_i = 1.0 - h[i]
    max_down_i = h[i]
    max_up_j = 1.0 - h[j]
    max_down_j = h[j]
    delta_low = max(-max_down_i, -max_up_j)
    delta_high = min(max_up_i, max_down_j)
    if delta_low >= delta_high:
        return None
    delta = np.random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, delta, i, j

    current_c5 = evaluate_c5(current_h)

    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 0.01

    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 0.1:

        new_h, _, _, _ = move_result

        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):

            current_c5 = new_c5
            
            # Update best if improved
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        iter_count += 1
        T = max(1e-10, T * alpha)
        
    return (best_h.tolist(), best_c5, n_points)
