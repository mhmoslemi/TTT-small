import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.fft import ifft, fft

def project_simplex(y, target_sum):
    """Project vector y onto the capped simplex: sum = target_sum, 0 <= x <= 1"""
    # Binary search for the threshold lambda
    l, u = np.min(y), np.max(y) + 1.0
    for _ in range(100):
        lamb = (l + u) / 2
        x = np.clip(y - lamb, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            l = lamb
        else:
            u = lamb
    lamb = (l + u) / 2
    x = np.clip(y - lamb, 0, 1)
    # Adjust to match target sum exactly
    scale = target_sum / np.sum(x)
    x *= scale
    return x

def compute_correlation(h):
    """Compute the correlation vector for full mode (no scaling)"""
    h_pad = np.concatenate([h, np.zeros_like(h)])
    h_conj = np.conj(np.concatenate([h, np.zeros_like(h)]))
    corr = ifft(fft(h_pad) * fft(h_conj)).real
    return corr[:len(h)]  # Only the first n values

def compute_gradient(h, c_s, alpha):
    """Compute gradient of log-sum-exp surrogate for max corr"""
    # Compute softmax weights (numerically stable)
    c_max = np.max(c_s)
    w = np.exp(alpha * (c_s - c_max))
    w = w / np.sum(w)
    
    # Compute gradient contribution from each shift
    grad = np.ones_like(h)
    # Forward and backward shifts
    for s in range(len(h)):
        # Left shift (for i = s, h[i] gets term from h[i-s])
        # For right shift: h[i] gets term from h[i+s] (with zero padding)
        forward_shift = np.roll(h, -s)
        backward_shift = np.roll(h, s)
        # Handle boundary correctly with zero padding by manual slicing
        if s > 0:
            forward_shift[:s] = 0
            backward_shift[-s:] = 0
        else:
            forward_shift[:] = h[:]
            backward_shift[:] = h[:]
        
        grad -= w[s] * (forward_shift + backward_shift)
    
    return grad

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_surrogate(h, alpha):
    """Compute the log-sum-exp surrogate"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    if alpha > 100:
        return c_max
    log_sum = np.log(np.sum(np.exp(alpha * (c - c_max)))) + c_max
    return log_sum / alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid parameters
    n_points = 96
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h_old = np.array(initial_h_values)
        h_resampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_old)), h_old)
        h = h_resampled.copy()
    else:
        h = np.random.rand(n_points)
        # Project onto simplex
        h = project_simplex(h, target_sum)
    
    # Algorithm parameters
    eta0 = 0.5
    alpha_start = 5
    alpha_mult = 1.03
    max_iters = 5000
    alpha_schedule_freq = 200
    
    alpha = alpha_start
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    iter_count = 0
    while iter_count < max_iters and (time.time() - start_time) < budget_s * 0.7:
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        c_s = corr * dx  # This gives all correlation values as in previous code
        
        # Compute gradient
        grad = compute_gradient(h, c_s, alpha)
        
        # Gradient step
        eta = eta0 / np.sqrt(iter_count + 1)
        y = h - eta * grad
        
        # Project onto feasible set
        h = project_simplex(y, target_sum)
        
        # Evaluate objective
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Update alpha
        if iter_count % alpha_schedule_freq == 0 and iter_count > 0:
            alpha *= alpha_mult
            if alpha > 200:
                alpha = 200
        
        iter_count += 1
        if (time.time() - start_time) > budget_s * 0.7:
            break
    
    # Local polishing with SLSQP
    def obj_polish(h):
        return evaluate_c5(h)
    
    def constr_polish(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constr_polish}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj_polish, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            polished_h = res.x
            polished_c5 = obj_polish(polished_h)
            if polished_c5 < best_c5:
                best_h = polished_h
                best_c5 = polished_c5
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
