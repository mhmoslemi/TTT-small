import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

# Constants
n_points = 100  # Small enough for exact solution with CBC
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def round_to_binary(h, n_points):
    """Round h to binary, preserving the integral."""
    h = np.array(h)
    # Make sure sum = n/2
    sum_h = np.sum(h)
    d = n_points // 2 - sum_h
    if abs(d) > 1e-6:
        # Adjust by rounding
        h = np.clip(h, 0, 1)
        # Use some rounding approach: rank and reset value
        sorted_indices = np.argsort(h)[::-1]
        # Add/Remove 1s from the top/bottom
        if d > 0:  # Need to remove some 1s
            for i in range(int(d)):
                if h[sorted_indices[i]] > 0.5:
                    h[sorted_indices[i]] = 0.0
        elif d < 0:  # Need to add some 1s
            for i in range(int(-d)):
                if h[sorted_indices[i]] < 0.5:
                    h[sorted_indices[i]] = 1.0
    # Rank and set top_half = 1, others 0:
    # Make sure exact sum = n/2
    ranked = np.argsort(h)[::-1]
    cut = n_points // 2
    h_new = np.zeros_like(h)
    h_new[ranked[:cut]] = 1.0
    # Special handling for odd n:
    if n_points % 2 == 1:
        if np.sum(h_new) < n_points/2:
            h_new[ranked[cut]] = 0.5
        elif np.sum(h_new) > n_points/2:
            h_new[ranked[cut-1]] = 0.5
    return h_new

def project_onto_simplex(h, target_sum=1.0):
    """Project h onto the feasible polytope {x ∈ [0,1]^n : sum(x)=target_sum}."""
    h = np.array(h)
    # Ensure bounds
    h = np.clip(h, 0., 1.)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-8:
        return h
    # Use waterfilling idea
    if current_sum > target_sum:
        # Need to reduce
        diff = current_sum - target_sum
        # Sort descending
        indices = np.argsort(h)[::-1]
        for i in indices:
            if diff <= h[i]:
                h[i] -= diff
                diff = 0
                break
            else:
                diff -= h[i]
                h[i] = 0
    elif current_sum < target_sum:
        # Need to increase
        diff = target_sum - current_sum
        indices = np.argsort(h)
        for i in indices:
            if diff <= 1 - h[i]:
                h[i] += diff
                diff = 0
                break
            else:
                diff -= 1 - h[i]
                h[i] = 1.0
    return h

def local_improvement(h, n_points):
    """Perform local hill-climbing to get near-binary."""
    # Step 1: Round to binary
    h = round_to_binary(h, n_points)
    
    # Step 2: Search for improving swaps
    best_h = h.copy()
    best_c5 = evaluate_h(best_h)
    changed = True
    count = 0
    max_iter = 5000
    while changed and count < max_iter:
        changed = False
        count += 1
        # Try all possible 1/0 swaps
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        for i in ones:
            for j in zeros:
                # Copy and swap h[i] and h[j]
                temp_h = h.copy()
                temp_h[i] = 0
                temp_h[j] = 1
                c5_new = evaluate_h(temp_h)
                if c5_new < best_c5:
                    best_h = temp_h
                    best_c5 = c5_new
                    changed = True
                    h = temp_h.copy()
                    break
            if changed:
                break
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main run function."""
    t0 = time.time()
    np.random.seed(seed)

    # Initialize
    initially = np.array(initial_h_values)
    h0 = initially.copy()
    best_h = h0.copy()
    best_c5 = evaluate_h(h0)

    print(f"Starting from initial solution with C5 = {best_c5:.6f}")

    # Try MILP first on small n
    MILP_time_limit = budget_s / 2.0
    try:
        print("Attempting MILP with CBC solver...")
        # Building MILP model
        # Variables
        h = cp.Variable(n_points, boolean=True)
        y = cp.Variable((n_points, 2 * n_points - 1), nonneg=True)
        t = cp.Variable(pos=True)

        # Constraints to create y_i,d = h_i * h_{i+d}
        constraints = []
        for d in range(-n_points+1, n_points):
            for i in range(n_points):
                # Check if both indices are valid
                if 0 <= i+d < n_points:
                    # Convex hull relaxation of product
                    idx = d + (n_points - 1)
                    constraints.append(y[i, idx] <= h[i])
                    constraints.append(y[i, idx] <= h[i+d])
                    constraints.append(y[i, idx] >= h[i] + h[i+d] - 1)
                    constraints.append(y[i, idx] >= 0)

        # Constraint on sum
        constraints.append(cp.sum(h) == n_points / 2.0)

        # Ensure that the mismatch count is bounded by t
        for d in range(-n_points+1, n_points):
            if d == 0:
                continue
            # Index mapping
            idx = d + (n_points - 1)
            # Precompute start and end indices for this d
            start = max(0, -d)
            end = min(n_points, n_points - d)
            # Sum_{ii in [start, end)} h[ii] - sum_{ii} y[ii, idx]
            m_d = cp.sum(h[start:end]) - cp.sum(y[start:end, idx])
            constraints.append(t >= m_d)

        # Objective
        objective = cp.Minimize(t)

        # Solve
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=CPLEX, verbose=False, time_limit=MILP_time_limit, presolve=False)

        print(f"MILP solution: t = {prob.value}")
        if prob.status in ["optimal", "optimal_inaccurate"]:
            # Extract optimal h
            h_best = np.array(h.value).astype(float)
            # Need to fix duality gap / rounding
            h_best = round_to_binary(h_best, n_points)
            c5_milp = evaluate_h(h_best)
            print(f"MILP rounding gives C5 = {c5_milp:.6f}")
            if c5_milp < best_c5:
                best_c5 = c5_milp
                best_h = h_best.copy()
        else:
            print(f"MILP non-optimal status = {prob.status}")
    except Exception as e:
        print(f"MILP error: {e}")

    # Multi-start heuristic using sub-gradient descent / Simulated Annealing
    num_starts = 3
    best_h_final = best_h.copy()
    best_c5_final = best_c5

    print("Starting continuous heuristic phase...")

    for n_start in range(num_starts):
        # Generate a new seed
        local_seed = seed + n_start + 1000
        np.random.seed(local_seed)
        print(f"Start {n_start} with seed {local_seed}")

        # Initial h (could be binary from MILP)
        if n_start == 0:
            h_current = best_h_final.copy()
        elif n_start == 1:
            h_current = np.random.rand(n_points)
            h_current = project_onto_simplex(h_current, n_points/2)
        else:
            # Thue-Morse pattern
            h_current = np.zeros(n_points)
            for i in range(n_points):
                h_current[i] = (bin(i).count('1') & 1)  # Simple pseudo-binary
            h_current = project_onto_simplex(h_current, n_points/2)

        # Optimization parameters
        max_iter = 20000
        alpha0 = 0.1
        T0 = 10.0
        gamma = 0.9995
        temperature = T0

        # Optimization
        iter_count = 0
        last_iter = 0

        while iter_count < max_iter:
            # Check time limit
            if time.time() - t0 > budget_s - 5:
                break

            # Compute all correlations
            try:
                corr_val = np.correlate(h_current, 1 - h_current, mode='full')
                # Find max index
                max_idx = np.argmax(corr_val)
                d_best = max_idx - (n_points - 1)  # convert full-index to shift

                # Compute gradient (pointwise - needed for multiple shifts)
                grad = np.zeros(n_points)
                for k in range(n_points):
                    k_plus = (k + d_best) % n_points
                    k_minus = (k - d_best) % n_points
                    grad[k] = 1 - h_current[k_plus] - h_current[k_minus]

                # Step
                alpha = alpha0 / np.sqrt(1 + iter_count)
                h_new = h_current - alpha * grad

                # Projection
                h_new = project_onto_simplex(h_new, n_points / 2)

                # Accept with probability (simulated annealing)
                c5_old = evaluate_h(h_current)
                c5_new = evaluate_h(h_new)
                if c5_new < c5_old:
                    h_current = h_new
                    if c5_new < best_c5_final:
                        best_c5_final = c5_new
                        best_h_final = h_new.copy()
                    iter_count += 1
                    continue
                # Else reject with probability
                delta = c5_new - c5_old
                prob_accept = np.exp(-delta / (temperature + 1e-8)) if temperature > 1e-8 else 0
                if np.random.rand() < prob_accept:
                    h_current = h_new
                # Cool down
                temperature *= gamma

            except Exception as e:
                print(f"Heuristic iteration {iter_count} error: {e}")
                break

            iter_count += 1
            if iter_count % 100 == 0:
                print(f"  Iter {iter_count} | Best C5 so far: {best_c5_final:.6f}")

    print(f"Final best C5: {best_c5_final}")

    # Final polishing step (local search on integer)
    best_h_final, best_c5_final = local_improvement(best_h_final, n_points)

    return list(best_h_final.astype(float)), best_c5_final, n_points
