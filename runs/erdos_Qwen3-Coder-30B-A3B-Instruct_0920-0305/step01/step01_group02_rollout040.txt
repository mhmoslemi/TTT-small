import numpy as np

from scipy.optimize import minimize

def project_onto_capped_simplex(y, s):
    """Project vector y onto {x : 0 <= x <= 1, sum x = s}"""
    lo = 0.0
    hi = 1.0
    # Initial bounds for lambda
    lam_low = np.min(y) - hi
    lam_high = np.max(y) - lo
    
    for _ in range(50):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(y - lam, lo, hi)
        total = x.sum()
        if total > s:
            lam_low = lam
        else:
            lam_high = lam
    x = np.clip(y - lam_high, lo, hi)
    return x

def compute_overlaps(h):
    """Compute overlap values for all shifts k = 0 to N-1"""
    n = len(h)
    one_minus_h = 1.0 - h
    overlaps = np.empty(n)
    for k in range(n):
        if k == 0:
            overlaps[k] = np.dot(h, one_minus_h)
        else:
            overlaps[k] = np.dot(h[:n-k], one_minus_h[k:])
    overlaps *= (2.0 / n)
    return overlaps

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    correlation *= dx
    return np.max(correlation)
