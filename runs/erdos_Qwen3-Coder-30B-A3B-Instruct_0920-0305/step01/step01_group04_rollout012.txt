def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5, max_correlation

def compute_projection(y, target_sum, bounds=(0, 1)):
    """Project y onto the feasible set {x : sum(x) = target_sum, bounds[0] <= x <= bounds[1]}"""
    n = len(y)
    y = np.clip(y, bounds[0], bounds[1])
    total = np.sum(y)
    diff = target_sum - total
    if abs(diff) < 1e-8:
        return y
    # Simple example: uniformly adjust
    scale = diff / n
    y += scale
    return np.clip(y, bounds[0], bounds[1])
