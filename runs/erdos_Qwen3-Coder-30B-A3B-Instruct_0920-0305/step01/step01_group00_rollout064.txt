import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_simplex(y, S=1.0):
    """Project vector y onto the capped simplex: sum = S, 0 <= y[i] <= 1."""
    # Ensure y is in [0,1], clamp if necessary
    y = np.clip(y, 0, 1)
    # Sort y in descending order
    u = np.sort(y)[::-1]
    # Compute cumulative sum
    cu = np.cumsum(u)
    # Find the threshold lambda
    idx = np.arange(1, len(u) + 1)
    cond = u - (S - cu) / idx > 0
    if not np.any(cond):
        # All entries are too small, just scale down to sum=S
        return y * S / np.sum(y)
    k = np.argmax(cond)
    l = (S - cu[k]) / (k + 1)
    # Return projected vector
    return np.clip(y - l, 0, 1)

def compute_correlations(h):
    """Compute correlations of h with 1-h, full mode."""
    n = len(h)
    corr = correlate(h, 1 - h, mode="full")
    return corr

def softmax_weights(corr_vals, alpha):
    """Compute softmax weights from correlation values."""
    # Subtract maximum for numerical stability
    cmax = np.max(corr_vals)
    exp_vals = np.exp(alpha * (corr_vals - cmax))
    return exp_vals / np.sum(exp_vals)

def compute_gradient(h, corr_vals, weights):
    """Compute gradient of the surrogate objective."""
    n = len(h)
    grad = np.zeros(n)
    # Loop over all shifts
    for s in range(-n + 1, n):
        if s >= 0:
            # shift h right by s (backward diff)
            shifted = np.concatenate([np.zeros(s), h[:-s]])
        else:
            # shift h left by abs(s) (forward diff)
            shifted = np.concatenate([h[-s:], np.zeros(-s)])
        grad += weights[s + n // 2] * shifted
    return 1 - grad

def line_search_step(h, grad, eta, alpha, corr_vals):
    """Line search for a stable step to ensure progress in soft max."""
    y = h - eta * grad
    y = project_simplex(y)
    new_corr_vals = compute_correlations(y)
    alt_obj = (1 / alpha) * np.log(np.sum(np.exp(alpha * new_corr_vals)))
    return y, alt_obj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Hyperparameters
    n_points = 96  # Must be even; dx=2/n_points = 0.0208
    dx = 2.0 / n_points
    target_sum = n_points / 2  # We want sum(h)*dx = 1, i.e., sum(h)=n/2

    # Initial function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to desired n_points
        x_orig = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n_points)
        h0 = np.interp(x_new, x_orig, h_init)
    else:
        # Initial random point on valid simplex
        h0 = np.random.rand(n_points)
        h0 = project_simplex(h0, S=target_sum)

    # Gradient descent loop
    best_h = h0.copy()
    best_C5 = np.inf
    alpha = 5.0
    r_alpha = 1.03
    max_alpha = 200.0
    max_iters_per_restart = 5000
    eta0 = 0.5  # base step size
    reset_iter = 200

    def eval_obj(h):
        corr_vals = compute_correlations(h)
        return np.max(corr_vals) * dx

    def eval_surrogate(h, alpha):
        corr_vals = compute_correlations(h)
        return (1 / alpha) * np.log(np.sum(np.exp(alpha * corr_vals)))

    num_restarts = 15
    restarts_done = 0
    
    while restarts_done < num_restarts and (time.time() - start_time) < budget_s - 5:
        h = h0.copy()
        if restarts_done > 0:
            # Generate new random start on simplex
            h = np.random.rand(n_points)
            h = project_simplex(h, S=target_sum)
        else:
            # First start with initial h0
            pass
            
        # Best tracked over this restart
        local_best_h = h.copy()
        local_best_C5 = eval_obj(h)
        # Main gradient loop
        for iter in range(max_iters_per_restart):
            if (time.time() - start_time) > budget_s - 5:
                break

            # Evaluate current correlations
            corr_vals = compute_correlations(h)
            # Compute weights (with numerical stability)
            weights = softmax_weights(corr_vals, alpha)
            # Compute gradient
            grad = compute_gradient(h, corr_vals, weights)
            # Step size
            eta = eta0 / np.sqrt(iter + 1)
            # Line search step
            y = h - eta * grad
            y = project_simplex(y, S=target_sum)
            # Quick objective check
            new_C5 = eval_obj(y)
            if new_C5 < local_best_C5:
                local_best_C5 = new_C5
                local_best_h = y.copy()
            h = y.copy()
            # Update alpha (annealing)
            if iter and iter % reset_iter == 0:
                alpha = min(alpha * r_alpha, max_alpha)
            if iter > 1000 and np.abs(alpha - max_alpha) < 0.1:
                break

        if local_best_C5 < best_C5:
            best_C5 = local_best_C5
            best_h = local_best_h.copy()
        restarts_done += 1

    # Final refinement with SLSQP
    def obj_slsqp(h):
        return eval_obj(h)

    def constraint_eq(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10})
    if res.success:
        refined_h = res.x
        final_C5 = obj_slsqp(refined_h)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = refined_h

    return (best_h.tolist(), best_C5, n_points)
