import numpy as np

import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure the constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (1.0 / h_sum)
        h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_c5_binary(bits):
    """Evaluate the C5 bound for binary h values"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

        np.random.shuffle(bits)
        
        curr_val = evaluate_c5_binary(bits)
        
        # iterative descent with limited attempts to avoid infinite loops
        attempts = 0
        max_attempts = 5000
        improved = True
        
        while improved and attempts < max_attempts and time.time() - start < time_limit:
            improved = False
            attempts += 1

            zeros_idx = np.flatnonzero(1 - bits)

                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)

        if curr_val < best_sum:
            best_sum = curr_val
            best_bits = bits.copy()
    
    return best_bits, best_sum
