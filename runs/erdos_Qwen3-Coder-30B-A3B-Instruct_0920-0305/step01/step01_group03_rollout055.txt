import numpy as np

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h to the capped simplex: 0 <= h_i <= 1, sum h_i = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    elif current_sum > target_sum:
        # Water filling: reduce until sum = target_sum
        # Sort descending
        h_sorted = np.sort(h)[::-1]
        # Find threshold
        cumulative = np.cumsum(h_sorted)
        n_exceed = np.searchsorted(cumulative, target_sum, side='right')
        if n_exceed == 0:
            return h * (target_sum / current_sum)
        else:
            threshold = (target_sum - cumulative[n_exceed-1]) / (len(h_sorted) - n_exceed)
            h_proj = np.minimum(h_sorted, threshold)
            # Project back to original indices
            h_proj = np.sort(h_proj)[::-1]
            # Normalize to correct sum
            h_proj = h_proj * (target_sum / np.sum(h_proj))
            return h_proj  
    else:
        # Sum is less than required, clamp at 1
        # Then increase by filling
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        # Increase uniformly
        h = h + diff / len(h)
        # Final projection to maintain bounds
        h = np.clip(h, 0, 1)
        # And renormalize
        h = h * (target_sum / np.sum(h))
        return h

def logsumexp_stable(log_weights):
    """Numerically stable logsumexp"""
    max_w = np.max(log_weights)
    return max_w + np.log(np.sum(np.exp(log_weights - max_w)))

def compute_gradient(h, R, lambda_param):
    """Compute gradient of soft minimum with respect to h"""
    n = len(h)
    # Compute log-weights w_s = exp(-lambda * R(s)) / sum(exp(-lambda * R(t)))
    log_ws = -lambda_param * R
    log_ws = log_ws - logsumexp_stable(log_ws)  # Normalize weights
    w = np.exp(log_ws)
    
    # Compute gradient: g_j = sum_s w_s * (h_{j+s} + h_{j-s})
    g = np.zeros_like(h)
    for s in range(-n+1, n):
        if s > 0:
            g += w[s] * np.concatenate([h[s:], np.zeros(s)])
        elif s < 0:
            g += w[s] * np.concatenate([np.zeros(-s), h[:s]])
        else:
            g += w[s] * h  # center term
            
    # g = (w + w_rev) * h
    # With full correlation, w_rev is reversed
    w_rev = w[::-1]
    # Full convolution of w_rev with h
    g = np.convolve(w_rev, h, 'full')[:len(h)]
    return g

def backtracking_line_search(h, g, lambda_param, step_size, target_sum, dx):
    """Perform backtracking line search for gradient step"""
    alpha = step_size
    min_alpha = 1e-8
    max_reduct = 5
    h_raw = h + alpha * g
    h_proj = project_to_capped_simplex(h_raw, target_sum)
    # Compute Phi_lambda of the new point
    R_new = np.correlate(h_proj, h_proj, "full")
    log_ws = -lambda_param * R_new
    log_ws = log_ws - logsumexp_stable(log_ws)
    Phi_new = logsumexp_stable(log_ws) / lambda_param
    R_old = np.correlate(h, h, "full") 
    log_ws = -lambda_param * R_old
    log_ws = log_ws - logsumexp_stable(log_ws)
    Phi_old = logsumexp_stable(log_ws) / lambda_param
    i_reduct = 0
    while Phi_new >= Phi_old and i_reduct < max_reduct:
        alpha *= 0.5
        i_reduct += 1
        if alpha < min_alpha:
            return h, alpha
        h_raw = h + alpha * g
        h_proj = project_to_capped_simplex(h_raw, target_sum)
        R_new = np.correlate(h_proj, h_proj, "full")
        log_ws = -lambda_param * R_new
        log_ws = log_ws - logsumexp_stable(log_ws)
        Phi_new = logsumexp_stable(log_ws) / lambda_param
    return h_proj, alpha
