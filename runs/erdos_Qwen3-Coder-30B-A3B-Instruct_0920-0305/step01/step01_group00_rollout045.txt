import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_simplex(y, s):
    """Project y onto the capped simplex: sum yi = s, 0 <= yi <= 1"""
    # Use bisection to find lambda
    y = np.asarray(y)
    lmin, lmax = y.min() - 1, y.max() + 1
    tol = 1e-12
    while lmax - lmin > tol:
        lamb = (lmin + lmax) / 2
        x = np.clip(y - lamb, 0, 1)
        if np.sum(x) > s:
            lmin = lamb
        else:
            lmax = lamb
    lamb = (lmin + lmax) / 2
    x = np.clip(y - lamb, 0, 1)
    return x

def compute_gradient(h, c, alpha):
    """Compute gradient of log-sum-exp surrogate."""
    n = len(h)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros_like(h)
    for i in range(n):
        grad[i] = 1 - np.sum(w * (np.roll(h, i) + np.roll(h, -i)))
    return grad

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Grid size and parameters
    n = 96  # points
    dx = 2.0 / n
    s = n / 2  # integral constraint sum

    # Initialize h
    h_current = np.array(initial_h_values)
    if len(h_current) >= n:
        h = h_current[:n]
    else:
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_current)), h_current)
    
    # Project to feasible set exactly
    h = project_simplex(h, s)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    # Annealing parameters
    alpha_start = 5.0
    alpha = alpha_start
    alpha_rate = 1.03
    alpha_max = 200.0
    alpha_update_interval = 200

    # Gradient descent
    max_iters = 5000
    eta0 = 0.5
    iter_count = 0
    restart_count = 0
    alpha_increase_counter = 0

    goal_c5 = 0.3807

    while (time.time() - start_time) < (budget_s * 0.7) and iter_count < max_iters:
        if iter_count % 1000 == 0 and iter_count > 0:
            # Check for early termination
            if best_c5 < goal_c5:
                break
        
        # Compute raw correlation vector
        c = correlate(h, 1 - h, mode="full")

        # Compute gradient
        grad = compute_gradient(h, c, alpha)

        # Gradient step with diminishing step size
        eta = eta0 / np.sqrt(iter_count + 1)
        y = h - eta * grad

        # Project onto simplex
        h = project_simplex(y, s)

        if iter_count % 100 == 0:
            c5_trial = evaluate_c5(h)
            if c5_trial < best_c5:
                best_h = h.copy()
                best_c5 = c5_trial
            
        # Anneal alpha
        alpha_increase_counter += 1
        if alpha_increase_counter >= alpha_update_interval:
            alpha = min(alpha * alpha_rate, alpha_max)
            alpha_increase_counter = 0

        iter_count += 1

        # Time check
        if (time.time() - start_time) > (budget_s * 0.95):
            break

    # Final polishing
    def objective(h):
        return evaluate_c5(h)

    # Local refinement - SLSQP with sum constraint
    bounds = [(0, 1) for _ in range(n)]
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0

    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            refined_h = res.x
            refined_c5 = objective(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    # Final projection to enforce sum constraint exactly
    best_h = project_simplex(best_h, s)
    final_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), final_c5, n)
