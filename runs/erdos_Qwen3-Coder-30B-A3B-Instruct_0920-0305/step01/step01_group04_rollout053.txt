fft_size = 2 * n_points - 1

def evaluate_h(h_values):
    """Evaluate C_5 bound for given h values."""
    h = np.array(h_values)
    # Use FFT-based correlation for efficiency with larger n
    corr = np.real(fft.ifft(fft.fft(h, fft_size) * np.conj(fft.fft(1 - h, fft_size)))).real
    max_corr = np.max(corr)
    return max_corr * dx

def evaluate_h_naive(h_values):
    """Naive correlation computation (for small n)."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def project_onto_sum_constraint(h, target_sum):
    """Project h onto the feasible simplex { h : sum(h) = target_sum, 0 ≤ h ≤ 1 }."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Add to smallest elements to increase sum
        idx = np.argsort(h)
        for i in range(len(h)):
            if diff <= 0: break
            delta = min(diff, 1 - h[idx[i]])
            h[idx[i]] += delta
            diff -= delta
    else:
        # Remove from largest elements to decrease sum
        idx = np.argsort(h)[::-1]
        for i in range(len(h)):
            if diff >= 0: break
            delta = min(-diff, h[idx[i]])
            h[idx[i]] -= delta
            diff += delta
    return h

def generate_initial_binary():

def max_mismatch_with_fft(h):
    """Compute max mismatch using FFT for speed."""
    h = np.array(h)
    corr = np.real(fft.ifft(fft.fft(h, fft_size) * np.conj(fft.fft(1 - h, fft_size)))).real
    return np.max(corr)

def get_worst_shift(h):
    """Return the shift index that maximizes overlap."""
    h = np.array(h)
    corr = np.real(fft.ifft(fft.fft(h, fft_size) * np.conj(fft.fft(1 - h, fft_size)))).real
    best_shift = np.argmax(corr)
    return best_shift, corr[best_shift]

def random_binary_h():
    """Return a random binary h vector."""
    h = np.zeros(n_points)
    ones = np.random.choice(n_points, size=n_points // 2, replace=False)
    h[ones] = 1.0
    return h

def adjust_sum(h, target_sum):
    """Ensure sum of h equals target sum by adjusting entries near average."""
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) < 1e-8:
        return h.copy()
    if h_sum > target_sum:
        # Need to reduce h_sum
        indices = np.arange(n_points)
        sort_idx = np.argsort(h)
        for i in range(len(indices)):
            if h_sum <= target_sum:
                break
            j = sort_idx[i]
            to_reduce = min(h[j], h_sum - target_sum)
            h[j] -= to_reduce
            h_sum -= to_reduce
    else:
        # Need to increase h_sum
        indices = np.arange(n_points)
        sort_idx = np.argsort(h)
        for i in range(len(indices)-1, -1, -1):
            if h_sum >= target_sum:
                break
            j = sort_idx[i]
            to_increase = min(1.0 - h[j], target_sum - h_sum)
            h[j] += to_increase
            h_sum += to_increase
    return h.copy()

def round_and_polish(h):
    """Round h to binary, then hill-climb."""
    # Set first half of sorted to 1, others to 0
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[n_points//2]
    h_rounded = (h >= threshold).astype(float)
    # Adjust to sum = n/2
    current_sum = np.sum(h_rounded)
    diff = n_points//2 - current_sum
    if diff != 0:
        if diff > 0:
            # Add ones to lowest values
            low_idx = np.argsort(h_rounded)
            for i in range(diff):
                if h_rounded[low_idx[i]] == 0:
                    h_rounded[low_idx[i]] = 1
        else:
            # Remove ones from highest values
            high_idx = np.argsort(h_rounded)[::-1]
            for i in range(-diff):
                if h_rounded[high_idx[i]] == 1:
                    h_rounded[high_idx[i]] = 0
    h_final = h_rounded
    while True:
        improvement = False
        for i in range(n_points):
            if h_final[i] == 1:
                for j in range(n_points):
                    if h_final[j] == 0:
                        h_temp = h_final.copy()
                        h_temp[i] = 0
                        h_temp[j] = 1
                        if max_mismatch_with_fft(h_temp) < max_mismatch_with_fft(h_final):
                            h_final = h_temp
                            improvement = True
                            break
        if not improvement:
            break
    return h_final

def subgradient_descent_step(h, step_size=0.2):
    """Perform a single step of projected subgradient descent."""
    shift, _ = get_worst_shift(h)
    grad = np.zeros_like(h)
    # Gradient computation
    for i in range(n_points):
        i_plus = (i + shift) % n_points
        i_minus = (i - shift) % n_points
        grad[i] = 1 - h[i_plus] - h[i_minus]
    h_new = h - step_size * grad
    h_proj = project_onto_sum_constraint(h_new, n_points/2)
    return h_proj
