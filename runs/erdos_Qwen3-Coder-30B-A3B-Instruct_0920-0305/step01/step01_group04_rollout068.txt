import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time


def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * 2.0 / len(h)
    return c5


def build_shift_matrices(n):
    """Build the Toeplitz shift matrices C_d for all non-zero shifts d."""
    C_d_list = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        C_d = np.zeros((n, n))
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                C_d[i, j] = 1
                C_d[j, i] = 1
        C_d_list.append(C_d)
    return C_d_list


def solve_sdp(h0, n, R=200):
    # Construct shift matrices
    C_d_list = build_shift_matrices(n)
    
    # SDP variables
    X = cp.Variable((n, n), symmetric=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == 1,  # Diagonal entries 1
    ]
    
    for C_d in C_d_list:
        constraints.append(cp.trace(C_d @ X) >= tau)
    
    # Objective
    objective = cp.Maximize(tau)
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)
        if prob.status == cp.OPTIMAL:
            X_opt = X.value
            tau_opt = tau.value
            best_h = None
            best_score = float('inf')
            
            # Round the solution
            for _ in range(R):
                # Eigenvalue decomposition
                try:
                    V, W = eigh(X_opt)
                    V = V[V > 1e-8]
                    W = W[:, V > 1e-8]

                        s[idx] *= -1
                    
                    h = (s + 1) / 2.0
                    
                    # Evaluate
                    score = evaluate_h(h)
                    if score < best_score:
                        best_score = score
                        best_h = h.copy()
                except:
                    continue
            return best_h, best_score
        else:
            return h0, evaluate_h(h0)
    except:
        return h0, evaluate_h(h0)


def upscale_pattern(h_small, n_target):
    # Method: concatenate with complement
    h_big = np.concatenate([h_small, 1 - h_small])
    while len(h_big) < n_target:
        h_big = np.concatenate([h_big, 1 - h_big])
    if len(h_big) > n_target:
        h_big = h_big[:n_target]
    return h_big


def hill_climb_h(h_start, max_iter=2000):
    h = h_start.copy()
    best_h = h.copy()
    best_score = evaluate_h(h)
    
    # Perturb at most a few times each iteration
    n = len(h)
    iter_idx = 0
    while iter_idx < max_iter:
        improved = False
        # Try all pairs to swap
        for i in range(n):
            for j in range(i+1, n):
                if h[i] != h[j]:  # Only swap if they are different
                    # Temporarily swap
                    h[i], h[j] = h[j], h[i]
                    new_score = evaluate_h(h)
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                        improved = True
                    else:
                        # Revert
                        h[i], h[j] = h[j], h[i]
        if not improved:
            break
        iter_idx += 1
    return best_h, best_score
