```python
import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if current_sum > 0:
        scale = target_sum / current_sum
        x = x * scale
    return x

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    correlation *= dx
    return np.max(correlation)

def create_initial_solution():
    """Create initial solution using a better construction"""
    n = 47
    h = np.zeros(n)
    
    # Create a piecewise constant function that's roughly 1/2
    # For better convergence, make it more structured
    # Use a construction that makes the correlation integral smaller
    
    # Split the domain into 3 parts: 0-0.5, 0.5-1.5, 1.5-2.0
    # This has been empirically better for avoiding high overlaps
    
    # First part: 0 to 1
    h[:15] = 0.5
    # Second part: 1 to 2  
    h[15:30] = 0.0
    # Third part: 2 to 3 (but we only have 47 points so goes to 2.0)
    h[30:] = 1.0
    
    # Make it feasible
    actual_sum = np.sum(h)
    target_sum = n / 2
    h = h * (target_sum / actual_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a better initial solution
    h_init = create_initial_solution()
    n_points = len(h_init)
    target_sum = n_points / 2
    
    # Ensure feasible
    h_current = project_to_feasible(h_init, target_sum)
    current_value = evaluate_erdos_solution(h_current)
    
    best_h = h_current.copy()
    best_value = current_value
    
    # Use a hybrid approach: global optimization + local refinement
    # Try differential evolution first for global search
    def objective(h):
        h = project_to_feasible(h, target_sum)
        return evaluate_erdos_solution(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create custom function for tracking time
    def timed_optimize():
        # Run differential evolution if time allows
        if time.time() - start_time < budget_s * 0.6:
            try:
                de_result = differential_evolution(
                    objective, 
                    bounds, 
                    seed=seed,
                    maxiter=1000,
                    popsize=15,
                    mutation=(0.5, 1.0),
                    recombination=0.7,
                    disp=False
                )
                
                if not de_result.success:
                    return
                
                h_opt = project_to_feasible(de_result.x, target_sum)
                opt_value = evaluate_erdos_solution(h_opt)
                
                if opt_value < best_value:
                    best_h[:] = h_opt
                    best_value = opt_value
            except:
                pass
        
        # Local refinement with Nelder-Mead
        if time.time() - start_time < budget_s * 0.95:
            try: 
                # Start from current best
                x0 = best_h.copy()
                result = minimize(
                    objective,
                    x0,
                    method='Nelder-Mead',
                    options={'maxiter': 2000, 'xatol': 1e-6, 'fatol': 1e-6}
                )
                
                h_opt = project_to_feasible(result.x, target_sum)
                opt_value = evaluate_erdos_solution(h_opt)
                
                if opt_value < best_value:
                    best_h[:] = h_opt
                    best_value = opt_value
                    
            except:
                pass
        
        # Heuristic random search refinement
        attempts = 0
        max_attempts = 10000
        while attempts < max_attempts and (time.time() - start_time) < budget_s - 0.1:
            # Random change to one element
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            
            # Ensure move is feasible
            old_val = best_h[i]
            new_val = old_val + delta
            if new_val < 0:
                new_val = 0
            elif new_val > 1:
                new_val = 1
            
            # Compute new vector
            h_proposed = best_h.copy()
            h_proposed[i] = new_val
            
            # Adjust others to keep sum constraint
            current_sum = np.sum(h_proposed)
            if np.abs(current_sum - target_sum) > 1e-10:
                adjustments = (target_sum - current_sum) / (n_points - 1)
                for j in range(n_points):
                    if j != i:
                        h_proposed[j] += adjustments
                        # Clamp
                        h_proposed[j] = np.clip(h_proposed[j], 0, 1.0)
            
            h_proposed = project_to_feasible(h_proposed, target_sum)
            proposed_value = evaluate_erdos_solution(h_proposed)
            
            if proposed_value < best_value:
                best_h[:] = h_proposed
                best_value = proposed_value
            
            attempts += 1
    
    # Run the optimization
    timed_optimize()
    
    return (best_h, best_value, len(best_h))
```
