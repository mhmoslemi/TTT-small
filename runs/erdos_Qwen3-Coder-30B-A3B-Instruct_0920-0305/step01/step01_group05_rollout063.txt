import numpy as np

def project_to_simplex(y, target_sum):
    """Project vector y onto the simplex with sum = target_sum and bounds [0,1]"""
    y = np.asarray(y)
    lower, upper = 0.0, 1.0
    # Bisection on alpha to find the right shift
    l, u = np.min(y) - (target_sum - np.sum(y)), np.max(y) - (target_sum - np.sum(y))
    for _ in range(50):  # ensure convergence
        alpha = (l + u) / 2.0
        z = np.clip(y - alpha, lower, upper)
        s = np.sum(z)
        if s > target_sum:
            l = alpha
        else:
            u = alpha
    alpha = (l + u) / 2.0
    z = np.clip(y - alpha, lower, upper)
    return z

def compute_fs(h, s, dx):
    """Compute f_s(h) = sum h_i*(1-h_{i+s}) for given shift s"""
    if s >= 0:
        h_shifted = np.concatenate([np.zeros(s), h[:-s]])
    else:
        h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
    return np.sum(h * (1 - h_shifted)) * dx

def smooth_max_p(h, p, dx):
    """Compute smooth approximation of max over all shifts"""
    n = len(h)
    # Compute all f_s(h) values (shifts from -(n-1) to n-1)
    f_vals = []
    for s in range(-(n-1), n):
        f = compute_fs(h, s, dx)
        f_vals.append(f)
    f_vals = np.array(f_vals)
    # Apply smooth max
    total = np.sum(np.abs(f_vals)**p)
    if total == 0:
        return 0.0
    return total**(1.0/p)

def gradient_smooth_max(h, p, dx):
    """Compute gradient of smooth_max_p with respect to h"""
    n = len(h)

    f_vals = []

            h_shifted = np.concatenate([np.zeros(s), h[:-s]])
        else:
            h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
        f = np.sum(h * (1 - h_shifted)) * dx
        f_vals.append(f)
    f_vals = np.array(f_vals)
    
    # Compute gradient
    for i in range(n):
        for j, s in enumerate(range(-(n-1), n)):
            f_s = f_vals[j]
            if s >= 0:
                h_shifted = np.concatenate([np.zeros(s), h[:-s]])
            else:
                h_shifted = np.concatenate([h[-s:], np.zeros(-s)])
            
            if i+s < n and i+s >= 0:
                grad[i] += (p * (abs(f_s)**(p-1)) * (1 - h_shifted[i])) * dx
            else:
                grad[i] += (p * (abs(f_s)**(p-1)) * 1.0) * dx
    return grad / (smooth_max_p(h, p, dx)**(p-1) + 1e-12)

def evaluate_h(h):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h)
    h = np.array(h)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def interpolate_h(h_coarse, n_fine):
    """Linear interpolation from coarse to fine grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
