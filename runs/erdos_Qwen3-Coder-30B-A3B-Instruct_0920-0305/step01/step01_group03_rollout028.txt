import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx=2.0/100):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def build_and_solve_lp(n_points):
    """Build and solve the linear program relaxation for the Erdos problem"""
    # Create variables
    x = cp.Variable(n_points)
    t = cp.Variable()
    
    # Build the matrix of indices for each shift
    num_shifts = 2 * n_points - 1
    w_vars = {}
    # For each shift s from -(n-1) to +(n-1)
    for s in range(-n_points + 1, n_points):
        w_vars[s] = {}
        for i in range(n_points):
            if 0 <= i + s < n_points:
                w_vars[s][i] = cp.Variable()
    
    # Setup constraints
    constraints = []
    
    # Mass constraint
    constraints.append(cp.sum(x) == n_points / 2)
    
    # Box constraints
    constraints.extend([x[i] >= 0 for i in range(n_points)])
    constraints.extend([x[i] <= 1 for i in range(n_points)])
    
    # McCormick inequalities
    for s in range(-n_points + 1, n_points):
        for i in range(n_points):
            if 0 <= i + s < n_points:
                w = w_vars[s][i]
                constraints.extend([
                    w <= x[i],
                    w <= x[i + s],
                    w >= x[i] + x[i + s] - 1,
                    w >= 0
                ])
    
    # Constraint for target minimal correlation t
    for s in range(-n_points + 1, n_points):
        if s in w_vars:
            expr = sum(w_vars[s][i] for i in range(n_points) if 0 <= i + s < n_points)
            constraints.append(expr >= t)
    
    # Objective: maximize t
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=ECOS, verbose=False, reltol=1e-6, abstol=1e-8)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return x.value, t.value
        else:
            return None, None
    except:
        return None, None

def deterministic_rounding(x_values, n_points):
    """Rounding scheme that selects the n/2 indices with largest values"""
    indices = np.argsort(x_values)[::-1]
    binary = np.zeros(n_points)
    binary[indices[:n_points//2]] = 1
    return binary

def swap_improvement(h, max_iter=20000, time_budget=500):
    """Local search using swaps to decrease C5"""
    start_time = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute initial correlation for each shift to optimize updates
    def compute_correlations(h_arr):
        corr = np.correlate(h_arr, 1 - h_arr, mode="full")
        return corr
    
    corr_full = compute_correlations(h)
    shifted_corr = {s: corr_full[n_points-1+s] for s in range(-n_points+1, n_points)}
    
    # Precompute which pairs contribute to each correlation
    corr_pairs = {}
    for s in range(-n_points+1, n_points):
        corr_pairs[s] = []
        for i in range(n_points):
            if 0 <= i + s < n_points:
                corr_pairs[s].append((i, i+s))

    # Shuffle to avoid bias
    np.random.shuffle(np.arange(n_points))
    
    iter_no_improvement = 0
    for _ in range(max_iter):
        if time.time() - start_time > time_budget:
            break
            
        # Pick a random 1 position and 0 position
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform swap
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        
        # Compute new correlations incrementally (optimization)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            if iter_no_improvement > 1000:
                break
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Test different resolutions, starting from smaller to allow more iterations
    resolutions = [80, 100, 120]
    best_solution = None
    best_c5 = float('inf')
    best_n = 0
    
    for n_points in resolutions:
        if time.time() - start_time > 0.96 * budget_s:
            break
            
        # Solve LP relaxation
        x_sol, t_sol = build_and_solve_lp(n_points)
        if x_sol is None:
            continue
        x_sol = np.array(x_sol)
        
        # Rounding
        h_init = deterministic_rounding(x_sol, n_points)
        # Evaluate initial solution
        c5_init = evaluate_erdos_solution(h_init)
        if c5_init < best_c5:
            best_c5 = c5_init
            best_solution = h_init.copy()
            best_n = n_points
            
    # Perform further local search if time allows
    if best_solution is not None and time.time() - start_time < 0.9 * budget_s:
        h_final, c5_final = swap_improvement(best_solution, max_iter=20000, time_budget=0.8 * budget_s)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_solution = h_final.copy()
    
    return (best_solution.tolist(), best_c5, best_n)
