import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project onto the intersection of a box [0,1] and a hyperplane (sum = target_sum)"""
    # Use water-filling to project onto the sum constraint while respecting box constraints
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Water filling to adjust to the target sum
    # Set up a descending sort to fill from high to low
    indices = np.argsort(h)[::-1]
    h_sorted = h[indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(h_sorted)
    n_sorted = len(h_sorted)
    
    # Find the breakpoint
    total = 0
    for i in range(n_sorted):
        if i == 0:
            sum_with_i = h_sorted[i]
        else:
            sum_with_i = h_sorted[i] + cumsum[i-1]
        
        # If we can make elements up to index i equal to h_sorted[i], 
        # without exceeding target_sum, then break
        if sum_with_i <= target_sum:
            total = sum_with_i
            continue
        else:
            # We are at the point where we start using a different approach
            # Calculate how much we can add to elements above index i-1
            if i > 0:
                average_to_i = (target_sum - cumsum[i-1]) / (n_sorted - i)
                break
            else:
                average_to_i = target_sum / n_sorted
                break
    else:
        # All elements fit into average
        average_to_i = target_sum / n_sorted
    
    # Set values
    h_new = h.copy()
    h_new[indices] = np.where(np.arange(n_sorted) < i, h_sorted[:i], average_to_i)
    
    # So far h_new has values proportional to the water filling but we should make sure
    # that the sum is correct and not just proportionally set in each element
    # For values <- i, they are also not equal, so we're going to adjust them carefully.
    # Actually the properly water-fully clipped updated h is not so easy,
    # so let's use the cycle approach:
    # Grow the h: adding a lambda to elements below threshold
    
    def compute_sum_for_lambda(lam):
        hnew = np.clip(h + lam, 0, 1)
        return np.sum(hnew)
    
    # This is a fixed-point like equation: if we add some lambda > 0,
    # we must increase the elements (and elements that were full were not changed).
    # So it's reducing to estimating the right scaler factor (lambda)
    low = -np.max(h)
    high = 1 - np.min(h)
    # Binary search or Bisection for lambda
    for _ in range(60):
        mid = (low + high) / 2
        new_sum = compute_sum_for_lambda(mid)
        if abs(new_sum - target_sum) < 1e-10:
            break
        elif new_sum < target_sum:
            high = mid
        else:
            low = mid
    h = np.clip(h + (low + high) / 2, 0, 1)
    
    return h

def soft_min_grad(h, R, lam):
    """Compute the gradient of soft-minimum Φ_λ(h)"""
    n = len(h)
    # Safely compute log-sum-exp, using max trick
    R_max = np.max(R)
    exp_R = np.exp(-lam * (R - R_max))
    exp_sum = np.sum(exp_R)
    w = exp_R / exp_sum  # Soft weights
    
    # Compute gradient: g_j = sum_s w_s * (h_{j+s} + h_{j-s})
    g = np.zeros(n)
    
    # Matrix form to compute full convolution of w with h
    # since indexing for negative arrays is tricky...
    # Compute g = (w + w_{reverse}) * h (convolution-like)
    
    # Not using convolution, but directly compute all slices
    for j in range(n):
        for s in range(len(R)):
            # s index in R corresponds to shift (index 0 means no shift)
            # Convert s to shift; shift = s - (n-1)
            shift = s - (n-1)
            k_fwd = j + shift
            k_bwd = j - shift
            if 0 <= k_fwd < n:
                g[j] += w[s] * h[k_fwd]
            if 0 <= k_bwd < n and k_bwd != k_fwd:
                g[j] += w[s] * h[k_bwd]
    
    return g

def backtracking_line_search(h, g, alpha_init, lam, target_sum, dx):
    """Backtracking line search"""
    def surrogate_value(h_val):
        n = len(h_val)
        R = np.correlate(h_val, h_val, mode="full")
        R_max = np.max(R)
        exp_R = np.exp(-lam * (R - R_max))
        exp_sum = np.sum(exp_R)
        return np.log(exp_sum) / lam + R_max / lam

    alpha = alpha_init
    h_new = project_to_capped_simplex(h - alpha * g, target_sum)
    new_surrogate = surrogate_value(h_new)
    old_surrogate = surrogate_value(h)
    
    # Backtrack if it doesn't improve
    num_reductions = 0
    while new_surrogate >= old_surrogate and num_reductions < 5:
        alpha = alpha / 2
        h_new = project_to_capped_simplex(h - alpha * g, target_sum)
        new_surrogate = surrogate_value(h_new)
        num_reductions += 1
    
    return h_new, alpha

def symmetric_project(h):
    """Symmetrize h about x=1"""
    n = len(h)
    h_new = np.zeros(n)
    for i in range(n):
        h_new[i] = (h[i] + h[n-1-i]) / 2
    return h_new
