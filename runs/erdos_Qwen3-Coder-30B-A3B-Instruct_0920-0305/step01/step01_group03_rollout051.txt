import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """
    Project vector v onto the set {x : lower <= x_i <= upper, sum_i x_i = target_sum}
    """
    v = np.asarray(v)
    n = len(v)
    # Check if feasible
    if target_sum < lower * n or target_sum > upper * n:
        raise ValueError("Sum constraint incompatible with bounds")

    # Start with the simplest projection: clip and then adjust.
    # Fallback to iterative projection via binary search if needed.

    # Use iterative approach to find shift theta such that sum = target_sum
    # \sum_i min(upper, max(lower, v_i + theta)) = target_sum
    # We'll do a binary search on theta:
    max_iter = 10000
    low, high = -10.0, 10.0
    
    # Refine bounds
    min_v, max_v = np.min(v), np.max(v)
    low = max(low, lower - max_v)
    high = min(high, upper - min_v)
    
    for _ in range(max_iter):
        theta = (low + high) / 2.0
        projected = np.clip(v + theta, lower, upper)
        current_sum = np.sum(projected)
        if current_sum <= target_sum:
            low = theta
        else:
            high = theta
        if abs(current_sum - target_sum) < 1e-10:
            break
    
    return np.clip(v + low, lower, upper)

def smooth_max_objective(h, lambda_val, dx):
    """Smoothed version of the objective to avoid non-smooth max"""
    # Total sum constraint: np.sum(h) * dx = 1
    # So sum(h) should equal n_points / 2 almost (up to floating point)  
    # Initialize correlation array
    n = len(h)
    # Now compute correlation of h itself
    corr_hh = np.correlate(h, h, mode='full')
    # Softmin approximation:  - (1/lambda) * log sum_i (exp(-lambda * corr_i))
    # Our loss function needs to minimize:  (1/lambda) * log sum_i exp(-lambda * corr_i)
    max_corr = np.max(corr_hh)  # shifted to avoid overflow
    exp_term = np.exp(-lambda_val * (corr_hh - max_corr))
    sum_exp = np.sum(exp_term)
    smooth_loss = np.log(sum_exp) / lambda_val + max_corr
    return smooth_loss

def compute_smooth_gradient(h, lambda_val, dx):
    n = len(h)
    corr_hh = np.correlate(h, h, mode='full')
    max_corr = np.max(corr_hh)
    
    # Compute weights (log-sum-exp gradient)
    exp_term = np.exp(-lambda_val * (corr_hh - max_corr))
    sum_exp = np.sum(exp_term)
    weights = exp_term / sum_exp
    
    # Compute gradient for each index
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        idx = s + (n-1)  # adjust index to be from 0 to 2n-2
        if weights[idx] == 0.0:
            continue
        w_s = weights[idx]
        # Compute contributions from shifting indices
        # Add h_{i+s} contribution to grad_i
        if s >= 0:
            grad[:n-s] += w_s * h[s:]  # h_{i+s}
        else:
            grad[-s:] += w_s * h[:-s]  # h_{i+s}
        # Add h_{i-s} contribution to grad_i
        if s <= 0:
            grad[-s:] += w_s * h[:n+s]  # h_{i-s}
        else:
            grad[:n-s] += w_s * h[s:]   # h_{i-s}
    # Gradient of loss = - (1 / lambda_value) * sum_s weight_s * (h_{i+s} + h_{i-s}) 
    return grad / lambda_val
    

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a larger number of sample points for better resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Target sum (since integrate h(x)dx = 1)
    target_sum = n_points / 2.0

    # Start with initial construction if available
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() and len(initial_h_values) == n_points else None

    best_c5 = float('inf')
    best_h = None

    # Generate diverse initial points for multi-start
    restarts = 30
    for restart in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Strategy for initialization:
        # 1. Use DC interpolation of original solution if available  
        # 2. Random with projection to unit simplex [0,1], sum=target
        # 3. Binary distribution (random assignment of 0/1)
        # 4. Bimodal distribution (various block shapes)
        
        h = None
        
        if initial_h is not None and np.random.rand() < 0.4:
            # Use interpolation from initial solution
            x_orig = np.linspace(0, 2, len(initial_h), endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h = np.interp(x_new, x_orig, initial_h)
        elif np.random.rand() < 0.3:
            # Random binary pattern with fixed count of 1s
            count_ones = int(target_sum)
            h = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, count_ones, replace=False)
            h[ones_indices] = 1.0
        else:
            # Random continuous with normalization
            h = np.random.uniform(0, 1, n_points)
            h *= target_sum / np.sum(h)
            
        # Ensure initial mass is correct and bounds respected
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
        
        # Ensure sum constraint is satisfied
        try:
            h = project_onto_box_simplex(h, target_sum)
        except:
            h = np.clip(h, 0,1)
            h = h * target_sum / np.sum(h)
            h = project_onto_box_simplex(h, target_sum)
        
        # Now find optimal solution by smooth gradient steps
        x = h.copy()
        
        # Stages: 1 (slow convergence) to 5 (fast stepping). Decrease lambda gradually.
        lambdas = [1., 5., 10., 20., 40.]
        max_iter_per_stage = 200
        step_size = 0.05
        
        for L in lambdas:
            if time.time() - start_time > budget_s * 0.95:
                break
            for _ in range(max_iter_per_stage):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                g = compute_smooth_gradient(x, L, dx)
                # Perform projected gradient descent using fixed step size
                x = x - step_size * g
                x = project_onto_box_simplex(x, target_sum)
                # Optionally add a small amount of noise to escape local minima
                if np.random.rand() < 0.05:
                    x += np.random.normal(0, 0.001, len(x))
                    x = project_onto_box_simplex(x, target_sum)
                    
        # Evaluate final state
        c5 = evaluate_erdos_solution(x, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
    
    # If no valid solution was found, we fallback to the best intial h
    if best_h is None:
        h = np.array(initial_h_values) if initial_h_values is not None and len(initial_h_values) == n_points else np.ones(n_points) * 0.5
        h_sum = np.sum(h)
        h = h * target_sum / h_sum 
        h = project_onto_box_simplex(h, target_sum)
        best_c5 = evaluate_erdos_solution(h, dx)
        best_h = h
        
    # Final refinement with actual optimizer to fine-tune
    def obj_func(h_vec):
        # Compute overlap integral directly and return it
        return evaluate_erdos_solution(h_vec, dx) 

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Define constraint
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - 1.0
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        obj_func,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-8}
    )

    # final refinement
    if result.success:
        final_c5 = obj_func(result.x)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = result.x.copy()

    # Ensure constraints are enforced
    best_h = project_onto_box_simplex(best_h, target_sum)
    return (best_h.tolist(), best_c5, n_points)
