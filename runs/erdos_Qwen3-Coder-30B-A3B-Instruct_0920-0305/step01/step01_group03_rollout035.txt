def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: sum(h) = target_sum, 0 <= h <= 1"""
    # Sort in descending order
    sorted_h = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_h)
    # Find the index where we should truncate
    idx = np.where(sorted_h > (target_sum - cumsum + sorted_h))[0]
    if len(idx) == 0:
        # All values should be zero (but shouldn't happen in practice)
        return np.maximum(0, np.minimum(1, h))
    # Set threshold
    threshold = (target_sum - cumsum[idx[-1] - 1]) / (idx[-1] + 1) if idx[-1] > 0 else target_sum / (idx[-1] + 1)
    # Project
    projected = np.maximum(0, np.minimum(1, h - threshold))
    # Renormalize to target sum
    scale = target_sum / np.sum(projected)
    return projected * scale

def compute_gradient(h, R, lambda_val):
    """Compute gradient of log-sum-exp approximation of min."""
    n = len(h)
    # Avoid numerical issues with log-sum-exp
    max_R = np.max(R)
    exp_terms = np.exp(-lambda_val * (R - max_R)) 
    w = exp_terms / np.sum(exp_terms)
    # Create gradient direction
    g = np.zeros_like(h)
    # Use manual loop for clarity and numerical control
    for s in range(len(R)):
        offset = s - (len(R) - 1) // 2
        if offset >= 0:
            g += w[s] * np.concatenate([h[offset:], np.zeros(offset)])
        else:
            g += w[s] * np.concatenate([np.zeros(-offset), h[:offset]])
    return g

def backtracking_line_search(h, g, lambda_val, alpha_init, max_iter=5):
    """Backtracking line search for step size alpha"""
    h_old = h.copy()
    # Get initial value
    R_old = np.correlate(h_old, h_old, mode="full")
    Phi_old = np.log(np.sum(np.exp(-lambda_val * R_old))) / lambda_val
    # Try decreasing alpha until sufficient decrease
    alpha = alpha_init
    for _ in range(max_iter):
        h_new = h_old + alpha * g
        # Project to capped simplex
        h_new = project_to_capped_simplex(h_new, len(h_new) / 2)
        # Compute new Phi
        R_new = np.correlate(h_new, h_new, mode="full")
        Phi_new = np.log(np.sum(np.exp(-lambda_val * R_new))) / lambda_val
        if Phi_new <= Phi_old:
            return alpha, h_new
        else:
            alpha /= 2.0
    return alpha, h_new

def symmetry_projection(h):
    """Apply symmetry constraint h[i] = (h[i] + h[-1-i])/2"""
    n = len(h)
    h_sym = np.zeros_like(h)
    for i in range(n):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
    return project_to_capped_simplex(h_sym, len(h) / 2)
