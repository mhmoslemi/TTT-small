import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points = 200  # Even number for binary pattern
dx = 2.0 / n_points
target_bound = 0.3807

# Generate initial binary pattern using quadratic residues (approximation)
def generate_quadratic_residue_pattern(n):
    # Use quadratic residues of a prime p where p ≡ 3 (mod 4)
    # For simplicity, using known good patterns for even n
    # We'll use a method to generate near-optimal binary pattern
    
    half_n = n // 2
    h = np.zeros(n)
    
    # Start with a balanced pattern, e.g., systematic placement
    # We will use a method that tries to keep overlap low  
    # One way: place ones with some spacing
    
    # Simple approach: binary pattern with ~half ones, but ensure roughly uniform overlap
    # Place 1s in a structured manner to reduce max correlation
    
    # Fill half positions with 1s, but distribute intelligently
    ones_indices = []
    # Try to avoid local clustering that causes high overlaps
    stride = max(1, n // (half_n + 1))  # spacing between ones
    for i in range(half_n):
        idx = (i * stride) % n
        ones_indices.append(idx)
    
    # Add a few more to make exactly half (ensure even)
    for idx in ones_indices:
        h[idx] = 1
    
    # Adjust to ensure sum = n/2 exactly
    current_sum = np.sum(h)
    diff = int(half_n - current_sum)
    
    # Add or remove ones to correct sum
    if diff > 0:
        # Need to add ones
        zeros_indices = np.where(h == 0)[0]
        for i in range(diff):
            if i < len(zeros_indices):
                h[zeros_indices[i]] = 1
    elif diff < 0:
        # Need to remove ones
        ones_indices = np.where(h == 1)[0]
        to_remove = -diff
        for i in range(to_remove):
            if i < len(ones_indices):
                h[ones_indices[i]] = 0

    return h

# Standard evaluation function
def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

# Constraint for integral = 1
def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

# Objective to minimize
def objective(h_values):
    return evaluate_h(h_values)

# Heuristic optimization using simulated annealing
def optimize_step_function(seed=42):
    """Optimize a step function to minimize C_5."""
    np.random.seed(seed)
    
    # Initial solution: better structured rand pattern
    h0 = generate_quadratic_residue_pattern(n_points)
    initial_c5 = evaluate_h(h0)
    best_h = h0.copy()
    best_c5 = initial_c5
    
    # Store best over iterations
    best_found_h = h0.copy()
    best_found_c5 = initial_c5
    
    # Time tracking
    start_time = time.time()
    budget_s = 600
    
    # Clone current state to avoid updating during search
    current_h = h0.copy()
    current_c5 = initial_c5
    
    # Local search parameters
    max_iter = 1000000  # Cap iterations for time limit
    temp = 1.0
    temp_decay = 0.9999
    min_temp = 1e-6
    acceptance_ratio_threshold = 0.1  # For adaptive cooling
    
    # Try local search with temperature
    iter_count = 0
    accepted_count = 0
    
    while time.time() - start_time < budget_s - 1.0 and iter_count < max_iter:
        iter_count += 1
        
        # Create candidate by swapping one 1 with one 0
        ones_indices = np.where(current_h == 1)[0]
        zeros_indices = np.where(current_h == 0)[0]
        
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break
            
        # Randomly choose one 1 and one 0 to swap
        i1 = np.random.choice(ones_indices)
        i0 = np.random.choice(zeros_indices)
        new_h = current_h.copy()
        new_h[i1] = 0
        new_h[i0] = 1
        
        # Compute new correlation
        new_c5 = evaluate_h(new_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h.copy()
            current_c5 = new_c5
            
            # Update best if improved
            if current_c5 < best_found_c5:
                best_found_h = current_h.copy()
                best_found_c5 = current_c5
            accepted_count += 1
        
        # Cooling schedule
        temp *= temp_decay
        
        # Adaptive cooling
        if iter_count % 1000 == 0:
            if accepted_count > 0:
                ratio = accepted_count / 1000.0
                if ratio < acceptance_ratio_threshold and temp > min_temp:
                    temp *= 0.95  # Cool faster if too few accepts
                accepted_count = 0  # reset for next batch
    
    # Return best found so far
    return best_found_h, best_found_c5

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    start_time = time.time()
    
    # Try to optimize starting from a structured initial solution
    h_values, c5_bound = optimize_step_function(seed)

    final_h = np.clip(h_values, 0, 1)

    actual_sum = np.sum(final_h)
    required_sum = n_points / 2
    
    # Rescale if needed
    if abs(actual_sum - required_sum) > 1e-6:
        scale = required_sum / actual_sum
        final_h *= scale
    
    # Boundary check
    final_h = np.clip(final_h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_h(final_h)
    
    return list(final_h), final_c5, n_points
