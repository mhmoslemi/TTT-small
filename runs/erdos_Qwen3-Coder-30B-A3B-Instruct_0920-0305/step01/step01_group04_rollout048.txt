import numpy as np
import time
from scipy.linalg import eigh
import cvxpy as cp

# Constants
n_points = 200
dx = 2.0 / n_points
target_bound = 0.3807
n0 = 61  # prime ≡ 1 mod 4, small enough for SDP


def compute_full_correlation(h):
    """Compute the full correlation vector (not scaled by dx) and return the maximum value."""
    h = np.array(h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlation)
    return max_corr


def build_shift_matrices(n):
    """Build Toeplitz shift matrices C_d for all non-zero shifts d."""
    C_d_list = {}
    for d in range(-n+1, n):
        if d == 0:
            continue
        C_d = np.zeros((n, n))
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                C_d[i, j] = 1
                C_d[j, i] = 1
        C_d_list[d] = C_d
    return C_d_list


def solve_sdp(n, C_d_list):
    """Solve the SDP relaxation for the minimum overlap problem."""
    X = cp.Variable((n, n), symmetric=True)
    tau = cp.Variable()

    constraints = [X >> 0, cp.diag(X) == 1]
    for d, C_d in C_d_list.items():
        constraints.append(cp.trace(C_d @ X) >= tau)

    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)

    if prob.status != cp.OPTIMAL:
        return None, None

    return X.value, tau.value


def round_sdp_matrix(X, n):
    """Round the SDP solution using hyperplane rounding."""
    # Eigenvalue decomposition
    try:
        eigenvals, eigenvecs = eigh(X)
        # Keep only positive eigenvalues
        keep = eigenvals > 1e-8
        V = eigenvecs[:, keep]
        eigenvals = eigenvals[keep]
        # Compute square root of eigenvalues
        sqrt_eigenvals = np.sqrt(eigenvals)
        # V is the square root factor
        V_sqrt = V * sqrt_eigenvals[None, :]

        # Generate a random Gaussian vector
        g = np.random.randn(V_sqrt.shape[1])
        # Project V onto g
        projections = V_sqrt.T @ g
        # Get signs
        s = np.sign(projections)

        # If we have an odd number of elements, adjust the sum
        s_sum = np.sum(s)
        if s_sum != 0:
            # Find smallest projection (by absolute value)
            idx = np.argmin(np.abs(projections))
            s[idx] = -s[idx]

        # Convert to binary h
        h = (s + 1) / 2
        h = np.clip(h, 0, 1)
        return h
    except Exception:
        # fallback: just return a random binary vector with correct weight if possible
        h = np.random.rand(n) > 0.5
        # adjust so that sum is n/2
        current_sum = np.sum(h)
        diff = int(n / 2) - current_sum
        if diff > 0:
            # flip some 0s to 1s
            indices_0 = np.where(h == 0)[0]
            flip_indices = np.random.choice(indices_0, diff, replace=False)
            h[flip_indices] = 1
        elif diff < 0:
            # flip some 1s to 0s
            indices_1 = np.where(h == 1)[0]
            flip_indices = np.random.choice(indices_1, -diff, replace=False)
            h[flip_indices] = 0
        return h


def scale_to_target_length(h_small, n_target):
    """Scale h_small up to n_target using simple concatenation with complement."""
    h_double = np.concatenate([h_small, 1 - h_small])
    if len(h_double) >= n_target:
        return h_double[:n_target]
    else:
        # Keep scaling up
        while len(h_double) < n_target:
            h_double = np.concatenate([h_double, 1 - h_double])
        return h_double[:n_target]


def hill_climb(h_start, C_d_list):
    """Perform hill climbing to refine h."""
    h = np.array(h_start.copy())
    n = len(h)
    max_corr = compute_full_correlation(h)
    improved = True
    max_passes = 20
    passes = 0

    while improved and passes < max_passes:
        improved = False
        for i in range(n):
            for j in range(n):
                if h[i] == 1 and h[j] == 0:
                    # Try swapping
                    h[i], h[j] = 0, 1
                    new_max_corr = compute_full_correlation(h)
                    if new_max_corr < max_corr:
                        max_corr = new_max_corr
                        improved = True
                    else:
                        h[i], h[j] = 1, 0
        passes += 1
    return h, max_corr

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5
