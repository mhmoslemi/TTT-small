import numpy as np
from cvxpy import *
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def compute_C5(h_vals, dx):
        """Evaluate C5 for given h using the provided evaluator"""
        return evaluate_erdos_solution(np.array(h_vals), dx)

    def build_and_solve_lp(n):
        """Build and solve the LP relaxation for the given n"""
        dx = 2.0 / n
        
        # Decision variables
        x = Variable(n)
        t = Variable(1)
        
        # Constraints
        constraints = [x >= 0, x <= 1, sum(x) == n/2]
        
        # Add auxiliary variables and constraints for all valid (i,s) pairs
        # This is a simplified approach - we'll compute the correlation matrix manually
        # For LP formulation, we need to be careful with indexing
        # All w_{i,s} are added as variables (total O(n^2) variables)  
        # To avoid excessive memory, let's build constraints directly
        # Calculate parameters of shifts
        shifts = list(range(-(n-1), n)) # -99 to 99 for n=100
        
        # Generate list of valid (i,s) pairs for which w_{i,s} is relevant
        valid_pairs = []  # List of (i,s) pairs
        for s in shifts:
            for i in range(n):
                if 0 <= i+s < n:
                    valid_pairs.append((i,s))
        
        # Create all w_{i,s} variables at once, but dict indexing is more readable
        w_vars = {}
        for i, s in valid_pairs:
            w_vars[(i,s)] = Variable(1)
            
        # Add McCormick constraints
        for i, s in valid_pairs:
            w = w_vars[(i,s)]
            constraints.append(w <= x[i])
            constraints.append(w <= x[i+s])
            constraints.append(w >= x[i] + x[i+s] - 1)
            constraints.append(w >= 0)
            
        # Add min correlation constraints for all shifts
        for s in shifts:
            # We must compute correlation sum for this s
            # This is the sum of valid w_{i,s} 
            shift_sum = 0
            for i in range(n):
                if 0 <= i+s < n:
                    shift_sum += w_vars[(i,s)]
            constraints.append(shift_sum >= t)
            
        # Objective
        objective = Maximize(t)
        
        # Create problem and solve
        prob = Problem(objective, constraints)
        
        try:
            # Use ECOS solver which is fast and handles this scale well
            prob.solve(solver=ECOS, abstol=1e-8, reltol=1e-6, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Retrieve fractional solution
                x_sol = x.value
                t_sol = t.value
                return x_sol, t_sol, False  # Not a rounding failure
            else:
                return None, None, True  # Solver failure
        except:
            return None, None, True  # Solver exception
            
    def deterministic_rounding(x_frac, n):
        """Round the fractional solution to a binary vector maintaining mass"""
        # Sort indices by decreasing x values
        indices_sorted = np.argsort(-x_frac)
        # Make binary vector with exactly n/2 ones
        b = np.zeros(n)
        b[indices_sorted[:n//2]] = 1
        return b
            
    def local_search_binary(h_binary, dx, max_swaps=30000):
        """Perform a swap-based local search"""
        h_best = h_binary.copy()
        C5_best = compute_C5(h_best, dx)
        
        swaps_done = 0
        no_improvement_count = 0
        max_no_improvement = 2000
        temp = 0.5
        cooling_rate = 0.99995
        start_temp = temp
        
        while swaps_done < max_swaps and (time.time() - start_time) < 0.95 * budget_s:
            swaps_done += 1
            
            # Pick a random 1 and a random 0
            ones = np.where(h_best == 1)[0]
            zeros = np.where(h_best == 0)[0]
            
            if len(ones) == 0 or len(zeros) == 0:
                break
                
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            # Try the swap
            h_trial = h_best.copy()
            h_trial[i] = 0
            h_trial[j] = 1
            
            C5_trial = compute_C5(h_trial, dx)
            
            if C5_trial < C5_best:
                h_best = h_trial.copy()
                C5_best = C5_trial
                no_improvement_count = 0
            else:
                # Simulated annealing acceptance
                delta = C5_trial - C5_best
                if delta > 0 and np.random.rand() < np.exp(-delta / temp):
                    h_best = h_trial.copy()
                    C5_best = C5_trial
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
            # Cool down
            temp = max(1e-4, temp * cooling_rate)
            
            # Early stopping
            if no_improvement_count > max_no_improvement:
                break
                
        return h_best, C5_best
            
    
    # Multi-resolution loop
    resolutions = [80, 100]
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    for n in resolutions:
        if (time.time() - start_time) > 0.9 * budget_s:
            break
            
        x_sol, t_sol, fail = build_and_solve_lp(n)
        if fail or x_sol is None:
            continue
            
        # Multiple random roundings (few to limit time)
        for i in range(3):
            if (time.time() - start_time) > 0.9 * budget_s:
                break
                
            h_rounded = deterministic_rounding(x_sol, n)
            dx = 2.0 / n
            
            h_final, C5_val = local_search_binary(h_rounded, dx)
            
            if C5_val < best_c5:
                best_c5 = C5_val
                best_h = h_final.copy()
                best_n = n
    
    # Final polishing
    if (time.time() - start_time) < 0.99 * budget_s and best_h is not None:
        dx = 2.0 / best_n
        # Try short gradient loop on best solution
        try:
            # Use simple projected gradient approach with small steps
            h = best_h.copy()
            lr = 0.01
            num_iter = 5
            for _ in range(num_iter):
                if (time.time() - start_time) > 0.99 * budget_s:
                    break
                    
                # Simple gradient estimate via finite differences
                grad = np.zeros(best_n)
                eps = 1e-4
                for k in range(best_n):
                    h_plus = h.copy()
                    h_minus = h.copy()
                    h_plus[k] = min(1.0, h[k] + eps)
                    h_minus[k] = max(0.0, h[k] - eps)
                    
                    C5_plus = compute_C5(h_plus, dx)
                    C5_minus = compute_C5(h_minus, dx)
                    
                    grad[k] = (C5_plus - C5_minus) / (2*eps)
                    
                # Update and project
                h = h - lr * grad
                
                # Project to feasible set [0,1] with sum = n/2
                h = np.clip(h, 0, 1)
                h = h * (n/2 / np.sum(h))
                
            # Final check
            final_C5 = compute_C5(h, dx)
            if final_C5 < best_c5:
                best_c5 = final_C5
                best_h = h.copy()
                
        except Exception:
            pass  # Skip if gradient polish fails

    if best_h is None:
        # Fallback to initial construction if needed
        h = np.array(initial_h_values)
        n = len(h)
        dx = 2.0 / n
        best_c5 = compute_C5(h, dx)
        best_h = h.tolist()
        best_n = n
            
    # Final evaluation
    if best_h is not None:
        dx_final = 2.0 / best_n
        best_c5 = compute_C5(best_h, dx_final)
        
    return (best_h.tolist(), best_c5, best_n)
