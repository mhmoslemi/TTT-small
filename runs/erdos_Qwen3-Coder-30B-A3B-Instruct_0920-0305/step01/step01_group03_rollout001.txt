import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx=2.0/200):
    """Evaluate the C5 value for a given h array"""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    correlation = np.correlate(h, 1-h, mode="full")
    # Find maximum value over all shifts
    max_corr = np.max(correlation)
    # Convert to actual C5 value
    return max_corr * dx

def project_onto_box_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the set {x: lower <= x_i <= upper, sum x_i = target_sum}"""
    # This is water-filling algorithm
    v = np.copy(v)
    n = len(v)
    # We solve shift theta such that sum_i clamp(v_i + theta, lower, upper) = target_sum
    # First, shift to account for lower bounds
    v_shifted = v - lower
    target_shifted = target_sum - n * lower
    
    # Binary search for theta
    low = -np.max(v_shifted) - 1.0
    high = np.max(1.0 - v_shifted) + 1.0
    
    # Additional safety check - if target is out of bounds, clip it to valid range
    if target_shifted < 0:
        target_shifted = 0.0
    if target_shifted > n:
        target_shifted = n
        
    # Convert to clipped values under upper bounds and target
    def evaluate_shift(theta):
        # For v_shifted, clamp to [0, upper-lower] = [0, 1]
        clipped = np.clip(v_shifted + theta, 0.0, 1.0)
        return np.sum(clipped)
    
    # Binary search for the right theta
    epsilon = 1e-10
    max_iter = 50
    for _ in range(max_iter):
        mid = (low + high) / 2
        val_mid = evaluate_shift(mid)
        if np.abs(val_mid - target_shifted) < epsilon:
            break
        if val_mid < target_shifted:
            low = mid
        else:
            high = mid
            
    theta = (low + high) / 2
    # Apply the shift then clip
    result = np.clip(v + theta, lower, upper)
    # Re-adjust for exact sum if needed:
    diff = target_sum - np.sum(result)
    if np.abs(diff) > 1e-10:
        # Perform adjustment by distributing difference evenly where possible
        indices = np.argsort(result)
        delta = diff / len(result)
        for idx in reversed(indices):
            if diff == 0:
                break
            old_val = result[idx]
            new_val = result[idx] + delta
            # New value clamped to bounds
            clamped = np.clip(new_val, lower, upper)
            if old_val != clamped:
                result[idx] = clamped
                diff -= (clamped - old_val)
    return result

def smooth_max_objective(h, corr_hh, lambda_val, target_sum):
    """Compute smooth approximation to 1 - min corr for optimization"""
    # corr_hh is correlation of h with itself
    # L = logsumexp( -lambda * corr_hh ) / lambda
    max_corr = np.max(corr_hh)
    # We compute exp(-lambda * (corr_hh - max_corr)) for numerical stability
    shifted_corr = corr_hh - max_corr
    exp_vals = np.exp(-lambda_val * shifted_corr)
    sum_exp = np.sum(exp_vals)
    log_sum_exp = np.log(sum_exp) + max_corr * lambda_val
    return log_sum_exp / lambda_val

def compute_gradient(h, corr_hh, lambda_val):
    """Compute gradient of smooth objective"""
    n = len(h)
    # Compute softmax weights for each shift
    max_corr = np.max(corr_hh)
    shifted_corr = corr_hh - max_corr
    exp_vals = np.exp(-lambda_val * shifted_corr)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    
    # Compute gradient
    grad = np.zeros_like(h)
    for offset in range(-(n-1), n):
        idx = offset + n - 1
        w = weights[idx]
        if w == 0:
            continue
        if offset >= 0:
            grad[:n-offset] += w * h[offset:]
            grad[offset:] += w * h[:n-offset]
        else:
            off = -offset
            grad[off:] += w * h[:n-off]
            grad[:n-off] += w * h[off:]
                
    return -grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum h[i] * dx = 1 => sum h[i] = target_sum
    lambda_schedule = [1, 5, 10, 20, 40]
    iterations_per_lambda = 200
    
    # Initial solutions
    initial_solutions = []
    
    # 1. From pre-imported initial_h_values (n=42)
    try:
        h_base = np.array(initial_h_values)
        h_upsampled = np.interp(np.linspace(0, 2, n_points, endpoint=False), 
                                np.linspace(0, 2, len(h_base), endpoint=False), 
                                h_base)
        h_upsampled = project_onto_box_simplex(h_upsampled, target_sum)
        initial_solutions.append(("upsampled_initial", h_upsampled.copy()))
    except:
        pass

    # 2. Random uniform initialization
    h_random = np.random.uniform(0, 1, n_points)
    h_random = project_onto_box_simplex(h_random, target_sum)
    initial_solutions.append(("random_uniform", h_random.copy()))
    
    # 3. Binary pattern (half ones, half zeros)
    h_binary = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, int(n_points/2), replace=False)
    h_binary[ones_indices] = 1.0
    h_binary = project_onto_box_simplex(h_binary, target_sum)
    initial_solutions.append(("binary_pattern", h_binary.copy()))
    
    # 4. Deterministic pattern (flat at center region)
    h_deterministic = np.ones(n_points) * 0.5
    center_block_size = n_points // 4  # Set central 1/4 to 1.0
    start_idx = 3*n_points//8
    h_deterministic[start_idx:start_idx+center_block_size] = 1.0
    h_deterministic = project_onto_box_simplex(h_deterministic, target_sum)
    initial_solutions.append(("deterministic", h_deterministic.copy()))

    # Optimize all initial solutions
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start optimization with smooth approximation
    num_restarts = 10  # Adjust based on remaining time
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Try different initial solutions
        for name, h_start in initial_solutions:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            h = h_start.copy()
            # Smooth optimization for each lambda in schedule
            for lambda_val in lambda_schedule:
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                for iter_batch in range(iterations_per_lambda):
                    if time.time() - start_time > budget_s * 0.95:
                        break
                        
                    # Compute correlation of h with h
                    corr_hh = np.correlate(h, h, mode='full')
                    min_corr = np.min(corr_hh)
                    # Evaluate raw objective for tracking
                    raw_obj = target_sum - min_corr
                    
                    if raw_obj < best_c5:
                        best_c5 = raw_obj
                        best_h = h.copy()
                    
                    # Compute gradient of smooth loss
                    grad = compute_gradient(h, corr_hh, lambda_val)
                    
                    # Gradient step
                    step_size = 0.01
                    h_new = h + step_size * grad
                    
                    # Project back onto feasible set
                    h = project_onto_box_simplex(h_new, target_sum)
                    
    # Final fine-tuning using SLSQP with exact objective
    def objective_func(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    if best_h is not None:
        result = minimize(
            objective_func,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500}
        )
        
        if result.success:
            final_c5 = objective_func(result.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result.x.copy()

    if best_h is None:
        # Fallback: use random solution
        best_h = np.random.uniform(0, 1, n_points)
        best_h = project_onto_box_simplex(best_h, target_sum)
        best_c5 = evaluate_erdos_solution(best_h, dx)

    return (best_h.tolist(), best_c5, n_points)
