```python
import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n = 80
    dx = 2.0 / n
    target_c5 = 0.3807

    # Build shift matrices and constraints
    def build_shift_matrix(n, shift):
        # Create a sparse matrix that selects h[i] * h[i+shift]
        # For indexing, we use Python conventions
        if shift == 0:
            return np.zeros((n, n))
        elif shift > 0:
            # X[i, i+shift] = 1 where valid
            rows = np.arange(n - shift)
            cols = np.arange(shift, n)
            data = np.ones_like(rows)
            return np.zeros((n, n)) + np.asarray(data, dtype=float)
        else:
            # shift < 0 -> X[i-shift, i] = 1 where valid
            rows = np.arange(-shift, n)
            cols = np.arange(n + shift)
            data = np.ones_like(rows)
            return np.zeros((n, n)) + np.asarray(data, dtype=float)

    # Create the SDP model
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2,
        cp.diag(X) == h,
        X >> 0,  # PSD
        t >= 0
    ]

    # McCormick Envelope constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints += [
                X[i, j] >= 0,
                X[i, j] <= h[i],
                X[i, j] <= h[j],
                X[i, j] >= h[i] + h[j] - 1
            ]

    # Overlap constraints for each shift k
    for s in range(-(n-1), n):
        if s == 0:
            continue
        shift_mat = build_shift_matrix(n, s)
        constraints += [t >= n/2 - cp.sum(cp.multiply(shift_mat, X))]

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    # Solver settings
    try:
        prob.solve(solver='SCS', eps=1e-5, max_iters=5000, verbose=False)
        if prob.status != cp.OPTIMAL:
            raise Exception("SDP did not converge")
    except Exception as e:
        print("SDP Solver Error:", e)
        # Fallback to start from centered initial h
        h0 = np.ones(n) * 0.5
        return (h0.tolist(), 0.5, n)
    
    h_sdp = h.value
    t_sdp = t.value
    X_sdp = X.value

    # Output lower bound (不太精确，因为不是实际 C5，仅用于 Follow-up 了解模型)
    # 这意味着: lower bound_target = t_sdp * dx

    # Rounding step - do many samples for good candiates
    best_h = None
    best_c5 = float('inf')
    n_samples = 2000

    # Compute U = V * sqrt(Lambda) to generate samples from the covariance matrix
    try:
        eigenvals, eigenvecs = eigh(X_sdp)
        positive_idx = eigenvals > 1e-8
        eigenvals = eigenvals[positive_idx]
        eigenvecs = eigenvecs[:, positive_idx]

        U = eigenvecs @ np.diag(np.sqrt(eigenvals))
    except Exception:
        U = np.eye(n) * np.sqrt(1.0)

    # Random sampling with projection onto sum constraint
    for _ in range(n_samples):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate sample from covariance
        z = U @ np.random.randn(U.shape[1])
        z = np.clip(z, 0, 1)
        
        # Project to sum constraint
        original_sum = np.sum(z)
        alpha = (n/2 - original_sum) / n
        y = np.clip(z + alpha, 0, 1)
        needed = n / 2 - np.sum(y)
        if abs(needed) > 1e-6:
            # Adjust by modifying largest values
            diff_arr = np.ones_like(y) * needed
            y = np.clip(y + diff_arr, 0, 1)
            while abs(np.sum(y) - n/2) > 1e-6:
                k = np.argmax(y)
                y[k] = min(y[k] + 0.0001, 1)
                if abs(np.sum(y) - n/2) < 1e-6:
                    break
        
        y = y / np.sum(y) * (n/2)
        h_candidate = np.clip(y, 0, 1)

        # Evaluate C5
        c5_value = evaluate_h(h_candidate)
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h_candidate.copy()

    # If we did not find a good candidate, fallback to deterministic (good) start
    if best_h is None:
        # Try deterministic rounding
        h_rnd = h_sdp.copy()
        # Binary rounding: set to 1 if > 0.5, else 0. Then fix the sum
        h_bin = (h_rnd > 0.5).astype(float)
        cnt = np.sum(h_bin)
        if cnt > n/2:
            # Need to reduce
            indices = np.argsort(h_rnd)[::-1]  # Sort by magnitude descending
            for i in range(int(cnt - n/2)):
                h_bin[indices[i]] = 0
        elif cnt < n/2:
            # Need to increase
            indices = np.argsort(h_rnd)  # ascending
            for i in range(int(n/2 - cnt)):
                h_bin[indices[i]] = 1
        best_h = h_bin
        best_c5 = evaluate_h(best_h)

    # Final optimization with local search if time allows
    if time.time() - start_time < budget_s - 10:
        V = np.copy(best_h)
        best_c5 = evaluate_h(V)
        # Simple hill-climbing -- swap one 0 with a 1 each time
        for _ in range(10000):
            if time.time() - start_time > budget_s - 10:
                break
            # Find all 0s and 1s
            zeros = np.where(V == 0)[0]
            ones = np.where(V == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # Try swap
            V_new = V.copy()
            V_new[i], V_new[j] = V_new[j], V_new[i]
            c5_new = evaluate_h(V_new)
            if c5_new < best_c5:
                V = V_new
                best_c5 = c5_new
        best_h = V

    return (best_h.tolist(), best_c5, n)
```
