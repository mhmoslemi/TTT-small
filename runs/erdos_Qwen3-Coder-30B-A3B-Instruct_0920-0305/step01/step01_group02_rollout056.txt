import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def project_to_feasible(x, target_sum):
    """Project vector x onto the simplex with sum = target_sum and bounds [0,1]"""
    x = np.array(x, dtype=float)
    # Simple projection: clip and normalize
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale = target_sum / current_sum if current_sum > 0 else 1.0
    x = x * scale
    return x

def evaluate_erdos_solution(h_values):
    """Evaluate the maximum overlap for given h values"""
    h = np.array(h_values)
    # Compute correlations for all lags
    correlation = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx = 2/N
    dx = 2.0 / len(h)
    correlation *= dx
    # Return maximum correlation value
    return np.max(correlation)

def compute_perturbed_solution(h, i, j, delta):
    """Compute a perturbed solution"""
    h_proposed = h.copy()
    h_proposed[i] += delta
    h_proposed[j] -= delta
    return h_proposed

def refine_solution(h_init, target_sum, max_iter=20000):
    """Use Nelder-Mead optimization to refine solution"""
    def objective(h):
        # Project to feasible set
        h_feasible = project_to_feasible(h, target_sum)
        return evaluate_erdos_solution(h_feasible)
    
    # Use scipy.optimize.minimize with Nelder-Mead
    result = minimize(objective, h_init, method='Nelder-Mead', 
                      options={'maxiter': max_iter, 'xatol': 1e-6, 'fatol': 1e-6})
    
    if result.success:
        h_opt = project_to_feasible(result.x, target_sum)
        return h_opt, evaluate_erdos_solution(h_opt)
    else:
        return h_init, evaluate_erdos_solution(h_init)
