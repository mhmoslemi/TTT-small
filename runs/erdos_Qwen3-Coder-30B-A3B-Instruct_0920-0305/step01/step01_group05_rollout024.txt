import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    
    # Compute the overlap integral using correlation
    # This is equivalent to max_k int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def build_shift_masks(n):
    """Precompute shift masks for efficient constraint building"""
    masks_fwd = {}
    masks_bwd = {}
    for s in range(-(n-1), n):
        mask_fwd = np.arange(n)
        if s >= 0:
            mask_fwd = mask_fwd[:-s]
        else:
            mask_fwd = mask_fwd[-s:]
        masks_fwd[s] = mask_fwd
        
        mask_bwd = np.arange(n)
        if s <= 0:
            mask_bwd = mask_bwd[-s:]
        else:
            mask_bwd = mask_bwd[:-s]
        masks_bwd[s] = mask_bwd
    return masks_fwd, masks_bwd

def compute_gradient_gd(h, s, masks_fwd, masks_bwd):
    """Compute gradient of g_d^s at h"""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        grad[masks_fwd[s]] += 2 * (h[masks_fwd[s]] + h[masks_fwd[s] - s])
    else:
        grad[masks_bwd[s]] += 2 * (h[masks_bwd[s]] + h[masks_bwd[s] + s])
    return grad

def ccp_solve(h_prev, n, masks_fwd, masks_bwd, dx):
    """Solve the CCP subproblem using CVXPY"""
    h = cp.Variable(n)
    tau = cp.Variable()
    
    # Constraint for all shifts
    constraints = []
    for s in range(-(n-1), n):
        # g_s^c(h) = sum(h) + 0.5*sum(h[i]^2 + h[i+s]^2)
        if s >= 0:
            convex_term = cp.sum(h) + 0.5 * cp.sum_squares(h[masks_fwd[s]]) + 0.5 * cp.sum_squares(h[masks_fwd[s] - s])
        else:
            convex_term = cp.sum(h) + 0.5 * cp.sum_squares(h[masks_bwd[s]]) + 0.5 * cp.sum_squares(h[masks_bwd[s] + s])
        
        # g_s^d(h) = 0.5 * sum((h[i] + h[i+s])^2)
        if s >= 0:
            grad_s = compute_gradient_gd(h_prev, s, masks_fwd, masks_bwd)
            dual_term = 0.5 * cp.sum_squares(h[masks_fwd[s]] + h[masks_fwd[s] - s])
        else:
            grad_s = compute_gradient_gd(h_prev, s, masks_fwd, masks_bwd)
            dual_term = 0.5 * cp.sum_squares(h[masks_bwd[s]] + h[masks_bwd[s] + s])
        
        dual_term_prev = 0.5 * cp.sum_squares(h_prev[masks_fwd[s]] + h_prev[masks_fwd[s] - s]) if s >= 0 else 0.5 * cp.sum_squares(h_prev[masks_bwd[s]] + h_prev[masks_bwd[s] + s])
        affine_term = cp.sum(grad_s * (h - h_prev)) + dual_term_prev
        constraints.append(tau >= convex_term - affine_term)
    
    # Box constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    # Integral constraint
    constraints.append(cp.sum(h) == n / 2)
    
    # Solve the convex program
    prob = cp.Problem(cp.Minimize(tau), constraints)
    try:
        prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iters=1000)
        h_new = h.value
    except:
        # If solver fails, return previous solution
        h_new = h_prev.copy()
    return h_new
