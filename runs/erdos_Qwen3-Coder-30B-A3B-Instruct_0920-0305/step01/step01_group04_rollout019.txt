def full_correlation(h):
    """Compute full correlation vector and return max value."""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return corr, max_corr

def solve_sdp(n):
    """
    Solve SDP relaxation:
    max tau
    s.t. trace(C_d @ X) >= tau for all d != 0, X >> 0, diag(X)=1
    """
    X = cp.Variable((n, n), symmetric=True)
    tau = cp.Variable()

    # Construct Toeplitz shift matrices
    C_d_list = []
    for d in range(-n+1, n):
        if d == 0:
            continue
        C_d = np.zeros((n, n))
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                C_d[i, j] = 1
                C_d[j, i] = 1
        C_d_list.append(C_d)

    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == 1,  # Diagonal is 1
    ]

    # Add autocorrelation constraints
    for C_d in C_d_list:
        constraints.append(cp.trace(C_d @ X) >= tau)

    # Objective
    objective = cp.Maximize(tau)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4, max_iters=5000)

    if prob.status != cp.OPTIMAL:
        return None, None

    return X.value, tau.value

def round_and_balance(X, n):
    """Apply Goemans-Williamson rounding and adjust for balanced weight."""
    # Eigen-decomposition of X
    try:
        w, V = eigh(X)
        # Keep only positive eigenvalues
        eps = 1e-8
        V = V[:, w > eps]
        w = w[w > eps]
        sqrt_w = np.sqrt(w)
        # V * sqrt_w is the factorization V * sqrt_w * sqrt_w^T = V * w * V^T = X
        # So we sample random vector g and round s = sign(V^T * g)
        g = np.random.randn(V.shape[1])
        s = np.sign(V.T @ g)
        # Ensure s sums to 0 (so h = (s+1)/2 sums to n/2)
        S = np.sum(s)
        if abs(S) > 0:
            # Flip the sign of the entry with smallest |s| => smallest change in s^T X s
            flip_idx = np.argmin(np.abs(s))
            s[flip_idx] *= -1
        h = (s + 1) / 2
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-6:
            # Adjust using small flips
            while abs(h_sum - 1.0) > 1e-6:
                if h_sum > 1:
                    idx = np.argmax(h)
                    h[idx] -= dx
                else:
                    idx = np.argmin(h)
                    h[idx] += dx
                h_sum = np.sum(h) * dx
        return h
    except Exception:
        return None

def upscale_scale(h, n_target):
    """Combine with complement to reach required length."""
    # Pad with complement
    if len(h) >= n_target:
        return h[:n_target]
    # Double sequence by concatenating with complement
    h = np.array(h)
    h_big = np.concatenate([h, 1 - h])
    if len(h_big) >= n_target:
        return h_big[:n_target]
    # If still not large enough, we could downscale with trimming or pad at later stage
    else:  # Double again
        h_big = np.concatenate([h_big, 1 - h_big])
        if len(h_big) > n_target:
            h_big = h_big[:n_target]
        return h_big

def hill_climb(h, n_iter=10):
    """Local binary hill climbing to reduce C5."""
    h = np.array(h)
    corr, best_M = full_correlation(h)
    for _ in range(n_iter):
        improved = False
        for i in range(len(h)):
            if h[i] == 0:
                continue
            for j in range(len(h)):
                if h[j] == 1:
                    continue
                # Swap i and j
                h[i], h[j] = h[j], h[i]
                corr_new, M_new = full_correlation(h)
                if M_new < best_M:
                    best_M = M_new
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]  # Revert if not better
        if not improved:
            break
    return list(h), best_M
