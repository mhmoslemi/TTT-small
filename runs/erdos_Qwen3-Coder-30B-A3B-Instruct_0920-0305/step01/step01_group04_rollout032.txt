def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_feasible(h):
    """Project h onto [0,1]^n and sum = n/2."""

    h = np.clip(h, 0, 1)

    h_sum = np.sum(h)
    epsilon = (n_points / 2.0 - h_sum) / n_points
    h += epsilon
    h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, d_star, alpha):

        grad[i] = 1 - h[i_plus] - h[i_minus]
    h_new = h - alpha * grad  # Potential next h
    h_new = project_onto_feasible(h_new)
    return h_new
