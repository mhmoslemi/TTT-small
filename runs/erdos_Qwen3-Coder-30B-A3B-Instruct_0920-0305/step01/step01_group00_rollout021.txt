import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def proj_onto_simplex(v, target_sum, lower=0.0, upper=1.0):
    # v: array
    # return x solving min ||x - v||^2 subject to sum x = target_sum, lower <= x <= upper
    # Use bisection on lambda.
    n = len(v)
    # clamp to lower and upper? Actually we need to find lambda such that x = clamp(v - lambda, lower, upper) sums to target.
    # Define function f(lambda) = sum(clamp(v - lambda, lower, upper)) - target_sum.
    # f is decreasing in lambda (since clamp of v - lambda). So we can binary search.
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, lower, upper)
        s = np.sum(x)
        if s > target_sum:
            lo = lam  # need larger lambda to reduce sum
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, lower, upper)
    return x

def propose_move(h, target_sum):
    # pick two distinct random indices
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # compute max increase/decrease at i
    max_up_i = 1 - h[i]
    max_down_i = h[i]
    # compute max increase/decrease at j
    max_up_j = 1 - h[j]
    max_down_j = h[j]
    # We want delta such that h[i] + delta ∈ [0,1], h[j] - delta ∈ [0,1].
    # So delta ∈ [ -max_down_i, max_up_i] intersect [-max_up_j, max_down_j]
    delta_low = max(-max_down_i, -max_up_j)
    delta_high = min(max_up_i, max_down_j)
    if delta_low >= delta_high:
        return None
    delta = np.random.uniform(delta_low, delta_high)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    # Ensure sum constraint is preserved exactly
    sum_diff = np.sum(h_new) - target_sum
    if abs(sum_diff) > 1e-10:
        # Redistribute difference to make sum correct
        # More robustly, just adjust so that sum matches target_sum
        h_new = h_new - sum_diff / n
    return h_new, delta, i, j
