import numpy as np

from scipy.linalg import eigh
from scipy.sparse import diags
import time

def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / len(h_values)
    h = np.array(h_values)
    if np.sum(h) * dx != 1.0:
        h = h * (1.0 / (np.sum(h) * dx))
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5
