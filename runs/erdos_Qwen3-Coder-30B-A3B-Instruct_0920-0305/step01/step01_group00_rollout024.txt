import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time


def project_simplex(y, s=0.5):
    """Project y onto the simplex {x : sum x = s, 0 <= x <= 1} using bisection."""
    # Ensure that we're working with a copy
    y = np.array(y, dtype=float)
    n = len(y)
    # Ensure the projection is within [0, 1]
    y = np.clip(y, 0, 1)
    # Binary search for lambda
    l, u = 0.0, np.max(y)
    for _ in range(100):
        if l >= u - 1e-10:
            break
        lam = (l + u) / 2.0
        x = np.clip(y - lam, 0, 1)
        if np.sum(x) < s:
            u = lam
        else:
            l = lam
    lam = (l + u) / 2.0
    x = np.clip(y - lam, 0, 1)
    return x


def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx


def soft_max_approx(h, alpha):
    """Compute (1/alpha) * log(sum of exp(alpha * c_s))"""
    # Correlation full mode
    corr = correlate(h, 1 - h, mode="full")
    c_max = np.max(corr)
    c_shifted = corr - c_max
    exp_c = np.exp(alpha * c_shifted)
    return (1.0 / alpha) * np.log(np.sum(exp_c))


def compute_gradient(h, alpha):
    """Compute the gradient of the smooth max objective using the weighted shifts."""
    # Compute correlations (unscaled)
    corr = correlate(h, 1 - h, mode="full")

    # Softmax weights (subtracted from max to prevent overflow)
    c_max = np.max(corr)
    c_shifted = corr - c_max
    exp_c = np.exp(alpha * c_shifted)
    weights = exp_c / np.sum(exp_c)

    # Compute gradient
    n = len(h)
    grad = np.zeros_like(h)

    # Loop over all shifts
    n_corr = len(corr)
    for s in range(n_corr):
        # Equivalent to one full shift of h -> h[s:]
        if s < n:
            h_shifted = np.roll(h, -s, axis=0)
        else:
            h_shifted = np.roll(h, -(s - n + 1), axis=0)

        # For s=0 shifted by 0 to h, for s=1 h[1:], s=2 h[2:] etc.
        # The forward shift and backward
        if s > 0:
            # Shift h forward by s, pad with zeros at the end
            left_acc = np.zeros_like(h)
            right_acc = np.zeros_like(h)
            if n - s > 0:
                left_acc[s:] = h[:-s]
                right_acc[:-s] = h[s:]
            else:
                left_acc = np.zeros_like(h)
                right_acc = h
        else:
            # s = 0 -> identity, no shift
            left_acc = h
            right_acc = h

        # Just sum of left and right (shifted) copies
        grad_component = 1.0 - (left_acc + right_acc)
        grad += weights[s] * grad_component

    return grad


def step_size_line_search(h, grad, step_size, alpha, current_c5):
    """Line search to compute a valid step size."""
    for _ in range(10):  # max 10 backtracks
        h_try = h - step_size * grad
        h_try = project_simplex(h_try, 0.5)
        # Evaluate objective with soft max
        new_c5 = soft_max_approx(h_try, alpha)
        if new_c5 < current_c5 * 0.999:
            return h_try, step_size
        step_size *= 0.5
    return h, step_size


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 96
    dx = 2.0 / n_points
    s = n_points / 2.0  # target sum

    # Initial h from the `initial_h_values` global variable if available
    try:
        initial_h = np.array(initial_h_values)
    except NameError:
        initial_h = np.random.rand(n_points)
        initial_h = project_simplex(initial_h, s)

    # Linear interpolation from initial_h to current grid
    if len(initial_h) != n_points:
        x_orig = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(x_new, x_orig, initial_h)

    # Project to correct sum
    initial_h = project_simplex(initial_h, s)

    # Parameters
    alpha0 = 5.0
    alpha_rate = 1.03
    max_alpha = 200.0
    max_iter_per_restart = 5000
    patience = 100  # early stop if no improvement for this many steps

    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)

    # Multi-restart loop
    restart_count = 0
    while (time.time() - start_time) < 0.7 * budget_s and restart_count < 100:
        current_h = initial_h.copy() if restart_count == 0 else project_simplex(np.random.rand(n_points), s)

        alpha = alpha0
        prev_c5 = np.inf
        patience_counter = 0

        iter_count = 0
        while iter_count < max_iter_per_restart and (time.time() - start_time) < budget_s:

            if alpha < max_alpha:
                alpha = min(alpha * alpha_rate, max_alpha)

            # Compute gradient (one per restart)
            grad = compute_gradient(current_h, alpha)
            # Normalize gradient for step size
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step_size = 0.5 / np.sqrt(iter_count + 1)
            current_h = current_h - step_size * grad
            current_h = project_simplex(current_h, s)

            # Evaluate true C5
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                patience_counter = 0
            else:
                patience_counter += 1

            if patience_counter > patience:
                break

            iter_count += 1

        restart_count += 1
        if best_c5 < 0.3807:
            break

    # Final polishing with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        def obj_func(h):
            return evaluate_c5(h)

        def cons_eq(h):
            return np.sum(h) * dx - 1.0

        cons = {'type': 'eq', 'fun': cons_eq}
        res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            polished_h = res.x
            polished_c5 = evaluate_c5(polished_h)
            if polished_c5 < best_c5:
                best_h = polished_h
                best_c5 = polished_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
