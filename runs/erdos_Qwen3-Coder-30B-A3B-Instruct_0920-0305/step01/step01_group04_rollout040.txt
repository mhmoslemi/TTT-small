```python
import numpy as np
from scipy.optimize import dual_annealing
from scipy.optimize import minimize
import time
import math

# Constants
n_points = 200
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def objective(h_values):
    """Objective function to minimize: C_5 bound."""
    return evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def generate_initial_solution():
    """Generate an initial solution based on known good constructions."""
    # Try to use a construction near ideal based on quadratic residues
    # or a carefully designed pattern
    
    h = np.zeros(n_points)
    
    # Simple heuristic: try pattern with 99 ones and 101 zeros
    # Place 1s in a way to minimize overlap
    
    # We use a simple pattern for initial construction 
    # avoid clustering ones too much
    mid = n_points // 2
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            h[i] = 1.0
        # leave others as 0.0 initially
    
    # But we also need to respect constraint sum*h*dx = 1
    # So we need to fix the sum to be exactly 100 = n_points / 2
    
    actual_sum = np.sum(h)
    required_sum = n_points / 2
    
    if actual_sum < required_sum:
        # Add some ones to make up the difference
        diff = int(required_sum - actual_sum)
        for i in range(diff):
            h[i] = 1.0
    elif actual_sum > required_sum:
        # Remove some ones
        diff = int(actual_sum - required_sum)
        for i in range(n_points-1, n_points-1-diff, -1):
            h[i] = 0.0
    
    return h

def local_search_optimize(initial_h):
    """Perform a local search to improve initial solution."""
    # Use the scipy optimizer with better tolerance
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(objective, initial_h, method='SLSQP', 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            return result.x.copy()
    except Exception:
        pass
    
    return initial_h

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    np.random.seed(seed)
    start_time = time.time()

    # Generate initial solution
    best_h = generate_initial_solution()
    best_c5 = evaluate_h(best_h)
    
    # Try to improve via local optimization
    improved_h = local_search_optimize(best_h)
    improved_c5 = evaluate_h(improved_h)
    
    if improved_c5 < best_c5:
        best_h = improved_h
        best_c5 = improved_c5

    # Run global optimization using different strategies
    try:
        # Try dual annealing on the improved solution
        bounds = [(0, 1) for _ in range(n_points)]
        result_da = dual_annealing(objective, bounds, maxiter=500, 
                                   initial_temp=100, restart_temp_ratio=1e-4,
                                   maxfun=1500, seed=seed)
        if result_da.success:
            new_h = result_da.x
            new_c5 = evaluate_h(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
    except Exception as e:
        pass
    
    # Final local optimization to refine
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            final_h = result.x
            final_c5 = evaluate_h(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass
    
    # Ensure integral is satisfied
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if abs(sum_h * dx - 1.0) > 1e-6:
        scale = 1.0 / (sum_h * dx)
        best_h *= scale
    
    return list(best_h), best_c5, n_points
```
