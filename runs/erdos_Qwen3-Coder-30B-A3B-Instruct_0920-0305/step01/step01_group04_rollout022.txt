import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

# -- Global constants --
n_points = 100  # Good balance between detail and solvability
dx = 2.0 / n_points
target_bound = 0.3807

# Use a high-resolution initialization if possible
try:
    h0_init = np.array(initial_h_values)
except NameError:
    # Default binary init with exact average area
    h0_init = np.zeros(n_points)
    h0_init[:n_points // 2] = 1.0

# Precomputed core setup
def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5, max_correlation

def project_onto_box_plus_sum(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    diff = target_sum - h_sum
    if abs(diff) < 1e-10:
        return h.copy()
    # Adjust using water-filling approach
    if diff > 0:
        # Need to increase values
        indices = np.argsort(h)
        for i in reversed(indices):
            if diff <= 0:
                break
            incr = min(diff, 1.0 - h[i])
            h[i] += incr
            diff -= incr
    else:
        # Need to decrease values
        indices = np.argsort(h)
        for i in indices:
            if diff >= 0:
                break
            decr = min(-diff, h[i])
            h[i] -= decr
            diff += decr
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    t0 = time.time()

    # --- Phase 1: Exact MILP (when possible) ---
    best_h = h0_init.copy()
    best_c5, best_M = evaluate_h(best_h)
    best_h = np.clip(best_h, 0, 1)
    if np.abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        scale = 1.0 / (np.sum(best_h) * dx)
        best_h *= scale
    best_c5, best_M = evaluate_h(best_h)

    # Run optimizations here only if there's sufficiently large budget.
    remaining_time = budget_s - (time.time() - t0)
    if remaining_time < 30:
        return list(best_h), best_c5, n_points

    # --- Phase 2: Heuristic Subgradient / Simulated Annealing ---
    def subgradient_step(h, d_star, alpha=0.5):
        """Performs one subgradient step with given shift d_star."""
        g = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            j_plus = (i + d_star) % n
            j_minus = (i - d_star) % n
            g[i] = 1 - h[j_plus] - h[j_minus]
        h_new = h - alpha * g
        return project_onto_box_plus_sum(h_new, n_points / 2.0)

    def evaluate_shifts(h):
        """Evaluate all m_d values efficiently."""
        h_rev = np.flip(h)
        corr = correlate(h, h_rev, mode='full')
        # The h * (1 - h) correlation is what we want
        ar = np.array(1 - h)
        corr2 = correlate(h, ar, mode='full')
        return corr2

    # Multi-start with deterministic and random starts
    best_total_h = best_h.copy()
    best_total_c5, best_total_M = best_c5, best_M

    start_h_list = [best_total_h]  # From previous phase
    # Add random binary start
    h_rand = np.zeros(n_points)
    h_rand[:n_points // 2] = 1.0
    np.random.shuffle(h_rand)
    start_h_list.append(h_rand)
    # Add Thue-Morse pattern
    tm_seq = []
    for i in range(n_points):
        t = sum(int(x) for x in bin(i)[2:]) % 2
        tm_seq.append(float(t))
    start_h_list.append(np.array(tm_seq))

    for start_idx, h_start in enumerate(start_h_list):
        h = h_start.copy()
        prev_c5, prev_M = evaluate_h(h)
        T = 1.0   # initial temperature
        cooling = 0.999
        alpha = 0.1  # learning rate for gradient descent step
        iter_count = 0
        max_iter = 50000
        while iter_count < max_iter:
            if time.time() - t0 > budget_s - 5.0:
                break

            # get all shifts score and select d*
            corr_vals = evaluate_shifts(h)
            # Remove index 0 (shift 0), which gives 0
            corr_vals = corr_vals[len(corr_vals)//2:]  # Not correct indexing axis trick,
            # let's do a real loop on shifts for now:

            d_star = 0
            max_val = -np.inf
            for d in range(-n_points+1, n_points):
                i_start = max(0, d)
                i_end = min(len(h), len(h) - d)
                if i_start >= i_end:
                    continue
                corr_val = np.sum(h[i_start:i_end] * (1 - h[i_start+d:i_end+d]))
                if corr_val > max_val:
                    d_star, max_val = d, corr_val

            # Perform update
            h_new = subgradient_step(h, d_star, alpha)

            _, new_M = evaluate_h(h_new)

            # Accept with probability
            delta = new_M - prev_M
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                prev_M = new_M
                _, prev_c5 = evaluate_h(h)
                if prev_M < best_total_M:
                    best_total_h = h.copy()
                    best_total_M = prev_M
                    best_total_c5 = prev_c5

            T *= cooling
            iter_count += 1

    # --- Phase 3: Final Rounding + Local Improvement ---
    h = best_total_h.copy()
    # Rounding to binary after initial continuous solve
    sorted_indices = np.argsort(h)[::-1]
    h_rnd = np.zeros_like(h)
    for i in range(n_points // 2):
        h_rnd[sorted_indices[i]] = 1.0

    if n_points % 2 == 1:
        # Ensure integral is correct
        h_rnd[sorted_indices[n_points // 2]] = 0.5

    # Local hill-climb swap for binary structure
    h_local = h_rnd.copy()
    improved = True
    while improved:
        improved = False
        prev_c5, _ = evaluate_h(h_local)
        for i in range(n_points):
            if h_local[i] != 1:
                continue
            for j in range(n_points):
                if h_local[j] != 0:
                    continue
                # Try swap
                h_local[i] = 0
                h_local[j] = 1
                new_c5, _ = evaluate_h(h_local)
                if new_c5 < prev_c5:
                    prev_c5 = new_c5
                    improved = True
                else:
                    h_local[i] = 1
                    h_local[j] = 0

    final_h = h_local.copy()
    final_c5, final_M = evaluate_h(final_h)

    return list(final_h), final_c5, n_points
