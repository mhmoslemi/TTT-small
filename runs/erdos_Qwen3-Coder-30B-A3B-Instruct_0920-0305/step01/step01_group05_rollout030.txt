def evaluate_h(h_values):
    """Evaluate the C5 bound for given h values"""
    # Normalize h to have integral = 1
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (1.0 / (current_sum * dx))
    
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def evaluate_c5_binary(bits):
    """Evaluate C5 for binary vector"""
    n = len(bits)
    dx = 2.0 / n
    h = bits.astype(float)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

        np.random.shuffle(bits)
        
        # Local descent
        improved = True
        while improved and time.time() - start < time_limit:
            improved = False
            # Try swaps
            for _ in range(2 * n):
                if time.time() - start >= time_limit:
                    break

                new_val = evaluate_c5_binary(bits)

                    bits[i], bits[j] = bits[j], bits[i]

                    # Accept improvement
                    improved = True

        best_bits[:half] = 1
        np.random.shuffle(best_bits)
        best_sum = evaluate_c5_binary(best_bits)
        
    return best_bits, best_sum
