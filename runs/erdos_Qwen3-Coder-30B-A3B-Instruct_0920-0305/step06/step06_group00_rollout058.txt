def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        grad = grad / (np.linalg.norm(grad) + 1e-12)

                h[j] += delta

            i = candidates[np.random.randint(len(candidates))]

            neighbors = [(i-1) % n_points, (i+1) % n_points]
            j = min(neighbors, key=lambda x: grad[x] if grad[x] < 0 else float('inf'))

            delta = min(h[i], 1 - h[j], step_size)
            if delta > 1e-6:
                h[i] -= delta
                h[j] += delta

        h = project_to_sum_box(h, TARGET_SUM)

        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
