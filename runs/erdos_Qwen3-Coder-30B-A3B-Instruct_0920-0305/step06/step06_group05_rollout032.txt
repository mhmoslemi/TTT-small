import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h_values):
    """Evaluate C5 using FFT for given h values"""
    h = np.array(h_values)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary h variables"""
    # Setup
    M = n // 2  # required number of ones
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Auxiliary variables for pairwise products
    num_pairs = n * (n - 1) // 2
    y = cp.Variable(num_pairs, boolean=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == M,    # Exact mass constraint
        t >= 0             # t is non-negative
    ]
    
    # Linking AND constraints
    y_index = 0
    for i in range(n):
        for j in range(i + 1, n):
            # y_ij = h_i AND h_j
            constraints.append(y[y_index] <= h[i])
            constraints.append(y[y_index] <= h[j])
            constraints.append(y[y_index] >= h[i] + h[j] - 1)
            y_index += 1
    
    # Lag constraints
    y_index = 0
    for k in range(1, n):
        sum_y = 0
        for i in range(n - k):
            sum_y += y[y_index]
            y_index += 1
        constraints.append(t >= M - sum_y)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.rint(h.value).astype(int)
            t_sol = t.value
            return h_sol, t_sol
    except:
        pass
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h_start):
    """Perform local continuous refinement using SLSQP"""
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - len(h) / 2
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = cp.minimize(
            objective, 
            h_start, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x, _ : None
        )
        if res.success:
            h_refined = res.x
            return project_to_sum_box(h_refined, len(h_start) / 2)
    except:
        pass
    return h_start

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    
    # Try different grid sizes
    for n in [60, 70, 80]:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        remaining_time = max(0.1, budget_s - (time.time() - start_time))
        if remaining_time < 0.1:
            break
            
        # Solve MILP
        h_bin, t_val = solve_milp(n, remaining_time * 0.5)
        if h_bin is not None:
            # Check the solution
            h_float = h_bin.astype(float)
            c5_eval = evaluate_c5(h_float)
            if c5_eval < best_c5:
                best_h = h_float.copy()
                best_c5 = c5_eval
                n_points = n
                
    # If no solution from ILP, fallback to the gradient-based method
    if best_h is None:
        # Run initial method
        from .initial_h_values import initial_h_values
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        n_points = 120
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, h0)
        h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
        best_h = h_interpolated.copy()
        best_c5 = evaluate_c5(best_h)
        
    # Optional continuous polishing
    if time.time() - start_time < 0.95 * budget_s:
        h_refined = local_continuous_refine(best_h)
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    return (best_h.tolist(), float(best_c5), n_points)
