import numpy as np
import time
from scipy.signal import correlate

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed):
    """Generate a random binary vector with exact sum constraint"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full correlation of x with itself"""
    return correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros, n):
    """Incrementally update overlap array after swapping positions i and j"""
    # Copy the current overlap array
    ov_new = ov.copy()
    
    # Compute shift indices for current 1-positions
    for k in ones:
        if k == i: continue
        # i is losing a 1 at position k -> remove overlaps from i
        if 0 <= k+i-j < 2*n-1:
            ov_new[k+i-j] -= 1
        if 0 <= i+k-j < 2*n-1:
            ov_new[i+k-j] -= 1
    # j is gaining a 1 at position k -> add overlaps to j
    for k in ones:
        if k == i: continue
        if 0 <= k+j-i < 2*n-1:
            ov_new[k+j-i] += 1
        if 0 <= j+k-i < 2*n-1:
            ov_new[j+k-i] += 1
    
    # Special case for the self-overlap (i,i) and (j,j)
    # Already handled: contribution of 1 at i to all others (from the two loops above)
    # contribution of 1 at j to all others (same)
    
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Run simulated annealing with swap moves"""
    start_time = time.time()
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = ov.min()
    
    # Initial temperature setting
    T0 = 0.5
    lambda_cool = 1.0 / 5000  # After 5000 steps, T drops to 0.5e-4
    
    # Time management
    max_iter = 1000000
    
    # Iteration tracking
    last_improvement = 0
    plateu_counter = 0
    max_plateau = 5000

    for iteration in range(max_iter):
        if time.time() - start_time > time_limit * 0.95:
            break
            
        # Temperature schedule
        T = T0 * np.exp(-lambda_cool * iteration)
        if iteration > 1000 and T < 1e-4:
            break

        # Pick a 1-index and a 0-index at random
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Propose the swap
        x_new = x.copy()
        x_new[i] = 0
        x_new[j] = 1
        
        # Update overlaps incrementally
        ov_new = swap_update(ov, i, j, ones, zeros, n_points)
        new_min = ov_new.min()
        
        # Compute delta
        delta = new_min - best_min
        
        # Acceptance rule
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            ov = ov_new
            best_min = new_min
            
            # Update indices
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            
            # Update best solution
            if new_min > best_min:
                best_x = x_new.copy()
                best_ov = ov_new
                best_min = new_min
                last_improvement = iteration
                
        # Plateau detection
        if iteration - last_improvement > max_plateau:
            break

    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Deterministic improvement leveraging the worst shift"""
    # Identify the worst shift
    s_star = np.argmin(ov) - (n_points - 1)
    
    # Set up the shifted indices
    ones_s = set((i+s_star) % n_points for i in ones)
    # Identify all relevant 0 positions
    zeros_s = set((i+s_star) % n_points for i in zeros)
    
    # Positions that lose 1 at s* (currently 1 at i, 0 at i+s*)
    lose_ones = [i for i in ones if (i+s_star) % n_points not in ones_s]
    
    # Positions that gain 1 at s* (currently 0 at i, 1 at i+s*)
    gain_zeros = [j for j in zeros if (j+s_star) % n_points in ones_s]
    
    # Check for any beneficial swaps
    # Priority: interact with those that would increase overlap by 1
    # But also check for new minimal value (now the old min would be better)
    changed = True
    while changed:
        changed = False
        temp_ones = set(ones)
        temp_zeros = set(zeros)
        temp_ov = ov.copy()
        
        # Prefer swapping within the above lists in some order
        for i in lose_ones:
            for j in gain_zeros:
                # Check if swap i->j maintains the minimum
                # First get new overlap array after the swap
                temp_ov_new = swap_update(temp_ov, i, j, temp_ones, temp_zeros, n_points)
                new_min = temp_ov_new.min()
                
                if new_min >= ov.min():
                    # Swap accepted
                    temp_ones.remove(i)
                    temp_ones.add(j)
                    temp_zeros.remove(j)
                    temp_zeros.add(i)
                    temp_ov = temp_ov_new
                    changed = True
                    ones = temp_ones
                    zeros = temp_zeros
                    ov = temp_ov
                    # Update lose_ones and gain_zeros based on new state
                    break
            if changed:
                break
    
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    np.random.seed(seed)
    start_time = time.time()
    
    # We'll later try a few restarts:
    best_c5 = float('inf')
    final_x = None
    
    num_restarts = 10
    for run_id in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initial seed could be a binary rounding of continuous vector or random binary
        x = initial_binary(seed + run_id)
        ov = full_overlap(x)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        
        # Run SA
        x_final, min_val = simulated_annealing(x, ov, ones, zeros, budget_s / num_restarts)
        ones_final = set(np.where(x_final == 1)[0])
        zeros_final = set(np.where(x_final == 0)[0])
        ov_final = full_overlap(x_final)
        
        # Final refinement
        x_refined, ov_refined = worst_shift_refine(x_final, ov_final, ones_final, zeros_final)
        new_min = ov_refined.min()
        c5 = dx * (TARGET_SUM - new_min)
        
        if c5 < best_c5:
            best_c5 = c5
            final_x = x_refined
            
    # Final conversion to float and return
    h_values = final_x.astype(float).tolist()
    
    return (h_values, float(best_c5), n_points)
