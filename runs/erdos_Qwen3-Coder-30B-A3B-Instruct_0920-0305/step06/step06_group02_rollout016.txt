import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h_current, step_size=0.01):
    """Basic gradient descent step"""
    corr = compute_corr_fft(h_current)
    max_idx = np.argmax(corr)
    
    # Compute gradient approximation using finite differences
    epsilon = 1e-4
    grad = np.zeros_like(h_current)
    
    for i in range(len(h_current)):
        h_plus = h_current.copy()
        h_plus[i] += epsilon
        h_plus = project_to_sum_box(h_plus, TARGET_SUM)
        corr_plus = compute_corr_fft(h_plus)
        max_plus = c5_from_corr(corr_plus, n_points)
        
        h_minus = h_current.copy()
        h_minus[i] -= epsilon
        h_minus = project_to_sum_box(h_minus, TARGET_SUM)
        corr_minus = compute_corr_fft(h_minus)
        max_minus = c5_from_corr(corr_minus, n_points)
        
        grad[i] = (max_plus - max_minus) / (2 * epsilon)
    
    # Update step
    h_new = h_current - step_size * grad
    
    # Project back to constraints
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def block_coordinate_descent(h0, max_iter=1000):
    """Optimize by updating variables in blocks"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for iter_num in range(max_iter):
        # Update in blocks of 5 coordinates
        block_size = 5
        for start_idx in range(0, n_points, block_size):
            end_idx = min(start_idx + block_size, n_points)
            
            # Try to improve each element in the block
            orig_block = h[start_idx:end_idx].copy()
            best_block = orig_block.copy()
            best_block_c5 = best_c5
            
            for i in range(start_idx, end_idx):
                # Try small variations
                for delta in [0.005, -0.005]:
                    temp_h = h.copy()
                    temp_h[i] += delta
                    if temp_h[i] >= 0 and temp_h[i] <= 1:
                        temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                        new_c5 = evaluate_c5(temp_h)
                        
                        if new_c5 < best_block_c5:
                            best_block_c5 = new_c5

            # Apply best block update if it improved
            if best_block_c5 < best_c5:
                h[start_idx:end_idx] = best_block
                best_c5 = best_block_c5
    
    return h
