import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def smooth_max_objective(h, alpha=1.0):
    """Smoothed max of correlation terms"""
    n = len(h)
    # Compute all correlations: h * (1-h) with full convolution
    corr = compute_corr_fft(h)
    # Scale by dx = 2/n
    dx = 2.0 / n
    corr = corr * dx
    
    # Compute smooth max of (1 - corr) terms = smooth min of corr terms
    # Use smooth max = log-sum-exp of -alpha * corr
    # But we want to minimize smooth max of C_k = 1 - corr_k
    # So minimize smooth max over all k of (1 - corr_k) = smooth max of (1 - corr)
    
    # Actually we want to maximize min correlation
    # minimize (max of (1 - corr) terms) = maximize min corr
    # which is equivalent to minimize (smooth max of (corr-1) terms) = minimize (smooth max of (1 - corr) terms)
    
    # Compute terms exp(-alpha * (1 - corr)) = exp(alpha * corr - alpha)
    # Then we want to minimize log( sum(exp(alpha*corr - alpha)) )
    # To make min corr high, we penalize when any corr is low
    
    # Finally: minimize smooth max of (1 - corr) = minimize -smooth min of corr
    # We can approximate smooth max of (1 - corr) by logsumexp of -alpha*(1 - corr) = logsumexp(alpha * corr - alpha)
    smoothed = np.log(np.sum(np.exp(alpha * corr - alpha))) - alpha
    return smoothed

def compute_gradient_smooth(h, alpha=1.0):
    """Compute gradient for smooth objective"""
    n = len(h)
    dx = 2.0 / n
    
    # Compute all correlations  
    corr = compute_corr_fft(h)
    corr = corr * dx
    
    # Compute the smooth max and its gradient
    # Objective derivative of -logsumexp(alpha * corr - alpha) w.r.t. h
    # This requires computing the gradient of each correlation term and summing them weighted by exp scaled terms
    
    # Compute softmax weights for correlations: w_k = exp(alpha * corr_k - alpha) / sum(exp(alpha * corr_k - alpha))
    # Similar to: w_k = exp(alpha * corr_k) / sum(exp(alpha * corr_k))
    exp_corr_weights = np.exp(alpha * corr)
    weight_sum = np.sum(exp_corr_weights)
    weights = exp_corr_weights / weight_sum
    
    # Gradient computation
    # Each term of the smooth max with weight w_k contributes -w_k * (1 - corr_k)
    # But we compute gradient w.r.t. h_i of the logsumexp, using chain rule
    # That involves derivatives of each correlation w.r.t. h
    
    # Use differentiation approach:
    # d/dh_i [logsumexp(a_k)] = (sum_k w_k * da_k/dh_i) where w_k = exp(a_k) / sum(exp(a_k))
    # For logsumexp(alpha*corr - alpha) = logsumexp(alpha*corr) - alpha
    # So gradient is (sum_k w_k * (d/dh_i) (alpha*corr_k)) = alpha * sum_k w_k * (d/dh_i corr_k)
    
    # Gradient contribution for each correlation term needed
    # But we want d/dh_i (corr_k) = d/dh_i sum_j h_j (1 - h_{j+k})
    # = (1 - h_{i+k}) + (1 - h_{i-k}) for k≠0, 2*(1 - h_i) for k=0
    # But we can compute this more efficiently
    
    # Let's compute gradients wrt to each correlation via FFT
    # For each shift k, derivative of corr_k w.r.t. h_i:
    # corr_k = sum_j h_j h_{j+k} (linear in h_j h_{j+k})
    # So: d corr_k /d h_i = h_{i+k} + h_{i-k} (if indices valid)
    
    # So we compute weighted sum of these contributions, but distributed correctly
    # We need to compute for each i: sum_k w_k * (h_{i+k} + h_{i-k}) (if valid)
    
    # To compute efficiently:
    grad = np.zeros(n)
    
    # For k term, gradient contribution:
    # each h_i gets contribution weighted by (h_{i+k} + h_{i-k}) multiplied by w_k
    for k in range(len(corr)):
        if k == 0:
            # gradient for k=0 term: sum_j h_j^2
            # d/dh_i (sum_j h_j^2) = 2 h_i
            # And constraint anyway is sum h_i = 1 
            grad += weights[k] * 2 * h
            continue
            
        # For k != 0, we compute contributions 
        # h_{i+k} exists in term ∑ h_i h_{i+k}, so d/dh_i (h_i h_{i+k}) = h_{i+k} 
        # So to compute d corr_k / dh_i = h_{i+k} + h_{i-k} 
        
        # Handle left and right shifts: k and -k contributions 
        # left shift: h_{i-k} contribution if i >= k , and contribute to grad[i] as h_{i-k}
        # right shift: h_{i+k} contribute to grad[i] as.h_{i+k} 
        # So we need to compute two-shifted versions:
        shift_right = np.roll(h, -k)  # shifting h backwards (gradient to h[i+k] contributes to grad[i])
        shift_left = np.roll(h, k)   # shifting h forward (gradient to h[i-k] contributes to grad[i])
        
        # Only include valid contributions (do not roll beyond boundary)
        # Or compute directly via slicing:
        if k > 0:
            # For right shift: for i=0 to N-1-k, grad[i] += w_k * h[i+k]
            # For left shift: for i=k to N-1, grad[i] += w_k * h[i-k]
            grad[:n-k] += weights[k] * h[k:] 
            grad[k:] += weights[k] * h[:n-k]
        else:
            # Rerun inspection, key fact: I indexed differently:
            # k is index 0 to N-1 that means shift this many positions LEFT or RIGHT in the full length 2N-1 array  
            # but in real values, index above means signal shift  
            
            # WaitForImageReferece
            pass
    
    # Multiply by -alpha (since we're computing gradient of -logsumexp)
    return -alpha * grad

def local_refine_smooth(h_start, alpha_schedule=[1.0, 2.0, 5.0, 10.0]):
    """Refine solution using smooth gradient descent"""
    n = len(h_start)
    dx = 2.0 / n
    target_sum = TARGET_SUM
    
    h_current = h_start.copy()
    max_iter = 300
    step_size = 0.001
    
    for alpha in alpha_schedule:
        # Gibbs nächster Durchlauf fügt grimmac für niedrige alpha hinzu
        for iter in range(max_iter):
            # Compute gradient
            try:
                grad = compute_gradient_smooth(h_current, alpha)
                # Take gradient step
                h_new = h_current - step_size * grad
                # Project back
                h_new = project_to_sum_box(h_new, target_sum)
                h_current = h_new
            except Exception as e:
                pass
                
            # Occasionally evaluate and terminate if we're good
            if iter % 50 == 0:
                eval_val = evaluate_c5(h_current)
                if eval_val <= 0.3807:
                    return h_current, eval_val
    
    return h_current, evaluate_c5(h_current)

def almost_binary_h(n_points, target_sum):
    """Create almost binary h to guide optimization"""
    h = np.zeros(n_points)
    # Place 50% of it as 1s
    ones_count = int(target_sum)
    ones_idx = np.random.choice(n_points, ones_count, replace=False)
    h[ones_idx] = 1.0
    
    # Project to ensure constraint
    h = project_to_sum_box(h, target_sum)
    return h
