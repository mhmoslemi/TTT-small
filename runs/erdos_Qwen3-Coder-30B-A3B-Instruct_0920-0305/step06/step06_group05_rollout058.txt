import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SCS

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, M, time_left):
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        cp.trace(X) == M,
        h >= 0,
        h <= 1,
        X >> 0,
        cp.diag(X) == h
    ]

    for k in range(n):
        Pk = np.zeros((n, n))
        for i in range(n - k):
            Pk[i, i+k] = 1
        constraints.append(t >= M - cp.trace(Pk @ X))

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=SCS, time_limit=time_left, verbose=False)
    except:
        pass

    if prob.status in ['optimal', 'optimal_inaccurate']:
        return h.value, X.value
    return None, None

def threshold_round(h_rel, M):
    # Select indices with highest values
    idxs = np.argsort(h_rel)[::-1][:M]
    b = np.zeros_like(h_rel)
    b[idxs] = 1
    return b

def hyperplane_round(X_rel, h_rel, M, num_trials=20):
    eigenvals, eigenvecs = np.linalg.eigh(X_rel)
    V_plus = eigenvecs[:, eigenvals > 1e-8]
    best = None
    best_val = float('inf')
    
    for _ in range(num_trials):
        g = np.random.normal(0, 1, V_plus.shape[1])
        s = V_plus @ g
        b = (s >= 0).astype(float)
        if np.sum(b) != M:
            diff = int(np.sum(b) - M)
            if diff > 0:
                flip_idx = np.random.choice(np.where(b == 1)[0], diff, replace=False)
                b[flip_idx] = 0
            else:
                flip_idx = np.random.choice(np.where(b == 0)[0], -diff, replace=False)
                b[flip_idx] = 1
        val = evaluate_c5(b)
        if val < best_val:
            best_val = val
            best = b.copy()
    return best

def efficient_swap(h, n_points):
    """Improve by local swaps"""
    h = h.copy()
    improved = True
    nonzeros = np.where(h != 0)[0]
    zeros = np.where(h == 0)[0]
    
    def update_lag_values(old_lag, new_lag, i, j, corr_vals):
        for k in range(len(corr_vals)):
            if abs(k - old_lag) <= i and abs(k - old_lag) <= j:
                corr_vals[k] += h[i] + h[j] - 2 * h[i] * h[j]
        return corr_vals

    best = float('inf')
    for _ in range(5000):
        if not improved:
            break
        improved = False
        best_corr = compute_corr_fft(h)
        best_val = c5_from_corr(best_corr, n_points)
        if best_val < best:
            best = best_val
        for i in nonzeros:
            for j in zeros:
                new_h = h.copy()
                new_h[i] = 0
                new_h[j] = 1
                new_corr = compute_corr_fft(new_h)
                new_val = c5_from_corr(new_corr, n_points)
                if new_val < best_val:
                    h = new_h
                    best_val = new_val
                    improved = True
                    break
            if improved:
                break
    return h

def polish_with_slsqp(h, M, dx):
    bounds = [(0, 1) for _ in range(len(h))]
    
    def objective(h_vec):
        corr = compute_corr_fft(h_vec)
        return dx * np.max(corr)

    def constraint_eq(h_vec):
        return np.sum(h_vec) - M

    cons = {'type': 'eq', 'fun': constraint_eq}
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h = res.x
            h = project_to_sum_box(h, M)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [60, 80, 100]
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / 100  # base dx for final output
    n_points = 100

    # SDP solve stages
    remaining_budget = budget_s * 0.5
    for idx, n in enumerate(n_values):
        if time.time() - start_time > budget_s - 20:
            break
        M = n / 2
        time_limit = remaining_budget / len(n_values) if idx < len(n_values)-1 else remaining_budget
        h_rel, X_rel = solve_sdp(n, M, time_limit)
        if h_rel is not None:
            # Rounding
            h_thresh = threshold_round(h_rel, M)
            h_thresh = efficient_swap(h_thresh, n)
            c5_thresh = evaluate_c5(h_thresh)
            if c5_thresh < best_c5:
                best_c5 = c5_thresh
                best_h = h_thresh.copy()
                
            h_random = hyperplane_round(X_rel, h_rel, M)
            if h_random is not None:
                h_random = efficient_swap(h_random, n)
                c5_rnd = evaluate_c5(h_random)
                if c5_rnd < best_c5:
                    best_c5 = c5_rnd
                    best_h = h_random.copy()
            remaining_budget -= time_limit
            if time.time() - start_time > budget_s - 20:
                break

    if best_h is None:
        # Fallback to deterministic construction
        h_fallback = np.zeros(100)
        h_fallback[:50] = 1
        best_h = h_fallback.copy()
        best_c5 = evaluate_c5(best_h)

    # Final polishing phase with SLSQP
    if time.time() - start_time < budget_s - 0.1:
        best_h = polish_with_slsqp(best_h, n_points / 2, dx)
        best_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(best_c5), n_points)
