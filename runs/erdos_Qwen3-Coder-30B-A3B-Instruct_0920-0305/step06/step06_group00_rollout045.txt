import numpy as np
from scipy.signal import fftconvolve

def project_to_sum_box(h, target_sum):
    """Project h onto the box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(h + (a + b)/2, 0, 1)
    return result


def compute_softmax_weights(corr, alpha):
    """Compute softmax weights from correlation values"""
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / np.sum(exp_vals)
    return weights


def compute_gradient(h, weights):
    """Compute gradient from softmax weights using FFT-based convolution"""
    n = len(h)
    # Forward convolution: w * h
    forward_conv = fftconvolve(h, weights[:n], mode='full')[:n]
    # Backward convolution: w_rev * h
    backward_conv = fftconvolve(h, weights[n-1::-1], mode='full')[:n]
    
    # Gradient = 1 - forward - backward
    grad = 1 - forward_conv - backward_conv
    return grad


def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    n = len(h)
    corr = fftconvolve(h, 1 - h, mode='full')
    corr = corr[:2*n - 1]
    return np.max(corr) * 2.0 / n


def perform_gradient_step(h, target_sum, alpha, eta):
    """Perform a single gradient descent step"""
    corr = fftconvolve(h, 1 - h, mode='full')
    corr = corr[:2*len(h) - 1]
    
    weights = compute_softmax_weights(corr, alpha)
    grad = compute_gradient(h, weights)
    
    # Backtracking line search
    h_trial = h - eta * grad
    h_trial = project_to_sum_box(h_trial, target_sum)
    c5_trial = evaluate_c5(h_trial)
    
    # If not improved, shrink learning rate
    while c5_trial >= evaluate_c5(h):
        eta /= 2
        if eta < 1e-10:
            break
        h_trial = h - eta * grad
        h_trial = project_to_sum_box(h_trial, target_sum)
        c5_trial = evaluate_c5(h_trial)
        
    return h_trial, c5_trial, eta


def binary_refinement(h, target_sum, max_iter=2000, T0=0.02):
    """Binary refinement via simulated annealing"""
    n = len(h)
    T = T0
    T_min = 1e-6
    alpha = 0.9995
    
    # Initial binary version
    indices_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices_sorted[:target_sum]] = 1
    
    best_h = h_bin.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(max_iter):
        # Pick two random indices to swap
        i1 = np.random.choice(np.where(h_bin == 1)[0])
        i2 = np.random.choice(np.where(h_bin == 0)[0])
        
        # Try the swap
        h_trial = h_bin.copy()
        h_trial[i1] = 0
        h_trial[i2] = 1
        
        c5_trial = evaluate_c5(h_trial)
        
        # Accept with probability
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        else:
            delta = c5_trial - best_c5
            if np.random.random() < np.exp(-delta / T):
                best_c5 = c5_trial
                best_h = h_trial.copy()
                
        T *= alpha
        if T < T_min:
            break
            
    return best_h, best_c5
