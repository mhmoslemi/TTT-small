import numpy as np
import time
from scipy.optimize import minimize

# Global variables for optimization
CURRENT_TARGET_SUM = None
DX_GLOBAL = None

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    L = 2*n - 1
    h_rev = h[::-1]
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L))) * DX_GLOBAL
    return float(1.0 - corr.max())

            best_h = h.copy()

        if len(inds) > 10:
            inds = np.random.choice(inds, size=10, replace=False)
            
        # compute gradient
        grad = np.zeros_like(h)

        h = project_to_sum_box(h, target_sum)
        # decay step
        step *= decay
        
    return best_h, best_c5

def sum_constraint(h):
    return np.sum(h) - CURRENT_TARGET_SUM

def polish_solution(h_init):
    global CURRENT_TARGET_SUM, DX_GLOBAL
    n = len(h_init)
    DX_GLOBAL = 2.0 / n
    CURRENT_TARGET_SUM = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_constraint}
    try:
        res = minimize(evaluate_c5_fft, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter':500, 'ftol':1e-12, 'disp':False})
        if res.success:
            h_pol = project_to_sum_box(res.x, CURRENT_TARGET_SUM)
            return h_pol, evaluate_c5_fft(h_pol)
    except:
        pass
    return h_init, evaluate_c5_fft(h_init)
