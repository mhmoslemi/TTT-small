import numpy as np
from scipy.signal import correlate
import time

# Global variables for performance
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def evaluate_max_corr(h):
    """Evaluate max correlation using FFT for speed"""
    n = len(h)
    N = 2*n
    fa = np.fft.rfft(h, n=N)
    fb = np.fft.rfft(1 - h, n=N)
    conv = np.fft.irfft(fa * np.conj(fb), n=N)
    corr = conv[:2*n-1]
    return corr.max() * dx

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low+high)/2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def full_corr(a, b):
    """Compute full correlation using FFT with zero-padding"""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def gradient_max_soft(h, alpha):
    """Gradient of soft-max approximation to max correlation"""
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    w = np.exp(alpha * (corr - maxc))
    w = w / w.sum()
    # Compute weighted forward and backward convolutions
    w_rev = w[::-1]
    # Forward convolution: Σ w[l] * h[i+l]
    conv_fwd = np.fft.irfft(np.fft.rfft(h, 2*n) * np.fft.rfft(w, 2*n), 2*n)[:n]
    # Backward convolution: Σ w[l] * h[i-l] = Σ w_rev[-l] * h[i+l]
    conv_bwd = np.fft.irfft(np.fft.rfft(h, 2*n) * np.fft.rfft(w_rev, 2*n), 2*n)[:n]
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def optimize_continuous(start_h, time_budget, n, dx, target_sum):
    """Optimize continuous h using projected gradient descent with soft max objective"""
    best_h = start_h.copy()
    best_val = evaluate_max_corr(best_h)
    h = start_h.copy()
    alpha = 5.0  # soft max temperature
    eta = 0.1    # step size
    start_time = time.time()
    
    for step in range(5000):
        if time.time() - start_time > time_budget - 1:
            break
            
        grad = gradient_max_soft(h, alpha)
        step_size = eta
        
        # Try line search
        max_attempts = 10
        for _ in range(max_attempts):
            h_new = project_to_sum_box(h - step_size * grad, target_sum)
            val_new = evaluate_max_corr(h_new)
            if val_new < best_val - 1e-12:
                h = h_new.copy()
                if val_new < best_val:
                    best_val = val_new
                    best_h = h.copy()
                break
            step_size *= 0.5
            if step_size < 1e-6:
                break
        
        # Adaptive alpha increase
        if step % 100 == 0 and alpha < 20.0:
            alpha += 0.2
            
        # Occasionally perturb to escape local minima
        if step % 500 == 0:
            h = project_to_sum_box(h + np.random.normal(0, 0.005, n), target_sum)
    
    return best_h, best_val

def binary_local_search(h_bin, time_budget, n, dx):
    """Local search in binary space using simulated annealing"""
    start_time = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_max_corr(best_h)
    
    temp = 0.01
    cooling_rate = 0.95
    min_temp = 1e-6
    step = 0
    
    while time.time() - start_time < time_budget:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        # Compute new C5
        val = evaluate_max_corr(cand)
        if val < best_val - 1e-9:
            best_val = val
            best_h = cand.copy()
        elif np.random.rand() < np.exp(-(val - best_val) / temp):
            best_h = cand.copy()
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
        step += 1
    
    return best_h, best_val
