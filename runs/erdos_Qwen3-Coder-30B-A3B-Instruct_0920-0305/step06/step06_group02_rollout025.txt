import numpy as np
import time
import random

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed):
    """Generate initial binary vector with exactly TARGET_SUM ones."""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full overlap array O_s for all shifts s."""
    ov = np.correlate(x, x, 'full')
    return ov

def swap_update(ov, i, j, ones, zeros):
    """Return new overlap array after swapping x[i] and x[j]."""
    # Create a copy to avoid modifying original
    ov_new = ov.copy()
    # Remove contribution of i:
    # For all k in ones, update ov_new[k-i+n-1], ov_new[i-k+n-1]
    for k in ones:
        if k != i:
            idx1 = k - i + n_points - 1
            idx2 = i - k + n_points - 1
            ov_new[idx1] -= 1
            ov_new[idx2] -= 1
    # Add contribution of j:
    # For all k in ones\{i}, update ov_new[k-j+n-1], ov_new[j-k+n-1]
    for k in ones:
        if k != i:
            idx1 = k - j + n_points - 1
            idx2 = j - k + n_points - 1
            ov_new[idx1] += 1
            ov_new[idx2] += 1
    return ov_new

def get_sets(x):
    """Return (ones, zeros) indices."""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    return ones, zeros

def simulated_annealing(x0, ov0, ones0, zeros0, time_limit):
    """Run simulated annealing search for the binary solution."""
    start = time.time()
    
    # Start with initial solution
    x = x0.copy()
    ov = ov0.copy()
    ones = ones0.copy()
    zeros = zeros0.copy()
    
    # Track current best
    cur_min = np.min(ov)
    best_x = x.copy()
    best_min = cur_min
    
    # Parameters for SA
    T0 = 0.5
    lambda_decay = 0.0001
    max_iter = 1000000
    iter_count = 0
    
    while time.time() - start < time_limit:
        # Update temperature
        t = iter_count / max_iter
        T = T0 * np.exp(-lambda_decay * t)
        
        # Select a random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create new overlap array after swapping i and j
        ov_new = swap_update(ov, i, j, ones, zeros)
        new_min = np.min(ov_new)
        
        # Accept or reject the move
        delta = new_min - cur_min
        if delta > 0 or random.random() < np.exp(delta / T):
            # Accept the move
            x[i] = 0
            x[j] = 1
            # Update sets
            ones = np.setdiff1d(ones, [i])
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.append(zeros, i)
            ov = ov_new
            cur_min = new_min
            
            # Update best if improved
            if new_min > best_min:
                best_x = x.copy()
                best_min = new_min
        
        iter_count += 1
        if iter_count % 10000 == 0:
            if time.time() - start > time_limit * 0.95:
                break
    
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """
    Try to improve the worst shift by swapping positions.
    """
    min_shift_idx = np.argmin(ov)
    min_shift = min_shift_idx - (n_points - 1)
    
    # Identify deficient 1-positions (1 that need to become 0 to improve overlap)
    # and beneficial 0-positions (0 that should become 1 to improve overlap)
    A = []  # pairs (i, s) such that x[i] = 1, x[i+s] = 0, s = min_shift
    B = []  # pairs (j, s) such that x[j] = 0, x[j+s] = 1, s = min_shift
    
    for i in ones:
        if 0 <= i + min_shift < n_points and x[i + min_shift] == 0:
            A.append(i)
    
    for j in zeros:
        if 0 <= j + min_shift < n_points and x[j + min_shift] == 1:
            B.append(j)
    
    # Try all swaps between A and B to improve this shift
    for i in A:
        for j in B:
            # Check if swapping i <-> j improves shift min_shift overlap at least by 1
            if x[i] == 1 and x[j] == 0:
                # Precompute effect
                # Make a new overlap array to test what this swap does to min_shift
                ov_test = swap_update(ov, i, j, ones, zeros)
                # Check if the new min is better than current var
                if np.min(ov_test) > np.min(ov) or (np.min(ov_test) == np.min(ov) and min_shift == np.argmin(ov)):
                    # Applies the move
                    x[i] = 0
                    x[j] = 1
                    # Update sets
                    ones = np.delete(ones, np.where(ones == i)[0][0])
                    ones = np.append(ones, j)
                    zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                    zeros = np.append(zeros, i)
                    ov = ov_test
                    break
    
    return x, ov, ones, zeros
