import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points
n_points_upscaled = 400
dx_upscaled = 2.0 / n_points_upscaled

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    corr = fftconvolve(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    x = np.asarray(x)
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for the max approximation"""
    # Subtract max for numerical stability
    corr_max = np.max(corr)
    exp_corr = np.exp(alpha * (corr - corr_max))
    weights = exp_corr / np.sum(exp_corr)
    return weights

def compute_gradient(h, weights):
    """Compute gradient of the soft-max surrogate"""
    # Forward convolution
    forward_conv = fftconvolve(h, weights, mode='full')
    # Backward convolution
    backward_conv = fftconvolve(h, weights[::-1], mode='full')
    # Gradient
    grad = 1.0 - forward_conv[:len(h)] - backward_conv[:len(h)]
    return grad

def projected_gradient_descent(h0, max_iter=10000, alpha0=5.0, alpha_max=20.0, 
                              eta0=0.5, eta_max=1.0, time_limit=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha0
    eta = eta0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time_limit and time.time() > time_limit:
            break
            
        # Evaluate
        corr = fftconvolve(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if abs(prev_c5 - c5) < 1e-10:
            break
        prev_c5 = c5
        
        # Softmax weights
        weights = softmax_weights(corr, alpha)
        
        # Gradient
        gradient = compute_gradient(h, weights)
        
        # Line search
        eta_test = eta
        while True:
            h_trial = project_to_sum_box(h - eta_test * gradient, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5:
                h = h_trial
                break
            eta_test /= 2
            if eta_test < 1e-8:
                break
                
        # Update parameters
        eta = min(eta_max, eta * 1.02)
        alpha = min(alpha_max, alpha * 1.01)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Construct diverse initial seeds
    initial_seeds = []
    
    # Interpolated from initial_h_values
    h0_init = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0_init))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_orig, h0_init)
    initial_seeds.append(project_to_sum_box(h_interp, n_points / 2))
    
    # Random continuous
    h_rand = np.random.rand(n_points)
    h_rand = project_to_sum_box(h_rand, n_points / 2)
    initial_seeds.append(h_rand)
    
    # Random binary
    h_bin = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, int(n_points / 2), replace=False)
    h_bin[ones_indices] = 1
    initial_seeds.append(h_bin)
    
    # Low-autocorrelation binary (Legendre)
    legendre_seq = []
    p = 31   # Small prime, better to use a prime ≈ n_points + 1
    while p < n_points + 10:
        p += 1
        is_prime = all(p % i != 0 for i in range(2, int(p**0.5)+1))
        if is_prime:
            break
    for i in range(p):
        val = 1 if (i*i % p) < p/2 else -1
        legendre_seq.append(val)
    legendre_seq = legendre_seq[:n_points]
    h_legendre = np.array(legendre_seq) > 0
    count_ones = np.sum(h_legendre)
    if count_ones != n_points / 2:
        # adjust by swapping
        if count_ones < n_points / 2:
            to_flip = np.where(~h_legendre)[0][:int(n_points/2 - count_ones)]
        else:
            to_flip = np.where(h_legendre)[0][:int(count_ones - n_points/2)]
        h_legendre[to_flip] = ~h_legendre[to_flip]
    initial_seeds.append(h_legendre)
    
    best_h = None
    best_c5 = float('inf')
    
    # Multi-start optimization
    total_seeds = len(initial_seeds)
    time_per_seed = 0.12 * budget_s / total_seeds
    
    for i, h_seed in enumerate(initial_seeds):
        seed_start_time = time.time()
        time_limit = seed_start_time + time_per_seed
        
        h_opt, c5_opt = projected_gradient_descent(h_seed, time_limit=time_limit)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() > start_time + 0.95 * budget_s:
            break

    # Upscale and refine if we find a promising solution
    if best_c5 <= 0.381:
        h_upscale = np.interp(np.linspace(0, 2, n_points_upscaled), 
                             np.linspace(0, 2, n_points), best_h)
        h_upscale = project_to_sum_box(h_upscale, n_points_upscaled / 2)
        h_opt_upscale, c5_opt_upscale = projected_gradient_descent(
            h_upscale, max_iter=20, alpha0=20.0, eta0=0.1, time_limit=None)
        if c5_opt_upscale < best_c5:
            best_c5 = c5_opt_upscale
            best_h = h_opt_upscale.copy()
    
    # Binary simulated annealing refinement
    if best_h is not None:
        binary_h = np.zeros(n_points_upscaled)
        indices = np.argsort(best_h)[::-1]
        binary_h[indices[:n_points_upscaled // 2]] = 1.0
        binary_h = project_to_sum_box(binary_h, n_points_upscaled / 2)
        
        target = n_points_upscaled // 2
        original_c5 = evaluate_c5(binary_h)
        c5 = original_c5
        best_binary_h = binary_h.copy()
        
        time_left = max(5.0, budget_s - (time.time() - start_time))
        sweeps_per_second = 10
        total_sweeps = int(sweeps_per_second * time_left)
        T = 0.02 * original_c5
        
        for sweep in range(total_sweeps):
            if T < 1e-6:
                break
            # Swap a 1 with a 0
            ones = np.where(best_binary_h == 1)[0]
            zeros = np.where(best_binary_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            
            new_binary_h = best_binary_h.copy()
            new_binary_h[i] = 0
            new_binary_h[j] = 1
            
            new_c5 = evaluate_c5(new_binary_h)
            
            if new_c5 < c5:
                c5 = new_c5
                best_binary_h = new_binary_h.copy()
            else:
                prob = np.exp(-(new_c5 - c5) / T)
                if np.random.rand() < prob:
                    c5 = new_c5
                    best_binary_h = new_binary_h.copy()
            
            T *= 0.9995
    
    if best_h is None:
        # Return something reasonable if we fail to improve from the
        # initial values (should not normally happen unless solver dies early)
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(initial_h_values)), 
                            initial_h_values)
        h_interp = project_to_sum_box(h_interp, n_points / 2)
        best_h = h_interp.copy()
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points_upscaled)
