from scipy.fft import fft, ifft

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    # Use full correlation (zero-padded)
    N = 2 * n_points
    h_fft = fft(h, n=N)
    ih_fft = fft(1 - h, n=N)
    corr_fft = h_fft * np.conj(ih_fft)
    corr = ifft(corr_fft).real
    corr_full = corr[:2*n_points-1]
    return (2.0 / n_points) * np.max(corr_full)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max_weights(corr, alpha):
    """Compute softmax weights for correlations"""
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(alpha * corr_shifted)
    weights = exp_corr / np.sum(exp_corr)
    return weights

def compute_gradient(h, weights):
    """Compute gradient of soft-max approximation using FFTs"""
    # Pad weights to match full correlation size
    N = 2 * n_points
    w_padded = np.zeros(N)
    w_padded[:len(weights)] = weights
    
    # Forward convolution
    h_fft = fft(h, n=N)
    w_fft = fft(w_padded, n=N)
    forward_conv = ifft(h_fft * w_fft).real
    
    # Backward convolution (reverse weights)
    w_rev = w_padded[::-1]
    w_rev_fft = fft(w_rev, n=N)
    backward_conv = ifft(h_fft * w_rev_fft).real
    
    # Gradient computation
    grad = 1.0 - forward_conv - backward_conv
    return grad

def projected_gradient_descent(h0, max_iter=10000, alpha_start=5.0, alpha_max=20.0, eta_start=0.5, eta_max=1.0):
    """Perform projected gradient descent with soft-max surrogate"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha_start
    eta = eta_start
    
    for i in range(max_iter):
        # Compute full correlation via FFT
        N = 2 * n_points
        h_fft = fft(h, n=N)
        ih_fft = fft(1 - h, n=N)
        corr_fft = h_fft * np.conj(ih_fft)
        corr_full = ifft(corr_fft).real[:2*n_points-1]
        
        # Softmax weights
        weights = soft_max_weights(corr_full, alpha)
        
        # Gradient
        grad = compute_gradient(h, weights)
        
        # Line search with backtracking
        eta_test = eta
        for _ in range(20):
            h_trial = project_to_sum_box(h - eta_test * grad, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                break
            eta_test /= 2
            if eta_test < 1e-8:
                break
                
        h = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
            
        # Update parameters
        eta = min(eta_max, eta * 1.02)
        alpha = min(alpha_max, alpha * 1.01)
        
        # Early stopping criteria
        if i > 500 and i % 500 == 0:
            if abs(best_c5 - evaluate_c5(best_h)) < 1e-6:
                break
                
    return best_h, best_c5

def generate_legendre_binary(length):
    """Generate a Legendre (quadratic residue) sequence and adjust to match sum"""
    # Find smallest prime >= length
    def is_prime(n):
        if n < 2:
            return False
        for i in range(2, int(np.sqrt(n)) + 1):
            if n % i == 0:
                return False
        return True
    
    p = length
    while not is_prime(p):
        p += 1
    
    # Generate Legendre sequence (length p)
    legendre_seq = np.array([1 if (i*i % p) == 0 else -1 for i in range(p)])
    if p % 2 == 0:
        legendre_seq = legendre_seq[1:][::-1]  # Remove first, reverse
    
    # Truncate to desired length
    h = np.zeros(length)
    h[:len(legendre_seq)] = legendre_seq[:length]
    
    # Convert to binary with exactly TARGET_SUM ones
    # Count current ones (including -1)
    count_ones = np.sum(h > 0)
    
    if count_ones > TARGET_SUM:
        # Get indices of 1s, set some to -1
        ones_indices = np.where(h > 0)[0]
        indices_to_flip = ones_indices[:count_ones - TARGET_SUM]
        for idx in indices_to_flip:
            h[idx] = -1
    elif count_ones < TARGET_SUM:
        # Get indices of -1s, set some to 1
        minus_ones_indices = np.where(h < 0)[0]
        indices_to_flip = minus_ones_indices[:TARGET_SUM - count_ones]
        for idx in indices_to_flip:
            h[idx] = 1
    
    # Convert to actual binary
    h_binary = (h > 0).astype(float)
    return h_binary
