import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_for_shifts(h, shift_indices):
    """Compute average gradient for worst shifts"""
    grad = np.zeros_like(h)
    for idx in shift_indices:
        shift = idx - (n_points - 1)
        if shift >= 0:
            grad[:n_points-shift] -= h[shift:]
            grad[shift:] -= h[:n_points-shift]
        else:
            s = -shift
            grad[s:] -= h[:n_points-s]
            grad[:n_points-s] -= h[s:]
    return grad / len(shift_indices)

def subgradient_descent(h0, target_sum, max_iter=5000, step_size=0.05, decay=0.9995):
    """Perform projected subgradient descent"""
    h = h0.copy()
    step = step_size
    for i in range(max_iter):
        # Compute correlation (overlap) for all shifts
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        max_val = overlaps.max()
        
        # Find all indices with maximum overlap
        max_indices = np.where(np.abs(overlaps - max_val) < 1e-12)[0]
        
        # Compute gradient
        grad = compute_gradient_for_shifts(h, max_indices)
        
        # Update step
        h = h - step * grad
        
        # Project to constraints
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, target_sum)
        
        # Update step size
        step *= decay
        
        if i % 1000 == 0:
            # Periodic check of current C5 value
            c5_current = evaluate_c5(h)
            if i > 0 and c5_current > max_val:
                # If projected solution improves outcome, resume optimization (not sure if helpful)
                pass
    
    return h
