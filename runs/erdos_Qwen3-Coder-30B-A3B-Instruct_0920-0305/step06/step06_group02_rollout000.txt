import numpy as np
import time
import random

def compute_convolution(h):
    """Compute full convolution of h with itself"""
    return np.correlate(h, h, mode='full')

def evaluate_c5(h):
    """Evaluate C5 using FFT approach for speed"""
    # Use FFT-based correlation which is more efficient for large arrays
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def compute_c5_from_conv(conv, n):
    """Compute C5 from precomputed convolution array"""
    w = n // 2  # target sum
    min_overlap = np.min(conv)
    return (w - min_overlap) * (2.0 / n)

def apply_swap_incremental(conv, ones, i, j, n):
    """Apply swap of positions i (1->0) and j (0->1) to conv in O(n) time"""
    # Make a copy of conv to compute new values
    new_conv = conv.copy()
    offset = n - 1
    
    # Remove i's contributions
    # For each k where x[k] = 1, subtract those correlations
    for k in ones:
        shift = k - i
        if k == i:
            # Self pair contribution
            new_conv[offset] -= 1
        else:
            # Both ordered pairs (i,k) and (k,i)
            idx1 = shift + offset
            idx2 = -shift + offset
            new_conv[idx1] -= 1
            new_conv[idx2] -= 1
    
    # Update ones list (remove i, add j)
    new_ones = [idx for idx in ones if idx != i] + [j]
    
    # Add j's contributions
    for k in new_ones:
        shift = k - j
        if k == j:
            # Self pair - only one contribution to indexing
            new_conv[offset] += 1
        else:
            # Add both ordered pairs (j,k) and (k,j)
            idx1 = shift + offset
            idx2 = -shift + offset
            new_conv[idx1] += 1
            new_conv[idx2] += 1
    
    return new_conv, new_ones
