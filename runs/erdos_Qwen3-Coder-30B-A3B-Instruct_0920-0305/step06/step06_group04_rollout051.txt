import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_logsumexp_smoothed_max(h, alpha=1.0):
    """Approximate max_k (sum_i h_i (1 - h_{i+k})) with log-sum-exp smoothing"""
    n = len(h)
    corr = compute_corr_fft(h)
    # Use the fact that we want 1 - (corr_i / dx) = 1 - (corr_i * n / 2)
    # But simpler approach: we want to minimize the maximum of terms
    # Term for lag k is -sum_i h_i * h_{i+k} = -f_k. We aim to minimize maximum of -f_k.
    # That is same as maximizing minimum of f_k = sum_i h_i h_{i+k}
    logterms = -alpha * corr * (n / 2)  # Factor adjustment
    logsumexp = np.log(np.sum(np.exp(logterms))) # This is ln(sum of exp(-alpha * f_k))
    return logsumexp

def compute_smoothed_gradient(h, alpha=1.0):
    """Compute gradient of smooth max approximation"""
    n = len(h)
    # Compute correlation values for all lags
    corr = compute_corr_fft(h)
    
    # Compute weights: w_k = exp(-alpha * f_k) / sum(exp(-alpha * f_k))
    # But actual terms are as defined in the problem:
    # f_k = sum_i h_i * h_{i+k} 
    # We want gradient of max_k -f_k (which is same as minimizing max_k f_k)
    # Directly compute weights for smooth approximation
    f_vals = corr * (n / 2)  # scale back to original value
    logterms = -alpha * f_vals
    exp_logterms = np.exp(logterms)
    w_sum = np.sum(exp_logterms)
    weights = exp_logterms / w_sum if w_sum > 0 else np.ones_like(exp_logterms) / len(exp_logterms)
    
    # Compute gradient of smoothed objective 
    # Gradient is vector with terms: -sum_k w_k * (h_{i+shift_k} + h_{i-shift_k})
    # We have elements in the full convolution, but we want the specific i and such that the shifts
    # are concerning current index. So gradient[i] = - sum_k w_k (h_{i+k} + h_{i-k}) for valid indices 
    
    gradient = np.zeros(n)
    for k in range(n):
        if k == 0:
            factor = weights[k] * 2 * h 
        else:
            # Compute both left and right shifts
            shift_right = np.roll(h, -k)  
            shift_left = np.roll(h, k)   
            factor = weights[k] * (shift_right + shift_left)
        gradient += -factor
    return gradient

    h = np.zeros(n_points)

    h[ones_idx] = 1.0
    
    # Project to ensure constraint
    h = project_to_sum_box(h, target_sum)
    return h
