import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Reverse h to get 1-h
    h_rev = h[::-1]
    # Compute 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

    print(f"Starting MILP phase with {n_points} points...")

        h_vars = cp.Variable(n_points, boolean=True)
        y_vars = {}

        # Build y variables for all (i, l)
        for l in range(n_points):
            y_vars[l] = cp.Variable((n_points - l), boolean=True)
        
        # Constraints
        constraints = [cp.sum(h_vars) == target_sum]
        
        # Product constraints
        for l in range(n_points):
            for i in range(n_points - l):
                y = y_vars[l][i]
                h_i = h_vars[i]
                h_ip = h_vars[i + l]
                constraints.append(y <= h_i)
                constraints.append(y <= h_ip)
                constraints.append(y >= h_i + h_ip - 1)
        
        # Lag-overlap constraints
        for l in range(n_points):
            constraints.append(cp.sum(y_vars[l]) >= lambda_var)
        
        # Bound constraint on lambda
        constraints.append(lambda_var <= target_sum)
        
        # Objective
        objective = cp.Maximize(lambda_var)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
