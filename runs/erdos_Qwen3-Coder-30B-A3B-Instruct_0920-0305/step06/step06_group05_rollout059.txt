import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from sklearn.decomposition import TruncatedSVD

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, M, time_remaining):
    """Solve SDP relaxation for the given n and M"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.trace(X) == cp.sum(h),
        h >= 0,
        h <= 1,
        X >> 0,
        cp.diag(X) == h,
        t >= M - cp.trace(P @ X) for P in lag_matrices(n)
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        # Solve with time limit
        time_limit = min(time_remaining / 2.0, 30.0)
        prob.solve(solver=cp.SCS, time_limit=time_limit)
    except:
        # Fallback to solve with no time limit in case of error
        try:
            prob.solve(solver=cp.SCS)
        except:
            pass

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, X.value, prob.value
    else:
        # fallback to a feasible point
        if h.value is not None:
            return h.value, X.value, float('inf')
        else:
            # Fallback to trivial solution
            h_v = np.ones(n) * M / n
            return h_v, None, float('inf')

def lag_matrices(n):
    """Generate lag matrices P_k for all k = 0, ..., n-1"""
    P_list = []
    for k in range(n):
        P = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                P[i, i + k] = 1.0
        P_list.append(P)
    return P_list

def rounding_trial(h_rel, M, n):
    """Perform one random hyperplane rounding trial"""
    # Generate a random vector
    g = np.random.normal(0, 1, n)
    s = h_rel @ g  # inner product in R^n
    b = (s >= 0).astype(int)

    # Repair to make sum exactly M
    sum_b = np.sum(b)
    if sum_b > M:
        # We have too many 1s
        ones_indices = np.where(b == 1)[0]
        flip_indices = np.random.choice(ones_indices, size=sum_b - M, replace=False)
        b[flip_indices] = 0
    elif sum_b < M:
        # We have too few 1s
        zeros_indices = np.where(b == 0)[0]
        flip_indices = np.random.choice(zeros_indices, size=M - sum_b, replace=False)
        b[flip_indices] = 1

    return b.astype(float)

def threshold_rounding(h_rel, M):
    """Perform threshold rounding"""
    idxs = np.argsort(h_rel)[::-1]  # descending order of indices
    b = np.zeros_like(h_rel)
    b[idxs[:int(M)]] = 1.0
    return b

def swap_hill_climb(h, M):
    """Deterministic hill-climbing via swaps"""
    # Convert to full binary vector for manipulation
    h_b = (h > 0.5).astype(int)
    if np.sum(h_b) != M:
        # Repair
        ones_idx = np.where(h_b == 1)[0]
        zeros_idx = np.where(h_b == 0)[0]
        num_to_change = int(np.sum(h_b) - M)
        if num_to_change > 0:
            # Change some ones to zeros
            change_idx = ones_idx[:num_to_change]
            h_b[change_idx] = 0
        else:
            # Change some zeros to ones
            change_idx = zeros_idx[:abs(num_to_change)]
            h_b[change_idx] = 1

    h = h_b.astype(float)
    improved = True
    attempts = 0
    max_attempts = 10000  # Safety limit
    corr = compute_corr_fft(h)

    while improved and attempts < max_attempts:
        improved = False
        best_lag = np.argmax(corr)
        best_c5 = c5_from_corr(corr, len(h))

        for i in range(len(h)):
            if h[i] == 0:
                continue
            for j in range(len(h)):
                if h[j] == 1:
                    continue
                # Check if swapping improves
                new_h = h.copy()
                new_h[i] = 0
                new_h[j] = 1

                # Partial recomputation for correlation
                new_corr = corr.copy()
                for k in range(len(h)):
                    if k < len(new_h):
                        # Kernels involving i or j change
                        if i == k or i == k - best_lag or i == k + best_lag:
                            new_corr[best_lag] += h[i] - new_h[i]
                        if j == k or j == k - best_lag or j == k + best_lag:
                            new_corr[best_lag] += h[j] - new_h[j]
                    else:
                        # General case
                        delta = 0
                        if k < len(new_h):
                            new_k = new_h[k] * (1 - new_h[k + best_lag]) if k + best_lag < len(new_h) else 0
                            old_k = h[k] * (1 - h[k + best_lag]) if k + best_lag < len(new_h) else 0
                            delta += new_k - old_k

                new_c5 = c5_from_corr(new_corr, len(new_h))
                if new_c5 < best_c5:
                    h = new_h
                    corr = new_corr
                    best_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        attempts += 1
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 60
    h_projected = None

    # Phase A: SDP solving
    n_vals = [60, 80, 100]
    for n in n_vals:
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining < 5:
            break
        M = int(n / 2)
        dx = 2.0 / n
        try:
            h_rel, X, c5_approx = solve_sdp(n, M, time_remaining)
            if np.any(np.isnan(h_rel)) or np.any(np.isinf(h_rel)):
                continue
            h_rel = np.clip(h_rel, 0, 1)
        except:
            continue

        # Rounding
        candidates = []
        for trial in range(15):
            if trial % 10 == 0:
                # Deterministic threshold rounding
                b = threshold_rounding(h_rel, M)
            else:
                # Random rounding
                b = rounding_trial(h_rel, M, n)
            candidates.append(b)

        # Evaluate candidates
        for h_cand in candidates:
            # Check that the sum is exactly matching M due to rounding repairs
            h_cand = project_to_sum_box(h_cand, M)
            h_cand = swap_hill_climb(h_cand, M)
            c5_val = evaluate_c5(h_cand)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_cand.copy()
                best_n = n

        if best_c5 < 0.379:
            break

    # Phase B: Final polishing
    if best_h is not None:
        # First perform a rounded version from threshold rounding before SLSQP
        try:
            h_p = project_to_sum_box(best_h, best_n / 2)
            h_p = swap_hill_climb(h_p, best_n / 2)
            c5_p = evaluate_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                best_n = best_n
        except:
            pass

        # SLSQP fine-tuning
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining > 0.5:
            bounds = [(0, 1) for _ in range(best_n)]
            TARGET_SUM = best_n / 2
            try:
                def objective(x):
                    return evaluate_c5(x)

                def constraint_eq(x):
                    return np.sum(x) - TARGET_SUM

                cons = {'type': 'eq', 'fun': constraint_eq}

                res = minimize(
                    objective,
                    best_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-9, 'gtol': 1e-9},
                    callback=None
                )

                if res.success:
                    h_final = project_to_sum_box(res.x, TARGET_SUM)
                    c5_final = evaluate_c5(h_final)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final
            except:
                pass

    # Final projection
    if best_h is not None:
        final_h = project_to_sum_box(best_h, best_n / 2)
        final_c5 = evaluate_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    # Fallback if nothing found yet
    if best_h is None:
        best_h = np.zeros(120)
        best_h[:60] = 1.0
        best_h = project_to_sum_box(best_h, 60.0)
        best_c5 = evaluate_c5(best_h)
        best_n = 120

    return (best_h.tolist(), float(best_c5), int(best_n))
