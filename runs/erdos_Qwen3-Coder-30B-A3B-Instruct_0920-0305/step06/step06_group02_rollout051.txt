import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_convolution_update(conv, ones, i, j, n):
    """Compute new convolution after swapping i (1->0) and j (0->1)"""
    # Make a copy to avoid side effects
    conv_new = conv.copy()
    
    # Remove contributions of i
    for k in ones:
        shift = k - i
        idx = shift + n - 1
        if k == i:
            conv_new[idx] -= 1
        else:
            conv_new[idx] -= 1
            conv_new[-shift + n - 1] -= 1
    
    # Remove i from ones list
    ones_new = [idx for idx in ones if idx != i]
    
    # Add contributions of j
    for k in ones_new + [j]:
        shift = k - j
        idx = shift + n - 1
        if k == j:
            conv_new[idx] += 1
        else:
            conv_new[idx] += 1
            conv_new[-shift + n - 1] += 1
    
    return conv_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_h = None
    best_c5 = float('inf')
    
    # Use binary initial solution
    # Start with a reasonable candidate: uniform block pattern
    h0 = np.zeros(n_points, dtype=int)
    # Fill roughly half evenly distributed
    indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    h0[indices] = 1
    
    # Precompute initial correlation using numpy
    h_current = h0.astype(float)
    conv_current = np.correlate(h_current, 1 - h_current, mode='full')
    cur_min_overlap = np.min(conv_current)
    cur_c5 = (TARGET_SUM - cur_min_overlap) * dx
    
    best_h = h0.copy()
    best_c5 = cur_c5
    
    # Simulated Annealing parameters
    max_iter = 50000
    temp_start = 0.1
    temp_end = 0.001
    cooling_rate = 0.9999
    
    # Maintain current ones list for fast random swap selection
    ones_indices = [i for i in range(n_points) if h0[i] == 1]
    zero_indices = [i for i in range(n_points) if h0[i] == 0]
    
    # Simulated Annealing main loop
    for iteration in range(max_iter):
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        
        # Reduce temperature
        temp = temp_start * (cooling_rate ** iteration)
        if temp < temp_end:
            temp = temp_end
            
        # Select random indices to swap
        i = random.choice(ones_indices)
        j = random.choice(zero_indices)
        
        # Compute new convolution using updates
        ones_copy = ones_indices[:]
        conv_new = compute_convolution_update(conv_current, ones_copy, i, j, n_points)
        
        new_min_overlap = np.min(conv_new)
        new_c5 = (TARGET_SUM - new_min_overlap) * dx
        
        # Accept or reject the move
        delta = new_c5 - cur_c5  # This is positive if worse
        if delta < 0 or random.random() < np.exp(-delta / temp):
            # Accept the move
            h_current[i] = 0
            h_current[j] = 1
            conv_current = conv_new
            cur_c5 = new_c5
            cur_min_overlap = new_min_overlap
            
            # Update lists
            ones_indices.remove(i)
            ones_indices.append(j)
            zero_indices.remove(j)
            zero_indices.append(i)
            
            # Update best solution
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_current.copy()
        else:
            pass  # Reject move
            
    # Convert final solution to the required format
    h_final = best_h.tolist()
    
    return (h_final, float(best_c5), n_points)
