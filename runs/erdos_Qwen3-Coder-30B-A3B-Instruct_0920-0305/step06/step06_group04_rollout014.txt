import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n = 120, so target sum = 60
N_FINE = 800       # fine grid for FFT-based correlation
K = 12             # number of Fourier harmonics
N_TARGET = 120     # final grid size

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    scale = target_sum / current_sum
    x = x * scale
    x = np.clip(x, 0, 1)
    return x

def create_h_from_fourier(coeffs, n_points=N_TARGET, n_fine=N_FINE):
    """Generate a step-function h from Fourier coefficients."""
    # Construct frequencies: k = 1, 2, ..., K (cosine and sine)
    k_vals = np.arange(1, K+1)
    
    # Periodic domain
    x = np.linspace(0, 2, n_fine, endpoint=False)
    
    # Build the function from Fourier series
    h = 0.5 + np.sum(
        coeffs[0:K] * np.cos(np.pi * k_vals.reshape(-1,1) * x) +
        coeffs[K:2*K] * np.sin(np.pi * k_vals.reshape(-1,1) * x),
        axis=0
    )
    
    # Clip and project
    h = np.clip(h, 0, 1)
    h = project_to_sum_box(h, n_points / 2.0)
    
    # Downsample to target size
    h_downsampled = h[::n_fine // n_points]
    h_downsampled = project_to_sum_box(h_downsampled, n_points / 2.0)
    
    return h_downsampled

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT."""
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(ifft(fft(h_rev, 2*len(h) - 1) * fft(b, 2*len(h) - 1)))
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def de_objective(coeffs):
    h = create_h_from_fourier(coeffs, n_points=N_TARGET)
    return evaluate_c5_fft(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = np.inf
    best_h = None
    n_points = N_TARGET
    
    # Explore both resolutions for better results
    resolutions = [120, 200]
    for n_target in resolutions:
        if (time.time() - start_time) > 0.8 * budget_s:
            break
        n_points = n_target
        TARGET_SUM = n_points / 2.0
        
        bounds = [(-0.5, 0.5)] * (2 * K)
        
        # Phase 1: Coarse optimization
        res1 = differential_evolution(
            de_objective,
            bounds,
            maxiter=250,
            popsize=50,
            seed=seed,
            updating='deferred',
            disp=False
        )
        if res1.success:
            h = create_h_from_fourier(res1.x)
            c5 = evaluate_c5_fft(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Phase 2: Refinement with fewer iterations but better exploration
        best_of_best = []
        for _ in range(5):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            res2 = differential_evolution(
                de_objective,
                bounds,
                maxiter=100,
                popsize=30,
                seed=None,
                updating='deferred',
                disp=False
            )
            if res2.success:
                h = create_h_from_fourier(res2.x)
                c5 = evaluate_c5_fft(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                continue
                
        # Final local refinement
        if best_h is not None:
            def refine_objective(h_flat):
                h = h_flat.reshape(-1)
                h = project_to_sum_box(h, TARGET_SUM)
                return evaluate_c5_fft(h)
            
            # Try multiple local searches starting from best h
            h_vec = best_h.reshape(-1)
            initial_h = h_vec.copy()
            for _ in range(50):
                if (time.time() - start_time) > 0.98 * budget_s:
                    break
                perturbation = np.random.normal(0, 0.02, len(h_vec))
                h_test = h_vec + perturbation
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                c5_test = evaluate_c5_fft(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
    
    # Binary rounding and swap
    if (time.time() - start_time) > 0.99 * budget_s:
        return (best_h.tolist(), float(best_c5), n_points)
    
    if best_h is not None:
        h_binary = np.zeros_like(best_h)
        indices = np.argsort(best_h)[::-1]
        h_binary[indices[:len(best_h)//2]] = 1.0
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        
        # Fast swap improvement
        for _ in range(1000):
            if (time.time() - start_time) > budget_s:
                break
            worst_idx = np.argmin([sum(h_binary[i:] * h_binary[:len(h_binary)-i]) for i in range(1, len(h_binary)//2)])
            i1 = np.where(h_binary[:worst_idx] == 0)[0]
            i2 = np.where(h_binary[worst_idx+1:] == 1)[0] + worst_idx + 1
            if len(i1) == 0 or len(i2) == 0:
                continue
            i1 = i1[0]
            i2 = i2[0]
            if i1 >= len(h_binary) or i2 >= len(h_binary):
                continue
            h_test = h_binary.copy()
            h_test[i1] = 1.0
            h_test[i2] = 0.0
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5_fft(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
