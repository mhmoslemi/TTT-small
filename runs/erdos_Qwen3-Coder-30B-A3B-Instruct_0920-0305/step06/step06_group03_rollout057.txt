```python
import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution, minimize
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def generate_initial_h():
    """Generate a better initial guess for h"""
    h = np.zeros(n_points)
    
    # Create a piecewise constant pattern that is more likely to be optimal
    # Try to use a construction inspired by known good solutions for this type of problem
    # Use a pattern of alternating high and low values
    segment_size = n_points // 4
    for i in range(4):
        start_idx = i * segment_size
        end_idx = start_idx + segment_size
        if i % 2 == 0:
            h[start_idx:end_idx] = 0.7
        else:
            h[start_idx:end_idx] = 0.3
    
    # Make sure the sum is exactly right
    current_sum = np.sum(h)
    h = h * (TARGET_SUM / current_sum)
    
    # Add a bit of smoothing to make optimization easier
    h = np.convolve(h, np.ones(5)/5, mode='same')
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    TARGET_SUM = n_points / 2
    
    # Start with a better initial construction
    h0 = generate_initial_h()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Use global optimization approach with multiple restarts
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use differential evolution to find a global minimum first
    try:
        res_de = differential_evolution(
            objective,
            bounds,
            constraints=cons,
            maxiter=500,
            popsize=15,
            seed=seed,
            disp=False
        )
        if res_de.success:
            h_de = project_to_sum_box(res_de.x, TARGET_SUM)
            c5_de = evaluate_c5(h_de)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
    except:
        pass
        
    # Local refinement with multiple restarts
    max_restarts = 200
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb best solution with noise
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue
            
    # A final local optimization to polish
    try:
        res_final = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-15}
        )
        if res_final.success:
            h_final = project_to_sum_box(res_final.x, TARGET_SUM)
            new_c5 = evaluate_c5(h_final)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
