import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_improve(h, max_iter=1000):
    """Improved local search to find better solutions"""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        if i % 100 == 0:
            corr_full = compute_corr_fft(h)
        
        # Find the current maximum correlation index
        max_idx = np.argmax(corr_full)
        max_idx = max_idx - (len(h) - 1)  # Convert to shift index
        
        # Increase h(x) where it would help the maximum correlation
        # Specifically, try increasing h at positions that affect max correlation
        
        # Find all points that contribute to the maximum correlation
        candidates = []

            k = max_idx

        best_improvement = 0
        best_move = None

                    h_new = h.copy()
                    h_new[j] += delta
                    if h_new[j] < 0 or h_new[j] > 1:
                        continue
                    
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    new_c5 = evaluate_c5(h_new)
                    improvement = best_c5 - new_c5
                    
                    if improvement > best_improvement and improvement > 1e-12:
                        best_improvement = improvement
                        best_move = (j, delta)
        
        if best_move is not None:
            j, delta = best_move
            h[j] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
        else:
            # If no improvement, try random perturbations
            if np.random.random() < 0.3:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
            
            # Occasionally do gradient descent-like updates
            if np.random.random() < 0.05:
                # Estimate direction of steepest descent via finite differences
                eps = 0.005
                grad = np.zeros(n_points)
                c5_base = evaluate_c5(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                    c5_plus = evaluate_c5(h_plus)
                    
                    grad[i] = (c5_plus - c5_base) / eps
                
                # Update with momentum
                momentum = 0.1
                h -= momentum * grad
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def genetic_algorithm_optimize(h0, num_generations=50, population_size=20):
    """Use genetic algorithm to explore solution space"""
    # Create initial population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        individual = h0 + np.random.normal(0, 0.01, n_points)
        individual = np.clip(individual, 0, 1)
        individual = project_to_sum_box(individual, TARGET_SUM)
        population.append(individual)
    
    for gen in range(num_generations):
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//2]
        
        # Keep top performers
        new_population = [population[i] for i in top_indices]
        
        # Generate offspring from top performers
        while len(new_population) < population_size:
            # Tournament selection
            parent_a_idx = np.random.choice(top_indices)
            parent_b_idx = np.random.choice(top_indices)
            
            parent_a = population[parent_a_idx]
            parent_b = population[parent_b_idx]
            
            # Crossover - blend two individuals
            alpha = np.random.random()
            offspring = alpha * parent_a + (1 - alpha) * parent_b
            
            # Mutation
            if np.random.random() < 0.3:
                mut_strength = 0.02
                mutate_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                for idx in mutate_indices:
                    offspring[idx] += np.random.normal(0, mut_strength)
            
            # Project to feasible set
            offspring = np.clip(offspring, 0, 1)
            offspring = project_to_sum_box(offspring, TARGET_SUM)
            
            new_population.append(offspring)
        
        population = new_population
    
    # Return best from last generation
    population_fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(population_fitness)
    return population[best_idx]
