# Global variables
TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move to reduce C5"""

        corr = compute_corr_fft(h)

        # Compute gradient for each position using the correlation
        # We compute the gradient analytically
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Find best move
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
            
        # Move small amount
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta <= 0:
            break
            
        h[idx_min] -= delta
        h[idx_max] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check convergence
        if evaluate_c5(h) < 0.379:
            break
            
    return h

    # This is a greedy local search
    h_improved = h.copy()
    
    # Try to increase h[i] and decrease h[j] where i = j + max_idx or vice versa
    # Try a few near-indices
    for _ in range(10):

            h_improved[i] += delta
            h_improved[j] -= delta
    
    h_improved = project_to_sum_box(h_improved, TARGET_SUM)
