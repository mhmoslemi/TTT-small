from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move to reduce C5"""

        worst_lag = np.argmax(corr)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Find best move
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)

            break

        delta = min(h[idx_min], 1 - h[idx_max], 0.01)

            break

        h[idx_max] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check convergence
        if evaluate_c5(h) < 0.379:
            break
            
    return h

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def optimize_and_refine(h0, n_iter):
    """Optimize a given starting point"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    for i in range(n_iter):
        # Random perturbation
        h_perturbed = h + np.random.normal(0, 0.01, n_points) * np.random.random()
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        # Local refinement
        h_refined = local_refine(h_perturbed)
        
        # Gradient based move
        h_grad = gradient_based_move(h_refined.copy())
        
        # Evaluate
        c5_new = evaluate_c5(h_grad)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_grad.copy()
    
    return best_h, best_c5
