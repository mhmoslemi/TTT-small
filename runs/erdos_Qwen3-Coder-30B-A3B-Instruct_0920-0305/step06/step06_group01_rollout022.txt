# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_move(h):

    if new_val < 0:
        delta = -h[i]
    elif new_val > 1:
        delta = 1 - h[i]
    
    return (i, delta)

def local_optimize(h, max_iter=500):
    """Refine solution using scipy.optimize"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def perturbed_hillclimb(h, iterations=300):
    """Perform perturbed hill climbing"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for _ in range(iterations):
        # Random perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        candidate_h = current_h + perturbation
        candidate_h = project_to_sum_box(candidate_h, TARGET_SUM)
        
        # Evaluate
        candidate_c5 = evaluate_c5(candidate_h)
        
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            
        # Occasionally step towards the max correlation
        if np.random.rand() < 0.1:
            # Find which positions to adjust
            corr = compute_corr_fft(current_h)
            max_lag_idx = np.argmax(corr)

            adjust_idx = max_lag_idx % n_points if max_lag_idx < n_points else (max_lag_idx - n_points) % n_points

            current_h = project_to_sum_box(current_h, TARGET_SUM)
            current_c5 = evaluate_c5(current_h)
    
    return current_h, current_c5
