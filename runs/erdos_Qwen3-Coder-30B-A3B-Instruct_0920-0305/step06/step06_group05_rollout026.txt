import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move to reduce C5"""
    h = h.copy()
    for _ in range(max_iterations):
        # Compute gradients for each position
        grad = np.zeros(n_points)
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr)
        
        # Compute gradient for each position
        for i in range(n_points):
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Find best move
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
            
        # Move small amount
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta <= 0:
            break
            
        h[idx_min] -= delta
        h[idx_max] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check convergence
        if evaluate_c5(h) < 0.379:
            break
            
    return h

def hybrid_optimization(h_in, max_time):
    """Hybrid optimization approach combining multiple strategies"""
    start_time = time.time()
    best_h = h_in.copy()
    best_c5 = evaluate_c5(h_in)
    
    # Strategy 1: Gradient-based descent
    if time.time() - start_time < max_time * 0.4:
        h1 = gradient_based_move(best_h.copy(), 100)
        c5_1 = evaluate_c5(h1)
        if c5_1 < best_c5:
            best_h = h1
            best_c5 = c5_1
    
    # Strategy 2: Local refinement
    if time.time() - start_time < max_time * 0.6:
        h2 = local_refine(best_h.copy())
        c5_2 = evaluate_c5(h2)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
    
    # Strategy 3: Random perturbation + refinement
    if time.time() - start_time < max_time * 0.8:
        h3 = best_h + np.random.normal(0, 0.01, n_points) * 0.5
        h3 = project_to_sum_box(h3, TARGET_SUM)
        h3 = local_refine(h3)
        c5_3 = evaluate_c5(h3)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3
            
    # Strategy 4: Simulated Annealing-like moves
    if time.time() - start_time < max_time:
        h4 = best_h.copy()
        temp = 0.1
        for _ in range(500):
            if time.time() - start_time >= max_time:
                break
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # Compute change in objective
            old_val = h4[i]
            new_val = np.clip(old_val + np.random.normal(0, temp), 0, 1)
            delta = new_val - old_val
            
            # Store old value of j if necessary
            old_j = h4[j]
            
            # Try both moves
            h4[i] = new_val
            h4[j] = np.clip(h4[j] + delta, 0, 1)
            
            # Check if move improves solution
            c5_new = evaluate_c5(h4)
            if c5_new < best_c5:
                best_h = h4.copy()
                best_c5 = c5_new
            else:
                # Restore original values
                h4[i] = old_val
                h4[j] = old_j
                
            # Decrease temperature
            temp *= 0.999
            if temp < 1e-6:
                temp = 1e-6
    
    return best_h, best_c5
