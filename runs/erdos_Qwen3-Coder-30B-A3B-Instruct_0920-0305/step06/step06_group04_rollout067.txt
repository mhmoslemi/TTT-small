def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def build_milp_binary(N):
    """Build binary MILP model for overlap minimization"""
    # Create variables
    h = cp.Variable(N, integer=True)
    y = {}
    lambda_var = cp.Variable(integer=True)
    
    # Add constraints
    constraints = []
    
    # Weight constraint
    constraints.append(cp.sum(h) == N/2)
    
    # Product constraints and overlap constraints
    for ell in range(N):
        for i in range(N - ell):
            if (i, ell) not in y:
                y[(i, ell)] = cp.Variable(integer=True)
            # McCormick constraints
            constraints.append(y[(i, ell)] <= h[i])
            constraints.append(y[(i, ell)] <= h[i + ell])
            constraints.append(y[(i, ell)] >= h[i] + h[i + ell] - 1)
            # Ensure y is binary (lossless for integer h)
            
        # Lag overlap constraint
        if ell > 0:  # Skip ell = 0 since it doesn't involve overlap
            lag_sum = cp.sum([y[(i, ell)] for i in range(N - ell)])
            constraints.append(lag_sum >= lambda_var)
    
    # Bound on lambda
    constraints.append(lambda_var <= N/2)
    
    # Objective
    objective = cp.Maximize(lambda_var)
    
    # Build problem
    prob = cp.Problem(objective, constraints)
    return prob, h, y, lambda_var

def solve_milp_binary(N, timeout_s):
    """Solve the binary MILP model with time limit"""
    try:
        prob, h, y, lambda_var = build_milp_binary(N)
        # Solve with timelimit
        prob.solve(solver=cp.CBC, timelimit=timeout_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            lambda_sol = lambda_var.value
            return h_sol, lambda_sol
        else:
            return None, None
    except Exception as e:
        return None, None
