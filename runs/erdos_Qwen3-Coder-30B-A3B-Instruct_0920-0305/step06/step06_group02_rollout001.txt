import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_binary(x):
    """Evaluate C5 for binary vector"""
    # x should be binary: 0/1 entries
    conv = np.correlate(x, x, mode='full')
    min_overlap = np.min(conv)
    c5 = (TARGET_SUM - min_overlap) * dx
    return c5

def update_conv_for_swap(conv, ones, i, j, n):
    """Update convolution after swapping x[i] from 1 to 0 and x[j] from 0 to 1"""
    # Need to prevent modifying original conv array
    conv_copy = conv.copy()
    offset = n - 1
    
    # Remove i's contributions
    for k in ones:
        shift = k - i
        idx = shift + offset
        conv_copy[idx] -= 1
        if k != i:
            conv_copy[-shift + offset] -= 1
    
    # Update ones list
    ones_new = ones.copy()
    ones_new.remove(i)
    ones_new.append(j)
    
    # Add j's contributions  
    for k in ones_new:
        shift = k - j
        idx = shift + offset
        conv_copy[idx] += 1
        if k != j:
            conv_copy[-shift + offset] += 1
    
    # Add self-contribution for j
    conv_copy[offset] += 1
    
    return conv_copy
