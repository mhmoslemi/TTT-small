import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.linalg import eigh
import random

def compute_corr_fft(h):
    """Compute correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value."""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h, dx):
    """Fast evaluation of C5 using FFT."""
    corr = compute_corr_fft(h)
    return dx * c5_from_corr(corr, len(h))

def hyperplane_round(X, h_rel, M, n):
    """Generate binary vectors using random hyperplane rounding."""
    # Perform SVD
    U, s, Vt = np.linalg.svd(X)
    r = np.sum(s > 1e-10)
    Vp = Vt[:r].T  # r x r matrix of good eigenvectors

    candidates = []
    # Run random trials
    for _ in range(20):
        g = np.random.normal(0, 1, r)
        s = Vp @ g
        b = (s >= 0).astype(int)
        # Repair to correct number of 1s
        ones_count = np.sum(b)
        if ones_count > M:
            to_zero = np.random.choice(np.where(b == 1)[0], ones_count - M, replace=False)
            b[to_zero] = 0
        elif ones_count < M:
            to_one = np.random.choice(np.where(b == 0)[0], M - ones_count, replace=False)
            b[to_one] = 1
        candidates.append(b)
    return candidates

def threshold_round(h_rel, M):
    """Threshold rounding of h_rel to binary."""
    # Get indices of largest elements
    idxs = np.argsort(h_rel)[::-1]
    b = np.zeros_like(h_rel, dtype=int)
    b[idxs[:M]] = 1
    return b

def swap_improve(h, n, M, dx, max_swaps=5000):
    """Swap hill-climbing."""
    h = h.copy()
    improved = True
    swaps_attempted = 0
    while improved and swaps_attempted < max_swaps:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if swaps_attempted >= max_swaps:
                break
            for j in zeros:
                swaps_attempted += 1
                h[i], h[j] = h[j], h[i]
                corr = compute_corr_fft(h)
                new_c5 = dx * np.max(corr)
                if new_c5 < evaluate_c5(h, dx):
                    h[i], h[j] = h[j], h[i]
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
        if improved:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_points = 0
    dx = 0.0  # Will compute
    
    # Phase A: SDP solving
    sdp_time_limit = 0.5 * budget_s
    n_vals = [60, 80, 100]  # Try these sizes
    
    for n in n_vals:
        if time.time() - start_time > sdp_time_limit:
            break
            
        M = n // 2
        dx = 2.0 / n
        print(f"Running SDP at n = {n}...")

        # Formulate SDP
        h = cp.Variable(n)
        X = cp.Variable((n, n), symmetric=True)
        t = cp.Variable()
        
        constraints = [
            cp.sum(h) == M,
            0 <= h,
            h <= 1,
            cp.diag(X) == h,
            X >> 0
        ]
        
        # Lag constraints
        for k in range(n):
            Pk = np.zeros((n, n))
            for i in range(n):
                if i + k < n:
                    Pk[i, i + k] = 1
            constraints.append(t >= M - cp.trace(Pk @ X))
        
        prob = cp.Problem(cp.Minimize(t), constraints)
    
        # Solve with SCS
        try:
            prob.solve(solver=cp.SCS, time_limit=2 * (sdp_time_limit / len(n_vals)), eps=1e-6)
            if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                h_opt = h.value
                X_opt = X.value
                t_val = t.value
                
                # Evaluate lower bound
                lower_bound = dx * t_val
                print(f"SDP at n={n} gave lower bound: {lower_bound:.6f}")
                
                # Rounding strategies
                candidates = []
                
                # Hyperplane rounding
                candidates.extend(hyperplane_round(X_opt, h_opt, M, n))
                
                # Threshold rounding
                b_thresh = threshold_round(h_opt, M)
                candidates.append(b_thresh)
                
                # Evaluate candidates
                for cand in candidates:
                    if not np.isclose(np.sum(cand), M):
                        idx = np.argsort(h_opt)[::-1][:M]
                        b = np.zeros_like(h_opt, dtype=int)
                        b[idx] = 1
                        cand = b
                    
                    # Apply local swaps
                    cand = swap_improve(cand, n, M, dx)
                    c5_val = evaluate_c5(cand, dx)
                    
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = cand.copy()
                        n_points = n
                        print(f"Improved bound at n={n}: {best_c5:.6f}")
                        
        except Exception as e:
            print(f"SDP at n={n} failed: {e}")
            continue
    
    # Phase B: Continuous polishing
    if best_h is not None and (time.time() - start_time) < 0.9 * budget_s:
        # Apply SLSQP polish
        from scipy.optimize import minimize
        
        def obj(x):
            return evaluate_c5(x, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                best_h = res.x
                best_h = project_to_sum_box(best_h, n_points / 2)
                best_c5 = evaluate_c5(best_h, dx)
                print(f"SLSQP polished to: {best_c5:.6f}")
        except:
            pass
    
    # Fallback: If no good solution yet, use deterministic construction
    if best_h is None:
        n_points = 100
        dx = 2.0 / n_points
        h_fallback = np.zeros(n_points)
        mid = n_points // 2
        h_fallback[:mid] = 1
        h_fallback = project_to_sum_box(h_fallback, n_points / 2)
        best_h = h_fallback
        best_c5 = evaluate_c5(best_h, dx)
        print(f"Fallback construction: {best_c5:.6f}")
        
    return (best_h.tolist(), float(best_c5), n_points)
