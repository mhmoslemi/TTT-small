import numpy as np

import time

def solve_milp(n, time_limit):
    """Solve MILP formulation for binary h variables and auxiliary y variables"""
    M = n // 2  # required number of ones
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()  # bounds worst-lag overlap
    
    # Constraints
    constraints = [
        cp.sum(h) == M,  # exact mass constraint
        t >= 0  # t is non-negative
    ]
    
    # Linking AND constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.extend([
                y[i, j] <= h[i],
                y[i, j] <= h[j],
                y[i, j] >= h[i] + h[j] - 1
            ])
    
    # Lag constraints for each positive lag k
    for k in range(1, n):
        # Sum of products for this lag
        lag_sum = cp.sum(y[range(n-k), range(k, n)])
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURACY]:
            h_sol = np.rint(h.value).astype(int)
            t_sol = t.value
            return h_sol, t_sol
            
    except Exception:
        pass
        
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def local_continuous_refine(h_start, target_sum, max_iter=200):
    """Perform continuous local refinement using SLSQP"""
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = cp.minimize(
            objective, 
            h_start, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, target_sum)
            return h
    except:
        pass
    return h_start
