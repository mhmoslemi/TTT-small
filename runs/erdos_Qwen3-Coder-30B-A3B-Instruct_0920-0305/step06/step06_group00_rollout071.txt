TARGET_SUM = None

dx = 2.0 / n_points

def evaluate_max_corr(h):
    """Compute max correlation using FFT."""
    corr = full_corr(h, 1 - h)
    return np.max(corr) * dx

def full_corr(a, b):
    """Compute full zero-padded correlation."""
    n = a.size
    N = 2 * n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2 * n - 1]

def project_to_sum_box(x, target):
    """Project onto box constraints and fixed sum."""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_max_soft(h, alpha):
    """Compute the gradient of the soft maximum of correlations."""
    n = len(h)
    corr = full_corr(h, 1 - h)
    max_c = corr.max()
    w = np.exp(alpha * (corr - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    w = w / w_sum
    # Use FFT to compute weights * h shifts efficiently
    N = 2 * n
    w_fft = np.fft.rfft(w, N)
    h_fft = np.fft.rfft(h, N)
    # Forward shifts: w * h[i+l]
    conv_fwd = np.fft.irfft(h_fft * w_fft, N)
    conv_fwd = conv_fwd[:1 + 2 * n - 2]  # Keep valid part
    # Backward shifts: w * h[i-l] => related to reverse of h convoluted with reverse of w
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, N)
    # We have w_rev * h[i - l] = convolve(h, w_rev) indexed as h[i-l] we want i-th term
    # Direct formulation: pointwise convolution works if we pad ranges correctly.
    conv_bwd = np.fft.irfft(h_fft * w_rev_fft, N)
    conv_bwd = conv_bwd[:1 + 2 * n - 2]
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def line_search(h, grad, alpha, target_sum, eta=1e-2):
    """Backtracking line search on the projected next iterate."""
    step_size = eta
    while step_size > 1e-12:
        h_new = project_to_sum_box(h - step_size * grad, target_sum)
        cost_new = evaluate_max_corr(h_new)
        cost_old = evaluate_max_corr(h)
        if cost_new < cost_old:
            return h_new, step_size
        step_size *= 0.5
    return h, step_size

def optimize_continuous(start_h, time_limit, alpha_start=5.0, alpha_end=20.0):
    """Gradient descent optimizer from a start point."""
    start_time = time.time()
    best_h = start_h.copy()
    best_val = evaluate_max_corr(best_h)
    alpha = alpha_start
    h = start_h.copy()
    # Parameters for step size and update
    # We'll do a simple exponential decay
    for iteration in range(MAX_ITERATIONS):
        if time.time() - start_time > time_limit:
            break
        grad = gradient_max_soft(h, alpha)
        h, step_size = line_search(h, grad, alpha, TARGET_SUM)
        val = evaluate_max_corr(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # Adaptive alpha increment
        if iteration % 50 == 0:
            alpha = min(alpha_end, alpha + 0.2)
    return best_h, best_val

def binary_local_search(best_h, remaining_time):
    """Hill climbing from the best continuous solution to binary."""
    start_time = time.time()
    # Convert continuous to binary
    h_bin = np.zeros_like(best_h)
    indices = np.argsort(-best_h)  # descending
    h_bin[indices[:int(TARGET_SUM)]] = 1.0
    best_val = evaluate_max_corr(h_bin)
    best_h_bin = h_bin.copy()
    while time.time() - start_time < remaining_time:
        ones = np.where(best_h_bin > 0.5)[0]
        zeros = np.where(best_h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best_h_bin.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = evaluate_max_corr(cand)
        if val < best_val:
            best_val = val
            best_h_bin = cand
    return best_h_bin, best_val
