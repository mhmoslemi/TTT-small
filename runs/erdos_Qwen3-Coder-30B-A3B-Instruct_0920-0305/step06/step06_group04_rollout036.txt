import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, sum_entries, Problem, Int, Bool, Matrix

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def evaluate_milp_solution(h):
    """Evaluate a binary solution for the MILP"""
    # Compute correlation for binary h using fast method
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def build_and_solve_milp(N, max_time_s):
    """Use CVXPY to build and solve dense MILP for best λ"""
    # Create variables
    h_vars = Variable(N, boolean=True)
    y_vars = {}
    lambda_var = Variable(integer=True)
    
    # Create y variables indexed by (i, ℓ)
    # Use the constraint that y_{i,ℓ} = h_i h_{i+ℓ}
    constraints = []
    
    # Weight constraint
    constraints.append(sum_entries(h_vars) == N/2)
    
    # For each lag ℓ from 0 to N-1
    for l in range(N):
        y_vars[l] = {}
        if l == 0:
            # For lag 0, no previous element to check
            continue
        for i in range(N - l):
            y_vars[l][i] = Variable(boolean=True)
            # McCormick envelopes
            constraints.append(y_vars[l][i] <= h_vars[i])
            constraints.append(y_vars[l][i] <= h_vars[i+l])
            constraints.append(y_vars[l][i] >= h_vars[i] + h_vars[i+l] - 1)
            # Sum of y's for this lag should be ≥ lambda
            if l == 1:
                # Start accumulating y's for this lag
                try:
                    lag_sum = y_vars[l][i]
                except KeyError:
                    lag_sum = 0
            try:
                lag_sum += y_vars[l][i]
            except KeyError:
                if l not in y_vars:
                    y_vars[l] = {}
                y_vars[l][i] = Variable(boolean=True)
                constraints.append(y_vars[l][i] <= h_vars[i])
                constraints.append(y_vars[l][i] <= h_vars[i+l])
                constraints.append(y_vars[l][i] >= h_vars[i] + h_vars[i+l] - 1)
                lag_sum = y_vars[l][i]

    # Constraint for all lags at once
    for l in range(1, N):
        try:
            lag_sum = sum(y_vars[l][i] for i in range(N - l))
            constraints.append(lag_sum >= lambda_var)
        except:
            continue
            
    # Additional constraint on lambda:
    constraints.append(lambda_var <= N/2)
    
    # Objective
    objective = Minimize(-lambda_var)  # Minimize -lambda_var to maximize lambda_var
    
    # Curate constraints cleanly
    prob = Problem(objective, constraints) 

    try:
        prob.solve(solver=cp.CBC, max_seconds=max_time_s, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h_vars.value
            # Return binary solution as a list
            return np.round(h_sol).astype(int).tolist(), prob.value
        else:
            return None, float('inf')
    except:
        return None, float('inf')
