import numpy as np
import time
import random

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create an initial binary vector with exactly TARGET_SUM ones."""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full overlap array."""
    return np.correlate(x, x, mode='full')

def update_overlap(ov, x, i, j, ones, zeros):
    """Update overlap array after swapping x[i] and x[j]."""
    ov_new = ov.copy()
    
    # Remove contributions from i
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n_points - 1] -= 1
    # Add contributions from j (but not involving i)
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    
    # Adjust the shared indices (very few effect, so we handle them carefully)
    # Remove i's contribution from any shift with i by computing all pairs involving i
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n_points - 1] -= 1
    
    # Add j's contribution to shifts involving j
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    
    return ov_new

def get_sets(x):
    """Get indices of ones and zeros."""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    return ones, zeros

def swap_update(ov, x, i, j, ones, zeros):
    """Efficiently update overlap after swapping positions i and j."""
    # Create copy
    ov_new = ov.copy()
    
    # Compute shifts affected by removal of 1 at i and addition of 1 at j
    affected_shifts = set()
    # For each 1 before removal, check its contribution to the overlap
    for idx in ones:
        if idx != i:
            s = idx - i  # shift due to i's removal  
            affected_shifts.add(s)
            
    # For each 1 after removal, check its new contribution 
    for idx in ones:
        if idx != i:
            s = idx - j  # shift due to j's addition  
            affected_shifts.add(s)
                
    # Do the changes incrementally - each overlap affected
    for s in affected_shifts:
        # i was contributing to O_s because we had 1 at i and 1 at i+s
        # need to remove these
        if i+s >= 0 and i+s < n_points and x[i+s] == 1:
            ov_new[s + n_points - 1] -= 1
        if j+s >= 0 and j+s < n_points and x[j+s] == 1:
            ov_new[s + n_points - 1] += 1
            
    return ov_new

def simulated_annealing(initial_h, initial_ov, ones, zeros, time_limit):
    """Run simulated annealing with incremental updates."""
    T0 = 0.5
    lambda_decay = 0.0003
    best_x = initial_h.copy()
    best_ov = initial_ov.copy()
    cur_min = np.min(initial_ov)
    best_min = cur_min
    cur_x = initial_h.copy()
    cur_ov = initial_ov.copy()
    
    start = time.time()
    iteration = 0
    
    while time.time() - start < time_limit:
        # Generate neighbor by swapping a 1 and 0 randomly
        i = random.choice(ones)
        j = random.choice(zeros)

        # Compute new overlap after the swap
        ov_new = swap_update(cur_ov, cur_x, i, j, ones, zeros)
        new_min = np.min(ov_new)
        delta = new_min - cur_min

        # Accept or reject
        accept = False
        if delta > 0:
            accept = True
        else:
            T = T0 * np.exp(-lambda_decay * iteration)
            if random.random() < np.exp(delta / max(T, 1e-8)):
                accept = True

        if accept:
            cur_min = new_min
            if new_min > best_min:
                best_min = new_min
                best_x = cur_x.copy()
                best_ov = ov_new.copy()
            
            # Update 1's and 0's sets and current vector
            cur_x[i], cur_x[j] = cur_x[j], cur_x[i]
            ones = np.where(cur_x == 1)[0]
            zeros = np.where(cur_x == 0)[0]
            cur_ov = ov_new
        
        iteration += 1
        
        if iteration % 10000 == 0:
            if time.time() - start > 0.95 * time_limit:
                break

    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine solution by explicitly improving the currently worst overlap."""
    min_idx = np.argmin(ov)
    worst_shift = min_idx - (n_points - 1)
    
    # Identify 1's and 0's involved in the worst shift
    A = []
    B = []
    
    for i in ones:
        if 0 <= i + worst_shift < n_points:
            if x[i + worst_shift] == 0:
                A.append(i)
    
    for j in zeros:
        if 0 <= j + worst_shift < n_points:
            if x[j + worst_shift] == 1:
                B.append(j)
    
    # Try to improve by swapping 1's in A with 0's in B
    ok = True
    while ok and A and B:
        ok = False
        for i in list(A):
            for j in list(B):
                # Check if this swap increases the worst shift
                old_i = x[i]
                old_j = x[j]
                
                # Temporarily perform swap
                x[i] = old_j
                x[j] = old_i
                
                # Check if the overlap at the worst shift increases or stays same
                ov_new = full_overlap(x)
                new_min_idx = np.argmin(ov_new)
                new_worst_shift = new_min_idx - (n_points - 1)
                if new_worst_shift == worst_shift and ov_new[new_min_idx] > ov[min_idx]:
                    # Accept swap
                    ov[min_idx] = ov_new[new_min_idx]
                    A.remove(i)
                    B.remove(j)
                    ok = True
                    break
                
                # Revert swap
                x[i] = old_i
                x[j] = old_j
            if ok:
                break

    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Multiple runs to avoid local minima
    for run_id in range(10):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Initialize 
        x = initial_binary(seed + run_id)
        ones, zeros = get_sets(x)
        ov = full_overlap(x)
        cur_min = np.min(ov)
        
        # Run simulated annealing      
        final_x, final_ov, final_min = simulated_annealing(
            x, ov, ones, zeros, 
            min(500, budget_s * 0.1)
        )
        
        # Refine by worst shift
        refined_x, refined_ov = worst_shift_refine(final_x.copy(), final_ov, 
                                                  np.where(final_x == 1)[0], 
                                                  np.where(final_x == 0)[0])
        refined_min = np.min(refined_ov)
        
        if refined_min > cur_min:
            cur_min = refined_min
            final_x = refined_x
            final_ov = refined_ov
            
        c5 = dx * (TARGET_SUM - cur_min)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = final_x.copy()
            
        # Early stopping if we're close to target
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    if best_h is not None:
        ones, zeros = get_sets(best_h)
        ov = full_overlap(best_h)
        refined_x, refined_ov = worst_shift_refine(best_h.copy(), ov, ones, zeros)
        c5_refine = dx * (TARGET_SUM - np.min(refined_ov))
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = refined_x.copy()
    
    h_values = best_h.astype(float).tolist()
    return (h_values, float(best_c5), n_points)
