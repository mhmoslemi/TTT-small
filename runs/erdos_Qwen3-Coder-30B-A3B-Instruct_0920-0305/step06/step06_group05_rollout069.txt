import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.fft import fft, ifft
import math

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points
TIME_BUDGET = 600

def compute_corr_fft(h):
    """Fast evaluation of correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_fast(h):
    """Compute C5 value using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, target_sum, time_limit):
    """Solve the SDP relaxation"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.diag(X) == h,  # X_ii = h_i
        X >> 0,           # X is psd
        cp.sum(h) == target_sum,
        0 <= h,
        h <= 1,
        t >= target_sum - cp.trace(P @ X) for P in get_lag_matrices(n)
    ]

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=200000, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, X.value
        else:
            # Fallback to whatever was obtained
            return h.value, X.value
    except Exception:
        return h.value, X.value

def get_lag_matrices(n):
    """Get the lag matrices P_k for all k = 0, ..., n-1"""
    P_list = []
    for k in range(n):
        P = np.zeros((n, n))
        for i in range(n - k):
            P[i, i + k] = 1
        P_list.append(P)
    return P_list

def random_rounding(h_rel, X_rel, M, n, num_trials=15):
    """Perform random hyperplane rounding"""
    candidates = []

    # Compute factorization of X_rel
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_rel)
        keep = eigenvals > 1e-8
        V_plus = eigenvecs[:, keep]
        r = V_plus.shape[1]

        # Perform trials
        for _ in range(num_trials):
            g = np.random.randn(r)
            s = V_plus @ g
            b = (s >= 0).astype(int)

            # Repair weight to exactly M
            current_sum = np.sum(b)
            if current_sum > M:
                to_flip = np.random.choice(np.where(b == 1)[0], size=int(current_sum - M), replace=False)
                b[to_flip] = 0
            elif current_sum < M:
                to_flip = np.random.choice(np.where(b == 0)[0], size=int(M - current_sum), replace=False)
                b[to_flip] = 1

            candidates.append(b)
    except:
        # Fallback to threshold rounding
        sort_idx = np.argsort(h_rel)[::-1]
        b = np.zeros(n)
        b[sort_idx[:int(M)]] = 1
        candidates.append(b)

    return candidates

def deterministic_threshold_round(h_rel, M, n):
    """Deterministic threshold rounding"""
    sort_idx = np.argsort(h_rel)[::-1]
    b = np.zeros(n)
    b[sort_idx[:int(M)]] = 1
    return [b]

def hill_climb(h_candidate):
    """Deterministic hill-climbing with swap strategy"""
    h = h_candidate.copy()
    best_c5 = compute_c5_fast(h)
    improved = True

    # Precompute the correlation matrix to speed up update
    prev_corr = compute_corr_fft(h)
    corr_changes = np.zeros(2 * len(h) - 1)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    while improved:
        improved = False
        for i in ones:
            if improved:
                break
            for j in zeros:
                if improved:
                    break
                # Check the contribution to each lag term
                # Compute new C5 after swap
                # Only that part of the lags involving i or j will change
                new_h = h.copy()
                new_h[i] = 0
                new_h[j] = 1

                new_c5 = compute_c5_fast(new_h)
                if new_c5 < best_c5 - 1e-10:
                    h = new_h
                    best_c5 = new_c5
                    improved = True
                    break

    return h

def polish_candidate(h_in, max_iters=150):
    """Use SLSQP to polish a discrete candidate a bit"""
    from scipy.optimize import minimize
    
    h = h_in.copy()
    
    def objective(h):
        return compute_c5_fast(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iters, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    global TARGET_SUM, n_points, dx
    
    # Initialize best solution
    best_h = np.zeros(n_points)
    best_c5 = float('inf')
    
    # Phase A: SDP solving with successively larger n
    n_vals = [60, 80, 100]
    best_n = 100
    for n in n_vals:
        if time.time() - start_time > 0.45 * budget_s:
            break

        n_points = n
        dx = 2.0 / n
        TARGET_SUM = n / 2

        time_left = budget_s - (time.time() - start_time)
        if time_left <= 5:
            break

        h_rel, X_rel = solve_sdp(n, TARGET_SUM, time_left * 0.8)

        if h_rel is None:
            continue

        # Rounding and hill-climbing
        candidates = []

        # Do both random and threshold rounding
        random_candidates = random_rounding(h_rel, X_rel, TARGET_SUM, n, 15)
        candidates.extend(random_candidates)

        deterministic_candidates = deterministic_threshold_round(h_rel, TARGET_SUM, n)
        candidates.extend(deterministic_candidates)

        for h_can in candidates:
            if time.time() - start_time > 0.8 * budget_s:
                break
            if np.sum(h_can) != TARGET_SUM:
                h_can = project_to_sum_box(h_can, TARGET_SUM)

            h_can = hill_climb(h_can)
            c5_val = compute_c5_fast(h_can)
            if c5_val < best_c5:
                best_h = h_can.copy()
                best_c5 = c5_val
                best_n = n

        if time.time() - start_time > 0.45 * budget_s:
            break

    # Phase B: Polishing the best
    if time.time() - start_time < 0.9 * budget_s:
        best_h = polish_candidate(best_h, 100)

    return (best_h.tolist(), float(best_c5), best_n)
