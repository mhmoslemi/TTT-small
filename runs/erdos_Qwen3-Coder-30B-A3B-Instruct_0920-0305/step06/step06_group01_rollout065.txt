import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_weights(S, alpha):
    """Compute weights for soft minimum"""
    max_S = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * S - max_S)
    w = exp_vals / np.sum(exp_vals)
    return w

def compute_gradient(h, w):
    """Compute gradient using convolution"""
    # Create symmetric kernel W
    n = len(h)
    W = np.zeros(2 * n - 1)
    # w is symmetric, so W[0] = w[n-1], W[1] = w[n-2], etc.
    W[n-1:] = w[:n]  # Upper half
    W[:n-1] = w[1:][::-1]  # Lower half reversed
    
    # Convolution to compute gradient
    conv = np.convolve(h, W, mode='full')
    g = conv[n-1:n-1+n]
    
    # Add contribution from lag 0
    g += w[n-1] * h
    
    return g

    # Initial solution
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2
    
    # Optimisation parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30.0
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    stagnation_counter = 0
    
    h = h_interpolated.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute auto-correlations
        S = correlate(h, h, 'full')  # Length 2*n - 1
        
        # Compute soft minimum weights
        w = compute_soft_min_weights(S, alpha)
        
        # Compute gradient
        g = compute_gradient(h, w)
        
        # Candidate update with step size
        h_new = h + step * g
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate exact C5
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = c5_new
            step = min(step * step_inc, 0.1)
            stagnation_counter = 0
        else:
            step = max(step * step_dec, 1e-4)
            stagnation_counter += 1
        
        # Annealing temperature
        alpha = min(alpha * alpha_inc, alpha_max)
        
        # Random perturbation every 200 iterations
        if iter_count % 200 == 0:
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            h = h_perturbed
        
        # Restart if stagnation
        if stagnation_counter >= restart_interval:
            h_start = np.random.uniform(0, 1, n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            h = h_start.copy()
            step = 0.02
            alpha = 0.5
            stagnation_counter = 0

    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM

    cons = {'type': 'eq', 'fun': constraint_eq}

        res = minimize(
            objective, 
            best_h,
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,

            h_final = project_to_sum_box(h_final, TARGET_SUM)
            final_c5 = evaluate_c5(h_final)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
