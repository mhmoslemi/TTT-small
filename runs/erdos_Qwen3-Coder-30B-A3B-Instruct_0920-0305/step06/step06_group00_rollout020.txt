import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    # Ensure sum is correct due to floating point drift
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1-1e-12)
        if free.any():
            result[free] += diff / free.sum()
            result = np.clip(result, 0, 1)
    return result

def full_corr(a, b):
    """Zero-padded FFT-based full correlation"""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def gradient_max_soft(h, alpha):
    """Compute gradient of the soft-max approximation of max correlation."""
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w = w / w_sum

    # Gradient via efficient correlation-like convolutions
    conv_fwd = full_corr(h, w)
    conv_bwd = full_corr(h, w[::-1])
    grad = 1.0 - conv_fwd - conv_bwd

    return grad

def optimize_continuous(h0, remaining_time, n, alpha_start=5.0):
    """Run continuous optimization with projected gradient descent."""
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    alpha = alpha_start
    step_size = 0.1
    iter_count = 0
    start = time.time()

    while time.time() - start < remaining_time - 0.1 and iter_count < 5000:
        g = gradient_max_soft(h, alpha)
        step = step_size
        while step > 1e-6:
            h_new = project_to_sum_box(h - step * g, TARGET_SUM)
            val_new = evaluate_c5(h_new)
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                if iter_count % 100 == 0:
                    alpha = min(alpha + 0.02, 20.0)
                break
            step *= 0.5
        if step <= 1e-6:
            pass  # No improvement found; try perturbation or break

        iter_count += 1
        if iter_count % 100 == 0:
            # Check if stagnated and add a small perturbation
            if abs(np.mean(g)) < 1e-7:
                h += np.random.normal(0, 0.005, n)
                h = project_to_sum_box(h, TARGET_SUM)

    if best_val < evaluate_c5(best_h):
        return best_h, best_val
    else:
        return best_h, evaluate_c5(best_h)

def binary_local_search(h_bin, remaining_time, n):
    """Binary simulated annealing refinement"""
    best_h = h_bin.copy()
    best_val = evaluate_c5(best_h)
    start = time.time()

    temp = 0.01
    cooling_rate = 0.999
    while time.time() - start < remaining_time - 0.1:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = evaluate_c5(cand)
        if val < best_val - 1e-9:
            best_val = val
            best_h = cand
        elif np.random.rand() < np.exp(-(val - best_val) / temp):
            best_val = val
            best_h = cand
        temp *= cooling_rate

    return best_h, best_val
