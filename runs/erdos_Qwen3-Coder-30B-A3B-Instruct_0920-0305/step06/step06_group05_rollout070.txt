import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_gradient_descent(h, max_iterations=100):
    """Adaptive gradient descent with momentum and adaptive step size"""
    h = h.copy()
    momentum = np.zeros(n_points)
    velocity = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(max_iterations):
        # Compute current gradient
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr)
        grad = np.zeros(n_points)
        for k in range(n_points):
            if k + worst_lag < n_points:
                grad[k] += h[k + worst_lag]
            if k - worst_lag >= 0:
                grad[k] += h[k - worst_lag]
                
        # Adaptive momentum
        momentum = beta1 * momentum + (1 - beta1) * grad
        velocity = beta2 * velocity + (1 - beta2) * (grad ** 2)
        
        # Bias correction
        momentum_hat = momentum / (1 - beta1 ** (i + 1))
        velocity_hat = velocity / (1 - beta2 ** (i + 1))
        
        # Update step
        step_size = 0.01
        delta = step_size * momentum_hat / (np.sqrt(velocity_hat) + epsilon)
        
        # Apply update
        h_new = h - delta
        
        # Project onto box constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        diff = TARGET_SUM - np.sum(h_new)
        if abs(diff) > 1e-8:
            # Adjust all entries by small amount
            adjust = diff / n_points
            h_new = np.clip(h_new + adjust, 0, 1)
        
        # Check for improvement
        if evaluate_c5(h_new) < evaluate_c5(h):
            h = h_new.copy()
        else:
            step_size *= 0.5  # Reduce step size if no improvement
            
        if i % 10 == 0 and evaluate_c5(h) < 0.3795:
            break
            
    return h

def genetic_algorithm_iteration(h_pop, fitnesses):
    """Perform one generation of genetic algorithm"""
    # Selection (tournament)
    selected = []
    tournament_size = 3
    for _ in range(len(h_pop)):
        idx = np.random.choice(len(h_pop), tournament_size)
        winner = idx[np.argmin([fitnesses[i] for i in idx])]
        selected.append(h_pop[winner].copy())
        
    # Crossover (uniform)
    new_pop = []
    for i in range(0, len(selected), 2):
        if i + 1 < len(selected):
            parent1, parent2 = selected[i], selected[i+1]
            mask = np.random.random(n_points) < 0.5
            child = np.where(mask, parent1, parent2)
        else:
            child = selected[i].copy()
        new_pop.append(child)
        
    # Mutation
    for i in range(len(new_pop)):
        if np.random.random() < 0.1:
            idx = np.random.randint(n_points)
            new_pop[i][idx] = np.random.random()
            
    return new_pop
