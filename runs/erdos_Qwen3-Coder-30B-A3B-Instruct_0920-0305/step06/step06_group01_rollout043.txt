import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = get_initial_h()
    best_c5 = evaluate_c5(best_h)
    TARGET_SUM = n_points / 2
    
    # Multiple optimization strategies
    strategies = [

        ("sgd", 1000, 0.05, 200),

    ]
    
    # Main optimization loop
    strategy_idx = 0
    last_restart = 0
    
    while time.time() - start_time < budget_s - 10:
        strategy_name, max_iter, lr, restart_freq = strategies[strategy_idx % len(strategies)]
        
        # Generate a new starting point periodically
        if strategy_idx > 0 and (strategy_idx - last_restart) >= restart_freq:
            h_start = get_initial_h() + np.random.normal(0, 0.02, n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
            last_restart = strategy_idx
        else:
            h_start = best_h.copy()
            
        # Adaptive gradient descent
        h_current = h_start.copy()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute gradient approximation
            corr = compute_corr_fft(h_current)
            max_idx = np.argmax(corr)
            if max_idx >= n_points:
                lag = max_idx - n_points
            else:
                lag = max_idx
                
            # Simple gradient direction: if a peak is at a specific lag, 
            # adjust values around that lag to reduce correlation at that lag
            gradient = np.zeros_like(h_current)
            
            # Simple rule: reduce the value at the peak lag position
            if 0 <= lag < n_points:
                gradient[lag] = -1.0
                
            # Add noise to escape local minima
            noise = np.random.normal(0, 0.005, n_points)
            
            # Update step with adaptive learning rate
            new_h = h_current + lr * (gradient + noise)
            # Project back to feasible region
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            # Evaluate the candidate solution
            new_c5 = evaluate_c5(new_h)

    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):

            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP',

            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        if res.success:
            best_h = project_to_sum_box(res.x, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
