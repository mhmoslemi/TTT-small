import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # fine grid for accurate FFT
K = 12             # number of harmonics (fourier coefficients)
N = 120            # final number of points
dx = 2.0 / N

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50

    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def fourier_to_h(theta, n_points=N, n_fine=N_FINE):
    """Map low-dimensional fourier coefficients to a feasible h vector"""
    x_fine = np.linspace(0, 2, n_fine)
    # DC component is fixed at 0.5 to ensure integral = 1
    h_fine = 0.5 * np.ones_like(x_fine)
    a = theta[:K]
    b = theta[K:]
    for k in range(1, K+1):
        h_fine += a[k-1]*np.cos(np.pi*k*x_fine) + b[k-1]*np.sin(np.pi*k*x_fine)
    # clip to [0,1]
    h_clipped = np.clip(h_fine, 0, 1)
    # project to sum constraint
    h_projected = project_to_sum_box(h_clipped, n_points / 2.0)
    # downsample to final resolution (every n_fine/n_points entry)
    return h_projected[::n_fine//n_points]

def fourier_objective(theta):
    h = fourier_to_h(theta)
    return evaluate_c5(h)

    best_theta = result.x

    def gradient_descent_refine(h_start, max_iter=200):
        h = h_start.copy()
        alpha = 1.0
        for t in range(max_iter):
            # Run multiple gradient iterations over last few lags
            corr = compute_corr_fft(h)
            # We treat the complement correlation analogously to earlier 
            # but seek to maximize the *minimum* of those values. 
            # So create a "penalized" log-sum-exp version for "soft sorting"
            # or use "rank based" weights to boost worse lag contribution
            f = np.array([np.sum(h[i:] * h[:-i]) for i in range(N)])
            # Smoothed weights for gradient
            w = np.exp(-alpha * f)
            w = w / np.sum(w)  # normalize
            # Gradient from this weighted average
            grad = np.zeros_like(h)
            for i in range(N):
                for l in range(1, N):
                    if i+l < N:
                        grad[i] -= w[l] * h[i+l] 
                    if i-l >= 0:
                        grad[i] -= w[l] * h[i-l]
            
            # Circumvent the double computation of h_rev h & h(1-h)
            # by taking gradient of max ln (f_\ell) = -min (-ln f_\ell)
            # gradient direction is slightly different due  
            # to inconvenient form, so we take a single small step
            eta = 0.02 # learning rate
            h_trial = h - eta * grad
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_try = evaluate_c5(h_trial)
            
            if c5_try < best_c5:
                h = h_trial
                best_c5 = c5_try
                alpha = min(alpha * 1.05, 20.0)  # adaptive increase
            
            if t % 20 == 0:

                if elapsed > 0.95 * budget_s:

        # the last 10 indices are the worst lags

        while swapped < max_swaps and loop_count < 3*max_swaps:
            loop_count +=1
            # ignore ones in worst lag positions, keep the non-ones related to worst
            # we want to move mass from a 1 that hurts a large correlation to a 0
            # that reduces that conflict
            lags_are_worst = [False] * N
            for lag in worst_lags:
                if lag < N//2:
                    lags_are_worst[lag] = True
            # for now we simplify: just take top ranked entries that are 1 and switch 
            # with bottom ranked 0s.
            ones = np.where(h_swap == 1)[0]
            zeros = np.where(h_swap == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = ones[0]
            i0 = zeros[0]
            for i in ones:
                if i1 != i:
                    # check how much it helps
                    h_try = h_swap.copy()
                    h_try[i1], h_try[i] = h_try[i], h_try[i1]
                    c_try = evaluate_c5(h_try)
                    if c_try < best_c5:
                        h_swap = h_try
                        best_c5 = c_try
                        swapped += 1
                        break
            
        return h_swap, best_c5

    h_final, c5_final = binary_swap_refinement(h_after_grad)
    
    # Final call
    actual_c5 = evaluate_c5(h_final)
    return (h_final.tolist(), float(actual_c5), N)
