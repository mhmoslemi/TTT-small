import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT efficiency

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h = h_interpolated.copy()
    lambda_mix = 0.05  # Initial mixing weight
    alpha_lambda = 1.02  # Geometric increase factor
    beta_relax = 0.5  # Relaxation factor
    alpha_beta = 1.01  # Increase of beta
    phase_reset_interval = 300
    restart_counter = 0
    last_improvement = 0
    
    # Pad to power of 2 for FFT
    h_padded = np.pad(h, (0, L - n_points), 'constant')
    
    # Main optimization loop
    iterations = 0
    max_iterations = 50000  # Prevent infinite loops
    
    while time.time() - start_time < budget_s * 0.9 and iterations < max_iterations:
        iterations += 1
        
        # FFT computation
        H = np.fft.fft(h_padded)
        P = np.abs(H)**2
        
        # Target magnitude (flat off-DC)
        P_target = np.mean(P[1:]) if np.sum(P[1:]) > 0 else 0
        M_target = np.sqrt(P_target)
        M_target = np.full_like(np.abs(H), M_target)
        M_target[0] = np.abs(H[0])  # Keep DC component
        
        # Mix in spectral target
        M_new = (1 - lambda_mix) * np.abs(H) + lambda_mix * M_target
        
        # Keep phase
        H_new = M_new * np.exp(1j * np.angle(H))
        
        # Inverse FFT
        h_prime = np.fft.ifft(H_new).real
        h_cand = h_prime[:n_points]

        h_cand = project_to_sum_box(h_cand, TARGET_SUM)
        
        # Apply relaxation blending
        h_new = (1 - beta_relax) * h + beta_relax * h_cand

        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()

    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
