import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_minimum_weights(S, alpha):
    """Compute weights for soft minimum"""
    S_shifted = -alpha * S
    S_shifted -= np.max(S_shifted)  # numerical stability
    exp_S = np.exp(S_shifted)
    weights = exp_S / np.sum(exp_S)
    return weights

def compute_gradient(h, weights):
    """Compute gradient using convolution with symmetric kernel"""
    n = len(h)
    # Build symmetric kernel
    W = np.zeros(2 * n - 1)
    for k in range(n):
        W[n-1+k] = W[n-1-k] = weights[k]
    
    # Convolution (fast with FFT)
    conv = np.convolve(h, W, mode='full')
    g = conv[n-1 : n-1 + n]
    # Add double contribution of lag 0
    g += weights[0] * h
    
    return g

    return (best_h.tolist(), float(best_c5), n_points)
