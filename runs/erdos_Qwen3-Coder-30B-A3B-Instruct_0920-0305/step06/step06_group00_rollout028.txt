import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h, learning_rate=0.01):
    """Perform a gradient descent step with projection"""
    # Compute gradient approximation
    # For each position, we estimate how moving mass affects max correlation
    corr = correlate(h, 1 - h, mode='full')
    max_lag = np.argmax(corr)
    
    grad = np.zeros_like(h)
    
    # Approximate gradient based on where the maximum correlation occurs
    for i in range(n_points):
        # Contribution to max correlation at lag = max_lag
        j = (i + max_lag) % n_points
        grad[i] += (1 - h[j])
        
        # Also consider negative contribution
        j = (i - max_lag) % n_points
        grad[i] += (1 - h[j])
    
    # Apply gradient step
    h_new = h - learning_rate * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

        h_candidate = h.copy()

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            # Sometimes accept worse moves to escape local optima
            if np.random.random() < 0.05:
                h = h_candidate.copy()
    
    return best_h, best_c5
