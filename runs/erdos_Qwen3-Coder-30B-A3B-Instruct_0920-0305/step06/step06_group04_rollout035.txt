import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # For accurate FFT-based correlation
N_TARGET = 120     # Final discretization size
K = 12             # Number of harmonics (4K coefficients)
MAX_COEF = 0.5     # Bound on the amplitude of harmonics

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50

    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # fallback
    y = np.clip(x + low, 0, 1)
    return y

def fourier_to_h(theta):
    """Convert Fourier coefficients to h vector"""
    # Ensure the correct length
    assert len(theta) == 2 * K
    a = theta[:K]
    b = theta[K:]

    # Create the trigonometric series
    x = np.linspace(0, 2, N_FINE)
    h_raw = 0.5 + np.sum([a[k] * np.cos(np.pi * (k+1) * x) + b[k] * np.sin(np.pi * (k+1) * x)
                          for k in range(K)], axis=0)

    # Clip
    h_clipped = np.clip(h_raw, 0, 1)

    # Project to exact sum
    h_projected = project_to_sum_box(h_clipped, TARGET_SUM)

    # Downsample to target resolution
    h_final = h_projected[::N_FINE // N_TARGET]
    return h_final

def evaluate_c5(h):
    """Fast evaluation of C5 using full correlation with FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(ifft(fft(h_rev, 2*len(h)-1) * fft(b, 2*len(h)-1)))
    return dx * np.max(corr)

def fitness(theta):
    """Fitness function for DE to minimize C5"""
    try:
        h = fourier_to_h(theta)
        c5 = evaluate_c5(h)
        return c5
    except Exception:
        return 1e10  # penalize invalid solutions

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Hyperparameters
    bounds = [(-MAX_COEF, MAX_COEF)] * 2 * K  # Options for a_k and b_k

    # Phase 1: Global search with DE
    DE_evals = 0
    best_result = None
    best_c5 = 1e10

    # Try multiple strategies
    strategy_types = ['best1bin', 'best2bin', 'currenttobest1bin']

    for strat in strategy_types:
        if (time.time() - start_time) > 0.7 * budget_s:
            break

        try:
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(fourier_to_h(x)) - TARGET_SUM}
        except:
            continue

        res = differential_evolution(
            fitness, bounds,
            strategy=strat,
            maxiter=150,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            updating='immediate',
            disp=False,
            callback=lambda x, convergence: None,
            workers=1
        )

        # Check if result is better
        try:
            h_candidate = fourier_to_h(res.x)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_result = res.x.copy()
        except:
            continue

        DE_evals += res.nfev
        if (time.time() - start_time) > 0.7 * budget_s:
            break

    # Phase 2: Refinement with focused DEs
    if best_result is not None and (time.time() - start_time) < 0.9 * budget_s:
        # Refine the top candidates
        candidates = [best_result]
        for j in range(3):  # 3 refined runs
            if (time.time() - start_time) > 0.9 * budget_s:
                break
            try:
                res = differential_evolution(
                    fitness, bounds,
                    strategy='best1bin',
                    maxiter=100,
                    popsize=30,
                    mutation=(0.5, 0.8),
                    recombination=0.8,
                    seed=seed + j,
                    updating='immediate',
                    disp=False,
                    workers=1
                )
                try:
                    h_candidate = fourier_to_h(res.x)
                    c5_candidate = evaluate_c5(h_candidate)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_result = res.x.copy()
                except:
                    pass
            except:
                pass

    # Final polishing with local refinement
    def gradient_refine():
        nonlocal best_result, best_c5
        h_original = fourier_to_h(best_result)
        h_current = h_original.copy()
        c5_current = evaluate_c5(h_current)
        
        # Projected gradient ascent variation
        alpha = 1.0
        for _ in range(300):
            if (time.time() - start_time) > budget_s - 5:
                break
            
            dx = 2.0 / len(h_current)
            # Compute full correlation f_l
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            corr = corr * dx
            min_lag_idx = np.argmin(corr)
            
            # Calculate gradient
            grad = np.zeros_like(h_current)
            for i in range(len(h_current)):
                if i + min_lag_idx < len(h_current):
                    grad[i] -= h_current[i + min_lag_idx]
                if i - min_lag_idx >= 0:
                    grad[i] -= h_current[i - min_lag_idx]

            # Perform update
            h_new = h_current - 0.01 * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)

            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            else:
                break
                
        return h_current, c5_current

    # Run final polish if we have a candidate
    if best_result is not None:
        try:
            final_h, final_c5 = gradient_refine()
            if final_c5 < best_c5:
                best_c5 = final_c5
        except:
            pass  # fallback to previous

    # Binary swap optimization
    if best_result is not None:
        try:
            h_final = fourier_to_h(best_result)
            h_binary = np.zeros_like(h_final)
            # Select top values to 1
            indices = np.argsort(-h_final)
            for i in range(N_TARGET // 2):
                h_binary[indices[i]] = 1.0
            # Then project to sum
            h_binary = project_to_sum_box(h_binary, TARGET_SUM)
            
            # Try swaps to improve
            max_swaps = 2 * N_TARGET
            swaps = 0
            while swaps < max_swaps:
                if (time.time() - start_time) > budget_s - 10:
                    break
                changed = False
                # Find worst lag
                corr = np.correlate(h_binary, 1 - h_binary, mode='full')
                min_lag_idx = np.argmin(corr)
                min_val = corr[min_lag_idx]
                
                # Look for optimal swaps
                for i in range(N_TARGET):
                    # Find a 1 with low contribution to worst lag
                    if h_binary[i] == 1:
                        h_test = h_binary.copy()
                        h_test[i] = 0
                        # Find best 0 for swap
                        for j in range(N_TARGET):
                            if h_test[j] == 0:
                                h_test[j] = 1
                                # Evaluate new correlation
                                new_corr = np.correlate(h_test, 1 - h_test, mode='full')
                                new_min = np.min(new_corr)
                                if new_min > min_val:
                                    h_binary = h_test.copy()
                                    changed = True
                                    swaps += 1
                                    break
                                h_test[j] = 0
                        if changed:
                            break
                if not changed:
                    break
            final_c5 = evaluate_c5(h_binary)
            if final_c5 < best_c5:
                best_c5 = final_c5
                h_final = h_binary
        except:
            pass  # fallback to previous

    final_h_values = fourier_to_h(best_result).tolist() if best_result is not None else np.random.uniform(0, 1, N_TARGET).tolist()

    # Final validation
    try:
        final_h_values = [max(0, min(1, x)) for x in final_h_values]  # ensure [0,1]
        final_sum = sum(final_h_values)
        final_h_values = [x * (TARGET_SUM / final_sum) for x in final_h_values]
        final_h_values = [max(0, min(1, x)) for x in final_h_values]  # ensure [0,1]
        final_c5 = evaluate_c5(final_h_values)
    except:
        final_c5 = 1e5

    return (final_h_values, float(final_c5), N_TARGET)
