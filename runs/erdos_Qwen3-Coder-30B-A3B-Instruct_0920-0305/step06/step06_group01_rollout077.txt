import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1
MAX_PHASE_RESETS = 5

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_initial_h():

    global TARGET_SUM
    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interpolated = np.interp(x_new, x_orig, h0)
    h_interpolated = project_to_sum_box(h_interpolated, n_points / 2)
    TARGET_SUM = n_points / 2
    return h_interpolated

def compute_flat_magnitude_spectrum(H):
    """Compute magnitude target for flat off-DC spectrum"""
    P = np.abs(H)**2
    off_dc_sum = np.sum(P[1:])
    if off_dc_sum == 0:
        return np.abs(H)
    target_magnitude = np.sqrt(off_dc_sum / (L - 1))
    M_target = np.full_like(H, target_magnitude)
    M_target[0] = np.abs(H[0])
    return M_target

def alternating_projection(h, max_iter=10000, 
                          initial_lambda=0.05, lambda_factor=1.02,
                          beta_start=0.5, beta_factor=1.01,
                          phase_reset_interval=300):
    """Main algorithm using alternating projection in spectral domain"""
    lambda_k = initial_lambda
    beta = beta_start
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    phase_resets = 0
    no_improvement_count = 0
    
    for i in range(max_iter):
        if i > 0 and i % phase_reset_interval == 0 and phase_resets < MAX_PHASE_RESETS:
            # Phase reset
            H = np.fft.fft(np.pad(h, (0, L - n_points), 'constant'))
            M_target = compute_flat_magnitude_spectrum(H)
            # Use random phases instead of preserving old ones
            random_phases = np.exp(1j * np.random.uniform(0, 2*np.pi, L))
            H_new = M_target * random_phases
            h_cand = np.real(np.fft.ifft(H_new))[:n_points]
            phase_resets += 1
            no_improvement_count = 0
        else:
            # Standard alternating projection step
            H = np.fft.fft(np.pad(h, (0, L - n_points), 'constant'))
            
            # Compute target magnitude
            M_target = compute_flat_magnitude_spectrum(H)
            
            # Blend with current magnitude
            M = (1 - lambda_k) * np.abs(H) + lambda_k * M_target
            
            # Preserve phase, form new spectrum
            H_new = M * np.exp(1j * np.angle(H))
            
            # Inverse FFT
            h_cand = np.real(np.fft.ifft(H_new))[:n_points]
            
            # Project to feasibility
            h_cand = project_to_sum_box(h_cand, TARGET_SUM)
            
            # Relaxation to avoid oscillations
            h_cand = (1 - beta) * h + beta * h_cand
            
            # Update parameters
            lambda_k = min(lambda_k * lambda_factor, 1.0)
            beta = min(beta * beta_factor, 1.0)
            
            # Count no-improvement steps
            c5_new = evaluate_c5(h_cand)
            if c5_new < best_c5:
                best_h = h_cand.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Early termination if no significant improvement
            if no_improvement_count > 500 and lambda_k >= 0.95:
                break
            
        h = h_cand
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Build initial solution
    h_start = build_initial_h()
    
    # Run main alternating projection algorithm
    h_final, c5_final = alternating_projection(
        h_start,
        max_iter=int(1e5),
        initial_lambda=0.05,
        lambda_factor=1.02,
        beta_start=0.5,
        beta_factor=1.01,
        phase_reset_interval=300
    )
    
    # Final polishing with local optimization
    def objective(h):
        return evaluate_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    from scipy.optimize import minimize
    try:
        res = minimize(
            objective, 
            h_final, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint_eq},
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_final = res.x
            h_final = project_to_sum_box(h_final, TARGET_SUM)
            c5_final = evaluate_c5(h_final)
    except:
        pass
    
    # Ensure final evaluation
    c5_final = evaluate_c5(h_final)
    
    return (h_final.tolist(), float(c5_final), n_points)
