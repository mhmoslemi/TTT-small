import numpy as np
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    N = 120
    dx = 2.0 / N
    target_sum = N / 2.0

    # Initialize problem model
    h = Variable(N, boolean=True)
    y = {}
    for ell in range(N):
        for i in range(N - ell):
            y[(i, ell)] = Variable(boolean=True)
    lam = Variable(integer=True, lower_bound=0)

    # Constraint: sum of h equals target sum
    constraints = [sum(h) == target_sum]

    # Product constraints for all (i, ell)
    for ell in range(N):
        for i in range(N - ell):
            # y[(i, ell)] <= h[i]
            constraints.append(y[(i, ell)] <= h[i])
            # y[(i, ell)] <= h[i + ell]
            constraints.append(y[(i, ell)] <= h[i + ell])
            # y[(i, ell)] >= h[i] + h[i + ell] - 1
            constraints.append(y[(i, ell)] >= h[i] + h[i + ell] - 1)

    # Lag-overlap constraints
    for ell in range(N):
        constraints.append(sum(y[(i, ell)] for i in range(N - ell)) >= lam)

    # Objective function
    objective = Maximize(lam)

    # Build the model
    prob = Problem(objective, constraints)

    # Solve under time limit
    remaining_time = 0.9 * budget_s
    best_c5 = float('inf')
    best_h = None

    # Solve with time limit using solver's built-in timeout option
    solvers.options['MAXIT'] = 100000  # Set large iteration limit
    solvers.options['TIME_LIMIT'] = remaining_time - 10  # Keep 10s for safe exit

    prob.solve(solver=ECOS_BB, verbose=False)
    
    if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        # Extract solution
        h_val = np.array([round(h[i].value) for i in range(N)])
        # Reconstruct the correlation from the solution
        corr = np.zeros(2*N - 1)
        for (i, ell), y_var in y.items():
            if y_var.value is not None:
                corr[ell] += y_var.value
        # To compute C5 we maximize over all lags the overlap integral
        # C5 = 1 - dx * lambda, where lambda was achieved
        lambda_val = lam.value
        if lambda_val is not None:
            c5 = 1 - dx * lambda_val
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_val.tolist()
    else:
        # Fallback to grid search on block functions
        best_c5 = float('inf')
        best_h = None

        def compute_corr(h_vec):
            # Compute the correlation with full mode using numpy
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return corr * dx

        # Small number of blocks e.g., 4 to reduce dimensionality
        num_blocks = 6
        # Generate all partitions of N/2 into num_blocks parts
        def generate_partitions(remaining, num_parts, min_val=0, max_val=None):
            if num_parts == 1:
                if remaining >= min_val and (max_val is None or remaining <= max_val):
                    yield [remaining]
                return
            if max_val is None:
                max_val = remaining

            for i in range(min_val, min(max_val + 1, remaining + 1)):
                for p in generate_partitions(remaining - i, num_parts - 1, i, max_val):
                    yield [i] + p

        # Enumerate all valid block configurations and find their overlaps
        min_block_len = 1
        # Try all combinations
        for partition in generate_partitions(N//2, num_blocks):
            # Also ... ensure that constraints like total width = N
            # We do: partition represents how many h[i] = 1 in each block
            if sum(partition) == N//2:  # total mass = 1
                # Compute block widths: total width = N, no. blocks = num_blocks
                block_widths = np.floor(N * np.array(partition) / (N//2)).astype(int)
                if sum(block_widths) != N:  # Adjust to reach exact N
                    diff = N - sum(block_widths)
                    # Adjust one block to balance
                    block_widths[-1] += diff
                # Now create step function
                h_cand = np.zeros(N)
                begin = 0
                for i in range(num_blocks):
                    # ensure block is non-empty
                    w = block_widths[i] if block_widths[i] > 0 else 1
                    end = min(N, begin + w)
                    h_cand[begin:end] = partition[i] / w   # normalize to make it a valid height
                    begin = end
                # Project to ensure sum = N//2
                h_cand = np.clip(h_cand, 0, 1)
                # Force exact sum constraint
                current_sum = np.sum(h_cand)
                if current_sum > 0:
                    h_cand *= (N//2) / current_sum
                # Now evaluate C5
                corr = compute_corr(h_cand)
                c5 = np.max(corr)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cand.tolist()

    return (best_h, best_c5, N)
