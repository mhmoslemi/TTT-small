# Helper function for computing correlations
def compute_corr_fft(h):
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n-1) * np.fft.fft(b, 2*n-1)))
    return corr

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def binary_milp_approach(n_points=120, solver="CBC", time_limit=500):
    """Solve the MILP using cvxpy with integer constraints"""
    # Create variables
    h = Variable(n_points, boolean=True)
    y = {}
    for i in range(n_points):
        for l in range(i+1, n_points):
            if l < n_points:
                y[(i,l)] = Variable(boolean=True)
    
    # Constraint for total mass
    constraints = [cp.sum(h) == n_points / 2]

    # Product constraints
    for i in range(n_points):
        for l in range(i+1, n_points):
            if l < n_points:
                constraints += [
                    y[(i,l)] <= h[i],
                    y[(i,l)] <= h[l],
                    y[(i,l)] >= h[i] + h[l] - 1
                ]

    # Min overlap constraints
    lambda_var = Variable(integer=True, name="lambda")

    # Add constraints for min overlap
    for l in range(1, n_points):
        # For lag l, sum of y[i,i+l] >= lambda
        sum_y = cp.sum([y[(i, i+l)] for i in range(n_points-l)])
        constraints += [sum_y >= lambda_var]

    # Objective
    objective = Maximize(lambda_var)

    # Problem
    prob = Problem(objective, constraints)

    # Solve
    prob.solve(solver=solver, time_limit=time_limit, verbose=False)

    if prob.status == "optimal":
        final_h = np.array(h.value, dtype=float)
        return final_h, 1 - 2 * lambda_var.value / n_points
    else:
        return None, None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Try to solve using MILP approach
    best_c5 = float('inf')
    best_h = None

    # Process only for a portion of the budget
    remaining_time = budget_s * 0.9  # Use 90% for MILP
    try:
        h, c5 = binary_milp_approach(n_points=120, solver="CBC", time_limit=remaining_time)
        if h is not None and c5 is not None and c5 < best_c5:
            best_h = h
            best_c5 = c5
    except Exception as e:
        pass

    # If MILP didn't find a solution or didn't finish in time, fallback to combinatorial search
    if best_h is None:
        # Start with initial recommended values
        h_start = np.array(initial_h_values)
        # Interpolate to 120 points
        x_orig = np.linspace(0, 2, len(h_start))
        x_new = np.linspace(0, 2, 120)
        h_interp = np.interp(x_new, x_orig, h_start)
        # Ensure sum constraint
        target_sum = 60.0
        if np.sum(h_interp) != target_sum:
            h_interp = h_interp * target_sum / np.sum(h_interp)
        best_h = h_interp
        best_c5 = evaluate_c5(h_interp)
    
    # Phase 2: Local improvement
    refine_time = budget_s * 0.1  # 10% for refinement
    refine_start = time.time()
    while (time.time() - refine_start) < refine_time and best_c5 > 0.379:
        # Make small perturbations
        h_perturbed = best_h.copy()
        # Keep some portion fixed (e.g., 90%)
        fixed = int(0.9 * len(h_perturbed))
        indices_to_change = np.random.choice(len(h_perturbed), len(h_perturbed) - fixed, replace=False)
        for idx in indices_to_change:
            delta = np.random.uniform(-0.02, 0.02)
            h_perturbed[idx] += delta
            h_perturbed[idx] = max(0, min(1, h_perturbed[idx]))
        h_perturbed = h_perturbed * target_sum / np.sum(h_perturbed)
        c5 = evaluate_c5(h_perturbed)
        if c5 < best_c5:
            best_h = h_perturbed
            best_c5 = c5
            
    return (best_h.tolist(), float(best_c5), 120)
