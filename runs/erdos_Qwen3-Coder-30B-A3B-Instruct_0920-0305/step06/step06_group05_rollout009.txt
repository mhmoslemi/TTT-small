import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from numpy.linalg import eigh
from scipy.fft import fft, ifft

# Global variables
n_points = 100
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n):
    """Solve SDP relaxation for given n"""
    h = cp.Variable(n, name="h")
    X = cp.Variable((n, n), symmetric=True, name="X")
    t = cp.Variable(1, name="t")
    
    # Constraints
    constraints = [
        cp.sum(h) == TARGET_SUM,
        0 <= h,
        h <= 1,
        cp.diag(X) == h,
        X >> 0]  # X is positive semidefinite
    
    # Lag constraints
    for k in range(n):
        Pk = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                Pk[i, i + k] = 1
        constraints.append(t >= TARGET_SUM - cp.trace(Pk @ X))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, eps=1e-4)
    
    # Return the best feasible point we obtained
    h_val = h.value if h.value is not None else np.full(n, 0.5)
    X_val = X.value if X.value is not None else np.eye(n)
    t_val = t.value if t.value is not None else np.inf
    return h_val, X_val, t_val

def round_sdp_solution(h_rel, X_rel, n, M, num_rounds=20):
    """Round the SDP solution to binary vectors"""
    candidates = []
    
    # Deterministic threshold rounding
    h_thresh = np.zeros(n)
    sorted_indices = np.argsort(-h_rel)  # indices of decreasing values
    h_thresh[sorted_indices[:int(M)]] = 1.0
    if np.sum(h_thresh) < M:
        # set necessary zeros to one
        zeros = np.where(h_thresh == 0)[0]
        if len(zeros) > 0:
            fill_indices = zeros[:int(M - np.sum(h_thresh))]
            h_thresh[fill_indices] = 1.0
    elif np.sum(h_thresh) > M:
        # set necessary ones to zero
        ones = np.where(h_thresh == 1)[0]
        if len(ones) > 0:
            zero_indices = ones[int(M):]
            h_thresh[zero_indices] = 0.0
    candidates.append(h_thresh)
    
    # Random hyperplane rounding with num_rounds tries
    # Perform eigh on X_rel
    try:
        eigenvals, eigenvecs = eigh(X_rel)
        # Keep only positive eigenvals and corresponding eigenvectors
        V_plus = eigenvecs[:, eigenvals > 1e-6]
        r = V_plus.shape[1]
        if r == 0:
            # If all eigenvalues are close to zero
            h_binary = np.zeros(n)
            h_binary[:int(M)] = 1.0
            candidates.append(h_binary)
        else:
            for _ in range(num_rounds):
                g = np.random.randn(r)
                s = V_plus @ g
                h_binary = np.zeros(n)
                h_binary[s >= 0] = 1
                
                # repair
                while np.sum(h_binary) > M:
                    # flip a random 1 to 0
                    flipped_idx = np.random.choice(np.where(h_binary == 1)[0])
                    h_binary[flipped_idx] = 0
                while np.sum(h_binary) < M:
                    # flip a random 0 to 1
                    flipped_idx = np.random.choice(np.where(h_binary == 0)[0])
                    h_binary[flipped_idx] = 1
                candidates.append(h_binary)
    except:
        # fallback to deterministic rounding
        pass
    
    return candidates

def swap_hill_climb(h_bin):
    """Perform swap hill climbing to improve a binary solution"""
    h = h_bin.copy()
    improved = True
    max_swaps = 5000
    swaps = 0
    while improved and swaps < max_swaps:
        improved = False
        # Compute current correlation for optimization
        corr = compute_corr_fft(h)
        curr_max_lag = np.argmax(corr)
        max_val = corr[curr_max_lag]
        # Iterate over all (i in ones, j in zeros)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                if swaps >= max_swaps:
                    break
                # compute change in max corr
                old_val = corr[curr_max_lag]
                delta = 0
                # Consider which lags are affected by swapping i and j
                for k in range(n_points):
                    if i+k < n_points and k != curr_max_lag:
                        if h[k] == 1:
                            delta += h[i]  # i was added to a 1-term
                        if h[i+k] == 1:
                            delta -= h[i]  # i was removed from a 1-term
                    if j+k < n_points and k != curr_max_lag:
                        if h[k] == 1:
                            delta -= h[j]  # j was removed from a 1-term
                        if h[j+k] == 1:
                            delta += h[j]  # j was added to a 1-term
                if i == 0 or j == 0:
                    pass  # skip if one index is 0 due to boundary
                # Actually perform the swap and check proceeds
                h[i] = 0
                h[j] = 1
                new_corr = compute_corr_fft(h)
                if new_corr[curr_max_lag] < max_val:
                    max_val = new_corr[curr_max_lag]
                    improved = True
                    break
                h[i] = 1
                h[j] = 0
                swaps += 1
            if improved:
                break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    scales = [60, 80, 100]
    
    # Phase A: SDP solving and rounding
    budget_per_n = 0.5 * budget_s / len(scales)
    for n in scales:
        if time.time() - start_time > budget_s - 10:
            break
        n_points = n
        dx = 2.0 / n
        TARGET_SUM = n / 2
        try:
            h_rel, X_rel, bound = solve_sdp(n)
            if bound == float('inf'):
                continue
            candidates = round_sdp_solution(h_rel, X_rel, n, TARGET_SUM, num_rounds=20)
            for h in candidates:
                h = project_to_sum_box(h, TARGET_SUM)
                h = swap_hill_climb(h)
                c5_score = evaluate_c5(h)
                if c5_score < best_c5:
                    best_c5 = c5_score
                    best_h = h.copy()
        except:
            continue
        # If finish early we can break
        if time.time() - start_time > 0.8 * budget_s:
            break
        # Allow tech time
        if time.time() - start_time > budget_per_n:
            break
        
    # Phase B: Continuous polishing of best solution
    if best_h is not None:
        # SLSQP polishing
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Use project_to_sum_box to get feasible start
        h_start = project_to_sum_box(best_h, TARGET_SUM)
        try:
            res = minimize(
                objective, 
                h_start, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h = res.x
                h = project_to_sum_box(h, TARGET_SUM)
                c5_after_slsqp = evaluate_c5(h)
                if c5_after_slsqp < best_c5:
                    best_c5 = c5_after_slsqp
                    best_h = h.copy()
        except:
            pass
    
    # Fallback to simple construction if nothing worked
    if best_h is None:
        # Simple alternating block pattern of length 2
        block_len = 2
        h_array = np.ones(n_points)
        for i in range(0, n_points, 2*block_len):
            if i + block_len > n_points:
                h_array[i:] = 0
                break
            h_array[i:i + block_len] = 1
            h_array[i + block_len:i + 2*block_len] = 0
        h_array = project_to_sum_box(h_array, TARGET_SUM)
        best_c5 = evaluate_c5(h_array)
        best_h = h_array
    
    return (best_h.tolist(), float(best_c5), n_points)
