import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def solve_milp(n, time_limit):
    """Solve the MILP formulation for binary variables."""
    M = n // 2  # Required number of ones
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(pos=True)
    
    # Auxiliary variables for logical AND: y[i,j] = h[i] AND h[j]
    # We only create those for i < j, and for valid lags
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(boolean=True)
    
    # Constraints
    constraints = []
    
    # Exact mass constraint
    constraints.append(cp.sum(h) == M)
    
    # Linking AND constraints
    for i in range(n):
        for j in range(i+1, n):
            # y[i,j] <= h[i], y[i,j] <= h[j]
            constraints.append(y[(i,j)] <= h[i])
            constraints.append(y[(i,j)] <= h[j])
            # y[i,j] >= h[i] + h[j] - 1
            constraints.append(y[(i,j)] >= h[i] + h[j] - 1)
    
    # Lag constraints
    for k in range(1, n):
        # t >= M - sum of y_{i,i+k}
        lag_sum = cp.sum([y[(i, i+k)] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Symmetry breaking - fix first element to 1
    constraints.append(h[0] == 1)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.rint(h.value).astype(int)
            t_val = t.value
            return h_sol, t_val
        else:
            # Try to get a feasible solution
            h_sol = np.rint(h.value).astype(int) if h.value is not None else None
            if h_sol is not None:
                # Compute t from constraints
                t_val = 0
                for k in range(1, n):
                    lsum = sum(h_sol[i] * h_sol[i+k] for i in range(n-k))
                    t_val = max(t_val, M - lsum)
                return h_sol, t_val
            return None, None
    except:
        return None, None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Perform continuous local refinement using SLSQP."""
    bounds = [(0, 1) for _ in range(len(h))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - len(h) / 2
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, len(h) / 2)
            return h
        else:
            return h
    except:
        return h
