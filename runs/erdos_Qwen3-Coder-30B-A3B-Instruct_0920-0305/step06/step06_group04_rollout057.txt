import numpy as np
import time

def solve_milp_erdos(n_points, max_time_s):
    """Solve the Erdős problem as a binary MILP."""
    # Create the variables
    h = cp.Variable(n_points, boolean=True)
    y = cp.Variable((n_points, n_points), boolean=True)
    lambda_var = cp.Variable(integer=True)
    
    # Objective: maximize lambda
    objective = cp.Maximize(lambda_var)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Product constraints: y[i,l] == h[i] * h[i+l]
    for l in range(n_points):
        for i in range(n_points - l):
            constraints.append(y[i, l] <= h[i])
            constraints.append(y[i, l] <= h[i + l])
            constraints.append(y[i, l] >= h[i] + h[i + l] - 1)
            # Additional check: if l is 0, there is an issue with indexing
            # But [i,l] does not need to be considered for l=0, as that's just h_i*h_i=h_i
    
    # Lag overlap constraints
    for l in range(n_points):
        if l == 0:
            continue  # Skip lag 0 to avoid repeated indices or weakening
        # Potential bug: sum over valid i's for given l
        # i: 0..N-1-l 
        constraints.append(cp.sum([y[i, l] for i in range(n_points - l)]) >= lambda_var)
    
    # Upper bound on lambda
    constraints.append(lambda_var <= n_points / 2)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try using different solvers
    solver_options = {
        'time_limit': max_time_s,
        'mip_tolerances_mip_gap': 1e-4,
        'mip_tolerances_absmipgap': 1e-4
    }
    
    try:
        prob.solve(solver=cp.CBC, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            lambda_val = lambda_var.value
            return h_sol, lambda_val
        else:
            return None, None
    except Exception:
        return None, None

def evaluate_c5(h):
    """Compute C5 as 1 - (2/N)*lambda from the MILP."""
    n = len(h)
    dx = 2.0 / n
    # Using numpy's correlate to get a full correlation
    # This is actually the correct evaluation, but from the same correlation it derived the lambda
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum=target_sum under box constraints."""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y
