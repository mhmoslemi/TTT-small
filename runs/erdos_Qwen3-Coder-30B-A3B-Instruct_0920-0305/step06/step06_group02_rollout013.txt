def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)

def simulated_annealing(h0, T0=1.0, cooling_rate=0.999, max_iter=1000):
    """Simulated annealing optimization"""
    h_current = h0.copy()
    c5_current = evaluate_c5(h_current)
    h_best = h_current.copy()
    c5_best = c5_current
    
    T = T0
    
    for i in range(max_iter):
        # Generate neighbor solution
        h_neighbor = h_current.copy()
        # Randomly modify one element
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_neighbor[idx] += delta
        
        # Project to feasible region
        h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
        
        # Evaluate neighbor
        c5_neighbor = evaluate_c5(h_neighbor)
        
        # Accept or reject
        if c5_neighbor < c5_current or np.random.random() < np.exp((c5_current - c5_neighbor) / T):
            h_current = h_neighbor
            c5_current = c5_neighbor
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        
        # Cool down
        T *= cooling_rate
        
    return h_best, c5_best

    h = h0.copy()
    best_c5 = evaluate_c5(h)

        for idx in range(n_points):

                    continue

                    h = h_new
                    best_c5 = c5_new
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5
