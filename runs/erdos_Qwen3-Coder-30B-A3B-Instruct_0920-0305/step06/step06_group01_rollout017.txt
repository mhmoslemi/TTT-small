import numpy as np
from scipy.signal import correlate
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    iteration = 0

        h_padded = np.pad(h, (0, n_points - 1), 'constant')
        H = np.fft.fft(h_padded)
        P = np.abs(H)**2
        
        # Compute target flat spectrum (off-DC part)
        P_target_off_dc = (np.sum(P[1:]) / (L - 1)) if L > 1 else 1.0
        P_target = np.full_like(P, np.sqrt(P_target_off_dc))
        P_target[0] = np.abs(H[0])  # Keep DC
        
        # Mix
        M = (1 - lambda_current) * np.abs(H) + lambda_current * P_target
        
        # Keep phase and IFFT
        H_new = M * np.exp(1j * np.angle(H))
        h_new = np.fft.ifft(H_new).real[:n_points]

        h_projected = project_to_sum_box(h_new, target_sum)
        
        # Relaxation
        h_updated = (1 - beta) * h + beta * h_projected
        h = h_updated

        corr = compute_corr_fft(h)

            best_c5 = c5_value
            last_best = best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Annealing
        if lambda_current < lambda_max:
            lambda_current = min(lambda_current * lambda_rate, lambda_max)
        
        # Relaxation update
        if beta < 1.0:

        return np.sum(h) - target_sum

            final_C5 = c5_from_corr(corr_final, n_points)
            if final_C5 < best_c5:
                best_h = final_h
                best_c5 = final_C5
    except:
        pass

    return (best_h.tolist(), float(best_c5), n_points)
