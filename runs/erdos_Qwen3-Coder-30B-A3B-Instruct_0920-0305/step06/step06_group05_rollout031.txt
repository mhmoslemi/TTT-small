# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move using direction of largest correlation"""
    h = h.copy()
    
    for _ in range(max_iterations):
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr)
        
        # Get a better estimate of gradient direction
        # For each position, compute how moving it affects the worst correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Use a smarter gradient update
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
        
        # Move amount inversely proportional to gradient magnitude
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta <= 0:
            break
        
        h[idx_min] -= delta
        h[idx_max] += delta
        
        # Project back to feasible set
        h = project_to_sum_box(h, TARGET_SUM)
        
        if evaluate_c5(h) < 0.379:
            break
    
    return h
