def project_to_sum_box(h, target_sum, n):
    """Project h to exact sum via water-filling"""
    h = h.copy()
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for optimal shift
    low, high = -1.0, 1.0
    tol = 1e-12
    
    for _ in range(50):
        mid = (low + high) / 2.0
        h_test = np.clip(h + mid, 0, 1)
        sum_test = np.sum(h_test)
        if abs(sum_test - target_sum) < tol:
            return h_test
        elif sum_test > target_sum:
            high = mid
        else:
            low = mid
    
    h_test = np.clip(h + low, 0, 1)
    return h_test

def c5_from_corr_fft(h):

    n = len(h)

    b = 1.0 - h
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    dx = 2.0 / n
    return dx * np.max(corr)

        return c5_from_corr_fft(h)

        return 1e10  # Invalid evaluation
