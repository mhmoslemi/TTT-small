import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def trust_region_optimize(h0, max_iter=1000):
    """Optimize using trust region method"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='trust-constr', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def simulated_annealing(h0, max_iter=1000):
    """Simulated annealing optimization"""
    h = h0.copy()
    current_energy = evaluate_c5(h)
    best_h = h.copy()
    best_energy = current_energy
    temperature = 0.1
    
    for i in range(max_iter):
        # Generate neighbor
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.02)
        h_new[idx] += delta
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_energy = evaluate_c5(h_new)
        
        # Accept or reject
        if new_energy < current_energy or np.random.random() < np.exp(-(new_energy - current_energy) / temperature):
            h = h_new
            current_energy = new_energy
        
        if current_energy < best_energy:
            best_h = h.copy()
            best_energy = current_energy
            
        # Cool down
        temperature *= 0.999
        
    return best_h

def genetic_algorithm(max_generations=200, population_size=30):
    """Genetic algorithm for optimization"""
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for generation in range(max_generations):
        # Evaluate fitness
        fitnesses = [evaluate_c5(h) for h in population]
        sorted_indices = np.argsort(fitnesses)
        
        # Keep best
        new_population = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Generate new population
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(sorted_indices[:population_size//2], 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.random() < 0.1:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] += np.random.normal(0, 0.02)
                child = project_to_sum_box(child, TARGET_SUM)
            
            new_population.append(child)
        
        population = new_population
    
    # Return best from final population
    fitnesses = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitnesses)
    return population[best_idx]

def refine_solution(h0, max_iter=500):
    """Refine solution using multiple strategies"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Try different optimization strategies
    strategies = [
        lambda h: trust_region_optimize(h, max_iter=200),
        lambda h: simulated_annealing(h, max_iter=300),
    ]
    
    for strategy in strategies:
        h_test = strategy(h.copy())
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            h = h_test
            best_c5 = c5_test
    
    return h
