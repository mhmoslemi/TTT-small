import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def full_corr(a, b):
    """Compute full correlation using FFT, zero-padded"""
    n = a.size
    N = 2*n                       # zero-padded size
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]              # lag range -(n-1)…(n-1)

def soft_max_gradient(h, alpha):
    """Compute gradient of soft max of correlation values"""
    n = len(h)
    
    # Compute full correlation (correlation with 1-h)
    corr = full_corr(h, 1 - h)
    max_corr = corr.max()
    
    # Softmax weights
    w = np.exp(alpha * (corr - max_corr))
    w_sum = w.sum()
    w = w / w_sum
    
    # Compute convolutions for weighted gradient
    # Convolution for the forward shift h[i+lag]: w * h
    conv_fwd = full_corr(h, w)
    # Convolution for the backward shift h[i-lag]: w_rev * h
    conv_bwd = full_corr(h, w[::-1])
    
    # Full gradient
    grad = 1.0 - conv_fwd - conv_bwd
    
    return grad

def line_search(h, grad, alpha, target_sum, step_size=0.1):
    """Backtracking line search"""
    while step_size > 1e-8:
        h_new = project_to_sum_box(h - step_size * grad, target_sum)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old - 1e-12 or step_size < 1e-6:
            break
        step_size *= 0.5
    return h_new, c5_new, step_size

def continuous_optimize(initial_h, time_budget, n_points, dx, target_sum):
    """Optimize continuous solution"""
    start = time.time()
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    h = initial_h.copy()
    alpha = 5.0  # start low for better exploration
    step_size = 0.05
    
    while time.time() - start < time_budget:
        # Compute gradient
        grad = soft_max_gradient(h, alpha)
        
        # Perform line search
        h_new, c5_new, step_size = line_search(h, grad, alpha, target_sum, step_size)
        
        # Update
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Adapt step and alpha
        if step_size < 1e-6:
            # Perturb if stuck
            h += np.random.normal(0, 0.001, n_points)
            h = project_to_sum_box(h, target_sum)
        alpha = min(20.0, alpha + 0.02)  # Increase alpha over time
        if time.time() - start > time_budget * 0.95:
            break
    
    return best_h, best_c5

def binary_refinement(initial_h, time_budget, n_points, dx):
    """Binary refinement using simulated annealing"""
    start = time.time()
    
    # Select top half values to 1, rest 0
    idx_sorted = np.argsort(initial_h)[::-1]
    binary_h = np.zeros(n_points)
    binary_h[idx_sorted[:n_points//2]] = 1.0
    binary_h = project_to_sum_box(binary_h, n_points / 2)
    
    best_h = binary_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Cooling schedule parameters
    temp_initial = 0.05
    temp_decay = 0.999
    min_temp = 1e-6
    
    temp = temp_initial
    
    while time.time() - start < time_budget:
        # Pick random 1 and 0 indices
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Create candidate by swapping
        candidate_h = best_h.copy()
        candidate_h[i] = 0.0
        candidate_h[j] = 1.0
        
        # Evaluate candidate
        c5_candidate = evaluate_c5(candidate_h)
        
        # Accept or reject
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = candidate_h.copy()
        else:
            delta = c5_candidate - best_c5
            prob = np.exp(-delta / temp)
            if np.random.rand() < prob:
                best_c5 = c5_candidate
                best_h = candidate_h.copy()
        
        # Cool down
        temp = max(min_temp, temp * temp_decay)
    
    return best_h, best_c5
