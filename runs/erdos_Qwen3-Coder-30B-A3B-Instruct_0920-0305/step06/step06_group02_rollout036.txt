import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def improved_gradient_optimization(h0, max_iter=1000):
    """Improved gradient-based optimization with better constraints handling"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def local_improvement_search(h0, max_iter=500):
    """Local improvement using neighborhood search with smart perturbations"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Try small changes with varying magnitudes
        perturbations = [
            (0.01, 1), (0.01, -1), 
            (0.005, 1), (0.005, -1),
            (0.02, 1), (0.02, -1)]
        
        best_delta = None
        best_idx = None
        best_improvement = 0
        
        # Consider all indices
        indices = list(range(n_points))
        np.random.shuffle(indices)  #random order to avoid bias
        
        for idx in indices[:20]:  # limit to first 20 for efficiency
            for delta_val, direction in perturbations:
                h_new = h.copy()
                h_new[idx] += delta_val * direction
                
                if h_new[idx] < 0 or h_new[idx] > 1:
                    continue
                
                # Project to maintain sum constraint
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                new_c5 = evaluate_c5(h_new)
                improvement = best_c5 - new_c5
                
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_delta = delta_val * direction
                    best_idx = idx
        
        if best_improvement > 1e-12 and best_idx is not None:
            h[best_idx] += best_delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
        else:
            # Diversification step
            if np.random.random() < 0.3:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

    # Apply small random perturbations
    h += np.random.normal(0, 0.01, n_points)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = synthetic_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Multiple optimizations with different strategies
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(10):
        if time.time() > phase1_end:
            break
            
        # Strategies for better exploration
        strategies = [
            lambda: np.random.uniform(0, 1, n_points),
            lambda: np.random.choice([0, 1], size=n_points, p=[0.6, 0.4]),
            synthetic_initialization
        ]
        h_start = strategies[run_id % len(strategies)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Gradient optimization
        h_refined = improved_gradient_optimization(h_start, max_iter=800)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Local improvement searches on best solutions
    phase2_end = start_time + 0.8 * budget_s
    for run_id in range(5):
        if time.time() > phase2_end:
            break
        
        h_refined = local_improvement_search(best_h.copy(), max_iter=400)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 3: Fine-tuning
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Add small random noise around the global best
        noise = np.random.normal(0, 0.005, n_points)
        h_noise = best_h + noise
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        # Apply a few rounds of gradient-based optimization
        h_final = improved_gradient_optimization(h_noise, max_iter=300)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
