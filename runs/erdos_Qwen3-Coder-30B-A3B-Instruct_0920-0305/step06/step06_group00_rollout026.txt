from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

                    break
                
                if direction * grad[i] <= 0:
                    continue
                    
                # Try to move max amount we can while keeping constraints
                move_max = min(h_trial[i], 1 - np.sum(h_trial) + h_trial[i])
                if move_max <= 0:

                move_amount = min(step_size, move_max)
                if move_amount > 1e-6:

                    target_idx = np.argmax([h_trial[j] for j in range(n_points) if abs(i-j) == max_lag and h_trial[j] < 1])

        # Occasionally recompute gradient for better accuracy

            corr = correlate(h, 1 - h, mode='full')
            max_lag = np.argmax(corr)
        
        # Early stopping heuristic
        if iteration > 100 and np.abs(best_c5 - evaluate_c5(best_h)) < 1e-8:
            break
