def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h, lr=0.01):
    """Perform a gradient descent step"""
    # Compute correlation gradient
    corr = correlate(h, 1 - h, mode='full')
    max_lag = np.argmax(corr)
    
    # Compute gradient for each position (simplified gradient approach)
    grad = np.zeros_like(h)

        if i + max_lag < n_points:
            grad[i] += (1 - h[i + max_lag])
        if i >= max_lag:
            grad[i] += h[i - max_lag]
    
    # Normalize gradient
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 0:
        grad = grad / grad_norm
    
    # Apply update
    h_new = h - lr * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new
