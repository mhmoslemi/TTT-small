import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create initial binary vector with exact weight constraint"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full overlap array using numpy correlate"""
    return np.correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros):
    """Update overlap array in O(w) time for a 1↔0 swap"""
    ov_new = ov.copy()
    # For each 1-position k, update contributions to overlap shifts
    for k in ones:
        if k != i:
            # Remove contribution from i and k
            s1 = k - i
            ov_new[s1 + n_points - 1] -= 1
            # Add contribution from j and k
            s2 = k - j
            ov_new[s2 + n_points - 1] += 1
    
    # Handle i itself
    for k in ones:
        if k != i:
            s = i - k
            ov_new[s + n_points - 1] -= 1
            s = j - k
            ov_new[s + n_points - 1] += 1
    
    # Update contributions involving i and j directly
    s = i - j
    ov_new[s + n_points - 1] += 1
    
    return ov_new

def get_ones_zeros(x):
    """Get sets of indices where x=1 and x=0"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    return ones, zeros

def simulated_annealing(x0, ov0, ones0, zeros0, time_limit):
    """Run simulated annealing with binary swaps"""
    x = x0.copy()
    ov = ov0.copy()
    ones = ones0.copy()
    zeros = zeros0.copy()
    
    cur_min = ov.min()
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = cur_min
    start_time = time.time()
    
    # Temperature schedule
    T0 = 0.5
    lambda_decay = 1.0  # Adjust to vary cooling rate
    
    # Main loop
    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1
        # Temperature
        t = iteration / 10000.0
        T = T0 * np.exp(-lambda_decay * t)
        
        # Select swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform swap
        x_new = x.copy()
        x_new[i] = 0
        x_new[j] = 1
        
        # Incremental overlap update
        ov_new = swap_update(ov, i, j, ones, zeros)
        new_min = ov_new.min()
        
        # Acceptance probability
        delta = new_min - cur_min
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            x = x_new
            ov = ov_new
            cur_min = new_min
            
            # Update sets
            ones = np.setdiff1d(ones, [i])
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.append(zeros, i)
            
            # Update best
            if new_min > best_min:
                best_x = x.copy()
                best_ov = ov.copy()
                best_min = new_min
    
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine solution by fixing the worst shift"""
    # Find worst shift
    worst_shift = np.argmin(ov) - n_points + 1
    min_overlap = ov.min()
    
    # Identify beneficial swaps
    A = np.where((x == 1) & (np.roll(x, -worst_shift) == 0))[0]
    B = np.where((x == 0) & (np.roll(x, -worst_shift) == 1))[0]
    
    # Try all possible swaps between A and B
    for i in A:
        for j in B:
            # Make sure swap is actually beneficial
            new_ov = ov.copy()
            # Simulate update (this is approximate but fast)
            new_ov[worst_shift + n_points - 1] += 1
            if new_ov.min() >= min_overlap:
                # Perform the swap
                x[i], x[j] = x[j], x[i]
                ov = new_ov
                break
        else:
            continue
        break
    
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize best solution
    best_x = None
    best_min = float('inf')
    best_ov = None
    
    # Run multiple restarts of simulated annealing
    restarts = 10
    time_per_restart = budget_s * 0.9 / restarts
    
    for _ in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        x = initial_binary()
        ones, zeros = get_ones_zeros(x)
        ov = full_overlap(x)
        cur_min = ov.min()
        
        # Run SA
        x, ov, cur_min = simulated_annealing(x, ov, ones, zeros, time_per_restart)
        
        # Refine worst shift
        ones, zeros = get_ones_zeros(x)
        x, ov = worst_shift_refine(x, ov, ones, zeros)
        cur_min = ov.min()
        
        # Update best
        if cur_min < best_min:
            best_x = x.copy()
            best_min = cur_min
            best_ov = ov.copy()
    
    # Final evaluation
    c5 = dx * (TARGET_SUM - best_min)
    
    return (best_x.tolist(), float(c5), n_points)
