import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time
from math import sqrt

# Constants
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_TIME_FRACTION = 0.99  # leave some time for final processing

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_pad = np.pad(h, (0, n_points - 1), mode='constant')
    h_conj_pad = np.pad(1 - h, (0, n_points - 1), mode='constant')
    corr_fft = ifft(fft(h_pad) * np.conj(fft(h_conj_pad)))
    corr_full = np.real(corr_fft)
    # corr_full has length 2*n - 1
    # indices from 0 to 2*n-2
    # The actual shift values are from -(n-1) to (n-1)
    # corr_full[i] corresponds to shift (i - (n-1))
    # max correlation is at the center index or near it
    max_idx = np.argmax(corr_full)
    return corr_full[max_idx] * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def normalization(h):
    """Normalize h to have required sum"""
    return project_to_sum_box(h, TARGET_SUM)

def softmax_weights(corr, alpha):
    """Compute softmax weights"""
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_shifted = np.exp(alpha * shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights

def compute_gradient(h, weights, n):
    """Compute gradient of soft-max objective"""
    # Forward convolution
    h_padded = np.pad(h, (0, n - 1), mode='constant')
    w_padded = np.pad(weights, (0, n - 1), mode='constant')
    conv_forward = np.real(ifft(fft(h_padded) * np.conj(fft(w_padded))))
    conv_forward = conv_forward[:n]
    
    # Backward convolution
    w_reversed = weights[::-1]
    w_rev_padded = np.pad(w_reversed, (0, n - 1), mode='constant')
    conv_backward = np.real(ifft(fft(h_padded) * np.conj(fft(w_rev_padded))))
    conv_backward = conv_backward[:n]
    
    # Gradient computation
    grad = 1 - conv_forward - conv_backward
    return grad

def line_search(h, grad, target_sum, alpha, weight_func, n, max_iter=20):
    # Initial trial step
    eta = 0.5
    for _ in range(max_iter):
        h_trial = h - eta * grad
        h_projected = project_to_sum_box(h_trial, target_sum)
        # Compute full correlation
        corr = correlate(h_projected, 1 - h_projected, mode='full')
        max_corr = np.max(corr)
        if max_corr < np.max(correlate(h, 1 - h, mode='full')) * dx:
            return h_projected, eta
        eta /= 2.0
    return h, eta

def projected_gradient_descent(h_start, max_iter=5000, alpha0=5, alpha_max=20, step_size=0.5, 
                              step_multiplier=1.02, alpha_multiplier=1.01):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha0
    eta = step_size
    old_c5 = float('inf')
    stagnation_count = 0
    convergence_threshold = 1e-5

    for i in range(max_iter):
        # Full correlation
        corr = correlate(h, 1 - h, mode='full')
        # Weights for soft max
        weights = softmax_weights(corr, alpha)
        # Compute gradient
        grad = compute_gradient(h, weights, n_points)
        # Line search
        h_new, eta = line_search(h, grad, TARGET_SUM, alpha, softmax_weights, n_points)
        # Update h
        h = h_new.copy()
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Update schedule
        alpha = min(alpha_max, alpha * alpha_multiplier)
        eta = min(eta * step_multiplier, 0.1)
        # Convergence test
        if abs(c5 - old_c5) < convergence_threshold:
            stagnation_count += 1
            if stagnation_count > 200:
                break
        else:
            stagnation_count = 0
        old_c5 = c5
    return best_h, best_c5

def generate_legendre_sequence(length):
    """Generate a Legendre sequence"""
    if length < 2:
        return []
    seq = np.ones(length)
    # Generate Legendre sequence using quadratic residues
    for i in range(1, length):
        # Legendre symbol (i/length) mod 2
        quad_res = pow(i, (length - 1) // 2, length) % length
        if quad_res != 0:
            # Use quadratic non-residue
            seq[i] = -1
    return seq

    h_random = normalization(h_random)
    h_seeds.append(("random continuous", h_random.copy()))

    h_binary = np.zeros(n_points)
    idx = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h_binary[idx] = 1.0
    h_seeds.append(("random binary", h_binary.copy()))

        legendre = generate_legendre_sequence(n_points)
        # Take first n_points values and convert to 0,1 values
        h_legendre = np.array(legendre)
        # Convert to binary - replace -1 by 0, keep 1
        h_legendre = (h_legendre + 1) / 2
        # Ensure correct number of ones
        current_ones = np.sum(h_legendre)
        diff = int(TARGET_SUM - current_ones)
        if diff > 0:
            idx_zero = np.where(h_legendre == 0)[0]
            if len(idx_zero) >= diff:
                for i in range(diff):
                    h_legendre[idx_zero[i]] = 1
        elif diff < 0:
            idx_one = np.where(h_legendre == 1)[0]
            if len(idx_one) >= -diff:
                for i in range(-diff):
                    h_legendre[idx_one[i]] = 0
        h_seeds.append(("legendre binary", h_legendre.copy()))
    except:
        pass

    best_c5 = float('inf')

    if time.time() < target_time and best_h is not None:

        for i in range(num_restarts):

            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)

            h_run, c5_run = projected_gradient_descent(h_pert, max_iter=50,

            if c5_run < best_c5:
                best_c5 = c5_run
                best_h = h_run.copy()

    # Final evaluation with official function
    final_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(final_c5), n_points)
