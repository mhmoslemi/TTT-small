import numpy as np
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def full_overlap(x):
    """Compute full overlap array using numpy correlate"""
    return np.correlate(x, x, 'full')

def initial_binary(seed):
    """Create initial binary vector with exact sum constraint"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def swap_update(x, ov, i, j, ones, zeros):
    """Update overlap array in O(w) time for a swap of x[i] and x[j]"""
    ov_new = ov.copy()
    # Remove contribution from i (both directions)
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n_points - 1] -= 1
    # Add contribution from j (both directions)
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    # Special case for j itself
    for k in zeros:
        if k != j:
            s = k - j
            ov_new[s + n_points - 1] += 1
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Run simulated annealing with fast incremental updates"""
    start = time.time()
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = ov.min()
    
    T0 = 0.5
    lambda_decay = 0.001
    max_iter = 1000000
    iter_count = 0
    
    while time.time() - start < time_limit and iter_count < max_iter:
        iter_count += 1
        # Pick random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute new overlap
        ov_new = swap_update(x, ov, i, j, ones, zeros)
        new_min = ov_new.min()
        delta = new_min - best_min
        
        # Accept condition
        if delta > 0:
            accept = True
        else:
            T = T0 * np.exp(-lambda_decay * iter_count)
            if T < 1e-4:  # Early stop
                break
            accept = np.random.rand() < np.exp(delta / T)
        
        if accept:
            # Update current solution
            x[i] = 0
            x[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            ov = ov_new
            if new_min > best_min:
                best_x = x.copy()
                best_ov = ov.copy()
                best_min = new_min
    
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Improve solution by repairing worst shift"""
    best_ov = ov.copy()
    best_min = ov.min()
    best_x = x.copy()
    
    # Get current worst shift
    worst_shift_idx = np.argmin(ov)
    worst_shift = worst_shift_idx - (n_points - 1)
    
    # Identify deficient 1-positions and beneficial 0-positions
    deficient = []
    beneficial = []
    
    for i in ones:
        if 0 <= i + worst_shift < n_points:
            if best_x[i + worst_shift] == 0:
                deficient.append(i)
    
    for j in zeros:
        if 0 <= j + worst_shift < n_points:
            if best_x[j + worst_shift] == 1:
                beneficial.append(j)
    
    # Try swaps
    for i in deficient:
        for j in beneficial:
            # Check if this swap will improve overlap
            ov_test = best_ov.copy()
            
            # Remove i and add j contributions
            for k in ones:
                if k != i:
                    s = k - i
                    ov_test[s + n_points - 1] -= 1
            for k in ones:
                if k != i:
                    s = k - j
                    ov_test[s + n_points - 1] += 1
            for k in zeros:
                if k != j:
                    s = k - j
                    ov_test[s + n_points - 1] += 1
            
            new_min = ov_test.min()
            
            if new_min > best_min:
                x[i] = 0
                x[j] = 1
                ones.remove(i)
                zeros.remove(j)
                ones.add(j)
                zeros.add(i)
                best_ov = ov_test
                best_min = new_min
                break
        else:
            continue
        break  # Break from innermost loop only if we broke an outer loop too
        
    return x, best_ov, best_min

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    x = initial_binary(seed)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    ov = full_overlap(x)
    best_min = ov.min()
    best_x = x.copy()
    
    # Multiple restarts of simulated annealing
    restarts = 10
    remaining_time = budget_s * 0.9
    
    for r in range(restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        x_restart = initial_binary(seed + r)
        ones_restart = set(np.where(x_restart == 1)[0])
        zeros_restart = set(np.where(x_restart == 0)[0])
        ov_restart = full_overlap(x_restart)
        
        # Run one short SA run with 60s (or less)
        sa_run_time = min(remaining_time / restarts, 60.0)
        x_best, ov_best, min_val = simulated_annealing(
            x_restart, ov_restart, ones_restart, zeros_restart, sa_run_time
        )
        
        # Post-process with worst-shift repair
        x_refined, ov_refined, min_refined = worst_shift_refine(
            x_best, ov_best, set(np.where(x_best == 1)[0]), set(np.where(x_best == 0)[0])
        )
        
        if min_refined > best_min:
            best_min = min_refined
            best_x = x_refined.copy()
    
    # Final computation
    c5 = dx * (TARGET_SUM - best_min)
    h_values = best_x.astype(float).tolist()
    
    return (h_values, float(c5), n_points)
