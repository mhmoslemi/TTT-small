import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradients(h):
    """Compute gradient for each position by analyzing correlation"""
    corr = correlate(h, 1 - h, mode='full')
    max_lag = np.argmax(corr)
    grad = np.zeros(n_points)
    
    # Compute contribution to each position as if we move a small amount
    for i in range(n_points):
        # Find positions that affect the maximum correlation
        for j in range(n_points):
            if abs(i - j) == max_lag:
                grad[i] += h[j]
    
    return grad

        h_test = project_to_sum_box(h_test, TARGET_SUM)
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()

        if iter_num % 50 == 0:
            step_size = max(0.0001, step_size * 0.98)
        
        # Get current gradient
        grad = compute_gradients(h)
        
        # Find directions that decrease C5
        improvement_dir = np.argsort(grad)
        
        # Try to improve in most negative gradient directions
        for i in range(min(5, len(improvement_dir))):
            direction = improvement_dir[i]
            # Try moving 10% of current value or step_size
            move_amount = min(step_size, h[direction] * 0.1)
            if move_amount < 1e-6:
                move_amount = step_size
            
            # Move mass from neighbors toward direction
            best_move = 0
            best_to = -1
            for j in range(max(0, direction - 10), min(n_points, direction + 10)):
                if j != direction and h[j] < 1:
                    change = min(move_amount, 1 - h[j])
                    if change > best_move:
                        best_move = change
                        best_to = j
            
            if best_move > 0:
                h[direction] -= best_move
                h[best_to] += best_move
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

    # Normalize to make sum = 1
    h = project_to_sum_box(h, TARGET_SUM)
    return h
