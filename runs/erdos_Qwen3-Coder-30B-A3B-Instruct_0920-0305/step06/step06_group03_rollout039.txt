def project_to_sum_box(x, target):
    # find lambda such that sum(clamp(x+lambda, 0, 1)) = target
    def f(lam):
        return np.clip(x + lam, 0, 1).sum() - target
    lo, hi = -1.0, 1.0
    while f(lo) * f(hi) > 0:
        lo -= 1
        hi += 1
    for _ in range(60):
        mid = (lo + hi) / 2
        fm = f(mid)
        if abs(fm) < 1e-12:
            break
        if f(lo) * fm <= 0:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)

def evaluate_c5_fft(h):
    """Evaluate C5 using FFT for fast correlation computation"""
    n = len(h)
    L = 2*n - 1
    h_rev = h[::-1]
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L))) * DX_GLOBAL
    return float(1.0 - corr.max())

    step = step0
    best_h = h.copy()
    best_c5 = 1.0
    L = 2 * n - 1
    for it in range(max_iter):
        # compute correlation using FFT
        h_rev = h[::-1]
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L))) * dx
        overlaps = 1.0 - corr
        max_val = overlaps.max()
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        # find indices near top
        tol = 1e-12
        inds = np.where(np.abs(overlaps - max_val) <= tol)[0]
        # if too many (unlikely), sample up to 10
        if len(inds) > 10:
            inds = np.random.choice(inds, size=10, replace=False)
        # compute gradient
        grad = np.zeros_like(h)
        for idx in inds:
            shift = idx - (n - 1)
            if shift >= 0:
                grad[:n - shift] -= h[shift:]
                grad[shift:] -= h[:n - shift]
            else:
                s = -shift
                grad[s:] -= h[:n - s]
                grad[:n - s] -= h[s:]
        grad = grad / len(inds)
        # update
        h = h - step * grad
        # box clamp
        np.clip(h, 0.0, 1.0, out=h)
        # project to sum
        h = project_to_sum_box(h, target_sum)
        # decay step
        step *= decay
    return best_h, best_c5
