import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction from provided solution, interpolated to n_points
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, h0)
    else:
        h_interpolated = np.ones(n_points) * 0.5
    
    h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
    
    # Initial solution
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2
    
    # Simulated Annealing approach with mixed strategies
    current_h = h_interpolated.copy()
    current_c5 = best_c5
    
    # Temperature schedule parameters
    T_start = 0.1
    T_end = 0.001
    cooling_rate = 0.9995
    max_iter = 1000
    
    def perturb_solution(h, T):
        """Perturb solution using multiple strategies"""
        strategy = random.choice(['small', 'large', 'local', 'global'])
        h_new = h.copy()
        
        if strategy == 'small':
            # Small perturbation
            idx = random.randint(0, n_points-1)
            delta = random.uniform(-0.02, 0.02)
            h_new[idx] += delta
        elif strategy == 'large':
            # Large perturbation
            indices = random.sample(range(n_points), max(1, n_points//10))
            delta = random.uniform(-0.1, 0.1)
            for idx in indices:
                h_new[idx] += delta
        elif strategy == 'local':
            # Localized perturbation
            idx = random.randint(0, n_points-1)
            window = min(10, n_points//4)
            start_idx = max(0, idx - window//2)
            end_idx = min(n_points, idx + window//2 + 1)
            for i in range(start_idx, end_idx):
                h_new[i] += random.uniform(-0.05, 0.05)
        else:  # global
            # Global perturbation
            deltas = np.random.normal(0, 0.05, n_points)
            h_new += deltas
            
        return project_to_sum_box(h_new, TARGET_SUM)
    
    # Annealing loop
    T = T_start
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Generate neighbor solution
        h_new = perturb_solution(current_h, T)
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject the new solution
        if c5_new < current_c5 or random.random() < np.exp(-(c5_new - current_c5) / T):
            current_h = h_new
            current_c5 = c5_new
            
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        
        # Cool down
        T = max(T_end, T * cooling_rate)
        iter_count += 1
    
    # Additional local refinement
    while time.time() - start_time < budget_s - 5:
        # Try random perturbations and local optimizations
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        # Local optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            return evaluate_c5(h)
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 50, 'ftol': 1e-12})
            if res.success:
                h_candidate = res.x
                h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
