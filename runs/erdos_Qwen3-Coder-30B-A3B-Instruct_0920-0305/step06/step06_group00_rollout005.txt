import numpy as np

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h_full = np.concatenate([h, np.zeros_like(h)])
    h_complement = np.concatenate([1-h, np.zeros_like(1-h)])
    corr = fftconvolve(h_full, h_complement[::-1], mode='full')
    return 2.0 * np.max(corr[:2*len(h)-1]) / len(h)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for correlation values"""
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(alpha * corr_shifted)
    return exp_corr / np.sum(exp_corr)

def compute_gradient(h, weights):
    """Compute gradient of soft-max objective using FFT"""
    n = len(h)
    # Forward convolution (shifted weights)
    w = np.concatenate([weights[n-1:], np.zeros(n-1)])
    # Backward convolution (reversed shifted weights)
    w_rev = np.concatenate([np.zeros(n-1), weights[:n]])
    
    # FFT-based convolutions
    h_fft = np.fft.fft(h)
    w_fft = np.fft.fft(w)
    w_rev_fft = np.fft.fft(w_rev)
    
    # Compute the two convolutions
    c_f = np.real(np.fft.ifft(h_fft * w_fft))
    c_b = np.real(np.fft.ifft(h_fft * w_rev_fft))
    
    # Gradient is 1 - c_f - c_b
    grad = np.ones(n) - c_f - c_b
    return grad

def projected_gradient_descent(h0, target_sum, alpha_max=20.0, max_iter=10000, beta=5, eta_max=1.0):
    """Run projected gradient descent with soft-max objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = beta
    eta = 1.0
    
    for i in range(max_iter):
        corr = fftconvolve(np.concatenate([h, np.zeros_like(h)]), 
                          np.concatenate([1-h, np.zeros_like(1-h)])[::-1], 
                          mode='full')[:2*len(h)-1]
        
        weights = softmax_weights(corr, alpha)
        grad = compute_gradient(h, weights)
        
        # Backtracking line search
        eta_test = eta
        while True:
            h_trial = project_to_sum_box(h - eta_test * grad, target_sum)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                break
            eta_test /= 2
            if eta_test < 1e-8:
                break
        
        # Update step size and temperature
        eta = min(eta_max, eta * 1.02)
        alpha = min(alpha_max, alpha * 1.01)
        
        # Convergence check
        if i > 100 and abs(best_c5 - evaluate_c5(best_h)) < 1e-10:
            break
    
    return best_h, best_c5
