import numpy as np
import time

def initial_binary(seed, n):
    """Create a random binary vector with exact weight n//2"""
    np.random.seed(seed)
    x = np.zeros(n, dtype=np.int8)
    ones_indices = np.random.choice(n, size=n//2, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full correlation of x with itself"""
    return np.correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros):
    """
    Incrementally update overlap array for swapping x[i] and x[j]
    Return new overlap array (copy of ov)
    """
    ov_new = ov.copy()
    n = len(ones)
    # Remove contribution of i
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n - 1] -= 1
    # Add contribution of j
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n - 1] += 1
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Run simulated annealing with binary swaps"""
    start = time.time()
    n = len(x)
    max_iter = 1_000_000
    T0 = 0.5
    lambda_coeff = 5.0  # cooling rate
    cur_min = ov.min()
    best_x = x.copy()
    best_min = cur_min
    cur_ov = ov.copy()
    
    # Allow multiple restarts with different seeds if time permits
    for restart in range(50):
        if time.time() - start > time_limit * 0.95:
            break
            
        # Starting point
        run_x = x.copy()
        run_ov = ov.copy()
        run_ones = ones.copy()
        run_zeros = zeros.copy()
        run_min = cur_min
        t = 0
        no_improvement_count = 0
        
        for i in range(max_iter):
            if time.time() - start > time_limit * 0.95:
                break
                
            # Accept all improvements and sometimes worse moves
            # Pick a 1 and a 0 at random
            if len(run_ones) == 0 or len(run_zeros) == 0:
                break
                
            idx1 = np.random.choice(run_ones)
            idx2 = np.random.choice(run_zeros)
            
            # Compute overlap after swap
            new_ov = swap_update(run_ov, idx1, idx2, run_ones, run_zeros)
            new_min = new_ov.min()
            
            delta = new_min - run_min
            
            # Accept if improvement or with probability exp(-delta/T)
            T = T0 * np.exp(-lambda_coeff * t)
            if delta > 0 or (T > 1e-8 and np.random.rand() < np.exp(delta / T)):
                # Apply the move
                run_x[idx1] = 0
                run_x[idx2] = 1
                run_ones.remove(idx1)
                run_ones.add(idx2)
                run_zeros.remove(idx2)
                run_zeros.add(idx1)
                run_ov = new_ov
                run_min = new_min
                no_improvement_count = 0
                
                if new_min > best_min:
                    best_min = new_min
                    best_x = run_x.copy()
            else:
                no_improvement_count += 1
                if no_improvement_count > 10000:
                    # Exit if no progress for long time
                    break
                    
            t += 1
    
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Apply deterministic swaps to improve worst shift"""
    n = len(x)
    worst_shift = np.argmin(ov) - (n - 1)
    min_value = ov.min()
    
    # Find indices contributing to worst overlap
    # When doing the overlap computation using correlate, 
    # we compute the overlap for lag = -(n-1) to n-1
    # So for shift = -(n-1) to n-1, we check index k, k+shift
    # and check if x[k] == x[k+shift] == 1
    
    # Scanning all pairs
    all_pairs = []
    for i in range(n):
        j = i + worst_shift
        if 0 <= j < n:
            if x[i] == 1 and x[j] == 0:
                all_pairs.append((i, j))  # 1 at i, 0 at j
    
    # Now for our processing:
    # For each pair (i,j) where we have 1 at i and 0 at j,
    # and i+worst_shift corresponds to a 0, we can try swapping to see if 
    # that increases the overlap at the worst shift.
    # If x[i]=1, x[i+worst_shift]=0 and x[j]=0 and x[j+worst_shift]=1, 
    # then swap increases ov[worst_shift]
    
    # Faster but thorough version now:
    # Better logic:
    # Consider each i where x[i]=1
    # Look at position k = i+worst_shift
    # If x[k] = 0, we can consider swapping x[i] with a 0 elsewhere  
    # to increase overlap at shift worst_shift.
    # But specifically for shifting by worst_shift:
    # this is wrong. We can improve overlap by swapping such an i with 
    # a 0 positioned at j where x[j]=0 and x[j+worst_shift] = 1.
    # In other words, x[i]=1, x[j]=0, 
    # x[i+worst_shift]=0 and x[j+worst_shift]=1 should be the condition  
    # for beneficial swap.
    
    changed = True
    while changed:
        changed = False
        # Find allowed swaps
        for i in ones:
            k = i + worst_shift
            if 0 <= k < n and x[k] == 0:
                # Need to find a 0 j in the range [0,n-1] such that
                # x[j+worst_shift] = 1 and x[j] = 0  
                for j in zeros:
                    if j + worst_shift < n and x[j + worst_shift] == 1:
                        # Swap x[i] and x[j]
                        x[i] = 0
                        x[j] = 1
                        ones.remove(i)
                        zeros.remove(j)
                        ones.add(j)
                        zeros.add(i)
                        # Update overlaps:
                        # The move affects overlaps for shifts near worst_shift
                        temp_ov = ov.copy()
                        
                        # Update overlap array accordingly:
                        # For each i in current ones (excluding old i)
                        for k in ones:  
                            s = k - i
                            temp_ov[s + n - 1] -= 1
                        # Apply addition of new 1 at j
                        for k in ones:  # Can be same as i but confusing to repeat, just redo all:
                            s = k - j
                            temp_ov[s + n - 1] += 1
                        # Also fix the ones for i:
                        ov = temp_ov
                        cur_min = ov.min()
                        if cur_min > min_value:
                            min_value = cur_min  
                            changed = True
                            # Need to do again for the new overlap:
                            break
            if changed:
                break
            # Update set representations:
            ov = full_overlap(x)
            min_value = ov.min()
    
    # Final evaluation:
    ov_final = full_overlap(x)
    return x, ov_final.min()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 120
    dx = 2.0 / n
    target_sum = n // 2
    
    best_x = initial_binary(seed, n)
    while np.sum(best_x) != target_sum:
        # If by chance the initial having wrong number of 1s
        best_x = initial_binary(seed, n)
    
    # Sets of 1s and 0s for fast lookups
    ones = set(np.where(best_x == 1)[0])
    zeros = set(np.where(best_x == 0)[0])
    ov = full_overlap(best_x)
    cur_min = ov.min()
    
    # Search through many restarts of annealing:
    for restart_i in range(30):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Make a copy and setup
        x = initial_binary(seed + restart_i, n)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        cur_min = ov.min()
        
        # Run SA
        x_sa, new_min = simulated_annealing(x, ov, ones, zeros, budget_s * 0.8)
        if new_min > cur_min:
            cur_min = new_min
            best_x = x_sa
            
    # Post-process with worst shift refinement
    best_ov = full_overlap(best_x)
    # Refine via deterministic swap fixing
    best_x, post_min = worst_shift_refine(best_x, best_ov, 
                                           set(np.where(best_x == 1)[0]),  
                                           set(np.where(best_x == 0)[0]))
    
    c5 = dx * (0.5 * n - post_min)
    # Ensuring return is 120 length
    return (best_x.astype(float).tolist(), float(c5), n)
