import numpy as np

def evaluate_erdos_solution(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_milp(n, time_limit):
    """Solve the MILP for binary variables"""
    M = n // 2
    dx = 2.0 / n

    # Decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()

    # Auxiliary variables for AND operation
    # We create variables for all pairs (i,j) with i < j
    y_vars = {}
    for i in range(n):
        for j in range(i+1, n):
            y_vars[(i,j)] = cp.Variable(boolean=True)

    # Constraints
    constraints = []

    # Exact mass constraint
    constraints.append(cp.sum(h) == M)

    # Linking AND constraints
    for i in range(n):
        for j in range(i+1, n):
            y = y_vars[(i,j)]
            constraints.append(y <= h[i])
            constraints.append(y <= h[j])
            constraints.append(y >= h[i] + h[j] - 1)

    # Lag constraints
    # For each k from 1 to n-1
    # t >= M - sum_{i=0}^{n-1-k} y_{i,i+k}
    for k in range(1, n):
        sum_y = 0
        for i in range(n - k):
            sum_y += y_vars[(i, i+k)]
        constraints.append(t >= M - sum_y)

    # Objective
    objective = cp.Minimize(t)

    # Create problem
    prob = cp.Problem(objective, constraints)

    # Set parameters
    solver_options = {
        "seconds": time_limit,
        "threads": 1
    }

    try:
        prob.solve(solver=solver=solver=solver, verbose=False, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.rint(h.value).astype(int)
            t_val = t.value
            return h_val, t_val
    except:
        pass

    return None, None
