import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap occurs at the center (lag 0) including all terms
    # The valid lags are from -n+1 to n-1, but we want the actual maximum of the overlap
    # This is equivalent to the maximum of all terms, but shifted by n-1
    max_overlap = np.max(corr)
    return dx * max_overlap

def solve_milp(n, time_limit):
    """
    Solve the Erdős minimum overlap problem as a mixed-integer linear program (MILP)
    """
    # Parameters
    M = n // 2  # Required number of ones (integral constraint)
    dx = 2.0 / n
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(1, nonneg=True)
    
    # Auxiliary variables for the AND operation
    # y[i,j] represents h[i] AND h[j]
    y = cp.Variable((n, n), boolean=True)
    
    # Constraints
    constraints = []
    
    # Exact mass constraint: sum of h equals M
    constraints.append(cp.sum(h) == M)
    
    # Linking AND constraints for all pairs (i,j)
    for i in range(n):
        for j in range(i+1, n):  # Only lower triangular to avoid redundancy
            # y[i,j] <= h[i]
            constraints.append(y[i,j] <= h[i])
            # y[i,j] <= h[j]
            constraints.append(y[i,j] <= h[j])
            # y[i,j] >= h[i] + h[j] - 1
            constraints.append(y[i,j] >= h[i] + h[j] - 1)
            # y[i,j] = y[j,i] for symmetry
            # Assume y is symmetric (same variable is used for both)
            y[j,i] = y[i,j]  # For simplicity in loop
    
    # Lag constraints (one per valid lag k)
    # We compute: M - sum of y_{i, i+k} for each lag k = 1, ..., n-1
    for k in range(1, n):
        # For lag k, sum over i such that i+k < n
        sum_y = cp.sum([y[i, i+k] for i in range(n-k)])
        # t >= M - sum_y
        constraints.append(t >= M - sum_y)
    
    # Fix first entry to break symmetry (optional but helps
    # reduce search space significantly)
    constraints.append(h[0] == 1)
    
    # objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with a given time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, cbc_params={"seconds": time_limit})
    except Exception as e:
        # Sometimes solver might fail due to internal issues
        return None, None
        
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = np.rint(h.value).astype(int)
        t_val = t.value
        return h_sol, t_val
    else:
        # If problem status is not optimal, try a fallback
        return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto the box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h_init, max_iter=200):
    """
    Perform a short continuous local refinement on the binary solution
    """
    def objective(h):
        return evaluate_erdos_solution(h)
    
    def constraint_eq(h):
        return np.sum(h) - len(h)/2  # mass constraint

    bounds = [(0, 1) for _ in range(len(h_init))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h_init, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, len(h)/2)
            return h
    except:
        pass
    return h_init
