import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def construct_initial_h():
    """Construct an initial h using a pattern that balances exploration and balance"""
    # Start with a piecewise constant construction
    h = np.zeros(n_points)
    
    # Fill with segments
    segment_length = n_points // 4
    h[:segment_length] = 0.5
    h[segment_length:2*segment_length] = 0.8
    h[2*segment_length:3*segment_length] = 0.3
    h[3*segment_length:] = 0.6
    
    return project_to_sum_box(h, TARGET_SUM)

def compute_gradient(h):
    """Compute approximate gradient based on correlation function"""
    corr = compute_corr_fft(h)
    worst_lag = np.argmax(corr)
    
    grad = np.zeros(n_points)
    for i in range(n_points):
        if i + worst_lag < n_points:
            grad[i] += h[i + worst_lag]
        if i - worst_lag >= 0:
            grad[i] += h[i - worst_lag]
    
    return grad

def gradient_based_move(h, max_iterations=30):
    """Refine using a gradient-like approach"""
    h = h.copy()
    
    for _ in range(max_iterations):
        grad = compute_gradient(h)
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
            
        # More aggressive moves to escape local minimum
        delta = min(h[idx_min], 1 - h[idx_max], 0.02)
        if delta <= 0:
            break
            
        h[idx_min] -= delta
        h[idx_max] += delta
        
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Stop early if we reach target
        if evaluate_c5(h) < 0.379:
            break
            
    return h

def local_refine(h, max_iter=200):
    """Local SLSQP refinement with better initial guess"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-14}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize target sum
    TARGET_SUM = n_points / 2
    
    # Create initial h with better pattern than default
    h0 = construct_initial_h()
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase A: Multi-start gradient-based refinement
    phase_a_end = start_time + 0.7 * budget_s
    run_id = 0
    
    while time.time() < phase_a_end:
        if run_id % 5 == 0:
            h_in = h0.copy()
        elif run_id % 5 == 1:
            # Random smooth noise
            h_in = np.random.uniform(0, 1, n_points)
            h_in = project_to_sum_box(h_in, TARGET_SUM)
        elif run_id % 5 == 2:
            # Piecewise constant with different pattern
            h_in = np.zeros(n_points)
            h_in[:n_points//3] = 0.2
            h_in[n_points//3:2*n_points//3] = 0.7  
            h_in[2*n_points//3:] = 0.4
            h_in = project_to_sum_box(h_in, TARGET_SUM)
        elif run_id % 5 == 3:
            # Binary pattern  
            h_in = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
            h_in = project_to_sum_box(h_in, TARGET_SUM)
        else:  # run_id % 5 == 4
            # Random with noise
            h_in = h0 + np.random.normal(0, 0.05, n_points)
            h_in = project_to_sum_box(h_in, TARGET_SUM)
            
        h_in = local_refine(h_in, max_iter=50)
        h_in = gradient_based_move(h_in)
        
        c5_candidate = evaluate_c5(h_in)
        if c5_candidate < best_c5:
            best_h = h_in.copy()
            best_c5 = c5_candidate
            
        run_id += 1
        
        if time.time() - start_time > 0.7 * budget_s:
            break
    
    # Phase B: Enhanced local refinement
    phase_b_end = start_time + 0.9 * budget_s
    for _ in range(50):
        if time.time() >= phase_b_end:
            break
            
        # Generate new candidate with small random perturbations
        h_candidate = best_h + np.random.normal(0, 0.01, n_points)
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        h_candidate = local_refine(h_candidate, max_iter=100)
        
        c5_candidate = evaluate_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
    
    # Phase C: Exploit corridors
    while time.time() < start_time + budget_s:
        strategies = []
        
        # Strategy 1: Gradient direction refinement with stronger moves
        h1 = gradient_based_move(best_h.copy(), 20)
        strategies.append((h1, evaluate_c5(h1)))
        
        # Strategy 2: Direct noise injection with refinement
        h2 = best_h + np.random.normal(0, 0.02, n_points)
        h2 = project_to_sum_box(h2, TARGET_SUM) 
        h2 = local_refine(h2, max_iter=80)
        strategies.append((h2, evaluate_c5(h2)))
        
        # Strategy 3: Part of the population optimizations
        
        for h_candidate, c5_candidate in strategies:
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
        
        # Additional refinement to make sure we don't miss anything
        if abs(evaluate_c5(best_h) - best_c5) < 1e-10:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
