import numpy as np
import time
from scipy.optimize import minimize

# Global variables for optimization
CURRENT_TARGET_SUM = None
DX_GLOBAL = None

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_fft(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    L = 2*n - 1
    h_rev = h[::-1]
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L))) * DX_GLOBAL
    return float(1.0 - corr.max())

def subgradient_optimize(h0, n, target_sum, dx, max_iter=2000, step0=0.2, decay=0.999):
    """Perform subgradient optimization with decayed step size"""
    h = h0.copy()
    step = step0
    best_h = h.copy()
    best_c5 = 1.0  # high initial
    L = 2 * n - 1
    
    for it in range(max_iter):
        # compute correlation using FFT
        h_rev = h[::-1]
        corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L))) * dx
        overlaps = 1.0 - corr
        max_val = overlaps.max()
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
            
        # find indices near top
        tol = 1e-12
        inds = np.where(np.abs(overlaps - max_val) <= tol)[0]
        
        # if too many (unlikely), sample up to 10
        if len(inds) > 10:
            inds = np.random.choice(inds, size=10, replace=False)
            
        # compute gradient
        grad = np.zeros_like(h)
        for idx in inds:
            shift = idx - (n - 1)
            if shift >= 0:
                grad[:n - shift] -= h[shift:]
                grad[shift:] -= h[:n - shift]
            else:
                s = -shift
                grad[s:] -= h[:n - s]
                grad[:n - s] -= h[s:]
                
        grad = grad / len(inds)
        
        # update
        h = h - step * grad
        
        # box clamp
        np.clip(h, 0.0, 1.0, out=h)
        
        # project to sum
        h = project_to_sum_box(h, target_sum)
        
        # decay step
        step *= decay
        
    return best_h, best_c5

def sum_constraint(h):
    """Constraint for sum = target_sum"""
    return np.sum(h) - CURRENT_TARGET_SUM

def polish_solution(h_init):
    """Polish solution with SLSQP if time permits"""
    global CURRENT_TARGET_SUM, DX_GLOBAL
    n = len(h_init)
    DX_GLOBAL = 2.0 / n
    CURRENT_TARGET_SUM = n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = {'type':'eq', 'fun': sum_constraint}
    
    try:
        res = minimize(evaluate_c5_fft, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter':500, 'ftol':1e-12, 'disp':False})
        if res.success:
            h_pol = project_to_sum_box(res.x, CURRENT_TARGET_SUM)
            return h_pol, evaluate_c5_fft(h_pol)
    except:
        pass
    
    return h_init, evaluate_c5_fft(h_init)
