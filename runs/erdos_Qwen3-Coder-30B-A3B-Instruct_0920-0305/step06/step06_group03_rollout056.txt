```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial point from a simpler construction
    h_init = np.zeros(n_points)
    # Simple envelope-like construction to start
    h_init[:n_points//4] = 0.5
    h_init[n_points//4:n_points//2] = 1.0
    h_init[n_points//2:3*n_points//4] = 0.5
    h_init[3*n_points//4:] = 0.0
    
    # Adjust integral to match constraint exactly
    h_init = project_to_sum_box(h_init, n_points/2)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    TARGET_SUM = n_points / 2
    
    # Try multiple random restarts with different initial structures
    for restart in range(20):
        if time.time() - start_time > budget_s:
            break
            
        # Generate new random initial structure
        h0 = np.random.rand(n_points)
        # Make it so that it's more likely to be a good starting solution
        h0 = np.clip(h0 * 1.5, 0, 1)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        # Add some structure to avoid local minima
        h0[:n_points//6] = np.linspace(0, 0.5, n_points//6)
        h0[n_points//6:2*n_points//6] = np.linspace(0.5, 1.0, n_points//6)
        h0[2*n_points//6:3*n_points//6] = np.linspace(1.0, 0.5, n_points//6)
        h0[3*n_points//6:4*n_points//6] = np.linspace(0.5, 0.0, n_points//6)
        h0[4*n_points//6:5*n_points//6] = np.linspace(0.0, 0.25, n_points//6)
        h0[5*n_points//6:] = np.linspace(0.25, 0.0, n_points//6)
        
        h0 = project_to_sum_box(h0, TARGET_SUM)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                h0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue
    
    # Perform local random search to get better result
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
            
        # Slight perturbation with decreasing magnitude
        h_changed = best_h + np.random.normal(0, 0.003, n_points) 
        h_changed = project_to_sum_box(h_changed, TARGET_SUM)
        
        try:
            res = minimize(
                objective, 
                h_changed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_candidate = project_to_sum_box(res.x, TARGET_SUM)
                c5_candidate = evaluate_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            continue
    
    return (best_h.tolist(), float(best_c5), n_points)
```
