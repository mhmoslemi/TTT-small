def evaluate_erdos_solution(h):
    # Should compute: max_k ∫ h(x) (1 - h(x+k)) dx
    # Discretized: np.correlate(h, 1-h, mode="full") * dx
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - np.array(h), mode='full')
    # Return max correlation scaled by dx
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    # Binary search for lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    # adjust sum to be exact
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def full_corr(a, b):
    # Compute full correlation (zero-padded) using FFT
    n = len(a)
    N = 2 * n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def gradient_max_soft(h, alpha):
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = np.max(corr)
    # Compute soft weights
    weights = np.exp(alpha * (corr - maxc))
    weights_sum = np.sum(weights)
    weights /= weights_sum

    # Compute gradient via weighted convolution
    # For lag l, \partial C_l/\partial h[i] = (1 - h[i+l]) - h[i-l]
    # Gradient vector is [1 - sum_w[l] h[i+l] - sum_w[l] h[i-l]]
    # = 1 - conv_forward - conv_backward

    # Forward conv: w * h
    w_reversed = weights[::-1]
    conv_fwd = full_corr(h, w_reversed)
    # Backward conv: w_rev * h (which is w * h reversed)
    conv_bwd = full_corr(h, weights)
    # Gradient = 1 - (conv_fwd + conv_bwd) but offset to match index alignment
    # Just clipping for now to make it equivalent to manual loop version
    # Actual rectangular area should align the indices correctly
    # Now we ensure that the result has the right size: taking actual array index into account
    # simplify: just use radial symmetrized convolution, making sure arrays align
    # Use zero padding equivalent direction
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def line_search(h, grad, target_sum, alpha, eta, eps=1e-10):
    # Backtracking line search
    h_new = project_to_sum_box(h - eta * grad, target_sum)
    new_val = evaluate_erdos_solution(h_new)
    iteration = 0
    while new_val >= evaluate_erdos_solution(h) - eps and iteration < 10:
        eta *= 0.5
        h_new = project_to_sum_box(h - eta * grad, target_sum)
        new_val = evaluate_erdos_solution(h_new)
        iteration += 1
    return h_new, new_val, eta

def optimize_continuous(h0, remaining_time, n, target_sum, dx):
    best_h = h0.copy()
    best_val = evaluate_erdos_solution(h0)
    alpha = 5.0  # Starting soft max parameter
    eta = 0.1
    t0 = time.time()
    max_iter = 5000
    iter_count = 0
    while time.time() - t0 < remaining_time * 0.95 and iter_count < max_iter:
        grad = gradient_max_soft(best_h, alpha)
        new_h, new_val, eta = line_search(best_h, grad, target_sum, alpha, eta)
        if new_val < best_val:
            best_h = new_h
            best_val = new_val
            alpha = min(alpha + 0.02, 20.0)  # Gradually increase alpha
        else:
            # Perturb on stagnation
            p = np.random.normal(0, 0.001, n)
            perturbed = project_to_sum_box(best_h + p, target_sum)
            val_perturb = evaluate_erdos_solution(perturbed)
            if val_perturb < best_val:
                best_h = perturbed
                best_val = val_perturb
        iter_count += 1
    return best_h, best_val

def binary_local_search(h_binary, remaining_time, n, dx):
    start_time = time.time()
    best_h = h_binary.copy()
    best_val = evaluate_erdos_solution(best_h)
    temp = 0.01
    decay_rate = 0.998
    # For efficiency, we'll work with direct swaps without complex precomputation
    while time.time() - start_time < remaining_time:
        # Select two random indices: one with h=1, one with h=0
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Try swap: i=1->0, j=0->1
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = evaluate_erdos_solution(new_h)
        delta = new_val - best_val
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = new_h
            best_val = new_val
        temp *= decay_rate
    return best_h, best_val
