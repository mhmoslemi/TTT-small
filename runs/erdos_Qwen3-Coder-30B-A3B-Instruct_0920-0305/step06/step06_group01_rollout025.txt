# Global variables
TARGET_SUM = None

def compute_auto_correlation(h):
    """Compute auto-correlation using FFT"""
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(h, L)).real
    return corr

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_min_weights(S, alpha):
    """Compute soft-minimum weights w_k = exp(-alpha*S_k) / sum(exp(-alpha*S_k))"""
    S_max = np.max(-alpha * S)
    exp_neg_alpha_S = np.exp(-alpha * S - S_max)
    w = exp_neg_alpha_S / np.sum(exp_neg_alpha_S)
    return w

def compute_gradient(h, w):
    """Compute gradient g_j = sum_k w_k (h_{j+k} + h_{j-k}) 
    using symmetric convolution"""
    # Build symmetric kernel
    W = np.zeros(2 * len(h) - 1)
    for k in range(len(h)):
        if k == 0:
            W[len(h) - 1] = 2 * w[k]
        else:
            W[len(h) - 1 + k] = w[k]
            W[len(h) - 1 - k] = w[k]
    
    # Convolution with h
    conv = np.convolve(h, W, mode='full')
    g = conv[len(h) - 1 : len(h) - 1 + len(h)]
    return g
