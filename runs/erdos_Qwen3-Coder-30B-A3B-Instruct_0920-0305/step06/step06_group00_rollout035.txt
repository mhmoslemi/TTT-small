import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT-based correlation"""
    N = 2 * n_points
    h_fft = fft(h, N)
    h_complement_fft = fft(1 - h, N)
    corr_fft = h_fft * np.conj(h_complement_fft)
    corr = ifft(corr_fft).real
    corr_full = corr[:2 * n_points - 1]
    return (2 / n_points) * np.max(corr_full)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute soft-max weights from correlations"""
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    exp_corr = np.exp(alpha * corr_shifted)
    return exp_corr / np.sum(exp_corr)

def compute_gradient(h, weights):
    """Compute gradient using FFT-based convolution"""
    N = 2 * n_points
    
    # Forward convolution of h with weights
    h_fft = fft(h, N)
    w_fft = fft(weights, N)
    conv_forward = ifft(h_fft * w_fft).real
    
    # Backward convolution of h with reversed weights
    w_rev = weights[::-1]
    w_rev_fft = fft(w_rev, N)
    conv_backward = ifft(h_fft * w_rev_fft).real
    
    # Gradient computation
    grad = 1.0 - conv_forward - conv_backward
    return grad

def projected_gradient_descent(h0, max_iter=2000, alpha=5.0, alpha_max=20.0, eta=0.5, eta_max=1.0):
    """Projected gradient descent with soft max objective"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        corr = correlate(h, 1-h, mode='full') 
        corr = corr[:2*n_points-1]
        weights = softmax_weights(corr, alpha)
        grad = compute_gradient(h, weights)
        
        # Backtracking line search
        eta_test = eta
        while True:
            h_trial = project_to_sum_box(h - eta_test*grad, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                best_h = h.copy()
                break
                
            eta_test /= 2
            if eta_test < 1e-8:
                break
                
        alpha = min(alpha_max, alpha * 1.01)
        eta = min(eta_max, eta * 1.02)
    
    return best_h, best_c5
