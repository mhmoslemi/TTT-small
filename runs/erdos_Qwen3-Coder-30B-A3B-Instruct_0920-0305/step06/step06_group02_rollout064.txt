```python
import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed):
    """Initialize binary vector with exact weight constraint."""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full correlation O_s = sum_i x_i * x_{i+s} for all shifts s."""
    return np.correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros):
    """Update overlap array after swapping x[i] and x[j] in O(w) time."""
    ov_new = ov.copy()
    w = len(ones)
    
    # Remove contribution from i
    for k in ones:
        if k == i:
            continue
        s = k - i
        ov_new[s + (n_points - 1)] -= 1
    
    # Add contribution from j
    for k in ones:
        if k == i:
            continue
        s = k - j
        ov_new[s + (n_points - 1)] += 1

    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Simple simulated annealing with fast incremental updates."""
    start = time.time()
    T0 = 0.5
    T_min = 1e-4
    alpha = 0.99999
    max_iter = 1000000
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = ov.min()
    T = T0
    iter_count = 0

    while time.time() - start < time_limit and T > T_min:
        # Select two indices to swap (one 1 and one 0)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        ov_new = swap_update(best_ov, i, j, ones, zeros)
        new_min = ov_new.min()
        delta = new_min - best_min

        # Accept or reject
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            best_x[i] = 0
            best_x[j] = 1
            # Update sets
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            best_ov = ov_new
            best_min = new_min

        # Cool down
        T *= alpha
        iter_count += 1

        if iter_count % 10000 == 0:
            if time.time() - start > time_limit * 0.95:
                break
    
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine solution by improving the worst overlapping shift."""
    best_ov = ov.copy()
    while True:
        min_ov = best_ov.min()
        min_idx = np.argmin(best_ov)
        s_star = min_idx - (n_points - 1)
        
        # Find deficient 1s and beneficial 0s
        A = set()
        B = set()
        for i in ones:
            if 0 <= i + s_star < n_points:
                if x[i + s_star] == 0:
                    A.add(i)
        for j in zeros:
            if 0 <= j + s_star < n_points:
                if x[j + s_star] == 1:
                    B.add(j)
        
        best_swap_performed = False
        for i in A:
            for j in B:
                if i == j:
                    continue
                # Test the swap
                ov_test = swap_update(best_ov, i, j, ones.copy(), zeros.copy())
                new_min = ov_test.min()
                if new_min > min_ov:
                    # Perform the swap
                    x[i] = 0
                    x[j] = 1
                    ones.remove(i)
                    zeros.remove(j)
                    ones.add(j)
                    zeros.add(i)
                    best_ov = ov_test
                    best_swap_performed = True
                    break
            if best_swap_performed:
                break
        
        if not best_swap_performed:
            break
    
    return x, best_ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial binary vector with exact weight
    x = initial_binary(seed)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    ov = full_overlap(x)
    current_min = ov.min()
    
    # Local optimization loop with simulated annealing
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = current_min
    
    # Run multiple SA phases with different strategies
    n_sa_runs = 5
    for run_id in range(n_sa_runs):
        seed_val = seed + run_id
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Try different initializations
        x_init = initial_binary(seed_val)
        ones_init = set(np.where(x_init == 1)[0])
        zeros_init = set(np.where(x_init == 0)[0])
        ov_init = full_overlap(x_init)
        min_init = ov_init.min()
        
        # Run SA
        x_sa, ov_sa, min_sa = simulated_annealing(
            x_init, ov_init, ones_init, zeros_init,
            (budget_s * 0.95 - (time.time() - start_time)) / n_sa_runs
        )
        
        if min_sa > best_min:
            best_x = x_sa
            best_ov = ov_sa
            best_min = min_sa
    
    # Final refinement
    ones_final = set(np.where(best_x == 1)[0])
    zeros_final = set(np.where(best_x == 0)[0])
    best_x, best_ov = worst_shift_refine(best_x, best_ov, ones_final, zeros_final)
    
    # Compute final C5
    final_min = best_ov.min()
    final_c5 = dx * (TARGET_SUM - final_min)
    
    # Convert to float list for output
    h_values = best_x.astype(float).tolist()
    
    return (h_values, float(final_c5), n_points)
```
