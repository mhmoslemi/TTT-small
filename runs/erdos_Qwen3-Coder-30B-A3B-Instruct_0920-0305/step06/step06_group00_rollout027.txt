import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_descent_step(h, lr=0.01):
    """Perform one gradient descent step to reduce C5"""
    # Compute correlation and find the peak
    corr = correlate(h, 1 - h, mode='full')
    max_lag = np.argmax(corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(n_points):
        for j in range(n_points):
            if abs(i - j) == max_lag:
                grad[i] += h[j]
    
    # Apply gradient step
    h_new = h - lr * grad
    return project_to_sum_box(h_new, TARGET_SUM)
