import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points
n_upscale = 400
dx_upscale = 2.0 / n_upscale

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def evaluate_c5_upscale(h):
    """Evaluation at higher resolution"""
    corr = correlate(h, 1 - h, mode='full')
    return dx_upscale * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for max approximation"""
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exps = np.exp(alpha * shifted)
    return exps / np.sum(exps)

def compute_gradient(h, weights):
    """Compute gradient using FFT-based convolutions"""
    # Zero pad for full convolution
    n = len(h)
    n_fft = 2 * n - 1
    h_pad = np.pad(h, (0, n - 1), 'constant')
    w_pad = np.pad(weights, (0, n - 1), 'constant')
    
    # Forward convolution (h * w)
    f_conv = ifft(fft(h_pad) * fft(w_pad).conj())
    f_conv = np.real(f_conv[:n])
    # Backward convolution (h * w_rev)
    b_conv = ifft(fft(h_pad) * fft(w_pad[::-1]).conj())
    b_conv = np.real(b_conv[:n])
    # Gradient
    grad = 1.0 - f_conv - b_conv
    return grad

def projected_gradient_descent(h0, max_iter=2000, alpha_start=5.0, alpha_max=20.0, eta_start=0.5, eta_max=5.0):
    """Projected gradient descent with soft max surrogate"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha_start
    eta = eta_start
    last_improvement = 0
    
    for i in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        weights = softmax_weights(corr, alpha)
        grad = compute_gradient(h, weights)
        # Line search
        eta_test = eta
        h_trial = project_to_sum_box(h - eta_test * grad, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        
        while c5_trial >= best_c5 and eta_test > 1e-8:
            eta_test /= 2
            h_trial = project_to_sum_box(h - eta_test * grad, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
        
        h = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
            last_improvement = i
        
        # Schedule updates
        alpha = min(alpha_max, alpha * 1.01)
        eta = min(eta_max, eta * 1.02)
        
        if i - last_improvement > 200:
            break
            
    return best_h, best_c5

def binary_seed(n, target):
    """Generate binary seed with target ones"""
    indices = np.random.permutation(n)[:target]
    h = np.zeros(n)
    h[indices] = 1
    return h

def legendre_seed(n, target):
    """Generate Legendre-like sequence - simpler version"""
    # Generate approximate Legendre sequence with fixed length
    # We'll make a very simple binary pseudo-sequence
    h = np.zeros(n)
    # Just set the first target positions to 1
    h[:target] = 1
    np.random.shuffle(h)
    return h

    ]

        h_up = np.interp(np.linspace(0, 2, n_upscale),

            best_c5 = c5_up
            best_h = h_up.copy()

        indices = np.argsort(binary_h)[::-1][:n_points//2]
        binary_h.fill(0)
        binary_h[indices] = 1
        
        # Simulated annealing
        current_c5 = evaluate_c5(binary_h)
        T = 0.02 * current_c5
        alpha = 0.9995
        max_swaps = 10000
        swap_count = 0
        
        while T > 1e-6 and swap_count < max_swaps:
            if time.time() - start_time > 0.99 * budget_s:
                break

            i = np.random.randint(n_points)
            j = np.random.randint(n_points)

            # Swap
            temp = binary_h[i]
            binary_h[i] = binary_h[j]
            binary_h[j] = temp

            new_c5 = evaluate_c5(binary_h)
            if new_c5 < current_c5:
                current_c5 = new_c5
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    # Accept
                    current_c5 = new_c5
                else:
                    # Reject
                    binary_h[i] = temp
                    binary_h[j] = binary_h[i]
            
            T *= alpha
            swap_count += 1
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
    
    # Final polish
    if time.time() - start_time < 0.995 * budget_s:
        # Slight noise and a few iterations
        h_noisy = best_h + np.random.normal(0, 0.001, n_points)
        h_noisy = project_to_sum_box(h_noisy, TARGET_SUM)
        h_final, c5_final = projected_gradient_descent(h_noisy, 
                                                      max_iter=20,
                                                      alpha_start=20.0,
                                                      eta_start=0.1)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
