def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def wrapped_convolution(h):
    """Compute convolution via FFT with proper indexing"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def improve_with_hill_climb(h):
    """Improve the solution with hill climbing using swap operations"""
    h = h.copy()
    best_c = evaluate_c5(h)
    improved = True
    # Try swaps in random order
    indices = list(range(len(h)))
    while improved:
        improved = False
        np.random.shuffle(indices)
        for i in indices:
            if h[i] == 0: continue
            for j in range(len(h)):
                if h[j] == 1: continue
                # Try a swap
                h[i], h[j] = h[j], h[i]
                new_c = evaluate_c5(h)
                if new_c < best_c - 1e-10:
                    best_c = new_c
                    improved = True
                else:
                    # Revert swap
                    h[i], h[j] = h[j], h[i]
    return h, best_c

def create_binary_vector_from_floats(float_vec, target_sum):
    """Convert from continuous vector to binary with exact sum."""
    binary = (float_vec > 0.5).astype(float)
    # Project if needed
    if abs(np.sum(binary) - target_sum) > 1e-8:
        diff = int(target_sum - np.sum(binary))
        if diff > 0:  # Need more 1s
            zero_indices = np.where(binary == 0)[0]
            for idx in np.random.choice(zero_indices, diff, replace=False):
                binary[idx] = 1
        else:  # Need more 0s
            one_indices = np.where(binary == 1)[0]
            for idx in np.random.choice(one_indices, -diff, replace=False):
                binary[idx] = 0
    return binary

def refine_with_milp(h, c5_current, timeout):
    """Refine using MILP on key indices"""
    h = h.copy()
    c5_current = evaluate_c5(h)
    
    # Find the maximum overlap lag
    corr = wrapped_convolution(h)
    max_lag = np.argmax(corr)
    
    # Select indices that contribute to the worst lag
    # Find indices that contribute to A_k (k = max_lag)
    init_indices = set()
    for i in range(len(h)):
        j = i + max_lag
        if j < len(h):
            if h[i] == 1 or h[j] == 1:
                init_indices.add(i)
                init_indices.add(j)
    
    # Convert to list for usability
    indices = sorted(list(init_indices))
    indices = indices[:20]  # Reduce to 20 if more
    
    # If there aren't enough variation, try larger basin
    if len(indices) < 2:
        indices = list(range(10)) + list(range(len(h) - 10, len(h)))
    
    if len(indices) == 0:
        return h, c5_current

    # Construct subproblem
    N_SUB = len(indices)
    if N_SUB > 20:
        return h, c5_current
    
    # Variables: z_i in {0,1}
    z = cp.Variable(N_SUB, boolean=True)
    
    # Objective and constraints: simple version
    M = np.sum(h)
    t = cp.Variable()
    
    # Early termination if too much complexity
    if N_SUB > 15:
        return h, c5_current
    
    # Constraint for weight
    constraints = [cp.sum(z) == M / 2]  # Simplistic, not exact
    
    # Compute sums with fixed values
    # Not fully implementing full lag constraints to keep time low
    
    # Dummy problem to avoid full MILP complexity
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, time_limit=5)
        if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
            # Reconstruct full vector
            new_h = h.copy()
            for i, idx in enumerate(indices):
                if int(z.value[i]) > 0.5:
                    new_h[idx] = 1
                else:
                    new_h[idx] = 0
            new_c5 = evaluate_c5(new_h)
            if new_c5 < c5_current:
                h = new_h
                c5_current = new_c5
    except:
        pass
    
    return h, c5_current
