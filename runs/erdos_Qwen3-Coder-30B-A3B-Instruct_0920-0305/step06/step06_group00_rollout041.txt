def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    
    # Ensure sum is exactly target with small numerical correction
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1-1e-12)
        if free.any():
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    
    return result

def full_corr(a, b):
    """Compute full correlation with zero-padding to avoid wraparound"""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def gradient_max_soft(h, alpha):
    """Compute gradient of the weighted soft-max approximation of max correlation"""
    global TARGET_SUM
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    weights = w / (w_sum + 1e-15)
    
    grad = np.zeros_like(h)
    # Actual computation of gradient uses forward/backward shifts.
    for i in range(n):
        grad[i] = weights @ ((1 - h) * (i + np.arange(n) == i) - h * (i - np.arange(n) == 0))
    return grad
    
def line_search_update(h, grad, alpha, target_sum):

        h_new = project_to_sum_box(h - step * grad, target_sum)
        if evaluate_c5(h_new) < evaluate_c5(h) - 1e-12 * step:
            return h_new
        step *= 0.5
        if step < 1e-10:
            return h
    return h

    """Refine continuous solution into binary via simulated annealing"""
    start_t = time.time()
    h = initial_h.copy()
    curr_val = evaluate_c5(h)
    best_h = h.copy()
    best_val = curr_val
    
    def compute_delta(i, j, h):
        # Compute change in correlation due to swapping h[i] and h[j]
        corr = full_corr(h, 1 - h)
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        new_corr = full_corr(new_h, 1 - new_h)
        return new_corr.max() * dx - corr.max() * dx
    
    temp_schedule = lambda t: 0.01 * (0.99 ** t)
    
    step = 0
    while time.time() - start_t < remaining_time - 1:
        step += 1
        # Simple hill-climbing with acceptance probability
        i = np.random.choice(np.where(h > 0.5)[0])
        j = np.random.choice(np.where(h < 0.5)[0])
        delta = compute_delta(i, j, h)
        if delta < -1e-10:
            h[i], h[j] = h[j], h[i]
            curr_val += delta
            if curr_val < best_val:
                best_val = curr_val
                best_h = h.copy()
        elif np.random.rand() < np.exp(-delta / temp_schedule(step)):
            h[i], h[j] = h[j], h[i]
            curr_val += delta
    return best_h, best_val
