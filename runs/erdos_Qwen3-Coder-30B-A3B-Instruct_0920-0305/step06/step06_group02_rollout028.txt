import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def policy_gradient_optimization(h0, max_iter=2000, step_size=0.01):
    """Use policy gradient descent in a simple way to optimize"""
    h = h0.copy()
    
    # Store gradients for each point
    gradients = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if iteration % 100 == 0:
            # Recompute gradient approximately
            gradients = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                
                c5_plus = evaluate_c5(h_plus)
                c5_minus = evaluate_c5(h_minus)
                gradients[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        h += step_size * gradients
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Annealing schedule for step size
        step_size *= 0.99
        
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    TARGET_SUM = n_points / 2
    h0 = adaptive_sampling_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    phase1_end = start_time + 0.4 * budget_s

        for _ in range(100):
            if time.time() > phase2_end:
                break
            # Compute approximate gradient
            grad = np.zeros_like(current_h)

                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                
                h_minus = current_h.copy()
                h_minus[i] -= eps
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                
                grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)

            delta = 0.01 * grad
            proposed = current_h - delta
            proposed = project_to_sum_box(proposed, TARGET_SUM)
            new_c5 = evaluate_c5(proposed)
            
            if new_c5 < current_c5:
                current_h = proposed
                current_c5 = new_c5

                # Slight random perturbation if no improvement
                idx = np.random.randint(0, n_points)
                if np.random.random() < 0.1:
                    current_h[idx] += np.random.normal(0, 0.005)
                    current_h = project_to_sum_box(current_h, TARGET_SUM)
                    current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Phase 3: Final polishing with global stochastic search
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Introduce some randomness
        noise_strength = 0.01
        h_new = best_h + np.random.normal(0, noise_strength, n_points)
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Occasionally do a fine-tune with optimization
        if np.random.random() < 0.1:
            h_tuned = policy_gradient_optimization(best_h, max_iter=100)
            c5_tuned = evaluate_c5(h_tuned)
            if c5_tuned < best_c5:
                best_h = h_tuned.copy()
                best_c5 = c5_tuned
    
    return (best_h.tolist(), float(best_c5), n_points)
