def evaluate_c5(h):
    """Fast evaluation of C5 using correlation via FFT"""
    # Use the full convolution, no circular wraparound
    N = len(h) * 2 - 1
    h1 = np.concatenate([h, np.zeros(N - len(h))])
    h2 = np.concatenate([1 - h, np.zeros(N - len(h))])
    corr_full = fftconvolve(h1, h2, mode='full')[:N]
   _corr = corr_full[len(h)-1:]  # This is the valid convolution window
    return np.max(_corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search (max 50 iterations to be safe)
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, weights):
    """Compute gradient of soft max over correlation values."""
    # weights are already precomputed
    N = len(h)
    
    # Using FFT for fast convolution:
    # Forward convolution: c_f[i] = sum of w[ell] * h[i+ell]
    fw = np.zeros(2*N - 1)
    fw[:len(weights)] = weights
    fw = np.concatenate([fw, np.zeros(N - 1)])  # pad to make size equal
    
    h_padded = np.concatenate([h, np.zeros(N - 1)])
    c_f = fftconvolve(h_padded, fw, mode='valid')
    c_f = c_f[:N]
    
    # Backward convolution: c_b[i] = sum of w[ell] * h[i-ell]
    bw = fw[::-1]
    c_b = fftconvolve(h_padded, bw, mode='valid')
    c_b = c_b[:N]
    
    g = 1.0 - c_f - c_b
    return g

def softmax_weights(corr, alpha):
    """Compute softmax weights for the max over all lags"""
    # Ensure numerical stability
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    exps = np.exp(alpha * corr_shifted)
    weights = exps / np.sum(exps)
    return weights

def projected_gradient_descent(h_in, max_iter=2000):
    """Run projected gradient descent with soft max surrogate"""
    h = h_in.copy()
    best_h = h_in.copy()
    best_c5 = evaluate_c5(h)
    alpha = 5.0
    alpha_max = 20.0
    eta = 0.5
    eta_max = 5.0
    N = len(h)
    target_sum = N / 2
    t_start = time.time()
    
    for i in range(max_iter):
        if time.time() - t_start > 580:
            break
        corr = evaluate_c5(h) * np.ones(len(h) * 2 - 1)  # Simplification for gradient computation
        weights = softmax_weights(corr[N-1:], alpha)

        grad = compute_gradient(h, weights)

        # Back-tracking line search
        eta_test = eta
        while True:
            h_new = h - eta_test * grad
            h_new = project_to_sum_box(h_new, target_sum)

            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                break
            eta_test /= 2
            if eta_test < 1e-12:
                break

        # Update step size
        eta = min(eta_max, eta * 1.02)
        alpha = min(alpha_max, alpha * 1.01)

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):

    h_leg[indices] = 1.0

    # Run gradient descent on each seed with early stopping based on time
    t_end = time.time() + budget_s
    best_c5 = np.inf
    best_h = None
    h_to_start = [h_interp, h_rand, h_rand_bin, h_leg]

            best_c5 = c5_result
            best_h = h_result.copy()
    
    # Start again with perturbation of best solution if time remains
    if time.time() < t_end - 5 and best_h is not None:
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = project_to_sum_box(h_perturb, n_points / 2)
        h_result, c5_result = projected_gradient_descent(h_perturb, max_iter)
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result.copy()
    
    # Optional upsampling if we have time and achieved a good bound
    if time.time() < t_end - 5 and best_c5 < 0.381:
        n_up = 400
        dx_up = 2.0 / n_up
        x_up = np.linspace(0, 2, n_up)
        h_up = np.interp(x_up, np.linspace(0, 2, n_points), best_h)
        h_up = project_to_sum_box(h_up, n_up / 2)

        for _ in range(10):
            corr = evaluate_c5(h_up) * np.ones(n_up * 2 - 1)

            if evaluate_c5(h_new) < evaluate_c5(h_up):
                h_up = h_new.copy()
        best_c5 = evaluate_c5(h_up)
        best_h = h_up

    return (best_h.tolist(), float(best_c5), n_points)
