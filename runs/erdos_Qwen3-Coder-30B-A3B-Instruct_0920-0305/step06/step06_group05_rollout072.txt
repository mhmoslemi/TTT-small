import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate C5 using numpy correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = dx * np.max(corr)
    return c5

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_milp(n, time_limit):
    """Solve the binary ILP formulation for the Erdos problem"""
    M = n // 2  # Required number of ones
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)  # auxiliary variables for h_i * h_j
    t = cp.Variable()  # maximum overlap
    
    # Constraints
    constraints = [
        cp.sum(h) == M,  # exact mass constraint
        t >= 0  # t >= 0
    ]
    
    # Linking constraints for auxiliary variables
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(y[i, j] <= h[i])
            constraints.append(y[i, j] <= h[j])
            constraints.append(y[i, j] >= h[i] + h[j] - 1)
    
    # Lag constraints
    for k in range(1, n):
        lag_sum = 0
        for i in range(n - k):
            lag_sum += y[i, i + k]
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
            h_opt = np.rint(h.value).astype(int)
            t_val = t.value
            return h_opt, t_val
    except:
        pass
    
    return None, None
