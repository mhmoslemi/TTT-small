import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_smoothed_max(h, alpha=1.0):
    """Compute smoothed max of correlations for gradient descent"""
    n = len(h)
    corr = compute_corr_fft(h)
    # Convert to negative for minimization (minimize -max)
    neg_corr = -corr[:n]  # Use only first n values for clarity
    exp_neg_corr = np.exp(alpha * neg_corr)
    # Log-sum-exp for smooth max
    log_sum_exp = np.log(np.sum(exp_neg_corr))
    # Gradient using chain rule
    w = exp_neg_corr / np.sum(exp_neg_corr)
    # Compute gradient of log-sum-exp = sum(w * (gradient of neg_corr))
    # Gradient of -f_k = - (h_{i+k} + h_{i-k})
    grad = np.zeros(n)
    for k in range(n):
        if k == 0:
            # Gradient of h_i^2 wrt h_i is 2*h_i
            grad += -2 * w[k] * h
        else:
            # For k > 0, y_{i+k} + y_{i-k} 
            y_forward = np.concatenate([h[k:], np.zeros(k)])
            y_backward = np.concatenate([np.zeros(k), h[:-k]])
            grad += -w[k] * (y_forward + y_backward)
    return log_sum_exp, grad

def smoothed_max_descent(h_start, max_iter=1000, alpha_schedule=None):
    """Smoothed max descent optimization"""
    if alpha_schedule is None:
        alpha_schedule = np.logspace(0, 2, 10)  # 1 to 100
        
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for alpha in alpha_schedule:
        for i in range(max_iter):
            try:
                # Compute gradient
                _, grad = compute_smoothed_max(h, alpha)
                
                # Gradient step
                step_size = 0.001
                h_new = h - step_size * grad
                
                # Project to constraints
                h = project_to_sum_box(h_new, TARGET_SUM)
                
                # Evaluate
                current_c5 = evaluate_c5(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
            except Exception as e:
                break
                
    return best_h, best_c5

def local_refine(h_start):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    try:
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h_start, evaluate_c5(h_start)
