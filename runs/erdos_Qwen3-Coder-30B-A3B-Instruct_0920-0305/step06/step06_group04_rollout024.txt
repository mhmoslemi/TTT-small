import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50

    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def evaluate_c5_fft(h):
    """Evaluate C5 using FFT (as in original but verify)"""
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*len(h) - 1) * np.fft.fft(b, 2*len(h) - 1)))
    return np.max(corr)

def fourier_to_h(theta, n_points, n_fine=800):
    """Map Fourier coefficients to discretized h"""
    # Ensure n_points divides n_fine
    assert n_fine % n_points == 0
    scale = n_fine // n_points

    x_fine = np.linspace(0, 2, n_fine, endpoint=False)
    h = 0.5 + np.sum(
        [theta[2*i]*np.cos(np.pi*(i+1)*x_fine) + theta[2*i+1]*np.sin(np.pi*(i+1)*x_fine)
         for i in range(len(theta)//2)], axis=0)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Project to sum = TARGET_SUM (which is n_points/2)
    h = project_to_sum_box(h, TARGET_SUM)
    
    # Downsample to final size
    h = h[::scale]
    
    return h

def objective_fourier(theta, n_points):
    """Objective function to minimize using FFT"""
    h = fourier_to_h(theta, n_points)
    c5 = evaluate_c5_fft(h)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_c5 = float('inf')
    best_h = None
    n_points = 120
    n_fine = 800
    K = 12  # Number of harmonics per side
    
    # Total parameters
    n_params = 2 * K  # a_k and b_k for k=1..K
    
    # Bounds for each coefficient
    bounds = [(-0.5, 0.5)] * n_params

    # Phase A: Differential Evolution on low-dimensional space
    max_evals = int(0.5 * budget_s / (0.001))  # rough estimate
    maxiter = max(50, max_evals // 20)
    
    # Run evolutionary optimization
    res = differential_evolution(
        objective_fourier,
        bounds,
        args=(n_points,),
        maxiter=maxiter,
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        polish=True,
        workers=1
    )
    
    # Extract best solution and evaluate final bound
    theta_best = res.x
    h_best = fourier_to_h(theta_best, n_points, n_fine)
    c5_opt = evaluate_c5_fft(h_best)
    
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_best.copy()
        
    # Phase B: If we have some time left, do targeted refinement
    if (time.time() - start_time) < 0.9 * budget_s:
        # Refine with a smaller popsize and more generations
        # Take top 5 from the first run and refine them separately
        refs = []
        for _ in range(3):
            seed_new = np.random.randint(0, 1000)
            res_ref = differential_evolution(
                objective_fourier,
                bounds,
                args=(n_points,),
                maxiter=100,
                popsize=30,
                mutation=(0.5, 0.8),
                recombination=0.9,
                seed=seed_new,
                disp=False,
                polish=True,
                workers=1
            )
            theta_ref = res_ref.x
            h_ref = fourier_to_h(theta_ref, n_points, n_fine)
            c5_ref = evaluate_c5_fft(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Final polishing with local refinement (gradient-based)
    if (time.time() - start_time) < 0.95 * budget_s and best_h is not None:
        # Simple gradient ascent with clipped perturbations
        h_curr = best_h.copy()
        for iter_timed in range(200):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            corr = np.correlate(h_curr, 1 - h_curr, mode='full')
            c5_val = np.max(corr)
            # steepest descent step: adjust in direction that lowers C5
            # We directly estimate gradients with finite difference using
            # perturb-based approximations
            delta = 0.001
            h_new = h_curr.copy()
            best_improvement = 0
            best_idx = -1
            # Search over a few indices putatively on a large lag
            indices_to_try = np.argsort(corr[-len(corr)//2:])[:10]
            for i in range(len(h_curr)):
                h_trial = h_curr.copy()
                h_trial[i] = np.clip(h_trial[i] + delta, 0, 1)
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
                c5_trial = np.max(corr_trial)
                if c5_trial < c5_val:
                    improvement = c5_val - c5_trial
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_idx = i
                        h_new = h_trial
            if best_idx != -1:
                h_curr = h_new
            else:
                break  # No improvement found

        # Final check
        corr_final = np.correlate(h_curr, 1 - h_curr, mode='full')
        final_c5 = np.max(corr_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_curr.copy()

    # Post-process: Binary rounding + swaps
    if best_h is not None and (time.time() - start_time) < 0.98 * budget_s:
        # Perform a few rounds of greedy binary swaps
        # Make a binary copy
        h_bin = np.zeros_like(best_h)
        sorted_indices = np.argsort(best_h)[::-1]
        top_indices = sorted_indices[:len(best_h)//2]
        h_bin[top_indices] = 1.0
        h_bin = project_to_sum_box(h_bin, TARGET_SUM)
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        c5_bin = np.max(corr)
        best_c5 = c5_bin
        best_h = h_bin

        # Run one final fast local search using swaps
        h_swapped = h_bin.copy()
        for round_swaps in range(200):  # short number of rounds
            if (time.time() - start_time) > 0.98 * budget_s:
                break
            # Sample a "worst" lag (say, the one contributing most to max overlap)
            corr_swapped = np.correlate(h_swapped, 1 - h_swapped, mode='full')
            worst_lag_idx = np.argmax(corr_swapped)
            min_val = corr_swapped[worst_lag_idx]
            # Find one index that when swapped helps
            best_swap_gain = 0
            best_i, best_j = -1, -1
            for i in range(len(h_swapped)):
                if h_swapped[i] > 0.5:  # Is a one
                    for j in range(len(h_swapped)):
                        if h_swapped[j] < 0.5:  # Is a zero
                            # Test change
                            h_trial = h_swapped.copy()
                            h_trial[i] = 0
                            h_trial[j] = 1
                            corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
                            new_min = np.max(corr_trial)
                            gain = min_val - new_min
                            if gain > best_swap_gain:
                                best_swap_gain = gain
                                best_i, best_j = i, j

            if best_i != -1:
                h_swapped[best_i] = 0
                h_swapped[best_j] = 1
                h_swapped = project_to_sum_box(h_swapped, TARGET_SUM)
            else:
                break  # No more swaps help

        # Final check
        corr_final = np.correlate(h_swapped, 1 - h_swapped, mode='full')
        final_c5 = np.max(corr_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_swapped.copy()

    return (best_h.tolist(), float(best_c5), n_points)
