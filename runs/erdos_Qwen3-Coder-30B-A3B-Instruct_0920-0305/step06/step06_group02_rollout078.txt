import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_improvement(h0, max_iter=100):
    """Run a simple genetic algorithm to find better solutions"""
    pop_size = 50
    num_mutations = pop_size // 5  # Mutate top 20 percent
    
    h0 = h0.copy()
    
    def get_fitness(h):
        return -evaluate_c5(h)  # Minimize negative of C5 (maximize -C5)
    
    def mutate(h, rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < rate:
                # Small random change
                h[i] += np.random.normal(0, 0.01)
                h[i] = max(0, min(1, h[i]))
        return project_to_sum_box(h, TARGET_SUM)
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = h0 + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for _ in range(max_iter):
        # Evaluate all
        fitness_scores = [get_fitness(h) for h in population]
        sorted_indices = np.argsort(fitness_scores)[::-1]  # Sorted by fitness (descending)
        
        # Keep better solutions
        new_pop = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Extend with offspring
        for _ in range(pop_size//2):
            # Tournament selection
            parent1 = population[sorted_indices[np.random.randint(0, pop_size//2)]]
            parent2 = population[sorted_indices[np.random.randint(0, pop_size//2)]]
            
            # Simple crossover
            crossover_idx = np.random.randint(0, n_points)
            child = np.concatenate((parent1[:crossover_idx], parent2[crossover_idx:]))
            
            # Mutate
            child = mutate(child)
            new_pop.append(child)
        
        population = new_pop
    
    # Return the best from final population
    fitness_scores = [get_fitness(h) for h in population]
    best_idx = np.argmax(fitness_scores)
    return population[best_idx]
