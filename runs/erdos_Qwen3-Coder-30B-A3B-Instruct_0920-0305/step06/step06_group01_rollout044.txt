import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_soft_min(h, alpha):
    """Evaluate soft-minimum of auto-correlations"""
    s = correlate(h, h, mode='full')
    # Numerical stability trick
    s_max = np.max(-alpha * s)
    exp_s = np.exp(-alpha * s - s_max)
    w = exp_s / np.sum(exp_s)
    
    # Build symmetric kernel W (centered)
    W = np.zeros(2 * n_points - 1)
    for k in range(n_points):
        W[n_points - 1 + k] = w[k]
        W[n_points - 1 - k] = w[k]
    
    conv = np.convolve(h, W, mode='full')
    g = conv[n_points - 1 : n_points - 1 + n_points]
    g += w[n_points - 1] * h  # handle lag 0 (double counting)
    
    # evaluate soft min
    G = - (1.0 / alpha) * (np.log(np.sum(exp_s)) + s_max)
    return G, g

    TARGET_SUM = n_points / 2
    
    # Hyperparameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02
    step_inc = 1.05

    stagnation_counter = 0
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        # gradient step using soft minimum
        G, g = evaluate_soft_min(best_h, alpha)
        h_new = best_h + step * g
        
        # project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # evaluate true C5
        c5_new = evaluate_c5(h_new)

            best_h = h_new.copy()

        alpha = min(alpha * alpha_inc, alpha_max)

            alpha = 0.5
            stagnation_counter = 0
