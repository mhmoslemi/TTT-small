# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_step(h, target_sum):
    """Perform one local search step on h"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Try small perturbations at random points
    for _ in range(50):
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)

            continue

        delta = min(h[i], 1 - h[j], 0.01)

            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
