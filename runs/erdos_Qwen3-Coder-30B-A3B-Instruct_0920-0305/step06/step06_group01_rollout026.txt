import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT speed

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Step 1: FFT
        h_padded = np.pad(h, (0, L - n_points), mode='constant')
        H = np.fft.fft(h_padded)
        
        # Step 2: Compute target magnitude (flat off-DC)
        P = np.abs(H)**2
        P_target = np.mean(P[1:])  # Mean of non-DC powers
        M_target = np.full_like(np.abs(H), np.sqrt(P_target))
        M_target[0] = np.abs(H[0])  # Keep DC
        
        # Step 3: Blend magnitude with target
        M = (1 - lambda_current) * np.abs(H) + lambda_current * M_target
        
        # Step 4: Reconstruct with original phase
        H_new = M * np.exp(1j * np.angle(H))

        h_new_padded = np.fft.ifft(H_new).real

        h_new = project_to_sum_box(h_new, TARGET_SUM)

        if beta_current < 1.0:

            if polished_c5 < best_c5:
                best_h = polished_h
                best_c5 = polished_c5
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
