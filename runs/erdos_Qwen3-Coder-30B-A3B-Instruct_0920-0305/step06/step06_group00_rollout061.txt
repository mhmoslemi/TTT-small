import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    corr = full_corr(h, 1 - h)
    return dx * np.max(corr)

def full_corr(a, b):
    """Compute full correlation using FFT for speed."""
    n = a.size
    N = 2*n                       # zero-pad to avoid wrap-around
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]              # lag range -(n-1)…(n-1)

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low+high)/2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_max_soft(h, alpha):
    """Compute gradient of smooth approximation to max correlation."""
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    # Soft weights: softmax of scaled correlation
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w = w / w_sum
    # Gradient computation via two convolutions:
    # grad[i] = 1 - sum(w[l] * h[i+l]) - sum(w[l] * h[i-l])
    H = np.fft.rfft(h)
    W = np.fft.rfft(w)
    W_rev = np.fft.rfft(w[::-1])
    conv_fwd = np.fft.irfft(H * W, n=2*n-1)[:n]  # forward shift
    conv_bwd = np.fft.irfft(H * W_rev, n=2*n-1)[:n]  # backward shift
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def optimize_continuous(start_h, remaining_time, n, dx, target_sum):
    """Run projected gradient descent with backtracking line search."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    alpha = 5.0
    step_size = 0.01
    stagnation_count = 0
    max_stagnation = 50
    
    start = time.time()
    iteration = 0
    
    while time.time() - start < remaining_time and iteration < 5000:
        # Gradient with increasing α
        grad = gradient_max_soft(h, alpha)
        step = step_size
        
        # Backtracking line search
        h_trial = project_to_sum_box(h - step * grad, target_sum)
        trial_val = evaluate_c5(h_trial)
        while trial_val >= best_val - 1e-10 and step > 1e-6:
            step *= 0.5
            h_trial = project_to_sum_box(h - step * grad, target_sum)
            trial_val = evaluate_c5(h_trial)
        
        if step > 1e-6:
            h = h_trial
            val = trial_val
            if val < best_val - 1e-10:
                best_val = val
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        iteration += 1

        # Adaptive α: increase if improving
        if stagnation_count <= 10 and alpha < 20.0:
            alpha += 0.1
        elif stagnation_count > 30:
            # Occasionally perturb
            h += np.random.normal(0, 0.001, n)
            h = project_to_sum_box(h, target_sum)
            stagnation_count = 0
            
    return best_h, best_val

def binary_local_search(h_bin, time_budget, n, dx):
    """Refine binary solution using simulated annealing."""
    start = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_c5(best_h)
    temp = 0.01
    
    while time.time() - start < time_budget:
        # Random swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        
        # Evaluate new value
        val = evaluate_c5(cand)
        
        # Accept or not
        if val < best_val:
            best_val = val
            best_h = cand.copy()
        elif np.random.rand() < np.exp(-(val - best_val) / temp):
            best_h = cand.copy()
            
        temp *= 0.999  # Cooling
        
    return best_h, best_val
