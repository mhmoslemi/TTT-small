TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def coord_descent_step(h, max_iter=10):
    """Perform coordinate descent step"""
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for _ in range(max_iter):
        for i in range(n_points):
            # Try small perturbations
            for delta in [-0.05, -0.02, 0.02, 0.05]:
                h_trial = h.copy()
                h_trial[i] += delta
                # Ensure it's within bounds
                h_trial[i] = np.clip(h_trial[i], 0, 1)
                
                # Project to correct sum
                h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
        
        if best_c5 >= current_c5:
            break
        h = best_h.copy()
        current_c5 = best_c5
    
    return best_h, best_c5
