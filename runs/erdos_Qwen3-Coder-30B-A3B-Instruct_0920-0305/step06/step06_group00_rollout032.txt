import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Use a modified Nelder-Mead approach with adaptive steps
    # Create a simple landscape-aware local search
    for _ in range(max_iter):
        # Sample a random point to see its effect
        i = np.random.randint(0, n_points)
        
        # Try small perturbations around this point
        deltas = [0.005, -0.005, 0.01, -0.01]
        best_delta = 0.0
        best_local_c5 = best_c5
        
        for delta in deltas:
            h_test = h.copy()
            # Apply a delta to a point and neighboring points
            for j in range(max(0, i-2), min(n_points, i+3)):
                if abs(i - j) <= 2:
                    h_test[j] = np.clip(h_test[j] + delta * (1 - abs(i-j)/3), 0, 1)
            
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            
            if c5_test < best_local_c5:
                best_local_c5 = c5_test
                best_delta = delta
        
        if best_delta != 0:
            # Apply the best delta to neighbors of i
            for j in range(max(0, i-2), min(n_points, i+3)):
                if abs(i - j) <= 2:
                    h[j] = np.clip(h[j] + best_delta * (1 - abs(i-j)/3), 0, 1)
            h = project_to_sum_box(h, TARGET_SUM)
            
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = h.copy()
    
    return best_h, best_c5

def adaptive_perturb_search(h0, max_iter=1000):
    """Perturb with adaptive step size based on gradient estimation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Estimate gradient via finite differences
        grad = np.zeros(n_points)
        epsilon = 0.001
        
        # Compute a rough gradient estimate
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = np.clip(h[i] + epsilon, 0, 1)
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            
            h_minus = h.copy()
            h_minus[i] = np.clip(h[i] - epsilon, 0, 1)
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # For each position, move in the negative gradient direction
        step_size = 0.02
        h_new = h.copy()
        moved = False
        
        # Prefer moving more mass to areas with high negative gradient
        sorted_indices = np.argsort(-grad)
        for i in sorted_indices[:n_points//4]:  # Move in top 25% gradient directions
            old_val = h_new[i]
            new_val = np.clip(h_new[i] - grad[i] * step_size, 0, 1)
            delta = new_val - old_val
            
            if abs(delta) > 1e-6:
                h_new[i] = new_val
                moved = True
        
        if moved:
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        else:
            # If no significant move, perturb randomly with small strength
            h_pert = h + np.random.normal(0, 0.002, n_points)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
    
    return best_h, best_c5
