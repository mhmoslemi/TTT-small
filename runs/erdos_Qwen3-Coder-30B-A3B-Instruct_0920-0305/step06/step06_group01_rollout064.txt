import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_correlations(h):
    """Compute auto-correlations using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def evaluate_c5_exact(h):
    """Exact evaluation using FFT"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_kernel(w):
    """Build symmetric kernel from weights"""
    W = np.zeros(2 * n_points - 1)
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            W[n_points - 1 + k] = w[k]
        else:
            W[n_points - 1 + k] = w[-k]
    return W

def compute_gradient(h, w):
    """Compute gradient using convolution"""
    W = build_kernel(w)
    conv = np.convolve(h, W, mode="full")
    g = conv[n_points-1 : n_points-1 + n_points]
    g += w[n_points-1] * h  # add double contribution of lag 0
    return g

    h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
    
    # Set up optimization parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30.0
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    
    # Initialize variables
    h = h_interpolated.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5_exact(h)
    TARGET_SUM = n_points / 2
    stagnation_counter = 0
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Compute correlations
        S = compute_correlations(h)
        
        # Compute soft minimum weights (with numerical stability)
        max_S = np.max(-alpha * S)
        exp_S = np.exp(-alpha * S - max_S)
        exp_S_sum = np.sum(exp_S)
        w = exp_S / exp_S_sum
        w = w.real  # Ensure real numbers only

        # Compute gradient
        g = compute_gradient(h, w)
        
        # Candidate update
        h_new = h + step * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new C5
        c5_new = evaluate_c5_exact(h_new)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            if c5_new < best_c5:

            step = min(step * step_inc, 0.1)
            stagnation_counter = 0
        else:
            step = max(step * step_dec, 1e-4)
            stagnation_counter += 1
        
        # Anneal temperature
        alpha = min(alpha * alpha_inc, alpha_max)
        
        # Random perturbation occasionally
        if iteration % 200 == 0:
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5_exact(h_new)
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                best_h = h_new.copy()

        if stagnation_counter >= restart_interval:

        cons = {'type': 'eq', 'fun': constraint_eq}

            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, TARGET_SUM)
                final_c5 = evaluate_c5_exact(h_final)
                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
