n_points = 240

TARGET_SUM = n_points / 2

def evaluate_erdos_solution(h):
    """evaluate_erdos_solution is pre-imported function"""
    # This is a wrapper if it expects a list rather than np array
    corr = correlate(h, 1 - np.array(h), mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target using binary search"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target
    
    # Need to ensure a and b are on different sides of root
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    
    # Ensure exact sum due to numerical drift
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1 - 1e-12)
        if free.any():
            result[free] += diff / free.sum()
            result = np.clip(result, 0, 1)
    
    return result

def full_corr(a, b):
    """Zero-padded full correlation using FFT"""
    n = a.size
    N = 2*n                       # zero-pad to avoid circular wrap-around
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]              # lag range -(n-1)…(n-1)

def evaluate_max_corr(h):
    """Evaluate max of correlation of h and (1-h) using FFT"""
    corr = full_corr(h, 1 - h)
    return corr.max() * dx

def gradient_max_soft(h, alpha):
    """Compute gradient of smooth max of correlation sequence"""
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    
    # Compute softmax weights
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w = w / w_sum

    # Compute convolution terms
    conv_fwd = full_corr(h, w)          # forward part: sum_l w[l] * h[i+l]
    conv_bwd = full_corr(h, w[::-1])    # backward part: sum_l w[l] * h[i-l]

    # Gradient = 1 - conv_fwd - conv_bwd
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def adaptive_step_size(h, grad, current_val, max_iter=20):
    """Apply backtracking line search for step size"""
    step_size = 0.1
    eta = 0.5
    for _ in range(max_iter):
        h_new = project_to_sum_box(h - step_size * grad, TARGET_SUM)
        new_val = evaluate_max_corr(h_new)
        if new_val < current_val - 1e-10:
            return step_size
        step_size *= eta
    return step_size
