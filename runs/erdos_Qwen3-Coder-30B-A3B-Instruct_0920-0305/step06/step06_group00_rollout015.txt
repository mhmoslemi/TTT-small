import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Global settings
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points
n_points_upscale = 400
dx_upscale = 2.0 / n_points_upscale

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    # Full cross-correlation: h * (1 - h)
    h1 = np.concatenate([h, np.zeros(n_points)])
    h2 = np.concatenate([1 - h, np.zeros(n_points)])
    corr_full = fftconvolve(h1, h2[::-1], mode='full')
    correlation_values = corr_full[n_points-1:2*n_points-1]
    return np.max(correlation_values) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights from correlation values"""
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_values = np.exp(alpha * shifted)
    weights = exp_values / np.sum(exp_values)
    return weights

def compute_gradient(h, weights):
    """Compute the analytic gradient of the soft-max objective"""
    n = len(h)
    # Forward convolution of h with weights
    h1 = np.concatenate([h, np.zeros(n)])
    w1 = np.concatenate([weights, np.zeros(n)])
    conv_f = fftconvolve(h1, w1[::-1], mode='full')[:2*n-1]
    # Backward convolution of h with weights
    h2 = np.concatenate([h, np.zeros(n)])
    w2 = np.concatenate([weights[::-1], np.zeros(n)])
    conv_b = fftconvolve(h2, w2[::-1], mode='full')[:2*n-1]
    # Gradient is 1 - forward - backward
    grad = 1 - conv_f - conv_b
    return grad

def projected_gradient_descent(h0, max_iter=1000, alpha_init=5, alpha_max=20, step_size=0.5, eta_max=10):
    """Projected gradient descent with soft-max surrogate"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha_init
    
    for i in range(max_iter):
        # Evaluate full correlation
        h1 = np.concatenate([h, np.zeros(n_points)])
        h2 = np.concatenate([1 - h, np.zeros(n_points)])
        corr_full = fftconvolve(h1, h2[::-1], mode='full')
        correlation_values = corr_full[n_points-1:2*n_points-1]
        
        # Compute soft-max weights
        w = softmax_weights(correlation_values, alpha)
        
        # Compute gradient
        grad = compute_gradient(h, w)
        
        # Backtracking line search
        eta = step_size
        while True:
            h_trial = project_to_sum_box(h - eta * grad, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5 or eta < 1e-8:
                break
            eta /= 2
        
        h = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
        
        # Update parameters
        alpha = min(alpha_max, alpha * 1.01)
        step_size = min(eta_max, step_size * 1.02)
        
        if i % 50 == 0 and i > 0:
            if np.abs(alpha - alpha_max) < 0.5:
                break
                
    return best_h, best_c5

def binary_seed(h0, n_points, target_sum):
    """Create a binary seed based on the best continuous solution"""
    # Set the largest |target_sum| entries to 1, rest to 0
    indices = np.argsort(h0)[::-1][:int(target_sum)]
    binary_h = np.zeros_like(h0)
    binary_h[indices] = 1
    return project_to_sum_box(binary_h, target_sum)

def legendre_seed(n_points, target_sum):
    """Generate a binary Legendre sequence."""
    # Compute a prime number near n_points
    p = n_points + 1
    if p % 2 == 0:
        p += 1
    while True:
        is_prime = True
        for i in range(2, int(np.sqrt(p)) + 1):
            if p % i == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2
    
    # Generate the Legendre sequence
    legendre = np.zeros(p)
    for i in range(p):
        # By quadratic residue definition
        if (i * i) % p == 0:
            legendre[i] = 0
        elif (i * i) % p < (p + 1) // 2:
            legendre[i] = 1
        else:
            legendre[i] = -1
    
    # Adjust to all 0/1
    legendre = (legendre + 1) // 2

    # Now truncate to n_points and adjust to exactly target_sum ones
    if len(legendre) > n_points:
        legendre = legendre[:n_points]
    else:
        legendre = np.pad(legendre, (0, n_points - len(legendre)), constant_values=0)
    
    # Adjust count to target_sum
    count_ones = np.sum(legendre)
    if count_ones > target_sum:
        # need to turn some 1s into 0s
        idxs = np.where(legendre == 1)[0]
        to_be_zero = int(count_ones - target_sum)
        np.random.shuffle(idxs)
        legendre[idxs[:to_be_zero]] = 0
    elif count_ones < target_sum:
        # need to turn some 0s into 1s
        idxs = np.where(legendre == 0)[0]
        to_be_one = int(target_sum - count_ones)
        np.random.shuffle(idxs)
        legendre[idxs[:to_be_one]] = 1

    return legendre
