import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def initialize_with_pattern():
    """Initialize with a pattern that's known to work well"""
    # Try to create a piecewise constant function that might be good
    h = np.zeros(n_points)
    
    # Divide into 4 segments for flexibility
    seg_len = n_points // 4
    for i in range(4):
        start = i * seg_len
        end = (i+1) * seg_len if i < 3 else n_points
        if i % 2 == 0:
            h[start:end] = 0.5
        else:
            h[start:end] = 0.0
    
    # Adjust sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def mirror_reflect(h):
    """Perform a mirror reflection operation to explore the space"""
    # Flip the function and adjust to maintain constraint
    h_reflected = h[::-1]
    return project_to_sum_box(h_reflected, TARGET_SUM)

def adaptive_local_search(h, max_iter=100):
    """Adaptive local search that adjusts step size"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        # Try different optimization methods
        methods = ['SLSQP', 'L-BFGS-B']
        for method in methods:
            res = minimize(
                objective, 
                h, 
                method=method, 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                return project_to_sum_box(res.x, TARGET_SUM)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Initialize with guided scheme
    best_h = initialize_with_pattern()
    best_c5 = evaluate_c5(best_h)
    
    # Track best solution
    current_best_h = best_h.copy()
    current_best_c5 = best_c5
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 10:
        iter_count += 1
        
        # Create candidate via small random perturbation
        candidate_h = current_best_h + np.random.normal(0, 0.02, n_points)
        candidate_h = project_to_sum_box(candidate_h, TARGET_SUM)
        
        # Refine with optimization
        refined_h = adaptive_local_search(candidate_h, max_iter=50)
        refined_c5 = evaluate_c5(refined_h)
        
        if refined_c5 < current_best_c5:
            current_best_h = refined_h.copy()
            current_best_c5 = refined_c5
            
            # Update best if needed
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
        
        if iter_count % 20 == 0:
            # Occasional large perturbation to escape local optima
            perturb_h = current_best_h + np.random.uniform(-0.1, 0.1, n_points)
            perturb_h = project_to_sum_box(perturb_h, TARGET_SUM)
            # Try larger local refinement
            large_refined_h = adaptive_local_search(perturb_h, max_iter=100)
            large_refined_c5 = evaluate_c5(large_refined_h)
            
            if large_refined_c5 < current_best_c5:
                current_best_h = large_refined_h.copy()
                current_best_c5 = large_refined_c5
                
                if large_refined_c5 < best_c5:
                    best_h = large_refined_h.copy()
                    best_c5 = large_refined_c5
                    
    return (best_h.tolist(), float(best_c5), n_points)
