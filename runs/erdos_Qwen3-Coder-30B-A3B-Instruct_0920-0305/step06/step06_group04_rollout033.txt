import numpy as np

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
MAX_N = 250

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_auto_corr(h):
    n = len(h)
    corr = np.real(np.fft.ifft(np.fft.fft(h, 2*n - 1) * np.fft.fft(h, 2*n - 1)))
    return corr[:n]

def smooth_max_objective(h, alpha):
    n = len(h)
    f = compute_auto_corr(h)
    if alpha > 1e10:
        return np.min(f)
    # Compute smooth max and its gradient direction
    exp_terms = np.exp(-alpha * f)
    exp_terms /= np.sum(exp_terms)
    # Value of smooth max
    smooth_max = - np.log(np.sum(exp_terms)) / alpha
    # Compute the gradient of the smooth max w.r.t. h
    g = np.zeros(n)
    h2 = h**2
    for i in range(n):
        for l in range(n):
            if i < n-l:
                g[i] -= exp_terms[l] * h[i+l]
            if i >= l and l > 0:
                g[i] -= exp_terms[l] * h[i-l]
        g[i] *= 2.0
    return smooth_max, g

def line_search(h, g, alpha, max_backtracks=8):
    eta = 0.5
    for _ in range(max_backtracks):
        h_trial = project_to_sum_box(h - eta * g, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        c5 = evaluate_c5(h)
        if c5_trial < c5:
            return h_trial, eta
        eta *= 0.5
    return h, eta  # Return old one if no progress

def continuous_phase(h, alpha, remaining_time):
    start_time = time.time()
    max_iter = 1000
    for it in range(max_iter):
        if time.time() - start_time > remaining_time * 0.8:
            break
        try:
            smooth_max, g = smooth_max_objective(h, alpha)
            h_new, eta = line_search(h, g, alpha)
            if eta > 1e-6:
                h = h_new
            # Occasionally add stochastic perturbations
            if it % 20 == 0:
                h += np.random.normal(0, 0.005, len(h))
                h = project_to_sum_box(h, TARGET_SUM)
        except:
            pass
    return h

def binary_swap_phase(h, remaining_time):

    n = len(h)
    f = compute_auto_corr(h)
    h_bin = np.round(h).astype(int)
    while time.time() - start_time < remaining_time * 0.8:
        # Find worst lag
        min_f = np.min(f)
        worst_lag = np.argmin(f)
        
        # Compute effective ones and zeros
        # First, identify indices where h_bin[i+worst_lag] = 0 but h_bin[i] = 1
        candidates_one = []
        candidates_zero = []
        
        for i in range(n):
            if h_bin[i] == 1:
                if i + worst_lag < n and h_bin[i + worst_lag] == 0:
                    candidates_one.append(i)
                if i - worst_lag >= 0 and h_bin[i - worst_lag] == 0:
                    candidates_one.append(i)
            if h_bin[i] == 0:
                if i + worst_lag < n and h_bin[i + worst_lag] == 1:
                    candidates_zero.append(i)
                if i - worst_lag >= 0 and h_bin[i - worst_lag] == 1:
                    candidates_zero.append(i)
        
        if not candidates_one or not candidates_zero:
            break
            
        # Randomly pick one from each
        p = np.random.choice(candidates_one)
        q = np.random.choice(candidates_zero)
        
        # Compute delta f for each lag
        delta_f = np.zeros(n)
        for l in range(n):
            # Check how swapping affects lag l
            # h[p] becomes 0, h[q] becomes 1
            # effect is: (-1) * delta conditions for l=|q-p|
            # Note h_bin is binary, so it's just counting overlaps
            if l == 0:
                # Spectral value f_0
                delta_f[l] = 2 * (h_bin[p] - h_bin[q]) - 1
            else:
                # For non-zero lags
                if p + l < n:
                    delta_f[l] += h_bin[p + l] - h_bin[q + l]
                if p - l >= 0:
                    delta_f[l] += h_bin[p - l] - h_bin[q - l]
        
        # New f values
        new_f = f + delta_f
        
        # Check if better
        if np.min(new_f) > np.min(f):
            # Apply the swap
            h_bin[p] = 0
            h_bin[q] = 1
            f = new_f
        else:
            break  # No improvement found
        
    # Finalization (update)
    # Convert back to continuous values if needed
    return h_bin, np.min(f)
