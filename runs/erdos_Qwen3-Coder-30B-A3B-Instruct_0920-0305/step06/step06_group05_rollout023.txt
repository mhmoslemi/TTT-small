import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Create the correlation using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The maximum value in the correlation corresponds to the maximum overlap
    max_overlap = np.max(corr)
    return dx * max_overlap

def solve_milp(n, time_limit):
    """Solve the ILP formulation of the problem"""
    # Create decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()

    # Constraints
    # Exact mass constraint
    mass_constraint = cp.sum(h) == n // 2

    # Linking AND constraints
    and_constraints = []
    for i in range(n):
        for j in range(i+1, n):
            and_constraints.append(y[i, j] <= h[i])
            and_constraints.append(y[i, j] <= h[j])
            and_constraints.append(y[i, j] >= h[i] + h[j] - 1)

    # Lag constraints (excluding lag 0)
    lag_constraints = []
    for k in range(1, n):
        lag_expr = cp.sum([y[i, i+k] for i in range(n-k)])
        lag_constraints.append(t >= n//2 - lag_expr)

    # Objective
    objective = cp.Minimize(t)

    # Create and solve the problem
    prob = cp.Problem(objective, [mass_constraint] + and_constraints + lag_constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.CBC, verbose=False, time_limit=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h.value).astype(int)
            return h_sol, t.value
    except:
        pass
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h_start):
    """Perform continuous local refinement using SLSQP"""
    n = len(h_start)
    target_sum = n / 2
    dx = 2.0 / n

    def objective(h):
        # We want to minimize the maximum overlap, which is the same as minimizing t
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr)

    def constraint_eq(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n)]
    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        # Use SLSQP to refine the solution
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )

        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, target_sum)
            return h_refined
    except:
        pass
    return h_start
