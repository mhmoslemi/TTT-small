import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def full_corr(a, b):
    """Compute full correlation (zero-padded) via FFT"""
    n = a.size
    N = 2*n  # zero-pad to avoid wrap-around
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]  # lag range -(n-1) to (n-1)

def soft_max_grad(h, alpha=10.0):
    """Compute gradient of soft-max over correlations"""
    n = h.size
    corr = full_corr(h, 1-h)
    max_val = corr.max()
    w = np.exp(alpha * (corr - max_val))  # avoid large exponents
    w_sum = w.sum()
    w /= w_sum
    # two convolutions
    conv_fwd = full_corr(h, w)  # weight h forward for each lag
    conv_bwd = full_corr(h, w[::-1])  # backward
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def simple_line_search(h, grad, alpha=10.0, step=0.1, max_iter=10):
    """Simple backtracking line search"""
    val_current = evaluate_c5(h)
    for _ in range(max_iter):
        h_new = project_to_sum_box(h - step * grad, TARGET_SUM)
        val_new = evaluate_c5(h_new)
        if val_new < val_current - 1e-10:
            return h_new, val_new, step
        step *= 0.5
    return h, val_current, step

def optimize_continuous(h_start, remaining_time, n, dx, target_sum, max_iter=5000):
    """Run projected gradient descent on the soft-max approximation"""
    h = h_start.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    alpha = 5.0
    step = 0.1
    last_improvement = 0
    for i in range(max_iter):
        if time.time() > remaining_time:
            break
        grad = soft_max_grad(h, alpha)
        h_new, val_new, step = simple_line_search(h, grad, alpha, step)
        h = h_new
        if val_new < best_val:
            best_val = val_new
            best_h = h.copy()
            last_improvement = i
        # increase alpha if improving
        if i - last_improvement > 200 and alpha < 20.0:
            alpha += 1.0
    return best_h, best_val

def binary_refinement(h_cont, remaining_time, n, dx):
    """Refine the continuous solution to a binary one"""
    # Threshold and ensure exactly half ones
    if np.sum(h_cont) < n/2:
        indices = np.argsort(-h_cont)  # Descending order
        h_bin = np.zeros_like(h_cont)
        h_bin[indices[:int(n/2)]] = 1.0
    else:
        indices = np.argsort(-h_cont)
        h_bin = np.zeros_like(h_cont)
        h_bin[indices[:int(n/2)]] = 1.0
    # Fine tune with simulated annealing
    def evaluate_binary(h):
        corr = full_corr(h, 1-h)
        return dx * corr.max()

    best_h = h_bin.copy()
    best_val = evaluate_binary(best_h)
    start_time = time.time()
    temp = 0.02
    decay = 0.99
    
    while time.time() - start_time < remaining_time:
        # Random swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Perform swap
        new_h = best_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0
        new_val = evaluate_binary(new_h)
        delta = new_val - best_val
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = new_h.copy()
            best_val = new_val
        temp *= decay
    return best_h, best_val

def generate_legendre_binary(n):
    """Generate a binary sequence from Legendre symbol"""
    if n % 2 != 0:
        n = n - 1  # ensure even
    # For simplicity use a 0/1 approximation of the Legendre symbol
    # Generate a random binary with n/2 ones
    h = np.zeros(n)
    ones = np.random.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h
