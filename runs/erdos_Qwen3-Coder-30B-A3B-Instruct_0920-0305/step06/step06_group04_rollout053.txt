import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_f(h):
    """Compute all autocorrelations f_ell = sum h_i h_{i+ell}"""
    n = len(h)
    f = np.zeros(n)
    for ell in range(n):
        if ell == 0:
            f[ell] = np.sum(h * h)
        else:
            f[ell] = np.sum(h[:-ell] * h[ell:])
    return f

def compute_gradient(h, f, alpha):
    """Compute gradient of -log(sum(exp(-alpha * f)))"""
    n = len(h)
    # Compute softmax weights
    f_shifted = -alpha * f
    max_f = np.max(f_shifted)
    f_shifted -= max_f
    w = np.exp(f_shifted)
    w /= np.sum(w)
    
    # Compute gradient
    g = np.zeros(n)
    for ell in range(n):
        if ell == 0:
            g -= 2 * w[ell] * h
        else:
            # Add h_{i+ell} term
            if ell < n:
                g[:-ell] -= w[ell] * h[ell:]
            # Add h_{i-ell} term (i.e., h[i+ell])
            if ell < n:
                g[ell:] -= w[ell] * h[:-ell]
    return g

def line_search(h, grad, alpha, step_size=0.5, max_reductions=8):
    """Backtracking line search"""
    for _ in range(max_reductions):
        h_new = project_to_sum_box(h - step_size * grad, len(h) / 2)
        if evaluate_c5(h_new) < evaluate_c5(h):
            return h_new, step_size
        step_size *= 0.5
    return h, step_size * 0.5

def continuous_phase(h, alpha, max_time, time_left):
    """Smoothed gradient ascent phase"""
    start_t = time.time()
    while time.time() < start_t + max_time and time.time() < time_left - 1:
        f = compute_f(h)
        grad = compute_gradient(h, f, alpha)
        h, _ = line_search(h, grad, alpha)
        if np.allclose(h, 0.5):  # Probabilistic jitter if h = 0.5
            h += np.random.normal(0, 0.01, len(h))
            h = project_to_sum_box(h, len(h) / 2)
    return h

def binary_swap_phase(h_bin, n_points, time_left):
    """Binary swap refinement using a greedy approach"""
    # Compute initial f
    f = compute_f(h_bin)
    # Find worst lag
    ell_worst = np.argmin(f)
    min_f = np.min(f)

    while time.time() < time_left - 1:
        # Get candidate positions
        candidates_one = []
        candidates_zero = []
        for i in range(n_points - ell_worst):
            if h_bin[i] == 1 and h_bin[i + ell_worst] == 0:
                candidates_one.append(i)
            if h_bin[i] == 0 and h_bin[i + ell_worst] == 1:
                candidates_zero.append(i)
        if not candidates_one or not candidates_zero:
            break
        
        # Pick assignment
        p = np.random.choice(candidates_one)
        q = np.random.choice(candidates_zero)
        
        # Compute Δf
        delta_f = np.zeros(n_points)
        for ell in range(n_points):
            contribution = 0
            if ell != 0:
                if p + ell < n_points and h_bin[p + ell] == 1:
                    contribution -= 1
                if q + ell < n_points and h_bin[q + ell] == 1:
                    contribution += 1
                if p - ell >= 0 and h_bin[p - ell] == 1:
                    contribution -= 1
                if q - ell >= 0 and h_bin[q - ell] == 1:
                    contribution += 1
            else:
                contribution = -1 - 1  # both changes
            delta_f[ell] = contribution
        
        # Update if beneficial
        new_f = f + delta_f
        if np.min(new_f) > min_f:
            h_bin[p] = 0
            h_bin[q] = 1
            f = new_f
            min_f = np.min(new_f)
        else:
            # Small tolerance for stagnation
            if np.min(new_f) >= min_f - 1e-8:
                break
    
    return h_bin, c5_from_corr(compute_corr_fft(h_bin), n_points)
