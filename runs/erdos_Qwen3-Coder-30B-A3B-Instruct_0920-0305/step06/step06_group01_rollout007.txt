import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT speed

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Initial settings
    h = h_interpolated.copy()

    best_c5 = evaluate_c5(h)
    
    # parameters for alternating projection
    lambda_k = 0.05  # mixing weight from spectral flatness
    alpha_lambda = 1.02  # increase λ slowly
    beta = 0.5  # relaxation factor
    alpha_beta = 1.01  # increase beta slowly
    phase_reset_interval = 300  # iterations without improvement before reset
    restart_counter = 0
    no_improvement_count = 0
    TARGET_SUM = n_points / 2
    
    while time.time() - start_time < budget_s - 10:  # leave some time for final polishing
        # FFT
        h_padded = np.pad(h, (0, L - n_points), 'constant')
        H = np.fft.fft(h_padded)
        
        # compute and construct target power spectrum
        P = np.abs(H)**2
        P_target = np.mean(P[1:])  # mean off-DC power
        
        # create target magnitude (flat off-DC)
        M_target = np.full(L, np.sqrt(P_target))
        M_target[0] = np.abs(H[0])  # Keep DC magnitude unchanged
        
        # blend
        M = (1 - lambda_k) * np.abs(H) + lambda_k * M_target
        
        # keep phase
        H_new = M * np.exp(1j * np.angle(H))
        
        # inverse FFT
        h_new = np.fft.ifft(H_new).real[:n_points]
        
        # project onto feasibility
        h_cand = project_to_sum_box(h_new, TARGET_SUM)
        
        # relaxation
        h_cand = (1 - beta) * h + beta * h_cand
        
        # evaluate
        c5_candidate = evaluate_c5(h_cand)
        if c5_candidate < best_c5:
            best_h = h_cand.copy()
            best_c5 = c5_candidate
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # adapt parameters
        if lambda_k < 1.0 and no_improvement_count < 200:
            lambda_k = min(lambda_k * alpha_lambda, 1.0)
        if beta < 1.0:
            beta = min(beta * alpha_beta, 1.0)
        
        # phase reset if needed
        if no_improvement_count > phase_reset_interval and restart_counter < 5:

        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
