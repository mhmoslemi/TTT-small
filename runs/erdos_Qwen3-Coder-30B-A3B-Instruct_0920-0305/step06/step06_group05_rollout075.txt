import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import inf

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_TIME_S = 600

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def fast_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def exact_c5(h):
    """Exact C5 using integer convolution"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_float(h):
    """Convert binary vector to float vector"""
    return h.astype(float)

def float_to_binary(h, threshold=0.5):
    """Convert float vector to binary vector"""
    return (h >= threshold).astype(int)

def get_worst_lag(h):
    """Get the lag that produces the worst correlation"""
    corr = compute_corr_fft(h)
    return np.argmax(corr) - (len(h) - 1)

def generate_initial_h():
    """Generate initial binary vector and project to sum"""
    h = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_points)
    h = np.interp(x_new, x_orig, h)
    h = project_to_sum_box(h, TARGET_SUM)
    h_binary = float_to_binary(h)
    return h_binary

def optimize_binary_vector(binary_h):
    """Optimize the binary vector to minimize C5"""
    h = binary_h.copy()
    best_h = h.copy()
    best_c5 = fast_c5(h)
    max_iter = 1000
    
    for _ in range(max_iter):
        improved = False
        for i in range(n_points):
            if h[i] == 1:
                for j in range(n_points):
                    if h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                        new_c5 = fast_c5(h)
                        if new_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = new_c5
                            improved = True
                        h[i] = 1
                        h[j] = 0
                        if improved:
                            break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5

def solve_milp_on_subset(h_binary, time_limit_s=10):
    """Solve small MILP on critical indices"""
    worst_lag = get_worst_lag(h_binary)
    
    # Identify influential indices
    indices_influential = set()
    for i in range(n_points):
        if i + worst_lag < n_points:
            if h_binary[i] == 1 and h_binary[i + worst_lag] == 1:
                indices_influential.add(i)
                indices_influential.add(i + worst_lag)
        if i - worst_lag >= 0:
            if h_binary[i] == 1 and h_binary[i - worst_lag] == 1:
                indices_influential.add(i)
                indices_influential.add(i - worst_lag)
    
    if len(indices_influential) > 20:
        # Keep top 20 most influential
        def influence(i):
            if i + worst_lag < n_points:
                return h_binary[i] * (1 - h_binary[i + worst_lag])
            return 0
        influentials = sorted(indices_influential, key=influence, reverse=True)[:20]
        indices_influential = set(influentials)
    
    free_indices = sorted(list(indices_influential))
    
    if len(free_indices) == 0:
        return h_binary
    
    # Create MILP variables
    z = cp.Variable(len(free_indices), boolean=True)
    
    # Set fixed values from current binary solution
    fixed_vals = [h_binary[i] for i in free_indices]
    
    # Objective function (minimize max lag overlap)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(z) == TARGET_SUM - (np.sum(h_binary) - np.sum(fixed_vals))]
    
    # Add lag constraints
    for k in range(n_points):
        if k == 0:
            continue
        # Compute contribution to lag k
        lag_sum = 0
        for i, idx in enumerate(free_indices):
            if idx + k < n_points:
                # If both entries in the product contribute to this lag
                expr = cp.Minimize(z[i])
                if isinstance(expr, cp.Minimize):
                    lag_sum += expr.args[0]
                if h_binary[idx + k] == 1:
                    # Add this to lag_sum only if we're changing from 0 to 1
                    # For simplicity, we'll consider exact algebraic calculation
                    pass
        constraints.append(t >= TARGET_SUM - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem and solve
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, time_limit=time_limit_s)
        result = z.value
        for i, idx in enumerate(free_indices):
            if result[i] > 0.5:
                h_binary[idx] = 1
            else:
                h_binary[idx] = 0
        return h_binary
    except:
        return h_binary

def continuous_refinement(h_float):
    """Refine the solution with continuous optimization"""
    def evaluate_func(x):
        return fast_c5(x)
    
    # Need lower and upper bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint function
    def constraint_eq(x):
        return np.sum(x) - TARGET_SUM
    
    # Gradients for the system
    def gradient_func(x):
        # Get worst lag
        corr = compute_corr_fft(x)
        worst_lag = np.argmax(corr) - (len(x) - 1)
        gradient = np.zeros_like(x)
        for i in range(len(x)):
            if i + worst_lag < len(x):
                gradient[i] -= (1 - x[i + worst_lag])
            if i - worst_lag >= 0:
                gradient[i] -= (1 - x[i - worst_lag])
        return gradient
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            evaluate_func,
            h_float,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=gradient_func,
            options={'maxiter': 150, 'ftol': 1e-7, 'gtol': 1e-7}
        )
        if res.success:
            h_float = res.x
            h_float = project_to_sum_box(h_float, TARGET_SUM)
            return h_float
    except:
        pass
    return h_float

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Initialize binary vector
    h_binary = generate_initial_h()
    current_c5 = fast_c5(h_binary.astype(float))
    best_h = h_binary.copy()
    best_c5 = current_c5
    
    # Timeout budgeting
    remaining_time = lambda: budget_s - (time.time() - start_time)
    
    # Phase 2: Local hill climbing
    if remaining_time() > 10:
        h_binary, current_c5 = optimize_binary_vector(h_binary)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_binary.copy()
    
    # Phase 3: MILP refinement (small subset)
    if remaining_time() > 20:
        try:
            # Attempt MILP solve with small subset of variables
            h_binary = solve_milp_on_subset(h_binary.copy(), 10)
            current_c5 = fast_c5(h_binary.astype(float))
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_binary.copy()
        except:
            pass
    
    # Phase 4: Continuous refinement
    if remaining_time() > 10:
        h_float = binary_to_float(best_h)
        h_float = continuous_refinement(h_float)
        current_c5 = exact_c5(h_float)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = float_to_binary(h_float)
    
    # Final exact evaluation
    final_h = binary_to_float(best_h)  
    final_c5 = exact_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
