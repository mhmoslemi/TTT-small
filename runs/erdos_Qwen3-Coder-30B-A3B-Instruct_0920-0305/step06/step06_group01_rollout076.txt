import time

# Global variables
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # FFT length (next power of 2 for speed)

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def spectral_flatness_projection(h, lambda_mix, relaxation_beta=0.5):
    """Perform one iteration of the spectral flatness projection"""
    # 1. FFT of zero-padded h
    h_padded = np.zeros(L)
    h_padded[:n_points] = h
    H = np.fft.fft(h_padded)
    
    # 2. Compute target magnitude (flat off-DC)
    P = np.abs(H)**2
    P_target_mean = (np.sum(P[1:]) / (L - 1)) if L > 1 else 0
    
    # 3. Blend with target magnitude
    M_target = np.sqrt(P_target_mean)
    M_new = (1 - lambda_mix) * np.abs(H) + lambda_mix * M_target
    M_new[0] = np.abs(H[0])  # keep DC
    
    # 4. Reconstruct with original phases
    H_new = M_new * np.exp(1j * np.angle(H))
    
    # 5. Inverse FFT
    h_new = np.fft.ifft(H_new).real[:n_points]
    
    # 6. Project onto feasible set
    h_projected = project_to_sum_box(h_new, TARGET_SUM)
    
    # 7. Relaxation (optional) 
    if relaxation_beta > 0:
        h_projected = (1 - relaxation_beta) * h + relaxation_beta * h_projected

    return h_projected
