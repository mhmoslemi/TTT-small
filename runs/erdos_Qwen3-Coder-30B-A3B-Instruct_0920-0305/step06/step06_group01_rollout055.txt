import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_grad(h, alpha):
    """Compute gradient of soft minimum with parameter alpha"""
    # Compute auto-correlations with h * h for all lags
    corr_hh = correlate(h, h, 'full')
    # Use log-sum-exp trick for numerical stability
    max_term = np.max(-alpha * corr_hh)
    exp_terms = np.exp(-alpha * corr_hh - max_term)
    w = exp_terms / np.sum(exp_terms)
    
    # Build symmetric kernel for convolution
    W = np.zeros(3 * n_points - 2)
    W[n_points - 1:] = w[:n_points]  # First half of W corresponds to positive lags
    W[:n_points - 1] = w[n_points:][::-1]  # Second half corresponds to negative lags
    
    # Convolution of h with W
    conv = np.convolve(h, W, mode='full')
    
    # Extract gradient as middle portion
    g = conv[n_points - 1 : n_points - 1 + n_points]
    
    # Add double contribution of lag 0 (weight w[0])
    g += w[0] * h
    
    return g

    alpha = 0.5
    alpha_max = 30

    stagnation_counter = 0

        # Compute gradient of soft minimum
        grad = compute_soft_min_grad(h, alpha)
        
        # Update with gradient ascent
        h_new = h + step * grad

        h_new = project_to_sum_box(h_new, TARGET_SUM)

            stagnation_counter = 0

                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
