import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Minimize, Problem
import math

# Use low-dimensional search as fallback for small m (blocks)
def compute_c5_from_step_function(block_positions, block_heights, n_points=120):
    # Reconstruct h from blocks
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    prev_pos = 0
    for i, (pos, height) in enumerate(zip(block_positions, block_heights)):
        h[prev_pos:pos] = height
        prev_pos = pos
    h[prev_pos:] = block_heights[-1]  # Fill up last segment

    # Compute overlap integral
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

# Generate all possible partitions of N into m non-negative integers
def generate_partitions(n, m):
    if m == 1:
        yield [n]
    else:
        for i in range(n + 1):
            for rest in generate_partitions(n - i, m - 1):
                yield [i] + rest

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    target_sum = n_points / 2
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None

    # Phase 1: Try integer programming with MILP solver
    # Use CBC or GLPK as fallback (CVXPY wraps these)
    
    # Try solving MILP using cvxpy
    # Decision variables: h[i] binary, y[i,l] binary, lambda integer
    h_vars = cp.Variable(n_points, boolean=True)
    lambda_var = cp.Variable(integer=True)
    y_vars = {}
    constraints = []
    
    # Add weight constraint
    constraints.append(cp.sum(h_vars) == target_sum)
    
    # Add product constraints for all lags and pairs
    # To reduce number of variables, only compute y_{i,l} where i+ℓ < n
    for l in range(n_points):
        for i in range(n_points - l):
            # Create a new variable y[i,l] to represent h[i]*h[i+l]
            y = cp.Variable(boolean=True)
            y_vars[(i,l)] = y
            constraints.append(y <= h_vars[i])
            constraints.append(y <= h_vars[i+l])
            constraints.append(y >= h_vars[i] + h_vars[i+l] - 1)
            
    # Lag overlap constraints
    for l in range(n_points):
        overlapping_sum = cp.sum([y_vars[(i,l)] for i in range(n_points - l)])
        constraints.append(overlapping_sum >= lambda_var)
    
    # Add upper bound to lambda
    constraints.append(lambda_var <= target_sum)
    # Objective
    objective = cp.Maximize(lambda_var)
    prob = Problem(objective, constraints)
    
    # Try solving with different time limits
    remaining_time = 0.9 * budget_s
    try:
        prob.solve(solver='CBC', verbose=False, time_limit=remaining_time)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_vals = h_vars.value
            if h_vals is not None:
                h_vals = np.round(h_vals).astype(int)
                c5_val = 1 - dx * lambda_var.value
                best_c5 = float(c5_val)
                best_h = h_vals.tolist()
    except Exception as e:
        pass  # Fall back to method below
    
    # Phase 2: Fallback to block-wise search if no good solution from MILP
    if best_h is None or best_c5 > 0.3807:
        remaining_time = max(0, budget_s - (time.time() - start_time) - 10)
        
        # Try using a very small number of blocks
        m = 6  # number of blocks
        block_configs = []
        
        # Get all ways to split 120 into 6 integer parts
        for lengths in generate_partitions(n_points, m):
            if not any(part == 0 for part in lengths):
                # For simplicity, try only binary heights
                for heights in [[0, 1]] * m:
                    if sum(h * l for h, l in zip(heights, lengths)) == target_sum:
                        block_configs.append((lengths, heights))
        
        # Evaluate each configuration
        best_block_c5 = float('inf')
        best_block_h = None
        block_time = time.time() + remaining_time / 2  # Spend some time here
        
        for lengths, heights in block_configs:
            if time.time() > block_time:
                break
            try:
                # Find positions
                positions = [0]
                for l in lengths[:-1]:
                    positions.append(positions[-1] + l)
                
                c5_val = compute_c5_from_step_function(positions, heights, n_points)
                if c5_val < best_block_c5:
                    best_block_c5 = c5_val
                    h = np.zeros(n_points)
                    prev_idx = 0
                    for l, h_val in zip(lengths, heights):
                        h[prev_idx:prev_idx+l] = h_val
                        prev_idx += l
                    best_block_h = h.tolist()
            except Exception:
                continue
        
        if best_block_h is not None:
            best_h = best_block_h
            best_c5 = best_block_c5
            
    return (best_h, float(best_c5), n_points)
