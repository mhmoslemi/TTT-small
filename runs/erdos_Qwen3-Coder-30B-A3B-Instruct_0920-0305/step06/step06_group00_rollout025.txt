def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    h = np.asarray(h)
    # Compute full correlation using FFT (zero-padded)
    N = 2 * n_points - 1
    h_pad = np.pad(h, (0, n_points - 1), mode='constant')
    h_conj_pad = np.pad(1 - h, (0, n_points - 1), mode='constant')
    corr = fftconvolve(h_pad, h_conj_pad[::-1], mode='valid')
    return np.max(corr) * dx

def project_to_box_and_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def make_legendre_binary(n, target):
    """Generate a Legendre binary sequence and adjust to have exactly target ones"""
    # Find smallest odd prime >= n
    def is_prime(p):
        if p < 2:
            return False
        if p == 2:
            return True
        for i in range(3, int(p**0.5) + 1, 2):
            if p % i == 0:
                return False
        return True

    p = n + 1
    while not is_prime(p):
        p += 2
    
    # Generate Legendre sequence (length p)
    legendre = np.zeros(p)
    for i in range(p):
        legendre[i] = ((i * i) % p) < p/2
    
    # Convert to binary ref sequence (0,1)
    ref = (legendre * 2 - 1).astype(int)
    
    # Trim to n points
    if len(ref) > n:
        ref = ref[:n]
    elif len(ref) < n:
        ref = np.pad(ref, (0, n - len(ref)), mode='constant')
    
    # Adjust so that exactly 'target' ones
    count_ones = np.sum(ref)
    indices_zeros = np.where(ref == 0)[0]
    indices_ones = np.where(ref == 1)[0]

    if count_ones > target:
        # Remove zeros to make it fit
        for i in range(count_ones - target):
            idx = indices_zeros[i]
            ref[idx] = 1
    elif count_ones < target:
        # Add zeros
        for i in range(target - count_ones):
            idx = indices_ones[i]
            ref[idx] = 0

    return ref

def soft_max_corr(h, alpha=5.0):
    """Compute soft max approximation of C5"""
    # Compute full correlation using FFT
    h_padded = np.pad(h, (0, n_points - 1), mode='constant')
    h_conj_padded = np.pad(1 - h, (0, n_points - 1), mode='constant')
    corr = fftconvolve(h_padded, h_conj_padded[::-1], mode='valid')
    # Soft max of correlations
    max_corr = np.max(corr)
    corr_shift = corr - max_corr  # Avoid overflow
    exp_corr = np.exp(alpha * corr_shift)
    weights = exp_corr / np.sum(exp_corr)
    return np.sum(weights * corr) * dx

def compute_gradient(h, weights):
    """Compute gradient using FFT-based convolution"""
    # Apply forward and backward convolution to weights
    h_padded = np.pad(h, (0, n_points - 1), mode='constant')
    weights_padded = np.pad(weights, (0, n_points - 1), mode='constant')
    
    # Compute I - conv(h, w_rev) - conv(h_rev, w)
    forward_conv = fftconvolve(h_padded, weights[::-1], mode='valid')[:n_points]
    backward_conv = fftconvolve(h_padded[::-1], weights, mode='valid')[:n_points]
    grad = 1.0 - forward_conv - backward_conv    
    return grad * dx

def gradient_descent_step(h, alpha=5.0, eta=0.5, max_iter=50):
    """Perform one gradient descent iteration"""
    past_h = h.copy()
    past_c5 = evaluate_c5(h)
    
    for iter in range(max_iter):
        corr = fftconvolve(
            np.pad(h, (0, n_points-1), mode="constant"), 
            np.pad(1-h, (0, n_points-1), mode="constant")[::-1], 
            mode="valid")
        weights = np.exp(alpha * (corr - np.max(corr))) / np.sum(np.exp(alpha * (corr - np.max(corr))))
        grad = compute_gradient(h, weights)
        h_trial = h - eta * grad
        h_trial = project_to_box_and_sum(h_trial, target_sum)
        
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < past_c5:
            h = h_trial.copy()
            past_c5 = c5_trial
            # reduce step size if no improvement
            if iter > 1:
                eta *= 0.95
        else:
            # backtracking line search
            while eta > 1e-8 and iter < max_iter:
                eta *= 0.5
                h_trial = h - eta * grad 
                h_trial = project_to_box_and_sum(h_trial, target_sum)
                c5_trial = evaluate_c5(h_trial)
                if c5_trial < past_c5:
                    h = h_trial.copy()
                    past_c5 = c5_trial
                    break
            if iter >= max_iter or eta < 1e-8:
                break

    return h, past_c5
