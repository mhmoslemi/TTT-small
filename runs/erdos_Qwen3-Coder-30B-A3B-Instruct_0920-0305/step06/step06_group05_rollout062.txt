import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations using full mode
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap corresponds to the maximum correlation
    max_overlap = np.max(corr)
    return dx * max_overlap

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary variables"""
    # Parameters
    M = n // 2  # Required number of ones
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Auxiliary variables for AND operations
    # We'll compute these as needed in the constraints
    
    # Constraints
    constraints = [cp.sum(h) == M]
    
    # Add symmetry breaking
    constraints.append(h[0] == 1)
    
    # Compute all lag constraints
    for k in range(1, n):
        # Sum of y_{i,i+k} for i = 0, ..., n-1-k
        lag_sum = 0
        for i in range(n - k):
            # Introduce a binary variable y_{i,i+k} to represent h_i AND h_{i+k}
            y = cp.Variable(boolean=True)
            # The logical AND constraints
            constraints.extend([
                y <= h[i],
                y <= h[i + k],
                y >= h[i] + h[i + k] - 1
            ])
            lag_sum += y
        
        # The constraint t >= M - lag_sum
        constraints.append(t >= M - lag_sum)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, seconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_USING_APPROXIMATION]:
            h_opt = np.rint(h.value).astype(int)
            t_val = t.value
            return h_opt, t_val
        else:
            # Try to get a feasible solution even if not optimal
            if prob.status == cp.SOLUTION_PRESENT:
                h_opt = np.rint(h.value).astype(int)
                t_val = t.value
                return h_opt, t_val
    except Exception:
        pass
    
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Perform local continuous refinement around binary solution
    """
    # Convert to floating point with values 0 or 1
    h = h.astype(float)
    bounds = [(0, 1) for _ in range(len(h))]
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    def constraint_eq(h):
        return np.sum(h) - len(h) / 2
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, len(h)/2)
            return h
    except:
        pass
    return h
