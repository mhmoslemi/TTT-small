def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = h  # we want h * h, not h * (1-h)
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_overlap_fft(h):
    """Compute correlation between h and (1-h) using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_overlap_fft(h)
    return c5_from_corr(corr, n_points)

def evaluate_c5_binary(h):
    """Exact evaluation for binary vectors using discrete convolution"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search for lambda
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_project_to_sum(x, target_sum):
    """Project binary vector to exact target sum"""
    ones_pos = np.where(x > 0.5)[0]
    zeros_pos = np.where(x < 0.5)[0]
    
    # We need to convert some 1s to 0s and some 0s to 1s
    current_sum = len(ones_pos)
    diff = target_sum - current_sum
    
    if diff > 0:  # Need more 1s
        # Convert zeros to 1s
        for _ in range(diff):
            if len(zeros_pos) > 0:
                i = zeros_pos[0]
                x[i] = 1
                zeros_pos = zeros_pos[1:]
                ones_pos = np.append(ones_pos, i)
            else:
                break
    elif diff < 0:  # Need fewer 1s
        # Convert 1s to 0s
        for _ in range(-diff):
            if len(ones_pos) > 0:
                i = ones_pos[-1]
                x[i] = 0
                ones_pos = ones_pos[:-1]
                zeros_pos = np.append(zeros_pos, i)
            else:
                break
    
    return x

def make_binary(h):
    """Convert float vector to binary vector with same sum"""
    h = np.clip(h, 0, 1)
    temp = h.copy()
    target = TARGET_SUM
    temp = binary_project_to_sum(temp, target)
    return temp

def swap_move(h):
    """Performs a swap move in binary vector"""
    # Selection of a 1-index and a 0-index
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h
    
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    
    h_new = h.copy()
    h_new[i] = 0
    h_new[j] = 1
    
    return h_new

def block_move(h):
    """Performs a block move"""
    # Select block size and position
    block_len = np.random.choice([2, 3, 4])
    start_pos = np.random.randint(0, n_points - block_len + 1)
    
    h_new = h.copy()
    # Flip the block
    for i in range(start_pos, start_pos + block_len):
        h_new[i] = 1 - h_new[i]
    
    # Repair if necessary
    current_sum = np.sum(h_new)
    diff = TARGET_SUM - current_sum
    
    if diff != 0:
        # Flip diff bits to repair
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        
        if diff > 0:  # Need more 1s
            for _ in range(int(diff)):
                if len(zeros) > 0:
                    i = np.random.choice(zeros)
                    h_new[i] = 1
                    zeros = np.delete(zeros, np.where(zeros == i)[0])
        else:  # Need fewer 1s
            for _ in range(int(-diff)):
                if len(ones) > 0:
                    i = np.random.choice(ones)
                    h_new[i] = 0
                    ones = np.delete(ones, np.where(ones == i)[0])
    
    return h_new

def acceptance_probability(delta_e, T):
    """Compute acceptance probability for simulated annealing"""
    if delta_e <= 0:
        return 1.0
    return np.exp(-delta_e / T)

def run_simulated_annealing(h0, T0, ALPHA, MIN_TEMP, time_budget, start_time):
    """Run Simulated Annealing search for binary vectors"""
    h_best = h0.copy()
    c5_best = evaluate_c5_binary(h_best)
    h_current = h0.copy()
    c5_current = evaluate_c5_binary(h_current)
    
    T = T0
    moves_since_temp = 0
    
    elite_pool = []
    elite_size = 5
    
    while True:
        elapsed_time = time.time() - start_time
        if elapsed_time >= time_budget - 5:
            break
            
        if np.random.random() < 0.15:
            # Block move
            h_new = block_move(h_current)
        else:
            # Swap move
            h_new = swap_move(h_current)
        
        # Evaluate new solution
        c5_new = evaluate_c5_binary(h_new)
        
        delta_e = c5_new - c5_current
        
        if acceptance_probability(delta_e, T) > np.random.random():
            h_current = h_new
            c5_current = c5_new
            
            # Update best solution
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        
        moves_since_temp += 1
        if moves_since_temp >= SWEEPS_PER_TEMP:
            T *= ALPHA
            moves_since_temp = 0
            
            # Store in elite pool
            if len(elite_pool) < elite_size:
                elite_pool.append((h_best.copy(), c5_best))
                elite_pool.sort(key=lambda x: x[1])
            else:
                if c5_best < elite_pool[-1][1]:
                    elite_pool[-1] = (h_best.copy(), c5_best)
                    elite_pool.sort(key=lambda x: x[1])
            
            # Stop if temperature is too low
            if T < MIN_TEMP:
                break
    
    # Return the best found
    return h_best, c5_best

def hill_climb(h_start, max_iterations=1000):
    """Perform hill climbing with exhaustive search for best swap"""
    h = h_start.copy()
    c5_current = evaluate_c5_binary(h)
    improved = True
    
    iterations = 0
    while improved and iterations < max_iterations:
        improved = False
        best_h = h.copy()
        best_c5 = c5_current
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        
        for i in ones_pos:
            for j in zeros_pos:
                h_test = h.copy()
                h_test[i] = 0
                h_test[j] = 1
                
                c5_test = evaluate_c5_binary(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test
                    improved = True
                    
        if improved:
            h = best_h
            c5_current = best_c5
            iterations += 1
            
    return h, c5_current

def run_slsqp_refinement(h_start, target_sum):
    """Refine to continuous domain"""
    h = h_start.copy()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1-h_vec, mode='full')
        return np.max(corr) * dx
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-7, 'gtol': 1e-7})
        if res.success:
            h_refined = res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = project_to_sum_box(h_refined, target_sum)
            return h_refined
    except:
        pass
        
    return h
