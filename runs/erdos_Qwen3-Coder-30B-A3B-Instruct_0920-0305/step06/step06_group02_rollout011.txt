import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def compute_min_overlap_conv(h):
    """Fast computation of minimum overlap using convolution"""
    # Use convolution to compute overlaps for all shifts
    conv = np.correlate(h, h, mode='full')
    # The minimum overlap corresponds to the minimum value in this array
    # For binary h, conv[k] = sum_i h[i] * h[i+k] for shift k.
    # We want the minimum of this over all shifts (including negative)
    return np.min(conv)

def initialize_binary_solution():
    """Initialize a binary solution with exactly half ones"""
    h = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h[ones_indices] = 1
    return h, list(ones_indices)

def initialize_from_best():
    """Initialize using precomputed best values"""
    h = np.array(initial_h_values)
    if len(h) != n_points:
        # Resample to match n_points
        x_ref = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_ref, h)
        # Convert to binary
        h = (h > 0.5).astype(int)
    else:
        h = (h > 0.5).astype(int)
    ones_indices = [i for i in range(n_points) if h[i] == 1]
    return h, ones_indices

def update_convolution(conv, ones, i, j, n):
    """Fast update of convolution after a swap"""
    # Remove contributions of i (1->0)
    for k in ones:
        shift = k - i
        idx = shift + n - 1
        if k == i:
            conv[idx] -= 1
        else:
            conv[idx] -= 1
            conv[-shift + n - 1] -= 1
    
    # Remove i from ones list
    ones.remove(i)
    
    # Add contributions of j (0->1)
    for k in ones + [j]:
        shift = k - j
        idx = shift + n - 1
        if k == j:
            conv[idx] += 1
        else:
            conv[idx] += 1
            conv[-shift + n - 1] += 1
    
    # Add self-contribution
    conv[n - 1] += 1
    
    # Add j to ones
    ones.append(j)
    
    return conv, ones

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Use the best known initial solution
    h, ones_indices = initialize_from_best()
    
    # Convert to float
    h_float = h.astype(float)
    current_c5 = evaluate_c5(h_float)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_float.copy()
    
    iterations_per_restart = 10000
    max_restarts = 10
    
    # Simulated Annealing with fast updates
    for restart in range(max_restarts):
        if time.time() > start_time + 0.9 * budget_s:
            break
            
        # Start with a fresh binary solution
        h, ones_indices = initialize_binary_solution()
        h_float = h.astype(float)
        
        # Compute correlation efficiently
        conv = np.correlate(h, h, mode='full')
        cur_min_overlap = np.min(conv)
        cur_c5 = (TARGET_SUM - cur_min_overlap) * dx
        
        # Simulated Annealing parameters
        T = 0.1  # Start temperature
        cooling_rate = 0.99999
        
        i_iter = 0
        while i_iter < iterations_per_restart:
            if time.time() > start_time + budget_s * 0.95:
                break
            
            # Choose two random indices: one that's 1, one that's 0
            i = random.choice(ones_indices)
            zero_indices = [j for j in range(n_points) if h[j] == 0]
            if not zero_indices:
                break
            j = random.choice(zero_indices)
            
            # Create a copy of current convolution
            conv_new = conv.copy()
            
            # Temporary update
            old_ones = ones_indices.copy()
            old_ones.remove(i)
            old_ones.append(j)
            
            # Apply the swap to convolution
            for k in old_ones:
                shift = k - i
                idx = shift + n_points - 1
                if k == i:
                    conv_new[idx] -= 1
                else:
                    conv_new[idx] -= 1
                    conv_new[-shift + n_points - 1] -= 1
                    
            for k in old_ones:
                shift = k - j
                idx = shift + n_points - 1
                if k == j:
                    conv_new[idx] += 1
                else:
                    conv_new[idx] += 1
                    conv_new[-shift + n_points - 1] += 1
                    
            conv_new[n_points - 1] += 1
            
            # Update overlap and c5
            new_min_overlap = np.min(conv_new)
            new_c5 = (TARGET_SUM - new_min_overlap) * dx
            
            delta = cur_c5 - new_c5  # Negative if worse
            
            if delta > 0 or random.random() < np.exp(delta / T):
                # Accept the move
                conv = conv_new.copy()
                cur_c5 = new_c5
                cur_min_overlap = new_min_overlap
                
                # Update ones_indices
                ones_indices.remove(i)
                ones_indices.append(j)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_float.copy()
            else:
                # Move rejected
                pass
                
            T *= cooling_rate
            i_iter += 1
            
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    else:
        # If we didn't find any better solution, use the initial discretization
        best_h = initial_h_values.copy()
        best_h = [float(x) for x in best_h]
        best_c5 = evaluate_c5(np.array(best_h))
    
    return (best_h, float(best_c5), n_points)
