import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_initial_solution():
    """Create an improved initial solution based on handcrafted construction"""
    # Start with a piecewise constant construction
    h = np.zeros(n_points)
    
    # Create a construction with high density in a central region and low elsewhere
    # This mimics an optimal arrangement: high overlap on one side with low density on the other
    
    # 3 segments: low at start, high in middle, low at end
    segments = []
    segment_points = [0.2, 0.8, 1.2, 1.8]  # x locations of segment boundaries
    
    # First segment: 0 to 0.2
    segments.append((0, 0.2, 0.0))
    # Second segment: 0.2 to 1.2 (high density)
    segments.append((0.2, 1.2, 0.8))
    # Third segment: 1.2 to 1.8
    segments.append((1.2, 1.8, 0.0))
    
    # Fill the h array with piecewise constant values
    x_pts = np.linspace(0, 2, n_points)
    for i in range(n_points):
        x = x_pts[i]
        for start, end, val in segments:
            if start <= x < end:
                h[i] = val
                break
        if x >= 1.8:
            h[i] = 0.0
    
    # Add small noise to help escape local optima
    h += np.random.normal(0, 0.001, n_points)
    h = np.clip(h, 0, 1)
    
    return project_to_sum_box(h, n_points / 2)

def guided_local_search(h0, max_iter=500):
    """Refine solution with guided local search focusing on max correlation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Compute correlation and find max lag
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        
        # For the maximizing lag, compute better move estimates
        # Try small moves that reduce the max correlation
        
        # Simple gradient approximation using first differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            for j in range(n_points):
                lag = abs(i - j)
                if lag == max_lag_idx:
                    grad[i] += h[j]  # weighting by density at j
        
        # Heuristic: move mass from low-gradient areas toward high-gradient
        pos_to_move_from = np.argmin(grad)
        pos_to_move_to = np.argmax(grad)
        
        # Small step for now
        delta = min(h[pos_to_move_from], 1 - h[pos_to_move_to], 0.005)
        if delta > 1e-6:
            h[pos_to_move_from] -= delta
            h[pos_to_move_to] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Create a better starting point
    best_h = create_initial_solution()
    best_c5 = evaluate_c5(best_h)
    TARGET_SUM = n_points / 2
    
    # Phase 1: Better guided local search with enhanced strategies
    phase1_end = start_time + 0.3 * budget_s
    for i in range(30):
        if time.time() >= phase1_end:
            break
            
        # Random perturbations
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
        
        h_refined, c5_refined = guided_local_search(h_perturbed, 400)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Phase 2: Gradient-based optimization with direct updates
    phase2_end = start_time + 0.8 * budget_s
    while time.time() < phase2_end:
        # Take multiple steps of gradient-based descent
        h_current = best_h.copy()
        for step in range(20):
            if time.time() > phase2_end:
                break
            
            # Compute directional derivatives via finite differences
            eps = 1e-4
            grad = np.zeros(n_points)
            
            # For each pixel i, estimate its partial derivative wrt height at i
            orig_c5 = evaluate_c5(h_current)
            
            for i in range(n_points):
                h_test = h_current.copy()
                h_test[i] += eps
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                new_c5 = evaluate_c5(h_test)
                grad[i] = (new_c5 - orig_c5) / eps
            
            # Move in the negative gradient direction with adaptive step
            step_size = min(0.01, 0.1 / (np.linalg.norm(grad) + 1e-8))
            h_new = h_current - step_size * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                h_current = h_new.copy()
            else:
                h_current = h_new.copy()  # Accept anyway for exploration randomness

    # Phase 3: Refinement
    while time.time() < start_time + budget_s:
        h_refined, c5_refined = guided_local_search(best_h, 150)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
