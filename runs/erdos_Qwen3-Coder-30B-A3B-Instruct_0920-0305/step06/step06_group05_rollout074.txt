# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_TIME_S = 600

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def fast_exact_eval(h):
    """Exact evaluation using integer convolution"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def binary_from_float(fvals):
    """Convert float vector to binary vector keeping sum"""
    # Round elements to nearest binary values
    rounded = np.round(fvals)
    # Project to sum = TARGET_SUM
    fsum = np.sum(fvals)
    bsum = np.sum(rounded)
    diff = TARGET_SUM - bsum
    if abs(diff) > 0.1:  # Shouldn't happen with good float values  
        # Adjust to get exact sum
        idxs = np.argsort(np.abs(fvals - rounded))
        for i in range(int(abs(diff))):
            if diff > 0:
                rounded[idxs[i]] = 1.0
            else:
                rounded[idxs[i]] = 0.0
    return rounded

# Hill Climbing
def hill_climb(h, c5):
    """Exhaustive hill climbing to improve solution"""
    best_h = h.copy()
    best_c5 = c5
    
    improvement = True
    while improvement:
        improvement = False
        for i in range(n_points):
            for j in range(n_points):
                if i >= j:
                    continue
                if h[i] > 0.5 and h[j] < 0.5:
                    # Try swapping
                    tmp = h[i]
                    h[i] = h[j]
                    h[j] = tmp
                    
                    new_c5 = fast_exact_eval(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        improvement = True
                        break
                    
                    # Undo swap
                    h[i] = tmp
                    h[j] = h[i]
                    h[i] = tmp
                elif h[i] < 0.5 and h[j] > 0.5:
                    # Try swapping
                    tmp = h[i]
                    h[i] = h[j]
                    h[j] = tmp
                    
                    new_c5 = fast_exact_eval(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        improvement = True
                        break
                    
                    # Undo swap
                    h[i] = tmp
                    h[j] = h[i]
                    h[i] = tmp
        h = best_h.copy()
        
    return best_h, best_c5
