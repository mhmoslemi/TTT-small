import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from itertools import product

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return dx * np.max(corr)

def compute_corr_fft(h):
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use 120 points as specified
    n_points = 120
    target_sum = n_points / 2.0

    # Initialization from the benchmark
    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h0 = np.interp(x_new, x_orig, h0)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (target_sum / np.sum(h0))  # Normalize sum

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 1: Binary MILP approach with CBC
    # Set up the MILP model using cvxpy
    try:
        print("Attempting MILP formulation")
        h_vars = cp.Variable(n_points, boolean=True)
        lambda_var = cp.Variable(integer=True)

        # Constraint: sum of h equals target_sum 
        weight_constraint = cp.Equality(cp.sum(h_vars), target_sum)

        # Create product variables y[i, l] = h[i] * h[i+l] for valid pairs
        y_vars = {}
        for l in range(n_points):
            # Only for i, i+l valid
            for i in range(n_points - l):
                y_vars[(i,l)] = cp.Variable(boolean=True)
        
        # Product constraints (McCormick envelopes)
        prod_constraints = []
        for l in range(n_points):
            for i in range(n_points - l):
                y = y_vars[(i,l)]
                h1 = h_vars[i]
                h2 = h_vars[i+l]
                prod_constraints.append(y <= h1)
                prod_constraints.append(y <= h2)
                prod_constraints.append(y >= h1 + h2 - 1)
                
        # Overlap constraints
        overlap_constraints = []
        for l in range(n_points):
            # Sum over valid indices for lag l
            y_sum = cp.sum([y_vars[(i,l)] for i in range(n_points - l)])
            overlap_constraints.append(y_sum >= lambda_var)
            
        # Bound on lambda
        lambda_bound = lambda_var <= target_sum

        # Objective
        objective = cp.Maximize(lambda_var)
        
        # Create the model and solve
        constraints = [weight_constraint] + prod_constraints + overlap_constraints + [lambda_bound]
        prob = cp.Problem(objective, constraints)
        
        # We only need a decent solution, not the optimal
        # Set timeout to avoid hanging too long
        remaining_budget = budget_s * 0.9
        print(f"Solving MILP with {remaining_budget:.1f}s limit")
        prob.solve(solver=cp.CBC, time_limit=remaining_budget)

        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.array(h_vars.value, dtype=float)
            lambda_val = lambda_var.value
            c5_sol = 1.0 - (2.0/n_points)*lambda_val
            print(f"MILP returned approximate solution with C5 = {c5_sol:.6f}")
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
        else:
            print(f"MILP did not converge to an integer solution (status: {prob.status})")
            # Fallback to high-quality initialization
            h_eval = h0.copy()
            c5_val = evaluate_c5(h_eval)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_eval.copy()
    except Exception as e:

        # Fallback to local refinement with best initialization
        h_eval = h0.copy()
        c5_val = evaluate_c5(h_eval)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_eval.copy()

    # Phase 2: Local refinement step
    # Create a simple stochastic local search
    refine_time = 0.1 * budget_s
    refine_start = time.time()
    last_improvement = 0
    
    while (time.time() - refine_start) < refine_time:
        # Mutate a small random subset
        i = np.random.randint(0, n_points)
        step = np.random.choice([-0.05, 0.05])
        h_test = best_h.copy()
        h_test[i] += step
        
        # Project back to [0,1]
        h_test = np.clip(h_test, 0, 1)
        # Scale to make sum = target_sum
        h_test = h_test * (target_sum / np.sum(h_test))
        
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > 1000:
                break  # Stop if no progress for 1000 steps

    # Final check
    c5_final = evaluate_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final

    return (best_h.tolist(), float(best_c5), n_points)
