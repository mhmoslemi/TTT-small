def project_to_sum_box(x, target_sum):
    """Adjust x to have exactly target_sum, maintaining [0,1] bounds."""
    tol = 1e-10
    low, high = -1.0, 1.0

        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)

            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

    dx = 2.0 / n_points
