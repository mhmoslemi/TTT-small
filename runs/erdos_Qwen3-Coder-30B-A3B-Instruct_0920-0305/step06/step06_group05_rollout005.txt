import numpy as np

# Global variables
n_points = 120
dx = 2.0 / n_points
M = n_points / 2

def compute_corr(h):
    """Compute correlation using FFT for speed"""
    h_rev = h[::-1]
    b = h  # Since we want h(x) * h(x+k) 
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(M - corr[n_points-1:])

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def is_binary(h):
    """Check if h consists of only 0's and 1's"""
    return np.all((h == 0) | (h == 1))

def binary_aware_swap(h, i, j):
    """Swap bits i and j of binary vector h"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def binary_block_flip(h, start, length):
    """Flip a block of bits of fixed length (for binary vectors)"""
    h = h.copy()
    end = min(start + length, len(h))
    for i in range(start, end):
        h[i] = 1 - h[i]
    return h

def binary_repair_weight(h, target_weight):
    """Enforce target weight by swapping 0s with 1s"""
    current_weight = np.sum(h)
    diff = int(current_weight - target_weight)
    if diff > 0:
        # Need to make some bits 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(zeros) >= diff:
            # Flip diff ones to zeros
            indices = ones[:diff]
            for i in indices:
                h[i] = 0
        else:
            # Will be slightly off
            indices = ones[:len(ones)]
            for i in indices:
                h[i] = 0
    elif diff < 0:
        # Need to make some bits 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) >= -diff:
            # Flip -diff zeros to ones
            indices = zeros[:(-diff)]
            for i in indices:
                h[i] = 1
        else:
            # Will be slightly off
            indices = zeros[:len(zeros)]
            for i in indices:
                h[i] = 1
    # Critical check
    if abs(np.sum(h) - target_weight) > 0.1:
        # Force correct weight (shouldn't happen given algorithm)
        ones = np.where(h > 0.5)[0]
        if len(ones) > int(target_weight):
            # Too many 1's
            for i in range(len(ones) - int(target_weight)):
                h[ones[i]] = 0
        else:
            # Too few 1's
            zeros = np.where(h < 0.5)[0]
            for i in range(int(target_weight) - len(ones)):
                h[zeros[i]] = 1
    return h

def compute_increments(h):
    """Compute incremental change to C5 when swapping bits"""
    c0 = evaluate_c5(h)
    n = len(h)
    best_swap = None
    best_improvement = 0
    new_h = h.copy()
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    
    # Collect needed information for computing changes
    corr = compute_corr(h)
    worst_lag_idx = np.argmax(M - corr[n-1:])
    worst_lag = worst_lag_idx

    for i in ones:
        for j in zeros:
            if abs(i - j) == worst_lag:
                # This will definitely improve the correlation at worst lag
                h_swap = binary_aware_swap(h.copy(), i, j)
                c_new = evaluate_c5(h_swap)
                improvement = c0 - c_new
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
    return best_swap, best_improvement

def hill_climb_binary(h, max_iter=500):
    """Use exhaustive testing on pairs to improve binary solution"""
    current_h = h.copy()
    improved = True
    i_iter = 0
    while improved and i_iter < max_iter:
        improved = False
        ones = np.flatnonzero(current_h > 0.5)
        zeros = np.flatnonzero(current_h < 0.5)
        
        # Try all (i,j) swaps
        best_i, best_j = None, None
        best_change = 0
        current_c5 = evaluate_c5(current_h)
        
        for i in ones:
            for j in zeros:
                h_swap = binary_aware_swap(current_h, i, j)
                c5_swap = evaluate_c5(h_swap)
                improvement = current_c5 - c5_swap
                if improvement > best_change:
                    best_change = improvement
                    best_i, best_j = i, j
        
        if best_i is not None:
            current_h = binary_aware_swap(current_h, best_i, best_j)
            improved = True
        i_iter += 1
        
    return current_h

def compute_lag_contributions(h):
    """Compute an array of which indices contribute to which lags"""
    n = len(h)
    contributions = {}
    for i in range(n):
        for k in range(n):
            j = i + k
            if 0 <= j < n:
                contrib = h[i] * h[j]
                if k not in contributions:
                    contributions[k] = []
                contributions[k].append((i, contrib))
    return contributions

        impacts = []

            t = 0

            if 0 <= j < len(h):
                t = h[i] * (1 - h[j]) + (1 - h[i]) * h[j]
            impacts.append((i, t))
        impacts.sort(key=lambda x: x[1], reverse=True)
        involved_list = [x[0] for x in impacts[:20]]
        
    if len(involved_list) == 0:
        return h
        
    # Variables
    z = cp.Variable(len(involved_list), boolean=True)
    # Weight is fixed in this context (M)
    t = cp.Variable()

    # Exact sum equals M.  Let's re-evaluate what we're doing:
    current_weight = int(np.sum(h))

            fix_to_1.append(k)
        else:
            fix_to_0.append(k)
    
    # Define "MILP objective" and constraints
    # Varibles z in {0,1} represent choices for involved indices
    # Create auxiliary variable y_{i,i+k} = z_i AND z_{i+k}
    terms = []
    # Structured like: for each lag used in worst case connection
    for k in range(1, n_points):
        if k > len(involved_list) + 3:
            continue
        this_k_terms = []
        for i in range(len(involved_list)):
            i1 = involved_list[i]
            i2 = i1 + k
            if i2 in involved_list:
                idx2 = involved_list.index(i2)
                y = cp.Variable(boolean=True)
                this_k_terms.append(y)
                # Add AND constraints y <= z[i], y <= z[idx2], y >= z[i] + z[idx2] - 1
                constraints = [y <= z[i], y <= z[idx2], y >= z[i] + z[idx2] - 1]
                # c.add(constraints) # Have to add
        terms.extend(this_k_terms)
    
    constraints = []
    # Weight
    constraints.append(cp.sum(z) == current_weight - len(fix_to_1) + len(fix_to_0))
    
    # Compute new contributions
    # This is activated if no change is made below for now
    
    prob = Problem(Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, seconds=min(5, budget_remaining * 0.15))
        # Accept setup is too complex for now. 
        # Return original vector
        return h
    except:
        return h
