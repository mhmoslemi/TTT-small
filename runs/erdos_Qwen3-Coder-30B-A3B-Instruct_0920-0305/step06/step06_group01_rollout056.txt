import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT efficiency
FFT_LEN = 512  # Pad to next power of 2 for speed

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    # Pad to FFT_LEN for speed
    h_pad = np.pad(h, (0, FFT_LEN - len(h)), 'constant')
    b_pad = np.pad(b, (0, FFT_LEN - len(b)), 'constant')
    corr = np.fft.ifft(np.fft.fft(h_pad) * np.fft.fft(b_pad).conj()).real
    # Return the full correlation (size L) but only the relevant part
    return corr[:L]

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
