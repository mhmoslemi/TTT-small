import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def solve_lagrangian_subproblem(h, lag_index, step_size=0.01):
    """Solve for optimal adjustment along the gradient of correlation at lag_index"""
    n = len(h)
    # Compute gradient of h_i w.r.t. correlation at this lag
    # The correlation for lag \ell is sum_i h_i h_{i+\ell} 
    # Its derivative w.r.t. h_i is h_{i+\ell} + h_{i-\ell}
    # For the specific lag we're considering
    grad = np.zeros(n)
    for i in range(n):
        left_idx = i - lag_index
        right_idx = i + lag_index
        if left_idx >= 0:
            grad[i] += h[left_idx]
        if right_idx < n:
            grad[i] += h[right_idx]
    # We take the direction as the negative of the gradient to increase overlap (decrease C_k)
    adj = -step_size * grad
    
    # Apply to h and project onto constraints
    h_new = h + adj
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def guided_move(h, worst_lag):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    n = len(h)
    # Compute subgradient g_i = h_{i-\ell} + h_{i+\ell}
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Find indices with min and max g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None  # No improvement possible
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.01)  # Small fixed step
    h_new = h.copy()
    h_new[i_min] -= delta
    h_new[i_max] += delta
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

def local_refine(h_start):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    try:
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h_start, evaluate_c5(h_start)

def almost_binary_h(n_points, target_sum):
    """Create almost binary h to guide optimization"""
    h = np.zeros(n_points)
    # Place 50% of it as 1s
    ones_count = int(target_sum)
    ones_idx = np.random.choice(n_points, ones_count, replace=False)
    h[ones_idx] = 1.0
    
    # Project to ensure constraint
    h = project_to_sum_box(h, target_sum)
    return h

def improved_coord_descent(h0, target_sum, max_it=5000):
    """Improved coordinate descent with Lagrangian subproblem solving"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    dt = 0.001  # Small step for fine adjustment
    
    for it in range(max_it):
        # Compute the correlation vector directly
        corr = compute_corr_fft(h)
        # Find the worst lag (largest correlation)
        worst_lag = np.argmax(corr)
        # Minimize the C_k for that lag
        h_new = solve_lagrangian_subproblem(h, worst_lag, dt)
        h = h_new
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Occasionally adapt step size
        if it % 50 == 0 and dt > 1e-6:
            dt *= 0.99
    return best_h, best_c5
