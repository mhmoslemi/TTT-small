import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def pattern_based_search(h0, max_iter=500):
    """Enhanced local search with pattern recognition and adaptive moves"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Try to identify patterns that work well
    for i in range(max_iter):
        # Select a random index to modify
        idx = np.random.randint(0, n_points)
        
        # Try different step sizes
        step_sizes = [0.02, 0.01, -0.01, -0.02]
        
        # Evaluate all potential moves
        best_move = None
        best_improvement = 0
        
        for step in step_sizes:
            h_new = h.copy()
            h_new[idx] += step
            
            if h_new[idx] < 0 or h_new[idx] > 1:
                continue
                
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            improvement = best_c5 - new_c5
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_move = (idx, step)
        
        # Apply best move if beneficial
        if best_move is not None and best_improvement > 1e-12:
            idx, step = best_move
            h[idx] += step
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
        else:
            # Random small perturbation for escape
            if np.random.random() < 0.2:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.01, 0.01)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

def adaptive_initialization():
    """Create intelligent initial solution using insights from problem structure"""
    # Use piecewise constant function that is non-uniform to avoid symmetries
    h = np.zeros(n_points)
    
    # Start with a two-level structure that is inspired by fast decay patterns
    # 30% of points at 0.9, rest at 0.1 but redistributed for sum constraint
    # Create a more structured input
    segment_length = n_points // 3
    h[:segment_length] = 0.8
    h[segment_length:2*segment_length] = 0.2
    h[2*segment_length:] = 0.1
    
    # Normalize to desired sum
    current_sum = np.sum(h)
    scale_factor = TARGET_SUM / current_sum
    h *= scale_factor
    
    # Add small random variation to the structure to escape local minimums
    noise = np.random.normal(0, 0.01, n_points)
    h = np.clip(h + noise, 0, 1)
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = adaptive_initialization()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Slightly modified gradient optimizations
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(15):
        if time.time() > phase1_end:
            break
            
        # Try different initialization strategies
        initial_modes = [
            adaptive_initialization,
            lambda: np.random.uniform(0, 1, n_points),
            lambda: np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        ]
        h_start = initial_modes[run_id % len(initial_modes)]()
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Gradient-based optimization with more iterations
        h_refined = optimize_with_gradients(h_start, max_iter=1500)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Pattern-based local searches with more focus on structure
    phase2_end = start_time + 0.7 * budget_s
    for run_id in range(10):
        if time.time() > phase2_end:
            break
            
        h_refined = pattern_based_search(best_h.copy(), max_iter=500)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 3: Adaptive improvements using strategic perturbations
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Strategic local search
        h_noise = best_h + np.random.normal(0, 0.005, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        h_final = pattern_based_search(h_noise, max_iter=300)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
