import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_gradient(h, alpha):
    """Compute gradient of soft minimum auto-correlation"""
    # Compute auto-correlations
    S = correlate(h, h, 'full')
    
    # Numerically stable computation of weights
    max_S = np.max(-alpha * S)
    exp_neg_alpha_S = np.exp(-alpha * S - max_S)
    w = exp_neg_alpha_S / np.sum(exp_neg_alpha_S)
    
    # Build symmetric kernel for convolution
    W = np.zeros(2 * n_points - 1)
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            W[n_points - 1 + k] = w[k]
        else:
            W[n_points - 1 + k] = w[-k]
    
    # Convolution to get gradient
    conv = np.convolve(h, W, mode='full')
    g = conv[n_points - 1 : n_points - 1 + n_points]
    
    # Add contribution from zero lag (handled specially)
    g += w[n_points - 1] * h
    
    return g

    # Algorithm parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    stagnation_counter = 0

    while time.time() - start_time < budget_s - 10:  # Save a bit of time for final refinement

        g = compute_soft_min_gradient(h, alpha)
        
        # Update candidate
        h_candidate = h + step * g
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)

            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
