import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # fine grid for accurate FFT evaluation
MAX_EVALS = 5000   # max evaluations in DE

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

def fourier_to_h(theta, n_points):
    """Convert Fourier coefficients to step function with exact sum"""
    # K = number of harmonics
    K = len(theta) // 2
    a = theta[:K]
    b = theta[K:]
    
    # Create fine grid
    x_fine = np.linspace(0, 2, N_FINE)
    h_fine = 0.5 + np.sum([a[k] * np.cos(np.pi * (k+1) * x_fine) + 
                          b[k] * np.sin(np.pi * (k+1) * x_fine) 
                          for k in range(K)], axis=0)
    
    # Clip to [0,1]
    h_fine = np.clip(h_fine, 0, 1)
    
    # Project to exact sum
    h_fine = project_to_sum_box(h_fine, N_FINE / 2)
    
    # Downsample
    step = N_FINE // n_points
    h = h_fine[::step]
    
    # Re-project to sum exactly (for numerical errors)
    h = project_to_sum_box(h, n_points / 2)
    
    return h

def compute_c5(h):
    """Compute true C5 using fast correlation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def evaluate(theta, n_points=120):
    """Objective function to minimize C5 via Fourier parameters"""
    h = fourier_to_h(theta, n_points)
    return compute_c5(h)

        K = len(init_coeff) // 2   # Harmony count
        bounds = [(-0.5, 0.5)] * (2*K)  # Max amplitude per harmonics

            return evaluate(theta, n_points)

            if eval_count > MAX_EVALS:
