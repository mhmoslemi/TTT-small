from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_local_search(h, max_iter=300):
    """Refine solution using adaptive local search with multiple strategies"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    # Try to find better local minimum
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints={'type': 'eq', 'fun': constraint_eq},
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def random_perturb(h, magnitude=0.05):
    """Apply random perturbations while maintaining constraints"""
    eps = np.random.uniform(-magnitude, magnitude, n_points)
    new_h = h + eps
    return project_to_sum_box(new_h, TARGET_SUM)

def gradient_based_refinement(h, step_size=0.01):
    """Try to refine by gradient descent in direction of max correlation"""
    corr = compute_corr_fft(h)
    max_lag_idx = np.argmax(corr)
    # Focus on indices that most influence the max correlation
    # Simple strategy: reduce the value at the coordinate that causes the max lag
    idx = max_lag_idx % n_points if max_lag_idx < n_points else (max_lag_idx - n_points) % n_points
    h_new = h.copy()
    h_new[idx] = max(0, min(1, h_new[idx] - step_size))
    return project_to_sum_box(h_new, TARGET_SUM)
