import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def grid_search_initialization():
    """Create initial solution using grid search approach"""
    h = np.zeros(n_points)
    
    # Start with a coarse grid, then refine
    # Try a few different patterns
    patterns = [
        # Pattern 1: Increasing then decreasing
        lambda i: max(0, min(1, 2 * i / n_points - 0.5)),
        # Pattern 2: Two peaks
        lambda i: 0.5 * (1 + np.tanh(20 * (i / n_points - 0.3))) if i < n_points//2 else \
                  0.5 * (1 + np.tanh(20 * (0.7 - i / n_points))),
        # Pattern 3: Step function
        lambda i: 1 if i < n_points//3 else (0.5 if i < 2*n_points//3 else 0)
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    for pattern in patterns:
        h_candidate = np.array([pattern(i) for i in range(n_points)])
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        c5 = evaluate_c5(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate
    
    return best_h if best_h is not None else np.ones(n_points) * 0.5

def local_improvement(h, max_iter=500):
    """Try to improve current solution by local search"""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    for _ in range(max_iter):
        # Try small perturbations to each point
        improved = False
        for i in range(n_points):
            # Try several small changes
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h.copy()
                h_new[i] += delta
                if 0 <= h_new[i] <= 1:
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        h = h_new
                        best_c5 = c5_new
                        improved = True
        if not improved:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = grid_search_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Multiple local searches with different starting points
    phase1_end = start_time + 0.5 * budget_s
    for run_id in range(50):
        if time.time() > phase1_end:
            break
            
        # Random perturbations from good base
        h_start = best_h + np.random.normal(0, 0.02, n_points)
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Local search improvement
        h_refined = local_improvement(h_start, max_iter=200)
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 2: Gradient-based refinement using SLSQP
    phase2_end = start_time + 0.85 * budget_s
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        h_opt = best_h.copy()
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Try a few different initializations
        for _ in range(5):
            if time.time() > phase2_end:
                break
            h_candidate = best_h + np.random.normal(0, 0.01, n_points)
            h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
            
            res = minimize(
                objective, 
                h_candidate, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            
            if res.success:
                h_refined = project_to_sum_box(res.x, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
    except:
        pass
    
    # Phase 3: Final local search
    phase3_end = start_time + budget_s
    if time.time() < phase3_end:
        best_h = local_improvement(best_h, max_iter=500)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
