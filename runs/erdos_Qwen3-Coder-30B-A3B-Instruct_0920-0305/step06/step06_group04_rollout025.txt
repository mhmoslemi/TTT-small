import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # Fine grid for accurate FFT evaluation
MAX_EVALS = 5000   # Maximum evaluations for optimization

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def fourier_to_h(theta, K, N):
    """Convert Fourier coefficients to discrete step function"""
    # Build the frequency basis functions
    x = np.linspace(0, 2, N_FINE)
    h_raw = 0.5 * np.ones(N_FINE)
    for k in range(1, K+1):
        h_raw += theta[2*(k-1)] * np.cos(np.pi * k * x) + theta[2*(k-1)+1] * np.sin(np.pi * k * x)
    
    # Clip to [0,1]
    h_clipped = np.clip(h_raw, 0, 1)
    
    # Project to exact sum
    h_clipped = project_to_sum_box(h_clipped, TARGET_SUM)
    
    # Downsample
    h = h_clipped[::N_FINE//N]
    
    return h

def optimize_fourier(K, N, seed=42):
    """Optimize Fourier coefficients to minimize C5"""
    np.random.seed(seed)
    
    def objective(theta):
        h = fourier_to_h(theta, K, N)
        return evaluate_c5(h)
    
    bounds = [(-0.5, 0.5) for _ in range(2*K)]
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=MAX_EVALS//10,
        popsize=50,
        seed=seed,
        disp=False
    )
    
    return result

def local_refine_h(h_start, max_iter=300):
    """Refine solution using projected gradient ascent"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    alpha = 1.0
    
    for _ in range(max_iter):
        # Compute all correlations
        corr = compute_corr_fft(h)
        # Get min correlation (since we're minimizing P(h) = 1 - max(corr))
        min_corr = np.min(corr)
        # Weighted gradient for the worst lag
        lags = np.arange(len(corr))
        weights = np.exp(-alpha * corr)
        weights /= np.sum(weights)
        
        # Approximate gradient (simplified for speed)
        grad = np.zeros(n)
        for i in range(n):
            for j in range(n):
                if i + j < n:
                    grad[i] -= weights[j] * h[i + j]
                if i - j >= 0:
                    grad[i] -= weights[j] * h[i - j]
        
        # Gradient step with line search
        step_size = 0.01
        h_trial = np.clip(h - step_size * grad, 0, 1)
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        
        if evaluate_c5(h_trial) < evaluate_c5(h):
            h = h_trial
            alpha = min(20, alpha * 1.05)  # Increase smoothing
        else:
            break
    
    return h

def binary_round_and_swap(h):
    """Round to binary and swap locally to improve C5"""
    # Make a binary version
    h_binary = h.copy()
    indices_sorted = np.argsort(h_binary)[::-1]
    ones_count = int(TARGET_SUM)
    for i in range(ones_count):
        h_binary[indices_sorted[i]] = 1.0
    for i in range(ones_count, len(h_binary)):
        h_binary[indices_sorted[i]] = 0.0
    
    # This is an aggressive simplification for practicality
    # In real code, this would use more intelligent swaps
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Multi-scale approach: first try coarse, then fine
    n_points = 120  # Start with base size
    K = 10          # Fourier modes
    
    # Use a coarse base
    res = optimize_fourier(K, n_points, seed)
    best_h = fourier_to_h(res.x, K, n_points)
    best_c5 = evaluate_c5(best_h)
    
    # Optionally try finer resolution if time allows
    if time.time() - start_time < 0.6 * budget_s:
        n_points = 200
        res = optimize_fourier(K, n_points, seed)
        H_fine = fourier_to_h(res.x, K, n_points)
        H_fine = project_to_sum_box(H_fine, TARGET_SUM)
        C5_fine = evaluate_c5(H_fine)
        
        if C5_fine < best_c5:
            best_h = H_fine
            best_c5 = C5_fine
    
    # Local polishing with projected gradient
    if time.time() - start_time < 0.9 * budget_s:
        refined_h = local_refine_h(best_h, max_iter=200)
        C5_refined = evaluate_c5(refined_h)
        if C5_refined < best_c5:
            best_h = refined_h
            best_c5 = C5_refined
    
    # Binary swap optimization
    if time.time() - start_time < 0.98 * budget_s:
        final_h = binary_round_and_swap(best_h)
        C5_final = evaluate_c5(final_h)
        if C5_final < best_c5:
            best_h = final_h
            best_c5 = C5_final
    
    return (best_h.tolist(), float(best_c5), len(best_h))
