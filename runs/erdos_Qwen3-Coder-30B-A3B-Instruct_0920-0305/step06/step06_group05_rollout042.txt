import numpy as np

import time
import cvxpy as cp

# Global variables for the problem
TARGET_SUM = None
n_points = 60
dx = 2.0 / n_points

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary variables."""
    # Parameters
    M = n // 2  # Required number of ones
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()  # Worst-case overlap bound
    
    # Constraints
    constraints = []
    
    # Exact mass constraint
    constraints.append(cp.sum(h) == M)
    
    # Linking AND constraints for all pairs (i,j) with i < j
    for i in range(n):
        for j in range(i+1, n):
            # y[i,j] should be h[i] AND h[j]
            constraints.append(y[i,j] <= h[i])
            constraints.append(y[i,j] <= h[j])
            constraints.append(y[i,j] >= h[i] + h[j] - 1)
            # Also enforce symmetry y[j,i] = y[i,j]
            if i != j:
                y[j,i] = y[i,j]
    
    # Lag constraints for all positive lags k = 1, ..., n-1
    for k in range(1, n):
        # t >= M - sum_{i=0}^{n-1-k} y[i,i+k]
        lag_sum = cp.sum([y[i, i+k] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Symmetry breaking: fix h[0] = 1
    constraints.append(h[0] == 1)
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.round(h.value).astype(int)
            t_val = t.value
            return h_val, t_val
        else:
            return None, None
    except:
        return None, None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Convert back to C5 value
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Perform a continuous local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(len(h))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
