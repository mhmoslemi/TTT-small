def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr) * (2.0 / len(h))

def evaluate_c5_binary(x):
    """Evaluate C5 for binary vector using FFT"""
    corr = np.correlate(x, 1 - x, mode="full")
    return np.max(corr) * (2.0 / len(x))

    ones_copy = ones.copy()
    ones_copy.remove(i)

        shift = k - i
        idx = shift + len(x) - 1
        conv[idx] -= 1
        if k != i:
            conv[-shift + len(x) - 1] -= 1

    # Add contributions of position j
    for k in ones_copy:
        shift = k - j
        idx = shift + len(x) - 1
        conv[idx] += 1
        if k != j:
            conv[-shift + len(x) - 1] += 1

    # Add self contribution of j
    conv[len(x) - 1] += 1

    # Update index lists
    ones.remove(i)
    ones.append(j)
    zeros.remove(j)
    zeros.append(i)

    return conv
