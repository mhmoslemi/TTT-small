def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):

def initialize_binary_pattern(n_points):
    """Initialize binary pattern using a heuristic pattern"""
    # Initialize with zeros
    h = np.zeros(n_points, dtype=int)
    
    # Distribute 1s more uniformly to avoid clustering
    # Try to place 1s with period to get good autocorrelation
    ones_count = n_points // 2
    
    # Place 1s every few positions with phase shift
    period = max(3, n_points // (ones_count + 1))
    for i in range(ones_count):
        pos = (i * period) % n_points
        h[pos] = 1
        
    # If not exactly right number, adjust
    current_count = np.sum(h)
    if current_count < ones_count:
        # Add more 1s where needed
        zero_indices = np.where(h == 0)[0]
        add_count = ones_count - current_count
        for i in range(add_count):
            if i < len(zero_indices):
                h[zero_indices[i]] = 1
    elif current_count > ones_count:
        # Remove some 1s
        one_indices = np.where(h == 1)[0]
        remove_count = current_count - ones_count
        for i in range(remove_count):
            if i < len(one_indices):
                h[one_indices[i]] = 0
    
    return h

def update_convolution(conv, ones, i, j, n):
    """Update convolution array after swapping positions i (1->0) and j (0->1)""" 
    # Convert to lists for efficient removal/addition
    ones_list = ones.copy()
    
    # Remove contributions from i (as 1)
    for k in ones_list:
        shift = k - i
        if k == i:
            # Self pairing
            conv[n-1] -= 1
        else:
            # Regular pair
            idx1 = shift + n - 1
            idx2 = -shift + n - 1
            conv[idx1] -= 1
            conv[idx2] -= 1
    
    # Remove i from ones_list
    ones_list.remove(i)
    
    # Add contributions from j (as 1)  
    for k in ones_list:
        shift = k - j
        if k == j:
            # Self pairing
            conv[n-1] += 1
        else:
            # Regular pair
            idx1 = shift + n - 1
            idx2 = -shift + n - 1
            conv[idx1] += 1
            conv[idx2] += 1
    
    # Add self pair for j
    conv[n-1] += 1
    
    # Update ones list
    ones_list.append(j)
    
    # Make sure values don't go below zero
    conv = np.maximum(conv, 0)
    
    return conv, ones_list
