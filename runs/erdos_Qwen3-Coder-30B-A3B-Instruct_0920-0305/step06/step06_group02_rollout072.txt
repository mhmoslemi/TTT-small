import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_adam(h0, learning_rate=0.01, max_iter=2000):
    """Optimize using Adam optimizer with projection"""
    h = h0.copy().astype(np.float64)
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    def compute_gradient(h):
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            val_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            val_minus = evaluate_c5(h_minus)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad
    
    for t in range(1, max_iter + 1):
        # Compute gradient
        grad = compute_gradient(h)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update parameters
        h = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
    return h

def pattern_search(h0, max_iter=500):
    """Pattern based search improving local structure"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    # Try different improvement patterns
    patterns = [
        [0.05] * 10 + [-0.05] * 10,
        [0.03] * 5 + [-0.03] * 5 + [0.03] * 5 + [-0.03] * 5,
        [0.02] * 20,
        [-0.02] * 20,
        [0.05] * 4 + [-0.05] * 12 + [0.05] * 4,
        [0.01] * 50 + [-0.01] * 50,
    ]
    
    for i in range(max_iter):
        if i % 10 == 0:
            # Recompute for accuracy
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
        
        # Try pattern updates
        for pattern in patterns:
            h_new = h.copy()
            pattern_len = min(len(pattern), len(h_new))
            for j in range(pattern_len):
                h_new[j] += pattern[j]
            
            # Project and check
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5:
                h = h_new.copy()
                best_c5 = new_c5
        
        # Add small random perturbation
        if np.random.random() < 0.3:
            pert_idx = np.random.randint(0, len(h))
            h[pert_idx] += np.random.uniform(-0.02, 0.02)
            h = project_to_sum_box(h, TARGET_SUM)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    return h

def adaptive_sampling_initialization():
    """Create a more intelligent initial solution"""
    # Start with known good values
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        x_ref = np.linspace(0, 2, len(h_ref))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_ref, h_ref)
    else:
        # Alternative: make a rough approximation with fewer samples
        h_interp = np.ones(n_points) * 0.5
    
    # Add small random perturbation
    h_perturbed = h_interp + np.random.normal(0, 0.02, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
    return h_perturbed

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    h0 = adaptive_sampling_initialization()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Phase 1: Adam optimization with multiple restarts
    phase1_end = start_time + 0.4 * budget_s
    for run_id in range(10):
        if time.time() > phase1_end:
            break
            
        # Random restart
        h_start = adaptive_sampling_initialization()
        h_opt = optimize_with_adam(h_start, learning_rate=0.01, max_iter=1000)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Phase 2: Pattern search
    phase2_end = start_time + 0.7 * budget_s
    for run_id in range(5):
        if time.time() > phase2_end:
            break
            
        h_search = pattern_search(best_h.copy(), max_iter=500)
        c5_search = evaluate_c5(h_search)
        
        if c5_search < best_c5:
            best_h = h_search.copy()
            best_c5 = c5_search
    
    # Phase 3: Local fine tuning with gradient
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Add small noise and optimize  
        h_noise = best_h + np.random.normal(0, 0.005, n_points)
        h_noise = project_to_sum_box(h_noise, TARGET_SUM)
        
        h_final = optimize_with_adam(h_noise, learning_rate=0.005, max_iter=300)
        c5_final = evaluate_c5(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
