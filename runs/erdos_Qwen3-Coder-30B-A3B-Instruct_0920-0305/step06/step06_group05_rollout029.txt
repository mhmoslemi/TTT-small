import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.fft import fft, ifft

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def c5_from_corr(corr, n):
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, M, max_time_s):
    # Prepare P_k matrices
    P_k_list = []
    for k in range(n):
        P_k = np.zeros((n, n))
        for i in range(n-k):
            P_k[i, i+k] = 1
        P_k_list.append(P_k)
    
    # Setup variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(X) == M,
        h >= 0,
        h <= 1,
        X >> 0,  # PSD
        cp.diag(X) == h,
        t >= M - cp.trace(P_k @ X) for P_k in P_k_list
    ]
    
    # Solve
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, time_limit=max_time_s, verbose=False)
    except:
        return None
    return h.value, X.value

def random_hyperplane_rounding(h_frac, X_frac, M, num_trials=15):
    _, Vt = np.linalg.eigh(X_frac)
    Vt = Vt.T
    # Retain only positive eigenvalues
    V = Vt[:10] if Vt.shape[0] > 10 else Vt

    candidates = []
    for _ in range(num_trials):
        g = np.random.standard_normal(V.shape[0])
        s = V @ g
        b = (s >= 0).astype(int)
        
        # Repair weight
        current_sum = np.sum(b)
        if current_sum > M:
            flip_indices = np.random.choice(np.where(b == 1)[0], 
                                          size=current_sum - M, replace=False)
            b[flip_indices] = 0
        elif current_sum < M:
            flip_indices = np.random.choice(np.where(b == 0)[0], 
                                          size=M - current_sum, replace=False)
            b[flip_indices] = 1
        
        if np.sum(b) == M:
            candidates.append(b)
    return candidates

def deterministic_threshold_rounding(h_frac, M):
    h_sorted_indices = np.argsort(h_frac)[::-1]
    b = np.zeros_like(h_frac)
    b[h_sorted_indices[:M]] = 1
    return [b.astype(int)]

def hill_climb_swap(h, t):
    n = len(h)
    h = h.copy()
    prev_c5 = evaluate_c5(h)
    
    improved = True
    swap_attempts = 0
    max_attempts = 5000
    
    while improved and swap_attempts < max_attempts:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                swap_attempts += 1
                h[i], h[j] = h[j], h[i]
                c5 = evaluate_c5(h)
                if c5 < t:
                    t = c5
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]  # revert
            if improved:
                break
    return h, t

def polish_candidate(h, M, max_iter=150, time_limit=0.2):
    start_time = time.time()
    h = h.copy()
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - M
    
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12},
                       callback=lambda x: time.time() > start_time + time_limit)
        if res.success:
            h = res.x
        h = project_to_sum_box(h, M)
    except:
        pass
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    
    # Phase A: SDP solving
    sdps_time_left = 0.5 * budget_s
    for n in [60, 80, 100]:
        if time.time() - start_time > 0.5 * budget_s:
            break
        M = n // 2
        remaining_time = max(0.1, sdps_time_left - 5)
        result = solve_sdp(n, M, remaining_time)
        if result is None:
            continue
        h_frac, X_frac = result
        
        if h_frac is not None and np.all(h_frac >= 0) and np.all(h_frac <= 1):
            # Rounding
            candidates = []
            candidates.extend(random_hyperplane_rounding(h_frac, X_frac, M, 15))
            candidates.extend(deterministic_threshold_rounding(h_frac, M))
            
            for h_candidate in candidates:
                h_candidate = np.array(h_candidate, dtype=float)
                h_candidate, c5 = hill_climb_swap(h_candidate, float('inf'))
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate.copy()
                    n_points = n
        sdps_time_left -= remaining_time + 5
            
    # Phase B: Continuous polishing & final evaluation
    if best_h is not None:
        try:
            polished_h = polish_candidate(best_h, n_points // 2, 50, 0.1)
            polished_c5 = evaluate_c5(polished_h)
            if polished_c5 < best_c5:
                best_c5 = polished_c5
                best_h = polished_h
        except:
            pass

    if best_h is None:
        # Fallback: simple construction
        n_fallback = 100
        h_fallback = np.zeros(n_fallback)
        h_fallback[:n_fallback//2] = 1
        h_fallback = project_to_sum_box(h_fallback, n_fallback//2)
        best_c5 = evaluate_c5(h_fallback)
        best_h = h_fallback
        n_points = n_fallback
        
    return (best_h.tolist(), float(best_c5), n_points)
