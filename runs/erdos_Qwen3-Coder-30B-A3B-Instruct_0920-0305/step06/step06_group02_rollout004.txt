import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create initial binary solution with exact sum constraint"""
    np.random.seed(seed)
    # Start with a good initial pattern from the previous work
    x = np.array(initial_h_values)
    # Round to binary and project to exact weight
    x = (x > 0.5).astype(np.int8)
    current_sum = np.sum(x)
    if current_sum > TARGET_SUM:
        # Need to flip some 1s to 0s
        ones = np.where(x == 1)[0]
        flip_indices = np.random.choice(ones, size=current_sum - TARGET_SUM, replace=False)
        x[flip_indices] = 0
    elif current_sum < TARGET_SUM:
        # Need to flip some 0s to 1s
        zeros = np.where(x == 0)[0]
        flip_indices = np.random.choice(zeros, size=TARGET_SUM - current_sum, replace=False)
        x[flip_indices] = 1
    
    return x

def full_overlap(x):
    """Compute full correlation (overlap) array"""
    ov = np.correlate(x, x, 'full')
    return ov

def swap_update(ov, i, j, ones, zeros):
    """Incrementally update overlap array after swapping positions i and j"""
    # Make a copy to avoid modifying the original
    ov_new = ov.copy()
    # For each 1-position (including i), adjust ov values
    # Remove contribution from i 
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n_points - 1] -= 1
    
    # Add contribution from j
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Run simulated annealing to find better binary solution"""
    start = time.time()
    T0 = 0.5
    alpha = 0.999
    T_min = 1e-4
    max_iter = 1000000
    
    cur_min = np.min(ov)
    best_x = x.copy()
    best_min = cur_min
    T = T0
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start < time_limit:
        iter_count += 1
        if T < T_min:
            T = T_min
        
        # Pick two random indices (one 1 and one 0)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Make a copy of current overlap
        ov_new = swap_update(ov, i, j, ones, zeros)
        new_min = np.min(ov_new)
        delta = new_min - cur_min
        
        # Accept move with probability
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            # Update overlap array
            ov[:] = ov_new
            cur_min = new_min
            
            # Update sets and vector
            x[i] = 0
            x[j] = 1
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            
            # Check if this is the new global best
            if new_min > best_min:
                best_x[:] = x
                best_min = new_min
        
        # Cool down
        T *= alpha
    
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine solution by improving the worst shift"""
    min_shift_idx = np.argmin(ov)
    worst_shift = min_shift_idx - n_points + 1  # Convert index to shift value
    
    # Find positions where we can improve this overlap
    improved = False
    do_again = True
    while do_again:
        do_again = False
        # Positions that have 1 but neighboring position (with this shift) does not
        A = set()
        # Positions that have 0 but their neighbor with this shift does
        B = set()
        
        for i in ones:
            neighbor_pos = i + worst_shift
            if 0 <= neighbor_pos < n_points and x[neighbor_pos] == 0:
                A.add(i)
        
        for j in zeros:
            neighbor_pos = j + worst_shift
            if 0 <= neighbor_pos < n_points and x[neighbor_pos] == 1:
                B.add(j)
        
        # Try all pairs AxB to improve overlap for this shift
        for i in list(A):
            for j in list(B):
                if i == j:
                    continue
                
                # Create a temporary new overlap array
                ov_new = swap_update(ov, i, j, ones, zeros)
                min_new = np.min(ov_new)
                
                # Only accept if it doesn't hurt other overlaps
                if min_new >= np.min(ov):
                    # Accept the swap
                    ov[:] = ov_new
                    x[i] = 0
                    x[j] = 1
                    ones.remove(i)
                    ones.add(j)
                    zeros.remove(j)
                    zeros.add(i)
                    improved = True
                    do_again = True
                    break
            if do_again:
                break
        
        if not improved:
            break
    
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize binary solution
    x = initial_binary(seed)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    ov = full_overlap(x)
    cur_min = np.min(ov)
    
    # Run multiple restarts of SA with time capping
    best_x = x.copy()
    best_min = cur_min
    
    early_exit = False
    while time.time() - start_time < 0.95 * budget_s and not early_exit:
        # Run SA with time limit
        time_remaining = 0.95 * budget_s - (time.time() - start_time)
        if time_remaining < 1:
            break
            
        # Run SA on current solution
        x_new, min_new = simulated_annealing(x.copy(), ov.copy(), ones.copy(), zeros.copy(), time_remaining)
        if min_new > best_min:
            best_x = x_new.copy()
            best_min = min_new
        
        # Apply worst-shift repair to find even better
        x_rep, ov_rep = worst_shift_refine(x_new.copy(), ov.copy(), ones.copy(), zeros.copy())
        rep_min = np.min(ov_rep)
        if rep_min > best_min:
            best_x = x_rep.copy()
            best_min = rep_min
        
        # For next iteration, fresh restart
        x = initial_binary(seed + 1)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        cur_min = np.min(ov)
    
    # Final refinement step
    x_final, ov_final = worst_shift_refine(best_x.copy(), full_overlap(best_x), 
                                          set(np.where(best_x == 1)[0]), 
                                          set(np.where(best_x == 0)[0]))
    final_min = np.min(ov_final)
    
    # Convert to float array
    final_h = best_x.astype(float)
    
    # Compute final C5 bound
    c5_bound = dx * (n_points / 2 - final_min)
    
    return (final_h.tolist(), float(c5_bound), n_points)
