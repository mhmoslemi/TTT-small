import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N / 2.0
    
    # Build MILP model
    h = cp.Variable(N, boolean=True)
    y = {}
    constraints = []
    
    # Add weight constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Add product constraints and lag overlap constraints
    lambda_var = cp.Variable(integer=True)
    constraints.append(lambda_var <= target_sum)

    # Precompute valid (i, l) pairs
    pairs = []
    for l in range(N):
        for i in range(N - l):
            pairs.append((i, l))
    
    # Create y variables and constraints
    for i, l in pairs:
        y_key = (i, l)
        y[y_key] = cp.Variable(boolean=True)
        constraints.append(y[y_key] <= h[i])
        constraints.append(y[y_key] <= h[i + l])
        constraints.append(y[y_key] >= h[i] + h[i + l] - 1)
        constraints.append(y[y_key] >= 0)
    
    # Add lag overlap constraints
    for l in range(N):
        # Sum over all valid i for this lag l
        expr = sum(y[i, l] for i in range(N - l))
        constraints.append(expr >= lambda_var)

    # Objective
    objective = cp.Maximize(lambda_var)
    
    # Create and solve model
    prob = cp.Problem(objective, constraints)
    timeout = 0.9 * budget_s
    prob.solve(solver=cp.CBC, timeout=timeout)
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
        h_sol = h.value
        lambda_val = lambda_var.value
        C5 = 1 - dx * lambda_val
    else:
        # Fallback to heuristics if MILP fails
        # Use the initial_h_values as a baseline
        h_sol = np.array(initial_h_values)
        h_sol = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, len(initial_h_values)), h_sol)
        h_sol = np.clip(h_sol, 0, 1)
        h_sol = h_sol * target_sum / np.sum(h_sol)
        C5 = evaluate_erdos_solution(h_sol)
    
    # Final refinement
    if (time.time() - start_time) < 0.95 * budget_s:
        def evaluate_C5(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode="full")
            return np.max(corr) * dx

        def project_to_sum(x, target_sum):
            return x * target_sum / np.sum(x)

        # Simple optimization using SLSQP around the solution
        def objective(h_vec):
            return evaluate_C5(h_vec)

        bounds = [(0, 1) for _ in range(N)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        try:
            from scipy.optimize import minimize
            res = minimize(objective, h_sol, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                h_sol = project_to_sum(res.x, target_sum)
                C5 = evaluate_C5(h_sol)
        except:
            pass

    return (h_sol.tolist(), float(C5), N)
