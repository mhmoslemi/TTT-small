import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800  # For accurate FFT evaluations
N_TARGETS = [120, 200]  # Try both resolutions
K_HARMONICS = 12  # Number of Fourier coefficients
MAX_COEFF = 0.5  # Bound on individual coefficients to avoid clipping
TIME_FRACTION_LOCAL = 0.25
TIME_FRACTION_BINARY = 0.05

# Helper functions
def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce exact sum = target_sum, box constraints [0,1]"""
    x = np.array(x)
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

def compute_corr_fft_vectorized(h):
    """Vectorized FFT-based full correlation of h with (1 - h)"""
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return corr

def evaluate_C5_from_corr(corr, n):
    """Compute C5 from correlation, using dx=2.0/n"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_C5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft_vectorized(h)
    return evaluate_C5_from_corr(corr, len(h))

def fourier_to_h(theta, n_points, n_fine=N_FINE):
    """Convert Fourier coefficients to discrete function h"""
    # 1. Extend theta to full vector (including DC)
    k = np.arange(1, K_HARMONICS + 1)
    a = theta[0:K_HARMONICS]
    b = theta[K_HARMONICS:2*K_HARMONICS]

    # 2. Create periodic interpolation on fine grid
    x = np.linspace(0, 2, n_fine)
    h_raw = 0.5 + np.sum([a[i-1] * np.cos(np.pi*i*x) + b[i-1] * np.sin(np.pi*i*x)
                          for i in range(1, K_HARMONICS + 1)], axis=0)

    # 3. Clip and project to integral = 1
    h_clipped = np.clip(h_raw, 0, 1)
    # Project to correct sum using uniform shift
    h_projected = project_to_sum_box(h_clipped, n_points / 2.0)

    # 4. Downsample to target grid size (preserves exact sum)
    indices = np.linspace(0, n_fine - 1, n_points, dtype=int)
    return h_projected[indices]

def obj_fourier(theta):
    """Return C5 evaluated at Fourier coefficients"""
    h = fourier_to_h(theta, N_TARGETS[-1])
    return evaluate_C5(h)

def local_improve(h_start, budget_s, start_time):
    """Perform gradient-based local refinement (kind of projected gradient)"""
    h = h_start.copy()
    t0 = time.time()
    max_iter = 1000
    step_size = 0.01
    for i in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
        # Compute current minimum lag and its corresponding correlation
        corr = compute_corr_fft_vectorized(h)
        min_lag_idx = np.argmax(corr)
        min_corr = corr[min_lag_idx]
        if min_corr == 0:  # Special case
            break
        # Compute gradient on h for that lag only
        # Ignoring gradients for other lags
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            if i - min_lag_idx >= 0:
                gradient[i] += h[i - min_lag_idx]
            if i + min_lag_idx < len(h):
                gradient[i] += h[i + min_lag_idx]
        # Move in direction of negative gradient
        trial_h = h - step_size * gradient
        trial_h = project_to_sum_box(trial_h, TARGET_SUM)
        new_c5 = evaluate_C5(trial_h)
        if new_c5 < evaluate_C5(h):
            h = trial_h
    return h

def binary_swap_optimization(h, max_swaps=400):
    """Run binary swap heuristic"""
    h = np.array(h)
    n = len(h)
    indices_1 = np.argsort(h)[-n//2:]
    indices_0 = np.argsort(h)[:n//2]
    indices_1_sorted = indices_1[np.argsort(h[indices_1])[::-1]]
    indices_0_sorted = indices_0[np.argsort(h[indices_0])]
    # Precompute all correlations
    corr = compute_corr_fft_vectorized(h)
    worst_lag_index = np.argmax(corr)
    # Iterate swapping
    swaps = 0
    while swaps < max_swaps:
        swaps += 1
        # For each item at high value position, look for good candidate to swap
        for i in indices_1_sorted:
            if time.time() > 0.95: break
            potential_swap = -1
            best_improvement = 0
            for j in indices_0_sorted:
                # Temporarily swap
                h_temp = h.copy()
                h_temp[i], h_temp[j] = h_temp[j], h_temp[i]
                h_corr = compute_corr_fft_vectorized(h_temp)
                new_max_corr = np.max(h_corr)
                if new_max_corr > corr[worst_lag_index] + 1e-6:
                    if new_max_corr - corr[worst_lag_index] > best_improvement:
                        best_improvement = new_max_corr - corr[worst_lag_index]
                        potential_swap = j
            if potential_swap != -1:
                h[i], h[potential_swap] = h[potential_swap], h[i]
                corr = compute_corr_fft_vectorized(h)
                worst_lag_index = np.argmax(corr)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None

    # Step 1: Multi-scale optimization with DE
    for N in N_TARGETS:
        # Prepare for this resolution
        target_sum = N / 2.0
        bounds = [(-MAX_COEFF, MAX_COEFF) for _ in range(2*K_HARMONICS)]
        
        # DE with custom strategy
        de_result = differential_evolution(
            obj_fourier,
            bounds=bounds,
            maxiter=200,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.8,
            seed=seed + N,
            disp=False,
            polish=False
        )

        h_candidate = fourier_to_h(de_result.x, N)
        c5_candidate = evaluate_C5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # Step 2: Local refinement using gradient ascent
    if best_h is not None:
        refined_h = local_improve(best_h, budget_s, start_time)
        c5_refined = evaluate_C5(refined_h)
        if c5_refined < best_c5:
            best_h = refined_h
            best_c5 = c5_refined
    
    # Step 3: Binary swap polish
    if best_h is not None:
        silver_h = binary_swap_optimization(best_h, 2*len(best_h))
        silver_c5 = evaluate_C5(silver_h)
        if silver_c5 < best_c5:
            best_c5 = silver_c5
            best_h = silver_h

    return (best_h.tolist(), float(best_c5), len(best_h))
