import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move to reduce C5"""
    h = h.copy()
    # Compute current correlation to find worst lag
    corr = compute_corr_fft(h)
    worst_lag = np.argmax(corr)
    
    # Use analytic gradient of the worst correlation term
    for iteration in range(max_iterations):
        # Compute gradients for each position based on correlation to the worst lag position
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution to correlation at lag = worst_lag
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Find best move
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
            
        # Move small amount
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta <= 0:
            break
            
        h[idx_min] -= delta
        h[idx_max] += delta
        
        # Project to constraints and fix sum
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check convergence
        if evaluate_c5(h) < 0.379:
            break
            
    return h

def optimize_step(h, step_size=0.02):
    """Perform gradient free optimization step using neighborhood search"""
    h_new = h.copy()
    best_h = h_new.copy()
    best_c5 = evaluate_c5(h_new)
    
    # Sample small changes around current h
    for i in range(n_points):
        for delta in [-step_size, step_size]:
            h_trial = h_new.copy()
            h_trial[i] = max(0, min(1, h_trial[i] + delta))
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
    
    return best_h

# Different strategy: Use evolutionary approach with population
def evolutionary_search():
    """Use a simple evolutionary strategy to improve solution"""
    population_size = 30
    generations = 200
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    for generation in range(generations):
        # Evaluate population
        fitness = [evaluate_c5(h) for h in population]
        
        # Find best to keep
        best_idx = np.argmin(fitness)
        best = population[best_idx]
        
        # Create new population by mutation
        new_population = [best]  # Elitism
        for _ in range(population_size - 1):
            # Select parent
            parent = population[np.random.randint(0, population_size)]
            # Mutate
            h_mut = parent.copy()
            # Randomly perturb some elements
            num_mut = max(1, n_points // 20)
            indices = np.random.choice(n_points, num_mut, replace=False)
            for idx in indices:
                h_mut[idx] = max(0, min(1, h_mut[idx] + np.random.normal(0, 0.01)))
            h_mut = project_to_sum_box(h_mut, TARGET_SUM)
            new_population.append(h_mut)
        
        population = new_population
    
    # Return the best from final population
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx], fitness[best_idx]
