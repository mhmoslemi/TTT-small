import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints with exact sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_exact(h):
    """Exact evaluation using direct convolution"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def get_lag_indices(h, k):
    """Get indices for lag k in correlation"""
    n = len(h)
    if k == 0:
        return list(range(n))
    elif k > 0:
        return list(range(n-k))
    else:
        return list(range(-k, n))

def simulated_annealing(seed=42, time_limit=420):
    """Simulated Annealing with elitism and crossover"""
    np.random.seed(seed)
    
    # Initialize elite pool
    elites = []
    elite_size = 5
    
    # Initial random binary solution
    h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    h = project_to_sum_box(h, TARGET_SUM)
    current_c5 = evaluate_c5(h)
    
    # Temperature control
    T0 = 0.02
    alpha = 0.995
    T_min = 1e-5
    T = T0
    l = n_points   # sweep size
    
    # Time tracking
    start_time = time.time()
    restart_interval = 0.1 * time_limit
    crossover_interval = 0.05 * time_limit
    
    # Main loop
    move_count = 0
    while T > T_min and time.time() - start_time < time_limit:
        # Check for restarts
        if time.time() - start_time > restart_interval:
            # Add to elite if not full or better than worst
            if len(elites) < elite_size or current_c5 < elites[-1][0]:
                with_best = [(current_c5, h.copy())]
                if len(elites) < elite_size:
                    elites.extend(with_best)
                else:
                    elites[-1] = with_best[0]
                elites.sort(key=lambda x: x[0])
            # Restart with elite or fresh random solution
            if len(elites) > 0 and np.random.rand() < 0.3:
                h = elites[np.random.randint(len(elites))][1].copy()
                new_c5 = evaluate_c5(h)
            else:
                h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
                h = project_to_sum_box(h, TARGET_SUM)
                new_c5 = evaluate_c5(h)
            current_c5 = new_c5
        
        # Test crossover every crossover_interval seconds
        if time.time() - start_time > crossover_interval and len(elites) >= 2:
            # Crossover between two elites
            i, j = np.random.choice(len(elites), 2, replace=False)
            parent1 = elites[i][1]
            parent2 = elites[j][1]
            
            cutoff = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:cutoff], parent2[cutoff:]])
            # Repair weight
            current_weight = np.sum(child)
            diffs = np.sum(child) - TARGET_SUM
            if diffs > 0:
                # Remove extras
                indices = np.where(child == 1)[0]
                to_remove = np.random.choice(indices, int(diffs), replace=False)
                child[to_remove] = 0
            elif diffs < 0:
                # Insert extras
                indices = np.where(child == 0)[0]
                to_add = np.random.choice(indices, int(-diffs), replace=False)
                child[to_add] = 1
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_c5 = evaluate_c5(child)
            
            # Update elites
            if len(elites) < elite_size or new_c5 < elites[-1][0]:
                with_best = [(new_c5, child)]
                if len(elites) < elite_size:
                    elites.extend(with_best)
                else:
                    elites[-1] = with_best[0]
                elites.sort(key=lambda x: x[0])
            h = child.copy()
            current_c5 = new_c5
            
        # Make a move
        # Pick a 1-index and 0-index to swap
        one_idx = np.random.choice(np.where(h == 1)[0])
        zero_idx = np.random.choice(np.where(h == 0)[0])
        
        # Try to make the move
        h_new = h.copy()
        h_new[one_idx] = 0
        h_new[zero_idx] = 1
        
        old_c5 = current_c5
        new_c5 = evaluate_c5(h_new)

        # Accept or reject
        if new_c5 <= old_c5:
            h = h_new
            current_c5 = new_c5
        else:
            p_accept = np.exp(-(new_c5 - old_c5) / T)
            if np.random.rand() < p_accept:
                h = h_new
                current_c5 = new_c5
                
        move_count += 1
        
        # Cooling schedule
        if move_count % l == 0:
            T *= alpha
            
    return h, current_c5

def hill_climb(h_best):
    """Exhaustive hill climbing"""
    h = h_best.copy()
    improved = True
    while improved:
        improved = False
        best_improvement = 0
        best_i, best_j = None, None
        
        # Bin indices for efficiency
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try all swaps
        for i in ones:
            for j in zeros:
                # Compute impact of the swap without applying it
                # Recomputing C5 using FFT would be slower - use direct comparison when feasible
                h_test = h.copy()
                h_test[i] = 0
                h_test[j] = 1
                
                c5_new = evaluate_c5(h_test)
                if c5_new < evaluate_c5(h):
                    if c5_new < evaluate_c5(h) - best_improvement:
                        best_improvement = evaluate_c5(h) - c5_new
                        best_i = i
                        best_j = j
                        improved = True
                        
        if improved:
            h[best_i] = 0
            h[best_j] = 1
            
    return h

def focus_milp(h, time_limit=90):

    start_time = time.time()
    try:
        # Identify worst lag  
        corr = correlate(h, 1-h, mode="full")
        worst_lag = np.argmax(corr)
        if worst_lag >= len(h):
            worst_lag -= len(h) - 1
        # Get indices contributing to worst lag
        j_indices = []
        for i in range(len(h)):
            if i + worst_lag < len(h):
                j_indices.append(i)
            if i - worst_lag >= 0:
                j_indices.append(i)
        idx_set = set(j_indices)
        # Choose up to 20 best contributing positions
        if len(idx_set) > 20:
            sums = []

                products = []
                for j in range(len(h)):
                    if i + worst_lag == j:
                        products.append(h[i] * h[j])
                # Use dot product approximation for efficiency  
                score = h[i] * (1 - h[i+worst_lag]) if i+worst_lag < len(h) else 0
                sums.append((score, i))
            sums.sort(reverse=True)
            selected_indices = [i[1] for i in sums[:20]]
        else:
            selected_indices = list(idx_set)
        selected_indices.sort()
        
        if not selected_indices:
            return h
        
        # Build MILP model
        n_vars = len(selected_indices)
        h_fixed = h.copy()
        
        # Create integer variables for selected positions
        z = cp.Variable(n_vars, boolean=True)
        # Define a dummy variable t to capture maximum overlap constraint
        t = cp.Variable()
        
        # Weight constraint
        weight_constr = cp.sum(z) == TARGET_SUM - (np.sum(h_fixed) - len(selected_indices) * np.sum(h_fixed[selected_indices]))
        
        # Lag constraints - compute which pairs are relevant
        constraints = []
        
        # This is a simpler high-level version - direct constraint on all lags but only focus on worst
        # We just enforce:
        # t >= M - A_k(h) where A_k = sum over i where h_i * h_{i+k} == 1   
        # We simplify by focusing on worst lag and a small neighborhood
        
        # Simple approach - check with just a few key lag terms for MILP.
        # No full MILP in this version - just check if simple fixes help
        # MILP version would be complex, so we will look for simple improvements
        
        # Return a slightly perturbed version if a move helps
        worst_corr_val = np.max(correlate(h, 1-h, mode="full")) * dx
        h_candidate = h.copy()
        best_change = 0
        
        # Try small local modifications 
        for i in selected_indices:
            test_h = h.copy()
            test_h[i] = 1 - test_h[i]  # flip
            new_c5 = evaluate_c5(test_h)
            if new_c5 < worst_corr_val and new_c5 < worst_corr_val - best_change:
                best_change = worst_corr_val - new_c5
                h_candidate = test_h.copy()
        
        return h_candidate if best_change > 0 else h
    except Exception:

def continuous_polish(h):
    """SLSQP continuous optimization"""
    h_float = h.copy()
    
    def objective(x):
        corr = correlate(x, 1-x, mode="full")
        return dx * np.max(corr)
    
    def constraint_eq(x):
        return np.sum(x) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h_float, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-7, 'gtol': 1e-7})
        if res.success:
            h_float = res.x
            h_float = project_to_sum_box(h_float, TARGET_SUM)
            return h_float
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Step 1: Simulated Annealing
    h_sa, c5_sa = simulated_annealing(seed, 0.7 * budget_s)
    best_h = h_sa.copy()
    best_c5 = c5_sa
    
    # Step 2: Hill Climbing
    if time.time() - start_time < 0.75 * budget_s:
        h_hc = hill_climb(best_h)
        c5_hc = evaluate_c5(h_hc)
        if c5_hc < best_c5:
            best_h = h_hc
            best_c5 = c5_hc
            
    # Step 3: MILP-based refinement  
    if time.time() - start_time < 0.9 * budget_s:
        h_milp = focus_milp(best_h, 0.15 * budget_s)
        c5_milp = evaluate_c5(h_milp)
        if c5_milp < best_c5:
            best_h = h_milp
            best_c5 = c5_milp
    
    # Step 4: Continuous polishing
    if time.time() - start_time < 0.95 * budget_s:
        h_cont = continuous_polish(best_h)
        c5_cont = evaluate_c5(h_cont)
        if c5_cont < best_c5:
            best_h = h_cont
            best_c5 = c5_cont
            
    # Final exact evaluation  
    best_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
