import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft

# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for soft max approximation"""
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_shifted = np.exp(alpha * shifted)
    return exp_shifted / np.sum(exp_shifted)

def compute_gradient(h, weights):
    """Compute gradient using FFT for convolution"""
    n = len(h)
    # Forward convolution: sum of weights*shifted h
    w_f = np.zeros(2*n - 1)
    w_f[:len(weights)] = weights
    
    # Compute FFT of h and weight vector
    h_fft = fft(h, 2*n)
    w_fft = fft(w_f, 2*n)
    
    # Compute convolutions
    conv_forward = ifft(h_fft * w_fft).real[:n]
    conv_backward = ifft(h_fft * np.conj(w_fft)).real[:n]
    
    # Gradient: 1 - forward - backward
    grad = 1.0 - conv_forward - conv_backward
    return grad

def projected_gradient_descent(h0, max_iter=3000, alpha_start=5.0, alpha_max=20.0, eta_init=0.5, eta_max=1.0):
    """Perform projected gradient descent with soft max approximation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = alpha_start
    eta = eta_init
    last_improvement = 0
    
    for i in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        weights = softmax_weights(corr, alpha)
        grad = compute_gradient(h, weights)
        
        # Backtracking line search
        eta_test = eta
        h_trial = h - eta_test * grad
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        
        while c5_trial >= best_c5 and eta_test > 1e-8:
            eta_test /= 2
            h_trial = h - eta_test * grad
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
        
        h = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
            last_improvement = i
        
        # Adaptive updates
        eta = min(eta_max, eta * 1.02)
        alpha = min(alpha_max, alpha * 1.01)
        
        # Early stopping
        if i - last_improvement > 1000:
            break
    
    return best_h, best_c5
