import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def guided_move(h, worst_lag):
    """Guided deterministic move for the worst lag to minimize the correlation"""
    n = len(h)
    # Compute subgradient g_i = h_{i-\ell} + h_{i+\ell}
    g = np.zeros(n)
    for i in range(n):
        left_idx = i - worst_lag
        right_idx = i + worst_lag
        if left_idx >= 0:
            g[i] += h[left_idx]
        if right_idx < n:
            g[i] += h[right_idx]
    
    # Find indices with min and max g
    i_min = np.argmin(g)
    i_max = np.argmax(g)
    
    if i_min == i_max:
        return None  # No improvement possible
    
    # Move a small amount from i_min to i_max
    delta_max = min(h[i_min], 1.0 - h[i_max])
    if delta_max <= 0:
        return None
    
    delta = min(delta_max, 0.01)  # Small fixed step
    return i_min, i_max, delta

def local_refine(h_start):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    try:
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h_start, evaluate_c5(h_start)

def almost_binary_h(n_points, target_sum):
    """Create almost binary h to guide optimization"""
    h = np.zeros(n_points)
    # Place 50% of it as 1s
    ones_count = int(target_sum)
    ones_idx = np.random.choice(n_points, ones_count, replace=False)
    h[ones_idx] = 1.0
    
    # Project to ensure constraint
    h = project_to_sum_box(h, target_sum)
    return h

def compute_smoothed_max(h, alpha=5.0):
    """Compute smoothed approximation of max over all lags"""
    n = len(h)
    # Compute all correlation values with full convolution
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    dx = 2.0 / n
    
    # Apply smoothing to max
    corr_dx = corr * dx  # Scale by dx to match actual c5 values
    
    # Compute negative correlation values (we want to minimize max of -(corr))
    corr_neg = -corr_dx
    # Apply log-sum-exp with smoothing parameter alpha
    exp_terms = np.exp(alpha * corr_neg)
    # We want to minimize -log(∑ exp(α * -corr)) = log(∑ exp(-α * corr))
    smooth_max = np.log(np.sum(exp_terms)) / alpha
    return smooth_max

def compute_gradient(h, alpha=5.0):
    """Compute gradient of smoothed max with respect to h"""
    n = len(h)
    # Compute all correlation values
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    dx = 2.0 / n
    
    # Apply smoothing weights
    corr_scaled = corr * dx
    corr_neg = -corr_scaled
    exp_terms = np.exp(alpha * corr_neg)
    sum_exp = np.sum(exp_terms)
    
    if sum_exp == 0:
        return np.zeros(n)
    
    # Compute w_k weights
    weights = exp_terms / sum_exp
    
    # Compute gradient d/dh_i of sum_k w_k * (-corr_k) 
    # where corr_k = sum_j h_j * (1 - h_{j+k}) = sum_j h_j - sum_j h_j h_{j+k}
    # so d/dh_i corr_k = - d/dh_i (sum_j h_j h_{j+k})
    # for fixed k, the derivative of sum_j h_j h_{j+k} w.r.t h_i:
    # if i = j+k or i = j-k: we get h_{i-k} or h_{i+k}  
    # For each k: sum all contributions to grad
    grad = np.zeros(n)
    # We compute ∂/∂h_i (-∑_k w_k * corr_k) 
    # = ∑_k w_k * ∂/∂h_i corr_k
    # ∂/∂h_i corr_k = - ∂/∂h_i (sum_j h_j h_{j+k}) = 
    # - (∂/∂h_i h_{i+k} + ∂/∂h_i h_{i-k}) = - (h_{i+k} + h_{i-k})
    # Note in the sum correlation over all lags
    
    # Precompute sum of weights
    for k in range(2*n - 1):
        # k corresponds to lag: k = 0 to 2*n-2, represent different lags from -n+1 to n-1
        lag = k - (n - 1)  # Convert to lag: -n+1 <= lag <= n-1
        if lag == 0:
            # Special case: dif_{i,i+k} = di_{i,i} = 1
            for i in range(n):
                grad[i] += weights[k] * 2 * h[i]
        else:
            # The gradient of sum_i h_i h_{i+lag} w.r.t h_i = h_{i+lag} + h_{i-lag}
            # Have to be careful about boundary values
            if lag > 0:
                # Δ = sum_{i=0}^{n-1-lag} (h_{i+lag} + h_{i-lag})
                # independent of i (getattrῶδ)? No, each term in derivative depends on i.
                # Contribution to grad[i]: weight[k] * (h_{i+lag} + h_{i-lag})
                # All terms in gradient for i (valid i+lag < N and i-lag >= 0):
                # For indices where i+lag valid (i <= N-1-lag), we get h_{i+lag}
                # For indices where i-lag valid (i >= lag), we get h_{i-lag}
                if lag < n:
                    for i in range(n - lag):
                        grad[i] += weights[k] * h[i + lag]
                    for i in range(lag, n):
                        grad[i] += weights[k] * h[i - lag]
    
    return -grad

def project_gradient_step(h, grad, step_size=0.01):
    """Perform projected gradient descent step"""
    h_new = h - step_size * grad
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new
