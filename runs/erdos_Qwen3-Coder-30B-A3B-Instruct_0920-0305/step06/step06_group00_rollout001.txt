import numpy as np
from scipy.signal import correlate

TARGET_SUM = None

def evaluate_erdos_solution(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using binary search."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial search bounds
    a, b = -np.max(x) - 1.0, 1.0 - np.min(x) + 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    
    # Adjust to ensure exact sum
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        free = (result > 1e-12) & (result < 1-1e-12)
        if free.any():
            result[free] += diff / free.sum()
            result = np.clip(result, 0, 1)
    
    return result

def full_corr(a, b):
    """Compute full correlation of two sequences a, b with zero padding"""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def gradient_soft_max(h, alpha):
    """Compute gradient of soft max over all lags using FFT."""
    n = len(h)
    corr = full_corr(h, 1 - h)  # This is 2*n-1 correlations
    maxc = corr.max()
    # Soft weights
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w = w / w_sum
    
    # Convolution-based gradient computation
    # We compute: grad[i] = sum_l w[l] * (1-h[i+l] - h[i-l])
    # This involves shifting: forward h shift by l, backward h shift by -l
    # We'll repeat shifted operations using FFT
    conv_fwd = full_corr(h, w)
    # Reverse w and convolve for backward term
    w_rev = w[::-1]  # Reverse weight vector for backward convolution
    conv_bwd = full_corr(h, w_rev)
    
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def optimize_continuous(start_h, remaining_time, n, dx, target_sum, alpha_start=5.0, alpha_end=20.0):
    """Run projected gradient descent with soft max."""
    h = start_h.copy()
    best_h = h.copy()
    corr = full_corr(h, 1 - h)
    best_val = dx * np.max(corr)
    last_improvement = 0
    alpha = alpha_start
    eta = 0.05  # Step size
    
    iter_count = 0
    max_iter = 5000
    start_time = time.time()
    
    while time.time() - start_time < remaining_time and iter_count < max_iter:
        iter_count += 1
        # Update alpha gradually to increase focus
        if iter_count > 100:
            alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, iter_count / 1000.0)
        
        grad = gradient_soft_max(h, alpha)
        # Adaptive step size
        step_size = eta
        for _ in range(10):
            h_new = project_to_sum_box(h - step_size * grad, target_sum)
            corr_new = full_corr(h_new, 1 - h_new)
            new_val = dx * np.max(corr_new)
            if new_val < best_val - 1e-12:
                h = h_new
                corr = corr_new
                best_val = new_val
                if new_val < best_val:
                    best_h = h.copy()
                last_improvement = iter_count
                break
            step_size /= 2.0
            if step_size < 1e-8:
                break
        else:
            if iter_count - last_improvement > 50:
                # Perturb if no improvement in a while
                h += np.random.normal(0, 0.01, n)
                h = project_to_sum_box(h, target_sum)
                last_improvement = iter_count
    
    return best_h, best_val

def binary_local_search(h_bin, remaining_time, n, dx):
    """Run hill climbing on binary h"""
    start_time = time.time()
    best_h = h_bin.copy()
    corr = full_corr(best_h, 1 - best_h)
    best_val = dx * np.max(corr)
    
    while time.time() - start_time < remaining_time:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        # compute new C5
        corr_new = full_corr(cand, 1 - cand)
        val = dx * np.max(corr_new)
        if val < best_val:
            best_val = val
            best_h = cand
    return best_h, best_val
