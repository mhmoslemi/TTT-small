import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_gradients(h0, max_iter=2000):
    """Optimize using gradient-based method with projection"""
    h = h0.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
        h = project_to_sum_box(h, TARGET_SUM)
    except:
        pass
    return h

def local_search_improvement(h0, max_iter=200):
    """Improved local search that makes larger improvements"""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Find the index with highest contribution to the C5
        corr_full = compute_corr_fft(h)
        max_idx = np.argmax(corr_full)
        
        # Perturb around the max correlation index
        # Try to decrease it by increasing some values and decreasing others
        idx_start = max_idx - n_points + 1
        if idx_start < 0:
            idx_start = 0
            
        # Create a small set of candidates
        candidates = []
        for j in range(max(0, idx_start), min(len(h), idx_start + 20)):
            # Try small perturbations
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h.copy()
                h_new[j] += delta
                if 0 <= h_new[j] <= 1:
                    candidates.append((j, delta))
        
        best_improvement = 0
        best_move = None
        
        for j, delta in candidates:
            h_new = h.copy()
            h_new[j] += delta
            
            if h_new[j] < 0 or h_new[j] > 1:
                continue
                
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            new_c5 = evaluate_c5(h_new)
            improvement = best_c5 - new_c5

                best_move = (j, delta)

            j, delta = best_move
            h[j] += delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)

            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    # Phase 3: Refinement with hybrid approach
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Try a few local strategies to escape local minima
        strategies = [
            lambda h: optimize_with_gradients(h, max_iter=500),
            lambda h: local_search_improvement(h, max_iter=300),
        ]
        
        h_current = best_h.copy()
        for strategy in strategies:
            h_temp = strategy(h_current)
            c5_temp = evaluate_c5(h_temp)
            if c5_temp < best_c5 * 0.99999:  # Only accept significant improvements
                best_h = h_temp.copy()
                best_c5 = c5_temp

            # If no substantial improvement, add noise and reoptimize
            h_noise = best_h + np.random.normal(0, 0.005, n_points)
            h_noise = project_to_sum_box(h_noise, TARGET_SUM)
            h_final = optimize_with_gradients(h_noise, max_iter=500)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (best_h.tolist(), float(best_c5), n_points)
