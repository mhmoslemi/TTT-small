import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # FFT length (next power of 2 for speed)

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
    
    # Initial solution
    h = h_interpolated.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    TARGET_SUM = n_points / 2
    
    # Parameters
    lambda_start = 0.05
    lambda_rate = 1.02
    beta_start = 0.5
    beta_rate = 1.01
    reset_interval = 300
    max_resets = 5
    lambda_val = lambda_start
    beta = beta_start
    reset_count = 0
    no_improvement_count = 0
    
    # Precompute zero-padded
    h_padded = np.zeros(L)
    h_padded[:n_points] = h
    
    while time.time() - start_time < budget_s - 10:
        # FFT of h_padded
        H = np.fft.fft(h_padded)
        P = np.abs(H)**2
        
        # Calculate target magnitude
        P_offdc = P[1:]  # Exclude DC
        P_target = np.mean(P_offdc)  # Mean of off-DC powers
        M_target = np.full_like(np.abs(H), np.sqrt(P_target))
        M_target[0] = np.abs(H[0])  # Keep DC component unchanged
        
        # Blend magnitude
        M = (1 - lambda_val) * np.abs(H) + lambda_val * M_target
        
        # Phase preservation
        H_new = M * np.exp(1j * np.angle(H))
        
        # IFFT
        h_cand_padded = np.fft.ifft(H_new).real
        h_cand = h_cand_padded[:n_points]
        
        # Project onto feasibility constraints
        h_cand = project_to_sum_box(h_cand, TARGET_SUM)
        
        # Relaxation to avoid oscillations
        h_new = (1 - beta) * h + beta * h_cand
        h = h_new
        
        # Evaluate new C5
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adjust parameters
        lambda_val = min(lambda_val * lambda_rate, 1.0)
        beta = min(beta * beta_rate, 1.0)
        
        # Phase reset if no improvement
        if no_improvement_count > reset_interval and reset_count < max_resets:
            H = np.fft.fft(h_padded)
            # Set new random phases to the magnitude (keeping the magnitude fixed)
            H_random_phase = np.abs(H) * np.exp(1j * np.random.uniform(0, 2*np.pi, L))

            h_cand = project_to_sum_box(h_cand, TARGET_SUM)
            
            # Relax with original h
            h_new = (1 - beta) * h + beta * h_cand
            h = h_new
            
            no_improvement_count = 0
            reset_count += 1
        
        # Update padded version for next iteration
        h_padded[:n_points] = h
    
    # Final polishing with SLSQP
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_eq(h_vec):
        return np.sum(h_vec) - TARGET_SUM
    
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            best_h = project_to_sum_box(res.x, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
