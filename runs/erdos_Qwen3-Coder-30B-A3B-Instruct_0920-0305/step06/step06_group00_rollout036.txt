def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_pso(h0, max_iter=1000, swarm_size=30):
    """Particle Swarm Optimization with adaptive parameters"""
    global TARGET_SUM
    n = len(h0)
    # Initialize swarm
    swarm = np.random.uniform(0, 1, (swarm_size, n))
    for i in range(swarm_size):
        swarm[i] = project_to_sum_box(swarm[i], TARGET_SUM)
    
    velocities = np.zeros_like(swarm)
    personal_best = swarm.copy()
    personal_best_values = np.array([evaluate_c5(p) for p in personal_best])
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO parameters
    w_min, w_max = 0.4, 0.9
    c1, c2 = 2.0, 2.0
    inertia = w_max
    
    for iteration in range(max_iter):
        # Adapt inertia weight
        inertia = w_max - (w_max - w_min) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_sum_box(swarm[i], TARGET_SUM)
            
            # Evaluate fitness
            current_value = evaluate_c5(swarm[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_value
    
    return global_best, global_best_value
