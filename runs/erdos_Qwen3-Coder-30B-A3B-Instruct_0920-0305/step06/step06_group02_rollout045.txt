import numpy as np
import time
import random

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Generate a binary vector with exact sum constraint"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_idx = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_idx] = 1
    return x

def full_overlap(x):
    """Compute full overlap array O_s = sum h_i h_{i+s} for all shifts s"""
    return np.correlate(x, x, 'full')

def swap_update(ov, i, j, ones, zeros):
    """Incrementally update overlap array after swapping position i (1) with j (0)"""
    ov_new = ov.copy()
    n = len(ones)  # number of 1's
    # i is set to 0 (remove contributions)
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n - 1] -= 1
    
    # j is set to 1 (add contributions)  
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n - 1] += 1
    
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Perform simulated annealing with incremental updates"""
    start = time.time()
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = ov.min()
    
    # Set initial temperature
    temp0 = 0.5
    temp_min = 1e-4
    lambda_cool = -np.log(temp0 / temp_min) / 10000  # cool over 10k steps
    
    current_ov = ov.copy()
    current_min = ov.min()
    current_x = x.copy()
    iteration = 0
    
    while time.time() - start < time_limit:
        # Temperature cooling
        t = time.time() - start
        temp = temp0 * np.exp(-lambda_cool * iteration)
        if temp < temp_min:
            temp = temp_min
            
        # Propose a swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Perform incremental update
        new_ov = swap_update(current_ov, i, j, ones, zeros)
        new_min = new_ov.min()
        
        # Acceptance probability
        delta = new_min - current_min
        if delta > 0 or random.random() < np.exp(delta / temp):
            # Accept move
            current_ov = new_ov
            current_min = new_min
            current_x[i] = 0
            current_x[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            
            # Update best solution if improved
            if current_min > best_min:
                best_x = current_x.copy()
                best_ov = current_ov.copy()
                best_min = current_min
                
        iteration += 1
        
        # Terminate if no improvement in long time
        if iteration % 5000 == 0 and iteration > 0:
            if temp < temp_min:
                break
    
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Try all beneficial swaps for the current worst shift"""
    worst_shift = np.argmin(ov) - (n_points - 1)
    current_min = ov.min()
    
    # Identify deficient and beneficial positions
    A = set()
    B = set()
    
    for i in ones:
        j = i + worst_shift
        if 0 <= j < n_points and x[j] == 0:
            A.add(i)
    
    for i in zeros:
        j = i + worst_shift
        if 0 <= j < n_points and x[j] == 1:
            B.add(i)
    
    # Try all pairs (i,j) where i is in A and j in B  
    A_list = list(A)
    B_list = list(B)
    random.shuffle(A_list)
    random.shuffle(B_list)
    
    start = time.time()
    tries = 0
    early_exit = False
    
    for i in A_list:
        for j in B_list:
            tries += 1
            if tries > 1000 or time.time() - start > 0.5:
                early_exit = True
                break
                
            # Create new overlap array with swap
            new_ov = swap_update(ov, i, j, ones, zeros)
            
            # Check if new overlap is still no smaller than before
            if new_ov.min() >= current_min:
                # Apply swap and update
                ov = new_ov
                x[i] = 0
                x[j] = 1
                ones.remove(i)
                zeros.remove(j)
                ones.add(j)
                zeros.add(i)
                current_min = ov.min()
                break
                
        if early_exit:
            break
    
    return x, ov, current_min

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization loop"""
    # Initialize
    np.random.seed(seed)
    start_time = time.time()
    
    # Run multiple restarts
    best_c5 = float('inf')
    best_h = None
    
    # Take some samples for time management
    num_restarts = 10
    restart_time = budget_s * 0.95 / num_restarts
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize
        x = initial_binary(seed + restart)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        min_val = ov.min()
        
        # Simulated Annealing
        final_x, final_min = simulated_annealing(x, ov, ones, zeros, restart_time * 0.9)
        
        # Post-processing: refine worst shift
        ones_final = set(np.where(final_x == 1)[0])
        zeros_final = set(np.where(final_x == 0)[0])
        ov_final = full_overlap(final_x)
        _, ov_final, final_min = worst_shift_refine(final_x, ov_final, ones_final, zeros_final)
        
        # Compute C5
        c5 = (n_points / 2 - final_min) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = final_x.astype(float)
    
    # Early exit or return best
    if best_h is None:
        # Fallback to simple random
        x = initial_binary(seed)
        ov = full_overlap(x)
        c5 = (n_points / 2 - ov.min()) * dx
        best_h = x.astype(float)
        best_c5 = c5
        
    return (best_h.tolist(), float(best_c5), n_points)
