import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap corresponds to the maximum of the correlation
    # but we must be careful with indexing
    max_corr = np.max(corr)
    return dx * max_corr

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary variables"""
    # Create variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == n // 2,
        t >= 0
    ]

    # Add linking constraints: y[i,j] = h[i] AND h[j]
    for i in range(n):
        for j in range(i+1, n):
            # y[i,j] <= h[i], y[i,j] <= h[j]
            constraints.append(y[i,j] <= h[i])
            constraints.append(y[i,j] <= h[j])
            # y[i,j] >= h[i] + h[j] - 1
            constraints.append(y[i,j] >= h[i] + h[j] - 1)
            # y[j,i] = y[i,j]
            y[j,i] = y[i,j]

    # Add lag constraints: t >= n/2 - sum of matching pairs for each lag
    for k in range(1, n):
        # Sum of y[i, i+k] for valid indices
        lag_sum = 0
        for i in range(n - k):
            lag_sum += y[i, i+k]
        constraints.append(t >= n/2 - lag_sum)

    # Reduce symmetry by fixing h[0] = 1
    constraints.append(h[0] == 1)

    # Objective
    objective = cp.Minimize(t)

    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        # Check if solution is valid
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.rint(h.value).astype(int)
            t_val = t.value
            return h_sol, t_val
    except:
        pass
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result
