import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from scipy.linalg import eigh

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n, dx, M, time_limit_s):
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Matrices for lag constraints
    P_list = []
    for k in range(n):
        P = np.zeros((n, n))
        for i in range(n - k):
            P[i, i + k] = 1
        P_list.append(P)

    constraints = [
        cp.trace(X) == M,
        X >> 0,
        cp.diag(X) == h,
        1 @ h == M,
        h >= 0,
        h <= 1,
    ]

    for P_k in P_list:
        constraints.append(t >= M - cp.trace(P_k @ X))

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.SCS, time_limit=time_limit_s, eps=1e-5)
    except:
        pass

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        X_sol = X.value
        t_val = t.value
    else:
        # fallback if no solution from SCS
        h_sol = h.value if h.value is not None else np.ones(n) * 0.5
        X_sol = np.eye(n) if X.value is None else X.value
        t_val = t.value if t.value is not None else np.inf

    return h_sol, X_sol, t_val

def rounding_and_hill_climb(h_rel, X_rel, M, n, dx, num_rounding_trials=20):
    # Eigendecomposition of X_rel
    V, Lambda = eigh(X_rel)
    V = V[:, Lambda > 1e-8]
    Lambda = Lambda[Lambda > 1e-8]
    V = V @ np.diag(np.sqrt(Lambda))

    best_h = None
    best_c5 = np.inf

    # Deterministic threshold rounding
    h_threshold = np.zeros(n)
    idx_sort = np.argsort(-h_rel)
    h_threshold[idx_sort[:int(M)]] = 1
    h_threshold = project_to_sum_box(h_threshold, M)
    corr = compute_corr_fft(h_threshold)
    c5 = dx * c5_from_corr(corr, n)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_threshold.copy()

    # Randomized rounding
    for _ in range(num_rounding_trials):
        g = np.random.randn(V.shape[1])
        s = V @ g
        h_rounded = (s >= 0).astype(float)
        # Fix the sum to be M
        sum_h = np.sum(h_rounded)
        if sum_h > M:
            flip_indices = np.where(h_rounded == 1)[0]
            flip_indices = np.random.choice(flip_indices, size=int(sum_h - M), replace=False)
            h_rounded[flip_indices] = 0
        elif sum_h < M:
            flip_indices = np.where(h_rounded == 0)[0]
            flip_indices = np.random.choice(flip_indices, size=int(M - sum_h), replace=False)
            h_rounded[flip_indices] = 1
        corr = compute_corr_fft(h_rounded)
        c5 = dx * c5_from_corr(corr, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()

    # Hill climbing from the best found solution
    if best_h is not None:
        h_best = best_h.copy()
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if h_best[i] == 1:
                    for j in range(n):
                        if h_best[j] == 0:
                            # Make flip
                            h_flip = h_best.copy()
                            h_flip[i] = 0
                            h_flip[j] = 1
                            corr = compute_corr_fft(h_flip)
                            c5_flip = dx * c5_from_corr(corr, n)
                            if c5_flip < best_c5:
                                best_c5 = c5_flip
                                h_best = h_flip.copy()
                                improved = True
                                break
                    if improved:
                        break
        return h_best, best_c5

    return None, np.inf

def polish_solution(h_start, M, dx, n):
    def objective(h):
        corr = compute_corr_fft(h)
        return dx * c5_from_corr(corr, n)

    bounds = [(0, 1) for _ in range(n)]

    def constraint_eq(h):
        return np.sum(h) - M

    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        from scipy.optimize import minimize
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if res.success:
            h_final = res.x
            h_final = project_to_sum_box(h_final, M)
            corr = compute_corr_fft(h_final)
            c5_final = dx * c5_from_corr(corr, n)
            return h_final, c5_final
    except:
        pass
    return h_start, objective(h_start)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Phase A: SDP solving
    n_vals = [60, 80, 100]
    best_c5 = np.inf
    best_h = None
    final_n = 0

    for n in n_vals:
        if time.time() - start_time > 0.45 * budget_s:
            break

        dx = 2.0 / n
        M = n / 2
        remaining_time = 0.55 * budget_s - (time.time() - start_time)
        if remaining_time <= 1:
            break

        # Solve SDP
        h_rel, X_rel, t_star = solve_sdp(n, dx, M, remaining_time / 2)

        if t_star < np.inf:
            # Rounding and local refinement
            h_candidate, c5_candidate = rounding_and_hill_climb(h_rel, X_rel, M, n, dx, 20)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                final_n = n

    # Phase B: Final polishing
    if best_h is not None and time.time() - start_time < 0.9 * budget_s:
        best_h, best_c5 = polish_solution(best_h, final_n / 2, 2.0 / final_n, final_n)

    # Fallback to simple construction
    if best_h is None:
        n = 100
        dx = 2.0 / n
        M = n / 2
        h = np.zeros(n)
        h[:int(M)] = 1
        h = project_to_sum_box(h, M)
        corr = compute_corr_fft(h)
        c5 = dx * c5_from_corr(corr, n)
        best_h = h
        best_c5 = c5
        final_n = n

    return (best_h.tolist(), float(best_c5), int(final_n))
