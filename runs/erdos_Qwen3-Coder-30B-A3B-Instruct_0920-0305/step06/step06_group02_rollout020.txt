import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_c5_binary(h):
    """Fast evaluation of C5 for binary vector using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def update_convolution(conv, ones, i, j, n):
    """Update convolution array after swapping positions i and j"""
    # Note: conv is full correlation (2n-1 elements)
    # For indices 0 to 2n-2 (corresponding to shifts -n+1 to n-1)
    # conv[idx] corresponds to shift = idx - (n-1)
    
    # First remove i's contributions (x[i] = 1 -> 0)
    # For each k currently 1, compute pair contributions
    new_conv = conv.copy()
    
    # Remove i's contributions to all shifts
    for k in ones:
        shift = k - i
        idx = shift + (n - 1)
        if idx >= 0 and idx < 2 * n - 1:
            new_conv[idx] -= 1
            if k != i:  # Don't double-count self overlap
                new_conv[-shift + (n - 1)] -= 1
    # Remove j's contributions (x[j] = 0 -> 1) for really new pairs
    # But phases are a bit tricky, better to do it comprehensively:
    # Add j's contribution when it becomes 1
    new_ones = [k for k in ones if k != i] + [j]
    
    for k in new_ones:
        shift = k - j
        idx = shift + (n - 1)
        if idx >= 0 and idx < 2 * n - 1:
            new_conv[idx] += 1
            if k != j:
                new_conv[-shift + (n - 1)] += 1
    
    # Double-check self overlap for j
    new_conv[n - 1] += 1  # Add self-pair for j
    
    return new_conv

def simulate_annealing(seed, budget_s):
    """Run simulated annealing to minimize C5"""
    np.random.seed(seed)
    n = n_points
    
    # Generate initial binary solution with target sum
    while True:
        h = np.zeros(n, dtype=int)
        indices = np.random.choice(n, n//2, replace=False)
        h[indices] = 1
        if np.sum(h) == n//2:
            break
    
    # Compute initial correlation (full)
    conv = np.correlate(h, h[::-1], mode='full')
    
    # Compute initial minimum overlap
    min_overlap = np.min(conv)
    cur_c5 = (n//2 - min_overlap) * dx

    # Store current ones list for efficient swap
    ones = [i for i in range(n) if h[i] == 1]
    
    # Annealing parameters
    T0 = 0.1
    T_min = 1e-8
    alpha = 0.99999
    max_iter = budget_s * 10000  # Roughly 10k iterations/sec
    start_time = time.time()
    
    best_c5 = cur_c5
    best_h = h.copy()
    best_conv = conv.copy()
    
    # Precompute full ranges to avoid overhead
    shift_range = range(-n+1, n)
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Pick two random distinct indices: one with 1, one with 0
        i = random.choice(ones)
        j = random.choice([k for k in range(n) if h[k] == 0])
        
        # Compute new convolution after swap using update
        new_conv = update_convolution(conv, ones, i, j, n)
        
        # Compute new minimum overlap
        new_min_overlap = np.min(new_conv)
        new_c5 = (n//2 - new_min_overlap) * dx
        
        # Compute acceptance probability
        delta_c5 = cur_c5 - new_c5  # Negative if worse
        
        # Accept or not depending on temperature
        if delta_c5 > 0 or random.random() < np.exp(delta_c5 / T0):
            # Accept the move
            h[i] = 0
            h[j] = 1
            ones.remove(i)
            ones.append(j)
            conv = new_conv
            cur_c5 = new_c5
            min_overlap = new_min_overlap
            
            # Update best so far
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                best_conv = conv.copy()
        
        # Cool down temperature
        T0 = max(T_min, T0 * alpha)
        
    # Convert best_h to float array as required by function signature
    best_h = best_h.astype(float).tolist()
    return (best_h, float(best_c5), n_points)
