# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def adaptive_local_search(h0, max_iter=500):

    h = h0.copy()
    best_h = h.copy()

    # Initialize step size
    step_size = 0.01
    
    for i in range(max_iter):
        # Every 50 iterations, update step size based on performance
        if i % 50 == 0 and i > 0:
            step_size = max(0.001, step_size * 0.9)
        
        # Random sampling of potential improvements
        candidates = []
        for _ in range(10):
            # Try moving mass between random positions
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = min(h[i1], 1 - h[i2], step_size)
            if delta > 1e-6:
                candidates.append((i1, i2, delta))
        
        # Try these candidates
        for i1, i2, delta in candidates:
            h_test = h.copy()
            h_test[i1] -= delta
            h_test[i2] += delta
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                h = h_test.copy()
                step_size = min(0.1, step_size * 1.1)  # Increase step if successful
                break
    
    return best_h, best_c5
