TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):

def update_conv_for_swap(conv, ones_list, i, j, n):
    """Update convolution after swapping i and j (binary values)"""
    # i originally 1, j originally 0
    # Remove contributions of i
    conv_new = conv.copy()
    for k in ones_list:
        shift = k - i
        idx = shift + n - 1
        if k == i:
            conv_new[idx] -= 1
        else:
            conv_new[idx] -= 1
            conv_new[-shift + n - 1] -= 1
    
    # Remove i from ones list
    new_ones = ones_list.copy()
    new_ones.remove(i)
    # Add j to ones list (for new ones)
    new_ones.append(j)
    
    # Add contributions of j
    for k in new_ones:

        idx = shift + n - 1
        if k == j:
            conv_new[idx] += 1
        else:
            conv_new[idx] += 1
            conv_new[-shift + n - 1] += 1
    
    return conv_new
