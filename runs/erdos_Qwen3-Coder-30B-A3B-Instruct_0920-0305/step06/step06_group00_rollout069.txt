def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            h = h_new
        else:
            step_size *= 0.9  # Reduce step size if no improvement
            
        if step_size < 1e-6:
            break
    
    return best_h, best_c5

        [0.01, 0.0, -0.01, 0.0, 0.0],  # Another shift
    ]
    
    for _ in range(n_iter):
        # Try different patterns
        for pattern in patterns:
            h_trial = h.copy()
            for i in range(min(len(pattern), n_points)):
                h_trial[i] = h[i] + pattern[i % len(pattern)]
            
            # Ensure feasibility
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        
        # Perturb solution
        h = best_h + np.random.normal(0, 0.002, n_points)
        h = project_to_sum_box(h, TARGET_SUM)
    
    return best_h, best_c5
