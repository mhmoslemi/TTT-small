dx = 2.0 / n_points

def evaluate_c5_binary(h):
    """Fast evaluation of C5 using FFT for binary vectors"""
    # Compute auto-correlation using FFT
    h_rev = h[::-1]
    b = h  # For autocorrelation, we use h itself, not 1-h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    # The maximum overlap is at the center (lag 0) for regular auto-correlation
    # But our goal is to minimize max of integral h(x)(1 - h(x+k))
    # which is equivalent to maximizing min of h(x) h(x+k)
    # So we compute auto-correlation of h with itself (h * h)
    # and then we compute the worst case of h * (1 - h) shifted
    # That means we want correlations of h with (1-h) which is the opposite
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(1-h, L)).real
    
    return dx * np.max(corr)

def compute_corr(h):
    """Compute full correlation to get all lags"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value for max_k int h(x)(1 - h(x+k))"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.5
        b += 0.5
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def binary_hill_climb(h, max_iter=1000):
    """Hill climber for binary vectors"""
    h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        improved = False
        # Try all possible single bit swaps
        for i in range(n_points):
            if h[i] == 0:
                continue
            for j in range(n_points):
                if h[j] == 1:
                    continue
                # Try swapping bit i from 1 to 0 with bit j from 0 to 1
                h[i], h[j] = 0, 1
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
                else:
                    # Revert swap
                    h[i], h[j] = 1, 0
        if not improved:
            break
    return h, best_c5
