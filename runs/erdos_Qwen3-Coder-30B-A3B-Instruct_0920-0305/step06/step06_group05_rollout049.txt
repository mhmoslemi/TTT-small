import time
import numpy as np
from scipy.signal import correlate
import cvxpy as cp

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def round_binary(h_rel, M):
    """Round a real-valued vector to binary with correct sum"""
    n = len(h_rel)
    
    # Threshold rounding: set M largest to 1, rest to 0
    idx = np.argsort(h_rel)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:M]] = 1
    
    # Adjust if needed
    if np.sum(h_bin) > M:
        diff = int(np.sum(h_bin) - M)
        to_zero = idx[M:M+diff]
        h_bin[to_zero] = 0
    elif np.sum(h_bin) < M:
        diff = int(M - np.sum(h_bin))
        to_one = idx[M:M+diff]
        h_bin[to_one] = 1
    
    return h_bin

def swap_hill_climbing(h_init, M, max_iter=5000):
    """Hill climbing with local swaps"""
    h = h_init.copy()
    best_c5 = evaluate_c5(h)
    improved = True
    iter_count = 0
    while improved and iter_count < max_iter:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                iter_count += 1
                if iter_count >= max_iter:
                    break
                    
                # Swap i and j
                h[i] = 0
                h[j] = 1
                
                # Recompute C5 efficiently
                corr = compute_corr_fft(h)
                new_c5 = c5_from_corr(corr, len(h))
                
                # Revert swap
                h[i] = 1
                h[j] = 0
                
                if new_c5 < best_c5:
                    h[i] = 0
                    h[j] = 1
                    best_c5 = new_c5
                    improved = True
                    break
            if improved or iter_count >= max_iter:
                break
    return h, best_c5

def solve_sdp(n, M, time_limit_s):
    """Solve the SDP relaxation."""
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    constraints = [
        cp.matmul(np.ones(n), h) == M,
        0 <= h, h <= 1,
        cp.diag(X) == h,
        X >> 0,
        t >= M - cp.trace(P_k @ X) for P_k in P_lags  
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.SCS, time_limit=time_limit_s)
        h_opt = h.value if h.value is not None else np.random.uniform(0, 1, n)
        X_opt = X.value if X.value is not None else np.eye(n)
        t_opt = t.value if t.value is not None else np.inf
        return h_opt, X_opt, t_opt
    except:
        # If solver fails, still return best known point
        h_opt = np.random.uniform(0, 1, n)
        X_opt = np.eye(n)
        t_opt = np.inf
        return h_opt, X_opt, t_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_solution = None
    best_c5 = float('inf')
    n_points = 60
    
    # Generate lag matrices
    def lag_mat(k, n):
        P = np.zeros((n,n))
        for i in range(n):
            j = i + k
            if j < n:
                P[i,j] = 1
        return P
    
    P_lags = [lag_mat(k, n_points) for k in range(n_points)]
    
    # Solve SDP for increasing sizes
    n_values = [60, 80, 100]
    # Compute time budget per SDP solve
    remaining_time = budget_s * 0.5
    last_n = 60
    
    for n in n_values:
        if time.time() - start_time > budget_s - 5:
            break
        n_points = n
        P_lags = [lag_mat(k, n) for k in range(n)]
        
        # Approximate solver time for this size
        solve_time = min(remaining_time * 0.8, 60)
        remaining_time -= solve_time
        
        try:
            h_opt, X_opt, t_opt = solve_sdp(n, n/2, solve_time)
        except:
            try:
                h_opt, X_opt, t_opt = solve_sdp(n, n/2, solve_time)
            except:
                continue
                
        # Rounding: random hyperplane and threshold
        h_final = round_binary(h_opt, n//2)
        h_final, c5_val = swap_hill_climbing(h_final, n//2)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_solution = h_final.copy()
    
    # Final polishing
    if best_solution is not None and time.time() - start_time < budget_s - 0.2:
        # Use SLSQP (from scipy.optimize.minimize)
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return evaluate_c5(h_vec)
        
        def constraint_eq(h):
            return np.sum(h) - n_points / 2
    
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_solution, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_solution = res.x
        except:
            pass
    
    dx = 2.0 / n_points
    final_c5 = dx * best_c5
    
    return (best_solution.tolist(), float(final_c5), n_points)
