# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def swap_move(h, i, j):
    """Perform a swap move between positions i and j"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def block_move(h, start, block_len):
    """Perform block swap operation"""
    h = h.copy()
    # This is a simplified implementation assuming block_len is small
    block_end = start + block_len
    if block_end > len(h):
        return h
    
    # Flip the block
    for i in range(start, block_end):
        h[i] = 1 - h[i]
    
    # Need to restore the weight constraint by swapping bits
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Count number of 1s and 0s that are flipped
    flipped_count = 0
    for i in range(start, block_end):
        flipped_count += (1 - 2 * h[i])
    
    # Always fix with at most |flipped_count|/2 swaps
    ones_count = len(ones)
    required_ones = TARGET_SUM
    
    if ones_count > required_ones:
        # Too many 1s: need to flip some bits from 1 to 0
        flip_count = ones_count - required_ones
        to_flip = np.random.choice(ones, flip_count, replace=False)
        for idx in to_flip:
            h[idx] = 0
    elif ones_count < required_ones:
        # Not enough 1s: need to flip some bits from 0 to 1
        flip_count = required_ones - ones_count
        to_flip = np.random.choice(zeros, flip_count, replace=False)
        for idx in to_flip:
            h[idx] = 1
    
    return h

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def compute_autocorrelation(h):
    """Compute the autocorrelation manually for binary vectors"""
    n = len(h)
    A = np.zeros(n)
    for k in range(n):
        # Compute sum of h_i * h_{i+k} for i such that both are valid
        count = 0
        for i in range(n - k):
            count += h[i] * h[i + k]
        A[k] = count
    return A

def hill_climb(h):
    """Perform exhaustive hill climbing"""
    h = h.copy()
    iters = 0
    while iters < 5000: # Prevent infinite loops
        improved = False
        # Compute current worst lag
        corr = compute_autocorrelation(h)
        worst_lag = np.argmin(corr)
        
        # Try all possible swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Optimization: only try a subset of swaps to reduce computation
        # but still perform some exploration
        len_ones = len(ones)
        len_zeros = len(zeros)
        max_swaps = min(500, len_ones * len_zeros)
        
        swap_indices = []
        for i in range(min(max_swaps, len_ones * len_zeros)):
            idx = np.random.randint(len_ones)
            jdx = np.random.randint(len_zeros)
            swap_indices.append((ones[idx], zeros[jdx]))
            
        for i, j in swap_indices:
            h_new = swap_move(h, i, j)
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                improved = True
                break
        
        if not improved:
            break
        iters += 1

    return h

def find_sel_indices(h, k_star):
    """Find indices with significant contribution at specific lag"""
    n = len(h)
    # Get indices where at least one element in the pair contributes to lag k_star
    sel_indices = set()
    
    # Add indices where at least one element of pair (i, i+k_star) is set
    for i in range(n):
        if i + k_star < n:
            if h[i] == 1 or h[i+k_star] == 1:
                sel_indices.add(i)
                sel_indices.add(i+k_star)
        if i - k_star >= 0:
            if h[i] == 1 or h[i-k_star] == 1:
                sel_indices.add(i)
                sel_indices.add(i-k_star)
    
    # Limit to size 20 or less for efficient MILP
    if len(sel_indices) > 20:
        # Compute influence for selection based on (h[i] * (1-h[i+k_star])) + (h[i] * (1-h[i-k_star]))
        influence = []
        for i in sel_indices:
            influence.append((i, abs(h[i] * (1 - h[min(i+k_star, n-1)])) + abs(h[i] * (1 - h[max(i-k_star, 0)]))))
        influence.sort(key=lambda x: x[1], reverse=True)
        sel_indices = {x[0] for x in influence[:20]}
    
    return sorted(list(sel_indices))

def milp_refine(h, remaining_time):

    # First determine which lag(s) are problematic
    corr = compute_autocorrelation(h)
    k_star = np.argmin(corr)
    
    sel_indices = find_sel_indices(h, k_star)
    if not sel_indices or len(sel_indices) == 0:
        return h
    
    # Ensure not too many variables
    if len(sel_indices) > 20:
        sel_indices = sel_indices[:20]
    
    n = len(h)
    m = TARGET_SUM
    
    # Build model
    try:
        # Get fixed binary values (those not in sel_indices)
        fixed_vars = {}
        for i in range(n):
            if i not in sel_indices:
                fixed_vars[i] = h[i]
        
        # Define binary variables for selected indices
        z = cp.Variable(len(sel_indices), boolean=True)
        t = cp.Variable()
        
        # Construct full binary vector with fixed parts
        full_h = h.copy()
        for k, idx in enumerate(sel_indices):
            full_h[idx] = z[k]
        
        # Form the constraints
        constraints = [cp.sum(z) == m - sum(1 for i in fixed_vars if fixed_vars[i] == 1)]
        
        # Lag constraints
        # One more trick, build constraints directly from edge pairs
        for k in range(n):
            val = 0
            for i in range(n - k):
                if i in sel_indices or i + k in sel_indices:
                    if i in sel_indices and i + k in sel_indices:
                        val += z[sel_indices.index(i)] * z[sel_indices.index(i + k)]
                    elif i in sel_indices:
                        val += z[sel_indices.index(i)] * (1 - fixed_vars.get(i + k, 0))
                    elif i + k in sel_indices:
                        val += z[sel_indices.index(i + k)] * (1 - fixed_vars.get(i, 0))
                    else:
                        val += (1 - fixed_vars.get(i, 0)) * (1 - fixed_vars.get(i + k, 0))
            if k == 0:
                # This is just a low bound t >= M - sum of products
                constraints += [t >= m - val]
            else:
                constraints += [t >= m - val]
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve in given time
        prob.solve(solver=cp.CBC, time_limit=remaining_time * 0.1, verbose=False)
        
        if prob.status in ["optimal", "feasible"]:
            # Construct final solution using string rounding for z star
            new_z = [int(round(val.value)) for val in z]
            h_result = h.copy()
            for k, idx in enumerate(sel_indices):
                h_result[idx] = new_z[k]
            
            return h_result
        
    except Exception as e:
        pass
    
    return h
