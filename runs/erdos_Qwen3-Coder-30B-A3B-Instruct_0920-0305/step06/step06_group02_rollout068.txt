import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_search_step(h):
    """Perform a local search step"""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    improved = True
    
    # Try different moves
    for i in range(n_points):
        for delta in [-0.02, -0.01, 0.01, 0.02]:
            h_new = h.copy()
            h_new[i] += delta
            
            if h_new[i] < 0 or h_new[i] > 1:

            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < best_c5:

                best_h = h_new.copy()

def genetic_optimization(h0, population_size=30, generations=100):
    """Genetic algorithm optimization"""
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h_mut = h0 + np.random.normal(0, 0.02, n_points)
        h_mut = project_to_sum_box(h_mut, TARGET_SUM)
        population.append(h_mut)
    
    for _ in range(generations):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        # Keep top 50%
        sorted_indices = np.argsort(fitness)[:population_size//2]
        population = [population[i] for i in sorted_indices]
        
        # Create new population
        new_population = population.copy()
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            parent1 = population[idx1] if fitness[idx1] < fitness[idx2] else population[idx2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], 
                                  np.random.uniform(0, 1, n_points - crossover_point)])
            
            # Mutation
            if np.random.random() < 0.3:
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.01)
            
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
        
        population = new_population
    
    # Return best individual from final population
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx]

            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                h_current = h_pert
                best_c5 = c5_pert
    
    # Phase 2: Genetic optimization
    phase2_end = start_time + 0.75 * budget_s

        h_new = genetic_optimization(h_current, population_size=20, generations=20)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            h_current = h_new
            best_c5 = c5_new
    
    # Phase 3: Fine-grained optimization
    phase3_end = start_time + budget_s
    while time.time() < phase3_end:
        # Use sequential local search with restarts
        h_local = h_current.copy()
        local_improved = True
        while local_improved and time.time() < phase3_end:
            h_local, c5_local, local_improved = local_search_step(h_local)
            if c5_local < best_c5:
                h_current = h_local
                best_c5 = c5_local
        
        # Random restart from reference
        h_restart = initialize_from_ref()
        h_restart = project_to_sum_box(h_restart, TARGET_SUM)
        c5_restart = evaluate_c5(h_restart)
        if c5_restart < best_c5:
            h_current = h_restart
            best_c5 = c5_restart
    
    return (h_current.tolist(), float(best_c5), n_points)
