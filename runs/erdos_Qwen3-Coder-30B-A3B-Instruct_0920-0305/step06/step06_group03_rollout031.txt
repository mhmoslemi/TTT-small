import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    num_restarts = 30
    max_iter = 2000
    step_decay = 0.9995
    
    best_h = None
    best_c5 = np.inf
    
    # Run multiple restarts
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize h randomly
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, n_points / 2)
        
        # Subgradient descent
        step_size = 0.05
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute all overlaps using FFT
            corr = correlate(h, 1 - h, mode="full")
            overlaps = corr * dx
            
            # Find all indices with maximum overlap
            max_val = np.max(overlaps)
            max_indices = np.where(np.abs(overlaps - max_val) < 1e-12)[0]
            
            # Compute gradient for the maximum overlaps
            grad = np.zeros_like(h)
            for idx in max_indices:
                shift = idx - (n_points - 1)
                if shift >= 0:
                    grad[:n_points-shift] -= h[shift:]
                    grad[shift:] -= h[:n_points-shift]
                else:
                    s = -shift
                    grad[s:] -= h[:n_points-s]
                    grad[:n_points-s] -= h[s:]
            
            grad /= len(max_indices)
            
            # Update h
            h = h - step_size * grad
            
            # Project back onto constraints
            h = np.clip(h, 0, 1)
            h = project_to_sum_box(h, n_points / 2)
            
            # Decay step size
            step_size *= step_decay
            
        # Evaluate the final solution
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Refine best solution with SLSQP
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda hh: np.sum(hh) - n_points / 2}
            res = minimize(
                lambda hh: evaluate_c5(hh),
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                refined_c5 = evaluate_c5(res.x)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = res.x.copy()
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
