```python
import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum and scale by dx
    c5 = dx * np.max(corr)
    return float(c5)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_budget = budget_s
    remaining_time = 0.9 * total_budget
    
    # Problem setup
    n_points = 120
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Create model using CVXPY
    # Decision variables
    h = cp.Variable(n_points, boolean=True)
    lambda_var = cp.Variable(integer=True)
    
    # Variables for product terms
    y = {}
    for l in range(n_points):
        for i in range(n_points - l):
            y[(i,l)] = cp.Variable(boolean=True)
    
    # Build the model
    objective = cp.Maximize(lambda_var)
    constraints = []
    
    # Weight constraint
    constraints.append(cp.sum(h) == target_sum)
    
    # Product constraints
    for l in range(n_points):
        for i in range(n_points - l):
            # y[i,l] <= h[i]
            constraints.append(y[(i,l)] <= h[i])
            # y[i,l] <= h[i+l]
            constraints.append(y[(i,l)] <= h[i+l])
            # y[i,l] >= h[i] + h[i+l] - 1
            constraints.append(y[(i,l)] >= h[i] + h[i+l] - 1)
    
    # Lag overlap constraints
    for l in range(n_points):
        if l == 0:
            # For l=0, the constraint is just sum of h[i]
            # No need to add more since this is covered by the sum constraint
            # This is skipped as it's just a constant condition
            continue
        else:
            # sum_{i=0}^{N-1-l} y_{i,l} >= lambda
            constraints.append(cp.sum([y[(i,l)] for i in range(n_points-l)]) >= lambda_var)
    
    # Bound on lambda
    constraints.append(lambda_var <= n_points / 2)
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Initialize solution
    best_c5 = float('inf')
    best_h = []

    # Try solving with different time limits
    solver_options = {'time_limit': remaining_time}
    
    try:
        # Try solving the MILP
        prob.solve(solver=CPLEX, **solver_options)
        # Or alternatively use CBC or GLPK if CPLEX is not available:
        # prob.solve(solver=CBC, **solver_options)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            # Extract solution
            h_vals = h.value
            if h_vals is not None and isinstance(h_vals, np.ndarray):
                best_h = h_vals.astype(float).tolist()
                # Compute C5 directly using the formulas given
                c5 = evaluate_erdos_solution(best_h)
                best_c5 = c5
            else:
                print("No valid solution found from MILP.")
        else:
            print(f"Solver status: {prob.status}")
    except Exception as e:
        print(f"Exception during solving: {e}")

    # Fallback: Block Search Approach
    if best_c5 == float('inf') or best_c5 > 0.3807:
        # Try possible block-wise guesses
        # Use blocks of up to 6 intervals, with binary assignment to each block
        best_c5_block = float('inf')
        best_h_block = []
        
        # Sets of block counts to try (small for speed): m = 3,4,5,6
        BLOCK_CHOICES = [3, 4, 5, 6]
        
        for m in BLOCK_CHOICES:
            # Enumerate integer partition of n_points into m parts
            def gen_partitions(n, k, min_val=1, max_val=None):
                if max_val is None:
                    max_val = n
            
                # Use recursive generation
                def recurse(current_sum, start, remaining, partial):
                    if remaining == 0:
                        if current_sum == n:
                            yield partial
                        return
                    for i in range(start, min(max_val, n - current_sum + 1)):
                        if current_sum + i <= n:
                            yield from recurse(current_sum + i, i, remaining - 1, partial + [i])
                yield from recurse(0, min_val, k, [])
            
            # For each such partition, generate assignments for binary heights
            part_gen = gen_partitions(n_points, m, 1)
            
            for lengths in part_gen:
                if sum(lengths) != n_points:
                    continue
                # Now assign binary heights to each block (0 or 1)
                from itertools import product
                assignments = list(product([0,1], repeat=m))
                for assignment in assignments:
                    # Create a step function
                    h_block = []
                    for idx, (length, height) in enumerate(zip(lengths, assignment)):
                        h_block.extend([height]*length)
                    if len(h_block) != n_points:
                        continue
                    h_block = np.array(h_block)
                    # Ensure sum is exactly target_sum
                    actual_sum = np.sum(h_block)
                    if abs(actual_sum - target_sum) > 1e-4:
                        # Adjust the first few entries to make it exact
                        diff = target_sum - actual_sum
                        for i in range(int(abs(diff))):
                            if diff < 0 and h_block[i] == 1:
                                h_block[i] = 0
                            elif diff > 0 and h_block[i] == 0:
                                h_block[i] = 1
                    # Evaluate c5 for this h
                    c5_block = evaluate_erdos_solution(h_block.tolist())
                    if c5_block < best_c5_block:
                        best_c5_block = c5_block
                        best_h_block = h_block.tolist()
                        
        if best_c5_block < best_c5:
            best_c5 = best_c5_block
            best_h = best_h_block
            
    return (best_h, float(best_c5), n_points)
```
