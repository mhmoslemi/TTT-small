import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_convolution(h):
    """Compute convolution of h with itself using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    conv = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return conv

def compute_convolution_optimized(h, conv_cache=None):
    """Compute convolution of h with itself efficiently"""
    if conv_cache is not None:
        # Use precomputed
        return conv_cache
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    conv = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return conv

def apply_swap_with_conv_update(conv, ones, zeros, i, j, n):
    """Apply a swap of i and j, updating convolution efficiently"""
    # We batch update to avoid repeated operations
    offset = n - 1
    # Store original conv values to revert if needed
    old_values = conv.copy()
    
    # Remove contribution of i (1 -> 0)
    # Goes through all k such that x_k = 1 (before removal)
    for k in ones:
        shift = k - i
        idx = shift + offset
        # No need to check k == i here, since we skip i when removing
        conv[idx] -= 1
        if k != i and k != i + shift:  # To avoid double counting in edge case
            conv[-shift + offset] -= 1
    
    # Add contribution of j (0 -> 1) 
    # We now have ones list update, add j and compute its new contributions
    new_ones = ones.copy()
    new_ones.remove(i)
    new_ones.append(j)
    
    for k in new_ones:
        shift = k - j
        if shift:
            idx = shift + offset
            conv[idx] += 1
            conv[-shift + offset] += 1
        else:
            # self-case, to prevent double counting
            conv[offset] += 1
    
    return conv

def swap_indices(i, j, ones, zeros):
    """Swap indices i (1 -> 0) and j (0 -> 1) keeping track of positions"""
    ones.remove(i)
    zeros.remove(j)
    ones.append(j)
    zeros.append(i)

def simulate_annealing(h0, max_iter=50000, initial_temp=1.0, final_temp=0.0001, cooling_rate=0.9999):
    """Simulated Annealing approach to minimizing C5"""
    # Initialize state
    current_h = h0.copy()
    current_ones = [i for i in range(len(h0)) if h0[i] == 1]
    current_zeros = [i for i in range(len(h0)) if h0[i] == 0]
    
    # Precompute initial convolution
    conv = compute_convolution(current_h)
    current_min_overlap = np.min(conv)
    current_c5 = (TARGET_SUM - current_min_overlap) * dx
    
    best_h = current_h.copy()
    best_min_overlap = current_min_overlap
    best_c5 = current_c5
    
    # Initialize temperature and iteration variables
    temp = initial_temp
    iteration = 0
    
    while iteration < max_iter:
        if iteration % 1000 == 0:
            # Early termination based on time
            pass
        
        # Generate one neighbor: swap a 1 with a 0
        if len(current_ones) < 1 or len(current_zeros) < 1:
            break
            
        i = random.choice(current_ones)
        j = random.choice(current_zeros)
        
        # Create new convolution by applying swap
        new_conv = conv.copy()
        
        # Perform a swap efficiently
        # Remove contribution of i (1 -> 0)
        for k in current_ones:
            shift = k - i
            idx = shift + n_points - 1
            new_conv[idx] -= 1
            if k != i:
                new_conv[-shift + n_points - 1] -= 1
            # Ensure valid indices - might not be necessary due to bounds
            # Proper boundary checks are implicit due to gears of shift bounds being within [-(n-1), (n-1)]
            
        # Update ones and zeros lists
        current_ones.remove(i)
        current_zeros.remove(j)
        current_ones.append(j)
        current_zeros.append(i)
        
        # Add contribution of j (0 -> 1)
        for k in current_ones:
            shift = k - j
            if shift != 0:
                idx = shift + n_points - 1
                new_conv[idx] += 1
                new_conv[-shift + n_points - 1] += 1
            else:
                new_conv[n_points - 1] += 1  # Self-match case
                
        # Revert current_ones and current_zeros lists
        current_ones.remove(j)
        current_zeros.remove(i)
        current_ones.append(i)
        current_zeros.append(j)
        
        # Compute new overlap
        new_min_overlap = np.min(new_conv)
        new_c5 = (TARGET_SUM - new_min_overlap) * dx
        
        # Compute change of objective
        delta = current_c5 - new_c5  # Positive if improving
        
        # Decide whether to accept move
        if delta > 0 or random.random() < np.exp(delta / temp):
            # Accept move
            current_h[i] = 0
            current_h[j] = 1
            conv = new_conv
            current_min_overlap = new_min_overlap
            current_c5 = new_c5
            
            # Update best solution
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_min_overlap = current_min_overlap
                best_c5 = new_c5
                
        # Restore lists (since current_h is kept but not modified, just algorithmic consistency)
        # Most likely, we optimize swap control after minimization for final results
        
        iteration += 1
        
        # Update temperature
        temp = initial_temp * (cooling_rate ** iteration)
        
        if temp < final_temp:
            break
    
    return best_h, best_c5
