import numpy as np
import time
from scipy.signal import correlate

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create initial binary vector with exact weight"""
    np.random.seed(seed)
    # Start with the best continuous solution rounded to binary
    try:
        h_cont = np.array(initial_h_values)
        # Round to binary with threshold at 0.5, then project to exact sum
        x = (h_cont > 0.5).astype(np.int8)
        current_sum = np.sum(x)
        if current_sum > TARGET_SUM:
            # Need to flip some 1s to 0s
            flip_indices = np.where(x == 1)[0]
            np.random.shuffle(flip_indices)
            for i in range(current_sum - TARGET_SUM):
                x[flip_indices[i]] = 0
        elif current_sum < TARGET_SUM:
            # Need to flip some 0s to 1s
            flip_indices = np.where(x == 0)[0]
            np.random.shuffle(flip_indices)
            for i in range(TARGET_SUM - current_sum):
                x[flip_indices[i]] = 1
        return x
    except:
        # Fallback to random
        x = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        current_sum = np.sum(x)
        if current_sum > TARGET_SUM:
            flip_indices = np.where(x == 1)[0]
            np.random.shuffle(flip_indices)
            for i in range(current_sum - TARGET_SUM):
                x[flip_indices[i]] = 0
        elif current_sum < TARGET_SUM:
            flip_indices = np.where(x == 0)[0]
            np.random.shuffle(flip_indices)
            for i in range(TARGET_SUM - current_sum):
                x[flip_indices[i]] = 1
        return x

def full_overlap(x):
    """Compute full correlation array O(s) = sum_i x_i * x_{i+s}"""
    return correlate(x, x, mode='full')

def update_overlap(ov, x, i, j, ones, zeros):
    """Fast update of overlap array due to swap of x[i] and x[j]"""
    # Create a copy to avoid modifying the original
    ov_new = ov.copy()
    # i is a 1, j is a 0
    # Update for all positions k where i+k and j+k are valid
    # Remove contribution of old i and old j for all k
    for k in ones:
        if i != k:
            # old i was part of overlap with k
            s = k - i
            ov_new[s + n_points - 1] -= 1
    for k in zeros:
        if j != k:
            # old j was not part of overlap with k
            s = k - j
            ov_new[s + n_points - 1] -= 1

    # Add contribution of new positions
    for k in ones:  
        if k != i:
            # new i is part of overlap with k
            s = k - i
            ov_new[s + n_points - 1] += 1
    for k in ones:  # excluding i, because j is now placed there
        if k != i:
            # new j is part of overlap with k
            s = k - j
            ov_new[s + n_points - 1] += 1

    return ov_new

def get_ones_zeros(x):
    """Return indices of 1s and 0s"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    return ones, zeros

def simulated_annealing(x0, time_limit):
    """Simulated Annealing using binary swaps"""
    start_time = time.time()
    # Precompute initial overlap and indices
    ov = full_overlap(x0)
    cur_min = ov.min()
    cur_max = ov.max()  # used for acceptance probability
    
    ones, zeros = get_ones_zeros(x0)
    best_x = x0.copy()
    best_min = cur_min
    
    # Temperature settings
    T0 = 0.5
    lambda_t = 0.01
    iter_count = 0
    
    # Main annealing loop
    while time.time() - start_time < time_limit:
        iter_count += 1
        
        # Temperature
        t = iter_count / 10000
        T = T0 * np.exp(-lambda_t * t)
        
        # Pick a random 1 and 0 to swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Make new overlap array
        ov_new = update_overlap(ov, x0, i, j, ones, zeros)
        new_min = ov_new.min()
        
        # Accept or reject
        delta = new_min - cur_min
        
        if delta > 0 or np.random.rand() < np.exp(delta / (T + 1e-10)):
            # Accept the move
            x0[i] = 0
            x0[j] = 1
            ov = ov_new
            cur_min = new_min
            
            # Update indices (expensive, but required)
            ones, zeros = get_ones_zeros(x0)
            
            if new_min > best_min:
                best_x = x0.copy()
                best_min = new_min
        
        # Occasionally check if time is nearly up
        if iter_count % 1000 == 0:
            if time.time() - start_time > 0.95 * time_limit:
                break
                
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine by improving the current worst shift"""
    # Find worst shift
    worst_idx = np.argmin(ov)
    worst_shift = worst_idx - n_points + 1
    
    # Identify good spots to move (1 -> 0) that increase overlap with worst_shift
    # Candidates for 1
    candidates_i = []
    for i in ones:
        if i + worst_shift < n_points and i + worst_shift >= 0:
            if x[i + worst_shift] == 0:
                candidates_i.append(i)
    
    # Candidates for 0  
    candidates_j = []
    for j in zeros:
        if j + worst_shift < n_points and j + worst_shift >= 0:
            if x[j + worst_shift] == 1:
                candidates_j.append(j)
    
    # Try all combinations of upgrades
    improved = True
    while improved:
        improved = False
        best_delta = 0
        
        # Try swapping all i->j pairs that both upgrade overlap
        for i in candidates_i:
            for j in candidates_j:
                ov_new = ov.copy()
                # Update incrementally
                for k in ones:
                    if i != k:
                        s = k - i
                        ov_new[s + n_points - 1] -= 1
                for k in ones:
                    if k != i:
                        s = k - j
                        ov_new[s + n_points - 1] += 1
                for k in zeros:
                    if j != k:
                        s = k - j
                        ov_new[s + n_points - 1] -= 1
                
                new_min = ov_new.min()
                delta = new_min - ov.min()
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
        
        if best_delta > 0:
            x[best_i] = 0
            x[best_j] = 1
            ov = update_overlap(ov, x, best_i, best_j, ones, zeros)
            improved = True
                    
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Run multiple restarts
    best_x = None
    best_min = np.inf
    
    # Adjust timing for multiple restarts
    time_per_restart = budget_s / 5.0
    
    for i in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
        x = initial_binary(seed + i)
        x, min_val = simulated_annealing(x, time_per_restart)
        
        ones, zeros = get_ones_zeros(x)
        ov = full_overlap(x)
        x_refined, ov_refined = worst_shift_refine(x, ov, ones, zeros)
        min_refined = ov_refined.min()
        
        if min_refined < best_min:
            best_x = x_refined
            best_min = min_refined
    
    # Final calculation
    final_ov = full_overlap(best_x)
    final_c5 = dx * (TARGET_SUM - final_ov.min())
    
    # Convert binary to float for output
    h_float = best_x.astype(float)
    
    return (h_float.tolist(), float(final_c5), n_points)
