import numpy as np
import time
from scipy.signal import correlate

# Global parameters
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2
L = 2 * n_points - 1

def initial_binary(seed):
    """Generate an initial binary vector with exact sum constraint."""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    positions = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[positions] = 1
    return x

def full_overlap(x):
    """Compute full correlation of binary vector x with itself."""
    return correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros, n):
    """Update overlap array in O(w) time given a swap of x[i] and x[j]."""
    # Copy current overlap
    ov_new = ov.copy()
    # Remove contributions of i and add contributions of j
    # For every 1-position k, if i+k is valid and k exists, subtract 1 from overlap
    # Add 1 to overlaps corresponding to j+k for all remaining 1-positions
    # i is removed from ones, added to zeros 
    # j is removed from zeros, added to ones
    
    # Go over all 1's in current ones set (except i)
    # That's the old contribution of j
    for k in ones:
        if k == i:
            continue
        # If we're tracking the overlap, simply update
        # For overlap shift s, that's contribution from (k, k+s)
        if 0 <= k < n:
            # We know it is 1 here, so if k+s is also valid and was 1, this contributes
            pass # to handle in loop below
        s = k - i  # shift from k to i
        shifted_s = s + n - 1
        if 0 <= shifted_s < L:
            ov_new[shifted_s] -= 1  # remove k from i

    # Now handle j: add increments based on j and current ones (except i)
    for k in ones:
        if k == i:
            continue
        s = k - j  # shift from k to j
        shifted_s = s + n - 1
        if 0 <= shifted_s < L:
            ov_new[shifted_s] += 1  # add k from j

    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    # Initial values
    cur_min = ov.min()
    best_x = x.copy()
    best_min = cur_min
    start = time.time()
    
    # Cooling schedule
    T0 = 0.5
    lambda_cool = 0.9995
    T = T0
    
    max_iter = 1000000
    iter_count = 0
    plateau_count = 0
    plateau_threshold = 5000
    time_check = 0.95 * time_limit  # Check every 1000 iterations
    
    # Maintain a list of pairs considered for exploitation
    last_moves = []
    
    while time.time() - start < time_limit and iter_count < max_iter \
          and plateau_count < plateau_threshold:
        
        if iter_count % 1000 == 0 and time.time() - start > time_check:
            break

        # Pick random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        ov_new = swap_update(ov, i, j, ones, zeros, n_points)
        new_min = ov_new.min()
        
        delta = new_min - cur_min
        
        # Accept or reject
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            # Apply swap
            x[i] = 0
            x[j] = 1
            ones.remove(i)
            zeros.remove(j)
            ones.add(j)
            zeros.add(i)
            ov = ov_new
            cur_min = new_min
            
            # Update best solution
            if new_min > best_min:
                best_x = x.copy()
                best_min = new_min
                
            # Reset plateau counter
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Cool down
        T *= lambda_cool
        iter_count += 1

    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """
    Fix the worst shift s that caused cur_min = ov.min(),
    by finding 1s that would benefit from swapping with 0s in overlap shift s.
    """
    cur_min = ov.min()
    s_star = np.argmin(ov) - n_points + 1
    
    # Sets where overlap shift s_star failed
    A = set()  # 1-positions such that x[i]=1, x[i+s_star]=0 (i+s_star < n)
    B = set()  # 0-positions such that x[j]=0, x[j+s_star]=1 
    
    for i in ones:
        pos = i + s_star
        if 0 <= pos < n_points:
            if x[pos] == 0:
                A.add(i)
        else:
            # out of bounds, not really affecting the overlap, skip
            pass
    
    for j in zeros:
        pos = j + s_star
        if 0 <= pos < n_points:
            if x[pos] == 1:
                B.add(j)
        else: 
            pass
    
    # Try to apply as many beneficial swaps as possible
    A_list = list(A)
    B_list = list(B)
    
    # Iterate and recompute every time to maintain correctness
    ov_old = ov.copy()  # For delta computation
    
    for i in A_list:
        for j in B_list:
            # Simulate the swap to see if it helps at s_star
            ov_new = ov.copy()
            # Remove old indices from contribution
            # The overlap shift s_star includes contribution from (i, i+s_star) 
            # and (i+s_star, i)
            s = i - (i+s_star)
            shifted_s = s + n_points - 1
            if 0 <= shifted_s < L:
                ov_new[shifted_s] -= 1
            s = (i+s_star) - i
            shifted_s = s + n_points - 1
            if 0 <= shifted_s < L:
                ov_new[shifted_s] -= 1
                
            # Add new indices
            s = j - (j+s_star)
            shifted_s = s + n_points - 1
            if 0 <= shifted_s < L:
                ov_new[shifted_s] += 1
            s = (j+s_star) - j
            shifted_s = s + n_points - 1
            if 0 <= shifted_s < L:
                ov_new[shifted_s] += 1
                
            new_min = ov_new.min()
            if new_min > cur_min:  # Check if overall improvement
                # Perform actual swap
                x[i] = 0
                x[j] = 1
                ones.remove(i)
                zeros.remove(j)
                ones.add(j)
                zeros.add(i)
                ov = ov_new
                cur_min = new_min
                break  # one swap at a time
        else:
            continue
        break
    
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    x = initial_binary(seed)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    ov = full_overlap(x)
    cur_min = ov.min()
    
    best_x = x.copy()
    best_min = cur_min
    
    # Multiple restarts
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < 0.9 * budget_s and restart_count < max_restarts:
        if restart_count > 0:
            # Generate new seed and restart
            new_seed = seed + restart_count
            x = initial_binary(new_seed)
            ones = set(np.where(x == 1)[0])
            zeros = set(np.where(x == 0)[0])
            ov = full_overlap(x)
            cur_min = ov.min()
            
        x, cur_min = simulated_annealing(x, ov, ones, zeros, 0.5 * budget_s / max_restarts)
        x, ov = worst_shift_refine(x, ov, ones, zeros)
        
        if cur_min > best_min:
            best_x = x.copy()
            best_min = cur_min
        restart_count += 1

    # Final refinement using repeated compatible swaps  
    for _ in range(500):
        if time.time() - start_time > 0.95 * budget_s:
            break
        s_star = np.argmin(ov) - n_points + 1
        A = set()
        B = set()
        for i in ones:
            pos = i + s_star
            if 0 <= pos < n_points:
                if x[pos] == 0:
                    A.add(i)
        for j in zeros:
            pos = j + s_star
            if 0 <= pos < n_points:
                if x[pos] == 1:
                    B.add(j)
        if not A or not B:
            break
        for i in A:
            for j in B:
                ov_new = ov.copy()
                s = i - (i+s_star)
                shifted_s = s + n_points - 1
                if 0 <= shifted_s < L:
                    ov_new[shifted_s] -= 1  # remove i from shift s_star
                s = (i+s_star) - i
                shifted_s = s + n_points - 1
                if 0 <= shifted_s < L:
                    ov_new[shifted_s] -= 1  # remove (i+s_star) from matching
                s = j - (j+s_star)
                shifted_s = s + n_points - 1
                if 0 <= shifted_s < L:
                    ov_new[shifted_s] += 1  # add j to shift s_star
                s = (j+s_star) - j
                shifted_s = s + n_points - 1
                if 0 <= shifted_s < L:
                    ov_new[shifted_s] += 1  # add (j+s_star) to matching
                new_min = ov_new.min()
                if new_min > best_min:
                    # Apply
                    x[i] = 0
                    x[j] = 1
                    ones.remove(i)
                    zeros.remove(j)
                    ones.add(j)
                    zeros.add(i)
                    ov = ov_new
                    best_min = new_min
                    break
            else:
                continue
            break

    c5 = dx * (TARGET_SUM - best_min)
    
    # Return as float list to match return type
    h_values = [float(xi) for xi in best_x]
    
    return (h_values, float(c5), n_points)
