import numpy as np
from scipy.signal import correlate
import time
import random

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points // 2
    
    # Generate initial binary solution
    h = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    h[ones_indices] = 1
    
    # Precompute correlation for initial state
    corr = compute_corr_fft(h)
    best_c5 = c5_from_corr(corr, n_points)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    max_iter = 200000
    T0 = 0.1
    Tf = 1e-6
    alpha = 0.999
    
    # Keep track of current state
    cur_h = h.copy()
    cur_corr = corr.copy()
    cur_c5 = best_c5
    cur_ones = list(ones_indices)
    
    # Helper functions to update convolution efficiently
    def update_conv_swap(conv, ones_list, i, j):
        # Remove contributions of i
        for k in ones_list:
            shift = k - i
            idx = shift + n_points - 1
            conv[idx] -= 1
            if k != i:
                conv[-shift + n_points - 1] -= 1
                
        # Update ones list by removing i and adding j
        ones_list.remove(i)
        ones_list.append(j)
        
        # Add contributions of j
        for k in ones_list:
            shift = k - j
            idx = shift + n_points - 1
            conv[idx] += 1
            if k != j:
                conv[-shift + n_points - 1] += 1
        
        # Handle self-pair
        conv[n_points - 1] += 1
        
        return conv
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Generate candidate swap
        i = random.choice(cur_ones)
        zero_indices = [idx for idx in range(n_points) if cur_h[idx] == 0]
        j = random.choice(zero_indices)
        
        # Create new solution
        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        
        # Update convolution
        new_corr = cur_corr.copy()
        # Apply swap to convolution
        new_corr = update_conv_swap(new_corr, cur_ones, i, j)
        
        # Evaluate new solution
        new_c5 = c5_from_corr(new_corr, n_points)
        
        # Acceptance probability for simulated annealing
        delta = new_c5 - cur_c5
        T = T0 * (alpha ** iteration)
        
        if delta < 0 or random.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_corr = new_corr
            cur_c5 = new_c5
            # Update ones list for next iteration
            cur_ones = [idx for idx in range(n_points) if cur_h[idx] == 1]
            
            # Update best solution
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
    
    # Convert best solution to float array
    best_h_float = [float(x) for x in best_h]
    
    return (best_h_float, float(best_c5), n_points)
