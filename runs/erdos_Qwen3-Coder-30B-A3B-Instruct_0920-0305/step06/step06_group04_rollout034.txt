import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import ifft, fft
import time

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return corr

def evaluate_C5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def project_to_sum_box(h, target_sum):
    """Water-filling projection to enforce sum = target_sum, box constraints [0,1]"""
    # Sort h in descending order to find lambda efficiently
    h_sorted = np.sort(h)[::-1]
    h_cum = np.cumsum(h_sorted)
    n = len(h)
    # Find the largest index such that h_sorted[i] >= (target_sum - h_cum[i-1]) / (n-i)
    # This helps compute the shift lambda
    for i in range(n):
        if i == 0:
            if h_sorted[i] >= target_sum / n:
                continue
            else:
                beta = (target_sum - h_cum[i - 1]) / (n - i)
                break
        else:
            if h_sorted[i] >= (target_sum - h_cum[i - 1]) / (n - i):
                continue
            else:
                beta = (target_sum - h_cum[i - 1]) / (n - i)
                break
    else:
        beta = (target_sum - h_cum[-1]) / (n - n)

    # Apply shift and clip
    h_proj = np.clip(h - beta, 0, 1)
    return h_proj, beta

def spectral_h_from_coeffs(freq_coeffs, N_fine=800, N=120):
    """Generate h from frequency domain coefficients using trigonometric series"""
    # freq_coeffs = [a1, ..., aK, b1, ..., bK]
    K = len(freq_coeffs) // 2
    a = freq_coeffs[:K]
    b = freq_coeffs[K:]
    
    x = np.linspace(0, 2, N_fine, endpoint=False)  # [0, 2) with N_fine points
    h_base = 0.5 + np.sum([a[k]*np.cos(np.pi*(k+1)*x) + b[k]*np.sin(np.pi*(k+1)*x)
                         for k in range(K)], axis=0)
    
    # Clip to [0,1]
    h_clipped = np.clip(h_base, 0, 1)
    
    # Project to sum = N/2
    h_proj, _ = project_to_sum_box(h_clipped, N/2)
    
    # Down-sample to original N
    h_final = h_proj[::N_fine//N]
    
    return h_final

def objective_from_coeffs(freq_coeffs, N_fine=800, N=120):
    """Objective function to be minimized"""
    h = spectral_h_from_coeffs(freq_coeffs, N_fine, N)
    return evaluate_C5(h)
