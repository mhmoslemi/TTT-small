import numpy as np
from scipy.signal import correlate
import time
import math

# Global constants
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        proj = np.clip(x + lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            high = lam
        else:
            low = lam
    proj = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-12) & (proj < 1 - 1e-12)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def full_corr(a, b):
    """Compute full correlation: zero-padded FFT"""
    n = a.size
    N = 2 * n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2 * n - 1]

def smooth_max_grad(h, alpha):
    """Compute gradient of smooth approximation to max correlation"""
    c = full_corr(h, 1 - h)
    max_c = c.max()
    # compute softmax weights
    w = np.exp(alpha * (c - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w)
        w_sum = w.sum()
    w = w / w_sum
    
    # compute weighted convolutions
    conv_fwd = full_corr(h, w)
    conv_bwd = full_corr(h, w[::-1])
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def line_search_step(h, grad, step_size, target_sum):
    """Perform backtracking line search with projection"""
    while step_size > 1e-8:
        h_new = project_to_sum_box(h - step_size * grad, target_sum)
        val_new = evaluate_c5(h_new)
        if val_new < evaluate_c5(h) - 1e-10:
            return h_new, val_new
        step_size *= 0.5
    return h, evaluate_c5(h)

def optimize_continuous(h0, max_time, n, dx, target_sum):
    """Run projected gradient descent with smoothing and line search"""
    h = h0.copy()
    val = evaluate_c5(h)
    best_h = h.copy()
    best_val = val
    alpha = 5.0
    step_size = 0.01
    
    t_start = time.time()
    iter_count = 0
    while time.time() - t_start < max_time and iter_count < 5000:
        iter_count += 1
        grad = smooth_max_grad(h, alpha)
        h_new, val_new = line_search_step(h, grad, step_size, target_sum)
        if val_new < best_val - 1e-12:
            best_h = h_new.copy()
            best_val = val_new
            # Increase α to sharpen focus
            alpha = min(20.0, alpha + 0.05)
            step_size = min(0.05, step_size * 1.1)  # adapt step size
        else:
            # Reduce step size if stagnating
            step_size *= 0.5
            if step_size < 1e-8:
                break
        h = h_new
    
    return best_h, best_val

def binary_local_search(h_bin, time_budget, n, dx):
    """Perform simulated annealing on binary representation"""
    start = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_c5(best_h)
    temp = 0.01
    cooling_rate = 0.999
    
    while time.time() - start < time_budget:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = evaluate_c5(cand)
        if val < best_val - 1e-9:
            best_val = val
            best_h = cand
        elif np.random.rand() < np.exp(-(val - best_val) / temp):
            best_h = cand
            best_val = val
        temp *= cooling_rate
        if temp < 1e-8:
            temp = 1e-8
    
    return best_h, best_val

    seeds = [h0_proj]

        r = np.random.rand(n_points)

        seeds.append(r_proj)
    
    best_h = h0_proj.copy()

            best_val = val_opt
            best_h = h_opt.copy()
    
    # Binary refinement phase: 30% of time
    remaining = 0.3 * budget_s
    if remaining > 1.0:
        bin_h = (best_h >= np.median(best_h)).astype(float)
        idx_sort = np.argsort(-best_h)
        bin_h[:] = 0
        bin_h[idx_sort[:int(n_points/2)]] = 1.0
        bin_h_opt, bin_val = binary_local_search(bin_h, remaining, n_points, dx)
        if bin_val < best_val:
            best_val = bin_val
            best_h = bin_h_opt.astype(float)

    remaining = 0.2 * budget_s

        if val_up_opt < best_val:

            best_h = h_up_opt.copy()
