import numpy as np

from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_correlations(h):
    """Compute all auto-correlations using FFT for efficiency"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, w):
    """Compute gradient via convolution with symmetric kernel"""
    # The gradient is g_j = sum_k w_k (h_{j+k} + h_{j-k})
    # We can compute this efficiently with a convolution
    # W is symmetric, so we use the convolution trick
    
    # Build symmetric kernel W from w
    W = np.zeros(2 * n_points - 1)
    W[n_points-1:] = w[:n_points]  # w[0], w[1], ..., w[n_points-1]
    W[:n_points-1] = w[:0:-1]      # w[n_points-1], ..., w[1]

    # Convolution of h with W (this is the sum for each position)
    conv = np.convolve(h, W, mode='full')
    g = conv[n_points-1 : n_points-1 + n_points]
    
    # Add half contribution from lag 0 which is counted twice
    g += w[n_points-1] * h
    
    return g

    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02

            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
