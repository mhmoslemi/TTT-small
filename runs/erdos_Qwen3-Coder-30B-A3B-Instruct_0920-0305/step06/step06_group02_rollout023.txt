import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_to_sum_box(h_minus, TARGET_SUM)
        
        grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)
    
    return grad

def optimize_with_gradients_and_projection(h0, max_iter=800):

    h = h0.copy()
    
    for i in range(max_iter):
        # Compute gradient
        grad = evaluate_gradient(h)
        
        # Gradient descent step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check improvement
        if evaluate_c5(h_new) < evaluate_c5(h) - 1e-12:
            h = h_new
        else:
            # Try smaller steps
            for step in [0.001, 0.0001]:
                h_try = h - step * grad
                h_try = project_to_sum_box(h_try, TARGET_SUM)
                if evaluate_c5(h_try) < evaluate_c5(h) - 1e-12:
                    h = h_try
                    break
                    
    return h

    h = h0.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # Random local search
        if np.random.random() < 0.7:
            # Try a few moves
            for _ in range(10):
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.02, 0.02)
                h_try = h.copy()
                h_try[idx] += delta
                h_try = project_to_sum_box(h_try, TARGET_SUM)
                
                new_c5 = evaluate_c5(h_try)
                if new_c5 < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = new_c5

            grad = evaluate_gradient(h)

                h = h_new
                best_c5 = new_c5

        if np.random.random() < 0.05:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
    
    return h
