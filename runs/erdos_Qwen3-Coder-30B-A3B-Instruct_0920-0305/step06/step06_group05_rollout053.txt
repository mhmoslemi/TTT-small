def solve_milp(n, time_limit):
    """Solve the MILP formulation for the Erdős problem"""
    M = n // 2  # required number of ones
    dx = 2.0 / n
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(1, nonneg=True)
    
    # Auxiliary variables for AND operations
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(1, boolean=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == M,  # exact mass constraint
    ]
    
    # Linking constraints for y variables
    for i in range(n):
        for j in range(i+1, n):
            # y_ij <= h_i, y_ij <= h_j
            constraints.append(y[(i,j)] <= h[i])
            constraints.append(y[(i,j)] <= h[j])
            # y_ij >= h_i + h_j - 1
            constraints.append(y[(i,j)] >= h[i] + h[j] - 1)
    
    # Lag constraints: t >= M - sum of overlapping pairs for each lag
    for k in range(1, n):
        # For lag k, we have n-k overlapping pairs
        lag_sum = cp.sum([y[(i, i+k)] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Configure solver
    solver_options = {"seconds": time_limit}
    
    try:
        prob.solve(solver=solver, verbose=False, **solver_options)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = np.rint(h.value).astype(int)
            t_val = t.value
            return h_sol, float(t_val)
    except:
        pass
    
    return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Perform local refinement with SLSQP"""
    n = len(h)
    bounds = [(0, 1) for _ in range(n)]
    target_sum = n / 2
    
    def objective(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_result = res.x
            h_result = project_to_sum_box(h_result, target_sum)
            return h_result
    except:
        pass
    return h
