import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Generate initial binary vector with exact sum constraint"""
    np.random.seed(seed)
    # Start with the best known solution from previous work
    try:
        h_ref = np.array(initial_h_values)
        x_ref = np.linspace(0, 2, len(h_ref))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_ref, h_ref)
        # Round to binary values
        x = (h_interp > 0.5).astype(np.int8)
        # Adjust to exact sum
        current_sum = np.sum(x)
        diff = TARGET_SUM - current_sum
        
        if diff > 0:
            # Need to add more 1s
            zeros = np.where(x == 0)[0]
            flip_indices = np.random.choice(zeros, diff, replace=False)
            x[flip_indices] = 1
        elif diff < 0:
            # Need to remove 1s
            ones = np.where(x == 1)[0]
            flip_indices = np.random.choice(ones, -diff, replace=False)
            x[flip_indices] = 0
            
        return x
    except:
        # Fallback to random binary
        x = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        current_sum = np.sum(x)
        diff = TARGET_SUM - current_sum
        
        if diff > 0:
            zeros = np.where(x == 0)[0]
            flip_indices = np.random.choice(zeros, diff, replace=False)
            x[flip_indices] = 1
        elif diff < 0:
            ones = np.where(x == 1)[0]
            flip_indices = np.random.choice(ones, -diff, replace=False)
            x[flip_indices] = 0
            
        return x

def full_overlap(x):
    """Compute full overlap array"""
    correlation = np.correlate(x, x, mode='full')
    return correlation

def swap_update(ov, i, j, ones, zeros):
    """Perform O(w) update for swapping position i and j in binary vector"""
    ov_new = ov.copy()
    
    # Remove contribution from position i
    # For every 1-position k, subtract 1 if (i, k) is on the same shift
    for k in ones:
        s = k - i
        ov_new[s + n_points - 1] -= 1
    
    # Add contribution from position j
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    
    return ov_new

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Simulated annealing with incremental updates"""
    start_time = time.time()
    cur_min = np.min(ov)
    best_x = x.copy()
    best_min = cur_min
    best_ov = ov.copy()
    
    # Temperature parameters
    T0 = 0.5
    alpha = 0.9995
    T_min = 1e-4
    
    # Main loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50000
    
    T = T0
    while time.time() - start_time < time_limit and T > T_min and no_improvement_count < max_no_improvement:
        if iterations % 1000 == 0:
            T *= alpha
            
        # Pick random 1 and 0 positions
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute new overlap after swap
        ov_new = swap_update(ov, i, j, ones, zeros)
        new_min = np.min(ov_new)
        
        # Acceptance criterion
        delta = new_min - cur_min
        
        if delta > 0 or np.random.random() < np.exp(delta / T):
            # Accept move
            x[i] = 0
            x[j] = 1
            ones.remove(i)
            zeros.add(j)
            zeros.remove(j)
            ones.add(j)
            ov = ov_new
            cur_min = new_min
            
            if cur_min > best_min:
                best_min = cur_min
                best_x = x.copy()
                best_ov = ov.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        iterations += 1
    
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Deterministic repair for the worst shift"""
    current_min = np.min(ov)
    if current_min < TARGET_SUM / 2:  # Improvement possible
        s_star = np.argmin(ov) - (n_points - 1)
        
        # Find sets of beneficial swaps
        # 1's that should become 0 (deficient positions) in shift s_star  
        # 0's that should become 1 (beneficial positions) in shift s_star
        A = []
        B = []
        for i in ones:
            if i + s_star >= 0 and i + s_star < n_points and x[i + s_star] == 0:
                A.append(i)
        for j in zeros:
            if j + s_star >= 0 and j + s_star < n_points and x[j + s_star] == 1:
                B.append(j)
                
        # Perform swaps to raise overlap at s_star
        for i in A:
            for j in B:
                # Check that swapping doesn't hurt current minimum
                ov_new = swap_update(ov, i, j, ones, zeros)
                if np.min(ov_new) >= current_min:
                    x[i] = 0
                    x[j] = 1
                    ones.remove(i)
                    zeros.remove(j)
                    ones.add(j)
                    zeros.add(i)
                    ov = ov_new
                    current_min = np.min(ov)
                    break
            if np.min(ov) > current_min:
                break  # Check further improvements
            
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    start_time = time.time()
    x = initial_binary(seed)
    ones = set(np.where(x == 1)[0])
    zeros = set(np.where(x == 0)[0])
    ov = full_overlap(x)
    cur_min = np.min(ov)
    
    # The main optimization loop - restartable
    while time.time() - start_time < 0.95 * budget_s:
        # Run simulated annealing
        x, ov, cur_min = simulated_annealing(x, ov, ones, zeros, (0.95 * budget_s) / 5)
        
        # Refine using worst-shift repair
        x, ov = worst_shift_refine(x, ov, ones, zeros)
        
        # Update sets to reflect current state
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        
        # 100 restarts max during the time limit
        if len(ones) != TARGET_SUM:
            # Reinitialize to meet the constraint
            x = initial_binary(seed)
            ones = set(np.where(x == 1)[0])
            zeros = set(np.where(x == 0)[0])
            ov = full_overlap(x)
            cur_min = np.min(ov)
    
    c5 = dx * (TARGET_SUM - cur_min)
    
    # Convert to float format as expected for return
    h_values = [float(i) for i in x]
    
    return (h_values, float(c5), n_points)
