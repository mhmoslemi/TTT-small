import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT efficiency
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def fft_project_step(h, lambda_mix, best_c5=None):
    """Perform one iteration of the alternating projection"""
    # Extend h with zeros
    h_ext = np.zeros(L)
    h_ext[:n_points] = h
    
    # Forward FFT
    H = np.fft.fft(h_ext)
    
    # Compute current power spectrum
    P = np.abs(H)**2
    
    # Compute target power spectrum (flat off-DC)
    # DC component stays unchanged
    P_target = np.mean(P[1:])  # Mean of off-DC components
    M_target = np.sqrt(np.full(L, P_target))
    M_target[0] = np.abs(H[0])  # DC component unchanged
    
    # Mix old and target magnitudes
    M = (1 - lambda_mix) * np.abs(H) + lambda_mix * M_target
    
    # Preserve phase
    H_new = M * np.exp(1j * np.angle(H))
    
    # Inverse FFT
    h_new = np.fft.ifft(H_new).real
    
    # Extract central part
    h_candidate = h_new[:n_points]
    
    # Project to feasibility
    h_projected = project_to_sum_box(h_candidate, TARGET_SUM)
    
    return h_projected

    best_c5 = evaluate_c5(h)
    
    # Annealing schedule
    lambda_mix = 0.05
    alpha_lambda = 1.02
    beta = 0.5
    alpha_beta = 1.01
    last_improvement = 0
    patience = 300
    phase_resets = 0

    iteration = 0

        iteration += 1

        # Update relaxation blend
        beta = min(1.0, beta * alpha_beta)
        h = (1 - beta) * h + beta * h_next

        if c5 < best_c5:
            best_c5 = c5

            phase_resets += 1
            last_improvement = iteration

        if np.random.rand() < 0.001 and iteration > 1000:
            h = h_next + np.random.normal(0, 0.01, n_points)
            h = project_to_sum_box(h, TARGET_SUM)

        return evaluate_c5(h)

    cons = {'type': 'eq', 'fun': constraint_eq}

            objective,
            best_h,
