def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_binary_corr(h):
    """Compute correlation for binary vectors using convolution """
    h_rev = h[::-1]
    corr = correlate(h_rev, 1 - h, mode='full')
    return corr

def project_to_binary_sum(h, target_sum):
    """Project a float vector to binary with exact sum"""
    # Threshold to get binary values with desired sum
    sorted_indices = np.argsort(h)
    # Set smallest elements to 0 and largest to 1
    h_b = h.copy()
    # Get the indices that should be 1
    num_ones = int(target_sum)
    h_b[sorted_indices[:len(h)-num_ones]] = 0
    h_b[sorted_indices[len(h)-num_ones:]] = 1
    return h_b

def get_active_indices(h, k_star):
    """Get indices involved in a specific lag"""
    n = len(h)
    # Find active indices: positions where h_i=1 and h_{i+k}=1, or h_i=0 and h_{i+k}=0
    S = set()
    for i in range(n):
        j = i + k_star
        if 0 <= j < n:
            if h[i] == 1 and h[j] == 1:
                S.add(i)
                S.add(j)
            elif h[i] == 0 and h[j] == 0:
                S.add(i)
                S.add(j)
    return list(S)

def simulated_annealing(h_initial, max_time):
    """Simulated annealing with hybrid approach"""
    
    def get_neighbors_bit_flip(state, m, n):
        # Select two indices to swap
        while True:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if state[i] == 1 and state[j] == 0:
                return i, j 
    
    def get_block_flip(state, n, max_block_length=4):
        # Select a block and flip its content
        block_length = np.random.randint(2, max_block_length+1)
        start_pos = np.random.randint(0, n - block_length + 1)
        # Determine how many 1s and 0s in the block
        ones_count = sum(state[start_pos:start_pos+block_length])
        ones_positions = []
        zeros_positions = []
        for i in range(block_length):
            if state[start_pos + i]:
                ones_positions.append(start_pos + i)
            else:
                zeros_positions.append(start_pos + i)
        # Swap corresponding number to preserve weight
        if ones_positions and zeros_positions:
            min_count = min(len(ones_positions), len(zeros_positions))
            # Randomly pick pairings
            for i in range(min_count):
                idx1 = ones_positions[i]
                idx2 = zeros_positions[i]
                if idx1 < len(state) and idx2 < len(state):
                    # We can only do one swap at a time in our current structure
                    return idx1, idx2
        return None, None

    h_best = np.array(h_initial)

    value_curr = value_best

def hill_climbing(h_initial):
    """Exhaustive local search with swap improvement"""
    h = h_initial.copy()
    current_value = evaluate_c5(h)
    
    improvement = True
    while improvement:
        improvement = False
        for i in range(n_points):
            if h[i] == 1:
                for j in range(n_points):
                    if h[j] == 0:
                        # Try swapping positions i and j
                        h_swap = h.copy()
                        h_swap[i] = 0
                        h_swap[j] = 1
                        
                        h_swap_projected = project_to_binary_sum(h_swap, TARGET_SUM)
                        new_value = evaluate_c5(h_swap_projected)
                        
                        if new_value < current_value:
                            h = h_swap_projected
                            current_value = new_value
                            improvement = True
                            break
        if not improvement:
            # Done optimizing
            break
    
    return h, current_value
