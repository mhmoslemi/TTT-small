import numpy as np
from scipy.signal import correlate
import time
import math

# Global constants
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points
MAX_ITER = 5000

def evaluate_erdos_solution(h):
    # Using provided function as per task description
    return max(np.correlate(h, 1 - h, mode='full')) * dx

def project_to_sum_box(x, target_sum):
    """Returns x projected onto [0,1] with sum = target_sum"""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def full_corr(a, b):
    """Compute full correlation of a and b with zero-padding"""
    n = a.size
    N = 2 * n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def compute_soft_max_grad(h, alpha):
    """Compute gradient of soft-max of correlation with respect to h"""
    n = len(h)
    corr = full_corr(h, 1 - h)
    max_corr = corr.max()
    exp_term = np.exp(alpha * (corr - max_corr))
    w = exp_term / exp_term.sum()
    
    conv_fwd = full_corr(h, w)
    conv_bwd = full_corr(h, w[::-1])
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def line_search(h, grad, alpha, target_sum, max_iter=20):
    """Perform backtracking line search"""
    step_size = 0.1
    for _ in range(max_iter):
        h_new = project_to_sum_box(h - step_size * grad, target_sum)
        val_new = max(full_corr(h_new, 1 - h_new)) * dx
        val_old = max(full_corr(h, 1 - h)) * dx
        if val_new < val_old - 1e-10:
            return h_new, val_new, step_size
        step_size *= 0.5
    return h, max(full_corr(h, 1 - h)) * dx, step_size

def continuous_optimize(start_h, remaining_time, target_sum, alpha_schedule):
    """Run projected gradient descent with soft-max objective"""
    best_h = start_h.copy()
    best_val = max(full_corr(best_h, 1 - best_h)) * dx
    
    h = start_h.copy()
    alpha = alpha_schedule(0)
    t = 0
    
    start_time = time.time()
    for iteration in range(MAX_ITER):
        if time.time() - start_time > remaining_time - 5:
            break
            
        grad = compute_soft_max_grad(h, alpha)
        h_new, val_new, _ = line_search(h, grad, alpha, target_sum)
        
        if val_new < best_val:
            best_h = h_new.copy()
            best_val = val_new
            
        h = h_new
        alpha = alpha_schedule(iteration + 1)
        t += 1
    
    return best_h, best_val

def binary_refinement(h_bin, remaining_time, target_sum):
    """Perform simulated annealing on binary representation"""
    best_h = h_bin.copy()
    best_val = max(full_corr(best_h, 1 - best_h)) * dx
    start_time = time.time()
    
    # Initial temperature
    temp = 0.01
    
    while time.time() - start_time < remaining_time - 5:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # compute new correlation
        candidate_h = best_h.copy()
        candidate_h[i] = 0.0
        candidate_h[j] = 1.0
        
        new_val = max(full_corr(candidate_h, 1 - candidate_h)) * dx
        delta = new_val - best_val
        if delta < 0 or math.exp(-delta / temp) > np.random.rand():
            best_val = new_val
            best_h = candidate_h.copy()
        
        temp *= 0.999
        
    return best_h, best_val

def random_binary_h(target_sum):
    """Generate a random binary h with exactly target_sum ones"""
    h = np.zeros(n_points)
    ones_idx = np.random.choice(n_points, int(target_sum), replace=False)
    h[ones_idx] = 1.0
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = float('inf')
    
    # Seeds
    seeds = [interp_initial_h()]
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = project_to_sum_box(h_rand, n_points / 2)
        seeds.append(h_rand)
    seeds.append(random_binary_h(n_points / 2))

    # Continuous optimization
    phase_time = budget_s * 0.6
    for i, h0 in enumerate(seeds):
        if time.time() - start_time > budget_s - 5:
            break
        remaining_time = min(phase_time, budget_s - (time.time() - start_time))
        alpha_schedule = lambda t: 5.0 + 0.02 * t if t < 250 else 15.0 + 0.02 * (t - 250)
        h_opt, val_opt = continuous_optimize(h0, remaining_time, n_points / 2, alpha_schedule)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # Binary refinement
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start_time)
        h_bin = (best_h >= np.median(best_h)).astype(float)
        h_bin = project_to_sum_box(h_bin, n_points / 2)
        h_bin_opt, val_bin_opt = binary_refinement(h_bin, remaining_time, n_points / 2)
        if val_bin_opt < best_val:
            best_val = val_bin_opt
            best_h = h_bin_opt.copy()

    # Final evaluation
    final_val = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), float(final_val), n_points)
