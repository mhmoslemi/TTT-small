import numpy as np

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_LAG = n_points - 1

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    max_corr = np.max(corr)
    return dx * (TARGET_SUM - max_corr)

def compute_autocorrelation(h):
    """Compute autocorrelation for all lags."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def get_worst_lag_indices(h):
    """Get indices of worst lag for current h."""
    corr = compute_autocorrelation(h)
    max_corr = np.max(corr)
    worst_lag = np.argmax(corr)
    return worst_lag

def exact_c5(h):
    """Compute exact C5 using integer convolution."""
    # Redo computation with integer precision
    corr = np.correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return dx * (TARGET_SUM - max_corr)

def binary_to_float_vector(h_binary, n):
    """Convert binary vector to float with given constraints."""
    h = h_binary.astype(float)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def perturb_with_blocks(h, p_block=0.15, max_block_size=4):
    """Perturb with block moves."""
    h = h.copy()
    if np.random.random() < p_block:
        block_len = np.random.choice(range(2, max_block_size + 1))
        start_pos = np.random.randint(0, len(h) - block_len + 1)
        # Flip block
        for i in range(start_pos, start_pos + block_len):
            h[i] = 1 - h[i]
        # Restore weight
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > TARGET_SUM:
            # Remove some 1's
            to_remove = len(ones) - int(TARGET_SUM)
            remove_indices = np.random.choice(ones, size=to_remove, replace=False)
            for idx in remove_indices:
                h[idx] = 0
        elif len(ones) < TARGET_SUM:
            # Add some 0's
            to_add = int(TARGET_SUM) - len(ones)
            add_indices = np.random.choice(zeros, size=to_add, replace=False)
            for idx in add_indices:
                h[idx] = 1
    else:
        # Single swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h[i], h[j] = h[j], h[i]
    return h

def swap_move(h, i, j):
    """Perform a swap move."""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def apply_swap_and_check(h, i, j, current_c5):
    """Apply swap and return new C5 if improved."""
    h_new = swap_move(h, i, j)
    # Repair weight if needed
    ones_count = np.sum(h_new)
    if ones_count != TARGET_SUM:
        # Adjust with more swaps to restore exactness
        if ones_count > TARGET_SUM:
            to_remove = int(ones_count - TARGET_SUM)
            ones_pos = np.where(h_new == 1)[0]
            remove = np.random.choice(ones_pos, to_remove, replace=False)
            for idx in remove:
                h_new[idx] = 0
        else:
            to_add = int(TARGET_SUM - ones_count)
            zeros_pos = np.where(h_new == 0)[0]
            add = np.random.choice(zeros_pos, to_add, replace=False)
            for idx in add:
                h_new[idx] = 1
    new_c5 = evaluate_c5(h_new)
    return h_new, new_c5

def optimize_with_slsqp(h_start, max_iter=150):
    """Optimize with SLSQP to get better continuous solution."""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h_start, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_opt = res.x
            h_opt = project_to_sum_box(h_opt, TARGET_SUM)
            return h_opt
    except:
        pass
    return h_start

def elite_pool():
    """Initialize elite pool."""
    return []

def add_to_elite(elite_list, h, c5_value, max_size=5):
    """Add solution to elite pool."""
    elite_list.append((h.copy(), c5_value))
    elite_list.sort(key=lambda x: x[1])
    if len(elite_list) > max_size:
        elite_list.pop()  # Remove worst
    return elite_list

def crossover_elites(elite_list):
    """Crossover two elites."""
    if len(elite_list) < 2:
        return None
    parent1, _ = elite_list[np.random.randint(0, len(elite_list))]
    parent2, _ = elite_list[np.random.randint(0, len(elite_list))]
    
    point = np.random.randint(1, n_points - 1)
    child = np.concatenate([parent1[:point], parent2[point:]])
    
    # Repair exact weight
    ones_count = np.sum(child)
    if ones_count != TARGET_SUM:
        if ones_count > TARGET_SUM:
            to_remove = int(ones_count - TARGET_SUM)
            ones_pos = np.where(child == 1)[0]
            remove = np.random.choice(ones_pos, to_remove, replace=False)
            for idx in remove:
                child[idx] = 0
        else:
            to_add = int(TARGET_SUM - ones_count)
            zeros_pos = np.where(child == 0)[0]
            add = np.random.choice(zeros_pos, to_add, replace=False)
            for idx in add:
                child[idx] = 1
    return child
