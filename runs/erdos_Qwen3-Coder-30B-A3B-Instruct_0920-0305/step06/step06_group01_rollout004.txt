import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_weights(S, alpha):
    """Compute weights for soft minimum of correlations"""
    S_max = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * S - S_max)
    return exp_vals / np.sum(exp_vals)

def compute_gradient(h, w):
    """Compute gradient from soft minimum weights w"""
    # Create symmetric kernel from weights w
    W = np.zeros(2 * len(h) - 1)
    for k in range(len(h)):
        W[len(h) - 1 + k] = w[k]
        W[len(h) - 1 - k] = w[k]
    
    # Convolution
    conv = np.convolve(h, W, mode='full')
    g = conv[len(h) - 1:len(h) - 1 + len(h)]
    # Double the contribution from lag 0
    g += w[len(h) - 1] * h
    return g
