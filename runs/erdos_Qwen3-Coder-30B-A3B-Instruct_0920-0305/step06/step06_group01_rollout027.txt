```python
import numpy as np
from scipy.signal import correlate
import time

# Global constants and variables
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Next power of 2 for FFT efficiency
TARGET_SUM = n_points / 2
MAX_PHASE_RESETS = 5
PHASE_RESET_INTERVAL = 300


def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    # Zero pad to length L
    h_padded = np.pad(h_rev, (0, L - len(h_rev)), mode='constant')
    b_padded = np.pad(b, (0, L - len(b)), mode='constant')
    corr = np.fft.ifft(np.fft.fft(h_padded) * np.conj(np.fft.fft(b_padded))).real
    return corr


def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)


def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)


def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b) / 2, 0, 1)
    return result


def initialize_h():
    """Initialize h from the provided initial_h_values, interpolated to n_points"""
    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interpolated = np.interp(x_new, x_orig, h0)
    h_projected = project_to_sum_box(h_interpolated, TARGET_SUM)
    return h_projected


def compute_target_magnitude(P):
    """Compute target magnitude with flat off-DC part"""
    # Compute sum of off-DC elements
    off_dc_sum = np.sum(P[1:])
    if off_dc_sum == 0:
        # All power in DC, use current structure
        return np.sqrt(P)
    # Target magnitude for off-DC
    target_val = np.sqrt(off_dc_sum / (L - 1))
    M_target = np.full(L, target_val)
    M_target[0] = np.sqrt(P[0])
    return M_target


def spectral_project(h, lambda_mix, phase_reset_count):
    """Perform one step of the alternating projection: spectral flatness update, then inverse FFT, then projection."""
    # 1. Compute forward FFT
    h_padded = np.pad(h, (0, L - len(h)), mode='constant')
    H = np.fft.fft(h_padded)
    P = np.abs(H)**2
    
    # 2. Compute target magnitude
    M_target = compute_target_magnitude(P)
    
    # 3. Mix magnitude
    M = (1 - lambda_mix) * np.abs(H) + lambda_mix * M_target
    # Preserve DC magnitude
    M[0] = np.abs(H[0])
    
    # 4. Phase preservation
    H_new = M * np.exp(1j * np.angle(H))
    
    # 5. Inverse FFT
    h_new = np.fft.ifft(H_new).real[:n_points]  # Keep only the first n_points
    
    # 6. Projection onto feasibility
    h_projected = project_to_sum_box(h_new, TARGET_SUM)
    
    # Random phase reset with low probability
    if phase_reset_count > 0 and np.random.rand() < 0.01:
        H_new = M * np.exp(1j * np.random.uniform(0, 2*np.pi, L))
        h_new = np.fft.ifft(H_new).real[:n_points]
        h_projected = project_to_sum_box(h_new, TARGET_SUM)
        phase_reset_count -= 1  # Use up one reset
    
    return h_projected, phase_reset_count


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize solution
    h = initialize_h()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Iterative algorithm parameters
    lambda_mix = 0.05  # Initial mixing weight
    alpha_lambda = 1.02  # Geometric increase rate
    beta = 0.5  # Relaxation blend
    alpha_beta = 1.01  # Blend increase rate
    phase_reset_count = MAX_PHASE_RESETS
    
    iter_count = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 5:  # Reserve 5 seconds for refinement
        iter_count += 1
        
        # Update mixing weight and blend
        if lambda_mix < 1.0:
            lambda_mix = min(lambda_mix * alpha_lambda, 1.0)
        
        beta = min(beta * alpha_beta, 1.0)
        
        # Perform spectral projection
        h_new, phase_reset_count = spectral_project(h, lambda_mix, phase_reset_count)
        
        # Relaxation: blend with previous point
        h_new = (1 - beta) * h + beta * h_new
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Update best solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Conditionally reset phases
        if no_improve_count > PHASE_RESET_INTERVAL:
            # Important to not reset phase too often to prevent oscillation
            H = np.fft.fft(np.pad(h_new, (0, L - len(h_new)), mode='constant'))
            M = np.abs(H)
            H_new = M * np.exp(1j * np.random.uniform(0, 2*np.pi, L))
            h_new = np.fft.ifft(H_new).real[:n_points]
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            no_improve_count = 0
            if evaluate_c5(h_new) < best_c5:
                best_h = h_new.copy()
                best_c5 = evaluate_c5(h_new)
        
        h = h_new.copy()
        
        # Periodic monitoring
        if iter_count % 2000 == 0:
            print(f"{iter_count} iterations elapsed. Current C5 = {best_c5:.6f}")
    
    # Final polishing with SLSQP
    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-14}
        )
        if res.success:
            best_h = res.x
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
```
