TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    best_h = h.copy()

    # Build gradient approximation
    for _ in range(max_iter):

        corr = correlate(h, 1 - h, mode='full')

        # Compute approximate gradient using finite differences
        grad = np.zeros(n_points)

                c5_plus = evaluate_c5(h_test)
                grad[i] = (c5_plus - best_c5) / delta
            else:
                grad[i] = 1e10  # penalty for infeasible step
        
        # Use gradient to guide moves
        # Find points with best decrease in objective
        sorted_idx = np.argsort(grad)
        
        # Try moves from worst to best
        moved = False
        for i in sorted_idx:
            if not moved:

                # Find a good destination position (either the positive peak)
                for j in range(n_points):

                        delta = min(h[i], 1 - h[j], 0.01)

                            h[j] += delta
                            h = project_to_sum_box(h, TARGET_SUM)
                            c5 = evaluate_c5(h)
                            if c5 < best_c5:
                                best_c5 = c5
                                best_h = h.copy()
                                moved = True
                            break
            if moved:
                break
        
        # If no improvement, small random perturbation
        if not moved:
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
