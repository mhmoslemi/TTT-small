import numpy as np

from scipy.optimize import minimize

# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        # Compute gradient for all positions
        grad = np.zeros(n_points)

            # Find the maximum correlation contribution from position i
            contribution = 0

                lag = abs(i - j)
                if lag == max_lag:
                    contribution += h[j] * (1 - h[i])  # Simplified gradient approximation
            
            grad[i] = contribution
        
        # Find best indices to adjust
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if grad[idx_min] >= grad[idx_max] or idx_min == idx_max:
            # Small random adjustment
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)

                    h[j] += delta

                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
        else:
            # Move mass from min to max (favoring the direction with higher negative gradient)
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    return best_h, best_c5

def local_search_with_pattern(h0, pattern_size=30):
    """Use a pattern-based approach with local search"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Consider a pattern to move around
    for _ in range(500):
        # Randomly select a window size
        window_size = min(pattern_size, n_points)
        start = np.random.randint(0, n_points - window_size + 1)
        
        # Extract and modify the pattern
        pattern = h[start:start+window_size].copy()
        pattern = np.clip(pattern + np.random.normal(0, 0.01, window_size), 0, 1)
        pattern = project_to_sum_box(pattern, window_size/2)
        
        # Store new values
        h_new = h.copy()
        h_new[start:start+window_size] = pattern
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5
