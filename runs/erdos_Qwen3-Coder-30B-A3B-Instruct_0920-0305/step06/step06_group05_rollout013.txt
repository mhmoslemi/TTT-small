import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def solve_milp(n, time_limit):
    """Solve the ILP for binary variables to get exact solution"""
    dx = 2.0 / n
    M = n // 2  # required number of ones
    
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # auxiliary variables for y_ij = h_i AND h_j
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(boolean=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == M,
        t >= 0
    ]
    
    # Linking constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.extend([
                y[(i,j)] <= h[i],
                y[(i,j)] <= h[j],
                y[(i,j)] >= h[i] + h[j] - 1
            ])
    
    # Lag constraints
    for k in range(1, n):
        # sum of y_{i,i+k} over valid indices
        lag_sum = cp.sum([y[(i, i+k)] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, 
                   verbose=False,
                   cbc_params={"seconds": time_limit})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract results
            h_sol = np.round(h.value).astype(int)
            t_sol = t.value
            return h_sol, t_sol
        else:
            return None, None
            
    except Exception:
        return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
