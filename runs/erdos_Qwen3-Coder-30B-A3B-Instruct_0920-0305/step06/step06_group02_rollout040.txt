import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_convolution_update(conv, ones, i, j, n):
    """Compute new convolution after swapping elements at i and j"""
    # Compute offset for converting shift to array index
    offset = n - 1
    
    # Make a copy of the old convolution
    conv_new = conv.copy()
    
    # Remove contributions of i (which changes from 1 to 0)
    for k in ones:
        shift = k - i
        idx = shift + offset
        conv_new[idx] -= 1
        if k != i:
            conv_new[-shift + offset] -= 1
    
    # Update ones list
    ones.remove(i)
    ones.append(j)

    # Add contributions of j (which changes from 0 to 1)
    for k in ones:
        shift = k - j
        idx = shift + offset
        conv_new[idx] += 1
        if k != j:
            conv_new[-shift + offset] += 1
    
    # Add self-pair for j (just one copy)
    conv_new[offset] += 1
    
    return conv_new
