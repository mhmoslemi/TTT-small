import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_improve(h, target_sum):
    """Local improvement using gradient descent with projected updates"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            return project_to_sum_box(res.x, target_sum)
    except:
        pass
    return h

def clustering_based_search(seed_h, target_sum):
    """Use clustering to find promising regions in the solution space"""
    np.random.seed(42)
    best_h = seed_h.copy()
    best_c5 = evaluate_c5(seed_h)
    
    # Try various patterns based on clustering
    for trial in range(10):
        # Create a "clustered" pattern by grouping similar values
        h_trial = np.zeros(n_points)
        clusters = 3 + trial % 5  # vary number of clusters
        for i in range(clusters):
            start_idx = int(i * n_points / clusters)
            end_idx = int((i + 1) * n_points / clusters)
            if i % 2 == 0:
                h_trial[start_idx:end_idx] = 0.8
            else:
                h_trial[start_idx:end_idx] = 0.2
        h_trial = project_to_sum_box(h_trial, target_sum)
        
        # Local improvement
        h_opt = local_improve(h_trial, target_sum)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):

    # Generate initial solution
    h0 = generate_initial_guess()

            best_h = h_clust.copy()
