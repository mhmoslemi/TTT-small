import numpy as np
import cvxpy as cp
import time

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary h variables"""
    M = n // 2  # Required number of ones
    dx = 2.0 / n
    
    # Decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)  # y[i,j] = h[i] AND h[j]
    t = cp.Variable()

    # Constraints
    constraints = [cp.sum(h) == M]
    
    # Linking constraints for y variables
    for i in range(n):
        for j in range(i+1, n):
            constraints.extend([
                y[i, j] <= h[i],
                y[i, j] <= h[j],
                y[i, j] >= h[i] + h[j] - 1
            ])

    # Lag constraints
    for k in range(1, n):
        # Sum over valid indices where i+k < n
        sum_y = cp.sum([y[i, i+k] for i in range(n-k)])
        constraints.append(t >= M - sum_y)

    # Fix h[0] = 1 to break symmetry
    constraints.append(h[0] == 1)

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
    except:
        return None, None

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = np.rint(h.value).astype(int)
        t_val = t.value
        return h_sol, t_val
    else:
        return None, None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Perform continuous refinement using SLSQP"""
    n = len(h)
    TARGET_SUM = n / 2
    bounds = [(0, 1) for _ in range(n)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = cp.minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
