import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def evaluate_c5(h):
    """Evaluate C5 using FFT-based correlation"""
    corr = correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_weights(S, alpha):
    """Compute weights for soft minimum with temperature alpha"""
    S_max = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * S - S_max)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def compute_gradient(h, weights):
    """Compute gradient using convolution"""
    # Build symmetric kernel from weights
    n = len(h)
    W = np.zeros(2 * n - 1)
    for k in range(n):
        W[n-1+k] = weights[k]
        if k > 0:
            W[n-1-k] = weights[k]
    
    # Convolution to get gradient
    conv = np.convolve(h, W, mode='full')
    g = conv[n-1 : n-1 + n]
    # Add contribution from lag 0 (double counted)
    g += weights[0] * h
    return g

    # Initialize best solution
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2
    
    # Optimisation parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30.0
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    stagnation_counter = 0
    max_iter = 10000
    
    h = h_interpolated.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute auto-correlations
        corr = correlate(h, h, mode="full")
        # The auto-correlation values at lag k are corr[n-1+k] where k ranges from -(n-1) to n-1
        S = corr[n_points-1 : 2*n_points-1]
        
        # Compute soft minimum weights
        weights = compute_soft_min_weights(S, alpha)
        
        # Compute gradient
        g = compute_gradient(h, weights)
        
        # Update solution
        h_candidate = h + step * g
        h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
        
        # Evaluate exact bound
        c5_candidate = evaluate_c5(h_candidate)
        
        # Accept or reject move
        if c5_candidate < best_c5:
            h = h_candidate
            best_h = h_candidate
            best_c5 = c5_candidate
            step = min(step * step_inc, 0.1)
            stagnation_counter = 0
        else:
            step = max(step * step_dec, 1e-4)
            stagnation_counter += 1
            
        # Annealing
        alpha = min(alpha * alpha_inc, alpha_max)
        
        # Random perturbation occasionally
        if i % 200 == 0 and i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + noise
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            c5_perturbed = evaluate_c5(h_perturbed)
            if c5_perturbed < best_c5:
                h = h_perturbed
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        # Full restart if stagnated too long
        if stagnation_counter >= restart_interval:

            alpha = 0.5

                h_refined = project_to_sum_box(h_refined, TARGET_SUM)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
