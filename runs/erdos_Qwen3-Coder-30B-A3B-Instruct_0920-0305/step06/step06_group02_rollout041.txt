import numpy as np
from scipy.signal import correlate
import time
import random

def compute_convolution(h):
    """Compute convolution of h with itself using FFT for efficiency"""
    # h is binary vector of length n
    # correlation = h * reverse(h) with zero padding (full mode)
    # this gives us correlation for all possible shifts (including negative)
    # result length 2*n -1
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    # Use full convolution
    corr = np.correlate(h, h_rev, mode='full')
    return corr

def evaluate_c5(h):
    """Fast evaluation using precomputed correlation."""
    # Compute full correlation of h with itself
    corr = compute_convolution(h)
    n = len(h)
    w = n // 2  # target sum
    # max correlation = w - min(corr)
    # we want to minimize max correlation = minimize (w - min)
    # i.e., maximize min(corr)
    # because when we shift by k, number of times both h and shifted h are 1 is corr[k]
    return (w - np.min(corr)) * (2.0 / n)

def apply_swap(conv, ones, zeros, i, j, n):
    """
    Apply swap of positions i (1) and j (0) to update convolution efficiently.
    conv: array of length 2*n-1, indexed from -n+1 to n-1
    ones: list of positions where h=1
    zeros: list of positions where h=0
    i: position to change from 1 to 0
    j: position to change from 0 to 1
    """
    # Modify conv to reflect the swap of i and j.
    # Update: remove contributions from i, then add from j
    
    # Remove old contributions of i:
    # i contributes pairs (i, k) where k is any k with h[k]=1 (before swap)
    # For each such k, the pair contributes 1 to correlation at shift (k-i)
    # We also have pair (k, i) which contributes 1 to shift (i-k)
    # Still for shift s = k - i, pairs (k, i) and (i, k) are both counted in corr[s + (n-1)] and corr[-s + (n-1)]
    # We need to subtract from both positions to get correct shift correlation 
    
    # Remove from correlation at all shifts: simply go through all k in ones_set
    #   shift = k - i  -> corr[shift + (n-1)] -= 1 (add n-1 offset to map -n+1 to 0)
    #   if k != i, also subtract from corr[-(k-i) + (n-1)] = corr[(i-k) + (n-1)] 

    old_len = len(ones)
    new_ones = [k for k in ones if k != i]  # remove i
    new_ones.append(j)  # add j
    
    # Copy the convolution to modify
    new_conv = conv.copy()
    # Remove contributions of i (which was 1)
    for k in ones:
        shift = k - i
        # Index in array
        idx1 = shift + (n-1)
        new_conv[idx1] -= 1
        if k != i:  # Don't double count self
            idx2 = -shift + (n-1)
            new_conv[idx2] -= 1
    # Add contributions of j (which was 0)
    for k in new_ones:
        shift = k - j
        idx1 = shift + (n-1)
        new_conv[idx1] += 1
        if k != j:  # Don't double count self
            idx2 = -shift + (n-1)
            new_conv[idx2] += 1
    
    # Update ones and zeros
    new_ones_list = [k for k in ones if k != i]
    new_zeros_list = [k for k in zeros if k != j]
    new_ones_list.append(j)
    new_zeros_list.append(i)
    
    return new_conv, new_ones_list, new_zeros_list

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points // 2  # half of points should be 1
    best_h = None
    best_c5 = float('inf')
    
    # Initial solution using the provided reference
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        x = np.zeros(n_points, dtype=int)
        # Convert to binary approximately
        for i in range(n_points):
            x[i] = 1 if init_h[i] > 0.5 else 0
        # Ensure correct sum
        current_sum = np.sum(x)
        if current_sum != TARGET_SUM:
            # Swap some 1s to 0s and 0s to 1s to make sum correct
            ones_idx = np.where(x == 1)[0]
            zeros_idx = np.where(x == 0)[0]
            # Should not happen in most cases
            if len(ones_idx) > TARGET_SUM:
                change_indices = ones_idx[:len(ones_idx) - TARGET_SUM]
                for idx in change_indices:
                    x[idx] = 0
                # Add to zeros
                change_zero_indices = zeros_idx[:TARGET_SUM - len(ones_idx)]  # This will not happen if actual capacity allows
            elif len(ones_idx) < TARGET_SUM:
                change_indices = zeros_idx[:TARGET_SUM - len(ones_idx)]
                for idx in change_indices:
                    x[idx] = 1
        # Compute initial convolution
        recon = np.correlate(x, x[::-1], mode='full')
        init_c5 = (TARGET_SUM - np.min(recon)) * dx
        if init_c5 < best_c5:
            best_h = x.copy()
            best_c5 = init_c5
    else:
        # Create random binary pattern with correct sum
        x = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, TARGET_SUM, replace=False)
        x[indices] = 1
        recon = np.correlate(x, x[::-1], mode='full')
        init_c5 = (TARGET_SUM - np.min(recon)) * dx
        if init_c5 < best_c5:
            best_h = x.copy()
            best_c5 = init_c5
    
    # Multiple restarts with simulated annealing
    restart_count = 0
    max_restarts = 50
    
    while restart_count < max_restarts and time.time() < start_time + 0.9 * budget_s:
        # Random restart
        x = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, TARGET_SUM, replace=False)
        x[indices] = 1
        
        # Get initial conv and min_overlap
        conv = np.correlate(x, x[::-1], mode='full')
        w = TARGET_SUM  # weight
        min_overlap = np.min(conv)
        c5 = (w - min_overlap) * dx
        
        # Current best for this restart
        current_best = c5
        current_x = x.copy()
        current_conv = conv.copy()
        
        # SA parameters
        temp = 1.0
        cool_rate = 0.999
        max_iter = 10000
        iter_count = 0
        accept_count = 0

        # Local optimization with SA or hill climbing
        while iter_count < max_iter and time.time() < start_time + 0.9 * budget_s:
            iter_count += 1
            
            # Get current indices for valid swaps
            ones = list(np.where(current_x == 1)[0])
            zeros = list(np.where(current_x == 0)[0])
            
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            # Choose two random positions, one with 1, one with 0
            i = random.choice(ones)      # source of 1 
            j = random.choice(zeros)     # target of 0

            # Apply swap with update of conv
            new_conv, new_ones, new_zeros = apply_swap(
                current_conv, ones, zeros, i, j, n_points)
            
            new_min = np.min(new_conv)
            new_c5 = (w - new_min) * dx
            
            # Accept the move
            delta = new_c5 - c5
            
            if delta <= 0 or random.random() < np.exp(-delta / temp):
                # Accept this move
                current_x = x.copy()
                current_x[i] = 0
                current_x[j] = 1
                current_conv = new_conv
                c5 = new_c5
                min_overlap = new_min
                
                # Update current best
                if c5 < current_best:
                    current_best = c5
                    current_x = x.copy()
                    current_conv = new_conv.copy()
                    
            # Reduce temperature
            temp *= cool_rate
            
        # Store the best solution found in this restart
        if current_best < best_c5:
            best_c5 = current_best
            best_h = current_x.copy()
            
        restart_count += 1

    # Final conversion to float representation
    final_h = best_h.astype(float).tolist()
    return (final_h, float(best_c5), n_points)
