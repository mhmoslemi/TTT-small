import numpy as np
import time
from scipy.signal import correlate

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation using scipy signal correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    low  = -np.max(x) - 1.0
    high =  1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y   = np.clip(x + lam, 0.0, 1.0)
        s   = y.sum()
        if s > target_sum:
            high = lam
        else:
            low  = lam
    y = np.clip(x + (low+high)/2.0, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def full_corr(a, b):
    """Compute full correlation using FFT to avoid wrap-around."""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def grad_softmax_max(h, alpha):
    """Compute gradient of soft-max of correlations."""
    corr = full_corr(h, 1-h)
    maxc = corr.max()
    # Soft max weights
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w /= w_sum

    # Vectorized construction of gradient
    # shift_forward[i] = h[i+lag] for lag in range(-n+1,n)
    # shift_backward[i] = h[i-lag] for lag in range(-n+1,n)
    conv_fwd = full_corr(h, w)  # actually the convolution of h with w
    w_rev = w[::-1]
    conv_bwd = full_corr(h, w_rev)
    grad = 1.0 - conv_fwd - conv_bwd
    return grad

def backtrack_line_search(h, grad, alpha, eta, target_sum):
    """Simple backtracking line search to ensure sufficient decrease."""
    while eta > 1e-6:
        h_new = project_to_sum_box(h - eta * grad, target_sum)
        c5_new = evaluate_c5(h_new)
        c5_old = evaluate_c5(h)
        if c5_new < c5_old:  # accept if improved
            break
        eta *= 0.5
    return h_new, eta

def optimize_continuous(start_h, remaining_time, n, dx, target_sum, max_iter=5000):
    """Run projected gradient descent until time runs out."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    alpha = 5.0  # start with low α to avoid sharp gradients initially
    eta = 0.1
    iter_count = 0
    start_time = time.time()
    
    while iter_count < max_iter and (time.time() - start_time) < remaining_time:
        grad = grad_softmax_max(h, alpha)
        h, eta = backtrack_line_search(h, grad, alpha, eta, target_sum)
        val = evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        # gradually increase α for sharper focus on max
        if iter_count % 100 == 0 and alpha < 20.0:
            alpha += 0.2
        iter_count += 1
    return best_h, best_val

def binary_local_search(h_bin, remaining_time, n, dx):
    """Perform simulated annealing or hill climbing on binary solutions."""
    start_time = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_c5(best_h)
    # initial temperature
    T = 0.01
    
    while (time.time() - start_time) < remaining_time:
        # pick random 1 and 0 indices to swap
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        
        val = evaluate_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()
        # optionally accept worse with decreasing temp, but not for speed
        elif val < best_val + T * np.random.rand():  # fast simple noise
            best_h = cand.copy()
            best_val = val
        
        # cooling schedule
        T *= 0.99995
    return best_h, best_val

    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    TARGET_SUM = n_points / 2.0

    # Phase 1: Continuous optimization from multiple starting points
    seeds = [h_interpolated]
    for _ in range(4):
        # Random continuous starting
        h_rand = np.random.rand(n_points)
        h_rand = project_to_sum_box(h_rand, TARGET_SUM)
        seeds.append(h_rand)

    # Multi-start loop
    phase1_time = 0.3 * budget_s
    remaining = phase1_time
    for i, seed_h in enumerate(seeds):
        if remaining <= 0:

        h_opt, c5_opt = optimize_continuous(seed_h.copy(), remaining, n_points, dx, TARGET_SUM)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        remaining -= 0.05 * phase1_time  # distributed allocation

    # Phase 2: Additional random restarts with perturbations
    phase2_time = 0.3 * budget_s
    remaining = phase2_time
    iter_count = 0
    while remaining > 3 and iter_count < 5:
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = project_to_sum_box(h_perturb, TARGET_SUM)
        h_opt, c5_opt = optimize_continuous(h_perturb, 0.05 * phase2_time, n_points, dx, TARGET_SUM)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        iter_count += 1
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        
        if remaining < 10:  # if out of budget, stop early
            break

    # Phase 3: Binary refinement
    phase3_time = 0.25 * budget_s
    remaining = phase3_time
    
    # Convert to binary approximation
    bin_approx = (best_h >= np.median(best_h)).astype(float)
    # Ensure exact sum
    ones_pos = np.where(bin_approx == 1)[0]
    if len(ones_pos) > n_points // 2:
        to_zero = ones_pos[:len(ones_pos) - n_points // 2]
        bin_approx[to_zero] = 0
    elif len(ones_pos) < n_points // 2:
        to_one = np.where(bin_approx == 0)[0]
        to_one = to_one[:n_points // 2 - len(ones_pos)]
        bin_approx[to_one] = 1
    
    # Use binary local search
    final_h_bin, final_c5_bin = binary_local_search(bin_approx, remaining, n_points, dx)
    if final_c5_bin < best_c5:
        best_c5 = final_c5_bin
        best_h = final_h_bin.copy()

    if budget_s - (time.time() - start_time) > 20:
        n_upscale = min(400, n_points * 2)  # Increase resolution
        dx_up = 2.0 / n_upscale
        x_up = np.linspace(0, 2, n_upscale)
        h_upscale = np.interp(x_up, np.linspace(0, 2, n_points), best_h)
        h_upscale = project_to_sum_box(h_upscale, n_upscale / 2.0)
        
        # Run few more continuous iterations
        h_upscale_final, c5_upscale = optimize_continuous(
            h_upscale, 0.1 * (budget_s - (time.time() - start_time)), 
            n_upscale, dx_up, n_upscale / 2.0, max_iter=200
        )
        if c5_upscale < best_c5:
            best_c5 = c5_upscale
            best_h = h_upscale_final.copy()
            n_points = n_upscale
            dx = dx_up

    return (best_h.tolist(), float(best_c5), n_points)
