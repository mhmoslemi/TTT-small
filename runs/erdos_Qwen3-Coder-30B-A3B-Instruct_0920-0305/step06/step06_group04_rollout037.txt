import numpy as np

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return dx * np.max(corr)

def build_milp(N, max_time=300):
    """Build and solve the MILP formulation of the problem"""
    h = cp.Variable(N, integer=True)
    y = {}
    for ell in range(N):
        for i in range(N - ell):
            # make y[i, ell] a binary variable
            y[(i, ell)] = cp.Variable(integer=True)
    
    # Lambda is an integer variable for the minimal overlap
    lam = cp.Variable(integer=True)
    
    # Objective: maximize lambda
    objective = cp.Maximize(lam)
    
    constraints = []
    
    # Constraint 1: sum(h) = N/2  
    constraints.append(cp.sum(h) == N/2)
    
    # Constraint 2: for all (i,ell), y[i,ell] represents h[i] * h[i+ell]
    for ell in range(N):
        for i in range(N - ell):
            # y[i,ell] <= h[i],   y[i,ell] <= h[i+ell]
            constraints.append(y[(i, ell)] <= h[i])
            constraints.append(y[(i, ell)] <= h[i+ell])
            # y[i,ell] >= h[i] + h[i+ell] - 1
            constraints.append(y[(i, ell)] >= h[i] + h[i+ell] - 1)
            # y[i,ell] binary
            constraints.append(y[(i, ell)] >= 0)
            constraints.append(y[(i, ell)] <= 1)
    
    # Constraint 3: for all lags ell, sum over i of y[i,ell] >= lam
    for ell in range(N):
        # Only compute for valid i
        constraints.append(cp.sum([y[(i, ell)] for i in range(N - ell)]) >= lam)
    
    # Constraint 4: lam should not exceed N/2
    constraints.append(lam <= N/2)
    
    # Define the model
    prob = cp.Problem(objective, constraints)
    
    # Solve with a time limit
    try:
        prob.solve(solver=cp.CBC, timelimit=max_time)
    except Exception as e:
        print(f"Solving failed: {e}")
        return None, None
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        sol_h = h.value
        sol_lam = lam.value
        return sol_h, sol_lam
    elif prob.status in [cp.INFEASIBLE, cp.UNBOUNDED]:
        return None, None
    else:
        # For other status codes, fallback to solution from relaxation
        try:
            sol_h = [min(max(h_val, 0), 1) for h_val in h.value]
            h_rounded = [int(h_val + 0.5) for h_val in sol_h]  # round to nearest int
            sol_lam = None  # Not recovered yet, but solution is available
            return np.array(h_rounded), sol_lam
        except:
            return None, None
