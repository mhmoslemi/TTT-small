import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

            # Compute overlaps for all shifts
            corr = correlate(h_current, 1 - h_current, mode='full')
            overlaps = corr * dx
            max_val = overlaps.max()
            
            # Find all indices where overlap is approximately maximum
            idxs = np.where(np.abs(overlaps - max_val) < 1e-12)[0]
            
            # Compute gradient for worst shift(s)
            grad = np.zeros_like(h_current)

                if shift >= 0:
                    grad[:n_points-shift] -= h_current[shift:]

                    grad[s:] -= h_current[:n_points-s]

            h_new = h_current + step_size * grad
