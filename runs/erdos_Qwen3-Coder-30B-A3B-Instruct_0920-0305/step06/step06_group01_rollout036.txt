import numpy as np
from scipy.signal import correlate
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    while time.time() - start_time < budget_s - 10:  # Reserve time for refinement

        # Magnitude mixing
        M_new = (1 - lambda_current) * np.abs(H) + lambda_current * target_magnitude
        
        # Reconstruct signal with new magnitude and old phase
        H_new = M_new * np.exp(1j * np.angle(H))
        h_new_padded = np.fft.ifft(H_new)

        h_proj = project_to_sum_box(h_new, target_sum)

        # Final projection to ensure constraints are exactly met

    return (best_h.tolist(), float(best_c5), n_points)
