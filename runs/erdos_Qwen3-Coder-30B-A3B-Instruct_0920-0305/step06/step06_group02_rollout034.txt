import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create initial binary vector with exact sum constraint"""
    np.random.seed(seed)
    # Start with best solution from previous program
    h_ref = np.array(initial_h_values)
    x_ref = np.linspace(0, 2, len(h_ref))
    x_new = np.linspace(0, 2, n_points)
    h_interp = np.interp(x_new, x_ref, h_ref)
    
    # Round and project to exact sum
    x = (h_interp > 0.5).astype(np.int8)
    actual_sum = np.sum(x)
    diff = TARGET_SUM - actual_sum
    
    if diff > 0:
        # Need to flip some 0s to 1s
        zero_indices = np.where(x == 0)[0]
        flip_indices = np.random.choice(zero_indices, diff, replace=False)
        x[flip_indices] = 1
    elif diff < 0:
        # Need to flip some 1s to 0s
        one_indices = np.where(x == 1)[0]
        flip_indices = np.random.choice(one_indices, -diff, replace=False)
        x[flip_indices] = 0
    
    return x

def full_overlap(x):
    """Compute full overlap array"""
    return np.correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros):
    """Update overlap array in O(w) time after swapping positions i,j"""
    ov_new = ov.copy()
    w = len(ones)
    
    # Remove contributions of i (both directions)
    for k in ones:
        if k == i:
            continue
        s = k - i
        ov_new[s + n_points - 1] -= 1
    
    # Add contributions of j (both directions)
    for k in ones:
        if k == i:
            continue
        s = k - j
        ov_new[s + n_points - 1] += 1
    
    # Special case for 1->1 in same index
    if i != j:
        s = i - j
        ov_new[s + n_points - 1] += 1
    
    return ov_new

def get_ones_zeros(x):
    """Get indices of ones and zeros"""
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    return ones, zeros

def simulated_annealing(x0, time_limit):
    """Run simulated annealing with incremental updates"""
    start_time = time.time()
    
    # Initial setup
    x = x0.copy()
    ones, zeros = get_ones_zeros(x)
    ov = full_overlap(x)
    cur_min = np.min(ov)
    
    # Best so far
    best_x = x.copy()
    best_min = cur_min
    
    # Annealing parameters
    T0 = 0.5
    lambda_param = 5.0  # cooling rate
    max_iter = 200000  # limit iterations for finite time
    iteration = 0
    
    # Update timer for early exit
    while iteration < max_iter and time.time() - start_time < time_limit:
        # Temperature schedule
        t = iteration / max_iter
        T = T0 * np.exp(-lambda_param * t)
        
        # Pick random 1 and 0 positions
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Store old overlap state for possible revert
        old_ov = ov.copy()
        old_ones, old_zeros = ones.copy(), zeros.copy()
        
        # Perform incremental update
        ov_new = swap_update(ov, i, j, ones, zeros)
        new_min = np.min(ov_new)
        
        # Acceptance probability for simulated annealing
        delta = new_min - cur_min
        if delta > 0 or np.random.random() < np.exp(delta / T):
            # Accept the move
            x[i] = 0
            x[j] = 1
            ones = np.setdiff1d(ones, [i])
            ones = np.append(ones, j)
            zeros = np.setdiff1d(zeros, [j])
            zeros = np.append(zeros, i)
            ov = ov_new
            cur_min = new_min
            
            if new_min > best_min:
                best_x = x.copy()
                best_min = new_min
        else:
            # Revert (should not happen as we keep the unchanged array)
            pass
        
        iteration += 1
        
        # Check for earlyexit
        if time.time() - start_time > time_limit * 0.95:
            break
    
    return best_x, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine by improving the worst shift systematically"""
    w = len(ones)
    best_ov = ov.copy()
    best_x = x.copy()
    min_idx = np.argmin(ov)
    best_min = ov[min_idx]
    shift = min_idx - n_points + 1  # actual shift value
    
    # Determine_candidate swaps for this shift
    num_swaps = 0
    max_swaps = 500
    
    # Find deficiency and benefit sets
    # Deficient 1-positions (value 1 with no 1 in shifted position)
    def_set = np.where((best_x == 1) & (np.roll(best_x, shift) == 0))[0]
    # Beneficial 0-positions (value 0 with 1 in shifted position)
    ben_set = np.where((best_x == 0) & (np.roll(best_x, shift) == 1))[0]
    
    # Try all valid swaps from def_set to ben_set
    def_list = def_set.tolist()
    ben_list = ben_set.tolist()
    
    for i in def_list:
        for j in ben_list:
            if i != j:
                # Test if this swap improves the worst shift
                ov_test = ov.copy()
                ov_test[min_idx] += 1  # This is the change due to swapping
                
                # Try it on possible copy
                test_x = best_x.copy()
                test_x[i] = 0
                test_x[j] = 1
                
                # Update ov for this swap (simplified - use full correlation)
                test_ov = full_overlap(test_x)
                new_min = np.min(test_ov)
                
                if new_min > best_min and num_swaps < max_swaps:
                    best_x = test_x
                    best_ov = test_ov
                    best_min = new_min
                    num_swaps += 1
                
                # For speed, only consider a few pairs
                if num_swaps > 10:
                    break
        if num_swaps > 10:
            break
    
    # Put the best result back
    return best_x, best_ov, best_min

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    x = initial_binary(seed)
    best_x = x.copy()
    
    ones, zeros = get_ones_zeros(x)
    ov = full_overlap(x)
    cur_min = np.min(ov)
    best_min = cur_min
    
    # Phase 1: Multiple SA runs with different seeds and restarts
    n_restart = 10
    for restart in range(n_restart):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Randomize seed for each restart
        np.random.seed(seed + restart)
        x_start = initial_binary(seed + restart)
        x_result, min_result = simulated_annealing(x_start, budget_s * 0.9)
        
        if min_result > best_min:
            best_x = x_result.copy()
            best_min = min_result
    
    # Phase 2: Performance improvement via worst-shift repair
    ones, zeros = get_ones_zeros(best_x)
    ov = full_overlap(best_x)
    # Let it run with more time if possible
    if time.time() - start_time < 0.95 * budget_s:
        for _ in range(3):  # allow up to 3 repairs
            if time.time() - start_time > 0.95 * budget_s:
                break
            _, ov, best_min = worst_shift_refine(best_x, ov, ones, zeros)
            ones, zeros = get_ones_zeros(best_x)
    
    # Final evaluation
    ov_final = full_overlap(best_x)
    final_min = np.min(ov_final)
    final_c5 = dx * (TARGET_SUM - final_min)
    
    # Return as list of floats
    h_values = [float(xi) for xi in best_x]
    return (h_values, float(final_c5), n_points)
