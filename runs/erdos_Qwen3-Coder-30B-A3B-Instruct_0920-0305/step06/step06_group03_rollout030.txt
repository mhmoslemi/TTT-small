# Global variables
TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_overlaps(h):
    """Compute overlap for all shifts using FFT"""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def subgradient_descent(h0, target_sum, max_iter=5000, step0=0.05, decay=0.9995):

    h = h0.copy()
    step = step0

    for i in range(max_iter):
        # Compute overlaps for all shifts
        overlaps = compute_overlaps(h)
        
        # Find indices with maximum overlap
        max_val = overlaps.max()
        idxs = np.where(np.abs(overlaps - max_val) < 1e-12)[0]
        
        # Compute averaged gradient for worst shifts
        grad = np.zeros_like(h)

        # Update h using subgradient step
        h = h - step * grad
        
        # Project onto constraints
        h = np.clip(h, 0, 1)

        step *= decay
        
        # Update best solution
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
