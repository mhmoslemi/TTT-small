import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 256
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_minimum_gradient(h, alpha=0.5):
    """Compute gradient of soft minimum of auto-correlations"""
    # Compute auto-correlations S_k(h)
    S = correlate(h, h, 'full')
    # Shift indices to include negative lags
    n = len(h)
    S = S[n-1 : 2*n-1]
    
    # Numerical stability: subtract max before exponentiating
    S_max = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * (S - S_max))
    w = exp_vals / np.sum(exp_vals)
    
    # Build symmetric kernel for convolution
    W = np.zeros(2*n - 1)
    for i in range(n):
        W[n-1+i] = w[i]
        W[n-1-i] = w[i]
    
    # Convolution with symmetric kernel
    conv = np.convolve(h, W, mode='full')
    g = conv[n-1 : n-1 + n]
    
    # Correct for center contribution (double counting)
    g += w[n-1] * h
    
    return g

    h_interpolated = project_to_sum_box(h_interpolated, TARGET_SUM)
    
    # Initial solution
    best_h = h_interpolated.copy()
    best_c5 = evaluate_c5(h_interpolated)
    
    # Optimisation parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    stagnation_counter = 0
    
    h = h_interpolated.copy()
    current_c5 = best_c5

    # Gradient ascent
    while time.time() - start_time < budget_s * 0.95:
        # Compute gradient
        g = soft_minimum_gradient(h, alpha)
        
        # Candidate step
        h_new = h + step * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate true bound
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject
        if c5_new < current_c5:
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            step = min(step * step_inc, 0.1)
            stagnation_counter = 0
        else:
            step = max(step * step_dec, 1e-4)
            stagnation_counter += 1
            
        # Temperature annealing
        alpha = min(alpha * alpha_inc, alpha_max)
        
        # Random perturbation
        if stagnation_counter % 200 == 0:
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_sum_box(h_perturbed, TARGET_SUM)
            h = h_perturbed
            
        # Full restart
        if stagnation_counter >= restart_interval:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_sum_box(h, TARGET_SUM)
            step = 0.02
            alpha = 0.5
            stagnation_counter = 0
    
    # Local refinement near the end
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}

                    best_h = refined_h
                    best_c5 = c5_refined
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_points)
