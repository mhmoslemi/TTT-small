import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute full correlation: h * (1 - h) with zero padding
    h_pad = np.pad(h, (0, n_points - 1), mode='constant')
    h_rev = np.pad(1 - h, (0, n_points - 1), mode='constant')
    corr = fftconvolve(h_pad, h_rev, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr_vals, alpha):
    """Compute softmax weights for minimax objective"""
    max_val = np.max(corr_vals)
    shifted = corr_vals - max_val
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def compute_gradient(h, weights):
    """Compute gradient of soft-max objective using FFT"""
    # Forward and backward convolution with weights
    weights_full = np.pad(weights, (0, len(h) - 1), mode='constant')
    h_pad = np.pad(h, (0, len(weights_full) - 1), mode='constant')
    h_rev = np.pad(h, (len(weights_full) - 1, 0), mode='constant')
    
    # Convolution in frequency domain
    h_fft = np.fft.fft(h_pad)
    w_fft = np.fft.fft(weights_full)
    
    # Forward convolution
    forward_conv = np.fft.ifft(h_fft * w_fft).real
    forward_conv = forward_conv[:len(h)]
    
    # Backward convolution (pos shift) -> equivalent to conv with reversed weights 
    w_rev = weights_full[::-1]
    w_rev_fft = np.fft.fft(w_rev)
    backward_conv = np.fft.ifft(h_fft * w_rev_fft).real
    backward_conv = backward_conv[-len(h):]  # takes that part
    
    grad = 1.0 - forward_conv - backward_conv
    return grad

def projected_gradient_descent(h_seed, max_iter=5000, alpha_start=5.0, alpha_max=20.0,
                              eta0=0.5, eta_max=0.1):
    """Run projected gradient descent on soft-max surrogate"""
    h = h_seed.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha = alpha_start
    eta = eta0
    history = []
    
    for i in range(max_iter):
        if i > 0 and i % 200 == 0:
            if len(history) > 10 and all(c >= best_c5 - 1e-8 for c in history[-10:]):
                break
            history = []
        
        # Full correlation
        h_padded = np.pad(h, (0, n_points - 1), mode='constant')
        h_rev_padded = np.pad(1 - h, (0, n_points - 1), mode='constant')
        corr_full = fftconvolve(h_padded, h_rev_padded, mode='full')
        corr_vals = corr_full[:2 * n_points - 1]
        
        # Softmax weights with current alpha
        weights = softmax_weights(corr_vals, alpha)
        
        # Compute gradient
        grad = compute_gradient(h, weights)
        
        # Backtracking line search
        h_trial = h - eta * grad
        h_trial = project_to_sum_box(h_trial, TARGET_SUM)
        c5_trial = evaluate_c5(h_trial)
        
        attempts = 0
        while c5_trial >= best_c5 and attempts < 10:
            eta /= 2.0
            h_trial = h - eta * grad
            h_trial = project_to_sum_box(h_trial, TARGET_SUM)
            c5_trial = evaluate_c5(h_trial)
            attempts += 1
        
        h = h_trial
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h.copy()
            history = []
        else:
            history.append(c5_trial)
            
        # Update alpha and eta
        alpha = min(alpha_max, alpha * 1.01)
        eta = min(eta_max, eta * 1.02)
    
    return best_h, best_c5

def binary_seed(n, target):
    """Generate binary seed with exactly target ones"""
    indices = np.random.permutation(n)[:target]
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def legendre_seed(n, target):
    """Generate Legendre sequence and adjust to get exactly target ones"""
    # Generate Legendre sequence of length around n+1
    p = 2 * n + 1
    while not is_prime(p):
        p += 2
    # Legendre with 0s appended to get even length
    legendre = np.array([1 if pow(i, (p-1)//2, p) == 1 else -1 for i in range(1, p)])
    if len(legendre) % 2 != 0:
        legendre = legendre[:-1]  # Remove one to make even length
    legendre = (legendre + 1) // 2  # Convert to 0/1
    while len(legendre) < n:
        legendre = np.concatenate([legendre, legendre])
    legendre = legendre[:n]
    # Adjust ones to match target
    ones_count = np.sum(legendre)
    if ones_count > target:
        indices = np.where(legendre == 1)[0]
        np.random.shuffle(indices)
        for idx in indices[:ones_count - target]:
            legendre[idx] = 0
    elif ones_count < target:
        indices = np.where(legendre == 0)[0]
        np.random.shuffle(indices)
        for idx in indices[:target - ones_count]:
            legendre[idx] = 1
    return legendre

def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(np.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True
