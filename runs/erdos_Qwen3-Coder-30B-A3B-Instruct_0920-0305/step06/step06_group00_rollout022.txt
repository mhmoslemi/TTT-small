import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    for _ in range(max_iter):
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        # Create gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            for j in range(n_points):
                if abs(i - j) == max_lag:
                    grad[i] += h[j]
        
        # Use gradient to perform moves
        # Try to move mass from areas of high gradient to areas of low gradient
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min != idx_max and grad[idx_min] < grad[idx_max]:
            delta = min(h[idx_min], 1 - h[idx_max], 0.01)
            if delta > 1e-6:
                h[idx_min] -= delta
                h[idx_max] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

            # Gradient-based random search
            i = np.random.choice(n_points)

                delta = min(h[i], 1 - h[j], 0.005)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5

def heuristic_local_search(h0, max_iter=1000):
    """Apply heuristic-based local search with multiple strategies"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Strategy 1: Move toward decreasing the maximum correlation
    for _ in range(max_iter // 2):
        # Compute current correlation peak
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        # Try to decrease that correlation by redistributing mass
        # In the lag position where correlation is high, try moving mass
        for i in range(n_points):
            best_improvement = 0
            best_j = i
            
            # Try moving mass from i to various j locations
            for j in range(n_points):
                if abs(i - j) == max_lag:
                    # Create a small mass transfer
                    delta = min(h[i], 1 - h[j], 0.005)
                    if delta > 1e-6:
                        h_test = h.copy()
                        h_test[i] -= delta
                        h_test[j] += delta
                        h_test = project_to_sum_box(h_test, TARGET_SUM)
                        c5_test = evaluate_c5(h_test)
                        improvement = best_c5 - c5_test
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_j = j
            
            # Apply the best move found
            if best_improvement > 0 and best_j != i:
                delta = min(h[i], 1 - h[best_j], 0.005)
                h[i] -= delta
                h[best_j] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Strategy 2: Fine-tune with local improvement
    for _ in range(max_iter // 2):
        # Perturb small random amounts
        i = np.random.choice(n_points)
        delta = np.random.uniform(-0.005, 0.005)
        if 0 <= h[i] + delta <= 1:
            # Test small change
            h_test = h.copy()
            h_test[i] += delta
            h_test = project_to_sum_box(h_test, TARGET_SUM)
            c5_test = evaluate_c5(h_test)
            if c5_test < best_c5:
                h = h_test
                best_c5 = c5_test
                best_h = h.copy()
    
    return best_h, best_c5
