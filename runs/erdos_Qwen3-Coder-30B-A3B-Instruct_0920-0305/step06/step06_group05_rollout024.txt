import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from collections import deque
import cvxpy as cp

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):
    """Perform a local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def gradient_based_move(h, max_iterations=50):
    """Perform gradient-based move to reduce C5"""
    h = h.copy()
    for _ in range(max_iterations):
        # Compute gradients for each position
        grad = np.zeros(n_points)
        corr = compute_corr_fft(h)
        worst_lag = np.argmax(corr)
        
        # Compute gradient for each position
        for i in range(n_points):
            if i + worst_lag < n_points:
                grad[i] += h[i + worst_lag]
            if i - worst_lag >= 0:
                grad[i] += h[i - worst_lag]
        
        # Find best move
        idx_min = np.argmin(grad)
        idx_max = np.argmax(grad)
        
        if idx_min == idx_max:
            break
            
        # Move small amount
        delta = min(h[idx_min], 1 - h[idx_max], 0.01)
        if delta <= 0:
            break
            
        h[idx_min] -= delta
        h[idx_max] += delta
        
        # Project to constraints
        h = project_to_sum_box(h, TARGET_SUM)
        
        # Check convergence
        if evaluate_c5(h) < 0.379:
            break
            
    return h

def bit_flip(h, i, j):
    """Flip bits at positions i and j, preserving sum"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def get_binary_h(h):
    """Convert to binary vector with exact weight"""
    h_binary = np.zeros(n_points)
    sorted_indices = np.argsort(h)[::-1]
    for i in range(int(TARGET_SUM)):
        h_binary[sorted_indices[i]] = 1
    return h_binary

def compute_autocorr(h):
    """Compute autcorrelation directly for binary"""
    L = 2*n_points - 1
    H = np.fft.fft(h, L)
    R = np.fft.ifft(H * H).real
    return R[n_points-1:]

def get_worst_lags(A):
    """Get indices with largest autocorrelations"""
    return np.argsort(A)[::-1]

def simulate_annealing():
    """Simulated Annealing for binary solutions"""
    # Temperature management
    T0 = 0.02
    T = T0
    alpha = 0.995
    cooling_interval = n_points
    T_min = 1e-5
    
    # State management
    current_h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    current_h = project_to_sum_box(current_h, TARGET_SUM)
    current_h = get_binary_h(current_h)
    
    best_h = current_h.copy()
    best_C5 = evaluate_c5(best_h)
    
    elapsed_time = 0
    start_time = time.time()
    
    # Elite pool
    elite_pool = deque(maxlen=5)
    
    move_count = 0
    while T > T_min and elapsed_time < 400:
        # Perform one move (random swap or block move)
        if np.random.random() < 0.15:
            # Block move
            block_length = np.random.choice([2, 3, 4])
            start = np.random.randint(0, n_points - block_length + 1)
            old_vals = current_h[start:start+block_length]
            flipped = 1 - old_vals
            current_h[start:start+block_length] = flipped
            
            # Fix wrong weight
            ones_count = np.sum(current_h)
            if ones_count > TARGET_SUM:
                # Remove extra 1s
                zero_positions = np.where(current_h == 0)[0]
                for i in range(int(ones_count - TARGET_SUM)):
                    idx = np.random.choice(zero_positions)
                    current_h[idx] = 1
                    zero_positions = np.delete(zero_positions, np.where(zero_positions == idx)[0])
            elif ones_count < TARGET_SUM:
                # Add extra 1s
                one_positions = np.where(current_h == 1)[0]
                for i in range(int(TARGET_SUM - ones_count)):
                    idx = np.random.choice(one_positions)
                    current_h[idx] = 0
                    one_positions = np.delete(one_positions, np.where(one_positions == idx)[0])
                    
        else:
            # Random swap
            one_positions = np.where(current_h == 1)[0]
            zero_positions = np.where(current_h == 0)[0]
            if len(one_positions) > 0 and len(zero_positions) > 0:
                i = np.random.choice(one_positions)
                j = np.random.choice(zero_positions)
                current_h[i], current_h[j] = current_h[j], current_h[i]
        
        # Evaluate new C5
        new_C5 = evaluate_c5(current_h)
        
        # Accept or reject
        if new_C5 <= best_C5:
            best_h = current_h.copy()
            best_C5 = new_C5
        elif np.random.random() < np.exp(-(new_C5 - best_C5)/T):
            pass  # Accept worse solution
        else:
            # Restore previous solution
            pass
            
        move_count += 1
        if move_count % cooling_interval == 0:
            T *= alpha
            
        elapsed_time = time.time() - start_time
        if elapsed_time >= 400:
            break
            
        # Store best in elite pool
        if len(elite_pool) < 5 or new_C5 < elite_pool[-1][0]:
            elite_pool.append((new_C5, current_h.copy()))
            elite_pool = deque(sorted(elite_pool))
            
    return best_h, best_C5

def hill_climb(h):
    """Local search: exhaustive swaps"""
    best_h = h.copy()
    best_C5 = evaluate_c5(h)
    improved = True
    
    while improved:
        improved = False
        for i in np.where(best_h == 1)[0]:
            for j in np.where(best_h == 0)[0]:
                h_new = bit_flip(best_h, i, j)
                C5_new = evaluate_c5(h_new)
                if C5_new < best_C5:
                    best_h = h_new
                    best_C5 = C5_new
                    improved = True
                    break
            if improved:
                break
                
    return best_h, best_C5

        critical_indices = set()

        k = worst_lags[0]
        for i in range(n_points):
            if i + k < n_points:
                if h_binary[i] == 1 and h_binary[i+k] == 1:
                    critical_indices.add(i)
                if h_binary[i] == 0 and h_binary[i+k] == 0:
                    critical_indices.add(i)
                    
        if len(critical_indices) > 20:
            # Use heuristic to select most influential
            influences = []
            for i in critical_indices:
                if i + k < n_points:
                    influ = h_binary[i] * (1 - h_binary[i+k])
                    influences.append((influ, i))
            influences.sort(reverse=True)
            critical_indices = set([x[1] for x in influences[:20]])
            
        # Apply MILP refinement
        v = cp.Variable(len(critical_indices), boolean=True)
        t = cp.Variable()
        
        prob = cp.Problem(cp.Minimize(t), [
            cp.sum(v) <= TARGET_SUM, 
            cp.sum(1 - v) <= n_points - TARGET_SUM,
            t >= 0
        ])
        
        # Add lag constraints (this is a simplification)
        # Mostly approximate for this version
        if prob.solve(solver=cp.CBC, verbose=False) == cp.OPTIMAL:
            # Use lambda to select subset of the optimized solution
            selected = v.value >= 0.5
            for i, idx in enumerate(list(critical_indices)):
                if idx < n_points:
                    h[idx] = float(selected[i])
        
        # Re-evaluate after fixes
        h = project_to_sum_box(h, TARGET_SUM)
        C5_fixed = evaluate_c5(h)
        if C5_fixed < C5_val:
            h = get_binary_h(h)
            return h, C5_fixed
            
    except:
        pass
    return h, C5_val
