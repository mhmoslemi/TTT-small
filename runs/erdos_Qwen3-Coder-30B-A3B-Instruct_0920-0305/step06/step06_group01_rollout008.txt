import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def perturb_solution(h, strength=0.03):
    """Create a perturbed version of the solution"""
    # Add noise to a random subset of points
    indices = np.random.choice(n_points, size=max(1, int(n_points * 0.2)), replace=False)
    h_perturbed = h.copy()
    for i in indices:
        delta = np.random.uniform(-strength, strength)
        h_perturbed[i] = np.clip(h_perturbed[i] + delta, 0, 1)
    return project_to_sum_box(h_perturbed, TARGET_SUM)

def guided_refinement(h, max_iter=200):
    """Refine solution using scipy.optimize with local search"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def simple_pattern_construction():
    """Create a simple pattern-based initial solution"""
    h = np.zeros(n_points)
    # Create a pattern similar to previously optimized solutions
    # Place step function with two levels
    mid = n_points // 2
    h[:mid] = 1.0
    h[mid:] = 0.0
    
    # Slightly adjust to meet constraint
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def optimized_pattern_construction():
    """Create a more sophisticated pattern-based construction"""
    h = np.zeros(n_points)
    # Create an approximately optimal piecewise constant pattern
    # This mimics the expected structure of an optimal solution
    
    # One possible pattern: local maxima in the middle with
    # decay towards edges
    for i in range(n_points):
        x = i * dx
        # Start with smooth transition
        if x < 0.5:
            h[i] = 1.0
        elif x < 1.5:
            h[i] = 1.0 - 2 * (x - 0.5)  # Linear decay
        else:
            h[i] = 0.0
    
    # Normalize to correct sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up target sum
    TARGET_SUM = n_points / 2
    
    # Try several starting configurations
    methods = [
        ("simple_pattern", simple_pattern_construction()),
        ("optimized_pattern", optimized_pattern_construction()),
        ("random", np.random.uniform(0, 1, n_points)),
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    for method_name, h_start in methods:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Ensure sum constraint
        h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Local optimization
        h_refined = guided_refinement(h_start.copy())
        c5_refined = evaluate_c5(h_refined)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            continue
            
    # Fine-tune final solution with adaptive strategy
    while time.time() - start_time < budget_s - 5:
        # Random restart with better solution if available
        if np.random.rand() < 0.5 and best_h is not None:
            h_new = perturb_solution(best_h, strength=0.05)
        else:
            # Try another pattern
            h_new = simple_pattern_construction()
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
        # Optimize
        h_opt = guided_refinement(h_new)
        c5_opt = evaluate_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
        
        # Periodic improvement
        if np.random.rand() < 0.1:
            # Try to improve further with some random moves
            for _ in range(10):
                h_rand = perturb_solution(best_h, strength=0.01)
                h_rand_opt = guided_refinement(h_rand)
                c5_rand_opt = evaluate_c5(h_rand_opt)
                if c5_rand_opt < best_c5:
                    best_h = h_rand_opt
                    best_c5 = c5_rand_opt
    
    # Return the final solution
    if best_h is None:
        # fallback to simple construction
        h_final = simple_pattern_construction()
        best_h = project_to_sum_box(h_final, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
