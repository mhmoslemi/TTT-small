import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_soft_minimum(h, alpha):
    """Evaluate the soft minimum of auto-correlations with parameter alpha"""
    S = correlate(h, h, 'full')
    S = S[len(h)-1:]
    # Compute log-sum-exp
    max_val = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * S - max_val)
    log_sum_exp = max_val + np.log(np.sum(exp_vals))
    return -log_sum_exp / alpha

def compute_gradient(h, alpha):
    """Compute gradient of the soft minimum with respect to h"""
    # Compute S_k(h) = sum h_i * h_{i+k}
    S = correlate(h, h, 'full')
    S = S[len(h)-1:]
    
    # Compute weights w_k = exp(-alpha * S_k) / sum exp(-alpha * S_l)
    max_val = np.max(-alpha * S)
    exp_vals = np.exp(-alpha * S - max_val)
    w = exp_vals / np.sum(exp_vals)
    
    # Symmetrize weights for symmetric convolution
    W = np.zeros(2 * len(h) - 1)
    W[len(h)-1:] = w
    W[:len(h)-1] = w[:0:-1]
    
    # Compute gradient g_j = sum_k w_k (h_{j+k} + h_{j-k})
    conv = np.convolve(h, W, mode='full')
    g = conv[len(h)-1 : len(h)-1 + len(h)]
    
    # Add contribution from lag 0
    g += w[len(h)-1] * h
    
    return g

    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02

        iteration += 1
        
        # Compute the gradient
        g = compute_gradient(best_h, alpha)
        
        # Compute candidate
        h_candidate = best_h + step * g

        alpha = min(alpha * alpha_inc, alpha_max)
        
        # Random restart if stagnated

            best_c5 = evaluate_c5(h_start)
            stagnation_counter = 0
            step = 0.02
            alpha = 0.5

    if time.time() - start_time < budget_s - 10:
        # Use SLSQP for local refinement
        bounds = [(0, 1) for _ in range(n_points)]
        
        def obj_fn(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                obj_fn,
                best_h,
                method='SLSQP',

                c5_final = evaluate_c5(h_final)

                    best_c5 = c5_final

            pass

    return (best_h.tolist(), float(best_c5), n_points)
