import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # Fine grid for accurate FFT correlation
K = 12             # Number of harmonics (Fourier coefficients)
EPS = 1e-12        # Small tolerance for numerical operations


def project_to_sum_box(h, target_sum):
    """Water-filling projection to enforce sum = target_sum, box constraints [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < EPS:
        return h
    # Use binary search on lambda to adjust
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        h_trial = np.clip(h + mid, 0, 1)
        sum_trial = np.sum(h_trial)
        if abs(sum_trial - target_sum) < EPS:
            return h_trial
        elif sum_trial > target_sum:
            high = mid
        else:
            low = mid
    h_proj = np.clip(h + low, 0, 1)
    return h_proj


def fourier_to_h(theta, N):
    """Map Fourier coefficients to h vector"""
    # Build trigonometric series with given coefficients
    x = np.linspace(0, 2, N_FINE)
    h_raw = 0.5 + np.sum(
        [theta[2 * k] * np.cos(np.pi * (k + 1) * x) + 
         theta[2 * k + 1] * np.sin(np.pi * (k + 1) * x)
         for k in range(K)], axis=0
    )
    # Downsample to match N
    step = N_FINE // N
    h_downsampled = h_raw[::step]
    return project_to_sum_box(h_downsampled, TARGET_SUM)


def evaluate_c5(h):
    """Evaluate C5 using FFT, equivalent to the referenced code"""
    n = len(h)
    dx = 2.0 / n
    # Use full correlation (inverse of convolution with reversed h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)


def objective(theta, N):
    """Objective function to be minimised"""
    h = fourier_to_h(theta, N)
    return evaluate_c5(h)


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize best solution
    best_c5 = float('inf')
    best_h = None

    # For now try coarse + fine resolutions
    N_list = [120, 200]

    for N in N_list:
        global TARGET_SUM
        TARGET_SUM = N / 2.0

        # Define search bounds for theta (K*2 coefficients)
        bounds = [(-0.5, 0.5)] * (2 * K)

        def optimized_objective(theta):
            return objective(theta, N)

        # Phase A: Deep global search
        # Allow up to 5000 evaluations as a proxy for time budget
        max_evals = 5000 if N == 120 else 2500  # Use fewer for N=200

        result = differential_evolution(
            optimized_objective,
            bounds,
            maxiter=500,
            popsize=50,
            mutation=(0.5, 1.0),
            recombination=0.9,
            seed=seed,
            callback=lambda x, convergence: None,
            strategy='best1bin',
            tol=1e-8
        )
        theta_best = result.x
        h = fourier_to_h(theta_best, N)
        c5 = evaluate_c5(h)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Phase B: Refinement around best solution
        # Use tighter options for 100 evals on the best point
        def refined_objective(theta):
            return objective(theta, N)

        # Run 5 independent refinements from around current best
        for trial in range(5):
            # Perturb best solution slightly
            init_theta = theta_best + np.random.uniform(-0.02, 0.02, 2 * K)
            try:
                result_ref = differential_evolution(
                    refined_objective,
                    bounds,
                    maxiter=100,
                    popsize=30,
                    mutation=(0.5, 0.8),
                    recombination=0.7,
                    seed=seed + 100 + trial,
                    strategy='best1bin',
                    x0=init_theta
                )
                theta_ref = result_ref.x
                h_ref = fourier_to_h(theta_ref, N)
                c5_ref = evaluate_c5(h_ref)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
            except:
                pass

        if (time.time() - start_time) > 0.9 * budget_s:
            break

    # Final polishing using local gradient steps
    def smooth_grad_step(h, alpha=10.0, step_size=0.01, max_iter=100):
        n = len(h)
        dx = 2.0 / n
        h_mod = h.copy()
        for _ in range(max_iter):
            # Compute correlation with 1-h
            corr = correlate(h_mod, 1 - h_mod, mode='full')
            # Flatten it for easier indexing
            corr_vals = corr  # len = 2*n - 1

            # Compute weights based on smoothed minimum
            min_corr = np.min(corr_vals)
            weights = np.exp(-alpha * (corr_vals - min_corr))
            weights /= np.sum(weights)

            # Compute derivative (approx axis-aligned gradient)
            # - For each i, d/dh_i = - \sum_{\ell} w_{\ell} ( h_{i+\ell} + h_{i-\ell} )
            gradient = np.zeros_like(h_mod)
            for i in range(n):
                for l in range(n):
                    if i + l < n:
                        gradient[i] -= weights[l] * h_mod[i + l]
                    if i - l >= 0:
                        gradient[i] -= weights[l] * h_mod[i - l]
            # Update
            h_new = h_mod - step_size * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            h_mod = h_new
            if evaluate_c5(h_mod) < best_c5:
                best_c5 = evaluate_c5(h_mod)
                best_h = h_mod.copy()
            else:
                break  # No improvement
        return h_mod

    # Final gradient polishing pass
    polished_h = smooth_grad_step(best_h)

    # Convert to binary (try). This is extra polish — lower chance of helping.
    # But compute candidate for testing:
    h_binary = np.zeros_like(best_h)
    indices = np.argsort(best_h)[::-1]    # indices from highest to lowest
    top = len(best_h) // 2
    h_binary[indices[:top]] = 1.0
    h_binary = project_to_sum_box(h_binary, TARGET_SUM)
    c5_binary = evaluate_c5(h_binary)
    if c5_binary < best_c5:
        best_c5 = c5_binary
        polished_h = h_binary

    return (polished_h.tolist(), float(best_c5), len(polished_h))
