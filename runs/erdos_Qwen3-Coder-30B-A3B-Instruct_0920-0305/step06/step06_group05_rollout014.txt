import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

    import cvxpy as cp

    has_cvxpy = False

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def fast_evaluate_c5_binary(h):
    """Exact evaluation for binary vectors using integer ops"""
    h = np.array(h, dtype=int)
    L = 2 * len(h) - 1
    H = np.fft.fft(h, L)
    R = np.fft.ifft(H * H).real
    A_k = R[len(h)-1:]
    return dx * (TARGET_SUM - np.min(A_k))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_float(h_binary):
    """Convert binary vector to float (0 or 1)"""
    return np.array(h_binary, dtype=float)

def float_to_binary(h_float, threshold=0.5):
    """Convert float vector to binary"""
    return (h_float > threshold).astype(int)

def neighbor_swap(h):
    """Generate one neighbor by swapping a 1 and a 0"""
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h
    i = np.random.choice(ones)
    j = np.random.choice(zeros)
    h[i], h[j] = h[j], h[i]
    return h

def neighbor_block_flip(h, p_block=0.15):
    """Generate one neighbor by flipping a block"""
    if np.random.random() > p_block:
        return neighbor_swap(h)
    
    h = h.copy()
    max_block_size = 4
    block_size = np.random.choice(range(2, max_block_size+1))
    start_pos = np.random.randint(0, len(h) - block_size + 1)
    
    # Flip the block
    h[start_pos:start_pos+block_size] = 1 - h[start_pos:start_pos+block_size]
    
    # Repair weight if necessary
    ones_count = np.sum(h)
    if ones_count != TARGET_SUM:
        # Correct by swapping bits
        excess = int(ones_count - TARGET_SUM)
        if excess > 0:
            # Need to flip some 1s to 0s
            flip_indices = np.random.choice(np.where(h == 1)[0], size=excess, replace=False)
            h[flip_indices] = 0
        else:
            # Need to flip some 0s to 1s
            flip_indices = np.random.choice(np.where(h == 0)[0], size=-excess, replace=False)
            h[flip_indices] = 1
    
    return h

def compute_autocorr_updates(h, i, j):
    """Compute the change in autocorrelation if we swap bits at i and j"""
    # h is an integer array
    changes = []
    for k in range(len(h)):
        if i + k < len(h):
            delta = h[i] * h[i+k] - h[j] * h[j+k] if k != 0 else 0
            changes.append(delta)
        if j + k < len(h):
            delta = h[j] * h[j+k] - h[i] * h[i+k] if k != 0 else 0
            changes.append(delta)
    return changes

def local_improve(h):
    """Hill climbing on binary vectors"""
    h = h.copy()
    improved = True
    while improved:
        improved = False
        best_incr = 0
        best_i, best_j = None, None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                val = fast_evaluate_c5_binary(h)
                # Temporarily swap
                h[i], h[j] = h[j], h[i]
                val_new = fast_evaluate_c5_binary(h)
                incr = val_new - val
                if incr < best_incr:
                    best_incr = incr
                    best_i, best_j = i, j
                # Swap back
                h[i], h[j] = h[j], h[i]
                
        if best_i is not None:
            h[best_i], h[best_j] = h[best_j], h[best_i]
            improved = True
    return h

def run_milp_refinement(h, time_remaining):
    """Use cvxpy to refine a small subset of variables"""
    if not has_cvxpy or time_remaining < 10:
        return h
    
    h_bin = float_to_binary(h)
    
    # Find worst lag
    corr = compute_corr_fft(h_bin)
    k_star = np.argmin(corr[len(h)//2:]) + len(h)//2
    
    # Collect indices involved in worst lag
    S = set()
    for i in range(len(h_bin)):
        if i + k_star < len(h_bin) and (h_bin[i] == 1 or h_bin[i + k_star] == 1):
            S.add(i)
        if i + k_star < len(h_bin) and (h_bin[i] == 0 or h_bin[i + k_star] == 0):
            S.add(i)
    
    if len(S) > 20:
        # Use a scoring heuristic to select top bits
        scores = []
        for i in S:
            score = (h_bin[i] * (1 - h_bin[i + k_star]) + (1 - h_bin[i]) * h_bin[i + k_star])
            scores.append((score, i))
        scores.sort(reverse=True)
        selected = [i for _, i in scores[:20]]
        S = set(selected)
    
    S = sorted(list(S))
    if len(S) == 0:
        return h
    
    # Create the MILP
    N = len(S)
    z = cp.Variable(N, boolean=True)
    # Fixed values
    fixed = [h_bin[i] for i in S]
    # Constraint: sum of fixed bits + sum of free bits = TARGET_SUM
    count = np.sum(fixed) + cp.sum(z) - sum(fixed) 
    # Objective
    t = cp.Variable()
    
    # Create and add constraints
    constraints = [cp.sum(z) - sum(fixed) + np.sum(fixed) == TARGET_SUM,
                   t >= 0]
    for k in range(len(h_bin)):
        if k == 0: continue
        if k not in S and k + k_star not in S:
            continue # Don't add constraints for isolated pairs
        # For each lag, add the constraint
        y_ij = []
        for i, idx in enumerate(S):
            # Check if i+k_star is in S
            idx_k = idx + k_star
            if idx_k < len(h_bin) and idx_k in S:
                # Add constrs, different depending on if (idx,k_star) in S
                y = cp.Variable(boolean=True)
                y_ij.append((i, y)
                constraints.append(y <= z[i])
                constraints.append(y <= h_bin[idx_k] if idx_k in S else z[S.index(idx_k)] <= 1)
                constraints.append(y >= z[i] + h_bin[idx_k] - 1 if idx_k in S else y >= z[i] + z[S.index(idx_k)] - 1)
                constraints.append(t >= TARGET_SUM - cp.sum(y_ij)) # THIS LINE NEEDS FIXING
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, seconds=time_remaining*0.1)
        if prob.status in ["optimal", "feasible"]:
            sol = [z.value[i] if z.value[i] is not None else 0 for i in range(N)]
            new_h = h.copy()
            for i, j in enumerate(S):
                new_h[j] = sol[i]
            # Re-project to exact weight
            ones_count = np.sum(new_h)
            if ones_count != TARGET_SUM:
                diff = int(ones_count - TARGET_SUM)
                if diff > 0:
                    # Flip some 1s to 0s
                    flip_to_zero = np.random.choice(np.where(new_h == 1)[0], size=diff, replace=False)
                    new_h[flip_to_zero] = 0
                else:
                    # Flip some 0s to 1s
                    flip_to_ones = np.random.choice(np.where(new_h == 0)[0], size=-diff, replace=False)
                    new_h[flip_to_ones] = 1
            return new_h 
    except:
        pass
    return h
