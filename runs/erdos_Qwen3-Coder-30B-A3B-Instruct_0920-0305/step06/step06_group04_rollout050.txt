import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def smooth_max_objective(h, alpha=1.0):
    """Smooth approximation of max_k (1 - f_k) where f_k = sum h_i h_{i+k}"""
    n = len(h)
    # Compute correlation f_k = sum h_i h_{i+k} for all k
    forward_cross = np.convolve(h, h, mode='full') # auto-correlation
    # Extract correlation for negative lags (explored via mode full) but we keep only matching size
    # But we want f_k for k = 0 to n-1, where f_k = sum_{i=0}^{n-1-k} h_i h_{i+k} 
    # which is e.g., for k=0: sum_i h_i^2, k=1: sum_i h_i h_{i+1}
    # Since correlation length is 2n-1 and we focus on k=0,..n-1, 
    # we extract appropriate correlation values  
    corr_values = forward_cross[n-1:] #  f_k for k=0,...n-1
    # We want max(1 - corr_values) -- this was the previous version
    # But in optimization we want minimize (sum_k (1-f_k)) = minimize sum_k f_k, or to maximize min f_k
    # Smoothed version of max(-f_k)  is -logsumexp(-alpha * f_k)
    # For α=0, logsumexp(0) = 0. As α → ∞, logsumexp(-alpha * f_k) → max(-f_k)  
    logexp_sum = np.log(np.sum(np.exp(-alpha * corr_values))) if np.sum(np.exp(-alpha * corr_values)) > 0 else 0
    return -logexp_sum

def smooth_gradient(h, alpha=1.0):
    """Compute gradient of smooth max objective"""
    n = len(h)
    # Forward correlation for auto-correlation term due to h * h
    forward_cross = np.convolve(h, h, mode='full') 
    corr_values = forward_cross[n-1:]  # f_k = sum h_i h_{i+k} for k = 0 to n-1
    
    # Compute weights: w_k = exp(-alpha * f_k) / sum(exp(-alpha * f_k))
    exp_terms = np.exp(-alpha * corr_values)
    exp_sum = np.sum(exp_terms)
    weights = exp_terms / exp_sum if exp_sum > 0 else np.ones_like(exp_terms) / len(exp_terms)
    
    # Compute gradient
    grad = np.zeros_like(h)
    # For each lag k = 0 to n-1, we have d/dh_i (h_i h_{i+k}) = h_{i+k} + h_{i-k}
    for k in range(n):
        if k == 0:
            # derivative of h_i^2 is 2 h_i; but since we doing -alpha*sum(h_i^2) gradient
            # we have -alpha * 2 * h_i for shift k=0 (using the same square terms)
            # GRADIENT part: -sum_k (w_k * 2 h_i)
            grad += -2 * weights[k] * h        
        else:
            # k>0 pairs of i and i+k appear  
            # we sum d/dh_i (h_i h_{i+k}) = h_{i+k} for all i  
            # and d/dh_{i+k} (h_i h_{i+k}) = h_i for all i 
            # So gradient for h_i = sum_{i': i'=i+k} w_k * h_{i'} + sum_{i': i'+k=i} w_k * h_{i'}
            # which means: h_{i+k} + h_{i-k} have weights                
            # For each i, add weight contribution to sum of i+k and i-k if within bounds  
            if k < n:
                # Add to grad[i]   weights[k] * h[i+k] 
                grad[k:] += weights[k] * h[:-k] 
                # Add to grad[i-k] weights[k] * h[i]  
                grad[:-k] += weights[k] * h[k:] 
    # As the actual gradient to use: 
    # -sum_k w_k derivative: so multiply by -1 and we are almost good
    # BUT from matching indices and sum over all lag k, total entries
    # It is simplest to associate gradient contribution  
    # For shift k, we reach each h_i with influence from h_{i-k} and h_{i+k}  
    # So in total,    grad[i] += - weights[k] * (h[i+k] + h[i-k]) 
    # for all valid indices 0< (i-k) <n and 0< (i+k) <n:
    # We compute all with slicing
    
    # Direct compute of the full new approach, to find |h_i h_{i+k}| impact on gradient
    # For each k in 0..n-1, each h_i shows up in sum, 
    # so for each h_i gets contribution weights[k] * (h_{i+k} + h_{i-k}) 
    # (i.e. pairs (i,i+k) and (i-k,i))  
    from functools import partial
    # Loop method, otherwise vectorized using convolution
    from scipy.signal import correlate
    # We want [i] gradient = sum_{k} weights[k] *(h_{i+k} + h_{i-k})
    # Shift right and left contributions + their weights.
    # Implement using for loops for clarity: 
    grad = np.zeros_like(h)
    for k in range(1,n):
        w_k = weights[k]        
        # Shift h by +k and +k-and-get-left-part: this adds it to right-shift grad values
        if k < len(h):
            # left part correct index >= 0 (so when i+k < n)            
            shift_right_indices = slice(k, len(h)) 
            shift_left_indices = slice(0, len(h)-k)            
            grad[shift_right_indices] += w_k * h[shift_left_indices]  # h[i-k] contributes to grad[i]  
            grad[shift_left_indices] += w_k * h[shift_right_indices] # h[i+k] contributes to grad[i] 
        # Else for k >=n would not contribute direct pairs and we have zero-ts dense case
    # But some overlapping as well known in correlate via convolution approach  
    grad *= -1
    return grad

def refine_step(h, alpha=1.0, step_size=0.01):
    """Perform one projected gradient ascent step""" 
    new_h = h.copy()
    grad = smooth_gradient(new_h, alpha)
    new_h -= step_size * grad
    # Project to feasible region [0,1] with sum=target_sum
    new_h = project_to_sum_box(new_h, TARGET_SUM)
    return new_h

def local_refinement(h_start, max_iter=1000, tol=1e-12):
    """Run gradient-based refinement with significant smoothing"""
    h = h_start.copy()
    # We already care about max_k C_k which is also max_k (1 - overlap_k)
    # So minimize max_k (1 - overlap_k) = 1 - min_k overlap_k
    # This targets maximizing min overlap.
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    alpha_schedule = [0.1, 0.5, 1.0, 2.0, 5.0] # sequence of smooth sigmas
    for alpha in alpha_schedule:
        print(f"Refining with α = {alpha}, progress...")
        for i in range(max_iter):
            h_new = refine_step(h, alpha=alpha, step_size=0.001)  
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            if abs(c5_new - best_c5) < tol:
                break
            h = h_new
    return best_h, best_c5

def almost_binary_h(n_points, target_sum):
    """Create almost binary h to guide optimization"""
    h = np.zeros(n_points)
    # Place 50% of it as 1s (with approximation for uniform distribution)
    ones_count = int(target_sum)
    ones_idx = np.random.choice(n_points, ones_count, replace=False)
    h[ones_idx] = 1.0
    
    # Project to ensure constraint
    h = project_to_sum_box(h, target_sum)
    return h
