import numpy as np
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed):
    """Initial binary vector with exact weight"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full correlation O_s = sum_i h_i * h_{i+s}"""
    return np.correlate(x, x, mode='full')

def update_overlap_move(ov, i, j, ones, zeros):
    """Update overlap array in O(w) when swapping x[i] and x[j] using O(w) operations."""
    ov_new = ov.copy()
    ones_new = set(ones)
    zeros_new = set(zeros)
    
    # Remove contribution from i and add from j
    # For each k such that (i,k) contributes to overlap:
    for k in ones:
        if k == i:
            continue
        s = k - i
        ov_new[s + n_points - 1] -= 1
    
    # Add contribution from j to all k such that (j,k) contributes
    for k in ones_new - {i}:
        s = k - j
        ov_new[s + n_points - 1] += 1
    
    # Update indices
    ones_new.discard(i)
    ones_new.add(j)
    zeros_new.discard(j)
    zeros_new.add(i)
    
    return ov_new, ones_new, zeros_new

def simulated_annealing(x0, ov0, ones0, zeros0, time_limit):
    """Run simulated annealing to increase minimum overlap"""
    T0 = 0.5
    cooling_rate = 0.9999
    min_T = 1e-4
    
    x = x0.copy()
    ov = ov0.copy()
    ones = ones0.copy()
    zeros = zeros0.copy()
    
    best_x = x.copy()
    best_ov = ov.copy()
    best_min = ov.min()
    
    cur_min = best_min
    T = T0
    t = 0
    
    start_time = time.time()
    while time.time() - start_time < time_limit:
        # Time management
        if t > 0 and t % 10000 == 0:
            elapsed = time.time() - start_time
            if elapsed > 0.95 * time_limit:
                break
            
        # Generate neighbor: swap a 1 and a 0
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Update overlap efficiently
        ov_new, ones_new, zeros_new = update_overlap_move(ov, i, j, ones, zeros)
        new_min = ov_new.min()
        
        # Acceptance probability
        delta = new_min - cur_min
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            x[i] = 0
            x[j] = 1
            ov = ov_new
            ones = ones_new
            zeros = zeros_new
            cur_min = new_min
            
            # Update best solution
            if new_min > best_min:
                best_x = x.copy()
                best_ov = ov.copy()
                best_min = new_min
        
        # Cooling schedule
        T *= cooling_rate
        if T < min_T:
            T = min_T
            
        t += 1
        
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Iteratively improve worst shift by swapping 1's/0's that raise it"""
    # Find the current worst shift
    worst_shift = np.argmin(ov) - (n_points - 1)
    
    # Generate sets of positions involved
    A = set()
    B = set()
    
    for i in ones:
        j = i + worst_shift
        if 0 <= j < n_points:
            if x[j] == 0:
                A.add(i)
    
    for j in zeros:
        i = j - worst_shift
        if 0 <= i < n_points:
            if x[i] == 1:
                B.add(j)
    
    # Sort A and B to avoid randomization issues
    A = sorted(A)
    B = sorted(B)
    
    # Try all pairs that could improve the worst shift
    improved = True
    while improved:
        improved = False
        for i in A:
            # Find j ∈ B such that swapping increases worst_shift overlap
            for j in B:
                # Check that this swap improves the overlap or doesn't hurt
                # Evaluate new overlap array after exchanging the 1 at i and 0 at j
                ov_test, _, _ = update_overlap_move(ov, i, j, ones, zeros)
                new_min = ov_test.min()
                if new_min > ov.min():
                    # Perform the swap
                    x[i] = 0
                    x[j] = 1
                    ones.discard(i)
                    ones.add(j)
                    zeros.discard(j)
                    zeros.add(i)
                    ov = ov_test
                    cur_min = new_min
                    improved = True
                    # Update the A and B sets since indices have changed
                    A = set(i2 for i2 in ones if (i2 + worst_shift) < n_points and x[i2 + worst_shift] == 0)
                    B = set(j2 for j2 in zeros if (j2 - worst_shift) >= 0 and x[j2 - worst_shift] == 1)
                    break
            if improved:
                break
    return x, ov

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Setup
    np.random.seed(seed)
    
    # Start with a good initialization
    x = initial_binary(seed)
    ov = full_overlap(x)
    ones = {i for i in range(n_points) if x[i] == 1}
    zeros = {i for i in range(n_points) if x[i] == 0}
    
    # Initial estimate
    cur_min = ov.min()
    best_min = cur_min
    best_x = x.copy()
    
    # Mini-basic SA loop
    max_iters = int(2e6)
    t = 0
    
    while t < max_iters:
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        # Do one swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Update efficiently
        ov_new, ones_new, zeros_new = update_overlap_move(ov, i, j, ones, zeros)
        new_min = ov_new.min()
        delta = new_min - cur_min
        
        if delta > 0 or np.random.rand() < np.exp(delta / 0.2):
            x[i] = 0
            x[j] = 1
            ov = ov_new
            ones = ones_new
            zeros = zeros_new
            cur_min = new_min
            
            if new_min > best_min:
                best_x = x.copy()
                best_min = new_min
        
        t += 1
    
    # Refine using worst-shift repair
    try:
        x, ov = worst_shift_refine(best_x, ov, ones, zeros)
        best_min = ov.min()
    except:
        pass
    
    # Final result
    c5 = dx * (TARGET_SUM - best_min)
    return (best_x.tolist(), float(c5), n_points)
