import numpy as np
import time
from cvxpy import *

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation"""
    n = len(h)
    dx = 2.0 / n
    # Reverse h to compute correlation properly
    h_rev = h[::-1]
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    N = 120
    dx = 2.0 / N
    target_sum = N / 2.0
    
    # Load initial solution
    h_initial = np.array(initial_h_values)
    # Interpolate to N points
    x_orig = np.linspace(0, 2, len(h_initial))
    x_new = np.linspace(0, 2, N)
    h_interp = np.interp(x_new, x_orig, h_initial)
    h_interp = np.clip(h_interp, 0, 1)
    # Project to correct sum
    h_interp = h_interp * target_sum / np.sum(h_interp)
    
    # Try MILP solver
    best_h = h_interp.copy()
    best_c5 = evaluate_c5(h_interp)
    
    try:
        # Use a weaker mixed-integer programming approach via CVXPY
        # Build variables
        h = Variable(N, boolean=True)
        y = {}
        for l in range(N):
            y[l] = Variable((N - l), boolean=True)
        lam = Variable(integer=True)
        
        # Constraints
        constraints = []
        constraints.append(sum(h) == target_sum)
        
        # Product constraints
        for l in range(N):
            for i in range(N - l):
                # y[l][i] <= h[i]
                constraints.append(y[l][i] <= h[i])
                # y[l][i] <= h[i + l]
                constraints.append(y[l][i] <= h[i + l])
                # y[l][i] >= h[i] + h[i + l] - 1
                constraints.append(y[l][i] >= h[i] + h[i + l] - 1)
        
        # Overlap constraints
        for l in range(N):
            constraints.append(sum(y[l]) >= lam)
            
        # Bounds on lambda
        constraints.append(lam <= target_sum)
        
        # Objective
        objective = Maximize(lam)
        
        # Formulate and solve problem
        prob = Problem(objective, constraints)
        
        # Set time limit to 500 seconds
        solve_time = min(500, budget_s * 0.8)
        prob.solve(solver=GUROBI, time_limit=solve_time, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = np.array([v.value for v in h])
            # Compute resulting C5
            c5 = 1.0 - 2.0 * lam.value / (N * dx) 
            if c5 < best_c5:
                best_h = h_sol
                best_c5 = c5
    except Exception:
        # Fallback to previous method if MILP fails or is not available
        pass
    
    # Local optimization refinement just to make sure
    def obj(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(N)]
    initial_h = best_h
    try:
        from scipy.optimize import minimize
        res = minimize(obj, initial_h, method='SLSQP', 
                       bounds=bounds, constraints={'type': 'eq', 'fun': constraint_eq},
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            c5_refined = evaluate_c5(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (best_h.tolist(), float(best_c5), N)
