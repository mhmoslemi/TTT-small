import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time
import math

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT-based correlation"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant')
    h_conj = np.pad(1-h, (0, n-1), mode='constant')
    corr = ifft(fft(h_padded) * np.conj(fft(h_conj))).real
    return np.max(corr) * (2.0 / n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmax_weights(corr, alpha):
    """Compute softmax weights for the correlation values"""
    corr_shifted = corr - np.max(corr)
    exp_corr = np.exp(alpha * corr_shifted)
    return exp_corr / np.sum(exp_corr)

def compute_gradient(h, weights):
    """Compute gradient using FFT-based convolution for efficiency"""
    n = len(h)
    padded_h = np.pad(h, (0, n-1), mode='constant')
    padded_weights = np.pad(weights, (0, n-1), mode='constant')
    conv_forward = ifft(fft(padded_h) * np.conj(fft(padded_weights))).real[:n]
    conv_backward = ifft(fft(padded_h[::-1]) * np.conj(fft(padded_weights[::-1]))).real[:n][::-1]
    return 1.0 - conv_forward - conv_backward

def line_search(h, grad, target_sum, alpha, max_iter=100):
    """Perform backtracking line search"""
    eta = 0.5
    eta_min = 1e-8
    h_current = h.copy()
    c5_current = evaluate_c5(h_current)
    
    for _ in range(max_iter):
        h_trial = project_to_sum_box(h_current - eta * grad, target_sum)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current:
            return h_trial, c5_trial, eta
        eta /= 2
        if eta < eta_min:
            break
    return h_current, c5_current, eta

def projected_gradient_descent(h0, target_sum, max_iter=5000, budget_time=None):
    """Projected gradient descent with softmax surrogate"""
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = 5.0
    alpha_max = 20.0
    eta = 0.5
    eta_max = 10.0
    converged = False
    no_improvement_count = 0
    no_improvement_limit = 200
    
    for i in range(max_iter):
        if budget_time and time.time() - start_time > budget_time:
            break
        corr = evaluate_c5(h) * np.ones(2 * len(h) - 1)  # Placeholder for full correlation
        corr_full = ifft(fft(np.pad(h, (0, len(h)-1))) * np.conj(fft(np.pad(1-h, (0, len(h)-1))))).real
        weights = softmax_weights(corr_full, alpha)
        grad = compute_gradient(h, weights)
        h_new, c5_new, eta = line_search(h, grad, target_sum, alpha)
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count > no_improvement_limit:
            break
        alpha = min(alpha_max, alpha * 1.01)
        eta = min(eta_max, eta * 1.02)
    return best_h, best_c5

def seed_interpolated(n_points):

    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interpolated = np.interp(x_new, x_orig, h0)
    h_interpolated = project_to_sum_box(h_interpolated, n_points/2)
    return h_interpolated

def seed_random_continuous(n_points, seed):
    """Generate random continuous seed"""
    np.random.seed(seed)
    h = np.random.rand(n_points)
    h = project_to_sum_box(h, n_points/2)
    return h

def seed_random_binary(n_points, seed):
    """Generate random binary seed with exactly half ones"""
    np.random.seed(seed)
    h = np.zeros(n_points)
    indices = np.random.choice(n_points, size=n_points//2, replace=False)
    h[indices] = 1
    return h

def seed_legendre(n_points):
    """Generate Legendre (quadratic residue) binary seed"""
    p = 0
    while p < 2 * n_points + 1:
        p = next_prime(p)
    seq = []
    for i in range(p):
        seq.append(1 if pow(i, (p-1)//2, p) == 1 else 0)
    seq = seq[:n_points]  # Adjust length
    count_ones = sum(seq)
    if count_ones < n_points // 2:
        # Add more ones randomly
        zeros = np.where(np.array(seq) == 0)[0]
        assert len(zeros) >= n_points // 2 - count_ones
        for i in np.random.choice(zeros, size=n_points//2 - count_ones, replace=False):
            seq[i] = 1
    elif count_ones > n_points // 2:
        # Remove excess ones randomly
        ones = np.where(np.array(seq) == 1)[0]
        assert len(ones) >= count_ones - n_points // 2
        for i in np.random.choice(ones, size=count_ones - n_points // 2, replace=False):
            seq[i] = 0
    return np.array(seq)

def next_prime(n):
    """Find next prime >= n"""
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(math.sqrt(x)) + 1):
            if x % i == 0:
                return False
        return True
    while not is_prime(n):
        n += 1
    return n

def binary_annealing(h_start, target_sum, budget_time):
    """Binary refinement via simulated annealing"""
    start_time = time.time()
    h = h_start.copy()
    c5_current = evaluate_c5(h)
    T = 0.02 * c5_current
    T_min = 1e-6
    decay = 0.9995
    max_sweeps = int(2000)
    sweep_count = 0
    
    while T > T_min and sweep_count < max_sweeps and time.time() - start_time < budget_time:
        h_trial = h.copy()
        i, j = np.random.randint(0, len(h), size=2)
        if h[i] == h[j]:
            continue
        h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
        h_trial = project_to_sum_box(h_trial, target_sum)  # Ensure sum constraint
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current or np.random.rand() < np.exp(-(c5_trial - c5_current) / T):
            h = h_trial
            c5_current = c5_trial
        T *= decay
        sweep_count += 1
    return h, c5_current

def upscale_and_refine(h_coarse, target_sum, fine_points, budget_time):
    """Upscale to a finer grid and refine on that grid"""
    start_time = time.time()
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, fine_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_to_sum_box(h_fine, fine_points/2)
    # With a coarse upfront evaluation, it is just the fine grid that is in effect for the
    # refined vector; so we simply optimize one more round
    h_fine, c5_fine = projected_gradient_descent(
        h_fine, fine_points/2, 
        max_iter=20, 
        budget_time=budget_time - (time.time() - start_time)
    )
    return h_fine, c5_fine

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Use a lower resolution for faster initial canvas
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Multi-start with different seeds
    seeds = [
        seed_interpolated(n_points),
        seed_random_continuous(n_points, seed + 1),
        seed_random_binary(n_points, seed + 2),
        seed_legendre(n_points)
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    time_per_seed = budget_s * 0.12  # 12% per seed
    for i, h_seed in enumerate(seeds):
        if time.time() - start_time > budget_s - 10:
            break
        h_opt, c5_opt = projected_gradient_descent(
            h_seed, target_sum, 
            max_iter=3000,
            budget_time=time_per_seed
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # If we still have room, do extra random starts
    if time.time() - start_time < budget_s - 10:
        time_per_restart = budget_s * 0.1
        for i in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            h_start = project_to_sum_box(best_h + np.random.normal(0, 0.01, n_points), target_sum)
            h_opt, c5_opt = projected_gradient_descent(
                h_start, target_sum,
                max_iter=1000,
                budget_time=time_per_restart
            )
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Binary refinement
    if time.time() - start_time < budget_s - 5:
        binary_seed = np.zeros(n_points)
        indices = np.argsort(best_h)[-n_points//2:]  # Take largest entries
        binary_seed[indices] = 1
        h_binary, c5_binary = binary_annealing(
            binary_seed, target_sum, budget_s * 0.35
        )
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary.copy()
    
    # Optional upscaling if a relatively low C5 is achieved
    if best_c5 <= 0.381 and time.time() - start_time < budget_s - 10:
        fine_points = 400
        h_upscaled, c5_upscaled = upscale_and_refine(
            best_h, target_sum, fine_points, budget_s * 0.05
        )
        if c5_upscaled < best_c5:
            best_c5 = c5_upscaled
            best_h = h_upscaled.copy()
    
    # Final polishing if time remains
    if time.time() - start_time < budget_s - 5:
        best_h = project_to_sum_box(best_h + np.random.normal(0, 0.001, n_points), target_sum)
        best_h, best_c5 = projected_gradient_descent(
            best_h, target_sum, max_iter=500, 
            budget_time=5.0
        )
    
    return (best_h.tolist(), float(best_c5), n_points)
