import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import ceil

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2
MAX_TIME = 600
SA_TIME = 0.7 * MAX_TIME
HC_TIME = 0.05 * MAX_TIME
MILP_TIME = 0.15 * MAX_TIME
SLSQP_TIME = 0.05 * MAX_TIME
T_MIN = 1e-5
ALPHA = 0.995
L = n_points
RESTART_INTERVAL = 0.1 * SA_TIME
CROSSOVER_INTERVAL = 0.05 * SA_TIME

def c5_from_corr(corr_vec):
    return dx * (TARGET_SUM - np.min(corr_vec[n_points-1:]))

def exact_c5(h):
    """Compute exact C5 using integer convolution"""
    h_rev = h[::-1]
    L_full = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L_full) * fft(1 - h, L_full)).real
    return c5_from_corr(corr)

def compute_corr_fast(h):
    """Fast FFT-based correlation"""
    h_rev = h[::-1]
    L_full = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L_full) * fft(1 - h, L_full)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto sum = target_sum and box [0,1]"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def binary_to_float(h):
    return h.astype(float)

def float_to_binary(x):
    return (x > 0.5).astype(int)

def swap_10(h, i, j):
    """Swap h[i] and h[j] for binary h"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def block_flip(h, s, length):
    """Flip a block of binary h"""
    h = h.copy()
    for i in range(s, min(s + length, len(h))):
        h[i] = 1 - h[i]
    return h

def crossover(parent1, parent2):
    """One-point crossover of binary vectors"""
    n = len(parent1)
    p = np.random.randint(1, n)
    child = np.concatenate([parent1[:p], parent2[p:]])
    
    # Repair weight
    ones = np.sum(child)
    while ones < TARGET_SUM:
        # Find a 0 and flip it to 1
        z = np.random.choice(np.where(child == 0)[0])
        child[z] = 1
        ones += 1
    while ones > TARGET_SUM:
        # Find a 1 and flip it to 0
        o = np.random.choice(np.where(child == 1)[0])
        child[o] = 0
        ones -= 1
    return child

def simulated_annealing(seed=42):
    """Run simulated annealing with restarts and crossover"""
    np.random.seed(seed)
    sa_start = time.time()
    
    # Initialize elite pool
    elite_pool = []
    elite_size = 5
    
    # Main loop
    t = 0
    current_h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    current_h = project_to_sum_box(current_h, TARGET_SUM).round().astype(int)
    current_c5 = exact_c5(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = 0.01  # Initial temp: choose small value to allow early exploration
    k = 0
    last_restart = sa_start
    last_crossover = sa_start
    
    while True:
        if time.time() - sa_start > SA_TIME - 5:
            break

        # Attempt restart
        if time.time() - last_restart > RESTART_INTERVAL:
            # Add to elite pool
            if len(elite_pool) < elite_size:
                elite_pool.append((current_h.copy(), current_c5))
                elite_pool.sort(key=lambda x: x[1])
            else:
                # Replace worst if current is better
                if current_c5 < elite_pool[-1][1]:
                    elite_pool[-1] = (current_h.copy(), current_c5)
                    elite_pool.sort(key=lambda x: x[1])
            
            # Choose restart type
            if len(elite_pool) > 0 and np.random.rand() < 0.5:
                # Restart from elite
                chosen = elite_pool[np.random.randint(0, len(elite_pool))]
                current_h = chosen[0].copy()
                current_c5 = chosen[1]
            else:
                # New random binary
                current_h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
                current_h = project_to_sum_box(current_h, TARGET_SUM).round().astype(int)
            
            current_c5 = exact_c5(current_h)
            T = 0.01
            last_restart = time.time()
        
        # Attempt crossover
        if time.time() - last_crossover > CROSSOVER_INTERVAL and len(elite_pool) >= 2:
            parent1 = elite_pool[np.random.randint(0, len(elite_pool))][0]
            parent2 = elite_pool[np.random.randint(0, len(elite_pool))][0]
            child = crossover(parent1, parent2)
            child_c5 = exact_c5(child)
            if child_c5 < current_c5:
                current_h = child
                current_c5 = child_c5
            last_crossover = time.time()

        # Basic swap move
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            new_h = swap_10(current_h, i, j)
            new_c5 = exact_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                # Accept with probability
                delta = new_c5 - current_c5
                if T > T_MIN and np.random.rand() < np.exp(-delta / T):
                    current_h = new_h
                    current_c5 = new_c5
        
        # Block move (15% probability)
        if np.random.rand() < 0.15 and len(ones) > 0 and len(zeros) > 0:
            length = np.random.choice([2, 3, 4])
            s = np.random.randint(0, n_points - length + 1)
            new_h = block_flip(current_h, s, length)
            
            # Repair weight
            new_h = project_to_sum_box(new_h, TARGET_SUM).round().astype(int)
            new_c5 = exact_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                delta = new_c5 - current_c5
                if T > T_MIN and np.random.rand() < np.exp(-delta / T):
                    current_h = new_h
                    current_c5 = new_c5
        
        # Cool down
        k += 1
        if k % n_points == 0:
            T *= ALPHA
        
        # Early termination
        if best_c5 <= 0.379:
            break
    
    return best_h, best_c5

def hill_climb(h_start):
    """Exhaustive hill climbing on binary vector"""
    h = h_start.copy()
    current_c5 = exact_c5(h)
    improved = True
    
    while improved:
        improved = False
        # Try all swaps
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                new_h = swap_10(h, i, j)
                new_c5 = exact_c5(new_h)
                if new_c5 < current_c5:
                    h = new_h
                    current_c5 = new_c5
                    improved = True
                    break
            if improved:
                break
        
        # If one pass failed, do full search on one iteration,
        # then compute FFT for best improvement
        if not improved:
            g = compute_corr_fast(h)
            worst_lag = np.argmax(g[n_points-1:])
            diff = g[n_points-1+worst_lag] - TARGET_SUM
            # Refine only when needed
            if diff > 0.1:  # Threshold to save time
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                best_delta = float('inf')
                best_i, best_j = None, None
                # For small sizes, do limited search
                for _ in range(1000):
                    i = np.random.choice(ones)
                    j = np.random.choice(zeros)
                    new_h = swap_10(h, i, j)
                    new_c5 = exact_c5(new_h)
                    if new_c5 < current_c5:
                        h = new_h
                        current_c5 = new_c5
                        improved = True
                        break
    
    return h, current_c5

def milp_refinement(h_start, time_remaining):
    """Run focused MILP refinement on critical indices"""
    h = h_start.copy()
    # Find the worst lag
    corr = compute_corr_fast(h)
    worst_lag = np.argmax(corr[n_points-1:])
    # Get involved indices
    involved = set()
    for i in range(len(h)):
        if i + worst_lag < len(h):
            involved.add(i)
            involved.add(i + worst_lag)
    involved = sorted(list(involved))
    
    if len(involved) > 20:
        # Pick most influential
        scores = []
        for i in involved:
            if i + worst_lag < len(h):
                scores.append((i, h[i] * (1 - h[i+worst_lag])))
        scores.sort(key=lambda x: x[1], reverse=True)
        involved = [x[0] for x in scores[:20]]
    
    # Define MILP problem
    try:
        z = cp.Variable(len(involved), boolean=True)
        t = cp.Variable()
        
        # Constraint: sum = TARGET_SUM
        # Fix other indices to current values (needed in objective)
        fixed_sum = 0
        fixed_z = []
        for i, idx in enumerate(involved):
            if h[idx] == 1:
                fixed_sum += 1
                fixed_z.append(1)
            else:
                fixed_z.append(0)
        
        # Objective: minimize t (upper bound on worst overlap)
        # Define a small constraint set for existing lags
        constraints = []
        for k in range(0, n_points):
            # Only consider lags that involve at least one of involved indices
            if k == 0:
                # k=0 edges are always added
                continue
            product_sum = 0
            for i in involved:
                if i + k < len(h):
                    # In general, the expression for A_k would be a sum over all
                    # pairs (i, i+k) where both are in the domain and both have value 1
                    # Here, we build a constraint for sum of if both elements are 1
                    # but we need to account for fixed values correctly.
                    # Add constraint term for lag k
                    if i in involved and (i+k) in involved:
                        idx_i = involved.index(i)
                        idx_j = involved.index(i+k)
                        # Create dummy variable for 'and' of z[i] and z[j]...
                        # PuTTING PRODUCTS to constraints manually.
                        pass
                    elif i in involved and (i+k) not in involved and h[i+k] == 1:
                        # Term z[idx] * 1 = z[idx]
                        product_sum += z[involved.index(i)]
                    elif i in involved and (i+k) not in involved and h[i+k] == 0:
                        # term z[idx] * 0 = 0
                        pass
                    elif i not in involved and (i+k) in involved:
                        # 1 * z[idx_j] = z[idx_j]
                        product_sum += z[involved.index(i+k)]
                    elif i not in involved and (i+k) not in involved:
                        if h[i] == 1 and h[i+k] == 1:
                            product_sum += 1
                            # else 0
            sum_constraint = (sum(z) + (TARGET_SUM - fixed_sum)) - product_sum <= t
            constraints.append(sum_constraint)
        
        # Additional constraint that is actually needed: weight
        constraints.append(cp.sum(z) + (TARGET_SUM - fixed_sum) == TARGET_SUM)
        
        # Create a single objective to maximize to saturate std formulation technique
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver=cp.CBC, time_limit=min(30, time_remaining * 0.1))
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
            new_z = z.value.round()
            # Update the h vector with new values
            for i, idx in enumerate(involved):
                h[idx] = new_z[i]
            h = project_to_sum_box(h, TARGET_SUM).round().astype(int)
            return h, exact_c5(h)
    except:
        pass
    return h_start, exact_c5(h_start)

def continuous_polish(h_start, time_remaining):
    """Perform continuous refinement with SLSQP"""
    h = h_start.copy()
    corr = compute_corr_fast(h)
    worst_lag = np.argmax(corr[n_points-1:])
    h_float = binary_to_float(h)
    # Prepare the gradient function
    def objective(x):
        # Ensure it is binary vector when passed to correlation step
        h_test = project_to_sum_box(x, TARGET_SUM)
        corr_vec = compute_corr_fast(h_test)
        return c5_from_corr(corr_vec)
    
    def constraint_eq(x):
        return np.sum(x) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h_float, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-7, 'gtol': 1e-7})
        if res.success:
            h_opt = res.x
            h_opt = project_to_sum_box(h_opt, TARGET_SUM)
            return float_to_binary(h_opt), exact_c5(float_to_binary(h_opt))
    except:
        pass
    return h_start, exact_c5(h_start)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Phase 1: Simulated Annealing
    sa_h, sa_c5 = simulated_annealing(seed)
    best_h = sa_h.copy()
    best_c5 = sa_c5
    
    # Phase 2: Hill Climbing
    if time.time() - start_time < 0.7 * budget_s:
        hc_h, hc_c5 = hill_climb(sa_h)
        if hc_c5 < best_c5:
            best_h = hc_h.copy()
            best_c5 = hc_c5
    
    # Phase 3: MILP Refinement
    if time.time() - start_time < 0.85 * budget_s:
        milp_h, milp_c5 = milp_refinement(best_h, budget_s - (time.time() - start_time))
        if milp_c5 < best_c5:
            best_h = milp_h.copy()
            best_c5 = milp_c5
    
    # Phase 4: Continuous Polish
    if time.time() - start_time < 0.95 * budget_s:
        slsqp_h, slsqp_c5 = continuous_polish(best_h, budget_s - (time.time() - start_time))
        if slsqp_c5 < best_c5:
            best_h = slsqp_h.copy()
            best_c5 = slsqp_c5

    return (best_h.tolist(), float(best_c5), n_points)
