def evaluate_erdos_solution(h_list):
    # Makes use of the provided function evaluate_erdos_solution
    # This is a logic placeholder for whenever the correct function is available
    # We are writing a coherent solution respecting the environment
    h = np.array(h_list)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target):
    """Projects x onto box constraints with fixed target sum."""
    low = -np.max(x) - 1.0
    high = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            high = lam
        else:
            low = lam
    y = np.clip(x + (low + high) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def full_corr(a, b):
    """Compute full correlation via FFT for speed."""
    n = a.size
    N = 2 * n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2 * n - 1]

def evaluate_max_corr(h):
    """Compute C5 by max of correlation."""
    corr = full_corr(h, 1 - h)
    return corr.max() * (2.0 / len(h))

def gradient_max_soft(h, alpha):
    """Compute gradient of the soft-max."""
    n = len(h)
    L = 2 * n - 1
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    w = np.exp(alpha * (corr - maxc))
    w = w / w.sum()
    # Gradient contribution of each lag
    grad = np.zeros_like(h)
    # For each lag index offset
    for idx, lag in enumerate(range(-n + 1, n)):
        weight = w[idx]
        if weight == 0:
            continue
        # shift forward and backward using slicing
        if lag >= 0:
            fwd = np.concatenate([np.zeros(lag), h[:n - lag]])
        else:
            lag_abs = -lag
            fwd = np.concatenate([h[lag_abs:], np.zeros(lag_abs)])
        if lag <= 0:
            bwd = np.concatenate([np.zeros(-lag), h[:-lag]])
        else:
            bwd = np.concatenate([h[lag:], np.zeros(lag)])
        grad += weight * ((1.0 - fwd) - bwd)
    return grad

def optimize_continuous(h_start, remaining_time, n, dx):
    """Run gradient descent to optimize h."""
    best_h = h_start.copy()
    best_val = evaluate_max_corr(best_h)
    h = h_start.copy()
    alpha = 5.0  # Initial smoothness parameter

    start = time.time()
    max_iter = 5000
    for _ in range(max_iter):
        if time.time() - start > remaining_time - 2:
            break
        grad = gradient_max_soft(h, alpha)
        step_size = 0.05
        trials = 0
        while trials < 10:
            h_new = project_to_sum_box(h - step_size * grad, n / 2)
            val_new = evaluate_max_corr(h_new)
            if val_new < best_val - 1e-12:
                break
            step_size *= 0.5
            trials += 1
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h = h_new.copy()
        if alpha < 20:
            alpha += 0.01  # Slowly increase softness of max
    return best_h, best_val

def binary_local_search(h_bin, remaining_time, n, dx):
    """Apply binary simulated annealing."""
    start = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_max_corr(best_h)
    T = 0.01
    decay = 0.99  # Annealing rate
    while time.time() - start < remaining_time:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # propose swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = evaluate_max_corr(cand)
        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / T):
            best_val = val
            best_h = cand
        T *= decay
    return best_h, best_val

        if val_opt < best_val:

            best_h = h_opt.copy()

    # Phase 2: More random restarts
    phase_end = start + 0.6 * budget_s
    for i in range(6):
        if time.time() > phase_end:
            break
        r = np.random.rand(n)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    phase_end = start + 0.85 * budget_s
    if best_h is not None:

        if remaining > 1:
            # Convert to binary solution
            idx_sort = np.argsort(-best_h)
            bin_h = np.zeros_like(best_h)
            bin_h[idx_sort[:int(target_sum)]] = 1.0
            bin_h_opt, bin_val = binary_local_search(bin_h, remaining, n, dx)
            if bin_val < best_val:
                best_val = bin_val
                best_h = bin_h_opt.astype(float)

    phase_end = start + budget_s

                h_upscaled = project_to_sum_box(h_upscaled - 0.01 * grad, 400 / 2)

            best_val = evaluate_max_corr(best_h)

    return (best_h.tolist(), float(best_val), n)
