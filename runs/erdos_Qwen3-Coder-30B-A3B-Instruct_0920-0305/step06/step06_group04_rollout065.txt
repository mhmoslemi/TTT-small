import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def fourier_to_h(theta, n_points, k_max):
    """Transform a vector of Fourier coefficients to a valid step function"""
    # Create frequencies
    x = np.linspace(0, 2, n_points)
    h_raw = 0.5 * np.ones_like(x)  # DC component
    for k in range(1, k_max + 1):
        h_raw += theta[2*(k-1)] * np.cos(np.pi * k * x) + theta[2*(k-1)+1] * np.sin(np.pi * k * x)
    
    h_clipped = np.clip(h_raw, 0, 1)
    # Project to correct sum
    h_proj = project_to_sum_box(h_clipped, TARGET_SUM)
    # Downsample to original size (assumes n_points divides evenly)
    h_final = h_proj[::(len(h_clipped)//n_points)]
    return h_final

def objective_fourier(theta, n_points=120, k_max=12):
    """Objective function for DE to minimize C5"""
    try:
        h = fourier_to_h(theta, n_points, k_max)
        return evaluate_c5(h)
    except:
        return 1e10  # Large value indicating infeasibility

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    k_max = 12
    total_vars = 2 * k_max

    # Initial guess for Fourier coefficients
    initial_guess = np.zeros(total_vars)
    bounds = [(-0.5, 0.5) for _ in range(total_vars)]

    # DE solver options
    opts = {'maxiter': 250, 'popsize': 50, 'tol': 1e-6, 'mutation': (0.5, 1), 'recombination': 0.9, 'strategy': 'best1bin'}

    # Phase A: Global search with DE
    de_result = differential_evolution(objective_fourier, bounds, args=(n_points, k_max), **opts)
    best_theta = de_result.x
    best_c5 = de_result.fun
    
    # Phase B: If time remains, refine using more generations
    if (time.time() - start_time) < 0.8 * budget_s:
        opts['maxiter'] = 100
        opts['popsize'] = 30
        best_theta = differential_evolution(objective_fourier, bounds, args=(n_points, k_max), **opts).x

    # Convert to final h
    try:
        h_final = fourier_to_h(best_theta, n_points, k_max)
    except:
        h_final = np.ones(n_points) * 0.5
        
    # Local refinement of the h vector
    if (time.time() - start_time) < 0.95 * budget_s:
        def refine_obj(h_vec):
            try:
                return evaluate_c5(h_vec)
            except:
                return 1e10

        h_refined = h_final.copy()
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = differential_evolution(refine_obj, bounds, constraints=constraints, maxiter=50, popsize=20)
            if res.success:
                h_refined = project_to_sum_box(res.x, TARGET_SUM)
                best_c5 = refine_obj(h_refined)
        except:
            pass

    # Final binary swap operation for minor improvement
    if (time.time() - start_time) < 0.99 * budget_s:
        h_binary = h_refined.copy()
        # Round to binary
        indices = np.argsort(h_binary)[::-1]
        ones = indices[:n_points//2]
        zeros = indices[n_points//2:]
        h_binary[ones] = 1.0
        h_binary[zeros] = 0.0
        h_binary = project_to_sum_box(h_binary, TARGET_SUM)
        h_final = h_binary
        best_c5 = evaluate_c5(h_final)

    return (h_final.tolist(), float(best_c5), n_points)
