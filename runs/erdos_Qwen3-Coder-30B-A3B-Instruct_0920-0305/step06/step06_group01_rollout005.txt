import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, alpha):
    """Compute gradient of soft minimum with temperature alpha"""
    # Compute auto correlation of h with itself
    S = correlate(h, h, 'full')
    
    # Numerical stability
    S_scaled = -alpha * S
    S_scaled -= np.max(S_scaled)
    exp_S = np.exp(S_scaled)
    weights = exp_S / np.sum(exp_S)
    
    # Construct symmetric kernel W from weights
    W = np.zeros(2 * n_points - 1)
    for i in range(n_points):
        W[n_points - 1 + i] = weights[i]
        if i != 0:
            W[n_points - 1 - i] = weights[i]
    
    # Compute gradient via convolution
    conv = np.convolve(h, W, mode='full')
    g = conv[n_points - 1 : n_points - 1 + n_points]
    g += weights[0] * h
    
    return g
