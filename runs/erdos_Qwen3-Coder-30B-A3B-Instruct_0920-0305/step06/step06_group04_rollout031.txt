def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_f_gradients(h, alpha=1.0):
    """Compute gradients for smoothed max approach"""
    n = len(h)
    # Compute all f_k values
    f_vals = []
    for k in range(n):
        if k == 0:
            f_k = np.sum(h * h)
        else:
            f_k = np.sum(h[:-k] * h[k:])
        f_vals.append(f_k)
    
    # Compute weights (smoothed probabilities) for minimizing max_f
    f_vals = np.array(f_vals)
    exp_neg_alpha_f = np.exp(-alpha * f_vals)
    sum_exp = np.sum(exp_neg_alpha_f)
    if sum_exp == 0:
        weights = np.ones_like(f_vals) / n
    else:
        weights = exp_neg_alpha_f / sum_exp
    
    # Compute gradient ∇F = -∑_k w_k ∇f_k
    # ∇f_k[i] = h[i+k] + h[i-k] (with proper boundary handling)
    gradient = np.zeros_like(h)
    
    for k in range(n):
        if k == 0:
            # For k=0: f_0 = sum_i h_i^2, gradient = 2 h_i
            gradient += -2 * weights[k] * h
        else:
            # Add contributions from right shift (h_{i+k})
            if k < n:
                gradient[:n-k] += -weights[k] * h[k:]
                # Add contributions from left shift (h_{i-k})
                gradient[k:] += -weights[k] * h[:n-k]
    
    return gradient

def smooth_max_c5(h, alpha=1.0):

    n = len(h)

    f_vals = []

            f_k = np.sum(h[:-k] * h[k:])
        f_vals.append(f_k)
    
    f_vals = np.array(f_vals)
    
    # Smoothed max: log-sum-exp approximation
    exp_neg_alpha_f = np.exp(-alpha * f_vals)
    sum_exp = np.sum(exp_neg_alpha_f)
    if sum_exp == 0:
        return 0.0
    log_sum_exp = np.log(sum_exp)
    # The value we want to minimize is -log_sum_exp = log_sum_exp / alpha
    return log_sum_exp / alpha
