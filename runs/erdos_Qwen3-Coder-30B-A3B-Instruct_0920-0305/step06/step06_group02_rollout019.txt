import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def particle_swarm_optimization(h0, max_iter=3000, swarm_size=30):
    """Improved PSO-based optimization"""
    h = h0.copy()
    n = len(h)
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for _ in range(swarm_size):
        h_init = h + np.random.normal(0, 0.01, n)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        swarm.append(h_init.copy())
        velocities.append(np.random.uniform(-0.02, 0.02, n))
        personal_best.append(h_init.copy())
        personal_best_vals.append(evaluate_c5(h_init))
    
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    for iter_num in range(max_iter):
        if iter_num % 100 == 0:
            # Refresh if stagnated
            if iter_num > 0:
                swarm = [project_to_sum_box(h + np.random.normal(0, 0.01, n), TARGET_SUM) 
                        for h in swarm]
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_sum_box(swarm[i], TARGET_SUM)
            
            # Evaluate
            current_val = evaluate_c5(swarm[i])
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                if current_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = current_val
    
    return global_best, global_best_val

    h = h0.copy()
    best_c5 = evaluate_c5(h)

    corr_full = compute_corr_fft(h)
    
    for i in range(max_iter):
        if i % 50 == 0:
            corr_full = compute_corr_fft(h)
        
        # Find indices with highest correlation (worst-case)
        max_idx = np.argmax(corr_full)
        max_val = corr_full[max_idx]
        
        # Try to decrease this particular correlation by increasing h or diminishing it
        # Try from all positions with small steps
        best_improvement = 0
        best_delta = None
        best_pos = None
        
        delta_step = 0.01
        
        for pos in range(n_points):
            # Try increasing the value
            if h[pos] < 1.0:
                h_new = h.copy()
                h_new[pos] += delta_step
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                c5_new = evaluate_c5(h_new)
                improvement = best_c5 - c5_new
                
                if improvement > best_improvement and improvement > 1e-12:
                    best_improvement = improvement
                    best_delta = delta_step
                    best_pos = pos
                    
            # Try decreasing the value
            if h[pos] > 0.0:
                h_new = h.copy()
                h_new[pos] -= delta_step
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                c5_new = evaluate_c5(h_new)
                improvement = best_c5 - c5_new
                
                if improvement > best_improvement and improvement > 1e-12:
                    best_improvement = improvement
                    best_delta = -delta_step
                    best_pos = pos
        
        # Apply best move if beneficial
        if best_pos is not None and best_improvement > 1e-12:
            h[best_pos] += best_delta
            h = project_to_sum_box(h, TARGET_SUM)
            best_c5 = evaluate_c5(h)
        else:
            # Random small modification to escape local minima
            if np.random.random() < 0.1:
                idx = np.random.randint(0, n_points)
                delta = np.random.uniform(-0.01, 0.01)
                h[idx] += delta
                h = project_to_sum_box(h, TARGET_SUM)
                best_c5 = evaluate_c5(h)
    
    return h

        h_refined, c5_refined = particle_swarm_optimization(h_start, max_iter=1500)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
