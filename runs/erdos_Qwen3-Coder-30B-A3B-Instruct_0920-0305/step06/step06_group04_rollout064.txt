import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60
N_FINE = 800       # For accurate FFT evaluation
N = 120            # Final discretization size
K = 12             # Number of harmonics in Fourier expansion
dx = 2.0 / N
dx_fine = 2.0 / N_FINE

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50

    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def fourier_to_h(theta):
    """Convert Fourier coefficients to h vector"""
    a = theta[:K]
    b = theta[K:]
    x = np.linspace(0, 2, N_FINE)
    phi = 0.5 + np.sum([a[k]*np.cos(np.pi*(k+1)*x) + b[k]*np.sin(np.pi*(k+1)*x) for k in range(K)], axis=0)
    h = project_to_sum_box(phi, TARGET_SUM)
    # Downsample to N points
    h = h[::N_FINE//N]
    return h

def evaluate_c5(h):
    """Compute full correlation and return C5"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)

def evaluate_c5_fft(h):
    """Uses FFT for more accurate computation"""
    h_rev = h[::-1]
    b = 1.0 - h
    n = len(h)
    # Full convolution using FFT
    corr = np.real(ifft(fft(h_rev, 2*n - 1) * fft(b, 2*n - 1)))
    return dx * np.max(corr)
