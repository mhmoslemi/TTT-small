import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different n_points
    n_points_list = [120, 200]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:

        # Multi-start phase (exploratory) - first 30 seconds
        for _ in range(10):

                h_start = np.zeros(n_points)

                h_start = project_to_sum_box(h_start, TARGET_SUM)
            else:
                # Continuous starts
                h_start = np.random.uniform(0, 1, n_points)
                h_start = project_to_sum_box(h_start, TARGET_SUM)
            
            c5_start = evaluate_c5(h_start)
            if c5_start < best_c5:
                best_h = h_start.copy()
                best_c5 = c5_start

        # Main optimization phase (remaining time)
        # Adjusted time allocation based on n_points
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget <= 10:
            continue

        h = best_h.copy()
        alpha = 0.5
        iter_cnt = 0

        while (time.time() - start_time) < 0.9 * budget_s:
            iter_cnt += 1

            corr = compute_corr_fft(h)

            if iter_cnt % 10 == 0:

                    if i < len(h) and (i+j)<len(h):

                return evaluate_c5(h)

                return np.sum(h) - TARGET_SUM

                h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,

                return np.sum(h) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            res = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12},
                tol=1e-12
            )
            
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, TARGET_SUM)
                c5_final = evaluate_c5(h_final)
                
                if c5_final < best_c5:
                    best_h = h_final
