import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 256
dx = 2.0 / n_points
L = 2 * n_points - 1  # Length for zero-padding
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Initialize main variables
    h = h_interpolated.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Parameters
    lambda_init = 0.05
    lambda_incr = 1.02
    lambda_max = 1.0
    beta_init = 0.5
    beta_incr = 1.01
    phase_reset_interval = 300
    max_resets = 5
    restart_count = 0
    lambda_k = lambda_init
    beta_k = beta_init
    
    # Main iteration loop
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 10:  # Reserve 10s for final polishing
        # FFT step
        h_padded = np.pad(h, (0, n_points - 1), 'constant')
        H = np.fft.fft(h_padded)
        
        # Compute power spectrum
        P = np.abs(H)**2
        
        # Compute target magnitude
        P_target_val = np.mean(P[1:])  # Average of off-DC powers
        M_target = np.full(L, np.sqrt(P_target_val))
        M_target[0] = np.abs(H[0])  # Keep DC
        
        # Mix magnitude
        M = (1 - lambda_k) * np.abs(H) + lambda_k * M_target
        
        # Reconstruct signal with same phase
        H_new = M * np.exp(1j * np.angle(H))
        
        # Inverse FFT
        h_new = np.fft.ifft(H_new).real[:n_points]
        
        # Projection onto feasibility constraints
        h_proj = project_to_sum_box(h_new, TARGET_SUM)
        
        # Apply relaxation
        h = (1 - beta_k) * h + beta_k * h_proj
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update parameters
        lambda_k = min(lambda_k * lambda_incr, lambda_max)
        beta_k = min(beta_k * beta_incr, 1.0)
        
        # Phase reset if necessary
        if no_improvement_count >= phase_reset_interval and restart_count < max_resets:

        return np.sum(h) - TARGET_SUM

                best_h = h_refined

    except Exception:
        pass

    return (best_h.tolist(), float(best_c5), n_points)
