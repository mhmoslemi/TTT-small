import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

# Helper functions
def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def compute_hessian(h):
    # Compute gradient components for SLSQP
    corr = compute_corr_fft(h)
    k_max = np.argmax(corr)
    grad = np.zeros_like(h)
    n = len(h)
    for i in range(n):
        if i + k_max < n:
            grad[i] -= (1 - h[i + k_max])
        if i - k_max >= 0:
            grad[i] -= (1 - h[i - k_max])
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using binary search"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h, max_iter=100):
    """Perform local refinement with SLSQP"""
    bounds = [(0, 1) for _ in range(len(h))]
    target_sum = np.sum(h)

    def objective(h):
        return evaluate_c5(h)

    def constraint_eq(h):
        return np.sum(h) - target_sum

    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        from scipy.optimize import minimize
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, target_sum)
            return h
    except:
        pass
    return h

def solve_sdp(n, target_sum, max_time=100):
    """Solve SDP relaxation"""
    dx = 2.0 / n
    M = target_sum

    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == M,
        0 <= h,
        h <= 1,
        cp.diag(X) == h,
        X >> 0,
    ]

    # Lag constraints
    for k in range(n):
        P_k = np.zeros((n, n))
        for i in range(n - k):
            P_k[i, i + k] = 1
        constraints.append(t >= M - cp.trace(P_k @ X))

    # Problem
    prob = cp.Problem(cp.Minimize(t), constraints)

    # Solve
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, time_limit=max_time)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, X.value
    except:
        pass
    return None, None

def random_round(h_rel, X_rel, M, n):
    """Random hyperplane rounding"""
    # SVD of X_rel
    try:
        U, s, Vt = np.linalg.svd(X_rel)
        k = min(10, len(s))
        V_plus = U[:, :k] * np.sqrt(s[:k])
        g = np.random.randn(k)
        s = V_plus @ g
        b = (s >= 0).astype(float)
    except:
        return None
    # Adjust sum to M
    diff = np.sum(b) - M
    if diff < 0:
        idx = np.where(b == 0)[0]
        np.random.shuffle(idx)
        b[idx[:abs(diff)]] = 1
    elif diff > 0:
        idx = np.where(b == 1)[0]
        np.random.shuffle(idx)
        b[idx[:diff]] = 0
    return b

def threshold_round(h_rel, M):
    """Threshold rounding"""
    idx = np.argsort(h_rel)[::-1]
    b = np.zeros_like(h_rel)
    b[idx[:int(M)]] = 1
    return b

def swap_improve(h_candidate):
    """Perform deterministic swap improvements to reduce overlap"""
    n = len(h_candidate)
    h_candidate = h_candidate.copy()
    improved = True
    count = 0
    max_count = 10000  # Safety cap
    while improved and count < max_count:
        improved = False
        # Find positions of 1's and 0's
        ones = np.where(h_candidate == 1)[0]
        zeros = np.where(h_candidate == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Compute current correlations
        corr = compute_corr_fft(h_candidate)
        k_star = np.argmax(corr)
        # Remap indices to correlate with the worst lag
        for i in ones:
            for j in zeros:
                old_val = corr[k_star]
                # Compute how swap affects the worst term
                # Original correlation at lag k_star
                # The change in overlap at lag k_star due swapping i and j
                new_corr = corr.copy()
                # For the following terms, we are subtracting or adding the change in terms
                # set dummy values for indices we care about
                changed_terms = set()
                # Update change in at lag k_star
                if (i+k_star < n and j >= k_star) or (j+k_star < n and i >= k_star):
                    if i+k_star < n:
                        # This is old value in i position before swap
                        # and i+k_star becomes 0
                        new_corr[i] -= h_candidate[i+k_star]
                        changed_terms.add(i)
                    if j+k_star < n:
                        new_corr[j] -= h_candidate[j+k_star]
                        changed_terms.add(j)
                    if i >= k_star:
                        new_corr[i-k_star] -= h_candidate[i]
                        changed_terms.add(i-k_star)
                    if j >= k_star:
                        new_corr[j-k_star] -= h_candidate[j]
                        changed_terms.add(j-k_star)
                else:
                    # Compute full update for the worst lag term (convolution changes)
                    # Just run a full recalculation for better accuracy and simplicity
                    h_new = h_candidate.copy()
                    h_new[i], h_new[j] = h_new[j], h_new[i]
                    corr_new = compute_corr_fft(h_new)
                    new_val = np.max(corr_new)
                new_val = np.max(corr)
                if new_val < corr[k_star]:
                    h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
                    improved = True
                    break
        count += 1
    return h_candidate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Step 1: Solve SDP for increasing sizes
    solutions = []
    n_sizes = [60, 80, 100, 120]
    time_per_sdp = budget_s * 0.5 / len(n_sizes)

    for n in n_sizes:
        target_sum = n / 2
        h_sdp, X_sdp = solve_sdp(n, target_sum, time_per_sdp)
        if h_sdp is not None:
            # Rounding and hill climbing
            roundings = []
            # Probability of random vs threshold rounding
            for _ in range(15):
                b = random_round(h_sdp, X_sdp, target_sum, n)
                if b is not None:
                    roundings.append(b)
            # Speed up with threshold rounding
            b_thresh = threshold_round(h_sdp, target_sum)
            if b_thresh is not None:
                roundings.append(b_thresh)

            for b in roundings:
                # Swap improvement
                b_improved = swap_improve(b)
                # Check if we want it
                c5 = evaluate_c5(b_improved)
                solutions.append((c5, b_improved, n))
            # Early exit if time is almost up
            if time.time() - start_time > 0.7 * budget_s:
                break

    if not solutions:
        # Fallback to simple pattern
        h_fallback = np.zeros(100)
        # Assign length arbitrary blocks of 1's and 0's to total length
        h_fallback[::2] = 1
        h_fallback = project_to_sum_box(h_fallback, 50)
        c5_fallback = evaluate_c5(h_fallback)
        return (h_fallback.tolist(), float(c5_fallback), 100)

    # Find best by C5
    best_c5, best_h, best_n = min(solutions, key=lambda x: x[0])
    # Optional polishing
    best_h = local_refine(best_h, max_iter=100)
    if time.time() - start_time < budget_s - 10:
        best_h = local_refine(best_h, max_iter=150)
    final_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(final_c5), best_n)
