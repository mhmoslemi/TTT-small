import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)))
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_for_shift(h, s):
    """Compute gradient for correlation at shift s"""
    grad = np.zeros_like(h)
    if s >= 0:
        grad[:n_points-s] += h[s:]
        grad[s:] += h[:n_points-s]
    else:
        t = -s
        grad[t:] += h[:n_points-t]
        grad[:n_points-t] += h[t:]
    return grad

def subgradient_descent(h0, target_sum, max_iter, step0, decay):
    """Subgradient descent for minimizing max overlap"""
    h = h0.copy()
    step = step0
    
    for i in range(max_iter):
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        
        # Find indices with maximum overlap
        max_val = overlaps.max()
        idxs = np.where(np.abs(overlaps - max_val) < 1e-12)[0]
        
        # Average gradients from all maximum shifts
        grad = np.zeros_like(h)
        for idx in idxs:
            shift = idx - (n_points - 1)
            grad += gradient_for_shift(h, shift)
        
        grad /= len(idxs)
        
        # Update step
        h = h - step * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = project_to_sum_box(h, target_sum)
        
        # Decay step size
        step *= decay
        
        if i % 100 == 0:
            # Periodic evaluation to ensure progress
            current_c5 = evaluate_c5(h)
            if i % 1000 == 0:
                pass  # Report occasionally
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Systematically test multiple resolutions
    resolutions = [180, 240]
    best_c5 = np.inf
    best_h = None
    best_n = n_points
        
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        n_points = n
        dx = 2.0 / n_points
        TARGET_SUM = n_points / 2
        
        # Use interpolation from best previous solution if exists
        if best_h is not None and len(best_h) > n:
            # Interpolate
            old_x = np.linspace(0, 2, len(best_h))
            new_x = np.linspace(0, 2, n)
            h0 = np.interp(new_x, old_x, best_h)
        else:
            # Generate random initialization
            h0 = np.random.rand(n)
            h0 = project_to_sum_box(h0, TARGET_SUM)
        
        # Run multiple restarts with subgradient descent
        num_restarts = 30
        for restart in range(num_restarts):
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Random perturbation of initial solution
            h = h0 + np.random.normal(0, 0.02, n)
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Run subgradient descent
            h_final = subgradient_descent(h, TARGET_SUM, max_iter=3000, step0=0.05, decay=0.9995)
            c5 = evaluate_c5(h_final)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
                best_n = n
                
    # Final polishing with SLSQP if time remaining and solution exists
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        try:
            bounds = [(0, 1) for _ in range(best_n)]
            cons = {'type': 'eq', 'fun': lambda hh: np.sum(hh) - TARGET_SUM}
            res = minimize(
                lambda hh: evaluate_c5(hh), 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            )
            if res.success:
                c5_refined = evaluate_c5(res.x)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = res.x.copy()
        except:
            pass

    return (best_h.tolist(), float(best_c5), best_n)
