import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp
import time
import random

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using integer convolution"""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_binary_vector(n, target_sum):
    """Generate a random binary vector with specified sum"""
    h = np.zeros(n)
    ones = np.random.choice(n, int(target_sum), replace=False)
    h[ones] = 1
    return h

def get_worst_lags(h):
    """Get the worst lag indices that produce maximum overlap"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.argmax(corr)

def swap_move(h, i, j):
    """Perform a swap between positions i and j"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def block_move(h, start, length):
    """Perform a block flip move"""
    h = h.copy()
    end = min(start + length, len(h))
    for i in range(start, end):
        h[i] = 1 - h[i]
    
    # Repair weight
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) > TARGET_SUM:
        # Remove excess 1s
        excess = len(ones) - int(TARGET_SUM)
        remove_indices = np.random.choice(ones, size=excess, replace=False)
        for idx in remove_indices:
            h[idx] = 0
    elif len(ones) < TARGET_SUM:
        # Add needed 1s
        needed = int(TARGET_SUM) - len(ones)
        for _ in range(needed):
            if len(zeros) > 0:
                idx = np.random.choice(zeros)
                h[idx] = 1
                zeros = np.where(h == 0)[0]
    
    return h

def simulated_annealing(h_start, max_time, T0=0.02, alpha=0.995, l=100):
    """Simulated Annealing with restarts and elitism"""
    current_h = h_start.copy()
    current_c5 = evaluate_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = T0
    iterations = 0
    elapsed = 0
    
    # Elite pool for diversity
    elites = []
    
    time_marker = time.time()
    
    while elapsed < max_time:
        start_time = time.time()
        
        # Run 1000 iterations or until time expires
        for i in range(l):
            # Try either a simple swap or a block move
            if np.random.rand() < 0.15:
                # Block move
                length = np.random.choice([2, 3, 4])
                start = np.random.randint(0, n_points - length + 1)
                new_h = block_move(current_h, start, length)
            else:
                # Simple swap
                ones = np.where(current_h == 1)[0]
                zeros = np.where(current_h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i_pos = np.random.choice(ones)
                    j_pos = np.random.choice(zeros)
                    new_h = swap_move(current_h, i_pos, j_pos)
                else:
                    new_h = current_h.copy()
            
            # Ensure weight is preserved
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
        T *= alpha
        iterations += l
        elapsed = time.time() - start_time
        
        # Check time for restart 
        if time.time() - time_marker > 30:
            time_marker = time.time()
            # Add to elites
            if len(elites) < 5:
                elites.append((best_h.copy(), best_c5))
                elites.sort(key=lambda x: x[1])  # keep sorted by C5
            else:
                if best_c5 < elites[-1][1]:
                    elites[-1] = (best_h.copy(), best_c5)
                    elites.sort(key=lambda x: x[1])
            # Restart from a random elite or from scratch
            choice = np.random.rand()
            if choice < 0.9 and len(elites) > 0:
                current_h = elites[np.random.randint(len(elites))][0].copy()
            else:
                current_h = random_binary_vector(n_points, TARGET_SUM)
            current_c5 = evaluate_c5(current_h)
            
        if best_c5 < 0.379:
            break
    
    return best_h, best_c5

def hill_climb(h_start):
    """Perform hill climbing with exhaustive swap search"""
    h = h_start.copy()
    current_c5 = evaluate_c5(h)
    improved = True
    
    while improved:
        improved = False
        best_delta = 0
        best_i, best_j = None, None
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try various swaps
        for i in ones:
            for j in zeros:
                h_temp = swap_move(h, i, j)
                temp_c5 = evaluate_c5(h_temp)
                if temp_c5 < current_c5:
                    # Alpha beta search to avoid accepted local minima
                    delta = current_c5 - temp_c5
                    if delta > best_delta:
                        best_delta = delta
                        best_i, best_j = i, j
                        
        if best_i is not None:
            h = swap_move(h, best_i, best_j)
            current_c5 = evaluate_c5(h)
            improved = True
    
    return h, current_c5

            h_float,

            bounds=bounds,
