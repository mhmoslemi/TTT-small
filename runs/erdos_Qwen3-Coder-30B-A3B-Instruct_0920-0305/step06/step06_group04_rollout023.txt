N_POINTS_LIST = [120, 200]

def compute_c5_from_corr(corr, n):
    dx = 2.0 / n
    return dx * np.max(corr)

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return compute_c5_from_corr(corr, len(h))

def evaluate_c5_precise(h):
    """High-precision evaluation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project to sum=target_sum and [0,1] bounds using water-filling algorithm"""
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 100
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0, 1)
    return y

def compute_autocorrelation(h):
    """Compute autocorrelations for all lags using FFT"""
    n = len(h)
    h_rev = h[::-1]
    fft_h = np.fft.fft(h_rev, 2*n - 1)
    fft_h_conj = np.conj(fft_h)
    corr = np.real(np.fft.ifft(fft_h_conj * fft_h))
    return corr[:n]

def smooth_max_objective(h, alpha):
    """Compute smoothed max objective: log-sum-exp of -alpha*f"""
    f = compute_autocorrelation(h)
    f = f[numel(f) // 2 :]
    f_centered = -alpha * f
    f_centered -= np.max(f_centered)  # prevent overflow
    exp_f = np.exp(f_centered)
    return -np.log(np.sum(exp_f)) / alpha

def compute_gradient(h, alpha):
    """Compute gradient of smooth max objective"""
    n = len(h)
    f = compute_autocorrelation(h)
    f = f[numel(f) // 2 :]
    
    # Compute weights
    f_centered = -alpha * f
    f_centered -= np.max(f_centered)  # prevent overflow
    w = np.exp(f_centered)
    w /= np.sum(w)
    
    # Compute g = -sum(w_l * (shifted_h + shifted_h_rev))
    g = np.zeros(n)
    for l in range(len(f)):
        if l > 0:
            g[:-l] += w[l] * h[l:]
            g[l:] += w[l] * h[:-l]
        else:
            g += 2 * w[0] * h
    
    g = -g  # gradient
    return g

def line_search_step(h, g, alpha, max_steps=8):
    """Perform backtracking line search"""
    eta = 0.5
    f_orig = evaluate_c5(h)
    
    for _ in range(max_steps):
        h_trial = project_to_sum_box(h - eta * g, TARGET_SUM)
        f_trial = evaluate_c5(h_trial)
        if f_trial < f_orig:
            return h_trial, f_trial
        eta *= 0.5
    return h, f_orig

def discrete_phase(h, time_remaining):
    """Binary swap refinement"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Round to binary
    h_bin = np.round(h).astype(float)
    h_bin = project_to_sum_box(h_bin, TARGET_SUM)
    
    # Compute initial autocorrelations
    corr = compute_autocorrelation(h_bin)
    f = corr[numel(corr) // 2 :]
    
    swaps_made = 0
    while time.time() - start_time < time_remaining and swaps_made < 20000:
        if swaps_made > 0 and swaps_made % 500 == 0:
            f = compute_autocorrelation(h_bin)
            f = f[numel(f) // 2 :]
            
        # Get minimum lag
        min_idx = np.argmin(f)
        min_val = f[min_idx]
        
        # Find candidates to swap
        candidates_one = list()
        candidates_zero = list()
        for i in range(len(h_bin)):
            # Check if i can still improve f[min_idx]
            if h_bin[i] == 1:
                if i + min_idx < len(h_bin) and h_bin[i + min_idx] == 0:
                    candidates_one.append(i)
            elif h_bin[i] == 0:
                if i + min_idx < len(h_bin) and h_bin[i + min_idx] == 1:
                    candidates_zero.append(i)
                    
        if not candidates_one or not candidates_zero:
            break
            
        # Try swaps
        for _ in range(10):
            p, q = np.random.choice(candidates_one), np.random.choice(candidates_zero)
            # Compute Delta f
            delta_f = np.zeros(len(f))
            for l in range(len(f)):
                if l == 0:
                    delta_f[l] = -2 * h_bin[p] * h_bin[q]  # Only non-zero term
                else:
                    if p + l < len(h_bin) and h_bin[p + l] == 1:
                        delta_f[l] += 1
                    if p - l >= 0 and h_bin[p - l] == 1:
                        delta_f[l] -= 1
                    if q + l < len(h_bin) and h_bin[q + l] == 1:
                        delta_f[l] -= 1
                    if q - l >= 0 and h_bin[q - l] == 1:
                        delta_f[l] += 1
            new_f = f + delta_f
            new_min = np.min(new_f)

            if new_min > min_val:
                h_bin[p] = 0
                h_bin[q] = 1
                f = new_f
                swaps_made += 1
                break
    
    return h_bin, evaluate_c5(h_bin)

def continuous_phase(h, alpha, time_remaining):
    """Smooth, projected gradient ascent"""
    start_time = time.time()
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_c5(h_current)
    
    while time.time() - start_time < time_remaining:
        # Compute gradient
        try:
            g = compute_gradient(h_current, alpha)
        except:
            break
        # Line search
        h_trial, c5_trial = line_search_step(h_current, g, alpha)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            
        h_current = h_trial
        
    return best_h

        h = best_h.copy()

    c5_fallback = evaluate_c5(h_fallback)
    return (h_fallback.tolist(), float(c5_fallback), 120)
