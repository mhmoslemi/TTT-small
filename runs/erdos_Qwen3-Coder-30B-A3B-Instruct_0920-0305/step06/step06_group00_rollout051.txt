def full_corr(a, b):
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]  # lag range -(n-1)...(n-1)

def evaluate_c5(h):
    corr = full_corr(h, 1 - h)
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    low  = -np.max(x) - 1.0
    high =  1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y   = np.clip(x + lam, 0.0, 1.0)
        s   = y.sum()
        if s > target_sum:
            high = lam
        else:
            low  = lam
    y = np.clip(x + (low+high)/2.0, 0.0, 1.0)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def gradient_max_soft(h, alpha):
    n = len(h)
    corr = full_corr(h, 1 - h)
    maxc = corr.max()
    w = np.exp(alpha * (corr - maxc))
    w_sum = w.sum()
    w = w / w_sum
    # gradient vector for each i
    grad = np.zeros_like(h)
    # Edge case: modify this to avoid loops by vectorizing convolution?
    # For efficiency, we will use a convolution-based approach:
    # rev_w = w[::-1]  # this is needed when shifting weights back
    w_rev = np.flip(w)
    # forward: C[i] = sum(w[l]*h[i+l])
    h_pad = np.pad(h, (0, n-1), mode='constant', constant_values=0)
    w_pad = np.pad(w, (0, n-1), mode='constant', constant_values=0)
    conv1 = np.convolve(h_pad, w_pad, mode='valid')[:n]
    # backward: C[i] = sum(w[l]*h[i-l]) = sum(w_rev[-l]*h[i-l]) = simply convolve h with w_rev
    # but ensure correct truncation for length n:
    conv2 = np.convolve(np.pad(h, (n-1, 0), mode='constant', constant_values=0), 
                        w_rev, mode='valid')[n-1:]
    grad = 1.0 - conv1 - conv2
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Strategy: 
    # 1. Interpolated solution from initial_h_values
    # 2. Random start with same size
    # 3. Binary construction
    # 4. Use n-points = 250 for good resolution and speed within budget
    n = 250
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Initialize solution set with diverse starting points
    h_seeds = []
    
    # Seed 1: Interpolated solution
    h_interp = np.interp(np.linspace(0, 2, n), 
                         np.linspace(0, 2, len(initial_h_values)), 
                         initial_h_values)
    h_interp = project_to_sum_box(h_interp, target_sum)
    h_seeds.append(h_interp)
    
    # Seed 2: Random continuous solution
    h_rand = np.random.rand(n)
    h_rand = project_to_sum_box(h_rand, target_sum)
    h_seeds.append(h_rand)
    
    # Seed 3: Binary solution
    h_bin = np.zeros(n)
    indices = np.random.choice(n, int(target_sum), replace=False)
    h_bin[indices] = 1.0
    h_bin = project_to_sum_box(h_bin, target_sum)
    h_seeds.append(h_bin)
    
    # Seed 4: Alternative optimizing order
    # Try another pattern: step function - distribution of 0, 1, 0, 1, ... but not exactly
    h_step = np.zeros(n)
    h_step[:n//2] = 1.0
    h_step = project_to_sum_box(h_step, target_sum)
    h_seeds.append(h_step)
    
    # Parameters
    total_iterations = 0
    best_h = None
    best_c5 = float('inf')
    
    # --- Continuous optimization ---
    time_allocated = 0.55 * budget_s  # 55% of the time for continuous optimization
    iter_per_seed = 500
    
    for h in h_seeds:
        if time.time() - start_time > start_time + time_allocated:
            break
        h_current = h.copy()
        c5_current = evaluate_c5(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
        
        alpha = 5.0  # initial smoothness
        # Main iteration loop for optimizing a seed
        for i in range(iter_per_seed):
            if time.time() - start_time > start_time + time_allocated:
                break
            total_iterations += 1

            # Compute gradient with soft max
            grad = gradient_max_soft(h_current, alpha)
            
            # Backtracking line search
            step_size = 0.1
            while step_size > 1e-8:
                h_new = project_to_sum_box(h_current - step_size * grad, target_sum)
                c5_new = evaluate_c5(h_new)
                if c5_new < c5_current - 1e-12:
                    h_current = h_new
                    c5_current = c5_new
                    break
                else:
                    step_size *= 0.5
                    
            # Adaptive α
            if c5_current < best_c5:
                alpha = min(alpha + 0.1, 20.0)
            else:
                # Slow decay
                alpha = max(alpha - 0.02, 5.0)

            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()

    # --- Binary refinement ---
    time_allocated = budget_s * 0.85
    if time.time() - start_time > time_allocated:
        pass
    else:
        # Convert best to binary
        binary_h = (best_h >= np.median(best_h)).astype(float)
        idx_sort = np.argsort(-best_h)
        binary_h[:] = 0
        binary_h[idx_sort[:int(target_sum)]] = 1.0
        binary_h = project_to_sum_box(binary_h, target_sum)

        # Binary simulated annealing or local search
        binary_current = binary_h.copy()
        c5_current = evaluate_c5(binary_current)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = binary_current.copy()
        
        temp = 0.05
        cooling_rate = 0.9995
        iter_steps = 0
        max_iter = 50000
        
        while iter_steps < max_iter and time.time() - start_time < time_allocated:
            # Pick a 1-index and 0-index
            ones = np.where(binary_current > 0.5)[0]
            zeros = np.where(binary_current < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            # Explore swap
            binary_cand = binary_current.copy()
            binary_cand[i] = 0.0
            binary_cand[j] = 1.0

            c5_cand = evaluate_c5(binary_cand)
            if c5_cand < c5_current:
                binary_current = binary_cand
                c5_current = c5_cand
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = binary_cand.copy()
            else:
                # Accept with probability
                delta = c5_cand - c5_current
                if np.random.rand() < np.exp(-delta / temp):
                    binary_current = binary_cand
                    c5_current = c5_cand
            
            temp *= cooling_rate
            iter_steps += 1
        if iter_steps == max_iter:
            print("Warning: Binary phase reached max iterations")

    # --- Upscale and refine ---
    time_left = budget_s - (time.time() - start_time)
    if time_left > 10 and len(best_h) < 500:
        n_upscale = min(len(best_h) * 2, 500)
        dx_up = 2.0 / n_upscale
        h_up = np.interp(np.linspace(0, 2, n_upscale), np.linspace(0, 2, len(best_h)), best_h)
        # Refine again
        target_up = n_upscale / 2.0
        h_up = project_to_sum_box(h_up, target_up)
        
        # Final short gradient pass
        alpha = 20.0
        for it in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            grad = gradient_max_soft(h_up, alpha)
            step_size = 0.01
            h_new = project_to_sum_box(h_up - step_size * grad, target_up)
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h_up):
                h_up = h_new
            else:
                step_size *= 0.5
                h_new = project_to_sum_box(h_up - step_size * grad, target_up)
                if evaluate_c5(h_new) < evaluate_c5(h_up):
                    h_up = h_new

        best_h = h_up
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), len(best_h))
