import numpy as np
from scipy.optimize import minimize
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def full_corr(a, b):
    """Compute full correlation of a and b using FFT."""
    n = a.size
    N = 2*n
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    c = np.fft.irfft(A * np.conj(B), N)
    return c[:2*n-1]

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation."""
    corr = full_corr(h, 1 - h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_of_softmax_max(h, alpha):

    n = len(h)
    corr_vals = full_corr(h, 1 - h)
    max_corr = corr_vals.max()
    # Compute softmax weights
    w = np.exp(alpha * (corr_vals - max_corr))
    w_sum = w.sum()
    w = w / w_sum
    
    # Compute weighted gradient: per element, differentiates sum w[i]*C[i]
    # ∂C[i]/∂h[j] = (1 - h[i+lag]) - h[i-lag], (with zero padding)
    # So we sum over lags w[l] * ((1 - h[i+l]) - h[i-l])
    # This is same as convolving h with some weights.
    
    # conv_fwd[i] = Σ_l w[l] * h[i+l]
    # conv_bwd[i] = Σ_l w[l] * h[i-l]
    # grad = 1 - conv_fwd - conv_bwd 

    # We can do this with convolution machinery:
    # generalized from convolution of two sequences:
    # conv(a,b)[i] = Σ_j a[j] * b[i-j]
    # So:
    # conv_fwd = Σ_l w[l] * h[i+l] = conv(w_rev, h)[i]
    # conv_bwd = Σ_l w[l] * h[i-l] = conv(w_rev, h)[i] where w_rev = w[::-1]
    # For this we seek more direct shift than creating full kernel.

    # We zero-pad below to length 2n
    w_padded = np.zeros(2*n)
    w_padded[n-1:] = w  # w is centered at index 0, lags run -n+1..n-1

    h_y = np.pad(h, (n-1, n-1), mode='constant', constant_values=0.)

    # These convolutions
    def convolution(a, b):
        n = a.size
        N = 2*n
        A = np.fft.rfft(a, N)
        B = np.fft.rfft(b, N)
        c = np.fft.irfft(A * np.conj(B), N)
        return c[:2*n-1]

    # conv_fwd = w[::-1] uconvo h with appropriate indexing.
    # Prepare the correct weight vectors
    # w_rev at index 0 corresponds to lag -(n-1), so w_rev[0] is weight of lag -(n-1)
    # We index wg = w_rev; shift = 0 means lag 0 (so w_rev[0] = w[n-1])
    # So when conv(w_rev, h), we get sum over l of w_rev[l] * h[i-l]
    # But in many cases, using index convention means w_rev has less right alignment.
    # Easier: we construct two operations as individual transformations.
    
    # Actual way to compute it properly:

    # eval on index shifted out of range for sum ends
    # Use vectorized approach:
    # Translate the lags to indices
    # Genradient w.r.t. h[i]
    # = sum_l w[l] * (1 - h[i+l]) - sum_l w[l] * h[i-l]
    #   = sum_l w[l] - sum_l w[l]*h[i+l] - sum_l w[l]*h[i-l]
    #   = 1 * sum(w) - sum_l w[l]*h[i+l] - sum_l w[l]*h[i-l]
    # let wx = 1..n (reversed)  [but w is n-length, not 2n-1]
    # I just compute it manually using shifts, in a vectorized way.
    # Degenerate case when lags extend outside array => padded with 0
    # For lags:
    grad = np.zeros(n)
    w_shifted = np.zeros(2*n-1) # this will be w.
    w_shifted[n-1:] = w # place vector w at center
    w_rev = w_shifted   # will be shifted so element at index 0 always matches lag = -(n-1)
    # Actually more complex but sure thing:
    w_shifted = np.zeros(2*n - 1)
    # w[l] corresponds to lag l from -n+1 to n-1
    # so for iteration over every lag, effective shift of lag if we do just conv(h, w_rev) why?
    # w_rev[n-1 - offset] = w[-offset]
    # But for convolution (not reversed) it's more complex. Let’s just make a direct loop.

    # Do it similarly to https://stackoverflow.com/a/47060709 or via what the user does:
    # Instead, vectorize by masking, avoids intricate indexing.
    # But better / faster: do loop over lags, vectors, but aggregated in vectorized sum
    # Since O(n^2) is fine for n=240, we'll use explicit loops converting shift to slicing
    # Let us repeat previous notion with variable names explicit
    # w has size 2n-1 = 479 corresponding to lags from -n+1 to n-1

    # Initialize gradient components via loops
    grad = np.zeros(n)
    # compute gradients for each lag l  
    for l in range(-n+1, n):

        # We don't recompute this via convolution because too complex for ease.
        # But if we make vectors shifted correctly:
        if l >= 0:
            a_l = np.concatenate([ np.zeros(l), h[:n-l] ])
        else:
            a_l = np.concatenate([h[-l:], np.zeros(-l) ])
        # h[i-l]
        if l <= 0:  # need shift back from i to i-l, i.e. shift forward by -l
            b_l = np.concatenate([ np.zeros(-l), h[:n-(-l)] ])
        else:  # shift forward by l, pad with zeros 
            b_l = np.concatenate([h[l:], np.zeros(l) ])
        # So weight * (1-a_l) - b_l
        term = w[n-1+l] * (1.0 - a_l) - w[n-1+l] * b_l  # This is not correct.
        # The formula is: for each lag l, contribution to gradient at h[i] = (1-h[i+l]) * w[l] - h[i-l] * w[l]
        # Rearrange: (1-h[i+l]) w[l] - h[i-l] w[l] 
        # i.e. this is w[l] *(1-h[i+l]) - w[l]*h[i-l]
        ind = (np.arange(n) + l) % n  # should work, but indices out of bounds should be truncated
        # From 1 - h[i+l]: only non-zero for l such that i+l < n and i+l >= 0. 
        # Mask fix for indices:
        shifted_h_for = np.full_like(h, 0., dtype=float)
        shifted_h_back = np.full_like(h, 0., dtype=float)
        if l > 0:  # shift h forward by l elements (i.e., h[i+l] = shift[i]) 
           shifted_h_for[l:] = h[:n-l]
        elif l < 0:  # shift h backward by l elements (h[i-l] = shift[i])
            shifted_h_back[:l] = h[-l:]
        else: # l == 0, desired:
            shifted_h_for[:] = h
        # subtract contributions
        grad += w[n-1+l] * (1 - shifted_h_for - shifted_h_back)  

    return grad

def optimize_continuous(h0, remaining_time, n, dx, target_sum):
    """Perform gradient descent starting with h0 and adapted to remaining time."""
    start = time.time()
    best_h = h0.copy()
    best_val = evaluate_c5(h0)
    alpha = 5.0
    h = h0.copy()
    iter_count = 0
    eta_initial = 0.001
    while time.time() - start < remaining_time and iter_count < 5000:
        iter_count += 1
        grad = gradient_of_softmax_max(h, alpha)
        # Adaptive step size with backtracking line search
        eta = eta_initial
        max_iter_linesearch = 20
        for _ in range(max_iter_linesearch):
            h_next = project_to_sum_box(h - eta * grad, target_sum)
            val_next = evaluate_c5(h_next)
            if val_next < best_val - 1e-12:
                h = h_next
                break
            else:
                eta *= 0.5
        if iter_count % 100 == 0:
            alpha = min(alpha + 0.01, 20.0)
        if val_next < best_val:
            best_val = val_next
            best_h = h.copy()
    return best_h, best_val

def binary_local_search(h_bin, remaining_time, n, dx):
    """Refine binary solution with swaps."""
    start = time.time()
    best_h = h_bin.copy()
    best_val = evaluate_c5(best_h)
    while time.time() - start < remaining_time:
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap
        cand = best_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = evaluate_c5(cand)
        if val < best_val - 1e-9:
            best_val = val
            best_h = cand
    return best_h, best_val
