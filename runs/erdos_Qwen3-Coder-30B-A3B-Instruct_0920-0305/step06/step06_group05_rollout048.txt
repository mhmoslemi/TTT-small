import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.linalg import eigh

def compute_corr_fft(h):
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def random_hyperplane_rounding(X_rel, h_rel, M, num_trials=15):
    eigenvals, eigenvecs = eigh(X_rel)
    V_plus = eigenvecs[:, eigenvals > 1e-8]
    candidates = []

    for _ in range(num_trials):
        g = np.random.randn(V_plus.shape[1])
        s = V_plus @ g
        b = (s >= 0).astype(int)
        current_sum = np.sum(b)
        
        if current_sum > M:
            idxs_to_flip = np.random.choice(np.where(b == 1)[0], size=int(current_sum - M), replace=False)
            b[idxs_to_flip] = 0
        elif current_sum < M:
            idxs_to_flip = np.random.choice(np.where(b == 0)[0], size=int(M - current_sum), replace=False)
            b[idxs_to_flip] = 1
            
        candidates.append(b)
        
    return candidates

def threshold_rounding(h_rel, M):
    idxs = np.argsort(h_rel)[::-1]
    b = np.zeros_like(h_rel)
    b[idxs[:M]] = 1
    return [b]

def hill_climb_swap(h_candidate, M, dx):
    n = len(h_candidate)
    h = h_candidate.copy()
    improved = True
    attempts = 0
    max_attempts = 5000

    while improved and attempts < max_attempts:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break

        for i in ones:
            for j in zeros:
                attempts += 1
                if attempts >= max_attempts:
                    break
                
                h[i], h[j] = h[j], h[i]
                corr = compute_corr_fft(h)
                new_c5 = dx * c5_from_corr(corr, n)
                
                if new_c5 < dx * c5_from_corr(compute_corr_fft(h_candidate), n):
                    h_candidate = h.copy()
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved:
                break
    
    return h_candidate

def polish_candidate(h_initial, M, dx):
    bounds = [(0, 1) for _ in range(len(h_initial))]
    
    def objective(h):
        corr = compute_corr_fft(h)
        return dx * c5_from_corr(corr, len(h))
    
    def constraint_eq(h):
        return np.sum(h) - M

    cons = {'type': 'eq', 'fun': constraint_eq}

    try:
        res = minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12},
            callback=lambda x: None
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, M)
            return h
    except:
        pass
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 0
    dx = 0.0

    # Phase A: SDP solving
    sdps = [60, 80, 100, 120]
    solver_time_left = 0.5 * budget_s
    
    for n in sdps:
        if time.time() - start_time > solver_time_left:
            break
            
        M = n // 2
        dx = 2.0 / n
        
        try:
            h = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)
            t = cp.Variable()

            # Constraints
            constraints = [
                cp.sum(h) == M,
                h >= 0,
                h <= 1,
                cp.diag(X) == h,
                X >> 0,
            ]

            # Lag constraints: t >= M - Tr(P_k X) for all k
            for k in range(n):
                P = np.zeros((n, n))
                for i in range(n):
                    if i + k < n:
                        P[i, i+k] = 1
                constraints.append(t >= M - cp.trace(P @ X))

            # Problem
            prob = cp.Problem(cp.Minimize(t), constraints)

            # Solve
            prob.solve(solver=cp.SCS, seconds=solver_time_left - (time.time() - start_time), verbose=False)

            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                X_sol = X.value
                t_sol = t.value

                if t_sol < best_c5 / dx:
                    candidates = random_hyperplane_rounding(X_sol, h_sol, M, 15) + threshold_rounding(h_sol, M)
                    for h_candidate in candidates:
                        h_candidate = hill_climb_swap(h_candidate, M, dx)
                        h_candidate = project_to_sum_box(h_candidate, M)
                        corr = compute_corr_fft(h_candidate)
                        c5_val = dx * c5_from_corr(corr, n)
                        if c5_val < best_c5:
                            best_c5 = c5_val
                            best_h = h_candidate.copy()
                            n_points = n
        except:
            continue

    # Phase B: Rounding + hill climb (if no SDP succeed)
    if best_c5 == float('inf'):
        # Fallback strategy
        n = 100
        dx = 2.0 / n
        M = n // 2
        
        pattern = np.array([1] * (n//2) + [0] * (n//2))
        pattern = np.roll(pattern, 10)
        best_h = pattern.copy()
        corr = compute_corr_fft(best_h)
        best_c5 = dx * c5_from_corr(corr, n)
        n_points = n

    # Phase C: Final polish
    if best_h is not None and time.time() - start_time < budget_s - 10:
        best_h = polish_candidate(best_h, n_points // 2, dx)

    # Final validation
    if best_h is None:
        # Last-ditch fallback with deterministic pattern
        n_points = 100
        dx = 2.0 / n_points
        pattern = np.array([1] * (n_points//2) + [0] * (n_points//2))
        pattern = np.roll(pattern, 10)
        best_h = pattern.copy()
        corr = compute_corr_fft(best_h)
        best_c5 = dx * c5_from_corr(corr, n_points)
        
    return (best_h.tolist(), float(best_c5), n_points)
