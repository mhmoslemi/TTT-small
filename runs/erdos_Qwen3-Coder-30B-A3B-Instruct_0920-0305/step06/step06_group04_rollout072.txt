import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def evaluate_c5_fft_dist(h):
    """Evaluate C5 using FFT correlation for faster computation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.real(np.fft.ifft(np.fft.fft(h[::-1], 2*n - 1) * np.fft.fft(1-h, 2*n - 1)))
    return dx * np.max(corr)

def smooth_max_autocorrelation(h, alpha=0.5):
    """Smooth approximation to the maximum overlap"""
    n = len(h)
    corr = np.real(np.fft.ifft(np.fft.fft(h[::-1], 2*n - 1) * np.fft.fft(h, 2*n - 1)))
    corr = corr[:n]
    # Ensure it's non-negative (correlations should be non-negative)
    corr = np.maximum(corr, 0)
    # Smooth the max using log-sum-exp
    exp_terms = np.exp(-alpha * corr)
    Z = np.sum(exp_terms)
    if Z == 0:
        weights = np.ones_like(corr) / n
    else:
        weights = exp_terms / Z
    
    return weights

def gradient_smooth_max(h, alpha=0.5):
    """Compute gradient of the smooth max, approximating the worst lag"""
    n = len(h)
    corr = np.real(np.fft.ifft(np.fft.fft(h[::-1], 2*n - 1) * np.fft.fft(h, 2*n - 1)))
    corr = corr[:n]
    exp_terms = np.exp(-alpha * corr)
    Z = np.sum(exp_terms)
    if Z == 0:
        weights = np.ones_like(corr) / n
    else:
        weights = exp_terms / Z
    
    # Gradient of  - sum_i w_i * sum_j h_i * h_{i+j} 
    g = np.zeros(n)
    
    # part involving h_i * h_{i+j} for j=1 ... n-1
    for j in range(1, n):
        # left shift h by j
        h_l = np.roll(h, j)
        # right shift h by j  
        h_r = np.roll(h, -j)
        g += weights[j] * (h_l + h_r)
    
    # Also add j=0 contribution (h_i^2) 
    g += 2 * weights[0] * h
    
    # Applying the negative sign because the smoothing brings up the maximum, so gradient is of negative value
    return -g

def line_search(h_start, grad, alpha=0.5, max_iter=8):
    """Perform line search for the best descent step with backtracking"""
    step_size = 0.5
    for _ in range(max_iter):
        h_trial = project_to_sum_box(h_start - step_size * grad, TARGET_SUM)
        cost_trial = evaluate_c5_fft_dist(h_trial)
        if cost_trial < evaluate_c5_fft_dist(h_start):
            return h_trial, cost_trial, step_size
        step_size *= 0.5
    return h_start, evaluate_c5_fft_dist(h_start), 0.0

def random_binary_h(n_points, target_sum):
    """Create a random binary h vector"""
    h = np.zeros(n_points)
    ones_count = int(target_sum)
    ones_idx = np.random.choice(n_points, ones_count, replace=False)
    h[ones_idx] = 1.0
    h = project_to_sum_box(h, target_sum)
    return h

def nearly_binary_h(n_points, target_sum):
    """Create a nearly-binary h vector with few intermediate values"""
    h = np.random.uniform(0, 1, n_points)
    h = project_to_sum_box(h, target_sum)
    # Quantize to 0, 0.5, or 1
    h = np.round(h * 2) / 2.0
    h = project_to_sum_box(h, target_sum)
    return h

def compute_delta_f_binary(h, p, q, lags):
    """Compute change in overlap when swapping positions p and q"""
    # We compute the change for each lag.
    delta_f = np.zeros(len(lags))
    # For each lag, compute how the counts change
    # h[p] and h[q] change from 1 to 0 and 0 to 1 respectively.
    for i, lag in enumerate(lags): 
        # h[p] and h[q] change
        # h[p+lag] and h[q+lag] might change
        if p + lag < len(h) and h[p + lag] == 1:
            delta_f[i] -= 1
        if q + lag < len(h) and h[q + lag] == 1:
            delta_f[i] += 1
        if p - lag >= 0 and h[p - lag] == 1:
            delta_f[i] -= 1
        if q - lag >= 0 and h[q - lag] == 1:
            delta_f[i] += 1
    return delta_f

def binary_swap_phase(h, n_points, max_search_seconds=20):
    """More efficient binary swaps to improve the minimum overlap"""
    # Get n_points
    # Convert to binary
    h_bin = np.round(h).astype(int)
    h_bin = project_to_sum_box(h_bin, TARGET_SUM)
    
    # Compute current outcomes
    corr = np.real(np.fft.ifft(np.fft.fft(h_bin[::-1], 2*n_points - 1) * np.fft.fft(h_bin, 2*n_points - 1)))
    corr = corr[:n_points]
    
    best_c5 = evaluate_c5_fft_dist(h_bin)
    best_h = h_bin.copy()
    
    start_time = time.time()
    
    # For each lag, we want to maximize the overlap
    # But it's computationally expensive to recompute all overlaps
    # So we just keep focusing on the worst lag.
    
    while (time.time() - start_time) < max_search_seconds:
        # Compute current overlaps
        corr = np.real(np.fft.ifft(np.fft.fft(h_bin[::-1], 2*n_points - 1) * np.fft.fft(h_bin, 2*n_points - 1)))
        corr = corr[:n_points]
        
        # Find lag with the smallest overlap
        worst_lag = np.argmin(corr)
        min_overlap = corr[worst_lag]
        
        # Find all positions where there's a 1 in h_bin (h[i] == 1)
        # and h_bin[i+worst_lag] == 0 and i < len(h_bin) - worst_lag
        ones_positions = np.where(h_bin == 1)[0]
        zero_positions = np.where(h_bin == 0)[0]
        
        if len(ones_positions) == 0 or len(zero_positions) == 0:
            break
            
        # Try different combinations
        swap_attempts = 0
        while swap_attempts < 1000:  # Limit to reasonable number of attempts
            p = np.random.choice(ones_positions)
            q = np.random.choice(zero_positions)
            
            if p + worst_lag >= len(h_bin) or h_bin[p + worst_lag] == 0:
                if q + worst_lag < len(h_bin) and h_bin[q + worst_lag] == 1:
                    # Valid swap for the current worst lag if the condition avoids the conflict
                    pass
                else:
                    swap_attempts += 1
                    continue
                    
            # Compute delta in overlaps if we perform the swap
            delta_f = compute_delta_f_binary(h_bin, p, q, range(n_points))
            new_corr = corr + delta_f
            
            new_min_overlap = np.min(new_corr)
            if new_min_overlap > min_overlap:
                # Accept the swap
                h_bin[p] = 0
                h_bin[q] = 1
                corr = new_corr
                min_overlap = new_min_overlap
                break  # Got an improvement, stop
            else:
                # Check if swapping improves the C5 value
                h_trial = h_bin.copy()
                h_trial[p] = 0
                h_trial[q] = 1
                trial_c5 = evaluate_c5_fft_dist(h_trial)
                if trial_c5 < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = trial_c5
                    
            swap_attempts += 1
            
        if swap_attempts >= 1000:
            # Stop if we couldn't find a good swap in 1000 attempts.
            break
    
    return best_h, best_c5
