import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SCS
from numpy.linalg import eig

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def solve_sdp(n, target_sum, max_time_s):
    dx = 2.0 / n
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0, h <= 1,
        cp.diag(X) == h,
        X >> 0
    ]

    # Lag constraints
    for k in range(n):
        P_k = np.zeros((n, n))
        for i in range(n-k):
            P_k[i, i+k] = 1
        constraints.append(t >= target_sum - cp.trace(P_k @ X))

    # Objective
    objective = cp.Minimize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    start = time.time()
    try:
        prob.solve(solver=SCS, 
                   verbose=False,
                   max_iters=int(max_time_s * 1000), 
                   time_limit=max_time_s)
    except:
        pass

    h_opt = h.value
    X_opt = X.value
    t_opt = t.value

    if h_opt is None:
        return None, None, None

    # Return scaled solution
    return h_opt, X_opt, t_opt

def random_rounding(h_rel, X_rel, M, n, num_rounds=15):
    residuals = []
    V, Lambda = eig(X_rel)
    V_positive = V[:, Lambda > 1e-6]
    Lambda_positive = Lambda[Lambda > 1e-6]
    if len(Lambda_positive) == 0:
        V_positive = V
        Lambda_positive = Lambda

    for _ in range(num_rounds):
        g = np.random.randn(V_positive.shape[1])
        s = V_positive @ g
        b = (s >= 0).astype(int)
        b_sum = np.sum(b)
        if b_sum > M:
            flip_indices = np.random.choice(np.where(b == 1)[0], size=b_sum-M, replace=False)
            b[flip_indices] = 0
        elif b_sum < M:
            flip_indices = np.random.choice(np.where(b == 0)[0], size=M-b_sum, replace=False)
            b[flip_indices] = 1
        residuals.append(b)
    return residuals

def threshold_rounding(h_rel, M, n):
    indices = np.argsort(h_rel)[::-1]
    b = np.zeros(n)
    b[indices[:M]] = 1
    return [b]

def hill_climb_swap(h_guess, n, M):
    h = h_guess.copy()
    improved = True
    iteration = 0
    while improved and iteration < 5000:
        improved = False
        for i in range(n):
            if h[i] == 1:
                for j in range(n):
                    if h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                        corr = compute_corr_fft(h)
                        new_c5 = c5_from_corr(corr, n)
                        if new_c5 < evaluate_c5(h_guess):
                            h_guess[:] = h[:]
                            improved = True
                            break
                        h[i], h[j] = 1, 0
                if improved:
                    break
        iteration += 1
    return h_guess

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: SDP solving
    n_values = [60, 80, 100]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5:
            break
        dx = 2.0 / n
        M = n / 2
        remaining_time = time_left / 2.0
        h_opt, X_opt, t_opt = solve_sdp(n, M, min(remaining_time, 100))
        if h_opt is None:
            continue
            
        # Random rounding
        rounding_trials = random_rounding(h_opt, X_opt, M, n, num_rounds=15)
        rounding_trials.extend(threshold_rounding(h_opt, M, n))
        
        for h_rounded in rounding_trials:
            h_rounded = np.clip(h_rounded, 0, 1)
            h_rounded = project_to_sum_box(h_rounded, M)
            
            # Local improvement (swap hill climbing)
            h_refined = hill_climb_swap(h_rounded, n, M)
            
            c5_val = evaluate_c5(h_refined)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_refined
                best_n = n
                
        if time.time() - start_time > budget_s:
            break
            
    # Phase B: SLSQP polishing
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n)]
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - best_n / 2
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if res.success:
                best_h = res.x
                best_h = project_to_sum_box(best_h, best_n / 2)
                best_c5 = evaluate_c5(best_h)
        except:
            pass

    if best_h is None:
        # Fallback strategy
        h_fallback = np.zeros(100)
        h_fallback[:50] = 1
        best_h = h_fallback
        best_c5 = evaluate_c5(best_h)
        best_n = 100

    return (best_h.tolist(), float(best_c5), int(best_n))
