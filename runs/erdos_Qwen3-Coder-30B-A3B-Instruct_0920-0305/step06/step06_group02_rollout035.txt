import numpy as np
import time
from collections import defaultdict

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create an initial binary vector with exactly TARGET_SUM ones"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full overlap array using correlation"""
    return np.correlate(x, x, mode='full')

def update_overlap(ov, i, j, ones, zeros):
    """Update overlap array after swapping x[i] and x[j]"""
    ov_new = ov.copy()
    # Remove contribution of i (all pairs (i,k) and (k,i) for k in ones)
    for k in ones:
        if k != i:
            s = k - i
            ov_new[s + n_points - 1] -= 1
    
    # Add contribution of j (all pairs (j,k) and (k,j) for k in ones \ {i})
    for k in ones:
        if k != i:
            s = k - j
            ov_new[s + n_points - 1] += 1
    return ov_new

def update_sets(ones, zeros, i, j):
    """Update sets after swapping position i and j"""
    ones.remove(i)
    zeros.remove(j)
    ones.add(j)
    zeros.add(i)
    return ones, zeros

def simulated_annealing(x, ov, ones, zeros, time_limit):
    """Simulated annealing with fast incremental updates"""
    start = time.time()
    best_x = x.copy()
    best_ov = ov.copy()
    cur_min = ov.min()
    best_min = cur_min
    
    # Cooling parameters
    T0 = 0.5
    lambda_cool = 0.001
    max_iter = 2_000_000  # Will be limited by time
    iter_count = 0
    
    while time.time() - start < time_limit:
        # Temperature
        t = iter_count / max_iter
        T = T0 * np.exp(-lambda_cool * t)
        
        # Pick random 1 and 0 positions
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        
        # Create new overlap array
        ov_new = update_overlap(best_ov, i, j, ones, zeros)
        new_min = ov_new.min()
        delta = new_min - cur_min
        
        # Accept or reject
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            # Apply swap
            best_x[i] = 0
            best_x[j] = 1
            best_ov = ov_new
            cur_min = new_min
            
            # Update sets
            ones, zeros = update_sets(ones, zeros, i, j)
            
            # Update best
            if new_min > best_min:
                best_min = new_min
                best_x = best_x.copy()
                best_ov = best_ov.copy()
        
        iter_count += 1
    return best_x, best_ov, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine the solution by fixing the worst shift explicitly"""
    min_s = np.argmin(ov)
    worst_shift = min_s - (n_points - 1)
    
    # Improved implementation using sets
    # Get positions where we're missing a 1 to make up for the overlap
    # This is the most beneficial pair to swap given the current worst shift
    same_pos = None
    for i in ones:
        if i + worst_shift < n_points and i + worst_shift >= 0:
            if x[i + worst_shift] == 0:
                # Found a beneficial swap position
                for j in zeros:
                    if j + worst_shift < n_points and j + worst_shift >= 0:
                        if x[j + worst_shift] == 1:
                            # Make a swap that increases the overlap for worst shift
                            # also consider if it worsens any other shift
                            return i, j
    
    # If no place found for beneficial swap, try to improve any shift
    for s in range(len(ov)):
        if ov[s] == np.min(ov):
            continue  # Skip the current minimum shift
        shift = s - (n_points - 1)
        # find indices where 1 and 0 meet
        need_one = []
        need_zero = []
        for i in range(n_points):
            if x[i] == 1 and i + shift < n_points and i + shift >= 0 and x[i + shift] == 0:
                need_one.append(i)
            if x[i] == 0 and i + shift < n_points and i + shift >= 0 and x[i + shift] == 1:
                need_zero.append(i)
        
        if need_one and need_zero:
            # Pick a random beneficial swap
            i = np.random.choice(need_one)
            j = np.random.choice(need_zero)
            return i, j
    
    return None, None

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    
    start = time.time()
    best_c5 = float('inf')
    best_x = None
    
    # Run multiple SA instances with different seeds
    num_restarts = 50
    restart_time = budget_s / num_restarts
    
    for run_id in range(num_restarts):
        t_left = budget_s - (time.time() - start)
        if t_left < 1:
            break
            
        x = initial_binary(seed + run_id)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        cur_min = ov.min()
        
        # Run simulated annealing
        x_final, ov_final, min_final = simulated_annealing(x, ov, ones, zeros, restart_time * 0.9)
        
        # Try worst-shift refinement
        iter_count = 0
        improved = True
        while improved and iter_count < 50 and time.time() - start < budget_s:
            improved = False
            # Let's do this safely without accidentally changing order too much
            i, j = worst_shift_refine(x_final, ov_final, set(np.where(x_final == 1)[0]), set(np.where(x_final == 0)[0]))
            if i is not None and j is not None:
                # Perform correction  
                ov_updated = update_overlap(ov_final, i, j, set(np.where(x_final == 1)[0]), set(np.where(x_final == 0)[0]))
                min_new = ov_updated.min()
                if min_new > min_final:
                    x_final[i] = 0
                    x_final[j] = 1
                    ov_final = ov_updated
                    min_final = min_new
                    improved = True
                    iter_count += 1
                else:
                    break  # Don't accept it if it doesn't help
            else:
                break  # No beneficial swaps found
            
        c5 = dx * (TARGET_SUM - min_final)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x_final.copy()
    
    if best_x is None:
        # fallback to an initial seed
        x = initial_binary(seed)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        best_x, ov_best, best_min = simulated_annealing(x, ov, ones, zeros, budget_s)
        best_c5 = dx * (TARGET_SUM - best_min)
    
    h_values = best_x.astype(float).tolist()
    return (h_values, float(best_c5), n_points)
