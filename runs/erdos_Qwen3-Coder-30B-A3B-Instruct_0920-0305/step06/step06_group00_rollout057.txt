import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h):
    """Compute gradient for each point based on current correlation structure"""
    corr = correlate(h, 1 - h, mode='full')
    max_lag = np.argmax(corr)
    
    # Create gradient based on how moving mass at each position affects the maximum correlation
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For each position, compute the effect of moving mass contributing to the largest correlation
        for j in range(n_points):
            if abs(i - j) == max_lag:
                grad[i] += h[j]
    
    return grad

def adaptive_local_search(h0, max_iter=300):

    best_h = h.copy()

    for iteration in range(max_iter):
        # Compute gradient at current position
        grad = compute_gradient(h)
        
        # Try to improve by moving mass from low-gradient positions to high-gradient ones
        if np.max(grad) > np.min(grad):
            # Move from low to high gradient
            idx_min = np.argmin(grad)
            idx_max = np.argmax(grad)
            if idx_min != idx_max:
                delta = min(h[idx_min], 1 - h[idx_max], 0.02)  # Larger moves
                if delta > 1e-6:
                    h[idx_min] -= delta
                    h[idx_max] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
        else:
            # If gradients are ineffective, use random small perturbations
            i = np.random.choice(n_points)
            j = np.random.choice(n_points)
            if i != j:
                delta = min(h[i], 1 - h[j], 0.01)
                if delta > 1e-6:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
    
    return best_h, best_c5
