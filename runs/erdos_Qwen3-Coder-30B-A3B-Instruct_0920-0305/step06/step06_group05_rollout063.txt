import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def solve_milp(n, time_limit_s):
    """Solve the Erdős minimum overlap problem as an MILP."""
    M = n // 2  # Required number of ones
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Auxiliary variables for AND operations
    # We only need to define y_ij for i < j
    y_vars = {}
    y_list = []
    for i in range(n):
        for j in range(i+1, n):
            y = cp.Variable(boolean=True)
            y_vars[(i,j)] = y
            y_list.append(y)
    
    # Constraints
    constraints = [
        cp.sum(h) == M,  # Exact mass constraint
        t >= 0,  # t is non-negative
    ]
    
    # AND linking constraints
    for i in range(n):
        for j in range(i+1, n):
            y = y_vars[(i,j)]
            constraints.extend([
                y <= h[i],
                y <= h[j],
                y >= h[i] + h[j] - 1
            ])
    
    # Lag constraints
    for k in range(1, n):
        # Sum of y_{i, i+k} for valid i, i+k < n
        lag_sum = cp.sum([y_vars[(i, i+k)] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, eps=1e-6, seconds=time_limit_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h.value).astype(int)
            t_val = t.value
            return h_sol, t_val
    except:
        pass
    
    # If infeasible or timeout, return None
    return None, None

def evaluate_c5_discrete(h):
    """Evaluate C5 using discrete correlation for verification."""
    # The overlap integral is dx * max_k sum(h_i * (1 - h_{i+k}))
    # Using FFT for correlation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def project_to_sum_box(h, target_sum):
    """Project h to sum = target_sum with box constraints h[i] in [0,1]."""
    # This uses a simple iterative approach:
    # Binary search for lambda:
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h_initial, target_sum, max_iter=200):
    """Perform local continuous refinement using SLSQP."""
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        # Return max value scaled by dx
        return np.max(corr) * (2.0 / len(h))

    def constraint_eq(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(len(h_initial))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = cp.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            return res.x
    except:
        pass
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    best_h, best_c5 = None, float('inf')
    best_n = 0
    
    np.random.seed(seed)
    
    # Try different grid sizes
    for n in [60, 70, 80]:
        if time.time() - start_time > 0.9 * budget_s:
            break
        
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break
            
        # Solve the ILP
        h_bin, t_val = solve_milp(n, min(remaining_time * 0.5, 100))
        
        if h_bin is not None:
            # Elect the best binary solution so far
            dx = 2.0 / n
            c5_binary = dx * t_val
            
            # For verification
            h_float = h_bin.astype(float)
            corr_value = evaluate_c5_discrete(h_float) 
            validated_c5 = corr_value * dx
            
            # Try a local continuous refine with SLSQP
            h_refined = local_continuous_refine(h_float, n/2, 200)
            c5_refined = evaluate_c5_discrete(h_refined) * dx
            
            # Update best solution
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                best_n = n
                
    # If no binary solution found, fallback to heuristic method
    if best_h is None:
        # Fallback to previous approach if needed
        # Just call the old approach as a fallback
        from previous_solution import run as fallback_run
        return fallback_run(seed, budget_s)
    
    return (best_h.tolist(), float(best_c5), int(best_n))
