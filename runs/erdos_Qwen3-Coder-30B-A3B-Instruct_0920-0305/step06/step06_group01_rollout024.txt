import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256  # Power of 2 for efficient FFT
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, w):
    """Compute gradient using convolution method"""
    # Build symmetric kernel from weights
    W = np.zeros(2 * n_points - 1)
    for k in range(n_points):
        W[n_points-1+k] = w[k]
        W[n_points-1-k] = w[k]
    
    # Compute convolution
    conv = np.convolve(h, W, mode='full')
    g = conv[n_points-1 : n_points-1 + n_points]
    
    # Add contribution from lag 0
    g += w[n_points-1] * h
    
    return g

def compute_softmin_weights(S, alpha):
    """Compute softmin weights w_k = exp(-alpha * S_k) / sum(exp(-alpha * S_l))"""
    S_max = np.max(-alpha * S)
    exp_neg_alpha_S = np.exp(-alpha * S - S_max)
    return exp_neg_alpha_S / np.sum(exp_neg_alpha_S)

    h0_interp = project_to_sum_box(h0_interp, n_points/2)
    
    # Initial solution
    best_h = h0_interp.copy()
    best_c5 = evaluate_c5(h0_interp)
    TARGET_SUM = n_points / 2
    
    # Optimisation parameters
    step = 0.02
    alpha = 0.5
    alpha_max = 30
    alpha_inc = 1.02
    step_inc = 1.05
    step_dec = 0.5
    restart_interval = 500
    stagnation_counter = 0
    max_iter = 100000  # reasonable upper bound
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute auto-correlations
        S = correlate(best_h, best_h, 'full')
        
        # Compute softmin weights
        w = compute_softmin_weights(S, alpha)
        
        # Compute gradient
        g = compute_gradient(best_h, w)
        
        # Generate candidate
        h_new = best_h + step * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate candidate
        c5_new = evaluate_c5(h_new)
        
        # Accept/reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            step = min(step * step_inc, 0.1)
            stagnation_counter = 0
        else:
            step = max(step * step_dec, 1e-4)
            stagnation_counter += 1
        
        # Temperature annealing
        alpha = min(alpha * alpha_inc, alpha_max)

        if iteration % 200 == 0:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)

            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        # Full restart
        if stagnation_counter >= restart_interval:
            h_restart = np.random.uniform(0, 1, n_points)
            h_restart = project_to_sum_box(h_restart, TARGET_SUM)
            best_h = h_restart
            best_c5 = evaluate_c5(h_restart)
            step = 0.02
            alpha = 0.5
            stagnation_counter = 0

    bounds = [(0, 1) for _ in range(n_points)]

        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}

            objective, 
            best_h,

            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}

            refined_h = res.x
            refined_h = project_to_sum_box(refined_h, TARGET_SUM)
            refined_c5 = evaluate_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
