# Import required libraries
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from itertools import combinations

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return np.max(corr)

def solve_sdp(n, target_sum, time_limit):
    """Solve SDP relaxation for Erdős overlap problem"""
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Diagonal coupling constraint
    diag_constraint = cp.diag(X) == h
    
    # Lag constraints
    constraints = [
        cp.trace(X) == target_sum,  # mass: sum of h entries = target_sum
        cp.diag(h) >= 0,
        cp.diag(h) <= 1,
        X >> 0,  # X is PSD
        t >= 0
    ]
    
    # Add lag constraints for k = 0 to n-1
    for k in range(n):
        P = np.zeros((n, n))
        if k == 0:
            P[np.arange(n), np.arange(n)] = 1
        else:
            # Construct P_k matrix
            for i in range(n - k):
                P[i, i+k] = 1
                P[i+k, i] = 1
        constraints.append(t >= target_sum - cp.trace(P @ X))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, time_limit=time_limit)
    except:
        pass
    
    return h.value, X.value

def round_to_binary(h_rel, target_sum):
    """Round fractional SDP solution to binary vectors"""
    candidates = []
    
    # Threshold rounding
    h_sorted = sorted(enumerate(h_rel), key=lambda x: x[1], reverse=True)
    h_binary = np.zeros_like(h_rel)
    for i in range(target_sum):
        h_binary[h_sorted[i][0]] = 1
    candidates.append(h_binary)
    
    # Hyperplane rounding
    for _ in range(10):
        # Random projection
        g = np.random.normal(0, 1, len(h_rel))
        s = h_rel @ g
        h_binary = (s >= 0).astype(float)
        count = int(np.sum(h_binary))
        if count < target_sum:
            # Flip zeros to ones
            indices = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(indices, target_sum - count, replace=False)
            h_binary[flip_indices] = 1
        elif count > target_sum:
            # Flip ones to zeros
            indices = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(indices, count - target_sum, replace=False)
            h_binary[flip_indices] = 0
        candidates.append(h_binary)
    
    return candidates

def swap_hill_climb(h, target_sum):
    """Perform hill-climbing swaps"""
    improved = True
    while improved:
        improved = False
        corr = compute_corr_fft(h)
        max_k = np.argmax(corr)
        for i in range(len(h)):
            if h[i] == 1:
                for j in range(len(h)):
                    if h[j] == 0:
                        # Swap i and j
                        h_copy = h.copy()
                        h_copy[i] = 0
                        h_copy[j] = 1
                        corr_new = compute_corr_fft(h_copy)
                        max_k_new = np.argmax(corr_new)
                        
                        if corr_new[max_k_new] < corr[max_k]:
                            h = h_copy.copy()
                            corr = corr_new
                            max_k = max_k_new
                            improved = True
                            break
            if improved:
                break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase A: SDP solving
    best_c5 = float('inf')
    best_h = None
    n_points = 100
    dx = 2.0 / n_points
    
    # Time allocation for SDP solving
    sdp_time = budget_s * 0.5
    time_remaining = sdp_time
    
    for n in [60, 80, 100]:
        if time.time() - start_time > budget_s - 5:
            break
            
        t_start = time.time()
        h_rel, X_rel = solve_sdp(n, n / 2, time_remaining / 3)
        sdp_elapsed = time.time() - t_start
        
        if h_rel is None:
            continue
            
        # Rounding
        candidates = round_to_binary(h_rel, n / 2)
        
        # Hedge against numerical noise
        for i, h in enumerate(candidates):
            h = project_to_sum_box(h, n / 2)
            
            # Hill-climbing
            h = swap_hill_climb(h, n / 2)
            
            # Evaluation
            c5 = evaluate_c5(h)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        time_remaining -= sdp_elapsed
        if time.time() - start_time > budget_s - 5:
            break
    
    # Phase B: Final polishing with SLSQP
    if time.time() - start_time < budget_s - 10 and best_h is not None:
        try:
            h_initial = best_h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            res = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12})
            
            if res.success:
                h_final = res.x
                h_final = project_to_sum_box(h_final, n_points / 2)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except:
            pass
    
    # Return final result
    if best_h is None:
        # Fallback construction
        h_fallback = np.zeros(n_points)
        h_fallback[:n_points//2] = 1
        best_h = h_fallback
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5 * dx), n_points)
