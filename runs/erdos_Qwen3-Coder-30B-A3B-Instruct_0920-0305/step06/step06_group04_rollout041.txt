import numpy as np

def compute_c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return compute_c5_from_corr(corr, n)

def project_to_sum_box(x, target_sum):
    """Project to sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def smoothed_gradient(h, alpha=1.0):
    """Compute gradient of smooth max using log-sum-exp"""
    n = len(h)
    # Compute auto_corr = correlation of h with itself
    h_rev = h[::-1]
    auto_corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(h, 2*n - 1)))

    # Extract the auto-correlation for shifts 0 to N-1 
    # Index N-1 corresponds to shift 0 (lag = 0), N to N+N-2 = 2N-2 (shift N-1)  
    auto_corr_lags_0_to_N_minus_1 = auto_corr[N-1:2*N-1]  

    # Compute the smoothed approx max of -auto_corr_lags_0_to_N_minus_1
    # Gradient of -log(sum(exp(-alpha * auto_corr))) with respect to h_i
    # = -(1/sum(exp(-alpha * auto_corr))) * (sum_i -alpha * exp(-alpha * auto_corr_i) * ∂auto_corr_i/∂h_i)  
    # So we need to compute ∂auto_corr_i / ∂h_i  
    # Which depends on h on i-th position and neighbor positions due to fft multiplication.  
    # or I can think of auto_corr term as sum_i h_i h_{i+k}  
    # So ∂ auto_corr_k / ∂ h_i =  h_{i+k}, 形好似 for i=k (someone else makes it 1 or 0)  
    # Counter part ∂ auto_corr_k / ∂ h_{i+k} = h_i  

    # It's more direct to compute the coefficients directly:
    # df/dh_i = sum_k w_k (h_{i+k} + h_{i-k}) where w_k = exp(-alpha* auto_corr_k) / Z  
    # We'll compute the logsumexp components directly.
    # Compute the weights: w_k = exp(-alpha*auto_corr_k)  
    # Should compute all first so their sum is normalizer so that warmed = 1

    Z = np.sum(np.exp(-alpha * auto_corr_lags_0_to_N_minus_1))
    if Z == 0: Z = 1e-15   # avoid division by zero
    
    # This is our weight vector
    warm = np.exp(-alpha * auto_corr_lags_0_to_N_minus_1) / Z
    
    # Now compute gradient
    # This is simplified gradient terms: (each contributing group of h_{i+k} and h_{i-k} ) 
    # For simplicity of implementation, use naive loop to compute all terms:  
    # For each index i, compute the gradient contribution:  
    # we want sum over k of w_k* (h_{i+k} + h_{i-k})  
    # But we work with indices. 
    # Note the weighting is only for the lags from 0 to N-1
    # indexing of auto_corr: ranges from 0 to 2N-2. 
    # Index N-1 to 2N-2 correspond to lags 0 to N-1 respectively.
    # We keep this structure for code simplicity. We cache result though:
    
    # Let’s compute the full gradient in expanded loops - tally contributions:
    grad = np.zeros(n)
    for k in range(n):   # iterating over lags 0 to N-1
        w = warm[k] 
        # k corresponds to auto_corr index k + N-1. 
        for i in range(n):  
            # Add term from i being matched with i+k (or i-k due to symmetry in pairs)
            # Let’s actually compute the bilinear matrix gradient by loop 
            # Refer to basic bilinear gradient:
            # f_k = ∑ h_i h_{i+k} , gradient w.r.t h_i = h_{i+k} + h_{i-k}
            left_val = h[(i - k) % n] if (i-k >=0) else 0  
            right_val = h[(i + k) % n] if (i + k < n) else 0  
            
            # Hmm wrong indexing parts, let’s work out better process
            # direct approach:
            # The auto_corr indexed at [N-1+k] = ∫ h(x) h(x+k) dx  
            # There are contributions to gradient of term ∑ h_i h_{i+k} 
            # d (h_i h_{i+k})/dh_i = h_{i+k}
            # d (h_i h_{i+k}) / dh_{i+k} = h_i  
            # Thus gradient term for h_i is: ∑_{k=0}^{N-1} w_k (h_{i+k} + h_{i-k})
            # 
            # Let’s compute neighbor indices w.r.t k shifts:
            # For fixed i:
            #    i is term contributing to auto_corr_lags_0_to_N_minus_1[0] (k=0)
            #    i+k is weight to auto_corr_lags_0_to_N_minus_1[k] (k=0 to N-1)  
            #    i-k is negative k regardless of range needed.
            #       
            # so for index i in gradient, find all auto_corr contributing to i via h_i = h_{i-k} for k=0 to N-1?
            # let’s fix legacy (symbols above), detailed:
            # index i in h, when computing auto_corr_k = ∫ h(x) h(x+k) dx for k = 0 to N-1,  
            # i*k = h_i h_{i+k}.  
            # So we want with respect to h_i, we have contribution in k terms:  
            # ∑_{k=0}^{N-1} w_k h_{i+k}  
            # ∑_{k=0}^{N-1} w_k h_{i-k}  # account symmetry
            # Total = ∑_k w_k (h_{i+k} + h_{i-k})
            # Wrap around indices to preserve valid range of indices
            if k != 0:   
                left = h[(i - k) % n]  # h_{i-k}  
                right = h[(i + k) % n]  # h_{i+k} 
                grad[i] += -w * (left + right)
            else:  # k=0 is special, h_i h_i = h_i^2  
                grad[i] += - 2 * w * (h[i])

    return grad
