# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorr(h):
    """Compute full autocorrelation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_worst_lag(h):
    """Find the worst (maximum) lag"""
    corr = compute_autocorr(h)
    return np.argmax(corr)

def swap_bits(h, i, j):
    """Perform a swap between positions i and j"""
    h = h.copy()
    h[i], h[j] = h[j], h[i]
    return h

def block_swap(h, s, l):
    """Perform a block swap of length l starting at position s"""
    h = h.copy()
    if s + l > len(h):
        return h
    # Flip the block
    for i in range(s, s+l):
        h[i] = 1 - h[i]
    # Fix weights
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    if len(ones_indices) > TARGET_SUM:
        # Too many ones, make some zeros
        excess = len(ones_indices) - TARGET_SUM
        # Pick random ones to flip to zeros
        to_flip = np.random.choice(ones_indices, size=excess, replace=False)
        for idx in to_flip:
            h[idx] = 0
    elif len(ones_indices) < TARGET_SUM:
        # Not enough ones
        deficit = TARGET_SUM - len(ones_indices)
        # Pick random zeros to flip to ones
        to_flip = np.random.choice(zeros_indices, size=deficit, replace=False)
        for idx in to_flip:
            h[idx] = 1
    return h

def elite_pool_append(elites, h, c5_val):
    """Add a new solution to the elite pool, keeping the best"""
    elites.append((h.copy(), c5_val))
    elites.sort(key=lambda x: x[1])
    return elites[:5]
