def evaluate_c5(h):
    """Fast evaluation of C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_based_refinement(h0, max_iter=1000):

        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        # Compute gradient estimate
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Estimate effect of shifting mass from i to nearby points
            for j in range(n_points):
                lag = abs(i - j)
                if lag == max_lag:
                    grad[i] += h[j] * (1 - 2 * h[i])  # Approximate gradient
        
        # Apply gradient descent step
        step_size = 1e-3
        h_new = h - step_size * grad

        if best_c5 < 0.3795:
            break
