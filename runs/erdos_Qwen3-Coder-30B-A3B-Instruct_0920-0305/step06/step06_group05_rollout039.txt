import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def solve_sdp(n, target_sum, max_time=30.0):
    """Solve SDP relaxation for given n"""
    dx = 2.0 / n
    M = target_sum
    
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.trace(X) == M,  # Tr(X) = M
        cp.diag(X) == h,   # diag(X) = h
        X >> 0,            # X is psd
        cp.sum(h) == M,    # sum(h) = M
        0 <= h, h <= 1,    # box constraints
        t >= M - cp.trace(P @ X) for P in generate_lag_matrices(n)
    ]

    # Objective
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Solve with time limit
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000, time_limit=max_time)
    except:
        pass
        
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_rel = h.value
        X_rel = X.value
    else:
        # fallback: use the last available iterates
        h_rel = h.value if h.value is not None else np.ones(n) * 0.5
        X_rel = X.value if X.value is not None else np.eye(n) * 0.5

    return h_rel, X_rel

def generate_lag_matrices(n):
    """Generate matrices P_k for different lags""" 
    P_list = []
    for k in range(n):
        P = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                P[i, i + k] = 1
        P_list.append(P)
    return P_list

def rounding_routine(h_rel, X_rel, M, n_rounds=20):
    """Perform random hyperplane rounding"""
    candidates = []
    
    # Deterministic threshold rounding
    h_threshold = np.zeros(n)
    sorted_indices = np.argsort(-h_rel)
    h_threshold[sorted_indices[:M]] = 1
    candidates.append(h_threshold.copy())
    
    # Random hyperplane rounding
    try:
        # Factorize X_rel
        w, v = np.linalg.eigh(X_rel)
        v = v[:, w > 1e-6]
        r = v.shape[1]
        
        for _ in range(n_rounds):
            # sample Gaussian vector
            g = np.random.randn(r)
            scores = v @ g
            
            # round to binary
            h_bin = (scores >= 0).astype(float)
            
            # repair weight
            current_sum = np.sum(h_bin)
            if abs(current_sum - M) > 0.01 * M:
                if current_sum > M:
                    # flip some 1s to 0s
                    flip_indices = np.random.choice(np.where(h_bin == 1)[0], 
                                                   size=int(current_sum - M), 
                                                   replace=False)
                    h_bin[flip_indices] = 0
                else:
                    # flip some 0s to 1s
                    flip_indices = np.random.choice(np.where(h_bin == 0)[0], 
                                                   size=int(M - current_sum), 
                                                   replace=False)
                    h_bin[flip_indices] = 1
            
            candidates.append(h_bin)
    except:
        pass
    
    return candidates

def hill_climb_swap(h_bin, n):
    """Perform deterministic hill climb using swaps"""
    h = h_bin.copy()
    improved = True
    max_swaps = 10000
    
    while improved and max_swaps > 0:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_c5 = evaluate_c5(h)
        
        for i in ones:
            for j in zeros:
                # swap
                h[i], h[j] = h[j], h[i]
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    improved = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]  # undo
            if improved:
                break
            max_swaps -= 1
    
    return h, best_c5

def polish_with_slsqp(h_bin, M, dx, max_iter=150):
    """Apply local SLSQP refinement."""
    h = h_bin.copy()
    
    def objective(x):
        return dx * evaluate_c5(x)
        
    def constraint_eq(x):
        return np.sum(x) - M
        
    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h = res.x
    except:
        pass
    
    h = project_to_sum_box(h, M)
    return h, evaluate_c5(h) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    n_points = 80  
    dx = 2.0 / n_points
    M = n_points / 2  # target integral 
    remaining_time = budget_s
    
    # Phase A: SDP solving (0.5 budget = 300s)
    phase_a_end = start_time + 0.5 * budget_s
    n_values = [60, 80, 100]
    n_to_try = [n for n in n_values if time.time() < phase_a_end]
    
    for n in n_to_try:
        remaining_time = phase_a_end - time.time()
        if remaining_time < 30:
            break
            
        h_rel, X_rel = solve_sdp(n, M, remaining_time * 0.9)
        h_candidates = rounding_routine(h_rel, X_rel, M, n_rounds=20)
        
        for h in h_candidates:
            h_hill, c5 = hill_climb_swap(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_hill.copy()
    
    # Phase B: Post-processing polishing
    phase_b_end = start_time + 0.8 * budget_s
    if best_h is not None:
        best_h, best_c5 = polish_with_slsqp(best_h, M, dx, 150)
    
    # Phase C: Final fallback (ensures result)
    if best_h is None:
        # Simple construction: Alternate blocks of equal length
        a = int(np.ceil(n_points / 2))
        b = n_points - a
        h = np.zeros(n_points)
        h[:a] = 1
        best_h = h / np.sum(h) * M
        best_c5 = evaluate_c5(best_h) * dx
    
    # Final round-trip
    final_h = project_to_sum_box(best_h, M)
    final_c5 = evaluate_c5(final_h) * dx
    
    return (final_h.tolist(), float(final_c5), len(final_h))
