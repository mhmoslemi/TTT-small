TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def project_to_sum_box(x, target_sum):
    """Use water-filling to project to sum = target_sum with box [0,1]"""
    x = np.array(x)
    # Simple iterative approach
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return x
    # We can use a lambda search with bisection
    low, high = -1.0, 1.0
    for _ in range(50):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        new_sum = np.sum(y)
        if abs(new_sum - target_sum) < 1e-10:
            return y
        elif new_sum > target_sum:
            high = mid
        else:
            low = mid
    y = np.clip(x + low, 0.0, 1.0)
    return y

def fourier_to_h(theta, n_points, n_fine=800):
    """Generate step function from spectral coefficients (theta)"""
    K = len(theta) // 2  # number of harmonics
    a, b = theta[:K], theta[K:]
    
    # Generate fine grid
    x_fine = np.linspace(0, 2, n_fine)
    
    # Compute periodic signal using truncated Fourier series
    h = 0.5 * np.ones_like(x_fine)
    for k in range(1, K + 1):
        h += a[k - 1] * np.cos(np.pi * k * x_fine) + b[k - 1] * np.sin(np.pi * k * x_fine)
    
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    
    # Project to correct sum
    h = project_to_sum_box(h, TARGET_SUM * n_fine / n_points)
    
    # Downsample to final n_points
    step = n_fine // n_points
    h = h[::step]
    
    return h

def compute_c5_from_corr(corr, n):
    """Compute C5 from full correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(theta, n_points=120, n_fine=800):
    """Evaluate C5 given theta (Fourier coefficients)"""
    h = fourier_to_h(theta, n_points, n_fine)
    corr = correlate(h, 1 - h, mode='full')
    return compute_c5_from_corr(corr, n_points)
