import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def solve_milp(n, time_limit):
    """Solve the ILP formulation for binary h to minimize C5"""
    M = n // 2  # target sum
    dx = 2.0 / n
    
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(pos=True)
    
    # Auxiliary variables for AND operations
    y = {}
    for i in range(n):
        for j in range(i+1, n):
            y[(i,j)] = cp.Variable(boolean=True)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == M)
    
    # AND linking constraints
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(y[(i,j)] <= h[i])
            constraints.append(y[(i,j)] <= h[j])
            constraints.append(y[(i,j)] >= h[i] + h[j] - 1)
    
    # Lag constraints
    for k in range(1, n):
        # Sum of y_{i, i+k} over valid i
        lag_sum = cp.sum([y[(i, i+k)] for i in range(n-k)])
        constraints.append(t >= M - lag_sum)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   cbc_params={"seconds": time_limit})
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.rint(h.value).astype(int)
            t_val = t.value
            return h_opt, t_val
        else:
            # Try to get a feasible solution
            h_opt = np.rint(h.value).astype(int)
            t_val = t.value
            return h_opt, t_val
    except:
        return None, None

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_continuous_refine(h):
    """Refine the binary solution with continuous optimization"""
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    bounds = [(0, 1) for _ in range(n)]
    
    def constraint_eq(h):
        return np.sum(h) - M
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, M)
            return h
    except:
        pass
    return h
