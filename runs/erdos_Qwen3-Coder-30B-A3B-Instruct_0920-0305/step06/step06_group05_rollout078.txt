import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp
from numpy.linalg import eig


def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result


def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr


def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return 2.0 / n * np.max(corr)


def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))


def solve_sdp(n, dx):
    M = n // 2
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == M,
        0 <= h,
        h <= 1,
        cp.diag(X) == h,
        X >> 0  # PSD constraint
    ]

    # Lag constraints -- creating P_k matrices and enforcing t >= M - Tr(P_k X)
    for k in range(n):
        Pk = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                Pk[i, i + k] = 1
        constraints.append(t >= M - cp.trace(Pk @ X))

    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)

    prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_opt = h.value
        X_opt = X.value
        t_opt = t.value
        return h_opt, X_opt, t_opt
    else:
        # fallback to best feasible point if not optimal
        h_opt = h.value if h.value is not None else np.array([0.5] * n)
        X_opt = X.value if X.value is not None else np.eye(n)
        t_opt = t.value if t.value is not None else 1.0
        return h_opt, X_opt, t_opt


def threshold_round(h_rel, M):
    # sort indices and pick largest M
    sorted_indices = np.argsort(h_rel)[::-1]
    b = np.zeros_like(h_rel)
    b[sorted_indices[:M]] = 1
    # adjust to exact weight if needed
    diff = np.sum(b) - M
    if diff > 0:
        zero_indices = np.where(b == 0)[0]
        mask = np.random.choice(len(zero_indices), int(diff), replace=False)
        b[zero_indices[mask]] = 1
    elif diff < 0:
        one_indices = np.where(b == 1)[0]
        mask = np.random.choice(len(one_indices), int(-diff), replace=False)
        b[one_indices[mask]] = 0
    return b


def hyperplane_round(h_rel, X_rel, M, num_trials=15):
    # use eigendecomposition to get a low-rank factorization of X
    U, s, Vt = np.linalg.svd(X_rel, full_matrices=False)
    r = np.sum(s > 1e-10)
    V_plus = U[:, :r]
    g = np.random.randn(r)
    s = V_plus @ g
    
    # convert scores to binary vector
    b = (s >= 0).astype(float)
    # ensure correct weight
    diff = np.sum(b) - M
    if diff > 0:
        zero_indices = np.where(b == 0)[0]
        mask = np.random.choice(len(zero_indices), int(diff), replace=False)
        b[zero_indices[mask]] = 1
    elif diff < 0:
        one_indices = np.where(b == 1)[0]
        mask = np.random.choice(len(one_indices), int(-diff), replace=False)
        b[one_indices[mask]] = 0
    return b


def hill_climb(h_candidate, M):
    n = len(h_candidate)
    # Convert integer to indices
    ones = list(np.where(h_candidate > 0.5)[0])
    zeros = list(np.where(h_candidate < 0.5)[0])
    
    improved = True
    best = h_candidate.copy()
    best_c5 = evaluate_c5(h_candidate)
    
    iterations = 0
    max_iter = 5000
    while improved and iterations < max_iter:
        improved = False
        batch = []
        for i in ones:
            for j in zeros:
                batch.append((i, j))
        
        np.random.shuffle(batch)
        for i, j in batch:
            # make swap
            h_test = h_candidate.copy()
            h_test[i] = 0
            h_test[j] = 1
            c5 = evaluate_c5(h_test)
            if c5 < best_c5:
                best = h_test
                best_c5 = c5
                improved = True
                h_candidate = h_test.copy()
                ones = list(np.where(h_candidate > 0.5)[0])
                zeros = list(np.where(h_candidate < 0.5)[0])
                break
        iterations += 1

    return best, best_c5


def polish(h_bin, M, dx):
    bounds = [(0, 1) for _ in range(len(h_bin))]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - M
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective,
            h_bin,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, M)
            return h
    except:
        pass
    return h_bin


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_values = [60, 80, 100]  #, 120
    dx = 2.0 / 100
    M = 50
    best_h = None
    best_c5 = float('inf')
    
    # Phase A: SDP solving
    phase_a_end = start_time + 0.5 * budget_s
    h_dict = {}
    for n in n_values:
        if time.time() >= phase_a_end:
            break
        dx = 2.0 / n
        try:
            h_rel, X_rel, t_opt = solve_sdp(n, dx)
            h_dict[n] = (h_rel, X_rel, t_opt)
        except Exception as e:
            pass

    # Phase B: Rounding & hill climbing
    phase_b_end = start_time + 0.85 * budget_s
    candidates = []

    for n, (h_rel, X_rel, t_opt) in h_dict.items():
        if time.time() >= phase_b_end:
            break
        dx = 2.0 / n
        # deterministic threshold rounding
        b_thresh = threshold_round(h_rel, M)
        # try several hyperplane rounds
        for _ in range(15):
            if time.time() >= phase_b_end:
                break
            b_random = hyperplane_round(h_rel, X_rel, M)
            h_rounded = b_random
            h_rounded, c5 = hill_climb(h_rounded, M)
            candidates.append((h_rounded, c5, n))

    # Final candidates from best SDP solutions or earlier stages
    for h_candidate, c5, n in candidates:
        if c5 < best_c5:
            best_h = h_candidate
            best_c5 = c5

    # Phase C: Polishing
    if best_h is not None:
        dx = 2.0 / len(best_h)
        return (best_h.tolist(), float(best_c5), len(best_h))

    # Fallback: deterministic construction
    h_fallback = np.zeros(100)
    h_fallback[:50] = 1
    h_fallback = project_to_sum_box(h_fallback, 50)
    h_fallback, best_c5 = hill_climb(h_fallback, 50)
    return (h_fallback.tolist(), float(best_c5), 100)
