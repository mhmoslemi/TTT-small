def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def local_refine(h):

        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h

def evaluate_c5_integer(h):
    """Exact evaluation with integer arithmetic to avoid FP errors"""
    h_int = np.round(h).astype(int)
    # Ensure the sum constraint
    current_sum = np.sum(h_int)
    if current_sum != TARGET_SUM:
        # Find positions to adjust
        diff = int(TARGET_SUM - current_sum)
        if diff > 0:
            # Need to flip zeros to ones
            zero_indices = np.where(h_int == 0)[0]
            flip_indices = zero_indices[:diff]
            h_int[flip_indices] = 1
        else:
            # Need to flip ones to zeros
            one_indices = np.where(h_int == 1)[0]
            flip_indices = one_indices[:abs(diff)]
            h_int[flip_indices] = 0
            
    # Compute correlation
    corr = np.correlate(h_int, 1 - h_int, mode='full')
    return dx * np.max(corr)
