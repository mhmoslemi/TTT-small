import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

TARGET_SUM = 60.0  # n=120, so target sum = 120/2 = 60

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    # Reverse h
    h_rev = h[::-1]
    # Compute b = 1 - h
    b = 1.0 - h
    # Full convolution using FFT
    corr = np.real(np.fft.ifft(np.fft.fft(h_rev, 2*n - 1) * np.fft.fft(b, 2*n - 1)))
    return corr

def c5_from_corr(corr, n):
    """Compute C5 from correlation"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Shift-and-clip projection to enforce sum = target_sum, box constraints [0,1]"""
    # Binary search for lambda to make sum equal to target_sum
    low, high = -1.0, 1.0
    tol = 1e-12
    max_iter = 50
    
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        y = np.clip(x + mid, 0, 1)
        sum_y = np.sum(y)
        if abs(sum_y - target_sum) < tol:
            return y
        elif sum_y > target_sum:
            high = mid
        else:
            low = mid
    # If we don't converge, return a very close approximation
    y = np.clip(x + low, 0, 1)
    return y

def compute_gradient(h, alpha=10.0):
    """Compute smoothed gradient of min overlap"""
    n = len(h)
    # Compute all correlations: f_ell = sum h_i h_{i+ell}
    f_corr = np.correlate(h, h, mode='full')
    # Only keep positive lag correlations (0 to n-1)
    f_vals = f_corr[n-1:]
    
    # Compute weights for each lag
    # Smooth log-sum-exp approximation to max
    weights = np.exp(-alpha * f_vals)
    total_weight = np.sum(weights)
    if total_weight > 0:
        weights /= total_weight
    
    # Compute gradient: - sum_{ell} weights[ell] * (h[i+ell] + h[i-ell])
    grad = np.zeros(n)
    
    # For each lag, add contribution from shifted vectors
    # Reverse the computation to avoid multiple passes
    for ell in range(n):
        w = weights[ell]
        if ell > 0:
            # h[i+ell] contribution (forward shift)
            if ell < n:
                grad[:n-ell] += -w * h[ell:]
            # h[i-ell] contribution (backward shift)
            if ell < n:
                grad[ell:] += -w * h[:n-ell]
        else:
            # ell == 0 case
            grad += -2 * w * h
    
    return grad

def line_search_step(h, grad, step_size=0.01):
    """Perform line search along the gradient to find a good step size."""
    # Try several step sizes and pick best one
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    current_alpha = step_size
    for _ in range(10):  # Max 10 step sizes
        new_h = h - current_alpha * grad
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        c5 = evaluate_c5(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        if current_alpha < 1e-8:
            break
        current_alpha *= 0.5
    
    return best_h, best_c5

def local_refine(h_start):
    """Refine solution using SLSQP optimizer"""
    def objective(h):
        return evaluate_c5(h)
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    bounds = [(0, 1) for _ in range(len(h_start))]
    
    try:
        res = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            c5_refined = evaluate_c5(h_refined)
            return h_refined, c5_refined
    except:
        pass
    return h_start, evaluate_c5(h_start)
