import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def evaluate_c5_binary(x):
    """Fast evaluation of C5 for binary vector"""
    # Compute correlation using FFT of x and (1-x)
    corr = np.correlate(x, 1 - x, mode='full')
    # The max correlation is the max value of the correlation array
    return dx * np.max(corr)

def compute_convolution_fast(x):
    """Compute convolution array for binary x to determine overlap counts"""
    # The convolution in numpy terms gives correlation
    corr = np.correlate(x, x, mode='full')
    # For full mode, it is of length 2*n-1, centered at index n-1
    # corr[i] = sum_k x[k] * x[k-i] 
    # The contribution for shift s is at index s + (n-1) = s + n - 1
    return corr

def apply_swap_conv(conv, x, ones_list, i, j, n):
    """Incremental update of convolution array when swapping elements i,j"""
    # x[i]=1, x[j]=0 -> becomes x[i]=0, x[j]=1
    # These are the indices that are switched from 1 to 0 and 0 to 1
    
    # Store current conv as reference to revert if needed
    old_conv = conv.copy()

    # Remove contributions to conv (from indices for which x[i] was 1)
    # Updated: for each k in ones_list, adjust conv contributions
    for k in ones_list:
        shift = k - i
        # compute actual index in conv array
        idx = shift + (n - 1)
        if k == i:
            # Only self-pair contributes once
            conv[idx] -= 1
        else:
            # Pair contributes twice: once in offset, once negatively
            conv[idx] -= 1
            # symmetric index for -shift
            conv[-shift + (n - 1)] -= 1

    # Remove i from ones_list
    try:
        ones_list.remove(i)
    except ValueError:
        pass

    # Add j to ones_list
    ones_list.append(j)

    # Now add contributions for j being new 1
    # In a binary vector, we count ~pairs that contribute to each shift
    # After swapping i (to 0) and j (to 1):
    # Original contribution removed from i
    # New contribution added from j
    
    # For each k still in ones_list (and also j we just added), 
    # adjust the convolution 
    for k in ones_list:  
        shift = k - j
        idx = shift + (n - 1)
        if k == j:
            conv[idx] += 1
        else:
            conv[idx] += 1
            # symmetric -shift contribution
            conv[-shift + (n - 1)] += 1

    # Hyperparameter for SA cooling rate
    return old_conv

def get_binary_from_solution(h):
    """Convert continuous solution h to binary"""
    # Get 1's at the random threshold (could be 0.5 or better)
    threshold = 0.5
    return (h > threshold).astype(int)

def initialize_random_binary(n, target_sum):
    """Create random binary initialization"""
    x = np.zeros(n, dtype=int)
    ones_indices = np.random.choice(range(n), target_sum, replace=False)
    x[ones_indices] = 1
    return x, list(ones_indices)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points // 2
    
    best_h = None
    best_c5 = float('inf')
    
    # Use large number of restarts
    restarts = 50
    
    for restart in range(restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Initialize binary vector with uniform random weights
        x, ones_list = initialize_random_binary(n_points, TARGET_SUM)
        ones_set = set(ones_list)  # for efficient membership check
        
        # Compute initial convolution
        conv = compute_convolution_fast(x)
        min_overlap = np.min(conv)
        current_c5 = (TARGET_SUM - min_overlap) * dx
        
        # Simulated Annealing parameters
        T_initial = 0.1
        T_final = 0.0001
        iterations = 10000
        
        # Track current state
        cur_min_overlap = min_overlap
        cur_c5 = current_c5
        
        # Track best solution seen so far in this restart
        best_min_overlap_in_restart = cur_min_overlap
        best_c5_in_restart = cur_c5
        best_x_in_restart = x.copy()
        best_ones_in_restart = ones_list.copy()
        
        for i in range(iterations):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Temperature
            T = T_initial * (T_final / T_initial) ** (i / iterations)
            
            # Make a swap between two positions: one with 1, one with 0
            # Efficient sampling from list of ones
            i_pos = random.choice(ones_list)
            # Find zero positions
            zero_indices = [k for k in range(n_points) if x[k] == 0]
            if not zero_indices:
                continue
            j_pos = random.choice(zero_indices)
            
            # Update convolution efficiently
            old_conv = conv.copy()
            
            # Apply swap and update convolution
            for k in ones_list:
                shift = k - i_pos
                idx = shift + (n_points - 1)
                if k == i_pos:
                    conv[idx] -= 1
                else:
                    conv[idx] -= 1
                    conv[-shift + (n_points - 1)] -= 1
            ones_list.remove(i_pos)
            ones_list.append(j_pos)
            
            for k in ones_list:  
                shift = k - j_pos
                idx = shift + (n_points - 1)
                if k == j_pos:
                    conv[idx] += 1
                else:
                    conv[idx] += 1
                    conv[-shift + (n_points - 1)] += 1
            conv[n_points - 1] += 1  # self-pair contribution
            
            # Compute new overlap value  
            new_min_overlap = np.min(conv)
            new_c5 = (TARGET_SUM - new_min_overlap) * dx
            
            # Accept or reject the move
            delta = new_min_overlap - cur_min_overlap
            if delta > 0 or random.random() < np.exp(delta / T):
                # Accept the move
                cur_min_overlap = new_min_overlap
                cur_c5 = new_c5
                if cur_c5 < best_c5_in_restart:
                    best_c5_in_restart = cur_c5
                    best_x_in_restart = x.copy()
                    best_ones_in_restart = ones_list.copy()
            else: 
                # Revert the changes, restore previous conv state
                conv[:] = old_conv[:]  
                # Correct ones list byUndoing the actual swap were not accepted
                # These operations are already taken on the list & restored at revert
                
            # If improvement, save best of current restart
            if cur_c5 < best_c5_in_restart:
                best_c5_in_restart = cur_c5
                best_x_in_restart = x.copy()
                best_ones_in_restart = ones_list.copy()
                
        # After iteration, check restart for best improvement
        if best_c5_in_restart < best_c5:
            best_c5 = best_c5_in_restart
            best_h = best_x_in_restart.copy()
    
    # If no solution found, return a simple heuristic
    if best_h is None:
        h_uniform = np.full(n_points, 0.5)
        best_c5 = evaluate_c5(h_uniform)
        best_h = h_uniform.tolist()
    else:
        # Convert to float array as required
        best_h = (best_h == 1).astype(float).tolist()
        
    return (best_h, float(best_c5), n_points)
