import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 120
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Initialize with a better starting value from `initial_h_values`
    h0 = np.array(initial_h_values)
    x_orig = np.linspace(0, 2, len(h0))
    x_new = np.linspace(0, 2, n_points)
    h_interp_0 = np.interp(x_new, x_orig, h0)
    h0 = np.clip(h_interp_0, 0, 1)
    h0 = h0 * target_sum / np.sum(h0)
    
    # Use the MILP approach based on binary variables with product constraints
    best_h = None
    best_c5 = float('inf')

    # Attempt solving as a Binary Integer Program (MILP)
    try:
        # Create binary variables h_i
        h = cp.Variable(n_points, boolean=True)
        # Lambda is the minimal overlap value to be maximized
        lambda_var = cp.Variable(integer=True)
        
        # Lag-Overlap constraints: enforce sum(y_i,l) >= lambda for each l
        constraints = []
        # Add total mass constraint
        constraints.append(cp.sum(h) == target_sum)
        # Add bounds for lambda (use the fact that minimum overlap among all lags is at least 0)
        constraints.append(lambda_var >= 0)
        constraints.append(lambda_var <= target_sum)

        # Introduce auxiliary binary variables for products h_i * h_{i+l}
        y_vars = {}  # Map (i,l) to the binary variable y_{i,l}
        
        for l in range(n_points):
            for i in range(n_points - l):
                y_vars[(i,l)] = cp.Variable(boolean=True)
                # McCormick envelope constraints
                constraints.append(y_vars[(i,l)] <= h[i])
                constraints.append(y_vars[(i,l)] <= h[i+l])
                constraints.append(y_vars[(i,l)] >= h[i] + h[i+l] - 1)

        # Add lag-specific overlap constraints
        for l in range(n_points):
            term = cp.sum([y_vars[(i,l)] for i in range(n_points-l)])
            constraints.append(term >= lambda_var)
        
        # Objective: maximize lambda
        objective = cp.Maximize(lambda_var)
        
        prob = cp.Problem(objective, constraints)
        
        remaining_time = budget_s * 0.9  # Spend 90% of budget solving MILP
        
        prob.solve(solver=CPLEX, time_limit=remaining_time, verbose=False)
        
        # Extract solution
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol_h = h.value
            sol_lambda = lambda_var.value
            if sol_h is not None:
                h_binary = np.round(sol_h).astype(float)
                h_binary = np.clip(h_binary, 0, 1)
                current_c5 = 1 - dx * sol_lambda
                if current_c5 < best_c5:
                    best_h = h_binary
                    best_c5 = current_c5
        else:
            # Fallback to construction in case MILP fails
            pass
    except Exception as e:
        # fallback to heuristic if MILP fails
        pass

    # As fallback, run heuristic optimization 
    if best_h is None:
        # Revert to a lightweight local search with refinement
        best_h = h0.copy()
        best_c5 = evaluate_erdos_solution(best_h, n_points)
        n_iter = 0
        while time.time() - start_time < 0.8 * budget_s:
            # Small random perturbations for local refinement
            h_new = best_h.copy()
            p = np.random.rand(n_points) < 0.05  # ~5% change
            h_new[p] = 1 - h_new[p]  # Flip values
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * target_sum / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, n_points)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            n_iter += 1
    else:
        # Final re-evaluation
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), float(best_c5), n_points)

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute full correlation via convolution (same computation as in the task)
    h_rev = h[::-1]
    b = 1.0 - np.array(h)
    corr = np.convolve(h_rev, b, mode='full')
    return dx * np.max(corr)
