import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 256
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_soft_min_weights(S, alpha):
    """Compute weights for soft minimum of correlations"""
    Smax = np.max(-alpha * S)
    exp_S = np.exp(-alpha * S - Smax)
    return exp_S / np.sum(exp_S)

def compute_gradient(h, w):
    """Compute gradient via convolution with symmetric kernel"""
    # w is symmetric, centered at index n-1, so we extract the relevant part
    # w_k corresponds to lag k 
    # Build symmetric kernel W
    n = len(h)
    W = np.zeros(2 * n - 1)
    for i in range(n):
        k = i - (n - 1)
        if k == 0:
            W[n - 1] = w[i]
        elif k > 0:
            W[n - 1 + k] = w[i]
        else:  # k < 0
            W[n - 1 + k] = w[i]
    
    # Convolution (this includes reflection of h)
    conv = np.convolve(h, W, mode='full')
    g = conv[n - 1 : n - 1 + n]  # center of convolution
    
    # Add the double contribution of h_j at lag 0
    g += w[n - 1] * h
    return g

def local_optimize(h, max_iter=300):
    """Refine solution using scipy.optimize"""
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            return h
    except:
        pass
    return h
