import numpy as np
from scipy.signal import correlate
import scipy.io
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import SCS

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def make_binary_approx(h_rel, M):
    """Threshold rounding: choose M largest entries to be 1, others 0."""
    idx = np.argsort(-h_rel)
    h_bin = np.zeros_like(h_rel)
    h_bin[idx[:M]] = 1
    return h_bin

def round_hybrid(h_rel, X_rel, M, n_points):
    """Rounding with hybrid methods: deterministic + random."""
    candidates = []

    # Deterministic threshold rounding
    h_det = make_binary_approx(h_rel, M)
    candidates.append((h_det, 'det'))

    # Random hyperplane rounding with 20 trials
    if X_rel.shape[0] <= 100:
        V, Lambda = np.linalg.eigh(X_rel)
        V_positive = V[:, Lambda > 1e-8]
        Lambda_positive = Lambda[Lambda > 1e-8]
        r = V_positive.shape[1]
        for _ in range(20):
            g = np.random.randn(r)
            s = V_positive @ g
            h_trial = (s > 0).astype(float)
            # Repair sum
            diff = np.sum(h_trial) - M
            if abs(diff) > 0:
                idx = np.where(h_trial == 1)[0]
                if diff < 0:
                    # Need to flip some 0s to 1s
                    flip = np.random.choice(np.where(h_trial == 0)[0], size=int(-diff), replace=False)
                    h_trial[flip] = 1
                else:
                    # Need to flip some 1s to 0s
                    flip = np.random.choice(idx, size=int(diff), replace=False)
                    h_trial[flip] = 0
            candidates.append((h_trial, 'rand'))

    return candidates

def swap_hill_climb(h_initial, n_points, dx):
    """Exhaustive swap search to minimize C5."""
    h = h_initial.copy()
    improved = True
    max_swaps = 5000
    swaps = 0

    while improved and swaps < max_swaps:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Precompute lag contributions
        lags = []
        for i in range(n_points):
            for j in range(i + 1, n_points):
                k = j - i
                if k < n_points:
                    lags.append(k)

        for k in range(n_points):
            # Compute correlations efficiently
            if k == 0:
                lags_mask = [True] * n_points
            else:
                lags_mask = []
                for i in range(n_points):
                    if i + k < n_points and i - k >= 0:
                        lags_mask.append((True, i, k))
                if len(lags_mask) == 0:
                    continue
                lags_mask = np.array(lags_mask)

        # To optimize: compute all A_k for current h
        A = np.zeros(n_points)  # A_k = sum h_i * h_{i+k}
        for k in range(n_points):
            if k == 0:
                A[k] = np.sum(h)
            else:
                # Direct sum
                total = 0
                for i in range(n_points - k):
                    total += h[i] * h[i + k]
                A[k] = total
        worst_lag = np.argmin(A)

        # Now search for a pair one-j zero to swap and reduce worst_lag
        best_improvement = 0
        best_i, best_j = None, None

        for i in ones:
            for j in zeros:
                swaps += 1
                if swaps >= max_swaps:
                    break
                # Analytically compute new A values due to swap (i,j) at worst lag
                if worst_lag == 0:
                    # Just total sum, not involved in any lag
                    A_new = A.copy()
                else:
                    # As a vector: find changes in A_k
                    A_new = A.copy()
                    # i: in positions i,i+k, i-k minimum, j: i+k,i-k
                    # All shifted by one from indices calculate
                    changed = set()

                    for ki in list(range(n_points)):
                        # contribution of i
                        if i + ki < n_points and i - ki >= 0:
                            if ki in changed:
                                print(f"Warning: ki={ki}, i={i}")
                            changed.add(ki)
                            A_new[ki] -= h[i] * h[i + ki] - h[j] * h[i + ki]  # i is 1 now, j is 0
                            A_new[ki] += h[i] * h[i - ki] - h[j] * h[i - ki]
                        # contribution of j
                        if j + ki < n_points and j - ki >= 0:
                            if ki in changed:
                                print(f"Warning: ki={ki}, j={j}")
                            changed.add(ki)
                            A_new[ki] -= h[j] * h[j + ki] - h[i] * h[j + ki]  # j is 1 now, i is 0
                            A_new[ki] += h[j] * h[j - ki] - h[i] * h[j - ki]

                    # Shifted indices around i
                    for delta in range(ki + 1, n_points - ki):
                        changed.add(ki)
                        A_new[ki] -= h[i] * h[i + ki] - h[j] * h[i + ki]
                        A_new[ki] += h[i] * h[i - ki] - h[j] * h[i - ki]

                if np.argmin(A_new) < np.argmin(A):
                    gain = np.argmin(A) - np.argmin(A_new)
                    if gain > best_improvement:
                        best_improvement = gain
                        best_i, best_j = i, j

        if best_i is not None:
            h[best_i] = 0
            h[best_j] = 1
            improved = True
        else:
            break

    return h

def local_refine(h_initial, M, dx):
    """Apply SLSQP for local refinement with constraints."""
    h = h_initial.copy()
    bounds = [(0, 1) for _ in range(len(h))]
    def objective(h_steps):
        corr = compute_corr_fft(h_steps)
        return dx * np.max(corr)
        
    def constraint_eq(h_val):
        return np.sum(h_val) - M
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    try:
        res = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h = res.x
            h = project_to_sum_box(h, M)
            return h
    except:
        pass
    return h

def create_p_matrix(n, k):
    """Create matrix P_k for lag k."""
    P = np.zeros((n, n))
    for i in range(n):
        j = i + k
        if j < n:
            P[i, j] = 1.0
    return P

def solve_sdp(n_points, M, target_time_left):
    """Use SDP to relax the problem"""
    dx = 2.0 / n_points
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == M,
        h >= 0, h <= 1,
        X >> 0,  # Positive semidefinite
        cp.diag(X) == h,  # Diagonal is h
    ]

    # Lag constraints
    for k in range(n_points):
        P_k = create_p_matrix(n_points, k)
        constraints.append(t >= M - cp.trace(P_k @ X))

    # Problem
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Start with exact solves, then abort if too slow
    solve_opts = {'max_iters': 3000}
    prob.solve(solver=SCS, **solve_opts)
    if prob.status != cp.OPTIMAL:
        # Take the best solution available even if not proven optimal
        if not prob.status in [cp.OPTIMAL_INACCURATE, cp.SOLVER_ERROR]:
            print(f"SDP status: {prob.status}")
    
    if prob.status == cp.OPTIMAL:
        h_rel = h.value
        X_rel = X.value
    else:
        # Fall back to best iterate
        h_rel = h.value
        X_rel = X.value
        if h_rel is None or X_rel is None:
            # Fallback to uniform
            h_rel = np.ones(n_points) * (M / n_points)
            X_rel = np.eye(n_points) * (M / n_points)

    return h_rel, X_rel

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    sol_list = []

    # Phase A: SDP to get convex relaxation
    phases = [60, 80, 100]
    phases_times = [0.0] * len(phases)
    current_time = start_time
    for pi, n_points in enumerate(phases):
        if time.time() - current_time >= 0.48 * budget_s:
            break
        remaining = budget_s - (time.time() - start_time)
        if remaining < 20:
            break

        M = n_points // 2
        dx = 2.0 / n_points

        h_rel, X_rel = solve_sdp(n_points, M, remaining * 0.5)
        
        # Rounding step
        candidates = round_hybrid(h_rel, X_rel, M, n_points)

        # Evaluation
        for h_bin, rounding_type in candidates:
            corr = compute_corr_fft(h_bin)
            c5 = dx * np.max(corr)
            sol_list.append((c5, h_bin.copy(), n_points))

    # Phase B: Apply local swaps and equilibrium optimization
    if len(sol_list) > 0:
        sol_list.sort(key=lambda x: x[0])
        def evaluate_candidate(h_candidate):
            corr = compute_corr_fft(h_candidate)
            return dx * np.max(corr)

        # Service the best 5 candidates with hill-climbing
        for i in range(min(5, len(sol_list))):
            c5, h_bin, n = sol_list[i]
            h_best = swap_hill_climb(h_bin, n, dx)
            c5_new = evaluate_candidate(h_best)
            sol_list[i] = (c5_new, h_best, n)

    # Phase C: Final polish with SLSQP
    if len(sol_list) > 0:
        sol_list.sort(key=lambda x: x[0])
        best_c5, best_h, best_n = sol_list[0]
        # Polish with SLSQP
        M = best_n // 2
        best_h = local_refine(best_h, M, dx)
        corr = compute_corr_fft(best_h)
        best_c5 = dx * np.max(corr)
        sol_list[0] = (best_c5, best_h, best_n)

    if len(sol_list) == 0:
        # Fallback to simple construction
        n_points = 100
        M = n_points // 2
        dx = 2.0 / n_points
        # Alternating blocks
        h = np.zeros(n_points)
        block_size = 2
        for i in range(0, n_points, 2 * block_size):
            h[i:(i + block_size)] = 1
        h = project_to_sum_box(h, M)
        corr = compute_corr_fft(h)
        c5_val = dx * np.max(corr)
        sol_list.append((c5_val, h, n_points))

    sol_list.sort(key=lambda x: x[0])
    c5_bound, h_values, n_points = sol_list[0]

    return (h_values.tolist(), float(c5_bound), int(n_points))
