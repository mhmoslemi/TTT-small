import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 240
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def auto_corr(h):
    """Compute auto-correlation of h"""
    return correlate(h, h, mode='full')

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_step(h, alpha, step, target_sum):
    """Perform one gradient ascent step"""
    n = len(h)
    # auto-correlation
    S = auto_corr(h)
    # compute weights
    # avoid overflow
    exponents = -alpha * S
    max_ex = np.max(exponents)
    w = np.exp(exponents - max_ex)
    w /= np.sum(w)
    # convolution with weight kernel
    conv = np.convolve(h, w, mode='full')
    # central part
    start = n - 1
    g = conv[start:start+n].copy()
    w0 = w[start]  # zero lag weight
    g += w0 * h
    # step
    h_new = h + step * g
    # project to [0,1] sum constraints
    h_new = project_to_sum_box(h_new, target_sum)
    return h_new

def random_feasible_h(n, target_sum):
    """Generate a random feasible h vector"""
    h = np.random.uniform(0, 1, n)
    return project_to_sum_box(h, target_sum)
