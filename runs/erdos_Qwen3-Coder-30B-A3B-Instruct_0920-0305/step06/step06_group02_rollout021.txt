import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def compute_convolution(h):
    """Compute convolution for overlap counts"""
    # Use numpy correlate with 'full' mode to get overlap counts
    conv = correlate(h, h, mode='full')
    return conv

def compute_c5_from_conv(conv, n):
    """Convert convolution result to C5 value"""
    return dx * (n/2 - np.min(conv))

def apply_swap(conv, ones, zeros, i, j, n):
    """Apply swap of positions i (now 1) and j (now 0) with delta update"""
    # Store original conv before changes
    conv_new = conv.copy()
    
    # Remove contributions of i (1 -> 0)
    for k in ones:
        shift = k - i
        if shift >= -(n-1) and shift <= (n-1):
            idx = shift + (n-1)
            conv_new[idx] -= 1
            if k != i:
                conv_new[-shift + (n-1)] -= 1
    
    # Remove i from ones list and add j to zeros list
    ones.remove(i)
    zeros.append(i)
    
    # Add contributions of j (0 -> 1)
    for k in ones + [j]:
        shift = k - j
        if shift >= -(n-1) and shift <= (n-1):
            idx = shift + (n-1)
            conv_new[idx] += 1
            if k != j:
                conv_new[-shift + (n-1)] += 1
    
    # Add self contribution for j
    conv_new[n-1] += 1
    
    # Add j to ones list and remove from zeros list
    ones.append(j)
    zeros.remove(j)
    
    # Return updated convolution and min overlap
    return conv_new, np.min(conv_new)

    needed = TARGET_SUM - current_sum
    
    if needed > 0:
        # Add ones to positions where value is 0
        zeros = np.where(h_binary == 0)[0]
        add_indices = np.random.choice(zeros, min(needed, len(zeros)), replace=False)
        for idx in add_indices:
            h_binary[idx] = 1
    elif needed < 0:
        # Remove ones from positions where value is 1
        ones = np.where(h_binary == 1)[0]
        remove_indices = np.random.choice(ones, min(-needed, len(ones)), replace=False)
        for idx in remove_indices:

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_h = adaptive_sampling_initialization()
    
    # Compute initial convolution
    conv = compute_convolution(best_h)
    min_overlap = np.min(conv)
    best_c5 = dx * (TARGET_SUM - min_overlap)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.99995
    max_iter_per_restart = 50000
    
    # Track indices for efficient swap selection
    ones = list(np.where(best_h == 1)[0])
    zeros = list(np.where(best_h == 0)[0])
    
    # Run multiple restarts
    restart_count = 0
    while time.time() - start_time < budget_s * 0.95:
        restart_count += 1
        # Reset for restart
        h = best_h.copy()
        conv = compute_convolution(h)
        min_overlap = np.min(conv)
        current_c5 = dx * (TARGET_SUM - min_overlap)
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        T = T0
        iteration = 0
        inner_iter = 0
        
        while time.time() - start_time < budget_s * 0.95 and iteration < max_iter_per_restart:
            # Time-based early termination check
            if time.time() - start_time > budget_s * 0.95:
                break
                
            iteration += 1
            inner_iter += 1
            
            # Choose random 1 and 0 to swap
            if len(ones) > 0 and len(zeros) > 0:
                i = random.choice(ones)
                j = random.choice(zeros)
                
                # Apply swap
                new_conv, new_min_overlap = apply_swap(conv, ones, zeros, i, j, n_points)
                new_c5 = dx * (TARGET_SUM - new_min_overlap)
                
                # Acceptance decision
                delta = current_c5 - new_c5  # positive if improvement
                if delta > 0 or random.random() < np.exp(delta / T):
                    # Accept the swap
                    h[i] = 0
                    h[j] = 1
                    conv = new_conv
                    current_c5 = new_c5
                    min_overlap = new_min_overlap
                    
                    # Update best solution if needed
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        
            # Temperature update
            if T > 1e-10:
                T *= cooling_rate
                
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
