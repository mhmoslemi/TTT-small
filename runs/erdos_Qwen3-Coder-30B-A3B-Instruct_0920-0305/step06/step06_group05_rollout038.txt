import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_corr_fft(h):
    """Compute the correlation using FFT for overlap calculation."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5_from_corr(corr, dx):
    """Convert correlation result to C5 value."""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_fast(h, dx):
    """Fast evaluation of C5."""
    corr = compute_corr_fft(h)
    return compute_c5_from_corr(corr, dx)

def run_swap_hill_climb(h, dx, n_points, max_swaps=5000):
    """Apply deterministic local swaps to improve h."""
    h = h.copy()
    improved = True
    swaps_count = 0
    while improved and swaps_count < max_swaps:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                if swaps_count >= max_swaps:
                    break
                # Swap i and j
                h[i], h[j] = h[j], h[i]
                new_c5 = evaluate_c5_fast(h, dx)
                if new_c5 < evaluate_c5_fast(h.copy(), dx):
                    improved = True
                    swaps_count += 1
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            if improved or swaps_count >= max_swaps:
                break
    return h

def rounding_and_improvement(X_rel, h_rel, n, M, dx, target_sum):
    """Perform randomized rounding on the SDP solution and hill climb."""
    h_best = None
    c5_best = float('inf')
    
    # Deterministic threshold rounding
    h_thresh = np.zeros(n)
    idxs = np.argsort(h_rel)[::-1]
    h_thresh[idxs[:int(M)]] = 1.0
    h_thresh = project_to_sum_box(h_thresh, target_sum)
    h_thresh = run_swap_hill_climb(h_thresh, dx, n)
    c5_thresh = evaluate_c5_fast(h_thresh, dx)
    if c5_thresh < c5_best:
        h_best = h_thresh.copy()
        c5_best = c5_thresh
    
    # Randomized rounding trials
    for _ in range(20):  # 20 roundings
        V, Lambda = np.linalg.eigh(X_rel)
        r = np.sum(Lambda > 1e-6)
        V_plus = V[:, :r]
        Lambda_plus = Lambda[:r]
        
        g = np.random.normal(0, 1, r)
        s = V_plus @ g
        b = (s >= 0).astype(float)

        # Adjust the weight to target_sum
        actual_sum = np.sum(b)
        if actual_sum > target_sum:
            flip_indices = np.argsort(s)[:int(actual_sum - target_sum)]
            b[flip_indices] = 0
        elif actual_sum < target_sum:
            flip_indices = np.argsort(-s)[:int(target_sum - actual_sum)]
            b[flip_indices] = 1

        b = project_to_sum_box(b, target_sum)
        b = run_swap_hill_climb(b, dx, n)
        c5_current = evaluate_c5_fast(b, dx)
        if c5_current < c5_best:
            h_best = b.copy()
            c5_best = c5_current

    return h_best, c5_best

def solve_sdp(n, M):
    """Solve the SDP relaxation."""
    dx = 2.0 / n
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == M,
        h >= 0,
        h <= 1,
        cp.diag(X) == h,
        X >= 0,
    ]
    
    for k in range(n):
        P_k = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                P_k[i, i + k] = 1
        constraints.append(t >= M - cp.trace(P_k @ X))

    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    start_time = time.time()
    prob.solve(solver=cp.SCS, max_iters=10000, time_limit=300)  # 5 min
    duration = time.time() - start_time

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        X_sol = X.value
        t_val = t.value
        return t_val, h_sol, X_sol, dx, duration
    else:
        # Fallback to approximate solution if solver fails
        h_sol = h.value if h.value is not None else np.ones(n) * 0.5
        X_sol = np.outer(h_sol, h_sol)
        return np.inf, h_sol, X_sol, dx, duration

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_vals = [60, 80, 100]  # Will use one of these
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Phase A: SDP solving
    for n in n_vals:
        if time.time() - start_time > 0.4 * budget_s:
            break
        M = n // 2  # Since ∫h = 1, and dx = 2/n, so sum(h) * dx = 1 => sum(h) = n/2
        t_val, h_sol, X_sol, dx, duration = solve_sdp(n, M)
        if t_val < best_c5:
            h_best, c5_best = rounding_and_improvement(X_sol, h_sol, n, M, dx, M)
            if h_best is not None and c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
                best_n = n
                
    # Phase B: Final polishing if there's time
    if time.time() - start_time < 0.95 * budget_s and best_h is not None:
        def objective(h):
            return evaluate_c5_fast(h, dx)
        
        def constraint_eq(h):
            return np.sum(h) - M
        
        bounds = [(0, 1) for _ in range(best_n)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                           constraints=cons, options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                best_h = res.x
                best_h = project_to_sum_box(best_h, M)
                best_c5 = evaluate_c5_fast(best_h, dx)
        except:
            pass

    # Ensure we have result if any part failed
    if best_h is None:
        # Fallback to simple binary stepping
        n = 100
        M = n // 2
        dx = 2.0 / n
        h_fallback = np.zeros(n)
        h_fallback[:M] = 1.0
        best_h = h_fallback.copy()
        best_c5 = evaluate_c5_fast(best_h, dx)
        best_n = n

    return (best_h.tolist(), float(best_c5), best_n)
