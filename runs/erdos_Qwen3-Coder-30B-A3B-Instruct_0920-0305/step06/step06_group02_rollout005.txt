import numpy as np
import time

# Global parameters
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def initial_binary(seed=42):
    """Create an initial binary vector with exact weight TARGET_SUM"""
    np.random.seed(seed)
    x = np.zeros(n_points, dtype=np.int8)
    ones_indices = np.random.choice(n_points, TARGET_SUM, replace=False)
    x[ones_indices] = 1
    return x

def full_overlap(x):
    """Compute full correlation of x with itself"""
    return np.correlate(x, x, mode='full')

def swap_update(ov, i, j, ones, zeros):
    """Update overlap array after swapping x[i] and x[j]"""
    # Make copy to avoid modifying original
    ov_new = ov.copy()
    
    # Remove contribution of i from all shifts
    for k in ones:
        if k == i:
            continue
        shift = k - i
        ov_new[shift + n_points - 1] -= 1
    
    # Remove contribution of j as 1 from all shifts (because it was 0)
    for k in ones:
        if k == i:
            continue
        shift = k - j
        ov_new[shift + n_points - 1] += 1
    
    # Add contribution of j as 1 from all shifts
    for k in ones:
        if k == i:
            continue
        shift = k - j
        ov_new[shift + n_points - 1] += 1
    
    # Add contribution of i as 1 from all shifts (because it was 0)
    for k in ones:
        if k == i:
            continue
        shift = k - i
        ov_new[shift + n_points - 1] -= 1
        
    return ov_new

def sim_ann(x0, ov0, ones0, zeros0, time_limit):
    """Run simulated annealing with incremental update"""
    start_time = time.time()
    T0 = 0.5
    lambda_coeff = 0.001
    best_x = x0.copy()
    best_ov = ov0.copy()
    best_ones = ones0.copy()
    best_zeros = zeros0.copy()
    cur_x = x0.copy()
    cur_ov = ov0.copy()
    cur_ones = ones0.copy()
    cur_zeros = zeros0.copy()
    cur_min = np.min(cur_ov)
    best_min = cur_min

    iterations = 0
    while time.time() - start_time < time_limit * 0.95:
        iterations += 1
        # Temperature schedule
        T = T0 * np.exp(-lambda_coeff * iterations / 10000)
        if T < 1e-4:
            T = 1e-4

        # Pick a random 1 and a random 0
        if len(cur_ones) == 0 or len(cur_zeros) == 0:
            break
        i = np.random.choice(cur_ones)
        j = np.random.choice(cur_zeros)
        
        # Calculate new overlap after swapping
        try:
            new_ov = swap_update(cur_ov, i, j, cur_ones, cur_zeros)
            new_min = np.min(new_ov)
            
            # Acceptance probability
            delta = new_min - cur_min
            if delta > 0 or np.random.rand() < np.exp(delta / T):
                # Accept the move
                cur_x[i] = 0
                cur_x[j] = 1
                cur_ov = new_ov
                cur_min = new_min
                cur_ones.remove(i)
                cur_ones.add(j)
                cur_zeros.remove(j)
                cur_zeros.add(i)
                
                if cur_min > best_min:
                    best_x = cur_x.copy()
                    best_ov = cur_ov.copy()
                    best_ones = cur_ones.copy()
                    best_zeros = cur_zeros.copy()
                    best_min = cur_min
        except:
            continue
    
    return best_x, best_ov, best_ones, best_zeros, best_min

def worst_shift_refine(x, ov, ones, zeros):
    """Refine solution by finding and fixing the worst shift"""
    cur_min = np.min(ov)
    min_shift = np.argmin(ov) - (n_points - 1)
    if cur_min >= TARGET_SUM / 2:
        return x, ov, ones, zeros, cur_min
    
    # Identify amenable pairs for swap to increase overlap in min_shift
    # Positions i where x[i] = 1, but x[i+min_shift] = 0
    A = set()
    # Positions j where x[j] = 0, but x[j+min_shift] = 1
    B = set()
    
    for i in ones:
        if 0 <= i + min_shift < n_points and x[i + min_shift] == 0:
            A.add(i)
        if 0 <= i - min_shift < n_points and x[i - min_shift] == 0:
            A.add(i)
    
    for j in zeros:
        if 0 <= j + min_shift < n_points and x[j + min_shift] == 1:
            B.add(j)
        if 0 <= j - min_shift < n_points and x[j - min_shift] == 1:
            B.add(j)
            
    # Try best improvements greedily
    A_list = list(A)
    B_list = list(B)
    np.random.shuffle(A_list)
    np.random.shuffle(B_list)
    
    for i in A_list:
        for j in B_list:
            if i == j:
                continue
            # Check if the swap increases the overlap at this shift
            # Try not to decrease any overlap
            test_ov = swap_update(ov, i, j, ones, zeros)
            if np.min(test_ov) >= cur_min:
                # Accept the move
                x[i] = 0
                x[j] = 1
                ov = test_ov
                cur_min = np.min(ov)
                ones.remove(i)
                ones.add(j)
                zeros.remove(j)
                zeros.add(i)
                break
                
    return x, ov, ones, zeros, cur_min

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Run multiple restarts
    best_x = None
    best_min = float('inf')
    for _ in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Initialize binary vector
        x = initial_binary(seed=seed+_)
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        cur_min = np.min(ov)
        
        # Run simulated annealing
        x_res, ov_res, ones_res, zeros_res, min_res = sim_ann(
            x, ov, ones, zeros, budget_s * 0.8)
        
        # Refine using worst-shift repair
        x_fine, ov_fine, ones_fine, zeros_fine, min_fine = worst_shift_refine(
            x_res, ov_res, ones_res, zeros_res)
        
        if min_fine < best_min:
            best_min = min_fine
            best_x = x_fine.copy()
            
    if best_x is None:
        # Fallback to a single run
        x = initial_binary()
        ones = set(np.where(x == 1)[0])
        zeros = set(np.where(x == 0)[0])
        ov = full_overlap(x)
        x_res, ov_res, ones_res, zeros_res, min_res = sim_ann(
            x, ov, ones, zeros, budget_s * 0.8)
        best_x = x_res
        
    ov_final = full_overlap(best_x)
    final_min = np.min(ov_final)
    c5 = (TARGET_SUM - final_min) * dx
    
    # Convert to floats (0. or 1.)
    h_final = best_x.astype(float)
    return (h_final.tolist(), float(c5), n_points)
