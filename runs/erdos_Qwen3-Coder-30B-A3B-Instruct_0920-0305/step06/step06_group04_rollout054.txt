import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    N_fine = 800
    N = 120  # Target discretization
    K = 12   # Number of Fourier harmonics (real + imaginary)
    target_sum = N / 2.0
    dx = 2.0 / N
    # Enforce that all of our Fourier coeffs are bounded in [-0.5, 0.5]
    bounds_c = [(-0.5, 0.5)] * (2 * K)

    # Helper functions

    def fourier_to_h(theta_coeffs, n_fine=N_fine):
        a = theta_coeffs[:K]
        b = theta_coeffs[K:]
        x = np.linspace(0, 2, n_fine, endpoint=False)
        h_raw = 0.5 + np.dot(np.stack((np.cos(np.pi * np.arange(1, K+1) * x),
                                       np.sin(np.pi * np.arange(1, K+1) * x)), axis=0),
                             np.stack((a, b), axis=0))
        return np.clip(h_raw, 0, 1)

    def project_to_sum_box(h, target_sum):
        # Water-filling approach to maintain the sum exactly
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        # Add a constant shift to achieve the correct sum
        shift = (target_sum - current_sum) / len(h)
        h_shift = h + shift
        return np.clip(h_shift, 0, 1)

    def evaluate_c5_fourier(theta_coeffs, downsample_to=N):
        h_fine = fourier_to_h(theta_coeffs, N_fine)
        # Downsample
        step = N_fine // downsample_to
        h = h_fine[::step]
        # Project to satisfy sum constraint
        h = project_to_sum_box(h, target_sum)
        # FFT correlation
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def filled_h_from_theta(theta_coeffs):
        h = fourier_to_h(theta_coeffs, N_fine)
        step = N_fine // N
        h = h[::step]
        return project_to_sum_box(h, target_sum)

    # Global optimization
    def optimize_global():
        result = differential_evolution(
            evaluate_c5_fourier,
            bounds=bounds_c,
            seed=seed,
            maxiter=500,
            popsize=50,
            strategy='best1bin',
            disp=False,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.9,
        )
        return result

    # Local polishing phase using projected gradient ascent
    def project_to_box_sum(x, target_sum):
        # Ensure x is a valid density
        x = np.clip(x, 0, 1)
        current_sum = np.sum(x)
        if abs(current_sum - target_sum) < 1e-10:
            return x
        # Add scalar λ
        shift = (target_sum - current_sum) / len(x)
        x = x + shift
        return np.clip(x, 0, 1)

    def inplace_gradient_ascent(h, max_iter=300, alpha_start=1.0, alpha_end=20.0):
        n = len(h)
        # Precompute FFT of h
        H = fft(h)
        # The frequency domain version of 1-h
        H_conj = fft(1 - h)
        # Correlation is real inverse FFT of H_conj * H
        corr = np.real(ifft(H_conj * H))
        # For minimum lag, we want to increase this or keep it stable
        # Use smoothed log-sum-exp 
        # Strengthen the log-sum-exp weights
        alpha = np.linspace(alpha_start, alpha_end, max_iter)
        for i in range(max_iter):
            # Use current correlation
            a = alpha[i]
            w = np.exp(-a * corr)
            w /= np.sum(w)
            
            # Approximate gradient at each point using finite differences
            grad = np.zeros_like(h)
            # Left and right shifted versions of h* ones
            for l in range(0, n-1):
                if l >= 0: grad += w[l] * h[l+1:]  # Right shift
                if l < n-1: grad += w[l] * h[:n-1-l]  # Left shift
                
            # Subtract 2 * grad (gradient descent, we want to minimize)
            h_new = h - 0.01 * grad
            h_new = project_to_box_sum(h_new, target_sum)
            # Update if it improves C5
            corr_new = correlate(h_new, 1 - h_new, mode="full")
            if np.max(corr_new) * dx < np.max(corr) * dx:
                h[:] = h_new
                corr[:] = corr_new
        return h

    # Binary rounding and swap step
    def binary_round_and_swap(h):
        # Set top half to 1, rest to 0
        ind = np.argsort(h)[::-1]
        h_bin = np.zeros_like(h)
        h_bin[ind[:len(h)//2]] = 1
        # Fast swap improving C5
        corr_orig = correlate(h_bin, 1 - h_bin, mode="full")
        c5_orig = np.max(corr_orig) * dx
        
        # Now try swaps one by one, mostly around the worst lag
        # Find worst lag
        worst_lag_idx = np.argmin(corr_orig)
        worst_lag = worst_lag_idx - (len(h_bin) - 1) // 2
        
        max_swaps = 2 * len(h_bin)
        passes = 0
        improve_count = 0
        
        # Helpers
        def update_correlation_single_swap(i, j, corr):
            # Compute new correlation values for the pairs that change due to swap
            n = len(corr)
            old_val1 = corr[i]
            old_val2 = corr[j]
            new_val1 = corr[i+1] if i+1 < n else 0
            new_val2 = corr[j-1] if j > 0 else 0
            # Update in O(1) to compute correlation impact of a single swap
            # Dummy placeholder for simplicity; actual implementation should be robust
            return corr

        # Actual full correlation from scratch is expensive but minimal for binary change
        # We'll do it naively by looping over some candidates
        while improve_count < max_swaps and passes < 100:
            # Find positions to swap
            max_val = -np.inf
            best_i, best_j = None, None

            # Iterate through some positions to find best swap
            for i in range(len(h_bin)):
                if h_bin[i] == 1:
                    for j in range(len(h_bin)):
                        if h_bin[j] == 0:
                            # Copy and try swap
                            h_test = h_bin.copy()
                            h_test[i], h_test[j] = h_test[j], h_test[i]
                            corr_test = correlate(h_test, 1 - h_test, mode="full")
                            min_corr_test = np.max(corr_test) * dx
                            if min_corr_test < c5_orig:
                                if min_corr_test > max_val:
                                    max_val = min_corr_test
                                    best_i, best_j = i, j
            
            if best_i is None:
                break
            else:
                h_bin[best_i], h_bin[best_j] = h_bin[best_j], h_bin[best_i]
                corr_orig = correlate(h_bin, 1 - h_bin, mode="full")
                c5_orig = np.max(corr_orig) * dx
                improve_count += 1
                
            passes += 1
            
        return h_bin

    time_left = budget_s - (time.time() - start_time)
    if time_left <= 10:
        return ([0.5]*N, 0.5, N)

    # Phase 1: Large-scale global optimization
    res1 = optimize_global()
    h_best = filled_h_from_theta(res1.x)
    c5_best = evaluate_c5_fourier(res1.x)
    
    # Phase 2: Local polishing
    h_polished = inplace_gradient_ascent(h_best.copy())
    c5_polished = evaluate_c5_fourier(res1.x, downsample_to=N)  # or recompute
    if evaluate_c5_fourier(res1.x, downsample_to=N) < c5_best:
        c5_best = evaluate_c5_fourier(res1.x, downsample_to=N)
        h_best = filled_h_from_theta(res1.x)

    # Phase 3: Binary round and swap
    h_binary = binary_round_and_swap(h_polished)
    c5_binary = evaluate_c5_fourier(res1.x, downsample_to=N)  # intentional reuse for consistency with mechanics part
    
    # Final evaluation of binary candidate
    corr = correlate(h_binary, 1 - h_binary, mode="full")
    final_c5 = np.max(corr) * dx
    
    # Return best path
    if final_c5 < c5_best:
        final_h = h_binary
        final_c5 = final_c5
    else:
        final_h = h_best

    return (final_h.tolist(), float(final_c5), N)
