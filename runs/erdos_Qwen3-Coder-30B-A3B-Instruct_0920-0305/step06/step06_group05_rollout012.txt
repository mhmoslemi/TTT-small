import numpy as np

def evaluate_erdos_solution(h):
    """Evaluate C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_corr = np.max(corr)
    return dx * max_corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_milp(n, time_limit):
    """Solve the MILP formulation for binary variables"""
    # Required number of 1s
    M = n // 2
    
    # Create decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == M,
        t >= 0
    ]
    
    # Linking constraints: y[i,j] = h[i] AND h[j]
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(y[i, j] <= h[i])
            constraints.append(y[i, j] <= h[j])
            constraints.append(y[i, j] >= h[i] + h[j] - 1)
    
    # Lag constraints
    for k in range(1, n):
        # Sum over valid pairs (i, i+k)
        sum_y = cp.sum([y[i, i+k] for i in range(n-k)])
        constraints.append(t >= M - sum_y)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
    except:
        return None, None
    
    # If we got a solution, extract the best binary h
    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_bin = np.rint(h.value).astype(int)
        t_val = t.value
        return h_bin, t_val
    else:
        # Return the best feasible point if obtained
        try:
            h_bin = np.rint(h.value).astype(int)
            t_val = t.value
            return h_bin, t_val
        except:
            return None, None

def local_continuous_refine(h_binary, n_points):
    """Refine binary solution with continuous optimization"""
    h_in = h_binary.astype(float)
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    def objective(h):
        # Compute max correlation
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint_eq(h):
        return np.sum(h) - TARGET_SUM
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(
            objective, 
            h_in, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        if res.success:
            h_refined = res.x
            h_refined = project_to_sum_box(h_refined, TARGET_SUM)
            return h_refined
    except:
        pass
    return h_in
