import numpy as np
from scipy.signal import correlate
from scipy.fftpack import fft, ifft
import time

def evaluate_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def fft_convolve(h, kernel):
    """Perform convolution using FFT"""
    N = len(h) * 2 - 1
    h_padded = np.pad(h, (0, N - len(h)), 'constant')
    kernel_padded = np.pad(kernel, (0, N - len(kernel)), 'constant')
    conv_fft = ifft(fft(h_padded) * fft(kernel_padded).conj())
    return np.real(conv_fft[:len(h)])

def compute_gradient(h, corr_weights):
    """Compute gradient using FFTs for efficiency"""
    n = len(h)
    # Expansion of convolution in terms of truncated sum:
    # We want grad[i] = sum_{ell} w[ell] * ((1-h[i+ell]) - h[i-ell])
    # We will split into forward and backward convolution
    # Forward part (h shifted right by ell): sum_{ell} w[ell] * h[i+ell]
    forward_kernel = corr_weights
    forward_conv = fft_convolve(h, forward_kernel)

    # Backward part (h shifted left by ell): sum_{ell} w[ell] * h[i-ell]
    backward_kernel = corr_weights[::-1]
    backward_conv = fft_convolve(h, backward_kernel)

    # Gradient is: 1 - forward_conv - backward_conv
    gradient = np.ones_like(h) - forward_conv - backward_conv
    return gradient

def softmax_weights(corr, alpha):
    """Softmax on correlation values with temperature alpha"""
    # Numerically stable softmax
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_shifted = np.exp(alpha * shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return weights

def project_gradient_step(h, grad, eta):

    # Update h with gradient step
    h_trial = h - eta * grad
    # Project to feasible set
