import numpy as np

import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h_i <= 1, sum(h) = target_sum"""
    # Use bisection to find lambda such that sum(clamp(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) / 2
        new_h = np.clip(h - lam, 0, 1)
        s = np.sum(new_h)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    new_h = np.clip(h - lo, 0, 1)
    return new_h

def build_five_step(a, b, n):
    """Build a 5-step function with parameters a, b"""
    # 0-a-u-1-u-a-0
    dx = 2.0 / n
    x = np.arange(0.5, n + 0.5) * dx  # sample points
    u = (2 * b - 1) / (2 * (b - a))  # solve for u
    if not (0 <= u <= 1):
        return None, False
    h = np.zeros(n)
    # Fill in 5 regions
    a_idx = int(np.ceil(a/dx))
    b_idx = int(np.ceil(b/dx))
    # step 1: 0
    # step 2: a, u
    # step 3: 1
    # step 4: u
    # step 5: 0
    h[:a_idx] = 0
    h[a_idx:b_idx] = u
    # Central part: height 1
    center_start = int(np.ceil((1 - b) / dx)) + a_idx
    center_end = int(np.ceil((1 + b) / dx)) - a_idx
    if center_end > n:
        center_end = n
    if center_start < center_end:
        h[center_start:center_end] = 1
    # Right part: u
    h[center_end:] = 0
    sl = np.sum(h) * dx
    h *= target_sum / sl
    # Ensure values stay in [0,1]
    h = np.clip(h, 0, 1)
    return h, True

def smooth_max(c, alpha=1.0):
    """Softmax approximation of max"""
    c_shift = c - np.max(c)  # for numerical stability
    exp_c = np.exp(alpha * c_shift)
    w = exp_c / np.sum(exp_c)
    return np.sum(w * c)

def gradient_step_simple(h, alpha=0.5, eta=0.5):
    """Compute gradient and take one step"""
    n = len(h)
    delta = 1e-6
    # correlation list
    corr = correlate(h, 1-h, mode="full")
    # smooth max for gradient
    # weights w for smoothing -- from max(corr)
    exp_corr = np.exp(alpha * (corr - np.max(corr)))
    w = exp_corr / np.sum(exp_corr)
    
    # Compute gradient:
    # d Phi / d h_i = 1 - sum_s w_s (h_{i+s} + h_{i-s})
    # need to compute two convolutions
    # Create extended arrays for easy indexing
    h_ext = np.concatenate([h, [0], h, [0]])  # add one to both ends
    # forward conv: w folded (so reversed) with h
    grad1 = correlate(w[::-1], h_ext, mode='full')
    # backward conv: w with h
    grad2 = correlate(w, h_ext, mode='full')
    # take the center part (excluding padding)
    grad = 1.0 - (grad1[n-1:n-1+n] + grad2[n-1:n-1+n])
    
    # update  
    new_h = h - eta * grad
    return new_h, alpha, eta
