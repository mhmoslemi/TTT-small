import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Choose a more refined discretization for our DC approach
    # 120 points is good: dx = 2/120 ≈ 0.01667, manageable for QP
    n_points = 120
    dx = 2.0 / n_points
    
    # Initial solution: interpolate from the provided values to match n_points
    h_initial = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(h0)), h0)
    
    # Normalize to sum to 1 (since we do not know the requirement is exact sum = 1)
    h_initial = h_initial * (n_points / (2 * np.sum(h_initial)))
    
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(h_initial)
    
    # --- DC (Difference of Convex) Algorithm with LP subproblem ---
    max_iter = 3000
    patience = 50
    patience_counter = 0
    last_improvement = 0

    def compute_autocorr(h):
        # Compute auto-correlation for all shifts: q_s = sum of h_i * h_{i+s}
        # This corresponds to correlate(h, h, mode='full')
        # q_s evaluates to the sum of h_i h_{i+s} for shift s
        # The actual expression to minimize is const - max_s q_s
        # The max of q_s is the worst-case shift
        return correlate(h, h, mode='full')
    
    def gradient_autocorr(h, shift):
        # Gradient of q_s(h) = sum_i h_i * h_{i+s} w.r.t. h_i
        # For shift s: gradient for h_i = h_{i+s} if i+s < n else 0
        # and h_{i-s} if i-s >= 0 else 0
        grad = np.zeros_like(h)
        if shift >= 0:
            # Shift by +shift (h_{i+shift})
            grad[:-shift] += h[shift:]
        if shift <= 0:
            # Shift by -shift (h_{i-shift})
            grad[-shift:] += h[:-shift]
        return grad  # this is ∇q_s at h
    
    def solve_lp_direction(grad, sum_constraint=n_points/2.0):
        # Solve: minimize grad^T x  subject to 0 <= x <= 1, sum(x) = sum_constraint
        # This is the standard capped-simplex problem
        # The solution is to sort gradient entries, assign largest values to 1,
        # smallest to 0, and decide the remaining with the sum constraint.
        # Uses analytical solution
        indices = np.argsort(-grad)  # descending order of gradient
        x = np.zeros_like(grad)
        total = 0.0
        for i in indices:
            if total + 1.0 <= sum_constraint:
                x[i] = 1.0
                total += 1.0
            else:
                x[i] = sum_constraint - total
                total = sum_constraint
                break
        return x
    
    # Now execute DC loop
    for idx_iter in range(max_iter):
        if (time.time() - start_time) >= 0.8 * budget_s:
            break
            
        # Compute all q_s for this h: the full auto-correlation vector
        auto = compute_autocorr(best_h)
        # Reverse to get from shift -n+1 to n-1
        # But we must consider overlaps running from shift 0 to 2*n-2
        # For a shift k (0 to 2*n-2), the corresponding shift index is k - (n-1)
        # We want the maximum overlap (minimum q_s), and we have full descriptor
        # For each shift k (from 0 to 2*n-2), we want to get q_s = auto[k]
        # But we care about only shift indices from -n+1 to n-1
        # So shift_s = k - (n-1) for k=0,...,2*(n-1)
        # auto[k] corresponds to q_{k-(n-1)}
        # Estimate where offset is:
        # auto has length 2*n - 1 = 239 for n=120
        # q_s for shift s = k - (n-1) with k = 0,...,239
        # So shift = k - 119 (let's be careful)
        
        # Find the DIRECT index in auto corresponding to actual maximum overlap
        # This is the index that contributes to q_s the most, i.e.
        # maximizing sum h_i * h_{i+s}: i.e. minimize q_s
        min_idx = np.argmin(auto)
        best_shift = min_idx - (n_points - 1)
        
        # Compute gradient for best_shift
        grad = gradient_autocorr(best_h, best_shift)
        
        # Solve LP to get next iterate
        h_new = solve_lp_direction(grad, sum_constraint=n_points/2.0)
        
        # Evaluate for some safeguarding (sometimes numerical issues occur)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
            last_improvement = idx_iter
        else:
            patience_counter += 1
            if patience_counter >= patience:
                break
        if (idx_iter - last_improvement) > 200:
            break

    # --- Optional final refinement: local smooth max ascent using projected gradient ---
    # We'll just do a few more iterations with a projected gradient-like approach
    
    best_h_final = best_h.copy()
    best_c5_final = best_c5
    
    def projected_gradient_descent(h_start, n_iters=1000, lr=0.05):
        h = h_start.copy()
        for _ in range(n_iters):
            # Compute all correlations, the full matrix:
            corr = correlate(h, 1-h, mode="full")
            # Compute gradient based on max index (as in DC)
            max_index = np.argmax(corr)
            # Compute the overlap for the shift that gives max overlap in 
            # the approximated value. In true version,
            # The j-th component of gradient = d/dh_j [sum_i h_i (1-h_{i+j})] 
            # = (1-h_{j+i}) - h_{j+i} for i with j+i within range
            
            grad = np.zeros_like(h)
            # Correlation with 1-h (as convolution of h with 1-h)
            temp = correlate(h, (1 - h), mode="full")
            # We need active shift for max correlation; retrieve it
            max_shift = max_index - (len(h) - 1)
            # For shift s: g[i] = 1 - h_{i+s} (if exists) 
            #              But corr[i] = sum_j h_j * (1 - h_{i+j-s}) = sum h_j - sum h_j h_{i+j-s}
            # So for shift s, d/dh_i [sum h_j h_{i+j-s}] 
            # = h_{i+s} if i+s exists (else 0), plus h_{i-s} if i-s exists
            # But we’ll do it directly:
            # ∇ [sum_j h_j (1 - h_{j+i})] = 
            # 1 - h_{i+?} 
            # Actually the issue is precise shift: for error, just go back to previous.
            
            # Our approach: Instead use sampling from integral evasions
            
        return h

    # We simply run a few more constrained steps on our current best
    h_refined = best_h.copy()
    current_c5 = best_c5
    _iter = 0
    # Run more short iterations like projected gradient or weighted descent
    while (time.time() - start_time < 0.92 * budget_s) and _iter < 500:
        # Sample a few random vectors near current to keep exploring
        rand_h = np.random.uniform(0, 1, n_points)
        rand_h = rand_h * (n_points / (2 * np.sum(rand_h)))  # normalize
        candidate = np.average([h_refined, rand_h], axis=0, weights=[0.9, 0.1])
        c5_candidate = evaluate_c5(candidate)
        if c5_candidate < current_c5:
            h_refined = candidate
            current_c5 = c5_candidate
        _iter += 1

    return [float(x) for x in h_refined], float(current_c5), int(n_points)
