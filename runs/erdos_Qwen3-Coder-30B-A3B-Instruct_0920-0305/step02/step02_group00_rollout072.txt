import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points_list = [48, 56, 60]
dx = 2.0 / n_points_list[0]  # Initial dx for the first n

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def project_onto_feasible(h, target_sum):
    """Project h onto box constraints [0,1] and sum constraint."""
    def f(λ):
        # Clip h - λ and sum
        x = np.clip(h - λ, 0, 1)
        return np.sum(x) - target_sum

    a, b = -10.0, 10.0
    for _ in range(30):
        mid = (a + b) / 2.0
        if f(mid) > 0:
            a = mid
        else:
            b = mid

    λ = (a + b) / 2.0
    return np.clip(h - λ, 0, 1)


def compute_softmin_weights(Cs, beta):
    """Compute softmin weights w_s = exp(-beta * C_s) / sum(exp(-beta * C_t))."""
    C_max = np.max(Cs)
    # Subtract the max to avoid overflow in exp
    exp_neg_beta_C = np.exp(-beta * (Cs - C_max))
    sum_exp = np.sum(exp_neg_beta_C)
    return exp_neg_beta_C / sum_exp


def gradient_step(h, Cs, beta, step_size):
    """Perform one gradient step on the soft-min objective."""
    ws = compute_softmin_weights(Cs, beta)
    # Compute the gradient ∇Φ = 2*dx * Σ_s w_s * A^{(s)} h
    g = np.zeros_like(h)
    n = len(h)
    for s in range(n):
        if s == 0:
            # Unshifted component: h
            g += ws[s] * h
        else:
            # Shifted components
            # h_{i+s} : right shift by s, fill with 0, then multiply by 0.5
            right_shifted = np.roll(h, -s)
            right_shifted[-s:] = 0.0
            # h_{i-s} : left shift by s, fill with 0, then multiply by 0.5
            left_shifted = np.roll(h, s)
            left_shifted[:s] = 0.0
            g += 0.5 * ws[s] * (right_shifted + left_shifted)

    # Scale by 2 * dx
    g *= 2 * dx
    # Update h by gradient ascent
    h_new = h + step_size * g
    return h_new


def compute_Cs(h_values):
    """Compute all correlation values C_s(h) = dx * sum(h_i * h_{i+s})."""
    h = np.array(h_values)
    n = len(h)
    Cs = np.zeros(n)
    for s in range(n):
        if s == 0:
            # h_i^2, doesn't involve a shift
            Cs[s] = dx * np.sum(h * h)
        else:
            # shift s, i from 0 to n-s-1
            # corresponding h_{i+s} indices are s to n-1
            # take part of h that overlaps
            sel = slice(0, n - s)
            shifted = h[s:]
            orig = h[sel]
            Cs[s] = dx * np.sum(orig * shifted)
    return Cs


def line_search_step(h_current, Cs_current, beta, step_size=0.5):
    """Perform backtracking line search."""
    h_try = gradient_step(h_current, Cs_current, beta, step_size)
    h_try = project_onto_feasible(h_try, len(h_current) / 2)
    Cs_try = compute_Cs(h_try)
    c5_try = np.max(Cs_try)  # Approximate worst shift
    c5_current = np.max(Cs_current)
    
    for _ in range(10):  # Max 10 backtrack steps
        if c5_try < c5_current:
            return h_try, Cs_try, c5_try
        step_size /= 2
        h_try = gradient_step(h_current, Cs_current, beta, step_size)
        h_try = project_onto_feasible(h_try, len(h_current) / 2)
        Cs_try = compute_Cs(h_try)
        c5_try = np.max(Cs_try)
    
    return h_current, Cs_current, c5_current

        # Generate additional random starts

        # Project binary start onto feasible set
        starts[1] = project_onto_feasible(starts[1], n / 2)
