import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT"""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2 * n)
    G = np.fft.rfft(1.0 - h, 2 * n)
    conv = np.fft.irfft(H * np.conj(G))[:2 * n - 1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve binary MILP for the Erdos problem with given n"""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    # Product linearization and worst-lag constraints
    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
        if prob.status in ["optimal", "optimal_inaccurate", "feasible"]:
            h_val = np.ravel(h.value)
            T_val = float(T.value)
            return T_val, h_val
        else:
            return None, None
    except Exception:
        return None, None

def project_to_simplex(h, M):
    """Project h to sum = M and bounds [0,1] using water-filling"""
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if abs(total - M) < 1e-10:
        return h
    # Water-filling
    sorted_h = np.sort(h)[::-1]
    cumsum = np.cumsum(sorted_h)
    # Find threshold index
    idx = np.argmax(sorted_h * np.arange(1, len(sorted_h)+1) - (M - cumsum + sorted_h))
    if idx == 0:
        # All elements > threshold
        threshold = (M - 1.0) / len(h)
        h = np.clip(h - threshold, 0, 1)
    else:
        # Unnormalized threshold
        threshold = (M - cumsum[idx-1]) / (len(h) - idx)
        h = np.clip(h - threshold, 0, 1)
    return h

def anneal_binary(h0, max_iter=1000000, time_limit=10):

    current_score = evaluate_h(h)

    best_score = current_score
    T0 = 0.01
    T = T0
    cooling = 0.9995
    
    start_time = time.time()
    num_swaps = 0
    
    # Precompute the typical Δ energy from swaps
    if True:
        delta_list = []
        for _ in range(100):
            i = np.random.randint(n)
            j = np.random.randint(n)
            # Find specific pairs that are valid
            if h[i] == 1 and h[j] == 0:
                temp = h.copy()
                temp[i] = 0
                temp[j] = 1
                new_score = evaluate_h(temp)
                delta_list.append(new_score - current_score)
        if delta_list:
            T0 = max(1e-6, 2.0 * max(1e-8, np.std(delta_list)))  # Use momentum of change
            T = T0
            
    while T > 1e-10 and (time.time() - start_time < time_limit - 0.1) and num_swaps < max_iter:
        # Try a swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == 1 and h[j] == 0:
            temp = h.copy()
            temp[i] = 0
            temp[j] = 1
            new_score = evaluate_h(temp)
            delta = new_score - current_score
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                h = temp
                current_score = new_score
                if new_score < best_score:
                    best_h = h.copy()
                    best_score = new_score
        T *= cooling
        num_swaps += 1
        if num_swaps % 10000 == 0:
            if time.time() - start_time > time_limit - 0.1:
                break
    
    # Final check of exact mass constraint just to be safe
    total = np.sum(best_h) * dx
    scale = 1.0 / total
    best_h *= scale
    best_score = evaluate_h(best_h)
    
    return best_h, best_score
