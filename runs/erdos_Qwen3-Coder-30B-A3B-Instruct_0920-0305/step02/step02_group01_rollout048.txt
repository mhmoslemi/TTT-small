def compute_worst_shift(h, dx):
    """Compute the shift that gives minimum overlap and its value."""
    auto = correlate(h, h, mode='full')
    # Find the minimum (worst) shift for the correlation
    min_idx = np.argmin(auto)
    worst_shift = min_idx - (len(h) - 1)
    worst_value = auto[min_idx] * dx
    return worst_shift, worst_value

def project_onto_feasible(h, dx, target_sum=1.0):
    """Project h onto the constraint that sum(h)*dx = target_sum."""
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    scale = target_sum / current_sum
    return h * scale

def capped_simplex_projection(g, h_old, dx, target_sum=1.0):
    """Solve linear program to maximize g^T h subject to 0 <= h <= 1, sum(h)*dx = target_sum."""
    # Simple capped simplex: sort by gradient, allocate mass to largest first
    n = len(g)
    idxs = np.argsort(-g)
    h_new = np.zeros(n)
    
    remaining_mass = target_sum / dx  # total mass in terms of discrete units
    for i in range(n):
        if remaining_mass <= 0:
            break
        idx = idxs[i]
        max_allowed = 1.0
        allocate = min(remaining_mass, max_allowed)
        h_new[idx] = allocate
        remaining_mass -= allocate
    
    # Normalize if necessary to ensure correct sum
    h_new = project_onto_feasible(h_new, dx, target_sum)
    
    return h_new

def compute_gradient_for_shift(h, shift):
    """Compute gradient of q_s(h) = sum_i h_i h_{i+s} w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if shift == 0:
        grad = 2 * h  # since q_s = sum h_i^2
    elif shift > 0:
        # h[i] contributes to h[i+shift], so grad[i] += h[i+shift]; h[i+shift] contributes to h[i], so grad[i+shift] += h[i]
        for i in range(n - shift):
            grad[i] += h[i + shift]
            grad[i + shift] += h[i]
    elif shift < 0:
        shift_abs = -shift
        for i in range(n - shift_abs):
            grad[i] += h[i + shift_abs]
            grad[i + shift_abs] += h[i]
    return grad

def update_h_dc(h, dx, target_sum=1.0):
    """Perform one DC step using capped simplex."""
    # Compute the worst shift
    worst_shift, _ = compute_worst_shift(h, dx)
    
    # Compute the gradient of q_{worst_shift}
    grad = compute_gradient_for_shift(h, worst_shift)
    
    # Solve linear program to maximize grad^T h subject to bounds
    h_new = capped_simplex_projection(grad, h, dx, target_sum)
    return h_new

def compute_c5(h, dx):
    """Compute the exact C5 bound."""
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
