def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1, sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    diff = target_sum - current_sum
    if abs(diff) < 1e-12:
        return h

    # Use water-filling algorithm
    if diff > 0:
        # Need to increase values - distribute diff among h < 1
        indices = np.where(h < 1)[0]
        if len(indices) > 0:
            increment = diff / len(indices)
            for i in indices:
                new_val = h[i] + increment
                if new_val > 1:
                    diff -= (1 - h[i])
                    h[i] = 1.0
                else:
                    h[i] = new_val
                    diff = 0
                    break
            # If diff still > 0, add to remaining elements
            if abs(diff) > 1e-12:
                # Just add remainder where possible
                for i in indices:
                    if abs(diff) < 1e-12:
                        break
                    if h[i] < 1:
                        to_add = min(diff, 1 - h[i])
                        h[i] += to_add
                        diff -= to_add
    else:
        # Need to decrease values
        indices = np.where(h > 0)[0]
        if len(indices) > 0:
            decrement = -diff / len(indices)
            for i in indices:
                new_val = h[i] - decrement
                if new_val < 0:
                    diff += h[i]
                    h[i] = 0.0
                else:
                    h[i] = new_val
                    diff = 0
                    break
            if abs(diff) > 1e-12:
                # Just subtract remainder where possible
                for i in indices:
                    if abs(diff) < 1e-12:
                        break
                    if h[i] > 0:
                        to_subtract = min(diff, h[i])
                        h[i] -= to_subtract
                        diff -= to_subtract

    return h

def compute_subgradient(h, best_lag, n_points, dx):
    """Compute subgradient for the current best lag"""
    g = np.zeros(n_points)
    m = best_lag
    # If m >= 0, the lag means h[i] * (1 - h[i+m])
    if m >= 0:
        # i ranges from 0 to n-1-m 
        for i in range(n_points - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        # If m < 0, use symmetry by swapping indices
        m = -m
        for i in range(m, n_points):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    return g * dx
