def evaluate_erdos_solution(h_values):
    """Evaluate the overlap integral for given h values."""
    h = np.array(h_values)
    # Compute correlations
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h_values)
    c5 = max_correlation * dx
    return c5
