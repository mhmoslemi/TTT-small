import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

# Constants
n_points = 48  # Slightly smaller for speed and cleaner LP
dx = 2.0 / n_points
target_bound = 0.3807
max_iter = 200
tol_obj = 1e-6
budget_s = 600

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_sum_constraint(h, target_sum):
    """Project h onto the hyperplane sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    # Simple bisection to find λ
    if current_sum > target_sum:
        lower, upper = 0, 1
        while upper - lower > 1e-8:
            mid = (lower + upper) / 2
            h_test = np.clip(h - mid, 0, 1)
            sum_test = np.sum(h_test)
            if sum_test > target_sum:
                lower = mid
            else:
                upper = mid
        h = np.clip(h - (lower + upper) / 2, 0, 1)
    else:
        lower, upper = 0, 1
        while upper - lower > 1e-8:
            mid = (lower + upper) / 2
            h_test = np.clip(h + mid, 0, 1)
            sum_test = np.sum(h_test)
            if sum_test < target_sum:
                lower = mid
            else:
                upper = mid
        h = np.clip(h + (lower + upper) / 2, 0, 1)
    return h

def build_linear_constraints(h_current):
    """For each shift s, compute the coefficients a_s and b_s for linearization."""
    n = len(h_current)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0.0
        for i in range(n - s):
            # Index mapping for the shifted variables
            a_s[i] += 1 - h_current[i + s] - h_current[i - s] if (i - s) >= 0 else 1 - h_current[i + s]
        b_s = -np.sum(h_current[i] * h_current[i + s] for i in range(n - s))
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_slp_subproblem(a_s_list, b_s_list, n, target_sum=0.5 * n_points):
    """Solve the linear program from the linearization."""
    h = Variable(n)
    t = Variable()
    constraints = [h[i] >= 0 for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    constraints += [sum(h) == target_sum]

    for s in range(n):
        constraints += [a_s_list[s] @ h + b_s_list[s] <= t]

    objective = Minimize(t)
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False)
        return h.value, t.value
    except:
        # Fallback to SCS or other solver if needed
        try:
            prob.solve(solver=SCS, verbose=False)
            return h.value, t.value
        except:
            return None, None

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    # Try different n_points if needed, but stick to consistent schedule
    n = n_points
    
    # Varying restarts with different seeding strategies
    num_starts = 15
    restarts = []
    
    # 1. Binary seed: Exactly n/2 ones and zeros
    binary_h = np.zeros(n)
    binary_h[:n//2] = 1
    np.random.shuffle(binary_h)
    restarts.append(binary_h.copy())
    
    # 2. Uniform seed with projection
    uniform_h = np.random.uniform(0, 1, n)
    uniform_h = project_onto_sum_constraint(uniform_h, 0.5*n)
    restarts.append(uniform_h.copy())
    
    # 3. Slightly interpolate the initial values (if available)
    try:
        interp_h = np.array(initial_h_values)
        if len(interp_h) == n_points:
            interp_h = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_points), interp_h)
            interp_h = project_onto_sum_constraint(interp_h, 0.5*n)
            restarts.append(interp_h.copy())
    except:
        pass
    
    # 4. Random foreign [0,1] vectors + projection
    for _ in range(10):
        r_h = np.random.uniform(0, 1, n)
        r_h = project_onto_sum_constraint(r_h, 0.5*n)
        restarts.append(r_h.copy())
    
    # Multi-start SLP loop
    iter_count = 0
    while time.time() - start_time < (budget_s * 0.95):
        for i, h_start in enumerate(restarts):
            if time.time() - start_time > (budget_s * 0.95):
                break
                
            h_current = h_start.copy()
            c5_prev = float('inf')
            
            for k in range(max_iter):
                if time.time() - start_time > (budget_s * 0.95):
                    break
                    
                # Compute correlation (used for our assessment of the current solution)
                o_s_vals = [np.correlate(h_current, 1 - np.roll(h_current, s), mode="valid")[0] for s in range(n)]
                max_o_s = np.max(o_s_vals) * dx
                c5_current = 1 - max_o_s  # We want to minimize max of (1 - O_s)
                c5_current = 1 - max_o_s  # Re-evaluate C5 correctly
                if np.isnan(c5_current) or abs(c5_current) > 1e6:
                    break
                
                # Check convergence
                if abs(c5_current - c5_prev) < tol_obj:
                    break
                    
                # Compute linearized constraints
                a_s_list, b_s_list = build_linear_constraints(h_current)
                
                # Solve subproblem
                h_dual, t_dual = solve_slp_subproblem(a_s_list, b_s_list, n, 0.5*n)
                if h_dual is None:
                    break
                h_new = np.array(h_dual) if isinstance(h_dual, list) else h_dual
                h_new = project_onto_sum_constraint(h_new, 0.5*n)
                
                # Evaluate true objective
                true_obj = evaluate_h(h_new)
                
                # Update global best
                if true_obj < best_c5:
                    best_h = h_new.copy()
                    best_c5 = true_obj
                    if best_c5 < target_bound:
                        break
                
                h_current = h_new
                c5_prev = true_obj
                iter_count += 1
                
        # Optional polishing step
        if best_h is not None:
            bounds = [(0, 1) for _ in range(n)]
            def obj(h):
                return evaluate_h(h)
            try:
                result = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                  options={'ftol': 1e-12, 'gtol': 1e-10, 'maxiter': 100})
                if result.success:
                    polished_h = result.x
                    polished_c5 = evaluate_h(polished_h)
                    if polished_c5 < best_c5:
                        best_h = polished_h
                        best_c5 = polished_c5
            except:
                pass

    # Final valid projection
    if best_h is not None:
        best_h = project_onto_sum_constraint(best_h, 0.5 * n)
        best_c5 = evaluate_h(best_h)

    return list(best_h), best_c5, n
