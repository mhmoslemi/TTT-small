import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto box constraints [0,1] and sum constraint."""
    lo = 0.0
    hi = 1.0
    
    # First clip to bounds
    w = np.clip(v, lo, hi)
    total = w.sum()
    
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
            
        if diff > 0:
            # Need to increase
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        else:
            # Need to decrease
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
            
        total = w.sum()
    
    return w

def compute_max_overlap(h, dx):
    """Compute maximum overlap integral and corresponding shift."""
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    # Compute shift m from index (mode="full" has indices from -(n-1) to (n-1))
    shift_m = argmax - (len(h) - 1)
    return max_val, shift_m

def gradient_for_shift(h, m):
    """Compute gradient of overlap integral for given shift m."""
    n = len(h)
    g = np.zeros_like(h)
    
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
        
    return g * (2.0 / n)  # dx = 2/n

def interpolate_to_n(h0, n):
    """Interpolate h0 from old number of points to n points."""
    old_n = len(h0)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, n, endpoint=False)
    h_interp = np.interp(new_x, old_x, h0)
    
    # Adjust to have exact sum (since dx = 2/n)
    current_sum = np.sum(h_interp)
    target_sum = n / 2
    if abs(current_sum) > 1e-12:
        scale = target_sum / current_sum
        h_interp *= scale
    
    # Ensure 0 <= h <= 1
    h_interp = np.clip(h_interp, 0, 1)
    
    # Final projection to ensure sum is exact
    h_interp = project_to_bounds_and_sum(h_interp, target_sum)
    
    return h_interp

def evaluate_h(h):

    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val
