import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization: use a moderate number of points for reasonable memory and speed
    n_points = 120
    dx = 2.0 / n_points
    
    # The constraint that integral equals 1 translates to sum(h)*dx = 1 => sum(h) = n_points/2
    target_sum = n_points / 2.0
    
    def project_onto_constraint(h):
        """Project h onto the constraint sum(h) = target_sum"""
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.ones_like(h) * (target_sum / n_points)
        scale = target_sum / h_sum
        return h * scale
    
    def solve_linear_program(grad, target_sum, lower=0, upper=1):
        """
        Solve the linear program:
        minimize <grad, h>
        subject to: lower <= h <= upper, sum(h) = target_sum
        
        Return the optimal h as a numpy array
        """
        # Sort indices in descending order of |grad|
        idx_sorted = np.argsort(-np.abs(grad))
        
        h = np.zeros_like(grad)
        remaining_sum = target_sum
        
        # Assign values greedily from largest |grad| to smallest
        for i in idx_sorted:
            if remaining_sum <= 0:
                break
            # Allocate as much as we can to h[i] without violating bounds
            alloc = min(upper - h[i], remaining_sum)  # Max additional we can take
            if alloc > 0:
                h[i] += alloc
                remaining_sum -= alloc
            else:
                continue
        
        # Now set remaining variables to bound if needed
        # Get indices with remaining sum to distribute
        remaining_indices = np.where(h < upper)[0]
        if remaining_sum > 0 and len(remaining_indices) > 0:
            remaining_size = len(remaining_indices)
            if remaining_size > 0:
                # Distribute the remaining sum evenly among available slots
                delta = min((upper - h[remaining_indices]).min(), remaining_sum / remaining_size)
                for i in remaining_indices:
                    if remaining_sum <= 0:
                        break
                    add = min(delta, remaining_sum)
                    h[i] += add
                    remaining_sum -= add
                    
        # Enforce final constraint exactly
        if remaining_sum > 0:
            # Set remaining positions to their maximum values
            for i in remaining_indices:
                if remaining_sum <= 0:
                    break
                add = min(upper - h[i], remaining_sum)
                h[i] += add
                remaining_sum -= add
        
        # Make sure it's within bounds again
        h = np.clip(h, lower, upper)
        # Now make sure sum constraint holds exactly
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust by reprojecting
            h = project_onto_constraint(h)
        return h

    def dc_step(h_old):
        """Perform one DC step."""
        # Compute all correlation values for shifts
        corr = correlate(h_old, h_old, mode="full")
        # The overlap from h(i) * (1 - h(i+k)) is represented as:
        # corr[s] = sum over i of h[i] * h[i-s]
        # But we want sum over i of h[i] * (1 - h[i+s]) = sum h[i] - sum h[i+s] * h[i]
        # Which is: sum(h_old) - corr[s] (up to indexing)
        corr_h1 = correlate(h_old, 1-h_old, mode="full")
        min_idx = np.argmin(corr_h1)
        min_value = corr_h1[min_idx]
        
        # For shift index min_idx, compute the gradient of the negative quantity
        # i.e., terms are h[i+s] + h[i-s]  where s = min_idx - n_points/2 for refl.
        shift_val = min_idx - (n_points // 2)  # shift from 0
        grad = np.zeros_like(h_old)
        
        # Forward and backward shift
        if shift_val > 0:
            grad[shift_val:] += h_old[:-shift_val]
            grad[:-shift_val] += h_old[shift_val:]
        elif shift_val < 0:
            shift_abs = -shift_val
            grad[shift_abs:] += h_old[:-shift_abs]
            grad[:-shift_abs] += h_old[shift_abs:]
        else:
            grad += 2 * h_old  # case shift = 0
        
        # Now solve primal linear program: minimize grad^T h subject to sum h = target_sum
        h_new = solve_linear_program(grad, target_sum)
        return h_new

    seeds = []
    
    # Seed 1: Uniform + small random perturbations
    h_uni = np.ones(n_points) * 0.5
    h_uni = project_onto_constraint(h_uni)
    seeds.append(h_uni.copy())
    
    # Seed 2: Initial shape from literature (if meaningful)
    h_init = np.array(initial_h_values)
    if len(h_init) != n_points:
        # interpolate if necessary
        h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
    h_init = project_onto_constraint(h_init)
    seeds.append(h_init.copy())
    
    # Seed 3: Optimized five-step shape
    h_five = np.zeros(n_points)
    npoints = len(h_five)
    segments = [0.25, 0.25, 0.25, 0.25]
    for i in range(len(segments)):
        start = int(i * npoints / len(segments))
        end = int((i+1) * npoints / len(segments))
        h_five[start:end] = segments[i] * 2.0
    h_five = project_onto_constraint(h_five)
    seeds.append(h_five.copy())
    
    # Seed 4: Random regular
    for _ in range(3):
        h_rnd = np.random.uniform(0, 1, n_points)
        h_rnd = project_onto_constraint(h_rnd)
        seeds.append(h_rnd.copy())
    
    # Compute C5 for all initial seeds
    best_h = None
    best_c5 = float('inf')
    for h_seed in seeds:
        c5_seed = evaluate_c5(h_seed)
        if c5_seed < best_c5:
            best_c5 = c5_seed
            best_h = h_seed.copy()
    
    # DC loop
    iteration = 0
    no_improvement = 0
    last_improvement = 0
    max_no_improvement = 30
    
    while (time.time() - start_time) < 0.8 * budget_s and no_improvement < max_no_improvement:
        h_new = dc_step(best_h)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject based on improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement = 0
            last_improvement = iteration
        else:
            no_improvement += 1
        
        iteration += 1
    
    # Final polishing: small gradient descent from best solution
    if (time.time() - start_time) < 0.95 * budget_s:
        # Smooth MAX refinement
        def smooth_max_corr(h, beta=1000.0):
            """Approximate max by log-sum-exp for smoothing gradient"""
            corr = correlate(h, 1-h, mode="full")
            return np.log(np.sum(np.exp(beta * (corr - np.max(corr))))) / beta + np.max(corr)
        
        def smooth_corr_grad(h, beta=1000.0):
            """Finite difference approximation of gradient"""
            eps = 1e-5
            g = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_onto_constraint(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_onto_constraint(h_minus)
                g[i] = (smooth_max_corr(h_plus, beta) - smooth_max_corr(h_minus, beta)) / (2 * eps)
            return g
        
        # Do a few GD steps
        h_final = best_h.copy()
        for _ in range(500):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            grad = smooth_corr_grad(h_final)
            # Projection step
            step_size = 1e-4
            h_final = h_final - step_size * grad
            h_final = project_onto_constraint(h_final)
        
        final_c5 = evaluate_c5(h_final)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
