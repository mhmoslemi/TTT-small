import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project vector h to the capped simplex {x: 0 <= x_i <= 1, sum x_i = target_sum}"""
    if np.allclose(np.sum(h), target_sum):
        return np.clip(h, 0, 1)

    def F(lam):
        # Computes sum of clipped values
        x = np.clip(h - lam, 0, 1)
        return np.sum(x) - target_sum

    # Binary search for lambda
    low, high = np.min(h), np.max(h) + 1.0
    for _ in range(40):
        mid = (low + high) / 2.0
        if F(mid) > 0:
            low = mid
        else:
            high = mid
    lambda_star = (low + high) / 2.0
    x = np.clip(h - lambda_star, 0, 1)
    return x

def smooth_max(c, alpha):
    """Return smooth approximation of max using softmax with temperature alpha"""
    exp_c = np.exp(alpha * c)
    weights = exp_c / np.sum(exp_c)
    return weights

def compute_grad(h, weights):
    """Compute gradient of smooth objective using convolutions"""
    n = len(h)
    # Reverse weights and compute convolutions
    w_rev = weights[::-1]
    g_plus = correlate(w_rev, h, mode='full')[n-1:n-1+n]
    g_minus = correlate(weights, h, mode='full')[n-1:n-1+n]
    grad = 1.0 - (g_plus + g_minus)
    return grad

def local_search_refinement(h0, n_points, budget_s, start_time):
    # Parameters for gradient descent
    max_iter = 5000
    alpha = 0.5
    eta = 0.5
    alpha_mult = 1.5
    eta_decay = 0.9
    patience = 500
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    dx = 2.0 / n_points

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break

        # Compute current correlation values
        corr = correlate(best_h, 1 - best_h, mode='full')
        weights = smooth_max(corr, alpha)
        grad = compute_grad(best_h, weights)

        # Projected gradient descent step
        h_new = best_h - eta * grad
        h_new = project_to_capped_simplex(h_new, n_points / 2.0)
        h_new = np.clip(h_new, 0, 1)

        # Evaluate and update
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        # Adaptive scheduling
        if i % 200 == 0:
            alpha = min(alpha * alpha_mult, 20.0)
            eta *= eta_decay

    return best_h, best_c5
