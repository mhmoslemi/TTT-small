def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex {x: 0<=x<=1, sum(x)=target_sum}"""
    # Using bisection on the Lagrange multiplier lambda
    h = np.array(h)
    low, high = -1000.0, 1000.0
    for _ in range(40):
        lam = (low + high) / 2.0
        x = np.clip(h - lam, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = lam
        else:
            high = lam
    x = np.clip(h - low, 0, 1)
    return x

def smooth_max(c, alpha=1.0):
    """Softmax approximation of max with temperature alpha"""
    w = np.exp(alpha * c)
    w = w / np.sum(w)
    return w

def gradient_step(h, alpha=0.5, eta=0.5):
    """Compute and apply one gradient step"""
    n = len(h)
    dx = 2.0 / n
    # Correlation: C_s = sum h_i * (1-h_{i+s})
    c = correlate(h, 1 - h, mode="full")
    # Smooth version of the max
    w = smooth_max(c, alpha)
    # Compute gradient
    g_plus = correlate(w[::-1], h, mode="full")[n - 1:n - 1 + n]
    g_minus = correlate(w, h, mode="full")[n - 1:n - 1 + n]
    grad = 1.0 - (g_plus + g_minus)
    # Update step
    h_new = h - eta * grad
    # Project
    h_new = project_to_capped_simplex(h_new, n / 2)
    return h_new

def build_five_step(a, b, n):

    u = (2 * b - 1) / (2 * (b - a)) if b > a else 0.0
    if not (0 <= u <= 1):
        return None
    # The five segments:
    # 0 - a - u - 1 - u - a - 0 (symmetric about x=1)
    # Left side:
    left_border = a
    right_border = 2.0 - a
    if left_border > 0.0 and b > 0.0:
        # from 0 to a: 0
        # from a to b: u
        # from b to 2-b: 1
        # from 2-b to 2-a: u
        # from 2-a to 2: 0
        for i in range(n):
            x_i = x[i]
            if x_i <= a or x_i >= 2 - a:
                h[i] = 0.0
            elif x_i <= b:
                h[i] = u
            elif x_i <= 2 - b:
                h[i] = 1.0
            else:
                h[i] = u
    # Renormalize to integral = 1:
    integral = np.sum(h) * dx
    h = h / integral * (n / 2)
    return h
