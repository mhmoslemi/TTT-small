import numpy as np
from scipy.fft import fft, ifft
import time
import cvxpy as cp

def evaluate_h(h):
    # evaluate C5 for a given h, dilated by dx
    n = len(h)
    dx = 2.0 / n
    h_padded = np.pad(h, (0, n), mode='constant')
    g_padded = np.pad(1 - h, (0, n), mode='constant')
    H = fft(h_padded)
    G = fft(g_padded)
    corr = ifft(H * np.conj(G)).real[:2*n - 1]
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 48  # start with reasonable size
    dx = 2.0 / n
    M = n // 2  # mass constraint
    T_target = int(np.ceil(0.3807 * n / 2))  # target bound converts to integer T
    best_c5 = float('inf')
    best_h = None

    # Phase 1: obtain a good binary solution via MILP if time allows
    if budget_s - (time.time() - t0) > 30:
        start_time = time.time()
        time_limit = min(0.35 * budget_s, budget_s - (time.time() - t0) - 10)
        
        # Binary variables
        h = cp.Variable(n, boolean=True)
        T = cp.Variable(integer=True)
        
        # Product variables p[ℓ][i] = h[i] * h[i+ℓ]
        p = {}
        for ℓ in range(1, n):
            p[ℓ] = cp.Variable((n - ℓ), boolean=True)
            
        # Constraints
        constraints = [
            cp.sum(h) == M,
        ]
        
        for ℓ in range(1, n):
            for i in range(n - ℓ):
                # McCormick envelopes for p[ℓ][i] = h[i]*h[i+ℓ]
                constraints.append(p[ℓ][i] <= h[i])
                constraints.append(p[ℓ][i] <= h[i + ℓ])
                constraints.append(p[ℓ][i] >= h[i] + h[i + ℓ] - 1)
                constraints.append(p[ℓ][i] >= 0)
                
        # Worst-lag constraint
        for ℓ in range(1, n):
            # This is effectively: sum_{i} p[ℓ][i] + T >= M
            # But we write a constraint that says T >= M - sum_{i} p[ℓ][i]
            # Since T is integer and sum p[ℓ][i] is integer, we have the equivalence
            constraints.append(T >= M - cp.sum(p[ℓ]))
        
        # Objective minimal T
        objective = cp.Minimize(T)
        prob = cp.Problem(objective, constraints)
        
        # Solve with CBC and time limit
        prob.solve(solver=cp.CBC, maxSeconds=time_limit)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = h.value
            T_val = T.value
            if T_val <= T_target:
                h_opt = np.round(h_val).astype(int)
                c5_opt = evaluate_h(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        else:
            # if it failed or timed out, get the best point
            h_val = h.value
            if h_val is not None:
                h_opt = np.round(h_val).astype(int)
                c5_opt = evaluate_h(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        
        # report progress
        print(f"Phase 1 (MILP): {time.time() - start_time:.2f}s")

    # Phase 2: binary local search
    if best_h is not None and best_c5 > 0.3807:
        if budget_s - (time.time() - t0) > 20:
            # binary swap-based simulated annealing
            start_time = time.time()
            h_bin = best_h.copy()
            c5_current = evaluate_h(h_bin)
            h_best = h_bin.copy()
            c5_best = c5_current
            
            # compute initial temperature via burn-in
            burn_in = 500
            deltas = []
            for _ in range(burn_in):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if h_bin[i] == 1 and h_bin[j] == 0:
                    h_bin[i], h_bin[j] = 0, 1  # swap
                    c5_new = evaluate_h(h_bin)
                    deltas.append(c5_new - c5_current)
                    c5_current = c5_new
                    h_bin[i], h_bin[j] = 1, 0  # revert
        
            # initial temperature
            tau = 1.5 * np.median(np.abs(deltas)) if len(deltas) > 10 else 1.0
            cooling_rate = 0.999
            min_temp = 1e-6
            
            max_iter = 20000 # set a cap for time
            for _ in range(max_iter):
                if time.time() - start_time > 0.35 * budget_s: break
                
                i = np.random.randint(n)
                j = np.random.randint(n)
                if h_bin[i] == 1 and h_bin[j] == 0:
                    h_bin[i], h_bin[j] = 0, 1  # swap
                    c5_new = evaluate_h(h_bin)
                    delta = c5_new - c5_current
                    if delta <= 0 or np.random.rand() < np.exp(-delta / tau):
                        c5_current = c5_new
                        if c5_current < c5_best:
                            c5_best = c5_current
                            h_best = h_bin.copy()
                    else:
                        h_bin[i], h_bin[j] = 1, 0  # revert
                    tau *= cooling_rate
            
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
            
            print(f"Phase 2 (binary SA): {time.time() - start_time:.2f}s")

    # Phase 3: fractional smoothing with projected gradient descent
    if best_h is not None and budget_s - (time.time() - t0) > 5:
        start_time = time.time()
        if time.time() - start_time > 0.2 * budget_s: 
            print(f"Phase 3 skipped (time): {time.time() - start_time:.2f}s")
            return (best_h.tolist(), best_c5, n)
            
        h_smooth = best_h.astype(float)
        beta = 1.0
        eta = 0.01
        for _ in range(3): # run a few beta stages
            if time.time() - start_time > 0.2 * budget_s: break
            # evaluate log-sum-exp surrogate F_beta
            # get all L_ell for ell = 1,...,n-1
            h_padded = np.pad(h_smooth, (0, n), mode='constant')
            g_padded = np.pad(1 - h_smooth, (0, n), mode='constant')
            H = fft(h_padded)
            G = fft(g_padded)
            corr = ifft(H * np.conj(G)).real[:2*n - 1]  # length = 2n - 1
            L = corr * dx  # L[ell] = sum h[i] (1 - h[i+ell])
            
            # Softmax weights
            exp_beta_L = np.exp(beta * L)
            weights = exp_beta_L / np.sum(exp_beta_L)
            
            # Compute weight-sum of gradients ∇L_ℓ(i) = 1 - h[i+ℓ] - h[i-ℓ]
            grad = np.zeros(n)
            for ell in range(1, n):  # ell from 1 to n-1
                # (i -> i+ell)
    
                # Reverse and shift 'h_smooth' to compute -h[i-ell]
                shifted_h = np.zeros_like(h_smooth)
                if ell < n:
                    shifted_h[ell:] = h_smooth[:-ell]
                else:
                    shifted_h = np.zeros_like(h_smooth)
                # handling i-ell == -1, i-ell == -2 etc. in this is a hack,
                # but equivalent to padding h with space |h| extra points at each side.
                # But let's calculate explicitly:
                # for ell = 1, range(0, n-1): i+ell = [1, n-1] =: k from 1 to n-1
                # so we need to sum from shift 0 to n-ell-1: i=0,...,n-ell-1; k = i+ell
                # i can [0, n-ell)
                # Gradient term: 1 - h[i+ell] 
                offset_h = np.zeros_like(h_smooth)
                if ell < n:
                    offset_h[ell:] = h_smooth[:-ell]
                else:
                    offset_h = np.zeros_like(h_smooth)
                
                # Gradient term is -1 + h[i+ell] + h[i-ell]
                term_gradients = 1 - h_smooth - offset_h
                # Accumulate with soft weights:
                grad += weights[ell-1] * term_gradients
            
            # gradient step
            h_new = h_smooth - eta * grad
	
            # project onto box [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Must maintain sum = M = n/2
            # Do water filling to compute the correct lambda
            # Compute masses (it's only a summation in the matrix form anyway)
            current_sum = np.sum(h_new)
            diff = current_sum - M  # positive if too much volume in the vector
            if abs(diff) > 1e-10:  # do water-filling step (binary search for lambda)
                def waterfilling_obj(lam):
                    h_test = np.clip(h_new + lam, 0, 1)
                    return np.sum(h_test) - M
                
                # Bisection to find lambda such that sum(clip(h_new + lambda, 0, 1)) = M
                # O(n) doing this function...
                low, high = -1.0, 1.0
                for _ in range(50):
                    lam_test = (low + high) / 2
                    val = waterfilling_obj(lam_test)
                    if val > 0:
                        high = lam_test
                    else:
                        low = lam_test
                lam = (low + high) / 2
                h_smooth = np.clip(h_new + lam, 0, 1)
            else:
                h_smooth = h_new
                
            # increase beta slowly
            beta *= 2.0
        
        # Final rejection check for feasibility and terminate
        h_smooth = np.clip(h_smooth, 0, 1)
        current_mass = np.sum(h_smooth) * dx
        h_smooth = h_smooth * (1.0 / current_mass)  # renormalise to exactly mass = 1
        
        final_c5 = evaluate_h(h_smooth)
        if final_c5 < best_c5:
            best_c5 = final_c5
            h_final = h_smooth
        else:
            h_final = best_h.copy()
        print(f"Phase 3 (frac smooth): {time.time() - start_time:.2f}s")

    # final check with evaluation
    final_c5 = evaluate_h(h_final)
    best_c5 = min(best_c5, final_c5)
    
    return (h_final.tolist(), best_c5, n)
