import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def evaluate_h(h):
    """Evaluate the C5 constant using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    H = rfft(h, 2*n)
    G = rfft(1.0 - h, 2*n)
    conv = irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP for T_opt = max_k sum h_i(1-h_{i+k})."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    try:
        prob.solve(solver=cp.CBC, verbose=False,
                   maxSeconds=time_limit)
    except:
        pass

    if prob.status in ["optimal", "optimal_inaccurate", "feasible"]:
        h_val = np.ravel(h.value) if h.value is not None else None
        T_val = T.value if T.value is not None else float('inf')
        return T_val, h_val
    else:
        return None, None

def anneal_binary(h0, max_iter=None, time_limit=60):
    """Binary swap-based simulated annealing preserving sum(h)=n/2."""
    n = len(h0)
    M = n // 2
    current_h = h0.copy()
    current_score = evaluate_h(current_h)
    best_h = current_h.copy()
    best_score = current_score
    start_time = time.time()

    # Estimate initial temperature from first few moves
    T0 = 0.0
    T0_samples = 0
    for _ in range(100):
        if time.time() - start_time > time_limit * 0.3:
            break
        i = np.random.randint(0, n)
        if current_h[i] == 0:
            continue
        j = np.random.randint(0, n)
        if current_h[j] == 1:
            continue
        # Swap
        test_h = current_h.copy()
        test_h[i] = 0
        test_h[j] = 1
        test_score = evaluate_h(test_h)
        delta = test_score - current_score
        if delta > 0:
            T0 += delta
            T0_samples += 1
    if T0_samples > 0:
        T0 /= T0_samples
    else:
        T0 = 0.1
    T = T0

    # Main loop
    iter_num = 0
    while time.time() - start_time < time_limit:
        if max_iter is not None and iter_num >= max_iter:
            break
        iter_num += 1
        i = np.random.randint(0, n)
        if current_h[i] == 0:
            continue
        j = np.random.randint(0, n)
        if current_h[j] == 1:
            continue

        # Swap
        test_h = current_h.copy()
        test_h[i] = 0
        test_h[j] = 1
        test_score = evaluate_h(test_h)

        delta = test_score - current_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            current_h = test_h
            current_score = test_score
            if current_score < best_score:
                best_h = current_h.copy()
                best_score = current_score
        T *= 0.999
        if T < 1e-6:
            T = 1e-6

    return best_h, best_score

def smooth_descent(h, time_limit=30):
    """Refine a discrete solution using smooth log-sum-exp gradient descent."""
    n = len(h)
    M = n // 2
    dx = 2.0 / n

    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    h = h * (M / np.sum(h))  # normalize to exact mass

    # Helper functions
    def compute_lags(h):
        n = len(h)
        L = []
        for ell in range(1, n):
            length = n - ell
            term = 0
            for i in range(length):
                term += h[i] * (1 - h[i + ell])
            L.append(term)
        return np.array(L)

    def project_simplex(h, target=M):
        """Project h onto the simplex with sum = target using bisection."""
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            return h

        # Binary search for lambda
        lbd_low = -1.0
        lbd_high = 2.0
        for _ in range(30):
            lbd = (lbd_low + lbd_high) / 2.0
            # Compute new h with lambda
            h_new = np.clip(h + lbd, 0.0, 1.0)
            s_new = np.sum(h_new)
            if s_new < target:
                lbd_high = lbd
            else:
                lbd_low = lbd
        # Final
        lbd = (lbd_low + lbd_high) / 2.0
        h = np.clip(h + lbd, 0.0, 1.0)
        h = h * (target / np.sum(h))  # ensure sum
        return h

    def step_size(h, g, beta=1.0):
        """Backtracking line search for step size."""
        eta = 0.1
        h_new = h - eta * g
        h_new = project_simplex(h_new)
        for _ in range(10):
            L = compute_lags(h_new)
            F_new = np.log(np.sum(np.exp(beta * L))) / beta
            L_old = compute_lags(h)
            F_old = np.log(np.sum(np.exp(beta * L_old))) / beta
            if F_new < F_old:
                return eta
            eta *= 0.5
        return eta

    # Algorithm parameters
    # Schedule β increasing from 1 to 20
    beta_schedule = [1, 2, 5, 10, 20]
    max_iter_per_beta = 100
    start_time = time.time()

    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        for _ in range(max_iter_per_beta):
            if time.time() - start_time > time_limit:
                break
            L = compute_lags(h)
            weights = np.exp(beta * L)
            weights = weights / np.sum(weights)

            g = np.zeros_like(h)
            for ell in range(1, n):
                length = n - ell
                # gradient of L_ell w.r.t h_i
                if ell < n:
                    for i in range(length):
                        if i + ell < n:
                            g[i] += weights[ell - 1] * (1 - h[i + ell])
                if ell < n and ell > 0:
                    for i in range(length):
                        if i - ell >= 0:
                            g[i] -= weights[ell - 1] * h[i - ell]
            # apply step and project
            eta = step_size(h, g, beta)
            h = h - eta * g
            h = project_simplex(h)

    h = project_simplex(h)
    return h, evaluate_h(h)
