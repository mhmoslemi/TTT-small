import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]   # full correlation
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve MILP for binary formulation of the Erdos problem."""
    M = n // 2
    # Variables
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    constraints = [cp.sum(h) == M, T >= 0]
    
    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        
        # Product linearisation constraints
        constraints += [p_ell <= h[:length], p_ell <= h[ell:], 
                        p_ell >= h[:length] + h[ell:] - 1]

        # Worst-lag constraint
        constraints += [cp.sum(p_ell) + T >= M]
    
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    # Timeout protection
    try:
        prob.solve(solver=cp.CBC, verbose=False, 
                   maxSeconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.ravel(h.value)
            t_opt = T.value
            return t_opt, h_opt
        else:
            # Fallback to best feasible if timeout or infeasible
            h_opt = np.ravel(h.value)
            # Re-evaluate the actual T from constraints
            t_estimated = 0
            for ell in range(1, n):
                length = n - ell
                p_val = np.ravel(p[ell-1].value)
                # Convert back to max overlap integral
                sum_p = np.sum(p_val) 
                t_estimated = max(t_estimated, M - sum_p)
            return t_estimated, h_opt
    except:
        return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=10):
    """Run swap-based simulated annealing for binary h."""
    n = len(h_start)
    M = n // 2   # target mass
    dx = 2.0 / n
    h = h_start.copy()
    current_energy = evaluate_h(h)
    best_h = h.copy()
    best_energy = current_energy
    temp = 1.0
    alpha = 0.999
    min_temp = 1e-6
    step = 0
    start_time = time.time()
    
    # Estimate initial temperature from few swaps
    initial_deltas = []
    num_samples = min(500, max_iter // 10)
    for _ in range(num_samples):
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 1 and h[j] == 0:
            old = h[i]
            h[i] = 0
            h[j] = 1
            new_energy = evaluate_h(h)
            delta = new_energy - current_energy
            initial_deltas.append(delta)
            h[i] = old
            h[j] = 0
    if initial_deltas:
        temp = np.std(initial_deltas) * 3
    
    while step < max_iter and time.time() - start_time < time_limit:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 1 and h[j] == 0:
            old = h[i]
            h[i] = 0
            h[j] = 1
            new_energy = evaluate_h(h)
            delta = new_energy - current_energy
            
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                current_energy = new_energy
                if current_energy < best_energy:
                    best_energy = current_energy
                    best_h = h.copy()
            else:
                h[i] = old
                h[j] = 0
                
        temp *= alpha
        if temp < min_temp:
            # Reheat if stuck (restarting temp estimate)
            temp = 0.5 * max(temp, 1e-3)
        step += 1
        
    return best_h, best_energy

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum; clips to [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) <= 1e-9:
        return h
    # Water-filling approach to solve: find lambda such that 
    # sum_i(clip(h_i + lambda, 0, 1)) = target_sum
    h_sorted = np.sort(h)[::-1]
    n = len(h)
    
    # Binary search on lambda
    lam_lo, lam_hi = -1.0, 1.0  # pessimistic bounds
    for _ in range(50):
        lam = (lam_lo + lam_hi) / 2.0
        s = sum(max(0, min(1, hi + lam)) for hi in h_sorted)
        if s >= target_sum:
            lam_lo = lam
        else:
            lam_hi = lam
    lam = (lam_lo + lam_hi) / 2.0
    h_proj = np.clip(h + lam, 0, 1)
    return h_proj

def smooth_descent(h_start, time_limit=5.0):
    """Smooth log-sum-exp descent with increasing β."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    # Compute initial overlaps
    def compute_lags(h):
        L = []
        for ell in range(1, n):
            shift = np.roll(h, -ell)
            # Optional: drop last ell entries to avoid wrap-around
            L.append(np.sum(h[:n-ell] * (1 - shift[:n-ell])))
        return np.array(L)
    
    beta_schedule = [1, 2, 5, 10, 20]
    start_time = time.time()
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        # Gradient descent for this β
        for iter_step in range(50):
            if time.time() - start_time > time_limit:
                break
            L_vals = compute_lags(h)
            if np.allclose(L_vals, 0.0):
                # Avoid numerical issues
                break
            weights = np.exp(beta * L_vals)
            weights /= weights.sum()
            
            # Gradient computation
            grad = np.zeros(n)
            for ell in range(1, n):
                if ell < n:
                    g = np.zeros(n)
                    g[:-ell] += 1
                    g[ell:] += -1
                    grad += weights[ell-1] * g
            
            # Take step
            eta = 0.1
            h_new = h - eta * grad
            
            # Project back
            h_new = project_to_simplex(h_new, 1.0)
            h = h_new
            
    return h, evaluate_h(h)
