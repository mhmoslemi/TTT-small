import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP formulation."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []  # list of product variables
    constraints = [cp.sum(h) == M, T >= 0]
    
    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        
        # McCormick envelopes for p_ell = h[i] * h[i+ell]
        constraints += [p_ell <= h[:length],
                        p_ell <= h[ell:],
                        p_ell >= h[:length] + h[ell:] - 1]
        
        # Worst-lag constraint: sum of products + T >= M
        constraints += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
    except:
        pass
    
    if prob.status in ["optimal", "optimal_inaccurate", "feasible"]:
        h_val = np.ravel(h.value)
        T_val = T.value
        return T_val, h_val
    return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=100):
    """Binary swap-based simulated annealing."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    current_score = evaluate_h(h)
    best_h = h.copy()
    best_score = current_score
    temp = 1.0  # initial temperature
    cooling_rate = 0.999
    min_temp = 1e-6
    start_time = time.time()
    
    # Estimate initial Δ from several swaps
    initial_deltas = []
    for _ in range(200):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_swap = h.copy()
            h_swap[i] = 0
            h_swap[j] = 1
            score_new = evaluate_h(h_swap)
            initial_deltas.append(max(0, score_new - current_score))
    if initial_deltas:
        temp = np.mean(initial_deltas) * 2.0
    
    iter_count = 0
    while temp > min_temp and iter_count < max_iter:
        if time.time() - start_time > time_limit:
            break
        iter_count += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = np.random.choice(ones)
            j = np.random.choice(zeros) 
            h_new = h.copy()
            h_new[i] = 0
            h_new[j] = 1
            new_score = evaluate_h(h_new)
            delta = new_score - current_score
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_new
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
        temp *= cooling_rate
    
    return best_h, best_score

def smooth_descent(h_start, time_limit=30):
    """Smooth log-sum-exp gradient descent."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    dx = 2.0 / n
    start_time = time.time()
    
    # project to simplex with constraint sum h * dx = 1
    def project_h(x):
        # Clip to [0, 1]
        x = np.clip(x, 0.0, 1.0)
        total = np.sum(x) * dx
        if abs(total - 1.0) < 1e-9:
            return x
        # Water-filling to adjust sum
        def water_fill(lam):
            x_new = np.clip(x + lam, 0.0, 1.0)
            return np.sum(x_new) * dx - 1.0
        # Bisection for lambda
        lb, ub = -1.0, 1.0
        for _ in range(30):
            mid = (lb + ub) / 2.0
            if water_fill(mid) > 0:
                ub = mid
            else:
                lb = mid
        lam = (lb + ub) / 2.0
        return np.clip(x + lam, 0.0, 1.0)
    
    # For β schedule
    beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0]
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        
        # Gradient descent for few iterations
        for _ in range(50):
            if time.time() - start_time > time_limit:
                break
            # Evaluate overlaps for all shifts
            overlaps = []
            for ell in range(1, n):
                s = 0.
                for i in range(n - ell):
                    s += h[i] * (1.0 - h[i + ell])
                overlaps.append(s)
            max_lag = max(overlaps)
            # Smooth surrogate F_beta
            exp_terms = [np.exp(beta * (ol - max_lag)) for ol in overlaps]
            ws = np.array(exp_terms) / sum(exp_terms)
            
            # Compute per-lag gradient components
            grad = np.zeros(n)
            for ell, w in enumerate(ws):
                for i in range(n - ell):
                    grad[i] += w * (1.0 - h[i + ell] - h[i - ell] if i - ell >= 0 else 0.0)
            # subtract right-shifted term if needed
            for ell, w in enumerate(ws):
                for i in range(n - ell):
                    grad[i + ell] -= w * h[i]
            # Update step with backtracking line search
            step = 0.1

                h_new = h - step * grad
                h_new = project_h(h_new)
                new_score = evaluate_h(h_new)
                if new_score < evaluate_h(h):
                    h = h_new
                    break
                else:
                    step *= 0.5
            else:
                h = project_h(h - 0.1 * grad)
    return h, evaluate_h(h)
