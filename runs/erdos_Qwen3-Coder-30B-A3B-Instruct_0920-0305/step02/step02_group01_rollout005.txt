import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def build_five_step(a, b, n_points):
    """Build a five-step symmetric function: 0-u-1-u-0"""
    dx = 2.0 / n_points
    x = np.arange(0.5, n_points, 1) * dx  # sample points
    u = (2 * b - 1) / (2 * (b - a))
    h = np.zeros(n_points)
    
    # Add µ to account for finite sampling;
    # stretch intervals by a factor to ensure integral = 1.
    # But calculate the right u value that keeps u in [0,1].
    if not (0 <= u <= 1):
        return None
    # span [0, a), [a, b), [b, 2-b), [2-b, 2-a), [2-a, 2)
    
    h[x < a] = 0
    h[(x >= a) & (x < b)] = u
    h[(x >= b) & (x < 2 - b)] = 1.0
    h[(x >= 2 - b) & (x < 2 - a)] = u
    h[x >= 2 - a] = 0
    # Normalize to get integral = 1
    integral = np.sum(h) * dx
    h = h / integral * 0.5 * n_points  # the ∫h=1 constraint
    
    return np.clip(h, 0, 1)

def project_to_capped_simplex(h, target_sum):
    """Project vector onto the capped simplex with zero-one bounds and fixed sum"""
    h = np.asarray(h)
    n = len(h)
    lb = 0
    ub = 1
    low, high = np.min(h), np.max(h)
    if low >= lb and high <= ub:
        # Already feasible
        scale = target_sum / np.sum(h)
        return h * scale
    
    # Simple dichotomy for λ such that sum(clipped(h - λ)) == target_sum
    def F(lam):
        return np.sum(np.clip(h - lam, lb, ub))
    
    a, b = low - 1, high + 1
    tolerance = 1e-10
    for _ in range(40):
        mid = (a + b) / 2
        f = F(mid)
        if f > target_sum:
            a = mid
        else:
            b = mid
        if abs(f - target_sum) < tolerance:
            break
    lam = (a + b) / 2
    res = np.clip(h - lam, lb, ub)
    # Final correction to exact sum
    scale = target_sum / np.sum(res)
    return res * scale

def smooth_max_conv(h, alpha=0.5):
    """Approximate the max over shifts using softmax with smoothing parameter alpha"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    c = corr * dx  # make physical by dx
    exp_c = np.exp(alpha * c)
    w = exp_c / np.sum(exp_c)
    return w

def gradient_smooth_max(h, alpha=0.5):
    """Compute gradient of smooth max objective using convolution"""
    n = len(h)
    dx = 2.0 / n
    w = smooth_max_conv(h, alpha)
    # compute convolution of h with 1-h at all shifts
    corr = correlate(h, 1 - h, mode="full")
    c = corr * dx  # physical scale
    
    # Gradient via convolutions
    # Each h_i contributes to positions i+s where s ranges over all shifts
    h_extended = np.concatenate([h, h])
    # convolve h with itself (symmetric part of w); place one sliding window
    ones = np.ones(n)
    # Precompute
    w_rev = w[::-1]
    g_plus = correlate(h_extended[:2*n], w_rev, mode="valid")
    g_minus = correlate(h_extended[n:2*n], w, mode="valid")
    # Now extract valid part for indices 0 to n-1
    g_plus = g_plus[n-1:n-1+n]
    g_minus = g_minus[n-1:n-1+n]
    # Final gradient
    grad = 1.0 - (g_plus + g_minus)
    return grad

def projected_gradient_descent(h0, max_iter=5000, alpha_start=0.5, eta_start=0.5):
    """Perform gradient descent with projection"""
    h = h0.copy()
    n = len(h0)
    target_sum = n * 0.5  # integral constraint
    alpha = alpha_start
    eta = eta_start
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for i in range(max_iter):
        # compute smooth objective gradient
        try:
            grad = gradient_smooth_max(h, alpha)
        except:
            # Fallback if smoothing fails
            grad = np.random.randn(n)
        h_new = h - eta * grad
        h_new = project_to_capped_simplex(h_new, target_sum)
        h = h_new
        
        # Periodic updates on smoothening and step size
        if i % 200 == 0:
            alpha *= 1.5
            alpha = min(alpha, 20)
            eta *= 0.9
        # Early stopping based on best result
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # If no improvement for long, early stop
        if i > 500 and i % 50 == 0:
            if i > 1000 and i % 100 == 0:
                pass
    return best_h, best_c5

def differential_evolution_fallback(best_h, n_points, budget_s):
    """Run a short DE to enhance global search"""
    np.random.seed(42)
    # Population size: 8, 500 generations max
    popsize = 8
    n_generations = 500
    F = 0.6
    CR = 0.9
    
    # Initialize population
    def project_pop(pop):
        for i in range(len(pop)):
            tmp = project_to_capped_simplex(pop[i], 0.5 * n_points)
            pop[i] = tmp
        return pop
    pop = [np.random.rand(n_points) for _ in range(popsize)]
    pop = project_pop(pop)
    
    c5s = [evaluate_c5(ind) for ind in pop]
    
    best_ind = pop[np.argmin(c5s)]
    best_c5 = np.min(c5s)
    
    # Evolve
    for gen in range(n_generations):
        if (time.time() - time.time() + budget_s) < 5:
            break
        new_pop = []
        for i in range(popsize):
            idxs = np.random.choice(popsize, 3, replace=False)
            a, b, c = pop[idxs]
            # mutation
            v = a + F * (b - c)
            # crossover
            mask = np.random.rand(n_points) < CR
            u = np.where(mask, v, pop[i])
            # carry out project
            u = project_to_capped_simplex(u, 0.5 * n_points)
            new_pop.append(u)
        # Selection: keep best
        new_c5s = []
        for ind in new_pop:
            new_c5s.append(evaluate_c5(ind))
        # Check if any better
        if min(new_c5s) < best_c5:
            best_idx = np.argmin(new_c5s)
            best_c5 = new_c5s[best_idx]
            best_ind = new_pop[best_idx]
        pop = new_pop
        c5s = new_c5s
    return best_ind, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: coarse grid search on (a,b)
    best_h = None
    best_c5 = float('inf')
    n_points = 200
    
    a_vals = np.linspace(0, 0.5, 51)
    b_vals = np.linspace(0.5, 1.0, 51)
    for a in a_vals:
        for b in b_vals:
            h = build_five_step(a, b, n_points)
            if h is not None:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Phase 2: gradient search
    if best_h is not None:
        best_h, best_c5 = projected_gradient_descent(best_h, max_iter=5000)
    
    # Phase 3: DE fallback
    if time.time() - start_time < 0.9 * budget_s:
        best_h, best_c5 = differential_evolution_fallback(best_h, n_points, budget_s)
    
    # Final correction
    dx = 2.0 / n_points
    h = project_to_capped_simplex(best_h, 0.5 * n_points)
    final_c5 = evaluate_c5(h)
    
    return (h.tolist(), final_c5, n_points)
