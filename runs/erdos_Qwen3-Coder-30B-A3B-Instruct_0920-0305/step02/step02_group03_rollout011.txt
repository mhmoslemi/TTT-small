import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Try increasing n_points to 100 for better resolution
    n_points = 100  # This enables better resolution for the optimization
    h0 = np.zeros(n_points)
    # Spreading ones more evenly to achieve better C5
    # Use a "balanced" pattern that reduces maximum overlap
    # We want about 50 ones distributed to minimize overlap
    
    # Use a "difference set" type approach: 
    # put 1's at positions where (i * golden_ratio) % 1 < 0.5
    golden_ratio = (1 + np.sqrt(5)) / 2
    positions = np.arange(n_points) * golden_ratio
    fractional = positions % 1
    indices = np.argsort(fractional)
    h0[indices[:n_points//2]] = 1
    
    # Normalize to satisfy integral constraint
    dx = 2.0 / n_points
    scale = 1.0 / (np.sum(h0) * dx)
    h0 = h0 * scale

    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Local optimization using SLSQP with constraints
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_candidate = res.x
            # Normalize to satisfy constraint
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Simulated Annealing for additional improvement
    def acceptance_probability(delta, temperature):
        if delta < 0:
            return 1.0
        return np.exp(-delta / temperature)
    
    # Parameters for simulated annealing
    T = 0.005
    Tmin = 1e-8
    alpha = 0.99
    
    # Start with a quasirandom binary construction
    h_current = best_h.copy()
    c5_current = evaluate_c5(h_current)
    
    best_h_final = h_current.copy()
    best_c5_final = c5_current

    # Track iterations to make sure we don't exceed time budget
    iterations = 0
    max_iterations = 10000  # Prevents infinite loops
    
    while (time.time() - start_time) < budget_s * 0.9 and iterations < max_iterations:
        # Perturbation: randomly flip a pair of bits
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Create new solution
        h_new = h_current.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Normalize
        scale = 1.0 / (np.sum(h_new) * dx)
        h_new = h_new * scale
        
        # Evaluate
        c5_new = evaluate_c5(h_new)
        
        # Acceptance criterion
        delta = c5_new - c5_current
        if acceptance_probability(delta, T) > np.random.rand():
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < best_c5_final:
                best_h_final = h_current.copy()
                best_c5_final = c5_current
                
        # Lower temperature
        T = max(Tmin, T * alpha)
        iterations += 1
    
    # Final check with local optimization on the best found solution
    try:
        res = minimize(objective, best_h_final, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-14, 'gtol': 1e-14})
        if res.success:
            h_candidate = res.x
            # Normalize to satisfy constraint
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5_final:
                best_h_final = h_candidate.copy()
                best_c5_final = c5_candidate
    except:
        pass
    
    return (best_h_final.tolist(), best_c5_final, n_points)
