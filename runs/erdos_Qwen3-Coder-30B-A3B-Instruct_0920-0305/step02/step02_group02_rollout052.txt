import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import random

# Constants
MAX_N = 80
MIN_N = 40
STEP_N = 8
TARGET_BOUND = 0.3807
SMOOTHING_START = 1.0
SMOOTHING_INCREASE = 2.0
SMOOTHING_MAX = 500.0
MAX_ITER_PER_RUN = 5000
BACKTRACKING_FACTOR = 0.5
BACKTRACKING_MAX = 10
PROJECTION_TOL = 1e-12
GRAD_TOL = 1e-6
TIME_PER_RESTART = 0.5
TOTAL_RESTARTS = 100

def evaluate_c5(h_values):
    """Evaluation function using numpy correlation."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_onto_capped_simplex(x, target_sum):
    """Project x onto the capped simplex: 0 <= x_i <= 1, sum(x) = target_sum."""
    x = np.copy(x)
    n = len(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    # Find threshold
    i = np.argmax(x_sorted * (np.arange(1, n+1)) > cumsum - target_sum)
    if i == 0:
        theta = (cumsum[0] - target_sum) / (n + 1)
        x = np.maximum(x - theta, 0)
    else:
        theta = (cumsum[i-1] - target_sum) / i
        x = np.maximum(x - theta, 0)
    # Ensure exact sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        idx = np.argmin(x)
        x[idx] += diff
    return x

def compute_correlations(h):
    """Compute all correlations via FFT."""
    n = len(h)
    h_fft = fft(h)
    h_conj_fft = np.conj(h_fft)
    cross_fft = h_fft * h_conj_fft
    # 2n-1 results
    cross = ifft(cross_fft).real
    return cross[n-1:]

def gradient_of_rk(h, k, n):
    """Compute gradient of r_k(h)."""
    g = np.zeros(n)
    g += 1.0
    if k > 0:
        g[k:] -= h[:-k]
    if k < n - 1:
        g[:-k] -= h[k:]
    return g

def compute_grad_f_alpha(h, r_values, alpha):
    """Compute gradient of log-sum-exp surrogate."""
    n = len(h)
    r_values = np.array(r_values)
    r_max = np.max(r_values)
    shifted = r_values - r_max
    numerators = np.exp(alpha * shifted)
    denom = np.sum(numerators)
    weights = numerators / denom
    grad = np.zeros(n)
    for i, w in enumerate(weights):
        grad += w * gradient_of_rk(h, i, n)
    return grad

def backtracking_line_search(h, grad, dx, w, alpha, step_size, func):
    """Backtracking line search for sufficient decrease."""
    t = step_size
    for _ in range(BACKTRACKING_MAX):
        h_new = project_onto_capped_simplex(h - t * grad, w)
        val_new = func(h_new, alpha, dx)
        val_curr = func(h, alpha, dx)
        if val_new < val_curr - 0.5 * t * np.dot(grad, grad):
            return h_new, t
        t *= BACKTRACKING_FACTOR
    return h, step_size  # fallback

def projected_grad_descent(h0, dx, w, alpha0, alpha_factor, alpha_max, max_iter):
    """Accelerated projected gradient descent for max-min."""
    n = len(h0)
    
    # Initializations
    h = h0.copy()
    y = h0.copy()
    t = 1
    alpha = alpha0
    
    last_val = np.inf
    
    for i in range(max_iter):
        # Compute all r_k
        r_vals = compute_correlations(h)  # size n, shifted to output [0, n-1]
        # Only use first half for non-zero lag
        r_vals = r_vals[:n-1]
        
        # Function value
        r_max = np.max(r_vals)
        log_sum_exp = np.log(np.sum(np.exp(alpha * (r_vals - r_max))) + 1e-15) + r_max
        val = dx * log_sum_exp / alpha
        
        if abs(val - last_val) < GRAD_TOL:
            break
        last_val = val

        # Gradient computation
        grad = compute_grad_f_alpha(h, r_vals, alpha)
        grad *= dx

        # Step size
        step_size = 0.5
        h_new, actual_step = backtracking_line_search(h, grad, dx, w, alpha, step_size, lambda h_new, a, d: d * np.log(np.sum(np.exp(a * (compute_correlations(h_new)[:n-1] - np.max(compute_correlations(h_new)[:n-1])))) + 1e-15) / a)

        # Update
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t**2))
        y_new = h_new + ((t - 1) / t_new) * (h_new - h)
        h = h_new
        y = y_new
        t = t_new

        # Update α
        if alpha < alpha_max:
            alpha *= alpha_factor
            if i % 50 == 0:
                # Re-evaluate
                r_vals = compute_correlations(h)[:n-1]
                r_max = np.max(r_vals)
                log_sum_exp = np.log(np.sum(np.exp(alpha * (r_vals - r_max))) + 1e-15) + r_max
                val = dx * log_sum_exp / alpha
                last_val = val

    return h, val

def initialize_h(n, w, seed):
    """Initialise with heuristics."""
    np.random.seed(seed)
    # Random biased start
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, w)
    return h

def local_sqps_refine(h0, dx, w):
    """Refinement using SLSQP."""
    def obj(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def con_eq(h):
        return np.sum(h) * dx - 1.0

    lb = [0.0] * len(h0)
    ub = [1.0] * len(h0)

    result = minimize(
        obj,
        h0,
        method='SLSQP',
        bounds=list(zip(lb, ub)),
        constraints={'type': 'eq', 'fun': con_eq},
        options={'maxiter': 200, 'ftol': 1e-9}
    )
    if result.success:
        h = project_onto_capped_simplex(result.x, w)
        return h, obj(h)
    return h0, obj(h0)

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    n = None

    # Sample sizes
    n_list = list(range(MIN_N, MAX_N + 1, STEP_N))
   	n_list = [x for x in n_list if x > 0]
    n_list.sort()

    time_per_n = budget_s / (len(n_list) + 1)
    
    for n in n_list:
        dx = 2.0 / n
        w = n / 2
        n_time = 0
        restart_count = 0

        while time.time() - start_time + n_time < time_per_n * 0.95:
            if restart_count > TOTAL_RESTARTS:
                break
            restart_count += 1
            
            h0 = initialize_h(n, w, seed+restart_count)
            
            try:
                h_opt, c5_opt = projected_grad_descent(
                    h0, dx, w,
                    alpha0=SMOOTHING_START,
                    alpha_factor=SMOOTHING_INCREASE,
                    alpha_max=SMOOTHING_MAX,
                    max_iter=MAX_ITER_PER_RUN
                )
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
            except Exception as e:
                pass

        if time.time() - start_time > budget_s * 0.95:
            break

    # Final refinement
    if best_h is not None:
        h_refine, c5_refine = local_sqps_refine(best_h, 2.0/len(best_h), len(best_h)/2)
        if c5_refine < best_c5:
            best_h, best_c5 = h_refine, c5_refine

    # Ensure constraints
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, len(best_h)/2)
        final_c5 = evaluate_c5(best_h)
        return list(best_h), final_c5, len(best_h)

    return [0.0] * 48, 1.0, 48
