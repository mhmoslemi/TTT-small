def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx

    return c5

def solve_sdp(n):

    h = cp.Variable(n)

    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == cp.multiply(h, h),  # X_ii = h_i^2
        cp.sum(h) == w,
        h >= 0,
        h <= 1
    ]
    
    # Objective
    # For simplicity, we'll start with a trial value
    t = cp.Variable()
    # We want to maximize t subject to trace(T_k * X) >= t for all k
    # This is done by solving SDP for each k and taking max over all k
    # Here we assume it is numeric
    object = cp.Maximize(t)
    # For now, set a placeholder value by solving an approx
    try:
        prob = cp.Problem(object, constraints)

def project_to_simplex(v, w):
    """Project v onto the capped simplex with sum = w."""
    u = np.sort(v)[::-1]
    tmp = (np.cumsum(u) - w) / np.arange(1, len(u) + 1)
    idx = np.argmax(u - tmp)
    theta = tmp[idx]
    x = np.maximum(v - theta, 0)
    x = np.minimum(x, 1)
    # Adjust total to be exactly w
    diff = w - np.sum(x)
    if abs(diff) > 1e-10:
        if diff > 0:
            pos = np.where(x > 0)[0]
            if len(pos) > 0:
                x[pos[0]] += diff
        elif diff < 0:
            pos = np.where(x < 1)[0]
            if len(pos) > 0:
                x[pos[0]] += diff
    return x

def fourier_reconstruct(a, b, n):
    """Reconstruct h from Fourier coefficients."""
    m = len(a)
    h = 0.5 * np.ones(n)

        d = (i + 1) * np.pi
        h += a[i] * np.cos(d * np.arange(n) / n) + b[i] * np.sin(d * np.arange(n) / n)
    return np.clip(h, 0, 1)

def fourier_fit(h_in, m):
    """Fit h_in to a truncated Fourier expansion."""
    n = len(h_in)
    # Construct phase matrix
    Ph = np.zeros((n, 2 * m))
    for i in range(n):
        for j in range(m):
            Ph[i, 2*j] = np.cos((j+1)*np.pi*i/n)
            Ph[i, 2*j+1] = np.sin((j+1)*np.pi*i/n)
    # The offset is 0.5, so we adjust target
    y = h_in - 0.5

    # Use lstsq with a small box constraint or perform one step projection
    try:
        # Find coefficients by solving least square problem with constraints (0 <= h <= 1)
        a_b = np.linalg.lstsq(Ph, y, rcond=None)[0]
        return a_b
    except:
        a_b = np.zeros(2 * m)
        return a_b
