import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=48 for better resolution, manageable MILP size
    n_points = 48
    dx = 2.0 / n_points
    M = n_points // 2  # number of ones required
    
    # Attempt MILP approach first (shortest computation time)
    best_h = None
    best_c5 = float('inf')
    
    try:
        # Use CVXPY to solve MILP
        h = cp.Variable(n_points, boolean=True)
        T = cp.Variable(integer=True)
        
        constraints = [cp.sum(h) == M, T >= 0]
        p_vars = []

        # Product variables for each shift
        for ell in range(1, n_points):
            length = n_points - ell
            p_ell = cp.Variable(length, boolean=True)
            p_vars.append(p_ell)
            # Constrain p_ell to be h_i * h_{i+ell}
            constraints += [
                p_ell <= h[:length],
                p_ell <= h[ell:ell+length],
                p_ell >= h[:length] + h[ell:ell+length] - 1
            ]
            # Add constraint sum_i p_iℓ + T >= M for this shift
            constraints += [cp.sum(p_ell) + T >= M]
        
        objective = cp.Minimize(T)
        prob = cp.Problem(objective, constraints)
        
        # Allocate time for MILP solver
        remaining_time = budget_s - (time.time() - start_time) - 10  # leave 10s buffer
        if remaining_time > 0:
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=remaining_time)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_val = h.value
                T_val = T.value
                h_float = h_val.astype(float).tolist()
                c5_bound = T_val * dx
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_float
    except Exception as e:
        pass  # Fall through to heuristics if not feasible

    # If MILP fails or gives high bound, fallback to heuristic
    if best_h is None or best_c5 > 0.4:
        # Generate a random binary solution with M ones
        h_init = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, size=M, replace=False)
        h_init[ones_indices] = 1
        best_h = h_init.tolist()
        best_c5 = evaluate_h(best_h)
        
        # Local refinement with simulated annealing (binary swaps)
        current_h = np.array(best_h)
        current_c5 = best_c5
        temp = 0.1
        cooling_rate = 0.999
        min_temp = 1e-6
        max_iter = 50000
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Make a random swap between 0 and 1
            idx1 = np.random.randint(0, n_points)
            idx2 = np.random.randint(0, n_points)
            
            # Ensure we're swapping 0 and 1
            if current_h[idx1] == current_h[idx2]:
                continue
            
            # Try new solution
            h_new = current_h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            c5_new = evaluate_h(h_new)
            
            # Accept if better or with probability
            if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                current_h = h_new
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.tolist()
                    
            temp *= cooling_rate

    # Final refinement using continuous optimization if within time
    if time.time() - start_time < budget_s * 0.95:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_eq(x):
                return np.sum(x) * dx - 1.0
            constr = {'type': 'eq', 'fun': constraint_eq}
            
            # Use SLSQP to refine the solution
            result = minimize(evaluate_h, np.array(best_h), method='SLSQP', 
                              bounds=bounds, constraints=constr, 
                              options={'maxiter': 1000, 'ftol': 1e-9})
            if result.success:
                refined_h = result.x.tolist()
                refined_c5 = evaluate_h(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
    
    # Ensure values are within [0,1] 
    best_h = [min(1, max(0, val)) for val in best_h]
    
    return (best_h, best_c5, n_points)
