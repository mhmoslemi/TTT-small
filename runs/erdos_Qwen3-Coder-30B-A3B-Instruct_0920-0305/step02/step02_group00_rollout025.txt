def evaluate_h(h_values):
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h, target_sum):
    """Project h to the feasible set: [0,1]^n and sum equals target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Renormalize to match sum constraint using bisection
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if current_sum == 0:
        h = np.full_like(h, target_sum / len(h))
        return h
    # Simple incremental adjustment (can be improved)
    # To ensure sum = target_sum, we scale appropriately
    h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def solve_lp_step(h_curr, n):
    """
    Solve the linear program:
        minimize t
        subject to a_s^T * h + b_s <= t for all s,
                  0 <= h_i <= 1 for all i,
                  sum(h) = n/2
    """
    # Precompute for all shifts
    # For every s from 0 to n-1, we want to compute:
    # a_s[i] = 1 - h_curr[i+s] - h_curr[i-s] (ignore out of bounds with 0)
    # b_s = - sum(h_curr[i] * h_curr[i+s]) * dx  

    # Create data structures for a_s and b_s
    a_s_list = []
    b_s_list = []
    dx = 2.0 / n
    target_sum = n / 2

    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0.0
        # For shift s
        sum_term = 0.0
        for i in range(n):
            h_i = h_curr[i]
            h_i_plus_s = h_curr[i+s] if i+s < n else 0.0
            h_i_minus_s = h_curr[i-s] if i-s >= 0 else 0.0
            a_s[i] = 1.0 - h_i_plus_s - h_i_minus_s
            sum_term += h_i * h_i_plus_s

        b_s = -sum_term * dx
        a_s_list.append(a_s)
        b_s_list.append(b_s)

    # Now, solve the LP
    h_var = Variable(n)
    t_var = Variable(1)  # scalar variable

    # linear constraints
    constraints = []
    for i in range(n):
        constraints.append(a_s_list[i] @ h_var + b_s_list[i] <= t_var)
    constraints.append(0 <= h_var)
    constraints.append(h_var <= 1)
    constraints.append(sum_entries(h_var) == target_sum)

    # objective
    objective = Minimize(t_var)
    problem = Problem(objective, constraints)

    try:
        problem.solve(solver=cp.CBC, verbose=False)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            new_h = h_var.value.flatten()
            new_t = t_var.value
            return new_h, new_t
    except:
        pass

    # fallback to safe point
    # project h_curr
    return project_to_feasible(h_curr, target_sum), evaluate_h(h_curr)
