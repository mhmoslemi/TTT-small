def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)

    correlation = np.correlate(h, 1 - h, mode="full")

def build_linear_constraints(h_cur):

        b_s = -np.sum(h_cur[:n-s] * h_cur[s:])
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list
