def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def smooth_max(h_values, alpha=20.0):
    """Compute smooth approximation of max of correlations."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Use log-sum-exp for smooth approximation of max
    max_corr = np.max(correlation)
    exp_terms = np.exp(alpha * (correlation - max_corr))
    smoothed = max_corr + (1.0 / alpha) * np.log(np.sum(exp_terms))
    return smoothed

def compute_gradient(h_values, alpha=20.0):
    """Compute gradient of smooth max w.r.t. h."""
    h = np.array(h_values)
    # Compute correlation values for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # Compute weights for each shift
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient
    grad = np.zeros_like(h)
    n = len(h)
    
    # Compute contribution from each shift
    for s in range(-(n-1), n):
        if s >= 0:
            # Positive shifts: h[i] * h[i+s] contribution
            if s < n:
                # columns for indices i, i+s
                indices_i = np.arange(n - s)
                indices_is = np.arange(s, n)
                # edit coefficient in grad
                grad[indices_i] -= weights[s + (n-1)] * dx * h[indices_is]
                grad[indices_is] -= weights[s + (n-1)] * dx * h[indices_i]
        else:
            # Negative shifts: h[i] * h[i+s] contribution
            if -s < n:
                indices_i = np.arange(-s, n)
                indices_is = np.arange(n + s)
                grad[indices_i] -= weights[s + (n-1)] * dx * h[indices_is]
                grad[indices_is] -= weights[s + (n-1)] * dx * h[indices_i]
    
    # Add linear term contribution
    scaling = dx * np.sum(weights)
    grad += scaling - 2 * dx * h * weights[0 + (n-1)]
    
    return grad

def project_to_simplex(v, target_sum):
    """Project vector v onto simplex defined by sum = target_sum and bounds [0,1]."""
    v = np.array(v)
    n = len(v)
    
    # Simple iterative projection
    # Clip to [0,1] first 
    clipped = np.clip(v, 0, 1)
    current_sum = np.sum(clipped)
    
    # If already correct, done
    if abs(current_sum - target_sum) < 1e-8:
        return clipped
    
    if current_sum < target_sum:
        # We need to increase values
        # Find target increasing rate
        scale = target_sum / current_sum
        clipped *= scale
        clipped = np.clip(clipped, 0, 1)  # Clip again
        return clipped
    
    # current_sum > target_sum, need to decrease
    # Binary search on shift
    def project_with_shift(shift):
        temp = v - shift
        temp = np.clip(temp, 0, 1)
        return np.sum(temp)
    
    # upper bound for valid shift
    lower_bound = 0.0
    upper_bound = np.max(v) + 1.0
    
    # Binary search  
    tol = 1e-8
    while upper_bound - lower_bound > tol:
        mid = (lower_bound + upper_bound) / 2
        sum_mid = project_with_shift(mid)
        if sum_mid > target_sum:
            lower_bound = mid
        else:
            upper_bound = mid
    
    final_v = v - upper_bound
    final_v = np.clip(final_v, 0, 1)
    return final_v

def gradient_descent_step(h_values, learning_rate=0.01, alpha=20.0):
    """One gradient descent step."""
    grad = compute_gradient(h_values, alpha)
    new_h = np.array(h_values) - learning_rate * grad
    # Project onto feasible set
    new_h = project_to_simplex(new_h, n_points / 2.0)
    return new_h
