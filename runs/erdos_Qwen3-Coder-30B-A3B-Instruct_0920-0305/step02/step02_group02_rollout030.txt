def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def autocorr_h(h):
    """Compute auto-correlations of h for all shifts."""
    n = len(h)
    # correlation of h with itself
    corr = np.correlate(h, h, mode="full")
    # Extract correlation for non-negative shifts: 
    # shift 0 is at index n-1, shift k at index n-1+k
    S = corr[n-1:]
    return S

def grad_S_k(h, k):
    """Compute gradient of S_k with respect to h."""
    n = len(h)
    if k == 0:
        # S_0 = sum(h_i^2) => dS_0/dh_i = 2 * h_i
        return 2 * h
    else:
        # S_k = sum_{i=0}^{n-1-k} h_i * h_{i+k}
        # Gradient: dS_k/dh_i = h_{i+k} if i+k < n else 0 + h_{i-k} if i-k >= 0 else 0
        g = np.zeros_like(h)
        g[:n-k] += h[k:]   # Contribution from h_{i+k}
        g[k:] += h[:n-k]   # Contribution from h_{i-k}
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto [lower, upper] simplex with target sum."""
    v = np.array(v)
    n = len(v)
    # Shift to [0, upper-lower] simplex
    u = v - lower
    u_upper = upper - lower
    # Renormalize target sum
    target_u = target_sum - n * lower
    
    # Sort in descending order
    sorted_u = np.sort(u)[::-1]
    # Cumulative sum
    cumsum = np.cumsum(sorted_u)
    # Find threshold index
    idx = np.arange(1, n+1)
    threshold = (cumsum - target_u) / idx
    # Find largest index where threshold <= sorted_u[index]
    mask = sorted_u > threshold
    if not np.any(mask):
        # All elements must be 0 after projection
        return np.full(n, lower)
    # Get the threshold value
    i = np.argmax(mask) if np.any(mask) else 0
    if i == 0:
        tau = threshold[0]
    else:
        tau = threshold[i-1]
    # Project onto [0, upper-lower] simplex
    x = np.maximum(np.minimum(u - tau, u_upper), 0) 
    # Project back to [lower, upper]
    result = x + lower
    # Adjust to ensure sum = target_sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    # Adjust scalar
    if abs(diff) > 1e-10:
        adjustment = diff / n
        result = np.clip(result + adjustment, lower, upper)
    return result
