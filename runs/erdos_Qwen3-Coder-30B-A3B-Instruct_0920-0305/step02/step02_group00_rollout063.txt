n_points_list = [48, 56, 60]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to feasible(h, target_sum):
    """Project h onto the feasible set: box [0,1]^n and equality constraint sum=h_target."""
    # For each value v, find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    v = h.copy()
    
    def f(lam):
        clipped = np.clip(v - lam, 0, 1)
        return np.sum(clipped) - target_sum
    
    # Use bisection to find lambda s.t. f(lambda)=0.
    a, b = -2.0, 2.0  # range of possible lambda
    eps = 1e-10
    for _ in range(50):  # ~log2(4/eps)
        mid = (a + b) / 2.0
        val = f(mid)
        if val > 0:
            a = mid
        else:
            b = mid
    lambda_val = (a + b) / 2.0
    
    # Apply the projection
    proj = np.clip(v - lambda_val, 0, 1)
    return proj

def compute_shifted_sum(h, s):
    """Compute sum h_i * h_{i+s} for shift s efficiently."""
    if s == 0:
        return np.sum(h * h)
    elif s > 0:
        return np.sum(h[:-s] * h[s:])
    else:
        return np.sum(h[-s:] * h[:s])

def compute_all_shifted_sums(h):
    """Compute O_s = sum h_i * h_{i+s} for all shifts s=[0,n-1]."""
    n = len(h)
    O_s = np.zeros(n)
    for s in range(n):
        O_s[s] = compute_shifted_sum(h, s)
    return O_s

def compute_Cs(h):
    """Compute the full list of C_s values (each is dx * O_s)."""
    O_s = compute_all_shifted_sums(h)
    return dx * O_s

def compute_grad_Cs(h, C_s):
    """Compute the gradient of the soft-min objective w.r.t. h."""
    n = len(h)
    alpha = np.exp(-beta * C_s)
    Z = np.sum(alpha)
    if Z == 0:
        w_s = np.zeros_like(C_s)
    else:
        w_s = alpha / Z
    
    # Compute the sum of weighted shift operations (as matrix multiply)
    g = np.zeros(n)
    
    # For each shift s
    for s in range(n):
        if s == 0:
            # This is simply 2 * h
            g += 2 * w_s[s] * h
        else:
            if s > 0:
                # g[i] += w_s[s] * 0.5*(h[i-s] + h[i+s])
                g[s:] += 0.5 * w_s[s] * h[:-s]
                g[:-s] += 0.5 * w_s[s] * h[s:]
            else:
                # s < 0, same but negative shift
                s_abs = -s
                g[s_abs:] += 0.5 * w_s[s] * h[:-s_abs]
                g[:-s_abs] += 0.5 * w_s[s] * h[s_abs:]
    
    return 2 * dx * g

def backtracking_line_search(h, grad, C_s, beta, alpha=step_size_initial,

    t = evaluate_h(h)  # Current value of max L_s

def run_gradient_ascent(h_init, beta_schedule, max_inner_iter, n_points):
    """Main gradient ascent loop."""
    h = h_init.copy()
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_value = evaluate_h(h)
    h = project_to_feasible(h, target_sum)
    
    for beta in beta_schedule:
        for i in range(max_inner_iter):
            C_s = compute_Cs(h)
            grad = compute_grad_Cs(h, C_s)
            
            # Line search with backtracking
            h_new, t_new, step_size = backtracking_line_search(h, grad, C_s, beta)
            h = h_new
            
            # Project to feasible again
            h = project_to_feasible(h, target_sum)
            
            # Update best solution
            if t_new < best_value:
                best_h = h.copy()
                best_value = t_new
                
            # Early exit if no improvement
            if i > 10 and abs(t_new - evaluate_h(h)) < 1e-7:
                break
            
            # If step size becomes too small, break
            if step_size < 1e-10:
                break
    
    # Final evaluation
    best_value = evaluate_h(best_h)
    
    return best_h, best_value

def run_single_n(n_points, seed):
    """Run optimization for a specific n."""    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start points
    start_points = []
    
    # Interpolate initial_h_values (which is available globally)
    try:
        from initial_h_values import initial_h_values
        larger_h = np.array(initial_h_values)
        if len(larger_h) > n_points:
            h_interp = larger_h[:n_points]  # truncate
        elif len(larger_h) < n_points:
            h_interp = np.interp(np.linspace(0, 1, n_points), 
                                np.linspace(0, 1, len(larger_h)), 
                                larger_h)
        else:
            h_interp = larger_h.copy()
        start_points.append(project_to_feasible(h_interp, target_sum))
    except:
        # Fallback: make a uniform start
        h_interp = np.ones(n_points) * 0.5
        h_interp = project_to_feasible(h_interp, target_sum)
        start_points.append(h_interp)
    
    # Binary vector (count ones)
    h_binary = np.zeros(n_points)
    ones_count = int(target_sum)
    h_binary[:ones_count] = 1.0
    h_binary = project_to_feasible(h_binary, target_sum)
    start_points.append(h_binary.copy())
    
    # Random uniform
    np.random.seed(seed * 1000 + n_points)
    h_uniform = np.random.rand(n_points)
    h_uniform = project_to_feasible(h_uniform, target_sum)
    start_points.append(h_uniform.copy())
    
    # Tweak from previous
    try:
        # Pretend we had previous solution from a smaller n
        h_prev = np.array(initial_h_values)
        if len(h_prev) > n_points:
            h_tweak = h_prev[:n_points] 
        elif len(h_prev) < n_points:
            h_tweak = np.interp(np.linspace(0, 1, n_points), 
                                np.linspace(0, 1, len(h_prev)), 
                                h_prev)
        else:
            h_tweak = h_prev
        h_tweak = project_to_feasible(h_tweak, target_sum)
        start_points.append(h_tweak)
    except:
        pass
    
    # Run gradient ascent from each starting point
    best_h = None
    best_value = float('inf')
    
    for i, start_h in enumerate(start_points):
        try:
            h_opt, val = run_gradient_ascent(start_h, beta_schedule, max_inner_iter, n_points)
            if val < best_value:
                best_value = val
                best_h = h_opt
        except Exception as e:
            continue
            
    return best_h, best_value, n_points

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    n_used = n_points_list[0]
    if n_used % 2 == 1:
        n_used = n_used + 1
    if n_used not in n_points_list:
        n_points_list.append(n_used)
    
    # Multi-start over different n
    for n_points in n_points_list:
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        try:
            h_opt, c5_opt, final_n = run_single_n(n_points, seed)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = final_n
                
        except Exception as e:
            # Skip this n if it errors
            continue
    
    # Final refinement with SLSQP
    if best_h is not None and time.time() - start_time < 0.95 * budget_s - 5:
        def obj(h):
            return evaluate_h(h)
        
        try:
            bounds = [(0, 1) for _ in range(len(best_h))]
            c = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
            result = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=c,
                              options={'maxiter': 200, 'ftol': 1e-10})
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_h(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h
        except:
            pass
    
    if best_h is None:
        # Fallback to a standard initial solution
        try:
            from initial_h_values import initial_h_values
            best_h = np.array(initial_h_values)
            best_c5 = evaluate_h(best_h)
            best_n = len(best_h)
        except:
            best_h = np.ones(48) * 0.5
            best_h = project_to_feasible(best_h, 24)
            best_c5 = evaluate_h(best_h)
            best_n = 48
    
    return list(best_h), best_c5, best_n
