def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def solve_sdp(n_points, w):
    """Solve the SDP relaxation to get a fractional solution."""
    # Variables
    h = cp.Variable(n_points)
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.trace(X) == w,  # trace is w
        cp.diag(X) == cp.square(h),  # diag(X) = h^2
        cp.sum(h) == w,
        X >> 0,  # PSD
        h >= 0, h <= 1
    ]
    
    # For each correlation constraint
    for k in range(n_points):
        # Build Toeplitz matrix T_k
        T_k = np.zeros((n_points, n_points))
        for i in range(n_points-k):
            T_k[i, i+k] = 1
            T_k[i+k, i] = 1
        T_k = 0.5 * T_k  # Ensure symmetry
        constraints += [cp.trace(T_k @ X) >= t]
    
    # Objective
    objective = cp.Maximize(t)
    problem = cp.Problem(objective, constraints)

    try:
        problem.solve(solver=cp.SCS, verbose=False)
        if problem.status in ['optimal', 'optimal_inaccurate']:
            h_sdp = h.value
            X_sdp = X.value
            return h_sdp, X_sdp, t.value
        else:
            return None, None, None
    except Exception:
        return None, None, None

def round_sdp(h_sdp, X_sdp, n_points):
    """Randomized rounding of SDP solution."""
    # Get Cholesky factorization
    try:
        R = cholesky(X_sdp, lower=True)
        r = np.random.normal(0, 1, R.shape[0])
        h_round = R.T @ r
    except:
        # Fallback to dominant eigenvector
        try:
            _, v = eigh(X_sdp)
            h_round = np.sqrt(np.max(np.linalg.eigvals(X_sdp))) * v[:, -1]
        except:
            h_round = np.copy(h_sdp)

    # Project onto capped simplex
    def project_simplex(v, w):
        # Water-filling approach for capped simplex
        v = np.copy(v)
        v = np.clip(v, 0, 1)
        total = np.sum(v)
        if abs(total - w) < 1e-10:
            return v
        diff = w - total
        if diff > 0:
            # Scale up
            v += diff / len(v)
            v = np.clip(v, 0, 1)
        else:
            # Scale down
            v *= w / total
            v = np.clip(v, 0, 1)
        return v

    h_projected = project_simplex(h_round, w)
    return h_projected

def fit_fourier(h_vals, M):
    """Fit a truncated Fourier series to h_vals."""
    x = np.arange(len(h_vals))
    A = np.column_stack([
        np.ones_like(x),
        *[np.cos(np.pi * m * x / len(h_vals)) for m in range(1, M+1)],
        *[np.sin(np.pi * m * x / len(h_vals)) for m in range(1, M+1)]
    ])
    # Use least squares to fit
    try:
        coefs = np.linalg.lstsq(A, h_vals, rcond=None)[0]
    except:
        # fallback to standard method
        # Minimize ||Ax - h||^2 subject to 0 <= h_vals[i] <= 1
        # introduce penalty to violate constraints
        # Shifts to fit cosine/sine coefficients
        def f(coefs):
            h_test = A @ coefs
            h_test = np.clip(h_test, 0, 1)
            return np.sum((h_test - h_vals)**2)

        res = minimize(f, np.zeros(A.shape[1]), method='L-BFGSB', bounds=[(0,1)]*A.shape[1])
        coefs = res.x
    
    return coefs

def reconstruct_fourier(coefs, n_points, M):
    """Reconstruct h from Fourier coefficients."""
    x = np.arange(n_points)
    h = coefs[0] * np.ones_like(x)
    for i in range(1, M+1):
        h += coefs[i] * np.cos(np.pi * i * x / n_points)
        h += coefs[i+M] * np.sin(np.pi * i * x / n_points)
    return np.clip(h, 0, 1)

def smooth_optimization(h_initial, n_points):
    """Optimize via smooth surrogate with α continuation."""
    # Linear approximation of the overlap function - using FFT
    def compute_overlap_fft(h_vec):
        h_fft = fft(h_vec)
        one_minus_h_fft = fft(1 - h_vec)
        conv = ifft(h_fft * np.conj(one_minus_h_fft))
        return np.real(conv[:len(h_vec)])

    def compute_overlap_batch(h_vec):
        # For all shifts
        overlaps = compute_overlap_fft(h_vec)
        return max(overlaps)

    # Smooth surrogate
    def surrogate_loss(coefs, alpha):
        h = reconstruct_fourier(coefs, n_points, M=6)
        # Compute mutual overlaps
        if n_points <= 64:
            overlaps = compute_overlap_fft(h)
            total = np.sum(np.exp(alpha * overlaps))
            return np.log(total) / alpha
        else:
            return compute_overlap_batch(h)

    # Gradient of surrogate
    def gradient_surr(coefs, alpha):
        h = reconstruct_fourier(coefs, n_points, M=6)
        overlaps = compute_overlap_fft(h)
        weights = np.exp(alpha * overlaps) / np.sum(np.exp(alpha * overlaps))
        # Approximate gradient of negative log sum exp using weight vector
        grad = np.zeros(len(coefs))
        h_diff = h - (1 - h)
        for i in range(1, 6+1):
            grad[i] = np.sum(weights * h_diff * np.cos(np.pi * i * np.arange(n_points) / n_points))
            grad[i + 6] = np.sum(weights * h_diff * np.sin(np.pi * i * np.arange(n_points) / n_points))
        return grad

    # Optimization
    theta = fit_fourier(h_initial, M=6)
    
    # α continuation
    alpha_vals = [1, 2, 4, 8, 16, 32, 64, 128, 256]
    for i, alpha in enumerate(alpha_vals):
        def surrogate_f(c):
            return surrogate_loss(c, alpha)
        def surrogate_g(c):
            return gradient_surr(c, alpha)
        
        # Track learning
        try:
            result = minimize(surrogate_f, theta, method='L-BFGSB', jac=surrogate_g,
                              bounds=[(0, 1)] * len(theta), options={'gtol': 1e-6})
            if result.success:
                theta = result.x
            if i > 0 and i % 3 == 0:
                h_probe = reconstruct_fourier(theta, n_points, M=6)
                current_c5 = evaluate_h(h_probe)
                if current_c5 < target_bound:
                    return h_probe
        except:
            pass

    h_final = reconstruct_fourier(theta, n_points, M=6)
    return h_final

def refine_precise(h_final, w):

    h = h_final.copy()

        one_minus_h_fft = fft(1 - h)
        conv = ifft(h_fft * np.conj(one_minus_h_fft))
        return np.real(conv[:len(h)])

    def project_simplex_exact(h_vec, w):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec)
        if abs(total - w) < 1e-10:
            return h_vec
        diff = w - total
        h_vec = np.clip(h_vec + diff / len(h_vec), 0, 1)
        return h_vec

    for _ in range(500):
        overlaps = compute_overlaps(h)
        min_idx = np.argmin(overlaps)
        # Take gradient for active min shift (simple)
        grad = h * (1-h)
        # Add small gradient to try to improve
        step_size = 0.005
        h = h - step_size * grad
        h = project_simplex_exact(h, w)
        if evaluate_h(h) < target_bound:
            break
    return h
