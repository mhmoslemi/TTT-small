def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def compute_sdp_solution(h0, n_points):
    """Solve the SDP relaxation for the optimal h."""
    dx = 2.0 / n_points
    w = n_points / 2.0

    # Create the Toeplitz matrices
    T_matrices = [np.eye(n_points, k=-k) + np.eye(n_points, k=k) for k in range(n_points)]

    # Variables: h and X
    # Fixed dimension
    h = Variable(n_points)
    X = Semidefinite(n_points)
    
    # Constraints on h to be in [0, 1]
    constraints = [
        h >= 0,
        h <= 1,
        sum_entries(h) == w,
        X == h @ h.T
    ]
    
    # Add constraints for the diagonal
    for i in range(n_points):
        constraints.append(X[i, i] == h[i]**2)
    
    # Objective function: maximize t (here we just add it as a placeholder)
    # We'll iterate looking for the min corner
    # Not used directly, just for formulating the handcrafted constraints
    
    results = []
    for i in range(n_points):
        t = trace(T_matrices[i] @ X)
        results.append(t)
    
    # For initial feasibility check
    p = Problem(Minimize(0), constraints)
    
    try:
        # Just solve feasible version first for a good initial 
        p.solve(solver=cp.SCS)  # can change if sufficient memory/COM, e.g. MOSEK
        h_val = h.value
        X_val = X.value
        if h_val is not None:
            return True, h_val, X_val
        else:
            return False, h0, np.eye(n_points)
    except:
        return False, h0, np.eye(n_points)

def random_round_sdp(h_SDP, X_SDP, n_points):
    """Apply random rounding to the SDP solution."""
    try:
        R = sqrtm(X_SDP)
        if np.iscomplexobj(R):
            R = R.real
        g = np.random.randn(R.shape[0])
        h_round = R.T @ g
        h_round = np.clip(h_round, 0, 1)
        
        # Project to the capped simplex
        s = np.sum(h_round)
        if abs(s - n_points/2.0) > 1e-6:
            # Use water-filling algorithm to project
            h_sorted = np.sort(h_round)[::-1]
            cumsum = np.cumsum(h_sorted)
            t = np.argmax(h_sorted - (n_points/2.0 - cumsum) / np.arange(1, len(h_sorted) + 1))
            threshold = (n_points/2.0 - cumsum[t]) / (len(h_sorted) - t)
            h_round = np.clip(h_round - threshold, 0, 1)
            # Rescale if necessary
            s_new = np.sum(h_round)
            if abs(s_new) > 1e-6:
                h_round = h_round * (n_points/2.0) / s_new
                
        return h_round
    except:
        # Fallback to deterministic rounding
        try:
            eigenvals, eigenvecs = np.linalg.eigh(X_SDP)
            idx = np.argmax(eigenvals)
            v = eigenvecs[:, idx]
            h_det = np.sqrt(eigenvals[idx]) * v
            h_det = np.clip(h_det, 0, 1)
            s = np.sum(h_det)
            if abs(s - n_points/2.0) > 1e-6:
                h_det = h_det * (n_points/2.0) / s
            return h_det
        except:
            return np.ones(n_points) * 0.5

def fourier_fit(h_vec, n_modes=6):
    """Fit a truncated Fourier series."""
    n = len(h_vec)
    x = np.arange(n)
    A = np.column_stack([np.ones(n), np.cos(np.pi * np.arange(1, n_modes + 1)[:, None] * x[None, :] / n),
                         np.sin(np.pi * np.arange(1, n_modes + 1)[:, None] * x[None, :] / n)])
    
    # Regularization for numerical stability
    A += 1e-12 * np.random.randn(*A.shape)
    
    # Least squares avoiding boundaries
    def bound_penalty(x):
        return np.sum((np.clip(x, 0, 1) - x)**2)
    
    # Use geometric LH search for θ
    bounds = [(0, 1)] * (2 * n_modes + 1)  # a0, a1, b1, ..., an, bn
    def objective(theta):
        a0 = theta[0]
        a_coeffs = theta[1:n_modes+1]
        b_coeffs = theta[n_modes+1:]
        h_fourier = a0 + np.dot(A[:,1:], np.concatenate([a_coeffs, b_coeffs]))
        # Track residuals and penalty
        h_proj = np.clip(h_fourier, 0, 1)
        penalty = bound_penalty(h_fourier)
        return np.linalg.norm(h_vec - h_proj)**2 + penalty
    
    # Nelder-Mead algo (direct optimization on θ not h_vec directly)
    try:
        init_guess = np.ones(2 * n_modes + 1)
        result = minimize(objective, init_guess, method='Nelder-Mead', options={'disp': False})
        if result.success:
            return result.x
    except:
        pass
    return np.ones(2 * n_modes + 1)

def reconstruct_h_from_fourier(theta, n_points, n_modes=6):
    """Reconstruct h from Fourier coefficients theta."""
    x = np.arange(n_points)
    h = theta[0] + np.sum(
        theta[1:n_modes+1] * np.cos(np.pi * np.arange(1, n_modes + 1)[:, None] * x[None, :] / n_points) 
        + theta[n_modes+1:] * np.sin(np.pi * np.arange(1, n_modes + 1)[:, None] * x[None, :] / n_points),
        axis=0)
    return np.clip(h, 0, 1)

def refine_with_continuation_gradient(h_initial, n_points):
    """Refine h using a continuation gradient on soft-max overlap."""
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        # The slowest part — FFT-based correlation
        v1 = np.pad(h, (0, n_points), 'constant')
        v2 = np.pad(1 - h, (0, n_points), 'constant')
        corr = fftconvolve(v1, v2, 'valid')[:n_points]
        return corr
    
    def surrogate_c5(θ, alpha=1.0):
        h = reconstruct_h_from_fourier(θ, n_points)
        overlaps = compute_overlap(h)
        return -dx * np.log(np.sum(np.exp(alpha * overlaps))) / alpha

    # Gradient ascent with continuation from small α to large
    init_theta = fourier_fit(h_initial)
    alpha_list = [1, 2, 5, 10, 25, 50, 100, 200]
    
    h_current = reconstruct_h_from_fourier(init_theta, n_points)
    theta_current = init_theta.copy()
    
    for alpha in alpha_list:
        # Subproblem with α
        def surrogate_func(θ):
            h = reconstruct_h_from_fourier(θ, n_points)
            return -dx * np.log(np.sum(np.exp(alpha * compute_overlap(h)))) / alpha
            
        def surrogate_grad(θ):
            h = reconstruct_h_from_fourier(θ, n_points)
            overlaps = compute_overlap(h)
            weights = np.exp(alpha * overlaps) / np.sum(np.exp(alpha * overlaps))
            # Numerical gradients of overlap terms w.r.t theta
            # axis-dependent gradient in freq domain is simpler to compute via adjoints
            # For now, use finite differencing for this simplified version
            eps = 1e-5
            grad = np.zeros_like(θ)
            for i in range(len(θ)):
                theta_h = theta.copy()
                theta_h[i] += eps
                h_plus = reconstruct_h_from_fourier(theta_h, n_points)
                theta_h[i] -= 2 * eps
                h_minus = reconstruct_h_from_fourier(theta_h, n_points)
                grad[i] = (surrogate_func(theta_h + 2 * eps * np.eye(len(θ))[i]) - 
                           surrogate_func(theta_h)) / (2 * eps)
            return grad
            
        try:
            # Instead of full gradient, update using projected gradient step
            h_project = lambda x: np.clip(x, 0, 1)
            def project_sum(x, target_sum):
                current_sum = np.sum(x)
                if abs(current_sum) > 1e-12:
                    x = x * (target_sum / current_sum)
                return x
            # Simple steepest descent on theta
            for step in range(30):
                h_new = reconstruct_h_from_fourier(theta_current, n_points)
                overlaps = compute_overlap(h_new)
                weights = np.exp(alpha * overlaps)
                weights = weights / np.sum(weights)
                # Simple heuristic
                eta = 1e-3 * (1.0 + alpha / 1000.0)  # adaptive step size
                theta_current -= eta * surrogate_grad(theta_current)
                # Ensure it projects onto feasible set after each step
                theta_current = np.clip(theta_current, -10, 10)
                # Post-step project back to valid range
                h_tmp = reconstruct_h_from_fourier(theta_current, n_points)
                h_tmp = h_project(h_tmp)
                h_tmp = project_sum(h_tmp, n_points / 2.0)
                h_current = h_tmp
            
            # Check current best after refinement
            updated_h = reconstruct_h_from_fourier(theta_current, n_points)
            if evaluate_h(updated_h) < evaluate_h(h_current):
                h_current = updated_h
                theta_current = theta_current.copy()
        except:
            pass
            
        # Step size control / early convergence detection
        # Early stopping if weights concentrate too much upon high α
        overlaps = compute_overlap(h_current)
        weights = np.exp(alpha * overlaps)
        weights = weights / np.sum(weights)
        if np.max(weights) > 0.99:
            break
    
    return h_current
