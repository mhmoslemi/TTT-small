```python
import numpy as np
from cvxpy import *
import time
from scipy.fftpack import fft, ifft
from scipy.optimize import minimize
import math

def evaluate_h(h):
    """FFT-based evaluation of C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Pad to length 2*n for correlation
    h_padded = np.pad(h, (0, n), 'constant')
    g_padded = np.pad(1 - h, (0, n), 'constant')
    H = fft(h_padded)
    G = fft(g_padded)
    corr = ifft(H * np.conj(G)).real
    max_val = np.max(corr[:2*n-1]) * dx
    return max_val
    
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 48
    dx = 2.0 / n
    M = n // 2  # target mass = n/2
    
    # Step 1: Try to solve exact binary MILP with cvxpy + CBC
    T_target = math.ceil(0.3807 * n / 2)
    h_bin = Variable(n, boolean=True)
    p = {}
    T = Variable(integer=True)
    constraints = [sum(h_bin) == M]
    for l in range(1, n):
        p[l] = Variable(n - l, boolean=True)
        for i in range(n - l):
            p[l][i] <= h_bin[i]
            p[l][i] <= h_bin[i + l]
            p[l][i] >= h_bin[i] + h_bin[i + l] - 1
            p[l][i] >= 0
        # account for this shift's contribution to max
        constraints.append(sum(p[l]) + T >= M)
    # objective
    objective = Minimize(T)
    if time.time() - start_time > budget_s * 0.3:
        # skip expensive MILP if not enough time
        h_opt = np.array(initial_h_values)
        h_opt = h_opt[:n]
        h_opt = h_opt / np.sum(h_opt) * M
    else:
        prob = Problem(objective, constraints)
        prob.solve(solver=CPLEX, verbose=False, maxSeconds=budget_s * 0.3)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_opt = np.array(h_bin.value).astype(float)
        else:
            # fallback to the initial solution
            h_opt = np.array(initial_h_values)
            h_opt = h_opt[:n]
            h_opt = h_opt / np.sum(h_opt) * M
    
    c5_bound = evaluate_h(h_opt)
    
    # Step 2: Binary Local Search (annealing)
    if time.time() - start_time > budget_s * 0.65:
        # skip local refinement
        h_bin_best = h_opt.copy()
        c5_best = c5_bound
    else:
        # Convert continuous h_opt to binary
        h_bin_current = (h_opt > 0.5).astype(float)
        # ensure sum is exactly M
        current_sum = int(np.sum(h_bin_current))
        if current_sum != M:
            # adjust to make sum equal M
            # find index of 1's and 0's to swap
            ones = np.where(h_bin_current == 1)[0]
            zeros = np.where(h_bin_current == 0)[0]
            if current_sum > M:
                # need to change some 1s to 0s
                idxs_to_flip = ones[:current_sum - M]
                for i in idxs_to_flip:
                    h_bin_current[i] = 0
            elif current_sum < M:
                # need to change some 0s to 1s
                idxs_to_flip = zeros[:M - current_sum]
                for i in idxs_to_flip:
                    h_bin_current[i] = 1
        c5_current = evaluate_h(h_bin_current)
        h_bin_best = h_bin_current.copy()
        c5_best = c5_current
        # Simulated Annealing
        t0 = time.time()
        tau = 1.0
        t_max = 20000  # Roughly 20000 iterations to approximate a full run
        for iteration in range(t_max):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Find indices of 1s and 0s
            ones = np.where(h_bin_current == 1)[0]
            zeros = np.where(h_bin_current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            # swap h[i] and h[j]
            h_new = h_bin_current.copy()
            h_new[i] = 0
            h_new[j] = 1
            c5_new = evaluate_h(h_new)
            delta = c5_new - c5_current
            if delta <= 0 or np.random.rand() < np.exp(-delta / tau):
                h_bin_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_bin_best = h_new.copy()
                    c5_best = c5_current
            # cooling
            tau *= 0.999
        h_opt = h_bin_best.copy()
        c5_bound = c5_best
        
    # Step 3: Fractional refinement using smooth log-sum-exp descent
    if time.time() - start_time > budget_s * 0.95:
        # skip fractional refinement
        pass
    else:
        # smooth descent with log-sum-exp
        h_current = h_bin_best.copy()
        beta_vals = [1, 2, 5, 10, 20]
        for beta in beta_vals:
            if time.time() - start_time > budget_s * 0.95:
                break
            # 100 steps per beta
            for step in range(100):
                if time.time() - start_time > budget_s * 0.95:
                    break
                # compute all L_ell and their gradients
                n_corr = 2 * n - 1
                L_ell = np.zeros(n_corr)
                dL_ell = np.zeros_like(h_current)
                # Since we already have correlation structure
                # use corr - L_ell is sum over i of h[i]*(1-h[i+ell])
                # which is sum over i of h[i] - h[i]*h[i+ell]
                # thus we can use convolution to get h*[1-h]
                h_pad = np.concatenate([h_current, np.zeros(n)])
                g_pad = np.concatenate([1 - h_current, np.zeros(n)])
                H = fft(h_pad)
                G = fft(g_pad)
                corr = ifft(H * np.conj(G)).real
                # extract the first n_corr values
                corr = corr[:2*n-1]
                # corr[i] is sum over j of h[j]*h[j+i]
                L_ell = corr * dx # linearized shift integral
                # compute gradient: sum of dL_ell[i]*w[i] for i
                # where w[i] = exp(beta*L_ell[i]) / sum_k exp(beta*L_ell[k])
                # derivate of temp = theta*exp (beta*L_ell) w.r.t. h[i] is 
                # derivative of (i-L_ell) wrt h[i] i.e. 1-h[i+ell] - h[i-ell]
                dL_ell = np.zeros_like(h_current)
                for ell in range(1, n):
                    if ell < n:
                        dL_ell += (1 - h_current[ell:] - h_current[n-ell:]) * dx * (ell < n)
                # compute full gradient now
                weights = np.exp(beta * L_ell)
                # normalize
                weights /= np.sum(weights)
                # the final gradient is sum of weighted gradients
                # dL_ell is per point, and this is as if we have a way to
                # compute cross dependencies.
                # let's do more careful shift term-by-term
                grad = np.zeros_like(h_current)
                for ell in range(1, n):
                    if ell < n:
                        # dh[i]/dx = 1 - h[i+ell] - h[i-ell]
                        grad_part = np.zeros_like(h_current)
                        # handle i = ell ... n: 1 - h[i] - h[i-ell]
                        if ell > 0:
                            grad_part[ell:] -= h_current[:-ell]   # -h[i-ell]
                        # handle i = 0 to n - ell: 1 - h[i] - h[i+ell]
                        grad_part[:-ell] -= h_current[ell:]   # -h[i+ell] 
                        grad_part += 1
                        grad += weights[ell] * grad_part * dx
                # step
                eta = 0.01
                h_new = h_current - eta * grad
                # project to box [0, 1]
                h_new = np.clip(h_new, 0, 1)
                # project to sum = M
                diff = np.sum(h_new) - M
                alpha = diff / n   # scale adjust
                h_new = h_new - alpha  # constant shift that changes average
                h_new = np.clip(h_new, 0, 1)
                h_current = h_new
        # final update of h_opt
        h_opt = h_current
        
    # Final verification step: clip and renormalize
    h_opt = np.clip(h_opt, 0, 1)
    mass = np.sum(h_opt) * dx
    if abs(mass - 1) > 1e-10:
        h_opt = h_opt / mass * 1
    final_c5 = evaluate_h(h_opt)
    return (h_opt.tolist(), final_c5, n)
```
