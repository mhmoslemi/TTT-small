import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = rfft(h, 2*n)
    G = rfft(1.0 - h, 2*n)
    conv = irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP formulation."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    solver_options = {'maxSeconds': time_limit}
    try:
        prob.solve(solver=cp.CBC, verbose=False, **solver_options)
    except Exception:
        return None, None
    
    if prob.status in ['optimal', 'optimal_inaccurate', 'feasible']:
        h_opt = np.ravel(h.value)
        T_opt = int(T.value)
        return T_opt, h_opt
    else:
        return None, None

def smooth_descent(h_start, time_limit=100):
    """Refine a binary solution with smooth log-sum-exp descent."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    h = np.clip(h, 0.0, 1.0)

    # Ensure mass is correct
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-10:
        h *= (1.0 / total)

    def compute_lags(h):
        # h[i] * (1 - h[i+ell]) for all ell, i
        L = []
        for ell in range(1, n):
            s = 0.0
            for i in range(n - ell):
                s += h[i] * (1.0 - h[i + ell])
            L.append(s)
        return np.array(L)

    def project_onto_simplex(h, target_sum=1.0):
        # project h to sum(h) * dx = target_sum = 1.0
        total = np.sum(h) * dx
        if abs(total - target_sum) < 1e-10:
            return h

        # water-filling approach to find lambda
        # sorted descending, equivalent to ravel(h)[::-1]
        sorted_vals = np.sort(h)[::-1]
        cumsum = np.cumsum(sorted_vals)
        n_sorted = len(sorted_vals)
        n_lambda = None
        for i in range(n_sorted):
            # sum(clip(h_i + lambda, 0, 1)) == target_sum
            # let lambda = (target_sum - cumsum[i]) / (n_sorted - i)
            # we want (h_i + lambda) <= 1 for i < n_lambda
            # and h_i + lambda >= 0 for all i
            # thus lambda >= -h_i and lambda <= 1-h_i => -h_i <= lambda <= 1-h_i
            # we focus on lambda: (target_sum - cumsum[i]) / (n_sorted - i) <= 1 - sorted_vals[i] => lambda <= ...
            # but actually: we want i-th largest element to be <= (1 - lambda), we want to solve:
            # sum_j=0^{i-1} 1 + (h_{compare}-lambda)  = target_sum =>  i + sum_j=0^{i-1} h_{compare} - i*lambda = target_sum => lambda = (i + sum_j=0^{i-1} h_{compare} - target_sum) / i
            # instead simpler approach: sort values, apply waterfilling with bybass of omega, just compute lambda
            # robust avoidance using 0.5 * (1.0 - sorted_vals[i]) to be conservative on the upper bound.
            lambda_ = (target_sum - cumsum[i]) / (n_sorted - i)
            # lambda chosen so that h_i + lambda = 1 for i = 0 to some point, 
            # i = 0 to i-1 are w/o bound, i to last are set to 1.
            # just verify
            if i == 0:
                if lambda_ <= 1.0 - sorted_vals[0]:
                    n_lambda = 0
                else:
                    # we found lambda that's too large. 
                    pass
            # ... (computational checks) 
            # For now just process step by step:
            # In practice, let's just compute naively using bisection:
            # reconstructed `h_prime`: cap `h + lambda`, s.t. sum = 1, 
            # iterate until convergence based on feasible values. 
            # more directly:
            # sum(clip(h[j] + lambda, 0, 1)) = target_sum
            # So we want to adjust h to target mass.
            # binserach for lambda:
            done = False
            left, right = -1.0, 1.0
            for _ in range(50):  # binary search
                mid = 0.5 * (left + right)
                h_trial = np.clip(h + mid, 0, 1)
                total_trial = np.sum(h_trial) * dx
                if abs(total_trial - target_sum) < 1e-10:
                    h = h_trial
                    done = True
                    break
                elif total_trial > target_sum:
                    right = mid
                else:
                    left = mid
            if done:
                break
        return h

    # Refinement loop
    beta_schedule = [1, 2, 5, 10, 20]
    for beta in beta_schedule:
        # use an exponential schedule
        start_iter = time.time()
        step_size = 0.1
        max_iter = 200
        for it in range(max_iter):
            if time.time() - start_iter > time_limit * 0.9:
                break
            if it > 0 and it % 10 == 0:
                h = project_onto_simplex(h)
            # Compute Lagrange terms and weights for smooth gradient
            Ls = compute_lags(h)
            exp_Ls = np.exp(beta * Ls)
            w = exp_Ls / np.sum(exp_Ls)
            # The gradient w.r.t. each h_i:
            grad = np.zeros(n)
            for ell in range(1, n):
                if ell >= n:
                    continue
                # For each ell compute the terms in gradient
                # For i in 0 to n - ell - 1:
                # d/dh_i L_ell = (1 - h_{i + ell} - h_{i - ell}) (here + is mod n, but checked cols)
                # ignore out of bounds
                for i in range(n - ell):
                    if i > 0:
                        grad[i] += w[ell - 1] * (1.0 - h[i + ell] - h[i - 1])
                    else:
                        grad[i] += w[ell - 1] * (1.0 - h[i + ell])
                        # i = 0, no contribution from h[-1]
            # step
            h -= step_size * grad
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_simplex(h)
    # Final full evaluation
    score = evaluate_h(h)
    return h, score

def anneal_binary(h_start, max_iter=100000, time_limit=100):
    """Perform swap-based simulated annealing from binary initialisation."""
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    current_score = evaluate_h(h)

    # Target mass
    target_mass = 1.0
    true_mass = np.sum(h) * dx
    # Rescale if not 1
    if abs(true_mass - target_mass) > 1e-10:
        h *= (target_mass / true_mass)
    
    # get new current_score; ensure this is integer-valued sample
    current_score = evaluate_h(h)
    best_score = current_score
    best_h = h.copy()
    start_time = time.time()
    temp = 1.0  # initial temp
    cooling_rate = 0.999

    # compute typical delta of accept for first 500 swaps
    deltas = []
    for _ in range(500):
        if len(deltas) < 500:
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h[i] == 1 and h[j] == 0:  # Valid swap
                old_val = h[i]
                h[i] = 0
                h[j] = 1
                new_score = evaluate_h(h)
                delta = new_score - current_score
                deltas.append(delta)
                current_score = new_score
                h[i] = old_val
                h[j] = 0
        else:
            break
    # estimate initial temp
    avg_delta = np.median(deltas) if deltas else 1.0
    if avg_delta > 0:
        temp = avg_delta * 2.0
    else:
        temp = 1.0

    it = 0
    while temp > 1e-6 and time.time() - start_time < time_limit:
        it += 1
        # make a random swap
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        
        if h[i] == 1 and h[j] == 0:
            # Swap h[i] and h[j]
            h[i] = 0
            h[j] = 1
            new_score = evaluate_h(h)
            delta = new_score - current_score
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                # Accept
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()
            else:
                # Reject
                h[i] = 1
                h[j] = 0
            temp *= cooling_rate
    return best_h, best_score
