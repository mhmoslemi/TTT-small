import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(v, S):
    """
    Project vector v onto the set {x: sum(x)=S, 0 ≤ x ≤ 1}
    Using bisection on lambda
    """
    # Ensure v is numpy array
    v = np.asarray(v)
    # Copy to avoid modifying input
    v = v.copy()
    
    # Bisection on lambda to find right projection
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        sum_x = x.sum()
        if sum_x > S:
            lo = lam
        else:
            hi = lam
    
    # Final projection
    lam = (lo + hi) / 2.0
    x = np.clip(v - lam, 0, 1)
    
    # Ensure sum is close to S by adjusting values that are near bounds
    residual = S - x.sum()
    if abs(residual) > 1e-10:
        # Find free indices (not at bounds)
        free = (x > 1e-8) & (x < 1 - 1e-8)
        if free.any():
            # Add residual evenly to free positions
            x[free] += residual / free.sum()
            # Clamp to [0,1]
            x = np.clip(x, 0, 1)
            
    return x

def gradient_descent(h0, S, budget_s, start_time, max_iter=5000):

    n = len(h)
    dx = 2.0 / n
    
    # alpha controls smoothing parameter, increasing over time
    alpha = 0.5
    # learning rate
    lr = 0.5
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # compute full correlation
        conv = correlate(h, 1 - h, mode="full")
        
        # Compute weight distribution using softmax of conv with smoothing
        # Use stable computation
        max_conv = np.max(conv)
        shift = conv - max_conv
        exp_vals = np.exp(alpha * shift)
        p = exp_vals / exp_vals.sum()
        
        # compute gradient via convolution
        # p_rev and p need to be used effectively in the convolution
        # compute conv_plus and conv_minus
        p_rev = p[::-1]
        conv_plus = np.convolve(p_rev, h, mode='full')[n-1:n-1+n]  # indices n-1 to 2n-2
        conv_minus = np.convolve(p, h, mode='full')[n-1:n-1+n]
        
        # gradient
        g = 1.0 - (conv_plus + conv_minus)
        
        # Update
        h_new = h - lr * g
        
        # Project
        h = project_to_capped_simplex(h_new, S)
        
        # Evaluate C5
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        iteration += 1
        
        # Adaptive tuning of parameters
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 20)
            lr *= 0.95
        
        # Check elapsed time to prevent timeout
        if time.time() - start_time > budget_s - 5:
            break
            
    return best_h, best_c5
