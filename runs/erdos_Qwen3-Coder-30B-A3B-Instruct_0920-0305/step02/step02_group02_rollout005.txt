import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable log-sum-exp"""
    xmax = np.max(x, axis=axis, keepdims=True)
    return xmax + np.log(np.sum(np.exp(x - xmax), axis=axis, keepdims=keepdims))

def project_to_capped_simplex(v, w):
    """Project vector v onto the capped simplex {x: sum(x)=w, 0≤x≤1}"""
    n = len(v)
    if np.sum(v) == w:
        return np.clip(v, 0, 1)
    # Sort descending
    sorted_v = np.sort(v)[::-1]
    # Find threshold tau
    cumsum = np.cumsum(sorted_v)
    idx = np.arange(1, n+1)
    threshold = (cumsum - w) / idx
    # Find the largest index where the constraint holds
    mask = sorted_v > threshold
    if not np.any(mask):
        tau = sorted_v[0]  # All entries at upper bound
    else:
        k = np.argmax(mask)
        tau = threshold[k-1] if k > 0 else threshold[-1]
    # Compute projection
    x = np.clip(v - tau, 0, 1)
    # Adjust for sum constraint
    sum_diff = w - np.sum(x)
    x[0] += sum_diff
    x = np.clip(x, 0, 1)
    return x

def compute_rk(h, w, n):
    """Compute r_k = w - S_k where S_k = sum h_i * h_{i+k}"""
    # FFT-based auto-correlation
    h_pad = np.pad(h, (0, n), mode='constant')
    H = fft(h_pad)
    S_full = ifft(H * np.conj(H)).real
    S_k = S_full[n-1 : 2*n-1]
    r_k = w - S_k
    return r_k

def compute_gradient_rk(h, w, n, k):
    """Gradient of r_k = w - S_k"""
    grad = np.zeros(n)
    if k == 0:
        # r_0 = w - sum h_i^2
        grad = np.ones(n) - 2 * h
    else:
        # r_k = w - sum h_i h_{i+k}
        grad[:-k] -= h[k:]  # d/dh_i (h_{i+k})
        grad[k:] -= h[:-k]  # d/dh_{i+k} h_i
        grad += 1  # constant term
    return grad

def compute_F_alpha(h, w, n, alpha):
    """Smooth surrogate F_alpha(h) = (dx/alpha) * log(sum exp(alpha * r_k))"""
    r_k = compute_rk(h, w, n)
    return (w - logsumexp(alpha * r_k)) / alpha

def compute_gradient_F_alpha(h, w, n, alpha):
    """Gradient of F_alpha as weighted sum of gradients of r_k"""
    r_k = compute_rk(h, w, n)
    # Softmax weights
    r_max = np.max(r_k)
    exp_r = np.exp(alpha * (r_k - r_max))
    weights = exp_r / np.sum(exp_r)
    # Sum gradients of r_k with weights
    grad = np.zeros(n)
    for i in range(n):
        g = compute_gradient_rk(h, w, n, i)
        grad += weights[i] * g
    return grad

def accelerated_projected_gradient(h0, w, n, alpha_max=200, alpha_init=1, tol=1e-6, max_iter=500):
    """FISTA-style accelerated projected gradient for smooth surrogate"""
    dx = 2.0 / n
    h = h0.copy()
    y = h.copy()
    t_prev = 1
    alpha = alpha_init
    f_prev = compute_F_alpha(h, w, n, alpha)
    
    for iteration in range(max_iter):
        if alpha > alpha_max:
            break
        
        # Compute gradient
        g = compute_gradient_F_alpha(h, w, n, alpha)
        eta = 0.5  # Step size (can be adapted later)
        
        # Line search - Armijo condition (simplified)
        while True:
            h_try = project_to_capped_simplex(y - eta * g, w)
            f_try = compute_F_alpha(h_try, w, n, alpha)
            if f_try <= f_prev + 0.5 * eta * np.dot(g, h_try - h) or eta < 1e-10:
                break
            eta *= 0.5
        h = h_try
        
        # Acceleration step
        t = (1 + np.sqrt(1 + 4 * t_prev ** 2)) / 2
        y = h + ((t_prev - 1) / t) * (h - h_prev)
        h_prev = h.copy()
        t_prev = t
        
        # Check convergence
        if abs(f_try - f_prev) / abs(f_prev) < tol:
            break
        f_prev = f_try

    return h

def get_initialization(n, alpha):
    """Select and return initial guess"""
    if alpha < 10:
        # Uniform random with projection
        h = np.random.uniform(0, 1, n)
        h = project_to_capped_simplex(h, n / 2)
    else:
        # Binary-like initialization for higher alpha
        h = np.zeros(n)
        h[:n//2] = 1
        h = project_to_capped_simplex(h, n / 2)
    return h

def evaluate_c5_from_h(h):
    """Evaluate C5 using numpy correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def run(seed=42, budget_s=600):
    """Main optimization loop"""
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Try different resolutions
    candidates_n = [48, 56, 64, 80]
    total_time_per_n = budget_s * 0.2
    
    for n in candidates_n:
        if time.time() - start_time > budget_s * 0.9:
            break
        dx = 2.0 / n
        w = n / 2  # integral constraint
        
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 10:
            break

        max_evals_per_restart = 5000
        restarts = 0
        best_for_n = float('inf')
        best_h_for_n = None

        while True:
            if time.time() - start_time > budget_s * 0.95:
                break
            restarts += 1

            # Select initialization
            alpha = 1
            h = get_initialization(n, alpha)

            h_opt = accelerated_projected_gradient(h, w, n, alpha_max=200, alpha_init=1, tol=1e-6)
            c5 = evaluate_c5_from_h(h_opt)

            if c5 < best_for_n:
                best_for_n = c5
                best_h_for_n = h_opt.copy()

        if best_for_n < best_c5:
            best_c5 = best_for_n
            best_h = best_h_for_n.copy()
            best_n = n

    if best_h is not None:
        final_c5 = evaluate_c5_from_h(best_h)
        best_h_proper = project_to_capped_simplex(best_h, best_n / 2)
        final_c5 = evaluate_c5_from_h(best_h_proper)
        return list(best_h_proper), final_c5, best_n

    return list(np.ones(48) * 0.5), 0.381, 48
