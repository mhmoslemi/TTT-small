def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / n_points
    target_sum = 1.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Water-filling approach
    if diff > 0:  # Need to increase values
        # Find indices where h < 1
        indices = np.where(h < 1)[0]
        while diff > 1e-10 and len(indices) > 0:
            # Distribute diff equally among remaining indices
            distribute_per = diff / len(indices)
            for i in indices:
                if h[i] + distribute_per <= 1:
                    h[i] += distribute_per
                    diff -= distribute_per
                else:
                    diff -= (1 - h[i])
                    h[i] = 1.0
            indices = np.where(h < 1)[0]
    else:  # Need to decrease values
        # Find indices where h > 0
        indices = np.where(h > 0)[0]
        while diff < -1e-10 and len(indices) > 0:
            # Distribute diff equally among remaining indices
            distribute_per = -diff / len(indices)
            for i in indices:
                if h[i] - distribute_per >= 0:
                    h[i] -= distribute_per
                    diff += distribute_per
                else:
                    diff += h[i]
                    h[i] = 0.0
            indices = np.where(h > 0)[0]
    
    return h

            j = i - m
            g[i] += (1 - h[j])
            g[j] -= h[i]
    
    g *= dx
