import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto the capped simplex with target_sum using bisection."""
    # Ensure x is within [0, 1]
    x = np.clip(x, 0, 1)
    
    def F(lam):
        # Compute projection with Lagrange multiplier lambda
        proj = np.clip(x - lam, 0, 1)
        return np.sum(proj) - target_sum
    
    # Bisection search for lambda such that F(lambda) = 0
    low, high = -10, 10
    for _ in range(40):
        mid = (low + high) / 2
        val = F(mid)
        if val > 0:
            low = mid
        else:
            high = mid
    lambda_star = (low + high) / 2
    
    # Return the projected vector
    return np.clip(x - lambda_star, 0, 1)

def smooth_max(c, alpha):
    """Compute softmax approximation of max with temperature alpha."""
    exp_c = np.exp(alpha * c)
    weights = exp_c / np.sum(exp_c)
    return weights

def compute_gradient(h, w):
    """Compute gradient of smooth objective using convolutions."""
    n = len(h)
    # Weights for convolution (flipped for proper correlation)
    w_reversed = w[::-1]
    
    # Convolve to get the derivatives
    g_plus = np.convolve(w_reversed, h, mode='full')[n-1:n-1+n]
    g_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]
    
    # Gradient = 1 - (g_plus + g_minus)
    grad = 1.0 - (g_plus + g_minus)
    
    return grad

def projected_gradient_descent(initial_h, max_iter=5000, step_size=0.5,
                              alpha_start=0.5, alpha_factor=1.5, alpha_max=20,
                              patience=500):
    """Project gradient descent with smooth objective."""
    # Setup
    h = initial_h.copy()
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    patience_counter = 0
    alpha = alpha_start
    
    for t in range(max_iter):
        # Evaluate the correlation vector for all shifts
        corr = correlate(h, 1 - h, mode="full")
        
        # Compute smooth max weights
        weights = smooth_max(corr, alpha)
        
        # Compute gradient
        grad = compute_gradient(h, weights)
        
        # Update with step size
        h_new = h - step_size * grad
        
        # Project onto capped simplex
        h = project_to_capped_simplex(h_new, target_sum)
        
        # Evaluate current C5
        current_c5 = evaluate_c5(h)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Step size decay
        if t > 0 and t % 200 == 0:
            step_size *= 0.9
        
        # Alpha increase
        if t > 0 and t % 200 == 0:
            alpha = min(alpha * alpha_factor, alpha_max)
            
        # Early stopping
        if patience_counter >= patience:
            break
            
    return best_h, best_c5

def differential_evolution(h0, n_points, max_iter=200):
    """Tiny DE fallback to escape local optima."""
    def evaluate_individual(h):
        # Project onto simplex
        h = project_to_capped_simplex(h, n_points / 2.0)
        return evaluate_c5(h)
    
    # Parameters
    F = 0.6
    CR = 0.9
    popsize = 8
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Initialize population
    population = []
    for _ in range(popsize):
        h = np.random.rand(n_points)
        h = project_to_capped_simplex(h, n_points / 2.0)
        population.append(h)
    
    for iter in range(max_iter):
        new_population = []
        
        for i in range(popsize):
            # Select indices for mutation
            idxs = [idx for idx in range(popsize) if idx != i]
            a, b, c = population[np.random.choice(idxs)], \
                      population[np.random.choice(idxs)], \
                      population[np.random.choice(idxs)]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Project
            trial = project_to_capped_simplex(trial, n_points / 2.0)
            
            # Selection
            trial_c5 = evaluate_individual(trial)
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
                
            new_population.append(trial)
            
        population = new_population
        
        if iter > 0 and iter % 50 == 0:
            # Check for convergence
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Fixed-grid params
    n_points = 200
    dx = 2.0 / n_points
    a_grid = np.linspace(0, 0.5, 51)
    b_grid = np.linspace(0.5, 1, 51) 
    
    # Convergence check
    best_h = None
    best_c5 = float('inf')
    
    # Grid search
    grid_time = 0.3 * budget_s
    grid_start = time.time()
    
    for a in a_grid:
        for b in b_grid:
            if b <= 0.5 or a >= 0.5:
                continue
                
            # Compute u
            u = (2*b - 1) / (2 * (b - a))
            
            # Check if u is valid
            if u < 0 or u > 1:
                continue
            
            # Build step function h
            x = np.arange(n_points) * dx + dx / 2.0
            h = np.zeros_like(x)
            
            # 0 - a - u - 1 - u - a - 0 symmetrical pattern
            mask1 = (x >= a) & (x < b)
            mask2 = (x >= b) & (x < 2 - b)
            mask3 = (x >= 2 - b) & (x < 2 - a)
            
            h[mask1] = u
            h[mask2] = 1.0
            h[mask3] = u
            
            # Ensure the integral constraint holds exactly
            h_sum = np.sum(h) * dx
            h = h * (n_points / 2.0) / h_sum
            h = np.clip(h, 0, 1)
            
            # Compute C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        if time.time() - grid_start > grid_time:
            break
    
    # Local refinement via projected gradient descent
    refine_time = 0.4 * budget_s
    ref_start = time.time()
    
    if time.time() - ref_start < refine_time:
        refined_h, refined_c5 = projected_gradient_descent(best_h, max_iter=5000)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
            
    # DE fallback if time allows
    de_time = budget_s - (time.time() - start_time)
    if de_time > 2:
        de_h, de_c5 = differential_evolution(best_h, n_points, max_iter=100)
        if de_c5 < best_c5:
            best_h = de_h
            best_c5 = de_c5
    
    # Final refinement/conversion
    best_h = project_to_capped_simplex(best_h, n_points / 2.0)
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)
    
    # Adjust final shape for missed integral constraint
    h_sum = np.sum(best_h) * dx
    best_h = best_h * (n_points / 2.0) / h_sum
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
