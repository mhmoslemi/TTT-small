def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def gradient_descent_with_smooth_max(h0, learning_rate=1e-3, max_iter=1000, tol=1e-10):
    """Use gradient descent with smooth max for optimization"""
    h = h0.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    
    # Precompute the indices for correlation
    # We only need to compute the correlation for one side
    # but need to filter out the non-overlapping parts
    corr_indices = np.arange(0, 2*n - 1) - (n - 1)
    
    for iteration in range(max_iter):
        # Compute current value and gradient
        c5_val = evaluate_c5(h)
        
        # Let's compute gradient numerically
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Rescale to satisfy constraint
            scale = 1.0 / (np.sum(h_plus) * dx)
            h_plus = h_plus * scale
            
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - c5_val) / eps
        
        # Update step
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        # Re-scale to satisfy constraint
        scale = 1.0 / (np.sum(h) * dx)
        h = h * scale
        
        if np.linalg.norm(grad) < tol:
            break
    
    return h

        iter_count += 1
        
    return best_h, best_c5
