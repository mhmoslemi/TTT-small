import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to target_sum
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    # Use bisection on lambda to find correct projection
    # F(lambda) = sum of clipped values at lambda
    def F(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # Binary search for lambda such that F(lambda) = target_sum
    l, r = -1, 1
    for _ in range(40):
        mid = (l + r) / 2
        if F(mid) > target_sum:
            l = mid
        else:
            r = mid
    # Final projection
    h = np.clip(h - l, 0, 1)
    # Ensure target sum exactly via scaling if needed
    scale = target_sum / np.sum(h)
    return h * scale

def smooth_max(h, alpha=0.5):
    """Compute smooth approximate of max over all shifts using softmax with temperature alpha"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full") 
    corr *= dx  # Scale by dx for correct units
    
    # Softmax weights
    w = np.exp(alpha * corr)
    w /= np.sum(w)
    return w, corr

def compute_gradient(h, w):
    """Compute gradient of smooth objective"""
    n = len(h)
    # Convolve w with h to get gradients
    # For indicies i, it's the contribution from previous and next indices
    # using the full convolution to get correct indices
    # w_rev = w[::-1] because we want convolution w^T * h (which is right-shifted)
    g_plus = correlate(w[::-1], h, mode="full")[n-1:n-1+n]
    g_minus = correlate(w, h, mode="full")[n-1:n-1+n]
    grad = 1.0 - (g_plus + g_minus)
    return grad

def gradient_step(h, grad, learning_rate):
    """Perform gradient step and return the new vector"""
    h_new = h - learning_rate * grad
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Part 1: Grid Search (use existing initial h as starting point)
    n_points = 200  # Moderate size for fast computation
    dx = 2.0 / n_points
    a_grid = np.linspace(0, 0.5, 51)
    b_grid = np.linspace(0.5, 1, 51)
    best_candidate = None
    best_grid_c5 = float('inf')
    
    max_grid_time = budget_s * 0.30  # 30% time for grid
    
    for a in a_grid:
        if time.time() - start_time > max_grid_time:
            break
        for b in b_grid:
            if time.time() - start_time > max_grid_time:
                break
            # Compute u
            if b <= a:
                continue
            u = (2*b - 1) / (2*(b - a))
            if u < 0 or u > 1:
                continue
            # Build the step function
            h = np.zeros(n_points)
            x = np.linspace(0 + 0.5*dx, 2 - 0.5*dx, n_points)
            # Segments: 0, a, u, 1, u, a, 0
            # Since symmetric, find where each transition happens
            # Segment lengths: first 0, then a, then u, then 1, then u, then a, then 0
            total_length = 2.0
            # See where transition points occur
            a_end = 0 + a
            u_start = a_end
            u_end = u_start + (b - a)
            one_start = u_end
            one_end = one_start + (2 - 2*b)
            u_start2 = one_end
            u_end2 = u_start2 + (b - a)
            a_end2 = u_end2 + a
            
            # Place values in h
            transitions = [
                (0, a_end, 0),
                (a_end, u_end, u),
                (u_end, one_end, 1),
                (one_end, u_end2, u),
                (u_end2, a_end2, 0)
            ]
            
            for i in range(n_points):
                x_i = x[i]
                for (start, end, val) in transitions:
                    if start <= x_i <= end:
                        h[i] = val
                        break
                    
            # Correct mass
            h = h * (0.5 * n_points / np.sum(h))
            h = np.clip(h, 0, 1)
            # Compute actual C5
            c5_candidate = evaluate_c5(h)
            if c5_candidate < best_grid_c5:
                best_grid_c5 = c5_candidate
                best_candidate = np.copy(h)
    
    if best_grid_c5 < best_c5:
        best_c5 = best_grid_c5
        best_h = best_candidate.copy()
    
    # Part 2: Local Gradient Descent on best from grid
    if time.time() - start_time > budget_s * 0.60:
        pass # No time left for refinement
    else:
        # Initialize with best from grid search
        h = best_h.copy()
        n = len(h)
        dx = 2.0 / n
        target_sum = 0.5 * n  # integral should be 1
        h = project_to_capped_simplex(h, target_sum)
        # Scale initial h to target integral
        scale = target_sum / np.sum(h)
        h = h * scale
        
        max_descent_time = budget_s * 0.50  # 50% for this phase
        alpha = 0.5
        eta = 0.5
        patience = 0
        nit = 0
        patience_limit = 500
        iter_limit = 5000
        best_exact_c5 = best_c5
        best_exact_h = h.copy()
        
        while (time.time() - start_time) < max_descent_time and nit < iter_limit:
            eta = max(0.0001, eta * 0.9995)  # decay
            if nit % 200 == 0:
                alpha = min(20.0, alpha * 1.5)  # Warm up the soft-max
            # Smooth approximation
            w, corr = smooth_max(h, alpha)
            # Compute gradient
            grad = compute_gradient(h, w)
            # Take step
            h_new = gradient_step(h, grad, eta)
            # Project back
            h_new = project_to_capped_simplex(h_new, target_sum)
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Compute exact C5
            exact_c5 = evaluate_c5(h_new)
            # Update best if improved
            if exact_c5 <= best_exact_c5:
                best_exact_c5 = exact_c5
                best_exact_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                if patience >= patience_limit:
                    break
            h = h_new
            nit += 1
            
        if best_exact_c5 < best_c5:
            best_c5 = best_exact_c5
            best_h = best_exact_h.copy()
    
    # Part 3: Final FDIF Evolutionary Search (Small DE)
    if time.time() - start_time < budget_s * 0.05:
        # Very small population size + limited generations
        n_pop = 8
        n_gen = 100
        best_feasible = best_h.copy()
        best_feasible_c5 = best_c5
        
        for gen in range(n_gen):
            if (time.time() - start_time) > budget_s - 1:
                break
            population = []
            for _ in range(n_pop):
                # Random population
                h = np.random.rand(n_points)
                h = project_to_capped_simplex(h, target_sum)
                h = np.clip(h, 0, 1)
                population.append(h)
            
            # Evaluate all individuals in population
            fitness = [evaluate_c5(h) for h in population]
            
            # Select best individual
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_feasible_c5:
                best_feasible_c5 = fitness[best_idx]
                best_feasible = population[best_idx].copy()
        
        # Final check
        if best_feasible_c5 < best_c5:
            best_c5 = best_feasible_c5
            best_h = best_feasible.copy()
    
    # Final processing
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    # Ensure the correct sum exactly
    scale = target_sum / np.sum(final_h)
    final_h = final_h * scale
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
