```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_h(h):
    # Use FFT computing all correlations quickly
    n = len(h)
    dx = 2.0 / n
    H = fft(h, 2*n)
    G = fft(1-h, 2*n)
    corr = ifft(H * np.conj(G))[:2*n-1]
    max_val = np.real(corr).max() * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Grid size and target
    n = 48
    dx = 2.0 / n
    M = n // 2
    
    time_left = lambda: budget_s - (time.time() - t0)
    
    h_best = None
    c5_best = float('inf')
    
    # Phase 1: MILP to find a good binary solution
    if time_left() > 30:
        print(f"Starting MILP with n={n}")
        # Variables
        h = Bool(n)
        p = [[Bool(n) for _ in range(n-1)] for _ in range(n-1)]
        T = Int(0, 100)
        
        # Constraints
        constraints = [sum(h) == M]
        
        for ℓ in range(n-1):
            for i in range(n-ℓ):
                # McCormick envelopes for p[ℓ][i] = h[i]*h[i+ℓ]
                constraints.append(p[ℓ][i] <= h[i])
                constraints.append(p[ℓ][i] <= h[i+ℓ])
                constraints.append(p[ℓ][i] >= h[i] + h[i+ℓ] - 1)
                # Linearizing h[i]*h[i+ℓ]
        
        # Worst lag constraints
        for ℓ in range(n-1):
            contrib = sum(p[ℓ][i] for i in range(n-ℓ))
            constraints.append(contrib + T >= M)
        
        # Objective
        prob = minimize(T)
        prob = Problem(prob, constraints)
        
        # Solve
        prob.solve(solver=cbc, maxSeconds=min(100, 0.35 * budget_s))
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            print(f"MILP succeeded, T = {T.value:.2f}")
            h_sol = [round(h[i].value) for i in range(n)]
            c5_sol = evaluate_h(h_sol)
            if c5_sol < c5_best:
                c5_best = c5_sol
                h_best = h_sol.copy()
        else:
            print("MILP did not converge")
        
    # Fallback: random binary
    if h_best is None:
        h_best = np.random.randint(0, 2, n).tolist()
        scale = 1.0 / (sum(h_best) * dx)
        h_best = (np.array(h_best) * scale).tolist()
        c5_best = evaluate_h(h_best)
        print(f"Used random fallback, C5 = {c5_best:.5f}")
    
    # Phase 2: Binary local refinement (simulated annealing on binary)
    if time_left() > 20:
        print("Starting binary local refinement...")
        current_h = h_best.copy()
        current_c5 = c5_best
        temp = 10.0
        cooling_rate = 0.999
        min_temp = 1e-6
        iters = 0
        
        while temp > min_temp and time_left() > 5:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if current_h[i] == current_h[j]: continue
            # Swap
            current_h[i], current_h[j] = current_h[j], current_h[i]
            new_c5 = evaluate_h(current_h)
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5-current_c5)/temp):
                current_c5 = new_c5
                if current_c5 < c5_best:
                    c5_best = current_c5
                    h_best = current_h.copy()
            else:
                # Revert swap
                current_h[i], current_h[j] = current_h[j], current_h[i]
            temp *= cooling_rate
            iters += 1
            if iters % 1000 == 0:
                print(f"Binary SA iter {iters} c5 = {c5_best:.5f}")
        print(f"Binary SA done, final C5 = {c5_best:.5f}")
    
    # Phase 3: Fractional smoothing with projected gradient descent
    if time_left() > 5 and c5_best > 0.3795:
        print("Starting fractional smoothing")
        h = np.array(h_best, dtype=float)
        scale = 1.0 / (np.sum(h) * dx)
        h = h * scale
        
        def smooth_max(h_vec, beta):
            n = len(h_vec)
            dx = 2.0 / n
            H = fft(h_vec, 2*n)
            G = fft(1-h_vec, 2*n)
            corr = ifft(H * np.conj(G))[:2*n-1]
            vals = np.real(corr) * dx
            
            # Softmax weighted average of max
            exp_vals = np.exp(beta * vals)
            weights = exp_vals / np.sum(exp_vals)
            return np.sum(weights * vals)
            
        def grad_smooth_h(h_vec, beta):
            n = len(h_vec)
            dx = 2.0 / n
            dL = np.zeros_like(h_vec)
            H = fft(h_vec, 2*n)
            G = fft(1-h_vec, 2*n)
            
            # Compute corr_L_1(h) = sum_i h[i]* (1 - h[i+1])
            # For each shift compute gradient of L_ℓ
            # We’ll approximate with FFT
            corr = ifft(H * np.conj(G))[:2*n-1]
            vals = np.real(corr) * dx
            
            exp_vals = np.exp(beta * vals)
            weights = exp_vals / np.sum(exp_vals)
            
            # Now compute ∇L_ℓ manually
            
            # Gradient along x direction (each lag contributes a row)
            grad_sum = np.zeros_like(h_vec)
            for ℓ in range(1, n):
                # For shift ℓ we approximate:
                # L_ℓ(h) = sum_i h[i] (1 - h[i+ℓ]) = sum_i h[i] - sum_i h[i] h[i+ℓ]
                # dL_ℓ/dh_k = 1 - h[k+ℓ] - h[k-ℓ] 
                # extend with zero padding
                h_lag = np.roll(h_vec, -ℓ)
                h_lag[-ℓ:] = 0
                h_lag = h_vec - h_lag  # approximation
                
                shifted = np.roll(h_vec, ℓ)  # for -h[k-ℓ]
                shifted[:ℓ] = 0
                h_lag += shifted
                
                grad_sum += weights[ℓ-1] * h_lag
        
            return grad_sum

        beta_vals = [1, 2, 5, 10, 20]
        h_best = h.copy()
        c5_best = evaluate_h(h_best)
        
        for beta in beta_vals:
            print(f"Setting beta={beta}")
            for i in range(100):
                grad = grad_smooth_h(h_best, beta)
                h_new = h_best - 0.01*grad
                
                # Box projection
                h_new = np.clip(h_new, 0, 1)
                
                # Mass projection using waterfilling
                def waterfill_func(lam):
                    new_h = np.clip(h_new + lam, 0, 1)
                    return np.sum(new_h) - M * dx
                
                lam_lower, lam_upper = -10.0, 10.0
                for _ in range(30):
                    lam_mid = (lam_lower + lam_upper) / 2
                    val = waterfill_func(lam_mid)
                    if val > 0:
                        lam_lower = lam_mid
                    else:
                        lam_upper = lam_mid
                lam = (lam_lower + lam_upper) / 2
                h_best = np.clip(h_new + lam, 0, 1)
                
                c5 = evaluate_h(h_best)
                if c5 < c5_best:
                    c5_best = c5
                if i % 100 == 0:
                    print(f"  iter {i}, beta={beta}, c5={c5:.5f}")
                if time_left() < 2:
                    break
            if time_left() < 2:
                break

    # Final projection
    h_best = np.clip(h_best, 0, 1)
    m = np.sum(h_best) * dx
    if abs(m - 1.0) > 1e-9:
        h_best = h_best / m
    final_c5 = evaluate_h(h_best)
    print(f"Final C5 = {final_c5:.5f}")
    
    return (h_best.tolist(), final_c5, n)
```
