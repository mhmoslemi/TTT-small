def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(h, target_sum):
    """Project h onto capped simplex: 0 <= h <= 1, sum(h) = target_sum."""

        if s > target_sum:

        else:

    lam = (low + high) / 2
    return np.clip(h - lam, 0, 1)

def generate_five_step(a, b, n):
    """Generate a symmetric five-step function with parameters a, b."""
    dx = 2.0 / n
    x = np.arange(0.5, n, 1.0) * dx  # sample locations
    # segment boundaries
    a_left = a
    a_right = 2 - a
    b_left = b
    b_right = 2 - b
    # Compute u from integral constraint
    if b <= a: u = 0
    else: u = (2 * b - 1) / (2 * (b - a))
    u = max(0, min(1, u))
    
    h = np.zeros(n)
    for i in range(n):
        pos = x[i]
        if pos < a_left or pos >= a_right:
            h[i] = 0
        elif pos < b_left:
            h[i] = u
        elif pos < b_right:
            h[i] = 1
        else:
            h[i] = u
    # Normalize to integral = 1
    integral = np.sum(h) * dx
    h = h / integral * 1.0   # scale to integral=1
    # Prevent numerical error pushing it out of [0,1]
    h = np.clip(h, 0, 1)
    return h, u
