def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) = target_sum"""
    n = len(h)
    w = np.clip(h, 0, 1)
    current_sum = np.sum(w)
    delta = target_sum - current_sum
    
    # Water-filling algorithm
    while abs(delta) > 1e-10:
        if delta > 0:
            # Distribute delta equally over indices with w_i < 1
            free_indices = np.where(w < 1)[0]
            if len(free_indices) == 0:
                break
            distribute_amount = delta / len(free_indices)
            for i in free_indices:
                w[i] = min(w[i] + distribute_amount, 1.0)
        else:
            # Subtract delta equally over indices with w_i > 0
            free_indices = np.where(w > 0)[0]
            if len(free_indices) == 0:
                break
            subtract_amount = -delta / len(free_indices)
            for i in free_indices:
                w[i] = max(w[i] - subtract_amount, 0.0)
        current_sum = np.sum(w)
        delta = target_sum - current_sum
        
    return w

def compute_subgradient(h, m):
    """Compute the subgradient for a specific lag m"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    
    if m >= 0:
        # i = 0 to n-1-m
        for i in range(n - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        # m < 0, flip the indices
        m = -m
        for i in range(m, n):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    
    return g * dx

def compute_average_subgradient(h, max_lags):
    """Compute average subgradient over all lags attaining the maximum"""
    n = len(h)
    g = np.zeros(n)
    count = 0
    
    for m in max_lags:
        g += compute_subgradient(h, m)
        count += 1
    
    if count > 0:
        g /= count
    
    return g

def symmetry_enforce(h):
    """Enforce symmetry by averaging h_i and h_{n-1-i}"""
    n = len(h)
    h_new = h.copy()
    for i in range(n // 2):
        avg = (h[i] + h[n - 1 - i]) / 2.0
        h_new[i] = h_new[n - 1 - i] = avg
    return h_new
