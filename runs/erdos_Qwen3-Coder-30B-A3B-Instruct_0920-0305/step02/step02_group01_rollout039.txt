import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a medium resolution for efficiency
    n_points = 120
    dx = 2.0 / n_points
    
    # Initial construction

    if len(h0) != n_points:

        h0_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h0)
        h0 = h0_interp

    seeds = []
    # Uniform
    seeds.append(np.ones(n_points) * 0.5)
    # Symmetric five-step shape from literature
    h_symmetric = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0 <= x < 0.4:
            h_symmetric[i] = 1.0
        elif 0.4 <= x < 0.8:
            h_symmetric[i] = 0.5
        elif 0.8 <= x < 1.2:
            h_symmetric[i] = 0.0
        elif 1.2 <= x < 1.6:
            h_symmetric[i] = 0.5
        else:
            h_symmetric[i] = 1.0
    seeds.append(h_symmetric)
    # Random start
    seeds.append(np.random.uniform(0, 1, n_points))
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # DC/CCP algorithm with capped-simplex subproblem
    max_iter = 2000
    patience = 30
    iter_no_improve = 0
    current_h = best_h.copy()
    current_c5 = best_c5
    
    for i in range(max_iter):
        if (time.time() - start_time) > 0.8 * budget_s:
            break
            
        # Compute autocorrelations for all shifts
        auto = correlate(current_h, current_h, mode='full')
        # The full correlation will be auto[-(n-1):] where the last n-1 terms correspond to different shifts
        # The term in \sum_i h_i * h_{i+k} for shift k corresponds to segment k in auto (k=0,...,n-1) and sign depends on mode
        # in fact h_i (1 - h_i+k) = h_i - h_i h_i+k so ...
        # the raw auto is \sum h_i h_{i+k}; needed is \sum h_i - auto = n*dx - auto        
        # but we mentioned that we are concerned with overlap integral = \int h(x)(1-h(x+k)) dx
        # that can be written as an integral over overlapping parts of two functions h(x) and h(x+k)
        # this leads to sum over all i h_i * (1 - h_{i+s}) = sum h_i - sum h_i h_{i+s}
        # after this, it's the max over shifts of this.
        
        # The shift functions contribute directly to integral n*dx - sum_{i} h_i h_{i+s}, which is just `n*dx - auto[s]`.
        # We want to min these.
        # so minimize max_s (n*dx - auto[s])  =  max_s (auto[s] - n*dx) = max_s auto[s] - n*dx
        # So we want to minimize max_s auto[s].
        # thus we want to find s that maximizes auto[s], and the direction to move for its decrease
        # is the negative gradient of that auto(s)
        # The gradient of this sum h_i h_{i+s} with respect to h_i is h_{i+s}, and we collect: gradient = [0, 0,...,0, h_n, h_{n-1},..., h_{s+1}] + [h_s, h_{s-1}, ..., h_1, 0, 0, ..., 0]
        
        # For shift s: gradient_i = h_{i+s} + h_{i-s} (considering proper boundary handling)
        # The auto sum over i is copied into a vector of length 2*n-1 to remove padding issues.
        auto_full = auto
        # Values of shifts 0 to n-1, which corresponds to shifts:
        # shift k in 0..n-1 -> elements (n-1) + k = full_idx = n-1+k
        # Want index of worst shift: shift that minimizes auto -> max(auto)
        worst_shift_idx = np.argmax(auto_full)
        # the actual shift in [-n+1, n-1]
        actual_shift = worst_shift_idx - (n_points - 1)
        
        # compute gradient of auto_s at current_h
        g = np.zeros(n_points)
        # g[i] = h[i+shift] + h[i-shift]
        for i in range(n_points):
            if i + actual_shift < n_points and i + actual_shift >= 0:
                g[i] += current_h[i + actual_shift]
            if i - actual_shift < n_points and i - actual_shift >= 0:
                g[i] += current_h[i - actual_shift]
                
        # now step in direction which maximizes gradient product (minimizes auto_s)
        # We want minimize gradient dot h, subject to sum h = sum(h0)*dx
        # (we are essentially minimizing sum h_i * g_i)
        
        # The LP problem (projection onto capped simplex, optimized via sorting)
        # sort g descending 
        sorted_indices = np.argsort(-g)
        total_mass = sum(current_h) * dx
        new_h = np.zeros_like(current_h)
        remaining_mass = total_mass
        
        for idx in sorted_indices:
            if remaining_mass <= 0:
                break
            util_mass = min(1.0, remaining_mass)
            new_h[idx] = util_mass
            remaining_mass -= util_mass            
        
        # Rescale to match integral constraint exactly: 
        new_h *= total_mass / (np.sum(new_h) * dx) if np.sum(new_h) * dx != 0 else 1
            
        # Test new solution
        candidate_c5 = evaluate_c5(new_h)
        if candidate_c5 < current_c5:
            current_h = new_h
            current_c5 = candidate_c5
            iter_no_improve = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            iter_no_improve += 1
            if iter_no_improve > patience:
                break
    
    # Final polishing with smooth max gradient descent
    if (time.time() - start_time) < 0.9 * budget_s:

            # Smoothly approximate max with smoothmax
            # max over n terms = log sum(exp(log_x)) where log_x is negated.
            # another approach: use Softmax (now on both directions).
            # Let U be a constant to scale very large values
            alpha = 10.0  # should be high enough for smooth approximation
            return -np.log(np.mean(np.exp(-alpha * x)))
            
        # Now optimize smooth version of C5 via projected gradient descent
        num_it = 2000
        # Projected gradient descent on a subset of the shifts:
        for i in range(num_it):
            if (time.time() - start_time) > budget_s:
                break
            # temporary version of e.g., computing gradient
            # For reasons of numeric stability, exact gradient is not provided for generic case.
            # Let's use a more robust and simple projected gradient, starting with a subsample
            # of directions with some code reuse.
            
            # Compute current overlap
            corr = correlate(current_h, 1 - current_h, mode="full")
            # Extract valid shifts (1 component less than full)
            corr_vals = corr[n_points-1:]  
            # To avoid degeneracy due to the range, consider only those with reasonable pattern
            idx_max_corr = np.argmax(corr_vals)
            
            # get the shift s that gives this overlap: 
            # from corr_vals[shift_idx] we use the shift idx from [-n+1, n-1]
            estimated_shift = idx_max_corr - (n_points-1)  
            step_vec = np.zeros(n_points)
            step_vec = np.zeros(n_points)
            for i in range(n_points):
                if i + estimated_shift >= 0 and i + estimated_shift < n_points:
                    step_vec[i] += current_h[i + estimated_shift]
                if i - estimated_shift >= 0 and i - estimated_shift < n_points:
                    step_vec[i] += current_h[i - estimated_shift]
                    
            # Update: negative gradient direction (trying to minimize individual overlap term)
            new_h = current_h - 0.1 * step_vec
            new_h = np.clip(new_h, 0, 1)
            
            # Correct to satisfy integral constraint
            total_c = np.sum(new_h) * dx
            new_h = new_h * (1.0 / total_c) if total_c > 1e-10 else new_h
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
    # Return best solution found:
    return (best_h.tolist(), best_c5, n_points)
