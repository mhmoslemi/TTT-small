import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

# Constants
n_list = [48, 56, 64, 80]
dx_list = [2.0 / n for n in n_list]
w_list = [n / 2 for n in n_list]
alpha_max = 500
alpha_init = 1.0
alpha_factor = 2.0
epsilon = 1e-6
max_iter = 1000
max_inner_iter = 1000

def project_to_capped_simplex(v, w):
    """Project vector v onto the capped simplex: 0 <= v_i <= 1, sum(v) = w."""
    v = np.array(v)
    n = len(v)
    if n == 0:
        return v
    
    # Get indices that sort v in descending order
    idx = np.argsort(-v)
    sorted_v = v[idx]
    
    # Initial estimate of threshold τ
    tau = 0.0
    cum_sum = 0.0
    
    # Find the correct threshold
    for i in range(n):
        cum_sum += sorted_v[i]
        if (i + 1) * sorted_v[i] - cum_sum > w:
            break
        tau = (cum_sum - w) / (i + 1)
    
    # Construct the projection
    projected = np.maximum(np.minimum(sorted_v - tau, 1.0), 0.0)
    # Reorder back to original positions
    result = np.empty_like(projected)
    result[idx] = projected
    
    # Ensure sum is exactly w
    diff = w - np.sum(result)
    if abs(diff) > 1e-12:
        # Add the difference to the first non-saturated element
        for i in range(n):
            if result[i] < 1.0:
                result[i] += diff
                break
        if abs(np.sum(result) - w) > 1e-12:
            # fallback to uniform
            result = np.full(n, w / n)
    
    return result

def compute_auto_correlation(h):
    """Use FFT to compute auto-correlation efficiently."""
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n)])
    h_fft = fft(h_padded)
    autocorr_fft = h_fft * np.conj(h_fft)
    autocorr_full = ifft(autocorr_fft).real
    return autocorr_full[n-1:2*n-1]

def log_sum_exp(v, alpha, max_v=None):
    """Compute log-sum-exp with numerical stability."""
    if max_v is None:
        max_v = np.max(v)
    shifted = v - max_v
    return max_v + np.log(np.sum(np.exp(alpha * shifted)))

def compute_r(h, w, k):
    """Compute r_k for a given shift."""
    n = len(h)
    if k == 0:
        return w - np.dot(h, h)
    else:
        # Convolution via slicing
        pad = np.zeros_like(h)
        pad[k:] = h[:-k] if k < n else np.zeros(n-k)
        return w - np.dot(h, pad)

def compute_r_all(h, w):
    """Compute all r_k values."""
    n = len(h)
    # Use FFT for efficiency
    ac = compute_auto_correlation(h)
    return w - ac

def compute_gradient_r_all(h, w):
    """Compute the gradient of r_k with respect to h."""
    n = len(h)
    grad = np.empty((n, n))  # [i,k] = d r_k / d h_i
    # For each shift k compute row of gradient
    for k in range(n):
        row = np.zeros(n)
        if k == 0:
            row = 1.0 - 2.0 * h
        else:
            # d r_k / d h_i = 1 - (h_{i+k} if exist) - (h_{i-k} if exist)
            if k < n:
                row[:n-k] += (1.0 if k == 0 else 0.0) - h[k:]
            if k > 0:
                row[k:] += (1.0 if k == 0 else 0.0) - h[:-k]
        grad[:, k] = row
    return grad

def smooth_objective(h, w, alpha, dx):
    """Compute the smooth surrogate objective."""
    r_all = compute_r_all(h, w)
    # soft_max_weights = np.exp(alpha * r_all) / np.sum(np.exp(alpha * r_all))
    # C5_approx = -dx * np.dot(soft_max_weights, r_all)
    log_sum = log_sum_exp(r_all, alpha)
    return -dx * log_sum / alpha

def smooth_gradient(h, w, alpha, dx):
    """Gradient of smooth objective with respect to h."""
    n = len(h)
    r_all = compute_r_all(h, w)
    
    # Log-sum-exp stability trick
    max_r = np.max(r_all)
    shifted = r_all - max_r
    exp_alpha_shifted = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_alpha_shifted)
    
    # Weights for the gradient combination
    weights = exp_alpha_shifted / sum_exp
    
    # Compute gradient using auto-correlation
    ac = compute_auto_correlation(h)  # [S_0, ..., S_{n-1}]
    r_all = w - ac  # [r_0, ..., r_{n-1}] (already used some indices)
    
    # d/dh_i of r_k = 1 - (if i+k < n) h_{i+k} - (if i-k >= 0) h_{i-k}
    # So we compute [grad] [h] = [constants that are 1] - corrs
    grad = np.zeros(n)
    
    # Compute total gradient for all shifts
    for k in range(n):
        if k == 0:
            # d r_0 / d h_i = 1 - 2 h_i
            grad += weights[k] * (1.0 - 2.0 * h)
        else:
            # d r_k / d h_i = 1 - (if i+k < n) h_{i+k} - (if i-k >= 0) h_{i-k}
            row = np.zeros(n)
            if k < n:
                if k == 0:
                    row[:] = 1.0 - 2.0 * h[:]
                else:
                    # add 1 to index i part, subtract h_{i+k} if exists
                    row[k:] -= h[:-k] if k > 0 else np.zeros(n - k)
                    row[:n-k] -= h[k:] if k < n else np.zeros(n - k)
                    row += 1.0
            else:
                # Default case or index ranges
                row = np.ones(n) - np.roll(h, k) - np.roll(h, -k)
                row = np.where(row > 0, row, 0)
                pass
            grad += weights[k] * row
    
    return -dx * grad / alpha * (alpha * 1.0)  # for formality use back propagation

def backtracking_line_search(g, h0, w, alpha, dx, eta=0.5, c=1e-4):
    """Backtracking line search with armijo condition."""
    # More a for exploration for simple direction case
    h_current = h0.copy()
    # Validate h_current before optimization
    if np.any(h_current < 0) or np.any(h_current > 1):
        h_current = np.clip(h_current, 0, 1)
    # Project to ensure sum is correct
    sum_h = np.sum(h_current)
    if abs(sum_h - w) > 1e-10:
        h_current = project_to_capped_simplex(h_current, w)
    for _ in range(20):  # limit line searches
        h_trial = h_current - eta * g
        h_trial = project_to_capped_simplex(h_trial, w)
        
        obj_current = smooth_objective(h_current, w, alpha, dx)
        obj_trial = smooth_objective(h_trial, w, alpha, dx)
        if np.isnan(obj_trial):
            eta *= 0.5
            continue
        if obj_trial < obj_current - c * eta * np.dot(g, h_current - h_trial):
            return h_trial, eta
        eta *= 0.5
    return h_current, eta  # fallback

def fista_step(h_current, y_current, t_current, g, w, alpha, dx):
    """FISTA (Nesterov) acceleration step."""
    h_trial, eta = backtracking_line_search(g, h_current, w, alpha, dx)
    t_new = (1 + np.sqrt(1 + 4 * t_current**2)) / 2
    y_new = h_trial + ((t_current - 1) / t_new) * (h_trial - h_current)
    return h_trial, y_new, t_new

def smooth_optimization(h_ini, w, dx, alpha_final, alpha_init=1.0):
    """Optimize using smooth surrogate with continuation."""
    n = len(h_ini)
    h_t = h_ini.copy()
    y_t = h_ini.copy()
    t_t = 1.0
    alpha = alpha_init
    
    while alpha <= alpha_final:
        # optimization loop
        for i in range(max_inner_iter):
            g = smooth_gradient(h_t, w, alpha, dx)
            h_next, y_next, t_next = fista_step(h_t, y_t, t_t, g, w, alpha, dx)
            
            # Stop if change is small
            delta = np.sum((h_next - h_t)**2)
            # print(f"Alpha {alpha}, iter {i}, norm delta {delta}")
            if delta < epsilon:
                break
            
            h_t = h_next
            y_t = y_next
            t_t = t_next
        
        alpha *= alpha_factor
    
    return h_t, smoothing_loss(h_t, w, alpha_final, dx)

def smoothing_loss(h, w, alpha, dx):
    """Return the smooth function value (for monitoring purposes)."""
    r_all = compute_r_all(h, w)
    r_max = np.max(r_all)
    shifted = r_all - r_max
    exp_shifted = np.exp(alpha * shifted)
    log_sum = np.log(np.sum(exp_shifted))
    return -dx * (log_sum / alpha + r_max)

def evaluate_exact(h, dx):
    """Exact C5 evaluation using full correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def generate_initialization(name, n, w):
    """Compute initial h based on the named strategy."""
    np.random.seed(42)  # fixed seed for consistency
    h = np.zeros(n)
    
    if name == "uniform":
        h = np.random.rand(n)
        h = project_to_capped_simplex(h, w)
    elif name == "binary":
        # Create binary pattern
        h = np.zeros(n)
        ones = int(w)
        h[:ones] = 1.0
        np.random.shuffle(h)
        h = project_to_capped_simplex(h, w)
    elif name == "three_block":
        # Place two anchor points
        a = np.random.uniform(0.1, 0.4)
        b = np.random.uniform(0.6, 0.9)
        if a >= b:

        idx_a = int(a * n)
        idx_b = int(b * n)
        h[:idx_a] = 1.0
        h[idx_b:] = 0.0

    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None
    # Global tracking of time
    time_left = budget_s
    restarts_per_n = 4
    
    for i, n in enumerate(n_list):
        dx = dx_list[i]
        w = w_list[i]
        
        # Time allocated to this n
        time_for_n = time_left / len(n_list)  # not perfect but better than nothing of fixed time
        inner_time_limit = time_for_n / restarts_per_n
        
        # Local search for this n
        best_h_for_n = None
        best_c5_for_n = np.inf
        
        # Generate random initiations and run multiple restarts
        for restart in range(restarts_per_n):
            init_strats = ["uniform", "binary", "three_block", "warm"]
            init_method = init_strats[restart % len(init_strats)]
            h0 = generate_initialization(init_method, n, w)
            
            try:
                h_opt, _ = smooth_optimization(h0, w, dx, alpha_max, alpha_init)
                
                exact = evaluate_exact(h_opt, dx)
                if exact < best_c5_for_n:
                    best_c5_for_n = exact
                    best_h_for_n = h_opt.copy()
            except:
                pass
            
            if time.time() - start_time > budget_s * 0.95:
                break
        
        # If found something improve global best
        if best_c5_for_n < best_c5:
            best_c5 = best_c5_for_n
            best_h = best_h_for_n
            best_n = n
        
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Final evaluation
    final_c5 = evaluate_exact(best_h, 2.0 / best_n)
    # Ensure it meets constraints properly
    if not (abs(np.sum(best_h) * (2.0 / best_n) - 1.0) < 1e-6):
        best_h = project_to_capped_simplex(best_h, 1.0 / (2.0 / best_n))
        final_c5 = evaluate_exact(best_h, 2.0 / best_n)
    
    # Return result
    return list(best_h), final_c5, best_n
