def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_to_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0, 1]
    w = np.clip(h, 0, 1)
    # Adjust to sum to n_points/2
    current_sum = np.sum(w)
    target_sum = n_points / 2.0
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return w
    
    # Water-filling approach (adapted for simplicity)
    # Sort indices by value and adjust
    if diff > 0:
        # Need to increase values
        # Find indices where w < 1
        indices = np.where(w < 1)[0]
        if len(indices) == 0:
            return w
        # Distribute the difference
        increasable = 1.0 - w[indices]
        to_distribute = diff
        if to_distribute <= np.sum(increasable):
            # Can distribute all
            increment = to_distribute / len(indices)
            w[indices] += increment
        else:
            w[indices] = 1.0
            # Recursively adjust
            return project_to_feasible(w, n_points)
    else:
        # Need to decrease values
        # Find indices where w > 0
        indices = np.where(w > 0)[0]
        if len(indices) == 0:
            return w
        # Distribute the difference
        decreasable = w[indices]
        to_distribute = -diff
        if to_distribute <= np.sum(decreasable):
            # Can distribute all
            decrement = to_distribute / len(indices)
            w[indices] -= decrement
        else:
            w[indices] = 0.0
            # Recursively adjust
            return project_to_feasible(w, n_points)
    return w

        g[i_vals] += 1.0 - h[j_vals]
        g[j_vals] -= h[i_vals]
    else:
        m = -m
        # i ranges from m to n_points - 1
        i_vals = np.arange(m, n_points)
        j_vals = i_vals - m
        g[i_vals] += 1.0 - h[j_vals]
        g[j_vals] -= h[i_vals]
    return g * dx

def compute_worst_lag(h, dx):
    """Compute the worst lag m and corresponding A_m"""
    n_points = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The valid range of lags is from -(n_points - 1) to (n_points - 1)
    max_val = np.max(conv)
    m_indices = np.where(conv == max_val)[0]
    # Convert to actual m values from the index
    # The index corresponds to lag m = index - (n_points - 1)
    m_vals = m_indices - (n_points - 1)
    return m_vals, max_val * dx

def update_with_symmetry(h):
    """Apply symmetry: h[i] = h[n-1-i] = average of both"""
    n_points = len(h)
    # Apply symmetry only to the first half
    for i in range(n_points // 2 + 1):
        j = n_points - 1 - i
        if i != j:
            avg = (h[i] + h[j]) / 2
            h[i] = h[j] = avg
    return h
