import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT for efficiency."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP formulation."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        # McCormick envelope for binary product
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]

        # Worst-lag constraint
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = np.ravel(h.value)
        T_opt = int(T.value)
        return T_opt, h_opt
    else:
        # Try to recover the best feasible solution
        if h.value is not None:
            h_opt = np.ravel(h.value)
            # Compute best feasible T (find minimum over ell of sum p_ell + T >= M)
            T_opt = 0
            for ell in range(1, n):
                p_ell = p[ell - 1]
                if p_ell.value is not None:
                    s = np.sum(p_ell.value)
                    T_opt = max(T_opt, M - s)
            return T_opt, h_opt
        return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=50):
    """Swap-based simulated annealing on binary vectors."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    current_c5 = evaluate_h(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Estimate initial temperature
    T0 = 0.01
    T = T0
    iter_count = 0
    
    start_time = time.time()
    
    while T > 1e-9 and iter_count < max_iter:
        if time.time() - start_time > time_limit:
            break
        iter_count += 1
        
        # Find indices where h[i] == 1 and h[j] == 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            continue
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
            
        # Swap values
        h[i], h[j] = 0, 1
        new_c5 = evaluate_h(h)
        
        delta = new_c5 - current_c5
        
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Revert swap
            h[i], h[j] = 1, 0
            
        T *= 0.9999

    return best_h, best_c5

def project_to_simplex(h, M):
    """Project h to the simplex {sum h_i = M, 0 <= h_i <= 1} using water-filling."""
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if abs(total - M) < 1e-9:
        return h_clipped
    elif total > M:
        # Need to reduce values (lower bounds)
        # Water-filling algorithm: sort and reduce
        y = np.sort(h_clipped)[::-1]
        z = np.zeros_like(y)
        height = (np.sum(y) - M) / len(y)
        for k in range(len(y)):
            if y[k] - height <= 0:
                z[k] = 0
            else:
                z[k] = y[k] - height
        # re-sort to match original order, but scaled correctly
        idxs = np.argsort(h_clipped)[::-1]
        new_h = h_clipped.copy()
        for i, idx in enumerate(idxs):
            new_h[idx] = z[i]
        return new_h
    else:
        # Add energy
        diff = M - total
        new_h = h_clipped + diff / len(h_clipped)
        return np.clip(new_h, 0, 1)

def smooth_descent(h_start, time_limit=30):
    """Perform gradient descent on log-sum-exp approximation of the max."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    dx = 2.0 / n
    # Ensure bounds are respected
    h = np.clip(h, 0.0, 1.0)
    # Ensure integral constraint
    total = np.sum(h) * dx
    h *= (1.0 / total)
    h = np.clip(h, 0.0, 1.0)
    
    start_time = time.time()
    
    # Stopping criteria
    β_schedule = [1, 2, 5, 10, 20]
    eta = 0.1
    max_iters_per_β = 100
    
    for β in β_schedule:
        if time.time() - start_time > time_limit:
            break
        # Compute all convolution values ahead of time
        conv_vals = []
        weights = []
        
        for ell in range(1, n):
            # Correlation: L_ell = sum h_i * (1 - h_{i+ell})
            s = 0.0
            for i in range(n - ell):
                s += h[i] * (1 - h[i + ell])
            conv_vals.append(s)
        
        # Compute weights for β-scaled version
        conv_vals = np.array(conv_vals)
        exp_vals = np.exp(β * conv_vals)
        weights = exp_vals / np.sum(exp_vals)
        # Weights sum to 1

        for i in range(max_iters_per_β):
            if time.time() - start_time > time_limit:
                break
            # Compute gradient w.r.t. h
            # ∂L_ell/∂h_i = 1 - h_{i+ell} - h_{i-ell}
            # We accumulate over all shifts:
            g = np.zeros(n)
            for ell, w in zip(range(1, n), weights):
                if ell < n:
                    g[:-ell] += w * (1.0 - h[ell:])  # h_{i+ell}
                if ell < n:
                    g[ell:] += w * (-h[:-ell])        # h_{i-ell}

            # Update h with gradient step
            h_new = h - eta * g
            # Project to feasible region:
            h_new = project_to_simplex(h_new, M)
            # Ensure ~(0,1) bounds
            h_new = np.clip(h_new, 0.0, 1.0)
            h = h_new
    return h, evaluate_h(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_score = float('inf')
    best_n = None

    # Try MILP on smaller grids (48, 56) to get strong bounds
    for n in [48, 56]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
            
        T_opt, h_opt = solve_milp(n, time_limit=remaining * 0.6)
        if h_opt is not None:
            score = evaluate_h(h_opt)
            if score < best_score:
                best_score = score
                best_h = h_opt.copy()
                best_n = n
            if best_score < 0.3807:
                break

    # Local search on binary if needed
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20 and best_h is not None:
        h_anneal, score = anneal_binary(best_h, max_iter=int(1e6), time_limit=remaining * 0.5)
        if score < best_score:
            best_score = score
            best_h = h_anneal.copy()
            # No change to n required

    # Smooth refinement if sufficient time
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5 and best_h is not None:
        h_cont, score = smooth_descent(best_h, time_limit=remaining * 0.4)
        if score < best_score:
            best_score = score
            best_h = h_cont.copy()

    # Final quality check and renormalization
    n = len(best_h) if best_h is not None else 48
    dx = 2.0 / n
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h) * dx
    if abs(total - 1.0) > 1e-9:
        best_h *= (1.0 / total)
    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n)
