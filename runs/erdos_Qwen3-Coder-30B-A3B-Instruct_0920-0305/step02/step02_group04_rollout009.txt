import numpy as np
from scipy.optimize import differential_evolution
from scipy.signal import fftconvolve
import time
import cvxpy as cp
from cvxpy import Bool, Int
import math

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def compute_c5_fft(h):
    """Fast computation of C5 with FFT for any vector h."""
    n = len(h)
    dx = 2.0 / n
    # zero pad to 2*n
    h_padded = np.pad(h, (0, n), mode='constant')
    h1_padded = np.pad(1-h, (0, n), mode='constant')
    cross_corr = fftconvolve(h_padded, h1_padded[::-1], mode='valid')
    c5 = np.max(cross_corr[:2*n-1]) * dx
    return c5

            T = cp.Variable(integer=True, name='T')
            # total mass constraint
            mass = cp.sum(h_bin) == M

            # product constraints: p[ℓ][i] = h[i] h[i+ℓ]
            # we'll make a matrix p with size n_points x (n_points-1) for shifts 1 to n_points-1
            # This is a slow double loop, but acceptable for this time budget
            # For each shift ℓ, we generate constraints p[ℓ][i] >= h[i] + h[i+ℓ] - 1, etc.

            p = cp.Variable((n_points, n_points - 1), boolean=True)

            prod_constraints = []
            for shift in range(1, n_points):
                for i in range(n_points - shift):
                    # p[shift][i] should be h[i]*h[i+shift]
                    prod_constraints.append(p[i, shift-1] <= h_bin[i])
                    prod_constraints.append(p[i, shift-1] <= h_bin[i + shift])
                    prod_constraints.append(p[i, shift-1] >= h_bin[i] + h_bin[i + shift] - 1)

            # worst lag constraint
            # for each fixed shift ℓ:
            # sum(p[ℓ]) + T >= M
            # because sum(p[ℓ]) = sum(h[i]*h[i+ℓ]) = M - (actual reduction in overlap)
            shift_overlaps = []
            for shift in range(1, n_points):
                sum_p = cp.sum(p[:, shift-1])
                # Since best_h currently is already a good solution, we can verify
                # that the shift overlaps are bounded by current value
                shift_overlaps.append(sum_p + T >= M)

            # Objective
            objective = cp.Minimize(T)

            # Build the problem
            prob = cp.Problem(objective, [mass] + prod_constraints + shift_overlaps)

            # Use CBC, but make a rough time limit
            time_limit_seconds = 0.3 * remaining
            try:
                prob.solve(solver=cp.CBC, maxSeconds=time_limit_seconds, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.SOLVER_ERROR]:
                    h_opt = h_bin.value
                    T_opt = T.value
                    print(f"MILP optimal T = {T_opt}")
                    if T_opt <= target_int:
                        # converged within target: save better h vector
                        best_h = h_opt.astype(float)
                        best_c5 = T_opt * dx
                        print(f"MILP gave final C5 = {best_c5}")

                    A[shift:, shift] = h_fract[:-shift]

                # Correlation matrix
                # each L_ℓ can be computed:
                # L_ℓ = sum_{i=0}^{n-shift-1} h[i] (1 - h[i+ℓ])
                # Expanding of h[i] - h[i]*h[i+ℓ] only depends on stepping through h vectors,
                # so get all lag sums with one FFT

                def compute_all_Ls(h):
                    Re = []
                    for shift in range(1, n):
                        clamped_h = np.pad(h, (shift, 0), mode='constant')
                        clamped_h1 = np.pad(1-h, (shift, 0), mode='constant')

                        val = np.sum(corr[:n-shift]) * dx
                        Re.append(val)
                    return np.array(Re)

                for _ in range(5):  # few iterations per beta
                    if time.time() - current_time > time_limit:
                        break

                    L_vals = compute_all_Ls(h_fract)
                    exp_L = np.exp(beta * L_vals)
                    wts = exp_L / np.sum(exp_L)
                    grad = np.zeros(n)
                    for shift in range(1, n):
                        # grad[i] += wts[shift-1] * (1 - h_fract[i + shift] - h_fract[i - shift])

                    # Apply grad
                    h_new = h_fract - eta * grad
                    # Box projection
                    h_new = np.clip(h_new, 0, 1)
                    # Mass projection - waterfilling
                    def solve_waterfilling(h_new, target_sum):
                        low, high = -10000, 10000
                        for _ in range(30):

                            new_h = np.clip(h_new + mid, 0, 1)
                            if np.sum(new_h) * dx < target_sum:
                                high = mid
                            else:
                                low = mid
                        new_h = np.clip(h_new + low, 0, 1)
                        return new_h

                    h_fract *= (1.0 / (np.sum(h_fract) * dx))
                    h_fract = np.clip(h_fract, 0, 1)
                    best_c5 = compute_c5_fft(h_fract)

    best_h = np.clip(best_h, 0, 1)

        best_h *= (1.0 / (np.sum(best_h)*dx))
    final_c5 = evaluate_h(best_h)

    return (best_h.tolist(), final_c5, n_points)
