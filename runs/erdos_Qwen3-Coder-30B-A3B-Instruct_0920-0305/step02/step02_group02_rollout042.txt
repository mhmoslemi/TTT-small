import numpy as np
from scipy.optimize import minimize
from scipy.fft import ifft, fft
import time

# Must be available in the environment
def evaluate_erdos_solution(h_values):
    # The official evaluation script from pre-imported
    # This will compute the overlap integral directly
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation via FFT for speed
    corr = ifft(fft(h) * fft(1 - h).conj()).real
    # Shift so that r_k = corr[n-1-k] corresponds to shift k
    # Since h is defined on [0,2], also the shift is from 0 to n-1
    # Extract all valid correlations (k from 0 to n-1)
    r_values = corr[n-1:]  # The first n elements of the correlation have the shifts we want
    max_r = np.max(r_values)
    return max_r * dx

def project_onto_capped_simplex(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=float)
    # Shift to make x >= 0
    x = np.maximum(x, 0.0)
    # now project onto sum = target_sum and 0 <= x_i <= 1
    # Convert to a capped simplex problem
    # We can leverage a water-filling like algorithm:
    # Sort descending, find cut point for sum constraint
    # Then clip at 0 and 1
    v_sorted = np.sort(x)[::-1]  # descending
    v_sorted_cumsum = np.cumsum(v_sorted)
    # Find threshold
    # We want to get closest to target_sum while maintaining boundary constraints.
    # The algorithm is modified to sequence x_back to satisfy sum constraint
    excess = v_sorted_cumsum[-1] - target_sum
    if excess <= 0:
        # No excess, just set all over 1 to 1.0
        x_clip = np.minimum(x, 1.0)
    else:
        # Need to adjust
        i = np.searchsorted(v_sorted_cumsum, target_sum, side='right')
        if i == 0:
            x_clip = np.zeros_like(x)
        else:
            # Cut point calculation
            tau = (v_sorted_cumsum[i-1] - target_sum) / i
            # Apply threshold
            y = np.maximum(v_sorted - tau, 0)
            y = np.minimum(y, 1)
            # Rearrange back to original indices
            idxs = np.argsort(x)[::-1]
            x_clip = np.empty_like(x)
            x_clip[idxs] = y[:len(idxs)]
    # Now ensure the constraint exactly
    sum_current = np.sum(x_clip)
    if abs(sum_current - target_sum) > 1e-10:
        # Adjust smallest non-1 to set exact sum
        diff = target_sum - sum_current
        # Find smallest non-zero/non-one element
        # but it's safer to just correct the first one that isn't saturated.
        # Simple approach: increase one element by diff
        # Implementation note:
        for i in range(len(x_clip)):
            if x_clip[i] < 1.0:
                x_clip[i] += diff
                break
    return x_clip

def compute_correlation(h, k):
    # correlation of h with (1-h) at shift k
    if k < 0 or k >= len(h):
        return 0
    return np.sum(h * np.roll(1 - h, k))

def compute_all_correlations(h):
    # Vectorized version with FFT
    n = len(h)
    # pad to avoid periodic wraparound
    h_padded = np.concatenate([h, np.zeros(n)])
    one_minus_h = 1 - h
    one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n)])
    corr = ifft(fft(h_padded) * fft(one_minus_h_padded).conj()).real
    # Take relevant portion (ends shifted to align with no wraparound)
    # The offset position is the one where 0-shift correlation is centered
    # We want correlations 0,...,n-1
    r_values = corr[n-1:n-1+n]
    return r_values

def smooth_max(h, alpha=1.0):
    n = len(h)
    dx = 2.0 / n
    r_values = compute_all_correlations(h)
    # Use log-sum-exp trick for numerical stability
    r_max = np.max(r_values)
    r_shifted = r_values - r_max
    log_sum_exp = r_max + np.log(np.sum(np.exp(alpha * r_shifted)))
    return dx * log_sum_exp

def gradient_smooth_max(h, alpha=1.0):
    """
    Gradient of the smooth surrogate log-sum-exp of correlations
    """
    n = len(h)
    dx = 2.0 / n
    h = np.asarray(h)
    # First compute correlations
    r_values = compute_all_correlations(h)
    # Stability improvement: shift
    r_max = np.max(r_values)
    r_shifted = r_values - r_max
    # softmax weights
    exp_r = np.exp(alpha * r_shifted)
    sum_exp = np.sum(exp_r)
    weights = exp_r / sum_exp  # (n,)
    # Now compute gradients of each r_k with respect to h_i
    # Column i: ∂r_k / ∂h_i
    # r_k = sum h_i * (1 - h_{i+k})
    # so ∂r_k / ∂h_i = 1 - (h_{i+k} + h_{i-k}) where boundaries are 0
    # define a helper to compute directional partials efficiently
    grad_r = np.zeros((n, n))
    for k in range(n):
        # For shift k, note h_{i+k} wraps around
        # If k = 0: ∂r_0 / ∂h_i = 1 - 2 * h_i
        if k == 0:
            grad_r[k, :] = 1 - 2 * h
        else:
            # Left shift similar to np.roll
            shifted_left = np.roll(h, -k)
            # Right shift via np.roll
            shifted_right = np.roll(h, k)
            grad_r[k, :] = 1 - shifted_left - shifted_right

    # Gradient of the log-sum-exp is weighted average of gradients of r_k
    # Vstack gradients for broadcasting
    # shape (n, n)
    grad_smooth = dx * np.sum(weights.reshape(-1, 1) * grad_r, axis=0)
    return grad_smooth

def backtracking_line_search(h, grad, alpha, func, step_size=0.5, reduction_factor=0.5, max_iter=10):
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_onto_capped_simplex(h_new, target_sum=len(h)/2)
        f_new = smooth_max(h_new, alpha)
        f_old = smooth_max(h, alpha)
        # Armijo condition: f_new <= f_old - c*step_size*|grad|²
        # For simplicity, use a strong Armijo-like condition
        if f_new <= f_old - 0.5 * step_size * np.dot(grad, grad):
            return step_size, h_new
        step_size *= reduction_factor
    return step_size, h_new

def accelerated_projected_gradient_descent(h0, dx, target_sum, alpha0=1.0, alpha_factor=2.0, alpha_max=float('inf'), max_iter=5000):
    """Minimizes smooth approximated max through APG """

    h = h0.copy()
    t = 1.0
    prev_h = h.copy()
    prev_grad = np.zeros_like(h)
    
    # warm start lambda iteration
    alpha = alpha0
    n = len(h)

    for iteration in range(max_iter):
        # compute current objective and gradient
        r_vals = compute_all_correlations(h)
        r_max = np.max(r_vals)
        r_shifted = r_vals - r_max
        exp_r = np.exp(alpha * r_shifted)
        sum_exp = np.sum(exp_r)
        weights = exp_r / sum_exp

        # Gradient of log-sum-exp
        grad = np.zeros_like(h)
        for k in range(n):
            if k == 0:
                local_grad = 1 - 2 * h
            else:
                shifted_left = np.roll(h, -k)
                shifted_right = np.roll(h, k)
                local_grad = 1 - shifted_left - shifted_right

            grad += weights[k] * local_grad

        # Gradient update with step size
        step_size = 0.5  # good starting point
        # First try backtracking line search
        try:
            step_size, h_trial = backtracking_line_search(h, grad, alpha, smooth_max, step_size)
        except Exception:
            # fallback to
            h_trial = h - step_size * grad
            h_trial = project_onto_capped_simplex(h_trial, target_sum=target_sum)

        # Nesterov extrapolation
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y_new = h_trial + ((t - 1) / t_new) * (h_trial - prev_h)
        prev_h = h_trial
        h = y_new
        t = t_new

        # Check
        if iteration % 500 == 0:
            # print(f"alpha={alpha}, step={step_size}, iter={iteration}, smooth_objective={smooth_max(h, alpha)}")
            if iteration > 0:
                pass

        # Adaptive alpha
        if iteration > 0 and iteration % 500 == 0:
            if alpha < alpha_max:
                alpha *= alpha_factor
                # Reset for re-start
                t = 1.0
                prev_h = h.copy()

        # stopping based on smooth objective change
        if iteration > 0 and iteration % 1000 == 0:
            f_old = smooth_max(h, alpha)
            # small tolerance of change
            if abs(f_old - smooth_max(h, alpha)) < 1e-6:
                break

    # Final projection for exact feasibility
    h_proj = project_onto_capped_simplex(h, target_sum=target_sum)
    return h_proj

def initialize_h(n, scheme='uniform'):
    """Initialize a vector satisfying sum(h)*dx=1; thus sum(h)=n/2"""
    dx = 2.0 / n
    w = n / 2
    if scheme == 'uniform':
        h = np.full(n, 0.5)
    elif scheme == 'binary':
        h = np.zeros(n)
        indices = np.random.permutation(n)[:int(w)]
        h[indices] = 1.0
    elif scheme == 'three_block':
        # sample center intervals
        np.random.seed(time.time_ns() % (2**32))
        a = np.random.uniform(0, 0.5)
        b = np.random.uniform(a, 2)
        # len of middle block
        middle_len = (b - a) * n / 2
        if middle_len >= 1:
            p = w - (a * n/2)
            if p < 0:
                p = 0
            else:
                p = p / middle_len
            if p > 1:
                p = 1
            h = np.zeros(n)
            for i in range(n):
                x = 2 * i / n
                if x < a:
                    h[i] = 1.0
                elif x > b:
                    h[i] = 0.0
                else:  # a <= x <= b
                    h[i] = p
        else:
            # fall back to uniform
            h = np.full(n, 0.5)
    else:
        h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum=w)
    return h

        w = n / 2
        time_per_restart = 0.5
        max_restarts = int(total_time_per_n / time_per_restart)

            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Try different schemes

                    best_c5 = c5_opt
                    best_h = h_opt
