import numpy as np
from scipy.optimize import minimize

def evaluate_h(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_to_constraint(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1, sum h_i = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        # Simple bisection to find lambda to adjust h to correct sum
        h = np.clip(h - (current_sum - n_points / 2) / n_points, 0, 1)
    return h

def build_linear_constraints(h):
    """Build linear constraints for all shifts based on current h."""
    n = len(h)
    dx = 2.0 / n
    As = []
    bs = []
    # For each shift s
    for s in range(n):
        a = np.zeros(n)
        b = 0.0
        # Compute linearized term for shift s
        for i in range(n):
            hi = h[i]
            # h_{i+s} where s is the shift index
            if i + s < n:
                h_is = h[i + s]
            else:
                h_is = 0
            # h_{i-s} where s is the shift index
            if i - s >= 0:
                h_i_s = h[i - s]
            else:
                h_i_s = 0
            # Linearization: h_i * h_{i+s} = h_i^k * h_{i+s} + h_{i+s}^k * h_i - h_i^k * h_{i+s}^k
            # Contribution to coefficient a_i:
            # We calculate: combine the (1 - h_{i+s}^k - h_{i-s}^k) * h_i terms
            if i < n - s:
                # Contribution from the bilinear term (h_i h_{i+s})
                a[i] += 1 - h_is - h_i_s
                # Constant term is - h_i^k * h_{i+s}^k
                b -= hi * h_is
        As.append(a)
        bs.append(b)
    return As, bs

def solve_subproblem(As, bs, n_points):
    """Solve the LP subproblem."""
    # Variables
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Constraints
    constraints = []
    # Lin constraints
    for i in range(n_points):
        constraints.append(As[i] @ h + bs[i] <= t)
        
    # Box constraints
    constraints.extend([0 <= h, h <= 1])
    
    # Integral constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, t.value
        else:
            # fallback to projecting the previous feasible point
            return None, None
    except:
        return None, None
