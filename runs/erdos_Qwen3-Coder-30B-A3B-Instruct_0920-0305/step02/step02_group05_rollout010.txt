import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto [0,1] hyperplane with target sum"""
    # Clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
            
        if diff > 0:
            # Need to increase values
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            # Need to decrease values
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
            
        total = w.sum()
        
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of A_m with respect to h"""
    n = len(h)
    g = np.zeros_like(h)
    
    if m >= 0:
        # Overlapping region: indices from 0 to n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # Negative shift: indices from -m to n-1
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    return g

def compute_max_overlap(h, dx):
    """Compute maximum overlap integral and the corresponding shift"""
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(conv)
    max_idx = np.argmax(conv)
    shift = max_idx - (len(h) - 1)
    return max_val, shift

    h_interp = np.interp(new_x, old_x, h0)
    return h_interp
