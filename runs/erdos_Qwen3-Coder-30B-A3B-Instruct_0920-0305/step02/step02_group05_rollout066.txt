def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) = n/2"""
    n = len(h)

    h_proj = np.clip(h, 0, 1)

    diff = target_sum - current_sum
    
    # Water-filling algorithm
    if abs(diff) < 1e-10:
        return h_proj
    
    # Sort indices by h value (descending) to identify which elements to adjust
    sort_indices = np.argsort(h_proj)[::-1]
    
    # If diff > 0 (need to add), we increase low values
    # If diff < 0 (need to subtract), we decrease high values
    remaining = diff
    while abs(remaining) > 1e-10:

        # For diff < 0, we can subtract from values that are > 0
        if remaining > 0:
            # Try to increase elements with current values < 1

                    i += 1
    
    return h_proj

def compute_subgradient(h, m):
    """Compute subgradient for a specific lag m."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    
    if m >= 0:
        # Lagging direction: h_i * (1 - h_{i+m})
        for i in range(n - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        # Leading direction: h_{i+m} * (1 - h_i)
        m_abs = -m
        for i in range(m_abs, n):
            j = i - m_abs
            g[i] += 1 - h[j]
            g[j] -= h[i]
    
    g *= dx
    return g

    # Convert offset indices back to lags
    m_values = max_indices - (n - 1)
    return max_val, m_values.tolist()
