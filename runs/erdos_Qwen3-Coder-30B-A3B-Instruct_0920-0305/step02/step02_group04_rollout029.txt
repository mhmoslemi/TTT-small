```python
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_h(h):
    """Evaluate the C5 constant using FFT for fast correlation computation."""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h) using FFT
    H = fft(h, 2*n)
    G = fft(1 - h, 2*n)
    corr = ifft(H * np.conj(G))[:2*n-1]
    max_val = np.real(corr).max() * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step 1: Discretisation choice
    n = 48  # Modest even grid size
    dx = 2.0 / n
    M = n // 2  # Target mass
    T_target = int(np.ceil(0.3807 * M))  # Target integer bound
    
    # Step 2: MILP formulation
    # Define variables
    h = cp.Variable(n, boolean=True)
    p = cp.Variable((n, n-1), boolean=True)  # p[i][l] = h[i]*h[i+l]
    T = cp.Variable(integer=True)  # Upper bound on max overlap

    # Constraints
    constraints = [
        cp.sum(h) == M,
    ]
    # Product constraints
    for l in range(1, n):
        for i in range(n - l):
            # McCormick envelopes for binary variables:
            # p[i][l] <= h[i], p[i][l] <= h[i+l], p[i][l] >= h[i]+h[i+l]-1
            constraints.append(p[i, l-1] <= h[i])
            constraints.append(p[i, l-1] <= h[i+l])
            constraints.append(p[i, l-1] >= h[i] + h[i+l] - 1)
            constraints.append(p[i, l-1] >= 0)
    
    # Worst-lag constraint
    for l in range(1, n):
        i_start = 0
        i_end = n - l
        total = cp.sum(p[i_start:i_end, l-1])
        constraints.append(total + T >= M)

    # Objective: minimize T
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    # Solve MILP within time limit
    time_limit = min(0.35 * budget_s, 60)  # at most 60s for MILP
    prob.solve(solver=cp.CBC, maximumSeconds=time_limit, verbose=False)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # Fallback to random binary vector
        h_bin = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        while np.sum(h_bin) != M:
            h_bin = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        h_opt = h_bin
    else:
        h_opt = h.value.astype(int)
        
    c5_opt = evaluate_h(h_opt)
    
    # Check if we've hit the target early
    if c5_opt <= 0.3807 and time.time() - start_time < 0.9 * budget_s:
        # Final processing and return
        h_scaled = h_opt.astype(float) * dx
        scale = np.sum(h_scaled) / 1.0
        h_final = h_scaled / scale
        c5_final = evaluate_h(h_final)
        return (h_final.tolist(), c5_final, n)
    
    # Step 4: Binary local refinement (swap based simulated annealing)
    def binary_swap(h_arr, i, j):
        new_h = h_arr.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]
        return new_h

    def binary_annealing(h_start, max_iter=5000, temp_start=1.0, cooling_rate=0.999):
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current)
        h_best = h_current.copy()
        c5_best = c5_current
        
        temp = temp_start
        for iteration in range(max_iter):
            if time.time() - start_time > 0.7 * budget_s:
                break
            # pick two random indices to swap
            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new = binary_swap(h_current, i, j)
            c5_new = evaluate_h(h_new)
            delta = c5_new - c5_current
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
            temp *= cooling_rate
        return h_best, c5_best
    
    h_bin_best, c5_bin_best = binary_annealing(h_opt)
    
    # Step 5: Fractional smoothing via projected gradient descent
    def smooth_max(h, beta=1.0):
        n = len(h)
        dx = 2.0 / n
        # Compute all lags efficiently using FFT
        H = fft(h, 2*n)
        G = fft(1 - h, 2*n)
        corr = ifft(H * np.conj(G))[:2*n-1]
        corr = np.real(corr) * dx
        exp_vals = np.exp(beta * corr)
        weights = exp_vals / np.sum(exp_vals)
        # Compute gradient from weighted norms
        grad = np.zeros_like(h)
        # For each lag compute weight and backpropagate
        for l in range(1, n):
            if l < n:
                # Gradient for that lag influence
                for i in range(n - l):
                    grad[i] += weights[l-1] * (1 - h[i + l] - h[i - l if i - l >= 0 else 0]) * dx
                    if i - l >= 0:
                        grad[i - l] += weights[l-1] * (-h[i]) * dx  # treat as residual if exists
        return corr, weights, grad

    def project_to_simplex(y, target):
        # Project y onto {x : x ≥ 0, sum(x) = target}
        p = np.sort(y)[::-1]
        temp = np.cumsum(p) - target
        idx = np.where(p > temp / np.arange(1, len(p)+1))[0]
        theta = temp[idx[-1]] / (idx[-1] + 1) if len(idx) > 0 else 0.0
        return np.maximum(y - theta, 0.0)

    def projected_gradient_descent(h_start, max_iter=300, beta_schedule=[1.0, 2.0, 5.0, 10.0]):
        h = h_start.astype(float)
        for beta in beta_schedule:
            for _ in range(max_iter // len(beta_schedule)):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                corr, weights, grad = smooth_max(h, beta)
                step_size = 0.01
                h_new = h - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h = project_to_simplex(h_new, M * dx)
        return h
    
    if time.time() - start_time < 0.85 * budget_s:
        h_smooth = projected_gradient_descent(h_bin_best)
        c5_smooth = evaluate_h(h_smooth)
        if c5_smooth < c5_bin_best:
            h_final = h_smooth
            c5_final = c5_smooth
        else:
            h_final = h_bin_best
            c5_final = c5_bin_best
    else:
        h_final = h_bin_best
        c5_final = c5_bin_best

    # Step 6: Final scaling and verification
    h_final = np.clip(h_final, 0, 1)
    integral = np.sum(h_final) * dx
    if abs(integral - 1.0) > 1e-9:
        h_final = h_final / integral
    c5_final = evaluate_h(h_final)

    return (h_final.tolist(), c5_final, n)
```
