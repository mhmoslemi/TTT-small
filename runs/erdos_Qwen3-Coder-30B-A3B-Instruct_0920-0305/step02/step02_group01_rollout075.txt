def gauss_smooth(c, alpha):
    """ Smooth approximation of max for vector c """
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / np.sum(exp_c)
    return w

def compute_c5_smooth(h, alpha=1.0):
    """ Compute smooth C5 approximation """
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    w = gauss_smooth(corr, alpha)
    c5_approx = np.sum(w * corr) * dx
    return c5_approx

def compute_c5_exact(h):
    """ Compute exact C5 value """
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    b2_idx = int((2-b) / dx)
    a2_idx = int((2-a) / dx)
    
    # Fill the function symmetrically
    for i in range(n):
        x = (i + 0.5) * dx
        if x < a or x >= (2-a):
            h[i] = 0
        elif x < b:
            h[i] = u
        elif x < (2-b):
            h[i] = 1
        else:
            h[i] = u
            
    # Apply mass correction
    integral = np.sum(h) * dx
    if integral <= 0:
        return None
    h = (h / integral) * (n / 2.0)
    
    # Ensure h stays in [0,1] after correction
    h = np.clip(h, 0, 1)
    
    return h
