import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    # Project h onto {x : 0 <= x_i <= 1, sum x_i = target_sum}
    # Using bisection to find lagrange multiplier lambda
    h = np.array(h)
    low, high = -1000.0, 1000.0
    for _ in range(40):
        lambda_mid = (low + high) / 2.0
        x = np.clip(h - lambda_mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = lambda_mid
        else:
            high = lambda_mid
    lambda_opt = (low + high) / 2.0
    return np.clip(h - lambda_opt, 0, 1)

def build_five_step_shape(a, b, n):
    # build on [0,2) with n points
    dx = 2.0 / n
    x = np.arange(n) * dx + dx/2.0  # midpoints
    
    # parametric bounds:
    # a -> [0,a]  
    # (a,b) -> [a,b]  height u
    # (b,2-b) -> [b,2-b] height 1
    # (2-b, 2-a) -> [2-b, 2-a] height u  
    # (2-a, 2) -> [2-a, 2] height 0
    
    # Compute u = (2b - 1) / (2*(b-a))
    # This is valid for b>a, and 0<=u<=1 iff u is in [0,1]
    if b <= a:
        return np.full(n, 0.0)
    u = (2*b - 1) / (2*(b-a))
    if u < 0 or u > 1:
        return np.full(n, 0.0)
    
    h = np.full(n, 0.0)
    # Compute intervals: [0, a), [a, b), [b, 2-b), [2-b, 2-a), [2-a, 2)
    for i, xi in enumerate(x):
        if xi < a:
            h[i] = 0
        elif xi <= b:
            h[i] = u
        elif xi <= 2-b:
            h[i] = 1
        elif xi <= 2-a:
            h[i] = u
        else:  # xi > 2-a
            h[i] = 0
    
    return h

def smooth_max_convolution(h, alpha=1.0):
    # Returns (approximation of) the max over shifts of S(h)
    corr = correlate(h, 1 - h, mode="full")
    # compute softmax weights for the 'correlation' values
    weights = np.exp(alpha * corr)
    weights /= np.sum(weights)
    return corr, weights

def compute_gradient(h, weights):
    # d/dh_i ( sum_s weights_s * (h_i - h_i * h_{i+s}) )
    # = 1 - sum_s weights_s * ( h_{i+s} + h_{i-s} )
    
    n = len(h)
    g_plus = correlate(weights[::-1], h, mode="full")[n-1:n-1+n]
    g_minus = correlate(weights, h, mode="full")[n-1:n-1+n]
    grad = 1 - (g_plus + g_minus)
    return grad
