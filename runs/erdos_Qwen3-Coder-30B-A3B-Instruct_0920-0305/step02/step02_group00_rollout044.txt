def build_linear_constraints(h_cur, n):
    """Build linearized constraints for each shift."""
    a_s = []
    b_s = []
    
    for s in range(n):
        # Compute the linear coefficients for shift s
        a = np.zeros(n)
        for i in range(n):
            # Account for boundary conditions (treat out-of-range as 0)
            if i + s < n:
                a[i] = 1.0 - h_cur[i + s] - (h_cur[i - s] if i - s >= 0 else 0.0)
            else:
                a[i] = 1.0 - h_cur[i + s - n] - (h_cur[i - s] if i - s >= 0 else 0.0)
        
        # Compute constant term for shift s
        b = -np.sum(h_cur * np.concatenate([h_cur[s:], np.zeros(s)]))
        a_s.append(a)
        b_s.append(b)
    
    return a_s, b_s

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)

    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5
