import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP formulation for the Erdos problem."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        # McCormick envelopes for products h[i] * h[i+ell]
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]
        # Worst-lag constraint
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    try:
        prob.solve(solver=cp.CBC, verbose=False,
                   maxSeconds=time_limit)
    except:
        pass

    if prob.status in ["optimal", "optimal_inaccurate"]:
        h_sol = np.ravel(h.value)
        T_sol = T.value
    elif prob.status == "infeasible":
        return None, None
    else:
        # Take best feasible solution found
        h_sol = np.ravel(h.value) if h.value is not None else None
        if h_sol is None:
            return None, None
        # Compute T manually from constraints
        T_sol = 0
        for ell in range(1, n):
            length = n - ell
            sum_p = np.sum(np.minimum(h_sol[:length], h_sol[ell:]))
            T_sol = max(T_sol, M - sum_p)
        if not isinstance(T_sol, (int, np.integer)):
            T_sol = int(np.ceil(T_sol))

    if h_sol is None or np.any(np.isnan(h_sol)):
        return None, None

    return T_sol, h_sol

def smooth_descent(h_init, time_limit=100):
    """Refine with smooth gradient descent using log-sum-exp."""
    max_iter = 5000
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = np.clip(h, 0.0, 1.0)
    T = 100  # dummy value to start
    beta_schedule = [1, 2, 5, 10, 20]

    start = time.time()
    for beta in beta_schedule:
        if time.time() - start > time_limit:
            break
        for it in range(max_iter):
            if time.time() - start > time_limit:
                break
            # Compute all L_ell
            HH = np.fft.rfft(h, 2*n)
            GG = np.fft.rfft(1.0 - h, 2*n)
            conv_full = np.fft.irfft(HH * np.conj(GG))[:2*n-1]
            L = conv_full.real  # L[ell] = sum h[i] (1 - h[i+ell])

            # Compute weights
            L_max = np.max(L)
            L_shifted = L - L_max
            exp_L = np.exp(beta * L_shifted)
            total = np.sum(exp_L)
            if total <= 0:
                break
            w = exp_L / total

            # Compute gradient
            grad = np.zeros(n)
            for ell in range(1, n):
                if ell < n:
                    # Forward difference hh[i] - hh[i+ell]
                    diff = h - np.concatenate([h[ell:], np.zeros(ell)])
                    # Backward difference hh[i+ell] - hh[i]
                    diff_rev = np.concatenate([np.zeros(ell), h[:-ell]]) - h
                    grad += w[ell-1] * (1 - diff - diff_rev)

            # Gradient step
            eta = 0.1
            while True:
                h_trial = h - eta * grad
                h_trial = np.clip(h_trial, 0.0, 1.0)
                # Project to sum = M
                total_h = np.sum(h_trial) * dx
                if abs(total_h - 1.0) < 1e-9:
                    h_new = h_trial
                    break
                # Water-filling projection
                h_trial_sorted = np.sort(h_trial)[::-1]  # descending
                cumsum = np.cumsum(h_trial_sorted)
                n_above = np.searchsorted(cumsum, 1.0, side='right')
                lambda_guess = (1.0 - cumsum[n_above-1]) / n_above if n_above > 0 else 0.0
                lambda_lo = -1e-9
                lambda_hi = 1.0 + 1e-9
                for _ in range(30):
                    lambda_mid = (lambda_lo + lambda_hi) / 2
                    h_proj = np.clip(h_trial - lambda_mid, 0.0, 1.0)
                    if np.sum(h_proj) * dx >= 1.0:
                        lambda_lo = lambda_mid
                    else:
                        lambda_hi = lambda_mid
                lambda_opt = (lambda_lo + lambda_hi) / 2
                h_new = np.clip(h_trial - lambda_opt, 0.0, 1.0)
                if np.abs(np.sum(h_new) * dx - 1.0) < 1e-9:
                    break
                eta *= 0.5
                if eta < 1e-10:
                    break
            h = h_new

    return h, evaluate_h(h)

def anneal_binary(h_init, max_iter=500000, time_limit=100):
    """Binary swap-based simulated annealing."""
    n = len(h_init)
    M = n // 2
    h = h_init.copy()
    score = evaluate_h(h)
    best_h = h.copy()
    best_score = score

    start = time.time()
    T = 0.1

    # Estimate typical Δ
    sum_delta = 0.0
    count_delta = 0
    for _ in range(500):
        if time.time() - start > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        # Temporarily flip
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_score = evaluate_h(new_h)
        delta = new_score - score
        if delta > 0:
            sum_delta += delta
            count_delta += 1
    if count_delta > 0:
        T = 1.5 * sum_delta / count_delta

    temp = T
    alpha = 0.999
    last_improvement = 0

    for i in range(max_iter):
        if time.time() - start > time_limit:
            break
        # Swap a 1 for a 0
        ones = np.nonzero(h)[0]
        zeros = np.nonzero(1 - h)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i_1 = np.random.choice(ones)
            j_0 = np.random.choice(zeros)
            # Try the swap
            new_h = h.copy()
            new_h[i_1] = 0
            new_h[j_0] = 1
            new_score = evaluate_h(new_h)
            delta = new_score - score
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = new_h
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                    last_improvement = i
        temp *= alpha
        if temp < 1e-6 or i - last_improvement > 10000:
            temp = T

    return best_h, best_score
