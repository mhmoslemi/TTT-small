import numpy as np
from scipy.optimize import minimize
from scipy.fft import ifft, fft
import time
import random

# Constants
n_candidates = [48, 56, 64, 80]
dx = 2.0 / n_candidates[0]  # Starting point
w = n_candidates[0] / 2
target_bound = 0.3807

def evaluate_c5(h):
    """Evaluate the C_5 bound for given h values."""
    # Make sure h is a numpy array
    h = np.array(h)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex {x : 0 <= x_i <= 1, sum x_i = target_sum}"""
    v = np.array(v)
    n = len(v)
    # Water-filling algorithm
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    # Find threshold
    threshold = 0
    for i in range(n):
        if v_sorted[i] > (target_sum - cumsum[i]) / (i+1):
            threshold = (target_sum - cumsum[i]) / (i+1)
            break
    else:
        threshold = (target_sum - cumsum[-1]) / n

    # Apply threshold and clip
    result = np.clip(np.maximum(v - threshold, 0), 0, 1)
    # Adjust to ensure sum = target_sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Correct by adjusting the smallest non-zero value
        inds = np.where(result > 1e-10)[0]
        if len(inds) > 0:
            idx = inds[0]
            result[idx] += diff
    return result

def compute_r_k(h, k):
    """Compute r_k = sum_i h_i * (1 - h_{i+k}) for a given k."""
    # Pad the array for circular convolution
    h_padded = np.concatenate([h, np.zeros(k)])
    h_shifted = np.concatenate([np.zeros(k), h])
    r_k_value = np.sum(h * (1 - h_shifted[:len(h)]))
    return r_k_value

def combined_correlations(h):
    """Return all r_k values for all k."""
    n = len(h)
    r_values = []
    for k in range(n):
        r_values.append(compute_r_k(h, k))
    return np.array(r_values)

def compute_gradient_single_rk(h, k):
    """Compute gradient of r_k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # r_k = sum_i h_i * (1 - h_{i+k})
    for i in range(n):
        grad[i] = 1.0
        # Add contribution from h_{i+k}
        i_k = (i + k) % n
        grad[i] -= h[i_k]
        # Add contribution from h_{i-k}
        i_negk = (i - k) % n
        grad[i] -= h[i_negk]
    return grad

def compute_gradient_rk_h(h):
    """Compute the gradient of the full correlation sum vector."""
    n = len(h)
    g = np.zeros(n)
    for k in range(n):
        grad_k = compute_gradient_single_rk(h, k)
        g += grad_k
    return g / n  # Average over k to get overall effect

def compute_gradient_f_alpha(h, r_vals, alpha):
    """Compute gradient of log-sum-exp surrogate."""
    n = len(h)
    # Compute softmax of alpha * r_vals
    r_max = np.max(r_vals)
    exp_r = np.exp(alpha * (r_vals - r_max))
    sum_exp_r = np.sum(exp_r)
    weights = exp_r / sum_exp_r
    # Compute weighted gradient sum
    g = np.zeros(n)
    for k in range(n):
        grad_k = compute_gradient_single_rk(h, k)
        g += weights[k] * grad_k
    return dx * g

def smooth_max_gradient_descent(h0, dx, w, alpha0=1, alpha_factor=2, alpha_max=300, max_iters=5000):
    """Run accelerated gradient descent with smoothed surrogate."""
    h = h0.copy()
    t = 1
    alpha = alpha0
    
    last_f = np.inf
    for iteration in range(max_iters):
        # Compute r_vals (efficiently using FFT)
        r_vals = combined_correlations(h)
        # Smooth surrogate function value
        r_max = np.max(r_vals)
        exp_r = np.exp(alpha * (r_vals - r_max))
        sum_exp_r = np.sum(exp_r)
        f_alpha = (dx / alpha) * (np.log(sum_exp_r) + r_max)
        
        # Early exit if converged
        if abs(f_alpha - last_f) < 1e-10:
            break
        last_f = f_alpha
        
        # Compute gradient
        g = compute_gradient_f_alpha(h, r_vals, alpha)
        
        # Nesterov-style extrapolation
        y = h + ((t - 1) / (t + 1)) * (h - h)
        # Gradient step
        eta = 0.5
        h_tilde = project_onto_capped_simplex(y - eta * g, w)
        
        # Update t
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        
        # Coordinate descent step
        h = h_tilde + ((t - 1) / t_new) * (h_tilde - h)
        h = project_onto_capped_simplex(h, w)
        
        # Increase alpha if needed
        if iteration % 100 == 0 and alpha < alpha_max:
            alpha *= alpha_factor
    
    return h, evaluate_c5(h)

def initial_h_values_method(seed):
    """Generate initial h values using three-block approach."""
    np.random.seed(seed)
    n = n_candidates[0]
    dx = 2.0 / n
    w = n / 2

    # Generate two random cut points
    a = np.random.rand() * 0.8  # Between 0 and 0.8
    b = a + 0.1 + np.random.rand() * 0.5  # Ensures b - a > 0.1 and some extra spread
    if b > 1.0:
        b = 1.0
        a = b - 0.1 - np.random.rand() * 0.5

    # Value for middle part
    p = (w - 2 * a * n / 2.0) / (b * n / 2.0 - a * n / 2.0) if (b - a) > 0 else 0.5 * w / (n / 2.0)

    h = np.zeros(n)
    x_vals = np.linspace(0, 2, n)
    for i, x in enumerate(x_vals):
        if x < a:
            h[i] = 1.0
        elif x < b:
            h[i] = p
        else:
            h[i] = 0.0

    # Perturb slightly and project
    h += np.random.normal(0, 0.02, n)
    h = np.clip(h, 0, 1)
    h = project_onto_capped_simplex(h, w)
    return h

def run(seed=42, budget_s=600):
    """Main optimization routine with acceleration and restart."""
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    best_n = None

    # Create a list of the relative computation budget to each n
    budget_list = [budget_s / len(n_candidates) * 0.95 for _ in n_candidates]

    for i, n in enumerate(n_candidates):
        current_budget = budget_list[i]
        if time.time() - start_time >= budget_s * 0.95:
            break

        dx = 2.0 / n
        w = n / 2
        restart_count = int(current_budget / 0.5)  # Allow ~ half seconds per restart for this n

        for _ in range(restart_count):
            if time.time() - start_time >= budget_s * 0.95:
                break
            # Random restart with three-block pattern
            h0 = initial_h_values_method(seed + _)
            h_opt, c5_opt = smooth_max_gradient_descent(h0, dx, w, max_iters=5000)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    # Final validation
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, w)
        final_c5 = evaluate_c5(best_h)
        return list(best_h), final_c5, best_n

    return list(np.ones(n_candidates[0]) * 0.5), 1.0, n_candidates[0]
