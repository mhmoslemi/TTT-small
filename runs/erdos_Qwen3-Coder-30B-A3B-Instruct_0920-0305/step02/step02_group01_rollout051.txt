import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(v, S):
    """Project onto the set {x: sum(x)=S, 0 ≤ x ≤ 1}"""
    # Use bisection on lambda to find projection
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        sum_x = x.sum()
        if sum_x > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Ensure sum is close to S by distributing residual
    residual = S - x.sum()
    if abs(residual) > 1e-10:
        free = (x > 1e-8) & (x < 1-1e-8)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0, 1)
    return x

def gradient_descent(h0, S, budget, start_time):
    """Run gradient descent with projection onto feasible set"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    dx = 2.0 / n
    alpha = 0.5
    lr = 0.5
    iteration = 0
    while time.time() - start_time < budget:
        conv = correlate(h, 1 - h, mode='full')
        shift = conv - np.max(conv)
        exp_vals = np.exp(alpha * shift)
        p = exp_vals / exp_vals.sum()
        p_rev = p[::-1]
        conv_plus = np.convolve(p_rev, h, mode='full')[n-1:n-1+n]
        conv_minus = np.convolve(p, h, mode='full')[n-1:n-1+n]
        g = 1.0 - (conv_plus + conv_minus)
        h_new = h - lr * g
        h = project_to_capped_simplex(h_new, S)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        iteration += 1
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 20)
            lr = lr * 0.9
        if iteration > 5000:
            break
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    total_budget = budget_s
    n = 200
    dx = 2.0 / n
    S = n/2
    
    # Step 1: Parameter grid search
    best_h = None
    best_c5 = np.inf
    
    # Use a finer grid for parameter search
    a_vals = np.linspace(0, 0.5, 101)
    b_vals = np.linspace(0.5, 1.0, 101)
    
    for a in a_vals:
        if time.time() - start > total_budget - 5:
            break
        for b in b_vals:
            if b <= a: continue
            denom = 2*(b - a)
            u = (2*b - 1) / denom if denom != 0 else 0
            if not (0 <= u <= 1):
                continue
            # Build h
            x = (np.arange(n) + 0.5) * dx
            h = np.zeros(n)
            mask2 = (x >= a) & (x < b)
            mask3 = (x >= b) & (x < 2-b)
            mask4 = (x >= 2-b) & (x < 2-a)
            h[mask2] = u
            h[mask3] = 1
            h[mask4] = u
            # Ensure sum constraint
            sumh = h.sum()
            if sumh * dx != 1:
                scale = 1.0 / (sumh * dx)
                h = np.clip(h * scale, 0, 1)
            # evaluate
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Step 2: Local gradient descent refinement
    elapsed = time.time() - start
    remaining = total_budget - elapsed
    if remaining > 5 and best_h is not None:
        h_ref, c5_ref = gradient_descent(best_h, S, remaining*0.9, start)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    # Step 3: Random restart gradient descent
    while time.time() - start < total_budget - 1:
        h0 = np.random.rand(n)
        h0 = project_to_capped_simplex(h0, S)
        h_r, c5_r = gradient_descent(h0, S, 30, start)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        if time.time() - start > total_budget - 1:
            break

    # Final scaling to exactly meet sum constraint
    best_h = best_h * (S / best_h.sum())
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_c5(best_h)

    return (best_h.tolist(), float(final_c5), n)
