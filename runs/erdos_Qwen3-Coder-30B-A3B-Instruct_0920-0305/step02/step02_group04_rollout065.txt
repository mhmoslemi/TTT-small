import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the MILP formulation of the binary Erdos problem."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    constraints = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        constraints += [
            p_ell <= h[:length],
            p_ell <= h[ell:],
            p_ell >= h[:length] + h[ell:] - 1
        ]
        constraints += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
        if prob.status in ['optimal', 'optimal_inaccurate', 'feasible']:
            h_val = np.ravel(h.value)
            T_val = int(T.value)
            return T_val, h_val
        return None, None
    except:
        return None, None

def project_to_simplex(h, target_sum):
    """Project h onto the simplex with sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-9:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # need to add diff in total, increase all entries equaly
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    else:
        # need to subtract |diff| in total, decrease all entries equally
        h -= -diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_descent(h_start, time_limit=20):
    """Smooth gradient descent Refinement."""
    n = len(h_start)
    M = n // 2
    dx = 2.0 / n
    h = h_start.copy()
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        h = project_to_simplex(h, M)
    
    def compute_lags(h):
        # Compute all L_ell(h) = sum h_i (1 - h_{i+ell})
        H = np.fft.rfft(h, 2*n)
        G = np.fft.rfft(1.0 - h, 2*n)
        conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
        return conv.real

    def compute_weights(L_vals, beta):
        # Compute softmax weights
        L_vals = np.array(L_vals)
        exp_vals = np.exp(beta * L_vals)
        w = exp_vals / np.sum(exp_vals)
        return w

    # Schedule of betas
    betas = [1, 2, 5, 10, 20]

    start_time = time.time()
    for beta in betas:
        if time.time() - start_time > time_limit * 0.9:
            break
        # get all lag values
        L_vals = compute_lags(h)
        weights = compute_weights(L_vals, beta)
        # construct gradient
        grad = np.zeros(n)
        # For each lag ell compute gradient contribution
        for ell in range(1, n):
            weight = weights[ell-1]
            if ell < n:
                shifted = np.roll(h, -ell)
                grad += weight * (1.0 - shifted - np.roll(h, ell))
        # simple gradient step
        eta = 0.1
        h_new = h - eta * grad
        # adopt step
        h = h_new.copy()
        h = np.clip(h, 0.0, 1.0)
        # Project to sum constraint
        h = project_to_simplex(h, M)
    return h, evaluate_h(h)

def anneal_binary(h_start, max_iter=int(1e6), time_limit=10):
    """Use binary simulated annealing to improve a binary solution."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    current_score = evaluate_h(h)
    best_h = h.copy()
    best_score = current_score

    # compute initial temperature by sampling
    temp = 1.0
    initial_deltas = []
    for _ in range(500):
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == 1 and h[j] == 0:
            h[i], h[j] = 0, 1
            new_score = evaluate_h(h)
            delta = new_score - current_score
            if delta > 0:
                initial_deltas.append(delta)
            h[i], h[j] = 1, 0  # undo
    if initial_deltas:
        temp = 5.0 * np.mean(initial_deltas)
    else:
        temp = 1.0

    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100000

    while iter_count < max_iter and time.time() - start_time < time_limit:
        if no_improvement_count > max_no_improvement:
            # reheat
            temp = max(temp * 0.5, 1e-8)
            no_improvement_count = 0

        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == 1 and h[j] == 0:
            h[i], h[j] = 0, 1
            new_score = evaluate_h(h)
            delta = new_score - current_score
            if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                h[i], h[j] = 1, 0  # undo
        iter_count += 1
        temp *= 0.999

    return best_h, best_score
