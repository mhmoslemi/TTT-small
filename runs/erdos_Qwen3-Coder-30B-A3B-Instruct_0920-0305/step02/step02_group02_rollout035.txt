import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import math

# Constants
n_points_list = [48, 56, 64, 80]
dx_list = [2.0 / n for n in n_points_list]
w_list = [n / 2.0 for n in n_points_list]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation using FFT
    h_pad = np.concatenate([h, np.zeros_like(h)])
    corr = ifft(fft(h_pad) * np.conj(fft(h_pad))).real[:len(h)]
    # Compute r_k = w - S_k
    r = np.array([w - s for s, w in zip(corr, w_list)])
    # Take maximum over all shifts (we take max over a few dominant shifts)
    max_r = np.max(r)
    # Scale by dx
    c5 = dx_list[0] * max_r
    return c5

def smooth_h(h_values, alpha):
    """Compute smooth surrogate F_alpha(h) = (dx/alpha) * log(sum(exp(alpha*r_k)))"""
    h = np.array(h_values)
    h_pad = np.concatenate([h, np.zeros_like(h)])
    corr = ifft(fft(h_pad) * np.conj(fft(h_pad))).real[:len(h)]
    r = np.array([w - s for s, w in zip(corr, w_list)])
    # Log-sum-exp trick for numerical stability
    r_max = np.max(r)
    r_shifted = r - r_max
    logsumexp = r_max + np.log(np.sum(np.exp(alpha * r_shifted)))
    return (dx_list[0] / alpha) * logsumexp

def compute_smooth_gradients(h_values, alpha):
    """Compute gradient of smooth surrogate."""
    h = np.array(h_values)
    h_pad = np.concatenate([h, np.zeros_like(h)])
    corr = ifft(fft(h_pad) * np.conj(fft(h_pad))).real[:len(h)]
    r = np.array([w - s for s, w in zip(corr, w_list)])
    
    # Compute weights (soft-max)
    r_max = np.max(r)
    r_shifted = r - r_max
    exp_alpha_r = np.exp(alpha * r_shifted)
    weights = exp_alpha_r / np.sum(exp_alpha_r)
    
    # Gradient of each r_k wrt h: (1 - h_{i+k}) - (1 - h_{i-k})
    # For k=0: 1 - 2*h_i
    # For k>0: -h_{i+k} - h_{i-k}    
    grad = np.zeros_like(h)
    for k in range(len(h)):
        if k == 0:
            grad += weights[k] * (1 - 2 * h)
        else:
            h_shift = np.zeros_like(h)
            h_shift[k:] = h[:-k]
            h_shift[:k] = 0
            h_shift_rev = np.zeros_like(h)
            h_shift_rev[:-k] = h[k:]
            h_shift_rev[-k:] = 0
            grad += weights[k] * (-h_shift - h_shift_rev)
    return grad

def project_capped_simplex(v, w):
    """Project v onto the capped simplex with sum = w."""
    n = len(v)
    # Sort descending
    sort_idx = np.argsort(v)[::-1]
    sorted_v = v[sort_idx]
    # Try to find threshold τ such that Σ min(max(v_i - τ, 0), 1) = w
    cumulative = np.cumsum(sorted_v)
    # Augment with zeros for index consistency
    cumulative_padded = np.concatenate([[0], cumulative])
    n_padded = np.arange(n + 1)
    # List of possible indices and thresholds
    # Inner while loop
    i = 0
    while i < n and (sorted_v[i] > 1):
        i += 1
    if i > 0 and (cumulative[i-1] >= w):
        # We can place everything at 0 or 1
        while i and cumulative[i-1] >= w:
            i -= 1
        tau = (cumulative[i] - w) / (n - i)
        result = np.zeros_like(v)
        result[sort_idx[:i]] = 1.0
        result[sort_idx[i]] = 1 - tau
        return result
    # Traverse from the top to find α that violates sum = w
    condition = np.where(cumulative_padded[:n] - (n_padded-1) * sorted_v[:] > w)[0]
    if len(condition) == 0:
        # All can be set to 1
        result = np.ones_like(v)
        result[sort_idx] = sorted_v
        return result
    # Now we have first violation, so find τ directly
    j = condition[0]  # index where sum is too small
    # instead, compute directly
    # For key point: cum_value - (j) * sorted_v[j] = w
    # i = j position, immediately we take an answer:
    # we set all earlier to 1, then [j] to some value so that
    # relation holds
    cli = sorted_v[j]
    # Slope of function sum - j*sorted_v[j] vs j
    if j > 0:
        # Sum_j-1 gives cumulative[j-1]
        # So: cum[j-1] - j*sorted_v[j] = w => sorting_v[j] = (cum[j-1] - w)/j
        if j > 0:
            inv = j
            tau = (cumulative[j-1] - w) / inv
        else:
            tau = (0 - w) / 1  # implies negative
    else:
        tau = -w  # not feasible either, set all to 0 or 1
        # avoid computing by hand
        # U can say lower is also problematic under constraints:
        # Would mean tau > max(sorted_v[0]), which implies being more than 1
        # Check edge cases carefully.
        # Let us instead just compute: all checked
        # tau = 0 first see if cum[0] is less than w
        if j == 0:
            result = np.zeros_like(v)
            # Option 1: Neutral
            # Option 2: Try values
            if cumulative[0] >= w:
                # v[0] >= (w/1)
                result[sort_idx[0]] = w
            else:
                # Infeasible, reduce to sum = w with 1
                result[sort_idx] = np.minimum(sorted_v, np.ones_like(sorted_v))
            return result
        else:
            # Not perhaps very accurate, but okay for now:
            result = np.clip(sorted_v - tau, 0, 1)
            total = np.sum(result)
            # normalize it to match w
            scale = w / total
            result = result * scale
            result = result[sort_idx]
            result = result[sort_idx]
            return result
    # End of detailed case
    # Instead, compute final value
    # Indeed, the threshold τ = (cumulative[j-1] - w)/(j)
    if j == 0:
        tau = w / (n + 1)  # improper case
    else:
        # General case
        # Here we have j obverse points to be 1, one to be some value:
        sum_before = cumulative[j - 1]
        # use sum >= w case
        # if sum of s[0..j-1] >= w then we alter:
        if sum_before >= w:
            while j > 0 and cumulative[j - 1] >= w:
                j -= 1
            if j == 0:
                tau = 0
            else:
                tau = (cumulative[j - 1] - w) / (n - j)
        else:
            tau = (sum_before - w) / (n - j)
    # Reconstruct result
    # Finalized algo:
    result = np.zeros_like(v)
    for i in range(n):
        if i < j:
            result[sort_idx[i]] = 1.0
        else:
            z = sorted_v[i] - tau
            if z > 1:
                result[sort_idx[i]] = 1.0
            elif z < 0:
                result[sort_idx[i]] = 0.0
            else:
                result[sort_idx[i]] = z
    return result

def accelerated_optimize(h0, n, alpha_start=1.0, alpha_final=500.0, max_iter=1000, tol=1e-6):
    """Optimize the smooth surrogate with accelerated gradient."""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    alpha = alpha_start
    objective_history = []
    while alpha < alpha_final * 1.1:
        prev_obj = np.inf
        for it in range(max_iter):
            # Compute surrogate
            current_obj = smooth_h(h, alpha)
            if abs(current_obj - prev_obj) / abs(prev_obj) < tol:
                break
            prev_obj = current_obj
            # Gradient step
            grad = compute_smooth_gradients(h, alpha)
            step_size = 0.5
            backtrack = True
            while backtrack and step_size > 1e-10:
                h_new = y - step_size * grad
                h_new = project_capped_simplex(h_new, w_list[0])
                # Check Armijo condition
                obj_new = smooth_h(h_new, alpha)
                if obj_new <= current_obj - 0.5 * step_size * np.dot(grad, h_new - y):
                    h = h_new
                    backtrack = False
                else:
                    step_size *= 0.5
            # Polyak step
            t_old = t
            t = 0.5 * (1 + math.sqrt(1 + 4 * t_old * t_old))
            y = h + ((t_old - 1) / t) * (h - h0)
            h0 = h
            objective_history.append(current_obj)
            if len(objective_history) > 30 and abs(objective_history[-1] - objective_history[-2]) < tol:
                break
        alpha *= 2.0
    return h

def generate_initial_h(n, method="uniform"):
    """Generate initial feasible starting point."""
    if method == "uniform":
        h = np.random.uniform(0, 1, n)
        h = project_capped_simplex(h, w_list[n_points_list.index(n)])
    elif method == "binary":
        # Generate binary vector with w ones
        h = np.zeros(n)
        h[:int(w_list[n_points_list.index(n)])] = 1
        np.random.shuffle(h)
        h += np.random.normal(0, 0.02, n)
        h = project_capped_simplex(h, w_list[0])
    elif method == "three_block":
        # Simple heuristic
        a = np.random.rand() * 0.5
        b = a + np.random.rand() * 0.5
        h = np.zeros(n)
        switch_a = int(a * n)
        switch_b = int(b * n)
        h[switch_a:switch_b] = 1  # or something better
        # Adjust to sum w
        current_sum = np.sum(h)
        h[:] = h * (w_list[n_points_list.index(n)] / current_sum)
        h = project_capped_simplex(h, w_list[n_points_list.index(n)])
    else:
        h = np.ones_like(h) * 0.5
        h = project_capped_simplex(h, w_list[0])
    return h
