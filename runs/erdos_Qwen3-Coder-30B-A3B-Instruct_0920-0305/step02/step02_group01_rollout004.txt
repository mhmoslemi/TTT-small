import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def construct_five_step(a, b, n):
    """Construct a 5-step symmetric function with parameters a, b"""
    dx = 2.0 / n
    x = np.linspace(dx/2, 2-dx/2, n)
    u = (2*b - 1) / (2*(b - a))
    if u < 0 or u > 1:
        return None, False
    h = np.zeros(n)
    # Slide along x grid to set h hard values
    for i in range(n):
        xi = x[i]
        if (xi <= a) or (xi >= 2-a):
            h[i] = 0
        elif (xi <= b) or (xi >= 2-b):
            h[i] = u
        else:
            h[i] = 1
    # Adjust the mass to match constraint
    total_mass = np.sum(h) * dx
    if total_mass == 0:
        return None, False
    h = h * (n/2) / total_mass
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    return h, True

def project_simplex(h, target_sum):
    """Project h onto the capped simplex: sum=desired, 0<=h<=1"""
    # From the standard algorithm for projection onto a simplex
    # Find lambda such that sum of clipped values equals target sum
    h = np.array(h)
    s = np.sort(h)[::-1]
    cumsum = np.cumsum(s)
    n = len(h)

    lambda_lo = 0.0
    lambda_hi = s[0] + 1.0

        lambda_m = (lambda_lo + lambda_hi) / 2.0
        # Compute sum of clipped values
        clipped = np.clip(h - lambda_m, 0, 1)
        total = np.sum(clipped)
        if total > target_sum:
            lambda_lo = lambda_m
        else:
            lambda_hi = lambda_m
    lambda_f = (lambda_lo + lambda_hi) / 2.0
    h_proj = np.clip(h - lambda_f, 0, 1)
    return h_proj

def smooth_max_corr(h, alpha=0.5):
    """
    Smooth approximation of max over all lags
    """
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    corr = corr * dx  # scale
    # Softmax weights
    exp_corr = np.exp(alpha * corr)
    weights = exp_corr / np.sum(exp_corr)
    return np.sum(weights * corr), weights

def gradient_smooth_c5(h, weights):
    """Compute gradient of smooth C5"""
    n = len(h)
    dx = 2.0 / n
    # convolution of weights with itself should be neutral
    w = weights
    # Compute d/dh_i of sum_s w_s * h_i * (1 - h_{i+s})
    # It's 1 - sum_s w_s (h_{i+s} + h_{i-s})
    g_plus = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]
    g_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]
    grad = 1.0 - (g_plus + g_minus)
    return grad

def gradient_descent_refinement(h_trial, n_points, max_iter=5000, 
                                alpha_start=0.5, alpha_mult=1.5,

    """

    h = h_trial.copy()
    dx = 2.0 / n_points
    
    alpha = alpha_start
    eta = initial_eta
    best_h = h.copy()
    best_c5_exact = evaluate_c5(h)

        # Update alpha
        if it > 0 and it % 200 == 0:
            alpha = min(alpha * alpha_mult, 20.0)
        
        # Compute smooth objective and its weights
        val_smooth, weights = smooth_max_corr(h, alpha)

        # Gradient step
        h_new = h - eta * grad
        
        # Project to feasible set

        h_new = np.clip(h_new, 0, 1)
        
        # Check if better (must compute exact C5)
        c5_val = evaluate_c5(h_new)
        if c5_val < best_c5_exact:
            best_h = h_new.copy()
            best_c5_exact = c5_val
            
        h = h_new
        
        # Adapt eta
        if it % 200 == 0:
            eta *= eta_decay
            
    return best_h, best_c5_exact
