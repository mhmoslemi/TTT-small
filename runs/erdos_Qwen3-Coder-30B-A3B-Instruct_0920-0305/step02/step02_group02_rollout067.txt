import numpy as np
from scipy.optimize import minimize
from scipy.linalg import eigh
from cvxpy import *
import time
import cvxpy as cp

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * (2.0 / len(h))
    return c5

def solve_sdp(n):
    """Solve SDP relaxation for given n."""
    dx = 2.0 / n
    w = n / 2
    
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    
    # Objective: maximize t
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD
        cp.diag(X) == cp.multiply(h, h),  # diag(X) = h^2
        cp.sum(h) == w,
        0 <= h,
        h <= 1,
    ]

    # Add constraints for each k
    for k in range(n):
        # Toeplitz matrix T_k - 1 if i = j+k, else 0
        T_k = np.zeros((n, n))
        T_k = np.eye(n-k, k=0) if k == 0 else np.eye(n-k, k=k) 
        # Adjust for symmetry around diagonal
        T_k = T_k + T_k.T - np.diag(np.diag(T_k))
        # Align properly with the actual correlations
        T_k_full = np.zeros((n,n))
        for i in range(n):
            for j in range(n):
                if i + k == j:
                    T_k_full[i,j] = 1
                    
        constraints.append(cp.trace(T_k_full @ X) >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return h.value, X.value, t.value
    except:
        pass  # Return None if infeasible or error
    return None, None, None

def project_to_simplex(v, w):
    """Project vector v onto capped simplex with sum = w"""
    v = np.array(v)
    # Sort values in descending order
    idx = np.argsort(-v)
    v_sorted = v[idx]
    
    # Find threshold
    cumsum = np.cumsum(v_sorted)
    threshold = (cumsum - w) / np.arange(1, len(v) + 1)
    # Find largest k such that threshold[k] >= v_sorted[k]
    k = np.argmax(v_sorted[idx] > threshold)
    if k == 0 and v_sorted[0] <= threshold[0]:
        k = -1
    else:
        k = np.argmax(v_sorted[idx] > threshold)
    tau = threshold[k] if k >= 0 else threshold[-1]
    
    # Reconstruct
    x = np.maximum(np.minimum(v - tau, 1), 0)
    # Correct sum if needed due to projection
    x_sum = np.sum(x)
    scale_factor = w / x_sum if x_sum > 0 else 1
    return x * scale_factor

def fourier_fit(h_vec, M):
    """Fit h_vec using a truncated Fourier series"""
    n = len(h_vec)
    # Create design matrix for cosine/sine basis
    idx = np.arange(n)
    # We use cos(pi*m*i/n) and sin(pi*m*i/n)
    # But we add a constant term, so we have M+1 terms: 1, cos, sin, cos2, sin2, ...
    A = np.ones((n, 2*M+1))
    A[:, 0] = 1.0  # constant term
    for m in range(1, M+1):
        A[:, 2*m-1] = np.cos(np.pi * m * idx / n)
        A[:, 2*m] = np.sin(np.pi * m * idx / n)
    
    # Fit coefficients (rectified to [0,1])
    # Solve Ax = b for least-squares subject to box constraint
    def objective(x):
        return np.linalg.norm(A @ x - h_vec)**2
    
    res = minimize(objective, np.zeros(2*M+1), method='L-BFGS-B', bounds=[(-np.inf, np.inf)]*(2*M+1))
    return res.x

def evaluate_fourier(x_coefficients, n):
    """Evaluate the Fourier reconstruction and compute C5"""
    # cos(PI * m * i / n)  and sin(pi*m*i/n)
    # reconstruct h from coeffs
    M = (len(x_coefficients) - 1) // 2
    idx = np.arange(n)
    h = x_coefficients[0] + sum(x_coefficients[2*m-1] * np.cos(np.pi * m * idx / n) + 
                                x_coefficients[2*m] * np.sin(np.pi * m * idx / n)
                                for m in range(1, M+1))
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # normalize to integral equal n/2
    h = h * (n/2 / np.sum(h))
    return evaluate_h(h)

def optimized_objective(x_coefficients, n):
    """Smooth surrogate objective"""
    M = (len(x_coefficients) - 1) // 2
    idx = np.arange(n)
    h = x_coefficients[0] + sum(x_coefficients[2*m-1] * np.cos(np.pi * m * idx / n) + 
                                x_coefficients[2*m] * np.sin(np.pi * m * idx / n)
                                for m in range(1, M+1))
    h = np.clip(h, 0, 1)
    h = h * (n/2 / np.sum(h))
    c5 = evaluate_h(h)
    return c5

                    h_final = h_final * (best_n/2 / np.sum(h_final))
