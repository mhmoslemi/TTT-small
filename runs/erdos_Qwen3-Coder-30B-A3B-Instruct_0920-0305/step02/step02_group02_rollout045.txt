import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def project_to_capped_simplex(v, w):
    """Project vector v onto the capped simplex: 0 <= x_i <= 1, sum x_i = w"""
    n = len(v)
    if np.abs(np.sum(v) - w) < 1e-12:
        return np.clip(v, 0, 1)
    
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find the threshold tau
    cumsum = np.cumsum(sorted_v)
    thresholds = (cumsum - w) / np.arange(1, n + 1)
    i = np.argmax(sorted_v >= thresholds)
    if i == 0 and sorted_v[0] >= thresholds[0]:
        tau = thresholds[0]
    else:
        tau = thresholds[i-1]
    projected = np.clip(sorted_v - tau, 0, 1)
    return projected

def compute_auto_correlation(h):
    """Compute auto-correlation S_k = sum h_i * h_{i+k} using FFT."""
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n)])
    H = fft(h_padded)
    S_full = ifft(H * np.conj(H)).real
    return S_full[n - 1 : 2 * n - 1]

def compute_all_r_k(h, w):
    """Compute for all shifts k: r_k = w - S_k"""
    S = compute_auto_correlation(h)
    return w - S

def compute_gradient_r_k(h, k, w):
    """Compute ∇r_k(h) = 1 - h_{i+k} - h_{i-k} for each i."""
    n = len(h)
    if k == 0:
        return np.ones_like(h) - 2 * h
    else:
        g = np.zeros_like(h)
        # h_{i+k}
        if k < n:
            g[:-k] += h[k:]
        # h_{i-k}
        if k > 0:
            g[k:] += h[:-k]
        return np.ones_like(h) - g

def evaluate_smooth_objective(h, alpha, w):
    """F_α(h) = (dx/α)*log( sum_k exp(α*r_k) )"""
    r = compute_all_r_k(h, w)
    r_max = np.max(r)
    if r_max > 50:  # Prevent overflow in exp
        r = r - r_max
    exp_terms = np.exp(alpha * r)
    return (1.0 / alpha) * np.log(np.sum(exp_terms))

def compute_smooth_gradient(h, alpha, w):
    """Compute ∇F_α(h) using soft-max weights."""
    r = compute_all_r_k(h, w)
    r_max = np.max(r)
    if r_max > 50:
        r = r - r_max
    exp_terms = np.exp(alpha * r)
    weights = exp_terms / np.sum(exp_terms)
    grad = np.zeros_like(h)
    for k in range(len(h)):
        g_k = compute_gradient_r_k(h, k, w)
        grad += weights[k] * g_k
    return grad

def accelerated_projected_gradient(h0, alpha_max, eps=1e-8, max_iter=1000):
    """FISTA-style update for smooth surrogate F_α"""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    w = np.sum(h)
    for i in range(max_iter):
        # compute current objective and gradient
        obj_old = evaluate_smooth_objective(h, alpha_max, w)
        grad = compute_smooth_gradient(h, alpha_max, w)

        # backtracking line search
        eta = 0.5
        for _ in range(10):
            h_guess = project_to_capped_simplex(y - eta * grad, w)
            obj_guess = evaluate_smooth_objective(h_guess, alpha_max, w)
            if obj_guess <= obj_old - 0.5 * eta * np.dot(grad, h_guess - h):
                break
            eta *= 0.5

        h_new = h_guess
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y_new = h_new + ((t - 1) / t_new) * (h_new - h)

        if np.abs(obj_old - evaluate_smooth_objective(h_new, alpha_max, w)) < eps:
            break

        h = h_new
        y = y_new
        t = t_new

    return h

def generate_initial_h(n, method='uniform', seed=None):
    """Generate initial h according to a specific method"""
    np.random.seed(seed)
    w = n / 2.0
    if method == 'uniform':
        h = np.random.uniform(0, 1, n)
    elif method == 'binary':
        h = np.zeros(n)
        h[:int(w)] = 1.0
        h = np.random.permutation(h)
        h += np.random.normal(0, 0.02, n)
    elif method == 'three_block':
        a, b = 0.5, 1.5
        h = np.zeros(n)
        a_idx = int(n * a / 2.0)
        b_idx = int(n * b / 2.0)
        h[:a_idx] = 1.0
        h[a_idx:b_idx] = 0.5
        # adjust to sum to w
        h = np.clip(h, 0, 1)
        scale = w / np.sum(h)
        h *= scale
    else:
        h = np.random.uniform(0.0, 1.0, n)

    h = project_to_capped_simplex(h, w)
    return h

def compute_c5_bound(h_values):
    """Compute C5 bound using standard method."""
    h = np.asarray(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / len(h_values)
    c5 = dx * np.max(corr)
    return c5

def run(seed=42, budget_s=600):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    min_c5 = np.inf
    best_h = None
    best_n = None
    # Try specific resolutions for performance
    candidates_n = [48, 56, 64, 80]
    time_per_n = budget_s / len(candidates_n)
    for n in candidates_n:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        w = n / 2.0
        global_best_h = None
        global_best_c5 = np.inf
        n_restart = 0
        remaining_time = time.time() - start_time
        restart_budget = min(30, max(1, int(budget_s / 500)))
        while n_restart < restart_budget and time.time() - start_time < 0.9 * budget_s:
            n_restart += 1
            methods = ['uniform', 'binary', 'three_block']
            method = methods[np.random.randint(0, len(methods))]
            h_start = generate_initial_h(n, method, seed + n_restart)

            h_opt = accelerated_projected_gradient(h_start, alpha_max=200, eps=1e-7)
            h_opt = project_to_capped_simplex(h_opt, w)
            c5 = compute_c5_bound(h_opt)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h_opt.copy()
        
        if global_best_c5 < min_c5:
            min_c5 = global_best_c5
            best_h = global_best_h
            best_n = n

    return list(best_h), min_c5, best_n
