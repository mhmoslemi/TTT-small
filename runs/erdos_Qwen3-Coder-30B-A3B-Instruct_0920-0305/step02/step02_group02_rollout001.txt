n_points = 60

def evaluate_c5(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def autocorr_h(h):
    """Compute all S_k = sum_i h_i * h_{i+k} for k in [0, n-1]."""
    n = len(h)
    # Use np.correlate with mode='full' and extract appropriate indices
    corr = np.correlate(h, h, mode='full')
    # By index mapping (negative lags to positive), [n-1] is lag 0, [n-1+k] is lag k for k>=0
    # So S_k = corr[n-1 + k]
    S_k = corr[n-1:]
    return S_k

def compute_gradient_S_k(h, k):
    """Compute gradient of S_k = sum_i h_i * h_{i+k} w.r.t h."""
    n = len(h)
    if k == 0:
        return 2 * h
    else:
        g = np.zeros_like(h)
        # Forward term
        if k < n:
            g[:n-k] += h[k:]
        # Backward term
        if k < n:
            g[k:] += h[:n-k]
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto the simplex with bounds [lower, upper] and sum = target_sum."""
    v = np.array(v)
    n = len(v)
    # Check bounds
    if target_sum < lower * n or target_sum > upper * n:
        raise ValueError("Target sum out of bounds for given constraints.")
    # Strategy: water-filling algorithm
    # Try to make v_i = min(max(v_i - lambda, lower), upper) sum to target
    # This is equivalent to first shifting v by lower, then computing
    # projections of v-lower to [0, upper-lower] simplex, then shifting back
    v_shifted = v - lower
    upper_shifted = upper - lower  # 1

    v_sorted = np.sort(v_shifted)[::-1]
    
    # Use binary search for lambda in [v_sorted[-1], v_sorted[0]]
    # But can be optimized using efficient method: compute lambda step by step
    
    # Water-filling approach: start from sorted list, process from large to small
    sum_so_far = 0
    for i in range(n):
        sum_so_far += v_sorted[i]
        # If we threshold v_sorted[i] at lambda, i.e., take min(v_sorted[i], lambda) better than max(sum_so_far, target_point)?
        # Calculate boundary based on number of remaining items
        # Of remaining i+1 items, for how many variables = min(v_sorted[i], threshold),  sum = (i+1) * threshold
        # We must solve (i+1) * threshold >= sum_so_far and threshold <= v_sorted[i], sum_so_far <= target_point 
        # Actually I will avoid complex handling and perform binary search
        # Use similar approach to Project onto unit simplex 

    # Okay, go better with direct binary search over lambda in [0, max(v_shifted) + 1]
    lower_lam, upper_lam = 0, np.max(v_shifted) + 1
    tol = 1e-9
    p = 0
    for _ in range(100):
        lam = (lower_lam + upper_lam) / 2
        # Compute sum of clamp(v_shifted[i] - lam, 0, upper_shifted)
        x = np.clip(v_shifted - lam, 0, upper_shifted)
        sum_x = np.sum(x)
        if abs(sum_x - target_point) < tol:
            p = lam
            break
        elif sum_x > target_point:
            # We over-threshold: need larger lambda
            lower_lam = lam
        else:
            upper_lam = lam
    else:
        # Not converged. Just use last bound-based value
        p = (lower_lam + upper_lam) / 2

    # Final check: compute projected x from this lambda
    x = np.clip(v_shifted - p, 0, upper_shifted)
    # Then shift back to original bounds
    x += lower
    return x

def gradient_ascent_step(h, alpha=0.01):
    """Perform a gradient ascent step to improve min S_k."""
    n = len(h)
    S_k = autocorr_h(h)
    # Compute r_k = sum h_i - S_k 
    # We want to minimize max_r_k = max_k r_k = max_k (w - S_k) = w - min_k S_k
    # So to maximize min S_k
    r_k = n/2 - S_k  # w = n/2
    # Note: r_k[0] = w - S_k[0] is r_0 = w - sum h_i^2
    # So we don't want to make k=0 active for gradient
        
    active_k = np.argmin(S_k[1:]) + 1  # exclude k=0
    # Compute gradient of S_k at active_k
    grad = compute_gradient_S_k(h, active_k)
    
    # Step: h -= alpha * grad; then project to [0,1] simplex with sum = n/2
    h_new = h - alpha * grad
    try:
        h_proj = project_onto_simplex_with_bounds(h_new, target_sum=n/2)
    except Exception as e:
        # fallback to simple projection

        # rescale to sum n/2
        s = np.sum(h_proj)
        h_proj = h_proj * (n/2) / s
    return h_proj
