import numpy as np

# Constants
n_points = 48  # Even number
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h, target_sum=1.0):
    """Project h to the feasible set: 0 <= h <= 1 and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) > 1e-10:
        h *= (target_sum / current_sum)
    return h

def build_linear_constraints(h):
    """Build the linearized constraints for the current h."""
    n = len(h)
    A = np.zeros((n, n))
    b = np.zeros(n)
    for s in range(n):
        # Compute a_s and b_s coefficients
        a_s = np.zeros(n)
        for i in range(n):
            # If i + s is out of bounds, it contributes 0
            if i + s < n:
                a_s[i] = 1 - h[i + s] - (h[i - s] if i - s >= 0 else 0)
            else:
                a_s[i] = 1 - (h[i - s] if i - s >= 0 else 0)
        # Compute b_s
        b_s = -np.sum(h[i] * h[i + s] for i in range(n - s))
        A[s, :] = a_s
        b[s] = b_s
    return A, b

def run_scp_iteration(h_current):
    """Run one iteration of the sequential convex programming approach."""
    n = len(h_current)
    # Build linearized constraints
    A, b = build_linear_constraints(h_current)
    # Create variables for h and t
    h_vars = Variable(n)
    t_var = Variable(1)
    # Objective: minimize t
    objective = Minimize(t_var)
    # Constraints: A @ h + b <= t, 0 <= h <= 1, sum(h) = 1/dx
    constraints = [A @ h_vars + b <= t_var,
                   h_vars >= 0,
                   h_vars <= 1,
                   sum_entries(h_vars) == n / 2]
    # Solve the LP problem using ECOS
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_new = np.array(h_vars.value).flatten()
        t_new = float(t_var.value)
        h_new = project_to_feasible(h_new)
        return h_new, t_new
    except:
        return h_current, evaluate_h(h_current)

def run(seed=42, budget_s=600):
    """Main SCP optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    # Number of restarts and sweeps
    max_restarts = 30
    max_iter_per_restart = 200
    total_sweeps = 0
    
    for restart in range(max_restarts):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Try different strategies for generating start points
        strategies = ['binary', 'uniform', 'interpolated']
        h_start = generate_starting_point(np.random.choice(strategies))
        
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current)
        
        # SCP iteration

            if c5_new < best_c5:

                break
            h_current = h_new.copy()
            c5_current = c5_new
            total_sweeps += 1
