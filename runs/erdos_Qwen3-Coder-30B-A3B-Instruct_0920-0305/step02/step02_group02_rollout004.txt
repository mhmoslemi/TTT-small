from scipy.fft import fft, ifft
import time
import random

# Constants
n_candidates = [48, 56, 64, 80]
target_bound = 0.3807

alpha_max = 500
eps = 1e-6

def compute_correlation(h):
    """ FFT-based correlation computation """
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n)])
    h_fft = fft(h_padded)
    corr_full = ifft(h_fft * np.conj(h_fft)).real
    return corr_full[n-1 : 2*n-1]

def compute_r(h, w, dx, corr):
    """ r_k = w - S_k where S_k is the auto-correlation """
    return w - corr

def compute_r_grad(k, h, n):
    """ Gradient of r_k w.r.t. h, special case handling for k=0 """
    grad = np.zeros(n)
    if k == 0:
        grad = np.ones(n) - 2 * h
    else:
        # For k > 0, we subtract h_{i+k} and h_{i-k}
        grad[:n-k] -= h[k:]
        if k < n: # Only do this if it doesn't lead to negative indices
            grad[k:] -= h[:n-k]
    return grad

def project_simplex(v, w):
    """ Project v onto capped simplex with sum = w, 0 <= v[i] <= 1 """
    n = len(v)
    v = np.array(v)
    # Sort in decreasing order
    sorted_v = np.sort(v)[::-1]
    # Find amount to subtract
    cumsum = np.cumsum(sorted_v) 
    idx = np.arange(n) + 1
    tmp = (cumsum - w) / idx
    tau = tmp[np.argmax(sorted_v >= tmp)] if not np.all(sorted_v < tmp) else sorted_v[-1]

    # Apply projection
    projected = np.clip(v - tau, 0, 1)
    # Adjust to ensure sum = w
    diff = w - np.sum(projected)
    if abs(diff) > 1e-10:
        idx = np.argsort(projected)
        for i in range(len(projected)):
            j = idx[i]
            if diff > 0 and projected[j] < 1:
                delta = min(diff, 1 - projected[j])
                projected[j] += delta
                diff -= delta
            if diff <= 0:
                break

    return projected

def compute_logsumexp_grad(h, w, dx, corr, alpha):
    """ Compute gradient of log-sum-exp of (r_k) with weights """
    n = len(h)
    r = compute_r(h, w, dx, corr)
    # Apply log-sum-exp trick to avoid overflow
    max_r = np.max(r)
    shifted_r = r - max_r
    exp_r = np.exp(alpha * shifted_r)
    Z = np.sum(exp_r)
    weights = exp_r / Z

    grad = np.zeros(n)
    for k in range(n):
        g = compute_r_grad(k, h, n)
        # Only compute subgradient if we need it; here, just elements
        grad += weights[k] * g

    return grad

def accelerated_projected_gradient(h0, w, dx, n, alpha, max_iter=500):
    h = h0.copy()
    y = h0.copy()
    t_prev = 1
    for _ in range(max_iter):
        corr = compute_correlation(h)
        r = compute_r(h, w, dx, corr)

        if np.max(r) < eps:
            break

        # Compute gradient of log-sum-exp (surrogate)
        g = compute_logsumexp_grad(h, w, dx, corr, alpha)

        # Line search
        eta = 0.5
        while True:
            h_candidate = project_simplex(y - eta * g, w)
            corr_candidate = compute_correlation(h_candidate)
            r_candidate = compute_r(h_candidate, w, dx, corr_candidate)

            if np.max(r_candidate) + eps > np.max(r) or eta < 1e-10:
                h = h_candidate
                break
            eta /= 2
        
        t_next = (1 + np.sqrt(1 + 4 * t_prev * t_prev)) / 2
        y = h + ((t_prev - 1)/t_next) * (h - h0)
        h0 = h.copy()
        t_prev = t_next
        
        # Early stopping based on small change
        if np.linalg.norm(h - h0) < eps:
            break

    return h

def initialize_h(n, w, method='binary'):
    """ Initialize h according to different strategies """
    h = np.zeros(n)
    if method == 'uniform':
        h = np.random.rand(n)
        h = project_simplex(h, w)
    elif method == 'binary':
        # Create balanced 0/1 vector then tweak with small noise
        ones = np.ones(int(w))
        zeros = np.zeros(n - int(w))
        h = np.concatenate([ones, zeros])
        np.random.shuffle(h)
        h += np.random.normal(0, 0.02, n)  # Small Gaussian perturbation
        h = project_simplex(h, w)
    elif method == 'three_block':
        # Set three regions: [0, a), [a,b), [b, 2]
        a = np.random.rand() * 0.5
        b = a + np.random.rand() * 0.5
        p = (w - a*n - (2-b)*n) / ((b-a)*n) if (b-a)*n > 0 else 0.5 # Just a_placeholder for robustness
        if (b-a)*n > 0:
            # Adjust p to meet the sum requirement
            total = a*n + (b-a)*n*p + (2-b)*n
            if np.isclose(total, w, atol=1e-2):
                h = np.full(n, 1) if a == 0 else np.concatenate([np.zeros(int(a*n)), np.full(int((b-a)*n), p), np.zeros(int((2-b)*n))])
                h = project_simplex(h, w)
    else:
        # fallback uniform
        h = np.random.rand(n)
        h = project_simplex(h, w)

    return h
