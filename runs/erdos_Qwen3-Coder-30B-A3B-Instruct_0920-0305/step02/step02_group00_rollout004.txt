import numpy as np
from scipy.optimize import minimize
import time

def build_linear_constraints(h):
    """Build linear constraints for all shifts."""
    n = len(h)
    # Precompute shift indices to avoid expensive indexing in loop
    a_s = np.zeros((n, n))
    b_s = np.zeros(n)
    
    for s in range(n):
        # Compute O_s(h) = sum(h_i * h_{i+s})
        # For each i, compute coefficients for the linearisation
        # L_s(h) = sum h_i - sum h_i * h_{i+s}
        # Linearised: sum h_i - (h_i^(k) h_{i+s} + h_{i+s}^(k) h_i - h_i^(k) h_{i+s}^(k))
        # Simplifies to: sum h_i - h_i^(k)*h_{i+s} - h_{i+s}^(k)*h_i + h_i^(k)h_{i+s}^(k)
        # = sum h_i + h_i^(k)h_{i+s}^(k) - h_i^(k)*h_{i+s} - h_{i+s}^(k)*h_i
        # = sum h_i + h_i^(k)h_{i+s}^(k) - (h_i^(k) h_{i+s} + h_{i+s}^(k) h_i)
        
        # We want to factor out h_i terms:
        # a_s[i] = 1 - h[i+s]^(k) - h[i-s]^(k)  [boundary handling is key]
        # but we note that we need to construct: 1 - h_{i+s}^{(k)} - h_{i-s}^{(k)} for each i
        # For boundary: i+s beyond range, set to 0; same for i-s.
        # But multiplying by h_i, for each shift, we only loop i from 0 to n-1-s
        
        a_i = np.zeros(n)
        b_s[s] = 0.0
        for i in range(n - s):
            a_i[i] += 1.0
            a_i[i] -= h[i+s]  # h_{i+s}^{k} term
            a_i[i+s] -= h[i]  # h_i^{k} term
            b_s[s] += h[i] * h[i+s]
        a_s[s] = a_i
        b_s[s] = -b_s[s]
    
    return a_s, b_s

def solve_slp(a_s, b_s, n, integral_constraint):
    """Solve the linear program for given constraints."""
    # Variables
    h = cp.Variable(n)
    t = cp.Variable(1)
    
    # Constraints
    constraints = []
    for s in range(n):
        constraints.append(a_s[s] @ h + b_s[s] <= t)
    constraints.append(cp.sum(h) == integral_constraint)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Solve
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False, maximumSeconds=1.0)
    
    if prob.status == cp.OPTIMAL:
        return h.value, t.value
    else:
        return None, None

def project_to_constraint(h, integral_constraint):
    """Project h onto the constraint sum(h) = integral_constraint."""
    h_clipped = np.clip(h, 0, 1)
    
    def bisection_search(lam):
        h_proj = np.clip(h_clipped - lam, 0, 1)
        return np.sum(h_proj) - integral_constraint
    
    # Binary search for appropriate lambda
    left, right = -1.0, 1.0
    while right - left > 1e-10:
        mid = (left + right) / 2
        val = bisection_search(mid)
        if val < 0:
            right = mid
        else:
            left = mid
    
    lam = (left + right) / 2
    return np.clip(h_clipped - lam, 0, 1)

def evaluate_c5(h):
    """Evaluate the actual C_5 objective."""
    n = len(h)
    dx = 2.0 / n
    correlations = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations)
    return max_corr * dx
