import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Constants
target_bound = 0.3807

def evaluate_c5(h):
    """Exact evaluation of C5 using correlation."""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def compute_correlations(h):
    """Compute all S_k using FFT."""
    n = len(h)
    dx = 2.0 / n
    w = n / 2
    
    # FFT-based auto-correlation
    H = fft(h, 2 * n)
    S_full = ifft(H * np.conj(H)).real
    S = S_full[n-1 : 2*n-1]
    
    # r_k = w - S_k
    r = w - S
    
    return r

def gradient_rk(h, k, w):
    """Gradient of r_k = w - S_k with respect to h_j."""
    n = len(h)
    grad = np.zeros(n)
    
    # d/dh_j r_k = d/dh_j (w - h_j * h_{j+k})
    # For j+k < n: grad[j] -= h_{j+k}
    # For j-k >= 0: grad[j] -= h_{j-k}
    if k == 0:
        grad = np.ones(n) - 2 * h
    else:
        # j+k case
        if k < n:
            grad[:-k] -= h[k:]
        # j-k case
        if k < n:
            grad[k:] -= h[:-k]
    return grad

def projected_gradient_descent(h_init, r_seq, alpha, max_iter=500):
    """Projected gradient descent with acceleration."""
    n = len(h_init)
    w = n / 2
    h = h_init.copy()
    
    # Line search L
    L = 2.0  # Fixed upper bound on step size
    t = 1.0
    y = h.copy()
    
    for it in range(max_iter):
        # Compute current r_k values
        r = r_seq(h)
        r_scaled = r * alpha
        r_scaled -= np.max(r_scaled)  # numerical stability
        
        # Softmax weights
        exp_r = np.exp(r_scaled)
        wts = exp_r / np.sum(exp_r)
        g = np.zeros(n)
        
        for k in range(n):
            g += wts[k] * gradient_rk(h, k, w)
        
        # Line search
        eta = L
        h_new = y - eta * g
        
        # Project onto capped simplex
        h_new = project_simplex(h_new, w)
        
        # Adaptive step  
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y = h_new + ((t - 1) / t_new) * (h_new - h)
        h = h_new
        t = t_new
        
        # Check stopping criterion
        if it > 10:
            F_prev = np.sum(wts * r_scaled) - np.log(np.sum(exp_r)) / alpha
            F_curr = np.sum(wts * (r * alpha - np.max(r * alpha))) - np.log(np.sum(exp_r)) / alpha
            if abs(F_curr - F_prev) / (abs(F_prev) + 1e-12) < epsilon:
                return h
                
    return h

def project_simplex(v, w):
    """Project vector v onto capped simplex with sum = w."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    z = np.cumsum(v_sorted)
    
    # Find tau
    for i in range(n):
        if v_sorted[i] + (w - z[i]) / (i + 1) >= 0:
            tau = (w - z[i]) / (i + 1)
            break
    else:
        # All entries would be zero -> zero vector
        tau = w
    
    # Compute projection
    x = np.clip(v - tau, 0, 1)
    # Correct the sum
    extra = w - np.sum(x)
    x[0] += extra
    return x

def smooth_optimization(h0, n, alpha_start=1, alpha_end=500, step_factor=2):
    """Run smooth continuation optimization."""
    h = h0.copy()
    alpha = alpha_start
    
    while alpha <= alpha_end:
        # Recompute r values for this alpha
        def r_seq(h_vec):
            return compute_correlations(h_vec)
        
        h = projected_gradient_descent(h, r_seq, alpha, 300)
        alpha *= step_factor
        
    return h

def run(seed=42, budget_s=600):
    """Main optimization loop."""
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_values:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        dx = 2.0 / n
        w = n / 2
        
        for ridx in range(max_restarts):
            if time.time() - start_time > 0.99 * budget_s:
                break
                
            h0 = initialize_h(n, ridx)
            h_final = smooth_optimization(h0, n)
            c5 = evaluate_c5(h_final)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
                best_n = n
            
            # Early stopping if target achieved
            if best_c5 <= target_bound:

            break

        final_c5 = evaluate_c5(best_h)

        # Fallback to initial values

        c5 = evaluate_c5(final_h)
