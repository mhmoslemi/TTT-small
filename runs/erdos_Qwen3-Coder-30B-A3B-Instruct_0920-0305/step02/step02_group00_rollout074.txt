import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp
from cvxpy import Minimize, Variable, Problem, sum, norm
import math

# Parameters
n_values = [48, 56, 64]
target_bound = 0.3807
n_points = 48  # default, will be adjusted in loop

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h, target_sum):
    """Project h onto the feasible set: 0 ≤ h ≤ 1, sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-6:
        return h
    # Use bisection to find lambda for projection
    lo, hi = -1.0, 1.0
    for _ in range(20):
        mid = (lo + hi) / 2
        h_test = np.clip(h - mid, 0, 1)
        if np.sum(h_test) > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h - lo, 0, 1)
    return h_proj * (target_sum / np.sum(h_proj))

def build_linear_constraints(h_current):
    """Build the linear approximation coefficients for each shift"""
    n = len(h_current)
    # Precompute shift maps, treating out-of-range indices as 0
    # a_s[i] = 1 - h_{i+s}^{(k)} - h_{i-s}^{(k)} if i-s, i+s exist else 0
    a_s_list = []
    b_s_list = []
    for s in range(n):
        # build a_s vector where index 0 ≤ i ≤ n-1
        a_s = np.zeros(n)
        b_s = 0.0
        # For each valid i, update a_s[i] and add to b_s
        for i in range(n - s):  # i ranges from 0 to n-1-s
            # if i-s < 0 or i+s >= n, treat corresponding h terms as 0
            term1 = h_current[i + s] if (i + s) < n else 0 .
            term2 = h_current[i - s] if (i - s) >= 0 else 0 .
            a_s[i] = 1.0 - term2 - term1
            b_s -= h_current[i] * h_current[i + s]
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_slp_lp(a_s_list, b_s_list, n):
    """Solve the sequential linear programming relaxation"""
    # cvxpy variables
    h = Variable(n)
    t = Variable(1)
    # Constraints
    constraints = [0 <= h, h <= 1]
    # Add the linear constraints for each shift
    for i in range(n):
        constraints.append(a_s_list[i].T @ h + b_s_list[i] <= t)
    # Heuristic: use integral constraint as a linear constraint on h
    constraints.append(sum(h) == n / 2)
    # Objective
    objective = Minimize(t)
    # Create and solve the problem
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_val = h.value
            t_val = t.value
            return h_val, t_val
        else:
            # fallback to scipy linprog
            # Build manually as np arrays
            # Need to solve:
            # minimize c^T x
            # subject to A_ub @ x <= b_ub
            #         A_eq @ x == b_eq
            #         lb <= x <= ub
            # In our case:
            # x = [h[0], ..., h[n-1], t]
            # c = [0, ..., 0, 1] (minimize t)
            # A_ub = [ a_s^T -1 ] [ -1 0 ] etc.
            # b_ub = [ b_s ]
            # A_eq = [1,1,...,1] [0] [0,0]
            # b_eq = [ n/2 ]
            # lb = [0]*n + [-inf], ub = [1]*n + [inf]
            # But let us observe the constraints
            # For each shift s:
            #  a_s.T @ h + b_s <= t
            #  or:   -t + a_s.T @ h <= -b_s
            # we'll solve using linprog
            # m = n + 1 variables
            # objective vector
            c = np.zeros(n + 1)
            c[-1] = 1  # minimize last (t)
            # A_ub comes from -t + a_s.T @ h <= -b_s
            #              i.e., a_s^T @ h - t <= -b_s
            matrix_rows = [
                np.append(a_s_list[i], -1.0) for i in range(n)
            ]
            # Each row is [a_s_list[i]._T, -1]
            A_ub = np.vstack(matrix_rows)  # shape (n, n+1)
            b_ub = np.array([-b_s_list[i] for i in range(n)])  # shape (n,)
            # Equality constraint
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0  # sum h_i  = n/2
            b_eq = np.array([n / 2])
            # bounds
            bounds = [tuple([0, 1]) for _ in range(n)] + [(None, None)] # added last for t
            # solve
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if result.success:
                return result.x[:-1], result.x[-1]
            else:
                # fallback
                return np.full(n, 0.5), float('inf')
    except:
        # Try to fallback to linprog
        c = np.zeros(n + 1)
        c[-1] = 1  # minimize t
        # matrix_rows = [list(a_s) + [-1] for a_s in a_s_list]  # we'll wrap in np.array later
        # A_ub = np.vstack(matrix_rows)  # Shape (n, n+1)
        # b_ub = np.array([-b for b in b_s_list])
        # A_eq = np.array([[1] * n + [0]])
        # b_eq = np.array([n / 2])
        # bounds = [tuple([0, 1]) for i in range(n)] + [(None, None)]
        # result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        # if result.success:
        #     return result.x[:-1], result.x[-1]
        # Else return simple case
        return np.full(n, 0.5), float('inf')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    # Try different n values
    for n in n_values:
        dx = 2.0 / n
        max_iter_per_start = 200
        tol = 1e-6
        # Local function to evaluate current score
        def eval_h(h):
            return evaluate_h(h)
        # Local function to generate seed points
        def generate_seed_points(n, num_seeds=8):
            seeds = []
            # Binary seed
            h_bin = np.zeros(n)
            h_bin[:n//2] = 1.
            np.random.shuffle(h_bin)
            seeds.append(h_bin.copy())
            # Uniform seed (projected)
            h_uni = np.random.rand(n)
            h_uni = project_to_feasible(h_uni, n/2)
            seeds.append(h_uni.copy())
            # Interpolated seed (from initial_h_values if provided)
            if 'initial_h_values' in globals():
                init_vals = np.array(initial_h_values)
                if len(init_vals) >= n:  # use first n points
                    h_interp = init_vals[:n]
                    h_interp = project_to_feasible(h_interp, n/2)
                    seeds.append(h_interp.copy())
                else:
                    # downsample the initial one
                    indices = np.linspace(0, len(init_vals), n, dtype=int, endpoint=False)
                    h_interp = init_vals[indices]
                    h_interp = project_to_feasible(h_interp, n/2)
                    seeds.append(h_interp.copy())
            # Perturbed best seed if present
            # This is a bit clunky because we're running a bunch of tests with multistart
            # For simplicity here, skip simple perturbation of best. We use restarts via loops.
            return seeds
        # Now, the main loop
        remaining_time = budget_s * 0.95 - (time.time() - start_time)
        # We don't go above 1500 starts if time allows
        restarts = min(1500, max(10, int(remaining_time / 0.1)))
        restarts = min(restarts, 500)  # prevent overuse
        if restarts <= 0:
            continue
        for restart in range(restarts):
            if time.time() - start_time > budget_s * 0.9:
                break
            # Generate seed
            seeds = generate_seed_points(n, num_seeds=5)
            for seed_h in seeds:
                h = seed_h.copy()
                last_obj = float('inf')
                for k in range(max_iter_per_start):
                    # Compute "correlations"
                    o_s = np.correlate(h, h, mode="full")[:n]
                    # Build linear constraints
                    a_s_list, b_s_list = build_linear_constraints(h)
                    # Solve LP subproblem
                    h_new, t_new = solve_slp_lp(a_s_list, b_s_list, n)
                    # Project onto feasible set
                    h_new = project_to_feasible(h_new, n / 2)
                    # Evaluate true objective
                    corr = np.correlate(h_new, 1 - h_new, mode="full")
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        if best_c5 < target_bound:
                            return list(best_h), best_c5, n
                    if abs(c5 - last_obj) < tol:
                        break
                    last_obj = c5
                    h = h_new
                # If "improved" cutoff every now and then
                if time.time() - start_time > budget_s * 0.9:
                    break
    # Final reoptimization with SLSQP if we have balance
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        bounds = [(0, 1) for _ in range(len(best_h))]
        def obj_to_minimize(h):
            return evaluate_h(h)
        # Use one last refinement step
        try:
            from scipy.optimize import minimize
            result = minimize(obj_to_minimize, best_h, method='SLSQP',
                              bounds=bounds,
                              constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2},
                              options={'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                refactor_c5 = evaluate_h(result.x)
                if refactor_c5 < best_c5:
                    best_h = result.x
                    best_c5 = refactor_c5
        except:
            pass
    if best_h is None:
        # Return the default from previous run if no better was found
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_h(best_h)
    return list(best_h), best_c5, n_points
