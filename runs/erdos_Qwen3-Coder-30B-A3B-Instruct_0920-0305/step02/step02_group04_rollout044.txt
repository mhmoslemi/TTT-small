import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def evaluate_h(h):
    """Evaluate the C5 constant for a given h via FFT."""
    n = len(h)
    dx = 2.0 / n
    H = rfft(h, 2*n)
    G = rfft(1.0 - h, 2*n)
    conv = irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit):
    """
    Solve the binary MILP formulation.
    Returns (T_opt, h_opt) or (None, None) if timeout.
    """
    M = n // 2
    try:
        # Variables
        h = cp.Variable(n, boolean=True)
        T = cp.Variable(integer=True, nonneg=True)
        
        # Product variables for each shift
        p = []
        constraints = [cp.sum(h) == M, T >= 0]
        
        for ell in range(1, n):
            length = n - ell
            p_ell = cp.Variable(length, boolean=True)
            p.append(p_ell)
            
            # Product linearisation (McCormick envelopes)
            constraints += [
                p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1
            ]
            
            # Worst overlap constraint
            constraints += [cp.sum(p_ell) + T >= M]
        
        # Objective
        problem = cp.Problem(cp.Minimize(T), constraints)
        
        # Solve with CBC and time limit
        solver_options = {'maxSeconds': time_limit * 0.9}
        problem.solve(solver=cp.CBC, verbose=False, **solver_options)
        
        if problem.status in ['optimal', 'optimal_inaccurate']:
            h_opt = np.ravel(h.value)
            T_opt = int(T.value)
            return T_opt, h_opt
        elif problem.status == 'infeasible':
            return None, None
        else:  # likely 'timed_out', 'user_cancelled', or 'unknown'
            # Try to return best feasible (from .value)
            try:
                h_opt = np.ravel(h.value)
                # Compute the actual T for this solution
                T_val = 0
                for ell in range(1, n):
                    length = n - ell
                    if ell < len(p):
                        p_ell_val = np.ravel(p[ell-1].value)
                        overlap = np.sum(p_ell_val)
                        # T is such that overlap + T >= M
                        T_val = max(T_val, M - overlap)
                return int(T_val), h_opt
            except:
                return None, None
                
    except Exception as e:
        return None, None

def anneal_binary(h0, max_iter, time_limit):
    """Annealing on binary vectors preserving mass."""
    n = len(h0)
    M = n // 2
    dx = 2.0 / n
    
    np.random.seed(int(time.time()) % 1000000)  # for reproducibility
    h = np.array(h0, dtype=float)
    current_score = evaluate_h(h)
    best_h, best_score = h.copy(), current_score
    
    start_time = time.time()
    
    # Adapt temperature
    temp = 1.0
    n_swaps = 0
    sum_delta = 0.0
    n_updates = 0
    
    # Initial sampling of temperature
    for _ in range(500):
        if time.time() - start_time > time_limit * 0.8:
            break
        # Pick two indices: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Temporarily flip
        h[i] = 0
        h[j] = 1
        new_score = evaluate_h(h)
        delta = new_score - current_score
        n_swaps += 1
        sum_delta += delta
        n_updates += 1
        # Restore
        h[i] = 1
        h[j] = 0
    
    if n_updates > 20:
        avg_delta = sum_delta / n_updates
        temp = max(1e-6, avg_delta * 2.0)
    
    # Actual annealing
    temp = temp
    temp_decay = 0.9995
    T_min = 1e-12
    
    for i_iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # Pick two indices: one 1 and one 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Temporarily flip
        h[i] = 0
        h[j] = 1
        new_score = evaluate_h(h)
        delta = new_score - current_score
        
        # Acceptance criterion
        if delta <= 0:  # Better or equal
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
        else:
            if np.random.rand() < np.exp(-delta / temp):
                current_score = new_score
            else:
                # Revert the change
                h[i] = 1
                h[j] = 0
                continue  # without changing temp, as not accepted
        
        # Update temperature
        temp = max(T_min, temp * temp_decay)
    
    # Ensure mass constraint
    mass = np.sum(h) * dx
    if abs(mass - 1.0) > 1e-6:
        scale = 1.0 / mass
        h *= scale
    best_h = np.clip(best_h, 0.0, 1.0)
    # Correct mass arbitrarily near by scaling
    mass_best = np.sum(best_h) * dx
    if abs(mass_best - 1.0) > 1e-6:
        scale = 1.0 / mass_best
        best_h *= scale
    
    return best_h, best_score

def smooth_descent(h0, time_limit):
    """Gradient descent on smooth surrogate objective."""
    n = len(h0)
    dx = 2.0 / n
    M = n // 2
    
    # Find the discrete domains in expedient way
    h = np.array(h0, dtype=float)
    beta_schedule = [1.0, 2.0, 5.0, 10.0, 20.0]
    step = 0.1
    
    start_time = time.time()
    
    # Water-filling projection onto sum=h*0.5 (mass)
    def project_sum(h, target_sum):
        # (value, index)
        pairs = [(x, i) for i, x in enumerate(h)]
        pairs.sort()
        # Determine lambda
        def sum_at_lambda(lam):
            total = 0
            for val, i in pairs:
                if val >= lam:
                    total += val
                elif val <= lam:
                    total += lam
                else:
                    total += lam
            return total
        # Binary search on lambda
        low, high = 0.0, 1.0
        for _ in range(30):
            mid = (low + high) / 2.0
            if sum_at_lambda(mid) >= target_sum:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2.0
        for i in range(n):
            h[i] = max(0.0, min(1.0, h[i] + lam))
        # Final normalisation
        total = np.sum(h)
        if total > 0:
            h *= (target_sum / total)
        return h
    
    for beta in beta_schedule:
        if time.time() - start_time > time_limit * 0.9:
            break
        for iter in range(200):  # max iters per beta
            if time.time() - start_time > time_limit:
                break
            # Compute all L_ell via FFT
            L_vals = []
            for ell in range(1, n):
                H = rfft(h, 2*n)
                G = rfft(1.0 - h, 2*n)
                conv = irfft(H * np.conj(G))[:2*n-1]
                L_val = conv[ell-1] * dx  # L_ell for shift ell
                L_vals.append(L_val)
            
            if len(L_vals) == 0:
                break
            
            # Compute weights
            L_arr = np.array(L_vals)
            # Avoid overflow in exp
            L_arr = L_arr - np.max(L_arr)  # stabilize
            exp_L = np.exp(beta * L_arr)
            weights = exp_L / np.sum(exp_L)
            
            # Compute weighted gradient
            # It's ∇(sum w_ell * (1 - h_{i+ell} - h_{i-ell})) 
            # But to prevent O(n^2) eval, we compute on full FFT product.
            # Let's compute it per-scroll using circulant structure
            
            grad = np.zeros(n)
            for ell, w in enumerate(weights):
                ell_idx = ell + 1
                # gradient for i-th position
                # = w * (1 - h_{i+ell} - h_{i-ell})  [in circular boundary]
                # = w * (1 - h[np.mod(i+ell,n)] - h[np.mod(i-ell,n)])
                # = w - w*h[i+ell] - w*h[i-ell]
                for i in range(n):
                    grad[i] += w
                    grad[i] -= w * h[np.mod(i+ell_idx,n)]
                    grad[i] -= w * h[np.mod(i-ell_idx,n)]
            
            # Take gradient step
            h_new = h - step * grad
            
            # Project back to feasible set
            h_new = np.clip(h_new, 0.0, 1.0)
            # Ensure sum equals M * dx (i.e. 1.0)
            h_new = project_sum(h_new, M * dx)
            h = h_new
            
            # Optional: early stopping if not improving
            if iter % 10 == 0:
                new_score = evaluate_h(h)
                if new_score < 0.001:
                    break
    
    return h, evaluate_h(h)
