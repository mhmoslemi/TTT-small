import numpy as np

def project_to_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: 0 ≤ x_i ≤ 1, sum x_i = target_sum."""
    n = len(y)
    y = np.array(y)
    def F(lam):
        return np.sum(np.clip(y - lam, 0, 1)) - target_sum
    low, high = -np.max(y) - 1, np.max(y) + 1
    for _ in range(40):
        mid = (low + high) / 2
        if F(mid) > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    x = np.clip(y - lam, 0, 1)
    return x

def compute_c5(h):
    """Evaluate C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def solve_sdp(n, max_time_per_sdp=30):
    """Solve SDP relaxation for a given n."""
    # Define the variables
    X = Semidef(n)
    h = Variable(n)
    m = Variable()
    
    # Constraints
    constraints = [X[i,i] == h[i] for i in range(n)]
    constraints += [0 <= h[i] for i in range(n)]
    constraints += [h[i] <= 1 for i in range(n)]
    constraints += [sum(h) == n / 2]
    
    # Autocorrelations
    for s in range(-(n-1), n):
        if s >= 0:
            r_s = sum(X[i, i+s] for i in range(n-s))
        else:
            r_s = sum(X[i+s, i] for i in range(-s, n))
        constraints += [r_s >= m]
        
    # Objective
    objective = Maximize(m)
    
    # Solve
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=SCS, max_iters=5000, eps=1e-5)
        X_val = X.value
        m_val = m.value
        
        if X_val is not None and m_val is not None:
            # Extract h from X
            eigvals, eigvecs = np.linalg.eigh(X_val)
            h_sol = np.abs(eigvecs[:, np.argmax(eigvals)])
            h_sol = h_sol / np.sum(h_sol) * (n / 2)
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = project_to_capped_simplex(h_sol, n/2)
            return m_val, h_sol
        return None, None
    except:
        return None, None

        # Project to capped simplex and refine
        h_refined = project_to_capped_simplex(h_interp, n_target / 2)

                           best_h)
        best_h = project_to_capped_simplex(h_fine, n_target / 2)
