import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_subgradient(h, m, dx):
    """Compute subgradient for lag m"""
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        for i in range(n - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        m = -m
        for i in range(m, n):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    return g * dx

def project_onto_feasible(h, target_sum):
    """Project h onto box constraints and sum constraint using water-filling"""
    w = np.clip(h, 0, 1)
    current_sum = np.sum(w)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return w
    
    # Water-filling algorithm
    while abs(diff) > 1e-10:
        if diff > 0:  # Need to increase sum
            # Find indices where w < 1
            free_indices = np.where(w < 1)[0]
            if len(free_indices) == 0:
                break
            distribute = diff / len(free_indices)
            for i in free_indices:
                w[i] = min(1, w[i] + distribute)
            diff = target_sum - np.sum(w)
        else:  # Need to decrease sum
            # Find indices where w > 0
            free_indices = np.where(w > 0)[0]
            if len(free_indices) == 0:
                break
            distribute = -diff / len(free_indices)
            for i in free_indices:
                w[i] = max(0, w[i] - distribute)
            diff = target_sum - np.sum(w)
    
    return w

def evaluate_h(h):

    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val
