def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) = n/2"""
    # Clip to [0,1]
    w = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target_sum = n_points / 2.0
    current_sum = np.sum(w)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-12:
        return w.copy()
    
    # Water-filling algorithm
    while abs(diff) > 1e-12:
        if diff > 0:  # Need to increase sum
            # Find indices with w_i < 1
            indices = np.where(w < 1)[0]
            if len(indices) == 0:
                break
            # Distribute diff equally among them
            delta = diff / len(indices)
            for i in indices:
                if w[i] + delta <= 1:
                    w[i] += delta
                else:
                    old_w = w[i]
                    w[i] = 1.0
                    diff -= (1.0 - old_w)
            diff = target_sum - np.sum(w)
        else:  # Need to decrease sum
            # Find indices with w_i > 0
            indices = np.where(w > 0)[0]
            if len(indices) == 0:
                break
            # Subtract diff equally among them
            delta = -diff / len(indices)
            for i in indices:
                if w[i] - delta >= 0:
                    w[i] -= delta
                else:
                    old_w = w[i]
                    w[i] = 0.0
                    diff += old_w
            diff = target_sum - np.sum(w)
            
    return w

def compute_subgradient(h, m, dx):
    """Compute subgradient of A_m w.r.t. h"""
    g = np.zeros_like(h)
    n = len(h)
    
    if m >= 0:
        # i from 0 to n-1-m
        for i in range(n - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        # m < 0 => swap roles
        m = -m
        for i in range(m, n):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    return g * dx

def avg_subgradient(h, A_vals, dx, n_points):
    """Compute average subgradient over all maximizing lags"""
    max_val = np.max(A_vals)
    max_indices = np.where(np.abs(A_vals - max_val) < 1e-10)[0]
    
    if len(max_indices) == 0:
        return np.zeros_like(h)
        
    # indices are 0 to 2*n_points - 2
    # we need to map them back to actual lag indices from -n+1 to n-1
    m_indices = max_indices - (n_points - 1)
    
    # compute average subgradient
    g_total = np.zeros_like(h)
    for m in m_indices:
        g_total += compute_subgradient(h, m, dx)
    return g_total / len(m_indices)

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val
