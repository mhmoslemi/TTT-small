import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum, max_iter=40):
    """Project h onto the capped simplex: 0 <= h_i <= 1, sum(h_i) = target_sum"""
    def F(lam):
        x = np.clip(h - lam, 0, 1)
        return np.sum(x) - target_sum
    
    a, b = -1000.0, 1000.0
    for _ in range(max_iter):
        lam = (a + b) / 2.0
        if F(lam) > 0:
            a = lam
        else:
            b = lam
    lam_star = (a + b) / 2.0
    return np.clip(h - lam_star, 0, 1)

def build_five_step_shape(a, b, n):
    """Build a symmetric five-step shape with parameters (a, b) and n samples."""
    dx = 2.0 / n
    x = np.arange(0.5, n + 0.5) * dx  # sample locations
    
    if b <= 0.5:
        return None  # Invalid
    
    # Compute u from the constraint
    if a >= b:
        return None  # Invalid
    u = (2 * b - 1) / (2 * (b - a))  # u = (2*b-1) / (2*(b-a))
    
    if u < 0 or u > 1:
        return None  # Invalid u
    
    h = np.zeros(n)
    # intervals: [0, a], [a, b], [b, 2-b], [2-b, 2-a], [2-a, 2]
    # corresponding heights: 0, u, 1, u, 0
    
    # Element-wise conditions
    mask1 = (x >= 0) & (x <= a)
    mask2 = (x > a) & (x <= b)
    mask3 = (x > b) & (x <= 2 - b)
    mask4 = (x > 2 - b) & (x <= 2 - a)
    mask5 = (x > 2 - a) & (x <= 2)
    
    h[mask1] = 0
    h[mask2] = u
    h[mask3] = 1
    h[mask4] = u
    h[mask5] = 0
    
    # Ensure that the integral is close to 1 (sum * dx = 1)
    integral = np.sum(h) * dx
    h = h * (1.0 / integral)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Step 1: Grid search on two key parameters
    # a in [0, 0.5], b in [0.5, 1]
    n_grid = 51
    a_grid = np.linspace(0, 0.5, n_grid)
    b_grid = np.linspace(0.5, 1, n_grid)
    
    # Slice: keep a < b, u must be in [0,1]
    for a in a_grid:
        for b in b_grid:
            if b <= a:
                continue
            h = build_five_step_shape(a, b, n_points)
            if h is None:
                continue
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    if best_h is None or (time.time() - start_time) > 0.8 * budget_s:
        return (best_h.tolist(), best_c5, n_points)
    
    # Step 2: Local refinement with smooth projected gradient descent
    def smooth_max(c, alpha):
        # Softmax approximation to max
        exp_c = np.exp(alpha * c)
        w = exp_c / np.sum(exp_c)
        return np.sum(w * c)
    
    def smooth_gradient(h, alpha, target_sum):
        # Compute smooth gradient with respect to Φ(h) = sum(w_i * (sum h_i - sum h_i h_{i+s}))
        c = correlate(h, 1 - h, mode="full")
        w = np.exp(alpha * c)
        w = w / np.sum(w)
        # Now compute gradient ∂Φ/∂h_i = 1 - sum_s w_s (h_{i+s} + h_{i-s})
        # We compute via convolution for efficiency
        h_padded = np.pad(h, (n_points-1, n_points-1), mode='constant')
        # w is length 2n-1, h_padded is length 3n-2, we want h_i stripped from center
        g_plus = correlate(w[::-1], h_padded, mode='full')[n_points-1:n_points-1+n_points]
        g_minus = correlate(w, h_padded, mode='full')[n_points-1:n_points-1+n_points]
        grad = 1.0 - (g_plus + g_minus)
        return grad
    
    h_current = best_h.copy()
    alpha = 0.5
    eta = 0.5
    step = 0
    patience_counter = 0
    best_h_local = h_current.copy()
    least_c5_local = evaluate_c5(h_current)
    
    while (time.time() - start_time) < 0.9 * budget_s:
        # Compute smooth gradient
        try:
            grad = smooth_gradient(h_current, alpha, n_points / 2)
            # Gradient step
            h_new = h_current - eta * grad
            # Project back onto feasible set
            h_projected = project_to_capped_simplex(h_new, n_points / 2)
            # Update h_current
            h_current = h_projected
            # Evaluate
            current_c5 = evaluate_c5(h_current)
            if current_c5 < least_c5_local:
                least_c5_local = current_c5
                best_h_local = h_current.copy()
                patience_counter = 0
            else:
                patience_counter += 1
                if patience_counter > 500:
                    break
            
            # Adjust learning rate and temperature
            step += 1
            if step % 200 == 0:
                eta *= 0.9
                alpha = min(alpha * 1.5, 20.0)
        except Exception as e:
            break
    
    best_h = best_h_local
    best_c5 = least_c5_local
    
    # Step 3: Small population DE as fallback if time permits
    if (time.time() - start_time) < 0.99 * budget_s:
        try:
            pop_size = 8
            generations = 300
            F = 0.6
            CR = 0.9
            
            def project_individual(indiv):
                return project_to_capped_simplex(indiv, n_points / 2)
            
            def eval_individual(indiv):
                c5 = evaluate_c5(indiv)
                return c5
            
            population = []
            for _ in range(pop_size):
                h = np.random.rand(n_points)
                h = project_individual(h)
                population.append(h)
            
            best_in_pop = min(population, key=lambda h: evaluate_c5(h))
            best_c5_pop = evaluate_c5(best_in_pop)
            
            for gen in range(generations):
                if (time.time() - start_time) > 0.99 * budget_s:
                    break
                new_pop = []
                for h in population:
                    # Selection: pick three distinct individuals
                    idxs = np.random.choice(pop_size, size=3, replace=False)
                    a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
                    # Mutation
                    v = a + F * (b - c)
                    # Crossover
                    trial = np.zeros_like(v)
                    mask = np.random.rand(n_points) < CR
                    trial = np.where(mask, v, h)
                    # Project to feasible set
                    trial = project_individual(trial)
                    # Selection
                    trial_c5 = evaluate_c5(trial)
                    h_c5 = evaluate_c5(h)
                    if trial_c5 < h_c5:
                        new_pop.append(trial)
                    else:
                        new_pop.append(h)
                population = new_pop
                # Update best
                current_best = min(population, key=lambda h: evaluate_c5(h))
                c5_best = evaluate_c5(current_best)
                if c5_best < best_c5_pop:
                    best_c5_pop = c5_best
                    best_in_pop = current_best.copy()
            
            if best_c5_pop < best_c5:
                best_h = best_in_pop
                best_c5 = best_c5_pop
        except:
            pass
    
    # Final check: ensure integral is exactly 1, and that h values are in [0,1]
    final_h = project_to_capped_simplex(best_h, n_points / 2)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
