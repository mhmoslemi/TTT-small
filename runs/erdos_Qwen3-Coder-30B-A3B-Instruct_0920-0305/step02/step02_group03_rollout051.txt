import numpy as np
from scipy.optimize import dual_annealing, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # Create bounds for optimization (h[i] in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        # Integral constraint: sum(h) * dx = 1
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # First, try local optimization around the best solution so far using SLSQP
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            h_candidate = res.x
            # Normalize to satisfy constraint
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Try local optimization with bounds via dual annealing for better global search
    try:
        da_result = dual_annealing(objective, bounds, maxiter=1000, seed=seed, 
                                   no_local_search=False, initial_temp=1e-2)
        if da_result.success:
            h_candidate = da_result.x
            # Normalize to satisfy constraint
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass

    # Try random restarts with better initializations
    max_restarts = 200
    restart_count = 0
    
    while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.9:
        # Use a mixture of random and good heuristic starting points
        if restart_count % 3 == 0:
            # Try a simple symmetric construction
            new_h = np.ones(n_points) * 0.5
            # Add some variation to avoid all-halves
            for i in range(n_points // 4, 3 * n_points // 4):
                new_h[i] = 0.7
        elif restart_count % 3 == 1:
            # Use some sigmoid-like shaping
            x = np.linspace(0, 2, n_points)
            new_h = 1 / (1 + np.exp(-10 * (x - 1)))
        else:
            # Pure random
            new_h = np.random.uniform(0, 1, n_points)
        
        # Project onto constraint
        dx = 2.0 / n_points
        scale_factor = 1.0 / (np.sum(new_h) * dx)
        new_h = new_h * scale_factor

        try:
            res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons,

                h_candidate = res.x
                # Normalize to satisfy constraint
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate)

            pass

    # Final refinement with local optimization

        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,

            h_candidate = res.x
            # Normalize to satisfy constraint
            dx = 2.0 / n_points
