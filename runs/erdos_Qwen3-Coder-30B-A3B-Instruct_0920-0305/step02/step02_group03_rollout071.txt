import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    
    # First, use integer programming approach for a smaller n to find a better pattern
    # Try n=72 (as it's a good middle ground for optimization, and 36 ones to meet constraint)
    n_small = 72
    dx_small = 2.0 / n_small
    
    # Use cvxpy to solve small ILP optimization for binary pattern
    # Variables for binary h values
    h_vars = cp.Variable(n_small, boolean=True)
    
    # Constraints: sum of h = 36 (half of n_small)
    constraints = [cp.sum(h_vars) == 36]
    
    # Compute correlation terms for shifts (n_small-1 shifts total)
    # This is a very simplified approach - we actually compute the correlation efficiently
    # but since we cannot directly model the full max due to non-convexity, we try 
    # a heuristic approach to design a good pattern, and opt in to use idealized pattern design
    
    # Attempt to create waveform that alternates cleanly. 
    # Use a more structured approach: pattern like a sawtooth:
    # Start with a base pattern with alternating 1s and 0s
    # We'll fine-tune this pattern with optimization later
    
    # Pattern creation approach: generate approximately balanced sequence using a 
    # quasi-uniform distribution of 1s and 0s, ideally incorporating golden ratio properties
    
    # Generate a low-discrepancy sequence
    phi = (1 + np.sqrt(5)) / 2  # Golden ratio
    indices = np.arange(n_small)
    fractional_parts = (indices * phi) % 1.0
    # Choose 36 indices to set to 1 (ground truth)
    sorted_indices = np.argsort(fractional_parts)
    ones_indices = sorted_indices[:36]
    
    # Create initial binary vector
    h_init = np.zeros(n_small)
    h_init[ones_indices] = 1.0
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(h_init)
    
    # Now refine with local optimization - use this seed pattern for local search
    bounds = [(0, 1) for _ in range(n_small)]
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx_small - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # First employ local optimization (SLSQP) to fine-tune the binary pattern
    try:
        res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 1000, 'ftol': 1e-14, 'gtol': 1e-14})
        if res.success:
            h_candidate = res.x
            # Clamp to binary (0 or 1) values around original h after optimization
            h_candidate = np.round(h_candidate).clip(0, 1)
            # Normalize to satisfy constraint
            scale = 1.0 / (np.sum(h_candidate) * dx_small)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            
            # Recalculate if we're near exact constraint
            if abs(np.sum(h_candidate) * dx_small - 1.0) < 0.001:
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
    except:
        pass
    
    # Maybe add additional smoothing or perturbation for convergence
    h_smooth = np.copy(best_h)
    if best_c5 > 0.38:
        # Use heuristic modified pattern
        iteration = 0
        max_iter = 1000
        k_iter = 50 # Every K iterations try acceptance of a bitflip from correctness
        while iteration < max_iter and (time.time() - start_time) < budget_s * 0.9:
            # Try to introduce a small change at random pair of indices
            i = np.random.randint(0, n_small)
            j = np.random.randint(0, n_small)
            
            # Be careful to keep sum constraint (we may try a swap)
            if h_smooth[i] > 0.5 and h_smooth[j] < 0.5:  # Swapping 1 to 0 and 0 to 1
                h_new = h_smooth.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                
                # Re-normalize to satisfy the integral = 1 constraint (this is crucial!)
                dx = 2.0/n_small
                scale_factor = 1.0 / (np.sum(h_new) * dx)
                h_new = h_new * scale_factor
                
                new_c5 = evaluate_c5(h_new)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
            elif iteration % k_iter == 0:  # Random few
                # simple x% chance to accept a small change possibly improving overlap
                pass # No change for simplicity; retain best found
        
            iteration += 1
        
        if best_c5 > 0.38:
            # Do a few iterations with stochastic descent
            for _ in range(500):
                if np.random.rand() < 0.3:
                    # Shift one random part (but maintain completeness)
                    idxs = np.random.choice(n_small, size=2, replace=False)
                    h_new = best_h.copy()
                    h_new[idxs[0]] = 1.0 - h_new[idxs[0]]
                    h_new[idxs[1]] = 1.0 - h_new[idxs[1]]
                    
                    # Renormalize
                    dx = 2.0 / n_small
                    scale_factor = 1.0 / (np.sum(h_new) * dx)
                    h_new = h_new * scale_factor
                    
                    new_c5 = evaluate_c5(h_new)
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5

    # Retain best solution from entire run (if smaller)  
    # Final check if we improved
    try:
        # Additional attempt with top-level direct optimization if needed
        h_final = best_h.copy()
        res = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-14})
        if res.success:
            h_candidate = res.x
            scale = 1.0 / (np.sum(h_candidate) * dx_small)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except:
        pass
    
    # Retrace backwards to original size (n_points), re-scale if needed, return solution
    if len(best_h) < n_points:
        # Expand pattern with interpolation and adjust for sum constraint
        h_upscaled = np.interp(np.linspace(0, n_points - 1, n_points), 
                               np.linspace(0, n_small - 1, n_small), best_h)
        dx_final = 2.0 / n_points
        scale_sum = 1.0 / (sum(h_upscaled) * dx_final)
        final_h = h_upscaled * scale_sum
        final_c5 = evaluate_c5(final_h)
    else:
        final_h = best_h.copy()
        final_c5 = best_c5
        
    # Ensure final integral = 1 
    dx_final = 2.0 / n_points
    scale_sum = 1.0 / (sum(final_h) * dx_final)
    final_h = final_h * scale_sum
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
