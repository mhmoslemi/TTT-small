import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_h(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2n for full convolution
    h_padded = np.pad(h, (0, n), 'constant')
    one_minus_h = np.pad(1 - h, (0, n), 'constant')
    corr = signal.fftconvolve(h_padded, one_minus_h[::-1], mode='valid')[:n]
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 48
    dx = 2.0 / n
    M = n // 2  # target sum
    T_target = int(np.ceil(0.3807 * M))
    
    # 1. MILP formulation
    h_bin = cp.Variable(n, boolean=True)
    p = cp.Variable((n-1, n), boolean=True)  # p[ℓ][i] = h[i]*h[i+ℓ]
    T = cp.Variable(integer=True, lower_bound=0)
    
    # Constraints
    mass_constraint = cp.sum(h_bin) == M
    
    # McCormick envelopes for product h[i]*h[i+ℓ]
    constraints = [mass_constraint]
    for l in range(n-1):
        for i in range(n-l):
            # p[l,i] <= h[i], p[l,i] <= h[i+l], p[l,i] >= h[i] + h[i+l] - 1
            constraints.append(p[l,i] <= h_bin[i])
            constraints.append(p[l,i] <= h_bin[i+l])
            constraints.append(p[l,i] >= h_bin[i] + h_bin[i+l] - 1)
    
    # Worst-lag constraints to maximize T
    for l in range(n-1):
        # sum_{i: i+l < n} p[l,i] + T >= M
        constraints.append(cp.sum(p[l,:n-l]) + T >= M)
    
    # Objective: minimize T
    prob = cp.Problem(cp.Minimize(T), constraints)
    
    # solve with CB
    time_left = budget_s - (time.time() - start_time)
    if time_left <= 1:
        best_h_bin = np.zeros(n, dtype=int)
        best_h_bin[:M] = 1
        h_bin_sol = best_h_bin
        best_T = 0
    else:
        prob.solve(solver=cp.CBC, maxSeconds=time_left * 0.9)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_bin_sol = np.round(h_bin.value).astype(int)
            best_T = int(T.value)
        else:
            # fallback strategy
            h_bin_sol = np.zeros(n, dtype=int)
            h_bin_sol[:M] = 1
            best_T = M
    
    h_bin_best = h_bin_sol.copy()
    C5_best = evaluate_h(h_bin_best)
    
    # If we already beat
    if best_T <= T_target:
        h_final = h_bin_best.astype(float)
        C5_eval = evaluate_h(h_final)
        return (h_final.tolist(), C5_eval, n)

    # 2. Binary local refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left <= 1:
        h_final = h_bin_best.astype(float)
        C5_eval = evaluate_h(h_final)
        return (h_final.tolist(), C5_eval, n)
    
    # Binary swap-based simulated annealing
    current_h = h_bin_best.copy()
    current_c5 = C5_best
    T_init = 1.0
    temp = T_init
    cooling = 0.999
    min_temp = 1e-6
    
    # Initial evaluation
    T_estimate = current_c5  # This is a rough estimate of typical change
    if T_estimate <= 0.1:  # Avoid numerical issues
        T_estimate = 0.5
    
    # Set initial T based on estimate to get reasonable acceptance probability
    temp = 10.0 * T_estimate  # Start with larger temperature to accept various moves
    
    # 3000 step max for binary search
    n_iter = min(3000, int(time_left*100)) if time_left > 1 else 1000
    for step in range(n_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
        # pick a random 1 and 0
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h_new = current_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # evaluate new C5
        c5_new = evaluate_h(h_new)
        
        delta = c5_new - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = h_new
            current_c5 = c5_new
        
        # decay temp
        temp *= cooling
    
    # 3. Fractional smoothing using projected gradient descent
    time_left = budget_s - (time.time() - start_time)
    if time_left <= 0.5:
        # avoid taking time for fractional part if too little time left
        h_final = current_h.astype(float)
        C5_eval = evaluate_h(h_final)
        return (h_final.tolist(), C5_eval, n)
    
    h_current = current_h.astype(float)
    beta_schedule = [1, 2, 5, 10, 20]
    step_size = 0.01
    n_grad_steps = 100  # we keep this since each entire round is restricted by time_budget
    
    # Water-filling function for mass constraint
    def project_mass(h, target_sum):
        def objective(lam):
            h_proj = np.clip(h + lam, 0, 1)
            return np.sum(h_proj) - target_sum
        
        # bisection
        low, high = -2, 2
        for _ in range(30):
            mid = (low + high) / 2
            if objective(mid) < 0:
                high = mid
            else:
                low = mid
        lam = (low + high) / 2
        return np.clip(h + lam, 0, 1)
    
    # Execute smoothing steps
    for beta in beta_schedule:
        for _ in range(n_grad_steps):
            if time.time() - start_time > budget_s * 0.99:
                break
            
            dx_L = np.zeros(n)  # gradient of L_ell
            # For all lags, computing direction of gradient
            # But we use the correlation formula again, which maximises the inline lag sum
            # Полный вариант: convolutions over entire FFT space
            h_padded = np.pad(h_current, (0, n), 'constant')
            ones_padded = np.pad(1 - h_current, (0, n), 'constant')
            corr = signal.fftconvolve(h_padded, ones_padded[::-1], mode='valid')[:n]
            
            # compute 1 + d/dh_i for L_ell = h_i (1 - h_{i+ell}) in FFT way
            # for i, for l ignore edges for implementation, but it works just fine
            all_D = np.zeros_like(h_current)
            for l in range(1, n):
                conv_result = signal.fftconvolve(h_current, -np.ones_like(h_current), mode='full')
                # The forward lag l, so that h[i+l] * some constant goes to x[i] when convoluted with [1,1,...1]
                # More straightforward: just do direct indexing trick
                for i in range(n-l):
                    if i < n and i+l < n:
                        all_D[i] += (1.0 - h_current[i+l])  # subtract due to difference definition
                        all_D[i+l] -= h_current[i]  # due to negative sign
            # But truncated edge effects won't matter much, we use sparse sparse computation
            # Let simulate L_ell = sum of h[i] * (1 - h[i+l]), too noisy to approximation
            # Let just simulate normal gradient;
            # Let just recompute gradient over all lags to get gradient and modifier
            # Better forward algorithm per-lag
            grad = np.zeros_like(h_current)
            # plain way
            H1 = np.pad(h_current, (0,n), 'constant')
            G = np.pad(1-h_current, (0,n), 'constant')
            # create correlations for each lag, then sum in manner of formula
            normalized_corr = np.zeros(n)
            # for ell:
            E = 1e-8  # to avoid divide by zero
            
            # Recompute Correlation and renewables
            weights = np.zeros(n-1)
            all_h_lags = []  # safe to store on heap size n
            for l in range(1, n):
                corr_l = np.convolve(h_current, np.flip(1-h_current[:n-l]), mode='valid')
                all_h_lags.append(corr_l)
                if len(corr_l) > 0:
                    weights[l-1] = np.exp(beta * np.max(corr_l))
                else:
                    weights[l-1] = 0.
            
            norm = np.sum(weights) + E
            weights = weights / norm
            # Build aggregrate gradient by weighted sum across lags
            grad = np.zeros_like(h_current)
            for l in range(1, n):
                if len(all_h_lags[l-1]) > 0:
                    # create deltas for naive gradient
                    diff_l = 1 - np.concatenate([h_current[l:], np.zeros(l)])
                    temp = all_h_lags[l-1]
                    # compute for finite increments in the main index
                    delta_grad = weights[l-1] * diff_l[:len(temp)]
                    # accumulate into gradient
                    grad[:len(temp)] += delta_grad
                    
                    # the backstep part
                    diff_back = -np.pad(h_current[:n-l], (l,0), 'constant')
                    grad = grad + weights[l-1] * diff_back
            
            # Update using gradient, and project back
            h_next = h_current - step_size * grad
            h_current = project_mass(h_next, M * dx)
            
            # preserve bounds
            h_current = np.clip(h_current, 0, 1)
    
    # Final clean-up
    h_final = h_current.copy()
    h_final = np.clip(h_final, 0, 1)
    M_actual = np.sum(h_final) * dx
    if abs(M_actual - 1.0) > 1e-9:
        h_final = h_final / M_actual
    
    C5_eval = evaluate_h(h_final)
    return (h_final.tolist(), C5_eval, n)
