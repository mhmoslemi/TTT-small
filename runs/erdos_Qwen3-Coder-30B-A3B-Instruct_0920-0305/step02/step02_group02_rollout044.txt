import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

# Constants
n_candidates = [48, 56, 64, 80]
dx_candidates = [2.0 / n for n in n_candidates]
w_candidates = [n / 2.0 for n in n_candidates]
max_alpha = 500.0
alpha_schedule = [1, 2, 4, 8, 16, 32, 64, 128, 256, 512]
eps = 1e-6

def project_to_capped_simplex(v, w, eps_zero=1e-12):
    """Project v onto {x : 0≤x≤1, Σx=w} using water-filling."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    v_sorted = np.append(v_sorted, -np.inf)
    cumsum = np.cumsum(v_sorted)
    idx = np.arange(n) + 1
    condition = (v_sorted[:-1] - (cumsum - w) / idx) > 0
    k = np.sum(condition)
    if k == 0:
        tau = max(0, min(1, w / n))
        x = np.full(n, tau)
    else:
        tau = (cumsum[k-1] - w) / k
        x = np.clip(v - tau, 0, 1)
        # Adjust to make sum exactly w
        resid = w - np.sum(x)
        if abs(resid) > eps_zero:
            # Add to the first non-saturated
            first_non_1 = np.argmax(x < 1)
            x[first_non_1] += resid
    return x

def evaluate_overlap(h, w, dx):
    """Evaluate the overlap integral C5(h) = dx * max_k r_k(h)."""
    # corr = np.correlate(h, 1 - h, mode='full')
    # r = w - corr[corr.size//2:]
    # C5 = dx * np.max(r)
    # Instead, compute via FFT for efficiency
    n = len(h)
    h_flip = np.flip(h)
    corr = fftconvolve(h, h_flip, mode='full')
    corr = corr[len(h)-1:]
    r = w - corr
    C5 = dx * np.max(r)
    return C5

def smooth_max_alpha(h, w, dx, alpha):
    """Smooth max of r_k with log-sum-exp approximation."""
    n = len(h)
    # Compute r_k = w - S_k(h) where S_k(h) = h * h (shifted auto-correlation)
    h_flip = np.flip(h)
    corr_full = fftconvolve(h, h_flip, mode='full')
    corr = corr_full[len(h)-1:]
    r = w - corr  # r_k for k = 0, ..., n-1

    # Apply log-sum-exp
    max_r = np.max(r)
    r_centered = r - max_r
    exp_ratios = np.exp(alpha * r_centered)
    sum_exp = np.sum(exp_ratios)
    log_sum_exp = max_r + np.log(sum_exp) / alpha
    return dx * log_sum_exp

def gradient_of_r(h, k, w):
    """Return the gradient of r_k(h) = w - S_k(h) = w - sum h_i * h_{i+k}."""
    n = len(h)
    grad = np.zeros(n)
    # h_i * h_{i+k} terms
    if k == 0:
        grad = np.ones(n) - 2 * h
    else:
        # d/dh_i r_k = 1 - h_{i+k} - h_{i-k} (assuming indices are valid)
        for i in range(n):
            if i+k < n:
                grad[i] -= h[i+k]
            if i-k >= 0:
                grad[i] -= h[i-k]
        grad = np.ones(n) - grad
    return grad

def gradient_of_smooth_max(h, w, dx, alpha):
    """Compute gradient of smooth max with log-sum-exp."""
    n = len(h)
    # Compute all S_k
    h_flip = np.flip(h)
    corr_full = fftconvolve(h, h_flip, mode='full')
    corr = corr_full[len(h)-1:]
    r = w - corr

    # Softmax weights
    max_r = np.max(r)
    shifted = r - max_r
    exp_ratios = np.exp(alpha * shifted)
    wts = exp_ratios / np.sum(exp_ratios)

    # Sum ∂r_k/∂h_i times w_k
    grad = np.zeros(n)
    for k in range(n):
        grad_k = gradient_of_r(h, k, w)
        grad += wts[k] * grad_k

    return dx * grad

def fista_smooth_optim(n, w, dx, alpha_max=500.0, max_iter=1000):
    # Initialization
    h = np.random.uniform(0, 1, n)
    h = project_to_capped_simplex(h, w)
    y = h.copy()
    t = 1.0
    alpha = 1.0
    for alpha in [1, 2, 4, 8, 16, 32, 64, 128]:
        prev_f = np.inf
        for _ in range(max_iter):
            g = gradient_of_smooth_max(y, w, dx, alpha)
            step_size = 0.5  # Safe step size
            h_new = project_to_capped_simplex(y - step_size * g, w)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t**2)) / 2.0
            y_new = h_new + ((t - 1.0) / t_new) * (h_new - h)
            f = smooth_max_alpha(h_new, w, dx, alpha)
            if abs(f - prev_f) / (abs(f) + 1e-12) < eps:
                break
            h = h_new
            y = y_new
            t = t_new
            prev_f = f
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Preliminary sweep through n values, preferring smaller ones
    for i in range(len(n_candidates)):
        if time.time() - start_time > 0.9 * budget_s:
            break
        n = n_candidates[i]
        dx = dx_candidates[i]
        w = w_candidates[i]
        h_candidate = fista_smooth_optim(n, w, dx, alpha_max)
        c5_eval = evaluate_overlap(h_candidate, w, dx)
        if c5_eval < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_eval
            best_n = n

    # Run final local refinement
    if best_h is not None and time.time() - start_time < 0.95 * budget_s:
        bounds = [(0, 1) for _ in range(len(best_h))]
        def obj(h):
            return evaluate_overlap(h, w, dx)
        try:
            result = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                              options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-7})
            if result.success:
                h_final = result.x
                c5_final = evaluate_overlap(h_final, w, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    # Final projection and guarantee correctness of the sum
    if best_h is not None:
        best_h = project_to_capped_simplex(best_h, w)
        best_c5 = evaluate_overlap(best_h, w, dx)

    # Return best solution as standard list
    return list(best_h), best_c5, best_n
