import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def project_simplex(y, target_sum):
    """Project vector y onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    n = len(y)
    low, high = -max(y) - 1, max(y) + 1
    for _ in range(40):
        lam = (low + high) / 2
        x = np.clip(y - lam, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = lam
        else:
            high = lam
    x = np.clip(y - lam, 0, 1)
    # Scale to target sum
    scale = target_sum / np.sum(x)
    return x * scale

def evaluate_c5(h):
    """Evaluate C5 for step function h"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    n_target = 180

    # SDP approach
    n_list = [30, 35, 40, 45, 50]
    sdps = []
    
    for n in n_list:
        if (time.time() - start_time) > 0.8 * budget_s:
            break
            
        # Create SDP variables
        X = cp.Variable((n, n), symmetric=True)
        h = cp.Variable(n)
        
        # Define constraints
        constraints = [
            X >> 0,  # PSD
            cp.diag(X) == h,  # Diagonal elements
            h >= 0,
            h <= 1,
            cp.sum(h) == n / 2,
        ]
        
        # Add autocorrelation constraints
        m = cp.Variable()
        for s in range(-(n-1), n):
            if s >= 0:
                r_s = sum(X[i, i+s] for i in range(n-s))
            else:
                r_s = sum(X[i+s, i] for i in range(-s, n))
            constraints.append(r_s >= m)
        
        # Objective
        objective = cp.Maximize(m)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, eps=1e-5, max_iters=3000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                sdps.append((n, X.value, m.value, h.value))
        except:
            pass

    # Process SDP results
    for n, X_val, m_val, h_val in sdps:
        # Extract h from X
        if (time.time() - start_time) > 0.9 * budget_s:
            break
            
        # Leading eigenvector
        try:
            w, v = np.linalg.eigh(X_val)
            v = np.abs(v[:, -1])
            v = project_simplex(v, n/2)
            if np.all(v >= 0) and np.all(v <= 1):
                c5 = evaluate_c5(v)
                if c5 < best_c5:
                    best_h = v.copy()
                    best_c5 = c5
        except:
            pass

    # Refinement with pattern search on best low-dim h
    if best_h is not None and (time.time() - start_time) < 0.95 * budget_s:
        # Simple pattern search refinement
        def obj(h):
            return evaluate_c5(h)
        
        current_h = best_h.copy()
        # Project to simplex before optimization to maintain sum
        current_h = project_simplex(current_h, len(current_h)/2)
        current_c5 = obj(current_h)
        
        # Nelder-Mead local search
        try:
            from scipy.optimize import minimize
            
            def optimize_step(h0, func):
                bounds = [(0, 1) for _ in range(len(h0))]
                res = minimize(func, h0, method='Nelder-Mead', 
                              options={'fatol': 1e-10, 'xatol': 1e-10, 'maxiter': 500})
                if res.success:
                    hnew = res.x.copy()
                    hnew = project_simplex(hnew, len(hnew)/2)
                    return hnew, func(hnew)
                return h0, func(h0)
            
            for _ in range(100):
                if (time.time() - start_time) > 0.95 * budget_s:
                    break
                new_h, new_c5 = optimize_step(current_h, obj)
                if new_c5 < current_c5:
                    current_h = new_h
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                else:
                    # Perturb slightly
                    step = np.random.normal(0, 0.01, len(current_h))
                    current_h = project_simplex(current_h + step, len(current_h)/2)
                    current_c5 = obj(current_h)
        
        except:
            pass

    # Upsampling and final refinement
    if best_h is not None and (time.time() - start_time) < 0.99 * budget_s:
        # Interpolate to higher resolution
        original_h = best_h
        n_upsampled = n_target
        scaled_h = np.interp(np.linspace(0, 2, n_upsampled), 
                             np.linspace(0, 2, len(original_h)), original_h)
        
        # Project to simplex
        scaled_h = project_simplex(scaled_h, n_upsampled / 2)
        final_c5 = evaluate_c5(scaled_h)
        
        if final_c5 < best_c5:
            best_h = scaled_h
            best_c5 = final_c5

        # Final pattern search refinement

            from scipy.optimize import minimize
            
            def obj(h):
                return evaluate_c5(h)

                          options={'fatol': 1e-12, 'xatol': 1e-12, 'maxiter': 1000})
            if res.success:
                refined_h = res.x.copy()
                refined_h = project_simplex(refined_h, len(refined_h)/2)
                refined_c5 = obj(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
