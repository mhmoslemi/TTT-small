def project_simplex(y, target_sum):
    """Project y onto the capped simplex: 0 <= x <= 1, sum(x) = target_sum"""
    y = np.asarray(y)
    low, high = -np.max(y) - 1, np.max(y) + 1
    for _ in range(40):
        mid = (low + high) / 2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(y - low, 0, 1)
    scale = target_sum / np.sum(x)
    return x * scale

def evaluate_c5_final(h):
    """Evaluate C5 on the final vector"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
