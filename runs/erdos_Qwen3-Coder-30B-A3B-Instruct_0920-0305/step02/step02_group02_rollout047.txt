import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.linalg import eigh, cholesky
import time
import cvxpy as cp
from cvxpy import PSD, Variable, Minimize, sum_entries, trace, multiply
import math

# Constants
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * 2.0 / len(h)
    return c5

def solve_sdp(n_points):
    """Solve SDP relaxation for the Erdos overlap problem."""
    dx = 2.0 / n_points
    w = n_points / 2.0
    X = Variable((n_points, n_points), PSD=True)
    h = Variable(n_points)
    
    # Toeplitz constraints
    T_matrices = []
    for k in range(n_points):
        T = np.zeros((n_points, n_points))
        for i in range(n_points - k):
            T[i, i + k] = 1
        T_matrices.append(T)
        
    # SDP constraints
    constraints = [
        cp.diag(X) == h * h,
        cp.sum(h) == w,
        h >= 0,
        h <= 1,
    ]
    
    # Add Toeplitz constraints
    for k, T in enumerate(T_matrices):
        constraints.append(trace(T @ X) >= 0.0)
    
    # Objective: maximize minimal correlation
    objective = Minimize(-trace(T_matrices[0] @ X))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        if prob.status == 'optimal':
            h_sdp = h.value
            if h_sdp is not None:
                return h_sdp
    except:
        pass
    return None

def project_to_simplex(v, w):
    """Project vector v onto the capped simplex with sum = w."""
    v = np.array(v)
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    rho = 0
    for i in range(n):
        if (v_sorted[i] - (np.sum(v_sorted[:i+1]) - w) / (i+1)) > 0:
            rho = i + 1
    theta = (np.sum(v_sorted[:rho]) - w) / rho
    return np.clip(v - theta, 0, 1)

def random_round_h(h_sdp, X_sdp):
    """Round SDP solution using randomized rounding."""
    try:
        R = cholesky(X_sdp).T  # R^T R = X_sdp
        g = np.random.randn(R.shape[0])
        round_vec = R.T @ g
        h_round = project_to_simplex(round_vec, np.sum(h_sdp))
        return h_round
    except:
        return None

def fit_fourier(h_vec, M=8):
    """Fit h_vec with a truncated Fourier series."""
    n = len(h_vec)
    Phi = np.zeros((n, 2*M + 1))
    Phi[:, 0] = 1
    for j in range(1, M+1):
        Phi[:, 2*j - 1] = np.cos(np.pi * j * np.arange(n) / n)
        Phi[:, 2*j] = np.sin(np.pi * j * np.arange(n) / n)
    
    def constrained_least_squares(a):
        h_reconstructed = 0.5 + Phi @ a
        h_reconstructed = np.clip(h_reconstructed, 0, 1)
        return np.sum((h_vec - h_reconstructed)**2)
    
    a_init = np.zeros(2*M)
    result = minimize(constrained_least_squares, a_init, method='L-BFGS-B', 
                      bounds=[(-10, 10)] * (2*M))
    return result.x

def reconstruct_h_from_fourier(a_coeffs, n):
    """Reconstruct h from Fourier coefficients."""
    M = len(a_coeffs) // 2
    h = 0.5 + np.sum([a_coeffs[2*j-1]*np.cos(np.pi * (j) * np.arange(n) / n) + 
                      a_coeffs[2*j]*np.sin(np.pi * (j) * np.arange(n) / n)
                      for j in range(1, M+1)], axis=0)
    return np.clip(h, 0, 1)

def smooth_optimize(h_init, n_points, max_iter=500):
    """Smooth optimization of the overlap function using projected gradient."""
    def surrogate_objective(a_coeffs, alpha):
        h = reconstruct_h_from_fourier(a_coeffs, n_points)
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1-h, mode='full')
        S = np.max(correlation)
        return -S * 2.0 / n_points * alpha

    def gradient_ascent(a_coeffs, alpha, max_iter=200):
        step_size = 1e-3
        for _ in range(max_iter):
            h = reconstruct_h_from_fourier(a_coeffs, n_points)
            h = np.clip(h, 0, 1)
            correlation = np.correlate(h, 1-h, mode='full')
            k_min = np.argmax(correlation)
            shift = k_min - (n_points - 1) // 2  # appropriate shift for original dx=2/n
            grad_vector = np.zeros(n_points)
            for i in range(n_points - abs(shift)):
                if shift >= 0:
                    grad_vector[i] += (1 - h[i + shift])
                else:
                    grad_vector[i + (-shift)] += (1 - h[i])
            grad_vector = np.clip(grad_vector, 0, 1)
            grad_vector = grad_vector / np.sqrt(np.sum(grad_vector**2) + 1e-12)
            a_coeffs += step_size * grad_vector
            h = reconstruct_h_from_fourier(a_coeffs, n_points)
            h = np.clip(h, 0, 1)
            # Re-project onto simplex
            h = project_to_simplex(h, n_points/2)
            a_coeffs = fit_fourier(h, len(a_coeffs) // 2)
        return a_coeffs

    # Use a simple gradient ascent over a refined set of shift vectors
    # Start with the initial guess derived from Fourier
    a_coeffs = fit_fourier(h_init, 8)
    try:
        a_coeffs = gradient_ascent(a_coeffs, 100, max_iter=200)
        h_final = reconstruct_h_from_fourier(a_coeffs, n_points)
        return project_to_simplex(h_final, n_points/2)
    except:
        return h_init

def run(seed=42, budget_s=600):
    """Main function to run."""
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = []
    best_n = None
    sizes = [48, 56]

    for n_points in sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h_sdp = solve_sdp(n_points)
        if h_sdp is None:
            continue
            
        # Early break if no valid solution
        if not np.isfinite(h_sdp).all():
            continue
            
        # Rounded solution
        try:
            X_sdp = np.outer(h_sdp, h_sdp)
            h_round = random_round_h(h_sdp, X_sdp)
        except:
            continue
            
        if h_round is None:
            continue
            
        # Fourier fitting and smooth optimization
        h_opt = smooth_optimize(h_round, n_points)
        c5_bound = evaluate_h(h_opt)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt
            best_n = n_points
            
        # A few additional runs with different random seeds for smaller n
        for i in range(5):
            if time.time() - start_time > budget_s * 0.9:
                break
            try:
                h_sdp = solve_sdp(n_points)
                if h_sdp is None:
                    continue
                X_sdp = np.outer(h_sdp, h_sdp)
                h_round = random_round_h(h_sdp, X_sdp)
                if h_round is None:
                    continue
                h_opt = smooth_optimize(h_round, n_points)
                c5_bound = evaluate_h(h_opt)
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_opt
                    best_n = n_points
            except:
                continue
                
    return list(best_h), best_c5, best_n
