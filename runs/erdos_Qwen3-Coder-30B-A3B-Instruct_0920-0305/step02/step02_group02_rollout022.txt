import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def project_onto_capped_simplex(x, target_sum):
    """Project x onto the capped simplex {x : 0 <= x_i <= 1, sum x_i = target_sum}"""
    x = np.asarray(x)
    n = len(x)
    if np.isclose(np.sum(x), target_sum):
        return x
    # water-filling algorithm
    y = np.sort(x)[::-1]
    z = np.cumsum(y)
    idx = np.arange(1, n + 1)
    temp = (target_sum - z) / idx
    rho = np.max(np.where(y > temp)[0])
    if rho == 0:
        threshold = target_sum / n
        result = np.clip(x - threshold, 0, 1)
    else:
        threshold = temp[rho - 1]
        result = np.clip(x - threshold, 0, 1)
    # ensure exact sum
    scale = target_sum / np.sum(result)
    result = result * scale
    return result

def compute_correlations(h, dx):
    """Return correlations r_k = sum h[i] * (1 - h[i+k]) for k = 0, ..., n-1"""
    n = len(h)
    # Calculate sum h[i] * h[i+k] for all valid k
    r_k = np.zeros(n)
    S_k = np.zeros(n)
    for k in range(n):
        if k == 0:
            # h[i] * (1-h[i]) = h[i] - h[i]^2
            temp = h - h * h
        else:
            # shift h by k and compute dot product
            shift = np.zeros(n)
            shift[k:] = h[:-k]
            temp = h * shift
        S_k[k] = np.sum(temp)
        r_k[k] = np.sum(h) - S_k[k]
    return r_k

def log_sum_exp(x, alpha):
    """Compute log(sum(exp(alpha*x))) using numerically stable trick"""
    x = np.asarray(x)
    max_x = np.max(x)
    return max_x + np.log(np.sum(np.exp(alpha * (x - max_x))))

def gradient_r_k(h, k, dx):
    """Compute gradient of r_k with respect to h_i"""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        # r_k = sum(h_i) - sum(h_i^2) => grad = 1 - 2*h_i
        grad[:] = 1.0 - 2.0 * h
    else:
        # r_k = sum h_i * (1 - h_{i+k}) = sum h_i - sum(h_i * h_{i+k})
        grad += 1.0  # from sum h_i term
        # now subtract contribution from h_i * h_{i+k} terms:
        # 
        # For i , the k term contributes to grad[i] => - h_{i+k} * (i+k<n)
        # and for j=k+i, the contribution to grad[j] => - h_{i} * (j-k >= 0)
        if k < n:
            # d/dh_i (h_i * h_{i+k}) = h_{i+k}
            # ∂r_k / ∂h_i = 1 - h_{i+k} * 1_{i+k<n}
            grad -= np.concatenate([np.zeros(k), h[:-k]])
        if k > 0:
            #    d/dh_i (h_i * h_{i+k}) = h_{i-k} * 1_{i-k>=0}
            grad -= np.concatenate([h[k:], np.zeros(k)])
    return grad

def compute_gradient_of_logmax(h, dx, alpha):
    """Compute gradient of smooth approximation log-sum-exp of r_k"""
    r_k = compute_correlations(h, dx)
    # Normalize by max for numerical stability
    max_r = np.max(r_k)
    r_k_norm = r_k - max_r
    exps = np.exp(alpha * r_k_norm)
    Z = np.sum(exps)
    weights = exps / Z
    # init gradient
    g = np.zeros_like(h)
    for k in range(len(h)):
        g += weights[k] * gradient_r_k(h, k, dx)
    return dx * g

def backtracking_line_search(h_half, g, dx, alpha, func_eval, max_iter=10):
    """Simple backtracking line search"""
    step_size = 1.0
    c1 = 1e-4  # Armijo condition
    rho = 0.5
    h = h_half
    r_k = compute_correlations(h, dx)
    old_val = dx * log_sum_exp(r_k, alpha)
    for iter in range(max_iter):
        h_new = h - step_size * g
        h_new = project_onto_capped_simplex(h_new, np.sum(h))
        r_k_new = compute_correlations(h_new, dx)
        new_val = dx * log_sum_exp(r_k_new, alpha)
        if new_val <= old_val - c1 * step_size * np.dot(g, g):
            return h_new, step_size
        step_size *= rho
    return h_half, step_size

def accelerated_projected_gradient_descent(h0, dx, target_sum, alpha0=1.0, alpha_factor=2.0,
                                          alpha_max=300.0, max_iters=5000):
    """Run accelerated projected gradient descent."""
    h = h0.copy()
    h = project_onto_capped_simplex(h, target_sum)
    t = 1.0
    y = h.copy()
    alpha = alpha0
    best_c5 = np.inf
    best_h = h.copy()
    
    for it in range(max_iters):
        # Compute approx function value and its gradient
        r_k = compute_correlations(h, dx)
        f_val = dx * log_sum_exp(r_k, alpha)
        g = compute_gradient_of_logmax(h, dx, alpha)
        # Backtracking line search
        y_new = y - g
        y_new = project_onto_capped_simplex(y_new, target_sum)
        new_h, step_size = backtracking_line_search(y_new, g, dx, alpha, lambda x: log_sum_exp(compute_correlations(x, dx), alpha))
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = new_h + ((t - 1.0) / t_new) * (new_h - h)
        y = project_onto_capped_simplex(y, target_sum)
        h = new_h
        t = t_new
        
        # Check if we hit target
        c5_trial = dx * np.max(r_k)
        if c5_trial < best_c5:
            best_h = h.copy()
            best_c5 = c5_trial

        # Update alpha for continuation if needed
        if alpha < alpha_max:
            if it > 100 and abs(f_val) < 1e-10:
                alpha = min(alpha * alpha_factor, alpha_max)

        # Early stop if we improve less than a tiny amount
        if it >= 100 and abs(f_val) < 1e-12:
            break
    return best_h, best_c5

def initial_h_values(n_points):
    """Initial guess: binary switch pattern"""
    h = np.zeros(n_points)
    interval = 2.0 / n_points
    # Let's take a triangular shape near 1, rest near 0
    for i in range(n_points):
        x = i * interval
        if 0.5 < x < 1.5:
            h[i] = 1.0
        else:
            h[i] = 0.0
    h = project_onto_capped_simplex(h, n_points / 2.0)
    return h

def play_with_zero_positions(h, low, high):
    """Modify positions slightly."""
    h_modified = h.copy()
    mask = (h_modified > low) & (h_modified < high)
    indices = np.where(mask)[0]
    if len(indices) > 0:
        idx = np.random.choice(indices)
        # shift a tiny bit
        h_modified[idx] = np.clip(h_modified[idx] + 0.01, 0, 1)
    return project_onto_capped_simplex(h_modified, np.sum(h))

def run(seed=42, budget_s=600):
    best_h = None
    best_c5 = np.inf
    np.random.seed(seed)
    ideal_n_values = [48, 56, 64, 80]
    start_time = time.time()
    time_per_n = budget_s / len(ideal_n_values)
    
    for n in ideal_n_values:
        if time.time() - start_time > budget_s * 0.99:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        valid_starting_points = 20
        n_restart_iterations = 0
        while time.time() - start_time < time_per_n * 0.9 and n_restart_iterations < valid_starting_points:
            # Initialize from different sources
            if n_restart_iterations % 3 == 0:
                # Uniform uniform
                h0 = np.random.rand(n)
            elif n_restart_iterations % 3 == 1:
                # Binary-like
                h0 = initial_h_values(n)
            else:
                # Personal trick: Some binary clustering + varied points
                h0 = np.random.rand(n)
                h0 = h0 * (target_sum / np.sum(h0))  # normalize to match sum
            # ensure entry-wise values are in [0,1]  
            h0 = np.clip(h0, 0, 1)
            h0 = project_onto_capped_simplex(h0, target_sum)
            # refine slightly by reprojecting back and forth
            h0 = project_onto_capped_simplex(h0, target_sum)
            
            try:
                # Run accelerated descent
                h_opt, c5_opt = accelerated_projected_gradient_descent(h0, dx, target_sum)
                # Recompute the actual value with the corrected agressive time limit
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                n_restart_iterations += 1
            except:
                n_restart_iterations += 1
                pass

    # Final fix and evaluation
    final_h = project_onto_capped_simplex(best_h, np.sum(best_h))
    eval_c5 = np.max(np.correlate(final_h, 1 - final_h, mode="full")) * (2.0 / len(final_h))
    return list(final_h), float(eval_c5), len(final_h)
