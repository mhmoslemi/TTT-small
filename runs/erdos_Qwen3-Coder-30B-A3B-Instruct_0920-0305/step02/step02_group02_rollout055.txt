import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
from scipy.spatial.distance import cdist
import math
import time
from itertools import product

# Global variables for fixed parameters
n_points_list = [48, 56, 64, 80]
target_c5 = 0.3807
max_iter = 10000
trust_region_iter = 200

def evaluate_h(h_values):
    """Compute the C_5 bound using direct correlation."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(corr)
    return max_correlation * dx

def lse_smoothed_max(h, alpha, w):
    """Compute log-sum-exp approximation of max r."""
    # Compute all r_k values efficiently via FFT
    nh = len(h)
    hpad = np.pad(h, (0, nh))
    hconj = np.conjugate(fft(hpad))
    auto_corr = ifft(hconj * hconj).real[nh-1:]
    r_vals = w - auto_corr

    # Apply log-sum-exp smoothing with numerical stability
    max_r = np.max(r_vals)
    shifted = r_vals - max_r
    exp_term = np.exp(alpha * shifted)
    log_sum = np.log(np.sum(exp_term)) + max_r * alpha
    return (log_sum / alpha) / w

def projection_simplex(h, w):
    """Project h onto capped simplex: 0 <= h[i] <= 1, sum(h) = w."""
    # Waterfilling approach
    h_sorted = np.sort(h)[::-1]
    n = len(h_sorted)
    
    # Find the threshold tau
    cumsum = np.cumsum(h_sorted)
    tau_candidates = (cumsum - w) / np.arange(1, n + 1)
    pos = np.searchsorted(tau_candidates, h_sorted, side="right")
    pos = np.maximum(pos, 0)
    indices = np.arange(n)
    if np.any(indices < pos):
        tau = tau_candidates[np.max(indices[indices < pos])]
    else:
        tau = tau_candidates[-1] if len(tau_candidates) > 0 else 0
    
    # Apply thresholding and renormalize if necessary
    proj = np.maximum(h - tau, 0)
    proj = np.minimum(proj, 1)
    sum_proj = np.sum(proj)
    if abs(sum_proj - w) > 1e-10:
        # Adjust: evenly distribute residual to maintain sum
        diff = w - sum_proj
        for i in range(len(proj)):
            if proj[i] < 1 and diff > 0:
                gain = min(1 - proj[i], diff)
                proj[i] += gain
                diff -= gain
                if diff <= 0:
                    break
    return proj

def compute_gradient_r(h, k, w):
    """Compute d/dh_i of r_k(h) = w - S_k(h)."""
    n = len(h)
    grad_r_k = np.zeros(n)
    if k == 0:
        grad_r_k = -2 * h + 1
    else:
        # This is the finite difference approximation to the gradient
        # If doing it correctly, use slicing
        if k > 0:
            # Replace with actual derivatives
            grad_r_k[:n-k] -= h[k:]
            grad_r_k[k:] -= h[:n-k]
            grad_r_k -= 1
            grad_r_k = -grad_r_k  # because derivative of r = w - S
    return grad_r_k

def compute_all_gradients(h, w):
    """Compute d/dh_i of all r_k."""
    n = len(h)
    gradients = []
    for k in range(n):
        grad = compute_gradient_r(h, k, w)
        gradients.append(grad)
    return np.array(gradients)

def accelerated_gradient_step(h, alpha, w, eps=1e-8):
    """Compute one FISTA step for smooth surrogate."""
    n = len(h)
    # Compute all r_k values
    hpad = np.pad(h, (0, n))
    hconj = np.conjugate(fft(hpad))
    auto_corr = ifft(hconj * hconj).real[n-1:]
    r_vals = w - auto_corr

    # Log-sum-exp step
    max_r = np.max(r_vals)
    shifted = r_vals - max_r
    exp_vals = np.exp(alpha * shifted)
    Z = np.sum(exp_vals)
    weights = exp_vals / Z  # Softmax weights

    # Compute gradient of LSE
    grads = compute_all_gradients(h, w)
    gradient = np.dot(weights, grads)  # weighted average of gradients

    # Simple backtracking for descent step
    step_size = 0.5
    y = h - step_size * gradient
    h_new = projection_simplex(y, w)

    return h_new

def run_single_restart(n_points, seed, alpha_max, beta=2, eps=1e-8, steps=2000):
    np.random.seed(seed)
    dx = 2.0 / n_points
    w = n_points / 2.0

    # Initialize h using the binary/balanced heuristic
    h = np.zeros(n_points)
    frac = w / n_points
    h[:int(np.round(frac * n_points))] = 1.0
    # Add noise for variation
    h += np.random.normal(0, 0.001, n_points)

    # Project onto simplex
    h = projection_simplex(h, w)

    alpha = 1.0
    while alpha <= alpha_max:
        conv = 1
        its = 0
        h_k = h.copy()
        old_f = lse_smoothed_max(h_k, alpha, w)

        while conv > eps and its < steps:
            h_kp1 = accelerated_gradient_step(h_k, alpha, w)
            f_k = lse_smoothed_max(h_kp1, alpha, w)
            conv = abs(f_k - old_f) / (abs(old_f) + 1e-10)
            old_f = f_k
            h_k = h_kp1
            its += 1

        h = h_k
        alpha *= beta

    # Final check with LSE, then actual computation
    final_max = evaluate_h(h)
    return h, final_max

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = []
    best_n = 0
    alpha_max = 500
    delta_t = budget_s / len(n_points_list)

    np.random.seed(seed)
    for i, n in enumerate(n_points_list):
        if time.time() - start_time > budget_s * 0.95:
            break
        local_time = time.time()
        trial_c5 = best_c5
        trial_h = []

        # Multi restarts
        restarts = 30  # reasonable number for 120s
        for r in range(restarts):
            if time.time() - start_time > budget_s * 0.99:
                break
            seed_r = seed + i * 1000 + r
            try:
                h_r, c5_r = run_single_restart(n, seed_r, alpha_max)
                if c5_r < trial_c5:
                    trial_c5 = c5_r
                    trial_h = h_r.copy()
            except Exception as e:
                continue

        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = trial_h
            best_n = n

    # Final trust-region if available
    if len(best_h) > 0 and time.time() - start_time < budget_s * 0.99:
        try:
            h_final = best_h.copy()
            dx = 2.0 / len(h_final)
            w = len(h_final) / 2.0

            def objective(h):
                corr = np.correlate(h, 1 - h, mode="full")
                return -np.max(corr) * dx

            def constraint_eq(h):
                return np.sum(h) * dx - 1.0

            constraints = {'type': 'eq', 'fun': constraint_eq}
            bounds = [(0, 1) for _ in range(len(h_final))]

            result = minimize(objective, h_final, constraints=constraints,
                              bounds=bounds, method='SLSQP',
                              options={'maxiter': trust_region_iter, 'ftol': 1e-12, 'gtol': 1e-12})

            if result.success:
                final_h = result.x
                final_c5 = evaluate_h(final_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h[:]
                    best_n = len(final_h)
        except Exception as e:
            pass

    return list(best_h), best_c5, best_n
