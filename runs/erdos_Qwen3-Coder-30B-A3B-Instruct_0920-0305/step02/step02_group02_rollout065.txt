# Helper function to project onto capped simplex
def project_simplex(v, w, lbd=0.0, ubd=1.0):
    # v is the vector to project; w is the target sum
    # Project onto [lbd, ubd]^n and set sum to w
    sorted_v = np.sort(v)[::-1]  # descending
    n = len(v)
    # Find threshold tau
    cumsum = np.cumsum(sorted_v)
    tau = (cumsum - w) / np.arange(1, n + 1)
    # The largest index i such that x_i > tau_i
    i = np.argmax(sorted_v > tau)
    if i == 0 and sorted_v[0] <= tau[0]:
        # All values are <= tau_0, so project to threshold tau_0
        threshold = tau[0]
        x = np.clip(v - threshold, 0, 1)
        # Rescale to match integral constraint
        current_sum = np.sum(x)
        if abs(current_sum - w) > 1e-10:
            # Add residual to first component
            x[0] += (w - current_sum)
            x = np.clip(x, 0, 1)
    else:
        # Compute threshold as the i-th element
        threshold = tau[i]
        x = np.clip(v - threshold, 0, 1)
        # Rescale
        current_sum = np.sum(x)
        if abs(current_sum - w) > 1e-10:
            x[0] += (w - current_sum)
            x = np.clip(x, 0, 1)
    return x

def compute_correlation(h):
    """Compute all correlations r_k = w - S_k(h) using FFT"""
    n = len(h)
    # Zero-pad for full convolution
    padded_h = np.pad(h, (0, n), mode='constant')
    # fft of h and its conjugate
    H = fft(padded_h)
    # Compute auto-correlation
    S_full = ifft(H * np.conj(H)).real
    # Take non-negative lags
    S = S_full[n-1 : 2*n - 1]
    # Compute r_k = w - S_k
    r = w - S
    return r

def smooth_max(x, alpha=1.0):
    """Smooth log-sum-exp of x"""
    x = np.array(x)
    xmax = np.max(x)
    x_shift = x - xmax
    exp_terms = np.exp(alpha * x_shift)
    log_sum_exp = xmax + np.log(np.sum(exp_terms)) / alpha
    return log_sum_exp

def smooth_max_grad(r, alpha=1.0):
    """Compute gradient of smooth max given r (r_k = w - S_k)"""
    n = len(r)
    r = np.array(r)
    r_max = np.max(r)
    r_shift = r - r_max
    exps = np.exp(alpha * r_shift)
    prob = exps / np.sum(exps)
    # Gradient of each r_k
    grad_r = np.zeros_like(r)
    for k in range(n):
        if k == 0:
            # ∂r_0/∂h_j = 1 - 2 * h_j
            grad_r[k] = np.ones(n) - 2 * h
        else:
            # ∂r_k/∂h_j = -1 if j+k < n and j-k >= 0 else 0)
            grad_r[k] = np.zeros(n)
            grad_r[k,k:] = -1  # h_j+k terms
            grad_r[k,:-k] = grad_r[k,:-k] - 1  # h_j-k terms
    # Final gradient of smooth max
    g = np.sum(prob * grad_r, axis=0)
    return g

def evaluate_c5(h):
    """Compute C5 exactly using standard correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = dx * np.max(corr)
    return c5
