```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_simplex(y, S):
    """Project vector y onto the capped simplex: 0 <= x <= 1, sum x = S"""
    n = len(y)
    y = np.array(y)
    low, high = -np.max(y) - 1, np.max(y) + 1
    for _ in range(40):
        mid = (low + high) / 2
        x = np.clip(y - mid, 0, 1)
        f = np.sum(x) - S
        if f > 0:
            low = mid
        else:
            high = mid
    x = np.clip(y - low, 0, 1)
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    # Run SDP optimization first
    # List of small values of n to test
    n_list = [30, 35, 40, 45, 50]
    for n in n_list:
        # Check time limit
        if (time.time() - start_time) > 0.7 * budget_s:
            break
        try:
            # Create CVXPY variables
            X = Semidef(n)
            h = Variable(n)
            m = Variable()

            # Add constraints
            constraints = [
                X[i, i] == h[i] for i in range(n)  # Diagonal entries of X are h_i
            ]
            constraints.append(0 <= h)
            constraints.append(h <= 1)
            constraints.append(sum(h) == n / 2)
            # Add constraints for the r_s >= m
            for s in range(-(n-1), n):
                r_s = 0
                if s >= 0:
                    # sum_{i=0}^{n-1-s} X_{i, i+s}
                    r_s = sum(X[i,i+s] for i in range(n-s))
                else:
                    # sum_{i=-s}^{n-1} X_{i, i+s} = sum_{i=0}^{n-1+s} X_{i+s, i}
                    r_s = sum(X[i+s,i] for i in range(n+s))
                constraints.append(r_s >= m)
            
            # Objective
            objective = Maximize(m)
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, eps=1e-5, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_val = np.array(h.value).flatten()
                # Project to capped simplex
                h_val = project_simplex(h_val, n/2)
                # Evaluate C5
                c5_val = evaluate_c5(h_val)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_val.copy()
        except Exception as e:
            continue
            
    # If SDP failed to give better solution, use initial_h_values
    if best_h is None:
        best_h = np.array(initial_h_values)
        best_c5 = evaluate_c5(best_h)
        
    # Local refinement using Nelder-Mead
    # Optimize with a short local search
    if (time.time() - start_time) < 0.9 * budget_s:
        try:
            def obj(h_vec):
                return evaluate_c5(h_vec)
                
            # Project onto simplex after every iteration
            def refine(h_vec):
                h_proj = project_simplex(h_vec, len(h_vec)/2)
                return h_proj
                
            # Use Nelder-Mead directly with feasible projections
            res = minimize(obj, best_h, method='Nelder-Mead',
                           options={'maxiter': 1000, 'ftol': 1e-9, 'xatol': 1e-9})
            h_final = res.x if res.success else best_h
            h_final = refine(h_final)
            c5_final = evaluate_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
        except:
            pass

    # Perform upsampling to large resolution
    n_target = 180
    if (time.time() - start_time) < 0.95 * budget_s:
        # Interpolate from best_h to n_target
        h_upsampled = np.interp(np.linspace(0, 2, n_target), np.linspace(0, 2, len(best_h)), best_h)
        h_upsampled = project_simplex(h_upsampled, n_target / 2)
        c5_upsampled = evaluate_c5(h_upsampled)
        if c5_upsampled < best_c5:
            best_c5 = c5_upsampled
            best_h = h_upsampled.copy()
    else:
        best_h = project_simplex(best_h, n_target / 2)
        best_c5 = evaluate_c5(best_h)

    # Final time check
    if (time.time() - start_time) < 0.99 * budget_s:
        # Extra refinement on high-res vector
        try:
            def obj(h_vec):
                return evaluate_c5(h_vec)
            res = minimize(obj, best_h, method='Nelder-Mead', 
                           options={'maxiter': 1000, 'ftol': 1e-10, 'xatol': 1e-10})
            if res.success:
                h_final = project_simplex(res.x, n_target / 2)
                c5_final = evaluate_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
        except:
            pass

    return (best_h.tolist(), float(best_c5), n_target)
```
