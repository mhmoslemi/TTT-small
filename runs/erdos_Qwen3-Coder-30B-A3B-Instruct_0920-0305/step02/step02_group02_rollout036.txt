def compute_overlap(h):
    """Compute max overlap of h with (1 - h)"""
    h = np.asarray(h)
    n = len(h)
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1 - h, mode='full')
    # only take part where both overlap
    mid = len(corr) // 2
    overlap = corr[mid:]
    return np.max(overlap)

def compute_overlap_float(h):
    """Continuous version for evaluating"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mid = len(corr) // 2
    overlap = corr[mid:]
    return np.max(overlap) * dx

def sdp_relaxation(h0, n):
    """Solve SDP relaxation"""
    dx = 2.0 / n
    w = n / 2

    # SDP variable
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)

    # Constraints for SDP relaxation
    constraints = [
        X >> 0,
        cp.diag(X) == cp.multiply(h, h),
        cp.sum(h) == w,
        h >= 0,
        h <= 1,
    ]
    
    # Minimize max correlation (relax into sum over k)
    T = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if j >= i:
                T[i, j] = 1
    # For max_t that can be infeasible, now we're just minimizing sum of entries that are >= t
    
    # Since we're relaxing rank constraint, the best we can do with SDP is get a lower bound
    # we optimize the best t we can get from the relaxed constraint in the target problem
    # But since the constraint is ty to compute all correlations, we solve one reformulation
    # Let's make the minimal over all shifts, and minimize that
    # Let's keep things simple and detect frontier via empirical check.
    
    # Massage approach: recursive coloring of correlation indices to create small pseudodiesel
    # Same as old but smoother process of getting within range

    # For now, we just do the SDP once to get a good rough point
    obj = cp.Minimize(cp.sum(cp.multiply(X, T)))  # relu type approach
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver='SCS', verbose=False)
        if prob.status == cp.OPTIMAL:
            return h.value
    except:
        pass
    return h0.copy()

def round_sdp_solution(X, h_sdp, n):
    """Round the SDP solution by taking a random linear combination."""
    # Try Cholesky
    try:
        R = np.linalg.cholesky(X)
        g = np.random.randn(R.shape[0])
        h_round = R.T @ g
    except np.linalg.LinAlgError:
        # If cholesky fails, try eigendecomposition
        try:
            e_vals, e_vecs = eigh(X)
            h_round = e_vecs @ (np.sqrt(np.maximum(e_vals, 0)) * np.random.randn(n))
        except:
            return h_sdp.copy()
    
    # Project to feasible region: [0,1]^n with sum = w
    h_round = np.clip(h_round, 0, 1)
    sum_h = np.sum(h_round)
    scale = 1.0 / sum_h * (n / 2)
    h_round *= scale
    
    # Water-filling projection
    h_proj = project_to_unit_box_and_sum(h_round, n)
    return h_proj

def project_to_unit_box_and_sum(h, n):
    """Project h to [0,1]^n with sum = n/2 using water-filling."""
    w = n / 2.0
    h = np.array(h)
    sorted_h = np.sort(h)[::-1]
    # to find threshold tau, project from the largest values such that 
    # sum over all is exactly w
    tot = 0
    for i in range(len(sorted_h)):
        if tot + sorted_h[i] * (i + 1) <= w:
            tot += sorted_h[i]
        else:
            # Compute threshold
            diff = w - tot
            if i != 0:
                thresh = diff / (i + 1)
                for j in range(i):
                    if h[j] > thresh:
                        h[j] = thresh
                h[i] = thresh
                for j in range(i + 1, len(h)):
                    h[j] = 0
            else:
                thresh = w / len(h) if len(h) > 0 else 0
                h.fill(thresh)
            break
    else:
        # All values are equal to or below computed threshold
        if len(sorted_h) > 0:
            thresh = w / len(sorted_h)
            h.fill(thresh)
    return h

def fit_fourier_reconstruct(h, M=8):
    """Fit h as a truncated Fourier series."""
    n = len(h)
    freqs = np.arange(1, M + 1)
    C = np.zeros((n, 2 * M))
    
    for i in range(n):
        for m in range(1, M + 1):
            C[i, m - 1] = np.cos(np.pi * m * i / n)  # a_m
            C[i, m - 1 + M] = np.sin(np.pi * m * i / n)  # b_m
            
    # Solve least-squares with box constraints
    # Problem is of form: min || C @ theta - h ||^2 s.t. 0 <= h_i <= 1
    
    # Let's just do a non-linear least squares fit
    def objective(theta):
        h_recon = 0.5 + C @ theta
        h_recon = np.clip(h_recon, 0, 1)
        diff = h - h_recon
        return np.sum(diff**2)
        
    # Initial guess
    init = np.zeros(2 * M)
    # Solve optimization
    result = minimize(objective, init, method='L-BFGSB', bounds=[(-2, 2)] * (2 * M), options={'maxiter': 500})
    if result.success:
        theta = result.x
    else:
        theta = init
    
    # Reconstruct h
    h_recon = 0.5 + C @ theta
    h_recon = np.clip(h_recon, 0, 1)
    return h_recon, theta

def refine_on_fourier(theta, M, n):
    """Refine h from Fourier coefficients."""
    dx = 2.0 / n
    freqs = np.arange(1, M + 1)
    C = np.zeros((n, 2 * M))
    for i in range(n):
        for m in range(1, M + 1):
            C[i, m - 1] = np.cos(np.pi * m * i / n)
            C[i, m - 1 + M] = np.sin(np.pi * m * i / n)
    
    def fn(theta):
        h = 0.5 + C @ theta
        h = np.clip(h, 0, 1)
        return np.max(np.correlate(h, 1 - h, mode='full')[n//2:])
    
    def grad(theta):
        h = 0.5 + C @ theta
        h = np.clip(h, 0, 1)
        h = h.astype(float)
        # invert to replace deep sieve into composed convolution operation
        conv = np.correlate(h, 1 - h, "full")
        conv = np.array(conv)
        max_k = np.argmax(conv)
        # So a fixed pattern of shifts will trace difference (first is not always this)
        # take gradient under subproblem
        g = np.zeros_like(theta)
        for i in range(n):
            for j in range(n):
                k = j - i  
                if k >= 0:
                    if k != max_k:
                        g += (1 - h[j]) * C[i, :]
                else:
                    k = -k
                    if k != max_k:
                        g += (1 - h[i]) * C[j, :]
        return g
    
    # continuation optim min with stepsize
    step_size = 0.1
    for i in range(1000):
        h = 0.5 + C @ theta
        h = np.clip(h, 0, 1)
        f = np.max(np.correlate(h, 1 - h, mode='full')[n//2:])
        g = grad(theta)
        g /= np.linalg.norm(g) + 1e-8
        h_new = 0.5 + C @ (theta - step_size * g)
        h_new = np.clip(h_new, 0, 1)
        if np.sum(h_new) < n / 2 + 1e-8:
            h_new *= (n / 2) / np.sum(h_new)
        
        #_POLICY: if error dominates or not improving, go to next candidate
        theta = theta - step_size * g
        step_size *= 0.99 if i % 10 != 0 else 0.95
    
    h_final = 0.5 + C @ theta
    h_final = np.clip(h_final, 0, 1)
    return h_final

def evaluate_final(h):
    """Use discrete method for final eval"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    corr = corr[len(corr) // 2:]
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    iterations = 0
    
    # Load initial
    initial_h = np.array(initial_h_values)
    data_found = False
    
    # Only pick large enough sizes to fit full run
    for n, dx in zip(n_candidates, dxs):
        if time.time() - start_time > budget_s - 10:
            break
        # Skip large SDP if time is tight
        if n > 64:
            # interpolate or mimic SDP without it?
            tries = 10  # fewer multi-seeds if small 
            h0 = initial_h[:n].copy()
            h0 /= np.sum(h0) * dx
            h0 = np.clip(h0, 0, 1)
            m = 8
            for seed_round in range(10):
                h_f = h0.copy()
                h_f, _ = fit_fourier_reconstruct(h_f, m)
                h_final = refine_on_fourier(np.zeros(2*m), m, n)
                
                c5 = evaluate_final(h_final)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final.copy()
                    best_n = n
                if c5 < target_bound:
                    data_found = True
                    break
        else:
            # Using SDP at smaller sizes
            tries = 20
            for seed_round in range(tries):
                h0 = initial_h[:n].copy()
                h0 /= np.sum(h0) * dx
                h0 = np.clip(h0, 0, 1)
                h_sdp = sdp_relaxation(h0, n)
                if h_sdp is None or np.any(np.isnan(h_sdp)):
                    h_sdp = h0.copy()
                # Rounding
                try:
                    # Schur complement, or just identity
                    X_sdp = np.outer(h_sdp, h_sdp)
                    h_round = round_sdp_solution(X_sdp, h_sdp, n)
                except:
                    h_round = h0.copy()
                # Fit Fourier
                h_f, _ = fit_fourier_reconstruct(h_round, M=8)
                # Refine final
                h_final = refine_on_fourier(np.zeros(16), 8, n)
                c5 = evaluate_final(h_final)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final.copy()
                    best_n = n
                if c5 < target_bound:
                    data_found = True
                    break
    
    # If still not found, just afresh optimizing
    if best_c5 == float('inf'):
        h0 = initial_h
        h0 = np.clip(h0, 0, 1)
        sum_h = np.sum(h0)
        scale = (len(h0) / 2) / sum_h
        h0 *= scale
        tries = 30
        for i in range(tries):
            if time.time() - start_time > budget_s:
                break
            h = h0 + np.random.normal(0, 0.01, len(h0))
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (len(h) / 2) / sum_h
            h *= scale
            # refine a bit
            h_f, _ = fit_fourier_reconstruct(h, M=6)
            # Re-optimize
            h_final = refine_on_fourier(np.zeros(12), 6, len(h))
            c5 = evaluate_final(h_final)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
                best_n = len(h_final)

    return list(best_h), best_c5, best_n
