def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_sum(h, target_sum):
    """Project h onto [0, 1]^n and then adjust to make sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)  
    # Use bisection to find lambda such that sum(clip(h-lambda,0,1)) = target_sum
    # The function f(lambda) = sum(clip(h-lambda,0,1)) is decreasing in lambda
    # Start values
    a, b = -100, 100
    f_a = np.sum(np.clip(h - a, 0, 1)) - target_sum
    f_b = np.sum(np.clip(h - b, 0, 1)) - target_sum
    
    if f_a * f_b > 0:
        # This shouldn't happen with extreme values, but just in case,
        # if signs are same (f_a >=0, f_b >=0) then we want gamma to be smaller than a
        # but if f_a <0, f_b <0, we want larger gamma than b.
        # However, the bounds are too big, we want to shrink their range.
        # Return naive clipped h scaled if sum different
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h = h * (target_sum / current_sum)
        return h

    for _ in range(50):
        mid = (a + b) / 2
        f_mid = np.sum(np.clip(h - mid, 0, 1)) - target_sum
        if f_mid > 0:
            a = mid
        else:
            b = mid
        if abs(b - a) < 1e-10:
            break
    
    lam = (a + b) / 2
    h = np.clip(h - lam, 0, 1)
    return h

def objective_smooth(h_vals, β, all_O_s, all_weights):

    # Compute all O_s = sum h_i h_{i+s} (i =0 to n-1-s)
    # This is a bit expensive - do it in separate function (avoiding recompute if possible)
    # So pass in
    # O_s = [ O_s for s in range(n) ]  needed to define objective  
    c_s_vals = np.array(all_O_s)  # should be [O_s * dx for s]
    c_s_vals = c_s_vals * dx  # already in
    t = - (1.0 / β) * np.log(np.sum(np.exp(-β * c_s_vals)))
    return t

def gradient_smooth(h_vals, β, all_O_s, all_weights):
    """Return gradient of smooth objective w.r.t. h."""  
    h = np.array(h_vals) 
    n = len(h)
    current_g = np.zeros_like(h)  
    # For all s, w_s = exp(-β C_s) / sum(exp(-β C_t))  
    # This when we apply it w.r.t. h = sum_s w_s * D C_s / D h  
    # Want D O_s / D h_i for fixed s 
    # But for shift s, D O_s / D h_i = sum_j D(h_i h_{i+s}) / D h_i =  
    # if we consider h_i appears only in h_j = h_i if j = i so D(h_i h_{i+s})/D h_i = h_{i+s}  
    # → we get contributions to indices i and i+s:  
    # If shift s, contribution to derivative of O_s h is set of pairs {i,i+s}  
    # So structure: 
    # h^T * A_s h where A_s is shift matrix (with off-diagonal terms in pattern)
    # A_s[i, j] = 0.5 if |i-j| = s, 0 else  
    # Grad O_s = 2 A_s h  
    # Grad C_s = dx * 2 A_s h  
    # Grad Phi = sum_s w_s * Grad C_s = 2*dx * sum_s w_s * A_s h  
    # So it suffices to compute 2*dx * w_s * A_s h for each s, then sum
    
    if len(all_weights) == 0:
        # Be careful with computing w_s first below
        # Let's do the weight computation here the same way as in iterative
        # Compute all C_s 
        c_s_vals = [all_O_s[i] * dx for i in range(len(all_O_s))]
        # Exponentiate in a numerically stable way  
        c_minus_max = np.array(c_s_vals) - np.max(c_s_vals)
        w = np.exp(-β * c_minus_max)
        w = w / np.sum(w)
        all_weights = w
    else:
        w = all_weights
        
    # Computing 2*dx * w * A_s h for each s now.  
    # A_s h (n floats) = sum_ij h_i*A_s[i,j] = for each i, its contribution (if i-s valid) + (if i+s valid)
    # So for s, get A_s h of size n:
    # First collect most in one loop over all s = 0..n-1
    for s in range(n):
        # val = (h[i] + h[i+s]) * 0.5 for I s, i=0...n-1-s
        # shall compute (A_s h_s)[i]  
        # Use a vectorized approach:
        if s == 0:
            # This case is A_s h = h_i h_i = h_i^2 = h point-wize.  
            # But still putting 1 in proper field for gradient logic:
            # The A_s = diag matrix? No, not with 0.5 off-diag.  
            # Actually for s=0, it's = [h_0 h_0, h_1 h_1, ...] = h^2
            # Gradient for h_i h_i = 2 h_i  
            # So grad = 2 * h.  
            # But note this effectively means A_0 = diag with 1 on diagonal
            # Or more logically, this is sum h_i h_{i} = sum h_i^2  
            # Gradient w.r.t. h_i of sum h_i^2 = 2 h_i  
            # So directly: 
            current_g += 2 * w[s] * h * dx  
        else:
            # Shift: make A_s h which uses h_{i+s}, h_{i-s} for valid indices  
            # Wrk h_{i-s} and h_{i+s} for all i where both valid  
            # Taking A_s h
            g_s = np.zeros_like(h)
            # This is equivalent to:
            # shift A_s h by +s, so it offsets over by s → g_s = A_s h  
            # where (A_s h)_i is h_{i+s} for i = 0 to n-1-s   and h_{i-s} for i = s to n-1  
            # so first compute h[index] corresponding to indices where i = 0 to n-1-s for forward component so +s index  
            # We want to compute sums of h_{i+s} over indices i = 0 to n-1-s
            # and h_{i-s} over i = s to n-1.  
            # So effectively shift h_list by +s and take elements.  
            # Specifically:  
            # g_s[i] = 0.5 * (h[i-s] if i >= s else 0) + 0.5 * (h[i+s] if i+s < n else 0)  
            # In NumPy:
            # Make two small arrays:
            # s=0 create vals for each i of in regards to forward/ backward parameters  
            start_w = 0
            start_h = 0 
            g_s = np.zeros_like(h)  
            g_s[s:] += 0.5 * h[:-s]  # h_{i-s} for valid i  
            g_s[:-s] += 0.5 * h[s:]  # h_{i+s} for valid i  
            # Convolution on shift matrix product is nothing simple, but what does general g_s = A_s h evaluate to?
            # Slot at index i (i=0..n-1) sums:
            # for i to i+s valid, i += 1, leverage +s: i=0 -> i=0,0 maps 1 to 0 and 0 to 0 
            # take first in i to n-1-s: 
            # g_s[i] = 0.5 h_{i+s} + 0.5 h_{i-s}  
            # However this is in an elegant vectorized way of |
            # a)  g_s[s:n] += 0.5 h[:n-s]   for h[i-s]
            # b)  g_s[:n-s] += 0.5 h[s:]   for h[i+s]
            # Different value for start or data ends.  
            current_g += 2 * dx * w[s] * g_s  
    return current_g  

def optimize_for_n(n):
    """Return (best_h, best_c5, n_points) for given n."""
    global dx
    dx = 2.0 / n
    
    # Valid points for multiple restarts:  
    # 1. linear interpolation of initial_h_values to size n  
    if len(initial_h_values) < n:
        h_interp = np.interp(
            np.linspace(0, 1, n), 
            np.linspace(0, 1, len(initial_h_values)),
            initial_h_values
        )
    else:
        h_interp = np.array(initial_h_values[:n]) 
    
    # 2. Pre-encoded binary pattern (fix weight)  
    h_bin = np.zeros(n)
    # Exactly n/2 ones 
    ones_loc = np.random.choice(n, n//2, replace=False)
    h_bin[ones_loc] = 1.0  
    # Project onto box and sum  
    h_bin = project_onto_box_and_sum(h_bin, n/2)  
    
    # 3. Random uniform + projection  
    h_rand = np.random.rand(n)
    h_rand = project_onto_box_and_sum(h_rand, n/2)  
      
    # 4. Scaled from previous best (if any exist in setup time, programmatically could use)
    # But in this specific function, we don't have global scope access to "previous best"  
    # Just minimize list as current 
    
    init_list = [h_interp, h_bin, h_rand]
    best_h = None
    best_c5 = float('inf')
    
    # Timing for inner search
    start_inner = time.time()
    
    # Use multi-start, 3 at least.  
    for h_start in init_list:
        # Inner gradient steps  
        h_current = h_start.copy()
        # Project initial  
        h_current = project_onto_box_and_sum(h_current, n/2)  
        
        # Ensure we do tracking best  
        # C5 direct tracking instead of passing around 
        c5_current = evaluate_h(h_current) 
        
        # β schedule
        β_schedule = [5, 15, 30, 60, 120]
        K_max = 200   # For inner loop only 
        rescaling = 0.5 # line search
        for β in β_schedule:
            for k in range(K_max): 
                if time.time() - start_inner > 0.9 * budget_s:
                    break    
                
                # Compute all C_s (using evaluate_h) for gradient computation  
                # and exact objective evaluation  
                h = np.array(h_current)  
                corr = np.correlate(h, 1 - h, mode="full")  # even slower access 
                
                # The almost** full correlation computed  
                # Extract O_s using a direct short version.  
                O_s = []  
                for s in range(n):
                    # Sum from i = 0 to n-s-1 of h_i h_{i+s}
                    o = 0.0
                    for i in range(n - s):
                        o += h[i] * h[i + s]
                    O_s.append(o)  
                
                # Now convert to c_s with dx factor 
                C_s = [o * dx for o in O_s]  
                # compute soft weights for current O_s or directly for C_s  
                C_s_np = np.array(C_s)  
                C_minus_max = C_s_np - np.max(C_s_np)  
                w_np = np.exp(-β * C_minus_max)  
                w_np = w_np / np.sum(w_np)  
                
                # Gradients for smooth likelihood  
                g_h = gradient_smooth(h_current, β, O_s, w_np)  
                
                # Compute also current max
                L_s = 1 - np.array(C_s)  
                t_current = np.max(L_s) 
                alpha = 0.5
                
                # Line search -- heuristic single not repeated too much
                for ls_try in range(5):  
                    h_proposed = h_current + alpha * g_h  
                    h_proposed = project_onto_box_and_sum(h_proposed, n/2)  
                    # Ideal evaluate corresponding to h_proposed, final H, L_s  
                    # Update final measure  
                    corr_prop = np.correlate(h_proposed, 1 - h_proposed, mode="full")  
                    prop_O = []  
                    for s in range(n):
                        o = 0.0
                        for i in range(n - s):
                            o += h_proposed[i] * h_proposed[i + s]
                        prop_O.append(o)  
                    prop_C = [o * dx for o in prop_O]  
                    prop_L = [1 - c for c in prop_C]  
                    t_proposed = np.max(prop_L)  
                    if t_proposed < t_current:
                        h_current = h_proposed
                        c5_current = t_proposed
                        break
                    alpha *= 0.5 
                    if alpha < 1e-10:
                        break  
                
                # Record after update  
                if c5_current < best_c5:
                    best_c5 = c5_current  
                    best_h = h_current.copy()
                
                # Check convergence
                if c5_current < target_bound:
                    break
                
                if k > 50 and k % 20 == 0:
                    if k > 50:  # converge test is used after 50 iters
                        pass  # No easy check here
                        
            if c5_current < target_bound:
                break  
            
            # After leaving β loop, we could add some variable animation if way too much BOT  
            
        if c5_current < best_c5:
            best_c5 = c5_current  
            best_h = h_current.copy()
            
    # Final refinement with SLSQP  
    if best_h is not None:  
        bounds = [(0, 1) for _ in range(n)]
        def obj_f(x):
            return evaluate_h(x)
        try:
            res = minimize(obj_f, best_h, method='SLSQP', bounds=bounds, 
                           options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8})
            if res.success:
                best_h = res.x
                # double check best c5
                # SLSQP doesn't guarantee bounds or initial, we should re check C5  
                best_c5 = evaluate_h(best_h)  
        except:
            pass

    return list(best_h), best_c5, n
