def project_onto_capped_simplex(v, target_sum):
    """Project vector v onto capped simplex: 0 <= v_i <= 1, sum(v) = target_sum."""
    v = np.array(v).copy()
    v = np.clip(v, 0, 1)
    total = np.sum(v)
    if abs(total - target_sum) < 1e-12:
        return v
    elif total > target_sum:
        # Reduce entries from largest to smallest
        sort_idx = np.argsort(v)[::-1]
        remaining = target_sum
        for i in sort_idx:
            if v[i] <= remaining:
                remaining -= v[i]
                v[i] = 0
            else:
                v[i] = remaining
                remaining = 0
                break
    else:
        # Increase entries from smallest to largest
        sort_idx = np.argsort(v)
        remaining = target_sum - total
        for i in sort_idx:
            if remaining <= 0:
                break
            inc = min(1 - v[i], remaining)
            v[i] += inc
            remaining -= inc
    return v

def compute_correlations(h):
    """Compute all correlations r_k(h) = sum_i h_i * (1 - h_{i+k}) for k=0..n-1."""
    n = len(h)
    corr = np.zeros(n)
    h1 = 1 - h  # easier to compute correlations now
    for k in range(n):
        corr[k] = np.sum(h * np.roll(h1, k))
    return corr

def gradient_of_corr(h, k):
    """Compute gradient of correlation r_k = sum_i h_i * (1 - h_{i+k})."""
    n = len(h)
    grad = np.zeros(n)
    h1 = 1 - h
    # For each i, the partial derivative depends on h_{i+k} and h_{i-k}
    # This handles boundary conditions by truncation
    for i in range(n):
        grad[i] = 1.0
        if i + k < n:
            grad[i] -= h1[i + k]  # minus sign in corr function
        if i - k >= 0:
            grad[i] -= h1[i - k]
    return grad

def grad_logsumexp(h, alpha=20.0):
    """Compute gradient of smooth surrogate f_alpha(h) = (dx/alpha)*log(sum_k exp(alpha*r_k(h)))."""
    corr = compute_correlations(h)
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exps = np.exp(alpha * shifted)
    sum_exps = np.sum(exps)
    weights = exps / sum_exps
    grad_total = np.zeros_like(h)
    for k in range(len(h)):
        grad_corr_k = gradient_of_corr(h, k)
        grad_total += weights[k] * grad_corr_k
    return (dx / alpha) * grad_total

def backtracking_line_search(h, grad, alpha=20.0, eta=0.5, c=1e-4):
    """Simple backtracking line search for step size."""
    t = 1.0
    while True:
        h_new = project_onto_capped_simplex(h - t * grad, w)
        corr_old = compute_correlations(h)
        corr_new = compute_correlations(h_new)
        left = np.log(np.sum(np.exp(alpha * corr_new))) if alpha > 0 else np.max(corr_new)  # Approximate f_alpha
        right = np.log(np.sum(np.exp(alpha * corr_old))) if alpha > 0 else np.max(corr_old)
        if left <= right - c * t * np.dot(grad, grad):
            return t
        t *= eta

def projected_grad_descent(h0, dx, w, alpha0=1.0, alpha_factor=2.0, alpha_max=300.0, max_iters=5000):
    """Run accelerated projected gradient descent for maximizing t with smooth surrogates."""
    h = h0.copy()
    t = 1.0
    alpha = alpha0
    old_f = -np.inf
    for it in range(max_iters):
        r_vals = compute_correlations(h)
        if alpha > 0:
            max_r = np.max(r_vals)
            shifted = r_vals - max_r
            f = (dx / alpha) * np.log(np.sum(np.exp(alpha * shifted)))
        else:
            f = np.max(r_vals)
        if abs(f - old_f) < 1e-6 or abs(np.max(r_vals) - np.min(r_vals)) < 1e-6:
            break
        old_f = f
        g = grad_logsumexp(h, alpha)
        if np.linalg.norm(g) < 1e-8:
            break
        # Nesterov-style extrapolation
        h_hat = project_onto_capped_simplex(h - g, w)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h = h_hat + ((t - 1.0) / t_new) * (h_hat - h)
        h = project_onto_capped_simplex(h, w)
        t = t_new
        if alpha < alpha_max:
            alpha *= alpha_factor
    return h, evaluate_c5(h)

def initialize_feasible(n, method="random"):
    """Initialize a feasible vector, method can be 'random', 'binary', 'three_block'."""
    if method == "random" or method == "binary":
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        h = project_onto_capped_simplex(h, w)
        return h
    elif method == "three_block":
        # Create a piecewise structure: left (1), middle (some p), right (0)
        # length of left and right is fixed percent of total
        a = np.random.rand() * 0.3
        b = a + 0.3 + np.random.rand() * 0.2
        if b > 1.0:
            b = 1.0
        p = (w - a * n / 2.0) / ((b - a) * n / 2.0)
        if p > 1.0:
            p = 1.0
        h = np.ones(n)
        for i in range(n):
            x = i * dx
            if x < a:
                h[i] = 1.0
            elif x >= b:
                h[i] = 0.0
            else:
                h[i] = p
        h = project_onto_capped_simplex(h, w)
        return h
    else:
        # Default: random (projected)
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        h = project_onto_capped_simplex(h, w)
        return h

    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    try:
        result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': max_iter})
        if result.success:
            h_refined = result.x
            h_refined = project_onto_capped_simplex(h_refined, w)
            c5 = evaluate_c5(h_refined)
            return h_refined, c5
    except Exception:
        pass
    return h, evaluate_c5(h)
