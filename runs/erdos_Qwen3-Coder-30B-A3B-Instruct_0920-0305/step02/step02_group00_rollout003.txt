import numpy as np
from scipy.optimize import minimize
import time

# constants
n_points = 56      # even number, moderate size
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_hyperplane(v, target_sum):
    """Projects vector v onto [0,1]^n and adjusts to sum to target_sum."""
    # Ensure values are in [0, 1]
    v = np.clip(v, 0, 1)

    # Find lambda such that sum of clip(v - lambda, 0, 1) equals target_sum
    def f(lam):
        return np.sum(np.clip(v - lam, 0, 1)) - target_sum

    # bracketing step: find interval [a, b] such that f(a)*f(b) <= 0
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 1
        b += 1

    # binary search for root
    tol = 1e-12
    while abs(b - a) > tol:
        c = (a + b) / 2
        if f(c) == 0:
            break
        if f(a) * f(c) < 0:
            b = c
        else:
            a = c

    lambda_opt = (a + b) / 2
    result = np.clip(v - lambda_opt, 0, 1)
    return result

    if start_method == "interpolated":

        h = np.array(initial_h_values)
        if len(h) != n:
            # Resample to desired n using linear interpolation
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_old, h)
    elif start_method == "random_binary":
        h = np.zeros(n)
        ones = np.random.choice(n, size=n//2, replace=False)
        h[ones] = 1.0
    elif start_method == "random_uniform":
        h = np.random.rand(n)
    else:
        raise ValueError("Unknown initialization method")

    # normalize sum to target
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        h *= (1 / current_sum)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_used = n_points

    # Try different n values if allowed to do so
    n_values = [n_points]
    
    for n in n_values:
        dx_local = 2.0 / n
        target_sum = 1.0 / dx_local  # This matches sum(h) = n/2 when dx = 2/n

        # Set up for optimization
        restarts = 5  # keep low to ensure quick exploration
        
        for restart in range(restarts):
            # Three different methods for generating h0
            methods = [
                "interpolated",
                "random_binary",
                "random_uniform"
            ]
            
            h0 = initialize_h(methods[restart % len(methods)], n)
            h_current = h0.copy()
            c5_current = evaluate_h(h_current)
            
            # Store best seen so far for this restart
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                n_used = n

            # Perform multi-start gradient descent with soft min objective
            # For β schedule
            beta_schedule = [5.0, 15.0, 30.0, 60.0, 120.0]
            max_iter_per_beta = 500
            max_stagnation = 20
            alpha0 = 0.1

            # inner loop per beta
            for beta in beta_schedule:
                # check allowed time
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                stagnation_counter = 0
                prev_c5 = c5_current
                
                # Perform iterations
                for t in range(max_iter_per_beta):
                    if time.time() - start_time > 0.95 * budget_s:
                        break
                        
                    # If no improvement in recent steps break out early
                    if stagnation_counter > max_stagnation:
                        break
                        
                    # Compute C_s measures for each shift:
                    # We compute all of them once by convolution
                    h = h_current
                    corr = np.correlate(h, 1 - h, mode="full")
                    C_s = corr * dx   # already have all unique shifts
                    
                    # compute weights
                    C_s_normalized = C_s - np.max(C_s)   # avoid numerical overflow
                    exp_neg_beta_C = np.exp(-beta * C_s_normalized)
                    weights = exp_neg_beta_C / np.sum(exp_neg_beta_C)

                    # gradient is 2dx * sum(weights[i] * A_i * h)
                    # For fast computation with shift matrices,
                    # we will vectorize shift operators using NumPy rotation
                    
                    # Skip shift s=0 for cumulative gradient (simplified implementation)
                    # Build full gradient by sum over each shift s
                    # Shifting helper using np.roll
                    g = np.zeros(n)
                    for s in range(n):
                        if s == 0:
                            # corresponds to sum_i h_i^2 term
                            g += 2.0 * h
                        else:
                            h_shift_forward = np.roll(h, -s, axis=0)
                            h_shift_backward = np.roll(h, s, axis=0)
                            contrib = (h_shift_forward + h_shift_backward) * 0.5
                            g += weights[s] * contrib
                    
                    # Scale by dx
                    g = 2.0 * dx_local * g

                    # now perform descent: line search
                    alpha = alpha0
                    descent_tried = False
                    step_sizes_tried = 0
                    
                    while step_sizes_tried < 5 and not descent_tried:
                        h_new = h_current + alpha * g
                        h_new = project_onto_box_and_hyperplane(h_new, target_sum)
                        
                        # check whether this move improved the objective
                        c5_new = evaluate_h(h_new)
                        
                        if c5_new < c5_current:
                            h_current = h_new
                            c5_current = c5_new
                            descent_tried = True
                            stagnation_counter = 0
                        else:
                            alpha *= 0.5
                            step_sizes_tried += 1
                            stagnation_counter += 1

                    # After one iteration per beta, check if we improve
                    if abs(c5_current - prev_c5) < 1e-10:
                        stagnation_counter += 1
                    else:
                        stagnation_counter = 0
                    prev_c5 = c5_current

                    if (c5_current < best_c5):
                        best_h = h_current.copy()
                        best_c5 = c5_current
                        n_used = n

        if best_c5 < target_bound:
            break

    # Final check using scipy's own SLSQP to refine small final improvements
    if best_h is not None and time.time() - start_time < 0.9 * budget_s:
        bounds = [(0, 1) for _ in range(n_used)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        try:
            result = minimize(evaluate_h, best_h, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                refined_c5 = evaluate_h(result.x)
                if refined_c5 < best_c5:
                    best_h = result.x
                    best_c5 = refined_c5
        except Exception:
            pass
            
    return list(best_h), best_c5, n_used
