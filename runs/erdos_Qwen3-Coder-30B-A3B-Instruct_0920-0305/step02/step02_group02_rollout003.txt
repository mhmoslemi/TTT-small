import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def project_onto_capped_simplex(x, target_sum):
    """Project vector x onto the capped simplex {x : 0 <= x_i <= 1, sum x_i = target_sum}."""
    x = np.array(x, dtype=float)
    n = len(x)
    # Ensure bounds
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-12:
        return x
    # Use water filling
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    # Find threshold
    i = 0
    while i < n and x_sorted[i] > (target_sum - cumsum[i]) / (i + 1):
        i += 1
    if i == 0:
        threshold = target_sum / n
        x = np.full_like(x, threshold)
    else:
        threshold = (target_sum - cumsum[i - 1]) / i
        x = np.clip(x - threshold, 0, 1)
    # Adjust for sum
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) > 1e-12:
        diff = target_sum - current_sum
        # Find smallest non-zero element to add diff to
        nonzero = np.where(x > 1e-12)[0]
        if len(nonzero) > 0:
            idx = nonzero[0]
            x[idx] += diff
        else:
            # all elements were zero, just distribute to one
            x[0] += diff
    return x

def compute_correlations(h, dx):
    """Compute all correlations r_k(h) = w - sum h[i] h[i+k] for k=0..n-1."""
    n = len(h)
    w = np.sum(h) * dx
    h_full = np.concatenate([h, np.zeros(n-1)])
    h_flip = np.concatenate([np.zeros(n-1), h])
    corr = np.correlate(h_full, h_flip, mode='valid')[:n]  # Or via fft if faster.
    r = w - corr * dx
    return r

def compute_correlations_fast(h, dx):
    """Use FFT for computing correlations."""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant')
    corr_fft = ifft(fft(h_padded) * fft(h_padded).conj())
    corr = np.real(corr_fft[:n])
    w = np.sum(h) * dx
    r = w - corr * dx
    return r

def compute_gradient(h, r, alpha, dx):
    """Compute the gradient of the log-sum-exp approximation."""
    n = len(h)
    r_max = np.max(r)
    r_shift = r - r_max
    exp_r = np.exp(alpha * r_shift)
    Z = np.sum(exp_r)
    weights = exp_r / Z
    # Gradient of log-sum-exp
    # ∇f_α = dx * Σ_k w_k * ∇r_k
    grad = np.zeros(n)
    for k in range(n):
        if k == 0:
            # This is ∂r₀/∂h_i = 1 - 2 h_i
            grad += -2 * weights[k] * h
        else:
            # For h_{i+k} and h_{i-k}, equal contributions
            # But here we are doing vectorized gradient
            # ∂r_k / ∂h_i = 1 - h_{i+k} - h_{i-k}
            grad += weights[k] * (np.ones(n) - np.roll(h, k) - np.roll(h, -k))
    grad += weights[0]  # add 1 for the first term
    grad = dx * grad
    return grad

def backtracking_line_search(h, g, f_old, alpha, dx, w, max_backtrack=10, c=1e-4):
    """Backtracking line search to find step size η."""
    # Armijo condition: f(h - ηg) <= f(h) - c * η * g^T g
    eta = 1.0
    f_new = np.inf
    for _ in range(max_backtrack):
        h_new = h - eta * g
        h_new = project_onto_capped_simplex(h_new, w)
        r_new = compute_correlations_fast(h_new, dx)
        f_new = (dx / alpha) * np.log(np.sum(np.exp(alpha * (r_new - np.max(r_new)))))
        if f_new <= f_old - c * eta * np.dot(g, g):
            return eta, h_new
        eta *= 0.5
    return eta, h_new

def accelerated_projected_gradient(h0, dx, w, alpha=1, alpha_max=300, max_iters=5000):
    """Accelerated projected gradient descent for smooth surrogate."""
    h = h0.copy()
    # initial step size & momentum
    t = 1
    y = h.copy()
    f_old = np.inf
    for k in range(max_iters):
        # Compute r and f
        r = compute_correlations_fast(h, dx)
        f = (dx / alpha) * np.log(np.sum(np.exp(alpha * (r - np.max(r)))))
        if abs(f - f_old) < 1e-6:
            break
        f_old = f
        # Compute gradient
        g = compute_gradient(h, r, alpha, dx)
        # Backtracking line search
        eta, h_candidate = backtracking_line_search(h, g, f, alpha, dx, w)
        # Nesterov extrapolation
        h_new = project_onto_capped_simplex(h_candidate, w)
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y_new = h_new + ((t - 1) / t_new) * (h_new - h)
        h = h_new
        y = y_new
        t = t_new
    # Final correlate for precise C5 value
    final_r = compute_correlations_fast(h, dx)
    c5 = np.max(final_r)  # Our surrogate approximates max(r_k) = max( w - S_k )
    return h, c5

def random_initial_h(n, w):
    """Generate a random feasible h vector."""
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, w)
    return h

def binary_initial_h(n, w):
    """Generate a binary 0/1 initial vector with sum w."""
    h = np.zeros(n)
    indices = np.random.choice(n, int(w), replace=False)
    h[indices] = 1
    h = project_onto_capped_simplex(h, w)
    return h

def initialize_h(n, w, seed=42):
    """Choose initial h randomly from various strategies."""
    np.random.seed(seed)
    strategies = [
        lambda: random_initial_h(n, w),
        lambda: binary_initial_h(n, w),
    ]
    strategy = np.random.choice(strategies)
    return strategy()
