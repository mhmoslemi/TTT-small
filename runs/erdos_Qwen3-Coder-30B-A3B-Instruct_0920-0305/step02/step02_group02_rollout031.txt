def grad_S_k(h, k):
    """Compute gradient of S_k w.r.t h"""

            g[:n-k] += h[k:]   # h_i contribution: derivative includes h_{i+k}
            g[k:] += h[:n-k]   # h_i contribution: derivative includes h_{i-k}
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project v onto the simplex with bounds [lower, upper] and sum = target_sum"""
    n = len(v)

        raise ValueError("Invalid target_sum")
    
    # Shift to work with [0, 1] bounds (subtract lower)
    u = v - lower

    # Use water-filling algorithm
    sorted_u = np.sort(u)[::-1]
    cumsum = np.cumsum(sorted_u)
    # Find threshold
    threshold = 0
    for i in range(n):
        if sorted_u[i] - (target_sum_norm - cumsum[i]) / (i + 1) >= 0:
            threshold = sorted_u[i] - (target_sum_norm - cumsum[i]) / (i + 1)
    
    x = np.maximum(u - threshold, 0)
    x = np.minimum(x, upper - lower)
    return x + lower
