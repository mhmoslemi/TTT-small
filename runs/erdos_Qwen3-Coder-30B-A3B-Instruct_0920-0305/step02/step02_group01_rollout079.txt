import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def capped_simplex_projection(g, sum_constraint, bounds=(0, 1)):
    """Project gradient g onto the feasible set with sum constraint.
       Returns h that minimizes ||h - g||^2 subject to bounds and sum."""
    # Sort indices by descending gradient values (descending order)
    idxs = np.argsort(-g)
    
    # Initialize solution
    h = np.zeros_like(g)
    
    # Start with the largest gradient components
    remaining_sum = sum_constraint
    
    for i in idxs:
        if remaining_sum <= 0:
            break
        # The value we want to assign to h[i] = min(1, max(0, g[i]))
        # But adjusted to maintain the sum constraint
        hi = min(bounds[1], max(bounds[0], g[i]))
        if hi <= remaining_sum:
            h[i] = hi
            remaining_sum -= hi
        else:
            h[i] = remaining_sum
            remaining_sum = 0
    
    # If we still have mass, distribute it to the best remaining entries
    # but more importantly ensure the sum constraint is exactly maintained
    total_assigned = np.sum(h)
    if abs(total_assigned - sum_constraint) > 1e-10:
        adjusted_h = h.copy()
        # Rescale to match exactly
        if total_assigned > 0:
            adjusted_h *= sum_constraint / total_assigned
        else:
            adjusted_h.fill(0)
            # Not better to improve below: rather ensure exact sum
            if sum_constraint > 0:
                # Fill up the top components
                sorted_idx = np.argsort(-g)
                for idx in sorted_idx[:int(sum_constraint)]:
                    adjusted_h[idx] = 1.0
        h = adjusted_h
    
    return h

def dc_c5_optimization(h0, n_points, dx, max_iter=3000, tol=1e-10):
    """Difference of Convex (DC) optimization with LP subproblems."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    recent_improvements = 0
    convergence_threshold = 30
    
    for iteration in range(max_iter):
        # Compute all correlation values for all shifts (this is O(n^2) but manageable)
        # Note FFT-based approach is more efficient for large n or when many shifts matter,
        # but for fixed n=120 this simple correlation works fine
        
        # We need: C5 = max_k Int h(x) (1 - h(x+k)) dx = max_k (sum over i: h[i] * (1 - h[i+k])
        # Let us compute this via convolution: where kernel is 1-h, correlate h with it
        corr = correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(corr)
        max_corr_value = corr[max_corr_idx]
        
        # Get the shift index that gives the worst overlap (max of h(x)(1-h(x+k)))
        # The index corresponds to shift s : 
        # If full correlation array has elements [k=0, k=1, ..., k=2n-2]  
        # then the shift we want is in [0, 2n-2] -> shift = max_corr_idx - (n-1) = max_corr_idx - n + 1
    
        shift_idx = max_corr_idx - (n_points - 1)
        if shift_idx < 0:
            shift_idx += n_points  # wrap around for essential structure
        
        # Compute the gradient of q_s(h) = sum_i h_i h_{i+s}
        # ∂q_s / ∂h_i = h_{i+s} + h_{i-s} (periodic or zero padding)
        g = np.zeros_like(h)
        if shift_idx == 0:
            g = h  # since shift by 0 applies to all h_i
        else:
            # Shift indices appropriately
            if shift_idx > 0:
                # h_{i+s} is right-shifted by shift_idx
                g[:shift_idx] = 0
                g[shift_idx:] = h[:-shift_idx]
                # Add the left-shifted term too
                g_left = np.zeros_like(h)
                g_left[shift_idx:] = h[:-shift_idx]
                g += g_left
            else:
                # shift_idx < 0, left shift by |shift_idx|
                shift_neg = -shift_idx
                g[shift_neg:] = 0
                g[:-shift_neg] = h[shift_neg:]
                # Add right-shifted term
                g_right = np.zeros_like(h)
                g_right[:-shift_neg] = h[shift_neg:]
                g += g_right

        # Simple update based on projected gradient direction
        # The DC subproblem is min -g^T h subject to sum h = 1 and 0≤h≤1
        # This is a simple capped simplex projection


        # Project onto the feasible set: good choice for the gradient-based update
        h_new = capped_simplex_projection(g, np.sum(h) * dx)

        # Ensure exact sum (fix quenching)
        # Avoid unnecessary mass increment unless necessary
        total_new = np.sum(h_new) * dx
        h_new *= (1.0 / total_new) if total_new > 0 else 1.0
        
        # Evaluate new function value
        c5_new = evaluate_c5(h_new)
        
        # Accept if improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            recent_improvements = 0
        else:
            recent_improvements += 1

        # Simple line search: might be useful at some point, but for now we keep incrementally improving
        h = h_new.copy()
        
        if recent_improvements >= convergence_threshold:
            break

    return best_h, best_c5
