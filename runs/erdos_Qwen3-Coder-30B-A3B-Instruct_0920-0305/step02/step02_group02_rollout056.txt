def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dxs[len(h_values)]
    return c5

def solve_sdp(n):
    dx = dxs[n]
    w = n / 2
    # Create variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    
    # Constraints
    constraints = [
        cp.diag(X) == cp.multiply(h, h),  # X_ii = h_i^2
        h >= 0, h <= 1,
        cp.sum(h) == w,
        X >> 0  # PSD constraint
    ]
    
    # Objective: minimize the minimum overlap
    # Construct Toeplitz matrices
    T_k_list = []
    for k in range(n):
        T_k = np.zeros((n, n))
        if k == 0:
            np.fill_diagonal(T_k, 1)
        else:
            np.fill_diagonal(T_k[k:], 1)
            np.fill_diagonal(T_k[:, k:], 1)
        T_k_list.append(T_k)
    
    # Objective value
    # max t subject to trace(T_k * X) >= t for all k
    t = cp.Variable()
    for k in range(n):
        constraints.append(cp.trace(T_k_list[k] @ X) >= t)
    
    # Minimize -t to maximize t
    objective = cp.Minimize(-t)
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.MOSEK, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            X_sol = X.value
            return h_sol, X_sol, problem.value
        else:
            return None, None, None
    except:
        return None, None, None

def round_sdp_solution(h, X, n):
    # Cholesky decomposition
    try:
        R = np.linalg.cholesky(X)
        # Random rounding
        g = np.random.randn(R.shape[0])
        h_round = R.T @ g
        
        # Project onto capped simplex
        def project_simplex(v, w):
            v_sorted = np.sort(v)[::-1]
            cumsum = np.cumsum(v_sorted)
            idx = np.where(v_sorted - (w - cumsum + v_sorted) / np.arange(1, len(v) + 1) > 0)[0]
            if len(idx) == 0:
                return np.ones_like(v) * 0
            k = idx[-1] + 1
            theta = (cumsum[k-1] - w) / k
            v_proj = np.maximum(v - theta, 0)
            # Clip to [0, 1]
            v_proj = np.clip(v_proj, 0, 1)
            # Scale to maintain sum
            if np.abs(np.sum(v_proj) - w) > 1e-6:
                v_proj *= w / np.sum(v_proj)
            return v_proj
            
        h_proj = project_simplex(h_round, n/2)
        return h_proj
    except:
        return None

def fourier_fit(h, n, M=6):
    # Generate cosine/sine basis
    x = np.arange(n)
    A = np.column_stack([
        np.ones(n),
        *[np.cos(np.pi * m * x / n) for m in range(1, M+1)],
        *[np.sin(np.pi * m * x / n) for m in range(1, M+1)]
    ])
    
    # Transform constraints into optimization
    # Minimize ||Ah - h_target||^2 with box constraints
    
    def objective(params):
        # Reconstruct h from Fourier coefficients
        # params = [a1,...,aM,b1,...,bM]
        a_vals = params[:M]
        b_vals = params[M:]
        h_rec = 0.5 + np.dot(A[:, 1:M+1], a_vals) + np.dot(A[:, M+1:], b_vals)
        # Clip to [0, 1]
        h_rec = np.clip(h_rec, 0, 1)
        # Scale to match sum
        h_rec *= (n/2) / np.sum(h_rec)
        return np.linalg.norm(h_rec - h)**2
    
    # Start point: least squares
    opt_res = minimize(objective, np.zeros(2*M), method='L-BFGS-B', bounds=[(None, None)]*2*M)
    return opt_res.x

def eval_fourier_coeffs(coeffs, n, M):
    x = np.arange(n)
    h_rec = 0.5 + np.dot(np.column_stack([np.cos(np.pi * m * x / n) for m in range(1, M+1)]), coeffs[:M])
    h_rec += np.dot(np.column_stack([np.sin(np.pi * m * x / n) for m in range(1, M+1)]), coeffs[M:])
    # Ensure it's in [0,1]
    h_rec = np.clip(h_rec, 0, 1)
    # Scale to match integral
    h_rec *= (n/2) / np.sum(h_rec)
    return h_rec

def smooth_refinement(h_init, n):
    M = min(10, n//4)
    coeffs = fourier_fit(h_init, n, M)
    h_smooth = eval_fourier_coeffs(coeffs, n, M)
    return h_smooth

def refine_final(h, n, dx):
    # Apply point-wise projected gradient descent
    def compute_overlap(h_vals):
        h_vec = np.array(h_vals)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Project onto [0,1] and ensure sum is correct
    def project_step(h_vals, step_size=0.005):
        h_new = h_vals - step_size * np.gradient(compute_overlap(h_vals))
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new *= (n / 2) / np.sum(h_new)
        return h_new
    
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = compute_overlap(h_current)
    
    max_iter_refine = 500
    for i in range(max_iter_refine):
        h_current = project_step(h_current)
        c5 = compute_overlap(h_current)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()
            
    return best_h, best_c5
