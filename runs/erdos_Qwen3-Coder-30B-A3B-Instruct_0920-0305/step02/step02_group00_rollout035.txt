def evaluate_h(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def linearize_bilinear(h, s):
    """Linearize bilinear term h_i * h_{i+s} for shift s."""
    # Handle boundary conditions (treat as 0)
    a = np.zeros_like(h)
    b = 0.0
    for i in range(len(h) - s):
        a[i] = 1 - h[i + s] - h[i - s] if i - s >= 0 else 1 - h[i + s]
        b -= h[i] * h[i + s]
    return a, b

def solve_lp(h_prev, n):
    """Solve the linear program for current h_prev."""
    # Variable for h and t
    h = cp.Variable(n)
    t = cp.Variable(1)

    # Constraints
    constraints = [cp.sum(h) == n / 2.0, h >= 0, h <= 1]

    # Build linearized constraints
    for s in range(n):
        a_s, b_s = linearize_bilinear(h_prev, s)
        constraints.append(a_s @ h + b_s <= t)

    # Objective
    objective = cp.Minimize(t)

    # Solve problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, verbose=False)

    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, t.value
    else:
        return h_prev, np.inf

def project_to_feasible(h):
    """Project h to satisfy ∫₀² h = 1 (i.e., sum = n/2) and h ∈ [0,1]"""
    h_proj = np.clip(h, 0, 1)
    current_sum = np.sum(h_proj)
    if abs(current_sum - n/2) < 1e-10:
        return h_proj
    # Simple bisection to adjust
    # Compute weight θ so that  Σ_i min(1, max(0, h_i - θ)) = n/2
    lb, ub = -1.0, 1.0
    for _ in range(50):
        theta = (lb + ub) / 2.0
        new_h = np.clip(h_proj - theta, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > n / 2:
            lb = theta
        else:
            ub = theta
        if abs(new_sum - n/2) < 1e-8:
            break
    projected = np.clip(h_proj - (lb+ub)/2, 0, 1)
    return projected
