import numpy as np
from scipy.optimize import minimize
from scipy.linalg import eigh
from cvxpy import *
import time
import math

# Predefined initial values if needed (possibly from 'initial_h_values')
# Sample size n = 48
n_points = 48
dx = 2.0 / n_points

def evaluate_h(h_values):
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def solve_sdp(n, w):
    """Solve the SDP relaxation for the given n and w."""
    h = Variable(n)
    X = Variable((n, n), symmetric=True)
    
    # Define constraints: diag(X) = h .* h, i.e. X_ii = h_i^2
    constr = [diag(X) == h * h]
    
    # Add constraints for bounding h in [0,1]
    for i in range(n):
        constr.append(h[i] >= 0)
        constr.append(h[i] <= 1)
    
    # Sum constraint (total mass)
    constr.append(sum(h) == w)
    
    # PSD constraint (pos semidefinite)
    constr.append(X >> 0)
    
    # Form scalability: trace(T_k * X) >= t for all k
    # We'll handle this later, and infeasible points are better skipped
    # For now, just try to make X rank-1 feasible, so SDP works
    
    # Objective: Try to maximize t subject to trace(T_k * X) >= t for all shifts
    t = Variable()
    
    # Build Toeplitz matrices for shifts
    set_of_shifts = list(range(0, n))
    # consider a few critical ones to add to constraints, skip full for speed
    num_shifts = min(20, len(set_of_shifts))
    shifts = set_of_shifts[:num_shifts]
    
    for k in shifts:
        T_k = np.zeros((n,n))
        for i in range(n):
            if i + k < n:
                T_k[i, i+k] = 1
        T_k = (T_k + T_k.T) / 2  # Symmetrize
        constr.append(trace(T_k @ X) >= t)
    
    prob = Problem(Maximize(t), constr)
    
    try:
        prob.solve(solver=SCS, verbose=False, max_iters=5000)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            if np.isfinite(t.value):
                return h.value, X.value, t.value
    except:
        pass
    return None, None, None

def round_sdp_solution(h_sdp, X_sdp, n):
    """Round SDP solution h_sdp to a feasible rounded vector."""
    # Compute Cholesky decomposition of X_sdp
    try:
        L = np.linalg.cholesky(X_sdp)
        g = np.random.randn(L.shape[0])
        h_rnd = L.T @ g  # This creates a sampled vector from the factor
    except:
        # Fallback: use leading eigenvector of X_sdp
        try:
            _, v = eigh(X_sdp, eigvals=(n-1, n-1))  # Get largest eigenvalue/eigenvector
            h_rnd = np.abs(v.flatten()) * np.sqrt(np.max(np.diag(X_sdp)))
        except:
            # Give up
            h_rnd = h_sdp.copy()
    return h_rnd

def project_to_simplex(v, w):
    """Project vector v onto the capped simplex {x: 0≤x≤1, sum(x)=w}, using waterfilling."""
    # Sort v in descending order
    v_sorted = np.sort(v)[::-1]
    # Find the threshold tau using water filling
    n = len(v)
    tau = 0.0
    cumsum = 0.0
    for i, x in enumerate(v_sorted):
        cumsum += x
        if cumsum - (i+1)*tau >= w:
            if i < n-1:
                tau = (cumsum - w) / (i+1)
            else:
                tau = cumsum / (i+1)
            break
    # Project each component
    x = np.array([max(min(x - tau, 1.0), 0.0) for x in v])
    s = np.sum(x) - w
    if abs(s) > 1e-10:
        # Adjust small errors by adding residual to interior component
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            idx = np.where(mask)[0][0]
            x[idx] -= s
    return x

def solve_fourier_fit(h_round, M=8):
    """Fit h_round with a truncated Fourier basis (cos and sin)."""
    # Create mapping matrix for polynomial fit
    n = len(h_round)
    A = np.ones((n, 1 + 2*M))
    for i in range(n):
        x = i * 1.0 / (n-1) if n > 1 else 0
        A[i, 1] = np.cos(np.pi * x)  # a_1 * cos(π x)
        A[i, 2] = np.sin(np.pi * x)  # b_1 * sin(π x)
        for m in range(2, M+1):
            A[i, 1+2*(m-1)] = np.cos(np.pi * m * x)  # a_m * cos(π m x)
            A[i, 1+2*(m-1)+1] = np.sin(np.pi * m * x)  # b_m * sin(π m x)
    
    # Force 0 to 1 bounds:
    # CVXPY solves the least squares for coefficients with constraints, but
    # since we want low dimensional, solve directly by SQP (L-BFGS-B) on coeffs
    
    # We will optimize, confident about constraints on the final h
    def objective(coeffs):
        # build h from coeffs
        h_vals = A @ np.append(0.5, coeffs)  # 0.5 is the offset
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Project to correct mass
        h_projected = project_to_simplex(h_vals, n/2)
        # Evaluate the actual objective
        return evaluate_h(h_projected)
        
    # Initial guess: zeros
    initial_coeffs = np.zeros(2*M)
    
    # Optimize
    result = minimize(objective, initial_coeffs, method='L-BFGS-B')
    
    if result.success:
        best_coeffs = result.x
        # Compute final profile
        h_final_vals = A @ np.append(0.5, best_coeffs)
        h_final_vals = np.clip(h_final_vals, 0, 1)
        h_final_vals = project_to_simplex(h_final_vals, n/2)
        return h_final_vals
    else:
        # Revert to original rounded one and refine
        h_clip = np.clip(h_round, 0, 1)
        h_clip = project_to_simplex(h_clip, n/2)
        return h_clip

def optimize_with_fourier(seed=42):
    np.random.seed(seed)
    # Test different sizes
    sizes = [48, 56, 64]
    best_h = None
    best_c5 = float('inf')
    best_n = None
    for n in sizes:
        w = n / 2
        h_sdp, X_sdp, t_sdp = solve_sdp(n, w)
        if h_sdp is None or X_sdp is None or t_sdp is None:
            continue
        # Round
        h_round = round_sdp_solution(h_sdp, X_sdp, n)
        h_round = project_to_simplex(h_round, w)
        # Fourier fit
        if n <= 64:  # Only do it for reasonable sizes
            h_fourier = solve_fourier_fit(h_round, 8)
        else:
            h_fourier = h_round.copy()
        # Evaluate
        c5 = evaluate_h(h_fourier)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fourier.copy()
            best_n = n
    return best_h, best_c5, best_n

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = None
    trials = 0
    while time.time() - start_time < budget_s * 0.95 and trials < 50:
        trials += 1
        h, c5, n = optimize_with_fourier(seed + trials)
        if h is not None and c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n = n
    if best_h is None:
        # Fallback
        h_fallback = np.array(initial_h_values)
        best_h = h_fallback
        best_c5 = evaluate_h(h_fallback)
        best_n = n_points
    return list(best_h), best_c5, best_n
