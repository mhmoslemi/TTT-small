import numpy as np
from scipy.signal import correlate
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto box constraints [0,1] and sum constraint"""
    lo = 0.0
    hi = 1.0
    # First clip to bounds
    w = np.clip(v, lo, hi)
    # Compute sum
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h):
    """Compute max of overlap integral for all shifts"""
    n = len(h)
    dx = 2.0 / n
    # Compute convolution of h with (1-h) for all lags
    conv = correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    # Find the shift that produces max
    argmax = np.argmax(conv)
    # Convert from index to shift (center of full convolution)
    shift_m = argmax - (n - 1)
    return max_val, shift_m

def gradient_for_shift(h, m):
    """Compute gradient of overlap integral for specific shift"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        # For positive shift: overlapping indices i = 0 to n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # For negative shift: overlapping indices i = -m to n-1
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx factor
