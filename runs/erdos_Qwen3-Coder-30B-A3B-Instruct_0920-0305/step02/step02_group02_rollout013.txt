import numpy as np
from scipy.fft import ifft, fft
import time

# Import required functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def project_onto_capped_simplex(x, target_sum):
    """Project x onto the capped simplex {x | 0 <= x_i <= 1, sum x_i = target_sum}."""
    x = np.asarray(x).copy()
    n = len(x)
    if np.isclose(np.sum(x), target_sum):
        return x
    if target_sum <= 0:
        return np.zeros_like(x)
    if target_sum >= n:
        return np.ones_like(x)
    # sort in descending order
    idx = np.argsort(x)[::-1]
    sorted_x = x[idx]
    # find threshold
    cumsum = np.cumsum(sorted_x)
    threshold_idx = np.searchsorted(cumsum, target_sum, side="right")
    if threshold_idx == 0:
        threshold = 0.0
    else:
        threshold = (cumsum[threshold_idx - 1] - target_sum + sorted_x[threshold_idx - 1]) / threshold_idx
    # construct result
    result = np.zeros_like(x)
    for i in range(n):
        j = idx[i]
        if x[j] > threshold:
            result[j] = x[j]
        else:
            result[j] = max(0.0, threshold)
    # ensure exact sum
    err = np.sum(result) - target_sum
    if abs(err) > 1e-12:
        # adjust the smallest nonboundary entry
        non_zero = np.where(result > 1e-12)[0]
        if len(non_zero) > 0:
            i_min = non_zero[np.argmin(result[non_zero])]
            result[i_min] -= err
    return result

def compute_correlations(h, dx):
    """Compute all correlation contributions using FFT."""
    h = np.asarray(h)
    n = len(h)
    h_padded = np.zeros(2 * n - 1)
    h_padded[:n] = h
    # correlation with (1 - h)
    h_conj = np.conj(fft(h_padded))
    h1 = 1 - h
    h1_padded = np.zeros(2 * n - 1)
    h1_padded[:n] = h1
    h1_conj = fft(h1_padded)
    corr = ifft(h_conj * h1_conj).real[:n]
    return corr * dx

def cost_function_smooth(h, alpha=1.0, dx=1.0):
    """Smooth approximation of max correlation."""
    r = compute_correlations(h, dx)
    max_r = np.max(r)
    r_shift = r - max_r
    exp_terms = np.exp(alpha * r_shift)
    Z = np.sum(exp_terms)
    if Z == 0:
        return 0
    log_Z = np.log(Z) + max_r * alpha
    return -dx / alpha * log_Z

def gradient_smooth(h, alpha=1.0, dx=1.0):
    """Gradient of smooth surrogate."""
    r = compute_correlations(h, dx)
    max_r = np.max(r)
    r_shift = r - max_r
    exp_terms = np.exp(alpha * r_shift)
    w = exp_terms / np.sum(exp_terms)
    n = len(h)
    grad = np.zeros_like(h)
    # compute gradient contribution of each shift
    for k in range(n):
        d_r_k = np.zeros_like(h)
        # adjust for periodic boundary
        if k == 0:
            # r_0 = w - sum h_i^2
            for i in range(n):
                d_r_k[i] = -2 * h[i]
        else:
            for i in range(n):
                if i + k < n:
                    d_r_k[i] += -h[i + k]
                if i - k >= 0:
                    d_r_k[i] += -h[i - k]
        # fill in the gradient term
        grad += w[k] * d_r_k
    return dx * grad

def backtracking_line_search(h, grad, alpha, dx, max_iter=10):
    """Simple backtracking line search."""
    step_size = 0.5
    c1 = 1e-4
    f_prev = cost_function_smooth(h, alpha, dx)
    for _ in range(max_iter):
        h_new = h - step_size * grad
        # project
        h_new = project_onto_capped_simplex(h_new, target_sum=len(h)*dx/2)
        f_new = cost_function_smooth(h_new, alpha, dx)
        if f_new <= f_prev - c1 * step_size * np.dot(grad, grad):
            return h_new, step_size
        step_size *= 0.5
    return h_new, step_size

def accelerated_projected_gradient_descent(h0, dx, w, alpha0=1, alpha_factor=2, alpha_max=300, max_iters=5000, tol=1e-6):
    """Accelerated projected gradient descent with continuation."""
    h = h0.copy()
    t = 1
    alpha = alpha0
    best_c5 = float('inf')
    best_h = h0.copy()
    
    for i in range(max_iters):
        # compute gradient of surrogate
        grad = gradient_smooth(h, alpha, dx)
        # Nesterov step
        h_tilde = project_onto_capped_simplex(h - grad, w)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h_new = h_tilde + ((t - 1) / t_new) * (h_tilde - h)
        h = h_new
        t = t_new
        
        # evaluate surrogate
        cost = cost_function_smooth(h, alpha, dx)
        # periodic check for improvement
        if i % 100 == 0:
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if abs(cost) < 1e-12:
            break
            
    # evaluate final with real maximum
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5

def initialize_h(n, method="random"):
    """Initialize a feasible h vector."""
    if method == "binary":
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1 if i < n//2 else 0
        h = project_onto_capped_simplex(h, target_sum=n/2)
    elif method == "three_block":
        a = np.random.uniform(0.2, 0.4)
        b = np.random.uniform(0.6, 0.8)
        p = (n/2 - a * n/2) / ((b - a) * n/2)
        h = np.zeros(n)
        for i in range(n):
            x = i * 2.0 / n
            if x < a:
                h[i] = 1
            elif x < b:
                h[i] = p
            else:
                h[i] = 0
        h = project_onto_capped_simplex(h, target_sum=n/2)
    else:  # random
        h = np.random.rand(n)
        h = project_onto_capped_simplex(h, target_sum=n/2)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h, best_c5 = None, float('inf')
    n_list = [48, 56, 64, 80]

    # split budget
    remain_n = len(n_list)
    
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        w = n / 2
        time_per_n = (budget_s - (time.time() - start_time)) / remain_n
        iter_time = time.time()
        
        # restarts
        for i in range(50):  # 50 restarts per n
            if time.time() - iter_time > time_per_n * 0.95 or time.time() - start_time > budget_s * 0.95:
                break
            # try different initialization methods
            methods = ["random", "binary", "three_block"]
            init_method = methods[i % len(methods)]
            h0 = initialize_h(n, init_method)
            h_opt, c5_opt = accelerated_projected_gradient_descent(h0, dx, w, max_iters=500)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        
        remain_n -= 1

    # final projection and evaluation
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, target_sum=len(best_h)*2/len(best_h))
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        # fallback to initial solution
        best_h = np.array(initial_h_values)
        final_c5 = evaluate_erdos_solution(best_h)

    return list(best_h), final_c5, len(best_h)
