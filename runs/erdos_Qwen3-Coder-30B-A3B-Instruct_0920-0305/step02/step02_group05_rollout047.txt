def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / n_points
    target_sum = 1.0
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    delta = target_sum - current_sum
    if abs(delta) < 1e-10:
        return h
    if delta > 0:
        # Distribute delta equally among indices < 1
        num_free = np.sum(h < 1)
        if num_free > 0:
            increment = delta / num_free
            h[h < 1] += increment
            h = np.clip(h, 0, 1)
    else:
        # Subtract the excess equally among indices > 0
        num_free = np.sum(h > 0)
        if num_free > 0:
            decrement = -delta / num_free
            h[h > 0] -= decrement
            h = np.clip(h, 0, 1)
    return h

def compute_subgradient(h, m, n_points, dx):
    """Compute subgradient for the given lag m"""
    g = np.zeros(n_points)
    if m >= 0:
        # i = 0 .. n-1-m, j = i+m
        i_vals = np.arange(0, n_points - m)
        j_vals = i_vals + m
        g[i_vals] += 1 - h[j_vals]
        g[j_vals] -= h[i_vals]
    else:
        # m < 0: swap roles of i and j
        m_abs = -m
        i_vals = np.arange(m_abs, n_points)
        j_vals = i_vals - m_abs
        g[i_vals] += 1 - h[j_vals]
        g[j_vals] -= h[i_vals]
    return g * dx

def symmetry_enforce(h, n_points):
    """Enforce symmetry h[i] = h[n-1-i]"""
    for i in range(n_points // 2):
        avg_val = (h[i] + h[n_points - 1 - i]) / 2.0
        h[i] = avg_val
        h[n_points - 1 - i] = avg_val
    return h

    max_val = np.max(A)

    return max_val, max_indices
