import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project v onto the set of vectors with bounds [0,1] and sum = target_sum"""
    lo = 0.0
    hi = 1.0
    # First clip to bounds
    w = np.clip(v, lo, hi)
    # Compute sum
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    # We'll iteratively adjust
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                # cannot increase more, break
                break
            delta = diff / n_free
            w[free] += delta
            # Clip again
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h):
    """Compute maximum overlap integral and the shift that achieves it"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    max_idx = np.argmax(conv)
    # The shift that gives max is: max_idx - (n_points - 1)
    shift = max_idx - (n_points - 1)
    return max_val, shift

def gradient_for_shift(h, m):
    """Compute the gradient of the overlap integral for a given shift m"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        # i from 0 to n-m-1
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx = 2/n
