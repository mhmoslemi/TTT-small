import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

# Constants
n_points_list = [48, 56, 64]
dx = 2.0 / 48  # Default
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_h_to_feasible(h, n_points, dx):
    """Project h onto the feasible set: [0,1]^n and sum = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        # Use bisection to rescale to sum = 1.0
        # Linear rescaling in proportion to difference from target sum
        h = h * (1.0 / current_sum)
        h = np.clip(h, 0, 1)
    return h

def build_linear_constraints(h_current):
    """Build linearized constraints from current h."""
    n = len(h_current)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0
        # Compute O_s^{(k)} = sum h_i * h_{i+s}
        O_s = 0
        # For i = 0..n-1-s
        for i in range(n - s):
            a_s[i] += 1 - h_current[i + s] - h_current[i - s] if i - s >= 0 else 1 - h_current[i + s]
            O_s += h_current[i] * h_current[i + s]
        b_s = -O_s
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_subproblem(a_s_list, b_s_list, n_points, dx):
    """Solve the LP subproblem using cvxpy."""
    h = Variable(n_points)
    t = Variable(1)
    
    # Constraints
    constraints = []
    for i, (a_s, b_s) in enumerate(zip(a_s_list, b_s_list)):
        constraints.append(a_s @ h + b_s <= t)
    constraints.append(sum(h) == n_points / 2)
    constraints.append(0 <= h)
    constraints.append(h <= 1)
    
    # Objective
    objective = Minimize(t)
    
    # Solve
    prob = Problem(objective, constraints)
    prob.solve(solver=CPLEX, verbose=False)
    
    if prob.status == 'optimal':
        return h.value, t.value
    else:
        # fallback to simple clipping, but should not happen for valid setup
        return np.clip(np.random.rand(n_points), 0, 1), np.inf

def run(seed=42, budget_s=600):
    """Main optimization routine."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = np.inf
    n_used = 48
    
    for n in n_points_list:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        stop_time = start_time + 0.95 * budget_s

        # Try multiple random restarts for good multi-start
        max_restarts = 40
        restart_count = 0
        while restart_count < max_restarts and time.time() < stop_time:
            # Different seed types
            if restart_count == 0:
                # Interpolated from initial_h_values if length matches, otherwise uniform
                h_initial = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(initial_h_values)), initial_h_values)
            elif restart_count == 1:
                h_initial = np.random.rand(n)
            elif restart_count == 2:
                # Binary initialisation
                h_binary = np.zeros(n)
                h_binary[:n//2] = 1
                np.random.shuffle(h_binary)
                h_initial = h_binary
            else:
                # Perturb best
                h_initial = best_h + np.random.normal(0, 0.05, n)
            h_initial = project_h_to_feasible(h_initial, n, dx)
            
            # Now perform CCP loop
            h_current = h_initial.copy()
            prev_c5 = np.inf
            max_iter = 200
            
            for iteration in range(max_iter):
                if time.time() >= stop_time:
                    break
                a_s_list, b_s_list = build_linear_constraints(h_current)
                h_new, t_new = solve_subproblem(a_s_list, b_s_list, n, dx)
                if np.isinf(t_new) or t_new == np.inf:
                    break
                # Make it feasible
                h_projected = project_h_to_feasible(h_new, n, dx)
                # Evaluate true objective
                h_for_eval = h_projected
                correlation = np.correlate(h_for_eval, 1 - h_for_eval, mode="full")
                current_c5 = np.max(correlation) * dx
                
                if current_c5 < best_c5:
                    best_h = h_projected.copy()
                    best_c5 = current_c5
                    n_used = n
                    if best_c5 < target_bound:
                        break
                        
                # Check convergence
                if abs(current_c5 - prev_c5) < 1e-6:
                    break
                prev_c5 = current_c5
                h_current = h_projected

            restart_count += 1

    # Final polishing step with local solver
    if best_h is not None:
        bounds = [(0, 1) for _ in range(n_used)]
        def objective(h):
            return evaluate_h(h)

        try:
            # Scipy SLSQP refinement
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                             options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                best_h = result.x
        except:
            pass

        final_c5 = evaluate_h(best_h)
        return list(best_h), final_c5, n_used

    # Final step: return initial if nothing found
    init_c5 = evaluate_h(initial_h_values)
    return list(initial_h_values), init_c5, len(initial_h_values)
