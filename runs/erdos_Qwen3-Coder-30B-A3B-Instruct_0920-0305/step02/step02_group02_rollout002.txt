import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Constants
n_candidates = [48, 56, 64, 80]
dx = 2.0 / n_candidates[0]
target_sum = n_candidates[0] / 2
alpha_max = 300
max_iterations = 5000
step_size_init = 0.5
tolerance = 1e-6

def evaluate_c5(h):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h)
    # Compute correlation of h with (1 - h)
    # Use FFT for efficient convolution
    h_fft = fft(np.concatenate([h, [0] * (len(h) - 1)]))
    inv_h_fft = fft(np.concatenate([1 - h, [0] * (len(h) - 1)]))
    correlation = ifft(h_fft * np.conj(inv_h_fft)).real
    # Extract valid correlations
    valid_correlations = correlation[:len(h)]
    max_correlation = np.max(valid_correlations)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_capped_simplex(x, target_sum):
    """Project x onto the capped simplex {x | 0 <= x_i <= 1, sum(x) = target_sum}."""
    x = np.array(x).copy()
    # Avoid numerical issues
    if np.isclose(np.sum(x), target_sum, rtol=1e-12):
        return x
    # Sort in descending order
    x_sorted = np.sort(x)[::-1]
    # Water-filling algorithm
    n = len(x)
    cumulative = np.cumsum(x_sorted)
    # Find threshold
    gamma = 0.0
    for i in range(n):
        if (x_sorted[i] - (target_sum - cumulative[i]) / (n - i)) >= 0:
            gamma = (target_sum - cumulative[i]) / (n - i)
            break
    # Apply threshold and ensure exact sum
    projected = np.maximum(0, np.minimum(1, x - gamma))
    # Adjust to get exactly target_sum
    diff = target_sum - np.sum(projected)
    # Distribute error to a single element if difference is small
    if abs(diff) > 1e-12:
        idx = np.argmax(projected)
        projected[idx] = np.clip(projected[idx] + diff, 0, 1)
    return projected

def compute_correlations(h):
    """Compute all correlation terms r_k for each shift k."""
    h = np.array(h)
    # Use convolution to compute correlations
    h_fft = fft(np.concatenate([h, [0] * (len(h) - 1)]))
    inv_h_fft = fft(np.concatenate([1 - h, [0] * (len(h) - 1)]))
    correlation = ifft(h_fft * np.conj(inv_h_fft)).real
    corr_terms = correlation[:len(h)]
    return corr_terms

def compute_gradient(h, r_terms, alpha):
    """Compute gradient of the smooth surrogate."""
    # Compute softmax weights
    r_terms_shifted = r_terms - np.max(r_terms)
    weights = np.exp(alpha * r_terms_shifted)
    weights /= np.sum(weights)
    # Compute gradient via FFT for efficiency
    h_fft = fft(np.concatenate([h, [0] * (len(h) - 1)]))
    inv_h_fft = fft(np.concatenate([1 - h, [0] * (len(h) - 1)]))
    # Derivative of r_k(h) = w - sum h_i h_{i+k} is 1 - h_{i+k} - h_{i-k}
    # or 1 - h_{i+k} when h is circular
    freq_grads = []
    for k in range(len(h)):
        grad_rk = np.zeros_like(h)
        if k > 0:
            grad_rk[:-k] = -h[k:]
        if k < len(h) - 1:
            grad_rk[k+1:] = -h[:-k-1]
        grad_rk[0] -= h[-k] if k > 0 else 0
        grad_rk[-k] -= h[0] if k > 0 else 0
        freq_grads.append(grad_rk)
    g = np.zeros_like(h)
    for i, w in enumerate(weights):
        g += w * freq_grads[i]
    return dx * g

def projected_accelerated_gradient_descent(h0, dx, target_sum, alpha_start=1.0, alpha_factor=2.0, max_alpha=300, max_iter=5000):
    """Optimize using projected accelerated gradient descent."""
    h = h0.copy()
    t_prev = 1.0
    y_prev = h.copy()
    
    alpha = alpha_start
    iterations = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    while alpha <= max_alpha and iterations < max_iter:
        # Mini-batch gradient descent with continuation
        last_f = float('inf')
        for _ in range(100):  # Inner loop for fixed smoothing
            r_terms = compute_correlations(h)
            f = (dx / alpha) * np.log(np.sum(np.exp(alpha * r_terms)))
            g = compute_gradient(h, r_terms, alpha)
            
            # Backtracking line search
            eta = step_size_init
            curr_c5 = evaluate_c5(h)
            while True:
                h_new = project_to_capped_simplex(y_prev - eta * g, target_sum)
                new_c5 = evaluate_c5(h_new)
                if new_c5 <= curr_c5:
                    h = h_new
                    break
                eta *= 0.5
            
            # Nesterov extrapolation
            t = (1.0 + np.sqrt(1.0 + 4.0 * t_prev ** 2)) / 2.0
            y = h + ((t_prev - 1.0) / t) * (h - y_prev)
            y_prev = y.copy()
            t_prev = t
            
            # Early stopping if improvement small
            if abs(last_f - f) < tolerance:
                break
            last_f = f
            
            iterations += 1
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
            if iterations >= max_iter:
                break
                
        # Continue with increased smoothing
        alpha *= alpha_factor
        if alpha > max_alpha:
            break
            
    return best_h, best_c5

def perturb_solution(h, magnitude=0.05):
    """Perturb solution slightly."""
    perturbed = h + np.random.normal(0, magnitude, len(h))
    return project_to_capped_simplex(perturbed, target_sum)

def local_refinement(h, dx, target_sum):
    """Refine with local SQP method."""
    def objective(x):
        return evaluate_c5(x)
    def constraint(x):
        return np.sum(x) * dx - 1.0

    bounds = [(0, 1) for _ in range(len(h))]
    cons = {'type': 'eq', 'fun': constraint}
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-12, 'gtol': 1e-12})
        if res.success:
            refined_h = project_to_capped_simplex(res.x, target_sum)
            refined_c5 = evaluate_c5(refined_h)
            return refined_h, refined_c5
    except:
        pass
    return h, evaluate_c5(h)

def run(seed=42, budget_s=600):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 48

    # Run on increasing n values
    for n in n_candidates:
        dx = 2.0 / n
        target_sum = n / 2
        allocate_time = budget_s * 0.8 / len(n_candidates)

        n_time = time.time()
        
        restarts = 0
        while time.time() - n_time < allocate_time:
            # Initialize one of multiple strategies
            rand_h = np.random.rand(n)
            rand_h = project_to_capped_simplex(rand_h, target_sum)
            binary_h = np.zeros(n)
            binary_h[:int(target_sum)] = 1
            np.random.shuffle(binary_h)
            binary_h = project_to_capped_simplex(binary_h, target_sum)
            three_block_h = np.ones(n) * 0.5
            mid_point = n // 2
            half = n // 4
            three_block_h[:half] = 1
            three_block_h[mid_point - half:mid_point + half] = 0.5
            three_block_h = project_to_capped_simplex(three_block_h, target_sum)
            
            # Use one of these initializers
            if restarts % 3 == 0:
                h = rand_h
            elif restarts % 3 == 1:
                h = binary_h
            else:
                h = three_block_h
            
            h_opt, c5_opt = projected_accelerated_gradient_descent(h, dx, target_sum, alpha_max=200)
            h_opt, c5_opt = local_refinement(h_opt, dx, target_sum)

            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n
                
            restarts += 1
            if restarts > 50:
                break # Safety check for too many restarts
            
            if time.time() - start_time > budget_s * 0.95:
                break
                
    # Final validation
    if best_h is not None:
        best_h = project_to_capped_simplex(best_h, target_sum)
        final_c5 = evaluate_c5(best_h)
        return list(best_h), final_c5, best_n
    else:
        return list(np.ones(48)*0.5), 0.5, 48
