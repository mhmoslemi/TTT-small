import numpy as np
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, norm, sum
import cvxpy as cp
import time
from scipy.spatial.distance import cdist

# Constants
target_bound = 0.3807
n_values = [48, 56, 64]
dx = 2.0 / 48  # Start with a default value
max_iter = 200
tol_obj = 1e-6
budget_s = 600

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values using correlation."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def project_onto_constraints(h, n_points, dx):
    """Project h onto the constraint set: 0 <= h_i <= 1, sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h) * dx
    if abs(h_sum - 1.0) > 1e-8:
        # Bisection method for α in [0, 1] such that sum(clamp(h - α, 0, 1)) = 1 / dx
        def func(alpha):
            h_proj = np.clip(h - alpha, 0, 1)
            return np.sum(h_proj) * dx - 1.0

        a, b = 0, 1
        while b - a > 1e-12:
            mid = (a + b) / 2
            if func(mid) > 0:
                a = mid
            else:
                b = mid
        alpha = a
        h = np.clip(h - alpha, 0, 1)
    return h


def build_linear_constraints(h_current):
    """Build the coefficient vectors a_s and constants b_s for each shift."""
    n = len(h_current)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0.0
        for i in range(n - s):
            a_s[i] = 1 - h_current[i + s] - h_current[i - s] if i - s >= 0 else 1 - h_current[i + s]
        b_s = -np.sum(h_current[i] * h_current[i + s] for i in range(n - s))
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list


def solve_subproblem(a_s_list, b_s_list, n_points, dx):
    """Solve the linear program for given constraints."""
    # Define decision variables
    h = Variable(n_points)
    t = Variable(1)
    
    # Constraints
    constraints = [0 <= h, h <= 1]
    constraints += [cp.sum(h) == n_points / 2]
    for a_s, b_s in zip(a_s_list, b_s_list):
        constraints += [a_s @ h + b_s <= t]
        
    # Minimize t (over all possible shifts)
    objective = Minimize(t)
    problem = Problem(objective, constraints)
    problem.solve(solver=cp.CBC, verbose=False)
    
    if problem.status in ["optimal", "optimal_inaccurate"]:
        h_sol = h.value
        t_sol = t.value
    else:
        # Fallback to current if not solved
        h_sol = np.zeros(n_points)
        t_sol = float('inf')
    return h_sol, t_sol


def solve_slp(h_initial, n_points, dx):
    """Sequential Linear Programming with projections."""
    h_current = h_initial.copy()
    best_c5 = float('inf')
    best_h = h_current.copy()
    
    for iteration in range(max_iter):
        # Compute current correlations
        O_s = []
        for shift in range(n_points):
            if shift == 0:
                O_s.append(np.sum(h_current * (1 - h_current)))
            else:
                corr = np.correlate(h_current, 1 - h_current, mode='full')
                O_s.append(corr[len(h_current) - 1 + shift] * dx)
        
        # Nonlinear correlation values
        c5_current = max(O_s)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
        
        # Build linear constraints
        a_s_list, b_s_list = build_linear_constraints(h_current)
        
        # Solve the LP
        h_new, t_new = solve_subproblem(a_s_list, b_s_list, n_points, dx)
        if t_new == float('inf'):
            break
        
        # Project h_new
        h_new = project_onto_constraints(h_new, n_points, dx)
        
        # Evaluate true objective with new h
        c5_eval = evaluate_h(h_new)
        
        # If improvement, update h
        if c5_eval < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_eval
        
        # Check for convergence
        if abs(c5_current - c5_eval) < tol_obj:
            break
        h_current = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main execution loop."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Calculate total time we allow for multiple n
    remaining_budget = budget_s * 0.95
    
    for n in n_values:
        if time.time() - start_time > remaining_budget:
            break
        dx = 2.0 / n
        
        # For this n, run multiple restarts
        num_restarts = 15  # Set a fixed trial count per n
        time_per_n = 0.8 * remaining_budget / len(n_values)
        local_start = time.time()
        time_per_restart = time_per_n / num_restarts
        
        for restart in range(num_restarts):
            if time.time() - start_time > remaining_budget:
                break
            if time.time() - local_start > time_per_n:
                break
                
            # Try different starting methods
            initial_methods = ["binary", "uniform", "interpolated"]
            start_method = initial_methods[restart % len(initial_methods)]
            h_start = generate_starting_point(n, method=start_method)
            h_proj = project_onto_constraints(h_start, n, dx)
            
            # Solve with SLP
            h_final, c5_final = solve_slp(h_proj, n, dx)
            
            # Check for improvement
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                if best_c5 < target_bound:
                    break
        
        # If within target go early exit
        if best_c5 < target_bound:
            break
    
    # Final polishing if needed
    if best_h is not None:
        bounds = [(0, 1) for _ in range(len(best_h))]
        def obj(h):
            return evaluate_h(h)
        try:
            # Local refinement
            result = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                              options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_h(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except Exception:
            pass
    
    original_n = len(best_h) if best_h is not None else 48
    return list(best_h) if best_h is not None else list(np.ones(48)*0.5), best_c5, original_n
