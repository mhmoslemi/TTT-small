import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_sum_constraint(h, target_sum):
    """Project h onto the constraint sum(h) * dx = target_sum"""
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    return h * scale

def capped_simplex_projection(grad, lb=0, ub=1, total_sum=1.0):
    """Project gradient onto the capped simplex: sum(h) = total_sum, lb <= h <= ub"""
    n = len(grad)
    dx = 2.0 / n
    # Scale to desired sum: note that we're solving for sum(h) = n*dx*target_sum (since dx = 2/n)
    # but we want sum(h) = target_sum, so h should be scaled accordingly
    
    # Actually let's work in the correct units, target_sum = 1, so sum(h) = 1
    # The gradient vector length corresponds to the line direction
    # Project to capped simplex: sum(h) = 1, 0 <= h_i <= 1
    # Sort gradient in descending order
    sorted_indices = np.argsort(-grad)
    h = np.zeros(n)
    remaining_sum = total_sum
    for i in range(n):
        idx = sorted_indices[i]
        if remaining_sum <= 0:
            h[idx] = 0
        else:
            cap = min(remaining_sum, ub - h[idx])
            if cap >= 0:
                h[idx] = min(ub, grad[idx] + h[idx])
                h[idx] = max(lb, h[idx])
                remaining_sum -= h[idx]
    # Adjust so we get exactly sum = 1
    actual_sum = np.sum(h)
    if abs(actual_sum) > 1e-12:
        h = h * (total_sum / actual_sum)
    return h

def compute_autocorrelation(h):
    """Compute all values of q_s(h) = sum h_i h_{i+s} for shift s"""
    n = len(h)
    # Simple correlation of h with itself gives sum h_i h_{i+s}
    auto = correlate(h, h, mode='full')
    # The middle term auto[n-1] corresponds to s=0, auto[n+1] to s=2, etc.
    # We want q_s for shifting in [0, 2]
    # Total indices: 2n-1
    # auto[0] to auto[2n-2]
    # For s = 0 to n-1: q_s = auto[n-1+s] (actual shift s = s)
    # We want to know q_s(h) for relevant significant s in 0..2.
    # use s from 0 to n-1 to cover all shifts, once + for simplicity we compute it
    # The actual connection is the linear index mapping for autocorrelation.
    # For convenience we will compute all the q_s as the full correlation between h and h
    # and look at entries corresponding to various shifts
    return auto

def compute_gradient_qs(h, shift_idx):
    """Compute the gradient of q_{shift_idx} at h"""
    n = len(h)
    # For q_s = sum_i h_i * h_{i+s}, the gradient is
    # d/dh_i [ sum_k h_k * h_{k+s} ] = h_{i+s} + h_{i-s}
    # if s < 0, that's h_{i+s}
    # if s > 0, that's h_{i-s}
    # This is a Tikhonov-style sum of shifted h by s.
    grad = np.zeros(n)
    if shift_idx == 0:
        # d/dh_i q_0 = h_i + h_i = 2 h_i?
        # Actually d/dh_i q_s(h) = h_{i+s} + h_{i-s}
        # For s = 0, this is just h_i + h_i = 2 h_i
        # but let's use good sense
        # d/dh_i q_0 = sum_k d/dh_i h_k h_{k+0} = h_i + h_i?
        # Actually need to be careful: when we consider just h_i in q_0 = sum_k h_k h_k = sum_k h_k^2
        # d/dh_i = 2 h_i (every k gives contribution 2 h_i if k = i)
        # So for q_s, we select the part where d/dh_i appears in the definition.
        # q_s = sum_k h_k h_{k+s}
        # Taking derivative:
        # d/dh_i q_s = { h_{i+s} } if i+s < n and i+s >= 0, { h_{i-s} } if i-s >= 0 and i-s < n, 0 otherwise
        # This can be computed with shifting
        r = np.roll(h, shift_idx)
        l = np.roll(h, -shift_idx)  # equivalent to h[-shift_idx:] + h[:-shift_idx] when shift_idx is pos
        # But just compute for uniform shift
        # Let's manually compute with rolling
        # n=4: h[0], h[1], h[2], h[3]
        # q_0 = h_0 h_0 + h_1 h_1 + h_2 h_2 + h_3 h_3
        # d/dh_0 q_0 = 2 h_0
        # d/dh_1 q_0 = 2 h_1 etc
        # q_1 = h_0 h_1 + h_1 h_2 + h_2 h_3
        # d/dh_0 q_1 = h_1
        # d/dh_1 q_1 = h_0 + h_2
        # d/dh_2 q_1 = h_1 + h_3
        # d/dh_3 q_1 = h_2
        # Correct approach: create two shifted versions, add them, but handle indices correctly
        left = np.roll(h, shift_idx)
        right = np.roll(h, -shift_idx)
        grad = left + right  # this is correct for the ith coefficient
        # Now we need to make sure we don't count reflections from out of bounds
        grad = np.where(np.arange(n) < n - shift_idx, grad, 0)  # extend to the right if available
        # For the right boundary we need both sides
        grad = np.where(np.arange(n) >= shift_idx, grad, 0)
        # So define properly for nonnegative index boundary
        grad = np.zeros_like(h)
        # Build for shift = s
        # Exact method assuming h defined in [0,n) and valid accordingly
        # For a fixed shift `s`, compute gradient of q_s = sum_i h_i h_{i+s} w.r.t h_i
        if shift_idx > 0:
            # The specific h_i influences the a_{i+s} term: h_i h_{i+s}
            # so they contribute: h_{i+s} to i, h_{i-s} to i+s if i >= s
            left = np.zeros_like(h)  # h_{i+s}
            right = np.zeros_like(h)  # h_{i-s}
            left[shift_idx:] = h[:-shift_idx]
            right[:-shift_idx] = h[shift_idx:]
            grad = left + right
        elif shift_idx == 0:
            grad = 2 * h
        grad = np.clip(grad, 0, 2)  # Should we clip it?
    else:
        left = np.zeros_like(h)
        right = np.zeros_like(h)
        left[shift_idx:] = h[:-shift_idx]
        right[:-shift_idx] = h[shift_idx:]
        grad = left + right
    return grad

def dc_update_step(h, target_shift=None):
    """Perform one DC update step based on target shift s* that minimizes q_s"""
    n = len(h)
    dx = 2.0 / n
    
    # A fast way to get correlation values and determine worst shift
    h_corr = correlate(h, h, mode="full")
    # The structure:
    # h_corr[k] is sum_{i} h[i] h[i - (k - n) ] = sum_{i} h[i] h[i - shift],
    # where shift = (k - n) (1-n) to n-1)
    # we're interested in sum_{i} h[i] h[i+s], that makes sense for s in [0, n-1]
    # h_corr look like:
    # [ ..., h_0 h_0 + h_1 h_1 + ... + h_{n-1} h_{n-1} (s=0),
    #    h_0 h_1 + h_1 h_2 + ... + h_{n-2} h_{n-1} (s=1),
    #    ..., h_0 h_0 + ... (s = n-1) ]
    # So h_corr[n-1] computes q_0, h_corr[n] computes q_1, down to 
    # h_corr [2n-2] corresponds to q_{n-1}
    # But we are actually computing 1-h, i.e., we do correlation between h_i and 1 - h_{i+s}
    # q_s = sum_i h_i (1 - h_{i+s}) = sum_i h_i - sum_i h_i h_{i+s} = 1 * n*dx - sum_i h_i h_{i+s}
    # So the new correlation term becomes
    # q_s(h) = 1 * n*dx - sum_i h_i h_{i+s}
    # By convention, the shifted function was h_{i+s} where index can go out-of-bound.
    # So compute for all shifts (0 to n-1) correlation of h with 1-h shifted (s = 0 to n-1)
    # But it's simpler: compute h o (1-h) (conventional correlation)
    # or (1-h) o h, or 1* h - h o h
    r_h = correlate(h, h, mode='full')
    r_1h = correlate(1-h, h, mode='full')
    r_h2 = correlate(h, 1-h, mode='full')
    # they should be same except maybe index alignment:
    # r_h2 = sum h[i](1-h[i+s]) = sum h[i] - sum h[i] h[i+s]
    # Thus final correlation for shift s is:
    # r_h2[n-1] to r_h2[n-1+n-1] = [n-1, 2n-2]
    # This gives the term for s = 0 to 2n-3, but actually is for shifts of h with (1-h)
    # also  (n-1+n-2) = 2n-3, so maximum real shift s is n-1
    # r_h2[n-1+i] for i = 0 to n-1 computes q_i
    corr_val = np.array(r_h2[n-1: 2*n-1])
    min_q = np.min(corr_val)
    s_min = np.argmin(corr_val)
    
    # Now compute the gradient for the minimum shift s_min  
    # This is d/dh_i q_{s_min}
    if s_min == 0:
        grad = 2*h
    else:
        left = np.zeros_like(h)
        right = np.zeros_like(h)
        left[s_min:] = h[:-s_min]
        right[:-s_min] = h[s_min:]
        grad = left + right
    # Apply fractal projection: capped-simplex projection to get new h that maximizes grad^T h
    # under 0 <= h <= 1, sum(h) = 1
    h_new = capped_simplex_projection(-grad)  # we want to minimize, so pick the direction of grad 
    # The gradient was actually 1*(1 - h) hence 1 - h contributes, if we think in positives
    # Change signs to reflect we want to reduce the correlation: gradient to be positive to indicate
    # what direction should we move to reduce the q_s
    # Normally gradient goes in the direction of increase, so to minimize, we drop it
    h_new = h_new * (1.0 / np.sum(h_new))  # Normalize to sum to 1
    return h_new, s_min

def dc_optimize_max_overlap(h0, max_iter=4000, tol=1e-8, time_limit=500):
    """Optimize through DC updates with capped simplex projection for each iteration""" 
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    c5_history = []
    iterations = 0
    start_time = time.time()

    h = project_sum_constraint(h, 1.0)  # ensure integral constraint

    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        h_old = h.copy()
        h, s_min = dc_update_step(h)
        h = project_sum_constraint(h, 1.0)  # renormalization
        c5 = evaluate_c5(h)
        c5_history.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        iterations += 1
        if len(c5_history) >= 2 and abs(c5_history[-1] - c5_history[-2]) < tol:
            # Stability check
            if i > 5:
                break
    return best_h, best_c5, iterations
