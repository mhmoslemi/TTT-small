import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def capped_simplex_projection(g, target_sum, bounds=(0, 1)):
    """Project gradient g onto the feasible set {h: sum(h)=target_sum, bounds[0] <= h[i] <= bounds[1]}"""
    # Sort indices by descending gradient
    idx = np.argsort(-g)
    h = np.zeros_like(g)
    remaining_sum = target_sum
    j = 0
    while j < len(idx) and remaining_sum > 0:
        i = idx[j]
        if remaining_sum >= bounds[1] - h[i]:
            h[i] = bounds[1]
            remaining_sum -= bounds[1] - h[i]
        else:
            h[i] = remaining_sum
            remaining_sum = 0
        j += 1
    if remaining_sum > 0:
        # Distribute remaining to the highest gradient components
        remaining portion = remaining_sum
        for i in idx[j:]:
            if remaining_portion <= 0:
                break
            delta = min(remaining_portion, bounds[1] - h[i])
            h[i] += delta
            remaining_portion -= delta
    return h

def dc_step(h, dx):
    """Perform one DC update step based on the worst shift"""
    n = len(h)
    # Compute full autocorrelation: q_s = sum h_i * h_{i+s}
    auto_corr = correlate(h, h, mode="full")
    # Find the shift that minimizes q_s, which maximizes overlap (1-q_s)
    # But we look at index which is the core of q_s, then compute shift
    # The full correlation has length 2n - 1, indexed from -(n-1) to (n-1)
    # For shift s=0 to n-1, we get q_s at index n-1+s (shifted)  
    # Therefore, normalize by dx to get actual q_s(h)
    # We look at the worst overlap: max(q_s) for all s -- best way to index?
    # We consider: for each shift s, q_s = sum_{i} h_i h_{i+s}
    
    # Select the index of the smallest q_s (most overlap)
    # q_s corresponds to index `n-1+s`, so `s = idx - (n-1)`
    # For s in [0, n-1], idx runs from n-1 to 2n-2, that is, 
    # automatch the worst contribution
    q_vals = auto_corr[n-1:]  # This indexes s = 0 to n-1 of the q_s values
    idx = np.argmin(q_vals)  # this is s*, the shift that minimizes q_s (maximizes overlap)
    s = idx
    # Now compute gradient of q_s(h) at current h: 
    # gradient wrt h_i is h_{i+s} + h_{i-s} (with wraparound in case of boundary)
    # We use finite difference, where h_{i+s} is h[(i+s)%n] and h_{i-s} is h[(i-s)%n]
    grad = np.zeros_like(h)
    for i in range(n):
        grad[i] = h[(i+s)%n] + h[(i-s)%n]
    # Now we want to solve min h^T (-grad) s.t. sum(h) = target_sum, 0 <= h <= 1
    h_new = capped_simplex_projection(-grad, target_sum=n/2)
    return h_new
