import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto bounds [0,1] and sum constraint"""
    lo, hi = 0.0, 1.0
    w = np.clip(v, lo, hi)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h, dx):
    """Compute maximum overlap integral and corresponding shift"""
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    argmax = np.argmax(conv)
    # Convert argmax index to shift m: m = argmax - (n-1)
    n = len(h)
    m = argmax - (n - 1)
    return max_val, m

def gradient_for_shift(h, m):
    """Compute gradient of overlap integral for given shift m"""
    n = len(h)
    g = np.zeros_like(h)
    
    if m >= 0:
        # Overlapping region for positive shift: i from 0 to n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # Negative shift case
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    return g * (2.0 / n)  # dx = 2/n

def evaluate_h(h):

    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val
