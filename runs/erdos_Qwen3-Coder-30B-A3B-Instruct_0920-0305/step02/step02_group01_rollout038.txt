import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

                    j = i + worst_shift_idx
                    if j < n_points:
                        grad[i] += h[j]

                    if j >= 0:
                        grad[i] += h[j]

                shift = worst_shift_idx - n_points + 1

                        grad[i] += h[i + shift]
                    if i - shift >= 0:

            # Solution is to rank the gradient and assign largest gradient entries to 1 or 0,

            indices_sorted = np.argsort(-grad)
