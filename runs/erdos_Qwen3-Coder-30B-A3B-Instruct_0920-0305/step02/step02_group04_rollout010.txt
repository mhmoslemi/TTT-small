import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(48) * 0.5
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Use MILP to find optimal binary solution if we have time
    best_c5 = evaluate_h(h0)
    best_h = h0.copy()
    
    # Try MILP optimization if we have reasonable time left
    if budget_s - (time.time() - start_time) > 60:
        try:
            # Define integer program with cvxpy
            n = n_points
            M = n // 2  # number of ones
            
            # Binary variables for h
            h = cp.Variable(n, boolean=True)
            T = cp.Variable(integer=True)  # integer variable for minimum sum
            
            # Constraints: sum of h equals M
            constraints = [cp.sum(h) == M, 
                           T >= 0]
            
            # Build linear constraints for product terms
            p_vars = []
            for ell in range(1, n):
                length = n - ell
                p_ell = cp.Variable(length, boolean=True)
                p_vars.append(p_ell)
                
                # Add linking constraints for p_ell: p_ell[i] ≈ h[i] * h[i+ell]
                # p_ell[i] <= h[i], p_ell[i] <= h[i+ell], p_ell[i] >= h[i] + h[i+ell] - 1
                constraints += [p_ell <= h[:length],
                                p_ell <= h[ell:ell+length],
                                p_ell >= h[:length] + h[ell:ell+length] - 1]
                
                # Constraint sum_i p_iell + T >= M
                constraints.append(cp.sum(p_ell) + T >= M)
            
            # Objective: minimize T
            objective = cp.Minimize(T)
            prob = cp.Problem(objective, constraints)
            
            # Solve with time limit
            remaining_time = min(50, budget_s - (time.time() - start_time) - 10)
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=remaining_time)
            
            # Extract solution if successful
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h.value
                if h_opt is not None:
                    # Convert to boolean numpy array
                    h_opt = np.round(h_opt).astype(int)
                    
                    # Ensure the integral constraint is respected by scaling
                    scale_factor = 1.0 / np.sum(h_opt * dx)
                    h_opt = h_opt * scale_factor
                    
                    # Evaluate the objective
                    test_c5 = evaluate_h(h_opt)
                    if test_c5 < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = test_c5
                        
        except Exception as e:
            # fallback to local optimization if MILP fails
            pass
    
    # Local optimization phase to refine the solution
    if time.time() - start_time < budget_s * 0.7:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_func(x):
            return np.sum(x) * dx - 1.0
        
        constr = {'type': 'eq', 'fun': constraint_func}

        result = None
        try:
            result = cp.minimize(evaluate_h, best_h, bounds=bounds, constraints=constr,

            h_opt = result.x
            c5_opt = evaluate_h(h_opt)

                best_h = h_opt.copy()
    
    # Final refinement with simulated annealing
    if time.time() - start_time < budget_s * 0.9:
        # Simulated Annealing
        current_h = best_h.copy()
        current_c5 = best_c5
        temp = 0.1
        cooling_rate = 0.995
        min_temp = 1e-6
        iteration_count = 0
        max_iter = 10000
        
        while temp > min_temp and iteration_count < max_iter and time.time() - start_time < budget_s * 0.99:

            i, j = np.random.randint(0, n_points, 2)
            h_new = current_h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            # Adjust to meet constraint
            scale = 1.0 / (np.sum(h_new) * dx)
            h_new *= scale
            
            try:
                c5_new = evaluate_h(h_new)
                
                # Accept or reject based on simulated annealing
                if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                    current_h = h_new
                    current_c5 = c5_new
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = current_h.copy()
                        
            except:
                pass
                
            temp *= cooling_rate
            iteration_count += 1
    
    # Ensure exact integral constraint
    scale = 1.0 / (np.sum(best_h) * dx)
    best_h = best_h * scale
    
    return (best_h.tolist(), best_c5, n_points)
