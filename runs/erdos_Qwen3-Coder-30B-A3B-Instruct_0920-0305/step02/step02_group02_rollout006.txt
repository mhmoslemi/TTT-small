def compute_C5(h_values):
    """Compute the overlap integral C5 for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * 0.0425531914893617
    return c5

def run_seed(seed, n_points, dx):
    """Run optimization for a specific seed and n_points."""
    np.random.seed(seed)
    
    # Generate initial solution using spectral method with optimal

    # Rescale to meet integral constraint exactly
    total = np.sum(h_init) * dx
    if abs(total - 1.0) > 1e-10:

    # Relaxation via SDP (not fully implemented in this solution due to time)

        integral = np.sum(h) * dx

    # Efficient local refinement
    def project_simplex(h, target_sum=None):
        if target_sum is None:
            target_sum = 1.0
        # Waterfilling projection onto simplex
        h_sorted = np.sort(h)[::-1]
        cumsum = np.cumsum(h_sorted)
        candidate_thresholds = (cumsum - target_sum) / np.arange(1, len(h)+1)
        threshold = candidate_thresholds[np.argmax(h_sorted - candidate_thresholds)]

    # Begin gradient-based refinement

    max_iter = 500

        eps = 1e-5
        h_l = project_simplex(np.clip(h_curr - eps, 0, 1))
        h_r = project_simplex(np.clip(h_curr + eps, 0, 1))
        c5_l = compute_C5(h_l)
        c5_r = compute_C5(h_r)
        grad = (c5_r - c5_l) / (2 * eps)
        step_size = 1e-3
        h_next = h_curr - step_size * grad
        h_next = project_simplex(h_next)
        
        # Update if improvement
        c5_next = compute_C5(h_next)

        h_curr = h_next
        if i > 10 and abs(c5_next - compute_C5(h_curr)) < 1e-6:
            break

    return best_h, best_c5
