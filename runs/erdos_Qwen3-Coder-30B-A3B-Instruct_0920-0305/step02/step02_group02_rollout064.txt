import numpy as np

def project_to_capped_simplex(v, w, tol=1e-12):
    """Project vector v onto the capped simplex with sum = w, 0 <= v_i <= 1."""
    v = np.clip(v, 0, 1)
    if abs(np.sum(v) - w) < tol:
        return v
    sorted_v = np.sort(v)[::-1]
    cumsum_sorted = np.cumsum(sorted_v)
    # Find threshold tau such that sum of min(v_i - tau, 1) = w
    # This is equivalent to finding tau where sum of (v_i - tau) if v_i > tau = w - sum of 1s
    # Or sum of (v_i - tau) if v_i > tau = w - n_ones
    # Let k0 be index where v_k0 > tau >= v_k0+1
    # Then: sum_{i=0}^{k0} (v_i - tau) = w - (k0 + 1)
    # Or sum_{i=0}^{k0} v_i - (k0 + 1) * tau = w - (k0 + 1)
    # Hence: tau = (sum_{i=0}^{k0} v_i - w + k0 + 1) / (k0 + 1)
    n = len(v)
    k0 = -1
    for k in range(n):
        if (cumsum_sorted[k] - w + k + 1) >= 0:
            k0 = k
            break
    if k0 == -1:
        # Rely on a uniform clipping, which is safe but extremely unlikely
        v = np.full_like(v, w / n)
        return v
    else:
        tau = (cumsum_sorted[k0] - w + k0 + 1) / (k0 + 1)
        projected = np.clip(v - tau, 0, 1)
        return projected

def compute_correlations(h):
    """Compute all auto-correlations using FFT."""
    nh = len(h)
    fft_h = fft(h, 2 * nh)
    corr_full = ifft(fft_h * np.conj(fft_h)).real
    corr = corr_full[nh - 1 : 2 * nh - 1]
    return corr

def compute_r_k(h, w):
    """Compute r_k = w - S_k where S_k is the auto-correlation."""
    correlations = compute_correlations(h)
    r_k = w - correlations
    return r_k

def compute_surrogate_gradient(h, r_k, alpha):
    """Compute gradient of the surrogate objective F_alpha."""
    # Compute softmax weights
    r_max = np.max(r_k)
    r_shifted = r_k - r_max
    exp_r = np.exp(alpha * r_shifted)
    sum_exp = np.sum(exp_r)
    weights = exp_r / sum_exp
    # Gradient is sum of weighted gradients of r_k
    g = np.zeros_like(h)
    nh = len(h)
    # r_k gradient for k = 0: 1 - 2*h_i
    g[0] += weights[0] * (1 - 2 * h[0])
    for i in range(1, nh-1):
        g[i] += weights[0] * (1 - 2 * h[i])  # r_k_grad for k=0
        g[i] += weights[i] * (-h[i-1])  # r_k_grad for k=i
        g[i-1] -= weights[i] * h[i]  # r_k_grad for k=i
    if nh > 1:
        g[nh-1] += weights[0] * (1 - 2 * h[nh-1])  # r_k_grad for k=0
        g[nh-2] -= weights[nh-1] * h[nh-1]  # r_k_grad for k=nh-1

    # Add terms for proper indexing
    tmp_g = np.zeros_like(h)
    for k in range(1, nh):
        shift = h[k:]
        tmp_g[k:] -= weights[k] * shift
        tmp_g[:nh-k] -= weights[k] * h[k:]
    # Add the derivative from r_k = w - S_k
    tmp_g[0] += weights[0]  # ablation for k=0

    # Add terms from k=0 (very simple)
    for i in range(nh):
        g[i] += weights[0]
        if i > 0:
            g[i-1] -= weights[1] * h[i]
            g[i] -= weights[1] * h[i-1]
        if i < nh - 1:
            g[i+1] -= weights[nh-1-i] * h[i]

    # Let's recompute everything properly using formulation:
    # F_alpha = (dx/alpha) * log(sum exp(alpha * r_k))
    # grad F_alpha = (dx/alpha) * sum_k w_k * gradient r_k
    # where w_k = exp(alpha * r_k) / sum_j exp(alpha * r_j)

    # Our simplified version:
    # r_k = w - S_k(h)
    # S_k(h) = sum_{i=0}^{n-k-1} h_i * h_{i+k}
    # So the derivative of S_k w.r.t. h_j
    #       = h_{j+k} if j <= n-k-1 and j+k < n
    #       = h_{j-k} if j>=k and j-k >= 0

    b = np.zeros_like(h)
    d = np.zeros_like(h)
    for k in range(1, nh):
        # Gradient of S_k with respect to h_j
        # k term: partial S_k / partial h_j = h_{j+k} if j+k < n else 0
        # k term: partial S_k / partial h_{j-k} = h_j if j >= k else 0
        hshift = np.zeros_like(h)
        hshift[k:] = h[:-k]  # h_{j+k}
        b += weights[k] * hshift
        if k < nh:
            hshift2 = np.zeros_like(h)
            hshift2[:-k] = h[k:]  # h_{j-k}
            b += weights[k] * hshift2

    g = weights[0] * np.ones_like(h) - b  # + 1 for first-term derivative of r_k = w - S_k
    # Simpler corret definition:
    # For the first term: r_k = w - S_k(h)
    # So grad r_k = 0 - grad S_k = - grad S_k
    # and we compute term by term in grad S_k, not through formula.
    # Obvious:  
    # d_r_k / d h_i = - (h_{i+k} if i+k < n) - (h_{i-k} if i-k >= 0)
    # which is valid for k >= 1
    # So for k=0: grad r_k = 0 - 2*h_i -> r_k = w - sum h_i^2 -> gradient -2*h
    # + 
    d = np.zeros_like(h)
    for k in range(1, nh):
        shift1 = np.zeros_like(h)
        shift2 = np.zeros_like(h)
        if k <= nh - 1:
            shift1[k:] = h[:-k]
        if k <= nh - 1:
            shift2[:-k] = h[k:]
        d = d - (weights[k] * shift1) - (weights[k] * shift2)
    d[0] = d[0] - (weights[0] * 2 * h[0])
    for i in range(1, nh):
        d[i] = d[i] - (weights[0] * 2 * h[i])
        if i + k < nh:
            d[i] -= (weights[k] * h[i+k])
        if i - k >= 0:
            d[i] -= (weights[k] * h[i-k])
    
    return d

def surrogate_objective(h, r_k, alpha, dx):
    """Compute surrogate objective F_alpha."""
    r_max = np.max(r_k)
    r_shifted = r_k - r_max
    exp_r = np.exp(alpha * r_shifted)
    sum_exp = np.sum(exp_r)
    return (dx / alpha) * np.log(sum_exp)

def run_fista_surrogate(h0, w, alpha, dx, max_iter=500, tol=1e-8):

        # Compute surrogate F_alpha

def initial_h_value(n):
    """Helper to get initial h values."""
    # Start with a constructed solution
    a = n // 4
    b = n // 2
    h = np.zeros(n)
    h[0:a] = 1
    h[a:b] = 0
    # Add a bit more control
    h = h + np.random.normal(0, 0.02, n)
    h = np.clip(h, 0, 1)
    h = project_to_capped_simplex(h, n / 2)
    return h

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    np.random.seed(seed)
    best_bound = 1e10
    best_h = None
    best_n = None
    start_time = time.time()
    
    n_idx = 0
    while time.time() - start_time < budget_s * 0.95:
        n = n_values[n_idx]
        dx = 2.0 / n
        w = n / 2
        # Try 3 restarts for this n
        max_restarts = 3
        restart_count = 0
        while restart_count < max_restarts and time.time() - start_time < budget_s * 0.95:
            h = sample_initial_h(n, w)
            # Run surrogate optimization with continuation
            alpha = 1.0
            alpha_max = 400.0
            while alpha < alpha_max and time.time() - start_time < budget_s * 0.95:
                h = run_fista_surrogate(h, w, alpha, dx, max_iter=200)
                alpha = min(alpha * 2, alpha_max)
            # Evaluate exact bound
            # Compute correlations and r_k
            corr = np.correlate(h, 1-h, mode='full')
            c5 = dx * np.max(corr)
            if c5 < best_bound:
                best_bound = c5
                best_h = h.copy()
                best_n = n
                if best_bound < 0.379:
                    break
            else:
                # Reduce number of restarts if we are stuck
                pass
            restart_count += 1
        n_idx = (n_idx + 1) % len(n_values)
        if best_bound < 0.379:
            break

    return list(best_h), best_bound, best_n
