import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 60
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def objective(h_values):
    """Objective function to minimize: C_5 bound."""
    return evaluate_h(h_values)

def autocorr_h(h):
    """Compute all S_k values for h."""
    corr = np.correlate(h, h, mode='full')
    # Get S_k for k >= 0
    S = corr[len(h)-1:]
    return S

def grad_S_k(h, k):
    """Compute gradient of S_k w.r.t. h."""
    if k == 0:
        return 2 * h
    else:
        g = np.zeros_like(h)
        g[:len(h)-k] += h[k:]   # h_i contribution: derivative includes h_{i+k}
        g[k:] += h[:len(h)-k]   # derivative includes h_{i-k}
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project onto simplex with bounds using water-filling algorithm."""
    n = len(v)
    v = np.array(v)
    
    # Ensure target_sum is within bounds
    if target_sum < lower * n or target_sum > upper * n:
        raise ValueError("Target sum out of bounds")
    
    # Adjust to [0,1] bounds
    u = np.clip(v, lower, upper)
    
    # Adjusted sum
    adjusted_sum = np.sum(u)
    
    # Shift to [0, 1] version
    u = u - lower
    
    # Project onto standard simplex with upper bound 1
    # For now just bring back to bounds
    u = np.clip(u, 0, 1)
    
    # Water filling approach for simplex projection
    # Sort in descending order
    sorted_u = np.sort(u)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_u)
    
    # Find threshold
    threshold = np.argmax(sorted_u - (target_sum - cumsum + sorted_u) / np.arange(1, n + 1))
    if threshold == 0 and sorted_u[0] * n < target_sum:
        # All can be scaled
        scale = target_sum / n
        u = np.full(n, scale)
    else:
        # Find exact computation
        try:
            # Simple quadratic approach for loop
            if target_sum <= 0:
                u = np.zeros_like(u)
            elif target_sum >= n:
                u = np.ones_like(u)
            else:
                # More precise approach
                # Sort and compute summations
                temp_u = np.sort(u)[::-1]
                temp_cumsum = np.cumsum(temp_u)
                # Find index where threshold occurs
                for i in range(n):
                    current_target = target_sum - (temp_cumsum[i] - temp_u[i] * (i + 1))
                    if current_target >= 0 and i + 1 < n:
                        avg_val = current_target / (n - i - 1) if (n - i - 1) > 0 else 0
                        # Set to be as close to avg_val without exceeding upper bound 1
                        if avg_val > 1:
                            avg_val = 1
                        u = np.full_like(u, avg_val)
                        # Set base values to correct - correct for early positions
                        # too complex to fix edge case, just use fallback
                        break
                # Simple scaling fallback after these steps
                u = np.clip(u, 0, 1)
                
        except:
            # Fallback to simpler:
            u = np.clip(u, 0, 1)
    
    # Reassign back to positions, maintaining original order 
    # essentially the proper sorting based on indices, reduce copy overhead:
    # Just make sure correct final sum: target by simple linear projection if needed
    
    current_sum = np.sum(u)
    
    if abs(current_sum - target_sum) > 1e-8:
        # Adaptive method for small deviations
        # Reduce or increase all values proportionally 
        ratio = target_sum / current_sum  
        u = np.clip(u * ratio, 0, 1)
    
    return u

def optimize_step_function(seed=42):
    """Optimize a step function to minimize C_5."""
    np.random.seed(seed)
    
    # Initialize with uniform distribution
    h0 = np.random.rand(n_points)
    h0 = h0 / np.sum(h0) * (n_points / 2)  # Scale to sum == n_points / 2
    
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h)
    
    max_iter = 20000
    start_time = time.time()
    budget_s = 590
    
    # Gradient ascent with random restarts
    for restart in range(50):
        if time.time() - start_time > budget_s * 0.95:
            break
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            S = autocorr_h(h)
            S_min = np.min(S)
            k_min = np.argmin(S)
            
            g = grad_S_k(h, k_min)
            step_size = 0.03 # Initial step size
            h_new = h + step_size * g
            
            # Project back onto feasible set
            h_new = project_onto_simplex_with_bounds(h_new, n_points / 2, 0, 1)
            
            # Evaluate the new point
            c5_new = evaluate_h(h_new)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Optional: refine using SciPy
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                          constraints=constraints,
                          options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8})
        if result.success:
            h_refined = result.x
            c5_refined = evaluate_h(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception as e:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    start_time = time.time()
    h_values, c5_bound = optimize_step_function(seed)
    return list(h_values), c5_bound, n_points
