import numpy as np
from scipy.linalg import eigh
from cvxpy import *
import time

def project_to_capped_simplex(y, target_sum):
    # Project y onto the capped simplex: {x: 0 <= x <= 1, sum x = target_sum}
    y = np.asarray(y)
    low, high = 0.0, 1.0
    n = len(y)
    if abs(np.sum(y) - target_sum) < 1e-10:
        return np.clip(y, low, high)
    
    def F(lam):
        return np.sum(np.clip(y - lam, low, high)) - target_sum
    
    # Binary search for lambda such that F(lambda) = 0
    # Initial bounds for lambda
    lam_low = -np.max(y) - 1.0
    lam_high = np.max(y) + 1.0
    
    for _ in range(40):
        lam = (lam_low + lam_high) / 2.0
        f_val = F(lam)
        if f_val > 0:
            lam_low = lam
        else:
            lam_high = lam
    lam = (lam_low + lam_high) / 2.0
    x = np.clip(y - lam, low, high)
    return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    # Use scipy.signal.correlate with mode="full", but manually compute
    # to avoid potential issues with correlation implementation for C5
    h = np.asarray(h)
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Constants
    ns = [30, 35, 40, 45, 50]  # Test different resolutions, small enough for SDP
    target_n = 180
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Phase 1: Solve SDP for several n values
    for n in ns:
        if time_left() < 30:
            break
        try:
            X = Variable((n, n), PSD=True)
            h = Variable(n)
            # Constraints for the SDP relaxation
            constraints = [X[i, i] == h[i] for i in range(n)]
            # Box constraints on h
            constraints += [0 <= h, h <= 1]
            # Integral constraint
            constraints += [sum(h) == n / 2]

            # Generate all possible shifts
            shifts = list(range(-n+1, n))  # -n+1 to n-1 (inclusive)
            r = {s: 
                 sum(X[i, i + abs(s)] for i in range(n - abs(s))) 
                 if s >= 0 else 
                 sum(X[i + abs(s), i] for i in range(n - abs(s))) 
                 for s in shifts}
            # Add constraints r_s >= m for all shifts
            m = Variable()
            for s in shifts:
                constraints += [r[s] >= m]
            # Maximize m
            objective = Maximize(m)
            prob = Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False, feastol=1e-5, reltol=1e-5)
            if prob.status == 'optimal':
                m_val = m.value
                X_val = X.value
                h_val = h.value
                # Extract feasible vector from low-dim X and h
                # (We're using same hijacking of h_val as solution)
                # positive eigenvector
                _, v = eigh(X_val)
                # Leading eigenvector
                w = v[:, -1]
                h_est = np.abs(w)
                # Scale to sum == n/2, then clip to [0,1]
                scale = (n / 2) / np.sum(h_est)
                h_clipped = np.clip(scale * h_est, 0, 1)
                # Exact projection to simplex
                h_feasible = project_to_capped_simplex(h_clipped, n / 2)
                c5 = evaluate_c5(h_feasible)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_feasible.copy()
                    best_n = n
        except Exception as e:
            # Skip if error occurs (solver failure, etc.)
            continue

    # If we didn't get a good initial solution, try the initial_h_values
    if best_h is None:
        h0 = np.array(initial_h_values)
        try:
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_n = len(h0)
        except:
            pass

    # Phase 2: Refine if we have a candidate
    if best_h is not None:
        # Local search: Nelder-Mead with projection
        best_h_refined = best_h.copy()
        current_c5 = evaluate_c5(best_h_refined)
        
        # Local refinement with projected Nelder-Mead
        # Simple boosting: add small perturbations
        n_ref = max(30, min(50, best_n))
        if time_left() > 100:
            try:
                from scipy.optimize import minimize
                
                def obj_h(v):
                    return evaluate_c5(v)
                
                def proj_h(v):
                    return project_to_capped_simplex(v, n_ref / 2)
                
                for i in range(5):
                    # Refine 5 times with small perturbations
                    h_perturbed = best_h_refined + np.random.normal(0, 0.01, n_ref)
                    # Apply projection step to enforce feasibility
                    h_perturbed = proj_h(h_perturbed)
                    v = h_perturbed
                    # Optimization step
                    try:
                        res = minimize(obj_h, v, method='Nelder-Mead', 
                                       options={'maxiter': 50, 'xatol': 1e-6, 'fatol': 1e-6})
                        if res.success:
                            v = res.x
                            v = proj_h(v)
                            c5_new = obj_h(v)
                            if c5_new < current_c5:
                                best_h_refined = v.copy()
                                current_c5 = c5_new
                    except:
                        pass
            except Exception as e:
                pass
            # Final evaluation
            final_c5 = evaluate_c5(best_h_refined)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = best_h_refined

    # Phase 3: Upsample to n_target using linear interpolation
    if best_h is not None:
        n_target = target_n
        # Linear interpolation from current points to target points
        # Assume [0,2] interval; grid points
        x_orig = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_target)
        h_upsampled = np.interp(x_new, x_orig, best_h)
        # Apply final projection
        h_upsampled = project_to_capped_simplex(h_upsampled, n_target / 2)
        # Final evaluation
        final_c5 = evaluate_c5(h_upsampled)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_upsampled
            best_n = n_target
    
    # Final output
    if best_h is None:
        # Fallback to something decent
        h0 = np.array(initial_h_values)
        best_h = h0
        best_n = len(h0)
        best_c5 = evaluate_c5(best_h)
    
    # Ensure the exact integral constraint
    scale = (best_n / 2) / np.sum(best_h)
    best_h = best_h * scale
    best_h = project_to_capped_simplex(best_h, best_n / 2)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), best_n)
