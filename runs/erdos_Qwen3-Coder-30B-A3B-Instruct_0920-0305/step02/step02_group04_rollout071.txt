import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

    start_time = time.time()

    dx = 2.0 / n

    # Create variables
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True)
    
    # Constraints
    constraints = [cp.sum(h) == M, T >= 0]
    
    # Product variables and constraints
    p_vars = []
    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p_vars.append(p_ell)
        # Add linking constraints: for i in range(length)
        constraints += [p_ell <= h[:length],
                        p_ell <= h[ell:ell+length],
                        p_ell >= h[:length] + h[ell:ell+length] - 1]
        constraints.append(cp.sum(p_ell) + T >= M)
    
    # Objective
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    remaining_time = budget_s * 0.3 - (time.time() - start_time)
    if remaining_time > 0:
        try:
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=remaining_time)
        except:
            pass
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h.value
        T_sol = T.value
        if h_sol is not None and T_sol is not None:
            # Convert to list of floats
            h_vals = h_sol.astype(float)
            c5_bound = T_sol * dx
            return (h_vals.tolist(), c5_bound, n)
    
    # If MILP fails, initialize with a known good construction
    # Use initial_h_values but rescale to sum to M
    h_vals = np.array(initial_h_values)
    h_vals = h_vals / np.sum(h_vals) * M
    c5_bound = evaluate_h(h_vals)
    return (h_vals.tolist(), c5_bound, len(h_vals))
