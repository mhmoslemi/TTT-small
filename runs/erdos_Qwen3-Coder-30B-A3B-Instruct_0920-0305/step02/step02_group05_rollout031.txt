import numpy as np
from scipy.signal import correlate
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project v onto the set of vectors with bounds [0,1] and given sum."""
    lo, hi = 0.0, 1.0
    # Clip to bounds
    w = np.clip(v, lo, hi)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a specific shift m."""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx factor

def compute_max_overlap(h, dx):
    """Compute the maximum overlap integral and the corresponding shift index."""
    conv = correlate(h, 1 - h, mode='full')
    conv *= dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    # Convert index to shift value
    shift = argmax - (len(h) - 1)
    return max_val, shift

def interpolate_to_n(h0, n):
    """Interpolate h0 to n points."""
    old_n = len(h0)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, n, endpoint=False)
    h = np.interp(new_x, old_x, h0)
    # Adjust to match sum constraint
    current_sum = np.sum(h)
    target_sum = n / 2
    h *= target_sum / current_sum
    return h
