import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def solve_milp_binary(n, time_limit):
    """Solve binary MILP for optimal h vector"""
    M = n // 2  # assume n even
    h = cp.Variable(n, boolean=True)
    # T real variable
    T = cp.Variable(integer=True)
    
    # Constraints for sum
    constraints = [cp.sum(h) == M,
                   T >= 0]
    
    # Build product variables p for each shift
    p_vars = []
    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p_vars.append(p_ell)
        # linking constraints
        constraints += [p_ell <= h[:length],
                        p_ell <= h[ell:],
                        p_ell >= h[:length] + h[ell:] - 1]
        # Sum constraint for each shift
        constraints.append(cp.sum(p_ell) + T >= M)
    
    # Objective: minimize T
    objective = cp.Minimize(T)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
        return h.value, T.value
    except:
        return None, None
