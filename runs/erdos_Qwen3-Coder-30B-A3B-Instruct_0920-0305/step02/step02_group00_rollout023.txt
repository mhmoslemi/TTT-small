import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_simplex(v, target_sum):
    """
    Project vector v onto the box [0,1]^n and then onto the simplex {x : sum(x) = target_sum}.
    Uses bisection to find the right λ for clipping.
    """
    # Clip to [0,1] with a shift
    def f(lam):
        clipped = np.clip(v - lam, 0, 1)
        return np.sum(clipped) - target_sum

    # Binary search for zero of f
    a, b = np.min(v), np.max(v) + 1.0
    for _ in range(100):
        mid = (a + b) / 2.0
        if f(mid) > 0:
            a = mid
        else:
            b = mid
    lam = (a + b) / 2.0
    x = np.clip(v - lam, 0, 1)
    return x

def compute_shifted_product(h, shift):
    """Compute the shifted product sum for a given shift.
       This simulates h^T A^{(shift)} h where A^{(shift)} is symmetric shift matrix.
    """
    n = len(h)
    if shift == 0:
        return np.sum(h * h)
    else:  # shift > 0
        # Simulate A^{(shift)} h efficiently
        # For a shift s, (A^{(shift)} h)_i = 0.5 * (h_{i-s} + h_{i+s}) if valid
        # For simplicity of implementation using roll and add
        h_shifted_back = np.roll(h, shift)
        h_shifted_forward = np.roll(h, -shift)
        # Only the overlapping part (middle part) should count for shift s
        # We do it as convolution / correlation way:
        # For s>0:
        if shift > 0:
            # h_shifted_backward will have h_{i-shift} at temp_location i
            # for i=shift to n-1. So sum_{i=0}^{n-1-shift} h_i * h_{i+shift}
            # corresponds to corr = np.correlate(h, h_shifted_back, 'valid')
            # But we also need to account for 0.5 factor after weighting
            # h has to be shifted forward to get h_{i-shift} (we have h_{i+shift})
            h_backward = np.roll(h, shift)
            # h_{i-s} for i >= s and h_{i+s} for i < n-s
            # But we want h_{i-s} + h_{i+s}
            h_backward[0:shift] = 0
            h_forward = np.roll(h, -shift)
            h_forward[-shift:] = 0
            return np.sum(0.5*(h_backward + h_forward)*h)
    raise ValueError("shift should be >= 0")

def soft_min_grad(h, beta=10.0):
    """Compute the gradient of soft minimum using shift structures."""
    n = len(h)
    dx = 2.0 / n
    # Precompute all C_s(h) = dx * O_s(h), O_s = sum_i h_i h_{i+s}
    # For simplification, use np.correlate (same convention as with evaluate_h)
    C_vals = np.correlate(h, 1-h, mode='full') * dx  # [n: 2n-1], peaks at n-1
    # Maximum indices with the index shift from 0 to n-1
    C_vals = C_vals[n-1:]  # Interval [0, n-1] as required for indices

    # Soft-min weight computation for current h; using shift way consistently
    max_C = np.max(C_vals)
    exp_neg_beta_C = np.exp(-beta * (C_vals - max_C))  # Avoid exp overflow
    w = exp_neg_beta_C / np.sum(exp_neg_beta_C)

    # Compute partial derivatives for each shift: dC_s/h = 2dx * A^(s) h
    # But precompute A^(s) h efficiently:
    # Use fast convolution + recombination technique

    # For s=0: partial derivative = 2*dx * h
    grad_total = 2*dx * w[0]*h

    # For s=1 to n-1:
    for s in range(1, n):
        # Shift h appropriately
        h_back = np.roll(h, s)
        h_front = np.roll(h, -s)
        # Apply shift matrix on both directions; effective: average of shift
        # But also take account that for symmetric form handles both directions
        # Efficiently compute as: 
        # FWD denote convolution_, BWD denote shift_back,
        # in this case symmetric matrix corresponds to careful indexing
        # We use np.correlate again; more cleanly:
        # A^(s) h = 0.5 (h_{i-s} + h_{i+s})
        # which we compute from rolled version: h_roll_inv = roll(h, -s)
        # and h_roll = roll(h, s)
        h_sl = np.roll(h, s)
        h_sr = np.roll(h, -s)
        part_h = h_sl + h_sr
        part_h[0:s] = 0  # Left, no part from left side; but problem is in middle
        part_h[-s:] = 0  # Right, no part from right side; 
        # Actually correct way: is to apply 0.5 * (h_{i+s} + h_{i-s})
        # For simplicity apply weighted average where index range matters:
        # indices <= s we don't get h_{i-s},
        # indices >= n-s we would miss h_{i+s}  
        # We reconstruct all manually:
        h_back = np.zeros_like(h)
        h_forw = np.zeros_like(h)
        # shift s bits right = left shift for h_i->h_{i+s}
        h_back[s:] = h[:-s]  # value shift left, org positions valid  
        h_forw[:-s] = h[s:]  # shift right, -> i -> i+s olds
        h_should = (h_back + h_forw)  # Correctly constructed form
        h_should[s:n-s] *= 0.5  # But only central valid
        grad_total += 2*dx * w[s] * h_should

    return -grad_total  # Minus sign because we want to maximize the soft-min.

def compute_gradient_mac(h, beta=10.0):
    """Much more efficient version of gradient computation """
    n = len(h)
    dx = 2.0 / n
    
    # Compute all C_s using np.correlate but carefully with full mode and extract
    h_vec = np.array(h)
    corr_vals = np.correlate(h_vec, 1 - h_vec, mode='full')
    # Retain only from shift = 0 to n-1 (that's what we care for)
    C_vals = corr_vals[n-1:] * dx
    
    # Soft weights based directly on actual values:
    # Better idea, just subtract max of C_vals to avoid numerical issues
    max_C = np.max(C_vals)
    exp_neg_beta_C = np.exp(-beta * (C_vals - max_C))
    w = exp_neg_beta_C / np.sum(exp_neg_beta_C)
    
    # Now reuse exact formulation of shift matrix derivative:
    # For shift s, it is: A^{(s)} h = sum_{i} h_{i} h_{i+s} = inner product
    # and hence d(h^T A s h) = 2 A^{(s)} h, so simplified
    # gradient = 2 dx * sum_s w_s A^{(s)} h  
    # Use numpy to vec-shift and sum:
    grad = np.zeros(n)
    
    # Fast walk through all shifts:
    for s in range(0, n):
        # Compute how A^(s) h must be computed
        if s == 0:
            # L_s = sum h_i^2
            deriv_L = 2.0 * h_vec
        else:
            # deriv_{h^T A^{s} h} = 2 * A^{s} h = 2 *[ h_{i+s} + h_{i-s} ] / 2 
            shifted_forward = np.roll(h_vec, -s)
            shifted_back = np.roll(h_vec, s)
            # careful to zero pad where needed
            deriv_L = shifted_forward + shifted_back
            # But handle shifting correctly using padding directly
            # Only valid range of inner products: exactly from offset s
            # But let np.roll do wrapping: 
            # That produces diagonal element identically for any length n shift matrix
            # But to make product manually:
            padded_left = np.pad(shifted_back, (s, 0), mode='constant', constant_values=0)
            padded_right = np.pad(shifted_forward, (0, s), mode='constant', constant_values=0)
            # Now truncate to valid indices for summation:
            deriv_L = (0.5 * padded_left + 0.5 * padded_right)[s:n-s]  
            # [s, n-s] is good enough for valid sums
            # But overall it still would be computed with proper alignment
            # Just make sure cumulative result is right in case of aliasing
            # right approach: using the other method: just treat each s                        
            # Trick here: instead of fixed full shuffling:
            # work with pair of roll operations, but average symmetrically        
            deriv_L = np.zeros_like(h_vec)
            # Grow both sides effectively:
            # For i=s+1 to n-s: get h_{i-s} + h_{i+s}
            if s > 0:
                deriv_L[s:n-s] = (np.roll(h_vec, s) + np.roll(h_vec, -s))[s:n-s] * 0.5
            
        # handle special case correctly for i -s  when s=0 we take h_i itself (h^2), 
        # but actually it should be (0.5 * (0 + h_i*2)) = h_i
        # so we make sure derivative is always computed
        if s == 0:
            # special because internal loop missing 
            deriv_L = 2*dx * w[0] * h_vec
        else:
            # Derivative for s-th shift        
            answer_s = 2*dx * w[s] * deriv_L
            grad += answer_s
    return -grad  # Sharp min is max of negated
    
def gradient_ascent_step(h, beta, step_size=0.01):
    g = compute_gradient_mac(h, beta)
    h_new = h + step_size * g
    return h_new
