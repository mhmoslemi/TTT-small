def evaluate_c5(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto [lower,upper]^n with sum = target_sum."""
    v = np.array(v)
    # Project onto simplex
    u = v.copy()
    u = np.clip(u, lower, upper)
    # Adjust to meet exact sum
    current_sum = np.sum(u)
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return u
    # Add diff evenly (but clipped)
    if diff > 0:
        # Need to increase values
        indices = np.argsort(u)
        for i in indices:
            delta = min(diff, upper - u[i])
            if delta <= 0:
                continue
            u[i] += delta
            diff -= delta
            if diff <= 1e-10:
                break
    else:
        # Need to decrease values
        indices = np.argsort(u)[::-1]
        for i in indices:
            delta = min(-diff, u[i] - lower)
            if delta <= 0:
                continue
            u[i] -= delta
            diff += delta
            if diff >= -1e-10:
                break
    return u
