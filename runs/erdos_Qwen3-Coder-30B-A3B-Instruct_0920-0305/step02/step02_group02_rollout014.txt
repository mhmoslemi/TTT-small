import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import math

def project_simplex(v, w):
    """Project v onto the capped simplex: 0 <= v_i <= 1, sum(v) = w.
    Uses water-filling algorithm.
    """
    if np.sum(v) == w and np.all((v >= 0) & (v <= 1)):
        return v.copy()
    
    # sort in descending order (indices)
    idx = np.argsort(v)[::-1]
    sorted_v = v[idx]

    # compute cumulative sums in descending order
    cumsum = np.cumsum(sorted_v)
    
    # find threshold
    n = len(v)
    z = -np.inf
    for i in range(n):
        if sorted_v[i] - (w - cumsum[i])/(i+1) >= 0:
            z = (w - cumsum[i]) / (i+1)
            break

    # compute the projection
    x = np.maximum(np.minimum(sorted_v - z, 1.0), 0.0)

    # permute back to original index order
    result = np.zeros_like(v)
    result[idx] = x

    # fix floating point errors to exactly satisfy sum
    result = result * w / np.sum(result)

    return result

def compute_r(h, w, dx):
    """Compute r_k = w - S_k for all shifts k, where S_k is auto-correlation."""
    # z-pad to avoid wrap-around issues
    n = len(h)
    h_padded = np.concatenate([h, np.zeros(n)])
    h_conj = np.concatenate([h[::-1], np.zeros(n)])
    s_full = np.real(ifft(fft(h_padded) * np.conj(fft(h_conj))))
    s_k = s_full[n-1:n-1+n]
    r = w - s_k
    return r

def compute_grad_F_alpha(h, w, dx, alpha):
    """Compute gradient of F_alpha(h) = (dx/alpha)*log(sum(exp(alpha * r_k)))"""
    n = len(h)
    r = compute_r(h, w, dx)
    max_r = np.max(r)
    r_shifted = r - max_r
    exp_r = np.exp(alpha * r_shifted)
    Z = np.sum(exp_r)
    weights = exp_r / Z
    
    # j-th component of gradient: sum_k w_k * (1 - h_{j+k} - h_{j-k})
    grad = np.zeros(n)
    for k in range(n):
        if k == 0:
            grad += weights[k] * (1 - 2*h)  # special case of r_0
        else:
            # h_j * (1 - h_{j+k}) - (1 - h_{j+k}) * h_j = h_j (1 - h_{j+k}) - (1 - h_{j+k}) h_j
            # but actually ∂r_k/∂h_j = 1 - h_{j+k} - h_{j-k} for j-k>=0 and j+k<n
            # compute the chunk of the gradient due to this shift k
            if k < n:
                pad_left = np.concatenate([np.zeros(k), h[:-k]])
                pad_right = np.concatenate([h[k:], np.zeros(k)])
                diff = 1 - pad_left - pad_right
                grad += weights[k] * diff

    # scaled gradient
    return dx * grad

def fista_alpha_smoothed(h0, w, dx, alpha_max=500.0, alpha_init=1.0, eps=1e-8):
    """Run accelerated projected gradient on the log-sum-exp surrogate."""
    n = len(h0)
    alpha = alpha_init
    h_old = h0.copy()
    t_old = 1.0
    max_iter = 5000

    while alpha <= alpha_max:
        h_new = h_old.copy()
        t_new = t_old
        iter_count = 0
        batch_size = 0
        
        while iter_count < max_iter:
            # compute the current function value and gradient
            r = compute_r(h_new, w, dx)
            r_max = np.max(r)
            r_shifted = r - r_max
            exp_r = np.exp(alpha * r_shifted)
            Z = np.sum(exp_r)
            logF = (dx / alpha) * (np.log(Z) + r_max)
            
            grad = compute_grad_F_alpha(h_new, w, dx, alpha)
            
            # line search but check improvement in function value instead of Armijo
            eta = 0.5
            for _ in range(50):
                h_temp = project_simplex(h_new - eta * grad, w)
                r_temp = compute_r(h_temp, w, dx)
                r_temp_max = np.max(r_temp)
                r_temp_shift = r_temp - r_temp_max
                exp_r_temp = np.exp(alpha * r_temp_shift)
                Z_temp = np.sum(exp_r_temp)
                logF_temp = (dx / alpha) * (np.log(Z_temp) + r_temp_max)
                
                if logF_temp <= logF - 1e-12:
                    h_new = h_temp.copy()
                    break
                eta *= 0.5
            
            # update momentum
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_old**2)) / 2.0
            h_new = project_simplex(h_new - eta * grad, w)
            y_new = h_new + ((t_old - 1.0) / t_new) * (h_new - h_old)
            
            if abs(logF - logF_temp) / (abs(logF) + 1e-10) < eps:
                break
            
            h_old = h_new
            t_old = t_new
            iter_count += 1
            
        h_old = h_new
        alpha *= 2.0

    return h_new, compute_r(h_new, w, dx)

def evaluate_final(h, dx):
    """Evaluate the actual C5 using exact correlation."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Candidate values for n
    n_values = [48, 56, 64, 80]
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    restart_budget = 50
    n_budget = budget_s / len(n_values)  # per n
    
    for n in n_values:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        w = n / 2.0
        
        # run multiple restarts for each n
        best_h_n = None
        best_c5_n = float('inf')
        
        for seed_restart in range(restart_budget):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Generate different initializations
            choice = seed_restart % 4
            if choice == 0:
                # Uniform random
                h0 = np.random.rand(n)
                h0 = project_simplex(h0, w)
            elif choice == 1:
                # Binary biased
                h0 = np.zeros(n)
                ones = np.random.choice(n, size=int(w), replace=False)
                h0[ones] = 1.0
                h0 = project_simplex(h0 + np.random.normal(0, 0.02, n), w)
            elif choice == 2:
                # Three-block heuristic
                a = np.random.uniform(0, 1)  # split point 1
                b = np.random.uniform(a, 2)  # split point 2
                x1 = a / 2.0
                x2 = (b - a) / 2.0
                val = (w - x1 * n) / x2 if x2 > 0 else 0.5
                if val < 0:
                    p = 0.0
                elif val > 1:
                    p = 1.0
                else:
                    p = val
                h0 = np.zeros(n)
                for i in range(n):
                    x = dx * i
                    if x < a:
                        h0[i] = 1
                    elif x < b:
                        h0[i] = p
                    else:
                        h0[i] = 0
                h0 = project_simplex(h0, w)
            else:
                # Warm-start from best if exists, otherwise random
                if best_h_n is not None:
                    idx = np.random.randint(n)
                    delta = 0.05
                    new_h = best_h_n.copy()
                    if new_h[idx] > delta:
                        new_h[idx] -= delta
                        new_h[np.random.randint(n)] += delta
                    h0 = project_simplex(new_h, w)
                else:
                    h0 = project_simplex(np.random.rand(n), w)
                    
            try:
                h_final, r_vals = fista_alpha_smoothed(h0, w, dx)
                c5 = evaluate_final(h_final, dx)
                
                if c5 < best_c5_n:
                    best_c5_n = c5
                    best_h_n = h_final.copy()
                    
            except Exception as e:
                continue
                
        # evaluation of the best for this n
        if best_h_n is not None:
            final_c5 = evaluate_final(best_h_n, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = best_h_n.copy()
                best_n = n
                
        if time.time() - start_time > budget_s * 0.95:
            break
                
    if best_h is None:
        # fallback
        n = 48
        dx = 2.0 / n
        w = n / 2.0
        best_h = project_simplex(np.random.rand(n), w)
        best_c5 = evaluate_final(best_h, dx)
        best_n = n

    return list(best_h), best_c5, best_n
