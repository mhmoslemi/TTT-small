import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import cvxpy as cp
from cvxpy import Minimize, Problem, Bool, Int

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT-based correlation."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Pad for full correlation
    h_pad = np.pad(h, (0, n_points - 1), 'constant')
    h1_pad = np.pad(1 - h, (0, n_points - 1), 'constant')
    corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(h1_pad))))
    c5 = np.max(corr[:2*n_points - 1]) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    # 1. Solve MILP
    h_vars = cp.Variable(n, boolean=True)
    p_vars = {}
    for l in range(1, n):
        p_vars[l] = cp.Variable((n - l), boolean=True)
    T = cp.Variable(integer=True)
    
    # Mass constraint
    constraints = [cp.sum(h_vars) == M]
    # Product constraints for each lag
    for l in range(1, n):
        for i in range(n - l):
            p = p_vars[l][i]
            h_i = h_vars[i]
            h_i_plus_l = h_vars[i + l]
            constraints.extend([
                p <= h_i,
                p <= h_i_plus_l,
                p >= h_i + h_i_plus_l - 1,
                p >= 0
            ])
    # Worst-lag constraint
    for l in range(1, n):
        lhs = cp.sum(p_vars[l]) + T
        constraints.append(lhs >= M)
    
    # Objective
    prob = Problem(Minimize(T), constraints)
    
    time_for_milp = min(0.35 * budget_s, time_remaining - 10)
    prob.solve(solver=cp.CBC, max_seconds=time_for_milp)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h_vars.value.astype(float)
        T_sol = T.value
    else:
        # Fallback to random binary feasible solution
        h_sol = np.zeros(n)
        indices = np.random.choice(n, M, replace=False)
        h_sol[indices] = 1
        T_sol = 0  # Placeholder
        
    print(f"MILP solution: T* = {T_sol}, C5 bound = {T_sol * dx}")
    
    # Check if we are already done
    if T_sol <= T_target:
        h_sol = np.clip(h_sol, 0, 1)
        scale = 1.0 / (np.sum(h_sol) * dx)
        h_sol *= scale
        final_c5 = evaluate_h(h_sol)
        return (h_sol.tolist(), final_c5, n)

    def compute_C5_binary_swap(h, i, j):
        # Simulate swap positions i and j in binary vector
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        h_new[j] = 1 - h_new[j]
        return evaluate_h(h_new)
    
    h_bin = h_sol.copy()

    current_c5 = evaluate_h(h_bin)
    best_h_bin = h_bin.copy()
    best_c5_bin = current_c5

        def smooth_max(h_vals, beta=1.0):
            # Evaluate all lags with FFT
            h_pad = np.pad(h_vals, (0, n - 1), 'constant')
            h1_pad = np.pad(1 - h_vals, (0, n - 1), 'constant')
            corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(h1_pad))))
            lag_vals = corr[:2*n-1] * dx
            return (1/beta) * np.log(np.sum(np.exp(beta * lag_vals)))
        
        def grad_smooth_max(h_vals, beta=1.0):
            # Precompute shifted array
            h_pad = np.pad(h_vals, (0, n - 1), 'constant')
            h1_pad = np.pad(1 - h_vals, (0, n - 1), 'constant')
            h_corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.conj(np.fft.fft(h1_pad))))
            lag_vals = h_corr[:2*n-1] * dx

            w = np.exp(beta * lag_vals) / np.sum(np.exp(beta * lag_vals))
            g = np.zeros_like(h_vals)
            for l in range(1, n):
                step = 1 - h_vals if l == 1 else np.zeros_like(h_vals)
                if l > 0:
                    step[l:] += -h_vals[:-l]
                if l < n:
                    step[:-l] += -h_vals[l:]
                # Shift for negative index offsets
                g += w[l-1] * step
            return g * dx
    
        def project_h(h_vec, mass_target):
            # Project onto box [0,1] and then onto simple mass constraint
            h_vec = np.clip(h_vec, 0, 1)
            current_mass = np.sum(h_vec) * dx

                return np.clip(h_vec + mu, 0, 1)
            # Try bisection
            low, high = -0.1, 0.1
            while high - low > 1e-12:
                mid = (low + high) / 2
                test_h = adjust_mass(mid)
                test_mass = np.sum(test_h) * dx
                if test_mass > mass_target:
                    low = mid
                else:
                    high = mid
            return adjust_mass((low + high) / 2) 
        
        beta_schedule = [1, 2, 5, 10, 20]
        for beta in beta_schedule:
            if time.time() - start_time > 0.9 * budget_s:
                break
            # Run gradient descent steps
            steps = 100
            alpha = 0.005 / beta  # Step size adapts to beta
            for _ in range(steps):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                g = grad_smooth_max(h_final, beta)
                h_new = h_final - alpha * g
                h_final = project_h(h_new, M * dx)
        scaled_final = h_final  # Already scaled
        final_c5 = evaluate_h(scaled_final)
        return (scaled_final.tolist(), final_c5, n)
    
    # Final check for bounds
    h_final = best_h_bin.astype(float)
    scale = 1.0 / (np.sum(h_final) * dx)
    h_final *= scale
    final_c5 = evaluate_h(h_final)
    return (h_final.tolist(), final_c5, n)
