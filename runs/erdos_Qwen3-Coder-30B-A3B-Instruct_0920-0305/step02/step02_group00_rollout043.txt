```python
import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_values = [48, 56, 60]
dx = 2.0 / 48  # Default dx, will be recomputed per n
target_bound = 0.3807
budget_s = 600

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_hyperplane(v, target_sum):
    """Project vector v onto [0,1]^n AND sum = target_sum"""
    # Find lambda such that sum(clamp(v - lambda, 0, 1)) = target_sum
    def f(lam):
        # Clip v - lam between 0 and 1
        clipped = np.clip(v - lam, 0, 1)
        return np.sum(clipped) - target_sum

    # Binary search for lambda
    a, b = -10, 10
    for _ in range(30):
        mid = (a + b) / 2
        if f(mid) > 0:
            a = mid
        else:
            b = mid
    lam = (a + b) / 2
    return np.clip(v - lam, 0, 1)

def soft_min_weights(C_s_vals, beta):
    """Compute weights for soft-min."""
    # Subtract max to avoid overflow
    C_s_vals = np.array(C_s_vals)
    C_max = np.max(C_s_vals)
    exp_vals = np.exp(-beta * (C_s_vals - C_max))
    return exp_vals / np.sum(exp_vals)

def gradient_for_soft_min(h, beta, dx, n):
    """Compute the gradient of the soft-min correlation."""
    # This uses vectorized shifts
    # Compute all correlations C_s=h^T A_s h using convolution
    C_s_vals = []
    for s in range(n):
        # correlation for shift s: sum over i of h_i * h_{i+s}
        # Using numpy correlate: mode full, then extract relevant part
        # But to be safe we could do direct roll
        h_shifted = np.roll(h, s)
        # For shift s, we only take valid overlaps: exclude wraparound
        o_s = np.dot(h[:-s] if s > 0 else h, h_shifted[s:] if s > 0 else h)
        C_s = dx * o_s
        C_s_vals.append(C_s)

    # Soft-min weights
    weights = soft_min_weights(C_s_vals, beta)
    # Gradient: sum over s of weights_s * grad(C_s)
    # grad(C_s) = dx * 2 * A_s h = 2*dx * (h_shift_forward + h_shift_backward) / 2
    # A_s(h) = (h[.].roll(s) + h[.].roll(-s)) / 2 for valid indices
    grad_sum = np.zeros(n)
    for s in range(n):
        w = weights[s]
        if s == 0:
            # A_s h = h for s=0
            grad_sum += 2 * dx * w * h
        else:
            # Use shift to compute A_s h
            h_shifted_forw = np.roll(h, s)
            h_shifted_back = np.roll(h, -s)
            # Construct the 'weighted' shifted vector, only valid contributions
            shifted_sum = (h_shifted_forw + h_shifted_back) / 2
            # But we don't want to add elements from invalid boundaries
            # Recompute to ensure only valid pairs are counted:
            # We have: h_i * h_{i+s} for i = 0..n-1-s
            # So for shift s:
            shift_forward = np.zeros(n)
            shift_backward = np.zeros(n)
            shift_forward[s:n] = h[:n-s]
            shift_backward[:n-s] = h[s:n]
            A_s_h = (shift_forward + shift_backward) / 2
            # But shift_backward[:n-s] = h[s:n].
            # shift_forward[s:n] = h[:n-s].
            # So for i=0..n-1, (A_s h)_i = 0.5 * h_{i-s} + 0.5 * h_{i+s}
            # with h[i] := 0 if i out of bounds
            # So skip - the shift does it right
            # avoid infinities in case of indexing error:
            # effectively: compute h[.].roll(s) for +s, h[.].roll(-s) for -s,
            # but don't actually use roll because that pads with zeros, leading to artifacts.
            # We do it manually:
            grad_sum += 2 * dx * w * A_s_h
    return grad_sum

def project_to_feasible(h, target_sum):
    """Project h to satisfy 0 ≤ h[i] ≤ 1 and sum(h) * dx = 1."""
    # First clip to [0,1]
    h_clipped = np.clip(h, 0, 1)
    # Then project to hyperplane
    h_projected = project_onto_box_and_hyperplane(h_clipped, target_sum)
    return h_projected

def gradient_ascent_step(h, beta, dx, n, alpha=0.1, max_backtrack=10):
    """Perform gradient ascent step and backtracking line search."""
    grad = gradient_for_soft_min(h, beta, dx, n)
    # Try step
    h_new = h + alpha * grad
    h_proj = project_to_feasible(h_new, n/2)
    # Compute true overlap value
    corr = np.correlate(h_proj, 1 - h_proj, mode="full")
    c5_new = np.max(corr) * dx
    # Backtracking line search
    for _ in range(max_backtrack):
        if c5_new < evaluate_h(h):  # Accept move
            return h_proj, c5_new
        else:
            alpha /= 2
            h_new = h + alpha * grad
            h_proj = project_to_feasible(h_new, n/2)
            corr = np.correlate(h_proj, 1 - h_proj, mode="full")
            c5_new = np.max(corr) * dx
    return h_proj, c5_new  # Return even if not accepted

def run(seed=42, budget_s=600):
    """Run the gradient ascent algorithm on the multiple n's."""
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = np.inf
    best_h = None
    best_n = None

    for n in n_values:
        if time.time() - start_time > 0.9 * budget_s:
            break
        dx = 2.0 / n
        # Determine inner time slice
        inner_budget = max(50, int(0.1 * (budget_s * 0.95 - (time.time() - start_time))))
        
        # Generate several starting points
        start_points = []
        # 1. Interpolation of initial_h_values
        # Assume initial_h_values has 47 values. Interpolate to n.
        # We just re-sample it at same order.
        if len(initial_h_values) == 47:
            h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, 47), initial_h_values)
            start_points.append(h_interp)
        else:
            # Fallback if length differs
            h_interp = np.ones(n) * 0.5
            start_points.append(h_interp)
        
        # 2. Binary vector with sum n/2
        h_binary = np.zeros(n)
        ones_indices = np.random.choice(n, size=n//2, replace=False)
        h_binary[ones_indices] = 1.0
        start_points.append(h_binary)
        
        # 3. Random uniform vector
        h_random = np.random.uniform(0, 1, n)
        h_projected = project_onto_box_and_hyperplane(h_random, n / 2)
        start_points.append(h_projected)
        
        # If we have a previous best, interpolate it to current n
        if best_h is not None and best_n is not None:
            if best_n != n:
                h_interp_best = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, best_n), best_h)
                start_points.append(h_interp_best)
            else:
                start_points.append(best_h)
        
        # Now for each starting point
        for i_start, h_in in enumerate(start_points):
            # h = h_in.copy()
            h = h_in.copy()
            h = project_to_feasible(h, n/2)
            c5_current = evaluate_h(h)
            t_current = c5_current
            
            # Schedule of β: from smooth to sharp
            beta_schedule = [5, 15, 30, 60, 120]
            
            # Early stopping parameter
            last_improvement_iter = 0
            iter_since_improvement = 0
            
            inner_start = time.time()
            for beta in beta_schedule:
                if time.time() - inner_start > inner_budget * 0.8:
                    break
                K_max = 300
                for k in range(K_max):
                    if time.time() - inner_start > inner_budget:
                        break
                    if iter_since_improvement > 20:
                        break
                    h_proj, c5_new = gradient_ascent_step(h, beta, dx, n, alpha=0.1)
                    if c5_new < t_current:
                        t_current = c5_new
                        h = h_proj
                        last_improvement_iter = k
                        iter_since_improvement = 0
                    else:
                        iter_since_improvement += 1
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                        best_n = n
                    if c5_new < target_bound:
                        # Early return
                        break

        # Optional refinement using SLSQP
        if best_h is not None and (time.time() - start_time) < 0.95 * budget_s:
            # To prevent any采矿等 issues with non-smooth max in optimize.minimize
            def helper_obj(_h):
                return np.max(np.correlate(_h, 1 - _h, mode="full")) * dx
            try:
                bounds = [(0, 1)] * n
                constraints = {'type': 'eq', 'fun': lambda _h: np.sum(_h) * dx - 1.0}
                result = minimize(helper_obj, best_h, method='SLSQP',
                                bounds=bounds, constraints=constraints,
                                options={'maxiter': 200, 'ftol': 1e-9})
                if result.success and np.max(np.correlate(result.x, 1 - result.x, mode="full")) * dx < best_c5:
                    best_c5 = np.max(np.correlate(result.x, 1 - result.x, mode="full")) * dx
                    best_h = result.x.copy()
                    best_n = n
            except:
                pass

    # Final evaluation
    if best_h is not None:
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        final_c5 = np.max(corr) * dx
        return list(best_h), final_c5, best_n
    else:
        # Fallback
        h_default = np.array(initial_h_values)
        if len(h_default) != 48:
            h_default = np.interp(np.linspace(0, 2, 48), np.linspace(0, 2, len(h_default)), h_default)
        return list(h_default), evaluate_h(h_default), 48
```
