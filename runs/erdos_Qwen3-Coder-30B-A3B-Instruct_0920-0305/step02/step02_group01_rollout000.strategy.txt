    lo = -np.max(v) - 1.0   # search
    hi = np.max(v) + 1.0
    # adjust to guarantee sum clip(v - lo) >= S (since more negative lo => x_i=1)
    # Actually for lambda = lo (very negative), v_i - lo is large positive, clip to 1 => sum = n >= S.
    # For lambda = hi (very positive), v_i - hi negative => clip to 0 => sum = 0 <= S.
    # So the root exists.
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        sum_x = x.sum()
        if sum_x > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Ensure sum close to S; maybe adjust by scaling residual
    residual = S - x.sum()
    # Distribute residual among those not at bounds
    if abs(residual) > 1e-10:
        # find indices where 0 < x < 1
        free = (x > 1e-8) & (x < 1-1e-8)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0, 1)
    return x
