import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_constraint(h, n_points):
    """Project h onto the constraint that sum(h) * dx = 1"""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    if abs(current_sum * dx - 1.0) > 1e-10:
        scale = 1.0 / (current_sum * dx)
        h = h * scale
    return h

def capped_simplex_projection(grad, h_old, n_points):
    """Project gradient onto feasible set using capped-simplex method.
    This gives the optimal h for minimizing -grad^T h subject to sum h = n/2, 0 <= h <= 1.
    """
    dx = 2.0 / n_points
    target_sum = 1.0 / dx  # = n_points / 2
    
    # Sort indices by descending gradient values
    sorted_indices = np.argsort(-grad)
    
    # Allocate mass greedily
    h_new = np.zeros_like(h_old)
    remaining_mass = target_sum
    used = np.zeros_like(h_old, dtype=bool)
    
    for idx in sorted_indices:
        if remaining_mass <= 0:
            break
        # How much we can add to this component
        max_add = min(1.0 - h_old[idx], remaining_mass)
        if max_add > 0:
            h_new[idx] = min(1.0, h_old[idx] + max_add)
            remaining_mass -= max_add
            used[idx] = True
    
    # If we still haven't reached the target, pad with the remaining mass at the highest gradient components
    if remaining_mass > 1e-10:
        # This case should be rare in practice when our approximation is good
        for i in range(len(h_new)):
            if not used[i]:
                max_add = min(1.0 - h_new[i], remaining_mass)
                h_new[i] += max_add
                remaining_mass -= max_add
                if remaining_mass <= 1e-10:
                    break
    
    return h_new

def compute_worst_shift_autocorrelation(h, n_points):
    """Compute q_s(h) = sum_i h_i h_{i+s} for all shifts s, get the one that minimizes q_s."""
    dx = 2.0 / n_points
    # Get full correlation including all shifts
    auto = correlate(h, h, mode='full')
    
    # The valid range for shifts s=0..n_points-1 corresponds to indices n_points-1 .. 2*n_points-2 in auto
    # Only consider positive shifts in [0, n_points-1]
    valid_shifts = np.arange(n_points)
    relevant_correlations = auto[n_points-1:n_points-1 + n_points]
    # Compute the values for all shifts
    q_vals = relevant_correlations * dx  # Scale by dx
    # We want to minimize q_vals, so we look for the smallest one
    min_idx = np.argmin(q_vals)
    min_shift = valid_shifts[min_idx]
    return min_shift, q_vals[min_idx]

def compute_worst_shift_gradient(h, min_shift, n_points):
    """Compute ∇q_{s*}(h) where q_{s*} is the worst shift."""
    grad = np.zeros_like(h)
    # Gradient of q_s(h) = sum_i h_i h_{i+s} is ∂/∂h_i q_s = h_{i+s} + h_{i-s}
    # Note: circular boundary conditions for the correlations
    for i in range(n_points):
        # Forward shift i -> i + s
        j1 = (i + min_shift) % n_points
        # Backward shift i -> i - s
        j2 = (i - min_shift) % n_points
        grad[i] = h[j1] + h[j2]
    return grad

def dc_step(h, n_points, max_iter=1000):
    """Perform one run of the DC algorithm for minimizing C5."""
    old_h = h.copy()
    old_c5 = evaluate_c5(h)
    dx = 2.0 / n_points
    target_sum = 1.0 / dx  # = n_points / 2
    stagnation_counter = 0
    best_h = h.copy()
    best_c5 = old_c5
    
    converged = False
    for iteration in range(max_iter):
        min_shift, _ = compute_worst_shift_autocorrelation(old_h, n_points)
        grad = compute_worst_shift_gradient(old_h, min_shift, n_points)
        new_h = capped_simplex_projection(grad, old_h, n_points)
        # Project onto constraint: this shouldn't be necessary but let's be robust
        new_h = project_onto_constraint(new_h, n_points)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < old_c5:
            old_c5 = new_c5
            old_h = new_h.copy()
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter >= 50:
                break
        
    return best_h, best_c5
