def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto simplex with bounds."""
    v = np.array(v)
    n = len(v)
    # Shift
    u = v - lower
    target_sum_u = target_sum - n * lower
    
    # If all v_i are at the boundary
    if target_sum_u < 0:
        return np.ones_like(v) * lower
    if target_sum_u > n * (upper - lower):
        return np.ones_like(v) * upper
    
    # Use water-filling algorithm
    sorted_u = np.sort(u)[::-1]
    cumsum_sorted = np.cumsum(sorted_u)
    # Find threshold that gives us the correct sum
    thresholds = (cumsum_sorted - target_sum_u) / np.arange(1, len(sorted_u) + 1)
    # Find the maximum index where threshold <= sorted_u[i]
    idx = np.searchsorted(thresholds, sorted_u, side="right") - 1
    idx = max(0, min(idx, len(sorted_u) - 1))
    if sorted_u[idx] <= thresholds[idx] and idx < len(sorted_u) - 1:
        # user threshold at next one
        idx += 1
        
    lambda_ = thresholds[idx] if idx < len(thresholds) else 0
    
    # Compute projection
    result = np.maximum(np.minimum(u - lambda_, upper - lower), 0) + lower
    # Ensure sum is correct through small adjustment if needed
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-8:
        diff = target_sum - current_sum
        adjustment = diff / n
        result += adjustment
    return result

def gradient_step(h, k_min):

        g = 2 * h
    else:

        g = np.zeros_like(h)

        if k_min > 0:
            if k_min < n:
                g[:n-k_min] += h[k_min:]
                g[k_min:] += h[:n-k_min]
        else:
            # Negative k, reasonable for full correlation mode
            # Actually mode="full" shifts to left, so final indices
            g[:-k_min] += h[-k_min:]
            g[-k_min:] += h[:-k_min]
    return g

def gradient_ascent_step(h, k_min, alpha):
    """Perform one gradient ascent step to increase S_{k_min}."""
    g = gradient_step(h, k_min)
    h_new = h + alpha * g
    # Project back onto feasible region
    h_new = project_onto_simplex_with_bounds(h_new, len(h) / 2, 0, 1)
    return h_new
