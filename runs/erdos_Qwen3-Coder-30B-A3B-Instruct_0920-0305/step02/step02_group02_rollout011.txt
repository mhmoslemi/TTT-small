def evaluate_c5(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def compute_autocorrelation(h):
    """Compute S_k = sum_i h_i * h_{i+k} for all shifts k."""
    n = len(h)
    # S_k for positive k: from correlation with itself
    corr = np.correlate(h, h, mode="full")
    # Extract S_k for k=0 to n-1 (non-negative shifts)
    S = corr[n-1:n-1+n]  # indices n-1 to 2n-2, which is k from 0 to n-1
    return S

def compute_gradient_S_k(h, k):
    """Compute gradient of S_k with respect to h."""
    n = len(h)
    if k == 0:
        return 2 * h
    else:
        g = np.zeros(n)
        if k > 0:
            # h_i contributes to derivative of h_{i+k}
            g[:n-k] += h[k:]
            # h_{i+k} contributes to derivative of h_i
            g[k:] += h[:n-k]
        return g
