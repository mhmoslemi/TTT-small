import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the binary MILP formulation for C5."""
    M = n // 2    # mass constraint
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []        # product variables
    constr = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        constr += [p_ell <= h[:length],
                   p_ell <= h[ell:],
                   p_ell >= h[:length] + h[ell:] - 1]

        constr += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), constr)
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
        if prob.status in ['optimal', 'optimal_inaccurate', 'feasible']:
            T_opt = int(T.value)
            h_opt = np.ravel(h.value)
            return T_opt, h_opt
        else:
            # Fallback: return best known feasible solution
            return None, None
    except:
        # If MILP fails, return None
        return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=100):
    """Binary swap-based simulated annealing."""
    np.random.seed(int(time.time()) % (2**32))
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    current_score = evaluate_h(h)

    best_h = h.copy()
    best_score = current_score

    # Determine initial temperature from few random steps
    temps = []
    for _ in range(500):
        i = np.random.choice(np.where(h == 1)[0])
        j = np.random.choice(np.where(h == 0)[0])
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_score = evaluate_h(h_new)
        delta = new_score - current_score
        if delta > 0 and not np.isclose(delta, 0):
            temps.append(delta)
    if temps:
        T0 = np.mean(temps) * 2
    else:
        T0 = 1.0

    T = T0
    iter_count = 0
    last_improvement = 0

    start = time.time()
    while iter_count < max_iter and time.time() - start < time_limit:
        # Pick two indices to swap
        i = np.random.choice(np.where(h == 1)[0])
        j = np.random.choice(np.where(h == 0)[0])

        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        new_score = evaluate_h(h_new)

        delta = new_score - current_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_score = new_score

        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
            last_improvement = iter_count

        T *= 0.999
        if T < 1e-6 or iter_count - last_improvement > 20000:
            T = T0
            last_improvement = iter_count
        iter_count += 1

    return best_h, best_score

def smooth_descent(h0, time_limit=30):
    """Run smooth gradient descent using log-sum-exp approximation."""
    n = len(h0)
    dx = 2.0 / n
    M = n // 2

    h = h0.copy()

    def project_to_simplex(y, target_sum):
        """Project y onto the simplex with sum=y_sum, y in [0,1]."""
        # Water-filling
        y = np.clip(y, 0, 1)
        y_sorted = np.sort(y)[::-1]
        # Find lambda such that sum of clips is target_sum
        s = 0
        for i in range(n):
            s += y_sorted[i]
            if s - y_sorted[i] <= target_sum:
                lambda_star = (s - target_sum) / (i + 1)
                break
        else:
            # All are set to zero
            lambda_star = y_sorted[-1] - target_sum / n

        y_proj = np.clip(y - lambda_star, 0, 1)
        return y_proj

    # Compute all overlaps via FFT only once per β
    def get_all_overlaps(h_in):
        H = np.fft.rfft(h_in, 2*n)
        G = np.fft.rfft(1.0 - h_in, 2*n)
        conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
        return conv.real

    # Schedules
    betas = [1, 2, 5, 10, 20]
    eta = 0.1

    start = time.time()
    for beta in betas:
        if time.time() - start > time_limit:
            break

        # Run projected gradient descent
        for _ in range(100):  # max grad steps per beta
            if time.time() - start > time_limit:
                break

            overlaps = get_all_overlaps(h)
            weights = np.exp(beta * overlaps)
            weights /= np.sum(weights)

            # Construct gradient
            grad = np.zeros(n)
            for ell in range(1, n):
                if ell < n:
                    # Gradient: d/dx_t L_ell = 1 - h_{t+ell} - h_{t-ell}
                    # So we only move from "t" to "t+ell".
                    # Forward difference
                    g_forward = np.zeros(n)
                    g_forward[:n-ell] = 1.0 - h[ell:]  # correct by ell steps
                    # Backward difference
                    g_backward = np.zeros(n)
                    g_backward[ell:] = 1.0 - h[:n-ell]   # correct by ell steps

                    weight = weights[ell-1]
                    grad += weight * (g_forward + g_backward)

            # Step
            h_new = h - eta * grad

            # Project
            h_new = project_to_simplex(h_new, M)

            h = h_new

    return h, evaluate_h(h)
