def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def dc_step(h, n):

    dx = 2.0 / n
    # Compute all correlations (overlap = sum h_i * (1-h_{i+s}))
    # Using correlate with 'full' mode gives:
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts from -n+1 to n-1)
    start = n - 1
    # The overlap contributions for each shift s
    overlaps = corr[start : start + 2 * n - 1]
    
    # Find the shift with smallest overlap
    min_idx = np.argmin(overlaps)
    best_shift = min_idx - (n - 1)  # This is the actual shift value
    
    # Compute the gradient of q_s(h) = sum_i h_i * h_{i+s} at current h
    # Gradient of q_s(i) = h_{i+s} + h_{i-s} 
    # Let's form the full gradient vector for the specific shift s
    grad = np.zeros(n)
    if best_shift >= 0:
        grad[best_shift:] += h[:-best_shift]
        grad[:best_shift] += h[-best_shift:]
    else:
        grad[:abs(best_shift)] += h[best_shift:]
        grad[abs(best_shift):] += h[:best_shift]
    
    # Linear program minimize -grad^T h subject to:
    # 0 <= h_i <= 1
    # sum h_i = n/2 
    #   
    # This is a simple LP: minimize -grad^T h subject to h_i in [0,1], sum h_i = n/2
    # Solution is extremely simple: project gradient onto feasible region.
    # Sort gradient descending and assign as much as possible to top elements 
    # (up to 1) and total sum = n/2 (can be fractional if approx)
    
    # Module to write a simple solver for this

    # Sort indices by descending gradient
    idxs = np.argsort(-grad)
    new_h = np.zeros(n) 
    s = 0.0  # sum of assigned values
    for i in idxs:
        if s >= n/2:
            break
        # Assign min(1.0, n/2 - s) 
        assign = min(1.0, n/2 - s)
        new_h[i] = assign
        s += assign
    
    # If we did not reach exactly n/2 total, adjust the last allocation
    if abs(s - n/2) > 1e-8:
        # Re-adjust the last entry assigned (i.e., the last occupied index)
        remaining = n/2 - s
        # Find the last assignment
        for i in reversed(idxs):
            if new_h[i] > 0 and (new_h[i] - remaining) >= 0:
                new_h[i] -= remaining
                break
            elif new_h[i] > 0:
                #Leave as is 
                break
    
    return new_h
