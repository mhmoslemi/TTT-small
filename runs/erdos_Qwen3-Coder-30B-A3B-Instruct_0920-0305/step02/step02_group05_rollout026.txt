import numpy as np
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist
import time
import math

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h[i] <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / n_points
    target_sum = 1.0
    current_sum = np.sum(h)
    # Clip to [0, 1]
    w = np.clip(h, 0, 1)
    delta = target_sum - np.sum(w) * dx
    
    # Water-filling algorithm
    while abs(delta) > 1e-12:
        if delta > 0:
            # Distribute delta equally among indices with w[i] < 1
            count = np.sum(w < 1)
            if count == 0:
                break
            distribute = delta / count
            for i in range(n_points):
                if w[i] < 1:
                    w[i] += distribute
                    delta -= distribute
        else:
            # Subtract |delta| equally among indices with w[i] > 0
            count = np.sum(w > 0)
            if count == 0:
                break
            subtract = -delta / count
            for i in range(n_points):
                if w[i] > 0:
                    w[i] = max(0, w[i] - subtract)
                    delta += subtract
        # Re-clip
        w = np.clip(w, 0, 1)
        delta = target_sum - np.sum(w) * dx
    return w

def compute_subgradient(h, m, n_points, dx):
    """Compute subgradient for the worst lag m"""
    g = np.zeros(n_points)
    if m >= 0:
        for i in range(n_points - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        m = -m
        for i in range(m, n_points):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    return g * dx

def initialize_h(n_points, initial_h_values):
    """Initialize h with interpolated initial_h_values"""
    # Interpolate to desired resolution
    initial = np.array(initial_h_values)
    if len(initial) == n_points:
        h = initial.copy()
    else:
        # Linearly resample initial to new grid
        x_old = np.linspace(0, 2, len(initial))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, initial)
    # Project to feasible set
    h = project_onto_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_time = 0
    best_c5 = float('inf')

    h = initialize_h(n_points, initial_h_values)

    resolutions = [61, 101, 121, 201]
    # We start at 101 which is enough for a good start
    
    current_n = n_points
    
    # Initialize a counter for stagnation
    no_improve_count = 0
    max_no_improve = 400
    
    # Inner loop parameters
    alpha_0 = 5.0
    iter_count = 0
    
    while total_time < 0.99 * budget_s:
        # Evaluate objective
        c = evaluate_h(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Check if needed to break
        if no_improve_count > max_no_improve:
            # Perturb
            h += np.random.normal(0, 0.02, len(h))
            h = project_onto_feasible(h, len(h))
            no_improve_count = 0       
        
        # Get all lags from the convolution
        dx = 2.0 / len(h)
        A = np.correlate(h, 1 - h, mode="full")
        A *= dx
        c_max = np.max(A)
        m_indices = np.where(A == c_max)[0]
        m = m_indices[0] - (len(h) - 1)  # Convert index to lag
        # Compute subgradient
        g = compute_subgradient(h, m, len(h), dx)
        
        # Update step size
        alpha = alpha_0 / math.sqrt(iter_count + 1)
        h -= alpha * g
        
        # Enforce symmetry (this is known to work well)
        h = 0.5 * (h + h[::-1])
        
        # Project to feasible set again
        h = project_onto_feasible(h, len(h))
        
        # Occasionally do a global restart if we've stuck for a while or probabilistically
        if (no_improve_count > max_no_improve or np.random.rand() < 0.01):
            h_start = np.random.rand(len(h))
            h_start = project_onto_feasible(h_start, len(h))
            h = h_start
            no_improve_count = 0
            
        iter_count += 1
        total_time = time.time() - start_time

    # Final check and computation
    if best_h is None:

        best_c5 = evaluate_h(best_h)
        
    best_c5 = evaluate_h(best_h)
    best_h = project_onto_feasible(best_h, len(best_h))
    best_c5 = evaluate_h(best_h)
    return (best_h.tolist(), best_c5, len(best_h))
