import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import random

# Constants
n_points_list = [48, 50, 52, 54, 56, 58, 60]
target_bound = 0.3807
dx = 2.0 / 48  # default to 48 points for consistency with imported initial_h_values

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def compute_c5_from_autocorr(h_values):
    """Compute C5 from autocorrelation (used in gradient method)."""
    h = np.array(h_values)
    # Compute S_k = sum_i h_i * h_{i+k} for all shifts
    n = len(h)
    correlations = np.correlate(h, h, mode="full")
    # Extract valid S_k (k from 0 to n-1)
    S_values = correlations[n-1:]
    # Minimize w - min(S_k), so we want to maximize min(S_k)
    # But we don't count S_0 = sum(h_i^2), which is not our objective being minimized
    # We want to minimize the overlap (w - min(S_k)), for k>=1
    min_S = np.min(S_values[1:])  # exclude k=0
    w = np.sum(h)  # sum of h
    c5 = (w - min_S) * dx
    return c5

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto the simplex with sum = target_sum and bounds [lower, upper]."""
    v = np.array(v)
    n = len(v)
    if n == 0:
        return np.array([])
    # shifted vector
    u = v - lower
    upper_shift = upper - lower
    
    # we actually want to find projection of u onto [0, upper_shift] simplex with sum = target_sum - lower * n
    # so just project onto [0,1] simplex with sum = target_sum - lower * n = s_target
    s_target = target_sum - lower * n
    
    if s_target <= 0:
        return np.full(n, lower)
    if s_target >= n * upper_shift:
        return np.full(n, upper)
    
    u_sorted = np.sort(u)[::-1]
    cumsum = np.cumsum(u_sorted)
    tau = np.zeros(n)
    
    for i in range(n):
        if u_sorted[i] - (cumsum[i] - u_sorted[i]) / (i+1) >= 0:
            tau[i] = (cumsum[i] - s_target) / (i+1)
    
    if len(tau) == 0:
        return np.full(n, lower)
    
    t = np.max(tau) if len(tau) > 0 else 0
    x = np.clip(u - t, 0, upper_shift)
    x = x + lower  # fix back to original bounds
    return x

def gradient_step(h, S_k_min_index, learning_rate=0.02):
    """Perform one gradient step to increase the minimal correlation."""
    n = len(h)
    grad = np.zeros(n)
    if S_k_min_index == 0:
        # gradient is 2 * h(i)
        grad = 2 * h
    else:
        # gradient based on neighbors
        if S_k_min_index > 0:
            # forward neighbors
            if S_k_min_index < n:
                grad[:-S_k_min_index] += h[S_k_min_index:]
            
            # backward neighbors
            if S_k_min_index < n:
                grad[S_k_min_index:] += h[:-S_k_min_index]
    # update with gradient ascent
    h_new = h + learning_rate * grad
    # project onto feasible set (box and sum constraint)
    # exact sum preserved now by adjusting the general direction for target sum
    # but we need a good way to maintain exact sum
    # First clip to bounds
    h_new = np.clip(h_new, 0, 1)
    # Then project to exact sum
    target_sum = np.sum(h)  # original sum
    h_new = project_onto_simplex_with_bounds(h_new, target_sum, 0, 1)
    return h_new

def local_search(h_values, max_iter=2000, tolerance=1e-8):
    """Improved local search using projected gradient ascent."""
    h = np.array(h_values)
    n = len(h)
    target_sum = np.sum(h)
    dx = 2.0 / n
    
    # Iterate with projected gradient
    for iter_ in range(max_iter):
        # Compute all S_k
        correlations = np.correlate(h, h, mode="full")
        S_values = correlations[n-1:]  # S_k for k from 0 to n-1
        
        # We ignore S_0 (mean squared)
        min_S_index = np.argmin(S_values[1:]) + 1  # k values from 1 to n-1
        min_S = S_values[min_S_index]
        
        # Calculate new C5
        w = np.sum(h)
        c5 = (w - min_S) * dx
        
        if c5 < tolerance:
            break
            
        # Perform gradient step
        h = gradient_step(h, min_S_index, learning_rate=0.01 * (1.0 / (1 + iter_ / 1000)))
    
    return list(h)

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / 48  # use standard dx to be compatible with initial construction
    
    # First, let's start with the initial_h_values if available
    try:
        initial_h = np.array(initial_h_values)
        c5_initial = evaluate_h(initial_h)
        if c5_initial < best_c5:
            best_h = initial_h.copy()
            best_c5 = c5_initial
    except:
        pass
    
    max_iter_per_n = 500  # per function evaluation complexity
    time_per_n = budget_s / len(n_points_list)  # time per n value
    
    # Try different n values
    for n in n_points_list:
        if time.time() - start_time > budget_s * 0.95:
            break
        iter_start = time.time()
        dx = 2.0 / n
        n_h = n
        
        # Random restarts starting from random points
        for restart in range(5):
            # Start with uniformly distributed h values, then project onto sum constraint
            h = np.random.rand(n) * 0.5 + 0.25  # Bounds [0.25, 0.75]
            sum_h = np.sum(h)
            h = h * (n/2) / sum_h  # Scale to ensure sum = n/2
            
            # Local optimization: projected gradient descent
            h_opt = local_search(list(h), max_iter=200)
            c5_opt = compute_c5_from_autocorr(h_opt)
            
            if c5_opt < best_c5:
                best_h = np.array(h_opt)
                best_c5 = c5_opt
                
            if time.time() - iter_start > time_per_n * 0.95:
                break
                
    # Final polishing with scipy SLSQP constrained optimization
    if best_h is not None:
        bounds = [(0, 1)] * len(best_h)
        
        def objective(h):
            return compute_c5_from_autocorr(h)
            
        def constraint_eq(h):
            return np.sum(h) * dx - 1.0  # integral = 1
            
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                             constraints={'type': 'eq', 'fun': constraint_eq},
                             options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.success:
                best_h = result.x
                best_c5 = compute_c5_from_autocorr(best_h)
        except:
            pass
            
    # Final evaluation to make sure it's accurate
    if best_h is not None:
        final_c5 = evaluate_h(best_h)
        # Double check it's still good
        final_c5 = compute_c5_from_autocorr(best_h)
        return list(best_h), final_c5, len(best_h)
    else:
        # Fall back to initial solution if nothing worked
        return list(initial_h_values), evaluate_h(initial_h_values), 48
