import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5


def project_to_box_and_simplex(v, target_sum):
    """Project vector v onto [0,1]^n and then onto the hyperplane sum x_i = target_sum."""
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    # Adjust to get correct sum
    current_sum = np.sum(v)
    if abs(current_sum - target_sum) < 1e-10:
        return v
    # Find lambda via bisection
    def f(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - target_sum

    a, b = -np.max(v), 1 - np.min(v)
    for _ in range(30):
        c = (a + b) / 2
        if f(c) > 0:
            a = c
        else:
            b = c
    lam = (a + b) / 2
    x = np.clip(v - lam, 0, 1)
    return x


def smooth_min_gradient(h, C_s, beta):
    """Compute gradient of smooth min with β, using precomputed C_s."""
    exp_neg_beta_C = np.exp(-beta * C_s)
    sum_exp = np.sum(exp_neg_beta_C)
    if sum_exp == 0:
        # fallback to uniform
        weights = np.ones_like(C_s) / len(C_s)
    else:
        weights = exp_neg_beta_C / sum_exp

    grad = np.zeros_like(h)
    for s in range(len(C_s)):
        # A_s * h: shifted average
        if s == 0:
            shift_h = h
        else:
            shift_h = np.zeros_like(h)
            if s < len(h):
                shift_h[s:] = h[:-s]
            if s > 0:
                shift_h[:-s] = h[s:]
        # Average
        shift_h = (shift_h + np.roll(shift_h, s)) / 2.0
        grad += weights[s] * shift_h
    return 2 * dx * grad


def compute_C_s(h_values):
    """Compute all correlation terms C_s = dx * sum h_i * h_{i+s}."""
    h = np.array(h_values)
    n = len(h)
    C_s = []
    for s in range(n):
        # Correlation at shift s
        corr = np.correlate(h, h, mode='full')
        corr_val = corr[n-1+s]  # Typical Python behavior for correlation
        C_s.append(dx * corr_val)
    return np.array(C_s)


def backtrack_line_search(h, grad, beta, max_iter=10):
    """Backtrack line search for step size."""
    alpha = 0.5  # Initial step size
    C_s = compute_C_s(h)
    current_val = max(1 - dx * np.sum(h * (1 - h)), 0)  # Approximate min C_s
    for _ in range(max_iter):
        h_new = project_to_box_and_simplex(h - alpha * grad, n_points / 2)
        C_s_new = compute_C_s(h_new)
        new_val = min(C_s_new)
        if new_val < current_val:
            return h_new, alpha
        else:
            alpha /= 2.0  # Halve step size
    return h, alpha


def gradient_ascent_loop(h_start, beta_schedule, max_inner_iter=300):
    """Perform gradient ascent with soft minimum."""
    h = h_start.copy()
    for beta in beta_schedule:
        for k in range(max_inner_iter):
            C_s = compute_C_s(h)
            # Compute gradient
            g = smooth_min_gradient(h, C_s, beta)
            # Line search (limited iterations to keep speed)
            h_new, _ = backtrack_line_search(h, g, beta, max_iter=5)
            # Check improvement
            old_max = max(C_s)
            C_s_new = compute_C_s(h_new)
            new_max = max(C_s_new)
            if new_max < old_max:
                h = h_new
            else:
                # If no improvement, break early
                if k > 20 and k % 20 == 0:
                    break
    return h

        h_bin[ones] = 1.0
        starts.append(project_to_box_and_simplex(h_bin, target_sum))

        h_rand = np.random.rand(n)
        starts.append(project_to_box_and_simplex(h_rand, target_sum))

            corr = np.correlate(h_final, 1 - h_final, mode="full")
            c5_final = np.max(corr) * dx

                best_c5 = c5_final
                best_h = h_final.copy()
