import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_candidates = [48, 56, 64, 80]
dxs = [2.0 / n for n in n_candidates]
targets = [n / 2 for n in n_candidates]

# Initial heuristic function
def initial_h_values(n):
    # Generate from previous solution with some randomness
    # This is just a placeholder - real versions replace with taken initial_h_values from environment
    np.random.seed(42)
    h = np.random.rand(n)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum * (2.0 / n) - 1.0) > 1e-6:
        scale = 1.0 / (h_sum * (2.0 / n))
        h *= scale
    return h

def project_onto_capped_simplex(x, target_sum):
    """Project x onto {y | 0 <= y_i <= 1, sum(y) = target_sum}."""
    x = np.asarray(x)
    n = len(x)
    # Sort values in descending order
    sorted_x = np.sort(x)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    # Find the threshold
    idx = np.where(sorted_x * np.arange(1, n + 1) > (target_sum - cumsum + sorted_x))[0]
    if len(idx) == 0:
        # All elements need to be 0 or 1
        tau = 0.0
    else:
        # Value of tau that makes sum = target_sum
        k = idx[-1] + 1
        tau = (target_sum - cumsum[k - 1]) / k
    # Compute x projection
    x_proj = np.maximum(np.minimum(x - tau, 1.0), 0.0)
    # Adjust to match target_sum exactly
    diff = target_sum - np.sum(x_proj)
    if abs(diff) > 1e-10 and np.any(x_proj > 0) and np.any(x_proj < 1):
        # Find smallest nonequal element to adjust
        idx_min = np.where(np.isclose(x_proj, 0.0))[0]
        if len(idx_min) > 0:
            idx_min = idx_min[0]
        else:
            idx_min = np.argmax(x_proj)
        x_proj[idx_min] += diff
    return x_proj

def compute_correlations(h, dx):
    """Compute all correlations r_k = sum h_i * (1 - h_{i+k})"""
    n = len(h)
    # Use FFT-based correlation
    H = np.fft.fft(h)
    H_conj = np.conj(H)
    cross_corr = np.fft.ifft(H * H_conj)
    cross_corr = np.real(cross_corr)
    # For the correct form we compute sum h_i * (1 - h_{i+k})
    # Which is sum h_i - sum h_i h_{i+k} 
    # We must handle the boundary effects carefully here.  
    # Below is a more precise approach.
    # r_k = sum h_i * (1 - h_{i+k}) = sum h_i - sum h_i h_{i+k}
    h_squared = np.convolve(h, h, mode='full')
    # Manual shift for edge
    r_vals = []
    total_sum = np.sum(h)
    for k in range(n):
        # Compute convolution sum over shifted h
        # At i, we take h_i * (1 - h_{i+k}) 
        # So sum h_i * (1 - h_{i+k}) = sum h_i  - sum h_i *h_{i+k}  
        # That means compare with shifted positions.
        if k == 0:
            s_k = np.sum(h * h)
        else:
            s_k = np.sum(h[:-k] * h[k:]) 
        r_k = total_sum - s_k
        r_vals.append(r_k)
    return np.array(r_vals)

def evaluate_c5_smooth(h, alpha=100.0):
    '''Evaluate smooth surrogate for C5 using log-sum-exp'''
    dx = 2.0 / len(h)
    r = compute_correlations(h, dx)
    r_max = np.max(r)
    r_shifted = r - r_max  # Prevent overflow
    exp_r = np.exp(alpha * r_shifted)
    sum_exp = np.sum(exp_r)
    if sum_exp <= 0:
        return np.inf
    log_sum_exp = np.log(sum_exp) + r_max * alpha 
    f = (dx / alpha) * log_sum_exp
    return f

def compute_gradient(h, alpha=100.0):
    '''Compute gradient using softmax weights for smooth log-sum-exp'''
    dx = 2.0 / len(h)
    n = len(h)
    r = compute_correlations(h, dx)
    r_max = np.max(r)
    r_shifted = r - r_max
    exp_r = np.exp(alpha * r_shifted)
    w = exp_r / np.sum(exp_r)
    
    # Now compute gradient per r_k:
    # g_k = -dx * sum_{i} h_i h_{i+k}
    # But we want ∇r_k = ∂r_k / ∂h_i = 1 - h_{i+k} - h_{i-k} (proper definition)
    
    # Better simple implementation with slices:
    grad = np.zeros_like(h)
    total_sum = np.sum(h)
    for k in range(n):
        # r_k = total_sum - sum h_i h_{i+k}
        # => ∂r_k / ∂h_i = - h_{i+k} for i+k < n
        #                      - h_{i-k} for i-k >= 0  
        # => make fv_len = 0th index loss in real coords
        if k == 0:
            g = np.full_like(h, 1.0)  # sum over all i of (1 - h_i)
            g -= 2 * h  # for case where h_{i+k} is just h_i  
        else:
            g = np.full_like(h, 1.0)
            if k < n:
                g[:-k] -= h[k:]
            if k > 0 and k < n:
                g[k:] -= h[:-k]
        grad += w[k] * g
    
    return dx * grad

def grad_projected_descent(h0, dx, w, alpha0=1.0, alpha_incr=2.0, alpha_max=100.0, max_iter=5000):
    h = h0.copy()
    t_prev = 1.0
    eta = 0.1

    for i in range(max_iter):
        # Evaluate r values
        r = compute_correlations(h, dx)
        # Compute max
        c5_value = np.max(r) * dx
        if i % 1000 == 0:
            print(f"Iter {i}: C5 = {c5_value:.6f}")
            if c5_value < 0.3800:
                print("Early stop at decent value!")
                break
        
        # Compute smoothed surrogate
        r_shifted = r - np.max(r)
        exp_r = np.exp(alpha0 * r_shifted)
        w = exp_r / np.sum(exp_r)
        if np.isnan(w).any():
            print(f'NaN in weights at iter {i}, setting to uniform')
            w = np.ones_like(w) / len(w)
        if np.isinf(np.sum(w)):
            print(f'Inf in weights')
            w = np.ones_like(w) / len(w)
        
        # Compute gradient:
        # Each component of g_k = - h_{i+k}
        g = np.zeros_like(h)
        for k in range(len(h)):
            if k == 0:
                g += w[k] * (1.0 - 2.0 * h)
            else:
                m = np.zeros_like(h)
                if k < len(h):
                    m[:-k] += h[k:]
                if k > 0 and k < len(h):
                    m[k:] += h[:-k]
                g += w[k] * (1.0 - m)
        g = -dx * g

        # Accelerated update
        t_new = (1 + np.sqrt(1 + 4 * t_prev ** 2)) / 2
        y = h - eta * g
        h_new = y + ((t_prev - 1) / t_new) * (y - h)
        h = h_new
        
        # Project back to valid set
        h = project_onto_capped_simplex(h, w)
        
        if np.any(h < -1e-10) or np.any(h > 1 + 1e-10):
            print("Projected h outside [0,1]. Adjusting.")
            h = np.clip(h, 0, 1)
            h = project_onto_capped_simplex(h, w)
            
        # Line search for eta
        while True:
            h_trial = project_onto_capped_simplex(h - eta * g, w)
            c5_trial = np.max(compute_correlations(h_trial, dx)) * dx
            if c5_trial < c5_value:
                break
            eta *= 0.5
            if eta < 1e-8:
                break
            
        if abs(c5_value - np.max(compute_correlations(h, dx)) * dx) < 1e-6:
            print(f"Stable after {i} iterations")
            break
            
        t_prev = t_new

        # Gradually increase alpha
        alpha0 = min(alpha0 * alpha_incr, alpha_max)

    c5_final = np.max(compute_correlations(h, dx)) * dx
    return h, c5_final

def local_refinement(h, dx, w):
    """Refine h using scipy trust-constr with exact objective"""
    def objective(h_var):
        return np.max(compute_correlations(h_var, dx)) * dx

    bounds = [(0, 1) for _ in range(len(h))]
    result = minimize(objective, h, method='trust-constr', bounds=bounds,
                      options={'maxiter': 300, 'gtol': 1e-8})
    if result.success:
        print(f"Local refinement improved to {result.fun:.6f}")
        return result.x, result.fun
    else:
        print("Local refinement failed")
        return h, np.max(compute_correlations(h, dx)) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # We'll only use the actual seed when initializing h0 repeatedly
    # But we do not reproducibly initialize from it.
    
    for idx, n in enumerate(n_candidates):
        dx = dxs[idx]
        w = targets[idx]
        print(f"Testing n={n}, dx={dx:.4f}, w={w:.2f}")
        
        # Stop early if time nearly over
        if time.time() - start_time > budget_s * 0.8:
            print("Time nearly expired")
            break
            
        num_restarts = 200
        for restart in range(num_restarts):
            if time.time() - start_time > budget_s * 0.98:
                print("Early exit due to time")
                break
                
            # Initialize and project
            h0 = initial_h_values(n)
            h0 = project_onto_capped_simplex(h0, w)

            h_opt, c5_opt = grad_projected_descent(h0, dx, w)
            
            # Refine using scipy if value reasonable
            if c5_opt < 0.4 and restart % 10 == 0:
                try:
                    h_ref, c5_ref = local_refinement(h_opt, dx, w)
                    if c5_ref < c5_opt:
                        h_opt, c5_opt = h_ref, c5_ref
                except:
                    pass

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                print(f"New best C5={c5_opt:.6f} with n={n}")

    # Final safety check
    if best_h is not None:
        final_c5 = np.max(compute_correlations(best_h, 2.0 / best_n)) * (2.0 / best_n)
        best_h = project_onto_capped_simplex(best_h, targets[n_candidates.index(best_n)])
        final_c5_eval = np.max(compute_correlations(best_h, 2.0 / best_n)) * (2.0 / best_n)
        print(f"Final annual return: C5 = {final_c5_eval:.6f}")
        return list(best_h), final_c5_eval, best_n
    else:
        print("Failed to find any solution")
        # Return a fallback
        fallback_n = 48
        fallback_h = np.ones(fallback_n) * 0.5
        fallback_h = project_onto_capped_simplex(fallback_h, fallback_n / 2)
        fallback_c5 = np.max(compute_correlations(fallback_h, 2.0 / fallback_n)) * (2.0 / fallback_n)
        return list(fallback_h), fallback_c5, fallback_n
