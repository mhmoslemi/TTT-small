import numpy as np
from scipy.optimize import dual_annealing, minimize
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK_MI
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try small n first for integer programming 
    n_points = 72  # This should allow for meaningful optimization
    
    # Start with good initial pattern - try to find a good binary pattern
    # Using a simple construction: place 1s evenly and scale to get sum=36
    h0 = np.zeros(n_points)
    
    # Use golden ratio distribution for initial pattern (quasi-uniform)
    golden = (1 + math.sqrt(5)) / 2
    for i in range(n_points):
        val = (i * golden) % 1
        if val < 0.5:  # Half of the points are 1
            h0[i] = 1.0
    
    # Adjust to exactly half ones
    current_sum = np.sum(h0)
    diff = 36 - current_sum
    
    if diff > 0:
        # Need to add more 1s at positions that are currently 0
        zeros = np.where(h0 == 0)[0]
        for i in range(int(diff)):
            if i < len(zeros):
                h0[zeros[i]] = 1.0
    elif diff < 0:
        # Need to remove 1s
        ones = np.where(h0 == 1)[0]
        for i in range(int(-diff)):
            if i < len(ones):
                h0[ones[i]] = 0.0
    
    # Now run optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_c5(h)
    
    # SB: Always normalize to ensure constraint sum(h) = n/2
    def constraint_eq(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Initialize best solution 
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Algorithm 1: Local search with random restarts from good patterns
    max_restarts = 300
    restart_count = 0
    while restart_count < max_restarts and (time.time() - start_time) < budget_s * 0.85:
        # Generate new random pattern but shifted for good coverage
        new_h = np.zeros(n_points)
        golden_offset = np.random.random()
        
        for i in range(n_points):
            val = (i * golden + golden_offset) % 1
            if val < 0.5:
                new_h[i] = 1.0

        # Adjust to exactly difference sum = n/2
        current_sum = np.sum(new_h)
        diff = 36 - current_sum
        
        if diff > 0:
            zeros = np.where(new_h == 0)[0]
            for i in range(int(diff)):
                if i < len(zeros):
                    new_h[zeros[i]] = 1.0
        elif diff < 0:
            ones = np.where(new_h == 1)[0]
            for i in range(int(-diff)):
                if i < len(ones):
                    new_h[ones[i]] = 0.0

        # Local optimize with SLSQP
        try:
            res = minimize(objective, new_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12, 'gtol': 1e-12})
            if res.success:
                h_candidate = res.x
                # Normalize to satisfy constraint
                dx = 2.0 / n_points
                scale = 1.0 / (np.sum(h_candidate) * dx)
                h_candidate = h_candidate * scale
                c5_candidate = objective(h_candidate) 
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except Exception as e:
            pass
            
        restart_count += 1
        if (time.time() - start_time) > budget_s * 0.9:
            break

    # Algorithm 2: Simulated Annealing refinement of best solution we found
    try:
        # We'll do a quick simulated annealing on best solution found so far
        def sa_objective(h):
            return evaluate_c5(h)
            
        # Set up bounds again for SA
        bounds_sa = [(0, 1) for _ in range(n_points)]
        
        # Run a very large simulated annealing to get a good shake 
        res_sa = dual_annealing(sa_objective, bounds_sa, maxiter=2000, 
                                initial_temp=0.01, maxfun=5000, seed=seed)
        if res_sa.success:
            h_candidate = res_sa.x
            # Normalize to satisfy constraint
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = sa_objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
    except Exception as e:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
