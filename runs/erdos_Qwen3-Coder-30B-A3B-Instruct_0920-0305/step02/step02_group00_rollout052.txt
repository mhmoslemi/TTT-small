import numpy as np
from scipy.optimize import minimize
import time

# Constants
TARGET_BOUND = 0.3807
INITIAL_N = 56
MAX_N = 80

def evaluate_c5(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_hyperplane(v, target_sum):
    """Project vector v onto [0,1]^n and then onto sum = target_sum."""
    # Clip to [0,1]
    v_clipped = np.clip(v, 0, 1)
    
    # If the current sum is already correct, no need to adjust
    current_sum = np.sum(v_clipped)
    if abs(current_sum - target_sum) < 1e-10:
        return v_clipped
    
    # Use bisection to find lambda such that clip(v - lambda, 0, 1) sums to target
    def sum_for_lambda(lam):
        return np.sum(np.clip(v - lam, 0, 1))
    
    # Bisection search for lambda
    low, high = -10.0, 10.0
    for _ in range(50):
        mid = (low + high) / 2.0
        s = sum_for_lambda(mid)
        if s < target_sum:
            high = mid
        else:
            low = mid
    
    # Apply final projection
    lam = (low + high) / 2.0
    projected = np.clip(v - lam, 0, 1)
    return projected

def get_shift_matrix(n, s):
    """Construct the shift matrix A_s for a given shift s."""
    A = np.zeros((n, n))
    # Forward shift
    if s > 0:
        A[:n-s, s:n] = 0.5
        A[s:n, :n-s] = 0.5
    elif s == 0:
        np.fill_diagonal(A, 1)
    else:
        # Backward shift
        A[s:n, :n+s] = 0.5
        A[:n+s, s:n] = 0.5
    return A

def compute_soft_min_weights(Cs, beta):
    """Compute soft-min weights given correlation values and β."""
    # Subtract the maximum to prevent numerical overflow
    Cs = np.array(Cs)
    max_C = np.max(Cs)
    exp_vals = np.exp(-beta * (Cs - max_C))
    weights = exp_vals / np.sum(exp_vals)
    return weights

def compute_correlations(h):
    """Compute all correlation values C_s for a given h."""
    dx = 2.0 / len(h)
    h = np.array(h)
    correlations = []
    for s in range(len(h)):
        # Use np.correlate to compute the correlation in the style of the problem
        # This computes sum_i h_i * h_{i+s} (circularly padded)
        corr = np.correlate(h, h, mode='full')
        # Take the middle part which corresponds to shift s
        idx = len(h) - 1 + s
        correlations.append(corr[idx])
    return np.array(correlations) * dx

def gradient_ascent_step(h, beta, step_size):
    """Perform one gradient ascent step with continuation."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute correlations C_s
    correlations = compute_correlations(h)  # Using the numpy method (full correl)
    Cs = correlations
    
    # Compute soft-min weights
    weights = compute_soft_min_weights(Cs, beta)
    
    # Form the gradient
    gradient = np.zeros(n)
    for s in range(n):
        A_s = get_shift_matrix(n, s)
        Ah = A_s @ h
        gradient += weights[s] * Ah
    
    # Scale the gradient by factor 2 * dx
    gradient *= 2 * dx
    
    # Perform gradient step
    h_new = h + step_size * gradient
    
    # Project back to feasible set
    h_new = project_onto_box_and_hyperplane(h_new, n / 2.0)
    
    return h_new

    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

        if len(orig_h) != n:
            # Interpolate

            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_orig, orig_h)
            h_interp = np.clip(h_interp, 0, 1)
            h_interp = project_onto_box_and_hyperplane(h_interp, n / 2.0)
            starts.append(h_interp)
        else:
            starts.append(project_onto_box_and_hyperplane(orig_h, n / 2.0))
            
        # 2. Random binary with correct sum
        h_binary = np.zeros(n)
        positions = np.random.choice(n, size=n//2, replace=False)
        h_binary[positions] = 1
        h_binary = project_onto_box_and_hyperplane(h_binary, n / 2.0)
        starts.append(h_binary)
        
        # 3. Random uniform then project
        h_random = np.random.uniform(0, 1, n)
        h_random = project_onto_box_and_hyperplane(h_random, n / 2.0)
        starts.append(h_random)
        
        # 4. Previous best if exists
        if best_h is not None and len(best_h) == n:
            starts.append(best_h.copy())
        
        # Multi-start gradient descent with continuation
        for i, h_start in enumerate(starts):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            h_current = h_start.copy()
            current_c5 = evaluate_c5(h_current)
            
            # Continuation schedule for β
            beta_schedule = [5, 15, 30, 60, 120]
            step_size = 0.005 * n  # Adjusted average step size
            
            for beta in beta_schedule:
                if time.time() - start_time > 0.9 * budget_s:

                        h_new = gradient_ascent_step(h_current, beta, step_size)
                        c5_new = evaluate_c5(h_new)
                        
                        # Accept only if improvement (or allow small noise)
                        if c5_new < current_c5 or (np.random.rand() < 0.01):
                            h_current = h_new
                            current_c5 = c5_new
                            
                            if current_c5 < best_c5:
                                best_c5 = current_c5
                                best_h = h_current.copy()

                    except Exception as e:

                # Early stopping if no improvement in several iterations
                if current_c5 < 0.3799:

    # Final Local Refinement
    if best_h is not None and time.time() - start_time < budget_s - 5:

        def objective(h):
            return evaluate_c5(h)  # Max of overlaps
        
        bounds = [(0, 1) for _ in range(len(best_h))]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * (2.0 / len(h)) - 1.0}

                refined_c5 = evaluate_c5(result.x)
                if refined_c5 < best_c5:

                    best_h = result.x
        except:
            pass
    
    # Final check
    if best_h is not None:

            best_c5 = evaluate_c5(best_h)
    else:
        # Fallback with initial_h_values
        h_init = np.array(initial_h_values)
        new_h = project_onto_box_and_hyperplane(h_init, INITIAL_N / 2.0)
        best_c5 = evaluate_c5(new_h)
        best_h = new_h
        best_n = INITIAL_N
        
    return list(best_h), best_c5, best_n
