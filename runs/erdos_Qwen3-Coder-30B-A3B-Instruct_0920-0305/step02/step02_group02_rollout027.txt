def solve_sdp(n):
    """Solve SDP relaxation for n points."""
    dx = 2.0 / n
    w = n / 2

    # Create symbolic variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)

    # Create Toeplitz matrices
    T_k = []
    for k in range(n - 1):
        t = np.zeros((n, n))
        np.fill_diagonal(t, 1)
        t = np.roll(t, k, axis=1)
        T_k.append(t)

    # Objective
    t = cp.Variable()
    constraints = [
        cp.trace(T_k[k] @ X) >= t for k in range(n - 1)
    ] + [
        cp.diag(X) == h * h,
        cp.sum(h) == w,
        X >> 0  # PSD
    ]

    # Add constraints for 0 <= h <= 1
    for i in range(n):
        constraints.append(0 <= h[i])
        constraints.append(h[i] <= 1)

    prob = cp.Problem(cp.Maximize(t), constraints)

    try:
        prob.solve(solver=cp.SCS, verbose=False)
        return h.value, X.value, t.value
    except:
        return None, None, None

def get_samples(h_sdp, n_points):
    h = h_sdp.copy()
    # Round the SDP solution
    try:
        from scipy.linalg import cholesky
        L = cholesky(h_sdp, lower=True)
        g = np.random.randn(L.shape[0])
        h_round = L.T @ g
        h_round = np.clip(h_round, 0, 1)
        # Project onto the feasible set
        r_sum = np.sum(h_round)
        scale = (n_points / 2) / r_sum
        h_round *= scale
        h_round = np.clip(h_round, 0, 1)
        return h_round
    except Exception:
        # fallback to exact projection
        return project_to_simplex(h)

def project_to_simplex(h):
    """Project h onto the simplex with ∫ h dx = 1."""
    n = len(h)
    dx = 2.0 / n
    target_sum = 1.0
    h_clipped = np.clip(h, 0, 1)

    # Water-filling algorithm for a scaled h
    # We reinterpret h as sample values for the integral = 1
    h_sorted = np.sort(h_clipped)[::-1]
    tmp_sum = 0.0
    threshold = 0.0

    for i in range(n):
        old_sum = tmp_sum
        tmp_sum += h_sorted[i]
        if tmp_sum / (i + 1) >= (target_sum + old_sum) / (i + 1):
            threshold = (target_sum + old_sum) / (i + 1)
            break
        else:
            threshold = tmp_sum / (i + 1)

    h_projected = np.clip(h_clipped - threshold, 0, 1)
    h_projected = h_projected * (target_sum / np.sum(h_projected))

    return h_projected

def fit_fourier(h, M=10):
    n = len(h)
    x = np.arange(n)
    A = np.zeros((n, 2 * M + 1))
    A[:, 0] = 1
    for m in range(1, M + 1):
        A[:, 2 * m - 1] = np.cos(np.pi * m * x / (n - 1))
        A[:, 2 * m] = np.sin(np.pi * m * x / (n - 1))

    # Solve least squares, ignoring box constraints
    coeffs = np.linalg.lstsq(A @ A.T @ A, A @ h, rcond=None)[0]
    return coeffs

def reconstruct_fourier(coeffs, n):
    M = (len(coeffs) - 1) // 2
    x = np.arange(n)
    h = coeffs[0] * np.ones(n)
    for m in range(1, M + 1):
        h += coeffs[2 * m - 1] * np.cos(np.pi * m * x / (n - 1))
        h += coeffs[2 * m] * np.sin(np.pi * m * x / (n - 1))
    return np.clip(h, 0, 1)

def evaluate_nonlinear(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    dx = 2.0 / len(h_values)
    c5 = max_correlation * dx
    return c5

def smooth_optimization(h_initial, M, n_points):
    def obj(coeffs):
        h = reconstruct_fourier(coeffs, n_points)
        return evaluate_nonlinear(h)

    coeffs = fit_fourier(h_initial, M=M)
    res = minimize(obj, coeffs, method='L-BFGS-B', options={'maxiter': 1000})
    return reconstruct_fourier(res.x, n_points)

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    for n in n_candidates:
        if time.time() - start_time > budget_s:
            break

        h_sdp, X_sdp, _ = solve_sdp(n)
        if h_sdp is None:
            continue

        h_round = get_samples(h_sdp, n)
        h_smooth = smooth_optimization(h_round, M=min(10, n // 8), n_points=n)
        c5 = evaluate_nonlinear(h_smooth)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_smooth.copy()
            best_n = n

        if best_c5 <= target_c5:
            break

    if best_h is None:
        best_h = initial_h_values()
    return list(best_h), best_c5, best_n
