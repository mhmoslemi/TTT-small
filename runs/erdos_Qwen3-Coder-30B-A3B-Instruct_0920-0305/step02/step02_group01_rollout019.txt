import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def compute_gradient(h, shift):
    """Compute gradient of q_shift(h) = sum h_i * h_{i+shift} """
    n = len(h)
    grad = np.zeros(n)
    if shift > 0:
        grad[shift:] += h[:-shift]
        grad[:-shift] += h[shift:]
    else:
        shift = -shift
        grad[shift:] += h[:-shift]
        grad[:-shift] += h[shift:]
    return grad

def capped_simplex_projection(grad, h_old, target_sum):
    """Project gradient onto feasible set with constraint sum(h) = target_sum
    using capped-simplex approach (analytical solution).
    """
    # Sort indices by decreasing gradient values
    idxs = np.argsort(-grad)
    h_new = np.zeros_like(h_old)
    remaining = target_sum
    i = 0
    
    for idx in idxs:
        if remaining <= 0:
            break
        if h_old[idx] < 1:
            take = min(1 - h_old[idx], remaining)
            h_new[idx] = h_old[idx] + take
            remaining -= take
        else:
            h_new[idx] = h_old[idx]
    # Renormalize to exact sum if needed (ensuring that the sum constraint is satisfied)
    current_sum = np.sum(h_new)
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-10:
        # Distribute the difference among elements with maximum gradient
        # but respecting bounds
        if diff > 0:
            for idx in idxs:
                if diff <= 0:
                    break
                if h_new[idx] < 1:
                    take = min(1 - h_new[idx], diff)
                    h_new[idx] += take
                    diff -= take
        else:
            for idx in idxs:
                if diff >= 0:
                    break
                if h_new[idx] > 0:
                    take = min(h_new[idx], -diff)
                    h_new[idx] -= take
                    diff += take
    return h_new

    # We can generate q_s values by using correlate(h,h).
    # But for a given shift s, we want sum_i h_i * (1-h_{i+s}) = sum_i h_i - sum_i h_i h_{i+s}.
    
    # We now compute cross-correlation of h with (1 - h) for different shifts.
    # Let's do this more accurately:
    # Actually, we want: evaluation is max_s ( integral of h * (1 - h[shift]) ) 
    # i.e. for each shift s, it's sum_i h_i * (1 - h_{i+s}) = sum h_i - sum h_i h_{i+s}.
    h_conj = 1 - h  # Each element of h_conj is (1 - h_i)
    corr_h_conj = correlate(h, h_conj, mode="full")
    
    max_corr_idx = np.argmax(corr_h_conj)
    # The shift for max correlation is offset: shift = max_corr_idx - (n-1)
    s = max_corr_idx - (n - 1)
    
    # Negative q_s(h) = - (sum h_i h_{i+s}) = sum h_i - overlap_s
    # To optimize the f(h) = max_s overlap_s = max_s (sum h_i - q_s(h)), 
    # its minimum is approximated by the worst shift
    
    # For the *worst* overlap:
    min_corr_idx = np.argmin(corr_h_conj)
    worst_shift = min_corr_idx - (n - 1)

    grad = compute_gradient(h, worst_shift)

    # That means minimize -grad * h subject to 0 <= h <= 1, sum h = target_sum
