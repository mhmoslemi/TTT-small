import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto [0,1] hyperplane with sum constraint"""
    lo, hi = 0.0, 1.0
    w = np.clip(v, lo, hi)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
        
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h, dx):
    """Compute correlation and find maximum overlap"""
    conv = np.correlate(h, 1 - h, mode="full")
    conv *= dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    shift = argmax - (len(h) - 1)
    return max_val, shift

def gradient_for_shift(h, m):
    """Compute gradient for given shift m"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)

    return h
