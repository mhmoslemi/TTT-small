def project_onto_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h_i <= 1, sum(h) = target_sum"""
    # Clip to [0, 1]
    w = np.clip(h, 0, 1)
    current_sum = np.sum(w)
    delta = target_sum - current_sum
    
    if abs(delta) < 1e-10:
        return w
    
    # Water filling algorithm
    # Sort indices by w_i value increasing
    sort_idx = np.argsort(w)
    remaining = delta
    num_fixed = 0
    i = 0
    
    while abs(remaining) > 1e-10 and i < len(w):
        # Find next boundary value
        if num_fixed == 0:
            threshold = 1.0
        else:
            threshold = w[sort_idx[num_fixed - 1]]
        
        # All elements with value <= threshold will be updated together
        # Find first element with w[i] > threshold
        while i < len(w) and w[sort_idx[i]] <= threshold:
            i += 1
        if i >= len(w):
            # All points have been processed
            break
            
        # Increment up to threshold
        if num_fixed == 0:
            # First batch: all are below 1, add to all
            update = remaining / (len(w) - num_fixed)
            if threshold - w[sort_idx[num_fixed]] > 0:
                max_update = (threshold - w[sort_idx[num_fixed]]) * (len(w) - num_fixed)
                update = min(update, max_update)
        else:
            # Later batches
            if i == len(w):
                update = remaining / (len(w) - num_fixed)
            else:
                update = remaining / (i - num_fixed)
                # Cap by next threshold
                val_to_update = w[sort_idx[i - 1]]
                update = min(update, (val_to_update - threshold) * (i - num_fixed))
        
        # Apply update
        for j in range(num_fixed, i):
            w[sort_idx[j]] += update
        remaining -= update * (i - num_fixed)
        # fix current batch and continue to next batch
        num_fixed = i
        # Move past threshold to find new quantities
        while i < len(w) and w[sort_idx[i]] == threshold:
            i += 1
    
    if abs(remaining) > 1e-8:
        # distribute remaining equally
        if remaining > 0: # need to set some elements to 1
            for i in range(len(w)):
                if w[i] < 1.0:
                    add = min(remaining, 1.0 - w[i])
                    w[i] += add
                    remaining -= add
                    if remaining <= 1e-8:
                        break
        else:  # need to set some elements to 0
            for i in range(len(w)):
                if w[i] > 0.0:
                    remove = min(-remaining, w[i])
                    w[i] -= remove
                    remaining += remove
                    if remaining >= -1e-8:
                        break
    
    return w

def compute_subgradient(h, m, dx):
    """Compute the subgradient for the max A_m."""
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        # h[i](1 - h[i+m]) for i in range(n-m)
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # Flip cases to handle negative lag correctly
        m = -m
        i = np.arange(m, n)
        j = i - m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * dx

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val
