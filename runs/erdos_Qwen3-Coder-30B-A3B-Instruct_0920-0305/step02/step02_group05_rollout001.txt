import numpy as np
from scipy.interpolate import interp1d
import time

def project_to_bounds_and_sum(v, target):
    """Project vector onto box constraints [0,1] and sum constraint"""
    lo = 0.0
    hi = 1.0
    # First clip to bounds
    w = np.clip(v, lo, hi)
    # Compute sum
    total = w.sum()
    if abs(total - target) < 1e-12:
        return w
    # We'll iteratively adjust
    while True:
        diff = target - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                # cannot increase more, break
                break
            delta = diff / n_free
            w[free] += delta
            # Clip again
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a specific shift"""
    n = len(h)
    g = np.zeros_like(h)
    dx = 2.0 / n
    
    if m >= 0:
        # Overlapping region: i from 0 to n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # Overlapping region: i from -m to n-1 (i.e., from |m| to n-1)
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    return g * dx

def compute_max_overlap(h, dx):
    """Compute maximum overlap integral"""
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    argmax = np.argmax(conv)
    shift_m = argmax - (len(h) - 1)
    return max_val, shift_m
