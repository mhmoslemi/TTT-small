import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def construct_five_step(a, b, n):
    """Construct a 5-step symmetric function with parameters a, b."""
    dx = 2.0 / n
    x = np.arange(0.5, n + 0.5) * dx  # sample points in [0,2)
    u = (2 * b - 1) / (2 * (b - a))
    if u < 0 or u > 1:
        return None, False
    # 0 - a - u - 1 - u - a - 0, symmetric around 1
    # Split the support into 5 pieces: [0,a), [a,b), [b,2-b), [2-b,2-a), [2-a,2)
    h = np.zeros(n)
    # Compute cumulative positions
    total_length = 2.0
    # based on symmetry, we need:
    # 0->a (length a)
    # a->b (length b-a)
    # b->(2-b) (length 2-2b)
    # (2-b)->(2-a) (length b-a)
    # (2-a)->2 (length a)
    # summed: a + (b-a) + (2-2b) + (b-a) + a = 2  
    # So all intervals are non-negative
    
    # Halve the width to get from 0 to a etc.
    # Lengths: 0->a (a), a->b (b-a), b->(2-b) (2-2b), (2-b)->(2-a) (b-a), (2-a)->2 (a)
    # We use whole intervals (n points) and some positional offset, so just use box integration
   
    # Based on x in [0,2)
    # Construct the vector h on subintervals
    # First compute positions 
    # interval: 0 to a, a to b, b to 2-b, 2-b to 2-a, 2-a to 2
    # For each sample point x in [0,2), check which interval it belongs to
    
    # No need for careful intervals, use the order as a 5-step function horizontally:
    # [0   ] [a   ] [u   ] [b   ] [1   ] [u   ] [2-b] [b   ] [1   ] [a   ] [2   ]
    # But because of symmetry, we can simply:
    # i) From i=0 to i=int(a/dx) -> 0
    # ii) Then from int(a/dx) to int(b/dx) -> u  
    # iii) Then from int(b/dx) to int((2-b)/dx) -> 1  
    # iv) Then from int((2-b)/dx) to int((2-a)/dx) -> u  
    # v) Then from int((2-a)/dx) to n → 0
    
    # Let's compute integrals of this to satisfy constraint exactly later
    # and then just use loops to build properly:
    
    # Optimize later by vectorizing; for now no worry about efficiency
    
    # Trick: just check segments:
    a_ind = int(a/dx)
    b_ind = int(b/dx)
    # center segment
    center_ind_start = int((2-b)/dx)
    center_ind_end = int((2-a)/dx)
    
    h[:a_ind] = 0.0
    h[a_ind:b_ind] = u
    h[b_ind:center_ind_start] = 1.0
    h[center_ind_start:center_ind_end] = u
    h[center_ind_end:] = 0.0
    
    # Adjust for rounding error:
    # ensure h is in [0,1] and integral is 1
    return h, True

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: h in [0,1] and sum(h) = target_sum"""
    # Use bisection on lambda to solve F(lambda) = target_sum
    # F(lambda) = sum clip(h_i - lambda, 0, 1)
    h_new = h.copy()
    low, high = -np.max(h), np.max(h - 1)
    for _ in range(40):
        lambda_cand = (low + high) / 2.0
        projected = np.clip(h_new - lambda_cand, 0, 1)
        current_sum = np.sum(projected)
        if current_sum > target_sum:
            low = lambda_cand
        else:
            high = lambda_cand
    return np.clip(h_new - (low + high) / 2.0, 0, 1)

def smooth_max_corr(h, alpha):
    """Compute soft-max over correlations"""
    n = len(h)
    corr = correlate(h, 1 - h, mode="full")
    weights = np.exp(alpha * corr)
    weights /= np.sum(weights)
    return corr, weights

def gradient_of_smooth_objective(h, weights):
    n = len(h)
    # Compute g_plus = convolve(weights[::-1], h, mode='full')[n-1:n-1+n]
    # and g_minus = convolve(weights, h, mode='full')[n-1:n-1+n]
    # These are convolved with reversed and non-reversed weights
    
    # scipy.signal.convolve(correlate with 1-h, h)
    # This uses padding on both ends
    
    # Py (rotate weights): 
    weights_rev = weights[::-1]
    g_plus = np.convolve(weights_rev, h, mode='full')
    g_minus = np.convolve(weights, h, mode='full')
    
    # Get center part
    g_plus = g_plus[n-1:n-1+n]
    g_minus = g_minus[n-1:n-1+n]
    
    # grad_i = 1 - (g_plus_i + g_minus_i)
    grad = 1.0 - (g_plus + g_minus)
    return grad
