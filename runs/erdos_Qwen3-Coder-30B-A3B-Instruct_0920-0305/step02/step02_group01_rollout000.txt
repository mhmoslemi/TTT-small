import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(v, S):
    """Project v onto the set {x: sum(x)=S, 0 ≤ x ≤ 1}"""
    # Use bisection on lambda.
    lo = -np.max(v) - 1.0   # search
    hi = np.max(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        sum_x = x.sum()
        if sum_x > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Ensure sum close to S
    residual = S - x.sum()
    if abs(residual) > 1e-10:
        free = (x > 1e-8) & (x < 1-1e-8)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0, 1)
    return x

def gradient_descent(h0, S, budget_s, start_time):
    """Run gradient descent with smoothing and projection"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    dx = 2.0 / n
    alpha = 0.5
    lr = 0.5
    iteration = 0
    max_iter = 5000
    max_iter_since_improvement = 1000
    last_improvement = 0
    
    while (time.time() - start_time < budget_s) and (iteration < max_iter):
        # compute conv and smoothed weights
        conv = correlate(h, 1 - h, mode='full')
        # compute p using softmax
        shift = conv - np.max(conv)
        exp_vals = np.exp(alpha * shift)
        p = exp_vals / exp_vals.sum()
        
        # compute gradient via convolution
        p_rev = p[::-1]
        conv_plus = np.convolve(p_rev, h, mode='full')[n-1:n-1+n]
        conv_minus = np.convolve(p, h, mode='full')[n-1:n-1+n]
        g = 1.0 - (conv_plus + conv_minus)
        
        # update
        h_new = h - lr * g
        # project onto capped simplex
        h = project_to_capped_simplex(h_new, S)
        
        # evaluate C5
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # adapt alpha and lr
        iteration += 1
        if iteration % 200 == 0:
            alpha = min(alpha * 1.5, 20)
            lr = lr * 0.9
            
        # early stopping
        if iteration - last_improvement > max_iter_since_improvement:
            break
    
    return best_h, best_c5
