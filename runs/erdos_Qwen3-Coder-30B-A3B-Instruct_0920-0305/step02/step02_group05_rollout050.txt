import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector v onto box [0,1] with sum constraint"""
    # First clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a given shift m"""
    n = len(h)
    g = np.zeros_like(h)
    dx = 2.0 / n
    
    if m >= 0:
        # For positive shift m: indices from 0 to n-m-1 overlap
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # For negative shift: indices from -m to n-1 overlap
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    return g * dx

def compute_max_overlap(h):
    """Compute maximum overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    k = argmax - (n - 1)  # convert index to actual shift
    return max_val, k, conv
