import numpy as np
from scipy.fft import fft, ifft
import time
from cvxpy import *

def evaluate_h(h):
    """Fast evaluation of C5 using FFT-based correlation."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2*n for full correlation
    h_pad = np.pad(h, (0, n), 'constant')
    g_pad = np.pad(1 - h, (0, n), 'constant')
    H = fft(h_pad)
    G = fft(g_pad)
    corr = ifft(H * np.conj(G)).real
    max_overlap = np.max(corr[:2*n-1])
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use fixed grid size for binary formulation
    n = 48
    dx = 2.0 / n
    M = n // 2  # Integral constraint: sum h = M
    
    # Determine target bound T via desired C5
    T_target = int(np.ceil(0.3807 * n / 2))  # Want C5 <= 0.3807
    T_target = max(T_target, 1)  # Avoid invalid small T
    
    # --- Phase 1: MILP for binary h ---
    if budget_s < 30:
        h_bin = np.array(initial_h_values)
        h_bin = h_bin[:n]  # truncate to n if needed
        h_bin = np.clip(h_bin, 0, 1)
        h_bin = h_bin / (np.sum(h_bin) * dx) * M  # rescale to exact mass
        best_h = h_bin.copy()
        best_c5 = evaluate_h(best_h)
    else:
        # Build MILP model using CVXPY
        h_bin_vars = Bool(n, name="h")
        p_vars = {}
        for l in range(1, n):
            p_vars[l] = Bool(n - l, name=f"p_{l}")
        
        T = Int(name="T")
        
        # Mass constraint
        constraints = [sum(h_bin_vars) == M]
        
        # Product constraints: p[ℓ][i] = h[i] * h[i+ℓ]
        for l in range(1, n):
            for i in range(n - l):
                p = p_vars[l][i]
                h1 = h_bin_vars[i]
                h2 = h_bin_vars[i + l]
                constraints.append(p <= h1)
                constraints.append(p <= h2)
                constraints.append(p >= h1 + h2 - 1)
        
        # Worst-lag constraint
        for l in range(1, n):
            # sum(p[l][i]) = total overlap for shift l
            # Total overlap is sum(h[i]*h[i+l]) == sum(p[l][i])
            # So overlap = M - sum(p[l][i]) for that shift
            # In the linearised version: sum(p[l][i]) + T >= M 
            # So one way to phrase it: 
            # Since this means we're modeling C5 = max of (M - sum(p[l][i]))
            # We have (M - sum(p[l][i])) <= T
            # This is indeed sum(p[l][i]) >= M - T
            # Or just sum(p[l][i]) + T >= M 
            constraints.append(sum(p_vars[l]) + T >= M)
        
        # Lower bound on T
        constraints.append(T >= 0)
        
        # Objective
        objective = Minimize(T)
        
        prob = Problem(objective, constraints)
        
        # Solve
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining < 5:
            # Skip if not enough time
            h_bin = np.array(initial_h_values)
            h_bin = h_bin[:n]
            h_bin = np.clip(h_bin, 0, 1)
            h_bin = h_bin / (np.sum(h_bin) * dx) * M
            best_h = h_bin.copy()
            best_c5 = evaluate_h(best_h)
        else:
            time_limit = min(time_remaining * 0.35, 100.0)  # Max 100s for MILP
            prob.solve(solver=CPLEX, verbose=False, maxSeconds=time_limit)
            
            # Extract solution
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = [float(h_bin_vars[i].value) for i in range(n)]
            else:
                # Fallback: find a feasible point manually
                h_val = np.copy(initial_h_values[:n]) if len(initial_h_values) >= n else np.ones(n) * 0.5
                h_val = np.clip(h_val, 0, 1)
                sum_h = np.sum(h_val) * dx
                h_val = h_val * M / sum_h
                h_sol = h_val
            
            # Final rescale to ensure mass is exactly M
            h_sol = np.array(h_sol)
            sum_h = np.sum(h_sol) * dx
            if abs(sum_h - 1.0) > 1e-9:
                h_sol = h_sol * (M / sum_h)
            best_h = h_sol
            best_c5 = evaluate_h(best_h)
            
            # Check if already good enough
            if best_c5 <= 0.3807:
                # We're done, no further refinement needed
                return (best_h.tolist(), best_c5, n)

    # --- Phase 2: Binary local search (Simulated Annealing) ---
    if budget_s - (time.time() - start_time) < 10:
        return (best_h.tolist(), best_c5, n)
    
    h_bin_best = best_h.copy()
    c5_best = best_c5
    
    # Convert to binary
    h_bin_current = np.round(h_bin_best).astype(int)
    h_bin_current = np.clip(h_bin_current, 0, 1)
    if np.sum(h_bin_current) != M:
        h_bin_current = np.zeros(n)
        h_bin_current[:M] = 1
        np.random.shuffle(h_bin_current)
        
    # Fast evaluation with FFT
    def compute_C5_fast(h_in):
        # Use FFT trick to compute all lags quickly
        h_pad = np.pad(h_in, (0, n), 'constant')  # pad to avoid wraparound
        g_pad = np.pad(1 - h_in, (0, n), 'constant')
        H = fft(h_pad)
        G = fft(g_pad)
        corr = ifft(H * np.conj(G)).real
        return np.max(corr[:2*n-1]) * dx
    
    def fast_C5_diff(h_old, h_new, iold, inew):
        # Compute difference in overlap if we swap positions iold with inew
        # This is partial indices only for CPU efficiency.
        # Simplified for speed.
        p1, p2 = h_old[iold], h_old[inew]
        if p1 == p2:
            return 0
        c_diff = 0.0
        # Introduce/update shifts to measure
        for l in range(1, min(5, n)):  # Cap at 5 shifts for speed
            i11 = iold - l
            i12 = iold + l
            i21 = inew - l
            i22 = inew + l
            #ycling terms - just approximate!
            for i_old_p in [i11, i12]:
                for i_new_p in [i21, i22]:
                    if 0 <= i_old_p < n and 0 <= i_new_p < n:
                        # Approximate the difference via one shift
                        c_diff += (
                            (h_old[iold] * (1 - h_old[i_old_p]) - h_new[iold] * (1 - h_new[i_old_p])) +
                            (h_old[inew] * (1 - h_old[i_new_p]) - h_new[inew] * (1 - h_new[i_new_p]))
                        )
        return c_diff * dx
        
    def compute_current_C5(h_in):
        return compute_C5_fast(h_in)
    
    # Simple simulated annealing on binary points
    T_current = 1.0
    
    # Sample random swaps
    warmup = 1000
    for _ in range(warmup):
        candidates_on = np.where(h_bin_current == 1)[0]
        candidates_off = np.where(h_bin_current == 0)[0]
        if len(candidates_on) == 0 or len(candidates_off) == 0:
            continue
        i = np.random.choice(candidates_on)
        j = np.random.choice(candidates_off)
        # Swap the bits
        h_bin_current[i], h_bin_current[j] = h_bin_current[j], h_bin_current[i]
        c5_new = compute_current_C5(h_bin_current)
        dC5 = c5_new - c5_best
        if dC5 < 0 or np.random.rand() < np.exp(-dC5 / T_current):
            # Accept
            if dC5 < 0:
                c5_best = c5_new
                h_bin_best = h_bin_current.copy()
        else:
            # Restore
            h_bin_current[i], h_bin_current[j] = h_bin_current[j], h_bin_current[i]
        T_current *= 0.9999
    
    # Final refinement of best solution
    best_h = h_bin_best.copy()
    best_c5 = c5_best
    
    # --- Phase 3: Smooth fractional refinement ---
    if budget_s - (time.time() - start_time) < 10:
        return (best_h.tolist(), best_c5, n)
        
    # Use log-sum-exp approximation to maximize smoothed C5
    max_iter = 500  # Total steps
    beta_schedule = [1, 2, 5, 10, 20]
    eta = 0.1  # Step size
    
    h = best_h.copy()
    
    # # Water-filling helper
    def project_mass(h_input, target_sum):
        h_vec = h_input.copy()
        h_vec = np.clip(h_vec, 0, 1)
        temp_sum = np.sum(h_vec) * dx  # In real units
        if abs(temp_sum - target_sum) < 1e-8:
            return h_vec
        # Water filling via bisection of lambda
        def water_fill_lambda(lam):
            w = np.clip(h_vec + lam, 0, 1)
            return np.sum(w) * dx - target_sum
        # Bisection
        low, high = -10, 10
        for _ in range(30):
            mid = (low + high) / 2
            val = water_fill_lambda(mid)
            if val > 0:
                high = mid
            else:
                low = mid
        final_lambda = (low + high) / 2
        h_final = np.clip(h_vec + final_lambda, 0, 1)
        return h_final
    
    try:
        for beta in beta_schedule:
            for _ in range(max_iter // len(beta_schedule)):
                # Compute all lags
                h_pad = np.pad(h, (0, n), 'constant')
                g_pad = np.pad(1 - h, (0, n), 'constant')
                H = fft(h_pad)
                G = fft(g_pad)
                corr = ifft(H * np.conj(G)).real
                overlaps = corr[:2*n-1] * dx
                
                # Gradient of smooth max using soft-argmax weights
                W_exps = np.exp(beta * overlaps)
                W_unnorm = W_exps / np.sum(W_exps)
                
                # Approximate gradient so it runs fast
                grad = np.zeros(len(h))
                for l in range(1, min(n-1, 10)):  # Very coarse approach
                    if l < len(overlaps):
                        # Go forward first
                        if l < len(h):
                            grad[:-l] += W_unnorm[l] * (-h[l:])
                        # Reverse
                        if (l-1) < len(h):
                            grad[l-1:] += W_unnorm[l] * (-h[:-l])
                # Backwards term (double counting due to both mappings)
                # just approximate as up to 1 hop to keep it fast!
                # Naive gradient with no kernel smoothing
                
                h_new = h - eta * grad
                # Project onto feasible set (clipping, rescaling)
                h_new = project_mass(h_new, M * dx)
                h = h_new
                # Final update of c5
                c5_current = np.max(overlaps)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h.copy()
    except:
        # Silently ignore gradient descent error
        pass
    
    # Final hard denormalization
    best_h = project_mass(best_h, 1.0)  # Should be good already
    
    # Report result
    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n)
