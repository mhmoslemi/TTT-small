def project_onto_box_and_sum(h, w):
    """Project h onto [0,1]^n and adjust sum to w."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - w) > 1e-6:
        scale = w / s
        h *= scale
    return h

def fit_fourier(h, M=6):
    """Fit h to a truncated cosine/sine basis."""
    n = len(h)
    # Simple Least-square fit using a test basis
    # θ = [a_1,...,a_M, b_1,...,b_M]
    x = np.arange(n) / (n - 1)
    
    # Create Fourier basis functions for least-squares
    A = np.ones((n, 1 + 2 * M))  # 1 + M*cos + M*sin
    for m in range(1, M + 1):
        A[:, 1 + m - 1] = np.cos(np.pi * m * x)
        A[:, 1 + M + m - 1] = np.sin(np.pi * m * x)

    # Solved by least squares
    coeff = np.linalg.lstsq(A, h, rcond=None)[0]
    
    return coeff

def reconstruct_fourier(coeff, n, M):
    """Reconstruct h from Fourier coefficients."""
    x = np.arange(n) / (n - 1)
    h = coeff[0] * np.ones(n)
    for m in range(1, M + 1):
        h += coeff[m] * np.cos(np.pi * m * x)
        h += coeff[M + m] * np.sin(np.pi * m * x)
    return h

def smooth_objective(coeff, n, M, alpha=1.0):
    """Smooth surrogate for auto-correlation sum."""
    h = reconstruct_fourier(coeff, n, M)
    h = project_onto_box_and_sum(h, n / 2)
    
    h_fft = fft(h)
    one_minus_h_fft = fft(1 - h)
    conv_fft = h_fft * np.conj(one_minus_h_fft)
    correlation = ifft(conv_fft).real
    max_correlation = np.max(correlation)
    return -max_correlation * 2.0 / n + np.log(np.sum(np.exp(alpha * correlation))) / alpha
