def project_onto_constraints(h, n_points):
    """Project h onto feasible set: 0 <= h <= 1, sum(h) = n/2"""
    # Clip to [0,1]
    w = np.clip(h, 0, 1)
    # Adjust to hit exact sum
    target = n_points / 2.0
    current_sum = np.sum(w)
    diff = target - current_sum
    
    # Water filling algorithm
    while abs(diff) > 1e-10:
        if diff > 0:  # Need to increase
            # Find indices where w < 1
            free_indices = np.where(w < 1)[0]
            if len(free_indices) == 0:
                break
            # Distribute diff equally
            distribute_per = diff / len(free_indices)
            for i in free_indices:
                if w[i] + distribute_per <= 1:
                    w[i] += distribute_per
                    diff -= distribute_per
                else:
                    diff -= (1 - w[i])
                    w[i] = 1.0
        else:  # Need to decrease
            # Find indices where w > 0
            free_indices = np.where(w > 0)[0]
            if len(free_indices) == 0:
                break
            # Distribute diff equally
            subtract_per = -diff / len(free_indices)
            for i in free_indices:
                if w[i] - subtract_per >= 0:
                    w[i] -= subtract_per
                    diff += subtract_per
                else:
                    diff += w[i]
                    w[i] = 0.0
                    
    return w

def compute_subgradient(h, m, n_points, dx):
    """Compute subgradient for lag m"""
    g = np.zeros(n_points)
    
    # Check if m >= 0
    if m >= 0:
        if m >= n_points:
            return g
        i_vals = range(0, n_points - m)
        j_vals = range(m, n_points)
    else:
        m = -m
        if m >= n_points:
            return g
        i_vals = range(m, n_points)
        j_vals = range(0, n_points - m)
    
    for i, j in zip(i_vals, j_vals):
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    g *= dx
    return g

def find_worst_lag(A):
    """Find all lags that achieve the maximum value in A"""
    max_val = np.max(A)
    indices = np.where(np.abs(A - max_val) < 1e-10)[0]
    return indices, max_val

    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val
