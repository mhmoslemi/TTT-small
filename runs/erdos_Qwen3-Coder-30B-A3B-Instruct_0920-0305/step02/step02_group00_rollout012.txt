import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807
n_candidates = [48, 56, 60]
dx_candidates = [2.0 / n for n in n_candidates]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_onto_box_and_sum(h, target_sum):
    """Project h onto [0,1]^n and then adjust to match target_sum."""
    # Clip to [0,1]
    h_proj = np.clip(h, 0, 1)
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) < 1e-10:
        return h_proj
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    def f(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target_sum
    # Bisection
    low, high = -1.0, 2.0
    for _ in range(50):
        mid = (low + high) / 2.0
        val = f(mid)
        if val > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    h_final = np.clip(h - lam, 0, 1)
    return h_final

def compute_shifted_vector(h, shift):
    """Compute the shift-apply vector efficiently."""
    n = len(h)
    result = np.zeros(n)
    if shift == 0:
        return h.copy()
    elif shift > 0:
        result[shift:] = h[:-shift]
    else:
        result[:shift] = h[-shift:]
    return result

def gradient_step(h, shifts, C_vals, weights, dx):
    """Compute gradient vector."""
    grad = np.zeros_like(h)
    for s in shifts:
        h_shifted = compute_shifted_vector(h, s)
        if s < 0:
            h_shifted = compute_shifted_vector(h, -s)
            # Reverse the definition to match Let's make it clean by cycling around shifts
        grad += weights[s] * h_shifted
    return 2 * dx * grad

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n, dx in zip(n_candidates, dx_candidates):
        if time.time() - start_time > 0.95 * budget_s:
            break

        target_sum = n / 2.0

        def compute_Cs(h):
            h_np = np.array(h)
            C_s = np.zeros(n)
            for s in range(n):
                # We compute O_s = sum h_i * h_{i+s}
                if s == 0:
                    O_s = np.sum(h_np ** 2)
                else:
                    h_shifted = np.roll(h_np, s)
                    O_s = np.sum(h_np * h_shifted)
                C_s[s] = dx * O_s
            return C_s

        # Multi-start with different initialization strategies
        starts = []

        # Interpolated from initial_h_values
        initial_h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        starts.append(project_onto_box_and_sum(initial_h_interp, target_sum))

        # Random binary with exact sum
        binary_h = np.zeros(n)
        ones = np.random.choice(n, size=int(target_sum), replace=False)
        binary_h[ones] = 1.0
        starts.append(project_onto_box_and_sum(binary_h, target_sum))

        # Random uniform with projected sum
        unif_h = np.random.uniform(0, 1, n)
        starts.append(project_onto_box_and_sum(unif_h, target_sum))

        # Previous best scaled
        scaled_h = np.array(initial_h_values) * (n / len(initial_h_values))
        starts.append(project_onto_box_and_sum(scaled_h, target_sum))

        for start_seed, start_h in enumerate(starts):
            if time.time() - start_time > 0.95 * budget_s:
                break

            h = start_h.copy()
            current_c5 = evaluate_h(h)
            best_in_start = current_c5
            best_in_start_h = h.copy()

            # Reconstruction with max-300 steps for either β schedule or iter
            beta_schedule = [5, 15, 30, 60, 120]
            max_inner_iter = 300

            for beta in beta_schedule:
                if time.time() - start_time > 0.95 * budget_s:
                    break
                for k in range(max_inner_iter):
                    if time.time() - start_time > 0.95 * budget_s:
                        break

                    # Step 1: Compute all C_s
                    C_vals = compute_Cs(h)

                    # Step 2: Compute soft-min weights
                    C_vals_max = np.max(C_vals)
                    if C_vals_max > 1e8:
                        C_vals -= C_vals_max
                    exp_vals = np.exp(-beta * C_vals)
                    exp_sum = np.sum(exp_vals)
                    weights = exp_vals / exp_sum
                    weights = np.nan_to_num(weights, nan=0.0, posinf=0.0, neginf=0.0)

                    # Step 3: Compute gradient
                    grad = np.zeros(n)
                    for s in range(n):
                        h_shifted = compute_shifted_vector(h, s)
                        grad += weights[s] * h_shifted
                    grad = 2 * dx * grad

                    # Step 4: Gradient ascent with backtracking
                    alpha = 0.5
                    step_size = 1e-3
                    alpha_t = alpha
                    for _ in range(10):  # At most 10 line search steps
                        h_new = h - step_size * grad
                        h_new = project_onto_box_and_sum(h_new, target_sum)
                        new_c5 = evaluate_h(h_new)
                        if new_c5 < current_c5:
                            h = h_new
                            current_c5 = new_c5
                            if new_c5 < best_in_start:
                                best_in_start = new_c5
                                best_in_start_h = h.copy()
                            break
                        alpha *= 0.5
                        if alpha < 1e-6:
                            break
                    if current_c5 < best_in_start:
                        best_in_start = current_c5
                        best_in_start_h = h.copy()

            # Local refinement using SLSQP
            if time.time() - start_time > 0.95 * budget_s:
                break
            bounds = [(0, 1) for _ in range(n)]
            def local_objective(h_vec):
                return evaluate_h(h_vec)
            try:
                result = minimize(local_objective, h, method='SLSQP', bounds=bounds,
                                 options={'maxiter': 200, 'ftol': 1e-12})
                if result.success and result.fun < best_in_start:
                    best_in_start = result.fun
                    best_in_start_h = result.x
            except:
                pass

            if best_in_start < best_c5:
                best_c5 = best_in_start
                best_h = best_in_start_h
                best_n = n

        # Early exit if target is met
        if best_c5 <= target_bound:
            break

    return list(best_h), best_c5, best_n
