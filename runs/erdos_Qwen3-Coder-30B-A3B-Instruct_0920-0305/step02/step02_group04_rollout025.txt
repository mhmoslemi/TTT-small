import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx using FFT"""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact MILP for binary h to minimize max overlap."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
    
    try:
        h_opt = np.ravel(h.value)
        T_opt = T.value
        score = T_opt * (2.0 / n)
        return T_opt, h_opt
    except:
        # fallback to best known feasible solution if timeout
        try:
            h_opt = np.ravel(h.value)
            # estimate T from constraints
            best_T = 0
            for ell in range(1, n):
                p_ell = p[ell-1].value
                if p_ell is not None:
                    T_est = M - np.sum(p_ell)
                    best_T = max(best_T, T_est)
            score = best_T * (2.0 / n)
            return best_T, h_opt
        except:
            return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=10):
    """Swap-based simulated annealing to improve binary solution."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    current_score = evaluate_h(h)
    best_h = h.copy()
    best_score = current_score
    
    t_start = time.time()
    T = 1.0
    T_min = 1e-6
    decay = 0.999
    
    # Try several random starting swaps
    acceptance_count = 0
    for i in range(1000):
        if time.time() - t_start > time_limit:
            break
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if len(ones) == 0 or len(zeros) == 0: break
        i_swap = np.random.choice(ones)
        j_swap = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i_swap] = 0
        h_new[j_swap] = 1
        new_score = evaluate_h(h_new)
        delta = new_score - current_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        if delta <= 0:
            T *= decay
        acceptance_count += 1
        if acceptance_count > 100 and T < 1e-4:
            break
    
    # Main simulation loop
    for iter in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        if len(ones) == 0 or len(zeros) == 0: break
        i_swap = np.random.choice(ones)
        j_swap = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i_swap] = 0
        h_new[j_swap] = 1
        new_score = evaluate_h(h_new)
        delta = new_score - current_score
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        T *= decay
        if T < T_min:
            break
    
    return best_h, best_score

def smooth_descent(h_start, time_limit=5.0):
    """Refine solution by smooth gradient descent on log-sum-exp."""
    n = len(h_start)
    M = n // 2
    h = h_start.copy()
    dx = 2.0 / n
    start_time = time.time()
    
    # Clip to [0,1], enforce exact mass
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h) * dx
    if abs(total - 1.0) > 1e-9:
        h *= (1.0 / total)
    
    def project_onto_mass(h, target):
        """Water-filling to enforce sum = target"""
        h = np.clip(h, 0.0, 1.0)
        def f(lam):
            return np.sum(np.clip(h + lam, 0.0, 1.0)) - target
        lo, hi = -1.0, 1.0
        for _ in range(100):
            mid = (lo + hi) / 2.0
            if f(mid) > 0: hi = mid
            else: lo = mid
        lam = (lo + hi) / 2.0
        return np.clip(h + lam, 0.0, 1.0)
    
    beta_schedule = [1, 2, 5, 10, 20]
    eta = 0.1
    
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        
        max_iter = 200
        for _ in range(max_iter):
            if time.time() - start_time > time_limit:
                break
            # compute all L_ell values (slower for large n)
            L_values = []
            for ell in range(1, n):
                corr = np.correlate(h, 1 - h, mode='full')[:n-ell]
                val = np.sum(corr)
                L_values.append(val)
            
            # compute weights
            w = np.exp(beta * np.array(L_values))
            w = w / np.sum(w)
            
            # gradient computation (simplified)
            grad = np.zeros(n)
            for ell, w_ell in enumerate(w):
                if ell == 0:
                    continue
                # ∂L_ell/∂h_i = -h_{i+ℓ} - h_{i-ℓ} + 1
                for i in range(n - ell):
                    grad[i] += w_ell * (1 - h[i + ell] - h[i - ell] if i - ell >=0 else 1 - h[i + ell])
                for i in range(ell, n):
                    grad[i] += w_ell * (1 - h[i - ell] - h[i + ell] if i + ell < n else 1 - h[i - ell])
            
            # line search for stepsize
            h_new = h - eta * grad
            h_new = project_onto_mass(h_new, M * dx)
            val_new = evaluate_h(h_new)
            if val_new < evaluate_h(h):
                h = h_new
            else:
                eta *= 0.5
        # projection
        h = project_onto_mass(h, M * dx)
    
    return h, evaluate_h(h)
