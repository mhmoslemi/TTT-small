import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807
n_candidates = [48, 56, 60]
dx_candidates = [2.0 / n for n in n_candidates]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_onto_box_and_simplex(v, target_sum):
    """Project vector v onto the intersection of [0,1]^n and {x: sum x = target_sum}."""
    # Bisection to find lambda such that sum(clamp(v - lambda, 0, 1)) == target_sum
    low, high = np.min(v - 1), np.max(v)
    tol = 1e-12
    max_iter = 50
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0, 1)
        s = np.sum(x)
        if abs(s - target_sum) < tol:
            break
        elif s > target_sum:
            low = lam
        else:
            high = lam
    return np.clip(v - lam, 0, 1)

def compute_soft_min_weights(C_s, beta):
    """Compute soft-min weights for given correlations and temperature β."""
    C_s = np.array(C_s)
    # Subtract max for numerical stability
    C_s_stable = C_s - np.max(C_s)
    exp_neg_beta_C = np.exp(-beta * C_s_stable)
    weights = exp_neg_beta_C / np.sum(exp_neg_beta_C)
    return weights

def compute_gradient(h, C_s, beta, dx):
    """Compute gradient of soft minimum correlation."""
    # Compute shift matrix products efficiently
    g = np.zeros_like(h)
    n = len(h)
    
    # For each shift s (from 0 to n-1)
    for s in range(n):
        if s == 0:
            # Direct multiplication by identity
            g += C_s[s] * h
        else:
            # Shifting forward and backward
            h_shifted_forward = np.roll(h, -s)
            h_shifted_backward = np.roll(h, s)
            contribution = 0.5 * (h_shifted_forward + h_shifted_backward)
            g += C_s[s] * contribution
    
    # Manually the weighting by soft-min
    weights = compute_soft_min_weights(C_s, beta)
    gradient = 2 * dx * np.sum([weights[i] * (np.roll(h, -i) + np.roll(h, i)) * 0.5 for i in range(n)], axis=0)
    return gradient

def compute_correlations(h, dx):
    """Compute auto-correlations for all shifts."""
    h = np.array(h)
    n = len(h)
    correlations = []
    for s in range(n):
        if s == 0:
            corr = dx * np.sum(h * h)
        else:
            # Compute correlation with shift
            h_shifted = np.roll(h, s)
            h_shifted[:s] = 0
            corr = dx * np.sum(h * h_shifted)
        correlations.append(corr)
    return correlations

def gradient_ascent_step(h, beta, dx, max_iter=100, tol=1e-10):
    """Perform gradient ascent on soft-min correlation."""
    h = np.array(h)
    n = len(h)
    target_sum = n / 2.0
    
    current_c5 = evaluate_h(h)
    for iteration in range(max_iter):
        # Compute correlations for each shift
        C_s = compute_correlations(h, dx)
        
        # Compute soft-min weights
        weights = compute_soft_min_weights(C_s, beta)
        
        # Compute gradient (approximation to avoid full matrix mult)
        g = np.zeros(n)
        for s in range(n):
            if s == 0:
                g += 2 * dx * weights[s] * h
            else:
                h_shifted_forward = np.roll(h, -s)
                h_shifted_backward = np.roll(h, s)
                g += 2 * dx * weights[s] * 0.5 * (h_shifted_forward + h_shifted_backward)
        
        # Step size with backtracking line search
        alpha = 0.1
        step_count = 0
        max_backtrack = 5
        
        h_new = project_onto_box_and_simplex(h + alpha * g, target_sum)
        new_c5 = evaluate_h(h_new)
        
        # Initialize line search
        old_c5 = current_c5
        success = False
        while step_count < max_backtrack and not success:
            if new_c5 < old_c5:
                h = h_new.copy()
                current_c5 = new_c5
                success = True
            else:
                alpha *= 0.5
                h_new = project_onto_box_and_simplex(h + alpha * g, target_sum)
                new_c5 = evaluate_h(h_new)
                step_count += 1
        
        if not success:
            h = h_new.copy()
            current_c5 = new_c5
         
        if iteration % 20 == 0:
            # Periodic progress check
            pass
        
        # Early termination if improvement stagnates 
        if abs(old_c5 - new_c5) < tol:
            break
        
    return h, current_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    n_used = None
    
    # Determine which n to use
    for n in n_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        
        # Generate diverse starting points
        starts = []
        
        # Interpolated from initial values if available

                                initial_h_values)
            starts.append(h_interp)
        except:

        # Binary start
        h_binary = np.zeros(n)
        h_binary[:n//2] = 1
        np.random.shuffle(h_binary)
        h_binary_projected = project_onto_box_and_simplex(h_binary, n / 2)
        starts.append(h_binary_projected)
        
        # Random uniform start
        h_uniform = np.random.rand(n)
        h_uniform_projected = project_onto_box_and_simplex(h_uniform, n / 2)
        starts.append(h_uniform_projected)
        
        # Random binary start (not projected)
        h_rand_binary = np.random.choice([0, 1], size=n, p=[0.5, 0.5])
        h_rand_binary = project_onto_box_and_simplex(h_rand_binary, n / 2)
        starts.append(h_rand_binary)

        # Try multiple starts
        for i_start, h_start in enumerate(starts):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Run gradient ascent with continuation for β values
            h_current = h_start.copy()
            c5_current = evaluate_h(h_current)
            
            # Schedule of β values
            beta_schedule = [5, 15, 30, 60, 120]
            
            for beta in beta_schedule:
                if time.time() - start_time > 0.95 * budget_s:
                    break
                h_current, c5_current = gradient_ascent_step(h_current, beta, dx)
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
                    n_used = n
                    if best_c5 < target_bound:
                        # Early termination if target already reached
                        break
            
        # Final refinement with SLSQP
        bounds = [(0, 1) for _ in range(n)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1}
        
        try:
            result = minimize(evaluate_h, best_h, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-12})
            if result.success:
                refined_c5 = evaluate_h(result.x)
                if refined_c5 < best_c5:
                    best_h = result.x
                    best_c5 = refined_c5
                    n_used = n
        except:
            pass
    
    if best_h is None:
        # Fallback to initial if no good candidate found
        try:
            h_fallback = np.array(initial_h_values)
            best_c5 = evaluate_h(h_fallback)
            best_h = h_fallback
            n_used = len(h_fallback)
        except:
            best_h = np.random.rand(48)
            best_h = project_onto_box_and_simplex(best_h, 24)
            best_c5 = evaluate_h(best_h)
            n_used = 48
    
    return list(best_h), best_c5, n_used
