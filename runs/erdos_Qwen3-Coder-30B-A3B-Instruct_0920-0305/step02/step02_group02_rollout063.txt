def evaluate_c5(h):
    """Original evaluation using convolutions for exact C5."""
    h = np.array(h)
    # Compute correlations: h(x) * (1 - h(x+k))
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * (2.0 / len(h))

def project_onto_capped_simplex(v, target_sum):
    """
    Project vector v onto capped simplex: 0 <= v_i <= 1, sum v_i = target_sum
    Uses water-filling algorithm.
    """
    v = np.array(v, dtype=float)
    n = len(v)
    # Ensure the vector is in [0, 1]
    v = np.clip(v, 0, 1)
    vs = np.sort(v)[::-1]  # descending
    sum_v = np.sum(v)
    if abs(sum_v - target_sum) < 1e-12:
        return v
    # Water filling
    # We find threshold tau such that:
    # sum_i max(0, v_i - tau) = target_sum
    # Let rank_j = # of elements >= tau
    #sum_over_j = sum(v_i - tau for i where v_i >= tau) = tau * rank_j + sum_j,v_j-tau
    # If we use summation over sorted list, sort-wise:
    # We traverse sorted list and compute cumulative sum up to prefix sum_i
    # We want sum_i - rank_i * tau = target_sum for some point i
    # i such that sum_i - rank_i * tau = target_sum
    # target_sum = sum_i - rank_i * tau
    # tau = (sum_i - target_sum) / rank_i
    prefix_sum = 0.0
    for i in range(n):
        prefix_sum += vs[i]
        rank = i + 1
        temp_tau = (prefix_sum - target_sum) / rank
        if rank == n or temp_tau <= vs[i+1]:  # temp_tau valid
            tau = temp_tau
            break
    else:
        # must be tau = (prefix_sum - target_sum) / n  but last element is vs[n-1]
        tau = (prefix_sum - target_sum) / n
    # Apply threshold
    x = np.clip(v - tau, 0, 1)
    # Fix numerical errors: enforce exact sum
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-12:
        # Increment smallest non-unit entry
        mask = (x < 1)
        if np.any(mask):
            idx = np.argmin(x[mask])
            x[mask][idx] += diff
    return np.clip(x, 0, 1)

def correlation_term_for_shift(h, shift):
    """Efficiently compute sum h_i * h_{i+shift} with circular wraparound."""
    n = len(h)
    if shift == 0:
        return np.sum(h * h)
    # Circular convolution with reverse h (for correlation)
    h_extended = np.concatenate([h, h[:shift]])
    h_rev = np.flip(h)
    # Compute correlation at shift placement
    # return sum h_i * h_{i+shift}
    # To compute:
    # h_rev[0] * h[shift] + h_rev[1] * h[shift+1] ...
    # is same as correlation of h_rev with h shifted to the left by shift
    h_shifted = h_extended[shift:shift+n]
    return np.sum(h * h_shifted)

def correlation_terms(h):
    """More efficient by reusing FFT if needed."""
    n = len(h)
    # Compute all correlations with circular wraparound
    h_fft = fft(h)
    h2_fft = fft(h * h)
    cross = ifft(h_fft * np.conj(h_fft))
    # cross contains correlations: [r_{-n+1}, ..., r_{0}, ..., r_{n-1}]
    # r_k = sum_i h_i h_{i+k} (circular)
    cross_real = cross.real
    # Take positive indices (correspond to shift 0 to n-1)
    return cross_real[n-1:n-1+n]

def get_gradient_for_shift(h, k, n):
    """Given the correlation shift k (non-negative),
    compute the gradient of r_k = sum_i h_i * (1 - h_{i+k})="r_k".
    This is: 1 - h_{i+k} - h_{i-k} for i not boundary"
    """
    grad_rk = np.zeros_like(h)
    # Gradient = 1 - h_{i+k} - h_{i-k}
    # but pad with 0 so it's circular or truncated?
    # maybe due to circular wrap case, let's use numpy step
    # This formula: 
    # if k=0, then r_k = sum h_i^2 
    #  grad r_k = 2 * h_i for i
    # if k>0:
    #  r_k = sum h_i h_{i+k}
    #  grad r_k = h_{i+k} (in i-th entry) + 0 (firing x_{i+k} comprehesive valid)
    # So: 
    # Grad = ones vector minus h_{i+k} terms (when k>0), 
    # but with valid ranges (fixed boundary handling)
    h_shift = np.roll(h, k) if k > 0 else h.copy()
    h_shift_back = np.roll(h, -k) if k > 0 else h.copy()  
    # mask out boundary arithmetically if needed
    # this privileged loop func due to padding?
    # Let's implement with explicit checks
    if k == 0:
        # grad r0 = 1 - 2*h
        grad_rk = 1 - 2 * h
    else:
        # grad r_k is 1 - h_{i+k} - h_{i-k}
        # Replace in r_k : shift2 = h[i+k], shift2_rev = h[i-k]
        grad_rk = np.ones_like(h)
        # set h[i+k] where k>0, i is within bounds
        if k < n:
            grad_rk -= h_shift
        # now shift back: h[i-k] 
        if k < n:
            grad_rk -= h_shift_back
    return grad_rk

def inner_gradient_for_logsumexp(h, alpha):
    """Given shifting to compute logsumexp of [r_k(h)], compute
    gradient continuous values for gradient step using surrogate.
    """
    n = len(h)
    # compute all r_k = sum h_i * h_{i+k}
    # Alternatively: correlate with 1-h
    # much fast version: use FFT
    h_reversed = np.flip(h)
    h_fft = fft(h, 2 * n)
    ones_minus_h = 1 - h
    h_minus_fft = fft(ones_minus_h, 2 * n)
    # full correlation = conv(h, 1-h)
    conv_full = ifft(h_fft * h_minus_fft.conj()).real[:n]  # just first n
    r = conv_full  # raw terms r_k 
    # stabilize it with subtracting the maximum for numerical overflow
    r_max = np.max(r)
    if r_max > 1000:
        r_shift = r - r_max
    else:
        r_shift = r
    exp_r = np.exp(alpha * r_shift)
    sum_exp_r = np.sum(exp_r)
    w = exp_r / sum_exp_r
    # compute weighted sum of gradients for each r_k 
    grad = np.zeros_like(h)
    for k in range(n):
        g = get_gradient_for_shift(h, k, n)
        # Add weighted, scaled gradient
        grad += w[k] * g
    grad *= (2.0 / n)  # or let the dx scale
    return r, grad, w

def accelerated_projected_gradient_descent(h0, dx, w, alpha0=1.0, alpha_factor=2.0, alpha_max=500.0, max_iters=5000):
    """
    Perform accelerated projected gradient descent.
    Returns (h, c5) of final iterate.
    """
    h = h0.copy()
    t = 1.0
    alpha = alpha0
    tol = 1e-6
    prev_f = float('inf')
    for i in range(max_iters):
        r, grad, w_weights = inner_gradient_for_logsumexp(h, alpha)
        # compute surrogate:
        f_val = (2.0 / len(h)) * np.log(np.sum(np.exp(alpha * (r - np.max(r)))))  # smooth approx.
        if np.abs(f_val - prev_f) < tol:
            break
        prev_f = f_val
        # Nesterov-style extrapolation update
        y_prev = h.copy()
        eta = 0.5
        while True:
            h_tilde = project_onto_capped_simplex(h - eta * grad, w)
            # jitter gradient slightly with small noise
            r_tilde, grad_tilde, _ = inner_gradient_for_logsumexp(h_tilde, alpha)
            f_tilde = (2.0 / len(h)) * np.log(np.sum(np.exp(alpha * (r_tilde - np.max(r_tilde)))))
            # Armijo condition
            bound = f_val + 0.5 * eta * np.dot(grad, h_tilde - h)
            if f_tilde <= bound:
                break
            eta *= 0.5
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        h = h_tilde + ((t - 1) / t_new) * (h_tilde - y_prev)
        # update alpha if it's not too recent
        if i > 10 and alpha < alpha_max:
            alpha *= alpha_factor
        # boundary check
        h = np.clip(h, 0, 1)
    # final evaluation of true objective C5 after softening is done.
    final_c5 = evaluate_c5(h)
    return h, final_c5

            h[start_idx:end_idx] = 1.0
        else:
