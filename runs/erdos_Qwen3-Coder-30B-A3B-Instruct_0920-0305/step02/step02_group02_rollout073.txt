def project_onto_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex {x | 0 <= x_i <= 1, sum x_i = target_sum}."""
    v = np.asarray(v)
    n = len(v)
    # Two-phase projection:
    # 1. Sort descendingly to find the projection threshold
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]

    # Compute prefix sums
    prefix_sums = np.concatenate([[0], np.cumsum(sorted_v)])
    # Find the threshold (like water filling)
    i = np.searchsorted(prefix_sums[:-1] + sorted_v[1:] * np.arange(1, n+1), target_sum, side="right")

    if i == 0:
        # All entries should be 1 => but sum is n > target => impossible if target < n
        # Just threshold at 1.
        projection = np.clip(v, 0, 1)
        s = np.sum(projection)
        if abs(s - target_sum) > 1e-10:
            # Correct by adjusting largest components
            if s > target_sum:
                delta = s - target_sum
                # Reduce largest components to zero
                for j in range(n):
                    idx = sorted_indices[j]
                    if v[idx] > 0:
                        dv = min(v[idx], delta)
                        projection[idx] -= dv
                        delta -= dv
                        if delta <= 0:
                            break
            else:
                delta = target_sum - s
                for j in range(n):
                    idx = sorted_indices[j]
                    if v[idx] < 1:
                        dv = min(1 - v[idx], delta)
                        projection[idx] += dv
                        delta -= dv
                        if delta <= 0:
                            break
        return projection

    # i represents position of threshold, with i-1 top values ≥ threshold t,
    # and remaining values < t. We compute (i-1)th one and see where to split.
    # The effective upper bound is such that:
    #    (i-1) * (threshold) + sum of all others = target
    # i.e., threshold = (target - sum_others) / (i-1)

    # To generalise slightly:
    # project to sorted top i elements with value = some_threshold
    # projected[i:] = 0 or value < threshold, but set to 0 and redistribute
    # To simplify: reconsider. Simpler fix here.
    # Just compute directly how much mass can be between threshold rows (i half-ways)
    # and shift the elements.

    # Correct method:
    # Peel off enough to make sum of top i equal to target
    # If sum of top i is more than target, then average is modulated
    # of the i largest elements to sum to target.
    # Compute contributions and then fill.

    # G simpler, after all:
    # We mimic what we would do with average of going over total.
    # Use basic water filling:
    threshold = 1e15  # We'll compute properly.
    total = 0
    for k in range(n):
        if k == 0:
            total = sorted_v[k]
        else:
            total += sorted_v[k]
        if total >= target_sum:
            # We use exactly the top k entries, average it.
            # The k elements are:
            # sorted_v[0], ..., sorted_v[k-1]
            # desired sum = target_sum
            mean_val = target_sum / k if k > 0 else 0
            break
    else:
        # All entries are included (or only one element)
        k = n
        mean_val = target_sum / k if k > 0 else 0

    result = np.zeros(n)
    for j in range(k):
        result[sorted_indices[j]] = mean_val
    for j in range(k, n):
        result[sorted_indices[j]] = 0
    return result

def correlation_with_h_shift(h, shift):
    """Compute r(shift) = sum h[i] * (1 - h[i+shift])."""
    # Wrap at both ends using modulo
    # h extended with zeros for efficiency?
    # Or just do normal correlation
    folded_h = np.concatenate([h, np.zeros(shift)])
    c = np.correlate(folded_h, 1 - h, 'valid')
    # Subsample the index correctly
    return c[shift]

def compute_all_correlations(h, dx):
    # Import at top not allowed?
    n = len(h)
    r_vals = []
    for k in range(n):
        if k == 0:
            # r_0 = sum h_i * (1 - h_i)
            r_val = np.sum(h * (1 - h))
        else:
            # r_k = sum h_i * (1 - h_{i+k})
            r_val = 0
            for i in range(n):
                j = (i + k) % n
                r_val += h[i] * (1 - h[j])
        r_vals.append(r_val)
    # Note: This might be slow -- FFT version faster:
    return np.array(r_vals) * dx

def smooth_max_gradient(h, dx, alpha=1.0):
    """Compute the gradient of smooth max based on correlations."""
    # Simplified for now
    n = len(h)
    # Overestimate   (we'll still correct this later with full functional)
    r_vals = []
    for k in range(n):
        if k == 0:
            r_val = np.sum(h * (1 - h))
        else:
            r_val = 0
            for i in range(n):
                j = (i + k) % n
                r_val += h[i] * (1 - h[j])
        r_vals.append(r_val)

    r_vals = np.array(r_vals) * dx
    max_r = np.max(r_vals)

    # Subtract maximum for numerical stability
    r_shifted = r_vals - max_r

    # Softmax weight
    exp_r = np.exp(alpha * r_shifted)
    Z = np.sum(exp_r)
    weights = exp_r / Z

    # Gradient: weighted sum of individual correlation gradients
    grad = np.zeros(n)
    for k in range(n):
        for i in range(n):
            # Dependency of r_k on h[i]
            # r_k = sum h_j (1 - h_{j+k})
            # When j = i: h_i contributes (-h_i) to r_k
            # When j = i-k: h_{i-k} contributes (-h_{i-k}) to r_k
            # Add 1 for everywhere it was included:
            # we compute potential d(r_k)/d(h_i) = sum_{j} neighbors of i
            # in h_j (1 - h_{j+k})  and j = i only:
            g_ij = 1  # 1 for h_i term
            if (i + k) % n < n:
                g_ij += -1 * h[(i + k) % n]
            g_ij += 1 if (i - k) >= 0 else 0
            if (i - k) >= 0:
                g_ij += -1 * h[i - k]
            # Unit inverse norm-wise
            if k == 0:
                # d/dh_i (sum h_j (1 - h_j)) = 1 - 2 h_i
                grad[i] += weights[k] * (1 - 2 * h[i])
            else:
                # (-h_{i-k}) + 1 for the j={i-k} and h_i term gives its usual term
                if i-k < 0:
                    g_ij = 1  # Predominant part in anyway
                if i+k < n:
                    g_ij = 1 - h[i + k]  # from h_i * (1 - h_{i+k})
                else:
                    # we include based on roll
                    pass
                # Do not add weights correction manually. Easy to get it wrong.
                grad[i] += weights[k] * (
                    1 - h[(i + k) % n] - h[(i - k) % n]  # simple
                )
    return grad * dx

def log_sum_exp_subtract_max(x):
    # Compute log sum exp with numerical subtraction of max
    x_max = np.max(x)
    x_shifted = x - x_max
    return x_max + np.log(np.sum(np.exp(x_shifted)))

def evaluate_c5_surrogate(h_values, dx, alpha):
    # Create correlations and return its smooth maximum
    h = np.array(h_values)
    n = len(h)
    r_vals = []
    for k in range(n):
        if k == 0:
            r_val = np.sum(h * (1 - h))
        else:
            r_val = 0
            for i in range(n):
                j = (i + k) % n
                r_val += h[i] * (1 - h[j])
        r_vals.append(r_val)
    r_vals = np.array(r_vals) * dx
    # log sum exp
    return (dx / alpha) * log_sum_exp_subtract_max(alpha * r_vals)

def projected_grad_descent(h_start, dx, w, alpha0=1.0, alpha_factor=2.0, alpha_max=300.0, max_iter=5000):
    # Argmax t subject to total sum = w, 0 <= h <= 1, r_k(h) >= t for all k
    h = h_start.copy()
    n = len(h)
    h = project_onto_capped_simplex(h, w)
    t_current = -np.inf
    alpha = alpha0
    last_f = 1e10
    it = 0
    for it in range(max_iter):
        # Compute surrogate f_alpha(h)
        f_smooth = evaluate_c5_surrogate(h, dx, alpha)
        # Compute gradient wrt h of smooth function
        g = smooth_max_gradient(h, dx, alpha)

        # Gradient step, with backtracking line search
        eta = 0.5  # Step size
        x_old = h.copy()
        while True:
            # Try step with eta
            h_new = x_old - eta * g
            h_new = project_onto_capped_simplex(h_new, w)
            # Actual function evaluation (not smooth surrogate)
            # Adjacent tests are relatively expensive, so only at end.
            # But we must keep a very close approximation
            # for smooth termination condition.
            f_new = evaluate_c5_surrogate(h_new, dx, alpha)
            # Armijo backtracking
            if f_new >= f_smooth - 0.5 * eta * g.dot(h_new - x_old):
                # Accept
                h = h_new
                break
            else:
                eta *= 0.5  # Retry with smaller step size

        # Update dual variable method?
        # Instead keep alpha increasing, might not be required.
        if abs(f_smooth - last_f) < 1e-6:
            # Converged smoothly
            break
        if alpha < alpha_max:
            alpha *= alpha_factor

        last_f = f_smooth

    # Final evaluation
    r = compute_all_correlations(h, dx)
    c5_final = np.max(r)
    return h, c5_final
