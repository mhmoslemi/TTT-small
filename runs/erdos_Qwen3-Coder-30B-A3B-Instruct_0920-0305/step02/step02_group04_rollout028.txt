import numpy as np
from scipy.fft import fft, ifft

def evaluate_h(h):
    """Fast evaluation of C5 using FFT correlation"""
    n = len(h)
    dx = 2.0 / n
    h_pad = np.pad(h, (0, n), mode='constant')
    g_pad = np.pad(1 - h, (0, n), mode='constant')
    H = fft(h_pad)
    G = fft(g_pad)
    corr = ifft(H * np.conj(G)).real[:2*n-1]
    return np.max(corr) * dx

        cp.sum(h_var) == M,

            constraints.extend([
                p_var[i, ℓ] <= h_var[i],
                p_var[i, ℓ] <= h_var[i + ℓ],
                p_var[i, ℓ] >= h_var[i] + h_var[i + ℓ] - 1,
