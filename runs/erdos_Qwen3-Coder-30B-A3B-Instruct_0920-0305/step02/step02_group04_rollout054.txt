import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT"""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the binary MILP formulation for the Erdos problem"""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        # product linearisation via McCormick envelopes
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]

        # worst-lag constraint
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    solver_options = {'maxSeconds': time_limit}
    prob.solve(solver=cp.CBC, verbose=False, **solver_options)

    if prob.status in ['optimal', 'optimal_inaccurate']:
        T_opt = int(T.value)
        h_opt = np.ravel(h.value)
        return T_opt, h_opt
    elif prob.status in ['infeasible', 'unbounded']:
        return None, None
    else:
        # continue with whatever is available
        # if T has a value and h has a value
        try:
            T_val = int(T.value) if hasattr(T.value, '__len__') else int(T.value)
            h_val = np.ravel(h.value)
            return T_val, h_val
        except:
            return None, None

def smooth_descent(h_start, time_limit=5):
    """Refine h using smooth log-sum-exp descent"""
    n = len(h_start)
    M = n // 2
    dx = 2.0 / n
    h = h_start.copy()
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h) * dx
    h *= (1.0 / total)

    # weights for the smooth objective
    beta_schedule = [1, 2, 5, 10, 20]

    # helper for shifting
    def shift_forward(v, k):
        return np.concatenate([v[k:], np.zeros(k)])
    def shift_backward(v, k):
        return np.concatenate([np.zeros(k), v[:-k]])

    def compute_Ls(h):
        # compute all the L_ell values quickly using FFT
        Ls = []
        for ell in range(1, n):
            # mask to avoid out-of-bounds indices
            length = n - ell
            L_ell = np.sum(h[:length] * (1 - h[ell:]))
            Ls.append(L_ell)
        return np.array(Ls)

    remaining = time_limit

    # main inner loop
    for beta in beta_schedule:
        # to terminate early, we'll sample time
        iter_start = time.time()
        if time.time() - iter_start > remaining:
            break

        iter_count = 0
        eta = 0.1
        max_iter = 1000

        while iter_count < max_iter:
            # check time budget
            if time.time() - iter_start > remaining:
                break

            # compute all L_ell values
            Ls = compute_Ls(h)

            # compute exp(β * L_ell) with some numerical scaling
            Ls_scaled = beta * Ls
            max_L = np.max(Ls_scaled)
            weights = np.exp(Ls_scaled - max_L)
            Z = np.sum(weights)

            # compute weighted gradient
            grad = np.zeros(n)
            for ell in range(1, n):
                length = n - ell
                if ell <= n // 2:
                    forward = shift_forward(h, ell)
                    backward = shift_backward(h, ell)
                else:
                    forward = np.zeros(n)
                    backward = np.zeros(n)
                    forward[:n - ell] = h[ell:]
                    backward[n - ell:] = h[:-ell]
                grad[:length] += weights[ell - 1] * (1 - forward[:length] - backward[:length])
            
            # normalize weights for numerical stability
            weights /= Z
            
            # rescale by β to account for direction
            grad *= (1.0 / beta)
            
            # take gradient step (line search)
            prev_h = h.copy()
            prev_val = np.max(Ls)
            while True:
                h_new = h - eta * grad
                h_new = np.clip(h_new, 0, 1)
                # enforce sum constraint
                total_new = np.sum(h_new) * dx
                h_new *= (1.0 / total_new)
 
                Ls_new = compute_Ls(h_new)
                new_val = np.max(Ls_new)
                if new_val <= prev_val:
                    h = h_new
                    break
                eta /= 2.0
                if eta < 1e-10:
                    h = h_new
                    break
            
            iter_count += 1

    # final recomputation and return
    return h, evaluate_h(h)

def anneal_binary(h_start, max_iter=1000000, time_limit=20):
    """Swap-based simulated annealing for binary h"""
    n = len(h_start)
    M = n // 2
    dx = 2.0 / n
    h = h_start.copy()
    score = evaluate_h(h)
    best_h = h.copy()
    best_score = score

    # set initial temperature adaptively
    init_temp = 0.1
    temp = init_temp

    # keep track of how many evaluations have been performed
    iter_count = 0
    start_time = time.time()
    
    while iter_count < max_iter:
        if time.time() - start_time > time_limit:
            break
            
        # choose a 1 and a 0 at random
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # compute change in overlap
        old_val = h[i] * (1 - h[i]) + h[j] * (1 - h[j])
        
        h[i], h[j] = 0, 1
        new_score = evaluate_h(h)
        delta = new_score - score
        
        # accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0  # undo swap
        
        iter_count += 1
        # decrease temp
        temp *= 0.999

    return best_h, best_score
