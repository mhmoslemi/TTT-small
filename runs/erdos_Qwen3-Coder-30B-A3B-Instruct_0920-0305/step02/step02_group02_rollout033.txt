import numpy as np
from scipy.optimize import minimize
from scipy.fft import ifft, fft
import time

def project_onto_capped_simplex(v, target_sum):
    """Project v onto {x : 0 <= x_i <= 1, sum x_i = target_sum}"""
    v = np.asarray(v)
    v = np.clip(v, 0, 1)
    s = np.sum(v)
    if np.isclose(s, target_sum):
        return v
    # Use water-filling algorithm
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    n = len(v_sorted)
    # Find threshold
    threshold = 0.0
    for i in range(n):
        if cumsum[i] + (n - i - 1) < target_sum:
            continue
        else:
            threshold = (target_sum - cumsum[i]) / (n - i - 1)
            break
    # Reconstruct
    result = np.clip(v_sorted - threshold, 0, 1)
    # Adjust to ensure exact sum
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-12:
        idx = np.argmax(result)
        result[idx] += residual
    return np.array([v[np.argwhere(v == x)[0][0]] if x in v else 0 for x in result])

def correlation(h, k):
    """Compute correlation S_k = sum h_i h_{i+k} for shift k"""
    n = len(h)
    if k == 0:
        return np.dot(h, h)
    elif k > 0:
        return np.sum(h[k:] * h[:-k])
    else:
        return np.sum(h[:k] * h[-k:])

def evaluate_c5(h):
    """Compute C5 with direct method"""
    n = len(h)
    dx = 2.0 / n
    corrs = [correlation(h, k) for k in range(n)]
    # r_k = w - S_k, where w = sum h * dx = 1.0
    r_vals = [1.0 - s for s in corrs]
    return dx * max(r_vals)

def evaluate_c5_smooth(h, alpha=100.0):
    """Evaluate smooth surrogate of max_k r_k"""
    n = len(h)
    dx = 2.0 / n
    # Compute all r_k values
    corrs = [correlation(h, k) for k in range(n)]
    r_vals = [1.0 - s for s in corrs]
    max_r = max(r_vals)
    shifted = [r - max_r for r in r_vals]
    exps = np.exp(alpha * np.array(shifted))
    log_sum_exp = np.log(np.sum(exps)) + max_r
    return dx * log_sum_exp

def compute_grad_r(h, k):
    """Gradient of r_k = 1 - sum h_i h_{i+k} w.r.t h_i"""
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        if i + k < n:
            grad[i] += -h[i + k]
        if i - k >= 0:
            grad[i] += -h[i - k]
        grad[i] += 1.0
    return grad

def gradient_of_smooth_max(h, alpha=100.0):
    """Gradient of the log-sum-exp approximation to max_k r_k"""
    n = len(h)
    dx = 2.0 / n
    corrs = [correlation(h, k) for k in range(n)]
    r_vals = [1.0 - s for s in corrs]
    max_r = max(r_vals)
    shifted = [r - max_r for r in r_vals]
    exps = np.exp(alpha * np.array(shifted))
    weights = exps / np.sum(exps)
    grad = np.zeros_like(h)
    for k in range(n):
        g = compute_grad_r(h, k)
        grad += weights[k] * g
    return dx * grad

def backtracking_line_search(h, grad, step_size, alpha=100.0, c=0.5, tau=0.5):
    """Simple backtracking line search for sufficient decrease"""
    f_prev = evaluate_c5_smooth(h, alpha)
    while True:
        h_new = project_onto_capped_simplex(h - step_size * grad, 1.0)
        f_new = evaluate_c5_smooth(h_new, alpha)
        if f_new <= f_prev - c * step_size * np.dot(grad, grad):
            return h_new, step_size
        step_size *= tau

def accelerated_projected_gradient(h0, dx, target_sum, alpha0=1.0, alpha_factor=2, alpha_max=300.0, max_iters=5000):
    """Accelerated gradient descent with continuation on alpha"""
    h = h0.copy()
    t = 1.0
    alpha = alpha0
    step_size = 0.5
    last_f = np.inf
    
    # warm-start
    for _ in range(max_iters):
        if alpha > alpha_max:
            break
        r_vals = [correlation(h, k) for k in range(len(h))]
        r_vals = [1.0 - s for s in r_vals]
        f = evaluate_c5_smooth(h, alpha)
        if abs(f - last_f) / (1e-8 + abs(last_f)) < 1e-6:
            break
        last_f = f
        
        g = gradient_of_smooth_max(h, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t**2)) / 2.0
        h_tilde = project_onto_capped_simplex(h - step_size * g, target_sum)
        h = h_tilde + ((t - 1.0) / t_new) * (h_tilde - h)
        h = project_onto_capped_simplex(h, target_sum)
        t = t_new
        
        # backtracking line search
        h_new, step_size = backtracking_line_search(h, g, step_size, alpha)
        h = h_new

        if abs(f - last_f) / (1e-8 + abs(last_f)) < 1e-6:
            break

    return h, evaluate_c5(h)

def reinitialise_h(n, method='binary'):
    """Generate restart starting configuration"""
    if method == 'uniform':
        h = np.random.uniform(0, 1, n)
        return project_onto_capped_simplex(h, 1.0)
    elif method == 'binary':
        h = np.zeros(n)
        num_ones = int(np.round(n / 2.0))
        h[:num_ones] = 1
        np.random.shuffle(h)
        return project_onto_capped_simplex(h, 1.0)
    elif method == 'three_block':
        a = np.random.uniform(0, 1)
        b = np.random.uniform(a, 2)
        h = np.zeros(n)
        mid = int(np.round(a * n / 2))
        end = int(np.round(b * n / 2))
        h[mid:end] = 1.0
        h = project_onto_capped_simplex(h, 1.0)
        return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_values = [48, 56, 64, 80]
    best_n = 0
    
    for n in n_values:
        dx = 2.0 / n
        w = 1.0  # integral = 1
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget < 1: break
        
        # allocate small portion of time to this n
        time_per_n = remaining_budget / len(n_values)
        start_n = time.time()
        
        # multiple restarts for each n
        restarts = 0
        while time.time() - start_n < time_per_n:
            restarts += 1
            for init_method in ['uniform', 'binary', 'three_block']:
                h = reinitialise_h(n, init_method)
                try:
                    h_opt, c5_opt = accelerated_projected_gradient(h, dx, w, max_iters=2000)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        best_n = n
                except Exception:
                    pass

    # Final refinement with trust-constr
    if best_h is not None:
        try:
            def obj(h):
                return evaluate_c5(h)
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * (2.0 / len(h)) - 1.0}
            bounds = [(0, 1)] * len(best_h)
            res = minimize(obj, best_h, method='trust-constr', bounds=bounds, constraints=cons, options={'maxiter': 100})
            if res.success and res.fun < best_c5:
                best_h = res.x
                best_c5 = res.fun
        except Exception:
            pass
    
    # Final projection to ensure constraints
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, 1.0)
        final_c5 = evaluate_c5(best_h)
        return list(best_h), final_c5, len(best_h)
    else:
        return list(np.ones(56) / 56), 0.5, 56
