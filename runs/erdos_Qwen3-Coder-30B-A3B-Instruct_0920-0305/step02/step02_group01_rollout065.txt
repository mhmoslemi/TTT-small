import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def build_five_step(a, b, n):
    """Build a 5-step function with parameters a, b"""
    dx = 2.0 / n
    x = np.linspace(0.5*dx, 2-dx/2, n) # sample points
    h = np.zeros(n)
    u = (2*b - 1)/(2*(b-a))
    if not (0 <= u <= 1):
        return None

    intervals = [(0, a), (a, b), (b, 2-b), (2-b, 2-a), (2-a, 2)]
    heights = [0, u, 1, u, 0]
    for i in range(5):
        start, end = intervals[i]
        height = heights[i]
        indices = np.where((x >= start) & (x < end))[0]
        h[indices] = height

    # Correct integral to exactly 1
    integral = np.sum(h) * dx
    h = h * (1.0 / integral)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0<=h_i<=1, sum(h)=target_sum"""
    # Objective is to find lambda such that projection_sum = target_sum
    def proj_sum(lam):
        return np.sum(np.clip(h - lam, 0, 1))
    
    # Bisection for lambda
    low, high = -1000, 1000
    for _ in range(40):
        mid = (low + high) / 2
        if proj_sum(mid) > target_sum:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    proj_h = np.clip(h - lam, 0, 1)
    # Renormalize
    proj_h *= target_sum / np.sum(proj_h)
    return proj_h

def smooth_max(c, alpha=1.0):
    """Compute a smooth approximation to max function with temperature alpha"""
    exp_c = np.exp(alpha * c)
    w = exp_c / np.sum(exp_c)
    return w

def compute_gradient(h, w):
    """Compute gradient with given weights w"""
    n = len(h)
    # For each i, grad_i = 1 - sum_s w_s * (h_{i+s} + h_{i-s})
    # Do this using 2 convolutions:
    g_plus = np.convolve(w[::-1], h, mode='full')[n-1:n-1+n]
    g_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]
    grad = 1.0 - (g_plus + g_minus)
    return grad

def smooth_projected_gradient_descent(h0, n, max_iter=5000):
    """Perform smoothed projected gradient descent"""
    h = h0.copy()
    target_sum = n / 2
    alpha = 0.5
    eta = 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iter):
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        # Use a soft maximum with increasing alpha
        w = smooth_max(corr, alpha=alpha)
        
        # Compute gradient
        grad = compute_gradient(h, w)
        
        h_new = h - eta * grad
        
        # Project back to feasibility
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        h = h_new
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adaptive learning rate and alpha
        if it % 200 == 0:
            eta *= 0.9
            alpha = min(alpha * 1.5, 20)
            
        # Early stopping on no improvement
        if it % 500 == 0:
            if evaluate_c5(h) >= best_c5:
                break
                
    return best_h, best_c5

def diff_evolution(global_best, n, max_iter=200):
    """Tiny differential evolution fallback"""
    target_sum = n / 2
    pop_size = 8
    # Initialize population
    pop = []
    for _ in range(pop_size):
        x = np.random.rand(n)
        x = project_to_capped_simplex(x, target_sum)
        pop.append(x)
    
    best_h = global_best.copy()
    best_c5 = evaluate_c5(best_h)
    
    for gen in range(max_iter):
        for i in range(pop_size):
            # Select three different individuals
            idxs = [j for j in range(pop_size) if j != i]
            a, b, c = [pop[j] for j in np.random.choice(idxs, 3, replace=False)]
            F = 0.6
            CR = 0.9
            v = a + F * (b - c)
            # Boundary check
            v = np.clip(v, 0, 1)
            # Crossover
            j_rand = np.random.randint(n)
            u = np.array([v[j] if np.random.rand() < CR or j == j_rand else pop[i][j] for j in range(n)])
            # Project
            u = project_to_capped_simplex(u, target_sum)
            # Selection
            if evaluate_c5(u) < evaluate_c5(pop[i]):
                pop[i] = u
                if evaluate_c5(u) < best_c5:
                    best_h = u.copy()
                    best_c5 = evaluate_c5(u)
                    
    return best_h, best_c5
