import numpy as np
from scipy.optimize import minimize
from scipy.linalg import eigh

import time

# Constants
n_list = [48, 56, 64, 80]
target_bound = 0.3807
budget_s = 600.0
max_rounding_trials = 20
max_fourier_modes = 10
max_iter_refine = 500

def evaluate_overlap(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def solve_sdp(n):
    dx = 2.0 / n
    w = n / 2.0
    X = Semidef(n)
    h = Variable(n)
    constraints = [
        trace(X) == w,
        h >= 0,
        h <= 1,
        sum(h) == w
    ]
    
    # Add diagonal constraints
    for i in range(n):
        constraints.append(X[i,i] == h[i]**2)
    
    # Add Toeplitz constraints to model the correlations
    T_k_list = []
    for k in range(n):
        T_k = np.zeros((n, n))
        if k == 0:
            T_k = np.eye(n)
        else:
            T_k[np.arange(n-k), np.arange(k, n)] = 1
            T_k[np.arange(k, n), np.arange(n-k)] = 1
        T_k_list.append(T_k)
    
    # Assume first k=0 to max: full system is large 
    # For large n, approximate with a sample
    max_k = min(20, n//2)
    
    # Take the constraint for first few shifts
    for k_idx in range(max_k):
        T_k = T_k_list[k_idx]
        Tr = cp.trace(T_k @ X)
        constraints.append(Tr >= 0.99)  # Placeholder
    
    objective = cp.Maximize(0)

    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.array(h.value).flatten()
            return h_val
    except:
        return None
    return None

def project_to_simplex(y, w):
    # Water-filling to project y onto {x | 0 <= x <= 1, sum(x) = w}
    y = np.copy(y)
    n = len(y)
    y_sorted = np.sort(y)[::-1]
    v = y_sorted
    tmp_sum = np.cumsum(v)
    tmp_sum /= np.arange(1, n + 1)
    idx = np.argmax(tmp_sum > (w - np.sum(v) + tmp_sum) / np.arange(1, n + 1))
    threshold = tmp_sum[idx]
    x = np.maximum(y - threshold, 0)
    x = np.minimum(x, 1)
    # Adjust to match w
    total = np.sum(x)
    if abs(total - w) > 1e-10:
        ratio = w / total
        x *= ratio
        x = np.clip(x, 0, 1)
    return x

def fourier_basis(n, m):
    # Generate cos/sin basis functions on [0,1] domain (discrete)
    x = np.linspace(0, 1, n)
    basis = []
    basis.append(np.ones(n))  # Constant term
    for i in range(1, m+1):
        basis.append(np.cos(np.pi * i * x))  # cos terms
        basis.append(np.sin(np.pi * i * x))  # sin terms
    return np.array(basis).T

def fit_fourier_coeffs(h_data, max_modes=10):
    # Fit with least squares
    n = len(h_data)
    basis = fourier_basis(n, max_modes)
    # Solve for a = basis.T @ h_data (possibly with some regularization or box constraints)
    # For now, just solve normally with box constraints
    def objective(c):
        h_rec = basis @ c + 0.5 * np.ones(n)
        return evaluate_overlap(h_rec)
    
    W = basis.T
    try:
        # Use a constrained LS solver or trust region
        # We use one-step approximation of solution for simplicity:
        # The pseudo inverse
        h_center = h_data - 0.5
        c = np.linalg.lstsq(W, h_center, rcond=None)[0]
        # Constrain to feasible range
        h_rec = basis @ c + 0.5
        h_rec = np.clip(h_rec, 0, 1)
        return c, evaluate_overlap(h_rec)
    except:
        return None, None

def get_spectrum(h):
    n = len(h)
    dx = 2.0 / n
    h_fft = np.fft.fft(h)
    h_conj = np.conj(h_fft)
    cross = h_fft * h_conj
    # compute the first n-1 non-zero elements of spectrum
    return np.abs(cross[:n//2+1])

def local_refinement(initial_h, max_iter=100):
    # Use a low dimensional smoother to get a better approximation
    n = len(initial_h)
    dx = 2.0 / n
    # Fit with cosine/sine components
    M = min(max_fourier_modes, n//4)
    if M < 1:
        M = 1
    basis = fourier_basis(n, M)
    # project to the feasible region
    try:
        # Use direct optimization on coefficients
        def objective(c):
            h_rec = basis @ c + 0.5 * np.ones(n)
            h_rec = np.clip(h_rec, 0, 1)
            # Ensure integral constraint
            total = np.sum(h_rec)
            h_rec *= 1.0 * (n / 2.0) / total
            h_rec = np.clip(h_rec, 0, 1)
            return evaluate_overlap(h_rec)
        
        c_init = np.random.normal(0, 0.1, 2*M)
        res = minimize(objective, c_init, method='L-BFGS-B', bounds=[(-5,5)] * (2*M))
        if res.success:
            h_final = basis @ res.x + 0.5 * np.ones(n)
            h_final = np.clip(h_final, 0, 1)
            # Re-scale
            total = np.sum(h_final)
            if abs(total - n/2.0) > 1e-10:
                h_final *= (n / 2.0) / total
            h_final = np.clip(h_final, 0, 1)
            return h_final, evaluate_overlap(h_final)
        else:
            return initial_h, evaluate_overlap(initial_h)
    except Exception as e:
        return initial_h, evaluate_overlap(initial_h)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    for n in n_list:
        if time.time() - start_time > 0.95 * budget_s:
            break
        
        h_sdp = None
        h_sdp = solve_sdp(n)
        if h_sdp is not None:
            # round the solution and process
            h_rounded = np.clip(h_sdp, 0, 1)
            h_rounded = project_to_simplex(h_rounded, n / 2.0)
            
            # Fit Fourier expansion
            c_fit, c5_fit = fit_fourier_coeffs(h_rounded, max_modes=6)
            if c_fit is not None:
                try:
                    best_h_local, c5_opt = local_refinement(h_rounded, max_iter=500)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = best_h_local
                        best_n = n
                    if best_c5 <= target_bound:
                        break
                except Exception:
                    continue
        
        # Optional fallback: random initialisation if SDP fails or is not stable
        trials = min(max_rounding_trials, 10 if n <= 50 else 2)
        for _ in range(trials):
            if time.time() - start_time > 0.95 * budget_s:
                break
            h_temp = np.random.rand(n)
            h_temp = project_to_simplex(h_temp, n / 2.0)
            h_temp = np.clip(h_temp, 0, 1)
            try:
                h_refined, c5_ref = local_refinement(h_temp, max_iter=100)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_refined
                    best_n = n
            except Exception:
                continue
        
        if best_c5 <= target_bound:
            break
        
        # Even skip search for very large n if no progress
        
        if n >= 64 and best_c5 > 0.381:
            continue  # Skip


    if best_h is None:
        # Default case (just use a simple construction)
        h = np.ones(48) * 0.5
        h[10:38] = 1.0
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h *= (48 / 2.0) / total
        h = np.clip(h, 0, 1)
        best_h = h
        best_c5 = evaluate_overlap(best_h)
        best_n = 48

    if time.time() - start_time > budget_s * 0.95:
        # Late stop, return best found
        pass

    return list(best_h), best_c5, best_n
