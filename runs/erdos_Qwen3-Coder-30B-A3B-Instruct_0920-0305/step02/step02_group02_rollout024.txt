def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * 2.0 / len(h)
    return c5

def project_to_capped_simplex(v, w):
    """Project vector v onto capped simplex: 0 <= v_i <= 1, sum v_i = w."""
    n = len(v)
    if w <= 0:
        return np.zeros(n)
    if w >= n:
        return np.ones(n)
    
    # Sort descending (reverse order) the values
    indices_sorted = np.argsort(v)[::-1]
    sorted_v = v[indices_sorted]
    
    # Find the threshold tau
    tau = 0.0
    cumsum = 0.0
    for i in range(n):
        cumsum += sorted_v[i]
        # Compare cumsum with w + (i+1) * sorted_v[i]
        if i + 1 < n:
            if cumsum <= w + (i+1) * sorted_v[i]:
                continue
            else:
                tau = (cumsum - w) / (i+1)
                break
        else:
            tau = (cumsum - w) / (i+1)
            break
    # Apply threshold to sorted values
    new_values = np.zeros_like(sorted_v)
    for i in range(n):
        new_values[i] = max(min(sorted_v[i] - tau, 1.0), 0.0)
    # Apply back to original indices
    result = v.copy()
    for i in range(n):
        result[indices_sorted[i]] = new_values[i]
    return result

def smooth_max_logsumexp(r_values, alpha):
    """Compute smooth approximation to max of r_values with parameter alpha."""
    r_values = np.array(r_values)
    max_r = np.max(r_values)
    shifted = r_values - max_r
    exp_terms = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_terms)
    if sum_exp == 0:
        return 0.0
    log_sum = np.log(sum_exp)
    return (log_sum / alpha) + max_r

def compute_r_shifts(h, w):
    """Compute r_k = w - S_k(h) efficiently using FFT."""
    n = len(h)
    # Zero-pad for full correlation (using length 2*n)
    h_padded = np.concatenate([h, np.zeros(n)])
    h_fft = fft(h_padded)
    sq_fft = h_fft * np.conj(h_fft)
    r_corr = ifft(sq_fft).real
    # Extract non-negative lags only
    r_k = r_corr[n-1 : 2*n-1]
    assert len(r_k) == n
    return w - r_k

def gradient_r_k(h, k, w):
    """Gradient of r_k = w - Σ h_i * h_{i+k} w.r.t h_j."""
    # ∂r_k/∂h_j = - h_{j+k} if j+k < n and j+k >= 0
    #              - h_{j-k} if j-k >= 0
    #              1 if j = 0
    n = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    else:
        # k >= 1; create gradient vector
        grad = np.zeros(n)
        for i in range(n):
            # Add term -h_{i+k} if valid
            if i + k < n:
                grad[i] -= h[i + k]
            # Add term -h_{i-k} if valid
            if i - k >= 0:
                grad[i] -= h[i - k]
        return grad

def backtracking_line_search(h, g, step_size=1.0, alpha=0.5, beta=0.9):
    """Backtracking to find step size that satisfies Armijo condition."""
    # Armijo condition: f(x - α g) ≤ f(x) - α c1 * g^T * g
    # Let's use a simple fallback
    return step_size * beta

def smooth_objective_gradient(h, w, alpha, smooth_alpha):
    """Return the smooth objective gradient (using log-sum-exp version)."""
    n = len(h)
    # Compute all r_k values
    r_k = compute_r_shifts(h, w)
    # Smooth version
    r_smooth = smooth_max_logsumexp(r_k, smooth_alpha)
    # Compute weights from soft-max
    max_r = np.max(r_k)
    shifted_r = r_k - max_r
    exp_r = np.exp(alpha * shifted_r)
    sum_exp_r = np.sum(exp_r)
    weights = exp_r / sum_exp_r
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(n):
        g_k = gradient_r_k(h, k, w)
        grad += weights[k] * g_k
    return grad

def accelerated_gradient_with_continuation(initial_h_values, w, n, dx, alpha_start=1., alpha_max=500., alpha_factor=2., eps_converge=1e-6):
    """Use FISTA like accelerated gradient with continuation."""
    # Initialize FISTA variables
    h = initial_h_values.copy()
    y = h.copy()
    t_current = 1.0
    step_size = 0.25  # initial learning rate
    smooth_alpha = alpha_start
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    iterations = 0
    
    while smooth_alpha <= alpha_max:
        # Solve for current alpha
        for _ in range(1000000):  # Avoid infinite loop
            # Store previous h and objective
            h_prev = h.copy()
            # Compute gradient of surrogate
            gradient = smooth_objective_gradient(h, w, smooth_alpha, smooth_alpha)
            # Compute h_tilde = project(y - step_size * gradient)
            h_tilde = y - step_size * gradient
            h_tilde = project_to_capped_simplex(h_tilde, w)
            # Compute t_{k+1}, y_{k+1}
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_current**2)) / 2.0
            y = h_tilde + ((t_current - 1.0) / t_new) * (h_tilde - h)
            h = h_tilde.copy()
            t_current = t_new
            
            # Check convergence
            if np.linalg.norm(h - h_prev) / (np.linalg.norm(h_prev) + 1e-8) < eps_converge:
                break
            iterations += 1
            if iterations > 2000:
                break
        # Compute final objective for current alpha
        h_new = h.copy()
        new_c5 = evaluate_h(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        # Increase alpha
        smooth_alpha *= alpha_factor
        # Warm start
        y = h.copy()
        t_current = 1.0
    return best_h, best_c5

def run_hill_climb(seed, h0, w, n, dx):
    """Run hill climb to refine the solution."""
    np.random.seed(seed)
    h = h0.copy()
    # Make a few passes of perturbation and proj
    for _ in range(100):
        i = np.random.randint(0, n)
        old_val = h[i]
        new_val = old_val + np.random.normal(0, 0.02)
        h[i] = new_val
        h = project_to_capped_simplex(h, w)
        # Re-evaluate
        new_c5 = evaluate_h(h)
        # Improve the solution in case of progress
    return h.copy(), evaluate_h(h)

def run(seed=42, budget_s=600):
    """Main optimization function."""
    start_time = time.time()
    np.random.seed(seed)
    best_bound = 1000.0
    best_h = None
    best_n = None

    # Try each value of n in turn with fair allocation of time
    time_per_n = budget_s / len(n_values)

    for idx, n in enumerate(n_values):
        dx = dx_values[idx]
        w = w_values[idx]
        remaining_time = budget_s - (time.time() - start_time)
        time_for_n = min(time_per_n, remaining_time * 0.9)
        n_start_time = time.time()
        
        # Try multiple random restarts
        num_restarts = 5
        for k in range(num_restarts):
            if time.time() - n_start_time > time_for_n:
                break
            try:
                # Strategy 1: initial_h_values + small random perturbation
                h_start = initial_h_values[:n].copy()
                if len(h_start) < n:
                    h_start = np.pad(h_start, (0, n - len(h_start)), 'constant', constant_values=(0, 0))
                if len(h_start) > n:
                    h_start = h_start[:n]
                h_start += np.random.normal(0, 0.01, n)
                h_start = np.clip(h_start, 0, 1)
                h_start = project_to_capped_simplex(h_start, w)

                # Run with smoother + continuation
                h_opt, c5_opt = accelerated_gradient_with_continuation(h_start, w, n, dx)
                if k == 0:
                    h_restart, c5_restart = run_hill_climb(seed + 1000 + k, h_opt, w, n, dx)
                    if c5_restart < c5_opt:
                        h_opt, c5_opt = h_restart, c5_restart
                # Final refinement with trust-region solver
                bounds = [(0, 1) for _ in range(n)]
                def obj(h):
                    return -evaluate_h(h)
                result = minimize(obj, h_opt, method='SLSQP', bounds=bounds, options={'maxiter': 100, 'ftol': 1e-10})
                if result.success:
                    h_final, c5_final = result.x, -result.fun
                    if c5_final < c5_opt:
                        h_opt, c5_opt = h_final, c5_final
                # Update best if needed
                if c5_opt < best_bound:
                    best_bound = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
            except Exception:
                continue
            if time.time() - start_time > 0.95 * budget_s:
                break

    # Final evaluation with official function
    if best_h is None:
        best_h = initial_h_values[:48]
        best_h = np.pad(best_h, (0, 48 - len(best_h)), 'constant', constant_values=(0, 0))
        best_n = 48
        best_bound = evaluate_h(best_h)
    else:
        h_eval = np.array(best_h)
        h_eval = np.clip(h_eval, 0, 1)
        h_eval = project_to_capped_simplex(h_eval, w_values[n_values.index(best_n)])
        best_bound = evaluate_h(h_eval)
    return list(h_eval), best_bound, best_n
