import numpy as np
from scipy.optimize import minimize
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto bounds [0,1] and sum constraint"""
    lo = 0.0
    hi = 1.0
    # First clip to bounds
    w = np.clip(v, lo, hi)
    # Compute sum
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    # We'll iteratively adjust
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                # cannot increase more, break
                break
            delta = diff / n_free
            w[free] += delta
            # Clip again
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h):
    """Compute the maximum overlap for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    argmax_idx = np.argmax(conv)
    # Compute shift corresponding to argmax (shift range is [-n+1, n-1])
    shift_m = argmax_idx - (n_points - 1)
    return max_val, shift_m

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a given shift m"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx factor
