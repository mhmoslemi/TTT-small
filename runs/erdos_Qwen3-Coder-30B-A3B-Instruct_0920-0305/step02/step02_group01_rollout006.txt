import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate

def project_to_capped_simplex(y, target_sum):
    """Project vector y onto the capped simplex: 0 ≤ x_i ≤ 1, sum x_i = target_sum"""
    y = np.asarray(y)
    n = len(y)
    # Bisection to find lambda such that sum(clamp(y - lambda, 0, 1)) == target_sum
    lambda_low = -np.max(y) - 1.0
    lambda_high = np.max(y) + 1.0
    tol = 1e-12
    for _ in range(40):
        lambda_mid = (lambda_low + lambda_high) / 2
        x = np.clip(y - lambda_mid, 0, 1)
        sum_x = np.sum(x)
        if sum_x > target_sum:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
    lambda_star = (lambda_low + lambda_high) / 2
    x = np.clip(y - lambda_star, 0, 1)
    scale = target_sum / np.sum(x)
    x = x * scale
    return x

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Early time check
    if time.time() - start_time > budget_s * 0.95:
        return initial_h_values, evaluate_c5(initial_h_values), len(initial_h_values)

    # Target resolution for final output
    n_target = 180
    target_sum = n_target / 2

    best_c5 = float('inf')
    best_h = np.array(initial_h_values)

    # Step 1: Solve SDP for several small n
    n_values = [30, 35, 40, 45, 50]
    sdps_solved = 0

    for n in n_values:
        if time.time() - start_time > budget_s * 0.7:
            break
        try:
            # Create CVXPY variables
            X = Semidefinite(n)
            h = Variable(n)
            m = Variable()

            # Constraints
            constraints = [
                X >> 0,                      # X is positive semidefinite
                [X[i, i] == h[i] for i in range(n)],  # Diagonal of X = h
                h >= 0,                      # Box constraints
                h <= 1,
                sum(h) == target_sum         # Integral constraint for n points
            ]

            # Autocorrelation constraints for all shifts s
            # But only add a few representative ones if SDP too heavy
            # For SDP, sample up to min(10, n) shifts (s=0 to n-1) for constraints
            max_shift = min(10, n)
            for s in range(max_shift):
                if s == 0: continue
                r_s = sum(X[i, i+s] for i in range(n-s))  # Positive shift
                constraints.append(r_s >= m)
                # Also add negative shift: X[i+s, i] (we'll handle this differently)
                r_neg = sum(X[i+s, i] for i in range(n-s))  # Negative shift
                constraints.append(r_neg >= m)

            # Objective
            prob = Problem(Maximize(m), constraints)
            prob.solve(solver=ECOS, tol=1e-5, max_iters=2000)
            m_opt = m.value
            X_opt = X.value
            h_opt = h.value.copy()
            
            # Extract projection
            h_prox = np.abs(np.linalg.eigh(X_opt)[1][:, -1])
            scale = target_sum / np.sum(h_prox)
            h_prox = h_prox * scale
            h_prox = project_to_capped_simplex(h_prox, target_sum)
            
            # Evaluate C5
            c5 = evaluate_c5(h_prox)
            if c5 < best_c5:
                best_h = h_prox.copy()
                best_c5 = c5
                
            sdps_solved += 1
        except Exception as e:
            continue

    # Step 2: Local refinement with pattern search (Nelder-Mead)
    if time.time() - start_time < budget_s * 0.9:
        try:
            from scipy.optimize import minimize
            bounds = [(0, 1) for _ in range(len(best_h))]
            
            def objective(x):
                return evaluate_c5(x)
                
            def constraint_eq(x):
                return np.sum(x) * 2.0 / len(x) - 1.0  # Normalized to 1
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Use a few final SLSQP local iterations
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'ftol': 1e-10, 'gtol': 1e-10}, maxiter=200)
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    # Step 3: Upsample to target n
    if time.time() - start_time < budget_s * 0.98:
        try:
            # Interpolate lowest dimensional solution to n_target
            h_low_dim = best_h
            h_upsample = np.interp(np.linspace(0, 2, n_target), np.linspace(0, 2, len(h_low_dim)), h_low_dim)
            h_upsample = project_to_capped_simplex(h_upsample, target_sum)
            c5_final = evaluate_c5(h_upsample)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_upsample
        except Exception:
            pass

    return (best_h.tolist(), float(best_c5), n_target)
