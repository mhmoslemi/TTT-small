import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT"""
    n = len(h)
    dx = 2.0 / n
    H = rfft(h, 2*n)
    G = rfft(1.0 - h, 2*n)
    conv = irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the binary MILP for the Erdos problem using CVXPY"""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        # McCormick envelopes for h_i * h_{i+ell}
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]

        # Constraint: sum of products + T >= M
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
    
    if prob.status in ["optimal", "optimal_inaccurate", "feasible"]:
        h_opt = np.ravel(h.value)
        T_opt = int(T.value)
        return T_opt, h_opt
    else:
        return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=30):
    """Binary simulated annealing"""
    n = len(h_start)
    M = n // 2
    h = h_start.copy().astype(float)
    current_score = evaluate_h(h)
    best_h = h.copy()
    best_score = current_score
    
    start_time = time.time()
    T = 1.0

    # Estimate initial temperature from first 500 moves
    sum_delta = 0.0
    count = 0
    for _ in range(500):
        if time.time() - start_time > time_limit * 0.5:
            break
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 1 and h[j] == 0:
            h[i], h[j] = 0, 1
            new_score = evaluate_h(h)
            delta = new_score - current_score
            sum_delta += delta
            count += 1
            if delta > 0:
                h[i], h[j] = 1, 0  # revert
            else:
                current_score = new_score
    if count > 0:
        T = max(1e-6, -sum_delta / count) * 3.0

    # Main annealing loop
    iter_count = 0
    while time.time() - start_time < time_limit * 0.8 and iter_count < max_iter:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if h[i] == 1 and h[j] == 0:
            h[i], h[j] = 0, 1
            new_score = evaluate_h(h)
            delta = new_score - current_score
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()
            else:
                h[i], h[j] = 1, 0  # revert

        T *= 0.999
        iter_count += 1
        if T < 1e-6:
            break

    return best_h, best_score

def smooth_descent(h_bin, time_limit=20):
    """Smoothed gradient descent with log-sum-exp surrogate"""
    n = len(h_bin)
    M = n // 2
    dx = 2.0 / n
    h = h_bin.copy().astype(float)
    T_start = time.time()
    
    # Define A: to compute shifted terms
    def shift_forward(h, k):
        return np.concatenate([h[k:], np.zeros(k)])
    def shift_backward(h, k):
        return np.concatenate([np.zeros(k), h[:-k]])
    
    # Gradient of L_ell with respect to h_i
    def grad_L_ell(h, ell):
        return 1.0 - shift_forward(h, ell) - shift_backward(h, ell)
    
    # Log-sum-exp surrogate
    def F_beta(h, beta):
        all_L = []
        for ell in range(1, n):
            L_ell = np.sum(h * (1.0 - shift_forward(h, ell)))
            all_L.append(L_ell)
        all_L = np.array(all_L)
        if np.any(np.isnan(all_L)):
            return np.inf
        weights = np.exp(beta * all_L)
        weights /= np.sum(weights)
        return (1.0 / beta) * np.log(np.sum(weights))
    
    # Gradient of log-sum-exp
    def grad_F_beta(h, beta):
        grad_total = np.zeros_like(h)
        all_L = []
        all_grads = []
        for ell in range(1, n):
            L_ell = np.sum(h * (1.0 - shift_forward(h, ell)))
            all_L.append(L_ell)
            g_ell = grad_L_ell(h, ell)
            all_grads.append(g_ell)
        all_L = np.array(all_L)
        weights = np.exp(beta * all_L)
        weights /= np.sum(weights)
        for ell in range(len(all_L)):
            grad_total += weights[ell] * all_grads[ell]
        return grad_total
    
    # Water-filling projection
    def project_on_simplex(h, target_sum):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-6:
            return h
        # Binary search for lambda to make sum = target_sum
        low, high = -10.0, 10.0
        for _ in range(30):
            lam = (low + high) / 2.0
            new_h = np.clip(h + lam, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if new_sum > target_sum:
                high = lam
            else:
                low = lam
        lam = (low + high) / 2.0
        h = np.clip(h + lam, 0.0, 1.0)
        return h
    
    # Schedule of beta values
    betas = [1, 2, 5, 10, 20]
    
    for beta in betas:
        if time.time() - T_start > time_limit * 0.5:
            break
        # Standard gradient descent steps
        for step in range(100):
            if time.time() - T_start > time_limit * 0.8:
                break
            g = grad_F_beta(h, beta)
            eta = 0.1
            h_new = h - eta * g
            h_new = project_on_simplex(h_new, M)
            h = h_new.copy()

    return h, evaluate_h(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h, best_score = None, float('inf')
    
    # Try MILP with n=48 first
    for n in [48, 56]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break

        T_opt, h_bin = solve_milp(n, time_limit=remaining*0.6)
        if h_bin is not None:
            score = evaluate_h(h_bin)
            if score < best_score:
                best_score, best_h = score, h_bin
            # if already below target, stop early
            if best_score < 0.3807:
                return (best_h.tolist(), best_score, n)

    # Local binary annealing
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20 and best_h is not None:
        h_anneal, score = anneal_binary(best_h, max_iter=int(1e6),
                                        time_limit=remaining*0.5)
        if score < best_score:
            best_score, best_h = score, h_anneal

    # Smooth fractional refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5 and best_h is not None:
        h_cont, score = smooth_descent(best_h, time_limit=remaining*0.4)
        if score < best_score:
            best_score, best_h = score, h_cont

    # Final sanity check
    n = len(best_h)
    dx = 2.0 / n
    best_h = np.clip(best_h, 0.0, 1.0)
    total = np.sum(best_h) * dx
    if abs(total - 1.0) > 1e-9:
        best_h *= (1.0 / total)

    final_score = evaluate_h(best_h)
    return (best_h.tolist(), final_score, n)
