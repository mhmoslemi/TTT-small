import numpy as np

from scipy.optimize import minimize
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the binary MILP to find T_opt and h_opt."""
    M = n // 2  # mass constraint
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []

    # Constraints
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        # McCormick envelope for binary product
        con += [p_ell <= h[:length],
                p_ell <= h[ell:],
                p_ell >= h[:length] + h[ell:] - 1]

        # Worst-lag constraint
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    solver_options = {'maxSeconds': time_limit}
    try:
        prob.solve(solver=cp.CBC, verbose=False, **solver_options)
    except:
        return None, None

    if prob.status in ['optimal', 'optimal_inaccurate', 'feasible']:
        try:
            h_opt = np.ravel(h.value)
            T_opt = int(T.value)
            return T_opt, h_opt
        except:
            return None, None
    else:
        return None, None

def anneal_binary(h_init, max_iter=1000000, time_limit=100):
    """Run binary simulated annealing from h_init to improve C5."""
    n = len(h_init)
    M = n // 2
    h = h_init.copy()
    h = np.clip(h, 0, 1)  # ensure bounds (though it should be binary)
    current_score = evaluate_h(h)
    best_h = h.copy()
    best_score = current_score
    T = 1.0
    start = time.time()
    decay_rate = 0.999

    # Precompute iterations for temperature profile
    for i in range(max_iter):
        if time.time() - start > time_limit:
            break

        # Try to find good initial temperature
        # Sample initial swaps to estimate Δ cost
        if i == 0:
            trial_swaps = []
            for _ in range(500):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i1 = np.random.choice(ones)
                i0 = np.random.choice(zeros)
                trial_swaps.append((i1, i0))
            # Compute Δs from swaps
            deltas = []
            for i1, i0 in trial_swaps:
                h_new = h.copy()
                h_new[i1] = 0
                h_new[i0] = 1
                score_new = evaluate_h(h_new)
                deltas.append(score_new - current_score)
            if deltas:
                avg_delta = np.mean(deltas)
                T = max(1e-3, 2.0 * abs(avg_delta))

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Choose swap indices
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)

        h_new = h.copy()
        h_new[i1] = 0
        h_new[i0] = 1

        new_score = evaluate_h(h_new)

        delta = new_score - current_score

        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_score = new_score
            if current_score < best_score:
                best_h = h.copy()
                best_score = current_score

        T *= decay_rate
        if T < 1e-6:
            # Reheat if too cold
            T = max(1e-3, 1.0)
            # Or stop if no improvement for a long time?

    return best_h, best_score

def smooth_descent(h_init, time_limit=100):
    """Run log-sum-exp gradient descent with warm-started β."""
    h = h_init.copy()
    n = len(h)
    M = n // 2
    dx = 2.0 / n
    # Use the same n for the smoother
    # Smooth surrogate with log-sum-exp
    def grad_logsumexp(h, beta=1.0):
        # Compute all overlaps L_ell
        # Will do FFT-based approach, since it's efficient
        H = np.fft.rfft(h, 2*n)
        G = np.fft.rfft(1.0-h, 2*n)
        convs = np.fft.irfft(H * np.conj(G))[:2*n-1]
        L_vals = convs.real  # length 2*n-1, but only ell=1 to n-1
        # L_vals[i] = L_{i+1} = sum_h_i * (1-h_{i+1})
        weights = np.exp(beta * L_vals)
        weights /= np.sum(weights)
        # Now compute weighted gradient of log sum exp w respect to each i
        # grad L_ell = 1 - h_{i+ell} - h_{i-ell}
        grad_full = np.zeros(n)
        # For each ell from 1 to n-1, we add weight * (1 - h_{i+ell} - h_{i-ell})
        for ell in range(1,n):
            w = weights[ell-1]
            grad_full += w * (1.0 - np.roll(h, ell) - np.roll(h, -ell))
        return grad_full

    # Gradient descent on the log-sum-exp surrogate function
    # Semismooth function F_beta = (1/beta)*log(sum over ell of exp(beta*L_ell))
    # Gradient is sum_w ell * grad L_ell
    # Stable or early stopping

    start = time.time()
    # Schedule beta values
    betas = [1, 2, 5, 10, 20]
    for beta in betas:
        if time.time() - start > time_limit:
            break

        # Start with h from previous step
        # Take many projected gradient steps
        iter_step = np.min([5000, max(200, int(100000 / beta))])
        old_h = h.copy()
        for _ in range(iter_step):
            if time.time() - start > time_limit:
                break
            grad = grad_logsumexp(h, beta)
            # Gradient step
            h_prev = h.copy()
            h -= 0.1 * grad
            # Project h onto unit box and equal sum
            h = np.clip(h, 0.0, 1.0)
            # Project to ensure sum h = M
            # Water-filling method
            total = np.sum(h) * dx  # account for dx
            diff = M - total
            if abs(diff) > 1e-10:
                # Need to adjust to sum to M (get clipped dx)
                adjusted = np.zeros_like(h)
                # Use bisection
                low, high = -1.0, 1.0
                for _ in range(30):
                    try:
                        mid = (low + high) / 2
                        temp = np.clip(h + mid, 0, 1)
                        s = np.sum(temp) * dx
                        if abs(s - M) < 1e-9:
                            break
                        if s < M:
                            low = mid
                        else:
                            high = mid
                    except:
                        break
                final = np.clip(h + (low + high) / 2, 0, 1)
                h = final
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) * dx - M
        h -= diff / n  # adjust by small amount evenly
        h = np.clip(h, 0.0, 1.0)

    return h, evaluate_h(h)
