import numpy as np
from scipy import signal
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto box [0,1] and hyperplane sum = target_sum."""
    # First clip to bounds [0,1]
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Use water-filling method to adjust sum
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def compute_max_overlap(h):
    """Compute the maximum overlap and its corresponding lag."""
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute correlation, result length = 2*n - 1
    conv = signal.correlate(h, 1 - h, mode="full")
    
    # Multiply by dx to account for discretization
    conv_scaled = conv * dx
    
    # Find maximum value
    max_val = np.max(conv_scaled)
    max_idx = np.argmax(conv_scaled)
    
    # Compute the corresponding lag (from -n+1 to n-1)
    lag = max_idx - (n_points - 1)
    
    return max_val, lag

def compute_gradient_for_shift(h, m):
    """Compute the gradient of the overlap integral for a given shift m."""
    n = len(h)
    g = np.zeros_like(h)
    dx = 2.0 / n
    
    if m >= 0:
        # Overlapping region indices [0, n-1-m]
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # For m < 0: positive shift in the opposite direction
        # Overlapping region indices [max(0, m), n-1] with shift (because we compute correlation with (1-h))
        # Equivalent to the case g[i] += 1 - h[i + |m|], g[i+|m|] -= h[i]
        m2 = -m  # Make it positive
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
        
    # Multiply gradient by dx factor
    return g * dx

    best_c5 = compute_max_overlap(best_h)[0]
    
    # Optimization loop

        # Compute the current best overlap
        max_val, max_lag = compute_max_overlap(h)
        
        # Update best solution if we found a better one
        if max_val < best_c5 - 1e-12:
            best_h = h.copy()
            best_c5 = max_val
            no_improve_count = 0  # Reset counter
        else:
            no_improve_count += 1
            
        # Check if we should restart due to stagnation
        if no_improve_count >= max_no_improve:
            # Generate a new random feasible solution
            h = rng.random(n_points)
            h = np.clip(h, 0, 1)
            h = project_to_bounds_and_sum(h, target_sum)
