import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the exact binary MILP for the Erdos problem."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    con = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)

        # McCormick envelopes for p_ell = h[i] * h[i+ell]
        con += [
            p_ell <= h[:length],
            p_ell <= h[ell:],
            p_ell >= h[:length] + h[ell:] - 1
        ]

        # Ensures sum p_ell + T >= M
        con += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), con)
    solver_options = {'maxSeconds': time_limit}
    try:
        prob.solve(solver=cp.CBC, verbose=False, **solver_options)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            T_opt = int(T.value)
            h_opt = np.ravel(h.value).astype(float)
            return T_opt, h_opt
        elif prob.status in [cp.INFEASIBLE, cp.UNBOUNDED]:
            return None, None
        else:
            # Try to recover best solution
            if h.value is not None:
                h_opt = np.ravel(h.value).astype(float)
                score = evaluate_h(h_opt)
                return int(T.value) if T.value else 0, h_opt
            return None, None
    except:
        return None, None

def anneal_binary(h_start, max_iter=1000000, time_limit=100):
    """Binary simulated annealing."""
    n = len(h_start)
    M = n // 2
    dx = 2.0 / n
    h = h_start.copy()
    curr_score = evaluate_h(h)
    best_h = h.copy()
    best_score = curr_score
    T = float('inf')
    initial_temp_factor = 5.0
    temp = 1.0
    iter_count = 0
    t_start = time.time()

    # Estimate initial temperature from random swaps
    if T == float('inf'):
        deltas = []
        for _ in range(500):
            i = np.random.randint(n)
            while h[i] == 0:
                i = np.random.randint(n)
            j = np.random.randint(n)
            while h[j] == 1:
                j = np.random.randint(n)
            orig = evaluate_h(h)
            h[i], h[j] = 0, 1
            new = evaluate_h(h)
            deltas.append(new - orig)
            h[i], h[j] = 1, 0
        temp = initial_temp_factor * np.std(deltas) if deltas else 1.0
        T = temp

    while iter_count < max_iter:
        t_now = time.time()
        if t_now - t_start > time_limit:
            break

        i = np.random.randint(n)
        while h[i] == 0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] == 1:
            j = np.random.randint(n)
        orig = curr_score
        h[i], h[j] = 0, 1
        new = evaluate_h(h)
        delta = new - orig
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            curr_score = new
            if new < best_score:
                best_score = new
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
        iter_count += 1
        T *= 0.999
        if T < 1e-6:
            T = 1e-6

    return best_h, best_score

def smooth_descent(h_start, time_limit=50):
    """Smoothed descent with log-sum-exp."""
    h = h_start.astype(float)
    n = len(h)
    dx = 2.0 / n
    M = n // 2
    max_beta = 20.0
    step_size = 0.1

    # Helper to shift h
    def shift_forward(x, l):
        return np.concatenate([x[l:], np.zeros(l)])
    def shift_backward(x, l):
        return np.concatenate([np.zeros(l), x[:-l]])

    # Run several beta values sequentially
    for beta in [1.0, 2.0, 5.0, 10.0, 20.0]:
        t_start = time.time()
        while True:
            # Time check
            if time.time() - t_start > time_limit * 0.8:
                break

            # Compute all L_ell for current h
            Ls = []
            for ell in range(1, n):
                shift_h = shift_forward(h, ell)
                L = np.sum(h[:n-ell] * (1 - shift_h[:n-ell]))
                Ls.append(L)
            Ls = np.array(Ls)

            # Compute energies and weights
            exp_L = np.exp(beta * Ls)
            Z = np.sum(exp_L)
            if Z == 0:
                break
            weights = exp_L / Z

            # Compute gradient with respect to h
            grad = np.zeros(n)
            for ell, w in enumerate(weights):
                ell_idx = ell + 1
                shift_h = shift_forward(h, ell_idx)
                back_h = shift_backward(h, ell_idx)
                g = 1.0 - shift_h[:n-ell_idx] - back_h[ell_idx:]
                grad[:n-ell_idx] += w * g

            # Step and project
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0., 1.)
            total = np.sum(h_new) * dx
            if abs(total - 1.0) > 1e-9:
                # Waterfilling to adjust
                scale = (1.0 / total)
                h_new *= scale
            h = h_new

    return h, evaluate_h(h)
