def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """Project h onto the capped simplex: 0 <= h_i <= 1 and sum(h) = target_sum"""
    # Simple bisection to find lambda such that projected sum = target_sum
    low, high = -100, 100
    for _ in range(40):
        mid = (low + high) / 2.0
        proj = np.clip(h - mid, 0, 1)
        current_sum = np.sum(proj)
        if current_sum > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(h - (low + high) / 2.0, 0, 1)

def smooth_max(c, alpha):
    """Softmax approximation of max"""
    exp_c = np.exp(alpha * c)
    return exp_c / np.sum(exp_c)

def compute_gradient(h, w):
    """Compute gradient for the smooth objective"""
    n = len(h)
    # Convolve with w to get the contribution of each h_i
    w_rev = w[::-1]
    g_plus = np.convolve(w_rev, h, mode='full')
    g_minus = np.convolve(w, h, mode='full')
    grad = 1.0 - (g_plus[n-1:n-1+n] + g_minus[n-1:n-1+n])
    return grad
