import numpy as np

def proj_simplex(y, S):
    """
    Project vector y onto the capped simplex: {x : 0 <= x_i <= 1, sum x_i = S}
    Using bisection on the lambda parameter.
    """
    y = np.asarray(y)
    n = len(y)
    # Ensure S is within valid range
    S = max(0, min(S, n))
    
    # Compute initial bracket for lambda
    lambda_low = -np.max(y) - 1
    lambda_high = np.max(y) + 1
    
    def F(lambda_val):
        x = np.clip(y - lambda_val, 0, 1)
        return np.sum(x)
    
    # Bisection
    for _ in range(40):
        lambda_mid = (lambda_low + lambda_high) / 2.0
        f_mid = F(lambda_mid)
        if f_mid >= S:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
            
    lambda_final = (lambda_low + lambda_high) / 2.0
    x = np.clip(y - lambda_final, 0, 1)
    # Adjust to exactly sum to S
    scale = S / np.sum(x)
    x *= scale
    return x

def solve_sdp(n, budget_s, start_time):
    """
    Solve SDP relaxation for n points.
    Return (best_h, c5_bound)
    """
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    h = cp.Variable(n)
    m = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,                          # X psd
        cp.diag(X) == h,                 # diag(X) == h
        0 <= h, h <= 1,                  # box constraints
        cp.sum(h) == n/2                 # integral constraint
    ]
    
    # Add constraints on r_s >= m for all shifts s
    for s in range(-(n-1), n):
        if s >= 0:
            # sum_{i=0}^{n-1-s} X_{i, i+s}
            expr = sum(X[i, i+s] for i in range(n-s))
        else:
            # sum_{i=-s}^{n-1} X_{i, i+s}
            expr = sum(X[i, i+s] for i in range(-s, n))
        constraints.append(expr >= m)
    
    # Objective
    objective = cp.Maximize(m)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, eps=1e-5, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            m_opt = m.value
            X_opt = X.value
            h_opt = h.value
            # Extract feasible h from X_opt:
            # use top eigenvector
            evals, evecs = np.linalg.eigh(X_opt)
            v = np.abs(evecs[:, -1])
            v = v / np.sum(v) * (n/2)
            hhat = np.clip(v, 0, 1)
            hhat = proj_simplex(hhat, n/2)
            # Compute C5 for this h
            c5 = evaluate_c5_discrete(hhat, n)
            return (hhat, c5)
    except Exception:
        pass
    return None, None

def evaluate_c5_discrete(h, n):
    """Evaluate C5 for discrete h."""
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

    # One final evaluation
