import numpy as np
from scipy.optimize import minimize
from scipy.fft import ifft, fft
import time
import random
from math import isclose

def project_onto_capped_simplex(v, target_sum):
    """Project vector v onto {x : 0 <= x_i <= 1, sum x_i = target_sum}"""
    v = np.asarray(v)
    n = len(v)
    x = v.copy()
    # Ensure it's feasible by clamping to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if isclose(current_sum, target_sum):
        return x
    # Use water-filling algorithm
    # sort values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_vals = x[sorted_indices]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_vals)
    # find threshold
    diff = target_sum - cumsum
    # Find index such that the cumulative sum is less than t
    threshold_idx = -1
    for i in range(n):
        if diff[i] >= 0 and (i == n-1 or diff[i+1] < 0):
            threshold_idx = i
            break
    # If all values are zero or all are one, adjust slightly to meet target
    if threshold_idx == -1:
        if current_sum > target_sum:
            # Remove excess from largest values
            # set smallest values to 0 and redistribute
            # largest exceed
            excess = current_sum - target_sum
            idx_order = np.argsort(x)
            for i in range(int(np.ceil(excess))):
                idx = idx_order[i]
                x[idx] = 0 if x[idx] <= excess else x[idx] - excess
            x = x / np.sum(x) * target_sum
        else:
            x = np.ones_like(x) * (target_sum / n)
        return x
    
    threshold = sorted_vals[threshold_idx]
    # Now, clip everything larger than threshold
    y = np.minimum(sorted_vals, threshold)
    # Ensure sum to target
    remaining_target = target_sum - np.sum(y)
    # Distribute to remaining
    num_remaining = n - threshold_idx - 1
    if num_remaining > 0:
        extra = remaining_target / num_remaining
        y[threshold_idx+1:] += extra
    # Now reindex back
    x_new = x.copy()
    x_new[sorted_indices] = y
    return x_new

def compute_correlations(h, dx):
    """Compute full correlations using FFT for speed."""
    h = np.array(h)
    n = len(h)
    # A bit of padding for proper full convolution
    h_padded = np.concatenate([h, np.zeros(n)])
    h_shifted = np.concatenate([np.zeros(n), h])
    conv_result = ifft(fft(h_padded) * fft(h_shifted.conj())).real
    # Take the middle part, padded with zeros
    correlations = conv_result[n-1:2*n-1] 
    # Multiply by dx for normalization
    return correlations * dx

def evaluate_c5_smooth(h, dx, alpha=50):
    """Evaluate smooth surrogate of C5 using log-sum-exp"""
    correlations = compute_correlations(h, dx)
    max_r = np.max(correlations)
    shifted = correlations - max_r
    exp_vals = np.exp(alpha * shifted)
    log_sum_exp = max_r + (1./alpha) * np.log(np.sum(exp_vals))
    return log_sum_exp

def compute_gradient(h, dx, alpha=50):
    """Compute gradient of smooth C5 with respect to h"""
    n = len(h)
    # correlations
    correlations = compute_correlations(h, dx)
    max_r = np.max(correlations)
    shifted = correlations - max_r
    exp_vals = np.exp(alpha * shifted)
    weights = exp_vals / np.sum(exp_vals)
    # compute each ∇r_k
    grad_sum = np.zeros(n)
    # For each k, compute d/dh_i of r_k = w - sum h_i h_{i+k}
    # ∂r_k / ∂h_i = 1 - h_{i+k} * (i+k < n) - h_{i-k} * (i-k >= 0)
    for k in range(len(correlations)):
        temp_grad = np.zeros_like(h)
        if k == 0:
            temp_grad = 1 - 2 * h
        else:
            # i+k < n
            if k < n:
                temp_grad[:-k] = 1 - h[k:]
                temp_grad[k:] = temp_grad[k:] - h[:-k]
            else:
                # i - k >=0
                temp_grad[k:] = 1 - h[:n-k]
                temp_grad[:n-k] = temp_grad[:n-k] - h[k:]


        grad_sum += weights[k] * temp_grad

    return dx * grad_sum

def backtracking_line_search(h, grad, dx, alpha, func_eval, max_iter=100):
    """Backtracking line search to pick step size"""
    gamma = 0.5
    c = 1e-4
    eta = 1.0
    for _ in range(max_iter):
        h_new = project_onto_capped_simplex(h - eta * grad, np.sum(h))
        new_val = evaluate_c5_smooth(h_new, dx, alpha)
        if new_val <= func_eval - c * eta * np.sum(grad * grad):
            return h_new, eta
        eta *= gamma
    return h, eta

def accelerated_projected_gradient_descent(h0, dx, w, alpha0=1.0, alpha_factor=2.0, alpha_max=300, max_iters=5000):
    """Run gradient descent with Nesterov acceleration and projection"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    alpha = alpha0

    prev_val = evaluate_c5_smooth(h, dx, alpha)
    converged = False

    for iteration in range(max_iters):
        grad = compute_gradient(h, dx, alpha)
        h_tilde, eta = backtracking_line_search(h, grad, dx, alpha, prev_val)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2.0
        y_new = h_tilde + ((t - 1) / t_new) * (h_tilde - h)
        h = y_new
        t = t_new
        curr_val = evaluate_c5_smooth(h, dx, alpha)
        rel_change = abs(curr_val - prev_val) / (abs(prev_val) + 1e-12)
        if rel_change < 1e-6 and iteration > 100:
            converged = True
        if iteration % 500 == 0:
            # Adjust smoothing parameter
            if alpha < alpha_max:
                alpha = min(alpha * alpha_factor, alpha_max)
                # restart
                t = 1
                y = h.copy()
        prev_val = curr_val

    return h, prev_val
