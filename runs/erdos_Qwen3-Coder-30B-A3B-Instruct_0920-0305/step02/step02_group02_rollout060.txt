def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def autocorr_h(h):
    """Compute correlation S_k = sum_i h_i * h_{i+k} for all k."""
    corr = np.correlate(h, h, mode='full')
    # Extract correlations for non-negative lags (k >= 0)
    S = corr[len(h)-1:] 
    return S

def gradient_S_k(h, k):
    """Compute gradient of S_k w.r.t. h."""
    n = len(h)
    if k == 0:
        return 2 * h
    else:
        g = np.zeros_like(h)
        if k < n:
            g[:n-k] += h[k:]   # h_i contribution from h_{i+k}
            g[k:] += h[:n-k]   # h_i contribution from h_{i-k}
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):

    assert lower * len(v) <= target_sum <= upper * len(v), "Sum out of bounds"

    v = np.array(v).copy()

    # Adjust to exact target sum
    if abs(v_sum - target_sum) < 1e-10:

    sorted_v = v[sorted_indices]
    
    # Find threshold
    remaining_sum = target_sum

    threshold = 0.0
    
    for i in range(len(sorted_v)):
        # Check if we can set this value to the threshold
        if sorted_v[i] <= threshold:
            break
        # Compute how much we can raise the threshold
        effective_value = sorted_v[i] 
        num_above_threshold = len(sorted_v) - i
        if i > 0:
            effective_value = min(sorted_v[i], threshold + (remaining_sum - threshold * i) / num_above_threshold)
        if effective_value < threshold:
            break
        # Set everything to threshold and part of this to effective_value
        # but we'll handle it via iteration
        threshold = effective_value
        num_used = i + 1
        remaining_sum -= threshold
    
    # Now adjust the remaining elements
    if num_used < len(sorted_v):
        avg_remaining = remaining_sum / (len(sorted_v) - num_used)
        # Only adjust those that are not constrained
        for i in range(num_used, len(sorted_v)):
            sorted_v[i] = max(min(avg_remaining, upper), lower)
            
    # Adjust elements used in thresholding
    for i in range(num_used):
        sorted_v[i] = max(min(threshold, upper), lower)
    
    # Re-sort to original positions
    result = np.zeros_like(v)
    result[sorted_indices] = sorted_v
    
    # Some adjustment may be needed due to numerical errors
    final_sum = np.sum(result)
    diff = target_sum - final_sum
    if abs(diff) > 1e-8:
        # Distribute the difference
        num_nonzero = len(result[result > 0])
        if num_nonzero > 0:
            step = diff / num_nonzero
            for i in range(len(result)):
                # Distribute evenly ensuring bounds
                if result[i] + step > upper:
                    result[i] = upper
                elif result[i] + step < lower:
                    result[i] = lower
                else:
                    result[i] += step
    
    assert abs(np.sum(result) - target_sum) < 1e-5
    return result
