import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto bounds [0,1] and maintain sum constraint."""
    # Clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Use water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase some values
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            # Need to decrease some values
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a given shift."""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx = 2/n factor

def compute_max_overlap(h):
    """Compute maximum overlap integral and return the maximizing shift."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    shift_m = argmax - (n - 1)
    return max_val, shift_m
