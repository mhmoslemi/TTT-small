    refine_time = 0.15 * budget_s    # 90s for local refinement

    for n in n_list:
        if time.time() - start_time > sdps_solve_time + start_time:
            break

            # For each n, we solve an SDP to find the best lower bound for r_s
            # introduce scalar var m, and var X (n x n), h (n)
            # minimize -m
            # subject to:
            #   X - h h^T = 0 (this is hard to write directly)
            #   X is PSD  
            #   X[i,i] = h[i] for all i
            #   0 <= h[i] <= 1
            #   sum(h) = n/2
            #   sum_{i} X[i, i+s] >= m for all shifts s

            # Formulate SDP with CVXPY for small n
            h = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)
            m = cp.Variable()

            constraints = [
                X >> 0,   # X is PSD
                cp.diag(X) == h,  # diagonal constraint
                h >= 0, h <= 1,  # box constraints
                cp.sum(h) == n/2,  # integral constraint
            ]
            
            # Add r_s >= m constraints for all s = -(n-1)...(n-1)
            # For now, add constraints only for necessary shifts to make SDP solvable
            for s in range(1, n):
                # Sum of X[i, i+s]
                sum_entries = 0
                for i in range(n - s):
                    sum_entries += X[i, i + s]
                constraints.append(sum_entries >= m)

            # solver settings
            problem = cp.Problem(cp.Maximize(m), constraints)

            # Solve
            problem.solve(solver=ECOS, verbose=False, max_iters=1000, feastol=1e-5, reltol=1e-5)
            
            if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                m_val = m.value
                h_val = h.value
                if m_val is not None:
                    # extract h: take eigenvec
                    X_val = X.value
                    # compute leading eigenvector
                    try:
                        if X_val is not None:
                            eigvals, eigvecs = np.linalg.eigh(X_val)
                            v = eigvecs[:, -1]
                            v = np.abs(v)
                            scale = (n/2) / np.sum(v)
                            h_extract = v * scale
                            h_extract = np.clip(h_extract, 0, 1)
                            
                            # Project to feasible simplex
                            def project_simplex(y, s):
                                # Algorithm from Constrained Optimization on the Unit Simplex
                                # by Weiran Wang and Vladimir Kuznetsov
                                y = np.array(y)
                                u = np.sort(y)[::-1]
                                tmp = (np.cumsum(u) - s) / np.arange(1, len(y)+1)
                                rho = np.argmax(u > tmp) if u[0] > tmp[0] else len(u) - 1
                                theta = tmp[rho - 1] if rho > 0 else tmp[-1] 
                                return np.clip(y - theta, 0, 1)

                            h_extract = project_simplex(h_extract, n/2)

                            # Evaluate C5 for n-size vector
                            dx = 2.0 / n
                            corr = correlate(h_extract, 1 - h_extract, mode="full")
                            max_corr = np.max(corr)
                            c5_est = max_corr * dx
                            if c5_est < best_c5:
                                best_c5 = c5_est
                                best_h = h_extract.copy()
                        else:
                            print(f"X is None for n={n}")
                    except Exception as e:
                        print(f"Eigendecomposition failed for n={n}, {e}")
                else:
                    print(f"m is None for n={n}")
            else:
                print(f"SDP {n} not optimal. Status: {problem.status}")

            dx = 2.0 / len(h)
            corr = correlate(h, 1-h, mode="full")
            return -np.max(corr) * dx  # negative because minimize

        # define constraints on simplex 
        def simplexeq(h):
            return np.sum(h) - len(h)/2

        cons = {'type': 'eq', 'fun': simplexeq}
        lb = [0] * len(best_h)
        ub = [1] * len(best_h)
        bounds = list(zip(lb, ub))

        # run simplex projection after each step (similar to batch procedure)
        MaxIters = 300
        try:
            res = minimize(objective_local, best_h, method='Nelder-Mead',
                           options={'maxiter': MaxIters, 'fatol': 1e-9, 'xatol': 1e-9})
            if res.success:
                h_new = res.x
                dx = 2.0 / len(h_new)
                corr = correlate(h_new, 1-h_new, mode="full")
                max_corr = np.max(corr)
                c5_new = max_corr * dx
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:

    # Phase 3: Upsampling to n_target = ~180
    if time.time() - start_time < budget_s:
        # choose target resolution
        n_target = 180

        # interpolate best_h to finer grid
        h_orig = best_h
        n_orig = len(h_orig)
        dt_orig = 2.0 / n_orig
        dt_target = 2.0 / n_target

        # resample uniformly
        indices_target = np.linspace(0, n_orig - 1, n_target)
        h_up = np.interp(indices_target, np.arange(n_orig), h_orig)

                corr = correlate(h, 1-h, mode="full")

                return np.sum(h) - n_target/2

                res = minimize(objective_fine, h_up, method='Nelder-Mead',
                               options={'maxiter': min(2000, int((budget_s - (time.time()-start_time)) / 0.05)),
                                        'fatol': 1e-10, 'xatol': 1e-10})
                if res.success:
                    candidate_h = res.x
                    dx = 2.0 / n_target
                    corr = correlate(candidate_h, 1 - candidate_h, mode="full")
                    max_corr = np.max(corr)
                    c5_candidate = max_corr * dx
                    if c5_candidate < final_c5:
                        final_c5 = c5_candidate

            except:
