import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    dx = 2.0 / len(h)
    c5 = max_correlation * dx
    return c5

def project_onto_feasible(v, target_sum):
    """Project vector v onto [0,1]^n and then correct the sum to target_sum."""
    def f(lam):
        x = np.clip(v - lam, 0, 1)
        return np.sum(x) - target_sum

    # Bisection to find lambda
    a, b = np.min(v), np.max(v) + (target_sum - np.sum(np.clip(v, 0, 1)))
    for _ in range(50):
        mid = (a + b) / 2
        if f(mid) > 0:
            b = mid
        else:
            a = mid

    lam = (a + b) / 2
    x = np.clip(v - lam, 0, 1)
    return x

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Try different n values
    n_candidates = [48, 56, 60]
    best_h = None
    best_c5 = float('inf')
    best_n = None

    for n in n_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        target_sum = n / 2.0

        # Initial starting points
        starts = []

        # Interpolated from initial_h_values (if available)
        try:
            original = np.array(initial_h_values)
            if len(original) <= n:
                h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(original)), original)
            else:
                h_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(original)), original)
            starts.append(h_interp)
        except:
            starts.append(np.ones(n) * 0.5)

        # Random binary pattern with correct weight
        h_binary = np.zeros(n)
        indices = np.random.choice(n, size=int(target_sum), replace=False)
        h_binary[indices] = 1
        h_binary = project_onto_feasible(h_binary, target_sum)
        starts.append(h_binary)

        # Random uniform + projection
        h_uniform = np.random.uniform(0, 1, n)
        h_uniform = project_onto_feasible(h_uniform, target_sum)
        starts.append(h_uniform)

        # Previous best input (if it's within this n)
        try:
            h_prev = np.array(initial_h_values)
            if len(h_prev) <= n:
                h_prev_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_prev)), h_prev)
            else:
                h_prev_interp = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_prev)), h_prev)
            starts.append(h_prev_interp)
        except:
            pass

        # Multi-start gradient ascent
        temp_schedule = [5, 15, 30, 60, 120]
        inner_iter_limit = 300
        for i, h_start in enumerate(starts):
            if time.time() - start_time > 0.95 * budget_s:
                break
            h_current = h_start.copy()
            h_current = project_onto_feasible(h_current, target_sum)

            current_c5 = evaluate_h(h_current)
            best_local_h = h_current.copy()
            best_local_c5 = current_c5

            # Gradient ascent with continuation
            for beta in temp_schedule:
                if time.time() - start_time > 0.95 * budget_s:
                    break
                for _ in range(inner_iter_limit):
                    if time.time() - start_time > 0.95 * budget_s:
                        break
                    # Compute all C_s values
                    corr = np.correlate(h_current, 1 - h_current, mode="full")
                    c_s = corr * dx
                    max_c = np.max(c_s)
                    c_s -= max_c  # Avoid overflow

                    # Soft min weights
                    exp_neg_beta_c = np.exp(-beta * c_s)
                    w_s = exp_neg_beta_c / np.sum(exp_neg_beta_c)

                    # Compute gradient
                    # A^{(s)} h: for each s, compute (h_{i-s} + h_{i+s}) / 2
                    grad = np.zeros(n)
                    for s in range(n):
                        if s == 0:
                            grad += w_s[s] * h_current
                        else:
                            # Shift left by s
                            shifted_left = np.roll(h_current, -s)
                            shifted_left[:s] = 0
                            # Shift right by s
                            shifted_right = np.roll(h_current, s)
                            shifted_right[-s:] = 0
                            grad += w_s[s] * (shifted_left + shifted_right) * 0.5
                    grad *= 2 * dx

                    # Line search with fixed α = 0.5
                    alpha = 0.5
                    h_trial = h_current + alpha * grad
                    h_trial = project_onto_feasible(h_trial, target_sum)
                    c_trial = evaluate_h(h_trial)

                    # Accept or reject
                    if c_trial < current_c5 - 1e-12:
                        h_current = h_trial
                        current_c5 = c_trial
                        if current_c5 < best_local_c5:
                            best_local_h = h_current.copy()
                            best_local_c5 = current_c5
                    # If rejected, stop early
                    else:
                        break

                # For subsequent β, we already have a good solution
                if beta == temp_schedule[-1]:
                    break

            # Local refinement with SLSQP
            bounds = [(0, 1) for _ in range(n)]
            def objective(h):
                return evaluate_h(h)
            try:
                result = minimize(objective, best_local_h, method="SLSQP",
                                 bounds=bounds, options={"maxiter": 200, "ftol": 1e-12})
                if result.success and result.fun < best_local_c5:
                    refined_h = result.x
                    refined_c5 = evaluate_h(refined_h)
                    if refined_c5 < best_local_c5:
                        best_local_c5 = refined_c5
                        best_local_h = refined_h
            except:
                pass

            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = best_local_h.copy()
                best_n = n

            if best_c5 < target_bound:
                break

        if best_c5 < target_bound:
            break

    # Final evaluate
    if best_h is not None:
        final_evaluate = evaluate_h(best_h)
        return list(best_h), final_evaluate, best_n
    else:
        # fallback to a simple pattern
        h = np.ones(48) * 0.5
        c5 = evaluate_h(h)
        return list(h), c5, 48
