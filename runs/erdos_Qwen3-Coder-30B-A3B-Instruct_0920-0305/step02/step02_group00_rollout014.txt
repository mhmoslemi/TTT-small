import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Problem, Minimize, Variable, sum, multiply, norm
import cvxpy as cp

# Constants
n_points_list = [48, 56, 64, 72]
dx = 2.0 / n_points_list[0]
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_constraints(h, sum_target):
    """Project h onto the box [0,1]^n and the hyperplane sum(h) = sum_target."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - sum_target) > 1e-10:
        # Use bisection to find the right scaling lambda
        a, b = 0.0, 1.0
        for _ in range(20):  # Bisection iterations
            lam = (a + b) / 2
            h_test = np.clip(h - lam, 0, 1)
            test_sum = np.sum(h_test)
            if test_sum > sum_target:
                a = lam
            else:
                b = lam
        h = np.clip(h - (a + b)/2, 0, 1)
        # Correct for possible numerical errors
        h = h * sum_target / np.sum(h)
    return h

def build_linear_constraints(h):
    """
    For each shift s, compute a_s (coefficient vector) and b_s (constant)
    such that sum_i h_i * h_{i+s} ≈ sum_i (a_s[i]*h[i] + b_s)
    """
    n = len(h)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0
        for i in range(n):
            # h_i * h_{i+s} ~ h_i^k * h_{i+s}^k + h_{i+s}^k * h_i^k - h_i^k * h_{i+s}^k
            # So h_i * h_{i+s} ~ a_s[i] * h[i] + b_s
            # Further: a_s[i] = 1 - h_{i+s}^k - h_{i-s}^k (those entries that exist)
            #     b_s = - sum_i h_i^k * h_{i+s}^k
            i_plus_s = i + s
            a_s[i] = 1  # base contribution from h_i
            if i_plus_s < n:
                a_s[i] -= h[i_plus_s]
            if i - s >= 0:
                a_s[i] -= h[i - s]
            # Additionally add - h_i^k * h_{i+s}^k (constant term)
        # Now compute the constant term
        correlation = np.correlate(h, h, mode='full')[n-1:]  # For shifted correlations
        # This is the bilinear sum for shift s
        b_s = -correlation[s] * dx  # dx is built into the correlation
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_subproblem(a_s_list, b_s_list, sum_target):
    """
    Solve the LP: min t subject to a_s^T h + b_s <= t for all s,
                      0 <= h_i <= 1, sum h_i = sum_target.
    """
    n = len(a_s_list[0])
    h = Variable(n)
    t = Variable()
    constraints = [0 <= h, h <= 1, sum(h) == sum_target]
    for s in range(len(a_s_list)):
        constraints.append(a_s_list[s] @ h + b_s_list[s] <= t)
    prob = Problem(Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
            t_sol = t.value
            return h_sol, t_sol
        else:
            # Fallback to basic approach
            return None, None
    except:
        return None, None

def slp_loop(h_start, max_iter=200, tol=1e-6):
    """Sequential linear programming loop."""
    h_current = h_start.copy()
    best_h = h_start.copy()
    best_obj = evaluate_h(h_current)
    for k in range(max_iter):
        # Compute O_s for current h
        correlation = np.correlate(h_current, h_current, mode="full")
        # Build constraints
        a_s_list, b_s_list = build_linear_constraints(h_current)
        # Solve LP
        h_new, t_new = solve_subproblem(a_s_list, b_s_list, len(h_current)/2)
        if h_new is None:
            break
        # Project h_new onto constraints
        h_projected = project_onto_constraints(h_new, len(h_current)/2)
        # Evaluate current true objective
        obj = evaluate_h(h_projected)
        if obj < best_obj:
            best_h = h_projected.copy()
            best_obj = obj
        # Check for convergence
        if k > 0 and abs(obj - prev_obj) < tol:
            break
        prev_obj = obj
        h_current = h_projected
    return best_h, best_obj

def perturb_solution(h_values, seed=42):
    """Perturb solution slightly."""
    np.random.seed(seed)
    h_new = h_values.copy()
    indices = np.random.choice(len(h_values), size=max(1, int(0.1 * len(h_values))), replace=False)
    for i in indices:
        h_new[i] = np.clip(h_new[i] + np.random.uniform(-0.1, 0.1), 0, 1)
    h_new = project_onto_constraints(h_new, len(h_values)/2)
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    n_used = 48  # Default from initial_h_values
    dx = 2.0 / n_used
    
    for n in n_points_list:
        dx = 2.0 / n
        restart_num = 15
        for run_idx in range(restart_num):
            if time.time() - start_time > budget_s * 0.9:
                break
            
            # Different starting points
            if run_idx == 0:
                # Interpolate initial_h_values
                target_len = n
                if len(initial_h_values) != target_len:
                    # Simple linear interpolation
                    x_orig = np.linspace(0, 2, len(initial_h_values))
                    x_new = np.linspace(0, 2, target_len)
                    h_start = np.interp(x_new, x_orig, initial_h_values)
                else:
                    h_start = np.array(initial_h_values)
                h_start = project_onto_constraints(h_start, target_len/2)
            elif run_idx == 1:
                # Random uniform
                h_start = np.random.uniform(0, 1, n)
                h_start = project_onto_constraints(h_start, n/2)
            elif run_idx < 5:
                # Binary like
                h_start = np.zeros(n)
                h_start[:n//2] = 1
                np.random.shuffle(h_start)
                h_start = project_onto_constraints(h_start, n/2)
            else:
                # Perturb best so far
                if best_h is not None:
                    h_start = perturb_solution(best_h, seed + run_idx)
                else:
                    h_start = np.random.uniform(0, 1, n)
                    h_start = project_onto_constraints(h_start, n/2)

            # Run SLP
            try:
                h_final, c5_final = slp_loop(h_start, max_iter=200)
            except:
                continue
            
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                best_n = n

            if best_c5 < target_bound:
                break

        if best_c5 < target_bound:
            break

    # Final polishing
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n)]
        def objective(h):
            return evaluate_h(h)
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                             constraints=constraints, options={'maxiter': 500,
                                                               'ftol': 1e-12,
                                                               'gtol': 1e-10})
            if result.success:
                best_h = result.x
                best_c5 = evaluate_h(best_h)
        except:
            pass

    return list(best_h), best_c5, best_n
