n_candidates = [48, 56, 64, 80]
dx_candidates = [2.0 / n for n in n_candidates]
w_candidates = [n / 2 for n in n_candidates]

def project_onto_capped_simplex(v, target_sum):
    """Project vector v onto the capped simplex {x | 0 <= x <= 1, sum(x) = target_sum}."""
    # Sort in descending order
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # Water-filling
    cumsum = np.cumsum(sorted_v)
    n = len(sorted_v)
    threshold = 0
    for i in range(n):
        if sorted_v[i] - (target_sum - cumsum[i]) / (n - i) >= 0:
            threshold = sorted_v[i] - (target_sum - cumsum[i]) / (n - i)
            break
    else:
        threshold = sorted_v[-1] - (target_sum - cumsum[-1]) / n

    x = np.maximum(np.minimum(sorted_v - threshold, 1.0), 0.0)
    x = x[np.argsort(sorted_indices)]
    
    # Ensure sum matches target
    actual_sum = np.sum(x)
    if not np.isclose(actual_sum, target_sum):
        # Adjust the smallest non-bound entry
        idx = np.argmax(x)
        x[idx] += (target_sum - actual_sum)
    
    return x

def correlation_all_k(h):
    """Compute all correlations r_k = sum h_i * (1 - h_{i+k}) for k = 0,...,n-1."""
    n = len(h)
    r_list = []
    for k in range(n):
        # Compute r_k = sum_i h_i * (1 - h_{i+k})
        # Use circular convolution for speed (last elements wrap to beginning)
        r = 0
        for i in range(n):
            j = (i + k) % n
            r += h[i] * (1.0 - h[j])
        r_list.append(r)
    return np.array(r_list)

def evaluate_c5_exact(h):
    """Evaluate C5 = max_k r_k by exact computation."""
    r = correlation_all_k(h)
    return np.max(r) * 2.0 / len(h)

def log_sum_exp(x, alpha):
    """Compute log(sum(exp(alpha * x))) with numerical stability."""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(alpha * (x - x_max))))

def gradient_correlation(h):
    """Compute gradient w.r.t. each h_i of sum of (1 - h_{i+k}) * h_i."""
    n = len(h)
    grad = np.zeros_like(h)
    for i in range(n):
        grad[i] = 1.0
        for k in range(1, n):
            j = (i + k) % n
            grad[i] -= h[j]
            j = (i - k) % n
            grad[i] -= h[j]
    return grad

def evaluate_surrogate_loss(h, dx, alpha):
    """Compute surrogate log-sum-exp of correlations."""
    r = correlation_all_k(h)
    # Compute surrogate f_α = (dx/α) * log(sum_k exp(α * r_k))
    f_alpha = dx * log_sum_exp(r, alpha)
    return f_alpha

def evaluate_surrogate_gradient(h, dx, alpha):
    """Compute gradient of surrogate log-sum-exp of correlations."""
    n = len(h)
    r = correlation_all_k(h)
    # Softmax weights
    r_max = np.max(r)
    exp_r = np.exp(alpha * (r - r_max))
    w = exp_r / np.sum(exp_r)
    # Gradient of each r_k w.r.t. h_i
    grad_all = []
    for k in range(n):
        g = np.zeros(n)
        for i in range(n):
            g[i] = 1.0
            # h_{i+k} term
            j = (i + k) % n
            g[i] -= h[j]
            # h_{i-k} term
            j = (i - k) % n
            g[i] -= h[j]
        grad_all.append(g)
    
    # Weighted sum
    grad = dx * np.sum([w[i] * grad_all[i] for i in range(n)], axis=0)
    return grad

def projected_grad_descent(h0, dx, w, alpha0=1.0, alpha_factor=2.0, alpha_max=300.0, max_iters=5000):
    """Accelerated projected gradient descent with continuation."""
    h = h0.copy()
    t_old = 1.0
    y = h.copy()
    
    alpha = alpha0
    t_prev = np.inf
    converged = False
    
    for it in range(max_iters):
        # Compute r_k values
        r_all = correlation_all_k(h)
        r_max = np.max(r_all)
        exp_r = np.exp(alpha * (r_all - r_max))
        w = exp_r / np.sum(exp_r)
        
        # Gradient of surrogate
        grads = [np.zeros_like(h) for _ in range(len(h))]
        for k in range(len(h)):
            for i in range(len(h)):
                grads[k][i] = 1.0
                j = (i + k) % len(h)
                grads[k][i] -= h[j]
                j = (i - k) % len(h)
                grads[k][i] -= h[j]
        
        g = dx * np.sum([w[i] * grads[i] for i in range(len(h))], axis=0)
        
        # Nesterov-style extrapolation
        h_tilde = project_onto_capped_simplex(y - 0.5 * g, w)
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_old**2)) / 2.0
        y = h_tilde + ((t_old - 1.0) / t_new) * (h_tilde - h)
        
        h = h_tilde
        t_old = t_new
        
        # Check convergence
        if it % 50 == 0:
            loss = evaluate_surrogate_loss(h, dx, alpha)
            if abs(loss - t_prev) < 1e-6:
                converged = True
            t_prev = loss

        # Increase alpha periodically
        if alpha < alpha_max and (it % 200 == 0 or converged):
            alpha *= alpha_factor
            if alpha > alpha_max:
                alpha = alpha_max

        if converged and alpha == alpha_max:
            break
    
    c5_val = evaluate_c5_exact(h)
    return h, c5_val

def initialize_h(n, method="binary_bias"):
    """Initialise h with one of several schemes."""
    w = n / 2.0
    if method == "random":
        h = np.random.rand(n)
        h = project_onto_capped_simplex(h, w)
    elif method == "binary_bias":
        h = np.zeros(n)
        h[:int(w)] = 1.0
        # Add tiny perturbation to create a soft distribution
        h += np.random.normal(0.0, 0.02, n)
        h = project_onto_capped_simplex(h, w)
    elif method == "three_block":
        # Create a 3-block pattern
        a = np.random.rand() * 0.5
        b = a + 0.2 + np.random.rand() * 0.3
        a = max(0.0, min(a, 1.0))
        b = max(0.0, min(b, 1.0))
        if a < b:
            p = (w - (b - a)) / (b - a)
            p = max(0.0, min(1.0, p))
            h = np.ones(n)
            for i in range(n):
                x = i * dx
                if x < a:
                    h[i] = 1
                elif x >= b:
                    h[i] = 0
                else:
                    h[i] = p
        else:
            h = np.ones(n) * 0.5
        h += np.random.normal(0.0, 0.01, n)
        h = project_onto_capped_simplex(h, w)
    return h
