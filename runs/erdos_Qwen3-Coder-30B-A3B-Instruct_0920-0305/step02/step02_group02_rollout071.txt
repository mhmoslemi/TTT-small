import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time
import random

# Constants
n_points_list = [48, 50, 52, 54, 56, 58, 60]  # Even numbers for binary feasibility
target_bound = 0.3807
dx = 2.0 / 60  # Using largest n for consistent dx

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def compute_autocorr(h):
    """Compute all correlation values S_k = sum_i h_i * h_{i+k}"""
    n = len(h)
    # Compute sum_i h_i * h_{i+k} for all k
    autocorr = np.correlate(h, h, mode="full")  # This computes h conv h
    S = autocorr[n-1:]  # Take only non-negative lags (k=0 to n-1)
    return S

def grad_S_k(h, k):
    """Gradient of S_k with respect to h"""
    n = len(h)
    if k == 0:
        return 2 * h
    else:
        g = np.zeros_like(h)
        if k < n:
            g[:n-k] += h[k:]   # h_i contribution: derivative includes h_{i+k}
            g[k:] += h[:n-k]   # derivative includes h_{i-k}
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector onto simplex with bounds [lower, upper] and sum = target_sum"""
    v = np.array(v).copy()
    n = len(v)
    # Shift by lower bound to make it non-negative
    u = v - lower
    target_sum = target_sum - n * lower
    
    if target_sum <= 0:
        return np.full(n, lower)
    
    # Sort in descending order
    u_sorted = np.sort(u)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(u_sorted)
    # Determine index where threshold is reached
    idx = np.argmax(u_sorted > (target_sum - cumsum + u_sorted) / np.arange(1, n+1))
    
    # Compute threshold
    if u_sorted[idx] > 0:
        # Use Newton's method or direct approach
        lambda_ = (cumsum[idx-1] - target_sum) / idx if idx > 0 else target_sum
        lambda_ = max(lambda_, 0)  # Ensure non-negative
    else:
        lambda_ = target_sum / n # Simple average case
    
    # Compute projected values
    x = np.maximum(np.minimum(u - lambda_, upper - lower), 0) + lower
    return x

def gradient_ascent_step(h, S, min_idx, step_size=0.01):
    """Perform a gradient ascent step to increase the minimal S_k"""
    n = len(h)
    min_k = min_idx
    g = grad_S_k(h, min_k)
    # Update h in direction that increases S_min
    h_new = h + step_size * g
    # Project back onto feasible region [0,1] with fixed sum
    h_projected = project_onto_simplex_with_bounds(h_new, np.sum(h), 0, 1)
    return h_projected

def optimize_step_function(seed=42, n_points=48):
    """Optimize a step function to minimize C_5."""
    np.random.seed(seed)
    h0 = np.full(n_points, 0.5)  # Start with uniform distribution
    h0 = project_onto_simplex_with_bounds(h0, n_points/2, 0, 1)
    best_h = h0.copy()
    initial_c5 = evaluate_h(h0)
    best_c5 = initial_c5
    
    # Parameters for gradient ascent
    max_iter = 5000
    step_size = 0.01
    decay_rate = 0.95  # For step size decay
    
    for i in range(max_iter):
        # Compute autocorrelations
        S = compute_autocorr(best_h)
        # Find the index of minimum S_k (excluding k=0)
        min_k_idx = np.argmin(S[1:]) + 1  # Avoid k=0
        min_S = S[min_k_idx]
        
        # Early termination if we're already good enough
        if best_c5 <= target_bound:
            break
            
        # Take gradient ascent step
        h_candidate = gradient_ascent_step(best_h, S, min_k_idx, step_size)
        
        # Evaluate the new solution
        c5_candidate = evaluate_h(h_candidate)
        
        # Accept if improvement or with some probability
        if c5_candidate <= best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            step_size = min(step_size * decay_rate, 0.1)
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = []
    best_c5 = float('inf')
    best_n_points = 48
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Try optimized solution first
        try:
            h_current, c5_current = optimize_step_function(seed, n_points)
            if c5_current < best_c5:
                best_h = h_current
                best_c5 = c5_current
                best_n_points = n_points
        except Exception:
            continue
    
    # Final refinement with SLSQP if time permits
    if time.time() - start_time < budget_s * 0.99:
        try:
            # Define bounds
            bounds = [(0, 1) for _ in range(best_n_points)]
            
            # Define constraints
            def constraint_eq(h):
                return np.sum(h) * (2.0 / best_n_points) - 1.0
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            # Refine using local optimization
            result = minimize(lambda x: evaluate_h(x), best_h, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8})
            
            if result.success:
                h_refined = result.x
                c5_refined = evaluate_h(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    return list(best_h), best_c5, best_n_points
