    def project_simplex(y, s):
        # Project y onto {x : 0 ≤ x_i ≤ 1, Σ x_i = s}
        # Using bisection as described https://arxiv.org/pdf/1101.6081.pdf
        y = np.array(y)
        low, high = -np.max(y)-1, np.max(y)+1
        def F(lam):
            return np.sum(np.clip(y - lam, 0, 1)) - s
        for _ in range(40):
            mid = (low + high) / 2
            val = F(mid)
            if val > 0:
                low = mid
            else:
                high = mid
        lam = (low + high) / 2
        x = np.clip(y - lam, 0, 1)
        return x
    
    # Function to compute C5 directly
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: SDP solve for multiple dimensions
    for n in n_values:
        if (time.time() - start_time) > (budget_s - 10):
            break
        print(f"Computing SDP for n={n}...")
        try:
            # Variable and problem for SDP relaxation
            X = Semidef(n)  # n x n PSD matrix
            h = Variable(n)
            m = Variable()
            
            # Constraints: X[i,i] = h[i]
            constraints = [X[i,i] == h[i] for i in range(n)]
            
            # Constraints: 0 <= h[i] <= 1
            constraints += [h[i] >= 0 for i in range(n)]
            constraints += [h[i] <= 1 for i in range(n)]
            
            # Constraints: sum(h) = n/2
            constraints.append(sum(h) == n/2)
            
            # Add r_s >= m for all shifts
            # r_s defined over index s in -n+1 to n-1. But we only
            # consider s from 0 to n-1. So compute full overlaps and enforce.
            # Max bounds on s: from -n+1 to n-1
            P = np.eye(n)  # For fast indexing
            for s in range(-(n-1), n): # for each possible shift s
                # Construct indices for valid pairs
                idx1 = np.arange(max(0, -s), min(n, n-s))  # i
                idx2 = np.arange(max(0, s), min(n, n+s))  # i+s
                if len(idx1) == 0:
                    continue
                # Sum over valid pairs: X[i, i+s]
                r_s = sum([X[idx1[i], idx2[i]] for i in range(len(idx1))])
                constraints.append(r_s >= m)
                
            # Objective: maximize m
            objective = Maximize(m)
            prob = Problem(objective, constraints)
            
            # Solve with ECOS or SCS; both should work in this environment
            prob.solve(solver=ECOS, verbose=False)

            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                m_opt = prob.value
                X_opt = X.value
                h_opt = h.value                
                # Project h_opt onto capped simplex to make it feasible

                # Compute C5 for this h_opt

                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
                    print(f"Improved C5 = {c5:.6f} for n={n}")
            else:
                print(f"SDP for n={n} failed: {prob.status}")
        except Exception as e:

            best_h[15:20] = 0.3
            best_h = project_simplex(best_h, 30/2)
            best_c5 = evaluate_c5(best_h)

    # Phase 2: Local Refinement
    # We'll do Nelder-Mead as a local search around the best vector
    def obj(h):

    if (time.time() - start_time) < (budget_s - 20):

            # Create initial small 1D search version
