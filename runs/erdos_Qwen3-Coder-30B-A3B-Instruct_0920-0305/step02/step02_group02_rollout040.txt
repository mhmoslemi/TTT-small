import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

# Constants
n_points = 60  # Selected larger n for better resolution
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def objective(h_values):
    """Objective function to minimize: C_5 bound."""
    return evaluate_h(h_values)

def constraint_eq(h_values):
    """Equality constraint: integral of h must be 1."""
    return np.sum(h_values) * dx - 1.0

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project vector v onto simplex with bounds [lower, upper] and sum = target_sum."""
    v = np.array(v)
    # Simplex projection algorithm
    # Shift to [0, 1] space first
    v_shifted = v - lower
    upper_shifted = upper - lower
    target_shifted = target_sum - len(v) * lower
    
    # Handle edge case where target_shifted is outside [0, len(v)]  
    if target_shifted < 0: target_shifted = 0
    if target_shifted > len(v) * upper_shifted: target_shifted = len(v) * upper_shifted

    # Constrained simplex projection:
    # Sort in descending order
    sorted_v = np.sort(v_shifted)[::-1]
    # We'll use geometric approach to find the threshold lambda
    # such that projected vector sums to target.
    n = len(sorted_v)
    # Weights sum to 1: sort weights in ascending order, then find threshold
    cumsum = np.cumsum(sorted_v)
    # Properly compute lambda
    lambda_vals = (cumsum - target_shifted) / np.arange(1, n+1)
    # Find largest index where lambda index is at least sorted_v[index-1]
    # Start from right, find last valid index
    idx = -1
    for i in range(n-1, -1, -1):
        if lambda_vals[i] <= sorted_v[i]:
            idx = i
            break
    # If found, compute lamba
    if idx >= 0:
        lambda_val = lambda_vals[idx]
    else:
        lambda_val = 0  # fallback
        
    # Prepare projected vector
    projected = np.maximum(np.minimum(v_shifted - lambda_val, upper_shifted), 0) + lower
    
    # Ensure exact sum
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Adjust to ensure sum is exactly target_sum
        # Try adjusting last element (single smallest perturbation)
        # But simpler: proportional correction
        adjusted = projected + diff / len(projected)  # Minimal correction evenly to all
        # Clamp to bounds (while keeping sum correct)
        projected = np.clip(adjusted, lower, upper)
        
    return projected

def gradient_ascent_step(h, k_min, step_size):
    """Perform gradient ascent step for minimal S_k."""
    n = len(h)
    if k_min == 0:
        # Gradient of S_0 = sum of squares
        grad = 2 * h
    else:
        grad = np.zeros_like(h)
        if k_min < n:
            # h_i contributes to gradient if i+k < n: h_{i+k}
            grad[:n-k_min] += h[k_min:]
            # and if i-k >=0: h_{i-k}
            grad[k_min:] += h[:n-k_min]
    # Update with step
    new_h = h + step_size * grad
    return new_h

def optimize_step_function(seed=42, max_iters=2000):
    """Optimize a step function to minimize C_5 using gradient ascent."""
    np.random.seed(seed)
    
    # Start with random initial solution
    h0 = np.random.rand(n_points)
    h0 = project_onto_simplex_with_bounds(h0, 1.0, 0.0, 1.0)
    initial_c5 = evaluate_h(h0)
    best_h = h0.copy()
    best_c5 = initial_c5

    start_time = time.time()
    overall_max_iters = max_iters
    
    iter_count = 0
    
    while iter_count < overall_max_iters and (time.time() - start_time) < 590:
        iter_count += 1
        
        # Compute all auto-correlations for each shift        
        # S_k = sum_i h_i * h_{i+k} 
        h = best_h
        full_corr = np.correlate(h, h, mode="full")
        # Extract S_k for non-negative shifts
        S_k_values = full_corr[len(h)-1:]  # includes k=0 to n-1
        
        # Find minimum S_k (excluding k=0)
        # k=0 gives sum of squares, not the min we want
        min_k_idx = np.argmin(S_k_values[1:]) + 1  # because we skip index 0
        min_S = S_k_values[min_k_idx]
        
        # Use gradient ascent on S_min to increase it
        step_size = 0.01 / (1.0 + iter_count / 100)
        new_h = gradient_ascent_step(h, min_k_idx, step_size)
        
        # Project back onto feasible simplex with bounds
        new_h = project_onto_simplex_with_bounds(new_h, 1.0, 0.0, 1.0)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h)
        
        # Accept new solution if better
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Refine solution using scipy's constrained minimizer with SLSQP
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                          constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-10})
        if result.success:
            final_h = result.x
            final_c5 = evaluate_h(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception as e:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    """Main function to run optimization."""
    start_time = time.time()
    
    # Run optimization
    h_values, c5_bound = optimize_step_function(seed)
    
    # Final evaluation to make sure constraints are satisfied
    h_values = np.clip(h_values, 0, 1)
    if abs(np.sum(h_values) * dx - 1.0) > 1e-6:
        scale = 1.0 / (np.sum(h_values) * dx)
        h_values *= scale
        
    final_c5 = evaluate_h(h_values)
    
    return list(h_values), final_c5, n_points
