def project_to_simplex(y, target_sum):
    """Project y onto the capped simplex: 0 <= x <= 1, sum x = target_sum"""
    n = len(y)
    y = np.copy(y)
    # Bisect on lambda to find the correct projection
    low = -np.max(y) - 1
    high = np.max(y) + 1
    tol = 1e-10
    max_iter = 40
    for _ in range(max_iter):
        mid = (low + high) / 2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(y - (low + high) / 2, 0, 1)
    scale = target_sum / np.sum(x)
    x = x * scale
    return x

            r = sum(X_val[i+s, i] for i in range(-s, n))
