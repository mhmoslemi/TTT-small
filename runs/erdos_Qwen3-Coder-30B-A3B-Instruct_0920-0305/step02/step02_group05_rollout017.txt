def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_to_feasible(h, n_points):
    """Project h to feasible set: 0 <= h_i <= 1 and sum(h) = n/2"""
    # Clip to [0, 1]
    w = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n_points / 2.0
    current_sum = np.sum(w)
    diff = target - current_sum
    
    # Water-filling algorithm
    while abs(diff) > 1e-10:
        if diff > 0:  # Need to add more mass
            # Find indices where w_i < 1
            free_indices = np.where(w < 1)[0]
            if len(free_indices) == 0:
                break
            # Distribute diff equally
            incr = diff / len(free_indices)
            for i in free_indices:
                new_val = w[i] + incr
                if new_val > 1:
                    diff -= (1 - w[i])
                    w[i] = 1
                else:
                    w[i] = new_val
                    diff = 0
        else:  # Need to subtract mass
            # Find indices where w_i > 0
            free_indices = np.where(w > 0)[0]
            if len(free_indices) == 0:
                break
            # Subtract diff equally
            decr = -diff / len(free_indices)
            for i in free_indices:
                new_val = w[i] - decr
                if new_val < 0:
                    diff -= w[i]
                    w[i] = 0
                else:
                    w[i] = new_val
                    diff = 0
    
    return w

        j_indices = i_indices - m_abs
        g[i_indices] += 1 - h[j_indices]
        g[j_indices] -= h[i_indices]
    return g * dx
