import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time
import random

# Parameters
n_values = [48, 56, 64, 80]
dx_values = [2.0 / n for n in n_values]
w_values = [n / 2 for n in n_values]
target_bound = 0.3807
alpha_max = 500.0
eps = 1e-6
max_iter = 500
n_restarts = 10
global_timeout = 600

def initial_h_uniform(n):
    """Generate uniform random initial h"""
    h = np.random.rand(n)
    h = h / np.sum(h) * (n / 2)
    return np.clip(h, 0, 1)

def initial_h_binary(n):
    """Generate binary initial h (0/1) with sum exactly n/2"""
    h = np.zeros(n)
    ones = np.random.choice(n, int(n / 2), replace=False)
    h[ones] = 1.0
    h = np.clip(h, 0, 1)
    return h

def initial_h_three_block(n):
    """Generate a three-block initial h: 0, p, 1"""
    a = np.random.uniform(0, 1)
    b = np.random.uniform(a, 2)
    p = (n / 2 - (1 - a) * n / 2) / (b - a)
    h = np.zeros(n)
    h[:int(a * n / 2)] = 0
    h[int(a * n / 2):int(b * n / 2)] = p
    h[int(b * n / 2):] = 1
    return np.clip(h, 0, 1)

def initial_h_perturbed_best(n, best_h):
    """Perturb the best h found so far"""
    h = best_h.copy()
    i, j = np.random.choice(n, 2, replace=False)
    delta = np.random.normal(0, 0.02)
    h[i] = np.clip(h[i] + delta, 0, 1)
    h[j] = np.clip(h[j] - delta, 0, 1)
    h = h / np.sum(h) * (n / 2)
    return np.clip(h, 0, 1)

def projection_simplex(h, w):
    """Project h onto capped simplex: 0 <= h[i] <= 1, sum(h) = w"""
    u = np.sort(h)[::-1]
    rho = 0
    for i in range(len(u)):
        if u[i] - (np.sum(u[:i+1]) - w) / (i+1) > 0.0:
            rho = i + 1
    theta = (np.sum(u[:rho]) - w) / rho
    h = np.clip(h - theta, 0, 1)
    return h

def calculate_S_k(h):
    """Compute auto-correlation S_k for all k via FFT"""
    n = len(h)
    h_fft = fft(h, 2*n)
    S_full = ifft(h_fft * np.conj(h_fft)).real
    S_k = S_full[n-1 : 2*n-1]
    return S_k

def calculate_r_k(h, w):
    """Given a vector h and width w, compute r_k = w - S_k"""
    S_k = calculate_S_k(h)
    return w - S_k

def lse_smooth(h, alpha, w):
    """Compute log-sum-exp of r_k with smoothness parameter alpha"""
    r_k = calculate_r_k(h, w)
    # Shift to prevent overflow
    max_r_k = np.max(r_k)
    shifted_r = r_k - max_r_k
    exps = np.exp(alpha * shifted_r)
    return np.log(np.sum(exps)) / alpha + max_r_k / alpha

def gradient_r_k(h, k):
    """Compute gradient of r_k with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        # r_0 = w - sum h_i * h_{i+0}
        grad = np.ones(n) - 2 * h
    else:
        # r_k = w - sum h_i * h_{i+k}
        grad -= np.ones(n)
        if k < n:
            grad[k:] += h[:-k]
        if k > 0:
            grad[:-k] += h[k:]
    return grad

def compute_gradient_F_alpha(h, alpha, w):
    """Compute gradient of smooth surrogate using softmax weights"""
    r_k = calculate_r_k(h, w)
    # Shift for numerical stability
    max_r_k = np.max(r_k)
    shifted_r = r_k - max_r_k
    w_k = np.exp(alpha * shifted_r)
    total_w = np.sum(w_k)
    w_k /= total_w
    
    g = np.zeros_like(h)
    for i in range(len(r_k)):
        g += w_k[i] * gradient_r_k(h, i)
    return g / alpha

def accelerated_projected_gradient(h0, alpha, w, max_iter=300, eps=1e-6):
    """FISTA-based accelerated projected gradient for smooth surrogate"""
    h = h0.copy()
    y = h.copy()
    t = 1
    
    for iteration in range(max_iter):
        g = compute_gradient_F_alpha(h, alpha, w)
        eta = 1.0
        while True:  # backtracking line search
            h_new = y - eta * g
            h_new = projection_simplex(h_new, w)
            if (lse_smooth(h_new, alpha, w) - lse_smooth(h, alpha, w) + 
                np.sum(g * (h_new - h)) + 0.5/eta * np.sum((h_new - h)**2) < 0):
                break
            eta *= 0.5
            
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y = h_new + ((t - 1) / t_new) * (h_new - h)
        
        if np.abs(lse_smooth(h_new, alpha, w) - lse_smooth(h, alpha, w)) / (1e-10 + lse_smooth(h, alpha, w)) < eps:
            break
            
        h = h_new.copy()
        t = t_new
    return h

def optimize_smoothed(h0, w, alpha_max, eps=1e-6):
    """Coordinate descent on alpha with continuation"""
    h = h0.copy()
    alpha = 1.0
    while alpha <= alpha_max:
        h = accelerated_projected_gradient(h, alpha, w, max_iter=max_iter, eps=eps)
        alpha *= 2.0
    return h

def compute_exact_C5(h, dx):
    """Compute the exact C5 with standard correlate"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_C5 = float('inf')
    best_h = None
    best_n = None
    
    random.seed(seed)
    np.random.seed(seed)
    n_idx = 0
    
    while time.time() - start_time < budget_s * 0.95 and n_idx < len(n_values):
        n = n_values[n_idx]
        dx = dx_values[n_idx]
        w = w_values[n_idx]
        n_idx += 1
        
        for _ in range(n_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Try several initialization strategies
            strategies = [
                initial_h_uniform,
                initial_h_binary,
                initial_h_three_block,
            ]
            
            strategy = random.choice(strategies)
            if strategy == initial_h_three_block:
                h = strategy(n)
            else:
                h = strategy(n)
                
            if strategy == initial_h_binary:
                h = np.clip(h, 0, 1)
                h = projection_simplex(h, w)
            elif strategy == initial_h_uniform:
                h = np.clip(h, 0, 1)
                h = projection_simplex(h, w)
            else:
                h = projection_simplex(h, w)
                
            # Continuation optimization
            h_opt = optimize_smoothed(h, w, alpha_max, eps)
            
            # Final evaluation
            C5 = compute_exact_C5(h_opt, dx)
            
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_opt
                best_n = n
                
    # Final adjustments and return
    if best_h is not None:
        final_h = projection_simplex(best_h, w)
        final_C5 = compute_exact_C5(final_h, 2.0 / best_n)
        return (list(final_h), final_C5, best_n)
    else:
        # Return some default if no solution found
        default_h = np.array(initial_h_uniform(48))
        default_C5 = compute_exact_C5(default_h, 2.0 / 48)
        return (list(default_h), default_C5, 48)
