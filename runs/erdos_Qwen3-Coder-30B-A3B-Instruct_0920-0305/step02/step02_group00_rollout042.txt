def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_onto_box_and_hyperplane(v, target_sum):
    """Project vector v onto [0,1]^n and then adjust to satisfy sum = target_sum."""
    # find lambda such that sum(clamp(v - lambda, 0, 1)) == target_sum
    # this uses bisection on the monotone decreasing function sum(clamp(v - lambda, 0, 1))  
    def f(lambda_val):
        clipped = np.clip(v - lambda_val, 0, 1)
        return np.sum(clipped) - target_sum

    a, b = np.min(v), np.max(v) + 1.0
    while b - a > 1e-12:
        m = (a + b) / 2.0
        if f(m) > 0:
            a = m
        else:
            b = m
    lambda_val = (a + b) / 2.0
    return np.clip(v - lambda_val, 0, 1)

def soft_min_gradient(h, β):
    """Evaluate gradient of soft minimum log-sum-exp of correlations for shift s = 0..n-1."""
    n = len(h)
    dx = 2.0 / n
    
    # Correlations for all shifts using convolution
    corr_coefs = np.correlate(h, 1 - h, mode='full')  # length 2*n - 1
    C_s = corr_coefs * dx
    
    # numerical stability
    C_s -= np.max(C_s)
    exp_neg_beta_C = np.exp(-β * C_s)
    Z = np.sum(exp_neg_beta_C)
    weights = exp_neg_beta_C / Z
    
    # compute weighted sum of gradients: for shift s, d/dh (h^T A^{(s)} h) = 2 * A^{(s)} h
    # A^{(s)} is symmetric and sparse: (A^{(s)})_{ij} = 0.5 if |i-j| = s, else 0
    # We'll compute 2*dx * sum_s w_s * (A^{(s)} h) efficiently:
    g = np.zeros(n)
    for s in range(n):
        w = weights[s]
        if s == 0:
            g += w * h
        else:
            # roll h to right by s, and vice versa
            h_shifted_l = np.roll(h, s)
            h_shifted_r = np.roll(h, -s)
            g += 0.5 * w * (h_shifted_l + h_shifted_r)
    
    return 2 * dx * g

def gradient_ascent_step(h, β, max_iter=100):
    """Perform gradient ascent step using backtracking line search."""
    dx = 2.0 / len(h)
    current_max_L = evaluate_h(h)
    g = soft_min_gradient(h, β)
    alpha = 0.5  # initial step size
    for _ in range(max_iter):
        h_proposed = h + alpha * g
        h_projected = project_onto_box_and_hyperplane(h_proposed, 1.0)
        proposed_max_L = evaluate_h(h_projected)
        if proposed_max_L < current_max_L:
            h = h_projected
            current_max_L = proposed_max_L
            break
        else:
            alpha *= 0.5  # backtrack 
    return h, current_max_L
