import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def project_to_capped_simplex(y, target_sum):
    """Project y onto the capped simplex: {x: 0 <= x_i <= 1, sum x_i = target_sum}"""
    y = np.array(y)
    low, high = -max(y) - 1, max(y) + 1
    tol = 1e-12
    for _ in range(40):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s >= target_sum:
            low = mid
        else:
            high = mid
    x = np.clip(y - low, 0, 1)
    scale = target_sum / np.sum(x)
    return x * scale

                X[i, i] == h[i] for i in range(n)

                if s >= 0:

                    r = sum(X[i, i+s] for i in range(n - s))
                else:
                    # s < 0, so shift is positive
                    r = sum(X[i+s, i] for i in range(-s, n))
