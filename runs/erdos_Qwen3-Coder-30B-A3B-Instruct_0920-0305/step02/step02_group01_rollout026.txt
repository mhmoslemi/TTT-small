def project_simplex(y, S):
    """Project y onto the capped simplex: {x | 0 <= x <= 1, sum x = S}"""
    y = np.asarray(y)
    low, high = -np.max(y) - 1, np.max(y) + 1
    for _ in range(40):
        mid = (low + high) / 2
        x = np.clip(y - mid, 0, 1)
        s = np.sum(x)
        if s > S:
            low = mid
        else:
            high = mid
    x = np.clip(y - low, 0, 1)
    scale = S / np.sum(x)
    return x * scale

def compute_C5(h):
    """Compute the exact C5 value for discrete h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
