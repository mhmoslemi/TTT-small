def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """Project vector y onto the capped simplex with sum = target_sum"""
    n = len(y)
    # Bisection to find lambda
    def F(lam):
        x = np.clip(y - lam, 0, 1)
        return np.sum(x) - target_sum
    
    # Initial bracketing
    low, high = -np.max(y) - 1, np.max(y) + 1
    for _ in range(100):
        mid = (low + high) / 2
        if F(mid) > 0:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2
    # Compute the final projection
    x = np.clip(y - lam, 0, 1)
    # Correct sum exactly (to avoid floating point errors)
    scale = target_sum / np.sum(x)
    x *= scale
    return x
