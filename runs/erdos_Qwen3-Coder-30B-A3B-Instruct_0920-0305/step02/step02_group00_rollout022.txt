import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_points_list = [48, 56, 60]
dx = 2.0 / n_points_list[0]
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_constraint(v, target_sum):
    """Project v onto [0,1]^n and then onto the constraint sum(h) = target_sum."""
    # Bisection to find lambda such that sum(clamp(v - lambda, 0, 1)) = target_sum
    l, u = np.min(v), np.max(v) + 1.0
    for _ in range(50):
        mid = (l + u) / 2.0
        x = np.clip(v - mid, 0, 1)
        if np.sum(x) > target_sum:
            l = mid
        else:
            u = mid
    return np.clip(v - l, 0, 1)

def compute_soft_min_weights(C_values, beta):
    """Compute softmax weights for the soft min."""
    C_values = np.array(C_values)
    # Subtract max for numerical stability
    C_shifted = C_values - np.max(C_values)
    exp_vals = np.exp(-beta * C_shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    return weights

def compute_correlations(h):
    """Compute all C_s values for given h."""
    # Using np.correlate in full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Return the overlapping parts (C_s values for s=0 to n-1)
    C_vals = corr[len(corr)//2:]
    C_vals = C_vals * dx
    return C_vals

def gradient_of_soft_min(h, C_vals, beta):
    """Compute gradient of soft-min of correlations."""
    n = len(h)
    if len(C_vals) != n:
        raise ValueError("Length of C_vals should equal n")
    
    weights = compute_soft_min_weights(C_vals, beta)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # For each shift s, compute A^{(s)} h efficiently
    for s in range(n):
        # A_s h = [0.5*(h_{i-s} + h_{i+s}) for valid i], with zero padding
        if s == 0:
            # For s=0, A_s = I (identity), so A_s h = h
            grad += weights[s] * h
        else:
            # A_s h = (h[s:] + h[:-s]) / 2
            # Pad to same length and add contribution
            h_shifted_forward = np.concatenate([np.zeros(s), h[:-s]])
            h_shifted_backward = np.concatenate([h[s:], np.zeros(s)])
            A_h = (h_shifted_forward + h_shifted_backward) * 0.5
            grad += weights[s] * A_h
    
    return 2 * dx * grad

def line_search(h_current, grad, max_iter=10, alpha_init=0.5):
    """Simple backtracking line search."""
    alpha = alpha_init
    h = h_current.copy()
    # Compute current objective value
    C_vals = compute_correlations(h)
    t_current = np.max(1 - C_vals)
    
    for _ in range(max_iter):
        h_new = h + alpha * grad
        h_new = project_onto_box_and_constraint(h_new, len(h)/2.0)
        C_vals_new = compute_correlations(h_new)
        t_new = np.max(1 - C_vals_new)
        if t_new < t_current:
            return h_new, alpha, t_new
        alpha *= 0.5
    return h, alpha, t_current

def multi_start_gradient_ascent(n_points, budget_s, seed=42):
    """Run gradient ascent from multiple random starting points."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = n_points
    dx = 2.0 / n_points
    
    # Generate diverse initial points
    initial_points = []
    
    # 1. Linearly interpolate the initial_h_values
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_interp = np.interp(np.linspace(0, len(h_init)-1, n_points), 
                             np.linspace(0, len(h_init)-1, len(h_init)), h_init)
        initial_points.append(("interp", h_interp))
    
    # 2. Random binary pattern with weight sum
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_proj = project_onto_box_and_constraint(h_rand, n_points / 2)
        initial_points.append(("random_bin", h_proj))
    
    # 3. Random uniform + projection
    for _ in range(2):
        h_rand = np.random.rand(n_points)
        h_proj = project_onto_box_and_constraint(h_rand, n_points / 2)
        initial_points.append(("uniform", h_proj))
    
    # Continue with our method
    # Load the initial_h_values if available
    n_steps = 100
    beta_schedule = [5.0, 15.0, 30.0, 60.0, 120.0]
    
    for name, h_init in initial_points:
        h = h_init.copy()
        
        # If time is almost up, break early
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # For this starting point, do full continuation
        for beta in beta_schedule:
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Inner loop for fixed beta
            for step in range(n_steps):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                C_vals = compute_correlations(h)
                grad = gradient_of_soft_min(h, C_vals, beta)
                
                h_updated, _, t_current = line_search(h, grad)
                h = h_updated
                
                # Early stopping if last few steps didn't improve
                if step > 10 and abs(t_current - best_c5) < 1e-8:
                    break
                    
                if t_current < best_c5:
                    best_c5 = t_current
                    best_h = h.copy()
                    best_n = n_points
                    
    return best_h, best_c5, best_n

def run(seed=42, budget_s=600):
    best_h = None
    best_c5 = float('inf')
    best_n = 48
    
    np.random.seed(seed)
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        h_now, c5_now, n_now = multi_start_gradient_ascent(n_points, budget_s, seed + n_points)
        
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h_now
            best_n = n_now
            
    if best_h is None:
        # If nothing found, fallback to initial construction
        best_h = np.array(initial_h_values)
        best_c5 = evaluate_h(best_h)
        best_n = len(best_h)
        
    # Final refinement with SLSQP
    if time.time() - start_time < 0.9 * budget_s:
        def objective(x):
            return evaluate_h(x)
        bounds = [(0, 1) for _ in range(best_n)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints,
                              options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
        except:
            pass
        
        # Final evaluation after SLSQP
        best_c5 = evaluate_h(best_h)
    
    return list(best_h), best_c5, best_n
