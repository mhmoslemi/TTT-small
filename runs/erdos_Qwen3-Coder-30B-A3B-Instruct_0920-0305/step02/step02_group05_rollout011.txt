import numpy as np

def project_to_bounds_and_sum(v, target_sum):
    """Project v onto the constraints: 0 <= v <= 1 and sum(v) = target_sum"""
    # Clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase values
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            # Need to decrease values
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient of A_m with respect to h"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        # Overlapping indices i = 0..n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # For m < 0
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx = 2.0 / n

    max_val = np.max(conv)
    argmax = np.argmax(conv)
    # Convert conv index to shift value
    shift = argmax - (len(h) - 1)
    return max_val, shift
