from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def optimize_with_gd(h0, bounds, n_points, max_iter=1000):
    """Optimize using gradient-based method"""
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_eq(h):
        dx = 2.0 / n_points
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-14})
        if res.success:
            dx = 2.0 / n_points
            scale = 1.0 / (np.sum(res.x) * dx)
            return res.x * scale, objective(res.x * scale)
    except:
        pass
    return h0, evaluate_c5(h0)
