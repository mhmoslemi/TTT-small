import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(y, target_sum):
    """Project vector y onto the capped simplex: 0 ≤ x_i ≤ 1, sum x_i = target_sum"""
    y = np.asarray(y)
    low, high = 0.0, 1.0
    n = len(y)
    
    def F(lam):
        x = np.clip(y - lam, low, high)
        return np.sum(x) - target_sum
    
    # Bisection to find lambda such that F(lambda) = 0
    # Bracket the root
    lam_low = np.max(-y) - 1
    lam_high = np.max(y) + 1
    
    for _ in range(40):
        lam = (lam_low + lam_high) / 2
        f_val = F(lam)
        if f_val > 0:
            lam_low = lam
        else:
            lam_high = lam
    
    lam = (lam_low + lam_high) / 2
    x = np.clip(y - lam, low, high)
    
    # Ensure exact sum by adjusting one element (if needed)
    diff = target_sum - np.sum(x)
    if abs(diff) > 1e-10:
        # Add diff uniformly to all
        x += diff / n
        x = np.clip(x, low, high)
    
    return x

def extract_h_from_X(X, target_sum):
    """Extract feasible h from PSD matrix X"""
    # Compute leading eigenvector
    vals, vecs = np.linalg.eigh(X.value)
    v = vecs[:, -1]
    # Rescale to match sum
    h = np.abs(v)
    scale = target_sum / np.sum(h)
    h = h * scale
    # Clip to [0,1] and adjust sum to match exactly
    h = np.clip(h, 0, 1)
    h = project_to_capped_simplex(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # System for solving SDP, extracting h, local refinement
    best_h = None
    best_c5 = float('inf')
    
    # For now, use small n that fits SDP
    n_list = [30, 35, 40, 45, 50]  # From strategy
    n_target = 180  # Final resolution
    
    # SDP based search
    for n in n_list:
        if budget_remaining() < 5:
            break

            # Make SDP for this n
            h_var = cp.Variable(n)
            X = cp.Variable((n, n), symmetric=True)

            constraints = [
                cp.diag(X) == h_var,
                X >> 0,
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n / 2
            ]
            
            # Define r_s (autocorrelations) - for all s
            # Add scalar variable m
            m = cp.Variable()
            # For each s compute relevant r_s
            for s in range(-(n - 1), n):
                # indices to sum
                if s >= 0:
                    r_s = cp.sum([X[i, i + s] for i in range(n - s)])
                else:
                    r_s = cp.sum([X[i, i - s] for i in range(-s, n)])
                constraints.append(r_s >= m)
            
            # Objective
            prob = cp.Problem(cp.Maximize(m), constraints)
            # Solve with SCS or ECOS -- default solver should be okay if available
            prob.solve(solver=cp.SCS, verbose=False, max_iters=3000, eps=1e-5)
            
            if prob.status != cp.OPTIMAL:
                continue
                
            # Extract feasible h
            h_extracted = extract_h_from_X(X, n / 2)
            c5_extracted = evaluate_c5(h_extracted)
            if c5_extracted < best_c5:
                best_h = h_extracted.copy()
                best_c5 = c5_extracted
                
        except Exception as e:
            pass  # Continue to next n

    # 1. Local refinement on best low-dim vector
    if budget_remaining() > 10:

        # Interpolate the best low-dim signal onto n_target points
        n_low = len(best_h)
        scale_factor = n_target / n_low
        x_low = np.linspace(0, 2, n_low)
        x_up = np.linspace(0, 2, n_target)
        # Using piecewise constant interpolation (step function)
        h_up = np.interp(x_up, x_low, best_h, left=0.0, right=0.0)
        # Ensure integral is correct by projecting to capped simplex
        h_up = project_to_capped_simplex(h_up, n_target / 2)
        # Final evaluation
        final_c5 = evaluate_c5(h_up)
        if final_c5 < best_c5:

    # Finally, fine-tune on highest resolution if still room
    if budget_remaining() > 30:

        try:
            def objective(h_vec):

            res_final = minimize(objective, h_final, method='Nelder-Mead',

                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
        except:
            pass  # If failed, keep previous best
    
    return (best_h.tolist(), float(best_c5), len(best_h))
