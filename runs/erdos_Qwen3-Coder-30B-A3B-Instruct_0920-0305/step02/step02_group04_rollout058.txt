import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Pad h and 1-h to length 2*n
    h_pad = np.pad(h, (0, n), 'constant')
    h1_pad = np.pad(1 - h, (0, n), 'constant')
    # FFT and correlation
    H = fft(h_pad)
    H1 = fft(h1_pad)
    corr = ifft(H * np.conj(H1)).real[:2*n-1]
    # Max over all lags
    max_val = np.max(corr) * dx
    return max_val

    # Note: enumerate all shift values from 1 to n-1
    # For each shift ℓ, associate constraint for product terms
    # p[ℓ][i] = h[i] * h[i+ℓ]

            p = cp.Variable(boolean=True)
            p_vars[(l, i)] = p
            # McCormick envelopes for p = h[i] * h[i+l]

            constraints.append(p <= h_vars[i + l])
            constraints.append(p >= h_vars[i] + h_vars[i + l] - 1)
            # For the max over lags:

            # where the sum is defined as ∑ p[l][i] over valid i

            # We'll compute this later for each l independently
        # At the end add the constraint that:
        # ∑ p[l][i] >= M - T

    # entries of ∑ p[l][i] over all lags l

    objective = cp.Minimize(T)

    # as a linear function of p[·] terms, and relate it via added constraint

            # Since the sum_expr is integer and M is integer
            # The = is not captured directly but we ensure that the min/max over lags are captured
            # We compute the sum and add undirectly that sum + T >= M via a big constraint per lag
            lag_sum.append(sum_expr)

    # Add a term to maximize by setting a fixed sum over all p for shift l.

    # We sum all such [i] for valid ℓ over all ℓ. For simplicity, represent it as sum over shifting pairs

                grad, value = smooth_max(h_current, beta)
                step = 0.001  # enough to capture small changes

                # Clip back

                # Maintain mass constraint

                h_prop = h_prop * (M / mass)
                h_current = h_prop

        h_final = h_current
        C5_final = evaluate_h(h_final)

    # Final verification

    final_mass = np.sum(h_final) * dx
    if abs(final_mass - 1.0) > 1e-9:
        h_final = h_final * (1.0 / final_mass)
    C5_final = evaluate_h(h_final)

    return (h_final.tolist(), C5_final, n)
