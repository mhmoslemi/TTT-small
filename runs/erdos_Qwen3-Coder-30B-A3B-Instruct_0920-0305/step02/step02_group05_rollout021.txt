import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto bounds [0,1] and sum constraint"""
    # Clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def compute_gradient_for_shift(h, m):
    """Compute gradient for the overlap at shift m"""
    n = len(h)
    g = np.zeros_like(h)
    
    if m >= 0:
        # Overlapping indices for positive shift
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # Negative shift case
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    
    # Scale by dx = 2/n
    return g * (2.0 / n)

def compute_max_overlap(h):
    """Compute maximum overlap integral and corresponding shift"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(conv)
    max_idx = np.argmax(conv)
    # Compute shift from index (centered at n-1)
    shift_m = max_idx - (n - 1)
    return max_val, shift_m
