import numpy as np
from scipy.fft import fft, ifft
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2n to avoid circular convolution effects
    h_pad = np.pad(h, (0, n), mode='constant')
    g_pad = np.pad(1 - h, (0, n), mode='constant')
    H = fft(h_pad)
    G = fft(g_pad)
    corr = ifft(H * np.conj(G))
    corr_real = np.real(corr[:2*n-1])
    max_val = np.max(corr_real) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n = 48  # Even number of points
    dx = 2.0 / n
    T_target = int(np.ceil(0.3807 * n / 2))  # Target integer bound

    best_c5 = evaluate_h(best_h)
    h_bin_best = None
    T_best = float('inf')
    
    # Phase 1: Solve MILP for binary h
    if time.time() - start_time < budget_s * 0.33:
        try:
            # CVXPY variables
            h = cp.Variable(n, boolean=True)
            p = cp.Variable((n, n-1), boolean=True)  # p[i][l] = h[i] * h[i+l]
            T = cp.Variable(integer=True)
            
            # Constraints
            constraints = []
            constraints.append(cp.sum(h) == n // 2)  # Mass constraint
            for l in range(1, n):
                for i in range(n - l):
                    # McCormick envelopes for p[i][l] = h[i]*h[i+l]
                    constraints.append(p[i, l - 1] <= h[i])
                    constraints.append(p[i, l - 1] <= h[i + l])
                    constraints.append(p[i, l - 1] >= h[i] + h[i + l] - 1)
                    constraints.append(p[i, l - 1] >= 0)
            
            # Worst lag constraint
            for l in range(1, n):
                # ∑ p[i][l] + T ≥ n/2
                constraints.append(cp.sum(p[:, l-1]) + T >= n // 2)
            
            # Bounds
            constraints.append(T >= 0)
            objective = cp.Minimize(T)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False, maxSeconds=budget_s * 0.32)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                T_sol = T.value
                if T_sol < T_best:
                    T_best = T_sol
                    h_bin_best = h_sol.copy()
            
        except Exception:
            pass

    # Fallback to binary heuristic if MILP failed
    if h_bin_best is None:
        h_bin_best = np.zeros(n, dtype=bool)
        # Start from initial guess, then randomly set half points to 1
        indices = np.random.choice(n, size=n//2, replace=False)
        h_bin_best[indices] = True
    
    # Phase 2: Binary local search (Simulated Annealing)
    if time.time() - start_time < budget_s * 0.66:
        current_h = h_bin_best.copy()
        current_c5 = evaluate_h(current_h.astype(float))
        t_start = time.time()
        c5_best = current_c5
        h_best = current_h.copy()
        
        # Initial temperature estimation
        delta_history = []
        for _ in range(500):
            i = np.random.randint(n)
            j = np.random.randint(n)
            while current_h[i] == 0:
                i = np.random.randint(n)
            while current_h[j] == 1:
                j = np.random.randint(n)
            new_h = current_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            new_c5 = evaluate_h(new_h.astype(float))
            delta = new_c5 - current_c5
            delta_history.append(delta)
            if delta < 0 or np.random.rand() < np.exp(-delta / (np.median(np.abs(delta_history)) * 5)):

                current_c5 = new_c5
                if current_c5 < c5_best:
                    c5_best = current_c5
                    h_best = current_h.copy()
        
        # Schedule
        temp = np.median(np.abs(delta_history)) * 5
        cooling_rate = 0.999
        
        while temp > 1e-6 and time.time() - t_start < budget_s * 0.33:
            i = np.random.randint(n)
            j = np.random.randint(n)
            while current_h[i] == 0:
                i = np.random.randint(n)
            while current_h[j] == 1:
                j = np.random.randint(n)

            new_h[j] = 1

                current_c5 = new_c5
                if current_c5 < c5_best:
                    c5_best = current_c5
                    h_best = current_h.copy()
            temp *= cooling_rate
            if time.time() - start_time > budget_s * 0.95:
                break

        h_bin_best = h_best.copy()
        best_c5 = c5_best

    # Phase 3: Fractional smoothing via projected gradient descent
    if time.time() - start_time < budget_s * 0.95:
        h_float = h_bin_best.astype(float)
        beta_schedule = [1, 2, 5, 10, 20]
        for beta in beta_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
            for step in range(20):
                # Compute gradient via FFT
                h_pad = np.pad(h_float, (0, n), mode='constant')
                g_pad = np.pad(1 - h_float, (0, n), mode='constant')
                H = fft(h_pad)
                G = fft(g_pad)
                corr_fft = H * np.conj(G)
                corr_real = np.real(ifft(corr_fft)[:2*n-1])
                lags = corr_real[:-1]  # L_0 ~ L_{n-2}
                w = np.exp(beta * lags)
                w /= w.sum()
                grad = np.zeros(n)
                for l, weight in enumerate(w):
                    if l < n-1:
                        grad[:n-l-1] += weight * (1 - h_float[l+1:] - h_float[:n-l-1])
                        
                # Perform projected gradient step
                alpha = 1e-1 / (beta + 1)
                h_new = h_float - alpha * grad
                
                # Box projection
                h_new = np.clip(h_new, 0, 1)
                
                # Mass projection via waterfilling
                mass = np.sum(h_new) * dx
                if abs(mass - 1) > 1e-9:
                    def waterfill(lam):
                        return np.sum(np.clip(h_new + lam, 0, 1)) * dx - 1
                    lam_hi = 1 - h_new.min()
                    lam_lo = -1 + h_new.max()
                    for _ in range(30):
                        lam_mid = (lam_lo + lam_hi) / 2
                        val = waterfill(lam_mid)
                        if val > 0:
                            lam_lo = lam_mid
                        else:
                            lam_hi = lam_mid
                    lam_opt = (lam_lo + lam_hi) / 2
                    h_new = np.clip(h_new + lam_opt, 0, 1)
                
                h_float = h_new
                if evaluate_h(h_float) < best_c5:
                    best_c5 = evaluate_h(h_float)
                    h_bin_best = h_float.copy()
                if time.time() - start_time > budget_s * 0.95:
                    break

    # Final scaling and cleanup
    h_float = h_bin_best.copy()
    mass = np.sum(h_float) * dx
    if abs(mass - 1) > 1e-9:
        h_float = h_float / mass
    c5_final = evaluate_h(h_float)
    return (h_float.tolist(), c5_final, n)
