import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def dc_lp_solver(h0, n_points, budget_s):
    """Use DC/Convex-Concave Procedure with LP subproblems to optimize the objective."""
    dx = 2.0 / n_points
    # Time tracking
    start_time = time.time()
    
    # Initial guess
    h = h0.copy().astype(float)
    h /= np.sum(h) * dx  # Scale to satisfy integral constraint
    h = np.clip(h, 0, 1)  # Clip to [0,1] as a safeguard
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    max_iter = 10000
    iter_count = 0
    
    # Define a small damping factor
    damping = 0.8
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.80:
            break
            
        iter_count += 1
        # Compute full autocorrelation
        auto_corr = correlate(h, h, mode="full")
        
        # the worst shift s is the one that minimizes h_i * h_{i+s}
        # We are effectively looking at negative overlap:
        # C5 = max_s [ sum(h_i) - sum(h_i * h_{i+s}) ] = 2 - min_s sum(h_i * h_{i+s})
        # Meaning: we want to (minimize) sum(h_i * h_{i+s}) over shifts
        # So we want to find shift s that minimizes sum(h_i * h_{i+s})
        # Note: sum h_i = 1*dx*n_points/n_points = 1 for consistency
        # The hcorr values are such that hcorr[shift] represents sum(h_i * h_{i+s})
        # Shifts range from -n+1 to n-1.
        # But here the correlation array from `correlate` has length 2*n - 1.
        # For example, with 4 points, the correlated array is of length 7. 
        # The linear index i corresponds to shift (i - n + 1). We are only interested in 
        # s=0 to s=n-1 to be included.
        relevant_corr = auto_corr[n-1:]  # The positive shifts from 0 to n-1
        s_star = np.argmin(relevant_corr)  # In 0, ..., n-1
        
        # Compute gradient of overlap = sum h_i * h_{i+s*}
        # In terms of `h`, we compute the gradient for shift s_star:
        h_shifted = np.roll(h, s_star, axis=0)
        gradient = h_shifted + np.roll(h, -s_star, axis=0)
        
        # Adjust for circular boundary: when s_star >= n or negative, extra handling
        # Actually, shift is now *positive*, so no need to worry here:
        # h_shifted has h_{i+s_star} for i=0..n-1
        # np.roll(h, -s_star) has h_{i-s_star} for i=0..n-1
        
        # Compute the gradient at h, i.e., compute ∂/∂ h_i (h_i * h_{i+s_star}) = h_{i+s_star} + h_{i-s_star}
        # Here we simply compute but not explicitly form it as a matrix, but instead we use
        # a linear programming approach via projecting the gradient to the feasible set.
        # We solve:
        # minimize - gradient · h  
        # subject to 0 <= h <= 1  
        #             sum(h) = n * dx (which is 1)
        gradient = gradient[:n_points]
        
        # Capped simplex projection onto { h : 0 <= h <= 1, sum(h) = 1/dx }
        # Here 1/dx = n_points/2, since dx = 2/n_points
        total_sum = n_points / 2
        sorted_grad_idx = np.argsort(gradient)[::-1]
        
        h_new = np.zeros_like(h)
        remaining_sum = total_sum
        i = 0
        
        # This is the sparse projection onto capped simplex
        while remaining_sum > 0 and i < n_points:
            gi = gradient[sorted_grad_idx[i]]
            j = sorted_grad_idx[i]
            
            # How much we can take
            if gi > 0:
                take = min(1.0, remaining_sum)
                h_new[j] = take
                remaining_sum -= take
            else:
                h_new[j] = 0.0  # won't increase it.
            i += 1
            
        # Fill the remaining
        if remaining_sum <= 0:
            pass
        else:
            # We're trying to make a non-zero sum, but we have bounded h.
            # so the set sum(h) = 1 should be close. Usually, it'd already be there.
            # But just to make sure let's distribute the leftover to some h units.
            # In practice, it should be zero unless we're slightly off due to numerical issues.
            pass
        
        # Need to still ensure sum is exactly n_points / dx = 1.0, stay within [0,1]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # Renormalization
        
        if np.isnan(np.sum(h_new)):
            print("NaN encountered in solution, stopping.")
            break
        
        # Damping if we want
        h = damping * h_new + (1 - damping) * h
        
        # Evaluate and keep improving solution
        c5_val = evaluate_c5(h)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
    
    return best_h, best_c5
