import numpy as np
from scipy.optimize import minimize
import time

n_points_list = [48, 56, 64, 80]
dx_list = [2.0 / n for n in n_points_list]
w_list = [n / 2 for n in n_points_list]
TARGET_C5 = 0.3807
MAX_ITER_PER_RUN = 5000
SMOOTHING_INC = 2
SMOOTHING_MAX = 500
TIME_PER_RUN = 0.5  # seconds

# ============= Helper Functions =============

def compute_correlations(h, dx):
    """Compute all correlations using fft for efficiency."""
    h_pad = np.pad(h, (0, len(h)), mode='constant', constant_values=0)
    h_conj = np.conj(np.fft.fft(h_pad))
    corr = np.fft.ifft(h_conj * np.fft.fft(h_pad)).real[:len(h)]
    return corr * dx

def compute_rk(h, k, w, dx):
    """Compute r_k = w - sum h_i * h_{i+k} for a given k."""
    if k == 0:
        return w - np.sum(h * h)
    else:
        h_shift = np.roll(h, k)
        return w - np.sum(h * h_shift)

def compute_all_rks(h, w, dx):
    """Vector: r_k for k = 0,...,n-1."""
    n = len(h)
    return np.array([compute_rk(h, k, w, dx) for k in range(n)])

def gradient_of_rk(h, k, w, dx):
    """Gradient of r_k w.r.t. h_i (represented as vector)."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if k == 0:
            grad[i] = 1 - 2 * h[i]
        else:
            if i + k < n:
                grad[i] -= h[i + k]
            if i - k >= 0:
                grad[i] -= h[i - k]
            grad[i] += 1
    return grad * dx

def gradient_of_surrogate(h, rks, alpha, dx):
    """Gradient of log-sum-exp surrogate."""
    max_r = np.max(rks)
    rks_shifted = rks - max_r
    exps = np.exp(alpha * rks_shifted)
    w = exps / np.sum(exps)
    grad = dx * np.sum([w[k] * gradient_of_rk(h, k, 1, dx) for k in range(len(rks))], axis=0)
    return grad

def project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0):
    """Project vector v onto the capped simplex defined by sum = target_sum, bounds [lower, upper]."""
    v = np.asarray(v)
    n = len(v)
    if n == 0:
        return v
    # Sort
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    # Water-filling algorithm
    sum_v = 0.0
    for i in range(n):
        sum_v += sorted_v[i]
        # Define threshold
        if i == n - 1:
            theta = (sum_v - target_sum) / (i + 1)
            break
        else:
            # Check if it's a valid partitioning point
            potential_theta = (sum_v - target_sum) / (i + 1)
            if sorted_v[i + 1] <= potential_theta:
                theta = potential_theta
                break
    else:
        theta = (sum_v - target_sum) / n
    # Clamp
    result = np.clip(sorted_v - theta, lower, upper)
    # Reorder
    inverse_indices = np.empty_like(sorted_indices)
    inverse_indices[sorted_indices] = np.arange(n)
    result = result[inverse_indices]
    # Adjust to fix floating-point issues
    current_sum = np.sum(result)
    delta = target_sum - current_sum
    if abs(delta) > 1e-10:
        # Add error to the first non-bound element
        for i in range(n):
            if 0 < result[i] < 1:
                result[i] += delta
                break
    return result

def run_projected_gradient(h0, dx, w, alpha_0=1.0, alpha_max=300, max_iter=5000):
    """Run accelerated projected gradient descent with log-sum-exp smoothing."""
    h = h0.copy()
    t = 1.0
    alpha = alpha_0
    loss_history = []

    # For gradient tracking
    y = h.copy()
    prev_h = h.copy()
    prev_loss = np.inf

    for it in range(max_iter):
        # compute all r_k
        rks = compute_all_rks(h, w, dx)
        # compute log-sum-exp
        max_r = np.max(rks)
        rks_shifted = rks - max_r
        exps = np.exp(alpha * rks_shifted)
        Z = np.sum(exps)
        loss = (dx / alpha) * np.log(Z)

        # check early stop
        if abs(loss - prev_loss) < 1e-6:
            break
        prev_loss = loss

        # gradient
        w_k = exps / Z
        grad = np.sum([w_k[k] * gradient_of_rk(h, k, w, dx) for k in range(len(w_k))], axis=0)

        # Nesterov acceleration
        t_new = (1 + np.sqrt(1 + 4 * t ** 2)) / 2
        h_tilde = project_onto_capped_simplex(y - 0.5 * grad, w)
        y = h_tilde + ((t - 1) / t_new) * (h_tilde - prev_h)
        prev_h = h.copy()
        h = h_tilde
        t = t_new

        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha * SMOOTHING_INC, alpha_max)

        # Store loss
        loss_history.append(loss)

    return h, np.min(rks) * dx

def random_init(n, w):
    """Initialize with random uniform values and project."""
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, w)
    return h

def three_block_init(n, w):
    """Initialize a three-block pattern."""
    a = np.random.rand() * 0.5
    b = a + np.random.rand() * 0.5
    x = np.linspace(0, 2, n)
    h = np.zeros(n)
    h[x < a] = 1
    h[(x >= b)] = 0
    # interpolate
    if a <= b:
        mid = np.argmax((x >= a) & (x < b))
        if mid < n:
            h[mid] = (w - np.sum(h)) / (b - a) * (x[mid] - a)
            h[(x >= a) & (x < b)] = h[mid]
        else:
            h[x >= a] = (w - np.sum(h)) / (2 - a)
    h = project_onto_capped_simplex(h, w)
    return h

def binary_init(n, w):
    """Initialize binary 0/1 pattern and add noise."""
    h = np.zeros(n)
    num_ones = int(w)
    indices = np.random.choice(n, num_ones, replace=False)
    h[indices] = 1
    h += np.random.normal(0, 0.02, n)
    h = project_onto_capped_simplex(h, w)
    return h

def get_initialization_methods():
    return [random_init, three_block_init, binary_init]

def evaluate_c5(h_values):
    """Evaluate C5 using its formal definition."""
    h = np.array(h_values)
    # compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # scale by dx
    c5 = max_correlation * 0.0425531914893617
    return c5

def local_sqp_refine(h, dx, w):
    """Refine solution using local SQP with bounds and fixed sum constraint."""
    def objective(h):
        return evaluate_c5(h)

    def my_constraint(h):
        return np.sum(h) * dx - 1.0

    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': my_constraint}
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
    if result.success:
        h_new = result.x
        return h_new, evaluate_c5(h_new)
    return h, evaluate_c5(h)
