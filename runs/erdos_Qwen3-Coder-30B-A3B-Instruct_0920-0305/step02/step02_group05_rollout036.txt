def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, target_sum):
    """Project h onto feasible set: 0 <= h_i <= 1 and sum(h) = target_sum"""
    # Clip to [0,1]
    w = np.clip(h, 0, 1)
    current_sum = np.sum(w)
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-12:
        return w
    
    # Water-filling algorithm
    iter_count = 0
    max_iter = len(h)
    while abs(diff) > 1e-10 and iter_count < max_iter:
        if diff > 0:
            # Distribute diff equally among indices with w_i < 1
            free_indices = np.where(w < 1)[0]
            if len(free_indices) == 0:
                break
            distribute = diff / len(free_indices)
            for i in free_indices:
                if w[i] + distribute > 1:
                    diff -= (1 - w[i])
                    w[i] = 1
                else:
                    w[i] += distribute
                    diff = 0
        else:
            # Reduce diff equally among indices with w_i > 0
            free_indices = np.where(w > 0)[0]
            if len(free_indices) == 0:
                break
            distribute = -diff / len(free_indices)
            for i in free_indices:
                if w[i] - distribute < 0:
                    diff += w[i]
                    w[i] = 0
                else:
                    w[i] -= distribute
                    diff = 0
        iter_count += 1
    
    return w

def compute_subgradient(h, m):
    """Compute subgradient for lag m"""
    n = len(h)
    g = np.zeros(n)
    dx = 2.0 / n
    
    if m >= 0:
        for i in range(n - m):
            j = i + m
            g[i] += (1 - h[j]) * dx
            g[j] -= h[i] * dx
    else:
        m = -m
        for i in range(m, n):
            j = i - m
            g[i] += (1 - h[j]) * dx
            g[j] -= h[i] * dx
    return g

def compute_worst_lag(A):
    """Find the lag(s) that achieve the maximum in A"""
    max_val = np.max(A)
    lag_indices = np.where(np.abs(A - max_val) < 1e-10)[0]
    return lag_indices, max_val

    """Apply symmetry constraint h[i] = h[n-1-i]"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n // 2 + 1):
        j = n - 1 - i
        h_sym[i] = h_sym[j] = (h[i] + h[j]) / 2
    return h_sym
