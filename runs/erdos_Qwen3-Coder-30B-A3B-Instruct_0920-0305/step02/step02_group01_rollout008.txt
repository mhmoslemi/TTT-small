def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def capped_simplex_projection(g, target_sum, lb=0, ub=1):
    """Project gradient g onto the capped simplex with sum constraint"""
    n = len(g)
    # Sort indices by decreasing absolute gradient values
    indices = np.argsort(-np.abs(g))
    
    # Initialize solution
    h = np.zeros(n)
    remaining_sum = target_sum
    
    for i in range(n):
        idx = indices[i]
        if remaining_sum <= 0:
            h[idx] = 0
        else:
            # Set to upper bound if it's less than remaining sum
            val = min(ub, remaining_sum)
            h[idx] = max(lb, val)
            remaining_sum -= h[idx]
            
    return h

def dc_step(h, dx):
    """Perform one DC iteration: solve the LP subproblem to update h"""
    n = len(h)
    # Compute full autocorrelation q_s(h) = sum_i h_i h_{i+s}
    auto_corr = correlate(h, h, mode="full")
    s_indices = np.arange(-n + 1, n)
    # We want the shift s that gives the *smallest* q_s 
    # That is, gives the *maximum* overlap (w.r.t. 1-h)
    # So we actually want to minimize the max of q_s or equivalently 
    # maximize min of -q_s 
    # For simplicity, let's use the shift with the smallest q_s
    q_values = auto_corr
    min_idx = np.argmin(q_values)
    s_opt = s_indices[min_idx]
    
    # Compute the gradient of q_s_opt at current h:
    # This is d/dh_i q_s(h) = h_{i+s_opt} + h_{i-s_opt} (with boundary handling)
    grad = np.zeros(n)
    # Shift h to the right (positive shift)
    if s_opt > 0:
        h_shifted = np.concatenate([np.zeros(s_opt), h[:-s_opt]])
    elif s_opt < 0:
        h_shifted = np.concatenate([h[-s_opt:], np.zeros(-s_opt)])
    else:
        h_shifted = h
    grad += h_shifted
    
    # Shift h to the left (negative shift) 
    if s_opt < 0:
        h_shifted = np.concatenate([np.zeros(-s_opt), h[-s_opt:]])
    elif s_opt > 0:
        h_shifted = np.concatenate([h[s_opt:], np.zeros(s_opt)])
    else:
        h_shifted = h
    grad += h_shifted
    
    # Now we solve:
    # minimize <grad, h>
    # subject to 0 <= h_i <= 1 for all i
    # and sum h_i = 1/dx
    # Since dx = 2.0/n, we want sum h_i = n/2 
    target_sum = n / 2
    
    # This is a linear program minimizing linear objective over box
    # Solution by sorting and taking largest components until sum = target_sum
    # This is equivalent to projecting the gradient negative onto the simplex
    # Minimize <grad, h>
    # = < -(-grad), h >
    # = < -g, h > where g = -grad
    g = -grad
    h_new = capped_simplex_projection(g, target_sum)
    
    # Ensure sum constraint 
    actual_sum = np.sum(h_new)
    h_new = h_new * (target_sum / actual_sum)
    
    return h_new
