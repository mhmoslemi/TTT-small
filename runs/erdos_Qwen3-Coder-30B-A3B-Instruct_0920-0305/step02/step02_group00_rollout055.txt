def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h_raw, n_points, dx):
    """Project h onto [0,1] and sum constraint."""
    h = np.clip(h_raw, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-6:
        # Bisection to adjust for sum constraint
        scale = 1.0 / current_sum
        h *= scale
        # Ensure sum is exactly 1.0
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-10:
            # Use bisection to adjust
            def scale_func(scale):
                h_trial = np.clip(h_raw * scale, 0, 1)
                return np.sum(h_trial) * dx
            # Binary search for the right scale
            low, high = 0.0, 2.0
            while high - low > 1e-8:
                mid = (low + high) / 2.0
                sum_trial = scale_func(mid)
                if abs(sum_trial - 1.0) < 1e-10:
                    low = high = mid
                    break
                elif sum_trial < 1.0:
                    low = mid
                else:
                    high = mid
            h = np.clip(h_raw * low, 0, 1)
            h *= (1.0 / (np.sum(h) * dx))
    return h

def build_linear_constraints(h_cur):
    """Build the vector a_s and scalar b_s for each shift s."""
    n = len(h_cur)
    a_s_list = []
    b_s_list = []
    for s in range(n):

        a_s = np.zeros(n)
        b_s = 0.0
        for i in range(n):
            # Determine how to handle out-of-bounds
            # For i+s, may be out of bounds
            if i + s < n:
                a_s[i] += 1.0
                a_s[i + s] -= h_cur[i]  # This is the coefficient of h_{i+s} in L_s
                b_s -= h_cur[i] * h_cur[i + s]
            else:
                a_s[i] += 1.0
                b_s -= h_cur[i] * 0.0  # h_{i+s} treated as 0 if out of bounds
            # Same for i-s (but make sure not to double count)
            if i - s >= 0:
                a_s[i] -= h_cur[i - s]
            # Add contribution of h_{i-s} in the linearization
            # if there's no term h_{i-s}, it should be just the -h_{i+s} effect arrived via linearity
            # In the linear term, we've already added -h_{i+s} * h_i, and -h_i * h_{i+s} from the symmetric term.
            if i - s >= 0:
                a_s[i - s] -= h_cur[i]
            else:
                # i - s < 0, no contribution
                pass
        # In the def a_s^T h + b_s, a_s is a vector and we maximize the overlap
        # The overlap - (a_s^T h + b_s) <= -t => a_s^T h + b_s >= t  => t <= a_s^T h + b_s
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_optimization_step(h_current, a_s_list, b_s_list):
    """Solve the linear program for the current linearization."""
    n = len(h_current)
    # Variables
    h = Variable(n)
    t = Variable(1)
    # Constraints
    constraints = [h >= 0, h <= 1]
    # Add equality constraint
    constraints.append(sum_expr(h) == n/2)
    # Add inequality constraints from a_s^T h + b_s <= t for all s
    for i, (a_s, b_s) in enumerate(zip(a_s_list, b_s_list)):
        constraints.append(a_s @ h + b_s <= t)
    # Objective
    objective = Minimize(t)
    # Problem
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cvxpy_solver, verbose=False)
        return h.value, t.value
    except:
        # Fallback: return current solution
        return h_current.copy(), evaluate_h(h_current)
