import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

n_points = 48  # chosen to be even for simplicity
dx = 2.0 / n_points
target_bound = 0.3807

def evaluate_h(h_values):
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h):
    """Project h to the set [0,1]^n and sum=h_sum."""
    h_proj = np.clip(h, 0, 1)
    h_sum = np.sum(h_proj)
    if abs(h_sum - n_points/2) < 1e-10:
        return h_proj
    # Bisection method to adjust h_proj to have sum = n_points/2
    # We rescale the h_proj to sum to n_points/2
    scale = (n_points/2) / h_sum
    h_proj = h_proj * scale
    return h_proj

def build_linear_constraints(h_cur):
    """Build coefficients for linear constraints a_s^T h + b_s <= t"""
    n = len(h_cur)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        a_s = np.zeros(n)
        b_s = 0.0
        for i in range(n):
            if i + s < n:
                a_s[i] += 1.0
                a_s[i + s] += -1.0
        b_s = -np.sum(h_cur * np.roll(h_cur, s)) * dx  # the O_s term
        a_s_list.append(a_s)
        b_s_list.append(b_s)
    return a_s_list, b_s_list

def solve_scp_subproblem(a_s_list, b_s_list):
    """Solve the linear program with cvxpy."""
    n = len(a_s_list[0])
    h = cp.Variable(n)
    t = cp.Variable(1)
    constraints = []
    for a_s, b_s in zip(a_s_list, b_s_list):
        constraints.append(a_s @ h + b_s <= t)
    constraints.append(cp.sum(h) == n_points/2)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.GLPK, verbose=False)  # Faster than CBC for these small problems
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h.value, t.value
    else:
        # fallback to standard solution when not successful
        return np.array([0.5] * n), 1.0

                h0[:n_points//2] = 1.0
