```python
import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_values = [48, 56, 60]
dx = 2.0 / 48  # Default to n=48, will adjust in loop
target_bound = 0.3807
max_inner_iter = 300
beta_schedule = [5, 15, 30, 60, 120]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_box_and_hyperplane(v, target_sum):
    """Project vector v onto [0,1]^n and then onto sum = target_sum."""
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    
    # Compute the projection: find λ such that sum of clipped(v - λ) = target
    # This works because f(λ) = sum clip(v - λ, 0, 1) is monotonically decreasing
    # We need λ such that sum of the clipped input is near target
    
    low, high = 0.0, 1.0
    while high - low > 1e-10:
        mid = (low + high) / 2
        projected = np.clip(v - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            low = mid
        else:
            high = mid
            
    return np.clip(v - low, 0, 1)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n_used = None

    for n in n_values:
        if time.time() - start_time > 0.95 * budget_s:
            break
        dx = 2.0 / n
        target_sum = n / 2
        print(f"Trying n = {n}")

        # Generate diverse starting points
        start_points = []

        # Interpolated from initial_h_values
        if len(initial_h_values) < n:
            h_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(initial_h_values)), initial_h_values)
        else:
            h_interp = np.array(initial_h_values[:n])
        h_interp = project_onto_box_and_hyperplane(h_interp, target_sum)
        start_points.append(("interpolated", h_interp))

        # Random binary with correct sum
        h_bin = np.zeros(n)
        h_bin[:int(target_sum)] = 1
        np.random.shuffle(h_bin)
        h_bin = project_onto_box_and_hyperplane(h_bin, target_sum)
        start_points.append(("binary", h_bin))

        # Random uniform
        h_unif = np.random.rand(n)
        h_unif = project_onto_box_and_hyperplane(h_unif, target_sum)
        start_points.append(("uniform", h_unif))

        # Previous best if exists
        if best_h is not None and len(best_h) == n:
            h_prev = np.array(best_h)
            h_prev = project_onto_box_and_hyperplane(h_prev, target_sum)
            start_points.append(("previous", h_prev))

        # Perform restarts for this n
        inner_budget = budget_s * 0.25 / len(n_values)  # Allocate part of budget
        for attempt, (start_name, h_current) in enumerate(start_points):
            if time.time() - start_time > 0.95 * budget_s:
                break

            # Run with continuation
            iter_count = 0
            h_best = h_current.copy()
            c5_best = evaluate_h(h_best)
            last_improvement_iter = 0

            for beta in beta_schedule:
                if iter_count > max_inner_iter or time.time() - start_time > 0.95 * budget_s:
                    break

                for k in range(100):  # Inner iterations for this beta
                    if time.time() - start_time > 0.95 * budget_s or iter_count > max_inner_iter:
                        break

                    iter_count += 1
                    # Compute all correlations for current h
                    corr = np.correlate(h_current, 1 - h_current, mode="full")
                    C_s = corr * dx

                    # Compute weights for soft min
                    max_C = np.max(C_s)
                    # Prevent overflow
                    C_s_shifted = C_s - max_C
                    exp_terms = np.exp(-beta * C_s_shifted)
                    w_s = exp_terms / np.sum(exp_terms)

                    # Compute gradient: 2 * dx * sum w_s * A_s h
                    grad = np.zeros(n)
                    # Instead of full matrices, compute its effects using convolution
                    # For shift s, the effect on entry i is h[i-s] + h[i+s] (centered at 0,1,...)
                    # Use roll to simulate shift, pad with zeros at boundaries
                    for s_idx in range(n):
                        if s_idx == 0:
                            grad += w_s[s_idx] * h_current
                        else:
                            shifted_forward = np.roll(h_current, -s_idx)
                            shifted_backward = np.roll(h_current, s_idx)
                            grad += w_s[s_idx] * 0.5 * (shifted_forward + shifted_backward)

                    grad *= 2 * dx

                    # Gradient ascent step
                    alpha = 0.3
                    h_new = h_current + alpha * grad
                    h_new = project_onto_box_and_hyperplane(h_new, target_sum)

                    # Evaluate exact
                    c5_new = evaluate_h(h_new)

                    if c5_new < c5_best:
                        h_best = h_new.copy()
                        c5_best = c5_new
                        last_improvement_iter = iter_count
                        if c5_best < target_bound:
                            break

                    h_current = h_new
                    if iter_count - last_improvement_iter > 10:
                        break

                if c5_best < target_bound:
                    break

            # Optional post-refinement with SLSQP
            if time.time() - start_time < 0.9 * budget_s:
                try:
                    def objective(h):
                        return evaluate_h(h)
                    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
                    bounds = [(0, 1) for _ in range(n)]
                    result = minimize(objective, h_best, method='SLSQP', bounds=bounds, constraints=cons,
                                     options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10})
                    if result.success:
                        h_final = result.x
                        c5_final = evaluate_h(h_final)
                        if c5_final < c5_best:
                            h_best = h_final
                            c5_best = c5_final
                except:
                    pass

            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
                best_n_used = n

            if best_c5 < target_bound:
                break

        if best_c5 < target_bound:
            break

    if best_h is None:
        best_h = np.array(initial_h_values[:48])
        dx = 2.0 / 48
        best_c5 = evaluate_h(best_h)
        best_n_used = 48

    return list(best_h), best_c5, best_n_used
```
