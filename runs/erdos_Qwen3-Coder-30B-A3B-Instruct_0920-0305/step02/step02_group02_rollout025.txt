def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Use FFT to efficiently compute auto-correlation
    fft_h = fft(h, 2 * len(h))
    conv = ifft(fft_h * np.conj(fft_h)).real
    S = conv[len(h) - 1 : 2 * len(h) - 1]
    r = np.sum(h) - S
    c5 = np.max(r) * 2.0 / len(h)
    return c5

def logsumexp(arr, alpha):
    """Numerically stable log-sum-exp."""
    max_val = np.max(arr)
    return max_val + np.log(np.sum(np.exp(alpha * (arr - max_val))))

def compute_surrogate_grad(h, alpha):
    """Compute gradient of smooth surrogate F_alpha(h)."""
    n = len(h)
    dx = 2.0 / n
    w = n / 2
    
    # Auto-correlation via FFT
    fft_h = fft(h, 2 * n)
    conv = ifft(fft_h * np.conj(fft_h)).real
    S = conv[n-1 : 2*n-1]
    r = w - S

    # Compute soft-max weights for gradient
    r_max = np.max(r)
    exp_r = np.exp(alpha * (r - r_max))
    weights = exp_r / np.sum(exp_r)
    
    # Gradient computation (from each r_k)
    g = np.zeros(n)
    for k in range(n):
        if k == 0:
            # ∂r_0/∂h_j = 1 - 2 * h_j
            g += weights[k] * (1 - 2 * h)
        else:
            # r_k =  w - S_k(h), ∂r_k/∂h_j = - ∂S_k/∂h_j
            grad_row = np.zeros(n)
            # Shift h to the right by k
            if k < n:
                grad_row[k:] = h[:-k]
            g -= weights[k] * grad_row
    
    return g * dx

def project_simplex(v, w):
    """Project v onto capped simplex with sum = w."""
    n = len(v)
    # Sort descending
    sorted_v = np.sort(v)[::-1]
    # Find threshold tau
    cumsum = np.cumsum(sorted_v)
    idx = np.arange(1, n+1)
    temp = (sorted_v - (w - cumsum) / idx)
    rejected = temp > 0
    if np.sum(rejected) == 0:
        # All values have been clamped to 1
        return np.clip(v, 0, 1)
    k = np.sum(rejected)
    tau = (w - cumsum[k-1]) / k
    
    result = np.clip(v - tau, 0, 1)
    # Adjust to ensure sum equals w
    diff = w - np.sum(result)
    if abs(diff) > 1e-10:
        # Add correction to first nonzero entry
        for i in range(n):
            if result[i] > 0:
                result[i] += diff
                break
    
    return result

def accelerated_gradient(h0, alpha, max_iter=500, epsilon=1e-6):
    """FISTA-style accelerated projected gradient."""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    for i in range(max_iter):
        # Compute smooth surrogate gradient
        g = compute_surrogate_grad(y, alpha)
        # Armijo-style line search
        eta = 0.1
        while True:
            h_new = project_simplex(y - eta * g, sum(h0))
            if np.all(h_new >= 0) and np.all(h_new <= 1):
                F_new = logsumexp(np.sum(h0) - np.correlate(h_new, h_new, mode='full')[len(h_new)-1:], alpha)
                # Simple backtracking
                if F_new < logsumexp(np.sum(h0) - np.correlate(y, y, mode='full')[len(h0)-1:], alpha):
                    break
            eta *= 0.5
        # Update FISTA
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y = h_new + ((t-1)/t_new) * (h_new - h)
        h = h_new
        t = t_new
        # Check convergence
        if i >= 50 and abs(F_new - logsumexp(np.sum(h0) - np.correlate(h, h, mode='full')[len(h)-1:], alpha)) < epsilon:
            break
    return h

def build_initial_point(n, strategy="uniform"):
    """Generate initial point with given strategy."""
    assert strategy in ["uniform", "binary", "three_block", "warm_start"]
    w = n // 2
    if strategy == "uniform":
        h = np.random.rand(n)
        h = project_simplex(h, w)
    elif strategy == "binary":
        h = np.zeros(n)
        ones = np.random.choice(n, size=w, replace=False)
        h[ones] = 1
        h += np.random.normal(0, 0.02, n)
        h = project_simplex(h, w)
    elif strategy == "three_block":
        a = np.random.rand() * 0.5
        b = a + np.random.rand() * 0.5
        p = (w - (b - a) * n) / (a + (1 - b)) / n if (a + (1 - b)) > 0 else 0.8
        h = np.ones(n) * p
        h[:int(a * n)] = 1
        h[int(b * n):] = 0
        h = project_simplex(h, w)
    else:  # warm_start
        h = np.random.rand(n)
        h = project_simplex(h, w)
    return h
