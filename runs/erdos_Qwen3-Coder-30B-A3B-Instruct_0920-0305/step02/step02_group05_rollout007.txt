def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1, sum h_i = n/2"""
    # Clip to [0, 1]
    w = np.clip(h, 0, 1)
    
    # Water-filling algorithm
    target = n_points / 2.0
    current_sum = np.sum(w)
    delta = target - current_sum
    
    # Ensure that tolerance is not too large
    tolerance = 1e-10
    while abs(delta) > tolerance:
        if delta > 0:
            # Distribute delta equally over all indices with w_i < 1
            mask = w < 1
            if np.sum(mask) == 0:
                break
            distribute_amount = delta / np.sum(mask)
            w[mask] += distribute_amount
            w = np.clip(w, 0, 1)
        else:
            # Subtract delta equally over all indices with w_i > 0
            mask = w > 0
            if np.sum(mask) == 0:
                break
            subtract_amount = -delta / np.sum(mask)
            w[mask] -= subtract_amount
            w = np.clip(w, 0, 1)
        current_sum = np.sum(w)
        delta = target - current_sum
        
    return w

def compute_subgradient(h, m, n_points, dx):
    """Compute subgradient for a specific lag m"""
    g = np.zeros(n_points)
    if m >= 0:
        for i in range(n_points - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        m_abs = -m
        for i in range(m_abs, n_points):
            j = i - m_abs
            g[i] += 1 - h[j]
            g[j] -= h[i]
    g *= dx
    return g

def find_max_lag_indices(A):
    """Find all indices that achieve the maximum value in A"""
    max_val = np.max(A)
    max_indices = np.where(np.abs(A - max_val) < 1e-10)[0]
    return max_indices

    """Symmetrize h to enforce mirror symmetry"""
    n = len(h)
    h_sym = np.zeros(n)
    for i in range(n):
        h_sym[i] = (h[i] + h[n - 1 - i]) / 2.0
    return h_sym
