import numpy as np
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807
max_n = 70
min_n = 40
n_values = [48, 56, 60, 64, 70]
dx_values = [2.0 / n for n in n_values]

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * (2.0 / len(h))
    return c5

def project_onto_constraints(v, target_sum):
    """Project v onto [0,1]^n and then onto the hyperplane sum=h."""
    # Clip to [0, 1]
    v = np.clip(v, 0, 1) 
    current_sum = np.sum(v)
    # If we're already at the correct sum, return
    if abs(current_sum - target_sum) < 1e-10:
        return v
    
    # Bisection to find lambda such that clip(v - lambda, 0, 1) sums to target
    low, high = -1.0, 1.0
    for _ in range(30):  # 30 iterations should be enough
        mid = (low + high) / 2.0
        proj = np.clip(v - mid, 0, 1)
        proj_sum = np.sum(proj)
        if proj_sum > target_sum:
            high = mid
        else:
            low = mid
    lambda_ = (low + high) / 2.0
    return np.clip(v - lambda_, 0, 1)

def compute_A_shifted(h, shift):
    """Compute A^{(s)} h where A^{(s)}[i][i+s] = 0.5 (and symmetric)"""
    n = len(h)
    result = np.zeros_like(h)
    if shift == 0:
        result = h
    else:
        # Element-wise operations
        if shift > 0:
            # i -> i + shift
            result[shift:] += h[:-shift] 
            result[:-shift] += h[shift:]
        else:  # shift < 0
            # shift is negative, i -> i - |shift|
            abs_s = -shift
            result[abs_s:] += h[:-abs_s]
            result[:-abs_s] += h[abs_s:]
        result *= 0.5
    return result

def smooth_min_gradient(h, beta=50.0):
    """Compute gradient of smooth min approximation."""
    n = len(h)
    dx = 2.0 / n
    # Correlations for each shift
    correlations = np.zeros(n)
    for s in range(n):
        shifted_h = np.roll(h, s)
        if s > 0:
            # We have to handle the zero-padding correctly 
            shifted_h[:s] = 0
        elif s < 0:
            shifted_h[s:] = 0
        correlations[s] = np.sum(h * shifted_h)
    correlations *= dx

    # Soft-min weights
    max_C = np.max(correlations)
    # Avoid arithmetic overflow
    exp_term = np.exp(-beta * (correlations - max_C))
    weights = exp_term / np.sum(exp_term)
    
    # Compute the gradient
    grad = np.zeros(n)
    for s in range(n):
        A_h = compute_A_shifted(h, s)
        grad += 2 * dx * weights[s] * A_h
    
    return grad

def gradient_ascent_step(h, beta, step_size=0.1):
    """Perform one gradient ascent step on smooth objective"""
    grad = smooth_min_gradient(h, beta)
    h_new = h + step_size * grad
    # Project back onto feasible region
    h_new = project_onto_constraints(h_new, len(h)/2)
    return h_new

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    # Try different n values
    for n, dx in zip(n_values, dx_values):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        target_sum = n / 2
        
        # Multiple restarts
        restarts = 5
        for _ in range(restarts):
            # Start with interpolated from initial_h_values
            initial_h = np.interp(np.linspace(0, 2, n), 
                          np.linspace(0, 2, len(initial_h_values)), 
                          initial_h_values)
            h = project_onto_constraints(initial_h, target_sum)
            
            # Gradient ascent with continuation
            max_iter = 500
            betas = [5, 15, 30, 60, 120]
            for beta in betas:
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                for i in range(max_iter):
                    if time.time() - start_time > 0.95 * budget_s:
                        break
                        
                    h_prev = h.copy()
                    h = gradient_ascent_step(h, beta)
                    
                    # Evaluate current solution
                    current_c5 = evaluate_h(h)
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                        best_n = n
                    # Simple stopping condition
                    if i > 50 and np.allclose(h, h_prev, atol=1e-6):
                        break
        
        # Final local refinement
        if best_h is not None and time.time() - start_time < 0.9 * budget_s:
            def obj(h_local):
                return evaluate_h(h_local)
            
            bounds = [(0, 1) for _ in range(n)]
            constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
            
            try:
                result = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                  constraints=constraints, options={'maxiter': 200})
                if result.success:
                    final_c5 = evaluate_h(result.x)
                    if final_c5 < best_c5:
                        best_h = result.x
                        best_c5 = final_c5
                        best_n = n
            except:
                pass
    
    if best_h is not None:
        return list(best_h), best_c5, best_n
    else:
        # Fallback to a simple construction
        n = 48
        dx = 2.0 / n
        h = np.array(initial_h_values)
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), h)
        h = project_onto_constraints(h, n/2)
        c5 = evaluate_h(h)
        return list(h), c5, n
