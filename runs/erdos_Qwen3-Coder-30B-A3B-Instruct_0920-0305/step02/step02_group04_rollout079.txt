```python
import numpy as np
import cvxpy as cp
from scipy.fft import fft, ifft
import time

def evaluate_h(h):
    """Evaluate the C5 constant using FFT-based correlation."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2n for full convolution
    h_padded = np.pad(h, (0, n), 'constant')
    g_padded = np.pad(1 - h, (0, n), 'constant')
    corr = ifft(fft(h_padded) * np.conj(fft(g_padded))).real[:2*n-1]
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 48  # Fix at 48 for speed
    dx = 2.0 / n
    M = n // 2  # Target mass
    T_target = int(np.ceil(0.3807 * M))  # Target integer bound
    
    # Step 1: Binary Mixed Integer Programming
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining < 30:
        # Too little time, skip MILP
        h_bin = np.zeros(n, dtype=int)
        h_bin[:M] = 1
        np.random.shuffle(h_bin)
    else:
        # MILP setup
        h_vars = cp.Variable(n, boolean=True)
        p_vars = [[cp.Variable(boolean=True) for i in range(n)] for l in range(n)]
        T = cp.Variable(integer=True, lb=0)
        
        # Constraints
        constraints = [cp.sum(h_vars) == M]
        
        # Product constraints (McCormick envelopes)
        for l in range(1, n):
            for i in range(n-l):
                # p[l][i] = h[i] * h[i+l]
                constraints.extend([
                    p_vars[l][i] <= h_vars[i],
                    p_vars[l][i] <= h_vars[i+l],
                    p_vars[l][i] >= h_vars[i] + h_vars[i+l] - 1,
                    p_vars[l][i] >= 0
                ])
        
        # Worst-lag constraints
        for l in range(1, n):
            lag_sum = cp.sum(p_vars[l][:n-l])
            constraints.append(lag_sum + T >= M)
        
        # Objective
        objective = cp.Minimize(T)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.CBC, maxSeconds=0.3 * t_remaining, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = np.round(h_vars.value).astype(int)
                T_opt = int(T.value)
            else:
                # Fallback to heuristic if solving failed
                h_opt = np.zeros(n, dtype=int)
                h_opt[:M] = 1
                np.random.shuffle(h_opt)
                T_opt = -1  # Indicate not solved
        except:
            # Fallback
            h_opt = np.zeros(n, dtype=int)
            h_opt[:M] = 1
            np.random.shuffle(h_opt)
            T_opt = -1
    
    # Step 2: Binary local refinement (annealing)
    if T_opt >= T_target:  # Only proceed if we need to improve
        h_curr = h_opt.copy()
        best_h = h_curr.copy()
        best_c5 = evaluate_h(h_curr)
        
        t_remaining = budget_s - (time.time() - start_time)
        if t_remaining < 10:
            pass  # Too little time, skip local search
        else:
            # Annealing
            n_swaps = int(1e6)
            T_start = 3.0 * np.median([np.abs(e) for e in np.random.choice(40, 1000, replace=True) if e != 0])
            T = T_start
            cooling = 0.999
            min_T = 1e-6
            
            for _ in range(n_swaps):
                if (time.time() - start_time) > 0.7 * budget_s:
                    break
                # Pick two random indices
                i, j = np.random.choice(n, size=2, replace=False)
                while h_curr[i] == 0:
                    i = np.random.randint(0, n)
                while h_curr[j] == 1:
                    j = np.random.randint(0, n)
                
                # Swap
                h_old = h_curr.copy()
                h_curr[i], h_curr[j] = 0, 1
                c5_new = evaluate_h(h_curr)
                c5_old = evaluate_h(h_old)
                
                delta = c5_new - c5_old
                if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                    # Accept
                    pass
                else:
                    # Reject, restore
                    h_curr = h_old.copy()
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_curr.copy()
                
                T *= cooling
                if T < min_T:
                    break
    
    # Final best
    h_final = best_h if 'best_h' in locals() else h_opt
    final_c5 = evaluate_h(h_final)
    
    # Step 3: Fractional smoothing (optional)
    t_remaining = budget_s - (time.time() - start_time)
    if t_remaining > 5 and final_c5 > 0.379:
        h_frac = h_final.astype(float)
        # Apply log-sum-exp descent
        beta_schedule = [1, 2, 5, 10, 20]
        steps_per_beta = 50
        
        for beta in beta_schedule:
            for _ in range(steps_per_beta):
                if (time.time() - start_time) > 0.9 * budget_s:
                    break
                # Compute all L_ell
                L_vals = np.zeros(n - 1)
                for l in range(1, n):
                    # This is a bit tricky for overlapping: use power of fft
                    h_padded = np.pad(h_frac, (0, n), 'constant')
                    g_padded = np.pad(1 - h_frac, (0, n), 'constant')
                    corr = ifft(fft(h_padded) * np.conj(fft(g_padded))).real[:2*n-1]
                    L_vals[l-1] = corr[l-1] * dx
                    
                # Softmax weights
                w = np.exp(beta * L_vals)
                w = w / np.sum(w)
                
                # Gradient of L_l
                grad_lag = np.zeros(n)
                for l in range(1, n):
                    for i in range(n-l):
                        grad_lag[i] += (1 - h_frac[i+l]) * dx
                        if i-l >= 0:
                            grad_lag[i] -= h_frac[i-l] * dx
                
                # Combined gradient
                grad = np.sum([w[l-1] * grad_lag for l in range(1, n)], axis=0)
                
                # Update
                eta = 0.001
                h_new = h_frac - eta * grad
                
                # Project to [0, 1]
                h_new = np.clip(h_new, 0, 1)
                
                # Project to sum = M * dx (water filling)
                def water_fill(h_local, target_sum):
                    min_h = 0
                    max_h = 1
                    while max_h - min_h > 1e-10:
                        mid = (min_h + max_h) / 2
                        filled = np.sum(np.clip(h_local + mid, 0., 1.))
                        if filled > target_sum:
                            max_h = mid
                        else:
                            min_h = mid
                    return np.clip(h_local + min_h, 0., 1.)
                
                h_frac = water_fill(h_new, M * dx)
                
                if np.sum(h_frac) * dx < M * dx - 1e-6 or np.sum(h_frac) * dx > M * dx + 1e-6:
                    h_frac = h_frac * (M * dx) / np.sum(h_frac)
                h_frac = np.clip(h_frac, 0, 1)
                
                final_c5 = evaluate_h(h_frac)
        
        h_final = h_frac
        
    # Final sanity check
    h_final = np.clip(h_final, 0, 1)
    m = np.sum(h_final) * dx
    if abs(m - 1.0) > 1e-9:
        h_final = h_final * (1.0 / m)
    final_c5 = evaluate_h(h_final)
    
    return (h_final.tolist(), final_c5, n)
```
