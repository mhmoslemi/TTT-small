import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

# Constants
n_candidates = [48, 56, 64, 80]
dx_candidates = [2.0 / n for n in n_candidates]
target_bound = 0.3807
max_alpha = 500
eps = 1e-6
max_iter_per_restart = 1000
max_restarts_per_n = 20
time_per_n = 600 / len(n_candidates)  # Roughly equal time per n

def soft_max_exp(x, alpha):
    """Numerically stable soft max computation."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(alpha * x_shifted)
    return exp_x / np.sum(exp_x)

def evaluate_corr(h):
    """Evaluate the auto-correlations using FFT for speed."""
    n = len(h)
    # Pad to 2*n and compute both auto-correlations
    fft_h = fft(h, 2 * n)
    corr = ifft(fft_h * np.conj(fft_h)).real
    # Take only non-negative lags k = 0, ..., n-1
    corr_k = corr[n-1:2*n-1]
    return corr_k

def evaluate_r_k(h, w):
    """For given h and w = sum(h) * dx, compute r_k = w - S_k."""
    S_k = evaluate_corr(h)
    r_k = w - S_k
    return r_k

def gradient_r_k(h, w, k):
    """Compute gradient of r_k = w - S_k with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # r_k = w - Σ_{i=0}^{n‑1‑k} h_i·h_{i+k}
    # ∂r_k/∂h_j = 0 (if index out of bounds) or - h_{j+k} or - h_{j-k}
    # If k=0: ∂r_0/∂h_j = - 2 h_j
    if k == 0:
        grad = - 2 * h
    else:
        # j+k >= 0, so h_{j+k}
        for i in range(n - k):
            grad[i] -= h[i + k]
        # j-k < n, so h_{j-k}
        for i in range(k, n):
            grad[i] -= h[i - k]
    # Add 1 for all terms (from constant w)
    grad += 1
    return grad

def project_simplex(x, w):
    """Projection onto the capped simplex with sum = w."""
    n = len(x)
    # Sort in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]

    # Compute thresholds and find where constraint binds
    cum_sum = np.cumsum(sorted_x)
    thresholds = (cum_sum - w) / np.arange(1, n+1)

    # Find largest index where the constraint holds
    valid_indices = np.where(sorted_x > thresholds)[0]
    if len(valid_indices) == 0:
        # All elements must be 0
        projected = np.zeros(n)
    else:
        k = valid_indices.max() + 1
        threshold = thresholds[k - 1]
        projected = np.clip(sorted_x[:k] - threshold, 0, 1)
        projected = np.concatenate([projected, np.zeros(n - k)])
        # Reorder to original indices
        final = np.zeros(n)
        final[sorted_indices] = projected
        projected = final

    # Ensure exact sum
    current_sum = np.sum(projected)
    diff = w - current_sum
    if abs(diff) > 1e-10:
        # Add difference to first non-saturated value
        for i in range(n):
            if projected[i] < 1:
                projected[i] += diff
                break

    return projected

def fista_smooth_surrogate(h0, w, dx, alpha_max, eps):
    """Minimize log-sum-exp approximation of the max-over-shift."""
    n = len(h0)
    h = h0.copy()
    y = h.copy()
    t = 1
    prev_F = np.inf
    alpha = 1

    while alpha <= alpha_max:
        for iteration in range(max_iter_per_restart):
            # Compute r_k for all k
            r_k = evaluate_r_k(h, w)

            # Compute log-sum-exp
            max_r = np.max(r_k)
            shifted_r = r_k - max_r
            exp_r = np.exp(alpha * shifted_r)
            sum_exp = np.sum(exp_r)
            F_alpha = (dx / alpha) * (np.log(sum_exp) + max_r)

            if abs(F_alpha - prev_F) < eps * abs(F_alpha):
                break
            prev_F = F_alpha

            # Compute weighted sum over ∇r_k
            weights = exp_r / sum_exp
            g = np.zeros(n)
            for k in range(n):
                grad = gradient_r_k(h, w, k)
                g += weights[k] * grad

            # Line search for η
            eta = 0.5
            while True:
                h_tilde = project_simplex(y - eta * g, w)
                r_k_tilde = evaluate_r_k(h_tilde, w)
                F_tilde = (dx / alpha) * (np.log(np.sum(np.exp(alpha * (r_k_tilde - np.max(r_k_tilde)))) + np.max(r_k_tilde))
                # Armijo condition with c=0.5
                if F_tilde <= F_alpha - 0.5 * eta * np.dot(g, h_tilde - y):
                    break
                eta *= 0.5

            # Update FISTA
            t_new = (1 + np.sqrt(1 + 4 * t ** 2)) / 2
            y = h_tilde + ((t - 1) / t_new) * (h_tilde - h)
            h = h_tilde
            t = t_new

        alpha *= 2

    # Final computation using exact max
    r_k_final = evaluate_r_k(h, w)
    C5 = dx * np.max(r_k_final)
    return h, C5

            break

        for restart in range(max_restarts_per_n):
            if time.time() - start_time > budget_s * 0.99:
                break
            restarts_this_n += 1

            # Heuristic initializations
            choice = restart % 4
            if choice == 0:  # Uniform random
                h = np.random.rand(n)
                h = project_simplex(h, w)
            elif choice == 1:  # Binary biased
                h = np.zeros(n)
                ones = int(w)
                h[:ones] = 1
                h = np.random.permutation(h)
                h += np.random.normal(0, 0.02, n)
                h = project_simplex(h, w)
            elif choice == 2:  # Three-block
                a = np.random.uniform(0, 0.5)
                b = np.random.uniform(0.5, 2)
                mid = int(a * n)
                end = int(b * n)
                h = np.zeros(n)
                h[:mid] = 1
                h[mid:end] = 0.5  # or set to some fractional value, e.g. 0.5
                h = project_simplex(h, w)
            else:  # Warm start from best
                h = init_h.copy()
                h = np.array(h)
                np.random.shuffle(h)
                h = project_simplex(h, w)

            h_opt, C5 = fista_smooth_surrogate(h, w, dx, max_alpha, eps)

            if C5 < best_C5:
                best_C5 = C5
                best_h = h_opt
                best_n = n

            if best_C5 < target_bound:
                break  # Achieved target early

    # Final evaluate using the robust formulation
    if best_h is not None:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_C5 = dx_candidates[n_candidates.index(best_n)] * np.max(corr)
    else:
        final_C5 = np.inf

    return list(best_h) if best_h is not None else [], final_C5, best_n or 48
