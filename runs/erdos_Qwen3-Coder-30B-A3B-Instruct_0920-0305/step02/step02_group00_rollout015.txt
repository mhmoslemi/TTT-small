import numpy as np
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, norm
import cvxpy as cp
from scipy.optimize import minimize
import time

def build_linear_constraints(h):
    """Build linear approximations of the bilinear terms for all shifts."""
    n = len(h)
    a_s_list = []
    b_s_list = []
    for s in range(n):
        # Build a_s
        a_s = np.zeros(n)
        for i in range(n):
            if i + s < n:
                a_s[i] = 1 - h[i + s] - (h[i - s] if i - s >= 0 else 0)
            else:
                a_s[i] = 1 - h[i + s]  # Reset for boundary
        a_s_list.append(a_s)

        # Build b_s
        b_s = 0
        for i in range(n - s):
            b_s -= h[i] * h[i + s]
        b_s_list.append(b_s)

    return a_s_list, b_s_list

def solve_lp(n, a_s_list, b_s_list, target_sum):
    """Solve the linear program for given coefficients."""
    h = Variable(n)
    t = Variable(1)
    
    # Objective: minimize t
    objective = Minimize(t)
    
    # Constraints
    constraints = []
    
    # Linear constraints from bilinear approximations
    for s in range(n):
        constraints.append(a_s_list[s] @ h + b_s_list[s] <= t)
    
    # Sum constraint
    constraints.append(cp_sum(h) == target_sum)
    
    # Bounds
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Solve
    prob = Problem(objective, constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_sol = h.value
        t_sol = t.value
    else:
        # Fallback solution: just return current h
        h_sol = np.zeros(n)
        t_sol = np.inf

    # Project the solution to feasible region
    h_proj = np.clip(h_sol, 0, 1)
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) > 1e-6:
        # Simple bisection to rescale
        def rescale(x, scale_factor):
            return np.clip(x * scale_factor, 0, 1)
        
        def target_diff(scale_factor):
            scaled_h = rescale(h_proj, scale_factor)
            return np.sum(scaled_h) - target_sum
            
        a, b = 0.9, 1.1
        for _ in range(100):
            mid = (a + b) / 2
            val = target_diff(mid)
            if abs(val) < 1e-6:
                break
            elif val < 0:
                a = mid
            else:
                b = mid
        h_proj = rescale(h_proj, (a + b) / 2)
    
    return h_proj, t_sol

def evaluate_h(h_values, dx):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        # Rescale
        h *= (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

            h_current = h0.copy()

            for k in range(200):  # max iterations

                    break
                
                # Build linear constraints
                a_s_list, b_s_list = build_linear_constraints(h_current)

                c5_current = evaluate_h(h_new, dx)

            return evaluate_h(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}

                best_c5 = evaluate_h(best_h, dx)

    return list(best_h), best_c5, n_points
