import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2*n for full convolution
    h_padded = np.pad(h, (0, n), 'constant')
    g_padded = np.pad(1 - h, (0, n), 'constant')
    H = fft(h_padded)
    G = fft(g_padded)
    conv = ifft(H * np.conj(G)).real[:2*n-1]
    max_val = np.max(conv) * dx
    return max_val

        return grad
