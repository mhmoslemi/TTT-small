def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h)*dx = 1"""
    dx = 2.0 / n_points
    target_sum = 1.0
    current_sum = np.sum(h)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    
    # Water-filling approach to enforce sum constraint
    if abs(current_sum - target_sum) > 1e-10:
        # Calculate difference
        diff = target_sum - current_sum
        # Distribute the difference
        if diff > 0:
            # Distribute to values < 1
            indices = np.where(h < 1)[0]
            if len(indices) > 0:
                distribute_per = diff / len(indices)
                for i in indices:
                    new_val = min(1.0, h[i] + distribute_per)
                    diff -= (new_val - h[i])
                    h[i] = new_val
        else:
            # Subtract from values > 0
            indices = np.where(h > 0)[0]
            if len(indices) > 0:
                subtract_per = -diff / len(indices)
                for i in indices:
                    new_val = max(0.0, h[i] - subtract_per)
                    diff += (h[i] - new_val)
                    h[i] = new_val

    return h

def compute_subgradient(h, m, dx):
    """Compute subgradient for a given lag m"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        # m >= 0 case
        indices = np.arange(n - m)
        for i in indices:
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        # m < 0 case, symmetric to m >= 0
        m = -m
        indices = np.arange(m, n)
        for i in indices:
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    return g * dx

def compute_best_lag(A):
    """Find the lag m* that maximizes A_m"""
    max_val = np.max(A)
    indices = np.where(A == max_val)[0]
    return indices, max_val

def symmetry_enforce(h):
    """Enforce symmetry h[i] = h[n-1-i]"""

    for i in range(n // 2):
        avg = (h[i] + h[n - 1 - i]) / 2
        h[i] = avg
        h[n - 1 - i] = avg
    return h
