import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_simplex(y, target_sum):
    """
    Project y onto the capped simplex: {x | 0 <= x_i <= 1, sum x_i = target_sum}
    """
    low, high = -np.max(y) - 1.0, np.max(y) + 1.0
    F = lambda l: np.sum(np.clip(y - l, 0, 1)) - target_sum
    for _ in range(40):
        mid = (low + high) / 2
        if F(mid) > 0:
            low = mid
        else:
            high = mid
    l = (low + high) / 2
    x = np.clip(y - l, 0, 1)
    return x

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_list = [30, 35, 40, 45, 50]
    best_h = None
    best_c5 = float('inf')
    
    # Phase 1: Solve SDP for varying n
    for n in n_list:
        if (time.time() - start_time) > 0.8 * budget_s:
            break
        try:
            # Variables
            X = Semidef(n)
            h = Variable(n)
            m = Variable()
            
            # Constraints
            constraints = [
                X >> 0,  # PSD
                h >= 0,
                h <= 1,
                sum(h) == n/2
            ]
            
            # Add autocorrelation constraints
            # Only compute for s in range(-(n-1), n-1)
            total_shifts = 2 * n - 2
            for s in range(-(n-1), n-1):
                if s >= 0:
                    r_s = sum([X[i, i+s] for i in range(n-s)])
                else:
                    r_s = sum([X[i, i+s] for i in range(-s, n)])
                constraints.append(r_s >= m)
            
            # Objective
            prob = Problem(Maximize(m), constraints)
            prob.solve(solver=ECOS, eps=1e-5, max_iters=500)
            if prob.status == "optimal":
                m_opt = m.value
                X_opt = X.value
                h_opt = h.value
                
                # Reconstruct h from X_opt (eigen-decomposition)
                evals, evecs = np.linalg.eigh(X_opt)
                v = np.abs(evecs[:, -1])  # take largest eigenvalue eigenvector
                scale = (n/2) / np.sum(v)
                h_reconstructed = v * scale
                h_clip = np.clip(h_reconstructed, 0, 1)
                h_clip = project_to_simplex(h_clip, n/2)
                
                c5_eval = evaluate_c5(h_clip)
                if c5_eval < best_c5:
                    best_h = h_clip.copy()
                    best_c5 = c5_eval
                    
        except Exception as e:
            # Continue with other n or skip
            continue
    
    # Phase 2: Local refinement with Nelder-Mead
    if best_h is not None and (time.time() - start_time) < 0.9 * budget_s:
        def objective(h):
            return evaluate_c5(h)
        
        # Refine using Nelder-Mead (includes projection)
        def fun(h):
            h_proj = project_to_simplex(h, len(h)/2)
            return objective(h_proj)
        
        try:
            res = minimize(fun, best_h, method='Nelder-Mead', options={'maxiter': 500})
            if res.success:
                h_refined = project_to_simplex(res.x, len(res.x)/2)
                c5_refined = evaluate_c5(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    # Phase 3: Upsample and fine-tune
    if (time.time() - start_time) < 0.95 * budget_s:
        # use a target n for final resolution
        n_target = 180
        dx = 2.0 / n_target
        # Interpolate low-res h to high-res box
        h_interp = np.interp(np.linspace(0, 2, n_target), 
                             np.linspace(0, 2, len(best_h)), 
                             best_h)
        # Project back to feasible region
        h_final = project_to_simplex(h_interp, n_target / 2.0)
        final_c5 = evaluate_c5(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final

    # Final sanity check
    dx = 2.0 / len(best_h)
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), float(final_c5), len(best_h))
