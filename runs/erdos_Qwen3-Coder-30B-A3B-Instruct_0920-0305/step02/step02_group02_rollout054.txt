def evaluate_h(h_values):
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute correlation of h with (1 - h) using FFT
    h_fft = fft(h, len(h)*2)
    corr = ifft(h_fft * np.conj(h_fft)).real
    corr = corr[len(h)-1 : 2*len(h)-1]
    r_values = len(h) / 2 - corr
    c5 = dx * np.max(r_values)
    return c5

def project_simplex(v, w):
    """Project vector v onto the capped simplex with sum = w."""
    v = np.array(v)
    n = len(v)
    if np.sum(v) == w:
        return v
    # Sort in descending order
    sorted_v = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(sorted_v)
    tau_vals = (cumsum - w) / np.arange(1, n+1)
    # Get the largest tau such that the projection is valid
    indices = np.arange(n)
    valid = (sorted_v - tau_vals) >= 0
    if np.sum(valid) == 0:
        # All projected values would be zero
        return np.zeros_like(v)
    k = np.argmax(valid)
    # Construct result
    theta = tau_vals[k]
    result = np.clip(v - theta, 0, 1)
    # Correct sum
    diff = w - np.sum(result)
    if abs(diff) > 1e-10:
        # Adjust by adding diff to first non-saturated coordinate
        for i in range(n):
            if result[i] < 1:
                incr = min(diff, 1 - result[i])
                result[i] += incr
                diff -= incr
                if abs(diff) < 1e-10:
                    break
    return result

def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable log-sum-exp."""
    x_max = np.max(x, axis=axis, keepdims=True)
    x_max = np.nan_to_num(x_max, nan=0)
    return np.log(np.sum(np.exp(x - x_max), axis=axis, keepdims=keepdims)) + x_max

def smooth_max_r(h, alpha=1.0):
    """Smooth approximation to max_k r_k with parameter alpha."""
    w = len(h) / 2
    n = len(h)
    # Compute r_k for all shifts k
    h_fft = fft(h, 2*n)
    corr = ifft(h_fft * np.conj(h_fft)).real
    corr = corr[n-1 : 2*n-1]
    r_vals = w - corr
    log_weights = alpha * r_vals
    log_weights -= np.max(log_weights)  # prevent overflow
    wts = np.exp(log_weights)
    wts = wts / np.sum(wts)
    smooth_max = np.sum(wts * r_vals)
    return smooth_max

def smooth_max_r_grad(h, alpha=1.0):
    """Compute gradient of smooth max approximation."""
    w = len(h) / 2
    n = len(h)
    h_fft = fft(h, 2*n)
    corr = ifft(h_fft * np.conj(h_fft)).real
    corr = corr[n-1 : 2*n-1]
    r_vals = w - corr # r_k = w - S_k
    log_weights = alpha * r_vals
    log_weights -= np.max(log_weights)  # stabilize
    wts = np.exp(log_weights)
    wts = wts / np.sum(wts)
    
    # Compute ∇r_k for all k
    one = np.ones_like(h)
    grad_list = []
    for k in range(n):
        if k == 0:
            g = one - 2. * h
        else:
            g = one.copy()
            g[k:] -= h[:-k]
            g[:k] -= h[-k:]
        grad_list.append(g)
    
    # Weighted sum
    grad = np.zeros_like(h)
    for i, g in enumerate(grad_list):
        grad += wts[i] * g
    return grad

def accelerated_projected_gradient(h0, alpha=1.0, max_iter=200, tol=1e-7):

    y = h0.copy()
