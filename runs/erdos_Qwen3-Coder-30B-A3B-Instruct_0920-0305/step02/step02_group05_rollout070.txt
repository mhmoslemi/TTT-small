import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto bounds [0,1] and sum constraint"""
    # First clip to bounds
    w = np.clip(v, 0.0, 1.0)
    total = w.sum()
    
    if abs(total - target_sum) < 1e-12:
        return w
    
    # Water-filling algorithm
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < 1.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        else:
            free = (w > 0.0)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        total = w.sum()
    return w

def interpolate_to_n(h0, n):
    """Interpolate h0 to new length n using linear interpolation"""
    old_n = len(h0)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, n, endpoint=False)
    h = np.interp(new_x, old_x, h0)
    # Adjust to meet exact sum constraint
    current_sum = np.sum(h)
    target_sum = n / 2
    scale = target_sum / current_sum if current_sum > 1e-12 else 1.0
    h *= scale
    return project_to_bounds_and_sum(h, target_sum)

def compute_max_overlap(h):
    """Compute overlap integral and return max and index"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    conv *= dx
    max_val = np.max(conv)
    argmax = np.argmax(conv)
    shift_m = argmax - (n - 1)
    return max_val, argmax, shift_m

def compute_gradient_for_shift(h, m):
    """Compute gradient of overlap integral for a given shift"""
    n = len(h)
    g = np.zeros_like(h)
    if m >= 0:
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:  # m < 0
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * (2.0 / n)  # dx factor
