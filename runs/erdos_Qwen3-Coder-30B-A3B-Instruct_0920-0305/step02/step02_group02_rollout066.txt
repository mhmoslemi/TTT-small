def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * (2.0 / len(h))
    return c5

def solve_sdp(n, dx):
    # Define variables for the SDP relaxation
    h = cp.Variable(n)
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Define constraints
    constraints = [
        cp.diag(X) == cp.multiply(h, h),  # diag(X) = h .* h
        0 <= h, h <= 1,
        cp.sum(h) == n / 2,
        X >> 0,  # X is positive semidefinite
    ]

    # Add constraints on each Toeplitz matrix
    T_matrices = []
    for k in range(n):
        T = np.zeros((n, n))
        for i in range(n):
            if i + k < n:
                T[i, i + k] = 1
        T_matrices.append(T)

    for T in T_matrices:
        constraints.append(cp.trace(T @ X) >= t)

    # Objective: maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value, X.value
        else:
            return None, None
    except Exception:
        return None, None

def project_to_capped_simplex(h_vec, w):
    """Project h_vec onto the capped simplex with sum = w."""
    # Sort values descending
    sorted_h = np.sort(h_vec)[::-1]
    # Water-filling to find the threshold tau
    cumulative = np.cumsum(sorted_h)
    idx = np.arange(1, len(sorted_h) + 1)
    # Find where the projected values sum to w
    masks = (sorted_h * idx - cumulative + w) > 0
    if np.any(masks):
        k = np.max(np.where(masks) + 1)
    else:
        k = 0
    # Compute the threshold
    if k == 0:
        tau = (np.sum(sorted_h) - w) / len(sorted_h) if len(sorted_h) > 0 else 0
    else:
        tau = (cumulative[k - 1] - w) / k
    # Project
    h_projected = np.clip(sorted_h - tau, 0, 1)
    # Re-sort back to original order
    h_projected_reshuffled = np.empty_like(h_vec)
    perm = np.argsort(h_vec)[::-1]
    h_projected_reshuffled[perm] = h_projected
    return h_projected_reshuffled

def round_sdp_solution(h_sdp, X_sdp, n):
    # Cholesky factorization
    try:
        R = cholesky(X_sdp, lower=True)
        # Generate random vector and compute rounded vector
        g = np.random.normal(0, 1, R.shape[0])
        h_round = R.T @ g
    except Exception:
        # Fallback to deterministic rounding using leading eigenvector
        try:
            lamb, V = eigh(X_sdp)
            v_best = V[:, -1]
            h_round = np.sqrt(np.max(lamb)) * v_best
        except Exception:
            # Fallback to projection of h_sdp
            h_round = h_sdp.copy()
    h_round = project_to_capped_simplex(h_round, n / 2)
    return h_round

def fit_fourier_coefficients(h_vals, M=10):
    n = len(h_vals)
    # Create basis matrix
    i_indices = np.arange(n)
    X = np.zeros((n, 2*M + 1))
    X[:, 0] = 1  # constant
    for m in range(1, M+1):
        X[:, 2*m - 1] = np.cos(np.pi * m * i_indices / n)
        X[:, 2*m] = np.sin(np.pi * m * i_indices / n)

    # Solve least-squares using pseudo-inverse
    try:
        coeffs = np.linalg.lstsq(X, h_vals, rcond=None)[0]
    except Exception:
        # Fallback to regularized solve
        coeffs, _ = np.linalg.lstsq(X, h_vals, rcond=1e-10)
    return coeffs

def construct_h_from_fourier(coeffs, n):
    M = (len(coeffs) - 1) // 2
    i_indices = np.arange(n)
    h_vals = coeffs[0] * np.ones(n)
    for m in range(1, M+1):
        h_vals += coeffs[2*m - 1] * np.cos(np.pi * m * i_indices / n)
        h_vals += coeffs[2*m] * np.sin(np.pi * m * i_indices / n)
    return h_vals

def compute_pairwise_correlations(h_vec):
    """Compute overlaps and return the maximum."""
    n = len(h_vec)
    S = np.zeros(n)
    for k in range(n):
        # Compute the overlap at shift k using FFT
        conv = np.correlate(h_vec, np.flip(1 - h_vec), mode='full')
        S[k] = conv[n-1+k]  # This is the shift k overlap
    return np.min(S)

def local_refinement_fourier(var0, h_fourier, n, dx):
    """Use projected gradient descent on the Fourier representation."""
    # Define the smooth surrogate function
    def surrogate(params, alpha=1):
        h_vals = construct_h_from_fourier(params, n)
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize to correct integral
        scale = (n / 2) / np.sum(h_vals)
        h_vals *= scale
        # Compute correlation-based surrogate
        if alpha < 1e-6:
            # Direct evaluation for small alpha - but not for large.
            # Here we use a large enough alpha to prevent numerical issues
            alpha = 1.
        # Use soft max
        corr_vals = []
        for k in range(n):
            corr = np.sum(h_vals * np.roll(1 - h_vals, k))
            corr_vals.append(corr)
        corr_vals = np.array(corr_vals)
        inv_alpha = 1.0 / alpha
        F = inv_alpha * np.log(np.sum(np.exp(alpha * corr_vals)))
        return F, corr_vals

    def gradient(params, alpha=1):
        h_vals = construct_h_from_fourier(params, n)
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize
        scale = (n / 2) / np.sum(h_vals)
        h_vals *= scale
        corr_vals = []
        for k in range(n):
            corr = np.sum(h_vals * np.roll(1 - h_vals, k))
            corr_vals.append(corr)
        corr_vals = np.array(corr_vals)
        # Compute normalized weights
        w = np.exp(alpha * corr_vals)
        w = w / np.sum(w)
        # Gradient is sum of partial derivatives w.r.t. each coefficient
        grad = np.zeros_like(params)
        # We approximate the gradient using finite differences since
        # the analytic expressions get complex here.
        eps = 1e-5
        for i in range(len(params)):
            pert = np.zeros_like(params)
            pert[i] = eps
            f_plus, _ = surrogate(params + pert, alpha)
            f_minus, _ = surrogate(params - pert, alpha)
            grad[i] = (f_plus - f_minus) / (2 * eps)
        return grad

    # Use BFGS with refinement
    result = minimize(surrogate, var0, args=(1.), method='L-BFGS-B',
                      jac=gradient, options={'maxiter': 200},
                      bounds=[(-10., 10.) for _ in range(len(var0))])

    if result.success:
        final_params = result.x
        h_final = construct_h_from_fourier(final_params, n)
        h_final = np.clip(h_final, 0, 1)
        # Normalize to correct sum
        scale = (n / 2) / np.sum(h_final)
        h_final *= scale
        return h_final
    else:
        return construct_h_from_fourier(var0, n)
