```python
import numpy as np
from scipy.fft import fft, ifft
import time
from cvxpy import *
import cvxpy as cp

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2n for full correlation
    h_pad = np.pad(h, (0, n), 'constant')
    g_pad = np.pad(1 - h, (0, n), 'constant')
    H = fft(h_pad)
    G = fft(g_pad)
    corr = ifft(H * np.conj(G)).real[:2*n - 1]
    max_val = np.max(corr) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n = 48  # Moderate even number works well
    dx = 2.0 / n
    T_target = int(np.ceil(0.3807 * n / 2))

    # Split budget: 35% for MILP, 35% for local search, 20% for smooth refinement
    time_left = budget_s
    time_milp = min(0.35 * time_left, 150)
    time_local = min(0.35 * time_left, 150)
    time_smooth = min(0.20 * time_left, 100)

    # Phase 1: Solve exact binary MILP
    h_bin = None
    T_opt = float('inf')

    try:
        h_vars = [Bool() for i in range(n)]
        p_vars = [[Bool() for i in range(n)] for l in range(n)]  # p[l][i]
        T = Int()

        # Sum of h[i] = n/2
        prob = Problem(Minimize(T), [
            sum(h_vars) == n//2,
            T >= 0
        ])

        # Product constraints: p[ℓ][i] = h[i]*h[i+ℓ]
        for l in range(1, n):
            for i in range(n - l):
                prob.constraints.extend([
                    p_vars[l][i] <= h_vars[i],
                    p_vars[l][i] <= h_vars[i + l],
                    p_vars[l][i] >= h_vars[i] + h_vars[i + l] - 1,
                    p_vars[l][i] >= 0
                ])
                
                # Lag sum constraint: ∑ p[ℓ][i] + T ≥ n/2
                # => T ≥ n/2 - ∑ p[ℓ][i]
                prob.constraints.extend([
                    T >= n//2 - sum(p_vars[l][i] for i in range(n-l))
                ])

        prob.solve(solver=CPLEX, verbose=False, maxSeconds=time_milp)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_bin = [int(h_var.value) for h_var in h_vars]
            T_opt = int(T.value)
        elif problem.status in [INFEASIBLE, UNBOUNDED]:
            # Fallback: No optimal solution found.
            pass
        else:
            # Fallback: Could not solve or found no feasible solution.
            pass
    except Exception:
        pass

    # If no binary MILP solution obtained, fall back to initial construction
    if h_bin is None or T_opt >= T_target:
        h_bin = np.array(initial_h_values, dtype=int)
        if len(h_bin) != n:
            h_bin = np.resize(h_bin, n)
    
    # Phase 2: Binary local search via simulated annealing
    if time.time() - start_time < time_local:
        best_h = h_bin.copy()
        best_c5 = evaluate_h(best_h)
        if best_c5 <= 0.3807:
            # Early stopping
            return (best_h.tolist(), best_c5, n)

        # Initial temperature estimate
        n_swaps = 500
        dE_list = []
        for _ in range(n_swaps):
            # Swap two random binary indices
            i1 = np.random.choice(np.where(best_h == 1)[0])
            i2 = np.random.choice(np.where(best_h == 0)[0])
            new_h = best_h.copy()
            new_h[i1], new_h[i2] = 0, 1
            c5_new = evaluate_h(new_h)
            dE_list.append(c5_new - best_c5)
            if c5_new < best_c5:
                best_h = new_h
                best_c5 = c5_new
        T0 = np.median(np.abs(dE_list)) * 2

        # Annealing schedule
        T = T0
        while T > 1e-6 and time.time() - start_time < time_local + start_time - time.time():
            # Try one swap
            i1 = np.random.choice(np.where(best_h == 1)[0])
            i2 = np.random.choice(np.where(best_h == 0)[0])
            new_h = best_h.copy()
            new_h[i1], new_h[i2] = 0, 1
            c5_new = evaluate_h(new_h)
            delta = c5_new - best_c5
            if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                best_h = new_h
                best_c5 = c5_new
                if best_c5 <= 0.3807:
                    break
            T *= 0.999
            
    # Phase 3: Fractional descent with log-sum-exp smoothing
    if time.time() - start_time < time_smooth + start_time - time.time() and len(best_h) == n:
        h_frac = best_h.astype(float)
        h_frac = h_frac / (np.sum(h_frac) * dx) * 1.0  # Ensure exact integral

        # Gradient-based descent with increasing beta
        def smooth_max_grad(h, beta):
            h_pad = np.pad(h, (0, n), mode='constant')
            g_pad = np.pad(1 - h, (0, n), mode='constant')
            H = fft(h_pad)
            G = fft(g_pad)
            corr = ifft(H * np.conj(G)).real[:2*n - 1]
            # evaluate log sum exp with all lags
            if np.min(corr) < 0:
                corr = corr - np.min(corr) + 1e-10
            weights = np.exp(beta * corr)
            weights /= np.sum(weights)

            # gradient of each lag
            grad_lags = []
            for l in range(1, n):  # lag l ranges from 1 to n-1
                g_i = np.zeros_like(h)
                g_i[:-l] = 1 - h[l:]  # same as 1 - h[i+l]
                g_i[l:] += 1 - h[:-l]  # same as 1 - h[i-l]
                grad_lags.append(g_i)

            # weighted sum of gradients
            g_smooth = np.zeros_like(h)
            for w, g in zip(weights, grad_lags):
                g_smooth += w * g
            return g_smooth

        # water-filling for mass projection
        def project_mass(h, target_mass):
            # h: array of values
            # target_mass: sum constraint
            def solve_waterfilling(h_clip, target):
                def waterfill_func(lam):
                    new_h = np.clip(h_clip + lam, 0, 1)
                    return np.sum(new_h) - target
                
                # find λ via binary search
                low, high = -1, 1
                for _ in range(50):
                    mid = (low + high) / 2
                    val = waterfill_func(mid)
                    if val < 0:
                        low = mid
                    else:
                        high = mid
                return (low + high) / 2

            target_mass = target_mass
            h_proj = h.copy()
            lambda_opt = solve_waterfilling(h_proj, target_mass)
            h_proj = np.clip(h_proj + lambda_opt, 0, 1)
            return h_proj

        # Anneal beta from 1 to high values
        beta_values = [1, 2, 5, 10, 20]
        for beta in beta_values:
            steps_per_beta = 50
            for _ in range(steps_per_beta):
                if time.time() - start_time > time_smooth + start_time - time.time():
                    break
                grad = smooth_max_grad(h_frac, beta)
                step_size = 0.01 / (beta ** 0.5)
                h_frac -= step_size * grad
                h_frac = project_mass(h_frac, n/2.0 * dx)
                # Ensure bounds
                h_frac = np.clip(h_frac, 0, 1)

        # Final evaluation with true h
        best_h = h_frac.copy()
        best_c5 = evaluate_h(best_h)

    # Finalize solution
    # Ensure weight constraint is satisfied exactly
    if time.time() - start_time < 5:
        # Clamp and reweight to exact mass
        mass_current = np.sum(best_h) * dx
        if abs(mass_current - 1.0) > 1e-9:
            best_h = best_h / mass_current

    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n)
```
