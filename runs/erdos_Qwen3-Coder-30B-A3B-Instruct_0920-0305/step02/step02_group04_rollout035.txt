import numpy as np
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT."""
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, 2*n)
    G = np.fft.rfft(1.0 - h, 2*n)
    conv = np.fft.irfft(H * np.conj(G))[:2*n-1]
    return np.max(conv.real) * dx

def solve_milp(n, time_limit=30):
    """Solve the MILP formulation for binary h to get min T."""
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True, nonneg=True)
    p = []
    constraints = [cp.sum(h) == M, T >= 0]

    for ell in range(1, n):
        length = n - ell
        p_ell = cp.Variable(length, boolean=True)
        p.append(p_ell)
        constraints += [
            p_ell <= h[:length],
            p_ell <= h[ell:],
            p_ell >= h[:length] + h[ell:] - 1,
        ]
        constraints += [cp.sum(p_ell) + T >= M]

    prob = cp.Problem(cp.Minimize(T), constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False, maxSeconds=time_limit)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = np.ravel(h.value)
            T_val = float(T.value)
            return T_val, h_val
        elif prob.status == cp.INFEASIBLE:
            return None, None
        else:
            # Fallback to best feasible point
            h_val = np.ravel(h.value)
            if h_val is None:
                return None, None
            # Compute approximate T from constraints (max of LP dual)
            # For demo simplicity, assume T is approximated by constraint
            # We compute a valid point by hand as a rough guess
            score = evaluate_h(h_val)
            return score / (2.0/n), h_val
    except:
        return None, None

def smooth_descent(h0, time_limit=10):
    """Refine a binary h to fractional values using smooth log-sum-exp."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    M = n // 2
    
    def proj_simplex(h, target_sum=M):
        """Project h to satisfy sum(h) = target_sum and h_i in [0,1]."""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        
        # water-filling approach
        h_sorted = np.sort(h)[::-1]
        cumsum = np.cumsum(h_sorted)
        idx = np.argmax(h_sorted * np.arange(1, len(h_sorted) + 1) > (target_sum - cumsum + h_sorted))
        if idx == 0 and h_sorted[0] * len(h_sorted) < target_sum:
            # All elements can be scaled up
            scale = target_sum / current_sum
            return h * scale
        else:
            # Use water-filling bisection on lambda
            low, high = 0.0, 1.0
            # Simple binary search approach:
            for _ in range(50):
                lam = (low + high) / 2.0
                new_h = np.clip(h - lam, 0, 1)
                if np.sum(new_h) > target_sum:
                    low = lam
                else:
                    high = lam
            lam = (low + high) / 2.0
            h = np.clip(h - lam, 0, 1)
            # Rescale if needed to hit exact sum
            total = np.sum(h)
            if abs(total - target_sum) > 1e-8:
                h *= target_sum / total
            return h
    
    # Main loop over β values
    beta_schedule = [1, 2, 5, 10, 20]
    # Compute initial overlaps
    def compute_lags(h):
        lags = []
        for ell in range(1, n):
            s = 0.0
            for i in range(n - ell):
                s += h[i] * (1 - h[i + ell])
            lags.append(s)
        return np.array(lags)
    
    def compute_grad(h, lags):
        # gradient of Lagrangian for maximize min overlap
        # We use lifted objective as max over lags
        # But for computation, use log-sum-exp surrogate
        weights = np.exp(beta * lags)
        weights = weights / np.sum(weights)
        g = np.zeros_like(h)        
        for ell in range(1, n):
            if ell >= n:
                continue
            for i in range(n - ell):
                g[i] += weights[ell - 1] * (1 - h[i + ell])
                g[i + ell] += weights[ell - 1] * (-h[i])  # terms from 1-h_{i+ell}
        return g
    
    h = proj_simplex(h)
    
    start_time = time.time()
    for beta in beta_schedule:
        if time.time() - start_time > time_limit:
            break
        # gradient descent with line search for direction and step size
        for step in range(20):
            if time.time() - start_time > time_limit:
                break
            lags = compute_lags(h)
            grads = compute_grad(h, lags)
            # gradient step
            eta = 0.1
            h_new = h - eta * grads
            # backtracking step size
            while True:
                h_test = proj_simplex(h_new)
                if evaluate_h(h_test) <= evaluate_h(h) + 1e-12:
                    h = h_test
                    break
                eta *= 0.5
                h_new = h - eta * grads
                if eta < 1e-8:
                    break
    return h, evaluate_h(h)

def anneal_binary(h0, max_iter=1000000, time_limit=20):
    """Anneal binary h with swaps using FFT for fast evaluation."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    M = n // 2

    initial_score = evaluate_h(h)
    current_score = initial_score
    best_h = h.copy()
    best_score = initial_score

    # Initial temperature estimation using 1000 swaps
    temp = 1.0
    count = 0
    sum_dE = 0.0
    while count < 1000 and time.time() < time.time() + time_limit * 0.1:
        # Pick a random 1 and a random 0
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap elements
        h[i], h[j] = h[j], h[i]
        new_score = evaluate_h(h)
        dE = new_score - current_score
        count += 1
        if abs(dE) > 1e-10:
            sum_dE += dE
        h[i], h[j] = h[j], h[i]  # Revert swap
    if count > 0:
        temp = sum_dE / count
    if temp <= 0:
        temp = 1.0

    start_time = time.time()
    cool_rate = 0.999
    iteration = 0
    stay_count = 0
    patience = 20000  # reheat after many iterations without progress

    h = h.copy()
    while iteration < max_iter and time.time() - start_time < time_limit:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # Swap elements
        h[i], h[j] = h[j], h[i]
        new_score = evaluate_h(h)
        dE = new_score - current_score
        if dE < 0 or np.random.rand() < np.exp(-dE / temp):
            current_score = new_score
            if current_score < best_score:
                best_h = h.copy()
                best_score = current_score
            stay_count = 0
        else:
            h[i], h[j] = h[j], h[i]
        temp *= cool_rate
        iteration += 1
        stay_count += 1
        if stay_count > patience:
            # Reheat
            temp = 1.0
            stay_count = 0

    return best_h, best_score
