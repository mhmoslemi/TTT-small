import numpy as np
from scipy.optimize import minimize
import time

# Constants
n_values = [48, 56, 60]
dx = 2.0 / 48  # default
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def project_onto_constraints(v, target_sum):
    """Project vector v onto the set {x | sum(x) = target_sum, 0 <= x <= 1}."""
    def f(λ):
        # Clip v - λ and compute sum
        x = np.clip(v - λ, 0, 1)
        return np.sum(x) - target_sum

    # Binary search for lambda such that f(λ) = 0
    a, b = np.min(v), np.max(v) + 1.0
    for _ in range(50):
        λ = (a + b) / 2.0
        if f(λ) > 0:
            a = λ
        else:
            b = λ
    λ = (a + b) / 2.0
    return np.clip(v - λ, 0, 1)

def compute_shifted_h(h, s):
    """Compute A^{(s)} h where A^{(s)} is the shift matrix."""
    n = len(h)
    if s == 0:
        return h
    elif s > 0:
        # (shifted forward + shifted backward) / 2
        shifted_f = np.concatenate([h[s:], np.zeros(s)])
        shifted_b = np.concatenate([np.zeros(s), h[:-s]])
        return 0.5 * (shifted_f + shifted_b)
    else:
        # For negative s, equivalent to s = -s
        return compute_shifted_h(h, -s)

def compute_all_Cs(h):
    """Compute all C_s values for given h."""
    n = len(h)
    Cs = []
    for s in range(n):
        # Compute O_s(h) = sum h_i * h_{i+s}
        shifted_h = compute_shifted_h(h, s)
        O_s = np.sum(h * shifted_h)
        C_s = dx * O_s
        Cs.append(C_s)
    return np.array(Cs)

def smooth_min_objective(h, β):
    """Compute Φ_β(h) = - (1/β) * log(sum_i exp(-β * C_i(h)))"""
    Cs = compute_all_Cs(h)
    # Subtract maximum to avoid overflow
    max_C = np.max(Cs)
    exp_neg_beta_Cs = np.exp(-β * (Cs - max_C))
    sum_exp = np.sum(exp_neg_beta_Cs)
    return -(1.0 / β) * np.log(sum_exp) + max_C / β

def compute_gradient(h, β):
    """Compute ∇Φ_β(h) = 2 * dx * Σ_s w_s * A^{(s)} h"""
    n = len(h)
    Cs = compute_all_Cs(h)

    # Compute weights
    max_C = np.max(Cs)
    exp_neg_beta_Cs = np.exp(-β * (Cs - max_C))
    Z = np.sum(exp_neg_beta_Cs)
    weights = exp_neg_beta_Cs / Z

    # Compute weighted sum of A^{(s)} h
    g = np.zeros_like(h)
    for s in range(n):
        w_s = weights[s]
        shifted_h = compute_shifted_h(h, s)
        g += w_s * shifted_h

    return 2 * dx * g

def backtracking_line_search(h, grad, β, α0=0.5, max_backtracks=10):
    """Perform backtracking line search for step size α"""
    α = α0
    h_current = h
    C_current = compute_all_Cs(h_current)
    t_current = np.min(C_current)

    for _ in range(max_backtracks):
        h_new = project_onto_constraints(h_current - α * grad, n/2)
        C_new = compute_all_Cs(h_new)
        t_new = np.min(C_new)
        if t_new < t_current:
            return h_new, α
        α /= 2.0
    return h_current, α

def gradient_ascent_with_continuation(h_start, n, max_iter=500):
    """Perform projected gradient ascent with β continuation."""
    h = h_start.copy()
    target_sum = n / 2
    β_schedule = [5, 15, 30, 60, 120]
    best_h = h.copy()
    best_t = np.min(compute_all_Cs(h))
    
    for β in β_schedule:
        for iter in range(200):  # Max iterations per β
            if iter > 10 and iter % 10 == 0:
                C_vals = compute_all_Cs(h)
                t_current = np.min(C_vals)
                if t_current >= best_t - 1e-6:
                    break  # No improvement

            grad = compute_gradient(h, β)
            h_new, α = backtracking_line_search(h, grad, β)
            C_vals = compute_all_Cs(h_new)
            t_new = np.min(C_vals)
            if t_new < best_t:
                best_t = t_new
                best_h = h_new.copy()
            h = h_new
    return best_h, best_t

        h_unif = np.random.rand(n)
        h_unif = project_onto_constraints(h_unif, target_sum)
