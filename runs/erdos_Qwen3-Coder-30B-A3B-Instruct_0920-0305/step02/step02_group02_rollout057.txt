import numpy as np
from scipy.optimize import minimize
from scipy.linalg import qr
import cvxpy as cp
from cvxpy import SDP, Variable, trace, sum_entries, diag, Promote, multiply, Minimize, Problem
from scipy.fftpack import fft, ifft
import time
import math

# Parameters for the algorithm
n_list = [48, 56, 64, 80]  # Discretisation sizes to try
max_iter = 500  # Iterations for local refinement
num_rounds = 20  # Number of rounding for each SDP result
max_budget_s = 600
initial_seed = 42

def reconstruct_h_from_fourier(freq_coeffs, n):
    """Reconstruct a function from low-frequency Fourier components."""
    a, b = freq_coeffs[:len(freq_coeffs)//2], freq_coeffs[len(freq_coeffs)//2:]
    x = np.arange(n)
    h = 0.5 * np.ones(n)
    for m in range(1, len(a)+1):
        h += a[m-1] * np.cos(np.pi * m * x / n) + b[m-1] * np.sin(np.pi * m * x / n)
    return np.clip(h, 0, 1)

def project_to_simplex(v, w):
    """Project vector v onto capped simplex: 0 <= v_i <= 1, sum v_i = w."""
    v_sorted = np.sort(v)[::-1]
    tmp_sum = 0
    for i in range(len(v_sorted)):
        tmp_sum += v_sorted[i]
        if tmp_sum > (i+1) * (w - (tmp_sum - v_sorted[i]) / (i+1)):
            break
    theta = (tmp_sum - w) / (i+1)
    return np.clip(np.maximum(v - theta, 0), 0, 1)

def solve_sdp(n, dx, w):
    """Solve the SDP relaxation of the problem."""
    # Matrix variable
    X = cp.Variable((n, n), PSD=True)
    h = cp.Variable(n)
    
    # Define Toeplitz matrices for correlation constraints
    T = []
    for k in range(n):
        T_k = np.zeros((n, n))
        for i in range(n - k):
            T_k[i, i+k] = 1
        T.append(T_k)
    
    constraints = [
        cp.diag(X) == cp.multiply(h, h),  # diag(X) = h^2
        cp.sum(h) == w,
        h >= 0,
        h <= 1
    ]
    
    for k in range(n):
        constraints.append(trace(T[k] @ X) >= 0)  # Trivial lower bound for now
    
    # Maximize a lower bound on the minimum overlap
    obj = cp.Maximize(0)  # Placeholder for optimization
    problem = Problem(obj, constraints)
    
    try:
        problem.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            return h.value, X.value
    except:
        pass
        
    return None, None

def run_rounding_sdp(h_sdp, X_sdp, n, w, num_rounds):
    """Create rounded versions of SDP solution."""
    try:
        U, s, Vt = np.linalg.svd(X_sdp)
        R = U @ np.sqrt(np.diag(s))
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(num_rounds):
            g = np.random.randn(R.shape[1])
            h_round = R.T @ g
            h_proj = project_to_simplex(h_round, w)
            
            # Compute actual C5
            c5 = compute_C5(h_proj, dx)
            if c5 < best_c5:
                best_h = h_proj
                best_c5 = c5
                
        return best_h, best_c5
    except:
        return None, float('inf')

def compute_C5(h_values, dx):
    """Compute the overlap integral for a given h."""
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5

def compute_spectral_coeffs(h_values, M):
    """Fit a truncated Fourier series."""
    n = len(h_values)
    x = np.arange(n)
    A = np.column_stack([np.ones(n), np.cos(np.pi * np.outer(x, np.arange(1, M+1)[:, None])), 
                            np.sin(np.pi * np.outer(x, np.arange(1, M+1)[:, None]))])
    # Solve least squares with box constraints
    coeffs = np.linalg.lstsq(A, h_values, rcond=None)[0]
    return coeffs

def compute_grad_norm(h_values, dx):
    """Compute approximate gradient norm based on differences."""
    h = np.array(h_values)
    n = len(h)
    grad_norm = 0.0
    for k in range(n):
        shift = np.concatenate([h[k:], h[:k]])
        diff = shift - (1 - h)
        grad_norm += np.sum(diff**2)
    return grad_norm

def smooth_refinement(theta, M, n, dx, w, max_iter=500):
    """Perform smooth refinement of the reconstruction from Fourier coefficients."""
    def surrogate_loss(theta, alpha):
        h_recon = reconstruct_h_from_fourier(theta, n)
        h = np.clip(h_recon, 0, 1)
        # Compute overlap integral using FFT for efficiency
        h_fft = fft(h)
        one_minus_h_fft = fft(1 - h)
        conv = ifft(h_fft * np.conj(one_minus_h_fft))
        corr_vals = np.real(conv[:n])
        max_corr = np.max(corr_vals)
        # Using soft max for smooth surrogate
        return - dx * max_corr
    
    theta_opt = theta.copy()
    
    # Full domain refinement
    for alpha in [1., 10., 50., 100.]:
        def loss_fn(t):
            return surrogate_loss(t, alpha)
        bounds = [(-np.inf, np.inf)] * len(theta_opt)
        result = minimize(loss_fn, theta_opt, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if result.success:
            theta_opt = result.x
    
    # Post-adjustment
    h_final = reconstruct_h_from_fourier(theta_opt, n)
    h_final = np.clip(h_final, 0, 1)
    # Adjust the sum to exact w
    current_sum = np.sum(h_final)
    h_final *= (w / current_sum)
    return h_final

def refine_with_gradient_descent(h_start, dx, n, max_iter=300):
    """Refine a solution using gradient of correlation."""
    h = h_start.copy()
    for _ in range(max_iter):
        # Compute overlaps efficiently using FFT
        h_fft = fft(h)
        one_minus_h_fft = fft(1 - h)
        conv = ifft(h_fft * np.conj(one_minus_h_fft))
        corr_vals = np.real(conv[:n])
        min_k = np.argmin(corr_vals)
        # Approximate gradient from max correlation
        shift = np.concatenate([h[min_k:], h[:min_k]])
        grad = shift - (1 - h)  # gradient of S_{min_k}
        h = h - 0.001 * grad
        h = np.clip(h, 0, 1)
        h = h * (np.sum(h) / (n/2))  # Force correct sum
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = None
    start_time = time.time()
    
    # Scheduled runs over different n sizes
    for n in n_list:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        w = n / 2
        
        # For larger n, skip SDP (less efficient)
        if n >= 80:
            # Direct Fourier approach without SDP
            # Start with a reasonable function
            h0 = np.array([0.5] * n)
            # Random Gaussian modulation + project
            h0 += 0.1 * np.random.randn(n)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (w / np.sum(h0))
            
            # Fit and refine
            M = min(10, n // 4)
            coeffs = compute_spectral_coeffs(h0, M)
            h_fourier = reconstruct_h_from_fourier(coeffs, n)
            h_fourier = h_fourier * (w / np.sum(h_fourier))
            
            h_refined = smooth_refinement(coeffs, M, n, dx, w)
            c5 = compute_C5(h_refined, dx)
            if c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5
                best_n = n                
        else:
            # SDP approach
            h_sdp, X_sdp = solve_sdp(n, dx, w)
            if h_sdp is not None and X_sdp is not None:
                h_rounded, c5_rounded = run_rounding_sdp(h_sdp, X_sdp, n, w, num_rounds)
                if h_rounded is not None and c5_rounded < best_c5:
                    # Refine with Fourier + smooth optimization
                    M = min(10, n // 4)
                    coeffs = compute_spectral_coeffs(h_rounded, M)
                    h_refined = smooth_refinement(coeffs, M, n, dx, w)
                    final_c5 = compute_C5(h_refined, dx)
                    if final_c5 < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = final_c5
                        best_n = n
                
            # If no SDP attempt, use random initialization on smaller scale
            h0 = np.array([0.5] * n)
            h0 += 0.1 * np.random.randn(n)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (w / np.sum(h0))
            M = min(10, n // 4)
            coeffs = compute_spectral_coeffs(h0, M)
            h_final = smooth_refinement(coeffs, M, n, dx, w)
            c5_current = compute_C5(h_final, dx)
            if c5_current < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_current
                best_n = n
    
    # Final polish with gradient descent
    if best_h is not None:
        best_h = refine_with_gradient_descent(best_h, dx, best_n)
        best_c5 = compute_C5(best_h, dx / 2 * 2 / best_n)
        
    return list(best_h), best_c5, best_n if best_h is not None else 48
