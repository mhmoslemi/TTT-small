def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def dc_step(h, dx):
    """Perform one step of the DC algorithm: minimize the worst correlation"""
    n = len(h)
    # Compute all autocorrelations (for each shift s)
    auto = correlate(h, h, mode="full")
    # Corresponding shifts: from -n+1 to n-1
    shifts = np.arange(-n + 1, n)
    
    # Find the shift s that minimizes auto (gives max overlap)
    min_idx = np.argmin(auto)
    s_min = shifts[min_idx]
    
    # Compute the gradient of q_s(h) for this shift
    # q_s(h) = sum_i h_i * h_{i+s}
    # Grad_i q_s = h_{i+s} + h_{i-s} (with boundary handling)
    grad = np.zeros(n)
    if s_min > 0:
        grad[s_min:] += h[:-s_min]
        grad[:-s_min] += h[s_min:]
    elif s_min < 0:
        grad[:s_min] += h[-s_min:]
        grad[-s_min:] += h[:s_min]
    else:  # s_min == 0
        grad += 2 * h
        
    # Solve the linear program: minimize grad * h subject to:
    # 0 <= h[i] <= 1 for all i, sum h = n/2
    # This is a capped simplex projection
    # Sort indices by decreasing gradient
    idxs = np.argsort(-grad)
    
    h_new = np.zeros(n)
    total = 0
    for i in idxs:
        if total >= n/2:
            break
        if total + 1 <= n/2:
            h_new[i] = 1.0
            total += 1
        else:
            frac = n/2 - total
            h_new[i] = frac
            total += frac
    return h_new
