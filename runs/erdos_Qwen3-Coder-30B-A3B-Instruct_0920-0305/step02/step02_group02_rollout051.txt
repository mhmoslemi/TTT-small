# Constants

def evaluate_c5(h):
    """Evaluate the C_5 bound for given h values."""
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def gradient_S_k(h, k):
    """Compute the gradient of S_k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        # S_0 = sum(h_i^2)
        g = 2 * h
    else:
        # S_k = sum_i h_i * h_{i+k}
        # Grad is h_{i+k} + h_{i-k}
        if k < n:
            g[:n-k] += h[k:]  # For i such that i+k < n
            g[k:] += h[:n-k]  # For i such that i-k >= 0
    return g

def project_simplex(v, target_sum):
    """Project onto the simplex defined by 0 <= v_i <= 1 and sum = target_sum."""
    v = np.array(v)
    n = len(v)
    g = np.copy(v)
    
    # Sort in ascending order
    sorted_indices = np.argsort(v)
    sorted_v = v[sorted_indices]
    
    # Compute the cumulative sums
    cumsum = np.concatenate([[0], np.cumsum(sorted_v)])
    
    # Find the threshold
    threshold = -1e10
    for i in range(n):
        lambda_i = (cumsum[i] - target_sum) / (n - i)
        if lambda_i <= sorted_v[i]:
            threshold = lambda_i
            break
    if threshold == -1e10:
        threshold = (cumsum[n] - target_sum) / (n)
    
    # Compute the projection
    projected = np.maximum(0, np.minimum(1, v - threshold))
    
    # Scale to match target_sum
    scale = target_sum / np.sum(projected)
    projected *= scale
    
    return projected
