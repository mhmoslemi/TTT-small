def evaluate_c5(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * dx
    return c5

def autocorr_h(h):
    """Compute S_k = sum_i h_i * h_{i+k} for all k."""
    n = len(h)
    # Compute correlation of h with itself
    corr = np.correlate(h, h, mode='full')
    # S_k for k = 0, 1, ..., n-1
    S = corr[n-1:]  # This corresponds to k from 0 to n-1
    return S

def grad_S_k(h, k):
    """Gradient of S_k = sum h_i h_{i+k} w.r.t h."""
    n = len(h)
    if k == 0:
        return 2 * h
    else:
        g = np.zeros_like(h)
        if k > 0:
            g[:n-k] += h[k:]   # h_{i+k} contribute to ∂S_k/∂h_i
            g[k:] += h[:n-k]   # h_{i-k} contribute to ∂S_k/∂h_i
        else:
            # This case should not occur for k > 0
            pass
        return g

def project_onto_simplex_with_bounds(v, target_sum, lower=0, upper=1):
    """Project point v onto simplex with bounds."""
    v = np.array(v).copy()
    target_sum = float(target_sum)
    n = len(v)
    
    # First shift to non-negative
    u = v - lower
    sum_u = np.sum(u)
    if sum_u < 0:
        u = np.zeros_like(u)
    elif sum_u > (upper-lower)*n:
        u = np.ones_like(u) * (upper-lower)
    else:
        # Use sorting-based projection
        u_sorted = np.sort(u)[::-1]
        tmp_sum = 0.0
       rho = 0
        for i in range(n):
            tmp_sum += u_sorted[i]

                pass
            else:
                rho = (tmp_sum - target_sum) / (i+1)
                break
        else:
            rho = (tmp_sum - target_sum) / n
                
        u = np.maximum(u - rho, 0)
        u = np.minimum(u, upper - lower)
    
    # Apply transformation back to original bounds
    return np.clip(u + lower, lower, upper)
