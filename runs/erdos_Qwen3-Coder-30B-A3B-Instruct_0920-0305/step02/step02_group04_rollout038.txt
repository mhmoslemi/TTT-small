import numpy as np
from scipy.fft import fft, ifft
import cvxpy as cp
import time

def evaluate_h(h):
    """Evaluate the C5 constant for a given h using FFT-based correlation."""
    n = len(h)
    dx = 2.0 / n
    # Pad to 2n for full convolution
    h_padded = np.pad(h, (0, n), mode='constant')
    h_conj = np.pad(1 - h, (0, n), mode='constant')
    corr = ifft(fft(h_padded) * np.conj(fft(h_conj)))
    corr_real = np.real(corr[:2*n-1])
    max_val = np.max(corr_real) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 48
    M = n // 2
    dx = 2.0 / n
    T_target = int(np.ceil(0.3807 * n / 2))
    
    # Phase 1: Binary MILP
    h_binary = None
    T_opt = None
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        try:
            # Model variables
            h = cp.Variable(n, boolean=True)
            p = cp.Variable((n, n-1), boolean=True)
            T = cp.Variable(boolean=True)
            
            # Constraints
            constraints = [
                cp.sum(h) == M
            ]
            for l in range(1, n):
                for i in range(n - l):
                    # p[l][i] = h[i] AND h[i+l]
                    constraints.extend([
                        p[i, l-1] <= h[i],
                        p[i, l-1] <= h[i+l],
                        p[i, l-1] >= h[i] + h[i+l] - 1,
                        p[i, l-1] >= 0
                    ])
            # Worst lag
            for l in range(1, n):
                constraints.append(
                    cp.sum(p[:, l-1]) + T >= M
                )
            
            # Objective
            objective = cp.Minimize(T)
            
            # Solve
            prob = cp.Problem(objective, constraints)
            # Set solver and time limit
            prob.solve(solver=cp.CBC, maxSeconds=0.30 * remaining)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_binary = h.value.astype(int)
                T_opt = int(T.value)
            else:
                # Fallback to a heuristic
                h_binary = np.zeros(n, dtype=int)
                h_binary[:M] = 1
                np.random.shuffle(h_binary)
                h_binary = np.array(h_binary)
                T_opt = T_target + 1  # Assume we haven't yet found the right bound
        except Exception:
            # Fallback to random
            h_binary = np.zeros(n, dtype=int)
            h_binary[:M] = 1
            np.random.shuffle(h_binary)
            h_binary = np.array(h_binary)
            T_opt = T_target + 1  # Assume we haven't yet found the right bound
    else:
        # Fallback heuristic
        h_binary = np.zeros(n, dtype=int)
        h_binary[:M] = 1
        np.random.shuffle(h_binary)
        h_binary = np.array(h_binary)
        T_opt = T_target + 1  # Assume we haven't yet found the right bound

    # Evaluate binary once
    if h_binary is not None:
        current_c5 = evaluate_h(h_binary)
    else:
        # This shouldn't happen
        current_c5 = 1.0
        h_binary = np.zeros(n, dtype=int)
        
    # Check if we already have a good enough binary solution
    if T_opt is not None and T_opt <= T_target:
        # Final resort: project towards integral
        h = h_binary.astype(float)
        scale = 1.0 / (np.sum(h) * dx)
        h *= scale
        return (h.tolist(), evaluate_h(h), n)

    # Phase 2: Binary local search (swap-based SA)
    if remaining > 20:
        # Annealing swaps
        h_current = h_binary.copy()
        current_c5 = evaluate_h(h_current)
        best_h = h_current.copy()
        best_c5 = current_c5
        T = 0.1 * current_c5
        cooling_rate = 0.999
        min_temp = 1e-6
        t_start = time.time()
        time_limit = 0.35 * remaining
        while time.time() - t_start < time_limit and T > min_temp:
            # Flip a 1 and a 0
            ones_indices = np.where(h_current == 1)[0]
            zeros_indices = np.where(h_current == 0)[0]
            if ones_indices.size > 0 and zeros_indices.size > 0:
                i = np.random.choice(ones_indices)
                j = np.random.choice(zeros_indices)
                h_new = h_current.copy()
                h_new[i] = 0
                h_new[j] = 1
                new_c5 = evaluate_h(h_new)
                delta = new_c5 - current_c5
                if delta <= 0 or np.random.rand() < np.exp(-delta / T):
                    h_current = h_new
                    current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
            T *= cooling_rate
        h_binary = best_h.copy()
        current_c5 = best_c5

    # Phase 3: Fractional smoother (projected gradient descent)
    if remaining > 5 and current_c5 > 0.379:
        # Good starting point
        h_smooth = h_binary.astype(float)
        scale = 1.0 / (np.sum(h_smooth) * dx)
        h_smooth *= scale
        current_c5 = evaluate_h(h_smooth)
        
        # Smooth surrogate
        def smooth_max(h, beta=1.0):
            n = len(h)
            h_padded = np.pad(h, (0, n), mode='constant')
            h_conj = np.pad(1 - h, (0, n), mode='constant')
            corr = ifft(fft(h_padded) * np.conj(fft(h_conj)))
            corr_real = np.real(corr[:2*n-1])
            corr_real *= dx  # scale by dx
            exps = np.exp(beta * corr_real)
            total = np.sum(exps)
            return np.log(total) / beta

        def grad_smooth(h, beta=1.0):
            n = len(h)
            h_padded = np.pad(h, (0, n), mode='constant')
            h_conj = np.pad(1 - h, (0, n), mode='constant')
            corr = ifft(fft(h_padded) * np.conj(fft(h_conj)))
            corr_real = np.real(corr[:2*n-1])
            corr_real *= dx
            exps = np.exp(beta * corr_real)
            w = exps / np.sum(exps)
            
            # Compute all gradient components
            grad = np.zeros(n)
            for l in range(1, n):
                for i in range(n):
                    if i + l < n:
                        grad[i] += w[l-1] * (1 - h[i+l] - h[i-l] if i-l >= 0 else 1 - h[i+l])
            return grad

        # Gradient descent loop
        beta_list = [1.0, 2.0, 5.0, 10.0, 20.0]
        time_limit = 0.2 * remaining
        t_start = time.time()
        iteration = 0
        
        # Water filling function
        def water_fill(h, M=M, dx=dx):
            h_clip = np.clip(h, 0, 1)
            def val(land):
                return np.sum(np.clip(h_clip + land, 0, 1)) - M
            low, high = -1.0, 1.0
            for _ in range(30):
                midpoint = (low + high) / 2
                if val(midpoint) > 0:
                    high = midpoint
                else:
                    low = midpoint
            return np.clip(h_clip + low, 0, 1)
        
        for beta in beta_list:
            if time.time() - t_start > time_limit:
                break
            for i in range(15):
                grad = grad_smooth(h_smooth, beta)
                step_size = 0.01
                h_smooth -= step_size * grad
                h_smooth = water_fill(h_smooth)
                iteration += 1
            current_c5 = evaluate_h(h_smooth)

    # Final check and renormalization
    h_final = h_smooth if 'h_smooth' in locals() else h_binary.astype(float)
    h_final = np.clip(h_final, 0, 1)
    current_mass = np.sum(h_final) * dx
    if abs(current_mass - 1.0) > 1e-9:
        h_final = h_final / current_mass
    final_c5 = evaluate_h(h_final)
    
    return (h_final.tolist(), final_c5, n)
