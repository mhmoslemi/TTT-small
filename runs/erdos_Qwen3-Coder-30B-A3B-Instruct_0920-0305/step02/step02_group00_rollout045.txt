import numpy as np
from cvxpy import Variable, Minimize, Problem, sum, norm, sqrt
import cvxpy as cp
from scipy.optimize import minimize
import time

# Constants
target_bound = 0.3807

def evaluate_h(h_values):
    """Evaluate the C_5 bound for given h values."""
    h = np.array(h_values)
    # Compute correlation of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value (over all shifts k)
    max_correlation = np.max(correlation)
    # Scale by dx
    c5 = max_correlation * 2.0 / len(h)
    return c5

def project_to_feasible(h_vec, n_points):
    """Project h onto [0,1] and sum constraint."""
    h_vec = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_vec)
    if abs(current_sum) > 1e-10:
        h_vec *= (n_points / 2.0) / current_sum
    return h_vec

def build_linear_constraints(h_current):
    """Compute linear coefficients for each shift."""
    n = len(h_current)
    a = np.zeros((n, n))
    b = np.zeros(n)
    for s in range(n):
        b[s] = -np.sum(h_current * np.concatenate((h_current[s:], np.zeros(s))))
        # Compute coefficients for the linearized term
        for i in range(n):
            # Handle boundary conditions - the part where h_{i+s} is out of bounds is treated as 0
            a[s,i] = 1.0 - (h_current[i+s] if i+s < n else 0.0) - (h_current[i-s] if i-s >= 0 else 0.0)
    return a, b

def solve_subproblem(a_batch, b_batch, n, t_init=1.0):
    """Solve the linear program for given coefficients."""
    # Variables
    h = Variable(n)
    t = Variable()
    # Constraints
    constraints = [0 <= h, h <= 1, sum(h) == n/2.0]
    for i in range(len(a_batch)):
        # a_batch[i] @ h + b_batch[i] <= t
        constraints.append(a_batch[i] @ h + b_batch[i] <= t)
    # Objective
    objective = Minimize(t)
    # Solve the problem
    prob = Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, verbose=False)
        h_solution = h.value
        t_solution = t.value
    except:
        # Fallback: return original
        h_solution = None
        t_solution = None
    return h_solution, t_solution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None

    # List of possible n values to try
    n_values = [48, 56, 64]  # Note: n must be even.
    max_iter = 200
    tol = 1e-6
    max_restarts = 15

    for n in n_values:
        dx = 2.0 / n
        if n > 70:
            continue  # Avoid too large problems
        time_per_n = budget_s / len(n_values)
        # Assign time
        time_for_current_n = start_time + time_per_n
        restart_count = 0
        early_exit = False

        while restart_count < max_restarts and time.time() < time_for_current_n:
            if early_exit:
                break

            # Generate initial feasible solution
            # start type 1: binary
            h_initial = np.zeros(n)
            h_initial[:n//2] = 1
            np.random.shuffle(h_initial)
            h_initial = project_to_feasible(h_initial, n)
            # start type 2: uniform
            h_initial2 = np.random.uniform(0, 1, n)
            h_initial2 = project_to_feasible(h_initial2, n)
            # start type 3: interpolated from initial_h_values
            if len(initial_h_values) == n:  # same size
                h_initial3 = np.array(initial_h_values)
            else:
                # interpolate initial_h_values to n points
                h_initial3 = np.interp(np.linspace(0, 1, n), 
                                       np.linspace(0, 1, len(initial_h_values)), 
                                       initial_h_values)
            h_initial3 = project_to_feasible(h_initial3, n)
            # pick the one with lower initial value or use a random pick
            h_current = h_initial
            initial_val = evaluate_h(h_current)
            if evaluate_h(h_initial) < initial_val:
                h_current = h_initial
            if evaluate_h(h_initial2) < initial_val:
                h_current = h_initial2
            if evaluate_h(h_initial3) < initial_val:
                h_current = h_initial3

            for it in range(max_iter):
                if time.time() > time_for_current_n:
                    break
                o_s = np.correlate(h_current, h_current, mode='full')
                o_s = o_s[len(o_s)//2:len(o_s)//2+n] * dx
                # changed to use convolution directly now: for this maximum
                h_sample = np.array(h_current)
                # Recompute coefficients
                A_batch = []
                B_batch = []
                for s in range(n):
                    b_s = -np.sum(h_sample * np.concatenate((h_sample[s:], np.zeros(s))))
                    a_s = np.zeros(n)
                    for i in range(n):
                        if i+s < n:
                            a_s[i] = 1.0 - h_sample[i+s]
                        else:
                            a_s[i] = 1.0
                        if i-s >= 0:
                            a_s[i] -= h_sample[i-s]
                    A_batch.append(a_s)
                    B_batch.append(b_s)

                # Solve linear program
                h_new, t_new = solve_subproblem(A_batch, B_batch, n)
                if h_new is None:
                    break
                h_new = np.array(h_new).flatten()
                h_new = project_to_feasible(h_new, n)
                # evaluate true objective
                c5_new = evaluate_h(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
                    if best_c5 < target_bound:
                        early_exit = True
                        break
                # Check convergence
                if abs(c5_new - initial_val) < tol:
                    break
                initial_val = c5_new
                h_current = h_new.copy()
            restart_count += 1
            if early_exit:
                break

    # Final polishing: run short SLSQP local optimization
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n)]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) * 2.0 / best_n - 1.0}
        def objective(h):
            return evaluate_h(h)
        try:
            result = minimize(objective, best_h, method='SLSQP', 
                              bounds=bounds, constraints=constraints,
                              options={'ftol': 1e-12, 'gtol': 1e-10})
            if result.success:
                best_h = result.x
                best_c5 = evaluate_h(best_h)
        except:
            pass

        best_h = np.array(initial_h_values)
        best_c5 = evaluate_h(best_h)
        best_n = len(best_h)
    return list(best_h), best_c5, best_n
