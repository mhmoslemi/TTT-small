def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution of h with (1-h) for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    # Max over all lags
    max_val = np.max(conv) * dx
    return max_val

def project_onto_feasible(h, n_points):
    """Project h onto the feasible set: 0 <= h_i <= 1 and sum(h) * dx = 1"""
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Water-filling approach
    current_sum = np.sum(h)
    diff = target_sum - current_sum

    if abs(diff) < 1e-10:
        return h
    
    # Sort indices to identify which variables to modify
    if diff > 0:
        # Distribute excess equally over values < 1
        indices = np.where(h < 1)[0]
        num_indices = len(indices)
        if num_indices > 0:
            delta = diff / num_indices
            for i in indices:
                h[i] = min(1.0, h[i] + delta)
    elif diff < 0:
        # Subtract deficit equally from values > 0
        indices = np.where(h > 0)[0]
        num_indices = len(indices)
        if num_indices > 0:
            delta = -diff / num_indices
            for i in indices:
                h[i] = max(0.0, h[i] - delta)

    return h

def compute_subgradient(h, m, dx):
    """Compute subgradient of A_m for given h and lag m"""
    n = len(h)
    g = np.zeros_like(h)
    
    if m >= 0:
        for i in range(n - m):
            j = i + m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    else:
        m = -m
        for i in range(m, n):
            j = i - m
            g[i] += 1 - h[j]
            g[j] -= h[i]
    
    g *= dx
    return g

def compute_gradient_for_max(h, dx):
    """Compute average gradient for all lags that achieve max A_m"""
    n = len(h)
    A = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(A)
    max_indices = np.where(np.abs(A - max_val) < 1e-10)[0]
    
    # Adjust indices to real lags
    m_vals = max_indices - (n - 1)
    
    g_total = np.zeros_like(h)
    num_max = len(m_vals)
    
    for m in m_vals:
        g = compute_subgradient(h, m, dx)
        g_total += g
    
    return g_total / num_max if num_max > 0 else np.zeros_like(h)

def symmetry_enforce(h):
    """Apply symmetry condition h[i] = h[n-1-i]"""
    n = len(h)
    h_sym = h.copy()
    for i in range(n // 2 + 1):
        if i < n - 1 - i:
            avg = (h[i] + h[n - 1 - i]) / 2.0
            h_sym[i] = avg
            h_sym[n - 1 - i] = avg
        else:
            # Middle point
            h_sym[i] = h[i]
    return h_sym
