def evaluate_h(h):
    """Evaluate the C5 constant using FFT for speed"""
    n = len(h)
    dx = 2.0 / n
    # Use FFT to compute all correlations efficiently
    h_padded = np.pad(h, (0, n-1), mode='constant')
    inv_h_padded = np.pad(1-h, (0, n-1), mode='constant')
    conv = fftconvolve(h_padded, inv_h_padded, mode='valid')
    max_val = np.max(conv) * dx
    return max_val
