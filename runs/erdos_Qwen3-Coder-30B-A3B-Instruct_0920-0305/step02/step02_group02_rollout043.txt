import numpy as np

from scipy.fft import fft, ifft
import time
import random

# Constants
n_candidates = [48, 56, 64, 80]
dx_candidates = [2.0 / n for n in n_candidates]
target_sum = 1.0
w_values = [n / 2 for n in n_candidates]

def project_onto_capped_simplex(x, target_sum):
    """Project x onto { y : 0 <= y_i <= 1, sum(y) = target_sum }"""
    x = np.array(x)
    n = len(x)
    # Sort in descending order with indices to track original positions
    sorted_idx = np.argsort(-x)
    sorted_x = x[sorted_idx]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    
    # Find threshold where sum of clipped values equals target
    # We want smallest k such that sum of max(sorted_x[i]-t,0) = target_sum
    threshold_idx = -1
    for i in range(n):
        # If we clip everything below the i-th element
        clipped = np.maximum(sorted_x - sorted_x[i], 0)
        if np.sum(clipped) >= target_sum:
            threshold_idx = i
            break
    
    if threshold_idx == -1:
        # All values are larger than target_sum / n, so make them equal
        threshold = target_sum / n
        x_proj = np.full_like(x, threshold)
    else:
        # threshold = sorted_x[threshold_idx]
        # Replace everything above with threshold
        x_proj = np.copy(x)
        # But this needs careful unpacking
        x_sorted = np.copy(sorted_x)
        # The relative part
        # q = (target_sum - cumsum[i-1]) / (n - i) if i > 0 else target_sum / n
        # Do binary search numerically
        if threshold_idx == 0:
            q = target_sum / n
            x_proj = np.full_like(x, q)
        else:
            # Find where cumulative sum drops below target_sum
            tmp = sorted_x[:threshold_idx]
            right_part = sorted_x[threshold_idx:]
            # Rescale right part: require sum right_part = target_sum - sum(tmp)
            s_tmp = np.sum(tmp)
            s_rest = target_sum - s_tmp
            if len(right_part) > 0:
                # min_right + (s_rest - min_right*len(right_part)) = sum(right_part)
                # i.e. min_right * (len(right_part)+1) = s_rest
                # But that isn't怎么。
                # Actually: if we want all right_part elements to be
                # equal: say each is q. Then len(right_part)*q = s_rest so q = s_rest / len(right_part).
                q = s_rest / len(right_part)
                q = max(q, 0)  # Clip to 0
                # Now the rest is what's needed to satisfy integrate to w
                x_sorted[threshold_idx:] = np.full_like(right_part, q)
                # Clamp to [0,1]
                x_sorted[threshold_idx:] = np.clip(x_sorted[threshold_idx:], 0, 1)
            # Clamp the left part
            x_sorted[:threshold_idx] = np.clip(x_sorted[:threshold_idx], 0, 1)
            # Now unsort
            x_proj = np.empty_like(x)
            x_proj[sorted_idx] = x_sorted
        
        # Adjust by small amount to exactly hit sum
        current_sum = np.sum(x_proj)
        if abs(current_sum - target_sum) > 1e-10:
            delta = target_sum - current_sum
            # Add to smallest non-bound entry
            idx = np.argmin(x_proj)
            if delta > 0:
                # stay within [0,1]
                x_proj[idx] = min(1, x_proj[idx] + delta)
            else:
                x_proj[idx] = max(0, x_proj[idx] + delta)
                
    return x_proj

def get_correlations(h, dx):
    """Calculates all correlation values and returns max."""
    h1 = np.array(h)
    h2 = 1 - h1
    n = len(h1)
    # Compute cross-correlation using FFT
    # For circular correlation, use ifft(fft(h)*fft(h2)) and shift
    corr = ifft(fft(h1) * np.conj(fft(h2)))
    corr = np.real(np.roll(corr, n//2))  # Shift halved for valid correlation
    # The full conv is n terms, and we want from k=0 to n-1
    # so we take first n terms
    return corr[:n] * dx

def evaluate_function(h, dx):
    """Evaluate the function value to minimize (C5)."""
    corr = get_correlations(h, dx)
    return -np.min(corr)

def evaluate_gradient(h, dx):
    """Evaluate the gradient of the surrogate loss."""
    n = len(h)
    # Compute C0 (r0)
    r0 = np.sum(h) - np.sum(h * h)  # since ∫ h(x)(1-h(x+k)) dx when k=0
    # Use FFT-based computation of correlations for all shifts k
    h1 = np.array(h)
    h2 = 1 - h1
    # FFT way
    cf = fft(h1) * np.conj(fft(h2))
    corr_full = ifft(cf)
    corr_full = np.real(np.roll(corr_full, n//2))
    corr = corr_full[:n] * dx
    # Now compute gradient of negative min
    min_idx = np.argmin(corr)
    # Gradient of r_k(h) = w - sum h_i * h_{i+k}
    # to compute -∂/∂h_i r_k(h) = h_{i+k} + h_{i-k} - 1
    # but we changes signs later to make it a gradient of loss
    # we use direct computation of gradient vector:
    grad = np.zeros_like(h)
    for i in range(n):
        # Partial derivative of r_k with respect to h_i
        # Take the source of influence on the k-th term:
        # γ_k = h_i * h_{i+k}  for all k, and we compute:
        # dγ_k/dh_i = (h_{i+k} if i+k< n) + (h_{i-k} if i-k≥0)
        for k in range(n):
            if i + k < n:
                grad[i] -= h[i+k]
            if i - k >= 0:
                grad[i] -= h[i-k]
        grad[i] += n  # since we subtracted thus from ∫ h(x)dx = 1, and add back from h_i terms in sum
    # Since: x = argmax_k r_k = w - min_k S_k and loss = -min_k S_k = min_k r_k
    # gradient of min_k r_k = ∇ r_{k_star} ... done correctly differently:
    # gradient of surrogate = sum_k w_k * ∇ r_k where w_k = exp(α r_k)/sum_j exp(α r_j)
    # But here not using smoothing; loss is not itself softly differentiable, but computing
    # actual gradients into which we feed duals... this is tricky for not-smooth loss.
    # Let's just compute the sign-based gradient for the analog.
    # So, we're using the fact that optimal t satisfies:
    # S_k(h*) = min_k S_k(h*) for some k*, i.e., r_k* = max_k r_k
    # The derivative ∇ r_k* = ∇ r_{k*} which is easy:
    # r_k = ∑ h_i h_{i+k}, ∂r_k/∂h_i = h_{i+k}
    grad = np.zeros_like(h)
    # Determine where the minimum happens
    min_k = min_idx
    # Gradient of r_min_k = h_{i+min_k}
    # ∇ r_k = [h_{i+k}, h_{i+k+1} ...]
    for i in range(n):
        if (i + min_k) < n:
            grad[i] += 1  # gradient of correlation term h_i * h_{i+k}
        else:
            pass # ignore boundary
    return -grad  # since we minimize future loss

def initialize_random(n, w):
    """Initialize a random feasible starting point."""
    x = np.random.uniform(0, 1, n)
    x = project_onto_capped_simplex(x, w)
    return x

def initialize_binary(n, w):
    """Initialize a binary-like pattern."""
    h = np.zeros(n)
    mid = n // 2
    # Binary block
    half = int(w)
    h[:half] = 1.0
    # Then randomize the rest sans violating w
    h = project_onto_capped_simplex(h, w)
    return h

def initialize_threeblock(n, w):
    """Initialize a three-block pattern."""
    h = np.zeros(n)
    a, b = np.random.uniform(0, 1, 2)
    a, b = sorted([a, b])
    a = int(a * n)
    b = int(b * n)
    if a == 0:
        a = 1
    if b == n:
        b = n - 1
    if a >= b:
        a = b - 1
    # Fill the middle with 0.5s
    a = max(0, a)
    b = min(n, b)
    mid_width = b - a
    d0 = w - mid_width * 0.5
    # Distribute 0 and 1
    if d0 > 0:
        # Make some entries = 1 and some 0.
        # Try to make 1s at both ends,
        left = (n - b) + a
        right = left
        if right + left < d0:
            # All are possible
            pass
        else:
            # Have to make it small enough
            right = int(d0/2)
            left = int(d0/2)
        # Fill first part with 1s
        for i in range(a):
            h[i] = 1
        # Currently filled gap, fill with 0.5
        for i in range(a, b):
            h[i] = 0.5
        # (Fill remaining with 0s)
        # Only need to fill positions [b,n) with 0s or 1s or 0.5s.
        # But we already have fixed above before, and we've specified
        # entire h already.
        # Make sure the integral is preserved.
        h = project_onto_capped_simplex(h, w)
    else:
        # Even less than w
        # Distribute 0s/1s at a, then b in the rest
        # Use optimal placement of zero, one, 0.5s to match w
        # Just fill middle with 0.5
        for i in range(a, b):
            h[i] = 0.5
        h = project_onto_capped_simplex(h, w)
    return h

def initialize_from_prev(h):
    """Analyze previous solution and create slightly perturbed version."""
    h = np.array(h)
    n = len(h)
    h += np.random.randn(n) * 0.01  # Very small noise
    h = np.clip(h, 0, 1)
    h = project_onto_capped_simplex(h, 1.0)
    return h

def initialize_h(seed, n, w):
    """Choose an initialization strategy at random."""
    random.seed(seed)
    strategies = [initialize_random, initialize_binary, initialize_threeblock]
    str_i = random.randrange(len(strategies))
    return strategies[str_i](n, w)

def line_search(f, grad_f, x, d, step_size, f_val):
    """Backtracking line search for Armijo condition."""
    # Back tracking line search: image it's a function to minimize eta
    # so define f(eta) = f(x - eta*d) = F(eta)
    # require F(eta) <= F(0) + c1 * eta * grad_f(x).dot(d)
    c1 = 1e-4
    max_iter = 30
    gamma = 0.5
    for _ in range(max_iter):
        x_new = x - step_size * d
        x_new = project_onto_capped_simplex(x_new, w)
        f_new = f(x_new)
        if f_new <= f_val - c1 * step_size * np.dot(grad_f, d):
            return step_size, x_new
        step_size *= gamma
    return step_size, x

def accelerated_projected_gradient_optimization(h0, dx, w, max_iters=5000):
    """Optimize h using accelerated projected gradient method with continuation."""
    h = h0.copy()
    y = h.copy()
    t = 1.0
    # Parameters
    alpha = 1
    max_alpha = 512
    alpha_factor = 2
    # For storage if stored version fails to find anything improve
    best_h = h.copy()
    best_c5 = evaluate_function(h, dx)

    for iter_no in range(max_iters):
        # Get updated values of correlation
        corr = get_correlations(h, dx)
        r_vals = corr / dx  # r_k = ∫ h(x) h(x+k) dx etc
        if len(r_vals) == 0:
            continue
        max_r = np.max(r_vals)
        min_r = np.min(r_vals)
        # smooth version of min r_k = log_sum_exp (but only for logical
        # behavior as it's not the discontinuous objective)
        r_best_idx = np.argmin(r_vals)
        r_best = r_vals[r_best_idx]
        # Use more precise gradient construction from actual derivatives
        # let s = sum h_i h_{i+k}, and r_k = w - s_k, so grad r_k = - grad s_k
        # grad s_k = [h_{i+k}] for index i
        grad_s = np.zeros(len(h))
        for k in range(len(h)):
            for i in range(len(h)):
                if i + k < len(h):
                    grad_s[i] -= h[i + k]
                # assume other boundary term contributions will occur as well
        # use direct gradient:
        grad_r = np.full_like(h, 1) - grad_s  # dual w.r.t b/c of neg sign
        grad_r = np.concatenate([grad_r, np.zeros(len(h))])
        
        # For numerical stability, shift by max
        shifted = alpha * r_vals  # 1D array
        max_shifted = np.max(shifted)
        shifted = shifted - max_shifted  # avoid overflow
        exps = np.exp(shifted)
        softmax_weights = exps / np.sum(exps)
        # Compute weighted average of all r_k's gradient
        # Actually better: descend directly:
        # Let maneuver = ∇r_k for the _k_min term
        maneuver = np.zeros_like(h)
        k_min = np.argmin(r_vals)
        for i in range(len(h)):
            if i + k_min < len(h):
                maneuver[i] += 1
            if i - k_min >= 0:
                maneuver[i] += 1
        # Gradient of smooth surrogate
        # Use the gradient of the contributions from all r_k
        # Using indicator based lookup for constructing ∇r_k at point k
        # end simplifying to most straightforward strategy:
        grad = np.zeros_like(h)
        # We're doing: min r_k = max_k (w - s_k)
        # So min_eqv = min_k s_k = min_k (w - r_k)
        # Gradient of surrogate loss = α sum w_k ∇ r_k
        # The gradient of r_k = 1 - ∑_i h_i h_{i+k}  → ∇ r_k = -h_{i+k}  for index i
        # so ∇ max_k r_k = ∇ r_{k_star} = -[h_{i+k_star}] for each i
        # but so that would correct match; instead, think of it by index of loss function:
        # Total: loss = - min_k (w - sum h_i h_{i+k}) = - w + max_k (sum h_i h_{i+k})
        # equivalently Minimize loss = - max_k S_k, so # loss = - (sum h_i h_{i+k_min})
        # The obbtd form is clearer to keep it simple:
        # minimize min_k r_k = w - max_k s_k, let g = grad [-s_k_min (x)]
        # Each ∂r_k/∂h_l = 1 − (i+l term above if l+k=pos)+ (i−k pos if l−k valid) ...
        # More basic perspective:
        # r_k = ∑ h_i (1−h_{i+k}) = ∑ h_i - ∑ h_i h_{i+k}
        # ∂r_k/∂h_l  = 1 - (h_{l+k} if l+k < n) - (h_{l−k} if l−k≥0)
        # Compute: ∇ item_k = - (h_{l+k} if l+k<n) - (h_{l−k} if l−k≥0) + 1
        # for gradient direction at k_min
        finalize_grad = np.zeros_like(h)
        for l in range(len(h)):
            val = 1
            if l + k_min < len(h):
                val -= h[l + k_min]
            if l - k_min >= 0:
                val -= h[l - k_min]
            finalize_grad[l] = -val  # opposite because - is taken in maximization proc
        
        # simple outer loop descent with diminishing step size
        # we want to minimize f(h) = -min_k r_k
        # Let's be DIrekt and form full F for all k for loss function:
        # surrogate loss for minimizing max
        # If the全资子 is robust, then compute:
        # let G_k = ∇r_k
        # surrogate loss = (dx/α) * log(Σ_k exp(α * r_k)) 
        # gradient = dx * Σ_k w_k * ∇r_k where w_k = exp(α r_k) / sum_j exp(α r_j)
        # step via Armijo condition:
        f_val = evaluate_function(h, dx)
        theta = 0.1  # step size guess
        
        # Gradient value from above directly
        sub_g = finalize_grad
        
        # perform step
        h_new = h - theta * sub_g
        h_new = project_onto_capped_simplex(h_new, w)
        
        # maybe also write (fast) projection at this point
        # simple line search:
        # assign smooth function
        # Line search on f:
        # f(h_new) < f(h_old) - c1*theta*grad^T*(h_1-h_0)
        f_h_new = evaluate_function(h_new, dx)
        if f_h_new < f_val - 1e-8:
            h = h_new
        else:
            h = h_new
        
        # Adjust for no progress:
        current_val = evaluate_function(h, dx)
        if current_val < best_c5:
            best_h = h.copy()
            best_c5 = current_val
        
        # optional continuation step too
        if alpha < max_alpha:
            alpha *= alpha_factor
            
        t_prev = t
        t = (1 + np.sqrt(1 + 4 * t_prev**2)) / 2
        y_new = h + ((t_prev - 1) / t) * (h - h_prev) if iter_no != 0 else h
        y = y_new
        
        h_prev = h
        
    # The final version of the best set
    return best_h, best_c5

def run(seed=42, budget_s=600):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = None
    remaining_time = budget_s

    for n_idx, n in enumerate(n_candidates):
        if time.time() - start_time > budget_s * 0.9:
            break
        dx = dx_candidates[n_idx]
        w = w_values[n_idx]
        # allocate approximately equal time per n
        time_per_n = remaining_time / len(n_candidates)
        n_start = time.time()
        total_restarts = 0
        num_restarts = min(max_restarts, int(time_per_n / time_per_restart))
        while time.time() - n_start < time_per_n and total_restarts < num_restarts:
            seed += 1
            h_init = initialize_h(seed, n, w)
            h_final, c5_final = accelerated_projected_gradient_optimization(h_init, dx, w)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                best_n = n
            total_restarts += 1

    if best_h is None:
        # fallback to some initialization
        h_default = np.array(initial_h_values)
        best_c5 = evaluate_function(h_default, 2.0 / 47)
        best_h = h_default.copy()
        best_n = 47

    # should now pass through final evaluation with original code
    final_h = project_onto_capped_simplex(best_h, 1.0)
    final_c5 = evaluate_function(final_h, 2.0 / best_n)

    # make sure we always return strings if any
    return list(final_h), -final_c5, best_n
