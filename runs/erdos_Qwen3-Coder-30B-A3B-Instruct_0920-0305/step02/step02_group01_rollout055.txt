import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """
    Projects h onto the capped simplex: 0 <= h_i <= 1, sum(h) = target_sum
    Uses bisection for the Lagrange multiplier.
    """
    # Ensure values are in [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-12:
        return h
    
    # Binary search for lambda
    low, high = -10.0, 10.0
    for _ in range(40):
        mid = (low + high) / 2.0
        new_h = np.clip(h - mid, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum > target_sum:
            low = mid
        else:
            high = mid
    lambda_opt = (low + high) / 2.0
    result = np.clip(h - lambda_opt, 0, 1)
    return result

def build_five_step_h(a, b, u, n_points):
    """Construct a 5-step symmetric function based on parameters a, b, u"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    x = np.arange(0.5, n_points, 1.0) * dx
    # Build the 5-step pattern: 0 - a - u - 1 - u - a - 0
    # Segments: [0,a), [a,b), [b, 2-b), [2-b, 2-a), [2-a, 2]
    # Values:     0,   u,   1,    u,   0
    # We use a closed lower bound; treat the last segment specially
    indices1 = (x >= a) & (x < b)
    indices2 = (x >= b) & (x < (2 - b))
    indices3 = (x >= (2 - b)) & (x < (2 - a))
    h[indices1] = u
    h[indices2] = 1.0
    h[indices3] = u
    # Final integral constraint scaling:
    current_sum = np.sum(h) * dx
    h = h * (n_points / 2) / current_sum
    return h

def smooth_max(corr, alpha):
    """Return the soft-max of corr with temperature alpha"""
    c = corr.copy()
    exp_c = np.exp(alpha * c)
    w = exp_c / np.sum(exp_c)
    return w

def compute_smooth_grad(h, alpha):
    """Compute gradient of the smooth approximation of max"""
    n = len(h)
    corr = correlate(h, 1 - h, mode="full")
    w = smooth_max(corr, alpha)
    # Incorporate the zeros from 1-h
    w_rev = w[::-1]
    # g_plus = convolve(w_rev, h, mode='full')[n-1:n-1+n]
    # g_minus = convolve(w, h, mode='full')[n-1:n-1+n]
    g_plus = np.convolve(w_rev, h, mode='full')[n-1:n-1+n]
    g_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]
    return 1.0 - (g_plus + g_minus)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 200  # Slightly higher resolution for better detail
    dx = 2.0 / n_points
    n_grid = 51  # Coarse grid resolution
    
    # Step 1: Grid Search over a, b parameters
    a_grid = np.linspace(0.0, 0.5, n_grid)
    b_grid = np.linspace(0.5, 1.0, n_grid)
    grid_start = time.time()
    for a in a_grid:
        if time.time() - start_time > 0.4 * budget_s:
            break
        for b in b_grid:
            if a >= b:
                continue
            u = (2*b - 1) / (2*(b - a))
            if 0 <= u <= 1:
                h_temp = build_five_step_h(a, b, u, n_points)
                if np.sum(h_temp) * dx < 0.999 or np.sum(h_temp) * dx > 1.001:
                    h_temp = project_to_capped_simplex(h_temp, n_points/2)
                c5 = evaluate_c5(h_temp)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_temp.copy()
    
    # Step 2: Smooth gradient descent
    if best_h is not None and (time.time() - start_time) < 0.9 * budget_s:
        h = best_h.copy()
        alpha = 0.5
        eta = 0.5
        iter_count = 0
        patience = 0
        last_best = best_c5
        prev_h = h.copy()
        max_iter = 5000
        while iter_count < max_iter and (time.time() - start_time) < 0.9 * budget_s:
            grad = compute_smooth_grad(h, alpha)
            h_new = h - eta * grad
            h_new = project_to_capped_simplex(h_new, n_points/2)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
            if patience > 500:
                break
            h = h_new
            iter_count += 1
            if iter_count % 200 == 0:
                alpha = min(20.0, alpha * 1.5)
                eta = max(1e-6, eta * 0.9)
    
    # Step 3: Small differential evolution fallback
    if (time.time() - start_time) < 0.95 * budget_s:
        pop_size = 8
        max_gen = 100
        pop = []
        for _ in range(pop_size):
            h_rnd = np.random.uniform(0, 1, n_points)
            h_rnd = project_to_capped_simplex(h_rnd, n_points/2)
            pop.append(h_rnd)
        for gen in range(max_gen):
            if (time.time() - start_time) > 0.95 * budget_s:
                break
            new_pop = []
            for i in range(pop_size):
                idxs = np.random.choice(pop_size, 3, replace=False)
                a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
                F = 0.6
                CR = 0.9
                v = a + F * (b - c)
                parent = pop[i]
                trial = np.where(np.random.rand(n_points) < CR, v, parent)
                trial = project_to_capped_simplex(trial, n_points/2)
                trial_c5 = evaluate_c5(trial)
                parent_c5 = evaluate_c5(parent)
                if trial_c5 < parent_c5:
                    new_pop.append(trial)
                else:
                    new_pop.append(parent)
            pop = new_pop
            # Update best
            for h_ind in pop:
                c5 = evaluate_c5(h_ind)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_ind.copy()
    
    # Final projection and evaluation
    if best_h is not None:
        best_h = project_to_capped_simplex(best_h, n_points/2)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
