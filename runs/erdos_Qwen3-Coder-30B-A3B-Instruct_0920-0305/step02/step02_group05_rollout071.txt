import numpy as np
import time

def project_to_bounds_and_sum(v, target_sum):
    """Project vector onto box [0,1] and sum constraint"""
    lo = 0.0
    hi = 1.0
    # First clip to bounds
    w = np.clip(v, lo, hi)
    # Compute sum
    total = w.sum()
    if abs(total - target_sum) < 1e-12:
        return w
    # We'll iteratively adjust
    while True:
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (w < hi)
            n_free = free.sum()
            if n_free == 0:
                # cannot increase more, break
                break
            delta = diff / n_free
            w[free] += delta
            # Clip again
            w = np.clip(w, lo, hi)
        else:
            free = (w > lo)
            n_free = free.sum()
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, lo, hi)
        total = w.sum()
    return w

def compute_max_overlap(h):
    """Compute max overlap and corresponding shift"""
    n_points = len(h)
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    argmax = np.argmax(conv)
    shift = argmax - (n_points - 1)
    return max_val, shift

def gradient_for_shift(h, m):
    """Compute gradient for a specific shift"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros_like(h)
    if m >= 0:
        # overlapping indices i from 0 to n-1-m
        i = np.arange(0, n - m)
        j = i + m
        g[i] += 1 - h[j]
        g[j] -= h[i]
    else:
        # overlapping indices i from -m to n-1
        m2 = -m
        i = np.arange(m2, n)
        j = i - m2
        g[i] += 1 - h[j]
        g[j] -= h[i]
    return g * dx
