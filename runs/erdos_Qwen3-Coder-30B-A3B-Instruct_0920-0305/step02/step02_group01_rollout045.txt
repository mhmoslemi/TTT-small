import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """
    Evaluate the C5 constant for a given step function h
    """
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_capped_simplex(h, target_sum):
    """
    Project h onto the capped simplex: 0 <= h[i] <= 1, sum(h) = target_sum
    """
    # Start with the current values
    h_new = h.copy()
    # Binary search for lambda
    low, high = -100.0, 100.0
    for _ in range(40):
        lam = (low + high) / 2.0
        # Compute x_i(lambda) = clip(h_new[i] - lam, 0, 1)
        x = np.clip(h_new - lam, 0, 1)
        sum_x = np.sum(x)
        if sum_x > target_sum:
            low = lam
        else:
            high = lam
    # Recompute final projection
    lam = (low + high) / 2.0
    x = np.clip(h_new - lam, 0, 1)
    return x

def smooth_max(c, alpha=0.5):
    """
    Soft max approximation to the max function
    Returns softmax weights w such that sum(w_i * c_i) approximates max(c)
    """
    exp_c = np.exp(alpha * c)
    return exp_c / np.sum(exp_c)

def compute_gradient(h, w):
    """
    Compute gradient of the smooth objective function with weights w
    """
    n = len(h)
    # Compute h_{i+s} and h_{i-s} sums via convolution
    # w is flipped because correlate in reverse order
    w_flip = w[::-1]
    g_plus = correlate(w_flip, h, mode='full')[n-1:n-1+n]
    g_minus = correlate(w, h, mode='full')[n-1:n-1+n]
    return 1.0 - (g_plus + g_minus)
