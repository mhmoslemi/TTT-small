import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import cvxpy as cp

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use larger number of samples for better resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Start with a promising construction - golden ratio based pattern
    # This creates a low-discrepancy sequence
    alpha = (np.sqrt(5) - 1) / 2  # Golden ratio
    h0 = np.array([1 if (int(i * alpha) % 2 == 0) else 0 for i in range(n_points)])
    sum_h = np.sum(h0)
    # Make sure it sums to exactly n_points/2
    target_sum = n_points // 2
    if sum_h != target_sum:
        # Adjust the values to get exactly the right sum
        if sum_h < target_sum:
            # Need to add more 1s
            indices_0 = np.where(h0 == 0)[0]
            for i in range(target_sum - sum_h):
                if i < len(indices_0):
                    h0[indices_0[i]] = 1
        else:
            # Need to remove some 1s
            indices_1 = np.where(h0 == 1)[0]
            for i in range(sum_h - target_sum):
                if i < len(indices_1):
                    h0[indices_1[i]] = 0

    # Initial evaluation
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Local optimization around the best solution: apply simulated annealing
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Parameters for simulated annealing
    max_iters = 100000
    temp_start = 0.005
    temp_decay = 0.9999
    temp_min = 1e-8
    
    iteration = 0
    while iteration < max_iters and (time.time() - start_time) < budget_s * 0.9:
        # Generate a candidate by swapping two elements
        i = np.random.randint(0, n_points)
        # Find a different element to swap with
        j = np.random.randint(0, n_points)
        while j == i:
            j = np.random.randint(0, n_points)
        
        # Swap values
        candidate_h = current_h.copy()
        candidate_h[i], candidate_h[j] = candidate_h[j], candidate_h[i]
        
        # Ensure we still have exactly n_points/2 ones
        if np.sum(candidate_h) == n_points // 2:
            candidate_c5 = evaluate_c5(candidate_h)
            # Accept if better, or with probability exp(-(candidate_c5 - current_c5)/temp)
            delta = candidate_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp_start):
                current_h = candidate_h
                current_c5 = candidate_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        temp_start *= temp_decay
        iteration += 1
        
        if temp_start < temp_min:
            temp_start = temp_min
    
    # Final refinement with local search using SLSQP
    try:
        # Create bounds for continuous optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            # Integral constraint: sum(h) * dx = 1
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        # Run SLSQP optimization
        res = differential_evolution(objective, bounds, seed=seed, maxiter=500, 
                                   popsize=20, mutation=(0.5, 1), recombination=0.7)
        if res.success:
            h_candidate = res.x
            scale = 1.0 / (np.sum(h_candidate) * dx)
            h_candidate = h_candidate * scale
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
    except Exception as e:
        pass
    
    # Final step: round to binary if all values are close to 0 or 1
    if (np.allclose(best_h, 0, atol=0.1) or np.allclose(best_h, 1, atol=0.1)):
        best_h = np.round(best_h)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
