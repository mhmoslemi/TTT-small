import numpy as np
from scipy.fft import rfft, irfft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
target_sum = n_points / 2

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = rfft(h_rev, n=L)
    B = rfft(b, n=L)
    corr = irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Evaluate C5 using FFT correlation"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute subgradient of correlation for shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # Forward contribution: (1 - h_{i+s})
        if s < N:
            g[:N-s] += 1 - h[s:]
        # Backward contribution: -h_{i-s}  
        if s > 0:
            g[s:] -= h[:N-s]
    else:
        # s < 0, reverse shift
        s_abs = -s
        if s_abs < N:
            g[s_abs:] += 1 - h[:N-s_abs]
        if s_abs > 0:
            g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute smooth gradient using top-K shifts"""
    corr = corr_fft(h)
    
    if T < 1e-12:
        # Use exact subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Use only top_k weights for efficiency
    top_idx = np.argpartition(weights, -top_k)[-top_k:]
    
    g = np.zeros_like(h)
    N = len(h)
    for idx in top_idx:
        w = weights[idx]
        s = idx - (N-1)
        grad_s = subgradient_for_shift(h, s)
        g += w * grad_s
    return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart with optional symmetry"""
    # Random initial h
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    h = h0.copy()
    
    T = T0
    step = step0
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    for it in range(max_iter):
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update step size
        step = step0 / np.sqrt(it + 1)
        
        # Update h
        h = h - step * g
        
        # Apply symmetry if requested
        if sym:
            h = (h + h[::-1]) / 2
            
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Update temperature
        T *= 0.9995
        
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local search using pair-wise swaps"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        # Pick two distinct indices
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_min = max(-best_h[i], best_h[j] - 1)
        delta_max = min(1 - best_h[i], best_h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Create new vector
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate
        c5_new = evaluate_c5_fast(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Coarse optimization (n=60) 
    coarse_n = 60
    coarse_dx = 2.0 / coarse_n
    coarse_target = coarse_n / 2
    
    h_coarse = np.random.rand(coarse_n)
    h_coarse = project_onto_capped_simplex(h_coarse, coarse_target)
    h_coarse, c5_coarse = gradient_descent_restart(
        coarse_n, coarse_target, 1000, 0.5, 0.1, 5, sym=False
    )
    
    # Upsample to target resolution
    if n_points > coarse_n:
        x_coarse = np.linspace(coarse_dx/2, 2 - coarse_dx/2, coarse_n)
        x_fine = np.linspace(dx/2, 2 - dx/2, n_points)
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        h_fine = project_onto_capped_simplex(h_fine, target_sum)
    else:
        h_fine = h_coarse.copy()
    
    # Multiple restarts
    restarts = 15
    remaining_time = budget_s * 0.8
    
    for i in range(restarts):
        if time.time() - start > budget_s * 0.9:
            break
        # Use triangular seed for one restart
        if i == 0:
            x = np.linspace(dx/2, 2 - dx/2, n_points)
            h_seed = np.maximum(0, 1 - np.abs(x - 1))
            h_seed = project_onto_capped_simplex(h_seed, target_sum)
        elif i == 1:
            # Symmetric random seed
            h_seed = np.random.rand(n_points)
            h_seed = project_onto_capped_simplex(h_seed, target_sum)
            h_seed = (h_seed + h_seed[::-1]) / 2
            h_seed = project_onto_capped_simplex(h_seed, target_sum)
        else:
            # Random seed
            h_seed = np.random.rand(n_points)
            h_seed = project_onto_capped_simplex(h_seed, target_sum)
            
        h_restart, c5_restart = gradient_descent_restart(
            n_points, target_sum, 2000, 0.5, 0.1, 10, sym=False
        )
        
        if c5_restart < best_c5:
            best_c5 = c5_restart
            best_h = h_restart.copy()
    
    # Local polishing if we have a solution
    if best_h is None:
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
    
    # Local swap refinement 
    remaining_time = budget_s - (time.time() - start)
    if remaining_time > 10:
        best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time - 5)
    
    # Ensure final projection on sum
    best_h = project_onto_capped_simplex(best_h, target_sum)
    best_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
