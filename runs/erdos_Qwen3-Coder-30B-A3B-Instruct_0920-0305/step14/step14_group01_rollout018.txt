# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_box_sum(u, target_sum):
    """Project u onto [0,1] with sum = target_sum using waterfilling"""
    u = np.array(u, dtype=float)
    if np.sum(u) == target_sum:
        return u
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_smooth_max_gradient(u, epsilon=0.02):
    """Compute gradient of smooth max using FFT-based convolution"""
    h = np.concatenate([u, u[::-1]])  # reconstruct full h
    corr = compute_correlation(h)
    
    # Compute weights for smooth max
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradients using convolution
    b = 1 - h
    L = len(h) * 2 - 1
    
    # conv(w, b) - conv(w_rev, h)
    w_full = np.concatenate([w, np.zeros(L - len(w))])
    b_full = np.concatenate([b, np.zeros(L - len(b))])
    h_full = np.concatenate([h, np.zeros(L - len(h))])
    
    conv1 = np.fft.ifft(np.fft.fft(w_full, L) * np.fft.fft(b_full, L)).real
    conv2 = np.fft.ifft(np.fft.fft(w_full[::-1], L) * np.fft.fft(h_full, L)).real
    
    grad = conv1 - conv2
    grad = grad[(len(h)-1):(len(h)-1)+len(h)]
    
    # Return only the first half (independent variables)
    return grad[:len(u)]

    corr = compute_correlation(h)
    max_corr = np.max(corr)
    
    # Smooth approximation to max
    smooth_max = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    
    # Add penalties for monotonicity and sum constraint

    penalty_sum = mu * (np.sum(u) - target_sum)**2
    
    # Gradient computation
    grad = compute_smooth_max_gradient(u, epsilon)

    grad_inc = np.zeros_like(u)
    grad_inc[1:] = -2 * lambda_inc * np.diff(u)
    grad_inc[:-1] += 2 * lambda_inc * np.diff(u)
    
    grad_sum = 2 * mu * (np.sum(u) - target_sum)
    grad += grad_inc + grad_sum
    
    return smooth_max + penalty_inc + penalty_sum, grad

    # Triangular shape

    triangular = np.minimum(x, 1-x)
    seeds.append(triangular)
    
    # Cosine squared shape
    x = np.linspace(0, 2, n_points)
    cosine = (1 + np.cos(np.pi * (x - 1))) / 2

    # Random monotone shape
    rand = np.random.rand(n_points//2)
    rand_sorted = np.sort(rand)
    seeds.append(rand_sorted)
    
    return seeds
