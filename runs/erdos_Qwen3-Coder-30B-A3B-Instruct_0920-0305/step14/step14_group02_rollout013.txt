import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return np.max(corr)

def build_triangular_seed(n):
    """Build triangular seed function"""
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.maximum(0, 1 - np.abs(x - 1))
    return h

def soft_max_gradient(corr, T, top_k=10):
    """Compute gradient using soft-max approximation with top-K shifts"""
    # Get top K shifts
    top_indices = np.argpartition(corr, -top_k)[-top_k:]
    
    # Soft-max weights
    corr_top = corr[top_indices]
    w = np.exp((corr_top - corr_top.max()) / T)
    w = w / w.sum()
    
    # Gradient computation
    grad = np.zeros_like(corr)
    for i, (s, weight) in enumerate(zip(top_indices, w)):
        if s >= 0:
            # Add contribution from h_i * (1 - h_{i+s})
            for j in range(len(corr) - s):
                grad[s + j] -= weight * (1 - corr[s + j])
        else:
            # Add contribution from h_{i+s} * h_i
            for j in range(-s, len(corr)):
                grad[j - s] -= weight * corr[j - s]
    
    return grad

def project_onto_capped_simplex(x, target_sum):
    """Project onto the capped simplex with sum constraint"""
    # Box clipping
    y = np.clip(x, 0, 1)
    # Sum correction
    current_sum = np.sum(y)
    if abs(current_sum - target_sum) < 1e-10:
        return y
    
    # Binary search for lambda
    def sum_with_lambda(lam):
        return np.sum(np.clip(y + lam, 0, 1))
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm - target_sum) < 1e-10:
            break
        if fm < target_sum:
            a = m
        else:
            b = m
    
    return np.clip(y + (a + b)/2, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Create triangular seed
    h0 = build_triangular_seed(n_points)
    h0 = project_to_sum_box(h0, target_sum)
    
    # Coarse optimization with N=60
    n_coarse = 60
    h_coarse = build_triangular_seed(n_coarse)
    h_coarse = project_to_sum_box(h_coarse, n_coarse / 2)
    
    # Continue optimization on coarse grid
    T = 0.1
    T_final = 1e-5
    tau_T = 0.9995
    alpha = 0.6
    tau_alpha = 0.999
    K = 10
    
    max_iter_coarse = 2000
    for t in range(max_iter_coarse):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute correlation (FFT-based)
        h_rev = h_coarse[::-1]
        b = 1 - h_coarse
        L = 2 * len(h_coarse) - 1
        corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
        
        # Top K shifts
        top_indices = np.argpartition(corr, -K)[-K:]
        corr_top = corr[top_indices]
        
        # Soft max
        w = np.exp((corr_top - corr_top.max()) / T)
        w = w / w.sum()
        
        # Gradient
        grad = np.zeros_like(corr)
        for i, (s, weight) in enumerate(zip(top_indices, w)):
            if s >= 0:
                for j in range(len(h_coarse) - s):
                    grad[s + j] += weight * (1 - h_coarse[j])
            else:
                for j in range(-s, len(h_coarse)):
                    grad[j - s] += weight * h_coarse[j]
        
        # Step
        alpha_t = alpha * (tau_alpha ** t)
        h_coarse = h_coarse - alpha_t * grad[:len(h_coarse)]
        
        # Project
        h_coarse = project_onto_capped_simplex(h_coarse, n_coarse / 2)
        
        # Update temperature
        T = max(T_final, T * tau_T)
        
        # Evaluate
        current_c5 = evaluate_c5(h_coarse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Interpolate to fine grid
            x_coarse = np.linspace(0, 2, n_coarse, endpoint=False)
            x_fine = np.linspace(0, 2, n_points, endpoint=False)
            h_fine = np.interp(x_fine, x_coarse, h_coarse)
            best_h = project_to_sum_box(h_fine, target_sum)
    
    # Fine optimization with restarts
    num_restarts = 15
    for restart in range(num_restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Random initialization (non-symmetric)
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, target_sum)
        
        # Restart with triangular seed
        h = build_triangular_seed(n_points)
        h = project_to_sum_box(h, target_sum)
        
        # Optimization loop
        T = 0.1
        alpha = 0.6
        tau_T = 0.9995
        tau_alpha = 0.999
        K = 10
        max_iter = 3000
        momentum = 0
        
        for t in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute correlation
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
            
            # Top K shifts
            top_indices = np.argpartition(corr, -K)[-K:]
            
            # Soft max weights
            corr_top = corr[top_indices]
            w = np.exp((corr_top - corr_top.max()) / T)
            w = w / w.sum()
            
            # Gradient
            grad = np.zeros(len(h))
            for i, (s, weight) in enumerate(zip(top_indices, w)):
                if s >= 0:
                    for j in range(len(h) - s):
                        grad[j] -= weight * (1 - h[j+s])
                else:
                    for j in range(-s, len(h)):
                        grad[j] -= weight * h[j+s]
            
            # Momentum
            momentum = 0.9 * momentum + 0.1 * grad
            step_size = alpha / np.sqrt(t+1)
            h = h - step_size * momentum
            
            # Project
            h = project_onto_capped_simplex(h, target_sum)
            
            # Update parameters
            T = max(T_final, T * tau_T)
            alpha = alpha * tau_alpha
            
            # Evaluate
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Local polishing with pair-swap
    if best_h is not None:
        h = best_h.copy()
        max_iter_polish = 100000
        for iter_num in range(max_iter_polish):
            if time.time() - start_time > budget_s:
                break
                
            # Pair swap
            i, j = np.random.choice(n_points, 2, replace=False)
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            if delta_low >= delta_high:
                continue
            
            delta = np.random.uniform(delta_low, delta_high)
            h[i] += delta
            h[j] -= delta
            
            # Project to ensure constraints
            h = project_onto_capped_simplex(h, target_sum)
            
            # Evaluate
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            elif np.random.rand() < 0.01:  # Accept worse with small probability
                h[i] -= delta
                h[j] += delta
                h = project_onto_capped_simplex(h, target_sum)
    
    # Final evaluation
    if best_h is not None:
        h_final = best_h.copy()
        h_final = project_onto_capped_simplex(h_final, target_sum)
        final_c5 = evaluate_c5(h_final)
        return (h_final.tolist(), float(final_c5), n_points)
    
    # Return initial solution if nothing better found
    h = build_triangular_seed(n_points)
    h = project_to_sum_box(h, target_sum)
    c5 = evaluate_c5(h)
    return (h.tolist(), float(c5), n_points)
