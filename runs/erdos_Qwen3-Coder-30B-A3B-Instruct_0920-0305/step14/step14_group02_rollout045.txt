import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
T0 = 0.12
TAU_T = 0.998
T_MIN = 5e-6
MAX_LBFGS_ITER = 2000
MAX_PAIRWISE_SWEEPS = 100
N_COARSE = 60

def evaluate_c5_exact(h):
    """Fast evaluation of C5 using FFT correlation"""
    # Compute full correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T):
    """Evaluate smooth surrogate of C5 (log-sum-exp)"""
    # Compute all correlation values using full correlation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Compute log-sum-exp with temperature T
    corr = corr * dx
    max_corr = np.max(corr)
    if max_corr > 100:  # Prevent overflow
        corr_shifted = corr - max_corr
    else:
        corr_shifted = corr
        
    log_sum_exp = max_corr + np.log(np.sum(np.exp(corr_shifted / T)))
    return log_sum_exp

def compute_gradient(h, T):
    """Compute gradient of smooth surrogate"""
    # Compute all correlation values using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr * dx
    
    # Compute weights for log-sum-exp
    max_corr = np.max(corr)
    corr_shifted = corr - max_corr
    exp_corr = np.exp(corr_shifted / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient from each correlation
    grad = np.zeros_like(h)
    for s in range(len(corr)):
        # Compute derivative of correlation [s] w.r.t h (using finite difference)
        if s < len(h):
            # Contribution from h_i to correlation shift s
            for i in range(len(h)):
                if 0 <= i <= len(h) - 1 - s:
                    grad[i] += weights[s] * (1 - h[i + s]) if i + s < len(h) else 0
                if i + s < len(h) and i + s >= 0:
                    grad[i] -= weights[s] * h[i + s] if i + s < len(h) else 0
    
    # Add gradient contribution for negative shifts (s < 0) 
    for s in range(-len(h)+1, 0):
        for i in range(len(h)):
            if 0 <= i - s < len(h) and i >= 0:
                grad[i] += weights[len(h)-1+s] * (1 - h[i - s]) if i - s < len(h) else 0
                grad[i] -= weights[len(h)-1+s] * h[i - s] if i - s < len(h) else 0
    
    return dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm for projection
    x = np.array(x)
    x_sorted = np.sort(x)[::-1]
    cum_sum = np.cumsum(x_sorted)
    n = len(x)  
    threshold = 0
    for i in range(n):
        if (cum_sum[i] + (i+1) * (1 - x_sorted[i])) < target_sum:
            threshold = x_sorted[i]
        else:
            break
    
    # Compute threshold
    threshold = (target_sum - cum_sum[i]) / (i+1) if i < n-1 else 0
    
    # Apply projection
    projected = np.clip(x - threshold, 0, 1)
    
    # Adjust for sum constraint
    current_sum = np.sum(projected)
    diff = target_sum - current_sum
    
    # Apply small adjustment to reach target sum exactly
    if abs(diff) > 1e-10:
        indices = np.argsort(projected)
        for i in range(min(int(np.ceil(abs(diff) * n)), n)):
            if diff > 0:
                # Increase values toward highest
                idx = indices[-1-i]
                delta = min(diff, 1 - projected[idx])
                projected[idx] += delta
                diff -= delta
            else:
                # Decrease values toward lowest
                idx = indices[i]  
                delta = min(-diff, projected[idx])
                projected[idx] -= delta
                diff += delta
            if abs(diff) < 1e-10:
                break
    
    return projected

def lbfgs_with_continuation(h0, max_iter=MAX_LBFGS_ITER):
    """Use L-BFGS with temperature continuation"""
    h = h0.copy()
    T = T0
    current_loss = evaluate_c5_smooth(h, T)
    
    # Store best solution
    best_h = h.copy()
    best_loss = current_loss
    
    # Store convergence history
    loss_history = []
    
    for iter_count in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, T)
        
        # Compute step size using line search (simplified version of L-BFGS)
        step_size = 1.0
        h_new = h - step_size * grad
        
        # Apply projection to box and sum constraints
        h_proj = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check for improvement
        loss_new = evaluate_c5_smooth(h_proj, T)
        if loss_new < current_loss:
            h = h_proj
            current_loss = loss_new
            if loss_new < best_loss:
                best_loss = loss_new
                best_h = h.copy()
        
        # Update temperature
        T = max(T_MIN, T * TAU_T)
        
        # Check for convergence based on minimal T and small gradient
        if T < T_MIN:
            break
        
    return best_h, best_loss

def pairwise_exchange_optimization(h, max_sweeps=MAX_PAIRWISE_SWEEPS):
    """Perform exact pairwise exchange with line search"""
    h_new = h.copy()
    old_c5 = evaluate_c5_exact(h_new)
    
    # Keep track of improvements  
    improvements = 0
    
    for sweep in range(max_sweeps):
        improved = False
        # Shuffle indices for random order
        indices = list(range(len(h_new)))
        np.random.shuffle(indices)
        
        # Sweep through indices in random order
        for i in indices:
            # Try moving small amounts from i to each other point
            for j in range(len(h_new)):
                if i == j:
                    continue
                    
                # Determine bounds for δ
                delta_min = max(-h_new[j], h_new[i] - 1)
                delta_max = min(h_new[i], 1 - h_new[j])
                
                if delta_min >= delta_max:
                    continue
                
                # Generate sample points for finding optimal δ
                candidates = []
                
                # Include boundary points
                candidates.append(delta_min)
                candidates.append(delta_max)
                
                # Include midpoints from smooth gradient (simplified)
                if delta_max - delta_min > 1e-10:
                    n_samples = min(10, int(1.0 / (delta_max - delta_min) * 20))
                    if n_samples > 2:
                        samples = np.linspace(delta_min, delta_max, n_samples)
                        candidates.extend(samples)
                
                # Compute slopes for each shift at current point
                c0 = np.zeros(len(h_new) * 2 - 1)
                for s in range(len(h_new) * 2 - 1):
                    if s < len(h_new):
                        # shift s positive
                        if s <= len(h_new) - 1:
                            c0[s] = np.sum(h_new[:len(h_new)-s] * (1 - h_new[s:]))
                        else:
                            c0[s] = 0
                    else:
                        # shift s negative 
                        s2 = s - (len(h_new) - 1)
                        if s2 >= 0:
                            c0[s] = np.sum(h_new[s2:] * (1 - h_new[:len(h_new)-s2]))
                        else:
                            c0[s] = 0
                        
                # Try candidates 
                best_delta = None
                min_max_corr = float('inf')
                for delta in candidates:
                    if delta < delta_min or delta > delta_max:
                        continue
                    h_test = h_new.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    h_test = project_to_sum_box(h_test, TARGET_SUM)
                    
                    # Calculate max correlation for this h_test
                    corr_val = evaluate_c5_exact(h_test)
                    if corr_val < min_max_corr:
                        min_max_corr = corr_val
                        best_delta = delta
                        
                # Apply best improvement if significant
                if best_delta is not None and min_max_corr < old_c5 - 1e-8:
                    h_new[i] -= best_delta
                    h_new[j] += best_delta
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    improvements += 1
                    improved = True
                    old_c5 = min_max_corr
        
        if not improved:
            break
            
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Use coarse to fine strategy
    # Part 1: Coarse resolution optimization
    print("Starting coarse resolution optimization...")
    h_coarse = np.zeros(N_COARSE)
    # Create ramp function (triangular shape)
    x = np.linspace(0, 2, N_COARSE)
    h_coarse = np.maximum(0, 1 - np.abs(x - 1))
    h_coarse = project_to_sum_box(h_coarse, N_COARSE / 2)
    
    # Run optimization with L-BFGS
    h_coarse_result, _ = lbfgs_with_continuation(h_coarse, max_iter=1200)
    h_coarse_result = project_to_sum_box(h_coarse_result, N_COARSE / 2)
    
    # Part 2: Finer grid interpolation and refinement
    print("Interpolating to finer grid...")
    # Interpolate from coarse to fine grid
    x_coarse = np.linspace(0, 2, N_COARSE)
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse_result)
    h_fine = project_to_sum_box(h_fine, TARGET_SUM)
    
    # Part 3: Multiple restarts with different initialization strategies
    best_h = h_fine.copy()
    best_c5 = evaluate_c5_exact(best_h)
    
    restart_count = 15  # About 15 restarts  
    for restart in range(restart_count):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Different initialization strategies
        init_type = restart % 4
        if init_type == 0:
            # Random initialization
            h_init = np.random.rand(n_points)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        elif init_type == 1:
            # Triangular symmetry 
            x = np.linspace(0, 2, n_points)
            h_init = np.maximum(0, 1 - np.abs(x - 1))
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        elif init_type == 2:
            # Sine wave
            x = np.linspace(0, 2, n_points)
            h_init = 0.5 + 0.3 * np.sin(3 * np.pi * x)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        else:  # init_type == 3
            # Interpolated coarse result 
            x_coarse = np.linspace(0, 2, N_COARSE)
            x_fine = np.linspace(0, 2, n_points)
            h_init = np.interp(x_fine, x_coarse, h_coarse_result)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        
        # Apply L-BFGS optimization to this initialization
        h_opt, _ = lbfgs_with_continuation(h_init, max_iter=1500)
        h_opt = project_to_sum_box(h_opt, TARGET_SUM)
        c5_opt = evaluate_c5_exact(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            
        # Check time budget
        if time.time() - start_time > 0.8 * budget_s:
            break
    
    # Final polishing with pairwise exchange
    print("Performing final polishing...")
    best_h = pairwise_exchange_optimization(best_h, max_sweeps=100)
    best_c5 = evaluate_c5_exact(best_h)
    
    # Final projection and evaluation
    best_h = project_to_sum_box(best_h, TARGET_SUM)
    final_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
