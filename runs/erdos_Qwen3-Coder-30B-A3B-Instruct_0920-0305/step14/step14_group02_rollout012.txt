import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlations(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), n=L)
    return corr

def compute_gradient(corr, top_k_indices, T):
    """Compute gradient using top-k shifts"""
    # Compute soft-max weights for top-k shifts
    corr_top = corr[top_k_indices]
    # For numerical stability
    corr_top_stable = corr_top - np.max(corr_top)
    exp_corr = np.exp(corr_top_stable / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros_like(corr)
    for i, (idx, weight) in enumerate(zip(top_k_indices, weights)):
        if idx >= 0:
            # Effect of shift idx on correlation
            if idx < len(h):
                grad[:len(h)-idx] += weight * (1 - h[idx:])
            else:
                grad[:len(h)-(idx-len(h))] += weight * (1 - h[idx-len(h):])
        else:
            # Effect of shift idx (negative) on correlation
            idx_abs = -idx
            if idx_abs < len(h):
                grad[idx_abs:] += weight * (1 - h[:-idx_abs])
            else:
                grad[:len(h)-(idx_abs-len(h))] += weight * (1 - h[:idx_abs-len(h)])
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120  # resolution
    dx = 2.0 / N
    target_sum = N / 2
    T0 = 0.1
    alpha0 = 0.6
    K = 10
    beta = 0.9
    max_iter = 3000
    
    # Coarse optimization
    N_coarse = 60
    dx_coarse = 2.0 / N_coarse
    target_sum_coarse = N_coarse / 2
    
    # Triangular seed for coarse resolution
    x_coarse = np.linspace(0, 2, N_coarse)
    h_coarse_initial = np.maximum(0, 1 - np.abs(x_coarse - 1))
    h_coarse_initial = project_to_capped_simplex(h_coarse_initial, target_sum_coarse)
    
    # Optimizer for coarse resolution
    h_coarse = h_coarse_initial.copy()
    T = T0
    alpha = alpha0
    best_coarse_c5 = float('inf')
    best_coarse_h = h_coarse.copy()
    
    # Momentum
    v = np.zeros_like(h_coarse)
    
    # Coarse loop
    for t in range(2000):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute correlations
        corr = compute_correlations(h_coarse)
        corr_max = np.max(corr)
        
        if corr_max < best_coarse_c5:
            best_coarse_c5 = corr_max
            best_coarse_h = h_coarse.copy()
            
        # Top-K shifts
        top_k_indices = np.argpartition(corr, -K)[-K:]
        
        # Compute gradient
        grad = compute_gradient(corr, top_k_indices, T)
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        step = alpha / np.sqrt(t + 1)
        
        # Update
        h_coarse = h_coarse - step * v
        
        # Project
        h_coarse = project_to_capped_simplex(h_coarse, target_sum_coarse)
        
        # Anneal
        T *= 0.9995
        alpha *= 0.999
        
    # Upsample the best coarse solution
    h_fine = np.interp(np.linspace(0, 2, N), np.linspace(0, 2, N_coarse), best_coarse_h)
    h_fine = project_to_capped_simplex(h_fine, target_sum)
    
    # Global restarts
    best_h = h_fine.copy()
    best_c5 = np.max(compute_correlations(h_fine)) * dx
    
    # Multiple restarts with different initialization strategies
    restarts = 15
    
    for restart in range(restarts):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Initialize with different strategies
        if restart % 3 == 0:
            # Random initialization
            h = np.random.rand(N)
        elif restart % 3 == 1:
            # Triangular
            x = np.linspace(0, 2, N)
            h = np.maximum(0, 1 - np.abs(x - 1))
        else:
            # Symmetric random
            h = np.random.rand(N)
            h = (h + h[::-1]) / 2
            
        h = project_to_capped_simplex(h, target_sum)
        
        # Main optimization loop
        T = T0
        alpha = alpha0
        v = np.zeros_like(h)
        
        for t in range(max_iter):
            if time.time() - start_time > 0.7 * budget_s:
                break
                
            # Compute correlations
            corr = compute_correlations(h)
            corr_max = np.max(corr)
            
            if corr_max < best_c5:
                best_c5 = corr_max
                best_h = h.copy()
            
            # Top-K shifts
            top_k_indices = np.argpartition(corr, -K)[-K:]
            
            # Compute gradient
            grad = compute_gradient(corr, top_k_indices, T)
            
            # Momentum
            v = beta * v + (1 - beta) * grad
            step = alpha / np.sqrt(t + 1)
            
            # Update
            h = h - step * v
            
            # Project
            h = project_to_capped_simplex(h, target_sum)
            
            # Anneal
            T *= 0.9995
            alpha *= 0.999
    
    # Local pair-swap polishing
    h = best_h.copy()
    max_swap_iter = 50000
    patience = 1000
    patience_count = 0
    
    for iter_num in range(max_swap_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Select two random indices
        i, j = np.random.randint(0, N, 2)
        
        # Determine feasible interval for transfer
        delta_low = max(-h[i], h[j]-1)
        delta_high = min(1-h[i], h[j])
        
        if delta_low >= delta_high:
            continue
            
        # Perform transfer
        delta = np.random.uniform(delta_low, delta_high)
        h[i] += delta
        h[j] -= delta
        
        # Compute new C5
        corr = compute_correlations(h)
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            patience_count = 0
        else:
            # Accept with small probability (Simulated Annealing)
            temperature = max(0.001, 1.0 - (time.time() - start_time) / budget_s)
            if np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
                best_c5 = c5_new
                best_h = h.copy()
                patience_count = 0
            else:
                # Revert the change
                h[i] -= delta
                h[j] += delta
                patience_count += 1
                
        if patience_count > patience:
            break
    
    # Final projection and evaluation
    final_h = project_to_capped_simplex(best_h, target_sum)
    final_corr = compute_correlations(final_h)
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), float(final_c5), N)
