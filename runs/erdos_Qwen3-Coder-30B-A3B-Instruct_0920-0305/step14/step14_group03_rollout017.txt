def evaluate_c5_reference(h):
    """Reference evaluation using np.correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute auto-correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr * dx

def compute_softmax_gradient(h, alpha=8.0):
    """Compute gradient using exact soft-max approximation of max overlap"""
    corr = compute_correlation(h)
    # Convert correlations to overlaps
    overlaps = np.array([1 - c for c in corr])
    
    # Softmax weights
    max_overlap = np.max(overlaps)
    if max_overlap > 0:
        weights = np.exp(alpha * (overlaps - max_overlap))
    else:
        weights = np.exp(alpha * overlaps)
    weights = weights / np.sum(weights)
    
    # Compute gradient
    grad = np.zeros_like(h)
    L = len(h)
    
    for s in range(-L+1, L):
        if s == 0:
            grad -= 2 * weights[s + L - 1] * h
        elif s > 0:
            # Forward part
            if s < L:
                grad[:L-s] -= weights[s + L - 1] * h[s:]
            # Backward part
            if L - s > 0:
                grad[s:] -= weights[s + L - 1] * h[:L-s]
        else:  # s < 0
            # Forward part
            if -s < L:
                grad[:L+s] -= weights[-s + L - 1] * h[-s:]
            # Backward part
            if L + s > 0:
                grad[-s:] -= weights[-s + L - 1] * h[:L+s]
                
    return grad
