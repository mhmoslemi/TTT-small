import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Build correlation index mapping
    def build_correlation_indices(n):
        indices = []
        for s in range(-n+1, n):
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    indices.append((i, j, s))
        return indices
    
    # Linear programming approach with McCormick relaxation
    def solve_feasibility(C):
        # Variables
        h = cp.Variable(n_points)
        w = cp.Variable(n_points * (2*n_points - 1))  # All possible w_{i,j} pairs
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # McCormick relaxation constraints for all correlations
        corr_indices = build_correlation_indices(n_points)
        corr_vars = {}
        w_idx = 0
        
        for i, j, s in corr_indices:
            # Index this w variable
            corr_vars[(i, j, s)] = w_idx
            w_idx += 1
            
            # McCormick constraints
            constraints.extend([
                w[w_idx-1] >= 0,
                w[w_idx-1] <= h[i],
                w[w_idx-1] <= h[j],
                w[w_idx-1] >= h[i] + h[j] - 1
            ])
        
        # Add correlation constraints
        rhs = target_sum - C / dx
        for s in range(-n_points+1, n_points):
            expr = 0
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    expr += w[corr_vars[(i, j, s)]]
            constraints.append(expr >= rhs)
        
        # Solve
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            return h.value
        except:
            return None
    
    # Binary search on C
    low, high = 0.30, 0.40
    best_C = high
    best_h = None
    best_h_values = None
    
    # Warm start for optimization
    while high - low > 1e-4 and time.time() - start_time < 0.9 * budget_s:
        C = (low + high) / 2
        h_solution = solve_feasibility(C)
        if h_solution is not None:
            # Verify solution - check if it's actually feasible
            h = np.array(h_solution)
            # Project to exact sum
            h = project_to_sum_box(h, target_sum)
            corr = compute_correlations(h)
            max_corr = np.max(corr)
            actual_C = dx * (target_sum - max_corr)
            
            if actual_C <= C:
                high = C
                best_C = C
                best_h = h
                # Interpolation to get better initial guess for next optimization
                if best_h_values is None:
                    best_h_values = h.copy()
            else:
                low = C
        else:
            low = C
    
    # Refinement with sub-gradient method
    if best_h is not None:
        h = best_h.copy()
        best_h_values = h.copy()
        best_C = evaluate_c5_custom(h)
        
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            
            # Compute all correlations
            corr = compute_correlations(h)
            max_corr_idx = np.argmax(corr)
            max_corr_val = corr[max_corr_idx]
            
            # Sub-gradient computation based on active constraints
            s = max_corr_idx - (n_points - 1)
            grad = np.zeros(n_points)
            
            # Compute gradient for worst shift
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    grad[i] += 1
                    grad[j] += 1
            
            # Update with step size
            step_size = 0.005
            h_new = h - step_size * grad
            h_new = project_to_sum_box(h_new, target_sum)
            
            new_C = evaluate_c5_custom(h_new)
            if new_C < best_C:
                h = h_new
                best_C = new_C
                best_h_values = h.copy()
            elif new_C > best_C and np.random.random() < 0.01:
                # Occasionally accept worse solutions to escape local optima
                h = h_new
                best_C = new_C
                best_h_values = h.copy()
    
    # Finer grid refinement
    if best_h_values is not None:
        n_fine = 200
        dx_fine = 2.0 / n_fine
        h_fine = np.interp(np.linspace(0, 2, n_fine), 
                          np.linspace(0, 2, n_points), 
                          best_h_values)
        
        # Project to correct sum
        h_fine = project_to_sum_box(h_fine, n_fine / 2)
        
        # Quick gradient refinement on fine grid
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            corr = compute_correlations(h_fine, n_fine)
            max_corr_idx = np.argmax(corr)
            max_corr_val = corr[max_corr_idx]
            s = max_corr_idx - (n_fine - 1)
            grad = np.zeros(n_fine)
            for i in range(n_fine):
                j = i + s
                if 0 <= j < n_fine:
                    grad[i] += 1
                    grad[j] += 1
            step_size = 0.002
            h_fine = h_fine - step_size * grad
            h_fine = project_to_sum_box(h_fine, n_fine / 2)
            
        best_h_values = h_fine
        best_C = evaluate_c5_custom(best_h_values, n_fine)
    
    return (best_h_values.tolist(), float(best_C), len(best_h_values))

def evaluate_c5_custom(h, n=None):
    """Fast evaluation using FFT"""
    if n is None:
        n = len(h)
    dx = 2.0 / n
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlations(h, n=None):
    """Compute all correlations using FFT"""
    if n is None:
        n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * (2.0 / n)
