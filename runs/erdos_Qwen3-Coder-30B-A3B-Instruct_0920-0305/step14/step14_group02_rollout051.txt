# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a specific shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # For shift s >= 0
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        # For shift s < 0 (equivalent to shift -s > 0 with reversed indices)
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smoothed maximum correlation"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just use the subgradient of the maximum shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert correlation values to weights using softmax
        max_corr = np.max(corr)
        # Shift for numerical stability
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (len(h)-1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run gradient descent with smooth max and optional symmetry"""
    # Random initial h
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    # Start with high temperature and step size
    T = T0
    step = step0
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    for it in range(max_iter):
        # Time check
        if it % 10 == 0:
            # Recompute exact C5 every 10 iterations
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update with step size
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        
        # Apply symmetry constraint if needed
        if sym:
            h = (h + h[::-1]) / 2
        
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Decay temperature
        T = max(1e-5, T * 0.9995)
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def local_swap_search(h, best_c5, time_limit):
    """Local search with pair-wise swaps"""
    start = time.time()
    N = len(h)
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        # Sample delta within feasible range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        c5 = evaluate_c5_fast(h_new)
        if c5 < best_c5:
            h = h_new
            best_c5 = c5
    return h, best_c5
