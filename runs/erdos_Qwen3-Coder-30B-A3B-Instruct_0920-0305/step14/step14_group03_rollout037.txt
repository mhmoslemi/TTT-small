def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    b = 1 - h
    # Use FFT for computation
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for gradient computation"""
    max_corr = np.max(corr)
    if max_corr > 0:
        exp_corr = np.exp(alpha * (corr - max_corr))
    else:
        exp_corr = np.exp(alpha * corr)
    return exp_corr / np.sum(exp_corr)

def compute_gradient(h, corr, alpha):
    """Compute exact gradient using softmax approximation of max"""
    p = compute_softmax_weights(corr, alpha)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # For each shift s, compute contribution to gradient
    n = len(h)
    for s in range(-n+1, n):
        if s == 0:
            # Contribution from shift 0
            grad += -2 * p[s + n - 1] * h
        else:
            # Forward shift contribution
            if s > 0:
                if s < n:
                    grad[:n-s] += -p[s + n - 1] * h[s:]
                # Backward shift contribution
                if s < n:
                    grad[s:] += -p[s + n - 1] * h[:n-s]
            else:  # s < 0
                # Forward shift contribution
                if -s < n:
                    grad[-s:] += -p[s + n - 1] * h[:-s]
                # Backward shift contribution
                if -s < n:
                    grad[:-s] += -p[s + n - 1] * h[-s:]
    
    return grad
