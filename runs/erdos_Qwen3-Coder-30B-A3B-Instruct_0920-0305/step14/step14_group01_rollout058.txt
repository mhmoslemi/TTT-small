import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_box_sum(u, target_sum):
    """Project u onto box constraints [0,1] with sum = target_sum"""
    if np.sum(u) == target_sum:
        return u
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_smooth_max_gradient(u, eps, lambda_inc=1e3, mu=1e4):
    """Compute gradient of smooth max with penalties"""
    # Reconstruct full h from symmetric u
    h = np.concatenate([u, u[::-1]])
    
    # Compute correlation
    corr = compute_correlation(h)
    
    # Compute weights for soft max
    w = np.exp(corr / eps)
    w = w / np.sum(w)
    
    # Gradient of smooth max
    L = len(corr)
    w_rev = w[::-1]
    
    # Convolution-based gradients
    conv1 = np.fft.irfft(np.fft.rfft(w, L) * np.fft.rfft(1-h, L)).real
    conv2 = np.fft.irfft(np.fft.rfft(w_rev, L) * np.fft.rfft(h, L)).real
    
    grad_smooth = conv1 - conv2
    grad_smooth = grad_smooth[(L-len(h))//2:(L+len(h))//2]
    
    # Gradients from penalties
    grad_penalty = np.zeros_like(u)
    
    # Monotonicity penalty (increasing)
    for i in range(1, len(u)):
        diff = u[i-1] - u[i]
        if diff > 0:
            grad_penalty[i-1] += 2 * lambda_inc * diff
            grad_penalty[i] -= 2 * lambda_inc * diff
    
    # Sum constraint penalty
    grad_sum = 2 * mu * (np.sum(u) - target_sum)
    
    # Combine
    return dx * grad_smooth[:len(u)] + grad_penalty + grad_sum

def compute_c5_bound(h):
    """Compute actual C5 bound using max correlation"""
    corr = compute_correlation(h)
    return np.max(corr)

def generate_initial_seed(seed):
    """Generate various initial shapes"""
    np.random.seed(seed)
    m = n_points // 2
    
    # Different initial shapes
    seeds = []
    
    # Uniform
    seeds.append(np.full(m, 0.5))
    
    # Triangular
    x = np.linspace(0, 1, m)
    tri = np.minimum(x, 1-x)
    seeds.append(tri)
    
    # Cosine squared
    x = np.linspace(0, 2, n_points)
    cos_sq = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cos_sq[:m])
    
    # Random monotone with noise
    rand = np.random.rand(m)
    rand_sorted = np.sort(rand)
    monotone = rand_sorted
    seeds.append(monotone)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial seeds
    initial_seeds = generate_initial_seed(seed)
    
    t_mid = start_time + 0.95 * budget_s
    
    for seed_idx, u0 in enumerate(initial_seeds):
        # Ensure initial is feasible
        u = project_to_box_sum(u0, target_sum)
        current_c5 = compute_c5_bound(np.concatenate([u, u[::-1]]))
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = np.concatenate([u, u[::-1]])
        
        # Continuation on epsilon
        for eps in epsilon_schedule:
            # Early exit
            if time.time() > t_mid:
                break
                
            def objective(u):
                return np.max(compute_correlation(np.concatenate([u, u[::-1]])))
            
            def gradient(u):
                return compute_smooth_max_gradient(u, eps)
            
            # Optimization with L-BFGS-B
            result = minimize(
                fun=objective,
                x0=u,
                jac=gradient,
                bounds=[(0, 1)] * len(u),
                method='L-BFGS-B',
                options={'maxfun': 250, 'disp': False}
            )
            
            # Project result
            u = project_to_box_sum(result.x, target_sum)
            
            # Check for better solution
            current_h = np.concatenate([u, u[::-1]])
            current_c5 = compute_c5_bound(current_h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        if time.time() > t_mid:
            break
    
    # Final refinement with projected subgradient descent
    if best_h is not None:
        h = best_h.copy()
        t_end = start_time + budget_s
        alpha0 = 0.01
        t = 0
        
        while time.time() < t_end:
            corr = compute_correlation(h)
            j_star = np.argmax(corr)
            
            # Gradient of the worst shift
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + j_star < len(h):
                    grad[i] += (1 - h[i + j_star])
                if i - j_star >= 0:
                    grad[i] -= h[i - j_star]
            
            # Update step
            alpha_t = alpha0 / np.sqrt(t + 1)
            h_new = h - alpha_t * grad
            h_new = project_to_box_sum(h_new[:len(h)//2], target_sum)
            h_new = np.concatenate([h_new, h_new[::-1]])
            
            # Check for improvement
            c5_new = compute_c5_bound(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            h = h_new
            t += 1

    return (best_h.tolist(), float(best_c5), n_points)
