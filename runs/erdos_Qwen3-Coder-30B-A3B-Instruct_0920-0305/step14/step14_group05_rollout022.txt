```python
import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 using FFT"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old grid to new grid"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def worst_shift(c):
    """Find the shift that gives maximum correlation"""
    idx = np.argmax(c)
    s = idx - (len(c) // 2)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:-s]
        h_minus[:-s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution schedule
    stages = [
        (80, 0.30),   # coarse
        (160, 0.35),  # medium
        (240, 0.35)   # fine
    ]
    
    best_h = None
    best_C = float('inf')
    
    for stage_idx, (n, time_fraction) in enumerate(stages):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        stage_time_limit = start_time + budget_s * time_fraction
        
        # Initialize the vector
        if stage_idx == 0:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, target_sum(n))
        else:
            h = interpolate(best_h, n)
            h = project_to_sum_box(h, target_sum(n))
        
        # Initialize best for this stage
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Simulated annealing
        T = 0.02
        alpha = 0.9997
        beta = 0.9999
        delta = 0.1
        
        iter_count = 0
        accepted_count = 0
        max_iter = 50000
        
        while time.time() < stage_time_limit and iter_count < max_iter:
            # Choose move type (targeted or random)
            if random.random() < 0.3:
                # Targeted move - focus on worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Find indices with low and high desirability
                sorted_indices = np.argsort(d)
                low_idx = sorted_indices[:max(1, len(sorted_indices) // 5)]
                high_idx = sorted_indices[-max(1, len(sorted_indices) // 5):]
                
                i = random.choice(low_idx)
                j = random.choice(high_idx)
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                if h[i] < delta or h[j] > 1 - delta:
                    iter_count += 1
                    continue
                
            # Perform move
            max_transfer = min(delta, h[i], 1 - h[j])
            if max_transfer <= 0:
                iter_count += 1
                continue
            
            transfer = random.uniform(0, max_transfer)
            
            h_new = h.copy()
            h_new[i] -= transfer
            h_new[j] += transfer
            
            # Clip to bounds
            h_new[i] = max(0, min(1, h_new[i]))
            h_new[j] = max(0, min(1, h_new[j]))
            
            # Project to sum constraint
            h_new = project_to_sum_box(h_new, target_sum(n))
            
            # Evaluate
            C_new = C5_value(h_new)
            C_old = C5_value(h)
            
            # Accept or reject
            if C_new <= C_old or random.random() < np.exp(-(C_new - C_old) / T):
                h = h_new
                if C_new < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = C_new
                accepted_count += 1
            
            # Update temperature and step size
            T *= alpha
            delta *= beta
            
            iter_count += 1
            
            # Progress check
            if accepted_count > 1000 and accepted_count % 1000 == 0:
                if time.time() > stage_time_limit:
                    break
        
        # Polishing with exact gradient descent
        h_polish = best_h_stage.copy()
        for k in range(2000):
            if time.time() > stage_time_limit or time.time() > start_time + budget_s * 0.95:
                break
                
            c = corr_fft(h_polish)
            max_val = np.max(c)
            eps = 1e-12
            
            # Find all shifts that are close to max
            s_indices = np.where(np.abs(c - max_val) < eps)[0]
            
            if len(s_indices) == 0:
                break
                
            # Compute gradient for each worst shift
            grad = np.zeros(len(h_polish))
            for s_idx in s_indices:
                s = s_idx - (len(c) // 2)
                grad += gradient_worst(h_polish, s)
            
            grad /= len(s_indices)
            
            # Step size
            eta = 0.5 / np.sqrt(k + 1)
            
            # Update
            h_polish = h_polish - eta * grad
            
            # Project
            h_polish = project_to_sum_box(h_polish, target_sum(n))
            
            # Check improvement
            C_new = C5_value(h_polish)
            if C_new < best_C_stage:
                best_C_stage = C_new
                best_h_stage = h_polish.copy()
        
        # Update global best
        if best_C_stage < best_C:
            best_C = best_C_stage
            best_h = best_h_stage.copy()
    
    return (best_h.tolist(), float(best_C), len(best_h))
```
