def evaluate_c5_exact(h):
    """Evaluate exact C5 using FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_surrogate(h, T=0.05):
    """Evaluate smooth surrogate of C5 using log-sum-exp"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Apply log-sum-exp smoothing
    corr = corr / T
    max_corr = np.max(corr)
    # Subtract max to avoid overflow
    exp_corr = np.exp(corr - max_corr)
    log_sum_exp = max_corr + np.log(np.sum(exp_corr))
    return dx * log_sum_exp

def gradient_c5_surrogate(h, T=0.05):
    """Compute gradient of the smooth surrogate"""
    # Compute all correlations with FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_fft = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Compute weights for log-sum-exp
    corr_fft = corr_fft / T
    max_corr = np.max(corr_fft)
    exp_corr = np.exp(corr_fft - max_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient contribution from each shift
    grad = np.zeros_like(h)
    for s in range(len(h)):
        # For shift s, correlation depends on h[i] and h[i+s]
        # Derivative is 1 - h[i+s] for h[i] term
        # And -h[i-s] for h[i+s] term
        if s == 0:
            grad += weights[s] * (1 - h)  # Note: negative sign from chain rule, but 1-h is correct
        else:
            # h[i] depends on h[i] and h[i+s]
            grad[:len(h)-s] += weights[s] * (1 - h[s:])
            # h[i+s] depends on h[i] and h[i+s] 
            grad[s:] -= weights[s] * h[:-s]
    return dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort values in descending order
    sorted_indices = np.argsort(-x)
    sorted_vals = x[sorted_indices]
    
    # Water-filling algorithm
    remaining_sum = target_sum
    num_used = 0
    
    # Find the point where we can fill to 1
    for i in range(len(sorted_vals)):
        if remaining_sum > (1 - sorted_vals[i]) * (i+1):
            remaining_sum -= (1 - sorted_vals[i])
            num_used = i + 1
        else:
            break
    
    # Fill the first num_used elements to 1, and distribute remaining
    result = x.copy()
    
    # Set first num_used to 1
    for i in range(num_used):
        result[sorted_indices[i]] = 1.0
    
    # Distribute the remaining sum among remaining elements
    if num_used < len(result) and remaining_sum > 0:
        remaining_sum = target_sum - num_used
        for i in range(num_used, len(result)):
            result[sorted_indices[i]] = min(1.0, remaining_sum / (len(result) - num_used))
    
    # If we didn't reach the target, adjust the remaining elements
    current_sum = np.sum(result)
    if current_sum < target_sum:
        adjustment = (target_sum - current_sum) / (len(result) - num_used)
        for i in range(num_used, len(result)):
            result[sorted_indices[i]] += adjustment
    elif current_sum > target_sum:
        adjustment = (current_sum - target_sum) / (len(result) - num_used)
        for i in range(num_used, len(result)):
            result[sorted_indices[i]] -= adjustment
    
    # Clamp to [0, 1]
    result = np.clip(result, 0, 1)
    
    return result
