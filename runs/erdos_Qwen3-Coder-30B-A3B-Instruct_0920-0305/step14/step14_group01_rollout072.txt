import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    harmony_memory = []

    # Harmony Search iterations
    t1 = start_time + 0.8 * budget_s
    for iter_num in range(max_iter):
        if time.time() >= t1:
            break
            
        # Create new harmony
        new_harmony = np.zeros(n_points)
        
        # For each decision variable
        for i in range(n_points):
            # Choose from memory with HMCR probability
            if np.random.rand() < HMCR:
                # Select from memory
                h_selected = harmony_memory[np.random.randint(HMS)][0]
                new_harmony[i] = h_selected[i]
                
                # Adjust pitch with PAR probability
                if np.random.rand() < PAR:
                    new_harmony[i] += np.random.normal(0, 0.05)
            else:
                # Random selection
                new_harmony[i] = np.random.rand()
        
        # Project to feasible region
        new_harmony = project_to_sum_box(new_harmony, TARGET_SUM)

        new_c5 = evaluate_c5(new_harmony)

        if new_c5 < best_c5:

            best_h = new_harmony.copy()
        
        # Replace worst harmony in memory if new one is better
        worst_idx = max(range(HMS), key=lambda i: harmony_memory[i][1])
        if new_c5 < harmony_memory[worst_idx][1]:
            harmony_memory[worst_idx] = (new_harmony, new_c5)
