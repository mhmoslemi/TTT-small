import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort the values in descending order to facilitate water filling
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Water filling algorithm
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the water level
    water_level = 0.0
    for i in range(n):
        if i == n - 1:
            water_level = (target_sum - cumsum[i-1]) / (n - i) if i > 0 else target_sum
        else:
            water_level = (target_sum - cumsum[i]) / (n - i)
            if water_level > sorted_x[i]:
                water_level = sorted_x[i]
        if water_level >= 0 and water_level <= 1:
            break
    
    # Apply the water level
    result = np.zeros_like(x)
    for i in range(n):
        if sorted_x[i] > water_level:
            result[sorted_indices[i]] = min(sorted_x[i], 1.0)
        else:
            result[sorted_indices[i]] = water_level
    
    # Adjust to ensure sum is exactly target_sum
    current_sum = np.sum(result)
    if abs(current_sum - target_sum) > 1e-10:
        # Scale by the ratio
        result = result * target_sum / current_sum
    
    return result

def compute_gradient(h, T):
    """Compute gradient of log-sum-exp approximation"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_fft = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Compute weights
    exp_corr = np.exp(corr_fft / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient per component
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if 0 <= i < len(h) - abs(s) and s >= 0:
                if i + s < len(h):
                    grad[i] += weights[s + len(h) - 1] * (1 - h[i + s])
            elif 0 <= i + s < len(h) and s < 0:
                if i >= -s:
                    grad[i] -= weights[-s + len(h) - 1] * h[i + s]
    
    return dx * grad

def smooth_max_minimize(h0, T0=0.12, tau_T=0.998, max_iter=3000, tol=1e-8):
    """Optimize using smooth max and L-BFGS-B with temperature continuation"""
    h = h0.copy()
    T = T0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Log sum exp optimized L-BFGS
    for iter_num in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, T)
        
        # Create objective function for L-BFGS
        def obj_func(h):
            # Compute correlations
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr_fft = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
            return dx * T * np.log(np.sum(np.exp(corr_fft / T)))
        
        # Gradient for L-BFGS (converted to full gradient + easy finite difference)
        def grad_func(h):
            return compute_gradient(h, T)
        
        try:
            # Use scipy.optimize.minimize with L-BFGS
            res = minimize(obj_func, h, method='L-BFGS-B', 
                          jac=grad_func, bounds=[(0, 1)]*len(h), 
                          options={'gtol': tol, 'ftol': tol})
            h = res.x
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Update best solution
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        except Exception:
            pass
        
        # Update temperature
        T = T * tau_T
        
        # Early stopping based on temperature
        if T < 5e-6:
            break
            
    return best_h, best_c5

        pairs = [(i, j) for i in range(len(h)) for j in range(i+1, len(h))]

            # Compute all correlations for current h
            h_rev = best_h[::-1]
            b = 1 - best_h

            # Walk through every shift to compute change coefficients

            delta_min = max(-best_h[j], best_h[i] - 1)

            if delta_min >= delta_max:

            shift_changes = {}

            for s in range(-len(h)+1, len(h)):

                if 0 <= i <= len(h) - 1 - abs(s) and s >= 0:
                    if i + s < len(h):
                        shift_changes[s][0] += 1  # d/dδ (h_i * (1 - h_{i+s})) = 1 - h_{i+s}
                        shift_changes[s][1] += -h[i + s]  # constant term

                    if j + s < len(h):
                        shift_changes[s][0] -= 1  # d/dδ (h_j * (1 - h_{j+s})) = -(1 - h_{j+s})
                        shift_changes[s][1] += h[j + s]  # constant term

                        shift_changes[s][0] += 1  # d/dδ (h_i * (1 - h_{i+s})) = 1 - h_{i+s}
                        shift_changes[s][1] += -h[i + s]  # constant term
                    
                    if 0 <= j + s <= len(h) - 1:
                        shift_changes[s][0] -= 1  # d/dδ (h_j * (1 - h_{j+s})) = -(1 - h_{j+s})
                        shift_changes[s][1] += h[j + s]  # constant term

            candidates = [delta_min, delta_max]
            
            # For each shift calculate how its correlation would change
            # We want to minimize the maximum shift correlation in some region
            
            # Let's iterate through all possible shifts to compute their affine functions
            affine_shifts = []
            for s, (coeff, const) in shift_changes.items():
                if abs(coeff) > 1e-12:  # Only consider non-zero coefficients
                    # Shift strength changed by: coeff * delta + const
                    affine_shifts.append((s, coeff, const))
            
            # Now find intersections
            # For intersections of f_s(δ) = coeff_s * δ + const_s and f_t(δ) = coeff_t * δ + const_t
            # where coeff_s ≠ coeff_t, set equal: coeff_s * δ + const_s = coeff_t * δ + const_t
            for i, (s1, c1, d1) in enumerate(affine_shifts):
                for j, (s2, c2, d2) in enumerate(affine_shifts[i+1:]):
                    if abs(c1 - c2) > 1e-12:
                        intersection = (d2 - d1) / (c1 - c2)
                        if delta_min <= intersection <= delta_max:
                            candidates.append(intersection)
            
            # Evaluate the objective at candidate points
            best_delta = 0.0
            best_val = float('inf')
            
            for delta in candidates:
                if delta_min <= delta <= delta_max:
                    # Evaluate how all shifts change with this delta
                    new_corr = corr_fft.copy()
                    for s, coeff, const in affine_shifts:
                        # Shift s value: coeff * delta + const
                        new_corr[s + len(h) - 1] += coeff * delta + const
                    
                    # Evaluate new maximum
                    new_max = dx * np.maximum.reduce(new_corr)
                    if new_max < best_val:
                        best_val = new_max
                        best_delta = delta
            
            # Change if beneficial
            if best_val < best_c5:
                new_h = best_h.copy()
                new_h[i] -= best_delta
                new_h[j] += best_delta
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                new_c5 = evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    improved = True
            
            if improved:
                break  # Start over for next round
        
        if not improved:
            break
            
    return best_h, best_c5
