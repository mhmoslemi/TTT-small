import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_reference(h):
    """Reference evaluation using np.correlate"""
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1-h, mode='full')) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for the correlation values"""
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    return w / np.sum(w)

def compute_gradient_exact(h, corr, weights):
    """Compute exact gradient using softmax weights"""
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each shift, compute weighted contribution
    for s in range(-n+1, n):
        if s == 0:
            grad -= 2 * weights[s + n - 1] * h
        elif s > 0:
            # Forward shift
            grad[:n-s] -= weights[s + n - 1] * h[s:]
            # Backward shift
            grad[s:] -= weights[s + n - 1] * h[:n-s]
        else:  # s < 0
            # Forward shift (negative index)
            grad[:n+abs(s)] -= weights[abs(s) + n - 1] * h[abs(s):]
            # Backward shift (negative index)
            grad[abs(s):] -= weights[abs(s) + n - 1] * h[:n-abs(s)]
    
    return grad
