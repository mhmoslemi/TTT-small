def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Fast correlation computation using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute subgradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute smooth gradient using top-K shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Return subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Use only top_k weights for efficiency
    top_idx = np.argpartition(weights, -top_k)[-top_k:]
    g = np.zeros_like(h)
    
    for idx in top_idx:
        w = weights[idx]
        s = idx - (len(h)-1)
        grad_s = subgradient_for_shift(h, s)
        g += w * grad_s
    
    return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Perform one gradient descent restart."""
    # Initial random h
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    T = T0
    step = step0
    
    for t in range(max_iter):
        g = smooth_gradient(h, T, top_k)
        step = step0 / np.sqrt(t + 1)
        
        h = h - step * g
        
        if sym:
            h = (h + h[::-1]) / 2
            
        h = project_onto_capped_simplex(h, target_sum)
        
        if t % 10 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        T *= 0.9995
        
        # Early stopping condition (check time budget)
        if t > 100 and best_c5 < 0.375:
            break
    
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local pair-swap hill climbing"""
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        i, j = np.random.choice(len(h), 2, replace=False)
        
        # Determine feasible delta
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5
