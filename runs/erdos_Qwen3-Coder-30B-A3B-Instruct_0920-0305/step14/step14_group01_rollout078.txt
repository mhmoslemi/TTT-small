```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_sum_box(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon):
    """Compute gradient of smooth max using FFT-based convolution"""
    corr = compute_correlation(h)
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradients via convolution
    b = 1 - h
    L = 2 * len(h) - 1
    
    #conv1 = np.convolve(w, b, mode='full')    # This method is too slow
    #conv2 = np.convolve(w[::-1], h, mode='full')
    
    # Use FFT for fast convolution
    w_fft = np.fft.fft(w, L)
    b_fft = np.fft.fft(b, L)
    h_fft = np.fft.fft(h, L)
    
    conv1 = np.fft.ifft(w_fft * b_fft).real
    conv2 = np.fft.ifft(w_fft[::-1] * h_fft).real
    grad = conv1 - conv2
    
    # Extract central part corresponding to h values (indices 0 to n-1)
    # The shift allows for proper indexing
    grad = grad[(L - len(h)) // 2 : (L - len(h)) // 2 + len(h)]
    return grad

def evaluate_smooth_c5(h, epsilon):
    """Evaluate smooth approximation to C5"""
    corr = compute_correlation(h)
    return epsilon * np.log(np.sum(np.exp(corr / epsilon)))

def evaluate_c5(h):
    """Evaluate true C5 using max correlation"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def get_initial_h_values(seed):
    """Generate initial vector based on seed"""
    np.random.seed(seed)
    # Various initial shapes
    initial_types = [
        np.ones(n_points) * 0.5,    # uniform
        np.minimum(np.arange(n_points), np.arange(n_points)[::-1]) / (n_points - 1),  # triangular
        (1 + np.cos(np.pi * np.arange(n_points) / (n_points - 1))) / 2,              # cosine
    ]
    # Random monotone
    random_u = np.sort(np.random.rand(n_points // 2))
    # Add some noise to make it less symmetric
    random_u += np.random.normal(0, 0.01, n_points // 2)
    random_u = np.clip(random_u, 0, 1)
    initial_types.append(np.concatenate([random_u, random_u[::-1]]))
    
    # Select one randomly
    return initial_types[seed % len(initial_types)]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Continue to improve global best until time runs out
    while time.time() - start_time < 0.95 * budget_s:
        # Start with a good initial seed
        h = get_initial_h_values(seed)
        u = h[:n_points//2]
        u = project_to_sum_box(u, target_sum / 2)
        
        # Continue with multistage optimization
        epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
        
        for epsilon in epsilon_schedule:
            # Define objective function and gradient
            def objective(u_vec):
                h_full = np.concatenate([u_vec, u_vec[::-1]])
                h_full = project_to_sum_box(h_full, target_sum)
                
                # Add penalty for non-monotonicity
                pen_inc = 1e3 * np.sum(np.maximum(0, np.diff(u_vec)) ** 2)
                pen_sum = 1e4 * (np.sum(u_vec) - target_sum/2) ** 2
                
                return evaluate_smooth_c5(h_full, epsilon) + pen_inc + pen_sum
            
            def gradient(u_vec):
                h_full = np.concatenate([u_vec, u_vec[::-1]])
                h_full = project_to_sum_box(h_full, target_sum)
                
                # Gradient of smooth max
                grad_smooth = compute_smooth_max_gradient(h_full, epsilon)
                # Take only gradient for first half (the independent variables)
                grad_smooth = grad_smooth[:len(u_vec)]
                
                # Add gradient from penalty terms
                grad_pen_inc = 2 * 1e3 * np.concatenate([
                    np.diff(u_vec), [0]
                ])
                grad_pen_sum = 2 * 1e4 * (np.sum(u_vec) - target_sum/2) * np.ones_like(u_vec)
                
                return dx * grad_smooth + grad_pen_inc + grad_pen_sum
            
            # Optimize with L-BFGS-B
            res = minimize(
                objective,
                u,
                method='L-BFGS-B',
                jac=gradient,
                bounds=[(0, 1)] * len(u),
                options={'maxfun': 200, 'disp': False}
            )
            u = res.x
            
            # Do final projection and evaluation
            h_full = np.concatenate([u, u[::-1]])
            h_full = project_to_sum_box(h_full, target_sum)
            c5 = evaluate_c5(h_full)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full.copy()
                
        seed += 1
    
    # Final refinement with exact max using projected subgradient descent
    if best_h is not None:
        h = best_h.copy()
        max_iter = 5000
        for i in range(max_iter):
            if time.time() - start_time > 0.99 * budget_s:
                break
            
            corr = compute_correlation(h)  
            j_star = np.argmax(corr)
            # The exact gradient of overlap w.r.t. h is just the shifted complement
            grad = np.zeros_like(h)
            if j_star < len(h):
                # This part could be more precise in interpreting j_star shift
                grad[j_star:] += (1 - h[:-j_star]) if j_star < len(h) else 0
                grad[:len(h)-j_star] -= h[j_star:]
            else:
                # This case might not happen with our implementation but keep for safety
                grad[j_star:] += (1 - h[:len(h)-(j_star-len(h))])
                grad[len(h)-(j_star-len(h)):] -= h[:(j_star-len(h))]
            
            # Projected gradient step
            alpha = 0.001 / np.sqrt(i + 1) 
            h_candidate = h - alpha * grad 
            h_candidate = project_to_sum_box(h_candidate, target_sum)
            
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            h = h_candidate
    
    if best_h is None:
        # Fallback to some default
        h0 = np.ones(n_points) * 0.5
        best_h = h0
        best_c5 = evaluate_c5(h0)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
