import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for the correlation values"""
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    return w / np.sum(w)

def compute_gradient(h, corr, weights):
    """Compute gradient using exact formula for softmax approximation"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift with non-zero weight
    for s in range(-n+1, n):
        ws = weights[s + n - 1]  # index into weights for shift s
        if ws > 1e-15:  # only consider significant weights
            # Handle shift s
            if s == 0:
                grad -= 2 * ws * h
            elif s > 0:
                # Forward shift: h[i] contributes to h[i+s]
                # For i from 0 to n-s-1, h[i+s] has been shifted from h[i]
                # For i from 0 to n-s-1, h[i] contributes to h[i+s] 
                grad[:n-s] -= ws * h[s:]
                grad[s:] -= ws * h[:n-s]
            else:  # s < 0
                # Backward shift: h[i] contributes to h[i+s]
                # Unpack indices properly: h[i] contributes to h[i+s] where i+s >= 0
                # i from -s to n-1, h[i+s] exists
                grad[-s:] -= ws * h[:-s] 
                grad[:-s] -= ws * h[-s:]
    return grad

def compute_grad_exact(h, alpha=8.0):
    """Compute exact gradient of the softmax approximation to max overlap"""
    corr = compute_corr(h)
    weights = compute_softmax_weights(corr, alpha)
    return compute_gradient(h, corr, weights)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Multi-resolution approach
    resolutions = [120, 200, 240]
    current_h = None
    
    for res_idx, n_points in enumerate(resolutions):
        dx = 2.0 / n_points
        TARGET_SUM = n_points / 2
        
        # Initialize from previous finest resolution or initial_h_values
        if res_idx == 0:
            h0 = np.array(initial_h_values)
            # Interpolate to this resolution if needed
            if len(h0) != n_points:
                new_x = np.linspace(0, 2, n_points)
                old_x = np.linspace(0, 2, len(h0))
                h0 = np.interp(new_x, old_x, h0)
            current_h = project_to_sum_box(h0, TARGET_SUM)
        else:
            # Interpolate from previous resolution
            old_x = np.linspace(0, 2, len(current_h))
            new_x = np.linspace(0, 2, n_points)
            h0 = np.interp(new_x, old_x, current_h)
            current_h = project_to_sum_box(h0, TARGET_SUM)
        
        # Three different starting points: interpolated, random, sinusoidal
        seeds = []
        seeds.append(current_h.copy())
        
        # Random seed
        random_h = np.random.rand(n_points)
        random_h = project_to_sum_box(random_h, TARGET_SUM)
        seeds.append(random_h)
        
        # Sinusoidal seed
        x = np.linspace(0, 2, n_points)
        sinusoidal_h = 0.5 + 0.1 * np.sin(np.pi * x)
        sinusoidal_h = project_to_sum_box(sinusoidal_h, TARGET_SUM)
        seeds.append(sinusoidal_h)
        
        # Run optimization for this resolution
        for seed_h in seeds:
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            h = seed_h.copy()
            current_c5 = np.max(compute_corr(h))
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Optimizer parameters
            alpha = 8.0
            eta = 0.12
            beta1, beta2 = 0.9, 0.999
            epsilon = 1e-8
            
            # Adam moment
            m = np.zeros(n_points)
            v = np.zeros(n_points)
            
            # Optimization loop
            max_iter = 2000 if res_idx < len(resolutions) - 1 else 3000
            patience_counter = 0
            
            for iteration in range(max_iter):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Compute gradient
                grad = compute_grad_exact(h, alpha)
                
                # Update Adam moments
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad**2
                
                # Bias correction
                m_hat = m / (1 - beta1**(iteration+1))
                v_hat = v / (1 - beta2**(iteration+1))
                
                # Proposed step
                proposed_h = h - eta * m_hat / (np.sqrt(v_hat) + epsilon)
                
                # Backtracking line search
                improved = False
                step_size = eta
                for _ in range(10):  # Maximum 10 backtracking steps
                    candidate_h = project_to_sum_box(proposed_h, TARGET_SUM)
                    candidate_c5 = np.max(compute_corr(candidate_h))
                    
                    if candidate_c5 < current_c5:
                        h = candidate_h
                        current_c5 = candidate_c5
                        improved = True
                        break
                    else:
                        step_size *= 0.5
                        proposed_h = h - step_size * m_hat / (np.sqrt(v_hat) + epsilon)
                
                if not improved:
                    # Reduce learning rate if no improvement
                    eta *= 0.7
                    if eta < 1e-6:
                        break
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    patience_counter = 0
                else:
                    patience_counter += 1
                
                # Adaptive temperature
                if iteration % 100 == 0:
                    alpha = min(alpha * 1.07, 200.0)
                
                # Adaptive learning rate
                if iteration % 200 == 0:
                    eta *= 1.01
                
                # Add jitter occasionally
                if iteration % 800 == 0 and iteration > 0:
                    jitter = np.random.normal(0, 0.001, n_points)
                    temp_h = h + jitter
                    temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                    temp_c5 = np.max(compute_corr(temp_h))
                    if temp_c5 < current_c5:
                        h = temp_h
                        current_c5 = temp_c5
                        if temp_c5 < best_c5:
                            best_c5 = temp_c5
                            best_h = h.copy()
                
                if patience_counter > 500:
                    break
        
        # Update current_h for next resolution
        current_h = best_h.copy() if best_h is not None else current_h
        
        # Early exit if time is running out
        if time.time() - start_time > 0.95 * budget_s:
            break
    
    # Final verification with proper evaluation
    if best_h is not None:
        c5_result = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
        return (best_h.tolist(), float(c5_result), n_points)
    else:
        # Fallback to initial solution if nothing better found
        h0 = np.array(initial_h_values)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        c5_result = np.max(np.correlate(h0, 1-h0, mode='full')) * dx
        return (h0.tolist(), float(c5_result), n_points)
