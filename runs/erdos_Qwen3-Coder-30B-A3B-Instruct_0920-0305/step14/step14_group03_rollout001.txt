def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    # Compute correlation using FFT
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    # O = dx * (target_sum - correlation)
    O = dx * (target_sum - corr)
    return O

def compute_gradient(h, w, dx):
    """Compute gradient using exact formula"""
    n = len(h)
    grad = np.zeros(n)
    
    # Loop through all shifts
    for r in range(2 * n - 1):
        s = r - (n - 1)
        ws = w[r]
        
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward contribution h[i+s]
            grad[:n-s] += -dx * ws * h[s:]
            # backward contribution h[i-s]
            grad[s:] += -dx * ws * h[:n-s]
        else:
            # s < 0
            s_abs = -s
            # forward contribution h[i+s] = h[i-s_abs]
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            # backward contribution h[i-s] = h[i+s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad
