import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, temperature=0.01):
    """Evaluate smooth surrogate of C5"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Apply log-sum-exp smoothing
    corr = corr * dx
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / temperature)
    log_sum_exp = max_corr + temperature * np.log(np.sum(exp_terms))
    return log_sum_exp

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm for projection onto capped simplex
    # Sort x in descending order
    x_sorted = np.sort(x)[::-1]
    
    # Find the threshold index
    cumsum = np.cumsum(x_sorted)
    threshold_index = -1
    for i in range(len(x_sorted)):
        if cumsum[i] <= target_sum:
            threshold_index = i
        else:
            break
    
    if threshold_index == -1:
        # All elements need to be reduced
        threshold = target_sum / len(x)
        projected = np.clip(x, 0, threshold)
    else:
        # Threshold is between elements
        threshold = (target_sum - cumsum[threshold_index]) / (len(x) - threshold_index - 1)
        projected = np.clip(x, 0, threshold)
        # Adjust if needed
        if np.sum(projected) < target_sum:
            # Add the remaining amount
            diff = target_sum - np.sum(projected)
            # Distribute among highest elements
            sorted_indices = np.argsort(x)[::-1]
            for idx in sorted_indices:
                if diff <= 0:
                    break
                add = min(diff, 1 - projected[idx])
                projected[idx] += add
                diff -= add
    
    return projected

def compute_gradient(h, temperature=0.01, top_k=12):
    """Compute gradient of smooth surrogate"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real * dx
    
    # Get top K shifts for gradient computation
    top_indices = np.argsort(-corr)[:top_k]
    
    # Compute weights for each shift
    max_corr = np.max(corr)
    exp_terms = np.exp((corr[top_indices] - max_corr) / temperature)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient for top K shifts
    grad = np.zeros_like(h)
    
    # For each top shift, compute derivative
    for i, s in enumerate(top_indices):
        if s >= 0:
            # Forward shift case
            for j in range(len(h) - s):
                grad[j] += weights[i] * (1 - h[j + s])
                if j + s < len(h):
                    grad[j + s] -= weights[i] * h[j]
        else:
            # Backward shift case
            s_abs = -s
            for j in range(len(h) - s_abs):
                grad[j + s_abs] += weights[i] * (1 - h[j])
                if j < len(h):
                    grad[j] -= weights[i] * h[j + s_abs]
    
    return grad

def lbfgsb_optimization(h0, max_iter=2000, tolerance=1e-6):
    """Optimize using L-BFGS-B with projected constraints"""
    def fun(h):
        return evaluate_c5_smooth(h)
    
    def jac(h):
        return compute_gradient(h)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(len(h0))]
    
    # Optimization with L-BFGS-B
    result = minimize(
        fun, h0,
        method='L-BFGS-B',
        jac=jac,
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': tolerance, 'gtol': tolerance}
    )
    
    return result.x
