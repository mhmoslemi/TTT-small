import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Precompute some harmonic basis functions for better exploration
    harmonics = []
    for i in range(1, 10):
        h = np.sin(np.pi * i * np.linspace(0, 2, n_points) / 2)
        harmonics.append(h)
    
    # Hybrid optimization approach
    # 1. Evolutionary algorithm
    popsize = 15
    population = [project_to_sum_box(np.random.rand(n_points), TARGET_SUM)

    # Add the best starting solution
    population[0] = h0.copy()
    
    # DE parameters
    F = 0.8
    CR = 0.7
    max_gen = 10000
    
    gen = 0
    t1 = start_time + 0.6 * budget_s
    while time.time() < t1 and gen < max_gen:
        gen += 1
        for i in range(popsize):
            # Differential evolution mutation
            r1, r2, r3 = np.random.choice(popsize, 3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            child = population[i].copy()
            jrand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == jrand:
                    child[j] = mutant[j]
            
            # Repair
            child = project_to_sum_box(child, TARGET_SUM)
            
            # Selection
            child_c5 = evaluate_c5(child)
            if child_c5 < evaluate_c5(population[i]):
                population[i] = child.copy()

    if best_h is not None:

            if iter_num % 3 == 0:

            eps = 1e-4

                h_plus[i] = min(h[i] + eps, 1.0)
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                
                h_minus = h.copy()
                h_minus[i] = max(h[i] - eps, 0.0)
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                
                c5_plus = evaluate_c5(h_plus)
                c5_minus = evaluate_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with adaptive learning rate
            learning_rate = 0.05 * (1.0 - iter_num / 1000.0)  # Decay learning rate
            h_new = h - learning_rate * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
