def evaluate_h(h_values, dx):
    """Evaluate the C5 bound for given h values"""

    correlation = np.correlate(h, 1 - h, mode='full')

    valid_correlations = correlation[len(h)-1:]
    c5_bound = np.max(valid_correlations) * dx
