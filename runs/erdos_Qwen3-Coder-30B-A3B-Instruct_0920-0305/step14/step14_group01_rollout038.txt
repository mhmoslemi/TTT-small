```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_box_sum(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum using waterfilling"""
    x = np.array(x, dtype=float)
    low, high = 0.0, 1.0
    tol = 1e-12
    
    def f(lam):
        projected = np.clip(x + lam, low, high)
        return np.sum(projected) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b) / 2, low, high)
    return result

def evaluate_correlation(h):
    """Compute correlation using FFT: h ⋆ (1 - h)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon):
    """Compute gradient of log-sum-exp approximation of max correlation"""
    # Compute correlation with all shifts
    corr = evaluate_correlation(h)
    
    # Compute smooth approximation weights
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Gradient computation via FFT convolutions
    b = 1 - h
    L = 2 * len(h) - 1
    
    # Convolution: w * b
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # Convolution: w[::-1] * h  
    conv2 = np.fft.ifft(np.fft.fft(w[::-1], L) * np.fft.fft(h, L)).real
    
    # Combine to get gradient  
    grad = conv1 - conv2
    
    # Extract central part (match length of original h)
    grad = grad[len(h)-1:2*len(h)-1]
    
    # Scale by dx
    return dx * grad

def evaluate_smooth_max(h, epsilon):
    """Evaluate smooth log-sum-exp approximation of max correlation"""
    corr = evaluate_correlation(h) 
    return epsilon * np.log(np.sum(np.exp(corr / epsilon)))

def compute_monotonicity_penalty(u):
    """Compute penalty for violating monotonicity (unimodal shape)"""
    penalty = 0.0
    lambda_pen = 1e3
    
    # Increase constraint only: u[i-1] <= u[i] for i=1...m-1
    for i in range(1, len(u)):
        diff = u[i-1] - u[i]
        if diff > 0:
            penalty += lambda_pen * diff**2
            
    return penalty

def compute_sum_penalty(u, target):
    """Compute penalty for violating sum constraint"""
    mu = 1e4
    diff = np.sum(u) - target
    return mu * diff**2

def compute_full_gradient(u, epsilon):
    """Compute complete gradient for the full objective"""
    # Reconstruct full h vector (symmetric)
    h = np.concatenate([u, u[::-1]])
    
    # Gradient of smooth max
    grad = compute_smooth_max_gradient(h, epsilon)
    
    # Only first half (independent variables)
    grad_u = grad[:len(u)]
    
    # Add penalty terms (monotonicity)
    lambda_pen = 1e3
    grad_penalty = np.zeros_like(u)
    
    # Monotonicity gradient (first order)
    for i in range(1, len(u)):
        if u[i-1] > u[i]:
            grad_penalty[i-1] += -2 * lambda_pen * (u[i-1] - u[i])
            grad_penalty[i] += 2 * lambda_pen * (u[i-1] - u[i])
    
    # Sum penalty gradient
    mu = 1e4
    grad_sum = 2 * mu * (np.sum(u) - target_sum) * np.ones_like(u)
    
    return grad_u + grad_penalty + grad_sum

def compute_initial_h(shape_type):
    """Generate initial h vectors of various shapes"""
    m = n_points // 2
    
    if shape_type == "uniform":
        return np.full(m, 0.5)
    elif shape_type == "triangular":
        x = np.linspace(0, 1, m)
        return np.minimum(x, 1 - x)
    elif shape_type == "cosine":
        x = np.linspace(0, 1, m)
        return (1 + np.cos(np.pi * (x - 1))) / 2
    elif shape_type == "random_monotone":
        # Generate random vector and sort to create monotone increasing
        r = np.random.rand(m)
        return np.sort(r)
    else:
        return np.full(m, 0.5)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Track best solution found
    best_c5 = float('inf')
    best_h = None
    
    # Multi-start optimization with different initial shapes
    initial_shapes = ["uniform", "triangular", "cosine", "random_monotone"]
    
    # Generate multiple restarts within time budget
    restart = 0
    while time.time() - start_time < 0.95 * budget_s:
        restart += 1
        if restart > 10:  # Safety break
            break
            
        # Choose random initial shape
        shape_type = np.random.choice(initial_shapes)
        u0 = compute_initial_h(shape_type)
        
        # Project to valid sum and bounds
        u0 = project_to_box_sum(u0, target_sum)
        
        current_u = u0.copy()
        current_c5 = float('inf')
        
        # Continuation on epsilon values
        for eps in epsilon_schedule:
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Optimization for current epsilon
            result = minimize(
                fun=lambda u: evaluate_smooth_max(np.concatenate([u, u[::-1]]), eps) + 
                              compute_monotonicity_penalty(u) + 
                              compute_sum_penalty(u, target_sum),
                x0=current_u,
                method='L-BFGS-B',
                bounds=[(0, 1)] * len(current_u),
                jac=lambda u: compute_full_gradient(u, eps),
                options={'maxfun': 250, 'disp': False}
            )
            
            if result.success:
                current_u = result.x
                
                # Project to ensure constraints
                current_u = project_to_box_sum(current_u, target_sum)
                
                # Evaluate true C5
                h_full = np.concatenate([current_u, current_u[::-1]])
                corr = evaluate_correlation(h_full)
                c5 = dx * np.max(corr)
                
                if c5 < current_c5:
                    current_c5 = c5
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_full.copy()
            
            # For next epsilon, use current result as starting point
            if not result.success:
                break
                
        # One final gradient-based refinement on exact max
        if best_h is not None and time.time() - start_time < 0.99 * budget_s:
            h = best_h.copy()
            lr = 0.1
            for _ in range(1000):
                if time.time() - start_time > 0.99 * budget_s:
                    break
                corr = evaluate_correlation(h)
                j_star = np.argmax(corr)
                
                # Compute subgradient for the j_star shift
                grad = np.zeros_like(h)
                for i in range(len(h)):
                    if i + j_star < len(h):
                        grad[i] += (1 - h[i + j_star])
                    if i - j_star >= 0:
                        grad[i] -= h[i - j_star]
                
                # Update
                h_new = h - lr * grad
                h_new = project_to_box_sum(h_new, target_sum)
                
                # Evaluate
                corr_new = evaluate_correlation(h_new)
                c5_new = dx * np.max(corr_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                
                h = h_new
                lr *= 0.995  # Decay learning rate
    
    # Final evaluation of best h
    if best_h is None:
        # Fallback to initial construction if nothing found
        u_init = compute_initial_h("cosine")
        u_init = project_to_box_sum(u_init, target_sum)
        best_h = np.concatenate([u_init, u_init[::-1]])
    
    # Final C5 evaluation
    final_corr = evaluate_correlation(best_h)
    final_c5 = dx * np.max(final_corr)
    
    return (best_h.tolist(), float(final_c5), n_points)
```
