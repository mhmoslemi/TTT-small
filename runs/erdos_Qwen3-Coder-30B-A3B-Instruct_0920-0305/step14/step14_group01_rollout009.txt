```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm
    x = np.array(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    
    # Find the threshold
    threshold = 0.0
    for i in range(len(x_sorted)):
        if cumsum[i] > target_sum:
            threshold = (cumsum[i] - target_sum) / (i + 1) 
            break
    else:
        # All values are zero or we're targeting sum=0
        threshold = 0.0
    
    # Project
    result = np.clip(x - threshold, 0, 1)
    return result

def compute_correlation(h, dx):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def gradient_of_smooth_max(h, epsilon):
    """Compute gradient of log-sum-exp approximation of the max overlap"""
    corr = compute_correlation(h, dx)
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)

    # Two convolutions needed for gradient
    b = 1 - h
    h_rev = h[::-1]
    L = len(h) * 2 - 1
    
    # conv(w, b) using FFT
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # conv(w_rev, h) using FFT
    w_rev = w[::-1]
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    grad = conv1 - conv2
    return grad[len(h)-1:2*len(h)-1] * dx

def total_objective_and_gradient(u, epsilon, lambda_inc=1e3, mu=1e4):
    """Compute objective value and gradient for a given u (first half)"""
    # Reconstruct full h
    h = np.concatenate([u, u[::-1]])
    
    # Gradient of smooth objective
    grad_smooth = gradient_of_smooth_max(h, epsilon)
    
    # Add penalty for sum constraint
    penalty_grad = 2 * mu * (np.sum(u) - target_sum) * np.ones_like(u)
    
    # Add penalty for monotonicity (increasing half)
    diff_u = np.diff(u)
    if len(diff_u) > 0:
        monotonicity_grad = lambda_inc * np.concatenate([np.zeros(1), -2 * diff_u])
        # Adjust for mirror effect (decreasing side)
        monotonicity_grad[:-1] += lambda_inc * np.concatenate([2 * diff_u, np.zeros(1)])
    else:
        monotonicity_grad = np.zeros_like(u)
    
    total_grad = grad_smooth[:len(u)] + penalty_grad + monotonicity_grad
    
    # Objective value (only the smooth max)
    corr = compute_correlation(h, dx)
    obj = epsilon * np.log(np.sum(np.exp(corr/epsilon))) * dx
    
    return obj, total_grad

def generate_initial_seeds():
    """Generate multiple initial seeds for restarts"""
    seeds = []
    
    # Uniform
    seeds.append(np.full(n_points//2, 0.5))
    
    # Triangular
    x = np.linspace(0, 1, n_points//2)
    tri = np.minimum(x, 1-x)
    seeds.append(tri)
    
    # Cosine squared
    x = np.linspace(0, 2, n_points)
    cos_sq = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cos_sq[:n_points//2])
    
    # Random monotone
    rand = np.random.rand(n_points//2)
    rand_sorted = np.sort(rand)
    seeds.append(rand_sorted)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate seeds
    initial_seeds = generate_initial_seeds()
    
    best_c5 = float('inf')
    best_h = None
    
    # Iterate over seeds and ε schedule
    max_iter_per_stage = 250
    
    for seed_idx, u0 in enumerate(initial_seeds):
        u = u0.copy()
        for epsilon in epsilon_schedule:
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # For the last stage, we might do more iterations
            current_max_iter = max_iter_per_stage if epsilon < 0.05 else max_iter_per_stage
            
            # Use L-BFGS-B to minimize smooth approximation
            res = minimize(
                lambda u: total_objective_and_gradient(u, epsilon),
                u,
                method='L-BFGS-B',
                bounds=[(0, 1)] * len(u),
                jac=True,
                options={'maxiter': current_max_iter, 'disp': False}
            )
            
            u = res.x
            # Project to constraint
            u = project_to_box_sum(u, target_sum)
            
            # Reconstruct full h
            h = np.concatenate([u, u[::-1]])
            
            # Evaluate true C5
            corr = compute_correlation(h, dx)
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if time.time() - start_time > 0.95 * budget_s:
            break
    
    # Final subgradient refinement
    if best_h is not None:
        h = best_h.copy()
        alpha_0 = 0.01
        max_refine_iter = 5000
        for t in range(max_refine_iter):
            if time.time() - start_time > budget_s:
                break
                
            corr = compute_correlation(h, dx)
            worst_shift_idx = np.argmax(corr)
            
            # Compute gradient for that shift
            grad = np.zeros_like(h)
            # Convolution approach for speed
            # Just compute for one shift, we'll take the direction
            # Not optimal but works for basic refinement
            if worst_shift_idx < len(h):
                for i in range(len(h)):
                    if i + worst_shift_idx < len(h):
                        grad[i] += (1 - h[i + worst_shift_idx])
                    if i - worst_shift_idx >= 0:
                        grad[i] -= h[i - worst_shift_idx]
            else:
                # Mirror the problem
                mirror_shift = 2 * len(h) - 2 - worst_shift_idx
                for i in range(len(h)):
                    if i + mirror_shift < len(h):
                        grad[i] += (1 - h[i + mirror_shift])
                    if i - mirror_shift >= 0:
                        grad[i] -= h[i - mirror_shift]
            
            alpha_t = alpha_0 / np.sqrt(t + 1)
            h_new = h - alpha_t * grad
            h_new = project_to_box_sum(h_new, target_sum)
            h = h_new
            
            # Evaluate bound
            corr = compute_correlation(h, dx)
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    if best_h is None:
        # Use the simplest initialization
        h = np.full(n_points, 0.5)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
