import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project x onto the capped simplex {x : sum(x) = target_sum, 0 <= x <= 1}"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlations(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), n=L)
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = compute_correlations(h)
    return np.max(corr) * dx

def gradient_corr_s(h, s):
    """Compute gradient of correlation for shift s"""
    n = len(h)
    grad = np.zeros(n)
    
    if s >= 0:
        # grad[i] = (1 - h[i+s]) - h[i-s] for i in valid range
        if s < n:
            grad[s:] = 1 - h[:n-s]
            grad[:n-s] -= h[s:]
    else:
        # s < 0, so shift is positive
        s_abs = -s
        if s_abs < n:
            grad[:n-s_abs] = 1 - h[s_abs:]
            grad[s_abs:] -= h[:n-s_abs]
    
    return grad

def compute_top_k_weights(corr, T, K):
    """Compute softmax weights for top K shifts"""
    # Subtract max for numerical stability
    corr_stable = corr - np.max(corr)
    exp_corr = np.exp(corr_stable / T)
    exp_corr_sum = np.sum(exp_corr)
    weights = exp_corr / exp_corr_sum
    
    # Get top K indices
    top_k_idx = np.argpartition(-weights, K)[:K]
    top_k_weights = weights[top_k_idx]
    
    # Re-normalize
    top_k_weights = top_k_weights / np.sum(top_k_weights)
    
    return top_k_idx, top_k_weights

def project_symmetric(h):
    """Make h symmetric by averaging with its reverse"""
    return (h + h[::-1]) / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup at finer resolution
    n_points = 120
    global dx
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Coarse optimization (N=60) to get a good starting point
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.clip(np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_coarse) - 1)), 0, 1)
    h_coarse = project_to_capped_simplex(h_coarse, target_sum / 2)
    
    # Run short continuation on coarse grid (2000 iterations)
    T0 = 0.1
    alpha0 = 0.6
    K = 10
    beta = 0.9
    max_iter_coarse = 2000
    
    h = h_coarse.copy()
    T = T0
    alpha = alpha0
    v = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = None
    
    for t in range(max_iter_coarse):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute correlations
        corr = compute_correlations(h)
        
        # Get top K shifts and weights
        top_k_idx, top_k_weights = compute_top_k_weights(corr, T, K)
        
        # Compute gradient using only top K shifts
        g = np.zeros_like(h)
        for i, s in enumerate(top_k_idx):
            grad_s = gradient_corr_s(h, s)
            g += top_k_weights[i] * grad_s
        
        # Momentum
        v = beta * v + (1 - beta) * g
        step = alpha / np.sqrt(t + 1)
        
        # Limit step size to prevent overshooting
        max_grad = np.max(np.abs(g))
        step = min(step, 0.5 / (max_grad + 1e-12))
        
        h = h - step * v
        
        # Project to feasible set
        h = project_to_capped_simplex(h, target_sum / 2)
        
        # Apply symmetry for symmetric restarts
        h = project_symmetric(h)
        
        # Update temperature and step size
        T *= 0.9995
        alpha *= 0.999
        
        # Periodic evaluation
        if t % 10 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Interpolate best coarse solution to fine grid
    if best_h is not None:
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, n_coarse), best_h)
        h_fine = project_to_capped_simplex(h_fine, target_sum)
    else:
        # Use triangular distribution as fallback
        h_fine = np.clip(np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1)), 0, 1)
        h_fine = project_to_capped_simplex(h_fine, target_sum)
    
    # Global optimization with multiple restarts
    n_restarts = 15
    best_c5_global = float('inf')
    best_h_global = None
    
    for restart in range(n_restarts):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Random starts with some structure
        if restart == 0:  # First restart: triangle
            h = np.clip(np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1)), 0, 1)
            h = project_to_capped_simplex(h, target_sum)
        else:  # Random starts
            h = np.random.rand(n_points)
            h = project_to_capped_simplex(h, target_sum)
            
        # Symmetrize for some restarts
        if restart % 3 == 0:
            h = project_symmetric(h)
            
        # Fine continuation (3000 iterations)
        T = T0
        alpha = alpha0
        v = np.zeros_like(h)
        
        for t in range(3000):
            if time.time() - start_time > 0.7 * budget_s:
                break
                
            # Compute correlations
            corr = compute_correlations(h)
            
            # Get top K shifts and weights
            top_k_idx, top_k_weights = compute_top_k_weights(corr, T, K)
            
            # Compute gradient using only top K shifts
            g = np.zeros_like(h)
            for i, s in enumerate(top_k_idx):
                grad_s = gradient_corr_s(h, s)
                g += top_k_weights[i] * grad_s
            
            # Momentum
            v = beta * v + (1 - beta) * g
            step = alpha / np.sqrt(t + 1)
            
            # Limit step size to prevent overshooting
            max_grad = np.max(np.abs(g))
            step = min(step, 0.5 / (max_grad + 1e-12))
            
            h = h - step * v
            
            # Project to feasible set
            h = project_to_capped_simplex(h, target_sum)
            
            # Apply symmetry for some restarts
            if restart % 3 == 0:
                h = project_symmetric(h)
            
            # Update temperature and step size
            T *= 0.9995
            alpha *= 0.999
            
            # Periodic evaluation
            if t % 10 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5_global:
                    best_c5_global = c5
                    best_h_global = h.copy()
    
    # Final polishing with pair-swap local search
    if best_h_global is not None:
        h = best_h_global.copy()
        current_c5 = best_c5_global
        patience = 0
        max_patience = 5000
        
        # Apply local search for remaining time
        remaining_time = budget_s - (time.time() - start_time)
        max_iterations = int(remaining_time * 10000)  # Rough estimate
        
        for i in range(max_iterations):
            if time.time() - start_time > budget_s:
                break
                
            # Try pair-swap moves
            i1, i2 = np.random.randint(0, n_points, 2)
            if i1 >= i2:
                i1, i2 = i2, i1
                
            # Compute feasible interval for Δ
            delta_low = max(-h[i1], h[i2] - 1)
            delta_high = min(1 - h[i1], h[i2])
            
            if delta_low < delta_high:
                # Random feasible delta
                delta = np.random.uniform(delta_low, delta_high)
                
                # Apply the swap
                h_new = h.copy()
                h_new[i1] += delta
                h_new[i2] -= delta
                
                # Project to simplex
                h_new = project_to_capped_simplex(h_new, target_sum)
                
                # Evaluate
                c5_new = evaluate_c5(h_new)
                
                # Accept or reject
                if c5_new < current_c5:
                    h = h_new
                    current_c5 = c5_new
                    patience = 0
                else:
                    # Accept with simulated annealing probability
                    if c5_new < current_c5 + 1e-10:  # For numerical stability
                        # Small temperature decay
                        temperature = max(0.001, 0.01 * (1 - (time.time() - start_time) / budget_s))
                        if np.random.rand() < np.exp(-(c5_new - current_c5) / temperature):
                            h = h_new
                            current_c5 = c5_new
                            patience = 0
                    else:
                        patience += 1
                        
                    if patience >= max_patience:
                        break
    
    # Final evaluation
    if best_h_global is not None and current_c5 < best_c5_global:
        final_h = h.copy()
        final_c5 = current_c5
    elif best_h_global is not None:
        final_h = best_h_global.copy()
        final_c5 = best_c5_global
    else:
        final_h = h_fine.copy()
        final_c5 = evaluate_c5(h_fine)
    
    # Final projection and exact evaluation
    final_h = project_to_capped_simplex(final_h, target_sum)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
