# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute smooth gradient using top-K shifts"""
    corr = corr_fft(h)
    
    if T < 1e-12:
        # Just use the subgradient of the maximum shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Use only top_k weights for efficiency
    top_idx = np.argpartition(weights, -top_k)[-top_k:]
    g = np.zeros_like(h)
    
    for idx in top_idx:
        w = weights[idx]
        s = idx - (len(h)-1)
        grad_s = subgradient_for_shift(h, s)
        g += w * grad_s
        
    return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Perform one gradient descent restart"""
    # Initial random h
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    h = h0.copy()
    T = T0
    step = step0
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    beta = 0.9  # momentum
    v = np.zeros_like(h)
    
    for t in range(max_iter):
        if t % 50 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update with momentum
        v = beta * v + (1 - beta) * g
        h = h - step * v
        
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Apply symmetry if requested
        if sym:
            h = (h + h[::-1]) / 2
            h = project_onto_capped_simplex(h, target_sum)
        
        # Decrease step size and temperature
        step = step0 / np.sqrt(t + 1)
        T *= 0.9995
        
        # Early stopping for time limits
        if t > 0 and t % 100 == 0:
            if time.time() > time.time() + 0.1:
                break
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def local_swap_search(h, best_c5, time_limit):
    """Perform local swap search"""
    start_time = time.time()
    N = len(h)
    
    while time.time() - start_time < time_limit:
        # Pick two distinct indices
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_min = max(-h[i], h[j] - 1)
        delta_max = min(1 - h[i], h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Update
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        
        # Occasionally accept worse moves (very rarely)
        elif np.random.rand() < 0.0001:
            h = h_new
    
    return h, best_c5
