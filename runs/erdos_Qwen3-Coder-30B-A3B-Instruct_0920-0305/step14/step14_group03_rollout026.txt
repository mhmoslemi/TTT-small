import numpy as np

from scipy.fft import fft, ifft
import time

def compute_corr(h):
    """Compute auto-correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for the correlation values"""
    # Shift correlation to be centered around 0 for numerical stability
    max_corr = np.max(corr)
    if max_corr > 0:
        shifted_corr = corr - max_corr
        exp_corr = np.exp(alpha * shifted_corr)
    else:
        exp_corr = np.exp(alpha * corr)
    
    weights = exp_corr / np.sum(exp_corr)
    return weights

def compute_gradient(h, corr, weights):
    """Compute gradient using weighted sum of individual gradients"""
    n = len(h)
    grad = np.zeros(n)
    
    # Add contribution from the shift s = 0 (self-correlation)
    grad -= 2 * weights[n-1] * h
    
    # Add contributions from all other shifts
    for s in range(1, n):
        # Forward shift contribution (h[i] contributes to h[i+s])
        if s < n:
            grad[:n-s] -= weights[n-1+s] * h[s:]
        # Backward shift contribution (h[i] contributes to h[i-s])  
        if s < n:
            grad[s:] -= weights[n-1-s] * h[:n-s]
    
    return grad

def project_to_sum_box(x, target_sum):

    a, b = -1.0, 1.0

        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
