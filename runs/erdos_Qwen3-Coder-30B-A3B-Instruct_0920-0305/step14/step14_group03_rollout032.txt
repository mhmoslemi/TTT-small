import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

# Global variables
TARGET_SUM = None
n_points = 100
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        init_strategy = project_to_sum_box(init_strategy, TARGET_SUM)

            best_h = current_h.copy()

    def objective(h):
        return evaluate_c5(h)

        return np.sum(h) - TARGET_SUM

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}

            worst_shift_idx = np.argmax(corr)
            worst_shift = worst_shift_idx - len(best_h) + 1
            
            gradient = np.zeros(n_points)
            for i in range(n_points):
                if 0 <= i + worst_shift < n_points:
                    gradient[i] -= (1 - best_h[i + worst_shift])
                if 0 <= i - worst_shift < n_points:
                    gradient[i] -= (1 - best_h[i - worst_shift])
            
            # Smooth step
            new_h = best_h - 0.001 * gradient
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    if best_h is None:

    return (best_h.tolist(), float(best_c5), n_points)
