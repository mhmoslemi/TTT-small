import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT with zero padding"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max correlation using top-K shifts"""
    corr = corr_fft(h)
    
    if T < 1e-12:
        # Subgradient of max shift in the limit
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert to exponentials with numeric stability
        max_corr = np.max(corr)
        # Shift to improve numeric stability
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        
        # Use only top_k weights for efficiency, ignoring negligible ones
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N-1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run gradient descent with continuation and optional symmetry"""
    # Random initial h
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    h = h0.copy()
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    T = T0
    step = step0
    
    for it in range(max_iter):
        if it % 100 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        
        # Apply symmetry constraint if needed
        if sym:
            h = (h + h[::-1]) / 2
            
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Update temperature
        T *= 0.9995
        
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local pair-swap hill climbing"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    no_improvement_count = 0
    
    while time.time() - start < time_limit and no_improvement_count < 5000:
        i, j = np.random.choice(N, 2, replace=False)
        # Sample delta with bounds
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        new_c5 = evaluate_c5_fast(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Allocate time budget with safety margin
    total_budget = budget_s * 0.95
    remaining_budget = total_budget
    
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: Coarse optimization (n=60)
    if remaining_budget > 10:
        n_coarse = 60
        dx_coarse = 2.0 / n_coarse
        target_sum_coarse = n_coarse / 2
        
        h_coarse, c5_coarse = gradient_descent_restart(
            n_coarse, target_sum_coarse, 1000, 0.5, 0.1, 5, sym=False
        )
        
        # Interpolate to main resolution
        t_coarse = np.linspace(dx_coarse/2, 2 - dx_coarse/2, n_coarse)
        t_main = np.linspace(dx/2, 2 - dx/2, n_points)
        h_main = np.interp(t_main, t_coarse, h_coarse)
        h_main = project_onto_capped_simplex(h_main, TARGET_SUM)
        
        # Start with interpolated vector
        h_start = h_main
        c5_start = evaluate_c5_fast(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
        
        remaining_budget -= 40  # ~40s for coarse run and upsampling
    
    # Phase 2: Multiple restarts at main resolution
    if remaining_budget > 100:
        restarts = 15
        max_iter = 2000
        step0 = 0.5
        T0 = 0.1
        top_k = 10
        
        # Try different initializations
        initializations = []
        
        # Random uniform
        for _ in range(10):
            h = np.random.rand(n_points)
            h = project_onto_capped_simplex(h, TARGET_SUM)
            initializations.append(h)
        
        # Symmetric random 
        for _ in range(2):
            h = np.random.rand(n_points)
            h = (h + h[::-1]) / 2
            h = project_onto_capped_simplex(h, TARGET_SUM)
            initializations.append(h)
        
        # Triangular shape initial
        x = np.linspace(dx/2, 2 - dx/2, n_points)
        h_tri = np.maximum(0, 1 - np.abs(x - 1))
        h_tri = project_onto_capped_simplex(h_tri, TARGET_SUM)
        initializations.append(h_tri)
        
        # Run restarts
        for i, h_start in enumerate(initializations[:restarts]):
            h, c5 = gradient_descent_restart(
                n_points, TARGET_SUM, max_iter, step0, T0, top_k, sym=False
            )
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Check if we should continue based on time
            if time.time() - start_time > budget_s * 0.9:
                break
                
        remaining_budget = total_budget - (time.time() - start_time)
    
    # Phase 3: Local search polishing
    if best_h is not None and (time.time() - start_time) < budget_s * 0.95:
        remaining_budget = budget_s - (time.time() - start_time)
        if remaining_budget > 5:
            best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_budget)
    
    # Final projection
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, TARGET_SUM)
        final_c5 = evaluate_c5_fast(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback: constant function
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
