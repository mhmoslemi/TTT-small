import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return np.max(corr)

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return corr

def compute_gradient(corr, h, top_k=10):
    """Compute gradient using top-K shifts"""
    # Get indices of top-K largest correlations
    idx_top = np.argpartition(corr, -top_k)[-top_k:]
    # Compute soft-max weights
    corr_top = corr[idx_top]
    max_corr = corr_top.max()
    exp_vals = np.exp((corr_top - max_corr) / 1e-5)  # temperature = 1e-5
    weights = exp_vals / exp_vals.sum()
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i, (s, w) in enumerate(zip(idx_top, weights)):
        if s >= 0:  # Positive shift
            # grad[j] += w * (1 - h[j+s]) for j in range(N-s)
            if s < len(h):
                grad[:len(h)-s] += w * (1 - h[s:])
            # grad[j] -= w * h[j-s] for j in range(s, N)  
            if s > 0:
                grad[s:] -= w * h[:len(h)-s]
        else:  # Negative shift
            s_abs = -s
            # grad[j] += w * (1 - h[j-s]) for j in range(s_abs, N)
            if s_abs < len(h):
                grad[s_abs:] += w * (1 - h[:len(h)-s_abs])
            # grad[j] -= w * h[j+s_abs] for j in range(N-s_abs)
            if s_abs > 0:
                grad[:len(h)-s_abs] -= w * h[s_abs:]
    
    return grad

def optimize_one_restart(h0, n_points, max_iter, T0=0.1, alpha0=0.6, beta=0.9, K=10):
    """Optimize one restart using projected gradient descent"""
    h = h0.copy()
    T = T0
    alpha = alpha0
    v = np.zeros_like(h)
    best_c5 = float('inf')
    best_h = h.copy()
    
    for t in range(max_iter):
        # Compute correlation
        corr = compute_correlation(h)
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using top-K shifts
        grad = compute_gradient(corr, h, K)
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        step = alpha / np.sqrt(t + 1)
        
        # Adaptive step size
        step = min(step, 0.5 / (np.abs(grad).max() + 1e-10))
        
        # Update
        h = h - step * v
        
        # Project
        h = project_to_sum_box(h, n_points / 2)
        
        # Annealing 
        T *= 0.9995
        alpha *= 0.999
        
        # Occasionally evaluate exact C5
        if t % 20 == 0:
            corr = compute_correlation(h)
            c5 = np.max(corr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

        x1 = min(x0 + 1, n_coarse - 1)

            h_fine[i] = coarse_h[x0]
        else:

            h_fine[i] = coarse_h[x0] * (1 - t) + coarse_h[x1] * t
    
    h_fine = project_to_sum_box(h_fine, target_sum)
    
    # Global optimization with multiple restarts
    best_h = h_fine.copy()
    best_c5 = evaluate_c5(h_fine)
    
    # Multiple restarts
    restarts = 12
    for restart in range(restarts):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Random initialization
        h0 = np.random.rand(n_points)
        h0 = project_to_sum_box(h0, target_sum)
        
        h_opt, c5_opt = optimize_one_restart(h0, n_points, 3000, T0=0.1, alpha0=0.6, K=10)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        
        # Symmetric restart
        h0_sym = (h0 + h0[::-1]) / 2
        h0_sym = project_to_sum_box(h0_sym, target_sum)
        
        h_opt_sym, c5_opt_sym = optimize_one_restart(h0_sym, n_points, 3000, T0=0.1, alpha0=0.6, K=10)
        
        if c5_opt_sym < best_c5:
            best_c5 = c5_opt_sym
            best_h = h_opt_sym.copy()
    
    # Local polishing with pair-swap
    current_h = best_h.copy()
    current_c5 = evaluate_c5(current_h)
    best_polished_h = current_h.copy()
    best_polished_c5 = current_c5
    
    t_remaining = budget_s * 0.3
    t_start_local = time.time()
    
    while time.time() - t_start_local < t_remaining:
        # Pair-swap local search
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
                
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Feasible interval for transfer
            delta_low = max(-current_h[i], current_h[j] - 1)
            delta_high = min(1 - current_h[i], current_h[j])
            
            if delta_low >= delta_high:
                continue
                
            delta = np.random.uniform(delta_low, delta_high)
            
            h_new = current_h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            
            # Project to feasible region
            h_new = project_to_sum_box(h_new, target_sum)
            
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_polished_c5:
                    best_polished_c5 = c5_new
                    best_polished_h = h_new.copy()
            else:
                # Accept with probability
                temperature = max(0.001, 0.1 * (1 - (time.time() - t_start_local) / t_remaining))
                if np.random.rand() < np.exp(-(c5_new - current_c5) / temperature):
                    current_h = h_new
                    current_c5 = c5_new
    
    # Final projection and evaluation
    final_h = project_to_sum_box(best_polished_h, target_sum)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
