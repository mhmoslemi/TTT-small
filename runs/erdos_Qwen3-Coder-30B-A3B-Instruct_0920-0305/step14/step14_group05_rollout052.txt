import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 using FFT"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old_h to new_n points"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def worst_shift(c):
    """Get the shift index that maximizes correlation"""
    idx = np.argmax(c)
    s = idx - (len(c) // 2)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[s:] = h[:n-s]
        h_minus[:s] = h[s:]
    else:
        h_plus[:s] = h[-s:]
        h_minus[-s:] = h[:s]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        # Initialize
        nonlocal TARGET_SUM
        TARGET_SUM = target_sum(n)
        
        # Initial vector
        if n == 80:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, TARGET_SUM)
        else:

            h = project_to_sum_box(h, TARGET_SUM)
        
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Simulated Annealing parameters
        T = 0.02
        alpha = 0.9997
        delta = 0.1
        beta = 0.9999
        max_iter = int(100000 * time_fraction / budget_s)
        
        # Acceptance tracking
        accepted_moves = 0
        total_moves = 0
        
        # Annealing loop
        for i in range(max_iter):
            if time.time() - start_time > budget_s * time_fraction:
                break
                
            total_moves += 1
            
            # Try targeted move (30% of the time)
            if random.random() < 0.3:
                # Calculate current correlation and find worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                
                # Compute desirability for each position based on worst shift
                h_plus, h_minus = shift_arrays(h, s)
                desirability = h_plus + h_minus
                
                # Find indices with low desirability (bottom 20%)
                n_low = max(1, int(0.2 * n))
                low_indices = np.argpartition(desirability, n_low)[:n_low]
                
                # Find indices with high desirability (top 20%)
                high_indices = np.argpartition(-desirability, n_low)[:n_low]
                
                # Select randomly from low and high
                i_low = random.choice(low_indices)
                i_high = random.choice(high_indices)
                
            else:
                # Pure random move
                i_low = random.randint(0, n - 1)
                i_high = random.randint(0, n - 1)
                while i_low == i_high:
                    i_high = random.randint(0, n - 1)
            
            # Compute transfer amount
            max_transfer = min(delta, h[i_low], 1 - h[i_high])
            if max_transfer <= 0:
                continue
                
            transfer_amount = random.uniform(0, max_transfer)
            
            # Propose move
            h_new = h.copy()
            h_new[i_low] -= transfer_amount
            h_new[i_high] += transfer_amount
            
            # Ensure the result is within [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project to sum constraint
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c_new = C5_value(h_new)
            c_old = C5_value(h)
            
            # Accept or reject
            if c_new <= c_old:
                h = h_new
                if c_new < best_C_stage:
                    best_h_stage = h_new.copy()
                    best_C_stage = c_new
                accepted_moves += 1
            else:
                # Accept with probability
                prob = np.exp(-(c_new - c_old) / T)
                if random.random() < prob:
                    h = h_new
                    if c_new < best_C_stage:
                        best_h_stage = h_new.copy()
                        best_C_stage = c_new
                    accepted_moves += 1
            
            # Update temperature and delta
            T = alpha * T
            delta = beta * delta

        eta0 = 0.5
        max_polish = min(2000, int(10000 * time_fraction / budget_s))

            if time.time() - start_time > budget_s * time_fraction:
                break
                
            c = corr_fft(h)
            max_val = np.max(c)
            epsilon = 1e-12
            
            # Find all shifts that achieve the maximum
            s_indices = np.where(np.abs(c - max_val) < epsilon)[0]
            
            # Compute gradient for all worst shifts
            g = np.zeros(len(h))
            for s_idx in s_indices:
                s = s_idx - (len(c) // 2)
                g += gradient_worst(h, s)
            
            g = g / len(s_indices)
            
            # Update step size
            eta = eta0 / np.sqrt(k + 1)
            
            # Update h
            h_new = h - eta * g
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c_new = C5_value(h_new)
            if c_new < best_C_stage:
                best_h_stage = h_new.copy()
                best_C_stage = c_new
                
            h = h_new
            
        return best_h_stage, best_C_stage
    
    # Run stages
    best_h_stage1, best_C_stage1 = make_stage(80, 0.30)
    best_h_stage2, best_C_stage2 = make_stage(160, 0.35)
    best_h_stage3, best_C_stage3 = make_stage(240, 0.35)
    
    # Return the best one
    if best_C_stage1 <= best_C_stage2 and best_C_stage1 <= best_C_stage3:
        return (best_h_stage1.tolist(), float(best_C_stage1), len(best_h_stage1))
    elif best_C_stage2 <= best_C_stage3:
        return (best_h_stage2.tolist(), float(best_C_stage2), len(best_h_stage2))
    else:
        return (best_h_stage3.tolist(), float(best_C_stage3), len(best_h_stage3))
