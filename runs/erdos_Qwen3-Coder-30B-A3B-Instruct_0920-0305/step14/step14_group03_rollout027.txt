import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = np.real(ifft(fft(h_rev, L) * fft(h, L)))
    return corr

def compute_softmax_gradient(h, corr, alpha=8.0):
    """Compute gradient of softmax approximation to max overlap"""
    n = len(h)
    # Compute softmax weights
    max_corr = np.max(corr)
    if max_corr > 0:
        weights = np.exp(alpha * (corr - max_corr))
    else:
        weights = np.exp(alpha * corr)
    weights = weights / np.sum(weights)
    
    # Compute gradient using vectorized operations
    grad = np.zeros(n)
    
    # Process all shifts
    for shift in range(-n+1, n):
        if shift == 0:
            grad += -2 * weights[shift + n - 1] * h
        elif shift > 0:
            # Forward part
            if shift < n:
                grad[:n-shift] += -weights[shift + n - 1] * h[shift:]
            # Backward part
            if shift < n:
                grad[shift:] += -weights[shift + n - 1] * h[:n-shift]
        else:  # shift < 0
            # Forward part - equivalent to shift > 0 case with |shift|
            abs_shift = -shift
            if abs_shift < n:
                grad[:n-abs_shift] += -weights[-shift + n - 1] * h[abs_shift:]
            # Backward part
            if abs_shift < n:
                grad[abs_shift:] += -weights[-shift + n - 1] * h[:n-abs_shift]
    
    return grad * 2.0 / n  # Scale by dx = 2/n
