import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation(h)
    return np.max(corr)

def smooth_max_correlation(h, T=1e-1):
    """Compute smooth approximation to max correlation"""
    corr = compute_correlation(h)
    max_corr = corr.max()
    exp_corr = np.exp((corr - max_corr) / T)
    return max_corr + T * np.log(exp_corr.sum())

def compute_gradient(h, corr, top_k=10):
    """Compute gradient using top-K shifts for efficiency"""
    # Find top-K shifts
    top_indices = np.argpartition(corr, -top_k)[-top_k:]
    
    # Compute weighted sum of gradients
    grad = np.zeros_like(h)
    for s in top_indices:
        if s >= 0:
            # Forward shift: h[i] contributes to corr_s when i+s is valid
            if s < len(h):
                grad[:len(h)-s] += 1 - h[s:s+len(h)-s]
                grad[s:s+len(h)-s] -= h[:len(h)-s]
        else:
            # Backward shift: h[i] contributes to corr_s when i-s is valid
            s_abs = -s
            if s_abs < len(h):
                grad[s_abs:] += 1 - h[:len(h)-s_abs]
                grad[:len(h)-s_abs] -= h[s_abs:]
    
    return grad

def project_onto_capped_simplex(x, target_sum):
    """Project onto the capped simplex with sum constraint"""
    # Box projection
    x = np.clip(x, 0, 1)
    
    # Sum correction using bisection
    def sum_with_lambda(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while sum_with_lambda(a) * sum_with_lambda(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < tol:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 120
    dx = 2.0 / n_points

    coarse_n = 60

    coarse_h = np.zeros(coarse_n)
    # Initialize with triangular function
    x_coarse = np.linspace(0, 2, coarse_n, endpoint=False)
    coarse_h = np.maximum(0, 1 - np.abs(x_coarse - 1))
    coarse_h = project_onto_capped_simplex(coarse_h, target_sum)

    T = 0.1

    best_c5_coarse = float('inf')
    best_h_coarse = coarse_h.copy()

    for iter_num in range(max_iter_coarse):
        if time.time() - start_time > 0.7 * budget_s:
            break

        top_indices = np.argpartition(corr, -K)[-K:]

        grad = np.zeros_like(coarse_h)
        for s in top_indices:
            if s >= 0:
                if s < len(coarse_h):
                    grad[:len(coarse_h)-s] += 1 - coarse_h[s:s+len(coarse_h)-s]
                    grad[s:s+len(coarse_h)-s] -= coarse_h[:len(coarse_h)-s]
            else:

                    grad[s_abs:] += 1 - coarse_h[:len(coarse_h)-s_abs]
                    grad[:len(coarse_h)-s_abs] -= coarse_h[s_abs:]
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        step_size = alpha / np.sqrt(iter_num + 1)
        
        coarse_h = coarse_h - step_size * v
        
        # Projection
        coarse_h = project_onto_capped_simplex(coarse_h, target_sum)

        T = max(T_final, T * tau_T)
        alpha = max(alpha_final, alpha * tau_alpha)
    
    # Upsample coarse solution to fine grid
    fine_h = np.interp(np.linspace(0, 2, n_points, endpoint=False),

    max_swap_iter = 100000

    # Final projection

    final_c5 = evaluate_c5(final_h)
