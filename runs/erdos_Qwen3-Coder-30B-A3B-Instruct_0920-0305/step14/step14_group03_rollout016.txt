import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_softmax_weights(corr, alpha):
    """Compute softmax weights for correlation values"""
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    return w / np.sum(w)

def compute_gradient(h, corr, weights):
    """Compute gradient using exact softmax weights"""
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each shift, accumulate the contribution
    for s in range(-n+1, n):
        if s == 0:
            # Diagonal contribution
            grad += -2 * weights[s + n - 1] * h
        elif s > 0:
            # Forward shift
            grad[:n-s] += -weights[s + n - 1] * h[s:]
            # Backward shift  
            grad[s:] += -weights[s + n - 1] * h[:n-s]
        else:
            # Backward shift (s < 0)
            s_abs = -s
            grad[:n-s_abs] += -weights[-s + n - 1] * h[s_abs:]
            # Forward shift
            grad[s_abs:] += -weights[-s + n - 1] * h[:n-s_abs]
    
    return grad

def interpolate_vector(h_coarse, n_fine):
    """Linearly interpolate coarse vector to finer grid"""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
