import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_raw_corr(h):
    """Compute raw correlation using FFT (without dx)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weights(corr_raw, alpha):
    """Compute softmax weights for correlation values"""
    max_c = np.max(corr_raw)
    exps = np.exp(alpha * (corr_raw - max_c))
    return exps / np.sum(exps)

def compute_ascend_weighted(h, weights):
    """Compute weighted ascend direction for gradient"""
    n = len(h)
    ascend = np.zeros_like(h)
    
    for idx, w_s in enumerate(weights):
        if w_s == 0:
            continue
        shift = idx - (n-1)
        if shift == 0:
            ascend += 2 * w_s * h
        elif shift > 0:
            ascend[:n-shift] += w_s * h[shift:]
            ascend[shift:] += w_s * h[:n-shift]
        else:
            s = -shift
            ascend[s:] += w_s * h[:n-s]
            ascend[:n-s] += w_s * h[s:]
    
    return ascend

def gradient_step(h, alpha, step):
    """Perform one gradient step"""
    corr_raw = compute_raw_corr(h)
    weights = compute_weights(corr_raw, alpha)
    ascend = compute_ascend_weighted(h, weights)
    
    # Normalize ascend to control step size
    ascend_norm = ascend / (np.max(np.abs(ascend)) + 1e-12)
    
    h_new = h + step * ascend_norm
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    # Enforce symmetry
    h_new = (h_new + h_new[::-1]) / 2
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Predefine initial constructions
    init_constructions = []
    
    # Triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    tri = np.minimum(x, 2 - x)
    init_constructions.append(tri)
    
    # Random uniform continuous
    for _ in range(2):
        rand = np.random.rand(n_points)
        rand_proj = project_to_sum_box(rand, TARGET_SUM)
        init_constructions.append(rand_proj)
    
    # Random binary
    for _ in range(2):
        idx = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        bin_vec = np.zeros(n_points)
        bin_vec[idx] = 1.0
        init_constructions.append(bin_vec)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            # Resample by interpolation to match n_points
            old_x = np.linspace(0, 2, len(h0), endpoint=False) + 1.0/len(h0)
            new_x = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
            h_resampled = np.interp(new_x, old_x, h0, left=0, right=0)
            h_resampled = project_to_sum_box(h_resampled, TARGET_SUM)
            init_constructions.append(h_resampled)
        else:
            init_constructions.append(project_to_sum_box(h0, TARGET_SUM))
    
    # Run optimizations for each initial construction
    for init_h in init_constructions:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        h = init_h.copy()
        current_c5 = evaluate_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent optimization
        alpha = 1.0
        step = 0.3
        max_iter = 5000
        patience = 0
        
        for it in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            h_new = gradient_step(h, alpha, step)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
            # Reduce step size if no improvement
            if patience > 20:
                step *= 0.5
                patience = 0
            
            # Increase alpha gradually
            if it % 50 == 0 and alpha < 200:
                alpha *= 1.05
        
        # Local refinement with simulated annealing
        h = best_h.copy()
        temp = 0.1
        for _ in range(2000):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Choose two random indices
            i, j = np.random.choice(n_points, 2, replace=False)
            delta = np.random.normal(0, 0.02)
            
            # Propose move
            h_new = h.copy()
            if h[i] + delta <= 1 and h[j] - delta >= 0:
                h_new[i] = np.clip(h[i] + delta, 0, 1)
                h_new[j] = np.clip(h[j] - delta, 0, 1)
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                
                c5_new = evaluate_c5(h_new)
                if c5_new < current_c5:
                    h = h_new
                    current_c5 = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    # Accept with probability
                    if np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                        h = h_new
                        current_c5 = c5_new
            else:
                # For infeasible move, try small delta that keeps boundaries
                delta = min(delta, 1 - h[i]) if delta > 0 else max(delta, -h[j])
                if abs(delta) > 1e-6:
                    h_new[i] = np.clip(h[i] + delta, 0, 1)
                    h_new[j] = np.clip(h[j] - delta, 0, 1)
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    
                    c5_new = evaluate_c5(h_new)
                    if c5_new < current_c5:
                        h = h_new
                        current_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
            
            temp *= 0.9995
    
    if best_h is None:
        # Fallback to initial_h_values if available
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values)
            if len(h0) != n_points:
                old_x = np.linspace(0, 2, len(h0), endpoint=False) + 1.0/len(h0)
                new_x = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
                h_resampled = np.interp(new_x, old_x, h0, left=0, right=0)
                best_h = project_to_sum_box(h_resampled, TARGET_SUM)
            else:
                best_h = project_to_sum_box(h0, TARGET_SUM)
        else:
            # Default initialization
            x = (np.arange(n_points) + 0.5) * dx
            best_h = np.minimum(x, 2 - x)
            best_h = project_to_sum_box(best_h, TARGET_SUM)
        
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
