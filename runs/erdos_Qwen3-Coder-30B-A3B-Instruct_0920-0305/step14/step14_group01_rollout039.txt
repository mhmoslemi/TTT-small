```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling approach
    x = np.array(x)
    sorted_x = np.sort(x)[::-1]
    
    # Find the threshold
    cumsum = np.cumsum(sorted_x)
    n_above = np.searchsorted(cumsum, target_sum, side='right')
    if n_above == 0:
        threshold = sorted_x[0]  # All elements are above the threshold
    else:
        threshold = (target_sum - cumsum[n_above-1]) / n_above
        # Ensure threshold is not negative
        threshold = max(0, threshold)
    
    # Apply the threshold
    result = np.clip(x, 0, threshold)
    
    # Adjust sum to exactly match target
    current_sum = np.sum(result)
    if current_sum > target_sum:
        # Reduce higher values
        diff = current_sum - target_sum
        indices = np.argsort(-result)
        for i in indices:
            if diff <= 0:
                break
            reduce = min(result[i], diff)
            result[i] -= reduce
            diff -= reduce
    elif current_sum < target_sum:
        # Increase lower values
        diff = target_sum - current_sum
        indices = np.argsort(result)
        for i in indices:
            if diff <= 0:
                break
            increase = min(1 - result[i], diff)
            result[i] += increase
            diff -= increase
    
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon):
    """Compute gradient of the smooth max approximation"""
    corr = compute_correlation(h)
    
    # Compute weights
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradient using convolution via FFT
    # The gradient of I_j = dx * corr_j is a weighted sum of shift gradients
    # Need to compute conv(w, 1-h) and conv(w_rev, h)
    b = 1 - h
    L = 2 * len(h) - 1
    
    # Convolution of w with b
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # Convolution of w_rev with h
    w_rev = w[::-1]
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Gradient is conv1 - conv2 (central part)
    grad = conv1 - conv2
    grad = grad[len(h)-1:2*len(h)-2]
    
    return dx * grad

def smooth_max_objective_and_gradient(u, epsilon):
    """Compute smooth max objective and its gradient"""
    # Reconstruct full h from symmetric u
    h = np.concatenate([u, u[::-1]])
    
    # Compute correlation and smooth max
    corr = compute_correlation(h)
    smooth_max = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    
    # Gradient of smooth max
    grad = compute_smooth_max_gradient(h, epsilon)
    
    # Add penalty for constraints
    # Monotonicity penalty (increasing part)
    penalty_grad = np.zeros_like(u)
    for i in range(1, len(u)):
        if u[i-1] - u[i] > 0:
            penalty_grad[i-1] += 2 * 1e3 * (u[i-1] - u[i])
            penalty_grad[i] -= 2 * 1e3 * (u[i-1] - u[i])
    
    # Sum constraint penalty
    sum_diff = np.sum(u) - target_sum
    penalty_grad += 2 * 1e4 * sum_diff
    
    # Return function value and gradient
    return smooth_max, penalty_grad

def generate_initial_seeds():
    """Generate different initial seeds for restarts"""
    seeds = []
    
    # Uniform
    seeds.append(np.full(n_points // 2, 0.5))
    
    # Triangular
    x = np.linspace(0, 1, n_points // 2)
    triangular = np.minimum(x, 1 - x)
    seeds.append(triangular)
    
    # Cosine squared
    x = np.linspace(0, 2, n_points)
    cosine_squared = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cosine_squared[:n_points // 2])
    
    # Random monotone
    random_vals = np.sort(np.random.rand(n_points // 2))
    seeds.append(random_vals)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial seeds
    seeds = generate_initial_seeds()
    
    # Time management - reserve 5% for final refinement
    max_time = start_time + 0.95 * budget_s
    
    for seed_idx, seed_u in enumerate(seeds):
        current_u = seed_u.copy()
        
        # Continuation on epsilon
        for epsilon in epsilon_schedule:
            if time.time() > max_time:
                break
                
            # Use scipy's L-BFGS-B optimizer
            result = minimize(
                lambda u: smooth_max_objective_and_gradient(u, epsilon)[0],
                current_u,
                method='L-BFGS-B',
                bounds=[(0, 1)] * len(current_u),
                jac=lambda u: smooth_max_objective_and_gradient(u, epsilon)[1],
                options={'maxiter': 250, 'disp': False}
            )
            
            current_u = result.x
            
            # Project to feasible region
            current_u = project_to_box_sum(current_u, target_sum)
            
            # Reconstruct full h
            h = np.concatenate([current_u, current_u[::-1]])
            
            # Evaluate true C5
            corr = compute_correlation(h)
            c5_val = np.max(corr) * dx
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        
        if time.time() > max_time:
            break
    
    # Final refinement with exact max
    if best_h is not None:
        h = best_h.copy()
        t_final = start_time + budget_s
        
        # Subgradient descent for final refinement
        lr0 = 0.1
        for i in range(10000):
            if time.time() >= t_final:
                break
                
            # Compute full correlation
            corr = compute_correlation(h)
            worst_shift = np.argmax(corr)
            
            # Compute subgradient for worst shift
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + worst_shift < len(h):
                    grad[i] += (1 - h[i + worst_shift])
                if i - worst_shift >= 0:
                    grad[i] -= h[i - worst_shift]
            
            # Update with step size
            lr = lr0 / np.sqrt(i + 1)
            h_new = h - lr * grad
            h_new = project_to_box_sum(h_new, target_sum)
            
            # Evaluate new C5
            corr_new = compute_correlation(h_new)
            c5_new = np.max(corr_new) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new.copy()
                best_h = h.copy()
    
    # If we didn't find a solution, use the initial seed
    if best_h is None:
        # Reconstruct from first seed with symmetry
        u0 = seeds[0]
        h0 = np.concatenate([u0, u0[::-1]])
        best_h = h0.copy()
        corr = compute_correlation(h0)
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), float(best_c5), n_points)
```
