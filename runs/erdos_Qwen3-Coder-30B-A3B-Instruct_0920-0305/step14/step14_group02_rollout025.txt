# Global variables
TARGET_SUM = None

dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation using full correlation"""
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def compute_gradient_F_T(h, T=0.05, K=12):
    """Compute gradient of log-sum-exp surrogate"""
    # Compute all correlations
    corr = np.correlate(h, 1-h, mode="full")
    # Keep only top K shifts
    top_k_indices = np.argsort(corr)[-K:][::-1]
    top_corr = corr[top_k_indices]
    
    # Compute softmax weights
    exp_corr = np.exp(top_corr / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i, s in enumerate(top_k_indices):
        # Compute gradient for this shift
        shift_grad = np.zeros_like(h)
        if s >= 0:
            for j in range(len(h) - s):
                shift_grad[j] += 1 - h[j + s]
                shift_grad[j + s] -= h[j]
        else:
            for j in range(-s, len(h)):
                shift_grad[j] += 1 - h[j + s]
                shift_grad[j + s] -= h[j]
        grad += weights[i] * shift_grad
    
    return dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling approach
    x = np.array(x)
    x_sorted = np.sort(x)[::-1]
    
    # Find lambda
    cumsum = np.cumsum(x_sorted)
    n = len(x)
    lambdas = (cumsum - target_sum) / np.arange(1, n+1)
    
    # Find the right lambda
    idx = np.argmax(lambdas >= x_sorted)
    if idx == 0 and lambdas[0] >= x_sorted[0]:
        lambda_star = lambdas[0]
    else:
        if idx == 0:
            lambda_star = np.infty
        else:
            lambda_star = lambdas[idx-1]
    
    # Apply projection
    result = np.clip(x - lambda_star, 0, 1)
    return result

def generate_triangular_seed(n):
    """Generate a triangular seed function"""
    x = np.linspace(0, 2, n)
    h = np.maximum(0, 1 - np.abs(x - 1))
    return h
