import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_exact(h):
    """Evaluate C5 using full FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_surrogate(h, T=0.05):
    """Evaluate smooth surrogate of C5 using log-sum-exp"""
    # Compute all correlations using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Apply log-sum-exp smoothing
    corr = corr * dx
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    return max_corr + T * np.log(np.sum(exp_terms))

def compute_gradient(h, T=0.05):
    """Compute gradient of the surrogate function"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    
    # Compute correlations for all shifts
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real * dx
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if 0 <= i <= len(h)-1-s:
                grad[i] += weights[s + len(h) - 1] * (1 - h[i + s])
            if s <= i <= len(h)-1:
                grad[i] -= weights[s + len(h) - 1] * h[i - s]
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def water_filling_projection(h, target_sum):
    """Fast projected onto the simplex using water-filling"""
    # Sort in descending order
    sorted_h = np.sort(h)[::-1]
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_h)
    # Find the threshold
    n = len(h)
    for i in range(n):
        if sorted_h[i] - (cumsum[i] - target_sum) / (i + 1) > 0:
            threshold = sorted_h[i] - (cumsum[i] - target_sum) / (i + 1)
            break
    else:  # All elements are too small
        threshold = sorted_h[-1] - (cumsum[-1] - target_sum) / n
    
    # Apply threshold
    result = np.clip(h - threshold, 0, 1)
    # Adjust to ensure sum constraint
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        # Add difference to smallest elements
        indices = np.argsort(result)
        i = 0
        while diff > 1e-10 and i < len(result):
            if result[indices[i]] < 1:
                add = min(diff, 1 - result[indices[i]])
                result[indices[i]] += add
                diff -= add
            i += 1
    
    return result

def smooth_optimization(h0, max_iter=3000, T0=0.12, decay=0.998, patience=50):
    """Optimize using projected L-BFGS with temperature continuation"""
    h = h0.copy()
    T = T0
    patience_count = 0
    prev_obj = float('inf')
    
    for i in range(max_iter):
        # Compute gradient of surrogate
        grad = compute_gradient(h, T)
        
        # L-BFGS step (simple gradient descent with step size)
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project onto box and sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate objective
        obj = evaluate_c5_surrogate(h_new, T)
        
        # Check improvement
        if obj < prev_obj:
            h = h_new.copy()
            prev_obj = obj
            patience_count = 0
        else:
            patience_count += 1
            if patience_count > patience:
                break
        
        # Temperature decay
        T *= decay
        if T < 1e-6:
            break
    
    return h

def pairwise_exchange_polish(h, max_iter=5000):
    """Exact local polishing using pairwise exchanges"""
    h = h.copy()
    best_c5 = evaluate_c5_exact(h)
    
    for iteration in range(max_iter):
        improved = False
        # Try all pairs in random order
        pairs = [(i, j) for i in range(len(h)) for j in range(len(h)) if i != j]
        np.random.shuffle(pairs)
        
        for i, j in pairs[:100]:  # Limit to 100 random pairs per iteration
            # Compute current correlations
            c = correlate(h, 1 - h, mode='full') * dx
            
            # Compute exact change for pair (i, j) for all possible δ
            # Only consider relevant shifts
            best_delta = None
            best_c5_new = best_c5
            
            # Simple greedy approach: try few step sizes
            for delta in np.linspace(-0.5, 0.5, 11):
                if delta == 0:
                    continue
                    
                # Check bounds
                if h[i] - delta < 0 or h[j] + delta < 0 or h[i] - delta > 1 or h[j] + delta > 1:
                    continue
                    
                # Create new vector
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                
                # Check sum constraint
                if abs(np.sum(h_new) - TARGET_SUM) > 1e-6:
                    continue
                    
                # Evaluate
                c5_new = evaluate_c5_exact(h_new)
                if c5_new < best_c5_new:
                    best_c5_new = c5_new
                    best_delta = delta
            
            if best_delta is not None:
                h[i] -= best_delta
                h[j] += best_delta
                best_c5 = best_c5_new
                improved = True
                
        if not improved:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate coarse initial solution (triangular)
    h_coarse = np.array([max(0, 1 - abs((i + 0.5) * 2 / 60 - 1)) for i in range(60)])
    h_coarse = project_to_sum_box(h_coarse, 30)  # 60/2 = 30
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, 60), h_coarse)
    h_fine = project_to_sum_box(h_fine, TARGET_SUM)
    
    best_h = h_fine.copy()
    best_c5 = evaluate_c5_exact(best_h)
    
    # Multiple restarts
    n_restarts = 12
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Generate different initializations
        if restart == 0:
            h_init = h_fine.copy()  # Starting from interpolated solution
        elif restart % 4 == 0:
            # Random initialization
            h_init = np.random.rand(n_points)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        elif restart % 4 == 1:
            # Symmetric random initialization
            h_init = np.random.rand(n_points)
            h_init = (h_init + h_init[::-1]) / 2
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        elif restart % 4 == 2:
            # Triangular (symmetric)
            h_init = np.array([max(0, 1 - abs((i + 0.5) * 2 / n_points - 1)) for i in range(n_points)])
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        else:  # restart % 4 == 3
            # Slightly perturbed version of previous best
            h_init = best_h + np.random.normal(0, 0.01, n_points)
            h_init = project_to_sum_box(h_init, TARGET_SUM)
        
        # Smooth optimization with temperature continuation
        h_opt = smooth_optimization(h_init.copy(), max_iter=1500, T0=0.12, decay=0.996)
        c5_opt = evaluate_c5_exact(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final polishing with pairwise exchanges
    if time.time() - start_time < budget_s * 0.95:
        best_h = pairwise_exchange_polish(best_h, max_iter=2000)
        best_c5 = evaluate_c5_exact(best_h)
    
    # Final projection
    best_h = water_filling_projection(best_h, TARGET_SUM)
    best_c5 = evaluate_c5_exact(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
