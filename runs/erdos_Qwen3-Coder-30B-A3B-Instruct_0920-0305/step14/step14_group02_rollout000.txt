import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # i from 0 to N-1-s
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s = -s
        # shift negative: i from s to N-1
        g[s:] += 1 - h[:N-s]
        g[:N-s] -= h[s:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max correlation"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert to exponentials
        max_corr = np.max(corr)
        # shift to improve numeric
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N-1)
            # compute grad for shift s
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart"""
    # Random initial h
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    
    h = h0.copy()
    T = T0
    best_c5 = float('inf')
    best_h = h0.copy()
    
    for it in range(max_iter):
        # Check time
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Adaptive step size
        step = step0 / np.sqrt(it + 1)
        
        # Update
        h = h - step * g
        
        # Optional symmetry enforcement
        if sym:
            h = (h + h[::-1]) / 2
            
        # Project
        h = project_onto_capped_simplex(h, target_sum)
        
        # Decay temperature
        T *= 0.9995
        
        # Early termination guard
        if it > 2000 and best_c5 < 0.385:
            break
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local pairwise swap search to refine solution"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute valid delta range
        delta_min = max(-best_h[i], best_h[j] - 1)
        delta_max = min(1 - best_h[i], best_h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Propose update
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Compute new C5
        new_c5 = evaluate_c5_fast(h_new)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Coarse optimization
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    coarse_target_sum = n_coarse / 2
    
    coarse_h, coarse_c5 = gradient_descent_restart(
        n_coarse, coarse_target_sum, 1000, 0.5, 0.1, 5, False
    )
    
    # Upsample to target resolution
    if n_points > n_coarse:
        x_coarse = np.linspace(dx_coarse/2, 2 - dx_coarse/2, n_coarse)
        x_target = np.linspace(dx/2, 2 - dx/2, n_points)
        upsampled_h = np.interp(x_target, x_coarse, coarse_h)
        upsampled_h = project_onto_capped_simplex(upsampled_h, TARGET_SUM)
        coarse_h = upsampled_h
    
    # Multiple restarts with different initializations
    restarts = 15
    remaining_time = budget_s * 0.8
    
    for i in range(restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Randomized restart
        h, c5 = gradient_descent_restart(
            n_points, TARGET_SUM, 2000, 0.5, 0.1, 10, False
        )
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Symmetric restart
        h_sym, c5_sym = gradient_descent_restart(
            n_points, TARGET_SUM, 2000, 0.5, 0.1, 10, True
        )
        
        if c5_sym < best_c5:
            best_c5 = c5_sym
            best_h = h_sym.copy()
            
        # Triangular seed restart
        x = np.linspace(dx/2, 2 - dx/2, n_points)
        triangular_h = np.maximum(0, 1 - np.abs(x-1))
        triangular_h = project_onto_capped_simplex(triangular_h, TARGET_SUM)
        h_tri, c5_tri = gradient_descent_restart(
            n_points, TARGET_SUM, 2000, 0.5, 0.1, 10, False
        )
        
        if c5_tri < best_c5:
            best_c5 = c5_tri
            best_h = h_tri.copy()
    
    # Local refinement with swaps
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 10:
            best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time - 5)
    
    # Ensure final feasibility
    if best_h is not None:
        best_h = project_onto_capped_simplex(best_h, TARGET_SUM)
        final_c5 = evaluate_c5_fast(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    # Fallback
    if best_h is None:
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
