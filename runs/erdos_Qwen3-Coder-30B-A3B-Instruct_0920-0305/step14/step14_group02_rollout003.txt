import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation(h)
    return np.max(corr)

def smooth_max_correlation(h, T=1e-1, K=10):
    """Compute smooth approximation of max correlation"""
    corr = compute_correlation(h)
    # Soft max with temperature T
    corr_centered = corr - np.max(corr)
    exp_corr = np.exp(corr_centered / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Get top K shifts
    idx_top = np.argpartition(corr, -K)[-K:]
    weights_top = weights[idx_top]
    weights_top = weights_top / np.sum(weights_top)
    
    # Compute gradient of smooth max
    grad = np.zeros_like(h)
    for i, s in enumerate(idx_top):
        if s >= 0:
            # Contribution from h_i * (1 - h_{i+s})
            grad[:-s] += weights_top[i] * (1 - h[s:])
            # Contribution from -h_{i-s} * h_i
            grad[s:] -= weights_top[i] * h[:-s]
        else:
            # Contribution from h_i * (1 - h_{i+s})
            grad[-s:] += weights_top[i] * (1 - h[:s])
            # Contribution from -h_{i-s} * h_i
            grad[:s] -= weights_top[i] * h[-s:]
    
    return corr, grad

def project_onto_capped_simplex(h, target_sum):
    """Project onto capped simplex using water filling"""
    # Box clipping
    h = np.clip(h, 0, 1)
    
    # Water filling
    def objective(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while objective(a) * objective(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-10:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 5000
    T0 = 0.1
    alpha0 = 0.6
    K = 10
    beta = 0.9
    tau_T = 0.9995
    tau_alpha = 0.999
    
    # Initialize
    best_h = None
    best_c5 = float('inf')
    
    # Generate initial triangular function
    x = np.linspace(0, 2, n_points)
    h_initial = np.maximum(0, 1 - np.abs(x - 1))
    h_initial = project_to_sum_box(h_initial, target_sum)
    
    # Try multiple restarts
    restarts = 15
    
    for restart in range(restarts):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Random restart
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, target_sum)
        
        # Symmetric restart
        if restart % 3 == 0:
            h = (h_initial + h_initial[::-1]) / 2
            h = project_to_sum_box(h, target_sum)
        
        # Optimisation loop
        T = T0
        alpha = alpha0
        v = np.zeros(n_points)
        
        for iteration in range(max_iter):
            if time.time() - start_time > 0.7 * budget_s:
                break
                
            # Compute correlation and gradient
            corr, grad = smooth_max_correlation(h, T, K)
            
            # Update with momentum
            v = beta * v + (1 - beta) * grad
            step = alpha / np.sqrt(iteration + 1)
            
            # Limit step size to avoid overshooting
            grad_norm = np.linalg.norm(grad, np.inf)
            if grad_norm > 0:
                step = min(step, 0.5 / grad_norm)
            
            h_new = h - step * v
            
            # Project onto feasible set
            h = project_onto_capped_simplex(h_new, target_sum)
            
            # Update temperature and step size
            T *= tau_T
            alpha *= tau_alpha
            
            # Periodic evaluation
            if iteration % 20 == 0:
                current_c5 = evaluate_c5(h) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    # Local combinatorial polishing
    if best_h is not None:
        h = best_h.copy()
        max_swaps = 100000
        temp_start = 0.1
        
        for swap in range(max_swaps):
            if time.time() - start_time > budget_s:
                break
                
            # Pick two random indices
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Compute feasible interval
            delta_low = max(-h[i], h[j] - 1)
            delta_high = min(1 - h[i], h[j])
            
            if delta_low >= delta_high:
                continue
                
            # Make a random feasible move
            delta = np.random.uniform(delta_low, delta_high)
            h[i] += delta
            h[j] -= delta
            
            # Compute new C5
            current_c5 = evaluate_c5(h) * dx
            
            # Accept or reject with simulated annealing
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            else:
                # Accept with probability based on temperature
                temperature = temp_start * (1 - (time.time() - start_time) / budget_s)
                if temperature > 1e-6:
                    if np.random.rand() < np.exp(-(current_c5 - best_c5) / temperature):
                        best_c5 = current_c5
                        best_h = h.copy()
                else:
                    # If temperature is too low, revert the change
                    h[i] -= delta
                    h[j] += delta
    
    # Final projection and evaluation
    if best_h is not None:
        final_h = project_onto_capped_simplex(best_h, target_sum)
        final_c5 = evaluate_c5(final_h) * dx
        return (final_h.tolist(), float(final_c5), n_points)
    else:
        # Fallback to initial solution
        return (h_initial.tolist(), evaluate_c5(h_initial) * dx, n_points)
