def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""

    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

    # Let's get the actual shifts that produce the maximum

    max_shift = max_idx - len(h) + 1
    
    # Gradient computation based on exact shifts
    # For each i, we need to compute how changing h[i] affects the correlation
    # Specifically, above shift contributes d/dh[i] * (1 - h[i+shift])
    
    grad = np.zeros_like(h)
    
    # For each possible shift
    for shift in range(-len(h)+1, len(h)):
        # Find where this shift contributes
        for i in range(len(h)):
            if 0 <= i + shift < len(h):
                # Contribution to correlation at position (i-shift) of shift
                if shift >= 0:
                    # Positive shift: h[i] contributes to corr[i+shift]
                    # We care about (1-h[i+shift])
                    grad[i] -= corr[i+shift] * (1 - h[i+shift])
                else:
                    # Negative shift: h[i-shift] contributes to corr[i]
                    # We care about (1-h[i-shift])  
                    if i - shift < len(h):
                        grad[i] -= corr[i] * (1 - h[i-shift])
    
    return grad
