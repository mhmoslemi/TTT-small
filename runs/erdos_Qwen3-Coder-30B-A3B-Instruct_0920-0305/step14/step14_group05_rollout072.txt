```python
import numpy as np
from scipy.fft import fft, ifft
import time
import random

# Global constants
TARGET_SUM = None
dx = None

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def C5_value(h):
    """Evaluate C5 with proper dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of a coarse solution onto a finer grid"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(c):
    """From a correlation vector return the shift attaining the maximum"""
    idx = np.argmax(c)
    s = idx - (len(c) - 1)
    return s

def shift_arrays(h, s):
    """Returns h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Exact sub-gradient of the worst term C_s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def make_stage(n, time_fraction, best_h_prev, start_time, budget_s):
    """Run a single stage of the optimization"""
    global TARGET_SUM, dx
    TARGET_SUM = target_sum(n)
    dx = dx(n)
    
    # Initialize
    if best_h_prev is None:
        h = np.random.uniform(0, 1, n)
        h = project_to_sum_box(h, TARGET_SUM)
    else:
        h = interpolate(best_h_prev, n)
        h = project_to_sum_box(h, TARGET_SUM)
    
    best_h_stage = h.copy()
    best_C_stage = C5_value(h)
    
    # Simulated annealing parameters
    T = 0.02
    alpha = 0.9997
    beta = 0.9999
    max_iter = int(100000 * time_fraction / 0.3)
    Δ = 0.1
    
    iter_count = 0
    acceptance_count = 0
    
    while time.time() < start_time + budget_s * time_fraction and iter_count < max_iter:
        # Decide move type
        if random.random() < 0.3:
            # Targeted move - improve worst shift
            c = corr_fft(h)
            s = worst_shift(c)
            
            # Compute desirability
            h_plus, h_minus = shift_arrays(h, s)
            d = h_plus + h_minus
            
            # Pick from low and high desirability indices
            sorted_indices = np.argsort(d)
            low_indices = sorted_indices[:max(1, n//5)]
            high_indices = sorted_indices[-max(1, n//5):]
            
            i = random.choice(low_indices)
            j = random.choice(high_indices)
        else:
            # Random move
            i, j = random.sample(range(n), 2)
        
        # Compute transfer amount
        max_delta = min(Δ, h[i], 1 - h[j])
        if max_delta <= 0:
            iter_count += 1
            continue
            
        delta = random.uniform(0, max_delta)
        
        # Propose move
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Clip to box constraints
        h_new[i] = max(0, min(1, h_new[i]))
        h_new[j] = max(0, min(1, h_new[j]))
        
        # Check if sum constraint is satisfied
        current_sum = np.sum(h_new)
        if abs(current_sum - TARGET_SUM) > 1e-10:
            # Re-project if necessary
            h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new state
        c_new = corr_fft(h_new)
        new_C = dx(n) * np.max(c_new)
        current_C = dx(n) * np.max(corr_fft(h))
        
        # Acceptance rule
        if new_C <= current_C or random.random() < np.exp(-(new_C - current_C) / T):
            h = h_new
            if new_C < best_C_stage:
                best_h_stage = h.copy()
                best_C_stage = new_C
            acceptance_count += 1
        
        # Update temperature and step size
        T *= alpha
        Δ *= beta
        
        iter_count += 1
        
        # Check if we should restart due to low acceptance rate
        if iter_count % 1000 == 0 and acceptance_count < 5:
            # Reset with random solution
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, TARGET_SUM)
            acceptance_count = 0
    
    # Deterministic polishing
    h_polish = best_h_stage.copy()
    best_C_polish = best_C_stage
    N_polish = min(2000, int(5000 * time_fraction / 0.3))
    
    for k in range(N_polish):
        if time.time() >= start_time + budget_s:
            break
            
        c = corr_fft(h_polish)
        max_val = np.max(c)
        # Find all shifts achieving max value within tolerance
        epsilon = 1e-12
        s_indices = np.where(np.abs(c - max_val) < epsilon)[0]
        
        if len(s_indices) == 0:
            break
            
        # Compute average gradient from all worst shifts
        g = np.zeros(n)
        for idx in s_indices:
            s = idx - (n - 1)  # Convert index back to shift
            g += gradient_worst(h_polish, s)
        
        g = g / len(s_indices)
        
        # Step size with diminishing learning rate
        eta = 0.5 / np.sqrt(k + 1)
        h_new = h_polish - eta * g
        
        # Project to constraints
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        c_new = corr_fft(h_new)
        new_C = dx(n) * np.max(c_new)
        
        if new_C < best_C_polish:
            best_C_polish = new_C
            h_polish = h_new.copy()
    
    return h_polish, best_C_polish

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Run optimization stages
    best_h = None
    best_C = float('inf')
    
    # Stage 1: coarse grid
    h_stage1, C_stage1 = make_stage(80, 0.30, None, start_time, budget_s)
    if C_stage1 < best_C:
        best_C = C_stage1
        best_h = h_stage1.copy()
    
    # Stage 2: medium grid
    h_stage2, C_stage2 = make_stage(160, 0.35, best_h, start_time, budget_s)
    if C_stage2 < best_C:
        best_C = C_stage2
        best_h = h_stage2.copy()
    
    # Stage 3: fine grid
    h_stage3, C_stage3 = make_stage(240, 0.35, best_h, start_time, budget_s)
    if C_stage3 < best_C:
        best_C = C_stage3
        best_h = h_stage3.copy()
    
    return (best_h.tolist(), float(best_C), len(best_h))
```
