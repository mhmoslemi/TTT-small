import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_simplex(x, target_sum):
    """Project vector x onto the capped simplex with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h, dx):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return dx * np.max(corr)

def compute_gradient(h, corr, top_k_indices, T):
    """Compute gradient using top-K shifts"""
    N = len(h)
    g = np.zeros(N)
    
    # For each top-k shift, compute the gradient contribution
    for s in top_k_indices:
        if s >= 0:
            # Forward shift case
            for i in range(N - s):
                g[i] += (1 - h[i + s]) * np.exp(corr[s] / T)
            for i in range(s, N):
                g[i] -= h[i - s] * np.exp(corr[s] / T)
        else:
            # Backward shift case
            s_neg = -s
            for i in range(N - s_neg):
                g[i] += (1 - h[i + s_neg]) * np.exp(corr[s] / T)
            for i in range(s_neg, N):
                g[i] -= h[i - s_neg] * np.exp(corr[s] / T)
    
    return g

def smooth_max_objective(h, T, top_k_indices):
    """Compute smooth approximation of the max objective"""
    N = len(h)
    corr = np.zeros(2 * N - 1)
    
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    corr_fft = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    
    # Soft max
    corr_values = corr_fft[top_k_indices]
    corr_max = np.max(corr_values)
    exp_corr = np.exp((corr_values - corr_max) / T)
    sum_exp = np.sum(exp_corr)
    
    return T * (np.log(sum_exp) + corr_max / T)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120  # Number of points
    dx = 2.0 / N
    target_sum = N / 2
    
    # Coarse optimization (N = 60)
    N_coarse = 60
    dx_coarse = 2.0 / N_coarse
    coarse_h = np.zeros(N_coarse)
    
    # Triangular seed for coarse optimization
    x_coarse = np.linspace(0, 2, N_coarse)
    coarse_h = np.maximum(0, 1 - np.abs(x_coarse - 1))
    coarse_h = project_to_simplex(coarse_h, target_sum)
    
    # Optimization for coarse resolution
    max_iter_coarse = 2000
    T = 0.1
    alpha = 0.8
    beta = 0.9
    K = 10
    tau_T = 0.9995
    tau_alpha = 0.999
    
    # Momentum
    v = np.zeros(N_coarse)
    
    best_c5_coarse = float('inf')
    best_h_coarse = coarse_h.copy()
    
    for t in range(max_iter_coarse):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute correlation (FFT)
        h_rev = coarse_h[::-1]
        b = 1 - coarse_h
        L = 2 * N_coarse - 1
        corr_fft = irfft(rfft(h_rev, L) * rfft(b, L), L).real
        
        # Get top K indices
        top_k_indices = np.argpartition(corr_fft, -K)[-K:]
        
        # Compute gradient using top K shifts
        g = np.zeros(N_coarse)
        corr_values = corr_fft[top_k_indices]
        corr_max = np.max(corr_values)
        exp_corr = np.exp((corr_values - corr_max) / T)
        exp_corr_sum = np.sum(exp_corr)
        
        for i, s in enumerate(top_k_indices):
            w_s = exp_corr[i] / exp_corr_sum
            if s >= 0:
                # Forward shift case
                for j in range(N_coarse - s):
                    g[j] += w_s * (1 - coarse_h[j + s])
                for j in range(s, N_coarse):
                    g[j] -= w_s * coarse_h[j - s]
            else:
                # Backward shift case
                s_neg = -s
                for j in range(N_coarse - s_neg):
                    g[j] += w_s * (1 - coarse_h[j + s_neg])
                for j in range(s_neg, N_coarse):
                    g[j] -= w_s * coarse_h[j - s_neg]
        
        # Update with momentum
        v = beta * v + (1 - beta) * g
        step_size = alpha / np.sqrt(t + 1)
        coarse_h = coarse_h - step_size * v
        
        # Project to feasible set
        coarse_h = project_to_simplex(coarse_h, target_sum)
        
        # Update temperature and step size
        T *= tau_T
        alpha *= tau_alpha
        
        # Periodic evaluation
        if t % 20 == 0:
            c5 = evaluate_c5(coarse_h, dx_coarse)
            if c5 < best_c5_coarse:
                best_c5_coarse = c5
                best_h_coarse = coarse_h.copy()
    
    # Interpolate to fine grid
    fine_h = np.zeros(N)
    coarse_x = np.linspace(0, 2, N_coarse)
    fine_x = np.linspace(0, 2, N)
    
    # Linear interpolation
    for i in range(N):
        x = fine_x[i]
        if x <= coarse_x[0]:
            fine_h[i] = coarse_h[0]
        elif x >= coarse_x[-1]:
            fine_h[i] = coarse_h[-1]
        else:
            # Find the right segment
            for j in range(N_coarse - 1):
                if coarse_x[j] <= x <= coarse_x[j+1]:
                    # Linear interpolation
                    t_frac = (x - coarse_x[j]) / (coarse_x[j+1] - coarse_x[j])
                    fine_h[i] = coarse_h[j] + t_frac * (coarse_h[j+1] - coarse_h[j])
                    break
    
    # Final projection
    fine_h = project_to_simplex(fine_h, target_sum)
    
    # Multiple restarts with fine resolution
    best_c5_fine = float('inf')
    best_h_fine = fine_h.copy()
    
    # Run multiple restarts
    n_restarts = 15
    max_iter_fine = 3000
    
    for restart in range(n_restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Random restart with Gaussian noise
        h_restart = fine_h + np.random.normal(0, 0.05, N)
        h_restart = project_to_simplex(h_restart, target_sum)
        
        # Optimization for fine resolution
        T = 0.1
        alpha = 0.6
        beta = 0.9
        K = 10
        tau_T = 0.9995
        tau_alpha = 0.999
        
        # Momentum
        v = np.zeros(N)
        
        for t in range(max_iter_fine):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute correlation (FFT)
            h_rev = h_restart[::-1]
            b = 1 - h_restart
            L = 2 * N - 1
            corr_fft = irfft(rfft(h_rev, L) * rfft(b, L), L).real
            
            # Get top K indices
            top_k_indices = np.argpartition(corr_fft, -K)[-K:]
            
            # Compute gradient using top K shifts
            g = np.zeros(N)
            corr_values = corr_fft[top_k_indices]
            corr_max = np.max(corr_values)
            exp_corr = np.exp((corr_values - corr_max) / T)
            exp_corr_sum = np.sum(exp_corr)
            
            for i, s in enumerate(top_k_indices):
                w_s = exp_corr[i] / exp_corr_sum
                if s >= 0:
                    # Forward shift case
                    for j in range(N - s):
                        g[j] += w_s * (1 - h_restart[j + s])
                    for j in range(s, N):
                        g[j] -= w_s * h_restart[j - s]
                else:
                    # Backward shift case
                    s_neg = -s
                    for j in range(N - s_neg):
                        g[j] += w_s * (1 - h_restart[j + s_neg])
                    for j in range(s_neg, N):
                        g[j] -= w_s * h_restart[j - s_neg]
            
            # Update with momentum
            v = beta * v + (1 - beta) * g
            step_size = alpha / np.sqrt(t + 1)
            h_restart = h_restart - step_size * v
            
            # Project to feasible set
            h_restart = project_to_simplex(h_restart, target_sum)
            
            # Update temperature and step size
            T *= tau_T
            alpha *= tau_alpha
            
            # Periodic evaluation
            if t % 20 == 0:
                c5 = evaluate_c5(h_restart, dx)
                if c5 < best_c5_fine:
                    best_c5_fine = c5
                    best_h_fine = h_restart.copy()
    
    # Final polishing with pair-swap
    h_polish = best_h_fine.copy()
    c5_polish = evaluate_c5(h_polish, dx)
    
    max_polish_iter = 100000
    temp = 0.1
    
    for i in range(max_polish_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Select two random indices
        idx1, idx2 = np.random.choice(N, 2, replace=False)
        
        # Compute feasible transfer range
        h1, h2 = h_polish[idx1], h_polish[idx2]
        delta_min = max(-h1, h2 - 1)
        delta_max = min(1 - h1, h2)
        
        if delta_min >= delta_max:
            continue
            
        # Make a random transfer
        delta = np.random.uniform(delta_min, delta_max)
        
        # Apply transfer
        h_proposed = h_polish.copy()
        h_proposed[idx1] += delta
        h_proposed[idx2] -= delta
        
        # Project to feasible set
        h_proposed = project_to_simplex(h_proposed, target_sum)
        
        # Compute new C5
        c5_proposed = evaluate_c5(h_proposed, dx)
        
        # Accept or reject
        if c5_proposed < c5_polish:
            h_polish = h_proposed
            c5_polish = c5_proposed
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_proposed - c5_polish) / temp):
                h_polish = h_proposed
                c5_polish = c5_proposed
                
        # Cooling schedule
        temp *= 0.9999
        
    # Final evaluation
    final_c5 = evaluate_c5(h_polish, dx)
    
    return (h_polish.tolist(), float(final_c5), N)
