import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_raw_corr(h):
    """Compute raw correlation using FFT (without dx factor)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weights(corr_raw, alpha):
    """Compute softmax weights for gradient computation"""
    max_c = np.max(corr_raw)
    exps = np.exp(alpha * (corr_raw - max_c))
    return exps / np.sum(exps)

def compute_ascend_weighted(h, weights):
    """Compute weighted ascend direction for gradient"""
    n = len(h)
    ascend = np.zeros_like(h)
    
    # Process each shift
    for idx, w_s in enumerate(weights):
        if w_s == 0:
            continue
        shift = idx - (n-1)
        
        if shift == 0:
            ascend += 2 * w_s * h
        elif shift > 0:
            # h_{i+s} term: for i = 0..n-1-s, add w_s * h[i+s] to ascend[i]
            ascend[:n-shift] += w_s * h[shift:]
            # h_{i-s} term: for i = s..n-1, add w_s * h[i-s] to ascend[i]
            ascend[shift:] += w_s * h[:n-shift]
        else:  # shift < 0
            s = -shift
            # h_{i+s} term: for i = s..n-1, add w_s * h[i-s] to ascend[i]
            ascend[s:] += w_s * h[:n-s]
            # h_{i-s} term: for i = 0..n-1-s, add w_s * h[i+s] to ascend[i]
            ascend[:n-s] += w_s * h[s:]
    
    return ascend

def gradient_step(h, alpha, step_size):
    """Perform one gradient step"""
    # Compute raw correlation
    corr_raw = compute_raw_corr(h)
    
    # Compute weights
    weights = compute_weights(corr_raw, alpha)
    
    # Compute weighted ascend direction
    ascend = compute_ascend_weighted(h, weights)
    
    # Normalize ascend to avoid large steps
    max_ascend = np.max(np.abs(ascend)) + 1e-12
    ascend_norm = ascend / max_ascend
    
    # Update h
    h_new = h + step_size * ascend_norm
    
    # Project to feasible set
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    # Enforce symmetry
    h_new = (h_new + h_new[::-1]) / 2
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Predefine initial constructions
    init_constructions = []
    
    # Triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    tri = np.minimum(x, 2 - x)
    init_constructions.append(tri)
    
    # Random uniform continuous
    for _ in range(2):
        rand = np.random.rand(n_points)
        rand_proj = project_to_sum_box(rand, TARGET_SUM)
        init_constructions.append(rand_proj)
    
    # Random binary
    for _ in range(2):
        idx = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        bin_vec = np.zeros(n_points)
        bin_vec[idx] = 1.0
        init_constructions.append(bin_vec)
    
    # Include initial_h_values if available and matching size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) == n_points:
            init_constructions.append(project_to_sum_box(h0, TARGET_SUM))
    
    # Run optimization for each initial construction
    for init_h in init_constructions:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Copy initial h
        h = init_h.copy()
        current_c5 = evaluate_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent loop
        alpha = 1.0
        step_size = 0.3
        max_iter = 5000
        patience = 0
        
        for iter_num in range(max_iter):
            if time.time() - start_time > 0.90 * budget_s:
                break
                
            # Perform gradient step
            h_new = gradient_step(h, alpha, step_size)
            c5_new = evaluate_c5(h_new)
            
            # Update if improved
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
            # Adapt parameters
            if patience > 100:
                step_size *= 0.9
                patience = 0
                
            if iter_num % 50 == 0 and alpha < 100:
                alpha *= 1.02
                
            # Occasionally perform additional refinement
            if iter_num % 200 == 0:
                # Local search with small perturbations
                h_local = h.copy()
                for _ in range(50):
                    if time.time() - start_time > 0.90 * budget_s:
                        break
                    idx1 = np.random.randint(n_points)
                    idx2 = np.random.randint(n_points)
                    delta = np.random.normal(0, 0.01)
                    h_local[idx1] = np.clip(h_local[idx1] + delta, 0, 1)
                    h_local[idx2] = np.clip(h_local[idx2] - delta, 0, 1)
                    h_local = project_to_sum_box(h_local, TARGET_SUM)
                    c5_local = evaluate_c5(h_local)
                    if c5_local < current_c5:
                        h = h_local
                        current_c5 = c5_local
                        if c5_local < best_c5:
                            best_c5 = c5_local
                            best_h = h.copy()
    
    # Final refinement
    if best_h is not None:
        h_final = best_h.copy()
        t_final = start_time + budget_s - 5
        for i in range(1000):
            if time.time() >= t_final:
                break
                
            # Local perturbation
            h_pert = h_final.copy()
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points) 
            delta = np.random.normal(0, 0.01)
            h_pert[idx1] = np.clip(h_pert[idx1] + delta, 0, 1)
            h_pert[idx2] = np.clip(h_pert[idx2] - delta, 0, 1)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < evaluate_c5(h_final):
                h_final = h_pert
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_final.copy()
    
    if best_h is None:
        # Fallback to triangular shape if no improvement
        x = (np.arange(n_points) + 0.5) * dx
        best_h = np.minimum(x, 2 - x)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
