import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_raw_corr(h):
    """Compute raw correlation without dx factor"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weights(corr_raw, alpha):
    """Compute softmax weights for gradient"""
    max_c = np.max(corr_raw)
    exps = np.exp(alpha * (corr_raw - max_c))
    return exps / np.sum(exps)

def compute_ascend_weighted(h, weights):
    """Compute weighted ascend direction"""
    n = len(h)
    ascend = np.zeros_like(h)
    
    # Process each shift
    for idx, w_s in enumerate(weights):
        if w_s == 0:
            continue
        shift = idx - (n - 1)
        
        if shift == 0:
            ascend += 2 * w_s * h
        elif shift > 0:
            # forward term: h[i] contributes to ascend[i-shift]
            if shift < n:
                ascend[shift:] += w_s * h[:-shift]
            # backward term: h[i] contributes to ascend[i+shift]  
            if shift < n:
                ascend[:-shift] += w_s * h[shift:]
        else:  # shift < 0
            s = -shift
            # forward term
            if s < n:
                ascend[s:] += w_s * h[:-s]
            # backward term
            if s < n:
                ascend[:-s] += w_s * h[s:]
                
    return ascend

def gradient_step(h, alpha, step):
    """Perform one gradient step"""
    corr_raw = compute_raw_corr(h)
    weights = compute_weights(corr_raw, alpha)
    ascend = compute_ascend_weighted(h, weights)
    
    # Normalize ascend to prevent large steps
    ascend_norm = ascend / (np.max(np.abs(ascend)) + 1e-12)
    
    h_new = h + step * ascend_norm
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    # Enforce symmetry
    h_new = (h_new + h_new[::-1]) / 2
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Predefine initial constructions
    init_constructions = []
    
    # triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    tri = np.minimum(x, 2 - x)
    init_constructions.append(tri)
    
    # random uniform continuous
    for _ in range(2):
        rand = np.random.rand(n_points)
        rand_proj = project_to_sum_box(rand, TARGET_SUM)
        init_constructions.append(rand_proj)
    
    # random binary
    for _ in range(2):
        idx = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        bin_vec = np.zeros(n_points)
        bin_vec[idx] = 1.0
        init_constructions.append(bin_vec)
    
    # Include initial_h_values if available and matches dimensions
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h0 = np.array(initial_h_values)
        init_constructions.append(project_to_sum_box(h0, TARGET_SUM))
    elif 'initial_h_values' in globals():
        # Resample if needed
        old_x = np.linspace(0, 2, len(initial_h_values), endpoint=False) + 1.0/len(initial_h_values)
        new_x = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
        h_resampled = np.interp(new_x, old_x, initial_h_values, left=0, right=0)
        init_constructions.append(project_to_sum_box(h_resampled, TARGET_SUM))
    
    # Main optimization loop
    for init_h in init_constructions:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Copy initial solution
        h = init_h.copy()
        current_c5 = evaluate_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent loop
        alpha = 1.0
        step = 0.3
        max_iter = 5000
        patience = 0
        
        for iter_num in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            h_new = gradient_step(h, alpha, step)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                patience = 0
            else:
                patience += 1
            
            # Adjust step size and alpha
            if patience > 50:
                step *= 0.9
                patience = 0
                
            alpha = min(alpha * 1.02, 200)
    
    # Final refinement with local search
    if best_h is not None:
        h = best_h.copy()
        t_final = start_time + budget_s - 5
        for i in range(1000):
            if time.time() >= t_final:
                break
                
            # Local perturbation
            h_pert = h.copy()
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points) 
            delta = np.random.normal(0, 0.01)
            h_pert[idx1] = np.clip(h_pert[idx1] + delta, 0, 1)
            h_pert[idx2] = np.clip(h_pert[idx2] - delta, 0, 1)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < evaluate_c5(h):
                h = h_pert
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h.copy()

    if best_h is None:
        # Fallback to initial construction
        h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
        best_h = h0
        best_c5 = evaluate_c5(h0)
    
    return (best_h.tolist(), float(best_c5), n_points)
