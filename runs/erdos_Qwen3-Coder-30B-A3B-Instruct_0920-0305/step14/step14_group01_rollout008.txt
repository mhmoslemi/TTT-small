import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon):
    """Compute gradient of smooth max using FFT"""
    corr = compute_correlation(h)
    
    # Compute weights
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Convolution for gradient computation
    b = 1 - h
    L = 2 * len(h) - 1
    
    # Compute conv(w, b)
    w_padded = np.pad(w, (0, L - len(w)), mode='constant')
    b_padded = np.pad(b, (0, L - len(b)), mode='constant')
    conv1 = np.fft.ifft(np.fft.fft(w_padded, L) * np.fft.fft(b_padded, L)).real
    
    # Compute conv(w_rev, h)
    w_rev = w[::-1]
    w_rev_padded = np.pad(w_rev, (0, L - len(w_rev)), mode='constant')
    h_padded = np.pad(h, (0, L - len(h)), mode='constant')
    conv2 = np.fft.ifft(np.fft.fft(w_rev_padded, L) * np.fft.fft(h_padded, L)).real
    
    # Extract central part and compute gradient
    grad = conv1[(L-len(h)):L] - conv2[(L-len(h)):L]
    
    return grad * dx

def compute_objective_and_gradient(u, epsilon, lambda_inc=1e3, mu=1e4):
    """Compute objective value and gradient for given u"""
    # Reconstruct full h vector (symmetric)
    h = np.concatenate([u, u[::-1]])
    
    # Apply monotonicity penalty
    penalty = 0
    grad_penalty = np.zeros_like(u)
    
    # Monotonicity penalty for increasing half (u[1:] >= u[:-1])
    for i in range(1, len(u)):
        diff = u[i-1] - u[i]
        if diff > 0:
            penalty += lambda_inc * diff * diff
            grad_penalty[i-1] += 2 * lambda_inc * diff
            grad_penalty[i] -= 2 * lambda_inc * diff
    
    # Ensure sum constraint with penalty term
    sum_diff = np.sum(u) - target_sum
    penalty += mu * sum_diff * sum_diff
    grad_penalty += 2 * mu * sum_diff
    
    # Compute smooth max gradient
    full_h = np.concatenate([u, u[::-1]])
    grad_smooth = compute_smooth_max_gradient(full_h, epsilon)
    
    # Extract gradient for the first half
    grad = grad_smooth[:len(u)] + grad_penalty
    
    # Compute smooth max value
    corr = compute_correlation(full_h)
    smooth_val = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    
    return smooth_val + penalty, grad[:len(u)]

    seeds.append(triangular)
    
    # Cosine squared
    x = np.linspace(0, 2, 240)
    cosine = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cosine[:120])
    
    # Random monotone
    for _ in range(2):
        rand = np.random.rand(120)
        rand_sorted = np.sort(rand)
        seeds.append(rand_sorted)
    
    return seeds
