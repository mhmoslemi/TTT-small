def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    max_overlap = 0
    
    # Compute correlation for all shifts
    for k in range(1, n):
        # For shift k, we compute integral over [0, 2-k*dx] 
        # But we're working with discrete values
        corr = 0
        for i in range(n - k):
            corr += h[i] * (1 - h[i + k])
        max_overlap = max(max_overlap, corr * dx)
    
    return max_overlap
