import numpy as np
from scipy.fft import rfft, irfft
import time

def project_onto_capped_simplex(x, target):
    """Project x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    H = rfft(h_rev, n=L)
    B = rfft(b, n=L)
    corr = irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    dx = 2.0 / len(h)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of corr_s w.r.t. h"""
    n = len(h)
    g = np.zeros_like(h)
    
    if s >= 0:
        # forward contribution: (1 - h_{i+s}) for i = 0..n-1-s
        if s < n:
            g[:n-s] += 1 - h[s:]
        # backward contribution: - h_{i-s} for i = s..n-1  
        if s > 0:
            g[s:] -= h[:n-s]
    else:
        # s < 0, reverse direction
        s_abs = -s
        # forward: (1 - h_{i+s}) for i = s_abs..n-1
        if s_abs < n:
            g[s_abs:] += 1 - h[:n-s_abs]
        # backward: - h_{i-s} for i = 0..n-1-s_abs
        if s_abs > 0:
            g[:n-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max correlation"""
    corr = corr_fft(h)
    
    if T < 1e-12:
        # Just return subgradient of the max shift
        idx = np.argmax(corr)
        shift = idx - (len(h) - 1)
        return subgradient_for_shift(h, shift)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Use only top_k weights
    top_idx = np.argpartition(weights, -top_k)[-top_k:]
    g = np.zeros_like(h)
    n = len(h)
    
    for idx in top_idx:
        w = weights[idx]
        s = idx - (n - 1)
        grad_s = subgradient_for_shift(h, s)
        g += w * grad_s
    
    return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run a single gradient descent optimization with continuation"""
    # Initial random h
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    
    h = h0.copy()
    T = T0
    step = step0
    best_c5 = float('inf')
    best_h = h0.copy()
    
    for t in range(max_iter):
        if t % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update step size
        step = step0 / np.sqrt(t + 1)
        
        # Update h
        h = h - step * g
        
        # Apply symmetry if required
        if sym:
            h = (h + h[::-1]) / 2
        
        # Project onto feasibility
        h = project_onto_capped_simplex(h, target_sum)
        
        # Update temperature
        T = max(1e-5, T * 0.9995)
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local hill-climbing using pair-swap moves"""
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        # Pick two indices
        i, j = np.random.choice(n, 2, replace=False)
        
        # Compute valid delta range
        delta_min = max(-best_h[i], best_h[j] - 1)
        delta_max = min(1 - best_h[i], best_h[j])
        
        if delta_min >= delta_max:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_min, delta_max)
        
        # Apply swap
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate new C5
        c5_new = evaluate_c5_fast(h_new)
        
        # Accept or probabilistically accept
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        elif np.random.rand() < 0.05 * np.exp(-(c5_new - best_c5) / (1e-3 + best_c5)):
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Coarse optimization first (n=60)
    coarse_n = 60
    coarse_target_sum = coarse_n / 2
    
    coarse_h, coarse_c5 = gradient_descent_restart(
        coarse_n, coarse_target_sum, 1000, 0.5, 0.1, 10, sym=False
    )
    
    # Upsample to higher resolution
    coarse_x = np.linspace(dx/2, 2-dx/2, coarse_n)
    fine_x = np.linspace(2/(n_points*2), 2-2/(n_points*2), n_points)
    fine_h = np.interp(fine_x, coarse_x, coarse_h)
    fine_h = project_onto_capped_simplex(fine_h, target_sum)
    
    # Global optimization runs
    best_h = fine_h.copy()
    best_c5 = evaluate_c5_fast(best_h)
    
    # Run multiple restarts
    restarts = 15
    restart_count = 0
    
    while restart_count < restarts and (time.time() - start_time) < budget_s * 0.8:
        if np.random.rand() < 0.7:
            # Random restart
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, target_sum)
        elif np.random.rand() < 0.5:
            # Symmetric random restart
            h0 = np.random.rand(n_points)
            h0 = (h0 + h0[::-1]) / 2
            h0 = project_onto_capped_simplex(h0, target_sum)
        else:
            # Triangular seed
            x = np.linspace(dx/2, 2-dx/2, n_points)
            h0 = np.maximum(0, 1 - np.abs(x-1)/1)
            h0 = project_onto_capped_simplex(h0, target_sum)
            
        h_best, c5_best = gradient_descent_restart(
            n_points, target_sum, 2000, 0.5, 0.1, 10, sym=False
        )
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            
        restart_count += 1
    
    # Local polishing
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        best_h, best_c5 = local_swap_search(
            best_h, best_c5, remaining_time * 0.9
        )
    
    # Verify final solution
    best_h = project_onto_capped_simplex(best_h, target_sum)
    final_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
