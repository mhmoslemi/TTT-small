import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 with dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of coarse solution to finer grid"""
    old_positions = np.linspace(0, 2, len(old_h))
    new_positions = np.linspace(0, 2, new_n)
    new_h = np.interp(new_positions, old_positions, old_h)
    return new_h

def worst_shift(c):
    """Return the shift s attaining maximum correlation"""
    idx = np.argmax(c)
    s = idx - (len(c) - 1)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient of the worst shift term"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        nonlocal TARGET_SUM
        TARGET_SUM = target_sum(n)
        dx_val = dx(n)
        
        # 1. Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
        else:
            h = interpolate(best_h_stage, n)
        h = project_to_sum_box(h, TARGET_SUM)
        current_C = C5_value(h)
        best_h_stage = h.copy()
        best_C_stage = current_C
        
        # 2. Simulated Annealing
        T = 0.02
        Δ = 0.1
        max_iter = int(100000 * time_fraction / budget_s)
        
        iter_count = 0
        while time.time() < start_time + budget_s * time_fraction and iter_count < max_iter:
            iter_count += 1
            
            # Generate move
            if random.random() < 0.3:  # Targeted move
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Select low desirability (lowest 20%) and high desirability (highest 20%)
                sorted_indices = np.argsort(d)
                low_indices = sorted_indices[:int(0.2 * n)]
                high_indices = sorted_indices[-int(0.2 * n):]
                
                if len(low_indices) > 0 and len(high_indices) > 0:
                    i = random.choice(low_indices)
                    j = random.choice(high_indices)
                    delta = min(Δ, h[i], 1 - h[j])
                    if delta > 0:
                        h[i] -= delta
                        h[j] += delta
                        h = project_to_sum_box(h, TARGET_SUM)
                        new_C = C5_value(h)
                        if new_C < current_C or random.random() < np.exp(-(new_C - current_C) / T):
                            current_C = new_C
                            if current_C < best_C_stage:
                                best_h_stage = h.copy()
                                best_C_stage = current_C
                        else:
                            # Revert
                            h[i] += delta
                            h[j] -= delta
                            h = project_to_sum_box(h, TARGET_SUM)
            else:  # Random move
                i, j = random.sample(range(n), 2)
                delta = min(Δ, h[i], 1 - h[j])
                if delta > 0:
                    h[i] -= delta
                    h[j] += delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    new_C = C5_value(h)
                    if new_C < current_C or random.random() < np.exp(-(new_C - current_C) / T):
                        current_C = new_C
                        if current_C < best_C_stage:
                            best_h_stage = h.copy()
                            best_C_stage = current_C
                    else:
                        # Revert
                        h[i] += delta
                        h[j] -= delta
                        h = project_to_sum_box(h, TARGET_SUM)
            
            # Update cooling parameters
            if iter_count % 10 == 0:
                T *= 0.9997
                Δ *= 0.9999
                if T < 1e-6:
                    T = 1e-6
                if Δ < 1e-6:
                    Δ = 1e-6
        
        # 3. Polishing
        N_polish = min(2000, int(5000 * time_fraction / budget_s))
        eta_0 = 0.5
        for k in range(N_polish):
            if time.time() >= start_time + budget_s * time_fraction:
                break
                
            c = corr_fft(h)
            max_corr = np.max(c)
            tolerance = 1e-12
            worst_shifts = np.where(np.abs(c - max_corr) < tolerance)[0]
            
            if len(worst_shifts) > 0:
                g = np.zeros(n)
                for s_idx in worst_shifts:
                    s = s_idx - (n - 1)
                    g += gradient_worst(h, s)
                g /= len(worst_shifts)
                
                eta = eta_0 / np.sqrt(k + 1)
                h = h - eta * g
                h = project_to_sum_box(h, TARGET_SUM)
                
                new_C = C5_value(h)
                if new_C < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = new_C
        
        return best_h_stage, best_C_stage
    
    # Run three stages
    best_h_stage, best_C_stage = make_stage(80, 0.30)
    best_h_stage, best_C_stage = make_stage(160, 0.35)
    best_h_stage, best_C_stage = make_stage(240, 0.35)
    
    return (best_h_stage.tolist(), float(best_C_stage), len(best_h_stage))
