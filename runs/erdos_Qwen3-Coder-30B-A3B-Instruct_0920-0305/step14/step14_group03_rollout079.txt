import numpy as np
from cvxpy import *
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    epsilon = 1e-4
    
    # Build index mapping for correlations
    def get_correlation_indices(n):
        corr_indices = []
        for s in range(-n+1, n):
            indices = []
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    indices.append((i, j))
            corr_indices.append(indices)
        return corr_indices
    
    corr_indices = get_correlation_indices(n_points)
    
    # Linear feasibility formulation with McCormick relaxation
    def solve_feasibility(C):
        # Variables
        h = Variable(n_points)
        w = {}
        
        # Create product variables
        for s in range(-n_points+1, n_points):
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    w[(i,s)] = Variable()
        
        # Constraints
        constraints = []
        
        # Box constraints
        constraints.extend([h[i] >= 0 for i in range(n_points)])
        constraints.extend([h[i] <= 1 for i in range(n_points)])
        
        # Sum constraint
        constraints.append(sum(h) == target_sum)
        
        # McCormick envelope constraints
        for s in range(-n_points+1, n_points):
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    w_ij = w[(i,s)]
                    constraints.extend([
                        w_ij >= 0,
                        w_ij <= h[i],
                        w_ij <= h[j],
                        w_ij >= h[i] + h[j] - 1
                    ])
        
        # Correlation constraints
        rhs = target_sum - C / dx
        for s, indices in enumerate(corr_indices):
            if indices:
                corr_expr = sum(w[(i,s-n_points+1)] for i,j in indices)
                constraints.append(corr_expr >= rhs)
        
        # Solve
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=GLPK, verbose=False, time_limit=10)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None
    
    # Binary search on C
    low = 0.30
    high = 0.40
    best_h = None
    best_C = float('inf')
    
    while high - low > epsilon and time.time() - start_time < 0.8 * budget_s:
        C_mid = (low + high) / 2
        h_sol = solve_feasibility(C_mid)
        
        if h_sol is not None:
            high = C_mid
            best_h = h_sol
            best_C = C_mid
        else:
            low = C_mid
    
    # Refinement with projected sub-gradient method
    if best_h is not None:
        h = best_h.copy()
        
        # Project to exact sum constraint
        def project_to_sum(x, target_sum):
            def f(lam):
                projected = np.clip(x + lam, 0, 1)
                return np.sum(projected) - target_sum
            
            # Binary search for lambda
            a, b = -1.0, 1.0
            while f(a) * f(b) > 0:
                a -= 0.1
                b += 0.1
            
            for _ in range(50):
                m = (a + b) / 2
                fm = f(m)
                if abs(fm) < 1e-10:
                    break
                if fm * f(a) < 0:
                    b = m
                else:
                    a = m
            
            return np.clip(x + (a + b)/2, 0, 1)
        
        h = project_to_sum(h, target_sum)
        
        def compute_c5(h):
            # Using correlate
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
            return dx * np.max(corr)
        
        # Sub-gradient optimization
        step_size = 0.001
        max_iter = 1000
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Compute correlations
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
            
            # Find worst shift
            max_corr_idx = np.argmax(corr)
            max_corr = corr[max_corr_idx]
            
            # Sub-gradient
            s = max_corr_idx - len(h) + 1
            g = np.zeros(len(h))
            
            # Gradient computation for the worst shift
            for i in range(len(h)):
                j1 = i + s
                j2 = i - s
                if 0 <= j1 < len(h):
                    g[i] += 1 - h[j1]
                if 0 <= j2 < len(h):
                    g[i] += 1 - h[j2]
            
            # Update
            h_new = h - step_size * g
            h_new = project_to_sum(h_new, target_sum)
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                h = h_new
                best_C = min(best_C, c5_new)
        
        # Upsample to higher resolution
        n_fine = 200
        dx_fine = 2.0 / n_fine
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_points), h)
        h_fine = project_to_sum(h_fine, n_fine / 2)
        
        # Final refinement on finer grid
        for iteration in range(500):
            if time.time() - start_time > budget_s:
                break
                
            # Compute correlations
            h_rev = h_fine[::-1]
            b = 1 - h_fine
            L = 2 * len(h_fine) - 1
            corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
            
            # Find worst shift
            max_corr_idx = np.argmax(corr)
            max_corr = corr[max_corr_idx]
            
            # Sub-gradient
            s = max_corr_idx - len(h_fine) + 1
            g = np.zeros(len(h_fine))
            
            # Gradient computation for the worst shift
            for i in range(len(h_fine)):
                j1 = i + s
                j2 = i - s
                if 0 <= j1 < len(h_fine):
                    g[i] += 1 - h_fine[j1]
                if 0 <= j2 < len(h_fine):
                    g[i] += 1 - h_fine[j2]
            
            # Update
            h_fine_new = h_fine - 0.0005 * g
            h_fine_new = project_to_sum(h_fine_new, n_fine / 2)
            
            # Check improvement
            c5_new = dx_fine * np.max(np.fft.ifft(np.fft.fft(h_fine_new[::-1], L) * np.fft.fft(1 - h_fine_new, L)).real)
            if c5_new < compute_c5(h_fine):
                h_fine = h_fine_new
                best_C = min(best_C, c5_new)
        
        h_final = h_fine
    else:
        # Fallback to initial solution if nothing worked
        h_final = np.array(initial_h_values)
        h_final = project_to_sum(h_final, target_sum)
        best_C = 0.380975  # Previous best
    
    # Final computation
    def evaluate_c5(h):
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return dx * np.max(corr)
    
    best_C = evaluate_c5(h_final)
    
    return (h_final.tolist(), float(best_C), len(h_final))
