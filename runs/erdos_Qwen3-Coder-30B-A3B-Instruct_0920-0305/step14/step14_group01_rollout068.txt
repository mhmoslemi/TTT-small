import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_box_sum(u, target_sum):
    """Project vector u onto box constraints [0,1] with sum = target_sum using water-filling"""
    u = np.array(u)
    u = np.clip(u, 0, 1)
    
    # Water-filling approach
    def objective(lam):
        proj = np.clip(u + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(100):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-12:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    lam = (a + b) / 2
    result = np.clip(u + lam, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def gradient_of_smooth_max(h, epsilon=0.02):
    """Compute gradient of log-sum-exp approximation to max"""
    corr = compute_correlation(h)
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradients using convolution
    h_rev = h[::-1]
    b = 1 - h
    
    # Forward convolution: w * b
    L = 2 * len(h) - 1
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # Backward convolution: w[::-1] * h
    conv2 = np.fft.ifft(np.fft.fft(w[::-1], L) * np.fft.fft(h, L)).real
    
    # Extract central part and compute gradient
    grad = conv1 - conv2
    grad = grad[len(h)-1:2*len(h)-1] * dx
    
    return grad

def compute_smooth_max(h, epsilon=0.02):
    """Compute smooth approximation to max"""
    corr = compute_correlation(h)
    return np.log(np.sum(np.exp(corr / epsilon))) * epsilon

            penalty_grad[i-1] += 2 * 1e3 * (u[i-1] - u[i])
            penalty_grad[i] -= 2 * 1e3 * (u[i-1] - u[i])
    
    # Add sum penalty
    sum_penalty = 2 * 1e4 * (np.sum(u) - target_sum)
    penalty_grad += sum_penalty
    
    return value, penalty_grad
