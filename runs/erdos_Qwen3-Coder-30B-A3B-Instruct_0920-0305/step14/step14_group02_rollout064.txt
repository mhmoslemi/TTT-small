import numpy as np

# Global variables
TARGET_SUM = None

dx = 2.0 / n_points
max_shifts = 12  # Top-K shifts to consider for gradient computation

def evaluate_c5_exact(h):
    """Evaluate C5 using exact FFT correlation"""
    # Compute full correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.01):
    """Evaluate smooth surrogate using log-sum-exp"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Use top-K shifts for numerical stability
    top_indices = np.argsort(corr)[-max_shifts:]
    top_corr = corr[top_indices]
    
    # Numerical stability: subtract max
    max_corr = np.max(top_corr)
    if max_corr > 0:
        exp_corr = np.exp((top_corr - max_corr) / T)
    else:
        exp_corr = np.exp(top_corr / T)
    
    # Regularize with a small value
    return dx * (T * np.log(np.sum(exp_corr)) + max_corr)

def correlation_gradient(h, T=0.01):
    """Compute gradient of the smooth surrogate"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Use top-K shifts for numerical stability
    top_indices = np.argsort(corr)[-max_shifts:]
    top_corr = corr[top_indices]
    
    # Compute softmax weights
    max_corr = np.max(top_corr)
    if max_corr > 0:
        exp_corr = np.exp((top_corr - max_corr) / T)
    else:
        exp_corr = np.exp(top_corr / T)
    
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient components
    grad = np.zeros_like(h)
    for idx, w in zip(top_indices, weights):
        # Compute partial derivative of correlation w.r.t h_i for shift idx
        # For shift s = idx - (n_points-1) (negative shifts are stored with positive indices)
        s = idx - (n_points - 1)
        if s >= 0:
            # shift s is positive: c_s = sum_{i=0}^{n_points-1-s} h[i](1-h[i+s])
            for i in range(n_points - s):
                grad[i] += w * (1 - h[i + s])
                if i + s < n_points:  # avoid double counting
                    grad[i + s] -= w * h[i]
        else:
            # shift s is negative: c_s = sum_{i=-s}^{n_points-1} h[i](1-h[i+s])
            # But since we use h_rev, we consider c_s = sum_{i=0}^{n_points-1-(-s)} h[i](1-h[i-s])
            s = -s  # Change to positive form
            for i in range(s, n_points):
                grad[i] += w * (1 - h[i - s])
                if i - s >= 0:
                    grad[i - s] -= w * h[i]
    
    return grad * dx

def triangular_seed(n_points):
    """Create a symmetric triangular seed"""
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 1))
    h = h / np.sum(h) * (n_points / 2)
    return h
