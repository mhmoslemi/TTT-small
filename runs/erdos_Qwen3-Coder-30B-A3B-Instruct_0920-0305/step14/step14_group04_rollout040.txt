```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_raw_corr(h):
    """Compute raw correlation using FFT (without dx)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weights(corr_raw, alpha):
    """Compute softmax weights for gradient computation"""
    max_c = np.max(corr_raw)
    exps = np.exp(alpha * (corr_raw - max_c))
    return exps / np.sum(exps)

def compute_ascend_weighted(h, weights):
    """Compute weighted ascend direction for gradient"""
    n = len(h)
    ascend = np.zeros_like(h)
    
    # For each shift, compute contribution
    for idx, w_s in enumerate(weights):
        if w_s == 0:
            continue
        shift = idx - (n-1)
        
        if shift == 0:
            ascend += 2 * w_s * h
        elif shift > 0:
            ascend[:n-shift] += w_s * h[shift:]
            ascend[shift:] += w_s * h[:n-shift]
        else:  # shift < 0
            s = -shift
            ascend[s:] += w_s * h[:n-s]
            ascend[:n-s] += w_s * h[s:]
    
    return ascend

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Predefine initial constructions
    init_constructions = []
    
    # Triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    tri = np.minimum(x, 2 - x)
    init_constructions.append(tri)
    
    # Random uniform continuous
    for _ in range(2):
        rand = np.random.rand(n_points)
        rand_proj = project_to_sum_box(rand, TARGET_SUM)
        init_constructions.append(rand_proj)
    
    # Random binary
    for _ in range(2):
        idx = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        bin_vec = np.zeros(n_points)
        bin_vec[idx] = 1.0
        init_constructions.append(bin_vec)
    
    # Include initial_h_values if available and length matches
    if 'initial_h_values' in globals():
        try:
            h0 = np.array(initial_h_values)
            if len(h0) == n_points:
                init_constructions.append(project_to_sum_box(h0, TARGET_SUM))
        except:
            pass
    
    # Run optimization for each initial construction
    for init_h in init_constructions:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Copy initial h
        h = init_h.copy()
        
        # Compute initial score
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent loop
        alpha = 1.0
        step_size = 0.3
        max_iter = 5000
        patience = 0
        
        for iter_num in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute raw correlation
            corr_raw = compute_raw_corr(h)
            
            # Compute weights for gradient
            weights = compute_weights(corr_raw, alpha)
            
            # Compute weighted ascend direction
            ascend = compute_ascend_weighted(h, weights)
            
            # Normalize ascend to avoid large steps
            ascend_norm = ascend / (np.max(np.abs(ascend)) + 1e-12)
            
            # Update with step
            h_new = h + step_size * ascend_norm
            
            # Project to feasible set
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Symmetrize (enforce symmetry about center)
            h_new = (h_new + h_new[::-1]) / 2
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate new solution
            c5_new = evaluate_c5(h_new)
            
            # Accept improvement
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                patience = 0
            else:
                patience += 1
                # Occasionally accept worse solution with probability
                if patience > 10 and np.random.rand() < 0.1:
                    h = h_new
                    current_c5 = c5_new
            
            # Adjust hyperparameters
            if iter_num % 50 == 0 and alpha < 100:
                alpha *= 1.02
                
            if patience > 20:
                step_size *= 0.9
                
            # Occasionally perform targeted moves
            if iter_num % 50 == 0:
                # Find worst shift
                max_corr_idx = np.argmax(corr_raw)
                worst_shift = max_corr_idx - (n_points - 1)
                
                # Perform a small local optimization
                h_pert = h.copy()
                # Move mass from a low to high position in worst shift area
                for _ in range(10):
                    i = np.random.randint(n_points)
                    j = (i + worst_shift) % n_points
                    if (h_pert[i] > 0.01 and h_pert[j] < 0.99):
                        delta = min(0.01, h_pert[i], 1 - h_pert[j])
                        h_pert[i] -= delta
                        h_pert[j] += delta
                
                h_pert = project_to_sum_box(h_pert, TARGET_SUM)
                c5_pert = evaluate_c5(h_pert)
                if c5_pert < current_c5:
                    h = h_pert
                    current_c5 = c5_pert
                    if c5_pert < best_c5:
                        best_c5 = c5_pert
                        best_h = h.copy()
    
    # Final refinement
    if best_h is not None:
        h = best_h.copy()
        t_final = start_time + budget_s - 5
        for i in range(1000):
            if time.time() >= t_final:
                break
                
            # Local perturbation
            h_pert = h.copy()
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points) 
            delta = np.random.normal(0, 0.005)
            h_pert[idx1] = np.clip(h_pert[idx1] + delta, 0, 1)
            h_pert[idx2] = np.clip(h_pert[idx2] - delta, 0, 1)
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < evaluate_c5(h):
                h = h_pert
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h.copy()

    if best_h is None:
        # If no good solution found, return the best from initial constructions
        best_h = init_constructions[0]
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
```
