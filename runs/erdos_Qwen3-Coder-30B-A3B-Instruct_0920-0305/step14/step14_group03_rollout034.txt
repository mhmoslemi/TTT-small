def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1

    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

    # Compute gradient for the specific shift that gives maximum correlation
    grad = np.zeros(len(h))
    
    # For shift k, the contribution to the gradient is:
    # d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} (if i+k within bounds)
    # Also [h_i * (1 - h_{i-k})] = 1 - h_{i-k} (if i-k within bounds)
    
    for i in range(len(h)):
        # Gradient contribution from shift k where max_corr occurs
        for k in range(-len(h) + 1, len(h)):
            idx_i = i
            idx_j = i + k

                    # Active shift - contributes to gradient
                    grad[i] += 1 - h[idx_j]
                else:
                    # Other shifts - their effect is negated with proper direction in max
                    grad[i] -= h[idx_j] if k > 0 else h[idx_j]
    
    return grad

def compute_smooth_max_gradient(h, corr, alpha=10.0):
    """Compute gradient using smooth max approximation"""
    # Compute softmax weights for the correlation
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    
    # Compute gradient from weighted sum of gradient contributions for each shift
    grad = np.zeros(len(h))
    
    # Consider all shifts
    for shift_idx in range(len(corr)):
        shift = shift_idx - len(h) + 1
        weight = w[shift_idx]
        
        # For this shift, compute gradient contribution
        for i in range(len(h)):
            # Contribution from shift to the right (h[i-k] from i)
            if shift > 0 and i - shift >= 0:
                grad[i] += weight * (1 - h[i - shift])
            # Contribution from shift to the left (h[i+k] from i)
            elif shift < 0 and i + shift < len(h):
                grad[i] += weight * (1 - h[i + shift])
            elif shift == 0:
                # Self-correlation term - diagonal
                grad[i] -= weight * h[i]
    
    return grad
