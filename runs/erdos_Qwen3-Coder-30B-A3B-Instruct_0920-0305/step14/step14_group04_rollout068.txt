# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def optimize_with_cvxpy(h0):
    """Use convex optimization to improve the solution"""
    # Set up the optimization problem
    h = cp.Variable(n_points)
    
    # Objective: minimize max correlation
    # We'll use a smooth approximation of the maximum
    alpha = 10.0
    
    # Create correlation vector with smooth max approximation
    corr_values = []
    for k in range(2 * n_points - 1):
        val = 0
        for i in range(n_points):
            j = i + k - (n_points - 1)
            if 0 <= j < n_points:
                val += h[i] * (1 - h[j])
        corr_values.append(val * dx)
    
    # Use log-sum-exp approximation to max
    max_corr_approx = cp.log_sum_exp(alpha * cp.hstack(corr_values)) / alpha
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == TARGET_SUM
    ]
    
    # Objective
    objective = cp.Minimize(max_corr_approx)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h.value
    except:
        pass
    
    return h0
