def project_to_simplex_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_soft_max_gradient(h, corr, alpha=10.0):
    """Compute exact gradient of soft max approximation of max overlap"""
    # Compute softmax weights for each shift
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    
    # Compute gradient efficiently using vectorized operations
    grad = np.zeros_like(h)
    
    # For each shift, compute contribution to gradient
    n = len(h)
    for s in range(-n+1, n):
        if s == 0:
            grad += -2 * w[s + n - 1] * h
        elif s > 0:
            # Forward contribution
            grad[:n-s] += -w[s + n - 1] * h[s:]
            # Backward contribution  
            grad[s:] += -w[s + n - 1] * h[:n-s]
        else:  # s < 0
            # Forward contribution (symmetric)
            grad[:n+abs(s)] += -w[abs(s) + n - 1] * h[abs(s):]
            # Backward contribution (symmetric)
            grad[abs(s):] += -w[abs(s) + n - 1] * h[:n-abs(s)]
    
    return grad

def interpolate_vector(h_coarse, n_fine):
    """Linearly interpolate a coarse vector to a finer grid"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_fine)
    
    # Linear interpolation
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
