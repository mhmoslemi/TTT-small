import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use a more sophisticated initialization approach
    # Start with a piecewise constant function that's balanced
    h0 = np.ones(n_points) * 0.5
    # Add some structured variation to avoid flat solutions
    for i in range(n_points):
        h0[i] += 0.1 * np.sin(2 * np.pi * i / n_points)
    
    h0 = project_to_sum_box(h0, TARGET_SUM)
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Use a combination of evolutionary and gradient-based optimization
    # Start with a population-based approach
    popsize = 30
    population = []
    
    # Generate diverse initial population
    for i in range(popsize):
        if i == 0:
            # First individual is our improved initialization
            h = h0.copy()
        elif i == 1:
            # Second individual is random with projection
            h = project_to_sum_box(np.random.rand(n_points), TARGET_SUM)
        else:
            # Remaining individuals use different strategies
            strategy = i % 3
            if strategy == 0:
                # Sine-based modulation
                h = np.ones(n_points) * 0.5
                for j in range(n_points):
                    h[j] += 0.2 * np.sin(4 * np.pi * j / n_points + np.random.rand() * np.pi)
            elif strategy == 1:
                # Step-like pattern
                h = np.zeros(n_points)
                step_pos = n_points // 3
                h[:step_pos] = 1.0
                h[step_pos:] = 0.3
            else:
                # Gaussian-like shape
                x = np.linspace(0, 2, n_points)
                h = np.exp(-((x - 1.0)**2) / (2 * 0.2**2))
                h = np.clip(h, 0, 1)
                
            h = project_to_sum_box(h, TARGET_SUM)
            
        population.append(h.copy())
        
    # Fitness-based evolution with adaptive parameters
    max_gen = 2000
    gen = 0
    t1 = start_time + 0.7 * budget_s
    while time.time() < t1 and gen < max_gen:
        gen += 1
        
        # Evaluate population
        fitness = [evaluate_c5(ind) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        
        # Keep top 50% and mutate rest
        top_indices = sorted_indices[:popsize//2]
        new_population = [population[i].copy() for i in top_indices]
        
        # Create offspring through crossover and mutation
        for i in range(popsize//2, popsize):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(top_indices, tournament_size, replace=False)
            winner = tournament_indices[np.argmin([fitness[j] for j in tournament_indices])]
            
            # Create offspring
            child = population[winner].copy()
            
            # Apply Gaussian mutation with adaptive rate
            mutation_rate = max(0.01, 0.1 * (1 - gen/max_gen))
            
            for j in range(n_points):
                if np.random.rand() < 0.1:  # Crossover probability
                    # Blend with another random individual
                    other = np.random.choice(popsize//2)
                    alpha = np.random.rand()
                    child[j] = alpha * child[j] + (1 - alpha) * population[other][j]
                    
                # Add mutation
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.normal(0, 0.05)
                    
            # Repair and add to population
            child = project_to_sum_box(child, TARGET_SUM)
            new_population.append(child)
            
        population = new_population
        
        # Update best solution
        current_fitness = [evaluate_c5(ind) for ind in population]
        min_fitness_idx = np.argmin(current_fitness)
        if current_fitness[min_fitness_idx] < best_c5:
            best_c5 = current_fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()

    # Local refinement with adaptive simulated annealing
    if best_h is not None:
        h = best_h.copy()
        max_iter = 2000
        temperature = 1.0
        min_temperature = 0.001
        cooling_rate = 0.995
        
        t2 = start_time + 0.9 * budget_s
        for iter_num in range(max_iter):
            if time.time() >= t2:
                break
                
            # Create neighbor
            h_neighbor = h.copy()
            idx = np.random.randint(n_points)
            
            # Try different perturbation strategies
            if np.random.rand() < 0.7:
                # Small random change
                change = np.random.normal(0, 0.02)
                h_neighbor[idx] = np.clip(h_neighbor[idx] + change, 0, 1)
            else:
                # Larger change
                change = np.random.normal(0, 0.08)
                h_neighbor[idx] = np.clip(h_neighbor[idx] + change, 0, 1)
                
            h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
            
            # Evaluate
            c5_neighbor = evaluate_c5(h_neighbor)
            c5_current = evaluate_c5(h)
            
            # Accept with probability
            if c5_neighbor < c5_current:
                h = h_neighbor
                if c5_neighbor < best_c5:
                    best_c5 = c5_neighbor
                    best_h = h.copy()
            else:
                # Accept with probability based on temperature
                if np.random.rand() < np.exp(-(c5_neighbor - c5_current) / temperature):
                    h = h_neighbor
                    
            # Cool down
            temperature *= cooling_rate
            if temperature < min_temperature:
                temperature = min_temperature

    # Final post-processing: try to find even better solutions by local search
    if best_h is not None:
        h = best_h.copy()
        
        # Try different local search strategies
        for strategy in range(3):
            h_test = h.copy()
            
            if strategy == 0:
                # Gradient-like search
                for i in range(500):
                    if time.time() >= start_time + budget_s:
                        break
                    idx = np.random.randint(n_points)
                    # Simple gradient estimation
                    h_plus = h_test.copy()
                    h_plus[idx] = min(h_test[idx] + 0.03, 1.0)
                    h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                    
                    h_minus = h_test.copy()
                    h_minus[idx] = max(h_test[idx] - 0.03, 0.0)
                    h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                    
                    c5_plus = evaluate_c5(h_plus)
                    c5_minus = evaluate_c5(h_minus)
                    
                    if c5_plus <= c5_minus:
                        h_test = h_plus
                    else:
                        h_test = h_minus
                        
                    if evaluate_c5(h_test) < best_c5:
                        best_c5 = evaluate_c5(h_test)
                        best_h = h_test.copy()
                        
            elif strategy == 1:
                # Pattern-based refinement
                for i in range(300):
                    if time.time() >= start_time + budget_s:
                        break
                    # Create a structured perturbation
                    perturbation = np.random.normal(0, 0.02, n_points)
                    # Apply some local correlation
                    for j in range(0, n_points, 8):
                        if j + 8 < n_points:
                            avg = np.mean(h_test[j:j+8])
                            perturbation[j:j+8] = np.full(8, avg)
                            
                    h_test_new = h_test + perturbation
                    h_test_new = project_to_sum_box(h_test_new, TARGET_SUM)
                    
                    if evaluate_c5(h_test_new) < evaluate_c5(h_test):
                        h_test = h_test_new
                        if evaluate_c5(h_test) < best_c5:
                            best_c5 = evaluate_c5(h_test)
                            best_h = h_test.copy()
                            
            else:
                # Randomized search
                for i in range(500):
                    if time.time() >= start_time + budget_s:
                        break
                    h_test_new = h_test.copy()
                    # Make a few random modifications
                    num_changes = np.random.randint(1, 10)
                    for _ in range(num_changes):
                        idx = np.random.randint(n_points)
                        h_test_new[idx] = np.clip(h_test_new[idx] + np.random.normal(0, 0.05), 0, 1)
                    h_test_new = project_to_sum_box(h_test_new, TARGET_SUM)
                    
                    if evaluate_c5(h_test_new) < evaluate_c5(h_test):
                        h_test = h_test_new
                        if evaluate_c5(h_test) < best_c5:
                            best_c5 = evaluate_c5(h_test)
                            best_h = h_test.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
