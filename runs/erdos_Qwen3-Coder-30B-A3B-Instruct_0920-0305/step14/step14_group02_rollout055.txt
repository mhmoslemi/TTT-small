def evaluate_c5_exact(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.05):
    """Evaluate smooth surrogate of C5 using log-sum-exp"""
    # Compute all correlations using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr[:len(corr)//2 + 1]  # Take only unique correlations
    corr = corr * dx
    
    # Apply log-sum-exp with temperature T
    # Shift for numerical stability
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(shifted_corr / T)
    log_sum_exp = max_corr + T * np.log(np.sum(exp_corr))
    return log_sum_exp

def evaluate_gradient(h, T=0.05):
    """Compute gradient of smooth surrogate"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr[:len(corr)//2 + 1] * dx
    
    # Compute softmax weights
    max_corr = np.max(corr)
    shifted_corr = corr - max_corr
    exp_corr = np.exp(shifted_corr / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        if i > 0:  # Positive shifts
            s = i
            # Derivative of c_s
            for j in range(len(h) - s):
                grad[j] += w * (1 - h[j+s])
                grad[j+s] -= w * h[j]
        else:  # Zero shift - special case
            # Derivative of c_0
            for j in range(len(h)):
                grad[j] += w * (1 - h[j])
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use water-filling algorithm for projection onto simplex
    x = np.array(x)
    # Sort x in descending order
    sorted_x = np.sort(x)[::-1]
    
    # Find the projection threshold lambda
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    lambda_vals = (cumsum - target_sum) / np.arange(1, n + 1)
    
    # Find the largest lambda such that all elements after it are >= 0
    # and all elements before it are >= lambda
    threshold_idx = np.argmax(sorted_x >= lambda_vals)
    
    if threshold_idx < len(lambda_vals):
        lambda_star = lambda_vals[threshold_idx]
    else:
        lambda_star = lambda_vals[-1]
    
    # Compute projection
    result = np.maximum(0, x - lambda_star)
    # Adjust for numerical errors
    if abs(np.sum(result) - target_sum) > 1e-10:
        # Handle case where we need to adjust again
        adjustment = (target_sum - np.sum(result)) / len(result)
        result = np.clip(result + adjustment, 0, 1)
        
    return result
