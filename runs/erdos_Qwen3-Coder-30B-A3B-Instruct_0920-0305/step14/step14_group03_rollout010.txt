import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    # Compute correlation using FFT
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    # O = dx * (target_sum - correlation)
    O = dx * (target_sum - corr)
    return O

def compute_gradient_from_w(h, w, dx):
    """Compute gradient using softmax weights"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Compute contributions for each shift
    for r in range(2*n-1):
        s = r - (n-1)
        w_s = w[r]
        if s == 0:
            grad += -dx * 2 * w_s * h
        elif s > 0:
            # forward contribution
            grad[:n-s] += -dx * w_s * h[s:]
            # backward contribution  
            grad[s:] += -dx * w_s * h[:n-s]
        else:
            # s < 0
            s_abs = -s
            grad[s_abs:] += -dx * w_s * h[:n-s_abs]
            grad[:n-s_abs] += -dx * w_s * h[s_abs:]
    
    return grad
