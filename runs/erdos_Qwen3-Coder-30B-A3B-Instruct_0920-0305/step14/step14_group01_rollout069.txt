def project_to_box_sum(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum using water-filling"""
    x = np.array(x)
    # Binary search for lambda
    def f(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT - returns full correlation vector"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

    h_extended[:len(h)] = h

    conv1 = np.fft.ifft(np.fft.fft(w_extended, L) * np.fft.fft(b_extended, L)).real
    # Convolution: w_rev * h

    # Extract the central part and adjust indices
    grad = conv1[len(h)-1:] - conv2[len(h)-1:]
    return dx * grad

    # Compute smooth max gradient
    grad_smooth = smooth_max_gradient(h, epsilon)
    
    # Compute penalized gradient
    grad = grad_smooth.copy()
    
    # Monotonicity penalty gradient for increasing part
    for i in range(1, len(u)):
        if u[i-1] > u[i]:
            grad[i] -= 2 * lambda_inc * (u[i-1] - u[i])
            grad[i-1] += 2 * lambda_inc * (u[i-1] - u[i])
    
    # Sum constraint penalty gradient
    grad_sum = 2 * mu * (np.sum(u) - target_sum)
    grad[:len(u)] += grad_sum
    
    # Objective value
    corr = compute_correlation(h)
    objective = epsilon * np.log(np.sum(np.exp(corr / epsilon)))  # LSE approximation
    
    return objective, grad
