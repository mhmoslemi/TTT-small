import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 value"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old grid to new grid"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(c):
    """Get the shift index that gives maximum correlation"""
    idx = np.argmax(c)
    return idx - (len(c) - 1)

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def make_stage(n, time_fraction, prev_best_h, start_time, budget_s):
    """Run a single optimization stage"""
    # Initialize
    if prev_best_h is None:
        h = np.random.uniform(0, 1, n)
        h = project_to_sum_box(h, target_sum(n))
    else:
        h = interpolate(prev_best_h, n)
        h = project_to_sum_box(h, target_sum(n))
    
    best_h_stage = h.copy()
    best_C_stage = C5_value(h)
    
    # Annealing parameters
    T = 0.02
    Δ = 0.1
    α = 0.9997
    β = 0.9999
    max_iter = int(80000 * time_fraction / budget_s)  # Scale based on time fraction
    
    # Track acceptance rate
    accepted_count = 0
    total_count = 0
    
    for iteration in range(max_iter):
        if time.time() >= start_time + budget_s * time_fraction:
            break
            
        total_count += 1
        
        # Decide on move type (targeted vs random)
        if random.random() < 0.3:
            # Targeted move - improve worst shift
            c = corr_fft(h)
            s = worst_shift(c)
            h_plus, h_minus = shift_arrays(h, s)
            d = h_plus + h_minus
            
            # Pick from low desirability (bottom 20%)
            low_indices = np.argsort(d)[:max(1, int(0.2 * n))]
            i = random.choice(low_indices)
            
            # Pick from high desirability (top 20%)
            high_indices = np.argsort(d)[-max(1, int(0.2 * n)):]
            j = random.choice(high_indices)
        else:
            # Random move
            i, j = random.sample(range(n), 2)
            # Ensure Δ is bounded by capacity
            Δ = min(Δ, h[i], 1 - h[j])
            
        if Δ <= 1e-10:
            continue
            
        # Propose transfer
        δ = random.uniform(0, Δ)
        h_new = h.copy()
        h_new[i] -= δ
        h_new[j] += δ
        
        # Clip to stay in box
        h_new[i] = max(0, min(1, h_new[i]))
        h_new[j] = max(0, min(1, h_new[j]))
        
        # Re-project to sum constraint
        h_new = project_to_sum_box(h_new, target_sum(n))
        
        # Evaluate new value
        c_new = C5_value(h_new)
        c_old = C5_value(h)
        
        # Accept or reject
        if c_new <= c_old:
            h = h_new
            if c_new < best_C_stage:
                best_C_stage = c_new
                best_h_stage = h_new.copy()
            accepted_count += 1
        else:
            # Accept with probability
            if random.random() < np.exp(-(c_new - c_old) / T):
                h = h_new
                if c_new < best_C_stage:
                    best_C_stage = c_new
                    best_h_stage = h_new.copy()
                accepted_count += 1
        
        # Update temperature and transfer size
        T = α * T
        Δ = β * Δ
        
        # Reset if acceptance rate is too low
        if total_count > 100 and accepted_count / total_count < 0.01:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, target_sum(n))
            accepted_count = 0
            total_count = 0
    
    # Polishing with exact gradient descent
    h_polish = best_h_stage.copy()
    for k in range(2000):
        if time.time() >= start_time + budget_s * time_fraction:
            break
            
        c = corr_fft(h_polish)
        max_val = np.max(c)
        
        # Find all shifts that achieve max value (within tolerance)
        tol = 1e-12
        s_indices = np.where(np.abs(c - max_val) < tol)[0]
        
        if len(s_indices) > 1:  # If there are ties
            # Average gradients from all worst shifts
            grad = np.zeros(len(h_polish))
            for s_idx in s_indices:
                s = s_idx - (len(c) - 1)
                grad += gradient_worst(h_polish, s)
            grad /= len(s_indices)
        else:
            s = s_indices[0] - (len(c) - 1)
            grad = gradient_worst(h_polish, s)
        
        # Step size schedule
        eta = 0.5 / np.sqrt(k + 1)
        h_polish = h_polish - eta * grad
        
        # Project to constraints
        h_polish = project_to_sum_box(h_polish, target_sum(n))
        
        # Update best if improved
        c_new = C5_value(h_polish)
        if c_new < best_C_stage:
            best_C_stage = c_new
            best_h_stage = h_polish.copy()
    
    return best_h_stage, best_C_stage

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization routine"""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Stage 1: coarse grid
    best_h, best_C = make_stage(80, 0.30, None, start_time, budget_s)
    
    # Stage 2: medium grid
    best_h, best_C = make_stage(160, 0.35, best_h, start_time, budget_s)
    
    # Stage 3: fine grid  
    best_h, best_C = make_stage(240, 0.35, best_h, start_time, budget_s)
    
    return (best_h.tolist(), float(best_C), len(best_h))
