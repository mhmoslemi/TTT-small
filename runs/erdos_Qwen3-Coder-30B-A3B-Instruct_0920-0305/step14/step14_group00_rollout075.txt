def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h values"""

    correlations = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts 1 to n-1)

    c5_bound = np.max(relevant_correlations) * dx
    return c5_bound
