import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Compute C5 value with proper dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Interpolate from coarse to fine grid"""
    old_centers = np.linspace(0.5, 1.5, len(old_h))
    new_centers = np.linspace(0.5, 1.5, new_n)
    new_h = np.interp(new_centers, old_centers, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(c):
    """Get shift s that achieves max correlation"""
    idx = np.argmax(c)
    s = idx - (len(c) // 2)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        nonlocal TARGET_SUM
        TARGET_SUM = target_sum(n)
        current_time = time.time()
        stage_deadline = start_time + budget_s * time_fraction
        
        # Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
        else:
            h = interpolate(best_h_stage, n)
        h = project_to_sum_box(h, TARGET_SUM)
        
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Annealing parameters
        T = 0.02
        Δ = 0.1
        α = 0.9997
        β = 0.9999
        max_iter = int(50000 * time_fraction / 0.35)
        accept_count = 0
        
        # Run simulated annealing
        for iteration in range(max_iter):
            if time.time() >= stage_deadline:
                break
                
            # Choose move type
            if random.random() < 0.3:
                # Targeted move to improve worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                d = np.zeros(n)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Select low desirability indices (bottom 20%)
                sorted_indices = np.argsort(d)
                low_indices = sorted_indices[:max(1, n//5)]
                i = random.choice(low_indices)
                
                # Select high desirability indices (top 20%)
                high_indices = sorted_indices[-max(1, n//5):]
                j = random.choice(high_indices)
                
                # Calculate transfer amount
                max_transfer = min(h[i], 1 - h[j])
                if max_transfer > 0:
                    δ = random.uniform(0, max_transfer)
                    h[i] -= δ
                    h[j] += δ
                    h = project_to_sum_box(h, TARGET_SUM)
                    
                    # Accept if better or with probability
                    new_C = C5_value(h)
                    if new_C <= best_C_stage:
                        best_h_stage = h.copy()
                        best_C_stage = new_C
                    elif random.random() < np.exp(-(new_C - best_C_stage) / T):
                        accept_count += 1
                    else:
                        # Revert
                        h[i] += δ
                        h[j] -= δ
                        h = project_to_sum_box(h, TARGET_SUM)
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                max_transfer = min(h[i], 1 - h[j])
                if max_transfer > 0:
                    δ = random.uniform(0, max_transfer)
                    h[i] -= δ
                    h[j] += δ
                    h = project_to_sum_box(h, TARGET_SUM)
                    
                    # Accept if better or with probability
                    new_C = C5_value(h)
                    if new_C <= best_C_stage:
                        best_h_stage = h.copy()
                        best_C_stage = new_C
                    elif random.random() < np.exp(-(new_C - best_C_stage) / T):
                        accept_count += 1
                    else:
                        # Revert
                        h[i] += δ
                        h[j] -= δ
                        h = project_to_sum_box(h, TARGET_SUM)
            
            # Update temperature and delta
            T *= α
            Δ *= β
            
            # Periodic restart if acceptance is too low
            if iteration > 0 and iteration % 1000 == 0 and accept_count == 0:
                h = np.random.uniform(0, 1, n)
                h = project_to_sum_box(h, TARGET_SUM)
            accept_count = 0
        
        # Polishing with exact sub-gradient descent
        for polish_iter in range(2000):
            if time.time() >= stage_deadline:
                break
                
            c = corr_fft(h)
            max_val = np.max(c)
            # Get the set of shifts achieving max value
            S = np.where(np.abs(c - max_val) < 1e-12)[0]
            
            # Compute gradient for all worst shifts (average)
            g = np.zeros(n)
            for s_idx in S:
                s = s_idx - (n - 1)
                g += gradient_worst(h, s)
            g /= len(S)
            
            # Step size diminishing
            eta = 0.5 / np.sqrt(polish_iter + 1)
            h_new = h - eta * g
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            new_C = C5_value(h_new)
            if new_C < best_C_stage:
                best_h_stage = h_new.copy()
                best_C_stage = new_C
            h = h_new
        
        return best_h_stage, best_C_stage
    
    # Run stages
    best_h, best_C = make_stage(80, 0.30)
    best_h, best_C = make_stage(160, 0.35)
    best_h, best_C = make_stage(240, 0.35)
    
    return (best_h.tolist(), float(best_C), len(best_h))
