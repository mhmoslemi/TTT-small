import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -2.0, 2.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    conv = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    O = dx * (target_sum - conv)
    return O

def compute_gradient_from_w(h, w, dx):
    """Compute gradient using softmax weights w"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Compute gradient contributions for each shift
    for r in range(len(w)):
        s = r - (n - 1)
        ws = w[r]
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:
            # s < 0
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad
