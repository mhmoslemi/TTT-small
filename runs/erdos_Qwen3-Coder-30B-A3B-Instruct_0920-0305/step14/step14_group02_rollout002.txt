import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlations(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
    return corr

def gradient_correlation_s(h, s):
    """Compute gradient of correlation for shift s"""
    n = len(h)
    grad = np.zeros(n)
    
    if s >= 0:
        # j from 0 to n-1-s
        for j in range(n - s):
            grad[j] += 1 - h[j + s]
        # j from s to n-1
        for j in range(s, n):
            grad[j] -= h[j - s]
    else:
        # j from 0 to n+ s-1
        for j in range(n + s):
            grad[j] -= h[j - s]
        # j from -s to n-1
        for j in range(-s, n):
            grad[j] += 1 - h[j + s]
    
    return grad

def top_k_indices(corr, k):
    """Get indices of top k largest values in corr"""
    return np.argpartition(corr, -k)[-k:]

def evaluate_c5(h):
    """Evaluate C5 using FFT method"""
    corr = compute_correlations(h)
    return np.max(corr) * (2.0 / len(h))
