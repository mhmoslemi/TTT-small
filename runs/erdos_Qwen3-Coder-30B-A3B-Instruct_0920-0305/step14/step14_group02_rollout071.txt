import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient vector for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s = -s
        g[s:] += 1 - h[:N-s]
        g[:N-s] -= h[s:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max using top-K shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert to exponentials with numerical stability
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N-1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart"""
    # Random initial h
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    T = T0
    step = step0
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    for it in range(max_iter):
        # Update step size (diminishing)
        step = step0 / np.sqrt(it + 1)
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        # Update
        h = h - step * g
        # Enforce symmetry if requested
        if sym:
            h = (h + h[::-1]) / 2
        # Project
        h = project_onto_capped_simplex(h, target_sum)
        # Evaluate
        c5 = evaluate_c5_fast(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Decay temperature
        T *= 0.9995
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local hill-climbing using pairwise swaps"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    patience = 0
    max_patience = 10000
    
    while time.time() - start < time_limit and patience < max_patience:
        i, j = np.random.choice(N, 2, replace=False)
        # Sample delta from Uniform(-d, d)
        delta = np.random.uniform(-0.1, 0.1)
        # Ensure h_i + delta in [0,1] and h_j - delta in [0,1]
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = np.clip(delta, delta_low, delta_high)
        if delta == 0:
            continue
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # No need for projection; sum preserved.
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_h = None
    best_c5 = float('inf')
    
    # Coarse optimization (n=60) for faster initial exploration
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    # Run coarse optimization
    coarse_h, coarse_c5 = gradient_descent_restart(
        n=n_coarse, 
        target_sum=target_sum_coarse, 
        max_iter=500, 
        step0=0.5, 
        T0=0.1, 
        top_k=8, 
        sym=False
    )
    # Interpolate to higher resolution
    x_coarse = np.linspace(dx_coarse/2, 2 - dx_coarse/2, n_coarse)
    x_fine = np.linspace(dx/2, 2 - dx/2, n_points)
    coarse_h_interp = np.interp(x_fine, x_coarse, coarse_h)
    coarse_h_interp = project_onto_capped_simplex(coarse_h_interp, target_sum)
    
    # Use the interpolated solution as a starting point for fine optimization
    fine_h, fine_c5 = gradient_descent_restart(
        n=n_points, 
        target_sum=target_sum, 
        max_iter=2000, 
        step0=0.5, 
        T0=0.1, 
        top_k=10, 
        sym=False  # Try both symmetric and asymmetric
    )
    
    if fine_c5 < best_c5:
        best_c5 = fine_c5
        best_h = fine_h.copy()
    
    # Try symmetric version
    sym_h, sym_c5 = gradient_descent_restart(
        n=n_points, 
        target_sum=target_sum, 
        max_iter=2000, 
        step0=0.5, 
        T0=0.1, 
        top_k=10, 
        sym=True
    )
    
    if sym_c5 < best_c5:
        best_c5 = sym_c5
        best_h = sym_h.copy()
    
    # Try the triangular initial vector
    x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
    triangular_h = np.maximum(0, 1 - np.abs(x - 1))
    triangular_h = project_onto_capped_simplex(triangular_h, target_sum)
    
    tri_h, tri_c5 = gradient_descent_restart(
        n=n_points, 
        target_sum=target_sum, 
        max_iter=1500, 
        step0=0.5, 
        T0=0.1, 
        top_k=10, 
        sym=False
    )
    
    if tri_c5 < best_c5:
        best_c5 = tri_c5
        best_h = tri_h.copy()
    
    # Try symmetric version of triangular
    tri_sym_h, tri_sym_c5 = gradient_descent_restart(
        n=n_points, 
        target_sum=target_sum, 
        max_iter=1500, 
        step0=0.5, 
        T0=0.1, 
        top_k=10, 
        sym=True
    )
    
    if tri_sym_c5 < best_c5:
        best_c5 = tri_sym_c5
        best_h = tri_sym_h.copy()
    
    # Additional restarts with random starts
    for i in range(8):
        if time.time() - start_time > budget_s * 0.85:
            break
        h_random = np.random.rand(n_points)
        h_random = project_onto_capped_simplex(h_random, target_sum)
        random_h, random_c5 = gradient_descent_restart(
            n=n_points, 
            target_sum=target_sum, 
            max_iter=1500, 
            step0=0.5, 
            T0=0.1, 
            top_k=10, 
            sym=False
        )
        
        if random_c5 < best_c5:
            best_c5 = random_c5
            best_h = random_h.copy()
    
    # Final local refinement using swap moves
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time)
    
    # Final projection to ensure constraints
    best_h = project_onto_capped_simplex(best_h, target_sum)
    best_c5 = float(evaluate_c5_fast(best_h))
    
    return (best_h.tolist(), best_c5, n_points)
