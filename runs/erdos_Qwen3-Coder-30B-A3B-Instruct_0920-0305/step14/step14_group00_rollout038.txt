    dx = 2.0 / n_points

    correlations = []
    for k in range(1, n_points):
        # Shift k: h(x) * (1 - h(x+k))
        corr = np.sum(h[:-k] * (1 - h[k:])) * dx
        correlations.append(corr)
    
    # For shift 0: h(x) * (1 - h(x)) 
    corr0 = np.sum(h * (1 - h)) * dx
    correlations.append(corr0)
    
    return max(correlations)
