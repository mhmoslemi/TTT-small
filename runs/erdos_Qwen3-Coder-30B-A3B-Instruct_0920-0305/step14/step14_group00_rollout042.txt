def evaluate_h(h_values, n_points):
    """Evaluate the C5 bound for given h values"""
    dx = 2.0 / n_points

    corr = correlate(h_values, 1 - np.array(h_values), mode='full')
    # Only consider the valid correlations (shifts from 1 to n-1)
    valid_corr = corr[len(h_values)-1:]
    c5_bound = max(valid_corr) * dx
    return c5_bound
