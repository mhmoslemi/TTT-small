import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using FFT correlation"""
    corr = correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort x in descending order and compute cumulative sums
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    
    # Find the threshold index
    n = len(x)
    threshold_idx = -1
    for i in range(n):
        if cumsum[i] > target_sum:
            threshold_idx = i
            break
    
    if threshold_idx == -1:
        # All elements can be set to 1
        return np.clip(x, 0, 1)
    elif threshold_idx == 0:
        # Need to scale down
        lambda_val = target_sum / n
        result = np.clip(x - lambda_val, 0, 1)
        return result
    else:
        # Use water-filling algorithm
        lambda_val = (target_sum - cumsum[threshold_idx-1]) / (n - threshold_idx)
        result = np.clip(x - lambda_val, 0, 1)
        return result

def smooth_max(h, shifts, T=0.05):
    """Evaluate log-sum-exp approximation to max over shifts"""
    c_s = []
    for s in shifts:
        if s >= 0:
            corr = np.sum(h[:-s] * (1 - h[s:])) * dx if s < len(h) else 0
        else:
            corr = np.sum(h[-s:] * (1 - h[:-s])) * dx if -s < len(h) else 0
        c_s.append(corr)
    
    c_s = np.array(c_s)
    # Subtract maximum for numerical stability
    max_c = np.max(c_s)
    exp_vals = np.exp((c_s - max_c) / T)
    return dx * (max_c + T * np.log(np.sum(exp_vals)))

def gradient_smooth_max(h, shifts, T=0.05):
    """Evaluate gradient of log-sum-exp approximation to max over shifts"""
    c_s = []
    grad_c_s = []
    
    for s in shifts:
        if s >= 0:
            corr = np.sum(h[:-s] * (1 - h[s:])) * dx if s < len(h) else 0
            grad = np.zeros_like(h)
            grad[:-s] += (1 - h[s:]) * dx
            grad[s:] -= h[:-s] * dx
        else:
            corr = np.sum(h[-s:] * (1 - h[:-s])) * dx if -s < len(h) else 0
            grad = np.zeros_like(h)
            grad[-s:] += (1 - h[:-s]) * dx
            grad[:-s] -= h[-s:] * dx
        c_s.append(corr)
        grad_c_s.append(grad)
    
    c_s = np.array(c_s)
    # Compute weights
    max_c = np.max(c_s)
    exp_vals = np.exp((c_s - max_c) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        grad += w * grad_c_s[i]
    return grad

def lbfgs_optimization(h0, T_init=0.12, T_decay=0.998, max_iter=3000):
    """Perform L-BFGS optimization with temperature continuation"""
    h = h0.copy()
    T = T_init
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Top K shifts to consider
    top_shifts = list(range(-n_points+1, n_points))
    
    for i in range(max_iter):
        # Compute smooth max and its gradient
        c5_smooth = smooth_max(h, top_shifts, T)
        
        # Gradient with respect to smooth max
        grad = gradient_smooth_max(h, top_shifts, T)
        
        # L-BFGS step (use scipy minimize with L-BFGS-B)
        result = minimize(
            lambda x: smooth_max(x, top_shifts, T),
            h,
            method='L-BFGS-B',
            jac=lambda x: gradient_smooth_max(x, top_shifts, T),
            bounds=[(0, 1)] * len(h),
            options={'maxiter': 1, 'disp': False}
        )
        
        h_new = result.x
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate exact C5
        c5_exact = evaluate_c5_exact(h_new)
        
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h_new.copy()
        
        # Update h
        h = h_new
        
        # Decrease temperature
        T *= T_decay
        if T < 5e-6:
            break
    
    return best_h, best_c5

def pairwise_exchange_optimize(h, max_iter=5000):
    """Perform exact pairwise exchange with line search"""
    h = h.copy()
    best_c5 = evaluate_c5_exact(h)
    
    # Keep track of correlation values
    corr_values = correlate(h, 1-h, mode='full') * dx
    
    for _ in range(max_iter):
        improved = False
        # Try pairs in random order
        indices = list(range(len(h)))
        np.random.shuffle(indices)
        
        for i in indices:
            # For each i, consider moving mass to all j
            for j in range(len(h)):
                if i == j:
                    continue
                
                # Compute the change in correlation for each shift
                # We will compute the effect of moving delta from i to j
                max_corr_before = np.max(corr_values)
                
                # Determine feasible range for delta
                delta_min = max(-h[j], h[i] - 1)
                delta_max = min(h[i], 1 - h[j])
                if delta_min >= delta_max:
                    continue
                
                # Compute the change in each correlation
                delta_corr = np.zeros(len(corr_values))
                
                # For each shift s, compute how much it changes when we move delta from i to j
                for s in range(-n_points+1, n_points):
                    if s >= 0:
                        if s < len(h) and i+s < len(h):
                            if s < len(h) and i < len(h) and i+s < len(h):
                                # Contribution to shift s: h[i] * (1 - h[i+s]) - (h[i]-delta) * (1 - h[i+s]) = delta * (1 - h[i+s])
                                delta_corr[n_points-1+s] += delta * (1 - h[i+s])
                                # Contribution from j: h[j] * (1 - h[j-s]) - (h[j]+delta) * (1 - h[j-s]) = -delta * (1 - h[j-s])
                                if j-s >= 0 and j < len(h):
                                    delta_corr[n_points-1+s] -= delta * (1 - h[j-s])
                        if i >= s and i-s < len(h):
                            # h[i-s] * (1 - h[i]) - (h[i-s]) * (1 - (h[i]-delta)) = delta * h[i-s]
                            delta_corr[n_points-1+s] += delta * h[i-s]
                # Compute the new correlation values with a small delta
                new_corr_values = corr_values + delta_corr
                
                # Ensure that we use the correct index mapping
                new_max_corr = np.max(new_corr_values)
                
                if new_max_corr < max_corr_before - 1e-12:
                    # Accept the move
                    h[i] -= delta
                    h[j] += delta
                    corr_values = new_corr_values
                    improved = True
                    best_c5 = new_max_corr
                    break
            
            if improved:
                break
                
        if not improved:
            break
    
    return h, best_c5
