import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 using FFT"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of coarse solution to finer grid"""
    old_positions = np.linspace(0, 2, len(old_h))
    new_positions = np.linspace(0, 2, new_n)
    new_h = np.interp(new_positions, old_positions, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(corr):
    """Return the shift s achieving maximum correlation"""
    idx = np.argmax(corr)
    s = idx - (len(corr) // 2)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    g = 1 - h_plus - h_minus
    return g

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        nonlocal TARGET_SUM
        TARGET_SUM = target_sum(n)
        dx_val = dx(n)
        
        # Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, TARGET_SUM)
        else:
            h = interpolate(best_h_stage, n)
            h = project_to_sum_box(h, TARGET_SUM)
            
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Simulated annealing
        T = 0.02
        delta = 0.1
        alpha = 0.9997
        beta = 0.9999
        max_iter = int(100000 * time_fraction / 0.35)
        
        iter_count = 0
        accepted_moves = 0
        T_min = 1e-6
        
        while time.time() < start_time + budget_s * time_fraction and iter_count < max_iter:
            iter_count += 1
            
            # Determine move type
            if random.random() < 0.3:
                # Targeted move - focus on worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Pick from lowest and highest desirability
                low_indices = np.argsort(d)[:max(1, len(d) // 5)]
                high_indices = np.argsort(d)[-max(1, len(d) // 5):]
                
                i = random.choice(low_indices)
                j = random.choice(high_indices)
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                j = random.choice([x for x in range(n) if x != i])
                
            # Make move
            max_delta = min(delta, h[i], 1 - h[j])
            if max_delta <= 0:
                continue
                
            delta_move = random.uniform(0, max_delta)
            h_new = h.copy()
            h_new[i] -= delta_move
            h_new[j] += delta_move
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize 
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate new state
            c_new = C5_value(h_new)
            c_old = C5_value(h)
            
            # Acceptance criterion
            if c_new <= c_old or random.random() < np.exp(-(c_new - c_old) / T):
                h = h_new
                if c_new < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = c_new
                accepted_moves += 1
                
            # Update temperature and delta
            if accepted_moves > 0 and accepted_moves % 10 == 0:
                T = max(T_min, T * alpha)
                delta = delta * beta
                
        # Polishing phase
        h_polish = best_h_stage.copy()
        best_C_polish = best_C_stage
        
        for k in range(2000):
            if time.time() >= start_time + budget_s * time_fraction:
                break
                
            c = corr_fft(h_polish)
            max_corr = np.max(c)
            epsilon = 1e-12
            
            # Find all shifts that achieve max correlation
            s_indices = np.where(np.abs(c - max_corr) < epsilon)[0]
            s_values = [s_idx - (len(c) // 2) for s_idx in s_indices]
            
            # Compute average gradient
            grad = np.zeros(len(h_polish))
            for s in s_values:
                grad += gradient_worst(h_polish, s)
            grad /= len(s_values)
            
            # Update step size
            eta = 0.5 / np.sqrt(k + 1)
            
            # Update and project
            h_new = h_polish - eta * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate and accept if improvement
            c_new = C5_value(h_new)
            if c_new < best_C_polish:
                best_C_polish = c_new
                h_polish = h_new.copy()
                
        return best_h_stage, best_C_stage
    
    # Run stages
    best_h, best_C = make_stage(80, 0.30)
    best_h, best_C = make_stage(160, 0.35)
    best_h, best_C = make_stage(240, 0.35)
    
    return (best_h.tolist(), float(best_C), len(best_h))
