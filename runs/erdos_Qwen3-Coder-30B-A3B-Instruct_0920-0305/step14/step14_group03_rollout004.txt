def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""

        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10

        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break

            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    shift = max_idx - len(h) + 1  # Convert index to actual shift

    grad = np.zeros_like(h)

        if 0 <= i - shift < len(h):
            grad[i] += (1 - h[i - shift])
    
    # Since we're minimizing, this should be negative gradient
    return -grad
