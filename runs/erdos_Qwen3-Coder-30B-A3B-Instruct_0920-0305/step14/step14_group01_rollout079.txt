import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    x = np.array(x)
    n = len(x)
    
    # Binary search for the water level
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_smooth_max_gradient(h, epsilon=0.02):
    """Compute gradient of log-sum-exp approximation to max correlation"""
    # Compute correlation vector
    corr = compute_correlation(h)
    
    # Compute weights for log-sum-exp
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradient using convolution
    h_rev = h[::-1]
    b = 1 - h
    
    # Convolution of weights with b
    L = 2 * len(h) - 1
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # Convolution of weights reversed with h
    w_rev = w[::-1]
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Combine gradients
    grad = conv1 - conv2
    
    # Extract central part
    grad = grad[len(h)-1:2*len(h)-1]
    
    return grad

def compute_objective_and_gradient(u, epsilon=0.02, lambda_inc=1e3, mu=1e4):
    """Compute smooth objective and its gradient"""
    # Reconstruct full h from symmetric u
    h = np.concatenate([u, u[::-1]])
    
    # Add constraints
    # Monotonicity penalty
    pen_inc = lambda_inc * np.sum(np.maximum(0, np.diff(u))**2)
    
    # Sum constraint penalty
    sum_penalty = mu * (np.sum(u) - target_sum/2)**2
    total_penalty = pen_inc + sum_penalty
    
    # Compute smooth max
    corr = compute_correlation(h)
    smooth_max = epsilon * np.log(np.sum(np.exp(corr/epsilon)))
    
    # Total objective
    f = smooth_max + total_penalty
    
    # Compute gradient
    grad_h = compute_smooth_max_gradient(h, epsilon)
    
    # Gradient wrt u (since h = [u, u[::-1]])
    grad_u = grad_h[:len(u)] + grad_h[-len(u):][::-1]
    
    # Add penalty gradients
    grad_u += 2 * lambda_inc * np.concatenate([np.diff(u), [0]])  
    grad_u += 2 * mu * (np.sum(u) - target_sum/2) * np.ones_like(u)
    
    return f, grad_u

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Epsilon schedule
    epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    
    # Multiple restarts
    restarts = 0
    max_restarts = 10
    
    while time.time() - start_time < 0.95 * budget_s and restarts < max_restarts:
        restarts += 1
        
        # Different initial seeds
        init_methods = [
            lambda: np.full(120, 0.5),  # Uniform
            lambda: np.minimum(np.arange(120), 120 - np.arange(120)),  # Triangular
            lambda: (1 + np.cos(np.pi * (np.arange(120) / 119.0 - 1))) / 2,  # Cosine-squared
        ]
        
        # Try a few different initializations
        for i, init_func in enumerate(init_methods):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Initial vector
            u0 = init_func()
            
            # Project to feasible set (box + sum)
            u0 = project_to_box_sum(u0, target_sum/2)
            
            best_u = u0.copy()
            best_c5_iter = float('inf')
            
            # Continuation on epsilon
            for epsilon in epsilon_schedule:
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Optimize with L-BFGS
                res = minimize(
                    lambda u: compute_objective_and_gradient(u, epsilon),
                    u0,
                    method='L-BFGS-B',
                    bounds=[(0, 1)] * len(u0),
                    jac=lambda u: compute_objective_and_gradient(u, epsilon)[1],
                    options={'maxiter': 250, 'disp': False}
                )
                
                u_final = res.x
                u_final = project_to_box_sum(u_final, target_sum/2)
                
                # Reconstruct full h
                h_final = np.concatenate([u_final, u_final[::-1]])
                
                # Evaluate true C5
                c5 = max(compute_correlation(h_final))
                
                if c5 < best_c5_iter:
                    best_c5_iter = c5
                    best_u = u_final.copy()
                
                u0 = u_final.copy()
                
            # Final refinement with subgradient method
            h = np.concatenate([best_u, best_u[::-1]])
            for _ in range(500):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Find worst shift
                corr = compute_correlation(h)  
                j_star = np.argmax(corr)
                
                # Compute subgradient for this shift in h
                grad = np.zeros_like(h)
                for i in range(len(h)):
                    if i + j_star < len(h):
                        grad[i] += (1 - h[i + j_star])
                    if i - j_star >= 0:
                        grad[i] -= h[i - j_star]
                
                # Steplength schedule
                alpha = 0.01 / np.sqrt(_ + 1)
                
                # Update h
                h_new = h - alpha * grad
                h_new = project_to_box_sum(h_new, target_sum)
                
                # Evaluate new C5
                c5_new = max(compute_correlation(h_new))
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                
                h = h_new.copy()
    
    if best_h is None:
        # Fallback: use cosine-squared initialization
        u0 = (1 + np.cos(np.pi * (np.arange(120) / 119.0 - 1))) / 2
        u0 = project_to_box_sum(u0, target_sum/2)
        h = np.concatenate([u0, u0[::-1]])
        best_c5 = max(compute_correlation(h))
        best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
