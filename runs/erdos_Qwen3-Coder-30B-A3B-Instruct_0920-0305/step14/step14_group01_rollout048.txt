```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

def project_to_box_sum(u, target_sum):
    """Project u onto box constraints [0,1] with sum = target_sum using water-filling"""
    u = np.array(u)
    if np.sum(u) == target_sum:
        return u.copy()
    
    # Water-filling algorithm
    sorted_u = np.sort(u)[::-1]
    cumsum = np.cumsum(sorted_u)
    n = len(u)
    
    # Find the cutoff index
    cutoff = 0
    for i in range(n):
        if sorted_u[i] * (i+1) > cumsum[i] - target_sum:
            cutoff = i
            break
    else:
        cutoff = n
    
    # Compute the threshold lambda
    if cutoff == 0:
        lmbda = (np.sum(u) - target_sum) / n
        result = np.maximum(u - lmbda, 0)
    else:
        lmbda = (cumsum[cutoff-1] - target_sum) / cutoff
        result = np.maximum(u - lmbda, 0)
        result = np.minimum(result, 1)
    
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon):
    """Compute gradient of smooth max using FFT"""
    corr = compute_correlation(h)
    
    # Compute weights
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Convolution for gradient
    L = len(h) * 2 - 1
    w_full = np.zeros(L)
    w_full[:len(w)] = w
    
    h_rev = h[::-1]
    b = 1 - h
    
    # Two convolutions
    conv1 = np.fft.ifft(np.fft.fft(w_full, L) * np.fft.fft(b, L)).real
    conv2 = np.fft.ifft(np.fft.fft(w_full[::-1], L) * np.fft.fft(h, L)).real
    
    grad = conv1 - conv2
    grad = grad[len(h)-1:2*len(h)-1]  # Extract central part
    
    return grad * dx

def compute_objective_and_gradient(u, epsilon, lambda_inc=1e3, mu=1e4):
    """Compute objective value and gradient for the smooth surrogate"""
    # Reconstruct full h vector
    h = np.concatenate([u, u[::-1]])
    
    # Compute smooth max gradient
    grad_smooth = compute_smooth_max_gradient(h, epsilon)
    
    # Add monotonicity penalty gradient
    grad_mono = np.zeros_like(u)
    for i in range(1, len(u)):
        if u[i-1] < u[i]:
            grad_mono[i-1] += 2 * lambda_inc * (u[i-1] - u[i])
            grad_mono[i] += 2 * lambda_inc * (u[i] - u[i-1])
    
    # Add sum constraint penalty gradient
    penalty_grad = 2 * mu * (np.sum(u) - target_sum)
    
    total_grad = grad_smooth[:len(u)] + grad_mono + penalty_grad
    
    # Compute objective value
    corr = compute_correlation(h)
    smooth_obj = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    
    return smooth_obj, total_grad

def evaluate_c5(h):
    """Evaluate the true C5 using the original routine"""
    return np.max(np.correlate(h, 1-h, mode="full")) * dx

def generate_initial_seeds():
    """Generate several initial seeds"""
    seeds = []
    
    # Uniform seed
    seeds.append(np.full(n_points // 2, 0.5))
    
    # Triangular seed
    x = np.linspace(0, 1, n_points // 2)
    tri = 1 - np.abs(2*x - 1)
    seeds.append(tri)
    
    # Cosine-squared seed
    x = np.linspace(0, 1, n_points // 2)
    cos2 = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cos2)
    
    # Random monotone seed
    rand = np.random.rand(n_points // 2)
    rand_sorted = np.sort(rand)
    seeds.append(rand_sorted)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial seeds
    initial_seeds = generate_initial_seeds()
    
    # Main optimization loop
    for seed_idx, u0 in enumerate(initial_seeds):
        # Copy initial seed and project onto valid set
        u = project_to_box_sum(u0, target_sum)
        
        for epsilon in epsilon_schedule:
            # Check time budget
            if time.time() - start_time > 0.95 * budget_s:
                break
            
            # Run L-BFGS optimization
            result = minimize(
                lambda u: compute_objective_and_gradient(u, epsilon)[0],
                u,
                method='L-BFGS-B',
                bounds=[(0, 1)] * len(u),
                jac=lambda u: compute_objective_and_gradient(u, epsilon)[1],
                options={'maxiter': 250, 'disp': False}
            )
            
            # Update u after optimization
            u = result.x
            u = project_to_box_sum(u, target_sum)
            
            # Evaluate using true C5
            h_full = np.concatenate([u, u[::-1]])
            current_c5 = evaluate_c5(h_full)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_full.copy()
        
        # Early exit if we've used most time
        if time.time() - start_time > 0.95 * budget_s:
            break
    
    # Final refinement using subgradient descent on the true objective
    if best_h is not None and time.time() - start_time < 0.99 * budget_s:
        h = best_h.copy()
        alpha0 = 0.05
        max_iter = 2000
        
        for i in range(max_iter):
            if time.time() - start_time > 0.99 * budget_s:
                break
                
            # Compute correlations and find worst shift
            corr = compute_correlation(h)
            worst_shift = np.argmax(corr)
            
            # Compute gradient for the worst shift
            grad = np.zeros_like(h)
            if worst_shift < len(h):
                grad[worst_shift:] += (1 - h[:-worst_shift])
            if worst_shift < len(h):
                grad[:-worst_shift] += -h[worst_shift:]
            
            # Update
            alpha_t = alpha0 / np.sqrt(i + 1)
            h_new = h - alpha_t * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_box_sum(h_new, target_sum)
            
            # Evaluate new value
            current_c5 = evaluate_c5(h_new)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
            h = h_new
    
    # Return final result
    if best_h is None:
        # Fallback
        h_full = np.ones(n_points)
        h_full = project_to_box_sum(h_full, target_sum)
        best_c5 = evaluate_c5(h_full)
        best_h = h_full
    
    return (best_h.tolist(), float(best_c5), n_points)
```
