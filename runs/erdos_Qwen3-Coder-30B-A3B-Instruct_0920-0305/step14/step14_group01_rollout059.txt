import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
L = 2 * n_points - 1

def project_to_box_sum(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling approach
    x = np.array(x)
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    
    # Find the threshold
    threshold_idx = -1
    for i in range(n):
        if cumsum[i] <= target_sum:
            threshold_idx = i
        else:
            break
    
    if threshold_idx == -1:
        # All elements are too large
        threshold = sorted_x[0] - (cumsum[0] - target_sum) / n
    else:
        # Some elements are below the threshold
        if threshold_idx == n - 1:
            threshold = (target_sum - cumsum[threshold_idx]) / (n - threshold_idx - 1)
        else:
            threshold = sorted_x[threshold_idx] - (cumsum[threshold_idx] - target_sum) / (threshold_idx + 1)
    
    # Apply thresholding and renormalize if needed
    result = np.clip(x, 0, 1)
    current_sum = np.sum(result)
    
    if abs(current_sum - target_sum) < 1e-10:
        return result
    
    # Adjust proportionally to get exact sum
    if current_sum > target_sum:
        # Reduce values
        diff = current_sum - target_sum
        scale = (current_sum - diff) / current_sum
        result *= scale
    else:
        # Increase values
        diff = target_sum - current_sum
        scale = (target_sum - diff) / target_sum
        result *= scale
    
    # Final adjust to get exactly target
    adjustment = target_sum - np.sum(result)
    if np.abs(adjustment) > 1e-12:
        # Add adjustment to the largest elements
        indices = np.argsort(result)[::-1][:int(np.ceil(np.abs(adjustment) / 1e-12))]
        for i in indices:
            if adjustment > 0 and result[i] < 1:
                add = min(adjustment, 1 - result[i])
                result[i] += add
                adjustment -= add
            elif adjustment < 0 and result[i] > 0:
                sub = min(-adjustment, result[i])
                result[i] -= sub
                adjustment += sub
            if np.abs(adjustment) < 1e-12:
                break
    
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max_gradient(u, epsilon=0.01):
    """Compute gradient of smooth max over correlations"""
    # Reconstruct full h
    h = np.concatenate([u, u[::-1]])
    
    # Compute correlation
    corr = compute_correlation(h)
    
    # Compute weights for smooth max
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Compute gradient via convolution
    b = 1 - h
    w_rev = w[::-1]
    
    # conv(w, b)
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    # conv(w_rev, h)
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Combine gradients
    grad = conv1 - conv2
    
    # Extract central portion (matching the length of h)
    grad = grad[n_points-1:2*n_points-1]
    
    # Scale by dx
    grad = dx * grad
    
    # Add monotonicity penalty gradient
    penalty_grad = np.zeros_like(u)
    for i in range(1, len(u)):
        penalty_grad[i-1] -= 2 * (u[i-1] - u[i])  # Decrease forces앞
        penalty_grad[i] += 2 * (u[i-1] - u[i])   # Increase force
        
    # Apply a large penalty for monotonic increase
    penalty_grad *= 1e3
    
    # Handle final correction
    final_grad = np.zeros(len(u))
    final_grad[:len(penalty_grad)] += penalty_grad
    
    return grad[:len(u)] + final_grad

def smooth_max_objective(u, epsilon=0.01):
    """Evaluate smooth max objective"""
    # Reconstruct full h
    h = np.concatenate([u, u[::-1]])
    
    # Compute correlation
    corr = compute_correlation(h)
    
    # Compute smooth max
    corr_scaled = corr / epsilon
    max_corr = epsilon * np.log(np.sum(np.exp(corr_scaled)))
    
    # Add penalty for sum constraint
    sum_penalty = 1e4 * (np.sum(u) - target_sum)**2
    
    # Add monotonicity penalty
    mono_penalty = 1e3 * sum((u[i-1] - u[i])**2 for i in range(1, len(u)))
    
    return max_corr + sum_penalty + mono_penalty

def evaluate_c5(h_values):
    """Use the pre-imported evaluation function"""
    corr = compute_correlation(h_values)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solutions to try
    initial_solutions = []
    
    # 1. Uniform
    h_uniform = np.full(n_points // 2, 0.5)
    initial_solutions.append(h_uniform)
    
    # 2. Triangular
    x = np.linspace(0, 1, n_points // 2)
    h_triangular = np.minimum(x, 1 - x) * 2
    initial_solutions.append(h_triangular)
    
    # 3. Cosine-squared
    x = np.linspace(0, 1, n_points // 2)
    h_cos = (1 + np.cos(np.pi * x)) / 2
    initial_solutions.append(h_cos)
    
    # 4. Random monotone
    h_random = np.sort(np.random.rand(n_points // 2))
    h_random = project_to_box_sum(h_random, target_sum)
    initial_solutions.append(h_random)
    
    best_c5 = float('inf')
    best_h = None
    
    # ε schedule
    eps_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    
    # Random restarts
    num_restarts = 0
    max_restarts = 10  # Limit number of restarts
    
    while time.time() - start_time < 0.95 * budget_s and num_restarts < max_restarts:
        num_restarts += 1
        
        # Choose a random initial solution
        init_idx = np.random.randint(len(initial_solutions))
        u = initial_solutions[init_idx].copy()
        
        # Optimize using smooth max with continuation
        for i, epsilon in enumerate(eps_schedule):
            if time.time() - start_time > 0.95 * budget_s:
                break
            
            def fun(u):
                return smooth_max_objective(u, epsilon)
            
            def jac(u):
                return smooth_max_gradient(u, epsilon)
            
            # Optimize with L-BFGS-B, limit to 200 iterations
            result = minimize(
                fun, 
                u, 
                method='L-BFGS-B', 
                jac=jac,
                bounds=[(0, 1)] * len(u),
                options={'maxiter': 200}
            )
            
            u = result.x
            
            # Project to feasible set
            u = project_to_box_sum(u, target_sum)
            
            # Evaluate the actual function
            h = np.concatenate([u, u[::-1]])
            current_c5 = evaluate_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early termination if we're close to target
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with exact max
    if best_h is not None:
        h = best_h.copy()
        max_iter = 5000
        alpha0 = 0.001
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.99:
                break
                
            corr = compute_correlation(h)
            worst_shift = np.argmax(corr)
            
            # Compute gradient for worst shift
            grad = np.zeros(n_points)
            for j in range(n_points):
                if j + worst_shift < n_points:
                    grad[j] += (1 - h[j + worst_shift])
                if j - worst_shift >= 0:
                    grad[j] -= h[j - worst_shift]
            
            # Update with adaptive learning rate
            alpha = alpha0 / np.sqrt(i + 1)
            h_new = h - alpha * grad
            
            # Clamp to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project to correct sum
            h_new = project_to_box_sum(h_new, target_sum)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    
    # If no good solution found, return the initial one
    if best_h is None:
        # Use the best initial solution
        u = initial_solutions[np.random.randint(len(initial_solutions))].copy()
        u = project_to_box_sum(u, target_sum)
        h = np.concatenate([u, u[::-1]])
        best_c5 = evaluate_c5(h)
        best_h = h
    
    return (best_h.tolist(), float(best_c5), n_points)
