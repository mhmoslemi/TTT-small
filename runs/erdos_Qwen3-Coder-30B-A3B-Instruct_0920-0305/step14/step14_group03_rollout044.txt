def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use Duchi's algorithm for projection
    # Sort x in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the threshold
    threshold = 0
    for i in range(n):
        if sorted_x[i] - (cumsum[i] - target_sum) / (i + 1) > 0:
            threshold = sorted_x[i] - (cumsum[i] - target_sum) / (i + 1)
        else:
            break
    
    # Project
    projected = np.maximum(x - threshold, 0)
    projected = np.minimum(projected, 1)
    
    # Adjust to get exact sum
    current_sum = np.sum(projected)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices where we can add diff
        idxs = np.argsort(projected)
        for i in range(len(idxs)):
            if projected[idxs[i]] < 1 and diff > 0:
                add = min(diff, 1 - projected[idxs[i]])
                projected[idxs[i]] += add
                diff -= add
            if diff <= 0:
                break
    return projected

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient(h, corr):
    """Compute gradient using the worst shift contribution"""
    worst_shift_idx = np.argmax(corr)
    shift = worst_shift_idx - len(h) + 1
    grad = np.zeros_like(h)
    
    # Back-projection from the worst shift
    for i in range(len(h)):
        # Add contribution to entry i
        if 0 <= i-shift < len(h):
            grad[i] -= h[i-shift]
        if 0 <= i+shift < len(h):
            grad[i] -= h[i+shift] 
    return grad
