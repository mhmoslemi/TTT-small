import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
MAX_ITER_LBFGS = 2000
TEMPERATURE_START = 0.12
TEMPERATURE_DECAY = 0.998
TEMPERATURE_MIN = 5e-6
MAX_PAIRWISE_SWEEPS = 50
PROJECTED_GRADIENT_TOLERANCE = 1e-6

def evaluate_c5_exact(h):
    """Evaluate the exact C5 using full FFT correlation"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, temperature=0.01):
    """Evaluate smooth surrogate of C5 using log-sum-exp"""
    # Compute all correlation values
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Use log-sum-exp to smooth the maximum
    corr_vals = corr[:len(corr)//2 + 1]  # Only positive shifts
    max_corr = np.max(corr_vals)
    shifted_corr = corr_vals / temperature
    # Handle numerical overflow
    shifted_corr = shifted_corr - max_corr / temperature
    log_sum_exp = max_corr + temperature * np.log(np.sum(np.exp(shifted_corr)))
    
    return dx * log_sum_exp

def gradient_c5_smooth(h, temperature=0.01):
    """Compute gradient of smooth C5 surrogate"""
    # Compute all correlation values
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Get positive correlation values
    corr_vals = corr[:len(corr)//2 + 1]
    
    # Compute softmax weights
    max_corr = np.max(corr_vals)
    shifted_corr = corr_vals / temperature
    shifted_corr = shifted_corr - max_corr / temperature
    weights = np.exp(shifted_corr)
    weights = weights / np.sum(weights)
    
    # Compute gradient of each correlation w.r.t h
    grad = np.zeros_like(h)
    
    for i, weight in enumerate(weights):
        if i == 0:  # No shift
            grad += weight * (1 - h)
        else:  # Positive shift
            shift = i
            grad[:len(h)-shift] += weight * (1 - h[shift:])
            grad[shift:] -= weight * h[:-shift]
    
    return dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm for projection onto simplex
    x_sorted = np.sort(x)[::-1]
    n = len(x)
    cumsum = np.cumsum(x_sorted)
    idx = np.arange(n)
    
    # Find the index where the cumulative sum crosses the target
    temp = (cumsum - target_sum) / (idx + 1)
    theta = np.max(temp[temp <= x_sorted])
    
    # Compute projected vector
    result = np.clip(x - theta, 0, 1)
    
    return result

def lbfgs_with_continuation(initial_h, max_iter=MAX_ITER_LBFGS):
    """L-BFGS with temperature continuation"""
    h = initial_h.copy()
    temperature = TEMPERATURE_START
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iter_num in range(max_iter):
        # Calculate current C5 value
        current_c5 = evaluate_c5_smooth(h, temperature)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = gradient_c5_smooth(h, temperature)
        
        # Update using projected L-BFGS step
        # Simple gradient descent step with projection
        h_new = h - 0.001 * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check convergence
        if np.linalg.norm(h_new - h) < PROJECTED_GRADIENT_TOLERANCE:
            break
            
        h = h_new
        
        # Decrease temperature
        temperature = max(TEMPERATURE_MIN, temperature * TEMPERATURE_DECAY)
        
        # Early stopping if temperature is too small
        if temperature <= TEMPERATURE_MIN:
            break
    
    return best_h, best_c5

def pairwise_exchange_polish(h_initial, max_sweeps=MAX_PAIRWISE_SWEEPS):
    """Polish using exact pairwise exchange with analytic line search"""
    h = h_initial.copy()
    best_c5 = evaluate_c5_exact(h)
    
    for sweep in range(max_sweeps):
        improved = False
        # Generate all pairs in random order
        pairs = [(i, j) for i in range(n_points) for j in range(i+1, n_points)]
        np.random.shuffle(pairs)
        
        for i, j in pairs[:200]:  # Limit number of pairs per sweep
            # Calculate the effect of moving mass from i to j
            h_copy = h.copy()
            old_corr = evaluate_c5_exact(h_copy)
            
            # Determine feasible range for δ
            delta_min = max(-h_copy[j], h_copy[i] - 1)
            delta_max = min(h_copy[i], 1 - h_copy[j])
            
            if delta_min >= delta_max:
                continue
            
            # Try several candidate values for δ
            candidate_deltas = np.linspace(delta_min, delta_max, 20)
            best_delta = 0
            best_new_c5 = old_corr
            
            for delta in candidate_deltas:
                h_test = h_copy.copy()
                h_test[i] -= delta
                h_test[j] += delta
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                
                new_c5 = evaluate_c5_exact(h_test)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_delta = delta
                    
            # Apply the best improvement
            if best_new_c5 < old_corr - 1e-8:
                h[i] -= best_delta
                h[j] += best_delta
                h = project_to_sum_box(h, TARGET_SUM)
                improved = True
                best_c5 = best_new_c5
        
        if not improved:
            break
    
    return h, best_c5
