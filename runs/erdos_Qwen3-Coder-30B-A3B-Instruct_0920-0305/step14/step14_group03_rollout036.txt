import numpy as np
from scipy.signal import correlate
from scipy.fft import ifft, fft
import time

def project_to_simplex_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_soft_max_gradient(h, corr, alpha):
    """Compute gradient of the soft-max approximation to max"""
    n = len(h)
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    
    grad = np.zeros(n)
    # Add contribution from each shift
    for s in range(-n+1, n):
        if s == 0:
            grad += -2 * w[s + n - 1] * h
        else:
            if s > 0:
                grad[:n-s] += -w[s + n - 1] * h[s:]
                grad[s:] += -w[s + n - 1] * h[:n-s]
            else:
                s_abs = -s
                grad[:n-s_abs] += -w[s + n - 1] * h[s_abs:]
                grad[s_abs:] += -w[s + n - 1] * h[:n-s_abs]
    
    return grad
