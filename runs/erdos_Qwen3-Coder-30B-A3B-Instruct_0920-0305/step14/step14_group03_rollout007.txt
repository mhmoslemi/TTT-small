import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def compute_corr(h):
    """Compute auto-correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = ifft(fft(h_rev, L) * fft(h, L)).real
    return corr

def compute_softmax_weights(corr_vals, alpha):
    """Compute softmax weights for the correlation values"""
    max_corr = np.max(corr_vals)
    if max_corr > 0:
        exp_vals = np.exp(alpha * (corr_vals - max_corr))
    else:
        exp_vals = np.exp(alpha * corr_vals)
    return exp_vals / np.sum(exp_vals)

def compute_gradient(h, corr, weights):
    """Compute exact gradient using softmax weights"""
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each shift, compute contribution to gradient
    for s in range(-n+1, n):
        if s == 0:
            grad -= 2 * weights[s + n - 1] * h
        elif s > 0:
            # Forward shift
            grad[:n-s] -= weights[s + n - 1] * h[s:]
            # Backward shift  
            grad[s:] -= weights[s + n - 1] * h[:n-s]
        else:  # s < 0
            # Forward shift
            grad[-s:] -= weights[s + n - 1] * h[:n+s]
            # Backward shift
            grad[:n+s] -= weights[s + n - 1] * h[-s:]
    
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Evaluate C5 using reference implementation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * dx
