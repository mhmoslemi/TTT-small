import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * (target_sum - corr)

def compute_gradient_from_w(h, w, dx):
    """Compute gradient using softmax weights"""
    n = len(h)
    grad = np.zeros(n)
    
    # For each shift, compute contributions
    for r in range(2*n-1):
        s = r - (n-1)
        ws = w[r]
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:
            # negative shift
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Try different values of n_points
    n_candidates = [120, 150, 180, 200]
    
    for n in n_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Set up variables for this n
        dx = 2.0 / n
        target_sum = n / 2
        
        # Multiple restarts for each n
        for restart in range(3):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Initialize with different strategies
            if restart == 0:
                # Use initial solution if available
                if 'initial_h_values' in globals() and len(initial_h_values) == n:
                    h0 = np.array(initial_h_values)
                else:
                    # Uniform initialization
                    h0 = np.full(n, 0.5)
            elif restart == 1:
                # Random initialization
                h0 = np.random.rand(n)
                h0 = h0 / np.sum(h0) * target_sum
            else:
                # Sinusoidal initialization
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h0 = 0.5 + 0.3 * np.cos(np.pi * x)
                h0 = np.clip(h0, 0, 1)
            
            # Project to sum constraint
            h0 = project_to_sum_box(h0, target_sum)
            
            # Optimization parameters
            current_h = h0.copy()
            current_c5 = evaluate_c5(current_h)
            
            # Track best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Adam optimizer parameters
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = np.zeros_like(current_h)
            v = np.zeros_like(current_h)
            
            # Optimization loop
            lr = 0.05
            alpha = 10.0
            t = 0
            
            for iteration in range(5000):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Compute overlap values and maximum
                O = compute_O(current_h, dx, target_sum)
                C5 = O.max()
                
                # Track best
                if C5 < best_c5:
                    best_c5 = C5
                    best_h = current_h.copy()
                
                # Compute softmax weights
                max_O = O.max()
                if max_O > 0:
                    exp_term = np.exp(alpha * (O - max_O))
                else:
                    exp_term = np.exp(alpha * O)
                w = exp_term / np.sum(exp_term)
                
                # Compute gradient
                grad = compute_gradient_from_w(current_h, w, dx)
                
                # Adam updates
                t += 1
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # Try update with backtracking line search
                step_factor = 1.0
                for _ in range(10):
                    new_h = current_h - step_factor * step
                    new_h = project_to_sum_box(new_h, target_sum)
                    new_O = compute_O(new_h, dx, target_sum)
                    new_C5 = new_O.max()
                    
                    if new_C5 < C5:
                        break
                    step_factor *= 0.5
                
                if new_C5 < C5:
                    current_h = new_h
                    current_c5 = new_C5
                else:
                    # Reduce learning rate
                    lr *= 0.5
                
                # Adaptive alpha
                if iteration % 200 == 0:
                    alpha = min(alpha * 1.05, 200.0)
                
                # Random perturbation occasionally
                if iteration % 500 == 0 and iteration > 0:
                    perturbation = np.random.normal(0, 0.001, n)
                    temp_h = current_h + perturbation
                    temp_h = project_to_sum_box(temp_h, target_sum)
                    temp_c5 = evaluate_c5(temp_h)
                    if temp_c5 < current_c5:
                        current_h = temp_h
                        current_c5 = temp_c5
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_c5(best_h)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # Return the initial solution as fallback
        h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
        c5_initial = evaluate_c5(h0)
        return (h0.tolist(), float(c5_initial), n_points)
