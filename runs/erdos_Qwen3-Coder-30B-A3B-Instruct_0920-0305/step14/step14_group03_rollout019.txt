import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    eps = 1e-4
    
    # Build index mapping for correlations
    def get_correlation_indices(n):
        # For each shift s in [-n+1, n-1], find valid pairs (i, i+s)
        shift_indices = {}
        for s in range(-n+1, n):
            pairs = []
            for i in range(n):
                j = i + s
                if 0 <= j < n:
                    pairs.append((i, j))
            shift_indices[s] = pairs
        return shift_indices
    
    # Linear feasibility formulation with McCormick relaxation
    def solve_feasibility(C_target):
        # Variables
        h = cp.Variable(n_points)
        w = cp.Variable((n_points, n_points), symmetric=False)
        
        # Constraints
        constraints = []
        
        # Box constraints
        constraints += [h >= 0, h <= 1]
        
        # Sum constraint
        constraints += [cp.sum(h) == target_sum]
        
        # McCormick relaxation constraints for w[i,j] = h[i] * h[j]
        shift_indices = get_correlation_indices(n_points)
        rhs = target_sum - C_target / dx
        
        for s in range(-n_points+1, n_points):
            if s in shift_indices:
                pairs = shift_indices[s]
                # Add constraint for each correlation term
                corr_expr = 0
                for i, j in pairs:
                    # w[i,j] <= h[i]
                    constraints.append(w[i,j] <= h[i])
                    # w[i,j] <= h[j]  
                    constraints.append(w[i,j] <= h[j])
                    # w[i,j] >= h[i] + h[j] - 1
                    constraints.append(w[i,j] >= h[i] + h[j] - 1)
                    corr_expr += w[i,j]
                # Constraint: correlation >= rhs
                constraints.append(corr_expr >= rhs)
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return None

        def f(lam):
            projected = np.clip(h + lam, 0, 1)

        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-10:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(h + (a + b)/2, 0, 1)
        return result

    def evaluate_c5(h):
        # Compute correlation using FFT
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return dx * np.max(corr)

    def subgradient_refinement(h0, max_iter=200):
        h = h0.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute all correlations
            corr = correlate(h, 1-h, mode='full')
            max_corr_idx = np.argmax(corr)
            max_corr_val = corr[max_corr_idx]
            
            # Identify worst shift
            worst_shift = max_corr_idx - len(h) + 1
            s = np.arange(len(h))
            t = s + worst_shift
            valid_mask = (t >= 0) & (t < len(h))
            i_indices = s[valid_mask]
            j_indices = t[valid_mask]
            
            # Compute subgradient
            g = np.zeros(len(h))
            for i, j in zip(i_indices, j_indices):
                # For shift s, gradient w.r.t h[i] is (1 - h[j])  
                g[i] += (1 - h[j])

            step_size = 1e-3
            new_h = h - step_size * g
            new_h = project_sum(new_h)
            
            # Accept improvement
            if evaluate_c5(new_h) < evaluate_c5(h):
                h = new_h
            else:
                # Small random jitter
                noise = np.random.normal(0, 1e-4, len(h))
                h = project_sum(h + noise)

        Cmid = (low + high) / 2
        h_feasible = solve_feasibility(Cmid)
        
        if h_feasible is not None:
            # Refine with sub-gradients and project to exact sum
            h_refined = subgradient_refinement(h_feasible, 100)
            c5_refined = evaluate_c5(h_refined)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
