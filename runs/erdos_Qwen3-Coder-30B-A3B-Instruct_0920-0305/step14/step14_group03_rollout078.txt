import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_linear_feasibility(h, n_points, dx, rhs):
    """Solve the linear feasibility problem using cvxpy"""
    # Variables
    h_var = cp.Variable(n_points)
    w_vars = cp.Variable((n_points, n_points), symmetric=True)
    
    # Constraints for McCormick envelopes
    constraints = []
    
    # Box constraints
    constraints.extend([h_var >= 0, h_var <= 1])
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # McCormick envelopes
    for i in range(n_points):
        for j in range(n_points):
            if abs(i - j) < n_points:  # Only consider valid shifts
                s = i - j
                if -n_points + 1 <= s <= n_points - 1:
                    # w[i,j] is h[i] * h[j] when shift s = i - j
                    constraints.append(w_vars[i,j] >= 0)
                    constraints.append(w_vars[i,j] <= h_var[i])
                    constraints.append(w_vars[i,j] <= h_var[j])
                    constraints.append(w_vars[i,j] >= h_var[i] + h_var[j] - 1)
    
    # Correlation constraints
    # For each shift, we need sum of w_{i,i+s} >= rhs
    for s in range(-n_points + 1, n_points):
        # Sum over valid i such that i+s is also valid
        if s >= 0:
            corr_sum = cp.sum([w_vars[i, i+s] for i in range(n_points-s)])
        else:
            corr_sum = cp.sum([w_vars[i+s, i] for i in range(-s, n_points)])
        
        constraints.append(corr_sum >= rhs)
    
    # Objective (dummy)
    prob = cp.Problem(cp.Minimize(0), constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INACCURATE]:
            return h_var.value
    except:
        pass
    
    return None
