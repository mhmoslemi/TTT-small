import numpy as np
from scipy.signal import correlate
import time
import random

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 with dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old grid to new grid"""
    old_positions = np.linspace(0, 2, len(old_h), endpoint=False) + dx(len(old_h))/2
    new_positions = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n)/2
    new_h = np.interp(new_positions, old_positions, old_h)
    return new_h

def worst_shift(c):
    """Return the shift corresponding to maximum correlation"""
    idx = np.argmax(c)
    s = idx - (len(c) - 1)
    return s

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient of the worst shift term"""
    h_plus, h_minus = shift_arrays(h, s)
    g = 1 - h_plus - h_minus
    return g

def make_stage(n, time_fraction, start_time, budget_s, prev_best_h=None):
    """Run one stage of the multi-resolution optimization"""
    global n_points, dx, target_sum
    
    # Update global variables for this stage
    n_points = n
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initialize
    if prev_best_h is None:
        h = np.random.uniform(0, 1, n)
        h = project_to_sum_box(h, target_sum)
    else:
        h = interpolate(prev_best_h, n)
        h = project_to_sum_box(h, target_sum)
    
    best_h_stage = h.copy()
    best_C_stage = C5_value(h)
    
    # Simulated Annealing parameters for this stage
    T = 0.02
    alpha = 0.9997
    beta = 0.9999
    delta = 0.1
    max_iter = int(80000 * time_fraction / budget_s)
    
    # Time management for this stage
    stage_end_time = start_time + time_fraction * budget_s
    
    # Annealing loop
    accepted_moves = 0
    total_moves = 0
    
    for iteration in range(max_iter):
        if time.time() >= stage_end_time:
            break
            
        total_moves += 1
        
        # Decide on move type
        if random.random() < 0.3:
            # Targeted move - improve worst shift
            c = corr_fft(h)
            s = worst_shift(c)
            h_plus, h_minus = shift_arrays(h, s)
            desirability = h_plus + h_minus
            
            # Select i from lowest 20% of desirability
            n_low = max(1, int(0.2 * n))
            low_indices = np.argpartition(desirability, n_low)[:n_low]
            i = random.choice(low_indices)
            
            # Select j from highest 20% of desirability  
            n_high = max(1, int(0.2 * n))
            high_indices = np.argpartition(-desirability, n_high)[:n_high]
            j = random.choice(high_indices)
            
            # Make transfer
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer > 0:
                delta_transfer = random.uniform(0, min(delta, max_transfer))
                h[i] -= delta_transfer
                h[j] += delta_transfer
                h = project_to_sum_box(h, target_sum)
                current_C = C5_value(h)
                
                # Accept or reject
                if current_C < best_C_stage:
                    best_C_stage = current_C
                    best_h_stage = h.copy()
        else:
            # Random move
            i, j = random.sample(range(n), 2)
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer > 0:
                delta_transfer = random.uniform(0, min(delta, max_transfer))
                h[i] -= delta_transfer
                h[j] += delta_transfer
                h = project_to_sum_box(h, target_sum)
                current_C = C5_value(h)
                
                # Accept or reject
                if current_C < best_C_stage:
                    best_C_stage = current_C
                    best_h_stage = h.copy()
                elif random.random() < np.exp(-(current_C - best_C_stage) / T):
                    # Accept worse solution with probability
                    pass  # keep current h (no change needed)
        
        # Update temperature and delta
        if total_moves % 10 == 0:
            T *= alpha
            delta *= beta
            
        # Occasionally reset temperature if too low
        if T < 1e-6 and accepted_moves > 0:
            T = 0.01
            delta = 0.1
            
        if total_moves % 100 == 0:
            # Check acceptance rate
            if total_moves > 1000 and accepted_moves / total_moves < 0.01:
                # Restart with higher temperature if acceptance is too low
                if T < 0.1:
                    T = 0.1
    
    # Polishing
    max_polish_iter = min(2000, int(5000 * time_fraction / budget_s))
    eta_0 = 0.5
    
    for k in range(max_polish_iter):
        if time.time() >= stage_end_time:
            break
            
        c = corr_fft(h)
        max_corr = np.max(c)
        epsilon = 1e-12
        s_indices = np.where(c >= max_corr - epsilon)[0]
        
        if len(s_indices) > 0:
            gradients = []
            for idx in s_indices:
                s = idx - (len(c) - 1)
                g = gradient_worst(h, s)
                gradients.append(g)
            
            avg_gradient = np.mean(gradients, axis=0)
            eta = eta_0 / np.sqrt(k + 1)
            h_new = h - eta * avg_gradient
            h_new = project_to_sum_box(h_new, target_sum)
            
            new_C = C5_value(h_new)
            if new_C < best_C_stage:
                best_C_stage = new_C
                best_h_stage = h_new.copy()
                h = h_new.copy()
        else:
            # If no good indices found, use single best shift 
            s = worst_shift(c)
            g = gradient_worst(h, s)
            eta = eta_0 / np.sqrt(k + 1)
            h_new = h - eta * g
            h_new = project_to_sum_box(h_new, target_sum)
            
            new_C = C5_value(h_new)
            if new_C < best_C_stage:
                best_C_stage = new_C
                best_h_stage = h_new.copy()
                h = h_new.copy()

    return best_h_stage, best_C_stage

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution stages
    stages = [(80, 0.30), (160, 0.35), (240, 0.35)]
    best_h = None
    best_C = float('inf')
    
    prev_h = None
    
    for n, time_fraction in stages:
        if time.time() >= start_time + budget_s:
            break
            
        stage_h, stage_C = make_stage(n, time_fraction, start_time, budget_s, prev_h)
        
        if stage_C < best_C:
            best_C = stage_C
            best_h = stage_h.copy()
            
        prev_h = stage_h.copy()
    
    return (best_h.tolist(), float(best_C), len(best_h))
