import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return corr

def compute_gradient(corr, h, top_k=10):
    """Compute gradient using top-k shifts"""
    # Find top-k shifts
    idx_top = np.argpartition(corr, -top_k)[-top_k:]
    
    # Compute soft-max weights
    corr_top = corr[idx_top]
    max_corr = corr_top.max()
    exp_corr = np.exp((corr_top - max_corr) / 1e-3)  # T=1e-3 for stability
    weights = exp_corr / exp_corr.sum()
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    for i, (s, w) in enumerate(zip(idx_top, weights)):
        if s >= 0:
            # First term: 1 - h_{i+s} for i in range(N-s)
            if s < len(h):
                grad[:len(h)-s] += w * (1 - h[s:])
        else:
            # Second term: h_{j-s} for j in range(-s, N)
            if -s < len(h):
                grad[-s:] += w * h[:-s]
    
    return grad
