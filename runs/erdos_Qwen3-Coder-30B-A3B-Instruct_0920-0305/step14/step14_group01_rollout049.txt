```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2
epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
max_iter_per_epsilon = 250
penalty_lambda = 1e3
penalty_mu = 1e4
final_refinement_steps = 1000

def project_to_box_sum(u, target_sum):
    """Project u onto box constraints [0,1] with sum = target_sum using waterfilling"""
    # Simple binary search for lambda
    def func(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while func(a) * func(b) > 0:
        a -= 0.1
        b += 0.1

    for _ in range(50):
        m = (a + b) / 2
        fm = func(m)
        if abs(fm) < 1e-12:
            break
        if fm * func(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, eps):
    """Compute gradient of smooth max using FFT"""
    corr = compute_correlation(h)
    w = np.exp(corr / eps)
    w = w / np.sum(w)
    
    # Convolution weights
    b = 1 - h
    L = 2 * len(h) - 1
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    conv2 = np.fft.ifft(np.fft.fft(w[::-1], L) * np.fft.fft(h, L)).real
    
    grad = conv1 - conv2
    return dx * grad[(len(h)-1):(2*len(h)-1)]

def compute_monotonicity_penalty(u):
    """Compute penalty for lack of monotonicity"""
    penalty = 0.0
    for i in range(1, len(u)):
        penalty += max(0, u[i-1] - u[i]) ** 2
    return penalty * penalty_lambda

def compute_monotonicity_gradient(u):
    """Compute gradient of monotonicity penalty"""
    grad = np.zeros_like(u)
    for i in range(1, len(u)):
        if u[i-1] > u[i]:
            grad[i-1] += -2 * penalty_lambda * (u[i-1] - u[i])
            grad[i] += 2 * penalty_lambda * (u[i-1] - u[i])
    return grad

def objective_with_gradient(u, eps):
    """Objective function with gradient for L-BFGS-B"""
    # Reconstruct full h from u
    h = np.concatenate([u, u[::-1]])
    
    # Apply sum constraint via projection
    h = project_to_box_sum(h, target_sum)
    
    # Compute smooth max gradient
    smooth_grad = compute_smooth_max_gradient(h, eps)
    
    # Get gradient w.r.t. u only (first half)
    grad_u = smooth_grad[:len(u)] + smooth_grad[len(u):][::-1]
    
    # Add monotonicity penalty
    penalty = compute_monotonicity_penalty(u)
    grad_u += compute_monotonicity_gradient(u)
    
    # Add sum constraint penalty
    sum_violation = np.sum(u) - target_sum
    grad_u += 2 * penalty_mu * sum_violation
    
    return penalty, grad_u

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def generate_seeds():
    """Generate diverse initial seeds"""
    seeds = []
    
    # Uniform
    seeds.append(np.full(n_points // 2, 0.5))
    
    # Triangular
    x = np.linspace(0, 1, n_points // 2)
    tri = np.minimum(x, 1-x)
    seeds.append(tri)
    
    # Cosine squared
    x = np.linspace(0, 2, n_points)
    cos_sq = (1 + np.cos(np.pi * (x - 1))) / 2
    seeds.append(cos_sq[:n_points // 2])
    
    # Random monotone
    rand = np.random.rand(n_points // 2)
    rand_sorted = np.sort(rand)
    seeds.append(rand_sorted)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial seeds
    initial_seeds = generate_seeds()
    
    restart_count = 0
    while time.time() - start_time < 0.95 * budget_s:
        restart_count += 1
        # Select random seed
        seed_idx = np.random.randint(len(initial_seeds))
        u = initial_seeds[seed_idx].copy()
        
        # Continue through epsilon schedule
        for eps in epsilon_schedule:
            # Optimization with L-BFGS-B
            result = minimize(
                lambda u: objective_with_gradient(u, eps),
                u,
                method='L-BFGS-B',
                jac=True,
                bounds=[(0, 1)] * len(u),
                options={'maxiter': max_iter_per_epsilon, 'disp': False}
            )
            
            u = result.x
            
            # Project to feasible set after each optimization step
            u = project_to_box_sum(u, target_sum)
            
            # Reconstruct full h and evaluate C5
            h = np.concatenate([u, u[::-1]])
            current_c5 = evaluate_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early termination if already good enough
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with subgradient descent on true max
    if best_h is not None:
        h = best_h.copy()
        t_end = start_time + budget_s
        alpha = 0.01
        
        for i in range(final_refinement_steps):
            if time.time() >= t_end:
                break
                
            # Find worst shift
            corr = compute_correlation(h)
            j_star = np.argmax(corr)
            
            # Gradient of the specific shift
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + j_star < len(h):
                    grad[i] += (1 - h[i + j_star])
                if i - j_star >= 0:
                    grad[i] -= h[i - j_star]
            
            # Update with subgradient descent
            h = h - alpha * grad
            h = project_to_box_sum(h, target_sum)
            
            # Evaluate and update best
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    if best_h is None:
        # Fallback to uniform
        h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5(h)
        best_h = h
    
    return (best_h.tolist(), float(best_c5), n_points)
```
