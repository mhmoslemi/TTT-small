def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use binary search for the projection
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search with higher precision
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient_exact(h):

    # (d/dh_i) ∫ h(x)(1 - h(x+k)) dx = (1 - h(x+k)) * δ_{x,i} - h(x) * δ_{x+k,i}

    # 1-{i+k} - 1_{i-k} (in the notation that it's 1 if the variable is involved and 0 otherwise)
    
    # Our implementation approach:
    # For index i:
    # ∂/∂h_i (max_k ∫ h(x)(1-h(x+k)) dx) = max_k ∂/∂h_i ∫ h(x)(1-h(x+k)) dx with k where max is achieved
    
    # Use specific coefficients for that shift
    # For kth shift we compute:
    # at position i: 1-h[i+k] terms where needed
    # at position i+k: -h[i] terms where needed
    
    gradient = np.zeros(len(h))
    
    # Find the position of max_correlation index
    # corr_length = 2*len(h) - 1
    actual_k = max_idx - (len(h) - 1)
    
    # At each position, compute how much the gradient contributes:
    # For shift k, position i, gradient = (1 - h_{i+k}) - h_{i-k} (if that index exists)
    # Account for boundary conditions and shifts properly
    
    # The gradient contribution at i for shift k is:
    # Evaluate index for backward shift contribution and forward shift contribution
    
    # After reflactoring to be more direct and safe:
    # For kth shift, we compute:
    # ∂/∂h_i [∫ h(x)(1-h(x+k)) dx] = 1_{i+k} - 1_{k-i} or similar interpretation
    
    for i in range(len(h)):
        # For optimization: check the region in correlation that is used
        if actual_k >= 0:
            # Positive shift: we look at h_{i-k} - h_{i+actual_k}
            if i - actual_k >= 0:
                gradient[i] += -h[i - actual_k] 
            if i + actual_k < len(h):
                gradient[i] += 1 - h[i + actual_k]
        else:
            # Negative shift: we look at h_{i+|shift|} - h_{i-|shift|}
            abs_k = -actual_k
            if i + abs_k < len(h):
                gradient[i] += 1 - h[i + abs_k]
            if i - abs_k >= 0:
                gradient[i] += -h[i - abs_k]
    
    return gradient
