import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return np.max(corr)

def smooth_max_correlation(h, T=1e-1, K=10):
    """Compute smooth approximation of max correlation with temperature T"""
    # Compute all correlations using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    
    # Select top K indices
    top_k_idx = np.argpartition(corr, -K)[-K:]
    
    # Compute soft-max weights
    corr_top = corr[top_k_idx]
    max_corr = corr_top.max()
    exp_corr = np.exp((corr_top - max_corr) / T)
    weights = exp_corr / exp_corr.sum()
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i, (s, w) in enumerate(zip(top_k_idx, weights)):
        if s >= 0:
            # First term: 1 - h_{i+s}
            if s < len(h):
                grad[:len(h)-s] += w * (1 - h[s:])
            # Second term: -h_{j-s} 
            if s > 0:
                grad[s:] -= w * h[:-s]
        else:
            # s < 0: First term - second term
            s_abs = -s
            if s_abs < len(h):
                grad[s_abs:] += w * (1 - h[:len(h)-s_abs])
            if s_abs < len(h):
                grad[:len(h)-s_abs] -= w * h[s_abs:]
    
    return corr, top_k_idx, weights, grad

def project_onto_capped_simplex(h, target_sum):
    """Project h onto the capped simplex with sum = target_sum"""
    h_proj = np.clip(h, 0, 1)
    sum_diff = np.sum(h_proj) - target_sum
    
    if abs(sum_diff) < 1e-10:
        return h_proj
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    def f(lam):
        temp = np.clip(h + lam, 0, 1)
        return np.sum(temp) - target_sum
    
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 120
    dx = 2.0 / N
    target_sum = N / 2
    max_iter = 3000
    T0 = 0.1
    alpha0 = 0.6
    K = 10
    beta = 0.9
    tau_T = 0.9995
    tau_alpha = 0.999
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Start with triangular seed
    x = np.linspace(0, 2, N)
    triangular = np.maximum(0, 1 - np.abs(x - 1))
    h0 = project_to_sum_box(triangular, target_sum)
    
    # Coarse optimization phase (N = 60)
    N_coarse = 60
    h_coarse = np.linspace(0, 2, N_coarse)
    triangular_coarse = np.maximum(0, 1 - np.abs(h_coarse - 1))
    h_coarse = project_to_sum_box(triangular_coarse, N_coarse / 2)
    
    # Gradient descent on coarse grid
    T = T0
    alpha = alpha0
    v = np.zeros_like(h_coarse)
    for t in range(2000):
        if time.time() - start_time > 0.7 * budget_s:
            break
        corr, top_k_idx, weights, grad = smooth_max_correlation(h_coarse, T, K)
        
        # Momentum
        v = beta * v + (1 - beta) * grad
        step = alpha / np.sqrt(t + 1)
        h_coarse = h_coarse - step * v
        
        # Projection
        h_coarse = project_onto_capped_simplex(h_coarse, N_coarse / 2)
        
        # Anneal
        T *= tau_T
        alpha *= tau_alpha
        
        # Check current solution
        current_c5 = evaluate_c5(h_coarse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best_coarse = h_coarse.copy()
    
    # Interpolate to finer grid
    x_coarse = np.linspace(0, 2, N_coarse)
    x_fine = np.linspace(0, 2, N)
    h_interp = np.interp(x_fine, x_coarse, h_best_coarse)
    h_fine = project_to_sum_box(h_interp, target_sum)
    
    # Global optimization with restarts
    restart_count = 15
    for restart in range(restart_count):
        if time.time() - start_time > 0.7 * budget_s:
            break
        # Random starting point
        h = np.random.rand(N)
        h = project_to_sum_box(h, target_sum)
        
        # Gradient descent
        T = T0
        alpha = alpha0
        v = np.zeros_like(h)
        for t in range(max_iter):
            if time.time() - start_time > 0.7 * budget_s:
                break
            corr, top_k_idx, weights, grad = smooth_max_correlation(h, T, K)
            
            # Momentum
            v = beta * v + (1 - beta) * grad
            step = alpha / np.sqrt(t + 1)
            h = h - step * v
            
            # Projection
            h = project_onto_capped_simplex(h, target_sum)
            
            # Anneal
            T *= tau_T
            alpha *= tau_alpha
            
            # Check current solution
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Fine-tune with pair-swap optimization
    if best_h is not None:
        h = best_h.copy()
        time_left = budget_s * 0.3
        start_polish = time.time()
        
        # Pair-swap local search
        num_swaps = 0
        max_swaps = 100000
        while time.time() - start_polish < time_left and num_swaps < max_swaps:
            if time.time() - start_time > budget_s:
                break
                
            i, j = np.random.choice(N, 2, replace=False)
            h_i, h_j = h[i], h[j]
            
            # Determine feasible transfer amounts
            delta_low = max(-h_i, h_j - 1)
            delta_high = min(1 - h_i, h_j)
            
            if delta_low >= delta_high:
                num_swaps += 1
                continue
                
            # Perform transfer
            delta = np.random.uniform(delta_low, delta_high)
            h[i] = h_i + delta
            h[j] = h_j - delta
            
            # Evaluate new C5
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            else:
                # Accept with small probability to escape local minima
                if np.random.rand() < 0.01:
                    # Revert the change
                    h[i] = h_i
                    h[j] = h_j
                else:
                    # Keep the change
                    pass
            
            num_swaps += 1
            
            # Occasionally check if we've finished
            if num_swaps % 1000 == 0:
                if time.time() - start_time > budget_s:
                    break
    
    # Final evaluation
    if best_h is None:
        best_h = h_fine
    
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), N)
