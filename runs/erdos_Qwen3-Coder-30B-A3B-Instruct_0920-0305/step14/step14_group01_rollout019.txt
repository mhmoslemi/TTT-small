```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_sum_box(u, target):
    """Project vector u onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(u + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(u + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(u, epsilon, lambda_inc=1e3, mu=1e4):
    """Compute gradient of smooth max with penalties"""
    h = np.concatenate([u, u[::-1]])
    corr = compute_correlation(h)
    
    # Smooth approximation to max
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Gradient contribution from correlation
    w_rev = w[::-1]
    # Full convolution using FFT
    conv1 = np.fft.ifft(np.fft.fft(w, 2*n_points-1) * np.fft.fft(1-h, 2*n_points-1)).real
    conv2 = np.fft.ifft(np.fft.fft(w_rev, 2*n_points-1) * np.fft.fft(h, 2*n_points-1)).real
    grad_corr = conv1[(n_points-1):] - conv2[(n_points-1):]
    
    # Gradient from monotonicity penalty (penalize decreasing slope)
    grad_mono = np.zeros_like(u)
    for i in range(1, len(u)):
        grad_mono[i] += lambda_inc * 2 * (u[i-1] - u[i])
        grad_mono[i-1] += lambda_inc * 2 * (u[i-1] - u[i])
    
    # Gradient from sum constraint
    grad_sum = 2 * mu * (np.sum(u) - target_sum)
    
    # Combine all components
    grad = dx * grad_corr[:len(u)] + grad_mono + grad_sum
    return grad

def evaluate_c5(h):
    """Evaluate the true C5 using max correlation"""
    corr = compute_correlation(h)
    return dx * np.max(corr)

def generate_initial_seeds():
    """Generate different initial seed vectors"""
    seeds = []
    
    # Uniform 0.5
    seeds.append(np.full(n_points//2, 0.5))
    
    # Triangular shape
    tri = np.array([min(i, n_points-1-i) for i in range(n_points//2)])
    tri = tri / np.sum(tri) * target_sum
    seeds.append(tri)
    
    # Cosine-squared shape (scaled to match sum)
    x = np.linspace(0, 1, n_points//2)
    cos_sq = (1 + np.cos(np.pi * x)) / 2
    cos_sq = cos_sq / np.sum(cos_sq) * target_sum
    seeds.append(cos_sq)
    
    # Random monotonic shape
    rand_u = np.sort(np.random.rand(n_points//2))
    rand_u = rand_u / np.sum(rand_u) * target_sum
    seeds.append(rand_u)
    
    return seeds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Schedule for smoothing parameter epsilon
    epsilon_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    max_iter_per_epsilon = 250
    
    best_c5 = float('inf')
    best_u = None
    
    # Generate initial seeds
    seeds = generate_initial_seeds()
    
    # Multiple restarts
    for restart in range(10):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Select a seed
        current_seed = seeds[restart % len(seeds)]
        u = current_seed.copy()
        
        # Continue with epsilon schedule
        for epsilon in epsilon_schedule:
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Optimization step
            def objective_and_gradient(u_flat):
                grad = compute_smooth_max_gradient(u_flat, epsilon)
                # Return objective value, gradient
                h = np.concatenate([u_flat, u_flat[::-1]])
                corr = compute_correlation(h)
                objective = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
                return objective, grad
            
            # Run optimization with L-BFGS
            result = minimize(
                objective_and_gradient,
                u,
                method='L-BFGS-B',
                jac=True,
                bounds=[(0, 1)] * len(u),
                options={'maxiter': max_iter_per_epsilon, 'disp': False}
            )
            
            u = result.x
            # Project onto feasible set
            u = project_to_sum_box(u, target_sum)
            
            # Evaluate true C5 for current solution
            h = np.concatenate([u, u[::-1]])
            current_c5 = evaluate_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_u = u.copy()
    
    # Final refinement with subgradient descent on true max
    if best_u is not None:
        h = np.concatenate([best_u, best_u[::-1]])
        t_end = start_time + budget_s - 0.05 * budget_s
        alpha0 = 0.01
        
        for iteration in range(10000):
            if time.time() >= t_end:
                break
                
            # Compute correlation and find worst shift
            corr = compute_correlation(h)
            worst_shift_idx = np.argmax(corr)
            
            # Compute subgradient for the worst shift
            grad = np.zeros_like(h)
            if worst_shift_idx < len(h):
                for i in range(len(h)):
                    if i + worst_shift_idx < len(h):
                        grad[i] += (1 - h[i + worst_shift_idx])
                    if i - worst_shift_idx >= 0:
                        grad[i] -= h[i - worst_shift_idx]
            
            # Update with step size decreasing as 1/sqrt(iteration+1)
            alpha_t = alpha0 / np.sqrt(iteration + 1)
            h_new = h - alpha_t * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Project onto sum constraint
            h_new = project_to_sum_box(h_new[:len(h)//2], target_sum)
            h_new = np.concatenate([h_new, h_new[::-1]])
            
            # Evaluate new C5
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h = h_new
            
            h = h_new
    
    if best_u is None:
        # Fallback to simple uniform distribution
        best_u = np.full(n_points//2, 0.5)
    
    h_final = np.concatenate([best_u, best_u[::-1]])
    
    return (h_final.tolist(), float(best_c5), n_points)
```
