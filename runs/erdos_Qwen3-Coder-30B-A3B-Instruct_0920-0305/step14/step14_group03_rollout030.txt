```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -2.0, 2.0
    while f(a) * f(b) > 0:
        a -= 0.5
        b += 0.5
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap values using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    # Compute convolution via FFT
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    # O_l = dx * (target_sum - correlation[l])
    O = dx * (target_sum - corr)
    return O

def compute_softmax_weights(O, alpha):
    """Compute softmax weights for gradient computation"""
    max_O = np.max(O)
    if max_O > 0:
        exp_term = np.exp(alpha * (O - max_O))
    else:
        exp_term = np.exp(alpha * O)
    w = exp_term / np.sum(exp_term)
    return w

def compute_gradient(h, w, dx):
    """Compute gradient using exact formula"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # For each shift, compute contribution to gradient
    for r in range(2 * n - 1):
        s = r - (n - 1)
        w_s = w[r]
        
        if s == 0:
            grad += -dx * 2 * w_s * h
        elif s > 0:
            # forward contribution h_{i+s}
            if s < n:
                grad[:n-s] += -dx * w_s * h[s:]
            # backward contribution h_{i-s} 
            if s < n:
                grad[s:] += -dx * w_s * h[:n-s]
        else:
            # s < 0, so negative shift
            s_abs = -s
            if s_abs < n:
                grad[s_abs:] += -dx * w_s * h[:n-s_abs]
                grad[:n-s_abs] += -dx * w_s * h[s_abs:]
    
    return grad

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Try different n points
    n_candidates = [120, 150, 180, 200]
    
    for n in n_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        max_iter = 5000
        
        # Multiple restarts
        for restart in range(3):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Initialize h
            if restart == 0:
                # Use initial solution if available, otherwise random
                if 'initial_h_values' in globals() and len(initial_h_values) == n:
                    h = np.array(initial_h_values)
                else:
                    # Random initialization
                    h = np.random.rand(n) * 0.8 + 0.1  # Random values between 0.1 and 0.9
                h = project_to_sum_box(h, target_sum)
            elif restart == 1:
                # Sine-based initialization
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h = 0.5 + 0.3 * np.sin(np.pi * x)  # Sine wave centered around 0.5
                h = np.clip(h, 0, 1)
                h = project_to_sum_box(h, target_sum)
            else:
                # Uniform initialization
                h = np.full(n, 0.5)
                h = project_to_sum_box(h, target_sum)
            
            # Initialize optimizer parameters
            lr = 0.1
            alpha = 10.0
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            
            # Adam parameters
            m = np.zeros_like(h)
            v = np.zeros_like(h)
            t = 0
            
            current_h = h.copy()
            current_c5 = evaluate_c5(current_h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_n = n
            
            # Optimization loop
            for iteration in range(max_iter):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                t += 1
                O = compute_O(current_h, dx, target_sum)
                C5 = O.max()
                
                if C5 < best_c5:
                    best_c5 = C5
                    best_h = current_h.copy()
                    best_n = n
                
                # Compute softmax weights
                w = compute_softmax_weights(O, alpha)
                
                # Compute gradient
                grad = compute_gradient(current_h, w, dx)
                
                # Adam updates
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # Update with projection
                new_h = current_h - step
                new_h = project_to_sum_box(new_h, target_sum)
                new_c5 = evaluate_c5(new_h)
                
                if new_c5 < C5:
                    current_h = new_h
                    current_c5 = new_c5
                else:
                    # Reduce learning rate if no improvement
                    lr *= 0.5
                    if lr < 1e-6:
                        lr = 1e-6
                
                # Adaptive alpha
                if t % 100 == 0:
                    alpha = min(alpha * 1.05, 200.0)
                
                # Occasionally add perturbation to escape local minima
                if t % 500 == 0 and t > 0:
                    noise = np.random.normal(0, 0.001, n)
                    temp_h = current_h + noise
                    temp_h = project_to_sum_box(temp_h, target_sum)
                    temp_c5 = evaluate_c5(temp_h)
                    if temp_c5 < current_c5:
                        current_h = temp_h
                        current_c5 = temp_c5
            
            # Final check for best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_n = n
    
    if best_h is None:
        # Fallback to initial solution if nothing found
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_c5 = evaluate_c5(best_h)
            best_n = len(initial_h_values)
        else:
            best_h = np.full(120, 0.5)
            best_c5 = evaluate_c5(best_h)
            best_n = 120
    
    return (best_h.tolist(), float(best_c5), best_n)
```
