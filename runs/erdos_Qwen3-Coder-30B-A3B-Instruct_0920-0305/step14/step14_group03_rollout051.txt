```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -2.0, 2.0
    while f(a) * f(b) > 0:
        a -= 0.5
        b += 0.5
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_fast(h, dx, target_sum):
    """Fast evaluation of C5 using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    O = dx * (target_sum - corr)
    return np.max(O)

def compute_O(h, dx, target_sum):
    """Compute O vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return dx * (target_sum - corr)

def compute_gradient_using_w(h, w, dx):
    """Compute gradient using softmax weights"""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute contributions for each shift s
    for r in range(2 * n - 1):
        s = r - (n - 1)
        ws = w[r]
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial solution as starting point
    if 'initial_h_values' in globals() and len(initial_h_values) == 200:
        h0 = np.array(initial_h_values)
    else:
        # Create uniform initial solution
        h0 = np.full(200, 0.5)
    
    h0 = project_to_sum_box(h0, 100)  # target_sum = n/2 = 100
    
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Different n_points to try
    n_candidates = [120, 150, 180, 200]
    
    for n in n_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        TARGET_SUM = n / 2
        dx = 2.0 / n
        
        # Multiple restarts for this n
        for restart in range(3):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Initialize h for this restart
            if restart == 0:
                h = h0.copy()
                if len(h) != n:
                    # Interpolate from 200 to n points
                    x_old = np.linspace(0, 2, 200, endpoint=False) + 1.0/200
                    x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
                    h = np.interp(x_new, x_old, h0)
                    h = np.clip(h, 0, 1)
            elif restart == 1:
                # Random initialization
                rand = np.random.rand(n)
                h = rand / np.sum(rand) * TARGET_SUM
                h = np.clip(h, 0, 1)
            else:
                # Sinusoidal initialization
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h = 0.5 + 0.3 * np.cos(np.pi * x / 2)
                h = np.clip(h, 0, 1)
            
            h = project_to_sum_box(h, TARGET_SUM)
            
            # Optimization parameters
            lr = 0.05
            alpha = 10.0
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            
            m = np.zeros_like(h)
            v = np.zeros_like(h)
            t = 0
            
            current_h = h.copy()
            current_c5 = evaluate_c5_fast(current_h, dx, TARGET_SUM)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Optimization loop
            max_iter = 10000
            for iteration in range(max_iter):
                if time.time() - start_time > 0.9 * budget_s:
                    break
                    
                t += 1
                
                # Compute O vector
                O = compute_O(current_h, dx, TARGET_SUM)
                C5 = O.max()
                
                # Update best
                if C5 < best_c5:
                    best_c5 = C5
                    best_h = current_h.copy()
                
                # Compute softmax weights
                max_O = O.max()
                if max_O > 0:
                    exp_term = np.exp(alpha * (O - max_O))
                else:
                    exp_term = np.exp(alpha * O)
                w = exp_term / np.sum(exp_term)
                
                # Compute gradient
                grad = compute_gradient_using_w(current_h, w, dx)
                
                # Adam updates
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # Try update with projection
                new_h = current_h - step
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                new_c5 = evaluate_c5_fast(new_h, dx, TARGET_SUM)
                
                if new_c5 < C5:
                    current_h = new_h
                else:
                    # Backtracking line search
                    step_size = lr
                    for _ in range(10):
                        step_size *= 0.5
                        test_h = current_h - step_size * grad
                        test_h = project_to_sum_box(test_h, TARGET_SUM)
                        test_c5 = evaluate_c5_fast(test_h, dx, TARGET_SUM)
                        if test_c5 < C5:
                            current_h = test_h
                            break
                
                # Schedule updates
                if t % 100 == 0:
                    alpha = min(alpha * 1.05, 200.0)
                
                # Occasionally random perturbation to escape local minima
                if t % 500 == 0 and t > 0:
                    perturbation = np.random.normal(0, 0.001, n)
                    temp_h = current_h + perturbation
                    temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                    temp_c5 = evaluate_c5_fast(temp_h, dx, TARGET_SUM)
                    if temp_c5 < current_c5:
                        current_h = temp_h
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5_fast(best_h, dx, TARGET_SUM)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
```
