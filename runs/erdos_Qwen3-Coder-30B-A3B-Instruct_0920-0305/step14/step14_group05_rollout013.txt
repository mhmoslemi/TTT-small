import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute raw correlation array using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 value including dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of coarse solution onto finer grid"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def worst_shift(c):
    """Return the shift index that achieves maximum correlation"""
    idx = np.argmax(c)
    return idx - (len(c) - 1)

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        nonlocal TARGET_SUM
        TARGET_SUM = target_sum(n)
        current_time = time.time()
        stage_deadline = start_time + budget_s * time_fraction
        
        # Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, TARGET_SUM)
        else:
            h = interpolate(best_h_stage, n)
            h = project_to_sum_box(h, TARGET_SUM)
            
        best_h_stage = h.copy()
        current_c5 = C5_value(h)
        best_C_stage = current_c5
        
        # Simulated Annealing
        T = 0.02
        delta = 0.1
        alpha = 0.9997
        beta = 0.9999
        max_iter = int(20000 * time_fraction)  # Scale with time fraction
        acceptance_count = 0
        total_trials = 0
        min_acceptance_rate = 0.01
        
        for iteration in range(max_iter):
            if time.time() >= stage_deadline:
                break
                
            total_trials += 1
            # Choose move type
            if random.random() < 0.3:
                # Targeted move
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Select indices from desirability
                sorted_indices = np.argsort(d)
                low_indices = sorted_indices[:max(1, len(d)//5)]
                high_indices = sorted_indices[-max(1, len(d)//5):]
                
                i = random.choice(low_indices)
                j = random.choice(high_indices)
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                j = random.choice(range(n))
                
            # Compute transfer amount
            max_transfer = min(delta, h[i], 1 - h[j])
            if max_transfer <= 0:
                continue
                
            delta_transfer = random.uniform(0, max_transfer)
            
            # Propose move
            h_new = h.copy()
            h_new[i] -= delta_transfer
            h_new[j] += delta_transfer
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            c5_new = C5_value(h_new)
            
            # Accept or reject
            if c5_new <= current_c5:
                h = h_new
                current_c5 = c5_new
                if c5_new < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = c5_new
                acceptance_count += 1
            else:
                if random.random() < np.exp(-(c5_new - current_c5) / T):
                    h = h_new
                    current_c5 = c5_new
                    if c5_new < best_C_stage:
                        best_h_stage = h.copy()
                        best_C_stage = c5_new
                    acceptance_count += 1
                    
            # Update parameters
            T *= alpha
            delta *= beta
            
            # Check acceptance rate
            if total_trials % 1000 == 0 and total_trials > 0:
                acceptance_rate = acceptance_count / total_trials
                if acceptance_rate < min_acceptance_rate:
                    # Restart
                    h = np.random.uniform(0, 1, n)
                    h = project_to_sum_box(h, TARGET_SUM)
                    current_c5 = C5_value(h)
                    acceptance_count = 0
                    total_trials = 0
        
        # Polishing with exact gradient descent
        max_polish_iter = 2000
        eta_0 = 0.5
        for k in range(max_polish_iter):
            if time.time() >= stage_deadline:
                break
                
            c = corr_fft(h)
            max_corr = np.max(c)
            eps = 1e-12
            indices = np.where(np.abs(c - max_corr) < eps)[0]
            
            if len(indices) == 0:
                indices = [np.argmax(c)]
            
            g = np.zeros(n)
            for s in indices:
                g += gradient_worst(h, s - (n-1))
            
            g /= len(indices)
            eta = eta_0 / np.sqrt(k + 1)
            h = h - eta * g
            h = project_to_sum_box(h, TARGET_SUM)
            
            new_c5 = C5_value(h)
            if new_c5 < best_C_stage:
                best_h_stage = h.copy()
                best_C_stage = new_c5
        
        return best_h_stage, best_C_stage
    
    # Run three stages
    best_h, best_C = make_stage(80, 0.30)
    best_h, best_C = make_stage(160, 0.35)
    best_h, best_C = make_stage(240, 0.35)
    
    return (best_h.tolist(), float(best_C), len(best_h))
