import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 240
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling approach
    x = np.array(x)
    tol = 1e-12
    low, high = 0.0, 1.0
    
    def f(lam):
        proj = np.clip(x + lam, low, high)
        return np.sum(proj) - target_sum
    
    # Binary search for lambda
    while f(low) * f(high) > 0:
        if f(low) > 0:
            low -= 1.0
        else:
            high += 1.0
    
    for _ in range(100):
        mid = (low + high) / 2
        fmid = f(mid)
        if abs(fmid) < tol:
            break
        if fmid * f(low) < 0:
            high = mid
        else:
            low = mid
    
    result = np.clip(x + (low + high) / 2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_smooth_max_gradient(h, epsilon=0.01):
    """Compute gradient of smooth max of correlations"""
    corr = compute_correlation(h)
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)
    
    # Gradient computation as convolutions
    b = 1 - h
    L = 2 * len(h) - 1
    w_rev = w[::-1]
    
    # conv(w, b)
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(b, L)).real
    
    # conv(w_rev, h)
    conv2 = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    grad = conv1 - conv2
    grad = grad[(len(h)-1):(2*len(h)-2)]
    
    return dx * grad

def compute_objective_and_gradient(u, epsilon=0.01):
    """Compute smooth objective and its gradient"""
    # Reconstruct full h from u (symmetric)
    h = np.concatenate([u, u[::-1]])
    
    # Compute smooth max
    corr = compute_correlation(h)
    smooth_max = epsilon * np.log(np.sum(np.exp(corr / epsilon)))
    
    # Gradient
    grad = compute_smooth_max_gradient(h, epsilon)
    
    # Monotonicity penalty
    pen_inc = 1e3 * np.sum(np.maximum(0, np.diff(u)) ** 2)
    grad_inc = np.zeros_like(u)
    grad_inc[1:] = -2 * 1e3 * np.maximum(0, np.diff(u)) 
    grad_inc[:-1] += 2 * 1e3 * np.maximum(0, np.diff(u)) 
    
    # Sum penalty
    penalty = 1e4 * (np.sum(u) - target_sum) ** 2
    grad_sum = 2 * 1e4 * (np.sum(u) - target_sum)
    
    return smooth_max + penalty + pen_inc, grad[:len(u)] + grad_inc + grad_sum
