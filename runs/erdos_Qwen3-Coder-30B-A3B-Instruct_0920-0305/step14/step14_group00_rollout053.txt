```python
import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist

def evaluate_h(h_values, dx):
    """Evaluate the C5 bound for given h values"""
    h = np.array(h_values)
    # Compute all correlations using numpy's correlate
    correlations = np.correlate(h, 1 - h, mode='full')
    # The valid correlations are from index n-1 to 2*n-2 (excluding the diagonal)
    valid_correlations = correlations[len(h)-1:]
    return np.max(valid_correlations) * dx

def initial_h_values():
    """Create initial construction - zigzag pattern"""
    n = 120
    h = np.zeros(n)
    # Create a zigzag pattern that satisfies constraints
    for i in range(n):
        if i < n//4:
            h[i] = 0.5
        elif i < n//2:
            h[i] = 1.0
        elif i < 3*n//4:
            h[i] = 0.0
        else:
            h[i] = 0.5
    # Adjust to make integral equal to 1
    h = h * (n/2) / np.sum(h)
    return h.tolist()

def objective(h_values):
    """Objective function to minimize"""
    dx = 2.0 / len(h_values)
    return evaluate_h(h_values, dx)

def constraint_sum(h_values):
    """Constraint that integral equals 1"""
    dx = 2.0 / len(h_values)
    return np.sum(h_values) * dx - 1.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction as starting point
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Method 1: Differential Evolution (global optimization)
    try:
        result_de = differential_evolution(
            objective, 
            bounds, 
            constraints=constraints,
            maxiter=int(budget_s/10),
            popsize=15,
            seed=seed,
            disp=False
        )
        
        if time.time() - start_time > budget_s - 1:
            best_h = initial_h
            best_c5 = evaluate_h(initial_h, dx)
        else:
            best_h = result_de.x
            best_c5 = result_de.fun
            
        # Method 2: Local optimization with L-BFGS-B if we have time
        if time.time() - start_time < budget_s - 2:
            # Refine with local optimization
            result_local = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': int(budget_s/20)},
                tol=1e-8
            )
            
            if result_local.success and result_local.fun < best_c5:
                best_h = result_local.x
                best_c5 = result_local.fun
                
    except Exception as e:
        # Fallback to initial construction
        best_h = initial_h
        best_c5 = evaluate_h(initial_h, dx)
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
```
