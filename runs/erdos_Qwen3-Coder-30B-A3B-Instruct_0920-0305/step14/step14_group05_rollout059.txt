def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: adjust values to meet sum constraint
    current_sum = np.sum(x)
    if current_sum == target_sum:
        return x.copy()
    
    # Linear scaling approach
    if current_sum > 0:
        scale = target_sum / current_sum
        projected = x * scale
    else:
        projected = np.ones_like(x) * (target_sum / len(x))
    
    # Clamp to [0,1] and adjust to meet sum exactly
    projected = np.clip(projected, 0, 1)
    current_sum = np.sum(projected)
    
    if abs(current_sum - target_sum) < 1e-10:
        return projected
    
    # If still not exact, adjust carefully
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need to increase values
            indices = np.argsort(projected)
            for i in range(len(indices)-1, -1, -1):  # Start from largest
                idx = indices[i]
                if diff <= 0:
                    break
                can_increase = 1.0 - projected[idx]
                increase = min(can_increase, diff)
                projected[idx] += increase
                diff -= increase
        else:
            # Need to decrease values
            indices = np.argsort(projected)
            for i in range(len(indices)):  # Start from smallest
                idx = indices[i]
                if diff >= 0:
                    break
                can_decrease = projected[idx]
                decrease = min(can_decrease, -diff)
                projected[idx] -= decrease
                diff += decrease
    
    return projected

    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx
