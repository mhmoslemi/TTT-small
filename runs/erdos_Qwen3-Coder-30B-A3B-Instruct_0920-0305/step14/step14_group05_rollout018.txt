import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: adjust linearly to get target sum
    current_sum = np.sum(x)
    if current_sum == target_sum:
        return x.copy()
    
    # Linear adjustment
    adjustment = (target_sum - current_sum) / len(x)
    result = np.clip(x + adjustment, 0, 1)
    
    # Fine tune to exactly match sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Distribute the difference to the largest elements
        idx = np.argsort(result)
        for i in range(int(abs(diff) / (1e-10))):
            if diff > 0:
                result[idx[i]] = min(1.0, result[idx[i]] + 1e-10)
            else:
                result[idx[-i-1]] = max(0.0, result[idx[-i-1]] - 1e-10)
                
    return result
