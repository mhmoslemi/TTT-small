def evaluate_h(h_values, dx):

    n = len(h)
    # For each shift k, compute integral of h(x)(1-h(x+k))
    correlations = []
    
    # Manual computation for all shifts
    for k in range(1, n):
        corr = 0.0
        # For shift k, we pair h[i] with (1-h[i+k]) for valid i
        for i in range(n - k):
            corr += h[i] * (1 - h[i + k])
        correlations.append(corr * dx)
    
    return max(correlations) if correlations else 0.0
