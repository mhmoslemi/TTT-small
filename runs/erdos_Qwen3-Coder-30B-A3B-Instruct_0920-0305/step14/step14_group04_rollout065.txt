TARGET_SUM = None

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def constrained_optimization(h0, max_iter=1000):
    """Use constrained optimization to refine solution"""

    h = cp.Variable(n)

    alpha = 10.0
    corr_exprs = []
    
    for i in range(2*n - 1):
        corr_val = 0
        for j in range(n):
            if 0 <= i - j < n:
                corr_val += h[j] * (1 - h[i - j])
        corr_exprs.append(corr_val * dx)
    
    # Smooth max approximation
    max_corr = cp.log_sum_exp(alpha * cp.hstack(corr_exprs)) / alpha
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == TARGET_SUM
    ]
    
    # Problem
    problem = cp.Minimize(max_corr)
    try:
        problem.solve(solver=cp.CVXOPT, verbose=False)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    return h0
