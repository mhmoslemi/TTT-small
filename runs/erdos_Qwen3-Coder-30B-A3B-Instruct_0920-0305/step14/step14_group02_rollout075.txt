import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_exact(h):
    """Exact evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.05):
    """Smooth evaluation using log-sum-exp approximation"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # Use log-sum-exp with temperature T
    corr_scaled = corr / T
    max_corr = np.max(corr_scaled)
    # Subtract max for numerical stability
    log_sum_exp = T * (np.log(np.sum(np.exp(corr_scaled - max_corr))) + max_corr)
    return dx * log_sum_exp

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling method for projection onto capped simplex
    # Sort values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find the threshold lambda
    cumulative_sum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Look for the point where the constraint binding occurs
    lambda_vals = (cumulative_sum - target_sum) / np.arange(1, n+1)
    
    # Find the largest lambda such that the projection is still <= 1
    lambda_max = 0
    idx = -1
    for i in range(n):
        if sorted_x[i] - lambda_vals[i] >= 0 and sorted_x[i] - lambda_vals[i] <= 1:
            lambda_max = lambda_vals[i]
            idx = i
        elif sorted_x[i] - lambda_vals[i] < 0:
            break
    
    # If this doesn't work, we need to find where the constraint becomes binding
    if idx == -1:
        # All elements would be <= 0, so make them all 0 except those that are 1 
        # But this is wrong, let's compute it correctly:
        # We want to solve this: we want lambda such that the sum of (x_i - lambda)_+ = target_sum
        # This is computed by water-filling method properly
        pass
    else:
        lambda_val = lambda_max
    
    # Now apply the projection
    projected = np.clip(x - lambda_val, 0, 1)
    
    # Adjust for numerical imprecision in sum
    current_sum = np.sum(projected)
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-10:
        # Add the difference to the largest elements
        # Find indices where we can add (i.e., elements < 1)
        add_idx = np.argsort(projected)[::-1]
        idx = 0
        while abs(diff) > 1e-10 and idx < len(add_idx):
            if projected[add_idx[idx]] < 1:
                add = min(diff, 1 - projected[add_idx[idx]])
                projected[add_idx[idx]] += add
                diff -= add
            idx += 1
    
    return projected

def compute_gradient(h, T=0.05):
    """Compute the gradient of the smooth surrogate"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real[:L]
    
    # Calculate weights using soft-max
    corr_scaled = corr / T
    max_corr = np.max(corr_scaled)
    corr_scaled -= max_corr  # numerical stability
    exp_corr = np.exp(corr_scaled)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # For each shift, compute derivative of c_s
    for s in range(len(corr)):
        # Compute the derivative for shift s
        if s >= len(h):
            continue
        # Gradient contributions from shift s   
        for i in range(len(h)):
            # derivative of h_i(1-h_{i+s}) w.r.t. h_i = 1 - h_{i+s}
            if i < len(h)-s:
                grad[i] += weights[s] * (1 - h[i+s]) * dx
            
            # derivative of h_i(1-h_{i+s}) w.r.t. h_{i+s} = -h_i 
            if i+s < len(h):
                grad[i+s] -= weights[s] * h[i] * dx
    
    return grad

def lbfgsb_optimization(initial_h, max_iter=2000):
    """Optimize using L-BFGS-B with projected gradient updates"""
    def objective(x):
        return evaluate_c5_smooth(x)
    
    def grad_objective(x):
        return compute_gradient(x)
    
    # Project the initial guess to satisfy sum constraint
    h0 = project_to_sum_box(initial_h, TARGET_SUM)
    
    # Use L-BFGS-B
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        jac=grad_objective,
        bounds=[(0, 1)] * len(h0),
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-12}
    )
    
    # Final projection to satisfy sum constraint exactly
    final_h = project_to_sum_box(result.x, TARGET_SUM)
    final_c5 = evaluate_c5_exact(final_h)
    
    return final_h, final_c5

def pairwise_exchange_polish(h, max_iterations=1000):

    # Generate a list of pairs to iterate through
    pairs = [(i, j) for i in range(n_points) for j in range(i+1, n_points)]
    # Shuffle for randomization
    np.random.shuffle(pairs)
    
    iteration = 0
    improved = True
    
    while improved and iteration < max_iterations:
        improved = False
        iteration += 1
        
        # Continue with random pairs
        for i, j in pairs:
            if iteration >= max_iterations:
                break
                
            if abs(current_h[i]) < 1e-10 and abs(current_h[j]) < 1e-10:
                continue
            
            # Compute the correlation vector for the current state
            current_corr = correlate(current_h, 1-current_h, mode='full')
            
            # Compute increment for each shift when moving mass from i to j
            # This is efficient because correlation is linear in h
            delta_corr = np.zeros(len(current_corr))
            
            # For shift s where i or j contributes (i in range [0, N-1-s])
            for s in range(len(current_corr)):
                # When moving from i to j
                if s == 0:
                    # contribution from i and j when s=0
                    if i < len(current_h) and j < len(current_h):
                        delta_corr[s] -= 1 - current_h[j]
                        delta_corr[s] += 1 - current_h[i]
                else:
                    # For other shifts involve the specific terms
                    if i-s >= 0 and i-s < len(current_h) and i < len(current_h):
                        delta_corr[s] -= current_h[i-s]  # h_{i-s} being reduced
                        delta_corr[s] += current_h[j-s] if 0 <= j-s < len(current_h) else 0
                    
                    # Add new contribution from j instead of i
                    if j >= 0 and j < len(current_h):
                        delta_corr[s] += current_h[j] if 0 <= i+s < len(current_h) else 0
                        delta_corr[s] -= current_h[i] if 0 <= i+s < len(current_h) else 0
                    
                    # Handle special shift zero case
                    if i + s < len(current_h) and 0 <= j+s < len(current_h):
                        delta_corr[s] -= current_h[i] if i >= 0 else 0

            # Compute the feasible range for delta
            max_delta = min(current_h[i], 1.0 - current_h[j])
            min_delta = max(-current_h[j], current_h[i] - 1.0)
            
            if max_delta <= min_delta + 1e-12:
                continue
                
            # Try to find the optimal delta analytically for the move from i to j
            # This is non-trivial due to the nature of the max operation
            # For this simplified version, we will sample and find min
            best_new_c5 = current_c5
            best_delta = 0
            
            # Determine some samples in the feasible range of delta
            step_size = (max_delta - min_delta) / 10.0
            if step_size < 1e-10:
                continue
                
            for delta in np.arange(min_delta, max_delta + step_size, step_size):
                # Create a new h with moved mass
                new_h = current_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
                
                # Project new h back to [0,1] box with equal sum
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                
                # Evaluate new c5
                new_c5 = evaluate_c5_exact(new_h)
                
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_delta = delta
                    break
                
            # Apply the best delta
            if best_new_c5 < current_c5 - 1e-10:
                current_h[i] -= best_delta
                current_h[j] += best_delta
                current_h = project_to_sum_box(current_h, TARGET_SUM)
                
                current_c5 = best_new_c5
                improved = True
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
    
    return best_h, best_c5
