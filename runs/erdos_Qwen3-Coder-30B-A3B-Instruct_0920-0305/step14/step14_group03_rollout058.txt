import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Build index mapping for all pairs (i, i+s) that are valid
    pairs = []
    for i in range(n_points):
        for s in range(-n_points+1, n_points):
            if 0 <= i+s < n_points:
                pairs.append((i, i+s, s))
    
    # Linear feasibility solving with McCormick relaxation
    # Variables
    h = cp.Variable(n_points)
    w = cp.Variable(len(pairs))  # Product variables
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    
    # McCormick constraints
    for idx, (i, j, s) in enumerate(pairs):
        constraints.extend([
            w[idx] >= 0,
            w[idx] <= h[i],
            w[idx] <= h[j],
            w[idx] >= h[i] + h[j] - 1
        ])
    
    # Define parameter for RHS of correlation constraints
    rhs_param = cp.Parameter()
    
    # Add correlation constraints
    shift_idx = {}
    for i in range(n_points):
        for s in range(-n_points+1, n_points):
            if 0 <= i+s < n_points:
                if s not in shift_idx:
                    shift_idx[s] = []
                shift_idx[s].append((i, i+s))
    
    for s, idx_pairs in shift_idx.items():
        corr_expr = cp.sum([w[pairs.index((i, j, s))] for i, j in idx_pairs if (i, j, s) in pairs])
        constraints.append(corr_expr >= rhs_param)
    
    # Solve using CVXPY with GLPK solver
    problem = cp.Problem(cp.Minimize(0), constraints)
    
    # Binary search on C
    low = 0.30
    high = 0.40
    epsilon = 1e-4
    
    best_h = None
    best_c5 = float('inf')
    
    while high - low > epsilon and time.time() - start_time < 0.9 * budget_s:
        Cmid = (low + high) / 2
        rhs_param.value = target_sum - Cmid / dx
        
        try:
            problem.solve(solver=cp.GLPK, verbose=False)
            
            if problem.status == cp.OPTIMAL:
                h_val = h.value
                h_val = project_to_sum_box(h_val, target_sum)
                computed_c5 = evaluate_c5(h_val)
                
                if computed_c5 < best_c5:
                    best_c5 = computed_c5
                    best_h = h_val.copy()
                
                high = Cmid
            else:
                low = Cmid
        except:
            low = Cmid
    
    # Refinement with projected subgradient method (short) 
    if best_h is not None:
        h = best_h.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s:
                break
            corr = correlate(h, 1-h, mode='full')
            max_corr_idx = np.argmax(corr)
            max_corr = corr[max_corr_idx]
            
            # Compute subgradient for max correlation
            worst_shift = max_corr_idx - len(h) + 1
            g = np.zeros_like(h)
            
            # Compute exact gradient for the max correlation
            for i in range(len(h)):
                if 0 <= i + worst_shift < len(h):
                    g[i] -= (1 - h[i + worst_shift])
                if 0 <= i - worst_shift < len(h):
                    g[i] -= (1 - h[i - worst_shift])
            
            # Projected gradient step
            step_size = 1e-3
            new_h = h - step_size * g
            new_h = project_to_sum_box(new_h, target_sum)
            
            new_c5 = evaluate_c5(new_h)
            if new_c5 < evaluate_c5(h):
                h = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
    
    # Upsample to higher resolution
    if best_h is not None:
        n_fine = 200
        dx_fine = 2.0 / n_fine
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_points), best_h)
        h_fine = project_to_sum_box(h_fine, n_fine / 2)
        
        # Final refinement with projected subgradient
        h = h_fine
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            corr = correlate(h, 1-h, mode='full')
            max_corr_idx = np.argmax(corr)
            max_corr = corr[max_corr_idx]
            
            worst_shift = max_corr_idx - len(h) + 1
            g = np.zeros_like(h)
            
            for i in range(len(h)):
                if 0 <= i + worst_shift < len(h):
                    g[i] -= (1 - h[i + worst_shift])
                if 0 <= i - worst_shift < len(h):
                    g[i] -= (1 - h[i - worst_shift])
            
            step_size = 1e-3
            new_h = h - step_size * g
            new_h = project_to_sum_box(new_h, n_fine / 2)
            
            new_c5 = evaluate_c5(new_h)
            if new_c5 < evaluate_c5(h):
                h = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
    
    return (best_h.tolist(), float(best_c5), len(best_h))
