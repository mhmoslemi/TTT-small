def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    max_overlap = 0
    
    # Compute all correlations
    for k in range(1, n):
        correlation = 0
        for i in range(n - k):
            correlation += h[i] * (1 - h[i + k])
        max_overlap = max(max_overlap, correlation)
    
    return max_overlap * dx

    """Objective function to minimize"""
    dx = 2.0 / len(h_values)
