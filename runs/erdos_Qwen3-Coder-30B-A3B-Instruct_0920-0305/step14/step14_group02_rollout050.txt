import numpy as np

import time

# Global variables
n_points = 120
dx = 2.0 / n_points
target_sum = n_points / 2

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # Forward contribution: (1 - h_{i+s}) for i = 0..N-1-s
        g[:N-s] += 1 - h[s:]
        # Backward contribution: -h_{i-s} for i = s..N-1
        g[s:] -= h[:N-s]
    else:
        # s < 0
        s_abs = -s
        # Forward contribution: (1 - h_{i+s}) for i = s_abs..N-1
        g[s_abs:] += 1 - h[:N-s_abs]
        # Backward contribution: -h_{i-s} for i = 0..N-1-s_abs
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient using soft-max of top-k shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Exact subgradient of max
        idx = np.argmax(corr)
        shift = idx - (len(h) - 1)
        return subgradient_for_shift(h, shift)
    else:
        # Smooth gradient using soft-max
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (len(h) - 1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart"""
    # Initialisation
    h0 = np.random.rand(n)
    h0 = project_onto_capped_simplex(h0, target_sum)
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    T = T0
    step = step0
    
    for it in range(max_iter):
        g = smooth_gradient(h, T, top_k)
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        
        if sym:
            h = (h + h[::-1]) / 2
            
        h = project_onto_capped_simplex(h, target_sum)
        
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        T *= 0.9995
        
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Apply pair-swap hill climbing"""
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        # Pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # Sample delta from Uniform(-d, d) but keep both in [0,1]
        d = 0.1
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
                
    return best_h, best_c5
