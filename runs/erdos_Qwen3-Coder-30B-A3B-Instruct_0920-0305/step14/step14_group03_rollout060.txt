def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -2.0, 2.0
    while f(a) * f(b) > 0:
        a -= 0.5
        b += 0.5
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5_fast(h, dx, target_sum):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    # O = dx * (target_sum - corr)
    O = dx * (target_sum - corr)
    return np.max(O)

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    O = dx * (target_sum - corr)
    return O

def compute_gradient_from_w(h, w, dx):
    """Compute gradient using weights w"""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute gradient: for each shift s, we add contributions to grad
    for r in range(2*n - 1):
        s = r - (n - 1)
        ws = w[r]
        
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:  # s < 0
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
            
    return grad
