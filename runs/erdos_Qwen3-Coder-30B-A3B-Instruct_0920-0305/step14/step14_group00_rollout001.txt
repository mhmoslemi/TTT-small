import numpy as np
import time

def project_onto_box_sum(v, target, lower=0.0, upper=1.0):
    # Project vector v onto box constraints [lower,upper] and sum=target
    # Using bisection method to find lambda
    low = -1e5
    high = 1e5
    for _ in range(60):
        lam = (low + high) / 2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    # Final projection
    x = np.clip(v - (low + high) / 2, lower, upper)
    # Fine-tune sum
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Distribute among free variables
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if len(free) > 0:
            x[free] += diff / len(free)
            x = np.clip(x, lower, upper)
    return x

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 180
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values)
        x_old = np.linspace(0, 2, len(init), endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_sum(h, target_sum, 0.0, 1.0)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Gradient descent parameters
    alpha0 = 0.5
    max_iter = 20000
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.5 and iteration < max_iter:
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        max_indices = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        # Compute subgradient
        grad = np.zeros_like(h)
        for idx in max_indices:
            shift = idx - (n_points - 1)
            if shift >= 0:
                s = shift
                i = np.arange(n_points - s)
                grad[i] += 1 - h[i + s]
                grad[i + s] -= h[i]
            else:
                s = -shift
                i = np.arange(n_points - s)
                grad[i + s] += 1 - h[i]
                grad[i] -= h[i + s]
        
        grad /= len(max_indices)
        
        # Step size
        alpha = alpha0 / np.sqrt(iteration + 1)
        
        # Gradient step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        h_new = project_onto_box_sum(h_new, target_sum, 0.0, 1.0)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Accept if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        
        # Occasional random perturbation
        if iteration % 1000 == 0:
            # Add small random noise
            noise = np.random.normal(scale=0.01, size=n_points)
            h_pert = h + noise
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_onto_box_sum(h_pert, target_sum, 0.0, 1.0)
            c5_pert = compute_c5(h_pert, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert
                
        iteration += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
