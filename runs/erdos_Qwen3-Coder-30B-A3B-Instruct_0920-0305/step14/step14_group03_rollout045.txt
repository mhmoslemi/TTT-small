def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use Duchi's algorithm for projection
    # Sort x in descending order
    idx = np.argsort(x)[::-1]
    x_sorted = x[idx]
    sum_x = 0
    rho = 0
    
    for i in range(len(x_sorted)):
        sum_x += x_sorted[i]
        temp = (sum_x - target_sum) / (i + 1)
        if temp >= x_sorted[i]:
            rho = i + 1
            break
    
    if rho == 0:
        theta = (np.sum(x) - target_sum) / len(x)
        return np.clip(x - theta, 0, 1)
    else:
        theta = (np.sum(x_sorted[:rho]) - target_sum) / rho
        result = np.clip(x - theta, 0, 1)
        return result

def compute_gradient(h):

    # with respect to h(i) is the sum of the contributions from shifts where h(i) appears.
    
    # Compute 1-h
    one_minus_h = 1 - h
    
    # For each i, we calculate how much it contributes to the maximum
    grad = np.zeros_like(h)
    
    # This is a simplified computation - in practice we'd compute
    # the index of maximum correlation efficiently.
    # We compute all correlations to find the max.
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    
    # Compute full correlation
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Find the maximum correlation and its shift index
    max_corr_idx = np.argmax(corr)
    worst_shift = max_corr_idx - len(h) + 1
    
    # For this specific shift, compute gradient contribution for each position
    for i in range(len(h)):
        # Check if h[i] appears in this correlation term
        if 0 <= i - worst_shift < len(h):
            # If h[i] appears at position i relative to shift -worst_shift, 
            # it contributes d/dh[i] of the integration
            grad[i] += 1 - h[i - worst_shift]
        
        if 0 <= i + worst_shift < len(h):
            grad[i] += 1 - h[i + worst_shift]
    
    return grad
