def evaluate_h(h, dx):
    """Evaluate the C5 constant for given h values"""
    n = len(h)
    # Compute all correlations with different shifts
    correlations = []
    for k in range(1, n):
        corr = 0
        for i in range(n - k):
            corr += h[i] * (1 - h[i + k])
        correlations.append(corr * dx)
    
    return max(correlations) if correlations else 0

    )
    
    h_opt = np.clip(result.x, 0, 1)
    h_opt = h_opt * (n/2) / np.sum(h_opt)
