def project_onto_box_and_sum(v, target, lower=0.0, upper=1.0):
    # Clip v to within bounds, then adjust sum with lambda
    # Use bisection
    low = -1e5
    high = 1e5
    for _ in range(60):
        lam = (low + high)/2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    # final x
    x = np.clip(v - (low+high)/2, lower, upper)
    # Might fine-tune: adjusting to exact sum by moving small amounts across variables not at bounds.
    # That ensures sum close to target.
    diff = target - x.sum()
    # If diff magnitude small, ignore.
    # else adjust by distributing among non-bound elements:
    if abs(diff) > 1e-10:
        # indices not at bounds
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if len(free) > 0:
            # distribute equally
            x[free] += diff / len(free)
            # clip again
            x = np.clip(x, lower, upper)
    return x
