import numpy as np
from scipy.fft import fft, ifft
import time
import random

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def C5_value(h):
    """Evaluate C5"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of a coarse solution onto a finer grid"""
    old_n = len(old_h)
    old_positions = np.linspace(0, 2, old_n, endpoint=False) + dx(old_n)/2
    new_positions = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n)/2
    new_h = np.interp(new_positions, old_positions, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(c):
    """Get the shift index that achieves the maximum correlation"""
    idx = np.argmax(c)
    return idx - (len(c) // 2)

def shift_arrays(h, s):
    """Returns h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Exact sub-gradient of the worst term C_s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        # 1. Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
        else:
            h = interpolate(best_h_stage, n)
        h = project_to_sum_box(h, target_sum(n))
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # 2. Simulated annealing
        T = 0.02
        Δ = 0.1
        max_iter = int(100000 * time_fraction / budget_s)
        acceptance_count = 0
        accepted_moves = 0
        max_acceptance_ratio = 0
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * time_fraction:
                break
                
            # Decide type of move
            if random.random() < 0.3:
                # Targeted move (focus on worst shift)
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Select low desirability indices (bottom 20%)
                low_indices = np.argsort(d)[:max(1, len(d) // 5)]
                if len(low_indices) == 0:
                    low_indices = [0]
                i = random.choice(low_indices)
                
                # Select high desirability indices (top 20%)
                high_indices = np.argsort(d)[-max(1, len(d) // 5):]
                if len(high_indices) == 0:
                    high_indices = [0]
                j = random.choice(high_indices)
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                # Limit Δ based on box constraints
                Δ = min(Δ, h[i], 1 - h[j])
                if Δ <= 0:
                    continue
                
            # Make move
            δ = random.uniform(0, Δ)
            h_new = h.copy()
            h_new[i] -= δ
            h_new[j] += δ
            
            # Clip to box constraints
            h_new[i] = max(0, min(1, h_new[i]))
            h_new[j] = max(0, min(1, h_new[j]))
            
            # Evaluate
            c5_new = C5_value(h_new)
            c5_old = C5_value(h)
            
            if c5_new <= c5_old:
                accept = True
            else:
                accept = random.random() < np.exp(-(c5_new - c5_old) / T)
            
            if accept:
                h = h_new
                if c5_new < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = c5_new
                accepted_moves += 1
                acceptance_count += 1
                
            # Update temperature and step size
            if (iteration + 1) % 100 == 0:
                T *= 0.9997
                Δ *= 0.9999
                
                # Restart if acceptance rate is too low
                if acceptance_count > 0:
                    acceptance_ratio = accepted_moves / acceptance_count
                    max_acceptance_ratio = max(max_acceptance_ratio, acceptance_ratio)
                    if acceptance_ratio < 0.01 and max_acceptance_ratio > 0.01:
                        h = np.random.uniform(0, 1, n)
                        h = project_to_sum_box(h, target_sum(n))
                        accepted_moves = 0
                        acceptance_count = 0
                        T = 0.02
                        Δ = 0.1
        
        # 3. Polishing with exact sub-gradient descent
        h = best_h_stage.copy()
        best_h_polish = h.copy()
        best_C_polish = best_C_stage
        
        # Set number of polishing iterations
        max_polish = min(2000, int(5000 * time_fraction / budget_s))
        eta_0 = 0.5
        for k in range(max_polish):
            if time.time() - start_time > budget_s * time_fraction:
                break
                
            c = corr_fft(h)
            max_c = np.max(c)
            # Get all shifts that are close to maximum
            eps = 1e-12
            s_indices = np.where(c >= max_c - eps)[0]
            s_list = [idx - (len(c) // 2) for idx in s_indices]
            
            # Compute average gradient
            g = np.zeros(len(h))
            for s in s_list:
                g += gradient_worst(h, s)
            g /= len(s_list)
            
            # Step size
            eta = eta_0 / np.sqrt(k + 1)
            
            # Update
            h_new = h - eta * g
            
            # Project onto constraints
            h_new = project_to_sum_box(h_new, target_sum(n))
            
            c5_new = C5_value(h_new)
            if c5_new < best_C_polish:
                best_C_polish = c5_new
                best_h_polish = h_new.copy()
                
            h = h_new
            
        return best_h_polish, best_C_polish
    
    # Run stages
    best_h, best_C = make_stage(80, 0.30)
    best_h, best_C = make_stage(160, 0.35)
    best_h, best_C = make_stage(240, 0.35)
    
    return (best_h.tolist(), float(best_C), len(best_h))
