def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_and_grad(h, alpha=10.0):
    """Compute correlation and its gradient"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    
    # FFT correlation
    corr_fft = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr_fft * dx
    
    # Compute smooth max gradient
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    
    # Gradient computation:
    # d/dh_i of sum_j w_j * h_i * (1 - h_{i+j}) 
    # = w_{-i} * (1 - h_0) + w_{i} * (1 - h_{2i}) 
    # but with proper indexing
    
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for s in range(-len(h)+1, len(h)):
            if 0 <= i + s < len(h):
                if s == 0:
                    grad[i] -= w[s + len(h) - 1] * (1 - h[i])
                else:
                    # Contribution from h_{i+s} term
                    grad[i] -= w[s + len(h) - 1] * h[i+s]
                    
    return corr, grad
