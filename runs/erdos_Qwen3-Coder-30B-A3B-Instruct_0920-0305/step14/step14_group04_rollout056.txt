```python
import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use a better initial solution based on mathematical insight
    # Start with a simple piecewise constant function that's likely to be good
    h0 = np.zeros(n_points)
    # Create a smooth step function with better overlap properties
    midpoint = n_points // 2
    # Try a function that increases linearly then decreases
    for i in range(n_points):
        if i < midpoint:
            h0[i] = i / midpoint
        else:
            h0[i] = 1 - (i - midpoint) / (n_points - midpoint)
    
    # Normalize to correct sum
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Main optimization loop with simulated annealing approach
    max_iter = 30000
    step_size = 0.1
    temp = 1.0
    temp_decay = 0.9995
    patience = 0
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Simulated annealing step
        # Generate neighbor by perturbing randomly
        h_new = current_h.copy()
        idx = np.random.randint(n_points)
        # Add small random perturbation
        delta = np.random.normal(0, 0.02)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject based on simulated annealing
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = current_h.copy()
            patience = 0
            last_improvement = iter_num
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp(-(c5_new - current_c5) / temp):
                current_h = h_new
                current_c5 = c5_new
                patience = 0
            else:
                patience += 1
        
        # Cool down temperature
        temp *= temp_decay
        
        # Reduce step size if stuck for too long
        if patience > 500:
            step_size *= 0.95
            patience = 0
            
        # Periodic refocusing
        if iter_num % 1000 == 0 and iter_num > 0:
            # Reset with a new random perturbation of best solution
            if best_h is not None:
                h_pert = best_h.copy()
                for i in range(n_points // 10):
                    idx = np.random.randint(n_points)
                    delta = np.random.normal(0, 0.03)
                    h_pert[idx] = np.clip(h_pert[idx] + delta, 0, 1)
                h_pert = project_to_sum_box(h_pert, TARGET_SUM)
                current_h = h_pert
                current_c5 = evaluate_c5(h_pert)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()

    # Final local search refinement
    if best_h is not None:
        h = best_h.copy()
        t_final = start_time + budget_s - 5
        for i in range(5000):
            if time.time() >= t_final:
                break
                
            # Local perturbation with larger steps for better exploration
            h_pert = h.copy()
            n_perturb = np.random.randint(1, 5)
            for _ in range(n_perturb):
                idx1 = np.random.randint(n_points)
                idx2 = np.random.randint(n_points) 
                delta = np.random.normal(0, 0.03)
                h_pert[idx1] = np.clip(h_pert[idx1] + delta, 0, 1)
                h_pert[idx2] = np.clip(h_pert[idx2] - delta, 0, 1)
            
            h_pert = project_to_sum_box(h_pert, TARGET_SUM)
            
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < evaluate_c5(h):
                h = h_pert
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
```
