import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project onto the capped simplex using bisection"""
    def sum_with_lambda(lam):
        proj = np.clip(x + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        if sum_with_lambda(a) * sum_with_lambda(b) > 0:
            a -= 0.1
            b += 0.1
        else:
            break
    
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m)
        if abs(fm) < 1e-12:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def evaluate_c5(h, dx):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
    return dx * np.max(corr)

def get_gradient(corr, h, T, K):
    """Compute gradient using top-K shifts"""
    # Get indices of top K shifts
    idx_top = np.argpartition(corr, -K)[-K:]
    # Compute soft-max weights
    corr_top = corr[idx_top]
    w = np.exp((corr_top - corr_top.max()) / T)
    w = w / np.sum(w)
    
    # Compute gradient for these shifts only
    grad = np.zeros_like(h)
    for i, s in enumerate(idx_top):
        if s >= 0:
            # Add contribution from h_i * (1 - h_{i+s})
            if s == 0:
                grad += w[i] * (1 - h)
            else:
                grad[:len(h)-s] += w[i] * (1 - h[s:])
                grad[s:] -= w[i] * h[:-s]
        else:
            # s < 0, so h_{i+s} is at position i+s (negative shift)
            s = -s
            grad[s:] += w[i] * (1 - h[:-s])
            grad[:len(h)-s] -= w[i] * h[s:]
    
    return grad

def smooth_max_objective(h, T, K):
    """Compute the smooth max objective"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
    
    # Get top K shifts
    idx_top = np.argpartition(corr, -K)[-K:]
    corr_top = corr[idx_top]
    
    # Soft max
    corr_top = corr_top - corr_top.max()
    w = np.exp(corr_top / T)
    w = w / np.sum(w)
    
    return T * np.log(np.sum(w * np.exp(corr_top)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 5000
    
    # Coarse optimization phase
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_coarse) - 1))
    h_coarse = project_to_capped_simplex(h_coarse, target_sum / 2)
    
    # Run coarse continuation
    T0 = 0.1
    alpha0 = 0.6
    K = 10
    beta = 0.9
    tau_T = 0.9995
    tau_alpha = 0.999
    
    T = T0
    alpha = alpha0
    v = np.zeros_like(h_coarse)
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = evaluate_c5(h_coarse, dx_coarse)
    
    for iter_num in range(2000):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Compute correlation
        h_rev = h_coarse[::-1]
        b = 1 - h_coarse
        L = 2 * len(h_coarse) - 1
        corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
        
        # Get gradient
        grad = get_gradient(corr, h_coarse, T, K)
        
        # Update with momentum
        v = beta * v + (1 - beta) * grad
        step_size = alpha / np.sqrt(iter_num + 1)
        step_size = min(step_size, 0.5 / (np.max(np.abs(grad)) + 1e-10))
        h_coarse_new = h_coarse - step_size * v
        
        # Project
        h_coarse_new = project_to_capped_simplex(h_coarse_new, target_sum / 2)
        
        # Update
        h_coarse = h_coarse_new
        T *= tau_T
        alpha *= tau_alpha
        
        # Evaluate
        c5 = evaluate_c5(h_coarse, dx_coarse)
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_h_coarse = h_coarse.copy()
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), best_h_coarse)
    h_fine = project_to_capped_simplex(h_fine, target_sum)
    
    # Global optimization with multiple restarts
    best_h = h_fine.copy()
    best_c5 = evaluate_c5(h_fine, dx)
    
    # Run multiple restarts
    num_restarts = 12
    for restart in range(num_restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Initialize with different strategies
        if restart == 0:
            # Start with triangular
            h_start = np.maximum(0, 1 - np.abs(np.linspace(0, 2, n_points) - 1))
        elif restart % 3 == 0:
            # Random start
            h_start = np.random.rand(n_points)
        else:
            # Mirror symmetric start
            h_start = np.random.rand(n_points)
            h_start = (h_start + h_start[::-1]) / 2
        
        h_start = project_to_capped_simplex(h_start, target_sum)
        
        # Run continuation for this restart
        h = h_start.copy()
        T = T0
        alpha = alpha0
        v = np.zeros_like(h)
        
        for iter_num in range(3000):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute correlation
            h_rev = h[::-1]
            b = 1 - h
            L = 2 * len(h) - 1
            corr = irfft(rfft(h_rev, L) * rfft(b, L), L)
            
            # Get gradient
            grad = get_gradient(corr, h, T, K)
            
            # Update with momentum
            v = beta * v + (1 - beta) * grad
            step_size = alpha / np.sqrt(iter_num + 1)
            step_size = min(step_size, 0.5 / (np.max(np.abs(grad)) + 1e-10))
            h_new = h - step_size * v
            
            # Project
            h_new = project_to_capped_simplex(h_new, target_sum)
            
            # Update
            h = h_new
            T *= tau_T
            alpha *= tau_alpha
            
            # Evaluate
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Local polishing with pair-swap
    h_polish = best_h.copy()
    current_c5 = best_c5
    patience = 0
    max_patience = 5000
    
    for _ in range(100000):
        if time.time() - start_time > budget_s:
            break
            
        # Pair swap
        i, j = np.random.choice(n_points, 2, replace=False)
        delta_low = max(-h_polish[i], h_polish[j] - 1)
        delta_high = min(1 - h_polish[i], h_polish[j])
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        h_new = h_polish.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Project to feasible set
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h_polish = h_new
            current_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
    
    # Final projection and evaluation
    final_h = project_to_capped_simplex(h_polish, target_sum)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), float(final_c5), n_points)
