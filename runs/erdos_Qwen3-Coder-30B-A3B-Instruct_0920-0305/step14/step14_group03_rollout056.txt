import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def compute_softmax_grad(h, corr, alpha=10.0):
    """Compute gradient using softmax approximation of max"""
    max_corr = np.max(corr)
    if max_corr > 0:
        weights = np.exp(alpha * (corr - max_corr))
    else:
        weights = np.exp(alpha * corr)
    weights = weights / np.sum(weights)
    
    n = len(h)
    grad = np.zeros(n)
    
    # Process all shifts
    for i in range(n):
        for s in range(-n+1, n):
            if s == 0:
                grad[i] -= 2 * weights[s + n - 1] * h[i]
            elif s > 0:
                if i + s < n:
                    grad[i] -= weights[s + n - 1] * h[i + s]
                if i - s >= 0:
                    grad[i] -= weights[-s + n - 1] * h[i - s]
            else:  # s < 0
                if i - s < n:
                    grad[i] -= weights[-s + n - 1] * h[i - s]
                if i + s >= 0:
                    grad[i] -= weights[s + n - 1] * h[i + s]
    
    return grad
