import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * (target_sum - corr)

def compute_gradient(h, w, dx):
    """Compute gradient using softmax weights"""
    n = len(h)
    grad = np.zeros(n)
    
    # Loop through all shifts
    for r in range(2 * n - 1):
        s = r - (n - 1)
        ws = w[r]
        
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:
            # s < 0
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Candidates for n_points
    n_candidates = [120, 150, 180, 200]
    
    for n in n_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        TARGET_SUM = n / 2
        dx = 2.0 / n
        
        # Multiple restarts
        for restart in range(3):
            if time.time() - start_time > 0.9 * budget_s:
                break
            
            # Initialize h
            if restart == 0:
                # Use initial solution if available
                if 'initial_h_values' in globals() and len(initial_h_values) == n:
                    h0 = np.array(initial_h_values)
                else:
                    # Create a reasonable initial solution
                    h0 = np.full(n, 0.5)
            elif restart == 1:
                # Random initialization
                h0 = np.random.rand(n)
                h0 = h0 / np.sum(h0) * TARGET_SUM
            else:
                # Sinusoidal initialization
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h0 = 0.5 + 0.3 * np.cos(np.pi * x)
                h0 = np.clip(h0, 0, 1)
            
            h0 = project_to_sum_box(h0, TARGET_SUM)
            current_h = h0.copy()
            current_c5 = evaluate_c5(h0)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_n = n
            
            # Optimization loop
            max_iter = 5000
            lr = 0.1
            alpha = 10.0
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            m = np.zeros_like(current_h)
            v = np.zeros_like(current_h)
            t = 0
            
            for iteration in range(max_iter):
                if time.time() - start_time > 0.9 * budget_s:
                    break
                    
                # Compute overlap values
                O = compute_O(current_h, dx, TARGET_SUM)
                C5 = O.max()
                
                # Keep track of best
                if C5 < best_c5:
                    best_c5 = C5
                    best_h = current_h.copy()
                    best_n = n
                
                # Compute softmax weights
                max_O = O.max()
                exp_term = np.exp(alpha * (O - max_O))
                w = exp_term / np.sum(exp_term)
                
                # Compute gradient
                grad = compute_gradient(current_h, w, dx)
                
                # Adam updates
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1**(t+1))
                v_hat = v / (1 - beta2**(t+1))
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # Update with projection
                new_h = current_h - step
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                new_c5 = evaluate_c5(new_h)
                
                if new_c5 < C5:
                    current_h = new_h
                    current_c5 = new_c5
                else:
                    # Reduce learning rate
                    lr *= 0.5
                
                # Adaptive parameters
                if iteration % 100 == 0:
                    alpha = min(alpha * 1.05, 200.0)
                
                t += 1
    
    if best_h is None:
        # Fallback to initial solution
        best_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
        best_c5 = evaluate_c5(best_h)
        best_n = n_points
    
    return (best_h.tolist(), float(best_c5), best_n)
