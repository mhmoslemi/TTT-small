import numpy as np
from scipy.signal import correlate
from scipy.fftpack import rfft, irfft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def smooth_max_correlation(h, T=0.1, K=10):
    """Compute smooth approximation of max correlation with temperature T"""
    # Compute full correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Get top K shifts
    top_k_indices = np.argpartition(corr, -K)[-K:]
    top_k_values = corr[top_k_indices]
    
    # Softmax weights
    max_corr = top_k_values.max()
    exp_values = np.exp((top_k_values - max_corr) / T)
    weights = exp_values / exp_values.sum()
    
    # Gradient computation (simplified)
    grad = np.zeros_like(h)
    for i, (s, w) in enumerate(zip(top_k_indices, weights)):
        if s >= 0:
            # Correlation with positive shift
            for j in range(len(h) - s):
                grad[j] += w * (1 - h[j + s])
            for j in range(s, len(h)):
                grad[j] -= w * h[j - s]
        else:
            # Correlation with negative shift
            s_abs = -s
            for j in range(len(h) - s_abs):
                grad[j] += w * (1 - h[j + s_abs])
            for j in range(s_abs, len(h)):
                grad[j] -= w * h[j - s_abs]
    
    return corr, grad, top_k_indices

def gradient_step(h, grad, step_size):
    """Perform gradient step with clipping"""
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 10000
    K = 10
    
    # Initial triangular seed
    x = np.linspace(0, 2, n_points)
    h0 = np.maximum(0, 1 - np.abs(x - 1))
    h0 = project_to_sum_box(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Coarse optimization phase
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = np.maximum(0, 1 - np.abs(x_coarse - 1))
    h_coarse = project_to_sum_box(h_coarse, n_coarse / 2)
    
    # Coarse optimization with fewer iterations
    T = 0.1
    alpha = 0.6
    beta = 0.9
    momentum = np.zeros_like(h_coarse)
    
    for i in range(1000):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        corr, grad, top_k_indices = smooth_max_correlation(h_coarse, T, K)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        step_size = alpha / np.sqrt(i + 1)
        step_size = min(step_size, 0.5 / np.abs(grad).max() if np.abs(grad).max() > 0 else 1.0)
        
        h_coarse_new = gradient_step(h_coarse, momentum, step_size)
        h_coarse_new = project_to_sum_box(h_coarse_new, n_coarse / 2)
        
        h_coarse = h_coarse_new
        
        # Periodic evaluation
        if i % 20 == 0:
            c5 = evaluate_c5(h_coarse)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_coarse.copy()
        
        # Anneal temperature
        T *= 0.9995
        alpha *= 0.999
    
    # Interpolate to fine grid
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_to_sum_box(h_fine, target_sum)
    
    # Multiple restarts with fine grid
    for restart in range(15):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Random initialization
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, target_sum)
        
        # Add symmetry for some restarts
        if restart % 3 == 0:
            h = (h + h[::-1]) / 2
            h = project_to_sum_box(h, target_sum)
        
        T = 0.1
        alpha = 0.6
        beta = 0.9
        momentum = np.zeros_like(h)
        
        for i in range(3000):
            if time.time() - start_time > budget_s:
                break
                
            corr, grad, top_k_indices = smooth_max_correlation(h, T, K)
            
            momentum = beta * momentum + (1 - beta) * grad
            step_size = alpha / np.sqrt(i + 1)
            step_size = min(step_size, 0.5 / np.abs(grad).max() if np.abs(grad).max() > 0 else 1.0)
            
            h_new = gradient_step(h, momentum, step_size)
            h_new = project_to_sum_box(h_new, target_sum)
            
            h = h_new
            
            if i % 20 == 0:
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            T *= 0.9995
            alpha *= 0.999
    
    # Local pair-swap polishing
    h = best_h.copy()
    T_final = 0.01
    max_swaps = 100000
    
    for i in range(max_swaps):
        if time.time() - start_time > budget_s:
            break
            
        # Pick two random indices
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        
        # Compute feasible interval for transfer
        delta_low = max(-h[i1], h[i2] - 1)
        delta_high = min(1 - h[i1], h[i2])
        
        if delta_low >= delta_high:
            continue
            
        # Make random transfer
        delta = np.random.uniform(delta_low, delta_high)
        
        # Apply transfer
        h_new = h.copy()
        h_new[i1] += delta
        h_new[i2] -= delta
        
        # Check constraints
        if np.any(h_new < 0) or np.any(h_new > 1):
            continue
            
        c5_new = evaluate_c5(h_new)
        c5_current = evaluate_c5(h)
        
        if c5_new < c5_current:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Accept with simulated annealing probability
            p_accept = np.exp(-(c5_new - c5_current) / T_final)
            if np.random.rand() < p_accept:
                h = h_new
    
    # Final projection and evaluation
    best_h = project_to_sum_box(best_h, target_sum)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
