import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 using FFT"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of a coarse solution onto a finer grid"""
    old_x = np.linspace(0, 2, len(old_h))
    new_x = np.linspace(0, 2, new_n)
    new_h = np.interp(new_x, old_x, old_h)
    return new_h

def worst_shift(corr):
    """Return the shift index that gives maximum correlation"""
    idx = np.argmax(corr)
    n = len(corr)
    s = idx - (n - 1)
    return s

def shift_arrays(h, s):
    """Returns h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    g = 1 - h_plus - h_minus
    return g

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    def make_stage(n, time_fraction):
        # Time allocation
        stage_start = time.time()
        stage_end = start_time + budget_s * time_fraction
        
        # Initialize
        if n == 80:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, target_sum(n))
        else:
            h = interpolate(best_h_stage, n)
            h = project_to_sum_box(h, target_sum(n))
            
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Simulated annealing parameters
        T = 0.02
        alpha = 0.9997
        beta = 0.9999
        delta = 0.1
        max_iter = int(10000 * time_fraction / 0.35) if time_fraction < 0.35 else 10000
        
        # Annealing loop
        for iter_num in range(max_iter):
            if time.time() > stage_end:
                break
                
            # Decide move type (targeted or random)
            if random.random() < 0.3:
                # Targeted move
                c = corr_fft(h)
                s = worst_shift(c)
                d = np.zeros(n)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Select low desirability indices
                low_indices = np.argsort(d)[:int(0.2 * n)]
                if len(low_indices) == 0:
                    low_indices = [0]
                i = random.choice(low_indices)
                
                # Select high desirability indices
                high_indices = np.argsort(d)[-int(0.2 * n):]
                if len(high_indices) == 0:
                    high_indices = [n-1]
                j = random.choice(high_indices)
                
                # Perform transfer
                max_transfer = min(h[i], 1 - h[j])
                delta_transfer = random.uniform(0, min(delta, max_transfer))
                h[i] -= delta_transfer
                h[j] += delta_transfer
                h[i] = max(0, h[i])
                h[j] = min(1, h[j])
            else:
                # Random move
                i, j = random.sample(range(n), 2)
                max_transfer = min(h[i], 1 - h[j])
                delta_transfer = random.uniform(0, min(delta, max_transfer))
                h[i] -= delta_transfer
                h[j] += delta_transfer
                h[i] = max(0, h[i])
                h[j] = min(1, h[j])
            
            # Compute new C5 value
            new_C = C5_value(h)
            
            # Accept with Metropolis probability
            if new_C <= best_C_stage or random.random() < np.exp(-(new_C - best_C_stage) / T):
                if new_C < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = new_C
            
            # Update temperature and delta
            T = alpha * T
            delta = beta * delta
        
        # Polishing
        c = corr_fft(best_h_stage)
        s = worst_shift(c)
        best_h_stage = best_h_stage.copy()
        
        # Few iterations of gradient descent
        for k in range(2000):
            if time.time() > stage_end:
                break
                
            c = corr_fft(best_h_stage)
            max_val = np.max(c)
            epsilon = 1e-12
            indices = np.where(np.abs(c - max_val) < epsilon)[0]
            s = indices[0] - (len(c) - 1)
            
            grad = gradient_worst(best_h_stage, s)
            
            # Adaptive step size
            eta = 0.5 / np.sqrt(k + 1)
            
            # Update
            new_h = best_h_stage - eta * grad
            
            # Project
            new_h = project_to_sum_box(new_h, target_sum(n))
            new_C = C5_value(new_h)
            
            if new_C < best_C_stage:
                best_h_stage = new_h
                best_C_stage = new_C
        
        return best_h_stage, best_C_stage
    
    # Run stages
    best_h, best_C = make_stage(80, 0.30)
    best_h, best_C = make_stage(160, 0.35)
    best_h, best_C = make_stage(240, 0.35)
    
    return (best_h.tolist(), float(best_C), len(best_h))
