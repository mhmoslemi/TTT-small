import numpy as np
from scipy.fft import rfft, irfft
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), L).real
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max_gradient(h, corr, T, K=10):
    """Compute gradient of soft max approximation"""
    # For numerical stability
    corr_stable = corr - np.max(corr)
    exp_corr = np.exp(corr_stable / T)
    weights = exp_corr / np.sum(exp_corr)
    
    # Select top K shifts
    top_k_indices = np.argpartition(weights, -K)[-K:]
    weights_top = weights[top_k_indices]
    weights_top = weights_top / np.sum(weights_top)
    
    # Compute gradient using top K indices
    grad = np.zeros_like(h)
    
    for i, s in enumerate(top_k_indices):
        if s >= 0:
            # First term: 1 - h_{i+s} for valid indices
            if s < len(h):
                grad[:len(h)-s] += weights_top[i] * (1 - h[s:])
            # Second term: -h_{i-s} for valid indices
            if s > 0:
                grad[s:] -= weights_top[i] * h[:len(h)-s]
        else:
            # s < 0, so h_{i+s} is h_{i+s} where i+s >= 0
            s_abs = -s
            if s_abs < len(h):
                grad[s_abs:] += weights_top[i] * (1 - h[:len(h)-s_abs])
            # Second term: -h_{i-s} 
            if len(h) > s_abs:
                grad[:len(h)-s_abs] -= weights_top[i] * h[s_abs:]
    
    return grad
