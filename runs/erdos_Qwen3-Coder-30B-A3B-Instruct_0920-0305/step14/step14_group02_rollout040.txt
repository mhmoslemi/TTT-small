import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # Forward part
        g[:N-s] += 1 - h[s:]
        # Backward part
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        # Forward part
        g[s_abs:] += 1 - h[:N-s_abs]
        # Backward part
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute smooth gradient using softmax weights"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just compute subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h) - 1)
        return subgradient_for_shift(h, shift)
    else:
        # Compute weights using softmax
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N - 1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart"""
    # Initialize
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    T = T0
    step = step0
    
    for t in range(max_iter):
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update with momentum
        step = step0 / np.sqrt(t + 1)
        h = h - step * g
        
        # Apply symmetry constraint if needed
        if sym:
            h = (h + h[::-1]) / 2
        
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Evaluate
        c5 = evaluate_c5_fast(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay temperature
        T *= 0.9995
        
        if t % 20 == 0:
            # Early stopping for very small improvement
            if np.abs(c5 - best_c5) < 1e-8:
                break
    
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local hill-climbing with pair-swaps"""
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    N = len(h)
    
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_min = max(-best_h[i], best_h[j] - 1)
        delta_max = min(1 - best_h[i], best_h[j])
        
        if delta_min >= delta_max:
            continue
            
        delta = np.random.uniform(delta_min, delta_max)
        
        # Apply the swap
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate new C5
        new_c5 = evaluate_c5_fast(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose resolution
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Coarse optimization first (n=60)
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    h_init_coarse = np.random.rand(n_coarse)
    h_init_coarse = project_onto_capped_simplex(h_init_coarse, target_sum_coarse)
    h_best_coarse, c5_coarse = gradient_descent_restart(
        n_coarse, target_sum_coarse, max_iter=1000, 
        step0=0.5, T0=0.1, top_k=10, sym=False
    )
    
    if c5_coarse < best_c5:
        best_c5 = c5_coarse
        best_h = h_best_coarse.copy()
    
    # Upsample coarse solution to target resolution
    if best_h is not None:
        x_coarse = np.linspace(dx_coarse/2, 2-dx_coarse/2, n_coarse)
        x_target = np.linspace(dx/2, 2-dx/2, n_points)
        h_upsampled = np.interp(x_target, x_coarse, best_h)
        h_upsampled = project_onto_capped_simplex(h_upsampled, TARGET_SUM)
        best_h = h_upsampled.copy()
        best_c5 = evaluate_c5_fast(best_h)
    
    # Multiple restarts at full resolution
    restarts = 15
    restart_count = 0
    
    while restart_count < restarts and (time.time() - start_time) < budget_s * 0.7:
        # Different initialization strategies: random, symmetric random, triangular
        if restart_count == 0:
            # Triangular shape
            x = np.linspace(dx/2, 2-dx/2, n_points)
            h0 = np.maximum(0, 1 - np.abs(x-1))
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        elif restart_count < 5:
            # Symmetric random
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
            h0 = (h0 + h0[::-1]) / 2
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        else:
            # Standard random
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        
        # Run gradient descent
        h_result, c5_result = gradient_descent_restart(
            n_points, TARGET_SUM, max_iter=2000, 
            step0=0.5, T0=0.1, top_k=10, sym=(restart_count % 3 == 0)
        )
        
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result.copy()
        
        restart_count += 1
    
    # Local refinement with swap search
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time)
    
    # Final evaluation
    if best_h is None:
        # Fallback
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
    
    # Ensure final projection
    best_h = project_onto_capped_simplex(best_h, TARGET_SUM)
    final_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
