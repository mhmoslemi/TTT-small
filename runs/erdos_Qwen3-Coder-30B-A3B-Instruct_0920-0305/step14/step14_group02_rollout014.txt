import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_correlation(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation(h)
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm for efficient projection
    # Sort the values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Compute cumulative sums from the back
    cumsum = np.cumsum(sorted_x[::-1])[::-1]
    
    # Find the threshold index using bisection
    def find_threshold():
        low, high = 0, len(sorted_x)
        while low < high:
            mid = (low + high + 1) // 2
            threshold = sorted_x[mid - 1] if mid > 0 else 0
            total = np.sum(sorted_x[:mid]) + threshold * (len(sorted_x) - mid)
            if total <= target_sum:
                low = mid
            else:
                high = mid - 1
        return low
    
    threshold_idx = find_threshold()
    
    # Adjust values
    result = x.copy()
    if threshold_idx == len(sorted_x):
        # All elements are below their thresholds
        # Add a scalar to adjust sum
        excess = target_sum - np.sum(result)
        result += excess / len(result)
        result = np.clip(result, 0, 1)
    else:
        # Threshold somewhere in the middle
        threshold = sorted_x[threshold_idx - 1] if threshold_idx > 0 else 0
        # Elements below threshold get adjusted
        for i in range(threshold_idx, len(sorted_x)):
            result[sorted_indices[i]] = threshold
        #Then we adjust the elements above threshold
        remaining_sum = target_sum - np.sum(result)
        # Distribute remaining_sum among elements >= threshold
        above_threshold_count = len(sorted_x) - threshold_idx
        if above_threshold_count > 0:
            adjustment = remaining_sum / above_threshold_count
            for i in range(threshold_idx):
                new_val = result[sorted_indices[i]] + adjustment
                result[sorted_indices[i]] = np.clip(new_val, 0, 1)
    
    return result

def compute_gradient(h, T=0.05):
    """Compute the gradient of the log-sum-exp surrogate function"""
    corr = compute_correlation(h)
    # Log-sum-exp smoothing
    max_corr = np.max(corr)
    exp_corr = np.exp((corr - max_corr) / T)
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    
    # Approximate using top K shifts for efficiency
    top_k = min(12, len(corr))
    top_shifts = np.argsort(corr)[-top_k:][::-1]
    
    grad = np.zeros_like(h)
    for s in top_shifts:
        # Compute gradient of c_s
        for i in range(len(h)):
            if 0 <= i <= len(h) - 1 - s:
                grad[i] += weights[s] * (1 - h[i + s])
            if s <= i <= len(h) - 1:
                grad[i] -= weights[s] * h[i - s]
    
    return grad

def smooth_optimize(h_start, T0=0.12, tau_T=0.998, max_iter=2000):
    """Optimize using projected L-BFGS-B with temperature continuation"""
    h = h_start.copy()
    
    # Initial projection
    h = project_to_sum_box(h, TARGET_SUM)
    
    T = T0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, T)
        
        # Use projected gradient descent step (simplified L-BFGS)
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project to box constraints and sum
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check if improvement
        current_c5 = evaluate_c5(h_new)
        if current_c5 >= prev_c5 and T < 5e-6:
            break
        elif current_c5 < prev_c5:
            h = h_new
            prev_c5 = current_c5
            
        # Update temperature
        T *= tau_T
        if T < 5e-6:
            break
    
    return h

def pairwise_exchange_optimize(h_start, max_iter=1000):
    """Perform exact line search pairwise exchanges"""
    h = h_start.copy()
    prev_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        improved = False
        # Shuffle indices for better exploration
        indices = list(range(len(h)))
        np.random.shuffle(indices)
        
        for i in indices:
            for j in indices:
                if i >= j:
                    continue
                # Get current correlation vector
                corr = compute_correlation(h)
                
                # Compute the increment for moving delta from i to j
                # For each shift, compute how corr changes with small delta
                def compute_delta_changes(delta):
                    new_c = corr.copy()
                    # update the correlations that depend on i and j
                    for s in range(len(corr)):  
                        if 0 <= i <= len(h) - 1 - s:
                            new_c[s] += delta * (1 - h[i + s])  # increase for h[i] decrease  
                        if s <= i <= len(h) - 1:
                            new_c[s] -= delta * h[i - s]  # decrease for h[i] decrease:
                        if 0 <= j <= len(h) - 1 - s:
                            new_c[s] -= delta * (1 - h[j + s]) # decrease for h[j] increase
                        if s <= j <= len(h) - 1:
                            new_c[s] += delta * h[j - s]  # increase for h[j] increase
                    return new_c
                
                # Find feasible range for delta
                delta_min = max(-h[j], h[i] - 1)
                delta_max = min(h[i], 1 - h[j])
                if delta_min >= delta_max:
                    continue
                    
                # Try multiple delta values
                best_delta = None
                min_max_corr = float('inf')
                for delta in np.linspace(delta_min, delta_max, 50):
                    new_c = compute_delta_changes(delta)
                    max_corr = np.max(new_c)
                    if max_corr < min_max_corr:
                        min_max_corr = max_corr
                        best_delta = delta
                        
                if best_delta is not None and min_max_corr < prev_c5 - 1e-7:
                    # Apply the exchange
                    h[i] -= best_delta
                    h[j] += best_delta
                    h = project_to_sum_box(h, TARGET_SUM)
                    prev_c5 = min_max_corr
                    improved = True
                    break
                    
            if improved:
                break
        
        if not improved:
            break
    
    return h

def triangular_seed(n):
    """Create a triangular shaped initial function"""
    x = np.linspace(0, 2, n)
    h = np.maximum(0, 1 - np.abs(x - 1))
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Coarse optimization phase
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    
    # Initialize with triangular function
    h0_coarse = triangular_seed(n_coarse)
    h0_coarse = project_to_sum_box(h0_coarse, TARGET_SUM / 2)
    
    # Interpolate to full grid
    h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h0_coarse)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    # Run optimization
    h_opt = smooth_optimize(h0)
    final_c5 = evaluate_c5(h_opt)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_opt.copy()
        
    # Local polishing with pairwise exchanges
    h_polish = pairwise_exchange_optimize(h_opt, 500)
    polish_c5 = evaluate_c5(h_polish)
    
    if polish_c5 < best_c5:
        best_c5 = polish_c5
        best_h = h_polish.copy()
        
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
