def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr_gradient(h):
    """Compute the gradient of the correlation with respect to h"""
    # Compute full convolution of h with (1-h) and its gradient
    L = 2 * len(h) - 1
    
    # This is a gradient calculation that should result in a computationally fast gradient
    # Let's compute the correlation explicitly and take partial derivatives
    # The observation is that for correlation with shift k, the
    # gradient w.r.t. h[i] is [1-h[i+k]] for k >= 0, and [1-h[i-k]] for k <= 0
    
    # We compute gradient pattern for the maximum correlation
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_shift = max_idx - len(h) + 1
    
    # Create gradient for the maximum shift case
    grad_corr = np.zeros(len(h))
    for i in range(len(h)):
        j_forward = i + max_shift
        j_backward = i - max_shift
        
        if 0 <= j_forward < len(h):
            grad_corr[i] += (1 - h[j_forward])  # Element of (1-h) at j_forward contributes to grad w.r.t. h[i]
        if 0 <= j_backward < len(h) and max_shift != 0:  # avoid double counting self
            grad_corr[i] += (1 - h[j_backward])
    
    return grad_corr

def clear_soft_max(h, alpha=50.0):
    """Compute smooth approximation to max using softmax with parameter alpha"""
    # compute correlations
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Compute softmax approximation to maximum
    max_value = np.max(corr)
    # Add small epsilon to prevent overflow
    exp_corr = np.exp(alpha * (corr - max_value))
    # Total: sum of exps
    total = np.sum(exp_corr)
    if total > 0:
        weights = exp_corr / total
    else:
        weights = np.ones_like(corr) / len(corr)
    
    # The following is an estimate of derivative using weights 
    return weights, corr

def compute_smooth_max_grad(h, weights=None, corr=None, alpha=50.0):
    """Correct and complete computation of smooth gradient"""
    if weights is None or corr is None:
        w, c = clear_soft_max(h, alpha)
        weights, corr = w, c
    
    # Compute gradient of weighted sum
    # This is a more accurate approach:
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    grad = np.zeros_like(h)
    
    # The sensitivity of max term with respect to h[i] depends on where it appears in the correlation
    # Compute the contribution of all shifts with weights
    
    # For each position i, since corr depends on distances:
    # At position s in corr, we have terms where h[i] is multiplied by (1-h[i+s]) and (1-h[i-s])
    # In fact, for full correlation, if we know at stage s we are solving for max shift s,
    # we should compute gradient using only that correlation contribution
    # But here, we compute gradient of the entire multi-shift problem
    
    # Efficient way: for smooth max, derivative w.r.t. h[i] involves relative shifts
    # Simplifying approach:
    # We compute both forward and backward contributions
    for i in range(len(h)):
        # Contributions to correlation at shifts where h[i] appears
        # This is angle between terms in the convolution
        for s in range(-len(h)+1, len(h)):
            j = i + s
            if 0 <= j < len(h):
                # Weighted gradient for this contribution
                # in the direction of shifted additions to corr
                if s > 0:
                    weight_idx = s + len(h) - 1
                else:
                    weight_idx = len(h) - 1 + s
                    
                if 0 <= weight_idx < len(weights):
                    if s > 0:
                        # weight is scaled to local derivatives
                        grad[i] += weights[weight_idx] * (-h[j])  # derivative of (1-h[j]) w.r.t. h[i]
                    elif s == 0:
                        grad[i] += weights[weight_idx] * (-1)  # derivative of (1-h[i]) w.r.t. h[i]
                    else:
                        grad[i] += weights[weight_idx] * (-h[j])  # derivative of (1-h[j]) w.r.t. h[i]
    
    return grad

def smooth_max_optimize(h0, max_iter=2000, lr=0.1, alpha=1.0, patience=50):
    """Optimization using smooth max approximation"""
    current_h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    nan_count = 0
    
    # Adaptive parameters

    for alpha_level, alpha in enumerate(alpha_schedule):

        for i in range(max_iter):
            if i % 100 == 0:

                print(f"Iteration {i}, C5 = {current_c5}, alpha = {alpha}")
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            # Compute soft max approximation and useful gradients
            weights, corr = clear_soft_max(current_h, alpha)
            
            # Compute gradient
            grad = compute_smooth_max_grad(current_h, weights, corr, alpha)
            
            # Check for numerical issues
            if np.any(np.isnan(grad)) or np.any(np.isinf(grad)):
                nan_count += 1
                print(f"Detected NaN/inf in gradient, count = {nan_count}")
                if nan_count > 5:
                    break
                # Add small noise to current solution
                current_h += np.random.normal(0, 0.001, len(current_h))
                current_h = project_to_sum_box(current_h, TARGET_SUM)
                continue

            new_h = current_h - lr * grad
            new_h = project_to_sum_box(new_h, TARGET_SUM)

            if new_c5 <= current_c5:
                current_h = new_h
            else:
                lr *= 0.5  # Reduce learning rate if no improvement
                        
    return best_h, best_c5
