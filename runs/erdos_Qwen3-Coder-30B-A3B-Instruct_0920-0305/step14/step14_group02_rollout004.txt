import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_exact(h):
    """Evaluate C5 exactly using FFT correlation"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.05):
    """Evaluate smooth surrogate for C5 using log-sum-exp"""
    # Compute all correlations efficiently
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Apply log-sum-exp smoothing
    scaled_corr = corr / T
    # Subtract max to avoid overflow
    max_corr = np.max(scaled_corr)
    exp_terms = np.exp(scaled_corr - max_corr)
    log_sum_exp = max_corr + np.log(np.sum(exp_terms))
    
    return dx * log_sum_exp

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using waterfilling"""
    # Waterfilling algorithm for projection onto simplex
    x = np.array(x)
    sorted_x = np.sort(x)[::-1]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the threshold
    # We want sum(max(0, x[i] - lambda)) = target_sum
    # This becomes sum(max(0, sorted_x[i] - lambda)) = target_sum
    # For lambda in range [sorted_x[n-1], sorted_x[0]], 
    # sum = (sorted_x[0] - lambda) + ... + (sorted_x[i] - lambda) where sorted_x[i] > lambda
    # For i such that sorted_x[i] > lambda
    # sum = cumsum[i] - (i+1)*lambda
    # We want sum = target_sum
    
    thresholds = sorted_x - (cumsum - target_sum) / np.arange(1, n+1)
    
    # Find largest threshold <= 0, i.e., largest i such that sorted_x[i] - (cumsum[i] - target_sum)/ (i+1) <= 0
    try:
        # Find largest index where is valid
        idx = np.where(thresholds <= 0)[0][-1]
        lambda_val = thresholds[idx]
        # Compute the projection
        result = np.maximum(0, x - lambda_val)
        # Adjust to ensure exact sum
        diff = target_sum - np.sum(result)
        if abs(diff) > 1e-10:
            # Simple adjustment: distribute error
            result += diff / len(result)
    except IndexError:
        # All values are positive (bounded by 1)
        lambda_val = np.minimum(0, np.max(sorted_x) - target_sum / n)
        # This is a bit tricky, use a more direct approach
        result = np.maximum(0, np.minimum(1, x - lambda_val))
        diff = target_sum - np.sum(result)
        if abs(diff) > 1e-10:
            if diff > 0:
                # Need to increase values slightly
                idx = np.argsort(result)[:int(diff / dx)]
                result[idx] += diff / len(idx)
            else:
                # Need to decrease values slightly
                idx = np.argsort(result)[-int(-diff / dx):]
                result[idx] += diff / len(idx)
            # Ensure within bounds
            result = np.clip(result, 0, 1)
    
    return result

def build_triangular_seed(n):
    """Build symmetric triangular seed"""
    x = np.linspace(0, 2, n, endpoint=False) + dx/2
    h = np.maximum(0, 1 - np.abs(x - 1))
    return h

def compute_gradient(h, T=0.05, K=12):
    """Compute gradient of smooth surrogate using top-K shifts"""
    # Compute all correlations with FFT as before
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Get top-K shifts by absolute value of correlation
    corr_abs = np.abs(corr)
    top_k_idx = np.argsort(corr_abs)[-K:][::-1] if K < len(corr_abs) else np.arange(len(corr_abs))
    
    # Compute weights for top shifts using log-sum-exp
    scaled_corr = corr[top_k_idx] / T
    max_corr = np.max(scaled_corr)
    exp_terms = np.exp(scaled_corr - max_corr)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient using top-K shifts
    grad = np.zeros_like(h)
    for i, s in enumerate(top_k_idx):
        # Gradient of correlation s = sum_{i} h_i * (1-h_{i+s}) 
        # This breaks into two parts:
        # d/dh_i = 1-h_{i+s} if 0 <= i <= N-1-s  
        # d/dh_j = -h_{j-s} if s <= j <= N-1

        # Part 1: 1-h_{i+s}
        if s >= 0:
            # When i+s < N, we differentiate h_i: 
            for j in range(len(h) - s):
                grad[j] += weights[i] * (1 - h[j+s]) * dx
        else:
            # When s < 0, we differentiate h_{i-s} with respect to h_i
            # where j = i-s, i=j+s
            for j in range(len(h) + s):
                grad[j+s] += weights[i] * (-h[j]) * dx
                
        # Part 2: -h_{j-s} 
        if s <= 0:
            # When j-s >= 0, we differentiate h_{j-s} with respect to h_i:
            # this happens when j=i+s, so we need: d/dh_i = -h_{i} for i+s >= 0 and i+s < N
            for j in range(-s, len(h)):
                grad[j+(s)] -= weights[i] * h[j] * dx
        else:
            # When s > 0, we need i-s >= 0, so threshold is s
            for j in range(len(h)):
                if j-s >= 0:
                    grad[j] -= weights[i] * h[j-s] * dx
    
    return grad

def lbfgs_b_optimize(initial_h, max_iter=5000, max_time=500):
    """L-BFGS-B optimization with continuation and projection"""
    start_time = time.time()
    
    h_current = initial_h.copy()
    T = 0.12  # Initial temperature
    tau_T = 0.998  # Cooling factor
    T_min = 5e-6  # Minimum temperature  
    tolerance = 1e-6
    prev_C5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        # Compute current objective and gradient  
        C5_current = evaluate_c5_smooth(h_current, T)
        grad = compute_gradient(h_current, T)
        
        # Check for convergence
        if abs(C5_current - prev_C5) < tolerance or T < T_min:
            break
        
        prev_C5 = C5_current
        
        # PyTorch-style L-BFGS but implemented manually with projections
        # Here we'll use a simple line search approach
        # Direction obtained from L-BFGS approximate Hessian (we're implementing simplified version)
        step_size = 0.01
        h_new = h_current - step_size * grad
        
        # Project onto feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Update
        h_current = h_new
        
        # Update temperature
        T *= tau_T
    
    return h_current

def pairwise_exchange(h_initial, max_iter=1000, max_time=50):
    """Pairwise exchange with exact line search"""
    start_time = time.time()
    h = h_initial.copy()
    prev_c5 = evaluate_c5_exact(h)
    
    # Cache the full correlation for faster access
    corr_cache = None
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
            
        improved = False
        
        # Shuffle indices
        indices = list(range(len(h)))
        np.random.shuffle(indices)
        
        for i in indices:
            if time.time() - start_time > max_time:
                break
                
            # For each i, try exchanging with j in a reasonable range
            for j in range(max(len(h) - 5, 0), len(h)):
                if i == j:
                    continue
                    
                # Compute the effect of moving mass from i to j
                # vary the amount of mass to move
                delta_values = np.linspace(-0.1, 0.1, 20)
                best_delta = 0
                best_c5 = prev_c5
                
                # Check if this exchange would be feasible
                if h[i] < 0:
                    continue
                if h[j] > 1:
                    continue
                    
                # Compute the effects on correlation
                temp_h = h.copy()
                
                # Load correlation if needed
                if corr_cache is None or iteration % 20 == 0:
                    corr_cache = []
                    h_rev = h[::-1]
                    b = 1 - h
                    L = 2 * len(h) - 1
                    all_corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
                    corr_cache = all_corr
                
                # Extract correlation terms for this specific exchange
                for delta in delta_values:
                    # Validate the move
                    if h[i] - delta < 0 or h[j] + delta > 1:
                        continue
                    temp_h[i] = h[i] - delta
                    temp_h[j] = h[j] + delta
                    temp_corr = corr_cache
                    
                    # The new c5 
                    c5_new = dx * np.max(temp_corr)
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_delta = delta
                        
                # If improvement, apply it
                if best_delta != 0 and best_c5 < prev_c5:
                    h[i] -= best_delta
                    h[j] += best_delta
                    prev_c5 = best_c5
                    improved = True
                    break
            
            if improved:
                break
                
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    
    # Initialize with triangular seed
    h0 = build_triangular_seed(n_points)
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    best_c5 = evaluate_c5_exact(h0)
    best_h = h0.copy()
    
    # Coarse optimization first (N=60)
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    
    # Build a coarse triangular seed
    h_coarse = build_triangular_seed(n_coarse)
    h_coarse = project_to_sum_box(h_coarse, n_coarse / 2)
    
    # Run coarse optimization
    h_coarse_opt = lbfgs_b_optimize(h_coarse, max_iter=1000, max_time=50)
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points, endpoint=False) + dx/2,
                       np.linspace(0, 2, n_coarse, endpoint=False) + dx_coarse/2,
                       h_coarse_opt)
    h_fine = project_to_sum_box(h_fine, TARGET_SUM)
    
    # Restart loop with various starting points
    restarts = 10
    for restart in range(restarts):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Try different initializations
        if restart == 0:
            h_start = h_fine.copy()
        elif restart % 3 == 0:
            # Random initialization
            h_start = np.random.rand(n_points)
            h_start = project_to_sum_box(h_start, TARGET_SUM)
        elif restart % 3 == 1:
            # Symmetric random 
            h_start = np.random.rand(n_points)
            h_start = (h_start + h_start[::-1]) / 2
            h_start = project_to_sum_box(h_start, TARGET_SUM)
        else:
            # Symmetric triangular
            h_start = build_triangular_seed(n_points)
            h_start = (h_start + h_start[::-1]) / 2
            h_start = project_to_sum_box(h_start, TARGET_SUM)
        
        # Optimize with L-BFGS
        h_opt = lbfgs_b_optimize(h_start, max_iter=2000, max_time=0.5 * budget_s / restarts)
        
        # Evaluate exact C5
        c5_eval = evaluate_c5_exact(h_opt)
        
        if c5_eval < best_c5:
            best_c5 = c5_eval
            best_h = h_opt.copy()
    
    # Final polishing with pairwise exchange
    if time.time() - start_time < 0.95 * budget_s:
        h_final = pairwise_exchange(best_h, max_iter=500, max_time=0.1 * budget_s)
        c5_final = evaluate_c5_exact(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    return (best_h.tolist(), float(best_c5), n_points)
