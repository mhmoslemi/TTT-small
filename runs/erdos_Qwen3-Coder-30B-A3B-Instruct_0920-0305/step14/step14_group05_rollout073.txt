```python
import numpy as np
from scipy.fft import fft, ifft
import time
import random

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def C5_value(h):
    """Evaluate C5 with proper dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old grid to new grid"""
    old_x = np.linspace(0, 2, len(old_h), endpoint=False) + dx(len(old_h))/2
    new_x = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n)/2
    interpolated = np.interp(new_x, old_x, old_h)
    return interpolated

def worst_shift(c):
    """Return the shift index that achieves maximum correlation"""
    idx = np.argmax(c)
    n = len(c)
    s = idx - (n - 1)
    return s

def shift_arrays(h, s):
    """Return h_plus[i] = h[i+s] and h_minus[i] = h[i-s] with zero padding"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:-s]
        h_minus[:-s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Exact sub-gradient of the worst shift term"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution schedule
    stages = [
        (80, 0.30),
        (160, 0.35), 
        (240, 0.35)
    ]
    
    best_h = None
    best_C = float('inf')
    
    for stage_n, time_fraction in stages:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        stage_budget = budget_s * time_fraction
        stage_start = time.time()
        
        # Initialisation
        if stage_n == 80:
            h = np.random.uniform(0, 1, stage_n)
            h = project_to_sum_box(h, target_sum(stage_n))
        else:
            h = interpolate(best_h, stage_n)
            h = project_to_sum_box(h, target_sum(stage_n))
        
        best_h_stage = h.copy()
        best_C_stage = C5_value(h)
        
        # Simulated Annealing
        T = 0.02
        delta = 0.1
        alpha = 0.9997
        beta = 0.9999
        max_iter = int(stage_budget * 10000) if stage_n == 80 else int(stage_budget * 20000)
        
        accepted_moves = 0
        for iteration in range(max_iter):
            if time.time() - stage_start > stage_budget:
                break
                
            # Decide type of move
            if random.random() < 0.3:
                # Targeted move - focus on worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                
                # Compute desirability for each index
                h_plus, h_minus = shift_arrays(h, s)
                desirability = h_plus + h_minus
                
                # Find indices with low/high desirability
                sorted_indices = np.argsort(desirability)
                low_indices = sorted_indices[:max(1, len(sorted_indices)//5)]
                high_indices = sorted_indices[-max(1, len(sorted_indices)//5):]
                
                i = random.choice(low_indices)
                j = random.choice(high_indices)
            else:
                # Random move
                i, j = random.sample(range(stage_n), 2)
            
            # Perform move
            move_amount = min(delta, h[i], 1 - h[j])
            if move_amount > 1e-10:
                h_new = h.copy()
                h_new[i] -= move_amount
                h_new[j] += move_amount
                h_new = np.clip(h_new, 0, 1)
                
                # Apply box sum constraint
                h_new = project_to_sum_box(h_new, target_sum(stage_n))
                
                # Evaluate
                C_new = C5_value(h_new)
                C_old = C5_value(h)
                
                # Accept or reject
                if C_new <= C_old:
                    h = h_new
                    if C_new < best_C_stage:
                        best_h_stage = h_new.copy()
                        best_C_stage = C_new
                    accepted_moves += 1
                else:
                    if random.random() < np.exp(-(C_new - C_old) / T):
                        h = h_new
                        if C_new < best_C_stage:
                            best_h_stage = h_new.copy()
                            best_C_stage = C_new
                        accepted_moves += 1
                
                # Update temperature and delta
                if accepted_moves % 10 == 0:
                    T *= alpha
                    delta *= beta
        
        # Polishing with exact sub-gradient descent
        polishing_budget = stage_budget * 0.2
        polishing_start = time.time()
        N_polish = 2000
        
        for k in range(N_polish):
            if time.time() - polishing_start > polishing_budget:
                break
                
            c = corr_fft(h)
            max_corr = np.max(c)
            eps = 1e-12
            # Find all shifts that achieve near-maximum
            s_indices = np.where(c >= max_corr - eps)[0]
            
            if len(s_indices) == 0:
                break
                
            # Compute average gradient from worst shifts
            avg_grad = np.zeros(len(h))
            for idx in s_indices:
                s = idx - (len(c) - 1)
                g = gradient_worst(h, s)
                avg_grad += g
                
            avg_grad /= len(s_indices)
            
            # Step size
            eta = 0.5 / np.sqrt(k + 1)
            
            # Update
            h_new = h - eta * avg_grad
            h_new = project_to_sum_box(h_new, target_sum(stage_n))
            
            # Evaluate and accept improvement
            C_new = C5_value(h_new)
            if C_new < best_C_stage:
                best_h_stage = h_new.copy()
                best_C_stage = C_new
                h = h_new.copy()
            else:
                h = h_new.copy()
        
        # Update global best
        if best_C_stage < best_C:
            best_C = best_C_stage
            best_h = best_h_stage.copy()
    
    return (best_h.tolist(), float(best_C), len(best_h))
```
