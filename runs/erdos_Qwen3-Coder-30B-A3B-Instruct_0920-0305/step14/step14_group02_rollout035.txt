def evaluate_c5_exact(h):
    """Evaluate C5 using full FFT correlation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.05):
    """Evaluate smooth surrogate for C5 using log-sum-exp"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Use log-sum-exp with temperature T
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    log_sum_exp = max_corr + T * np.log(np.sum(exp_terms))
    return dx * log_sum_exp

def evaluate_grad(h, T=0.05):
    """Evaluate gradient of smooth surrogate"""
    # Compute correlations for all shifts
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_fft = np.fft.fft(h_rev, L) * np.fft.fft(b, L)
    corr = np.fft.ifft(corr_fft).real
    
    # Compute weights
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient for each component
    grad = np.zeros_like(h)
    for s in range(len(corr)):
        if s < len(h):
            # Gradient for positive shift
            if s < len(h):
                grad[:len(h)-s] += weights[s] * (1 - h[s:])
            if s > 0:
                grad[s:] += weights[s] * (-h[:len(h)-s])

                grad[s_neg:] += weights[s] * (1 - h[:len(h)-s_neg])

                grad[:s_neg] += weights[s] * (-h[s_neg:])
    
    return dx * grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling method for projection onto capped simplex
    # Sort values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find the threshold lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find the largest k such that (cumsum[k] - k * lambda) <= target_sum
    # where lambda = (cumsum[k] - target_sum) / (k + 1) for index k
    thresholds = (cumsum - target_sum) / np.arange(1, n + 1)
    
    # Find the largest index where threshold <= sorted_x[index]
    valid_indices = np.where(sorted_x >= thresholds)[0]
    if len(valid_indices) == 0:
        lambda_val = np.max((cumsum[-1] - target_sum) / n, 0)
        result = np.clip(x - lambda_val, 0, 1)
    else:
        k = valid_indices[-1]
        lambda_val = thresholds[k]
        result = np.clip(sorted_x - lambda_val, 0, 1)
        # Restore original order
        result[sorted_indices] = result
    
    return result
