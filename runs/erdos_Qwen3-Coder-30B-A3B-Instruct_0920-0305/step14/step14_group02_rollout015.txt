# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_exact(h):
    """Evaluate C5 using FFT - exact computation"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def evaluate_c5_smooth(h, T=0.05):
    """Evaluate smooth surrogate using log-sum-exp"""
    # Compute all correlations
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real[:len(h)]
    
    # Use log-sum-exp with temperature T
    corr = corr * dx
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    return max_corr + T * np.log(np.sum(exp_terms))

def compute_gradient(h, T=0.05):
    """Compute gradient of smooth surrogate"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_fft = np.fft.fft(h_rev, L) * np.fft.fft(b, L)
    corr = np.fft.ifft(corr_fft).real[:len(h)] * dx
    
    # Compute weights
    max_corr = np.max(corr)
    exp_terms = np.exp((corr - max_corr) / T)
    weights = exp_terms / np.sum(exp_terms)
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Gradient of c_s = sum h_i (1-h_{i+s}) with respect to h_j
        for s in range(-len(h)+1, len(h)):
            if 0 <= i < len(h) - abs(s):
                if s >= 0:
                    if i + s < len(h):
                        if i == i + s:
                            grad[i] += weights[i] * (1 - h[i + s]) - weights[i] * h[i]
                        else:
                            grad[i] += weights[i] * (1 - h[i + s])
                else:
                    if i - s >= 0:
                        grad[i] -= weights[i] * h[i - s]
    return grad

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort values in descending order to do water-filling efficiently
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Compute cumulative sums
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Water-filling
    threshold = 0.0
    total_excess = 0.0
    
    for i in range(n):
        # Amount we can raise the water level to cover this index
        if i == 0:
            excess = target_sum - cumsum[i]
        else:
            excess = target_sum - cumsum[i] - (i * (sorted_x[i] - sorted_x[i-1]))
        
        if excess <= 0:
            # We've found the right level
            threshold = sorted_x[i] + (target_sum - cumsum[i]) / (i + 1)
            break
        else:
            # We'll raise the water level to this point
            threshold = sorted_x[i]
            total_excess = excess
    
    # Apply threshold to all values
    result = np.zeros_like(x)
    for i in range(n):
        if sorted_x[i] > threshold:
            result[sorted_indices[i]] = threshold
        else:
            result[sorted_indices[i]] = sorted_x[i]
    
    # Adjust to the exact sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-10:
        # Distribute the difference to the smallest elements
        sorted_indices_rev = np.argsort(result)
        for i in range(n):
            if diff > 1e-10:
                to_add = min(diff, 1 - result[sorted_indices_rev[i]])
                result[sorted_indices_rev[i]] += to_add
                diff -= to_add
            else:
                break
    
    return np.clip(result, 0, 1)

def lbfgs_with_continuation(h0, max_iter=2000, T0=0.12, decay_factor=0.998, min_T=5e-6):
    """Projected L-BFGS with temperature continuation to minimize smooth surrogate"""
    h = h0.copy()
    T = T0
    
    # Initialize history
    Hk = np.eye(len(h))  # Initial inverse Hessian
    
    for iteration in range(max_iter):
        # Compute gradient
        grad = compute_gradient(h, T)
        
        # Check if we should stop
        if T < min_T:
            break
            
        # Update with projected L-BFGS step
        # This is an approximation - for efficiency we use a simple step
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Update temperature
        T *= decay_factor
        
        h = h_new.copy()
        
    return h

    h_best = h.copy()
    c5_best = evaluate_c5_exact(h)
    
    for iteration in range(max_iters):
        improved = False
        
        # Randomize sweep order
        indices = list(range(len(h)))
        np.random.shuffle(indices)
        
        for i in indices:
            # Find best exchange partner j
            best_delta = None
            best_c5 = c5_best
            
            for j in range(len(h)):
                if i == j:
                    continue
                    
                # Compute valid range for delta
                delta_min = max(-h[i], h[j] - 1)
                delta_max = min(h[i], 1 - h[j])
                
                if delta_min >= delta_max:
                    continue
                    
                # Sample a few points in range for fast estimation of optimal delta
                candidates = [delta_min, delta_max]
                if delta_max - delta_min > 0.05:
                    candidates += np.linspace(delta_min, delta_max, 5).tolist()
                
                for delta in candidates:
                    if delta < delta_min or delta > delta_max:
                        continue
                        
                    # Apply exchange
                    h_trial = h.copy()
                    h_trial[i] -= delta
                    h_trial[j] += delta
                    h_trial = project_to_sum_box(h_trial, TARGET_SUM)
                    
                    c5_trial = evaluate_c5_exact(h_trial)

                        best_c5 = c5_trial
                        best_delta = delta
                        
            if best_delta is not None:
                # Apply the best exchange
                h[i] -= best_delta
                h[j] += best_delta
                h = project_to_sum_box(h, TARGET_SUM)
                c5_best = best_c5
                improved = True
        
        if not improved:
            break
            
    return h
