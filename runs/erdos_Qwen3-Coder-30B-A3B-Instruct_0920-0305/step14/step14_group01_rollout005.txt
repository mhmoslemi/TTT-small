import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import differential_evolution
import random

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    # Try different optimization strategies
    # Strategy 1: Differential Evolution with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        x_projected = project_to_sum_box(x, TARGET_SUM)
        return evaluate_c5(x_projected)

        # Use differential evolution with multiple restarts
        for restart in range(5):
            if time.time() - start_time > 0.8 * budget_s:

            np.random.seed(seed + restart)
            result = differential_evolution(
                objective, bounds, seed=seed + restart,

            )
            if not result.success:

            h_candidate = project_to_sum_box(result.x, TARGET_SUM)
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    except:

    # Strategy 2: Simulated Annealing with adaptive cooling
    if best_h is not None:
        h = best_h.copy()
        max_iter = 10000
        temperature = 1.0
        min_temperature = 1e-8
        cooling_rate = 0.9998
        
        t2 = start_time + 0.9 * budget_s
        for iter_num in range(max_iter):
            if time.time() >= t2:
                break
                
            # Try a neighbor solution
            h_neighbor = h.copy()

            c5_current = evaluate_c5(h)

            if c5_neighbor < c5_current or np.random.rand() < np.exp((c5_current - c5_neighbor) / temperature):

    return (best_h.tolist(), float(best_c5), n_points)
