import numpy as np
import time

def project_onto_box_sum(v, target, lower=0.0, upper=1.0):
    # Project vector v onto box constraints [lower, upper] and sum = target
    # Using bisection on lambda parameter
    low = -1e5
    high = 1e5
    for _ in range(60):
        lam = (low + high) / 2
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high) / 2, lower, upper)
    # Fine-tune sum
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Distribute diff among non-bound elements
        free = np.where((x > lower + 1e-12) & (x < upper - 1e-12))[0]
        if len(free) > 0:
            x[free] += diff / len(free)
            x = np.clip(x, lower, upper)
    return x

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 180
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from interpolated initial_h_values
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values)
        x_old = np.linspace(0, 2, len(init), endpoint=False)
        x_new = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        h = project_onto_box_sum(h, target_sum, 0.0, 1.0)
    else:
        # Fallback to uniform initialization
        h = np.full(n_points, 0.5)
        h = project_onto_box_sum(h, target_sum, 0.0, 1.0)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Gradient descent parameters
    alpha0 = 0.5
    max_iterations = 20000
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.5 and iteration < max_iterations:
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        
        # Find all indices that achieve maximum correlation
        max_indices = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        # Compute subgradient using all worst-case shifts
        grad = np.zeros_like(h)
        for idx in max_indices:
            shift = idx - (n_points - 1)
            if shift >= 0:
                # Positive shift case
                s = shift
                i = np.arange(n_points - s)
                grad[i] += 1 - h[i + s]
                grad[i + s] -= h[i]
            else:
                # Negative shift case
                s = -shift
                i = np.arange(n_points - s)
                grad[i + s] += 1 - h[i]
                grad[i] -= h[i + s]
        
        # Average gradients if multiple shifts contribute
        grad /= len(max_indices)
        
        # Step size schedule
        alpha = alpha0 / np.sqrt(iteration + 1)
        
        # Take gradient step
        h_candidate = h - alpha * grad
        
        # Project onto feasible set
        h_candidate = project_onto_box_sum(h_candidate, target_sum, 0.0, 1.0)
        
        # Compute new C5
        cand_c5 = compute_c5(h_candidate, dx)
        
        # Accept improvement
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate.copy()
            h = h_candidate
        elif cand_c5 <= max_corr * dx:
            # Accept if not much worse
            h = h_candidate
            
        # Occasional random perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            # Add small random noise and reproject
            noise = np.random.normal(scale=0.01, size=n_points)
            h_pert = h + noise
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_onto_box_sum(h_pert, target_sum, 0.0, 1.0)
            pert_c5 = compute_c5(h_pert, dx)
            if pert_c5 < best_c5:
                best_c5 = pert_c5
                best_h = h_pert.copy()
                h = h_pert
        
        iteration += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
