import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Create initial shape: cosine squared function
    x = np.linspace(0, 2, n_points)
    h0 = (1 + np.cos(np.pi * (x - 1))) / 2
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Subgradient descent with adaptive step size
    max_iter = 20000
    step_size = 0.1
    decay_rate = 0.999
    
    for iteration in range(max_iter):
        if time.time() >= start_time + 0.9 * budget_s:
            break
            
        # Compute correlation
        L = 2 * n_points - 1
        h_rev = current_h[::-1]
        b = 1 - current_h
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        
        # Find the worst shift
        idx_max = np.argmax(corr)
        shift = idx_max - (n_points - 1)
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(current_h)
        if shift >= 0:
            # i+shift index condition
            grad[:n_points-shift] += 1 - current_h[shift:]
            grad[shift:] -= current_h[:n_points-shift]
        else:
            s = -shift
            # shift negative
            grad[s:] += 1 - current_h[:n_points-s]
            grad[:n_points-s] -= current_h[s:]
        
        # Update with step size and project
        step_size *= decay_rate
        current_h = current_h - step_size * grad
        current_h = np.clip(current_h, 0, 1)
        current_h = project_to_sum_box(current_h, TARGET_SUM)
        
        # Evaluate and track best
        current_c5 = evaluate_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    # Final improvement with local search
    if best_h is not None:
        h = best_h.copy()
        for iteration in range(5000):
            if time.time() >= start_time + budget_s:
                break
                
            # Try a neighbor solution by swapping small amounts
            i, j = np.random.randint(0, n_points, 2)
            delta = np.random.uniform(-0.02, 0.02)
            
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            h_new[j] = np.clip(h_new[j] - delta, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
