import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    n_particles = 30
    particles = [project_to_sum_box(np.random.rand(n_points), TARGET_SUM)

    # Add the best starting solution
    particles[0] = h0.copy()
    
    # Initialize velocities (small random values)
    velocities = [np.random.normal(0, 0.01, n_points) for _ in range(n_particles)]
    
    # Best personal positions
    personal_best_positions = [p.copy() for p in particles]
    personal_best_c5 = [evaluate_c5(p) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    max_iter = 2000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.8:
        iter_count += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and projection
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_to_sum_box(particles[i], TARGET_SUM)

            current_c5 = evaluate_c5(particles[i])

            if current_c5 < personal_best_c5[i]:

                personal_best_positions[i] = particles[i].copy()

                    h_down = project_to_sum_box(h_down, TARGET_SUM)
                    candidates.append((h_down, evaluate_c5(h_down)))
                    
                    # Up direction  
                    h_up = h.copy()
                    h_up[idx] = min(h[idx] + step_size, 1.0)
                    h_up = project_to_sum_box(h_up, TARGET_SUM)
                    candidates.append((h_up, evaluate_c5(h_up)))
                    
                    # Find best candidate
                    best_candidate = min(candidates, key=lambda x: x[1])
                    if best_candidate[1] < evaluate_c5(h):
                        h = best_candidate[0]
                        if best_candidate[1] < best_c5:
                            best_c5 = best_candidate[1]
                            best_h = h.copy()

                        break

            if not any_improved:
                idx = np.random.randint(n_points)
                h_new = h.copy()
                h_new[idx] = np.clip(h_new[idx] + np.random.normal(0, 0.01), 0, 1)
                h_new = project_to_sum_box(h_new, TARGET_SUM)
                if evaluate_c5(h_new) < evaluate_c5(h):
                    h = h_new
                    if evaluate_c5(h) < best_c5:
                        best_c5 = evaluate_c5(h)
                        best_h = h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
