import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto capped simplex with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L), n=L)
    return corr

def compute_gradient(corr, idx_top, T):
    """Compute gradient using top-K shifts"""
    # Soft-max weights
    corr_top = corr[idx_top]
    max_corr = corr_top.max()
    w = np.exp((corr_top - max_corr) / T)
    w = w / w.sum()
    
    # Gradient computation
    g = np.zeros_like(corr)
    for i, s in enumerate(idx_top):
        if s >= 0:
            g[s:] += w[i] * (1 - np.pad(h[:-s], (s, 0), 'constant', constant_values=0))
            g[:-s] -= w[i] * np.pad(h[s:], (0, s), 'constant', constant_values=0)
        else:
            g[:s] += w[i] * (1 - np.pad(h[-s:], (0, -s), 'constant', constant_values=0))
            g[-s:] -= w[i] * np.pad(h[:s], (-s, 0), 'constant', constant_values=0)
    
    return g
