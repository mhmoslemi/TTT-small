import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_softmax_gradient(h, alpha=8.0):
    """Compute gradient of softmax approximation to max overlap"""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    
    # Softmax weights
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros_like(h)
    
    # For each shift s, add weighted contribution
    for s in range(-len(h)+1, len(h)):
        if s == 0:
            # Contribution from h_i * h_{i+s} for s=0
            grad -= 2 * weights[s + len(h) - 1] * h
        else:
            # Forward and backward shifts
            if s > 0:
                # h_i * h_{i+s} term 
                if s < len(h):
                    grad[:len(h)-s] -= weights[s + len(h) - 1] * h[s:]
                    grad[s:] -= weights[s + len(h) - 1] * h[:len(h)-s]
            else:  # s < 0
                # h_i * h_{i+s} term
                s_abs = -s
                if s_abs < len(h):
                    grad[s_abs:] -= weights[s + len(h) - 1] * h[:-s_abs]
                    grad[:len(h)-s_abs] -= weights[s + len(h) - 1] * h[s_abs:]
    
    return grad

            eps = 1e-8

            alpha = 8.0

            max_iter = 3000
            patience_counter = 0
            best_iter_c5 = current_c5

                grad = compute_softmax_gradient(current_h, alpha)
                
                # Adam update
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad**2
                
                # Bias correction
                m_hat = m / (1 - beta1**(iteration + 1))
                v_hat = v / (1 - beta2**(iteration + 1))
                
                # Proposed update
                step = eta * m_hat / (np.sqrt(v_hat) + eps)

                # Project
                projected_h = project_to_sum_box(proposed_h, TARGET_SUM)

                # Transfer small amount
                delta = 1e-4
                if current_h[idx1] >= delta:
                    current_h[idx1] -= delta
                    current_h[idx2] += delta
                    # Project
                    current_h = project_to_sum_box(current_h, TARGET_SUM)
            
            # Evaluate with reference function
            corr = np.correlate(current_h, 1-current_h, mode='full')
