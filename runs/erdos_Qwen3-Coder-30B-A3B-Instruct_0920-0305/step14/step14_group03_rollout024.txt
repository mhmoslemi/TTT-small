def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_gradient_exact(h):
    """Compute exact gradient"""
    # Compute correlation
    corr = compute_corr(h)
    max_idx = np.argmax(corr)
    
    # Gradient computation
    grad = np.zeros_like(h)
    
    # For the maximum correlation index, we need to compute partial derivatives
    # The derivative of the maximum of a set of functions is the derivative of the maximum function
    # In our case, if corr[i] is maximum, then d/dh_j of the maximum is equivalent to 
    # the derivative of the inner product satisfying the equality condition
    
    # This constraint propagates to all points. 
    # We treat it like computing subgradient of the max function  
    
    # Extract non-zero correlations at maximum point
    s = max_idx - len(h) + 1  # The shift value that gives max
    
    # For shift s, we have correlation contribution from h[i]*h[i+s] or h[i]*h[i-s] etc
    # So the partial derivative wrt h[i] is (1-h[i+s]) for i in range(len(h)) where i+s is valid
    # and (1-h[i-s]) for i in range(len(h)) where i-s is valid
    
    # So we can compute both types of derivative
    for i in range(len(h)):
        # derivative from h[i+s] (if valid)
        if 0 <= i + s < len(h):
            grad[i] += (1 - h[i+s])
        # derivative from h[i-s] (if valid)
        if 0 <= i - s < len(h):
            grad[i] += (1 - h[i-s])
                
    # Due to linearity we can subtract twice the gradient wrt sum constraint
    # to keep norm fixed (this is the projection gradient variant)
    # But actually since gradient is involved with correlation, and it's not that simple,
    # we adopt a fixed simple approach for now
    return grad
