import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def subgradient_step(h, step_size, target_sum):
    """Perform one subgradient step"""
    N = len(h)
    L = 2 * N - 1
    
    # Compute correlation
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Find the index that maximizes correlation
    idx_max = np.argmax(corr)
    shift = idx_max - (N - 1)
    
    # Compute gradient for the worst case shift
    g = np.zeros_like(h)
    if shift >= 0:
        # i+shift index condition
        g[:N-shift] += 1 - h[shift:]
        g[shift:] -= h[:N-shift]
    else:
        s = -shift
        # shift negative
        g[s:] += 1 - h[:N-s]
        g[:N-s] -= h[s:]
    
    # Update h
    h_new = h - step_size * g
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_sum_box(h_new, target_sum)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use cosine-based initial solution
    x = np.linspace(0, 2, n_points)
    h0 = (1 + np.cos(np.pi * (x - 1))) / 2
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Subgradient descent with decreasing step size
    max_iter = 20000
    init_step = 0.1
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        step = init_step / np.sqrt(i + 1)
        current_h = subgradient_step(current_h, step, TARGET_SUM)
        current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Local refinement with random swaps
    if best_h is not None:
        h = best_h.copy()
        max_iter = 5000
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Try random swap
            i1, i2 = np.random.randint(0, n_points, 2)
            delta = np.random.uniform(-0.02, 0.02)
            
            h_new = h.copy()
            h_new[i1] = np.clip(h_new[i1] + delta, 0, 1)
            h_new[i2] = np.clip(h_new[i2] - delta, 0, 1)
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
