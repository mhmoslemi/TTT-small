```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
T0 = 0.12
TAU_T = 0.998
T_MIN = 5e-6
MAX_ITER_LBFGS = 2000

def fast_correlation(h):
    """Fast evaluation of all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = fast_correlation(h)
    return np.max(corr)

def smooth_max(h, temp=0.05, top_k=12):
    """Compute smooth approximation to max of correlations"""
    corr = fast_correlation(h)
    # Keep only top K shifts for computational efficiency
    top_shifts = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[top_shifts]
    
    # Log-sum-exp with temperature
    exp_corr = np.exp(corr_top / temp)
    log_sum_exp = temp * np.log(np.sum(exp_corr))
    return log_sum_exp

def gradient_smooth_max(h, temp=0.05, top_k=12):
    """Gradient of smooth max approximation"""
    corr = fast_correlation(h)
    # Keep only top K shifts for computational efficiency
    top_shifts = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[top_shifts]
    
    # Compute weights
    exp_corr = np.exp(corr_top / temp)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient for top shifts only
    grad = np.zeros_like(h)
    for i, s in enumerate(top_shifts):
        # Compute gradient for shift s
        # d/dh_j c_s = 1 - h_{j+s} if j <= N-1-s, -h_{j-s} if j >= s
        for j in range(len(h)):
            if j <= len(h) - 1 - s and j >= 0:
                grad[j] += weights[i]
            if j >= s and j < len(h):
                grad[j] -= weights[i]
    
    return grad * dx

def project_to_simplex(x, target_sum):
    """Project vector x onto capped simplex with sum = target_sum"""
    # Water-filling algorithm
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    n = len(x)
    
    # Find index where threshold should be placed
    idx = -1
    for i in range(n):
        threshold = (cumsum[i] - target_sum) / (i + 1)
        if threshold <= sorted_x[i]:
            idx = i
        else:
            break
    
    if idx == -1:
        # All components are equal
        lambda_val = (np.sum(x) - target_sum) / n
        result = np.maximum(0, np.minimum(1, x - lambda_val))
    else:
        lambda_val = (cumsum[idx] - target_sum) / (idx + 1)
        result = np.maximum(0, np.minimum(1, x - lambda_val))
    
    return result

def lbfgs_with_continuation(seed_h, budget_s):
    """Use L-BFGS with temperature continuation to minimize smooth surrogate"""
    start_time = time.time()
    
    h = seed_h.copy()
    h = project_to_simplex(h, TARGET_SUM)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    temp = T0
    iter_count = 0
    
    while temp > T_MIN and iter_count < MAX_ITER_LBFGS and time.time() < start_time + budget_s * 0.7:
        # Compute smooth objective and its gradient
        def obj(h_vec):
            return smooth_max(h_vec, temp)
        
        def grad(h_vec):
            return gradient_smooth_max(h_vec, temp)
        
        # Run L-BFGS step
        res = minimize(
            obj, h, method='L-BFGS-B', 
            jac=grad,
            bounds=[(0, 1)] * len(h),
            options={'maxiter': 20, 'gtol': 1e-8}
        )
        
        h = res.x
        h = project_to_simplex(h, TARGET_SUM)
        
        # Check if we improved
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Continue with lower temperature
        temp *= TAU_T
        iter_count += 1
    
    return best_h, best_c5

def pairwise_exchange(h, max_iter=5000):
    """Exact pairwise exchange with analytical line search"""
    start_time = time.time()
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    improved = True
    
    iter_count = 0
    while improved and iter_count < max_iter and time.time() < start_time + 100:
        improved = False
        # Shuffle indices for randomness
        indices = list(range(len(current_h)))
        np.random.shuffle(indices)
        
        for i in indices:
            # Try exchanging with neighbors
            for j in range(max(0, i-10), min(len(current_h), i+11)):
                if i == j:
                    continue
                    
                # Compute the effect of moving small amount from i to j
                h_i_old, h_j_old = current_h[i], current_h[j]
                delta = min(h_i_old, 1 - h_j_old, h_j_old, 1 - h_i_old)
                if delta < 1e-8:
                    continue
                    
                # Perform a small step and compute the impact on all correlations
                step_sizes = np.linspace(-delta*0.1, delta*0.1, 21)
                best_step = 0
                best_corr = current_c5
                
                corr_base = fast_correlation(current_h)
                
                for s in step_sizes:
                    h_test = current_h.copy()
                    h_test[i] = h_i_old - s
                    h_test[j] = h_j_old + s
                    h_test = project_to_simplex(h_test, TARGET_SUM)
                    corr_test = fast_correlation(h_test)
                    test_c5 = np.max(corr_test)
                    
                    if test_c5 < best_corr:
                        best_corr = test_c5
                        best_step = s
                
                # Apply the best step
                if best_step != 0:
                    current_h[i] -= best_step
                    current_h[j] += best_step
                    current_h = project_to_simplex(current_h, TARGET_SUM)
                    current_c5 = best_corr
                    improved = True
                    break
            
            if improved:
                break
            
            iter_count += 1
    
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Initial triangle function
    x = np.linspace(0, 2, n_points)
    initial_h = np.maximum(0, 1 - np.abs(x - 1))
    initial_h = project_to_simplex(initial_h, TARGET_SUM)
    
    # Run L-BFGS with continuation from the initial solution
    h1, c5_1 = lbfgs_with_continuation(initial_h, budget_s * 0.6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Try with multiple restarts using different seeds
    for restart in range(10):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Random restart with improved projection
        h_start = np.random.rand(n_points)
        h_start = project_to_simplex(h_start, TARGET_SUM)
        h_restarted, c5_restarted = lbfgs_with_continuation(h_start, budget_s * 0.1)
        
        if c5_restarted < best_c5:
            best_c5 = c5_restarted
            best_h = h_restarted.copy()
    
    # Local polishing with pairwise exchange
    if best_h is not None:
        final_h, final_c5 = pairwise_exchange(best_h, 5000)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    # Final check
    if best_h is None:
        best_h = initial_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
```
