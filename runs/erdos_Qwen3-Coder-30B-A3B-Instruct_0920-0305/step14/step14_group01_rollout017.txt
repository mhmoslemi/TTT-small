import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        best_c5 = current_c5
        best_h = current_h.copy()

    # Use a hybrid approach combining particle swarm optimization with local refinement
    # Initialize particles
    n_particles = 30
    particles = []
    velocities = []
    
    # Create initial swarm around the best known solution
    for i in range(n_particles):
        if i == 0:
            # First particle is the best known solution
            particle = h0.copy()
        else:
            # Other particles are perturbed versions
            particle = project_to_sum_box(
                h0 + np.random.normal(0, 0.05, n_points), 
                TARGET_SUM
            )
        particles.append(particle)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Personal best tracking
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_c5(p) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    if gbest_c5 < best_c5:
        best_c5 = gbest_c5
        best_h = gbest.copy()

    max_iter = 3000
    iter_num = 0
    
    while iter_num < max_iter and time.time() - start_time < budget_s * 0.95:
        iter_num += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply bounds and projection
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_to_sum_box(particles[i], TARGET_SUM)
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i])
            
            # Update personal best
            if fitness < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = fitness
                
                # Update global best
                if fitness < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = fitness

                        best_c5 = gbest_c5

    if best_h is not None:

        learning_rate = 0.05
        max_iter_local = 2000
        
        for i in range(max_iter_local):
            if time.time() - start_time > budget_s * 0.99:
                break
            
            # Compute finite differences for gradient
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] = min(h[j] + eps, 1.0)
                h_plus = project_to_sum_box(h_plus, TARGET_SUM)
                
                h_minus = h.copy()
                h_minus[j] = max(h[j] - eps, 0.0)
                h_minus = project_to_sum_box(h_minus, TARGET_SUM)
                
                c5_plus = evaluate_c5(h_plus)
                c5_minus = evaluate_c5(h_minus)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_new = h - learning_rate * grad
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < evaluate_c5(h):
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                learning_rate *= 0.995  # Decrease learning rate

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
