import numpy as np
from scipy.signal import correlate
import time
import random

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Fast correlation computation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate C5 using FFT correlation"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation from old_h to new_n points"""
    old_positions = np.linspace(0, 2, len(old_h), endpoint=False) + dx(len(old_h))/2
    new_positions = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n)/2
    new_h = np.interp(new_positions, old_positions, old_h)
    return new_h

def worst_shift(corr):
    """Get the shift index that gives maximum correlation"""
    idx = np.argmax(corr)
    s = idx - (len(corr) - 1)
    return s

def shift_arrays(h, s):
    """Get shifted arrays h_plus and h_minus for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient for the worst shift s"""
    h_plus, h_minus = shift_arrays(h, s)
    g = 1 - h_plus - h_minus
    return g

def make_stage(n, time_fraction, start_time, budget_s, prev_best_h=None):
    """Run a single stage of the multi-resolution optimization"""
    # Initialize
    if prev_best_h is None:
        h = np.random.uniform(0, 1, n)
        h = project_to_sum_box(h, target_sum(n))
    else:
        h = interpolate(prev_best_h, n)
        h = project_to_sum_box(h, target_sum(n))
        
    best_h_stage = h.copy()
    best_C_stage = C5_value(h)
    
    # Parameters for this stage
    T = 0.02
    delta = 0.1
    alpha = 0.9997
    beta = 0.9999
    max_iter = int(80000 * time_fraction / 0.35)  # Scale with fraction
    iter = 0
    
    # Annealing loop
    while iter < max_iter and time.time() < start_time + budget_s * time_fraction:
        # Generate move
        if random.random() < 0.3:  # Targeted move
            c = corr_fft(h)
            s = worst_shift(c)
            h_plus, h_minus = shift_arrays(h, s)
            d = h_plus + h_minus
            
            # Find indices in bottom 20% and top 20%
            num_bottom = max(1, int(0.2 * n))
            num_top = max(1, int(0.2 * n))
            
            bottom_indices = np.argpartition(d, num_bottom)[:num_bottom]
            top_indices = np.argpartition(d, -num_top)[-num_top:]
            
            # Pick random ones from these sets
            i = random.choice(bottom_indices)
            j = random.choice(top_indices)
        else:  # Random move
            i, j = random.sample(range(n), 2)
            
        # Compute maximum transfer amount
        max_transfer = min(delta, h[i], 1 - h[j])
        if max_transfer <= 0:
            iter += 1
            continue
            
        # Make move
        delta_transfer = random.uniform(0, max_transfer)
        h[i] -= delta_transfer
        h[j] += delta_transfer
        h = project_to_sum_box(h, target_sum(n))
        
        # Evaluate
        current_C = C5_value(h)
        
        # Accept or reject
        if current_C <= best_C_stage:
            if current_C < best_C_stage:
                best_h_stage = h.copy()
                best_C_stage = current_C
        else:
            # Accept with probability
            if random.random() < np.exp(-(current_C - best_C_stage) / T):
                if current_C < best_C_stage:
                    best_h_stage = h.copy()
                    best_C_stage = current_C
            else:
                # Revert the move (but we already applied it since it's in-place)
                pass
        
        # Update parameters
        T = alpha * T
        delta = beta * delta
        iter += 1
    
    # Polishing with exact gradient descent
    polishing_iter = min(2000, int(10000 * time_fraction / 0.35))
    for k in range(polishing_iter):
        if time.time() >= start_time + budget_s:
            break
            
        c = corr_fft(h)
        max_val = np.max(c)
        epsilon = 1e-12
        worst_indices = np.where(np.abs(c - max_val) < epsilon)[0]
        
        # Compute average gradient
        g = np.zeros(len(h))
        for s_idx in worst_indices:
            s = s_idx - (len(h) - 1)
            g += gradient_worst(h, s)
        g = g / len(worst_indices)
        
        # Step size
        eta = 0.5 / np.sqrt(k + 1)
        
        # Update
        h = h - eta * g
        h = project_to_sum_box(h, target_sum(n))
        
        # Check improvement
        current_C = C5_value(h)
        if current_C < best_C_stage:
            best_C_stage = current_C
            best_h_stage = h.copy()
    
    return best_h_stage, best_C_stage
