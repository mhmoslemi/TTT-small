import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation via FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient vector for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        # Forward contribution: (1 - h_{i+s})
        if s < N:
            g[:N-s] += 1 - h[s:]
        # Backward contribution: - h_{i-s}
        if s > 0:
            g[s:] -= h[:N-s]
    else:
        s = -s
        # Forward contribution: (1 - h_{i+s}) 
        if s < N:
            g[s:] += 1 - h[:N-s]
        # Backward contribution: - h_{i-s}
        if s > 0:
            g[:N-s] -= h[s:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max using top-K shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just use the subgradient of the max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        max_corr = np.max(corr)
        # Shift for numerical stability
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency  
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (len(h)-1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent with continuation"""
    # Initialize
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    # Parameters
    T = T0
    step = step0
    
    for it in range(max_iter):
        # Time check
        if it % 100 == 0:
            # Periodic C5 evaluation
            c5 = evaluate_c5_fast(h) 
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute smooth gradient
        g = smooth_gradient(h, T, top_k)
        # Update step size
        step = step0 / np.sqrt(it + 1)
        # Gradient descent step
        h = h - step * g
        # Enforce symmetry if requested
        if sym:
            h = (h + h[::-1]) / 2
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Decay temperature
        T *= 0.9995
        
        # Avoid very small T
        if T < 1e-5:
            T = 1e-5
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Pair-wise swap hill climber for local refinement"""
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    N = len(h)
    
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range 
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        
        # Propose swap
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate new solution
        new_c5 = evaluate_c5_fast(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            
    return best_h, best_c5
