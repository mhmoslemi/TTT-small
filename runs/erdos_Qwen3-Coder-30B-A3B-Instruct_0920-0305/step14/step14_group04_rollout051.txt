import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_raw_corr(h):
    """Compute raw correlation using FFT (without dx factor)"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_weights(corr_raw, alpha):
    """Compute softmax weights for correlation values"""
    max_c = np.max(corr_raw)
    exps = np.exp(alpha * (corr_raw - max_c))
    return exps / np.sum(exps)

def compute_ascend_weighted(h, weights):
    """Compute weighted ascend direction"""
    n = len(h)
    ascend = np.zeros_like(h)
    
    # Process each shift
    for idx, w_s in enumerate(weights):
        if w_s == 0:
            continue
        shift = idx - (n-1)
        
        if shift == 0:
            ascend += 2 * w_s * h
        elif shift > 0:
            # forward term: h_{i+s}
            if shift < n:
                ascend[:n-shift] += w_s * h[shift:]
            # backward term: h_{i-s}  
            if shift < n:
                ascend[shift:] += w_s * h[:n-shift]
        else:  # shift < 0
            s = -shift
            # forward term: h_{i+s} == h_{i-s} (negative shift)
            if s < n:
                ascend[s:] += w_s * h[:n-s]
            # backward term: h_{i-s}
            if s < n:
                ascend[:n-s] += w_s * h[s:]
    
    return ascend

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Predefine initial constructions
    init_constructions = []
    
    # triangular shape
    x = (np.arange(n_points) + 0.5) * dx
    tri = np.minimum(x, 2 - x)
    init_constructions.append(tri)
    
    # random uniform continuous
    for _ in range(2):
        rand = np.random.rand(n_points)
        rand_proj = project_to_sum_box(rand, TARGET_SUM)
        init_constructions.append(rand_proj)
    
    # random binary
    for _ in range(2):
        idx = np.random.choice(n_points, int(TARGET_SUM), replace=False)
        bin_vec = np.zeros(n_points)
        bin_vec[idx] = 1.0
        init_constructions.append(bin_vec)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            # Resample by interpolation
            old_x = np.linspace(0, 2, len(h0), endpoint=False) + 1.0/len(h0)
            new_x = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
            h_resampled = np.interp(new_x, old_x, h0, left=0, right=0)
            h_resampled = project_to_sum_box(h_resampled, TARGET_SUM)
            init_constructions.append(h_resampled)
        else:
            init_constructions.append(project_to_sum_box(h0, TARGET_SUM))
    
    # Main optimization loop
    max_iter = 5000
    alpha = 1.0
    step_size = 0.3
    patience = 0
    
    for init_h in init_constructions:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        current_h = init_h.copy()
        current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Gradient descent with adaptive parameters
        for iter_num in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute raw correlation
            corr_raw = compute_raw_corr(current_h)
            
            # Compute weights with softmax
            weights = compute_weights(corr_raw, alpha)
            
            # Compute weighted ascend direction
            ascend = compute_ascend_weighted(current_h, weights)
            
            # Normalize ascend to control step size
            ascend_norm = ascend / (np.max(np.abs(ascend)) + 1e-12)
            
            # Update with step size
            h_new = current_h + step_size * ascend_norm
            
            # Project onto constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Apply symmetry enforcement
            h_new = (h_new + h_new[::-1]) / 2
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            c5_new = evaluate_c5(h_new)
            
            # Accept new solution
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = current_h.copy()
                patience = 0
            else:
                patience += 1
                
            # Adaptive step size
            if patience > 10:
                step_size *= 0.9
                patience = 0
                
            # Increase alpha gradually
            if alpha < 100 and iter_num % 20 == 0:
                alpha *= 1.02
                
            # Occasionally refine with additional perturbations
            if iter_num % 100 == 0 and time.time() - start_time < 0.9 * budget_s:
                h_pert = current_h.copy()
                # Perturb randomly
                for _ in range(10):
                    i = np.random.randint(n_points)
                    j = np.random.randint(n_points)
                    delta = np.random.normal(0, 0.01)
                    if abs(delta) > 0.001:
                        h_pert[i] = np.clip(h_pert[i] + delta, 0, 1)
                        h_pert[j] = np.clip(h_pert[j] - delta, 0, 1)
                h_pert = project_to_sum_box(h_pert, TARGET_SUM)
                c5_pert = evaluate_c5(h_pert)
                if c5_pert < current_c5:
                    current_h = h_pert
                    current_c5 = c5_pert
                    if c5_pert < best_c5:
                        best_c5 = c5_pert
                        best_h = current_h.copy()

    if best_h is None:
        # Fallback to initial construction
        best_h = init_constructions[0].copy()
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
