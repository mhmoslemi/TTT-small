def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Simple projection: adjust values to meet sum constraint
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-10:
        return np.clip(x, 0, 1)
    
    # Linear adjustment
    adjustment = (target_sum - current_sum) / len(x)
    result = np.clip(x + adjustment, 0, 1)
    
    # Fine-tune to exactly match sum
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-10:
        # Find indices to adjust
        idxs = np.argsort(result)
        if diff > 0:
            # Need to increase some values
            for i in range(int(diff / (1e-8))):
                if i < len(idxs) and result[idxs[i]] < 1:
                    result[idxs[i]] = min(1.0, result[idxs[i]] + 1e-8)
        else:
            # Need to decrease some values
            for i in range(int(-diff / (1e-8))):
                if i < len(idxs) and result[idxs[i]] > 0:
                    result[idxs[i]] = max(0.0, result[idxs[i]] - 1e-8)
    
    return result
