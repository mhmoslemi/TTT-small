import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx

    start_time = time.time()

    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None

    # Start with a piecewise constant function with alternating high/low values
    h0 = np.zeros(n_points)
    
    # Create a more targeted initial solution
    # Try a pattern: alternating high and low values
    for i in range(n_points):
        if i % 4 < 2:
            h0[i] = 0.8
        else:
            h0[i] = 0.2
    
    # Ensure the sum constraint
    h0 = project_to_sum_box(h0, TARGET_SUM)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Use a more sophisticated optimization algorithm - simulated annealing approach
    # with adaptive cooling schedule and mutation strategies
    max_iter = 100000
    temperature = 1.0
    cooling_rate = 0.9995
    mutation_strength = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Create a new candidate solution by perturbing current solution
        new_h = current_h.copy()
        
        # Apply random mutations
        # 70% chance to make small changes
        if np.random.random() < 0.7:
            # Make small perturbations to several random positions
            for _ in range(10):
                idx = np.random.randint(0, n_points)
                perturbation = np.random.normal(0, mutation_strength)
                new_h[idx] += perturbation
        
        # 30% chance to do larger perturbations for exploration
        else:
            # Select a random range and apply a larger change
            start_idx = np.random.randint(0, n_points - 10)
            end_idx = min(start_idx + 10 + np.random.randint(0, 20), n_points)
            change = np.random.normal(0, mutation_strength * 5)
            new_h[start_idx:end_idx] += change
        
        # Project to valid constraints
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h)
        
        # Accept or reject based on simulated annealing acceptance probability
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Accept worse solution with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.random() < np.exp(-delta / temperature):
                current_h = new_h
                current_c5 = new_c5
        
        # Update temperature
        temperature *= cooling_rate
        
        # Occasionally do a more aggressive search
        if iteration % 1000 == 0:
            # Try random restart with better initial solution
            restart_h = np.zeros(n_points)
            # Try a trapezoidal pattern that might work well
            mid = n_points // 2
            for i in range(n_points):
                if i < mid:
                    restart_h[i] = min(1.0, 0.3 + 0.7 * i / (mid - 1))
                else:
                    restart_h[i] = max(0.0, 1.0 - 0.7 * (i - mid) / (n_points - mid - 1))
            
            restart_h = project_to_sum_box(restart_h, TARGET_SUM)
            restart_c5 = evaluate_c5(restart_h)
            if restart_c5 < current_c5:
                current_h = restart_h
                current_c5 = restart_c5
                if restart_c5 < best_c5:
                    best_c5 = restart_c5

    # Final local refinement with gradient-based method if we have time
    if time.time() - start_time < 0.98 * budget_s and best_h is not None:
        h = best_h.copy()
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
                
            corr = np.correlate(h, 1-h, mode="full") * dx
            worst_shift_idx = np.argmax(corr)
            max_corr = corr[worst_shift_idx]
            
            # Simple gradient descent approach based on the correlation
            grad = np.zeros_like(h)
            for i in range(len(h)):
                # For each position i, compute how changing it affects the maximum correlation
                for j in range(len(h)):
                    shift = j - i
                    if shift >= -len(h)+1 and shift <= len(h)-1:
                        idx = shift + len(h) - 1
                        if idx < len(corr):
                            # Gradient contribution
                            if shift != 0:
                                grad[i] += corr[idx] * (1 - h[j] - h[i])  # Simplified approximation
                            else:
                                grad[i] -= 2 * corr[idx] * h[i]
            
            # Normalize and update
            if np.linalg.norm(grad) > 1e-10:
                grad = grad / np.linalg.norm(grad)
            
            # Apply gradient step
            step_size = 0.01
            new_h = h - step_size * grad
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            new_c5 = evaluate_c5(new_h)
            if new_c5 < evaluate_c5(h):
                h = new_h
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()

    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
