import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient of correlation for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smoothed max correlation"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert to exponentials, shift for numerical stability
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (len(h)-1)
            # compute grad for shift s
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent optimization restart"""
    # Initialize
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    T = T0
    step = step0
    
    for it in range(max_iter):
        if sym:
            # enforce symmetry
            h = (h + h[::-1]) / 2
            
        g = smooth_gradient(h, T, top_k)
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        h = project_onto_capped_simplex(h, target_sum)
        
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        T *= 0.9995
        if T < 1e-5:
            T = 1e-5
            
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Perform local swap optimization"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        
        # Calculate feasible delta range
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        
        if delta_low >= delta_high:
            continue
            
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different resolutions
    resolutions = [60, 120, 150]
    best_c5 = float('inf')
    best_h = None
    
    # Coarse optimization
    if time.time() - start_time < budget_s * 0.1:
        n = 60
        dx = 2.0 / n
        target_sum = n / 2
        h_coarse, c5_coarse = gradient_descent_restart(
            n, target_sum, 1500, 0.5, 0.1, 10, sym=False
        )
        
        if c5_coarse < best_c5:
            best_c5 = c5_coarse
            best_h = h_coarse.copy()
    
    # Upsample and refine at higher resolution
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.8:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        
        # Start from best solution found so far or interpolate coarse
        if best_h is not None and len(best_h) == 60:
            h_start = np.interp(
                np.linspace(0, 2, n, endpoint=False) + dx/2,
                np.linspace(0, 2, 60, endpoint=False) + dx/2,
                best_h
            )
            h_start = project_onto_capped_simplex(h_start, target_sum)
        else:
            h_start = np.random.rand(n)
            h_start = project_onto_capped_simplex(h_start, target_sum)
        
        # Run gradient descent restarts with symmetric and non-symmetric starts
        restarts = 10
        for i in range(restarts):
            if time.time() - start_time > budget_s * 0.8:
                break
                
            # Mixed initializations
            if i == 0:
                # Symmetric triangular
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h_init = np.maximum(0, 1 - np.abs(x - 1))
                h_init = project_onto_capped_simplex(h_init, target_sum)
                sym = True
            elif i == 1:
                # Symmetric random
                h_init = np.random.rand(n)
                h_init = project_onto_capped_simplex(h_init, target_sum)
                h_init = (h_init + h_init[::-1]) / 2
                h_init = project_onto_capped_simplex(h_init, target_sum)
                sym = True
            else:
                # Random start
                h_init = np.random.rand(n)
                h_init = project_onto_capped_simplex(h_init, target_sum)
                sym = False
            
            h, c5 = gradient_descent_restart(
                n, target_sum, 2000, 0.5, 0.1, 10, sym=sym
            )
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Local polishing with swap moves
    if best_h is not None and time.time() - start_time < budget_s * 0.95:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            best_h, best_c5 = local_swap_search(
                best_h, best_c5, remaining_time
            )
    
    if best_h is None:
        # Fallback
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
    
    # Final projection and evaluation
    best_h = project_onto_capped_simplex(best_h, TARGET_SUM)
    best_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
