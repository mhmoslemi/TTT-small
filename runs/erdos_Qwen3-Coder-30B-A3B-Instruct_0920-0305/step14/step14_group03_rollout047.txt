import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_reference(h):
    """Reference evaluation using correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_simplex_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute auto-correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_softmax_objective(h, alpha=10.0):
    """Compute the softmax approximation to max overlap"""
    corr = compute_correlation(h)
    # Convert to overlap values
    overlaps = dx * (TARGET_SUM - corr)
    # Apply softmax
    max_overlap = np.max(overlaps)
    if max_overlap > 0:
        w = np.exp(alpha * (overlaps - max_overlap))
    else:
        w = np.exp(alpha * overlaps)
    w = w / np.sum(w)
    # Return the weighted average
    return np.sum(w * overlaps)

def compute_gradient(h, alpha=10.0):
    """Compute exact gradient of softmax approximation"""
    corr = compute_correlation(h)
    # Compute overlaps
    overlaps = dx * (TARGET_SUM - corr)
    # Compute softmax weights
    max_overlap = np.max(overlaps)
    if max_overlap > 0:
        w = np.exp(alpha * (overlaps - max_overlap))
    else:
        w = np.exp(alpha * overlaps)
    w = w / np.sum(w)
    
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute contribution from each shift
    n = len(h)
    for s in range(-n+1, n):
        if s == 0:
            # Direct self-correlation term
            grad += -2 * dx * w[s + n - 1] * h
        else:
            # Cross-correlation terms
            if s > 0:
                # Forward shift: h[i] contributes to h[i+s]
                if s < n:
                    grad[:n-s] += -dx * w[s + n - 1] * h[s:]
                # Backward shift: h[i+s] contributes to h[i]  
                if s < n:
                    grad[s:] += -dx * w[s + n - 1] * h[:n-s]
            else:
                # Negative shift case (s < 0)
                s_abs = -s
                if s_abs < n:
                    grad[:n-s_abs] += -dx * w[s + n - 1] * h[s_abs:]
                if s_abs < n:
                    grad[s_abs:] += -dx * w[s + n - 1] * h[:n-s_abs]
    
    return grad

def adam_update(h, grad, m, v, beta1=0.9, beta2=0.999, epsilon=1e-8, eta=0.1):
    """Adam optimizer update with projection"""
    # Update moments
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    
    # Bias correction
    m_hat = m / (1 - beta1)
    v_hat = v / (1 - beta2)
    
    # Compute step
    step = eta * m_hat / (np.sqrt(v_hat) + epsilon)
    
    # Apply step and project
    new_h = h - step
    new_h = project_to_simplex_box(new_h, TARGET_SUM)
    
    return new_h, m, v

def backtracking_line_search(h, grad, m, v, alpha=10.0, eta=0.1, max_backtrack=8):
    """Backtracking line search to ensure improvement"""
    current_c5 = compute_softmax_objective(h, alpha)
    
    for _ in range(max_backtrack):
        new_h, new_m, new_v = adam_update(h, grad, m, v, eta=eta)
        new_c5 = compute_softmax_objective(new_h, alpha)
        
        if new_c5 < current_c5:
            return new_h, new_m, new_v, new_c5
            
        eta *= 0.5  # Reduce step size
    
    return h, m, v, current_c5

def interpolate_solution(h_coarse, n_fine):
    """Interpolate coarse solution to finer grid"""
    h_coarse = np.array(h_coarse)
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine
