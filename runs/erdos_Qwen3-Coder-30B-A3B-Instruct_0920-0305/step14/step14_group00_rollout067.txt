def evaluate_h(h_values, dx):
    """Evaluate the C5 bound for given h values"""
    h = np.array(h_values)
    n = len(h)
    
    # Compute all correlation values
    correlations = []
    for k in range(1, n):
        # Compute integral for shift k
        corr = 0
        for i in range(n - k):
            corr += h[i] * (1 - h[i + k])
        correlations.append(corr * dx)
    
    return max(correlations)
