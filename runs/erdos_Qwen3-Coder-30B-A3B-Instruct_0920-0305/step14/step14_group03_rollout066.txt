import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_softmax_grad(h, alpha=8.0):
    """Compute gradient using softmax approximation of max overlap"""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    
    # Use log-sum-exp trick for numerical stability
    exp_corr = np.exp(alpha * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient using vectorized operations
    grad = np.zeros_like(h)
    
    # For each shift s, compute contribution
    n = len(h)
    for s in range(-n+1, n):
        if s == 0:
            # s=0: contributes -2*weights[0]*h
            grad += -2 * weights[n-1] * h
        else:
            if s > 0:
                # Forward shift
                grad[:n-s] += -weights[n-1+s] * h[s:]
                # Backward shift
                grad[s:] += -weights[n-1+s] * h[:n-s]
            else:
                # Backward shift (positive s)
                s_abs = -s
                grad[s_abs:] += -weights[n-1+s] * h[:-s_abs]
                # Forward shift
                grad[:n-s_abs] += -weights[n-1+s] * h[s_abs:]
    
    return grad

def backtracking_line_search(h, grad, alpha=8.0, eta=0.12, max_backtracks=8):
    """Perform backtracking line search for step size selection"""
    for _ in range(max_backtracks):
        new_h = h - eta * grad
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        new_c5 = np.max(compute_corr(new_h))
        current_c5 = np.max(compute_corr(h))
        
        if new_c5 < current_c5:
            return new_h, eta
        eta *= 0.5
    
    return h, eta * 0.7

            m = np.zeros(n_points)

    if best_h is not None:

                delta = 1e-4
                # Try moving delta from i to j
                temp_h = h_refined.copy()
                temp_h[i] -= delta
                temp_h[j] += delta
                temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                temp_c5 = np.max(compute_corr(temp_h))
                if temp_c5 < np.max(compute_corr(h_refined)):
                    h_refined = temp_h
                    best_c5 = temp_c5

    result_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
    
    return (best_h.tolist(), float(result_c5), n_points)
