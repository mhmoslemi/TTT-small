import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_feasibility(n_points, target_sum, rhs_value, use_sdp=False):
    """Solve the linear feasibility problem or SDP relaxation"""
    dx = 2.0 / n_points
    
    if use_sdp:
        # SDP formulation
        h = cp.Variable(n_points)
        X = cp.Variable((n_points, n_points), symmetric=True)
        
        # Constraints
        constraints = [
            cp.diag(X) == h,
            h >= 0, h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Add PSD constraint
        constraints.append(X >> 0)
        
        # Add correlation constraints
        for s in range(-n_points+1, n_points):
            if s == 0:
                corr_expr = cp.sum(cp.diag(X, k=s))
            else:
                corr_expr = cp.sum(X[range(n_points-abs(s)), range(abs(s), n_points)])
            
            constraints.append(corr_expr >= rhs_value)
        
        # Objective (just feasibility)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None
    else:
        # Linear programming formulation with McCormick relaxation
        h = cp.Variable(n_points)
        
        # Variables for McCormick relaxation
        w = cp.Variable((n_points, n_points))
        
        # Constraints
        constraints = [
            h >= 0, h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # McCormick constraints
        for i in range(n_points):
            for j in range(n_points):
                if abs(i-j) < n_points:  # Valid correlation indices
                    s = j - i
                    if -n_points+1 <= s <= n_points-1:
                        # Constraint: w[i,j] >= h[i]*h[j] >= 0
                        constraints.append(w[i,j] >= 0)
                        constraints.append(w[i,j] <= h[i])
                        constraints.append(w[i,j] <= h[j])
                        constraints.append(w[i,j] >= h[i] + h[j] - 1)
        
        # Add correlation constraints
        for s in range(-n_points+1, n_points):
            # Compute correlation for shift s
            corr_expr = 0
            for i in range(n_points):
                j = i + s
                if 0 <= j < n_points:
                    corr_expr += w[i, j]
            
            constraints.append(corr_expr >= rhs_value)
        
        # Objective (just feasibility)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None

def compute_overlap(h):
    """Compute the maximum overlap of a step function"""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return dx * (n/2 - max_corr)
