def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""

    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1

    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_exact_gradient(h, alpha=10.0):
    """Compute exact gradient using softmax approximation of max"""

    corr = compute_corr(h)
    
    # Compute softmax weights for the correlation
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    
    # Compute gradient
    gradient = np.zeros_like(h)
    
    # Gradient computation: for each position i
    # d/dh_i = sum_j w_j * (delta_{i,j} - delta_{i,j+k})
    # where k is the shift
    for i in range(len(h)):
        for j in range(len(h)):
            if i == j:
                gradient[i] += w[j]  # weight for same position
            elif j < len(h) and i < len(h):
                # For shift j-i, compute contribution
                shift = j - i
                if shift < 0:
                    shift_idx = -shift - 1 + len(h) - 1
                else:
                    shift_idx = shift + len(h) - 1
                    
                if 0 <= shift_idx < len(corr):
                    gradient[i] -= w[shift_idx] * h[j]
    
    return gradient
