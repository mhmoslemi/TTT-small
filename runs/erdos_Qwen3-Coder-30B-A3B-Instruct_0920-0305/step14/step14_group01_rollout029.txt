def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(100):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def gradient_lse(h, epsilon):
    """Compute gradient of log-sum-exp approximation of max"""
    corr = compute_correlation(h)
    w = np.exp(corr / epsilon)
    w = w / np.sum(w)

    L = len(h) * 2 - 1

    # Forward convolution w * b
    conv1 = np.fft.ifft(np.fft.fft(w_pad, L) * np.fft.fft(b, L)).real
    # Reverse convolution w[::-1] * h
    conv2 = np.fft.ifft(np.fft.fft(w_pad[::-1], L) * np.fft.fft(h, L)).real

        penalty += lambda_inc * max(0, u[0] - 0)**2  # Boundary condition
