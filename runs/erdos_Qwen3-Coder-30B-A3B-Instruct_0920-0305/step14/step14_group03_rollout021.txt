import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_O(h, dx, target_sum):
    """Compute overlap vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * (target_sum - corr)

def compute_gradient_from_w(h, w, dx):
    """Compute gradient using softmax weights"""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute contributions for each shift s
    for r in range(2*n-1):
        s = r - (n-1)
        ws = w[r]
        if s == 0:
            grad += -dx * 2 * ws * h
        elif s > 0:
            # forward part
            grad[:n-s] += -dx * ws * h[s:]
            # backward part
            grad[s:] += -dx * ws * h[:n-s]
        else:
            s_abs = -s
            grad[s_abs:] += -dx * ws * h[:n-s_abs]
            grad[:n-s_abs] += -dx * ws * h[s_abs:]
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Try multiple n_points values
    n_candidates = [120, 150, 180, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    for n in n_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        TARGET_SUM = n / 2
        dx = 2.0 / n
        max_iters = 5000
        
        # Multiple restarts
        for restart in range(3):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Initialize h based on restart number
            if restart == 0:
                # Use the provided initial solution
                if 'initial_h_values' in globals() and len(initial_h_values) == n:
                    h0 = np.array(initial_h_values)
                else:
                    # Start with uniform distribution
                    h0 = np.full(n, 0.5)
            elif restart == 1:
                # Random initialization
                rand = np.random.rand(n)
                h0 = rand / np.sum(rand) * TARGET_SUM
            else:
                # Sinusoidal initialization
                x = np.linspace(0, 2, n, endpoint=False) + dx/2
                h0 = 0.5 + 0.3 * np.cos(np.pi * x / 2)
                h0 = np.clip(h0, 0, 1)
            
            h0 = project_to_sum_box(h0, TARGET_SUM)
            
            # Optimization parameters
            lr = 0.1
            alpha = 10.0
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            m = np.zeros_like(h0)
            v = np.zeros_like(h0)
            t = 0
            
            current_h = h0.copy()
            current_c5 = evaluate_c5(h0)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_n = n
            
            # Main optimization loop
            for iteration in range(max_iters):
                if time.time() - start_time > 0.95 * budget_s:
                    break
                    
                # Compute overlap values
                O = compute_O(current_h, dx, TARGET_SUM)
                C5 = O.max()
                
                # Update best solution
                if C5 < best_c5:
                    best_c5 = C5
                    best_h = current_h.copy()
                    best_n = n
                
                # Compute softmax weights
                max_O = O.max()
                exp_term = np.exp(alpha * (O - max_O))
                w = exp_term / np.sum(exp_term)
                
                # Compute gradient
                grad = compute_gradient_from_w(current_h, w, dx)
                
                # Adam updates
                t += 1
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                
                # Try update with projection
                new_h = current_h - step
                new_h = project_to_sum_box(new_h, TARGET_SUM)
                new_c5 = evaluate_c5(new_h)
                
                if new_c5 < C5:
                    current_h = new_h
                    current_c5 = new_c5
                else:
                    # Reduce learning rate
                    lr *= 0.5
                    if lr < 1e-6:
                        lr = 1e-6
                
                # Increase alpha gradually
                if iteration % 100 == 0:
                    alpha = min(alpha * 1.05, 200.0)
                
                # Occasionally add random perturbation
                if iteration % 500 == 0 and iteration > 0:
                    perturbation = np.random.normal(0, 0.001, n)
                    temp_h = current_h + perturbation
                    temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                    temp_c5 = evaluate_c5(temp_h)
                    if temp_c5 < current_c5:
                        current_h = temp_h
                        current_c5 = temp_c5

    if best_h is None:
        # Fallback to initial solution
        if 'initial_h_values' in globals() and len(initial_h_values) == 200:
            best_h = np.array(initial_h_values)
        else:
            best_h = np.full(200, 0.5)
        best_h = project_to_sum_box(best_h, 100)
        best_c5 = evaluate_c5(best_h)
        best_n = 200
    
    return (best_h.tolist(), float(best_c5), best_n)
