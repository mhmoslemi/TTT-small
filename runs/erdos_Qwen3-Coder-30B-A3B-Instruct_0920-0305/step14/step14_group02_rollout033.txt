import numpy as np
from scipy.fft import rfft, irfft
import time

def project_to_capped_simplex(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlations(h):
    """Compute all correlations via FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = irfft(rfft(h_rev, L) * rfft(b, L)).real
    return corr

def compute_gradient(corr, h, top_k=10):
    """Compute gradient using top-K shifts for efficiency"""
    # Get indices of top-K correlations
    top_indices = np.argpartition(corr, -top_k)[-top_k:]
    
    # Compute soft-max weights
    corr_top = corr[top_indices]
    max_corr = corr_top.max()
    weights = np.exp((corr_top - max_corr) / 1e-3)  # Very small temperature
    weights = weights / weights.sum()
    
    # Build gradient
    g = np.zeros_like(h)
    for i, (s, w) in enumerate(zip(top_indices, weights)):
        if s >= 0:
            # h_i (1 - h_{i+s})
            if s < len(h):
                g[:len(h)-s] += w * (1 - h[s:])
            # -h_{i-s} h_i
            if s > 0:
                g[s:] -= w * h[:-s]
        else:
            # s < 0, meaning shift backward
            s_abs = -s
            # h_i (1 - h_{i+s})
            g[s_abs:] += w * (1 - h[:len(h)-s_abs])
            # -h_{i-s} h_i
            if s_abs < len(h):
                g[:len(h)-s_abs] -= w * h[s_abs:]
    
    return g

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Grid parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Build triangular seed
    x = np.linspace(0, 2, n_points)
    h0 = np.maximum(0, 1 - np.abs(x - 1))
    h0 = project_to_capped_simplex(h0, target_sum)
    
    # Coarse optimization phase
    h_coarse = h0.copy()
    t_coarse = start_time + 0.1 * budget_s  # 10% for coarse
    
    # Coarse grid with 60 points
    n_coarse = 60
    x_coarse = np.linspace(0, 2, n_coarse)
    h_coarse = np.maximum(0, 1 - np.abs(x_coarse - 1))
    h_coarse = project_to_capped_simplex(h_coarse, n_coarse/2)
    
    # Run coarse optimization
    for i in range(2000):
        if time.time() > t_coarse:
            break
        
        # Compute correlations
        corr = compute_correlations(h_coarse)
        c5_value = dx * np.max(corr)
        
        # Gradient descent with top-K trick
        g = compute_gradient(corr, h_coarse, top_k=8)
        
        # Adaptive step size
        step_size = 1e-2 / (1 + i * 1e-4)
        h_coarse -= step_size * g
        
        # Project onto feasible set
        h_coarse = project_to_capped_simplex(h_coarse, n_coarse/2)
    
    # Interpolate to fine grid
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = project_to_capped_simplex(h_fine, target_sum)
    
    # Fine optimization with restarts
    best_h = h_fine.copy()
    best_c5 = dx * np.max(compute_correlations(h_fine))
    
    # Multiple restarts with different strategies
    restarts = 15
    
    for restart in range(restarts):
        if time.time() > start_time + 0.6 * budget_s:
            break
            
        # Random initialization
        h = np.random.rand(n_points)
        h = project_to_capped_simplex(h, target_sum)
        
        # Different optimization schedule
        if restart < 5:  # More iterations for first few restarts
            max_iter = 3000
        else:
            max_iter = 2000
            
        # Optimizer parameters
        T = 0.1
        alpha = 0.6
        beta = 0.9
        
        # Momentum
        v = np.zeros_like(h)
        
        for i in range(max_iter):
            if time.time() > start_time + 0.7 * budget_s:
                break
                
            if i % 20 == 0:  # Periodic exact evaluation
                corr = compute_correlations(h)
                current_c5 = dx * np.max(corr)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            
            # Compute correlations
            corr = compute_correlations(h)
            
            # Compute gradient
            g = compute_gradient(corr, h, top_k=10)
            
            # Momentum
            v = beta * v + (1 - beta) * g
            
            # Adaptive step size
            step_size = alpha / np.sqrt(i + 1)
            
            # Ensure step isn't too large
            g_norm = np.linalg.norm(g)
            if g_norm > 0:
                step_size = min(step_size, 0.1 / g_norm)
                
            h -= step_size * v
            
            # Project
            h = project_to_capped_simplex(h, target_sum)
    
    # Final exact evaluation
    corr = compute_correlations(best_h)
    final_c5 = dx * np.max(corr)
    
    # Local polishing with pair-swap
    current_h = best_h.copy()
    current_c5 = final_c5
    
    t_polish = time.time() + 0.2 * budget_s
    
    for i in range(100000):
        if time.time() > t_polish:
            break
            
        # Random pair swap
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        
        # Compute feasible range
        delta_min = max(-current_h[i1], current_h[i2] - 1)
        delta_max = min(1 - current_h[i1], current_h[i2])
        
        if delta_min >= delta_max:
            continue
            
        # Make swap
        delta = np.random.uniform(delta_min, delta_max)
        h_new = current_h.copy()
        h_new[i1] += delta
        h_new[i2] -= delta
        
        # Project to feasible set
        h_new = project_to_capped_simplex(h_new, target_sum)
        
        # Compute new C5
        corr_new = compute_correlations(h_new)
        new_c5 = dx * np.max(corr_new)
        
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
        else:
            # Accept with small probability to escape local minima
            temperature = max(0.001, 0.1 * (1 - (time.time() - start_time) / budget_s))
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
                current_h = h_new
                current_c5 = new_c5
                
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final projection and evaluation
    best_h = project_to_capped_simplex(best_h, target_sum)
    corr_best = compute_correlations(best_h)
    final_c5 = dx * np.max(corr_best)
    
    return (best_h.tolist(), float(final_c5), n_points)
