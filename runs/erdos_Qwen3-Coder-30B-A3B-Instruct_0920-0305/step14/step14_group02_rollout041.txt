import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    n = len(h)
    L = 2 * n - 1
    # use real FFT for speed
    h_rev = h[::-1]
    b = 1 - h
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    max_corr = np.max(corr)
    return max_corr * dx

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    b = 1 - h
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def subgradient_for_shift(h, s):
    """Compute subgradient for a single shift"""
    N = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        # i from 0 to N-1-s
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s = -s
        # i from s to N-1
        g[s:] += 1 - h[:N-s]
        g[:N-s] -= h[s:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute smooth gradient using top-K shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # use standard subgradient for max
        idx = np.argmax(corr)
        shift = idx - (len(h) - 1)
        return subgradient_for_shift(h, shift)
    else:
        # compute softmax weights
        max_corr = np.max(corr)
        # shift to improve numeric stability
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N-1)
            # compute grad for shift s
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart"""
    # Initialize
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    T = T0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for it in range(max_iter):
        if sym:
            h = (h + h[::-1]) / 2
        g = smooth_gradient(h, T, top_k)
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        h = project_onto_capped_simplex(h, target_sum)
        
        # Compute C5 every few iterations
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Decay T
        T *= 0.9995
        
        # Termination check
        if it > 1000 and T < 1e-4:
            break
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def local_swap_search(h, best_c5, time_limit):
    """Local swap search to refine the solution"""
    start = time.time()
    N = len(h)
    while time.time() - start < time_limit:
        # Pick two distinct indices
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_low, delta_high)
        
        # Apply swap
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate new solution
        c5_new = evaluate_c5_fast(h_new)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [60, 120]
    max_iter = 2000
    step0 = 0.5
    T0 = 0.1
    top_k = 10
    restarts = 15
    
    best_h = None
    best_c5 = float('inf')
    
    # Coarse phase
    n_coarse = 60
    dx_coarse = 2.0 / n_coarse
    target_sum_coarse = n_coarse / 2
    
    # Try triangular seed
    x = np.linspace(dx_coarse/2, 2 - dx_coarse/2, n_coarse)
    h_triangular = np.maximum(0, 1 - np.abs(x-1))
    h_triangular = project_onto_capped_simplex(h_triangular, target_sum_coarse)
    
    # Random restarts
    for restart in range(12):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Random initialization
        h0 = np.random.rand(n_coarse)
        h0 = project_onto_capped_simplex(h0, target_sum_coarse)
        
        # Run with triangular seed if restart 0
        if restart == 0:
            h, c5 = gradient_descent_restart(n_coarse, target_sum_coarse, max_iter, step0, T0, top_k, sym=False)
        else:
            h, c5 = gradient_descent_restart(n_coarse, target_sum_coarse, max_iter, step0, T0, top_k, sym=False)
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Upsample and refine on higher resolution
    if best_h is not None and n_points > n_coarse:
        # Interpolate to high resolution
        x_coarse = np.linspace(dx_coarse/2, 2 - dx_coarse/2, n_coarse)
        x_fine = np.linspace(dx/2, 2 - dx/2, n_points)
        h_upsampled = np.interp(x_fine, x_coarse, best_h)
        h_upsampled = project_onto_capped_simplex(h_upsampled, n_points / 2)
        
        # Refine at high resolution
        h, c5 = gradient_descent_restart(n_points, n_points / 2, max_iter, step0, T0, top_k, sym=False)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Local search on best solution
    if best_h is not None:
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time * 0.2)
    
    # Final evaluation
    if best_h is not None:
        best_c5 = evaluate_c5_fast(best_h)
        best_h = project_onto_capped_simplex(best_h, n_points / 2)
        best_c5 = evaluate_c5_fast(best_h)
    else:
        # Fallback to simple constant function
        best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
