```python
import numpy as np
from scipy.fft import fft, ifft
import time
import random

# Global constants
TARGET_SUM = None
dx = None

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def C5_value(h):
    """Evaluate C5 with dx scaling"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def interpolate(old_h, new_n):
    """Linear interpolation of a coarse solution onto a finer grid"""
    old_n = len(old_h)
    old_positions = np.linspace(0, 2, old_n, endpoint=False) + dx(old_n) / 2
    new_positions = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n) / 2
    new_h = np.interp(new_positions, old_positions, old_h)
    return project_to_sum_box(new_h, target_sum(new_n))

def worst_shift(c):
    """Return the shift index that achieves maximum correlation"""
    idx = np.argmax(c)
    return idx - (len(c) - 1)

def shift_arrays(h, s):
    """Return h_plus[i] = h_{i+s} and h_minus[i] = h_{i-s} with proper padding"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
        
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient of the worst shift term C_s"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Define stages
    stages = [
        {"n": 80, "time_fraction": 0.30},
        {"n": 160, "time_fraction": 0.35},
        {"n": 240, "time_fraction": 0.35}
    ]
    
    best_h = None
    best_C = float('inf')
    
    for stage_idx, stage in enumerate(stages):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        n = stage["n"]
        dx = 2.0 / n
        target_sum_n = target_sum(n)
        allocated_time = budget_s * stage["time_fraction"]
        
        # Initialize
        if stage_idx == 0:
            h = np.random.uniform(0, 1, n)
            h = project_to_sum_box(h, target_sum_n)
        else:
            h = interpolate(best_h, n)
            h = project_to_sum_box(h, target_sum_n)
            
        current_C = C5_value(h)
        best_h_stage = h.copy()
        best_C_stage = current_C
        
        # Simulated annealing
        T = 0.02
        delta_max = 0.1
        max_iter = int(1e6)
        accept_count = 0
        last_accept_count = 0
        warmup_iter = 1000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            if time.time() - start_time > allocated_time:
                break
                
            # Decide on move type
            if random.random() < 0.3 and i > warmup_iter:
                # Targeted move - focus on worst shift
                c = corr_fft(h)
                s = worst_shift(c)
                h_plus, h_minus = shift_arrays(h, s)
                d = h_plus + h_minus
                
                # Find low and high desirability indices
                sorted_indices = np.argsort(d)
                low_indices = sorted_indices[:max(1, len(d)//5)]
                high_indices = sorted_indices[-max(1, len(d)//5):]
                
                i_idx = random.choice(low_indices)
                j_idx = random.choice(high_indices)
            else:
                # Random move
                i_idx, j_idx = random.sample(range(n), 2)
                
            # Perform move
            delta = min(delta_max, h[i_idx], 1 - h[j_idx])
            if delta > 1e-10:
                delta = random.uniform(0, delta)
                h[i_idx] -= delta
                h[j_idx] += delta
                
                # Clip to bounds
                h[i_idx] = max(0, min(1, h[i_idx]))
                h[j_idx] = max(0, min(1, h[j_idx]))
                
                # Evaluate
                new_C = C5_value(h)
                
                # Accept or reject
                if new_C <= current_C or random.random() < np.exp(-(new_C - current_C) / T):
                    current_C = new_C
                    if new_C < best_C_stage:
                        best_h_stage = h.copy()
                        best_C_stage = new_C
                    accept_count += 1
                else:
                    # Revert move
                    h[i_idx] += delta
                    h[j_idx] -= delta
            
            # Update parameters
            if i % 50 == 0 and i > warmup_iter:
                T *= 0.9997
                delta_max *= 0.9999
                
                # Reset if acceptance rate is too low
                if accept_count < last_accept_count + 5 and i > warmup_iter + 1000:
                    T = min(0.05, T * 1.5)
                last_accept_count = accept_count
        
        # Polishing with exact gradient descent
        for k in range(2000):
            if time.time() - start_time > budget_s:
                break
                
            c = corr_fft(best_h_stage)
            max_val = np.max(c)
            tolerance = 1e-12
            worst_shifts = np.where(c >= max_val - tolerance)[0]
            
            if len(worst_shifts) == 0:
                break
                
            # Compute gradient for all worst shifts
            grad = np.zeros(n)
            for s_idx in worst_shifts:
                s = s_idx - (n - 1)
                g = gradient_worst(best_h_stage, s)
                grad += g
            
            grad /= len(worst_shifts)
            
            # Step size
            eta = 0.5 / np.sqrt(k + 1)
            
            # Update
            best_h_stage = best_h_stage - eta * grad
            best_h_stage = project_to_sum_box(best_h_stage, target_sum_n)
            
            new_C = C5_value(best_h_stage)
            if new_C < best_C_stage:
                best_C_stage = new_C
                best_h_stage = best_h_stage.copy()
        
        # Update global best
        if best_C_stage < best_C:
            best_C = best_C_stage
            best_h = best_h_stage.copy()
    
    return (best_h.tolist(), float(best_C), len(best_h))
```
