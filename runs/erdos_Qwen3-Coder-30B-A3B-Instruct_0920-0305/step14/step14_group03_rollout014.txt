def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient_exact(h):
    """Compute exact gradient of the objective function"""
    # We want to compute gradient of max_k ∫ h(x)(1 - h(x+k)) dx
    # This is equivalent to computing the gradient of the worst-case shift
    # Let's compute all correlations and find the worst one first
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    corr = corr * dx
    
    # Find the worst shift (max correlation)
    max_idx = np.argmax(corr)
    worst_shift = max_idx - len(h) + 1
    
    # Compute gradient at this worst shift
    grad = np.zeros_like(h)
    
    # For the worst shift k, we have contribution to gradient:
    # ∫ h(x)(1 - h(x+k)) dx, gradient is (1-h(x+k)) for x contributions
    # and -(1-h(x-k)) for (x+k) contributions
    
    # Let's compute with more care
    for i in range(len(h)):
        if 0 <= i + worst_shift < len(h):
            grad[i] += (1 - h[i + worst_shift])
        if 0 <= i - worst_shift < len(h):
            grad[i] -= (1 - h[i - worst_shift])
        
        # And we also need the term from h(x+k) derivative
        # The full gradient at position i with shift k
        # ∂/∂h_i ∫ h(x)(1 - h(x+k)) dx = 1 - h(x+k) with x = i
        # But we are not differentiating x, we are differentiating h_i
        # So we compute: the contribution to ∫ h(x)(1 - h(x+k)) dx from h_i
        # When x = i-k, we get h(i-k)(1-h(i)), when x = i, we get h(i)(1-h(i+k))
        # But we only take the derivative wrt h_i term: it appears in the term
        # h(i)(1 - h(i+k)) so ∂/∂h_i = 1 - h(i+k)
        
        # But the whole problem is more subtle; what we want is the derivative of 
        # max_k ∫ h(x)(1 - h(x+k)) dx with respect to h_i
        
        # Modeled directly with contribute to correlation term:
        # For each shift, the gradient of correlation with respect to h[i]
        # = 1 - h(i + k) for shift k and i where (i+k) in valid range (but this 
        # applies to the i+k term, so we compute it in reverse)
        
        # Simplification: we compute gradient for the worst shift directly:
        shift = worst_shift
        if shift >= 0:
            if 0 <= i - shift < len(h):
                grad[i] += (1 - h[i - shift])  # Additional from the h(i-shift) term in optimization
        else:  # shift < 0
            offset = -shift
            if 0 <= i + offset < len(h):
                grad[i] -= (1 - h[i + offset])  # This from h(i+offset) term (negative due to sign)
    
    return grad
