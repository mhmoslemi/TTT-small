import time
import random

# Global constants
TARGET_SUM = None
dx = 2.0 / 240  # Default value, will be updated in run

def dx(n):
    return 2.0 / n

def target_sum(n):
    return n / 2

def project_to_sum_box(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute correlation using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def C5_value(h):
    """Evaluate the objective C5"""
    c = corr_fft(h)
    return dx(len(h)) * np.max(c)

def shift_arrays(h, s):
    """Return h_plus and h_minus arrays for shift s"""
    n = len(h)
    h_plus = np.zeros(n)
    h_minus = np.zeros(n)
    
    if s >= 0:
        h_plus[:n-s] = h[s:]
        h_minus[s:] = h[:n-s]
    else:
        h_plus[-s:] = h[:n+s]
        h_minus[:n+s] = h[-s:]
    
    return h_plus, h_minus

def gradient_worst(h, s):
    """Compute exact sub-gradient of the worst term"""
    h_plus, h_minus = shift_arrays(h, s)
    return 1 - h_plus - h_minus

def worst_shift(corr):
    """Find shift that gives maximum correlation"""
    idx = np.argmax(corr)
    n = len(corr)
    return idx - (n - 1)

def interpolate(old_h, new_n):
    """Linear interpolation of coarse solution to finer grid"""
    old_n = len(old_h)
    old_positions = np.linspace(0, 2, old_n, endpoint=False) + dx(old_n) / 2
    new_positions = np.linspace(0, 2, new_n, endpoint=False) + dx(new_n) / 2
    new_h = np.interp(new_positions, old_positions, old_h)
    return new_h
