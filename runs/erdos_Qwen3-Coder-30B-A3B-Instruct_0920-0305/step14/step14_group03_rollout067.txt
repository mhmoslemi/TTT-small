import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5_reference(h):
    """Reference evaluation using correlate"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_grad(h, corr, alpha=10.0):
    """Compute exact gradient using softmax approximation of max"""
    # Compute softmax weights
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    
    # Normalize weights
    w = w / np.sum(w)
    
    # Compute gradient using vectorized operations
    grad = np.zeros_like(h)
    
    # Handle each shift weight
    n = len(h)
    for s in range(-n+1, n):
        if s == 0:
            grad += -2 * w[s + n - 1] * h
        elif s > 0:
            # Forward shift
            grad[:n-s] += -w[s + n - 1] * h[s:]
            # Backward shift
            grad[s:] += -w[s + n - 1] * h[:n-s]
        else:  # s < 0
            # Forward shift (symmetric to s>0 case)
            grad[-s:] += -w[-s + n - 1] * h[:-s]
            # Backward shift (symmetric to s>0 case)
            grad[:n+s] += -w[-s + n - 1] * h[-s:]
    
    return grad

def compute_softmax_objective(h, alpha=10.0):
    """Compute the soft-max approximation to max overlap"""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    if max_corr > 0:
        w = np.exp(alpha * (corr - max_corr))
    else:
        w = np.exp(alpha * corr)
    w = w / np.sum(w)
    return np.sum(w * corr)

def adam_step(h, grad, m, v, beta1=0.9, beta2=0.999, epsilon=1e-8, lr=0.1):
    """Perform Adam step with projection"""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1)
    v_hat = v / (1 - beta2)
    step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
    new_h = h - step
    return new_h, m, v

def backtracking_line_search(h, grad, m, v, alpha=10.0, lr=0.1, max_backtrack=8):
    """Perform backtracking line search for better step size"""
    for _ in range(max_backtrack):
        new_h, _, _ = adam_step(h, grad, m, v, lr=lr)
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        new_c5 = compute_softmax_objective(new_h, alpha)
        if new_c5 < compute_softmax_objective(h, alpha):
            return new_h, lr
        lr *= 0.5
    return h, lr
