import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def genetic_algorithm_strategy():
    """Genetic algorithm with adaptive parameters"""
    popsize = 30
    max_gen = 5000
    population = [project_to_sum_box(np.random.rand(n_points), TARGET_SUM) 
                  for _ in range(popsize)]

    population[0] = initial_h_values.copy()
    population[0] = project_to_sum_box(population[0], TARGET_SUM)
    
    for gen in range(max_gen):
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in population]
        
        # Selection: tournament selection
        new_population = []
        for _ in range(popsize):
            idx1, idx2 = np.random.choice(popsize, 2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                parent = population[idx1].copy()
            else:
                parent = population[idx2].copy()
            new_population.append(parent)
        
        # Crossover and mutation
        for i in range(0, popsize, 2):
            if i + 1 < popsize:
                # Uniform crossover
                child1, child2 = new_population[i].copy(), new_population[i+1].copy()
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child1[j], child2[j] = child2[j], child1[j]
                
                # Mutation
                for j in range(n_points):
                    if np.random.rand() < 0.05:
                        child1[j] = np.clip(child1[j] + np.random.normal(0, 0.02), 0, 1)
                    if np.random.rand() < 0.05:
                        child2[j] = np.clip(child2[j] + np.random.normal(0, 0.02), 0, 1)
                
                child1 = project_to_sum_box(child1, TARGET_SUM)
                child2 = project_to_sum_box(child2, TARGET_SUM)
                
                new_population[i] = child1
                new_population[i+1] = child2
        
        population = new_population
    
    # Return best individual
    fitness = [evaluate_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    return population[best_idx].copy(), fitness[best_idx]

def simulated_annealing_strategy():
    """Simulated Annealing with adaptive cooling"""
    h = initial_h_values.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    temperature = 1.0
    min_temperature = 1e-6
    cooling_rate = 0.999
    max_iter = 3000
    
    for iter_num in range(max_iter):
        # Generate neighbor solution
        h_neighbor = h.copy()
        idx = np.random.randint(n_points)
        change = np.random.normal(0, 0.03)
        h_neighbor[idx] = np.clip(h_neighbor[idx] + change, 0, 1)
        h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
        
        neighbor_c5 = evaluate_c5(h_neighbor)
        
        # Accept or reject
        if neighbor_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - neighbor_c5) / temperature):
            h = h_neighbor
            current_c5 = neighbor_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        temperature *= cooling_rate
        if temperature < min_temperature:
            temperature = min_temperature
    
    return best_h.copy(), best_c5

def gradient_based_strategy():
    """Gradient-based optimization"""
    h = initial_h_values.copy()
    h = project_to_sum_box(h, TARGET_SUM)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    max_iter = 2000
    learning_rate = 0.05
    
    for iter_num in range(max_iter):
        # Estimate gradient with finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(h[i] + eps, 1.0)
            h_plus = project_to_sum_box(h_plus, TARGET_SUM)
            
            h_minus = h.copy()
            h_minus[i] = max(h[i] - eps, 0.0)
            h_minus = project_to_sum_box(h_minus, TARGET_SUM)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient and momentum
        h_new = h - learning_rate * grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h.copy(), best_c5

def local_search_strategy():
    """Local search with various neighborhood structures"""
    h = initial_h_values.copy()

    current_c5 = evaluate_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    max_iter = 2000
    
    for iter_num in range(max_iter):
        # Try different neighborhoods
        if np.random.rand() < 0.7:
            # Single point perturbation
            idx = np.random.randint(n_points)
            h_neighbor = h.copy()
            h_neighbor[idx] = np.clip(h_neighbor[idx] + np.random.normal(0, 0.02), 0, 1)
            h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
        else:
            # Block perturbation
            start_idx = np.random.randint(n_points)
            block_size = min(10, n_points - start_idx)
            h_neighbor = h.copy()
            for i in range(start_idx, start_idx + block_size):
                h_neighbor[i] = np.clip(h_neighbor[i] + np.random.normal(0, 0.03), 0, 1)
            h_neighbor = project_to_sum_box(h_neighbor, TARGET_SUM)
        
        c5_neighbor = evaluate_c5(h_neighbor)
        
        if c5_neighbor < current_c5:
            h = h_neighbor
            current_c5 = c5_neighbor
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return best_h.copy(), best_c5
