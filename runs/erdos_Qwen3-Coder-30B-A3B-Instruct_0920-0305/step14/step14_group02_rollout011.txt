import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT - returns correlation values for all shifts"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient vector of correlation at shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient of smooth max correlation approximation"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Use subgradient of exact max
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    
    # Compute softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    weights = exp_vals / np.sum(exp_vals)
    
    # Use only top_k weights for efficiency
    top_idx = np.argpartition(weights, -top_k)[-top_k:]
    g = np.zeros_like(h)
    N = len(h)
    for idx in top_idx:
        w = weights[idx]
        s = idx - (N-1)
        grad_s = subgradient_for_shift(h, s)
        g += w * grad_s
    return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run one gradient descent restart with optional symmetry"""
    # Random initial h
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    # Gradient descent parameters
    T = T0
    step = step0
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    velocity = np.zeros_like(h)
    beta = 0.9  # momentum parameter
    
    for it in range(max_iter):
        # Compute smooth gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update with momentum
        velocity = beta * velocity + (1 - beta) * g
        h = h - step * velocity
        
        # Apply symmetry constraint if requested
        if sym:
            h = (h + h[::-1]) / 2
            
        # Project onto feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Periodically evaluate exact C5
        if it % 20 == 0:
            c5 = evaluate_c5_fast(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Update step size and temperature
        step = step0 / np.sqrt(it + 1)
        T *= 0.9995
        
        # Early exit if T is very small
        if T < 1e-5:
            break
    
    # Final evaluation
    c5 = evaluate_c5_fast(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return best_h, best_c5

def local_swap_search(h, c5, time_limit):
    """Local pairwise swap search"""
    start = time.time()
    N = len(h)
    best_h = h.copy()
    best_c5 = c5
    
    while time.time() - start < time_limit:
        # Pick two distinct indices
        i, j = np.random.choice(N, 2, replace=False)
        
        # Compute feasible delta range
        delta_low = max(-best_h[i], best_h[j] - 1)
        delta_high = min(1 - best_h[i], best_h[j])
        
        if delta_low >= delta_high:
            continue
            
        # Sample delta
        delta = np.random.uniform(delta_low, delta_high)
        
        # Create neighbor
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Evaluate
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine problem size
    n_list = [60, 120, 150]  # Order of preference
    
    best_h = None
    best_c5 = float('inf')
    
    # Coarse optimization first
    if n_list[0] <= n_points:
        h0 = np.random.rand(n_list[0])
        h0 = project_onto_capped_simplex(h0, n_list[0]/2)
        # Simple coarse optimization
        for _ in range(1000):
            g = smooth_gradient(h0, 0.1, 5)
            h0 = h0 - 0.5 * g
            h0 = project_onto_capped_simplex(h0, n_list[0]/2)
            c5 = evaluate_c5_fast(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                
    # Upsample if needed
    if best_h is not None and n_points > n_list[0]:
        x_coarse = np.linspace(0, 2, n_list[0], endpoint=False) + 1.0/n_list[0]
        x_fine = np.linspace(0, 2, n_points, endpoint=False) + 1.0/n_points
        best_h = np.interp(x_fine, x_coarse, best_h)
        best_h = project_onto_capped_simplex(best_h, n_points/2)
    
    # Multiple restarts with different initializations
    restarts = 15
    for restart in range(restarts):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Different initializations
        if restart == 0:
            # Triangular shape
            x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
            h0 = np.maximum(0, 1 - np.abs(x - 1))
            h0 = project_onto_capped_simplex(h0, n_points/2)
        elif restart < 3:
            # Symmetric shape
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, n_points/2)
            h0 = (h0 + h0[::-1]) / 2
        else:
            # Random
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, n_points/2)
        
        # Run gradient descent
        h, c5 = gradient_descent_restart(
            n_points, n_points/2, 
            max_iter=2000, 
            step0=0.5, 
            T0=0.1, 
            top_k=10
        )
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Local refinement
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        best_h, best_c5 = local_swap_search(
            best_h, best_c5, 
            time_limit=remaining_time * 0.8
        )
    
    # Final projection to ensure feasibility
    best_h = project_onto_capped_simplex(best_h, n_points/2)
    final_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
