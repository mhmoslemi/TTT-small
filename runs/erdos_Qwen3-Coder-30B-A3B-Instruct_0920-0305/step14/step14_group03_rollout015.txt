    h_rev = h[::-1]
    b = 1 - h

    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Using Duchi algorithm for projection
    x = np.array(x)
    u = np.sort(x)[::-1]
    css = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, len(u) + 1) > (css - target_sum))
    if rho == 0:
        # All elements non-negative, just scale
        lambda_val = (np.sum(x) - target_sum) / len(x)
        projected = np.clip(x - lambda_val, 0, 1)
        return projected
    else:
        # Use the projection formula from Duchi
        lambda_val = (css[rho - 1] - target_sum) / rho
        projected = np.clip(x - lambda_val, 0, 1)
        return projected

def compute_corr(h):
    """Compute correlation vector using FFT"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

    grad = np.zeros_like(h)
