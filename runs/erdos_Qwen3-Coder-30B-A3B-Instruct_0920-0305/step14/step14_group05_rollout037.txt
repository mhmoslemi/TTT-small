def evaluate_c5(h):

    h_rev = h[::-1]

        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10

        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
