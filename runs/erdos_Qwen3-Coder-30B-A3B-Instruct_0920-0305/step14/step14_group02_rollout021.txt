import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_onto_capped_simplex(x, target):
    """Project vector x onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def corr_fft(h):
    """Compute full correlation array using FFT"""
    N = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * N - 1
    H = np.fft.rfft(h_rev, n=L)
    B = np.fft.rfft(b, n=L)
    corr = np.fft.irfft(H * B, n=L)
    return corr

def evaluate_c5_fast(h):
    """Fast evaluation of C5 using FFT"""
    corr = corr_fft(h)
    max_corr = np.max(corr)
    return max_corr * dx

def subgradient_for_shift(h, s):
    """Compute gradient for a single shift s"""
    N = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:N-s] += 1 - h[s:]
        g[s:] -= h[:N-s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:N-s_abs]
        g[:N-s_abs] -= h[s_abs:]
    return g

def smooth_gradient(h, T, top_k=10):
    """Compute gradient using softmax weights over top_k shifts"""
    corr = corr_fft(h)
    if T < 1e-12:
        # Just subgradient of max shift
        idx = np.argmax(corr)
        shift = idx - (len(h)-1)
        return subgradient_for_shift(h, shift)
    else:
        # Convert to exponentials
        max_corr = np.max(corr)
        exp_vals = np.exp((corr - max_corr) / T)
        weights = exp_vals / np.sum(exp_vals)
        # Use only top_k weights for efficiency
        top_idx = np.argpartition(weights, -top_k)[-top_k:]
        g = np.zeros_like(h)
        N = len(h)
        for idx in top_idx:
            w = weights[idx]
            s = idx - (N-1)
            grad_s = subgradient_for_shift(h, s)
            g += w * grad_s
        return g

def gradient_descent_restart(n, target_sum, max_iter, step0, T0, top_k, sym=False):
    """Run gradient descent with continuation for a single restart"""
    # Generate random initial h
    h = np.random.rand(n)
    h = project_onto_capped_simplex(h, target_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_c5_fast(h)
    
    T = T0
    step = step0
    for it in range(max_iter):
        if sym:
            # Enforce symmetry after each update
            h = (h + h[::-1]) / 2
        
        # Compute gradient
        g = smooth_gradient(h, T, top_k)
        
        # Update with momentum
        step = step0 / np.sqrt(it + 1)
        h = h - step * g
        
        # Project back to feasible set
        h = project_onto_capped_simplex(h, target_sum)
        
        # Store best solution
        c5 = evaluate_c5_fast(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay temperature
        T *= 0.9995
        
        # Early exit if time limit nears
        if it % 50 == 0 and it > 0:
            # Check if we're close to time limit
            if it > max_iter * 0.8:
                break
    
    return best_h, best_c5

def local_swap_search(h, best_c5, time_limit):
    """Local hill-climbing using pair swaps"""
    start = time.time()
    N = len(h)
    while time.time() - start < time_limit:
        i, j = np.random.choice(N, 2, replace=False)
        # Sample delta from Uniform(-d, d)
        d = 0.1
        delta = np.random.uniform(-d, d)
        
        # Bound delta to keep both values in [0,1]
        delta_low = max(-h[i], h[j] - 1)
        delta_high = min(1 - h[i], h[j])
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # No need for projection; sum is preserved
        c5_new = evaluate_c5_fast(h_new)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = float('inf')
    
    # Coarse optimization first
    coarse_n = 60
    coarse_target_sum = coarse_n / 2
    coarse_h, coarse_c5 = gradient_descent_restart(
        coarse_n, coarse_target_sum, 1000, 0.5, 0.1, 10, sym=False
    )
    
    # Upsample to main resolution
    if coarse_c5 < best_c5:
        best_c5 = coarse_c5
        best_h = coarse_h.copy()
    
    # Refine at higher resolution
    upscaled_h = np.interp(
        np.linspace(0, 2, n_points, endpoint=False) + dx/2,
        np.linspace(0, 2, coarse_n, endpoint=False) + dx/2,
        coarse_h
    )
    upscaled_h = project_onto_capped_simplex(upscaled_h, TARGET_SUM)
    if evaluate_c5_fast(upscaled_h) < best_c5:
        best_c5 = evaluate_c5_fast(upscaled_h)
        best_h = upscaled_h.copy()
    
    # Multiple restarts with different strategies
    restarts = 15
    sym_restarts = 3
    for r in range(restarts):
        if time.time() - start_time > budget_s * 0.9:
            break
        
        # Choose initialization strategy
        if r == 0:
            # Triangular seed
            x = np.linspace(0, 2, n_points, endpoint=False) + dx/2
            h0 = np.maximum(0, 1 - np.abs(x-1))
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        elif r < sym_restarts:
            # Symmetric random
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
            h0 = (h0 + h0[::-1]) / 2
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        else:
            # Random uniform
            h0 = np.random.rand(n_points)
            h0 = project_onto_capped_simplex(h0, TARGET_SUM)
        
        h, c5 = gradient_descent_restart(n_points, TARGET_SUM, 2000, 0.5, 0.1, 10, sym=False)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Local refinement with pair swaps
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        best_h, best_c5 = local_swap_search(best_h, best_c5, remaining_time)
    
    # Final projection and evaluation
    best_h = project_onto_capped_simplex(best_h, TARGET_SUM)
    final_c5 = evaluate_c5_fast(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
