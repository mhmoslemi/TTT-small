def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute correlation using FFT
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use Duchi's algorithm for efficient projection
    x = np.array(x)
    # Sort in descending order
    idx = np.argsort(x)[::-1]
    x_sorted = x[idx]
    
    # Compute cumulative sums
    cumsum = np.cumsum(x_sorted)
    n = len(x)
    
    # Find the threshold
    t = (cumsum - target_sum) / np.arange(1, n + 1)
    t = np.maximum(t, 0)
    
    # Find the rightmost positive threshold
    pos = np.where(t >= 0)[0]
    if len(pos) == 0:
        tau = 0
    else:
        tau = np.max(t[pos])
    
    # Apply projection
    result = np.clip(x - tau, 0, 1)
    
    return result

def compute_gradient_exact(h):
    """Compute exact gradient of the maximum correlation"""
    L = 2 * len(h) - 1
    h_rev = h[::-1]
    b = 1 - h
    
    # Compute correlation to find maximum
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Find the worst shift
    max_idx = np.argmax(corr)
    worst_shift = max_idx - len(h) + 1
    
    # Compute gradient analytically
    grad = np.zeros_like(h)
    
    # For the specific shift that causes maximum correlation
    # The gradient w.r.t. h[i] is the contribution from shifts that involve h[i]
    if worst_shift >= 0:
        # For shift k, contributions to h[i] are from h[i-k] (and h[i+k] indirectly)
        # The derivative of correlation with respect to h[i] involves the 
        # contributions from terms where h[i] appears
        # Specifically, sum_{k} h[i-k](1-h[i]) (if i-k positive)
        for i in range(len(h)):
            # Contribution from shift worst_shift
            if i - worst_shift >= 0:
                grad[i] += (1 - h[i - worst_shift])
    else:
        # For negative shift
        for i in range(len(h)):
            # Contribution from shift worst_shift
            if i + worst_shift < len(h):
                grad[i] += (1 - h[i + worst_shift])
    
    # More accurate approach - explicit computation
    grad = np.zeros_like(h)
    for i in range(len(h)):
        for k in range(len(h)):
            shift = k - i
            if 0 <= k < len(h) and abs(shift) < len(h) - 1:
                # Add contribution to h[i] from shift that puts h[k] at position i
                # This creates the correlation term:
                # h[i] * (1 - h[i+shift]) 
                if abs(shift) == 0:
                    grad[i] += 1 - h[i]
                else:
                    idx_shift = i + shift
                    if 0 <= idx_shift < len(h):
                        grad[i] += 1 - h[idx_shift]
    
    # Normalize by dx
    return grad
