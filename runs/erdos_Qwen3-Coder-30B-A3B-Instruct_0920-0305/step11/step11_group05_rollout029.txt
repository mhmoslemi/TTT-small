import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Compute C5 using FFT correlation"""
    # Ensure correct length and normalization
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def smooth_max(h, tau=0.05):
    """Compute smooth max using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    S = n / 2
    # Compute all g_s values
    g_s = []
    for s in range(-n+1, n):
        if s >= 0:
            g = dx * (S - np.sum(h[:-s] * h[s:]))
        else:
            g = dx * (S - np.sum(h[-s:] * h[:s]))
        g_s.append(g)
    
    g_s = np.array(g_s)
    # Compute smooth max
    max_g = np.max(g_s)
    exp_vals = np.exp((g_s - max_g) / tau)
    smooth_max_val = max_g + tau * np.log(np.sum(exp_vals))
    return smooth_max_val

def compute_gradient(h, tau=0.05):
    """Compute gradient of smooth max using FFT"""
    n = len(h)
    dx = 2.0 / n
    S = n / 2
    
    # FFT of h and 1-h
    h_fft = fft(h)
    one_minus_h_fft = fft(1 - h)
    
    # Compute correlation (g_s values)
    corr = ifft(h_fft.conj() * one_minus_h_fft).real
    
    # Shift correlation to prepare for handling arbitrary shifts
    # corr[i] corresponds to g_{i-n+1} so we need to handle properly
    g_s = np.zeros(2*n - 1)
    g_s[:n] = corr[n-1:]
    g_s[n:] = corr[:n-1]
    
    g_s = g_s[:2*n-1]
    g_s = dx * (S - g_s)
    
    # Compute weights for the weighted average
    max_g = np.max(g_s)
    exp_vals = np.exp((g_s - max_g) / tau)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute weighted average of gradients
    grad = np.zeros(n)
    
    # Gradient of g_s(h) = dx * (S - sum h_i * h_{i+s})
    # d(g_s)/d(h_i) = dx * (1 - h_{i+s}) - dx * h_{i-s} 
    # So for weighted average, we compute:
    # sum_s w_s * (1 - h_{i+s} - h_{i-s})
    
    # Using FFT trick:
    # A = ifft(fft(weights) * fft(1-h)) 
    # B = ifft(fft(weights) * fft(h))
    weights_fft = fft(weights)
    A_fft = weights_fft * one_minus_h_fft
    B_fft = weights_fft * h_fft
    A = ifft(A_fft).real
    B = ifft(B_fft).real
    
    # Fix indices properly
    grad = dx * (A - B)
    
    return grad

def accelerated_projected_gradient(h0, max_iter=3000, tau=0.05, eta=1e-4, tol=1e-6):
    """Accelerated projected gradient method with backtracking line search"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    
    for k in range(max_iter):
        # Compute current value and gradient using FFT
        current_value = smooth_max(h, tau)
        gradient = compute_gradient(h, tau)
        
        # Backtracking line search
        alpha = 1.0
        max_step = 0.1
        gradient_norm_sq = np.sum(gradient**2)
        step_size = 1.0
        
        # Compute initial descent
        h_trial = project_to_sum_box(y - alpha * gradient, len(h) / 2)
        if np.sum(h_trial) > len(h) / 2 + 1e-10:
            h_trial = project_to_sum_box(y - alpha * gradient, len(h) / 2)
            
        trial_value = smooth_max(h_trial, tau)
        
        while trial_value > current_value - eta * alpha * gradient_norm_sq and alpha > 1e-8:
            alpha *= 0.5
            h_trial = project_to_sum_box(y - alpha * gradient, len(h) / 2)
            if np.sum(h_trial) > len(h) / 2 + 1e-10:
                h_trial = project_to_sum_box(y - alpha * gradient, len(h) / 2)
            trial_value = smooth_max(h_trial, tau)
        
        # Update
        h_new = h_trial
        t_new = (1 + np.sqrt(1 + 4*t**2)) / 2
        y_new = h_new + ((t - 1)/t_new) * (h_new - h)
        
        # Update variables
        h = h_new
        y = y_new
        t = t_new
        
        # Check convergence
        if k > 10 and abs(current_value - smooth_max(h, tau)) < tol:
            break
            
    return h

def symmetrize_and_project(h):
    """Symmetrize and project to feasible set"""
    h_sym = (h + h[::-1]) / 2
    return project_to_sum_box(h_sym, len(h) / 2)

        current_h = h0.copy()

        tau_values = [0.05, 0.02, 0.01, 0.005, 0.001, 1e-4]
        for tau in tau_values:
            if time.time() - start_time > budget_s - 10:
                break
                
            current_h = accelerated_projected_gradient(

                tau=tau,

            )

            current_h = symmetrize_and_project(current_h)
            
            # Evaluate with true C5

    if best_h is not None and time.time() - start_time < budget_s - 5:

        current_h = best_h.copy()
        current_c5 = evaluate_c5(current_h)
        
        # Simulated annealing approach with small step sizes
        for i in range(2000):

            idx1, idx2 = np.random.choice(n_points, size=2, replace=False)
            
            # Compute admissible step
            delta_max = min(current_h[idx1], 1 - current_h[idx2])
            if delta_max <= 1e-10:
                continue
                
            # Small random step
            delta = np.random.uniform(-delta_max*0.01, delta_max*0.01)
            
            # Make move
            new_h = current_h.copy()
            new_h[idx1] -= delta
            new_h[idx2] += delta
            
            # Project to ensure constraints
            new_h = project_to_sum_box(new_h, target_sum)
            
            # Evaluate
            new_c5 = evaluate_c5(new_h)
            
            # Accept or reject
            if new_c5 < current_c5:

            best_c5 = evaluate_c5(best_h)

        # Fallback to initial point
        best_h = initial_h_values
        best_c5 = evaluate_c5(initial_h_values)
    
    return (best_h.tolist(), float(best_c5), n_points)
