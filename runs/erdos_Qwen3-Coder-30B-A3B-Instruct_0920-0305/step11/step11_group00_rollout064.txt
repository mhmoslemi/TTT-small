def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum using binary search"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    return np.clip(v + (a + b)/2, 0, 1)

def corr_full(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute the gradient of c_k(h) with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1 - h[:-k] if k > 0 else 1 - h
        if k < n - 1:
            grad[:n-k] -= h[k:]
    else:
        k_abs = -k
        grad[k_abs:] = 1 - h[:-k_abs] if k_abs > 0 else 1 - h
        if k_abs < n - 1:
            grad[:n-k_abs] -= h[k_abs:]
    return grad

def max_overlap(h):
    """Compute max overlap with dx factor"""
    dx = 2.0 / len(h)
    corr = corr_full(h)
    return dx * np.max(corr)

def shift_mask(k, n):
    """Generate sparse matrix for shifting k"""
    M = np.zeros((n, n))
    for i in range(max(0, -k), min(n, n - k)):
        M[i, i + k] = 1
    return M

def build_sdp(n):
    """Build SDP problem for the relaxation"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
        
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau

def solve_sdp(prob):
    """Solve SDP and return (h, X, tau_opt)"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
            return prob.value, prob.variables()[0].value, prob.variables()[1].value
    except Exception:
        pass
    return np.inf, None, None

def round_from_sdp(h_sdp, X_sdp, n):
    """Round a solution to get a feasible step function"""
    # Method 1: Diagonal rounding
    h1 = h_sdp.copy()
    
    # Method 2: Leading eigenvector rounding
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(eigenvals)
        eigval = eigenvals[idx]
        eigvec = eigenvecs[:, idx]
        if eigval > 0:
            h2 = np.sqrt(eigval) * eigvec
            h2 = proj_sum_box(h2, n/2)
        else:
            h2 = h1.copy()
    except:
        h2 = h1.copy()
    
    # Method 3: Gaussian randomization
    h3_best = h1.copy()
    best_c5 = max_overlap(h3_best)
    for _ in range(5):
        z = np.random.multivariate_normal(np.zeros(n), X_sdp)
        h3 = np.abs(z)
        h3 = proj_sum_box(h3, n/2)
        c5 = max_overlap(h3)
        if c5 < best_c5:
            best_c5 = c5
            h3_best = h3
    
    # Return best among roundings
    c5_1 = max_overlap(h1)
    c5_2 = max_overlap(h2)
    if c5_1 <= c5_2 and c5_1 <= best_c5:
        return h1
    elif c5_2 <= best_c5:
        return h2
    else:
        return h3_best

            break
            
        corr = corr_full(h)
        Cmax = np.max(corr)
        Kmax = np.where(corr >= Cmax - 1e-12)[0]
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new)
        if c5_new < best_c5:
            step = min(step * 1.08, 0.2)
            best_h = h_new.copy()
            best_c5 = c5_new

            h = h_new.copy()
        else:

            delta = np.random.uniform(-0.02, 0.02)
            h_new = h.copy()
            # transfer delta from i to j
            old_i = h_new[i]
            old_j = h_new[j]
            h_new[i] = np.clip(old_i - delta, 0, 1)
            h_new[j] = np.clip(old_j + delta, 0, 1)

            c5_new = max_overlap(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
    
    return best_h, best_c5
