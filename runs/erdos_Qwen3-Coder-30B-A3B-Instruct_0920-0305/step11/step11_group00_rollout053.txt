import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    low = -np.max(x) - 1
    high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (low + high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(low) < 0:
            high = m
        else:
            low = m
    
    result = np.clip(x + (low + high)/2, 0, 1)
    return result

def grad_shift(h, k):
    """Compute gradient of correlation c_k(h) for shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def lse_surrogate(h, alpha):
    """Compute log-sum-exp surrogate of max correlation"""
    corr = compute_corr_fft(h)
    max_corr = np.max(corr)
    if max_corr == -np.inf:
        return 0, np.zeros_like(h)
    
    # Safe computation of log-sum-exp
    exp_vals = np.exp(alpha * (corr - max_corr))
    sum_exp = np.sum(exp_vals)
    log_sum_exp = np.log(sum_exp) + max_corr
    
    # Compute weights
    weights = exp_vals / sum_exp
    
    # Compute gradient
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        if w > 1e-15:  # Skip negligible weights
            k = i - (n_points - 1)  # Convert index to shift
            grad += w * grad_shift(h, k)
    
    return log_sum_exp, grad

def backtracking_line_search(h, grad, alpha, eta, f_old):
    """Perform backtracking line search for step size"""
    c1 = 0.5
    rho = 0.5
    max_reductions = 5
    
    for _ in range(max_reductions):
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        f_new, _ = lse_surrogate(h_new, alpha)
        if f_new <= f_old + c1 * eta * np.dot(grad, grad):
            return h_new, eta
        eta *= rho
        
    return h, eta

def random_perturbation(h, delta=0.02):
    """Perform a small random swap move to escape local minima"""
    h_new = h.copy()
    i, j = np.random.choice(n_points, 2, replace=False)
    move_amount = np.random.random() * delta
    h_new[i] -= move_amount
    h_new[j] += move_amount
    return project_to_sum_box(h_new, TARGET_SUM)

def apg_with_continuation(h0, max_time):
    """Main optimization routine"""
    start_time = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_from_corr(compute_corr_fft(h), n_points)
    
    # Configuration
    max_alpha = 40.0
    alpha_factor = 1.4
    apg_steps_per_alpha = 150
    eta_init = 0.1
    eta_min = 1e-6
    eta_max = 1.0
    
    # Initial APG parameters
    h_prev = h.copy()
    y = h.copy()
    eta = eta_init
    
    alpha = 0.5
    
    iteration = 0
    last_improvement = 0
    
    while alpha <= max_alpha and time.time() - start_time < max_time - 5:
        for _ in range(apg_steps_per_alpha):
            if time.time() - start_time > max_time - 5:
                break
                
            # Compute current surrogate
            f_curr, grad = lse_surrogate(h, alpha)
            
            # Nesterov acceleration
            beta = 0.9
            y = h + beta * (h - h_prev)
            
            # Backtracking line search
            h_tmp, eta = backtracking_line_search(y, grad, alpha, eta, f_curr)
            
            # Update
            h_prev = h.copy()
            h = h_tmp
            
            # Check for improvement in true C5
            curr_c5 = c5_from_corr(compute_corr_fft(h), n_points)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                last_improvement = iteration
            
            # Perturbation escape
            if iteration % 100 == 0 and iteration > last_improvement + 200:
                h_pert = random_perturbation(h)
                c5_pert = c5_from_corr(compute_corr_fft(h_pert), n_points)
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                    h = h_pert.copy()
                    last_improvement = iteration
            
            iteration += 1
            
            # Adaptive step size
            if iteration % 30 == 0:
                if time.time() - start_time > max_time - 5:
                    break
                eta = min(eta * 1.1, eta_max)
        
        # Increase alpha
        alpha *= alpha_factor
        if alpha > max_alpha:
            break
    
    # Sub-gradient polishing phase
    eta_sg = 0.08
    stall_count = 0
    for _ in range(1500):
        if time.time() - start_time > max_time - 5:
            break
            
        corr = compute_corr_fft(best_h)
        max_corr = np.max(corr)
        K_max = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        if len(K_max) == 0:
            break
            
        # Compute average gradient
        avg_grad = np.zeros_like(best_h)
        for k in K_max:
            shift = k - (n_points - 1)
            avg_grad += grad_shift(best_h, shift)
        avg_grad /= len(K_max)
        
        # Update with projected gradient step
        h_new = project_to_sum_box(best_h - eta_sg * avg_grad, TARGET_SUM)
        c5_new = c5_from_corr(compute_corr_fft(h_new), n_points)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall_count = 0
            # Increase eta if we make progress
            eta_sg = min(eta_sg * 1.05, 0.15)
        else:
            # Decrease eta if no progress
            eta_sg *= 0.5
            stall_count += 1
            if stall_count > 500:
                break
            
        # Random swap move every 200 steps
        if _ % 200 == 0:
            h_pert = random_perturbation(best_h, 0.015)
            c5_pert = c5_from_corr(compute_corr_fft(h_pert), n_points)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
    
    return best_h, best_c5
