from scipy.sparse import diags
import time

def proj_sum_box(v, target):
    """Project v onto box [0,1] with sum = target using binary search"""
    low, high = -1.0, 1.0
    for _ in range(30):
        lam = (low + high) / 2
        proj = np.clip(v + lam, 0, 1)
        if np.sum(proj) > target:
            high = lam
        else:
            low = lam
    return np.clip(v + (low + high)/2, 0, 1)

def corr_full(h):
    """Fast correlation using FFT - returns raw values, no dx scaling"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Compute max overlap with dx scaling"""
    corr = corr_full(h)
    return dx * np.max(corr)

def grad_shift(h, k):
    """Gradient of correlation term c_k(h) = sum h_i(1-h_{i+k})"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # h_i(1-h_{i+k}) gradient wrt h_i
        grad[:-k] += 1 - h[k:]
        # gradient wrt h_{i+k}
        grad[k:] -= h[:-k]
    else:
        # k < 0: h_i(1-h_{i+k}) gradient wrt h_i
        grad[-k:] += 1 - h[:k]
        # gradient wrt h_{i+k}
        grad[:k] -= h[-k:]
    return grad

def avg_sub_grad(h):
    """Average subgradient of max overlap"""
    corr = corr_full(h)
    max_val = np.max(corr)
    indices = np.where(corr >= max_val - 1e-12)[0]
    g = np.zeros(len(h))
    for k in indices:
        g += grad_shift(h, k)
    return g / len(indices)

def polish(h_start, n, dx, remaining_time):
    """Projected gradient descent polishing"""
    h = h_start.copy()
    h_prev = h_start.copy()
    step = 0.07
    beta = 0.9
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    
    start_time = time.time()
    
    for it in range(2000):
        if time.time() - start_time > remaining_time - 1:
            break
            
        corr = corr_full(h)
        c5 = dx * np.max(corr)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Subgradient
        Kmax = np.where(corr >= np.max(corr) - 1e-12)[0]
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        
        # Project
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new, dx)
        
        if c5_new < c5:
            step = min(step * 1.08, 0.2)
            h_prev = h
            h = h_new
        else:
            step *= 0.5
            
        # Swap move every 250 iterations
        if it % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            h_old = h.copy()
            h[i] = np.clip(h[i] - delta, 0, 1)
            h[j] = np.clip(h[j] + delta, 0, 1)
            h = proj_sum_box(h, n/2)
            c5_new = max_overlap(h, dx)
            if c5_new >= max_overlap(h_old, dx):
                h = h_old
    
    return best_h, best_c5

def shift_mask(k, n):
    """Create a sparse matrix representing shift k"""
    if k >= 0:
        return diags([1], [k], shape=(n, n), format='csr')
    else:
        return diags([1], [k], shape=(n, n), format='csr')

def build_sdp(n):
    """Build SDP problem for the Erdos overlap problem"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Build shift masks
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob

def solve_sdp(prob):
    """Solve the SDP with error handling"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL:
            return prob
    except:
        pass
    return None

def round_from_sdp(h_sdp, X_sdp, n):
    """Try multiple rounding strategies"""
    candidates = []
    
    # Strategy 1: diagonal rounding
    h1 = h_sdp.copy()
    candidates.append(h1)
    
    # Strategy 2: leading eigenvector rounding
    try:
        evals, evecs = np.linalg.eigh(X_sdp)
        lambda_max = max(0, evals[-1])
        u_max = evecs[:, -1]
        h2 = np.sqrt(lambda_max) * u_max
        h2 = np.abs(h2)  # Use absolute value to ensure positivity
        h2 = proj_sum_box(h2, n/2)
        candidates.append(h2)
    except:
        pass
    
    # Strategy 3: Gaussian randomization
    try:
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, n/2)
            candidates.append(h3)
    except:
        pass
    
    best_h = candidates[0]
    best_c5 = max_overlap(best_h, 2/n)
    
    for h in candidates[1:]:
        c5 = max_overlap(h, 2/n)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            
    return best_h
