import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.sparse import csr_matrix

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def corr_full(h):
    """Compute full correlation using scipy.signal"""
    return correlate(h, 1-h, mode="full")

def shift_mask(k, n):
    """Create shift mask matrix for shift k"""
    if k >= 0:
        data = np.ones(n-k)
        row = np.arange(n-k)
        col = np.arange(k, n)
    else:
        data = np.ones(n+k)
        row = np.arange(-k, n)
        col = np.arange(n+k)
    
    M = csr_matrix((data, (row, col)), shape=(n, n))
    return M

def build_sdp(n):
    """Build SDP problem for the given n"""
    h = cp.Variable(n)
    X = cp.Variable((n,n), PSD=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Overlap constraints for all shifts
    for k in range(-(n-1), n):
        M_k = shift_mask(k, n)
        constraints.append(cp.trace(M_k @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau

def solve_sdp(prob):
    """Solve SDP using SCS"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status == 'optimal':
            return True, prob.value
        else:
            return False, None
    except:
        return False, None

def round_from_sdp(h_sdp, X_sdp, n):
    """Try different rounding methods"""
    best_h = None

    # Method 1: Diagonal rounding
    h1 = h_sdp.copy()

    c5_1 = np.max(corr_full(h1)) * (2.0/n)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Method 2: Leading eigenvector
    try:
        w, v = np.linalg.eigh(X_sdp)
        if w[-1] > 0:
            h2 = np.sqrt(w[-1]) * np.abs(v[:, -1])
            h2 = proj_sum_box(h2, n/2)
            c5_2 = np.max(corr_full(h2)) * (2.0/n)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
    except:
        pass
    
    # Method 3: Gaussian randomization
    for _ in range(5):
        try:
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, n/2)
            c5_3 = np.max(corr_full(h3)) * (2.0/n)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
        except:
            continue
    
    return best_h, best_c5

def compute_gradient(h, k):
    """Compute gradient for shift k"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[:n-k] = 1 - h[k:n]
        grad[k:n] -= h[:n-k]
    else:
        grad[-k:] = 1 - h[:n+k]
        grad[:n+k] -= h[-k:]
    return grad

def polish_step(h_start, n, dx, budget_remaining):
    """Run gradient descent polishing"""
    h = h_start.copy()
    h_prev = h_start.copy()
    step = 0.07
    beta = 0.9
    best_h = h.copy()
    best_c5 = np.max(corr_full(h)) * dx
    
    start_time = time.time()
    
    for it in range(2000):
        if time.time() - start_time > budget_remaining - 1:
            break
            
        c = corr_full(h)
        Cmax = np.max(c)
        Kmax = np.where(np.abs(c - Cmax) < 1e-12)[0]
        
        # Compute average gradient
        g = np.zeros(n)
        for k in Kmax:
            g += compute_gradient(h, k - (n-1))
        g /= len(Kmax)
        
        # Nesterov-style extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = np.max(corr_full(h_new)) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.08, 0.2)
        else:
            step *= 0.5
            
        # Swap-move every 250 iterations
        if it % 250 == 0:
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            if delta > 0 and h_new[i] + delta > 1:
                delta = 1 - h_new[i]
            if delta < 0 and h_new[j] - delta > 1:
                delta = h_new[j] - 1
            if delta != 0:
                h_temp = h_new.copy()
                h_temp[i] += delta
                h_temp[j] -= delta
                h_temp = proj_sum_box(h_temp, n/2)
                c5_temp = np.max(corr_full(h_temp)) * dx
                if c5_temp < best_c5:
                    best_c5 = c5_temp
                    best_h = h_temp.copy()
        
        h_prev = h.copy()
        h = h_new.copy()
    
    return best_h, best_c5
