import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 using full correlation"""
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if abs(current_sum - target_sum) < 1e-12:
        return x
    
    # Binary search for lambda
    def objective(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-12:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b) / 2, 0, 1)
    return result

def round_to_feasible(r_opt, n, M, dx):
    """Rounding step: convert optimal autocorrelation to feasible h"""
    # Fourier-based rounding
    S = np.fft.fft(r_opt)
    A = np.sqrt(np.maximum(S, 0))  # Non-negative square root
    
    best_h = None
    best_c5 = float('inf')
    
    # Try 4 different phase vectors
    for seed in [0, 42, 123, 456]:
        np.random.seed(seed)
        phase = np.random.uniform(0, 2*np.pi, n)
        h_candidate = np.fft.ifft(A * np.exp(1j * phase)).real
        h_candidate = project_to_box_sum(h_candidate, M)
        
        c5 = evaluate_c5(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate
    
    return best_h, best_c5

def refine_solution(h, dx, max_iter=50, tol=1e-6):
    """Projected gradient ascent on the true objective"""
    h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Compute initial gradients for all correlations
    corr_full = np.correlate(h, 1 - h, mode="full")
    min_corr_idx = np.argmax(corr_full)
    min_corr_val = corr_full[min_corr_idx]
    
    for iter in range(max_iter):
        # Compute gradient for the minimal correlation term
        # Find all indices where correlation is near minimal (within tolerance)
        active_indices = np.where(abs(corr_full - min_corr_val) < 1e-8)[0]
        
        # Compute average gradient
        grad = np.zeros_like(h)
        for idx in active_indices:
            # Forward shift gradient
            if idx < len(corr_full) - len(h) + 1:
                forward_shift = np.concatenate([h[idx+1:], np.zeros(idx + 1)])
                grad += forward_shift
            
            # Backward shift gradient  
            if idx >= len(h) - 1:
                backward_shift = np.concatenate([np.zeros(len(h) - (idx - len(h) + 1)), h[:idx - len(h) + 1]])
                grad += backward_shift
        
        # Normalize gradient
        grad = grad / max(1e-12, np.linalg.norm(grad))
        
        # Gradient step
        step_size = 0.2 / np.sqrt(iter + 1)
        h_new = h + step_size * grad
        
        # Project back
        h_new = project_to_box_sum(h_new, len(h) / 2)
        
        # Update correlation and check improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            
            # Recompute correlations for next iteration
            corr_full = np.correlate(h, 1 - h, mode="full")
            min_corr_val = np.max(corr_full)
        else:
            # Early termination if improvement is small
            if iter > 10 and abs(best_c5 - new_c5) < tol:
                break
    
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 80
    dx = 2.0 / n
    M = n / 2
    
    # Create and solve the SDP
    r = cp.Variable(n)
    t = cp.Variable()
    
    # Construct Toeplitz matrix
    T = cp.Variable((n, n), symmetric=True)
    for i in range(n):
        for j in range(n):
            T[i,j] = r[abs(i-j)]
    
    # Constraints
    constraints = [
        T >> 0,  # PSD constraint
        cp.sum([(n-k) * r[k] for k in range(n)]) == M**2,  # Mass constraint
        r >= t,  # Lower bound on all autocorrelations
        r >= 0,  # Non-negativity
        r <= M,  # Box constraint
        r <= n - cp.arange(n)  # Box constraint
    ]
    
    # Solve SDP
    prob = cp.Problem(cp.Maximize(t), constraints)
    solver_time = 0
    
    # Try to solve with available solvers
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        solver_time = prob.solver_stats.solve_time
    except:
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
            solver_time = prob.solver_stats.solve_time
        except:
            # Fallback to simple heuristic
            h = np.full(n, 0.5)
            h = project_to_box_sum(h, M)
            c5 = evaluate_c5(h, dx)
            return (h.tolist(), float(c5), n)
    
    if prob.status != cp.OPTIMAL:
        # Fallback to heuristic
        h = np.full(n, 0.5)
        h = project_to_box_sum(h, M)
        c5 = evaluate_c5(h, dx)
        return (h.tolist(), float(c5), n)
    
    # Extract solution
    r_opt = r.value
    t_opt = t.value
    
    # Rounding to get feasible h
    h, c5_rounded = round_to_feasible(r_opt, n, M, dx)
    
    # Refinement with projected gradient descent
    remaining_time = budget_s - (time.time() - start_time) - 10  # Leave 10s margin
    
    if remaining_time > 10:
        max_iter = min(50, max(10, int(remaining_time / 2)))
        h_refined, c5_final = refine_solution(h, dx, max_iter=max_iter)
        c5 = c5_final
        h = h_refined
    else:
        c5 = c5_rounded
    
    return (h.tolist(), float(c5), n)
