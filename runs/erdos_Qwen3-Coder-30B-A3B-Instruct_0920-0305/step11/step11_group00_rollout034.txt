import numpy as np
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse import diags
import cvxpy as cp
import time
import math

def proj_sum_box(v, target):
    """Project vector v onto box constraints [0,1] with sum = target using binary search."""
    low, high = -1.0, 1.0
    while True:
        mid = (low + high) / 2
        projected = np.clip(v + mid, 0, 1)
        sum_val = np.sum(projected)
        if abs(sum_val - target) < 1e-10:
            return projected
        if sum_val > target:
            high = mid
        else:
            low = mid

def corr_full(h):
    """Compute all correlations via FFT."""
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h_rev, b, mode="full")
    return corr

def max_overlap(h, dx):
    """Compute max overlap with dx scaling."""
    corr = corr_full(h)
    return dx * np.max(corr)

def build_shift_mask(k, n):
    """Build sparse shift mask for shift k."""
    if k >= 0:
        data = np.ones(n-k)
        row = np.arange(n-k)
        col = np.arange(k, n)
    else:
        data = np.ones(n+k)
        row = np.arange(-k, n)
        col = np.arange(n+k)
    return csr_matrix((data, (row, col)), shape=(n, n))

def build_sdp(n):
    """Build SDP problem."""
    # Precompute shift masks efficiently
    M_masks = []
    for k in range(-(n-1), n):
        M_masks.append(build_shift_mask(k, n))
    
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Overlap constraints
    for k, M in enumerate(M_masks):
        # From paper: tr(M_k * X) >= n/2 * (1 - tau)
        constraints.append(cp.trace(M @ X) >= (n/2) * (1 - tau))
    
    # Objective
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau, M_masks

def solve_sdp(prob, solver='SCS'):
    """Solve SDP with safeguards."""
    try:
        prob.solve(solver=solver, max_iters=5000, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return True
        return False
    except:
        return False

def round_from_sdp(h_sdp, X_sdp, n):
    """Try different rounding strategies."""
    best_h = None
    best_c5 = float('inf')
    
    # Strategy 1: Diagonal
    h1 = h_sdp.copy()
    h1 = proj_sum_box(h1, n/2)
    c5_1 = max_overlap(h1, 2.0/n)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
    
    # Strategy 2: Leading eigenvector
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_sdp)
        if len(eigenvals) > 0:
            idx = np.argmax(eigenvals)
            eigval = max(eigenvals[idx], 0)  # Ensure non-negative
            u = eigenvecs[:, idx]
            h2 = np.sqrt(eigval) * u
            h2 = proj_sum_box(h2, n/2)
            c5_2 = max_overlap(h2, 2.0/n)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
    except:
        pass
    
    # Strategy 3: Gaussian randomization (5 draws)
    for _ in range(5):
        try:
            z = np.random.multivariate_normal(np.zeros(n), X_sdp, 1)[0]
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, n/2)
            c5_3 = max_overlap(h3, 2.0/n)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
        except:
            continue
    
    return best_h

def grad_shift(h, k):
    """Gradient of correlation term with respect to h for shift k."""
    n = len(h)
    grad = np.zeros(n)
    
    if k >= 0:
        # h_i * (1 - h_{i+k}) for i = 0..n-k-1
        for i in range(n - k):
            grad[i] += (1 - h[i + k])
        for i in range(k, n):
            grad[i] -= h[i - k]
    else:
        # h_i * (1 - h_{i+k}) for i = -k..n-1
        for i in range(-k, n):
            grad[i] += (1 - h[i + k])
        for i in range(n + k):
            grad[i] -= h[i - k]
    
    return grad

def polish(h_start, n, dx, budget_s):
    """Polish using projected gradient descent."""
    np.random.seed(42)
    h = h_start.copy()
    h_prev = h.copy()
    beta = 0.9
    step = 0.07
    c5_best = max_overlap(h, dx)
    it = 0
    max_iters = 2000
    
    while it < max_iters:
        it += 1
        corr = corr_full(h)
        Cmax = np.max(corr)
        Kmax = np.where(np.abs(corr - Cmax) < 1e-12)[0]
        
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k - (n-1))  # Adjust k to match indexing
        g /= len(Kmax)
        
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new, dx)
        if c5_new < c5_best:
            step = min(step * 1.08, 0.2)
            c5_best = c5_new
            h_prev = h.copy()
            h = h_new.copy()
        else:
            step *= 0.5
            
        # Swap moves every 250 iterations or if stuck
        if it % 250 == 0 or (it % 200 == 0 and abs(c5_best - max_overlap(h, dx)) < 1e-12):
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            if h[i] + delta >= 0 and h[i] + delta <= 1 and h[j] - delta >= 0 and h[j] - delta <= 1:
                h[i] += delta
                h[j] -= delta
                h = proj_sum_box(h, n/2)
                c5_new = max_overlap(h, dx)
                if c5_new < c5_best:
                    c5_best = c5_new
                
        if time.time() - start_time > budget_s - 10:
            break
    
    return h, c5_best

def run(seed=42, budget_s=600, **kwargs):
    global start_time
    start_time = time.time()
    np.random.seed(seed)
    
    best_c5 = float('inf')
    best_h = None
    n_points = 120
    dx = 2.0 / n_points
    
    # Try SDP + polish
    prob, h_var, X_var, tau_var, M_masks = build_sdp(n_points)
    
    success = solve_sdp(prob)
    
    if success:
        try:
            h_sdp = h_var.value
            X_sdp = X_var.value
            tau_opt = tau_var.value
            
            # Rounding
            h_seed = round_from_sdp(h_sdp, X_sdp, n_points)
            
            # Polish
            if h_seed is not None:
                h_opt, c5_opt = polish(h_seed, n_points, dx, budget_s)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                
        except Exception as e:
            pass
    
    # Try with different n if time permits
    if time.time() - start_time < budget_s - 10 and n_points == 120:
        n_points = 140
        dx = 2.0 / n_points
        try:
            prob, h_var, X_var, tau_var, M_masks = build_sdp(n_points)
            success = solve_sdp(prob)
            if success:
                h_sdp = h_var.value
                X_sdp = X_var.value
                h_seed = round_from_sdp(h_sdp, X_sdp, n_points)
                if h_seed is not None:
                    h_opt, c5_opt = polish(h_seed, n_points, dx, budget_s)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt
        except Exception as e:
            pass
    
    # Try random start if empty
    if best_h is None:
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = proj_sum_box(h_rand, n_points / 2)
        h_opt, c5_opt = polish(h_rand, n_points, dx, budget_s)
        best_h = h_opt
        best_c5 = c5_opt
    
    return (best_h.tolist(), float(best_c5), n_points)
