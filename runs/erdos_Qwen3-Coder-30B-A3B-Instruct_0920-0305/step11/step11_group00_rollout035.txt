import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time
from scipy.sparse import diags
from scipy.sparse import csr_matrix

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def shift_mask(k, n):
    """Create a sparse shift mask for shift k"""
    # Create a sparse matrix representing shift k
    if k >= 0:
        data = np.ones(n - k)
        row = np.arange(n - k)
        col = np.arange(k, n)
    else:
        data = np.ones(n + k)
        row = np.arange(-k, n)
        col = np.arange(n + k)
    
    return csr_matrix((data, (row, col)), shape=(n, n))

def max_overlap(h):
    """Calculate the maximum overlap given a step function h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr)

def corr_full(h):
    """Compute full correlation vector using fft for speed"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # j+k is in valid range but j is not if j < k
        if k < n:
            # h_{j+k} for j in [0, n-k-1]
            grad[:n-k] += 1 - h[k:n]  
            # h_{j-k} for j in [k, n-1]
            if k > 0:
                grad[k:] -= h[:n-k]
    else:
        # k < 0
        k = -k  # make it positive
        # h_{j+k} for j in [k, n-1]
        if k < n:
            grad[k:] += 1 - h[:n-k]
        # h_{j-k} for j in [0, n-k-1]
        if k > 0:
            grad[:n-k] -= h[k:]
    return grad

def find_active_shifts(corr):
    """Find shifts that attain the maximum correlation"""
    max_val = np.max(corr)
    return np.where(np.abs(corr - max_val) < 1e-12)[0]

def project_gradient(h, g, step_size):
    """Perform a projected gradient step"""
    # Try to move along gradient
    h_new = h - step_size * g
    
    # Project onto box constraints [0, 1]
    h_new = np.clip(h_new, 0, 1)
    
    # Project onto sum constraint
    h_new = project_to_sum_box(h_new, len(h)/2)
    
    return h_new

def polish_step(h_seed, n, dx, max_iter=1000, time_budget=500):
    """Polish using projected gradient descent"""
    h = h_seed.copy()
    step_size = 0.07
    beta = 0.9
    h_prev = h.copy()
    best_c5 = max_overlap(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > time_budget - 5:
            break
            
        corr = corr_full(h)
        c5 = dx * np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find active shifts
        active_shifts = find_active_shifts(corr)
        
        # Compute average gradient
        avg_grad = np.zeros(n)
        for k in active_shifts:
            avg_grad += grad_shift(h, k - (n-1))
        avg_grad /= len(active_shifts)
        
        # Nesterov-like step
        y = h + beta * (h - h_prev)
        
        # Gradient step
        h_new = y - step_size * avg_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum_box(h_new, n/2)
        
        new_c5 = dx * np.max(corr_full(h_new))
        
        if new_c5 < c5:
            # Accept the step and increase learning rate
            step_size = min(step_size * 1.08, 0.2)
            h_prev = h
            h = h_new
        else:
            # Backtrack
            step_size *= 0.5
            
        # Occasionally perform a swap move to escape plateaus
        if it % 250 == 0:
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            h_try = h.copy()
            h_try[i] = np.clip(h_try[i] - delta, 0, 1)
            h_try[j] = np.clip(h_try[j] + delta, 0, 1)
            h_try = project_to_sum_box(h_try, n/2)
            try_c5 = dx * np.max(corr_full(h_try))
            if try_c5 < c5:
                h = h_try
                
    return best_h, best_c5
