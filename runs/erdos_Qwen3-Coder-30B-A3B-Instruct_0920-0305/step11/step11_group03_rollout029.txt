import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 80
    dx = 2.0 / n_points
    M = n_points / 2
    
    # Solve SDP relaxation
    # Variables: r = autocorrelation vector, t = lower bound on min r_k
    r = cp.Variable(n_points)
    t = cp.Variable()
    
    # Build Toeplitz matrix
    T = cp.Variable((n_points, n_points), symmetric=True)
    for i in range(n_points):
        for j in range(n_points):
            T[i,j].value = r[abs(i-j)].value if i < n_points and j < n_points else 0
    
    # Constraints
    constraints = [
        cp.trace(T) == M,  # t is an entry of T
    ]
    
    # Add linear constraint sum_k (n-k) r_k = M^2
    linear_constraint = cp.sum([(n_points - k) * r[k] for k in range(n_points)]) == M**2
    
    # Add PSD constraint
    constraints.append(T >> 0)
    
    # Add bound constraints
    for k in range(n_points):
        constraints.append(r[k] >= t)
        constraints.append(r[k] >= 0)
        constraints.append(r[k] <= min(M, n_points - k))
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints + [linear_constraint])
    
    # Time allocation
    max_sdp_time = 250
    remaining_time = max_sdp_time
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, time_limit=remaining_time)
        # Check if solution is found
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            r_opt = r.value
            t_opt = t.value
        else:
            r_opt = None
            t_opt = None
    except Exception as e:
        r_opt = None
        t_opt = None
    
    # Fallback: use uniform vector if SDP fails
    if r_opt is None or t_opt is None:
        h_best = np.full(n_points, 0.5)
        h_best = project_to_sum_box(h_best, M)
        c5 = compute_c5_direct(h_best, dx)
        return (h_best.tolist(), float(c5), n_points)
    
    # Spectral factorization
    def spectral_round(r_vec):
        # Compute DFT
        S = np.fft.fft(r_vec)
        # Take square root
        A = np.sqrt(np.maximum(S.real, 0))
        # Try four phase choices
        best_h = None
        best_c5 = float('inf')
        for phase_seed in range(4):
            np.random.seed(phase_seed)
            phi = np.random.uniform(0, 2*np.pi, n_points)
            h_candidate = np.fft.ifft(A * np.exp(1j*phi)).real
            # Project to feasible set
            h_projected = project_to_sum_box(h_candidate, M)
            # Compute actual C5
            actual_c5 = compute_c5_direct(h_projected, dx)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_projected
        return best_h, best_c5

    # Project to feasible set with box constraints
    def project_to_sum_box(x, target_sum):
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        a, b = -1.0, 1.0
        while f(a) * f(b) > 0:
            a -= 0.1
            b += 0.1
        
        tol = 1e-10
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < tol:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    # Compute actual C5 value
    def compute_c5_direct(h, dx):
        corr = correlate(h, 1 - h, mode="full")
        return dx * np.max(corr)

    # Rounded solution
    h_rounded, c5_rounded = spectral_round(r_opt)

    # Local refinement with projected gradient ascent
    def project_to_box(h):
        return np.clip(h, 0, 1)
    
    def compute_grad_single(h, k):
        grad = np.zeros_like(h)
        if k >= 0:
            grad[k:] += h[:-k]
        if k <= 0:
            grad[:k] += h[-k:]
        return grad
    
    def compute_grad(h):
        # Find minimum value
        corr = correlate(h, 1 - h, mode="full")
        min_idx = np.argmax(corr)
        min_val = corr[min_idx]
        
        # Collect all gradients for active shifts
        active_shifts = []
        for i in range(len(corr)):
            if abs(corr[i] - min_val) < 1e-8:
                active_shifts.append(i - (len(h) - 1))
        
        grad = np.zeros_like(h)
        for k in active_shifts:
            grad += compute_grad_single(h, k)
        
        return grad / len(active_shifts)
    
    # Gradient ascent
    h_refined = h_rounded.copy()
    base_grad_iter = 30
    remaining_time = budget_s - (time.time() - start_time)
    
    max_iter = min(base_grad_iter, int(remaining_time / 3.0))
    
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute negative gradient (since we want to maximize overlap, minimize 1-C5)
        grad = compute_grad(h_refined)
        h_refined += 0.2 / np.sqrt(iter_idx + 1) * grad
        h_refined = project_to_box(h_refined)
        h_refined = project_to_sum_box(h_refined, M)
    
    # Final C5 evaluation
    c5_final = compute_c5_direct(h_refined, dx)
    
    return (h_refined.tolist(), float(c5_final), n_points)
