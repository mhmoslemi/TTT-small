import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_autocorr(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_corr(h):
    """Compute correlation with 1-h using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def evaluate_min_r(h):
    """Evaluate the minimum autocorrelation"""
    auto_corr = compute_autocorr(h)
    # Only consider positive shifts (first half of correlation)
    # Index 0 is shift 0, indices 1 to n-1 are positive shifts
    positive_shifts = auto_corr[n_points-1:]  # indices from n-1 to 2*n-2
    return np.min(positive_shifts) * dx

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def gradient_step(h, min_indices, step_size):
    """Perform gradient step to increase minimum autocorrelation"""
    # Initialize gradient
    grad = np.zeros_like(h)
    
    # Compute gradient from all minimum correlation shifts
    for k in min_indices:
        # Gradient contribution from shift k for the term h_i * h_{i+k}
        # For index i, derivative w.r.t. h_i is h_{i+k} 
        # But we need to handle boundaries carefully
        if k == 0:
            # For shift 0, the gradient is just sum of h including itself
            grad += 2 * h
        else:
            # Left shift contribution h_{i+k} for i in [0, n-k)
            if k < len(h):
                grad[:len(h)-k] += h[k:]
                # Right shift contribution h_{i-k} for i in [k, n)  
                grad[k:] += h[:len(h)-k]
    
    # Apply step and update h
    h_new = h + step_size * grad
    
    # Project back to constraints
    h_new = project_to_sum_box(h_new, TARGET_SUM)
    return h_new

                    best_h = optimized_h.copy()
        except:
            pass
    
    return (best_h.tolist(), float(best_c5), n_points)
