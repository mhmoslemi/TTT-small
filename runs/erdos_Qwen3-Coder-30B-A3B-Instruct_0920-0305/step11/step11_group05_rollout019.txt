def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(h, tau=0.05):
    """Smooth approximation of max using log-sum-exp"""
    corr = compute_corr_fft(h)
    corr = corr * dx
    # Use log-sum-exp trick for numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp((corr - max_corr) / tau)
    return tau * np.log(np.sum(exp_corr)) + max_corr

def smooth_max_gradient(h, tau=0.05):
    """Compute gradient of smooth max using FFT"""
    h_rev = h[::-1]
    one_minus_h = 1 - h
    L = 2 * len(h) - 1
    
    # Compute correlations
    fft_h = np.fft.fft(h_rev, L)
    fft_1mh = np.fft.fft(one_minus_h, L)
    corr = np.fft.ifft(fft_h * fft_1mh).real
    
    # Compute weights using soft max with temperature tau
    corr_scaled = corr / tau
    max_corr = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - max_corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute gradient components
    fft_w = np.fft.fft(weights, L)
    grad1 = np.fft.ifft(fft_w * fft_1mh).real
    fft_h = np.fft.fft(h, L)
    grad2 = np.fft.ifft(fft_w * fft_h).real
    
    gradient = dx * (grad1 - grad2)
    return gradient

def accelerated_projected_gradient(h0, tau, max_iter=3000, eta=1e-4):
    """Accelerated projected gradient with backtracking line search"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    
    for k in range(max_iter):
        # Compute gradient
        grad = smooth_max_gradient(h, tau)
        
        # Backtracking line search
        alpha = 1.0
        while True:
            h_candidate = project_to_sum_box(h - alpha * grad, TARGET_SUM)
            if smooth_max(h_candidate, tau) <= smooth_max(h, tau) - eta * alpha * np.linalg.norm(grad)**2:
                break
            alpha *= 0.5
            
        h_new = h_candidate
        t_new = (1 + math.sqrt(1 + 4 * t**2)) / 2
        y_new = h_new + ((t - 1) / t_new) * (h_new - h)
        
        h = h_new
        y = y_new
        t = t_new
        
        if k % 100 == 0:
            # Check early stopping criteria
            if np.linalg.norm(grad) < 1e-6:
                break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate restart points
    initials = generate_initials()
    
    # Continuation schedule
    taus = [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005, 0.0001]

        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_h = h0.copy()
        
        # Continuation loop
        for tau in taus:
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Use fewer iterations for later, smaller temperatures
            max_iter = max(500, int(3000 * tau / 0.05))
            try:
                current_h = accelerated_projected_gradient(current_h, tau, max_iter)
            except:
                pass
            
            # Check early stopping via C5 value
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5

        # Check final C5 value for current initial
        if best_h is None or evaluate_c5(current_h) < best_c5:
            best_c5 = evaluate_c5(current_h)
            best_h = current_h.copy()
    
    # Final refinement with pairwise exchange
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        current_h = best_h.copy()

            h_new = current_h.copy()
            h_new[i] -= delta
            h_new[j] += delta

            h_new = project_to_sum_box(h_new, TARGET_SUM)

                current_h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()

                # Accept with probability
                prob = np.exp(-(new_c5 - old_c5) / t_final)

                    current_h = h_new
                    if new_c5 < best_c5:
                        best_c5 = new_c5

            t_final *= 0.9995  # Cool down temperature

        final_c5 = evaluate_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback to initial
        best_h = np.array(initial_h_values)

        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
