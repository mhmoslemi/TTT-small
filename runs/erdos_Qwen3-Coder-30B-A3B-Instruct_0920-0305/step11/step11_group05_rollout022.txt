def project_onto_sum_box(v, target_sum):
    """Project vector onto box [0,1] with sum = target_sum using water-filling."""
    sorted_indices = np.argsort(-v)  # descending order
    v_sorted = v[sorted_indices]
    cumsum = np.cumsum(v_sorted)
    n = len(v)
    
    # Find the threshold
    threshold_idx = -1
    for i in range(n):
        if cumsum[i] <= target_sum:
            threshold_idx = i
        else:
            break
    
    if threshold_idx == -1:
        # All entries exceed target_sum, so use uniform
        proj = np.full(n, target_sum / n)
    else:
        # Use water-filling approach
        total_excess = target_sum - cumsum[threshold_idx]
        threshold = v_sorted[threshold_idx] - total_excess / (threshold_idx + 1)
        proj = np.clip(v_sorted, 0, threshold)
        # Adjust for sum
        current_sum = np.sum(proj)
        diff = target_sum - current_sum
        if diff > 0:
            proj[proj < threshold] = threshold
            proj[proj < 1] += diff / np.sum(proj < 1)
            proj = np.clip(proj, 0, 1)
        elif diff < 0:
            proj[proj > threshold] = threshold
            proj[proj > 0] += diff / np.sum(proj > 0)
            proj = np.clip(proj, 0, 1)
                
    # Reorder back to original indices
    proj_full = np.empty_like(v)
    proj_full[sorted_indices] = proj
    return proj_full

def corr_fft(h):
    """Compute full cross-correlation using FFT."""
    n = len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr[n - 1:]  # Take positive lags

def evaluate_c5(h):
    """Evaluate C5 using FFT."""
    corr = corr_fft(h)
    return np.max(corr) * 2 / len(h)

def lp_step(y):
    """Solve the linear program with frozen vector y, return optimal x and t."""
    n = len(y)
    # Variables
    x = Variable(n)
    t = Variable(1)
    
    # Build constraints
    constraints = []
    constraints.append(sum(x) == 1.0)  # Sum constraint
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # For each shift s, add constraint sum_i x_i * y_{i+s} >= t
    for s in range(-n+1, n):
        if s >= 0:
            shifted_y = y[s:] + [0]*s
        else:
            shifted_y = [0]*(-s) + y[:s]
        shifted_y = shifted_y[:n]  # Truncate to length n
        constraints.append(sum(x[i] * shifted_y[i] for i in range(n)) >= t)
    
    # Objective
    prob = Problem(Minimize(-t), constraints)
    try:
        prob.solve(solver='ECOS', verbose=False, max_iters=5000)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(x.value).flatten(), float(t.value)
        else:
            return None, None
    except:
        return None, None
