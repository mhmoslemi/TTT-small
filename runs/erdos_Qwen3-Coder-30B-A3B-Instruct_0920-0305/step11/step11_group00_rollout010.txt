import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n // 2
    # Linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        # Even: symmetric around center
        h = np.concatenate([first, first[::-1]])
    else:
        # Odd: center component is 0  
        h = np.concatenate([first, [0], first[::-1]])
    return h

def random_feasible_vector(n, target_sum):
    """Generate a random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h

def grad_shift(h, s):
    """Compute gradient for shift s"""
    n = len(h)
    # Pad array with zeros on both sides
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    
    # Compute indices for shifted elements
    base = np.arange(n) + (n-1)
    
    # Compute gradient for shift s
    # term1 = 1 - h_{j+s} 
    term1 = 1 - padded[base + s]
    # term2 = h_{j-s} 
    term2 = padded[base - s]
    
    return term1 - term2

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using smooth approximation"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    n = len(h)
    
    # Parameters for smooth optimization
    alphas = np.linspace(0.5, 30, 15)
    step = 0.5
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for _ in range(200):
            if time.time() - start_time > time_limit:
                break
            
            # Compute correlation
            c = compute_corr(h)
            
            # Compute weights with numerical stability
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute weighted gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n-1)
                shift_grad = grad_shift(h, s)
                grad += weight * shift_grad
            
            # Adaptive step size
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
            
            # Adjust step size based on improvement
            if step_cur < step * 0.1:
                step *= 0.9
            else:
                step = min(step * 1.05, 1.0)
                
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using subgradient method"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    n = len(h)
    
    step = 0.5
    max_iter = 1000
    
    for iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute correlation
        c = compute_corr(h)
        
        # Find indices with maximum correlation
        max_val = np.max(c)
        max_indices = np.where(np.abs(c - max_val) < 1e-10)[0]
        
        # Compute average gradient for all max indices
        grad = np.zeros_like(h)
        for idx in max_indices:
            s = idx - (n-1)
            shift_grad = grad_shift(h, s)
            grad += shift_grad
            
        grad /= len(max_indices)
        
        # Adaptive step size
        step_cur = step
        for _ in range(5):
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
                break
            else:
                step_cur *= 0.5
        
        # Adjust step size based on improvement
        if step_cur < step * 0.1:
            step *= 0.9
        else:
            step = min(step * 1.05, 1.0)
            
    return best_h, best_val
