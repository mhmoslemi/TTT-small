import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

                    optimized_c5 = evaluate_c5(optimized_h)
                    
                    if optimized_c5 < best_c5:
                        best_c5 = optimized_c5
                        best_h = optimized_h.copy()
            except:
                pass
            
        # Secondary: local optimization on clustered subset
        elif iteration % 4 == 0:
            # Select indices in clusters for targeted optimization
            cluster_size = max(15, n_points // 10)
            indices = []
            for i in range(0, n_points, cluster_size):

                cluster_indices = list(range(i, end_idx))

        if iteration % 30 == 0 and time.time() - start_time < budget_s * 0.9:

            h_new = np.zeros(n_points)
