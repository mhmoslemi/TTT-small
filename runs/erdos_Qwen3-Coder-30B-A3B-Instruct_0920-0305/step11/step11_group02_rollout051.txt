import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 300
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Use a more sophisticated construction: interpolate from previous best solution
    # and improve with gradient-based optimization
    if 'initial_h_values' in globals() and len(initial_h_values) >= 120:
        # Interpolate to larger resolution
        old_h = np.array(initial_h_values)
        new_h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(old_h)), old_h)
        h0 = project_to_sum_box(new_h, TARGET_SUM)
    else:
        # Random initialization with sum constraint
        h0 = np.random.random(n_points)
        h0 = project_to_sum_box(h0, TARGET_SUM)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Gradient-based optimization using projected gradient descent
    max_iterations = 10000
    iteration = 0
    
    # For gradient computation, we'll use finite differences for stability
    eps = 1e-6
    learning_rate = 0.05
    
    while time.time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        # Compute correlation
        corr = compute_corr_fft(current_h)
        # Find the worst shift
        max_corr_idx = np.argmax(corr)
        # Actual shift is symmetric around center
        shift = max_corr_idx - (n_points - 1)
        
        # Special case: don't optimize at shift 0 (should be 0 anyway)
        if shift == 0:
            shift = np.argmax(corr[1:]) + 1  # Skip index 0
            if shift >= n_points:
                shift = 1
                
        # Compute gradient for the worst shift
        # This approximates: partial derivative of sum_i h[i] * (1 - h[i+shift]) with respect to h[i]
        # = (1 - h[i+shift]) - h[i-shift] (if indices exist)
        gradient = np.zeros(n_points)
        if shift > 0:
            # Handle positive shift (h_{i+shift} term)
            gradient[shift:] += current_h[:-shift]
            # Handle negative shift (h_{i-shift} term)
            gradient[:-shift] += current_h[shift:]
        else:
            shift_abs = -shift
            # Handle negative shift (h_{i-shift} term)
            gradient[shift_abs:] += current_h[:-shift_abs]
            # Handle positive shift (h_{i+shift} term)
            gradient[:-shift_abs] += current_h[shift_abs:]
        
        # Add gradient of constant term
        gradient += 1.0
        
        # Normalize gradient to have reasonable step size
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 0:
            gradient = gradient / grad_norm
        
        # Try step with current gradient
        h_new = current_h - learning_rate * gradient
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            current_h = h_new.copy()
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Accept worse solution with some probability
            if np.random.random() < 0.01:
                current_h = h_new.copy()
                current_c5 = new_c5
                
        # Reduce learning rate over time
        if iteration % 100 == 0:
            learning_rate *= 0.98
            
        # Occasionally do more comprehensive local search
        if iteration % 500 == 0:
            # Do some random local improvements
            for _ in range(10):
                i = np.random.randint(0, n_points)
                delta = np.random.random() * 0.2 - 0.1
                
                # Try small adjustments
                temp_h = current_h.copy()
                temp_h[i] += delta
                
                if temp_h[i] < 0 or temp_h[i] > 1:
                    continue
                    
                temp_h = project_to_sum_box(temp_h, TARGET_SUM)
                temp_c5 = evaluate_c5(temp_h)
                
                if temp_c5 < current_c5:
                    current_h = temp_h.copy()
                    current_c5 = temp_c5
                    
                    if temp_c5 < best_c5:
                        best_c5 = temp_c5
                        best_h = current_h.copy()

    # Final polishing with more precise optimization
    if best_h is not None and time.time() - start_time < budget_s * 0.3:
        # Use scipy minimize to further optimize
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h)
        
        def constraint_eq(h):
            return np.sum(h) - TARGET_SUM
        
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            res = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-15, 'gtol': 1e-15}
            )
            if res.success:
                optimized_h = res.x
                optimized_h = project_to_sum_box(optimized_h, TARGET_SUM)
                optimized_c5 = evaluate_c5(optimized_h)
                
                if optimized_c5 < best_c5:
                    best_c5 = optimized_c5
                    best_h = optimized_h.copy()
        except:
            pass
    
    if best_h is None:
        best_h = h0.copy()
        
    return (best_h.tolist(), float(best_c5), n_points)
