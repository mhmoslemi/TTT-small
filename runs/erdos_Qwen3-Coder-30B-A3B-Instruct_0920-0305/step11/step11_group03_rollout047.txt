import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelations_fft(h):
    """Compute full autocorrelation using FFT"""
    # Pad with zeros to get sufficient length
    n = len(h)
    pad_len = 2 * n - 1
    h_padded = np.pad(h, (0, n-1), 'constant', constant_values=0)
    
    # Compute FFT and correlation
    H = np.fft.fft(h_padded)
    corr = np.fft.ifft(H * np.conj(H)).real
    
    # Extract the valid part (non-negative lags)
    return corr[n-1:]

def compute_softmin_weights(R_values, beta):
    """Compute soft-min weights"""
    exp_R = np.exp(-beta * R_values)
    return exp_R / np.sum(exp_R)

def compute_gradient_softmin(h, R_values, beta):
    """Compute gradient of softmin function"""
    # Compute weights
    w = compute_softmin_weights(R_values, beta)
    
    # Compute gradient via convolution trick
    n = len(h)
    w_rev = w[::-1]
    
    # Forward shift part
    forward_shift = np.convolve(w, h, mode='full')[n-1:2*n-1]
    
    # Backward shift part
    backward_shift = np.convolve(w_rev, h, mode='full')[0:n]
    
    # Combine to get full gradient
    grad = -(forward_shift + backward_shift)
    
    return grad

def compute_c5_exact(h):
    """Compute exact C5 using full correlation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)
