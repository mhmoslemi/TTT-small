import numpy as np
import time
from scipy.signal import correlate

# Global constants
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    lam_low = -np.max(x) - 1.0
    lam_high = 1.0 - np.min(x)
    
    # Binary search
    for _ in range(30):
        lam = (lam_low + lam_high) / 2
        if f(lam) > 0:
            lam_low = lam
        else:
            lam_high = lam
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_gradient_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    # Create padded version of h with zeros on both ends
    pad_h = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base_indices = np.arange(n) + (n - 1)
    
    grad_k = (1 - pad_h[base_indices + k]) - pad_h[base_indices - k]
    return grad_k

def compute_smooth_max(h, alpha):
    """Compute smooth max surrogate"""
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    log_sum_exp = max_corr + np.log(np.sum(exp_terms)) / alpha
    return log_sum_exp

def compute_weights(corr, alpha):
    """Compute softmax weights for smooth max"""
    max_corr = np.max(corr)
    exp_terms = np.exp(alpha * (corr - max_corr))
    weights = exp_terms / np.sum(exp_terms)
    return weights

def accelerated_projected_gradient(h, alpha, max_iter=150, eta_init=0.1):
    """APG with line search on smooth max"""
    h_prev = h.copy()
    y = h.copy()
    eta = eta_init
    beta = 0.9
    
    for i in range(max_iter):
        # Compute current correlation
        corr = compute_correlation(y)
        
        # Compute weights
        weights = compute_weights(corr, alpha)
        
        # Compute average gradient
        grad = np.zeros_like(h)
        for k in range(-n_points+1, n_points):
            grad_k = compute_gradient_shift(h, k)
            grad += weights[k + n_points - 1] * grad_k
        
        # Backtracking line search
        h_tmp = y - eta * grad
        h_new = project_to_sum_box(h_tmp, TARGET_SUM)
        
        # Check if step improved surrogate
        new_alpha = compute_smooth_max(h_new, alpha)
        old_alpha = compute_smooth_max(y, alpha)
        
        if new_alpha < old_alpha:
            h = h_new
            y = h + beta * (h - h_prev)
            h_prev = h_new
            eta = min(eta * 1.1, 1.0)  # Increase step size
        else:
            eta = max(eta * 0.5, 1e-6)  # Reduce step size
            y = h + beta * (h - h_prev)
    
    return h

def subgradient_polishing(h, max_iter=1500):
    """Subgradient polishing for exact max"""
    eta = 0.08
    for i in range(max_iter):
        corr = compute_correlation(h)
        max_val = np.max(corr)
        k_indices = np.where(np.abs(corr - max_val) < 1e-12)[0]
        
        # Compute average gradient
        grad = np.zeros_like(h)
        for k in k_indices:
            grad_k = compute_gradient_shift(h, k - n_points + 1)
            grad += grad_k
        
        grad = grad / len(k_indices)
        
        # Update and project
        h_tmp = h - eta * grad
        h_new = project_to_sum_box(h_tmp, TARGET_SUM)
        
        new_corr = compute_correlation(h_new)
        new_max = np.max(new_corr)
        old_max = np.max(corr)
        
        if new_max < old_max:
            h = h_new
            eta = min(eta * 1.05, 0.15)
        else:
            eta = max(eta * 0.5, 1e-6)
            
        # Random perturbation escape
        if i % 200 == 0:
            i1, i2 = np.random.choice(n_points, 2, replace=False)

                h[i1] -= delta
                h[i2] += delta
                h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def triangular_start():
    """Create triangular starting point"""
    h = np.zeros(n_points)
    mid = n_points // 2
    for i in range(mid):
        h[i] = 1 - i / mid
        h[n_points - 1 - i] = h[i]
    return h
