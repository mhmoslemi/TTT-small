import numpy as np
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Use analytic bounds for the initial interval
    lam_low = -np.max(x) - 1.0
    lam_high = 1.0 - np.min(x)
    
    # Safety bounds
    while f(lam_low) * f(lam_high) > 0:
        lam_low -= 0.1
        lam_high += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (lam_low + lam_high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = m
        else:
            lam_low = m
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def grad_shift(h, k):
    """Compute gradient of shift k for correlation"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def solve_continuation(h0, max_time):
    """Main optimization loop with continuation"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # For APG, track previous iterate
    h_prev = h.copy()
    y = h.copy()
    beta = 0.9
    
    # Continuation parameters
    alpha = 0.5
    alpha_increase = 1.4
    max_alpha = 40.0
    
    # APG parameters
    eta = 0.05
    max_apg_iter = 150
    apg_step = 0
    
    t_start = time.time()
    
    while alpha <= max_alpha and time.time() - t_start < 0.95 * max_time:
        apg_iter = 0
        last_improvement = 0
        
        while apg_iter < max_apg_iter and time.time() - t_start < 0.95 * max_time:
            if apg_iter % 30 == 0 and apg_iter > 0:
                # Check for stagnation
                current_c5 = evaluate_c5(h)
                if current_c5 >= best_c5 - 1e-12:
                    last_improvement += 1
                    if last_improvement > 10:
                        # Perturbation escape
                        i, j = np.random.choice(n_points, 2, replace=False)
                        delta = np.random.uniform(0.01, 0.03)
                        h[i] = np.clip(h[i] - delta, 0, 1)
                        h[j] = np.clip(h[j] + delta, 0, 1)
                        h = project_to_sum_box(h, TARGET_SUM)
                        current_c5 = evaluate_c5(h)
                        if current_c5 < best_c5:
                            best_h = h.copy()
                            best_c5 = current_c5
                        last_improvement = 0
                else:
                    last_improvement = 0
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5

            # Compute correlation
            corr = compute_corr_fft(h)
            
            # Soft max weights for gradient
            c_alpha = alpha * corr
            c_max = np.max(c_alpha)
            exp_c = np.exp(c_alpha - c_max)
            weights = exp_c / np.sum(exp_c)
            
            # Compute gradient
            grad = np.zeros_like(h)
            for k in range(-n_points+1, n_points):
                if k == 0:
                    continue
                grad_k = grad_shift(h, k)
                grad += weights[k+n_points-1] * grad_k
            
            # Nesterov accelerated gradient step
            y = h + beta * (h - h_prev)
            h_tmp = y - eta * grad
            h_new = project_to_sum_box(h_tmp, TARGET_SUM)
            
            # Backtracking line search
            test_c5 = c5_from_corr(compute_corr_fft(h_new), n_points)
            trial_c5 = c5_from_corr(compute_corr_fft(h), n_points)
            
            if test_c5 < trial_c5:
                h = h_new
                h_prev = h_new
                apg_step += 1
                apg_iter += 1
                continue
                
            # Line search failed: reduce step size
            old_eta = eta
            eta = eta * 0.5
            if eta < 1e-6:
                eta = old_eta
                # Accept even if no improvement, as a last resort
                h = h_new
                h_prev = h_new
                apg_step += 1
                apg_iter += 1
                continue
            else:
                # Try again with smaller step
                h_tmp = y - eta * grad
                h_new = project_to_sum_box(h_tmp, TARGET_SUM)
                test_c5 = c5_from_corr(compute_corr_fft(h_new), n_points)
                if test_c5 < trial_c5:
                    h = h_new
                    h_prev = h_new
                    apg_step += 1
                    apg_iter += 1
                else:
                    # Still no improvement
                    h = h_new
                    h_prev = h_new
                    apg_step += 1
                    apg_iter += 1
                    
        # Increase alpha for continuation
        alpha = min(alpha * alpha_increase, max_alpha)
        
        # Update best if needed
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Sub-gradient polishing phase
    eta = 0.08
    sub_iter = 0
    for i in range(1500):
        if time.time() - t_start > 0.95 * max_time:
            break
            
        corr = compute_corr_fft(best_h)
        max_val = np.max(corr)
        k_max = np.where(np.abs(corr - max_val) < 1e-12)[0]
        
        if len(k_max) == 0:
            break
            
        # Compute average gradient
        avg_grad = np.zeros_like(best_h)
        for k in k_max:
            avg_grad += grad_shift(best_h, k - n_points + 1)
        avg_grad = avg_grad / len(k_max)
        
        # Update with projected gradient step
        h_new = best_h - eta * avg_grad
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        new_c5 = c5_from_corr(compute_corr_fft(h_new), n_points)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            eta = min(eta * 1.05, 0.15)
        else:
            eta = max(eta * 0.5, 1e-6)
            
        # Random swap move for shakeout
        if i % 200 == 0:
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(0.01, 0.015)
            best_h[i1] = np.clip(best_h[i1] - delta, 0, 1)
            best_h[i2] = np.clip(best_h[i2] + delta, 0, 1)
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            new_c5 = c5_from_corr(compute_corr_fft(best_h), n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                
        sub_iter += 1
    
    return best_h, best_c5
