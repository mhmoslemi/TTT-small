import numpy as np

from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def project_onto_feasible_set(h, target_sum):
    """Project h onto [0,1]n with sum = target_sum using waterfilling"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    def objective(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while objective(a) * objective(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search to precision 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = objective(m)
        if abs(fm) < 1e-12:
            break
        if fm * objective(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(h + (a + b)/2, 0, 1)

def compute_autocorrelation(h):
    """Compute full autocorrelation using FFT"""
    n = len(h)
    # Pad to avoid circular convolution effects
    L = 2 * n - 1
    H = fft(h, L)
    R_full = ifft(H * np.conj(H)).real
    # Return only non-negative lags
    return R_full[n-1:]

def softmin_weights(R_vals, beta):
    """Compute soft-min weights"""
    exp_neg_beta_R = np.exp(-beta * R_vals)
    weights = exp_neg_beta_R / np.sum(exp_neg_beta_R)
    return weights

def gradient_via_convolution(h, weights):
    """Compute gradient via convolution trick"""
    n = len(h)
    # Reverse weights for backward shift
    weights_rev = weights[::-1]
    # Compute convolution with forward and backward shifts
    conv_forward = np.convolve(weights, h, mode='full')
    conv_backward = np.convolve(weights_rev, h, mode='full')
    
    # Extract relevant parts
    grad = - (conv_forward[n-1:2*n-1] + conv_backward[0:n])
    return grad

def evaluate_c5_exact(h):
    """Evaluate exact C5 using correlation"""
    dx = 2.0 / len(h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)
