import numpy as np

from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def correlation_fft(h):
    """Compute full correlation efficiently using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlation_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Convolve two arrays using FFT"""
    L = len(a) + len(b) - 1
    L = 1 << (L - 1).bit_length()  # next power of 2
    return np.fft.ifft(np.fft.fft(a, L) * np.fft.fft(b, L)).real

def smooth_max(corr, alpha):
    """Smooth approximation to max using log-sum-exp"""
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * corr)))

def smooth_max_grad(corr, alpha):
    """Gradient of smooth max with respect to corr"""
    w = np.exp(alpha * corr)
    w /= np.sum(w)
    return w

def compute_grad(h, corr, alpha):
    """Compute gradient of smooth max approximation w.r.t. h"""
    w = smooth_max_grad(corr, alpha)
    # convolve with h to get (w * h)
    conv1 = convolve_fft(w, h)
    # convolve w_rev [reversed weights] with h to get (w_rev * h)
    conv2 = convolve_fft(w[::-1], h)
    return 1.0 - conv1 - conv2

def line_search(h, grad, alpha):
    """Perform line search to find appropriate step size"""
    eta = 0.5
    h0 = h.copy()
    c0 = evaluate_c5(h0)
    
    while eta > 1e-5:
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c_new = evaluate_c5(h_new)
        if c_new < c0:
            return h_new, eta
        eta *= 0.5
    return h0, eta

def binary_round(h):
    """Convert continuous h to binary with correct count"""
    median_val = np.median(h)
    b = (h >= median_val).astype(float)
    
    current_count = np.sum(b)
    needed_count = TARGET_SUM
    
    if current_count == needed_count:
        return b
    
    # Fix the count
    if current_count > needed_count:
        # Too many ones, flip some to zeros
        flip_indices = np.argsort(h)[int(current_count - needed_count):]
        b[flip_indices] = 0
    else:
        # Too few ones, flip some to ones
        flip_indices = np.argsort(h)[int(needed_count - current_count):]
        b[flip_indices] = 1
    
    return b

def swap_improve(b, max_corr):
    """Improve binary sequence by swapping 1s and 0s"""
    h_b = b.copy()
    corr_b = max_corr.copy()
    best_c5 = evaluate_c5(h_b)
    best_b = h_b.copy()
    n = len(h_b)
    
    # Find all 1s and 0s
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    
    # Sample a few swaps
    for _ in range(5 * n):
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Swap values and update correlation
        h_b[i], h_b[j] = h_b[j], h_b[i]
        new_c5 = evaluate_c5(h_b)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_b = h_b.copy()
        else:
            # Undo the swap
            h_b[i], h_b[j] = h_b[j], h_b[i]
            
    return best_b, best_c5
