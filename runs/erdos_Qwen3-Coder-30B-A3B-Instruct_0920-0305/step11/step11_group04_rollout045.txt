def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    dx = 2.0 / len(h)
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def compute_shift_matrices(n):
    """Compute shift matrices for all valid shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create Toeplitz matrix with 1s on diagonal k
        diagonals = [1] * (n - abs(k))
        offset = max(0, k)
        shift_mat = diags(diagonals, k, shape=(n, n), format='csr')
        shift_matrices.append(shift_mat)
    return shift_matrices

def solve_sdp(n, shift_matrices, max_time=400):
    """Solve SDP relaxation for Erdős problem"""
    # Define variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD constraint
        cp.diag(X) == 1,  # diagonal constraint
        t >= 0,
        t <= 1
    ]
    
    # Balanced constraint (sum of entries = 0)
    J = np.ones((n, n))
    constraints.append(cp.trace(J @ X) == 0)
    
    # Shift constraints for all valid shifts
    for i, shift_mat in enumerate(shift_matrices):
        # shift_mat has shape (n,n), we want to compute trace(S_k @ X) >= 2*n*(1-t) - n_k
        n_k = 2*n - 1 - abs(i - (n-1))
        constraints.append(cp.trace(shift_mat @ X) >= 2*n*(1-t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS
    start_time = time.time()
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
        if prob.status == cp.OPTIMAL:
            return X.value, t.value, True
    except:
        pass
    
    return None, None, False

        eigenvals, eigenvecs = eigh(X)

        mask = eigenvals > 1e-8
        V = eigenvecs[:, mask]
        D = np.diag(np.sqrt(eigenvals[mask]))
        L = V @ D
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(R):
            # Random vector
            g = np.random.randn(L.shape[1])
            
            # Compute signs
            signs = np.sign(L @ g)
            h = (signs + 1) / 2
            
            # Ensure exactly half ones
            current_ones = np.sum(h)
            if current_ones > n/2:
                indices_to_flip = np.argsort(h)[:int(current_ones - n/2)]
                h[indices_to_flip] = 0
            elif current_ones < n/2:
                indices_to_flip = np.argsort(h)[int(n/2 - current_ones):]
                h[indices_to_flip] = 1
            
            # Evaluate
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    except:
        return None, float('inf')
