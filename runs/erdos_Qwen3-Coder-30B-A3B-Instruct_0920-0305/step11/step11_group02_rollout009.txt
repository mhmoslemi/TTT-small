import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorrelation(h):
    """Compute autocorrelation using FFT"""
    L = 2 * len(h)
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient from weights using FFT"""
    L = 2 * len(h)
    w_rev = w[::-1]
    
    # Convolution with h: w * h (shifted by +j)
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    
    # Convolution with h: w * h (shifted by -j)
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # The shifted indices need to be handled correctly for the gradient
    grad = -(1.0 / target_sum) * (conv_plus[(len(h)-1):] + conv_rev[(len(h)-1):])
    return grad

def compute_c5(h):
    """Compute C5 value exactly"""
    O = autocorrelation(h)
    return dx * np.max(target_sum - O)

    h_binary = project_to_sum_box(h_binary, target_sum)
    
    # Try both initializations
    for h0 in [h_upsampled, h_binary]:
        # Gradient descent with continuation
        h = h0.copy()
        alpha = 1.0      # smoothing parameter

        last_improvement = 0
        best_h_local = h.copy()
        best_c5_local = compute_c5(h0)

            if time.time() - start_time > budget_s * 0.95:

            O = autocorrelation(h)
            g = target_sum - O
            
            # Compute smooth max
            max_g = np.max(g)
            shift = g - max_g
            w_raw = np.exp(alpha * shift)
            w = w_raw / np.sum(w_raw)
            
            # Compute gradient
            grad = gradient_from_weights(h, w)
            
            # Projected gradient step
            y = h - eta * grad
            h = project_to_sum_box(y, target_sum)
