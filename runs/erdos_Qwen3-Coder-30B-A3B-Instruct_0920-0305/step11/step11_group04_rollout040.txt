import numpy as np

# Precomputed Thue-Morse sequence for n=128
def thue_morse_binary(n):
    # Generate Thue-Morse sequence of length n (binary 0/1)
    tm = [0] * n
    for i in range(1, n):
        tm[i] = (tm[i >> 1] + 1) % 2
    return np.array(tm, dtype=float)

def project_to_sum_box(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum using water filling"""
    # Sort values in descending order
    sorted_indices = np.argsort(-x)
    sorted_values = x[sorted_indices]
    
    # Water-filling to find threshold lambda
    cumsum = 0.0
    threshold = 0.0
    for i in range(len(sorted_values)):
        cumsum += sorted_values[i]
        # Check if we can make all values from 0 to i equal to min(1, sorted_values[i])
        # without exceeding the target sum
        if i + 1 > target_sum:
            break
        if cumsum + (len(sorted_values) - i - 1) <= target_sum:
            threshold = sorted_values[i]
        else:
            # Compute exact threshold for remaining elements
            new_sum = cumsum + (len(sorted_values) - i - 1) * min(1.0, sorted_values[i])
            if new_sum >= target_sum:
                # current threshold for remaining ones
                threshold = (target_sum - cumsum) / (len(sorted_values) - i - 1)
                break
    else:
        # If we get here, all values are already larger than threshold
        threshold = max(0.0, min(1.0, (target_sum - cumsum) / len(sorted_values)))
        
    # Apply threshold to all elements
    result = np.clip(x, 0.0, 1.0)
    # Adjust to reach exact sum
    below_threshold = np.where(result < threshold)[0]
    above_threshold = np.where(result > threshold)[0]
    
    # Determine how much to adjust
    current_sum = np.sum(result)
    remaining_adjustment = target_sum - current_sum
    
    if remaining_adjustment > 0:
        # Increase values below threshold
        sorted_below = np.sort(result[below_threshold])
        if len(sorted_below) > 0:
            # Increase the lowest ones
            for i in range(len(sorted_below)):
                inc = min(1.0 - sorted_below[i], remaining_adjustment)
                if inc > 0:
                    # Find corresponding index
                    idx = np.where(result == sorted_below[i])[0][0]
                    result[idx] += inc
                    remaining_adjustment -= inc
                    if remaining_adjustment == 0:
                        break
    elif remaining_adjustment < 0:
        # Decrease values above threshold
        sorted_above = np.sort(result[above_threshold])
        sorted_above = sorted_above[::-1]  # Descending
        if len(sorted_above) > 0:
            for i in range(len(sorted_above)):
                dec = min(sorted_above[i] - 0.0, -remaining_adjustment)
                if dec > 0:
                    # Find corresponding index
                    idx = np.where(result == sorted_above[i])[0][0]
                    result[idx] -= dec
                    remaining_adjustment += dec
                    if remaining_adjustment == 0:
                        break

    return result

def autocorr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_C5(h, dx):
    """Compute C5 using full correlation"""
    corr = autocorr(h) * dx
    return 1.0 - np.min(corr)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Perform greedy swap local search focused on minimizing worst shift"""
    n = len(h)
    target_sum = n // 2
    max_iter = 1000
    
    # Compute correlations
    corr = autocorr(h) * dx
    worst_shift = np.argmin(corr)
    # We just optimize the worst shift
    w = np.zeros(2 * n - 1)
    w[worst_shift] = 1.0  # To just focus on that shift for now

    iterations = 0
    while iterations < max_iter and time.time() - start_time < budget_s * 0.95:
        iterations += 1
        # Determine the shift k that gives the lowest correlation
        k_opt = np.argmin(corr)

        # Find indices where we can make a beneficial swap
        # We look for 0 at i and 1 at i + k
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        best_swap = None
        best_gain = -np.inf
        
        # For all zeros, check where becomes one at shift
        for i in zeros:
            j = i + k_opt
            if j >= 0 and j < n:
                # Swapping the zero at i to one would add +dx to correlation if h[j] == 1
                gain = 1.0 if h[j] == 1 else 0.0
                # Also subtract the gain from where we flip a one to zero
                loss = 0.0
                # We also might affect correlation at k_opt by removing a 1-1 product
                # Just estimate it simple way - use correlation coefficient
                if i + k_opt < n and i + k_opt >= 0:
                    if h[i + k_opt] == 1:
                        loss += 1.0  # if lose a 1-1 pair due to setting i to 1, we lose value
                        # Estimate degradation of shift k in hand by replacing one with zero
                        # But simple approach: check net influence on correlation of worst shift
                
                net_gain = gain - loss
                if net_gain > best_gain:
                    best_gain = net_gain
                    best_swap = (i, i+k_opt)

                    best_gain = net_gain
                    best_swap = (i, i-k_opt)

                opportunity = True
            elif j - k_opt < n and j - k_opt >= 0 and h[j - k_opt] == 1:
                opportunity = True
            if not opportunity:
                continue
                
            # Look at zero locations that paired with one at shift k_opt
            for i in zeros:
                if (i + k_opt == j) or (i - k_opt == j):  # If j pairs with i at k_opt shift
                    gain = 0.0
                    loss = 0.0
                    # Check gain if we swap them 
                    if i + k_opt == j:
                        # We would effectively add two coalitions to k_opt
                        # No net increase?
                        pass
                    elif i - k_opt == j:
                        # Similar reflection
                        pass
                    net_gain = gain - loss
                    # No need to compute lighting detail each

        # Simpler version with direct swap evaluation
        # Very crude nearest improvement:
        if best_swap is not None:
            i, j = best_swap
            # Execute swap with a small benefit test
            if h[i] == 0 and h[j] == 1:
                # Swap i and j
                temp_h = h.copy()
                temp_h[i] = 1
                temp_h[j] = 0
                # Recompute correlation for the worst shift only
                corr_new = autocorr(temp_h) * dx
                if corr_new[k_opt] > corr[k_opt]:
                    h = temp_h
                    corr = corr_new
                    continue  # Restart search with new args
            elif h[i] == 1 and h[j] == 0:
                # Swap i and j
                temp_h = h.copy()
                temp_h[i] = 0
                temp_h[j] = 1
                # Recompute correlation for the worst shift only
                corr_new = autocorr(temp_h) * dx
                if corr_new[k_opt] > corr[k_opt]:
                    h = temp_h
                    corr = corr_new
                    continue  # Restart search with new args

        # Try random swap to escape local minima
        if np.random.rand() < 0.05 and len(zeros) > 0 and len(ones) > 0:
            i = np.random.choice(zeros)
            j = np.random.choice(ones)
            temp_h = h.copy()
            temp_h[i] = 1
            temp_h[j] = 0
            # Only update if any improvement effect on correlation
            corr_new = autocorr(temp_h) * dx  
            if corr_new[k_opt] > corr[k_opt]:
                h = temp_h
                corr = corr_new
        else:
            break  # No improvement
