def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum using binary search."""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b) / 2, 0, 1)
    return result

def corr_full(h):
    """Compute full correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Compute the maximum overlap."""
    corr = corr_full(h)
    return dx * np.max(corr)

def grad_shift(h, k):
    """Compute the gradient of correlation term for shift k."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1 - h[:-k] if k > 0 else 1 - h
        if k < n:
            grad[:k] = 0
    else:
        grad[:k] = 1 - h[-k:] if k < 0 else 1 - h
        if k < 0:
            grad[k:] = 0
    return grad

def shift_mask(k, n):
    """Create sparse shift mask for shift k."""
    if abs(k) >= n:
        return csr_matrix((n, n))
    data = np.ones(n - abs(k))
    if k > 0:
        row = np.arange(n - k)
        col = np.arange(k, n)
    else:
        row = np.arange(-k, n)
        col = np.arange(n + k)
    return csr_matrix((data, (row, col)), shape=(n, n))

def build_sdp(n):
    """Build SDP relaxation for the problem."""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n / 2,
        tau >= 0
    ]
    for k in range(-(n - 1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n / 2) * (1 - tau))
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob

def solve_sdp(problem):
    """Solve the SDP and return solution."""
    try:
        problem.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if problem.status == cp.OPTIMAL:
            return problem.variables()[0].value, problem.variables()[1].value, problem.objective.value
        else:
            raise Exception("Problem is not optimal")
    except:
        raise Exception("SDP solver failed")

def round_from_sdp(h_sdp, X_sdp):
    """Round the SDP solution to get a feasible vector."""
    # Diagonal rounding
    h1 = h_sdp.copy()
    
    # Leading eigenvector rounding
    try:
        evals, evecs = np.linalg.eigh(X_sdp)
        v = evecs[:, -1]
        h2 = np.sqrt(evals[-1]) * v
        h2 = np.abs(h2)  # non-negative
        # Ensure correct sum
        h2 = proj_sum_box(h2, len(h2) / 2)
    except:
        h2 = h1.copy()
    
    # Randomized rounding
    h3s = []
    for _ in range(5):
        z = np.random.multivariate_normal(np.zeros(len(h_sdp)), X_sdp)
        h3 = np.abs(z)
        # Ensure correct sum
        h3 = proj_sum_box(h3, len(h3) / 2)
        h3s.append(h3)
    
    # Evaluate all candidates
    best_h = h1
    best_c5 = max_overlap(h1, 2.0 / len(h1))
    
    h2_c5 = max_overlap(h2, 2.0 / len(h1))
    if h2_c5 < best_c5:
        best_h = h2
        best_c5 = h2_c5
        
    for h3 in h3s:
        h3_c5 = max_overlap(h3, 2.0 / len(h1))
        if h3_c5 < best_c5:
            best_h = h3
            best_c5 = h3_c5
    
    return best_h, best_c5

        c = corr_full(h)
        cmax = np.max(c)
        Kmax = np.where(np.abs(c - cmax) < 1e-12)[0]
        
        # Compute average gradient
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n / 2)
        
        # Projected gradient update
        h_new_c5 = max_overlap(h_new, dx)
        if h_new_c5 < best_c5:
            best_c5 = h_new_c5
            best_h = h_new.copy()
            step = min(step * 1.08, 0.2)
        else:
            step *= 0.5
            
        h_prev = h.copy()
        h = h_new.copy()
        
        # Swap move every 250 iterations
        if it % 250 == 0 and it > 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            old_val_i, old_val_j = h[i], h[j]
            h[i] = np.clip(h[i] - delta, 0, 1)
            h[j] = np.clip(h[j] + delta, 0, 1)

            h_c5 = max_overlap(h, dx)
            if h_c5 <= best_c5:
                best_c5 = h_c5
                best_h = h.copy()
            else:
                h[i] = old_val_i
                h[j] = old_val_j

    return best_h, best_c5
