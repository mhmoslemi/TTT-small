def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for binary search
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Ensure bounds are correct
    while f(lam_low) > 0:
        lam_low -= 1.0
    while f(lam_high) < 0:
        lam_high += 1.0
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        fm = f(lam)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = lam
        else:
            lam_low = lam
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_grad_shift(h, k):
    """Compute gradient of shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def log_sum_exp_surrogate(c, alpha):
    """Compute log-sum-exp surrogate"""
    max_c = np.max(c)
    if max_c == -np.inf:
        return 0.0
    c_shifted = alpha * (c - max_c)
    exp_terms = np.exp(c_shifted)
    log_sum = max_c + np.log(np.sum(exp_terms)) / alpha
    return log_sum
