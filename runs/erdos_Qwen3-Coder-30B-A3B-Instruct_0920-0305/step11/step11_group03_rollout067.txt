import numpy as np

from scipy.fft import fft, ifft

def project_onto_feasible_set(h, target_sum):
    """Project h onto the set {h : 0 <= h <= 1, sum(h) = target_sum}"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def softmin_weights(R_vals, beta):
    """Compute softmin weights for given R_vals and temperature beta"""
    exp_R = np.exp(-beta * R_vals)
    return exp_R / np.sum(exp_R)

def compute_autocorrelation(h, dx):
    """Compute all autocorrelations using FFT"""
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n-1)])
    H = fft(h_extended)
    R_full = ifft(H * np.conj(H)).real
    # Extract the relevant part R_k for k>=0
    R_vals = R_full[n-1:]
    return R_vals * dx

def compute_gradient(h, R_vals, w, dx):
    """Compute gradient of softmin using convolution trick"""

    # Reverse the weights for the backward shift
    w_rev = w[::-1]
    # Forward shift contribution: conv(w, h, mode='full')[n-1:2*n-1]
    # Backward shift contribution: conv(w_rev, h, mode='full')[0:n]
    # But we need to be careful about modes and indices
    w_extended = np.concatenate([np.zeros(n-1), w, np.zeros(n-1)])
    h_extended = np.concatenate([h, np.zeros(2*n-2)])
    grad_part1 = ifft(fft(w_extended) * fft(h_extended)).real[n-1:2*n-1]
    grad_part2 = ifft(fft(w_rev) * fft(h_extended)).real[0:n]
    # Combine and negate for gradient of softmin
    grad = -(grad_part1 + grad_part2)
    return grad * dx

def compute_f_beta(R_vals, beta):
    """Compute softmin value"""
    exp_neg_beta_R = np.exp(-beta * R_vals)
    return np.log(np.sum(exp_neg_beta_R)) / beta
