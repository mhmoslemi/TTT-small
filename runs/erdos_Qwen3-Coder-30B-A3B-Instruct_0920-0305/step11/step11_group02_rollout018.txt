import numpy as np

import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    L = 2 * len(h)
    h_pad = np.pad(h, (0, L - len(h)), mode='constant')
    h_rev = h_pad[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev) * np.fft.fft(h_pad)).real
    return corr

def compute_gradient(h, weights):
    """Compute gradient from weights using FFT"""
    L = 2 * len(h)
    w_pad = np.pad(weights, (0, L - len(weights)), mode='constant')
    w_rev = w_pad[::-1]
    
    h_pad = np.pad(h, (0, L - len(h)), mode='constant')
    
    conv_plus = np.fft.ifft(np.fft.fft(w_pad) * np.fft.fft(h_pad)).real
    conv_rev = np.fft.ifft(np.fft.fft(w_rev) * np.fft.fft(h_pad)).real
    
    # Shift to center
    shift = len(h) - 1
    grad = -(conv_plus[shift:] + conv_rev[shift:])
    return grad[:len(h)]

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Compute autocorrelation for all lags
    O = compute_autocorrelation(h)
    # g_j = S - O_j where S = n/2
    S = len(h) / 2
    g = S - O
    return c5_from_corr(g, len(h))

def smooth_max_surrogate(h, alpha=1.0):
    """Compute smooth max surrogate with parameter alpha"""
    S = len(h) / 2
    O = compute_autocorrelation(h)
    g = S - O
    
    # Numerical stability: shift by max to avoid overflow
    max_g = np.max(g)
    shifted_g = g - max_g
    
    # Softmax weights
    exp_weights = np.exp(alpha * shifted_g)
    weights = exp_weights / np.sum(exp_weights)
    
    # Compute gradient with FFT
    grad = compute_gradient(h, weights)
    
    # Return the smooth surrogate value (negative since we're minimizing)
    smooth_value = np.log(np.sum(exp_weights)) / alpha + max_g
    
    return smooth_value, grad, weights
