import numpy as np
from scipy.signal import correlate

import time

# Global variables
n_points = 180
dx = 2.0 / n_points

def compute_cosine_basis(n, m):
    """Compute cosine basis matrix Phi"""
    i = np.arange(n)
    Phi = np.ones((n, m))
    for j in range(1, m):
        Phi[:, j] = np.cos(np.pi * j * i / (n - 1))
    return Phi

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def softmin_weights(R, beta):
    """Compute softmin weights for given correlation values"""
    R_shifted = R - np.min(R)  # Shift for numerical stability
    exp_R = np.exp(-beta * R_shifted)
    weights = exp_R / np.sum(exp_R)
    return weights

def compute_autocorrelations(h):
    """Compute full autocorrelations using FFT"""
    n = len(h)
    # Pad to avoid circular convolution effects
    h_padded = np.concatenate([h, np.zeros(2*n-1)])
    h_fft = np.fft.fft(h_padded)
    R_full = np.fft.ifft(h_fft * np.conj(h_fft)).real
    # Extract the non-negative lags (positive shifts)
    R = R_full[n-1:2*n-1] * dx
    return R

def gradient_from_weights(weights, h):
    """Compute gradient from softmin weights"""
    n = len(h)
    # Create the gradient from the convolution of weights with shifted h
    weights_rev = weights[::-1]
    # Forward shift part
    forward_shift = np.convolve(weights, h, mode='full')[n-1:2*n-1]
    # Backward shift part
    backward_shift = np.convolve(weights_rev, h, mode='full')[0:n]
    
    grad = -(forward_shift + backward_shift)
    return grad

    weights = softmin_weights(R, beta)
    
    # Objective value
    objective_value = np.log(np.sum(np.exp(-beta * R))) / beta
    
    # Gradient
    grad = gradient_from_weights(weights, h)
    grad = Phi.T @ grad
    
    return objective_value, grad
