```python
import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_fft(h):
    """Compute the correlation of h with (1-h) using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max(corr, alpha):
    """Compute smooth max of correlation with parameter alpha"""
    return (1/alpha) * np.log(np.sum(np.exp(alpha * corr)))

def compute_gradient(h, corr, alpha):
    """Compute gradient of smooth max w.r.t. h"""
    w = np.exp(alpha * corr)
    w = w / np.sum(w)

    # Convolution with w
    L = 2 * len(h) - 1
    conv1 = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real[:len(h)]
    conv2 = np.fft.ifft(np.fft.fft(w[::-1], L) * np.fft.fft(h, L)).real[:len(h)]
    
    return 1.0 - conv1 - conv2

def line_search(h, grad, target_sum, alpha, max_steps=10):
    """Line search along gradient direction using projected steps"""
    eta = 0.5
    eta_min = 1e-5
    
    for _ in range(max_steps):
        h_new = project_to_sum_box(h - eta * grad, target_sum)
        c_new = np.max(correlation_fft(h_new))
        c_old = np.max(correlation_fft(h))
        
        if c_new < c_old:
            return h_new, eta
        eta *= 0.5
        if eta < eta_min:
            break
    
    return h, eta

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population (DE)
    popsize = 30
    population = []
    for _ in range(popsize):
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c = np.max(correlation_fft(h))
        fitness.append(c)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # DE parameters
    F = 0.5
    CR = 0.9
    
    # Stage 1: Differential Evolution (60% of budget)
    t0 = time.time()
    while time.time() - t0 < 0.6 * budget_s and time.time() - start_time < budget_s:
        for i in range(popsize):
            # Mutation
            r1, r2, r3 = np.random.choice(popsize, 3, replace=False)
            v = population[r1] + F * (population[r2] - population[r3])
            
            # Crossover
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR or j == np.random.randint(0, n_points):
                    trial[j] = v[j]
            
            # Repair
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            c_trial = np.max(correlation_fft(trial))
            c_current = fitness[i]
            
            if c_trial < c_current:
                population[i] = trial
                fitness[i] = c_trial
                
                # Update best
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
        
        # Self-adaptation
        if np.random.rand() < 0.1:
            F = np.clip(F + np.random.normal(0, 0.1), 0, 1)
            CR = np.clip(CR + np.random.normal(0, 0.1), 0, 1)
    
    # Stage 2: Smoothed Max Gradient Descent (30% of budget)
    t1 = time.time()
    h = best_h.copy()
    alpha = 4.0
    
    while time.time() - t1 < 0.3 * budget_s and time.time() - start_time < budget_s:
        c = correlation_fft(h)
        grad = compute_gradient(h, c, alpha)
        h, _ = line_search(h, grad, TARGET_SUM, alpha)
        c_new = np.max(c)
        
        # Update best
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h.copy()
        
        # Increase alpha
        if int(time.time() - t1) % 30 == 0:
            alpha = min(alpha * 1.5, 200)
    
    # Stage 3: Binary rounding and swap hill climbing (10% of budget)
    t2 = time.time()
    
    # Convert to binary
    median_val = np.median(h)
    b = (h >= median_val).astype(float)
    
    # Fix sum to target
    ones_count = np.sum(b)
    if ones_count > TARGET_SUM:
        # Find indices to flip
        to_flip = np.where(b == 1)[0]
        indices_to_flip = np.random.choice(to_flip, int(ones_count - TARGET_SUM), replace=False)
        b[indices_to_flip] = 0
    elif ones_count < TARGET_SUM:
        # Find indices to flip
        to_flip = np.where(b == 0)[0]
        indices_to_flip = np.random.choice(to_flip, int(TARGET_SUM - ones_count), replace=False)
        b[indices_to_flip] = 1
    
    # Swap hill climbing
    c_current = np.max(correlation_fft(b))
    if c_current < best_c5:
        best_c5 = c_current
        best_h = b.copy()
    
    # Run multiple restarts of swap hill climbing
    for _ in range(100):
        if time.time() - t2 > 0.1 * budget_s:
            break
            
        # Randomly perturb the binary vector
        b_copy = b.copy()
        num_flips = int(0.05 * n_points)  # 5% of elements
        flip_indices = np.random.choice(n_points, num_flips, replace=False)
        
        for idx in flip_indices:
            b_copy[idx] = 1 - b_copy[idx]
        
        # Fix the sum
        ones_count = np.sum(b_copy)
        if ones_count > TARGET_SUM:
            to_flip = np.where(b_copy == 1)[0]
            indices_to_flip = np.random.choice(to_flip, int(ones_count - TARGET_SUM), replace=False)
            b_copy[indices_to_flip] = 0
        elif ones_count < TARGET_SUM:
            to_flip = np.where(b_copy == 0)[0]
            indices_to_flip = np.random.choice(to_flip, int(TARGET_SUM - ones_count), replace=False)
            b_copy[indices_to_flip] = 1
        
        # Evaluate
        c_new = np.max(correlation_fft(b_copy))
        if c_new < best_c5:
            best_c5 = c_new
            best_h = b_copy.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
```
