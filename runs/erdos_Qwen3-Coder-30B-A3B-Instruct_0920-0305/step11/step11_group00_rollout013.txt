import numpy as np

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for binary search
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        fm = f(lam)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = lam
        else:
            lam_low = lam
    
    result = np.clip(x + lam, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_grad_shift(h, k):
    """Compute gradient for a specific shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def log_sum_exp_surrogate(c, alpha):
    """Compute log-sum-exp surrogate of the max"""
    c_alpha = alpha * c
    c_max = np.max(c_alpha)
    exp_vals = np.exp(c_alpha - c_max)
    log_sum = c_max + np.log(np.sum(exp_vals))
    return log_sum / alpha

def gradient_from_surrogate(c, alpha):
    """Compute gradient of the log-sum-exp surrogate"""
    c_alpha = alpha * c
    c_max = np.max(c_alpha)
    exp_vals = np.exp(c_alpha - c_max)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros_like(c)
    for k in range(len(c)):
        grad_k = compute_grad_shift(h, k - (len(h) - 1))
        grad += weights[k] * grad_k
    return grad

def accelerated_projected_gradient(h0, max_iter=200, alpha_init=0.5, alpha_factor=1.4, 
                                  max_alpha=40, eta_init=0.1, backtracking_factor=0.5):
    """Accelerated projected gradient descent with continuation"""
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    alpha = alpha_init
    eta = eta_init
    best_c5 = float('inf')
    best_h = h0.copy()
    
    for iteration in range(max_iter):
        # Compute correlation
        c = compute_correlation(h)
        current_c5 = np.max(c)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update continuation parameter
        if iteration % 100 == 0 and alpha < max_alpha:
            alpha *= alpha_factor
            if alpha > max_alpha:
                alpha = max_alpha
        
        # Compute surrogate gradient
        if alpha < max_alpha:
            surrogate_grad = gradient_from_surrogate(c, alpha)
        else:
            # Use exact max gradient for final polishing
            max_indices = np.where(np.abs(c - np.max(c)) < 1e-12)[0]
            surrogate_grad = np.zeros_like(h)
            for k in max_indices:
                grad_k = compute_grad_shift(h, k - (len(h) - 1))
                surrogate_grad += grad_k
            surrogate_grad /= len(max_indices)
        
        # Nesterov momentum
        beta = 0.9
        y = h + beta * (h - h_prev)
        
        # Backtracking line search
        while True:
            h_tmp = y - eta * surrogate_grad
            h_new = project_to_sum_box(h_tmp, len(h) / 2)
            c_new = compute_correlation(h_new)
            new_c5 = np.max(c_new)
            
            if new_c5 <= current_c5 or eta < 1e-8:
                break
            eta *= backtracking_factor
        
        # Update step size
        h_prev = h.copy()
        h = h_new
        
        # Random perturbation escape
        if iteration % 100 == 0 and iteration > 0:
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = 0.02
            if h[i] > delta and h[j] < 1 - delta:
                h[i] -= delta
                h[j] += delta
                h = project_to_sum_box(h, len(h) / 2)
                c_new = compute_correlation(h)
                new_c5 = np.max(c_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        
        # Increase eta if we are making progress
        if alpha < max_alpha and iteration % 10 == 0:
            eta = min(eta * 1.1, 1.0)
    
    return best_h, best_c5
