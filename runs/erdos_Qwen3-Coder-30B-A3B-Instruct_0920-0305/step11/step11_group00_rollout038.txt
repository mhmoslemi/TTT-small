```python
import numpy as np
from scipy.signal import correlate
from scipy.interpolate import interp1d
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_spline_profile(z_values, n_points=120):
    """Build a symmetric spline profile from z values"""
    # Use 7 knots: 0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0  
    # This represents 4 free parameters for symmetric profile
    assert len(z_values) == 4
    knots = np.array([0.0, 0.2, 0.4, 0.5])
    y_vals = sigmoid(np.array(z_values))
    
    # Ensure monotonicity by mirroring
    full_y = np.concatenate([y_vals, y_vals[::-1]])
    full_x = np.concatenate([knots, 2 - knots[::-1]])
    
    # Create interpolating spline (not using scipy.interpolate directly to maintain control)
    x_vals = np.linspace(0, 1, 100)
    
    # Generate cubic spline using scipy interpolator for smoothness
    f = interp1d(full_x, full_y, kind='cubic', fill_value='extrapolate')
    y_vals_interp = f(x_vals)
    
    # Downsample to n_points
    h_vals = np.interp(np.linspace(0, 1, n_points), x_vals, y_vals_interp)
    
    # Apply symmetry on full interval [0,2]
    h_symmetric = np.concatenate([h_vals, h_vals[::-1]])
    
    # Ensure symmetry in the actual sampled points
    h_final = h_symmetric[:n_points]
    
    # Scale to satisfy mass constraint
    sum_h = np.sum(h_final)
    if sum_h > 0:
        alpha = TARGET_SUM / sum_h
        h_final = np.clip(alpha * h_final, 0, 1)
        # Final adjustment in case clipping pushed the sum
        h_final = project_to_sum_box(h_final, TARGET_SUM)
    
    return h_final

def calibrate_and_evaluate(z_values):
    """Compute C5 for given z parameters"""
    h = build_spline_profile(z_values)
    return evaluate_c5(h)

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Initial guess based on cubic spline with parametric shape
    initial_guess = np.array([0.0, 0.0, 0.0, 0.0])  # symmetric knots vs weights
    
    best_c5 = float('inf')
    best_h = None
    best_z = None
    
    def objective(z):
        try:
            return calibrate_and_evaluate(z)
        except:
            return float('inf')
    
    # Use an evolutionary strategy with multiple restarts and larger budget
    # Simple multi-start optimization to maximize diversity
    best_z = initial_guess.copy()
    best_c5 = objective(best_z)
    
    # CMA-ES-style inner loop
    outer_iterations = 0
    max_outer = 50
    while outer_iterations < max_outer and (time.time() - start_time) < (budget_s * 0.75):
        outer_iterations += 1
        # Deflate the current guess
        noise_scale = 0.3 + 0.02 * np.random.random()
        new_z = best_z + noise_scale * np.random.normal(0, 1, len(best_z))
        
        try:
            new_c5 = objective(new_z)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_z = new_z.copy()
        except Exception as e:
            continue
    
    # Generate the final solution from best z
    final_h = build_spline_profile(best_z)
    final_c5 = evaluate_c5(final_h)
    
    # Post-processing to fine-tune final solution
    h_current = final_h.copy()
    c5_current = final_c5
    
    # Short local optimization using projected gradient
    max_updates = 1200
    step_size = 0.08
    update_count = 0
    
    while update_count < max_updates and (time.time() - start_time) < (budget_s - 10):
        update_count += 1
        # Compute correlation
        corr = compute_corr_fft(h_current)
        k_worst = np.where(np.abs(corr - np.max(corr)) < 1e-12)[0]
        
        # Approximate gradient for worst overlaps
        grad = np.zeros_like(h_current)
        for k in k_worst:
            if k == 0:
                grad += 1 - h_current
            else:
                shifted_h = np.roll(h_current, k)
                grad += 1 - shifted_h - h_current
        
        # Normalized gradient descent step
        grad = grad / (np.linalg.norm(grad) + 1e-10)
        new_h = h_current - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        new_c5 = evaluate_c5(new_h)
        if new_c5 < c5_current:
            h_current = new_h
            c5_current = new_c5
            if update_count % 200 == 0:
                # Small random perturbation to escape plateaus
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                delta = np.random.uniform(-0.015, 0.015)
                old_val1, old_val2 = h_current[idx1], h_current[idx2]
                h_current[idx1], h_current[idx2] = np.clip(h_current[idx1] + delta, 0, 1), np.clip(h_current[idx2] - delta, 0, 1)
                h_current = project_to_sum_box(h_current, TARGET_SUM)
                new_c5 = evaluate_c5(h_current)
                if new_c5 < c5_current:
                    c5_current = new_c5
                else:
                    h_current[idx1], h_current[idx2] = old_val1, old_val2
        else:
            step_size *= 0.5
            if step_size < 1e-5:
                break
                
    return (h_current.tolist(), float(c5_current), n_points)
```
