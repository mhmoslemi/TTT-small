import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time
import cvxpy as cp

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """Project vector v onto [0,1]^n with sum = target_sum using sorting."""
    n = len(v)
    sorted_indices = np.argsort(-v)  # descending order indices
    sorted_v = v[sorted_indices]
    
    # Water-filling approach
    cumsum = np.cumsum(sorted_v)
    i = np.searchsorted(cumsum, target_sum, side='right')
    
    if i == 0:
        # target_sum <= sorted_v[0] * n
        threshold = target_sum / n
        result = np.clip(v, 0, threshold)
        # Try to scale up necessary entries
        scale_factor = target_sum / np.sum(result)
        result = np.clip(result * scale_factor, 0, 1)
    else:
        # target_sum <= cumsum[i-1]
        threshold = (target_sum - cumsum[i-1]) / (n - i)
        result = np.zeros_like(v)
        # Set values above threshold to 1
        result[sorted_indices[:i]] = 1.0
        # Scale the remainder
        remaining_sum = target_sum - i
        to_scale = sorted_indices[i:]
        if len(to_scale) > 0:
            scale_factor = remaining_sum / np.sum(sorted_v[i:])
            result[sorted_indices[i:]] = np.clip(sorted_v[i:] * scale_factor, 0, 1)
    
    return result

def corr_fft(h):
    """Compute full correlation of h with (1-h) using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT."""
    corr = corr_fft(h)
    return dx * np.max(corr)

def solve_linearized_step(y):
    """Solve linear subproblem given frozen y.
    
    Maximize t subject to:
    sum(x_i * y_{i+s}) >= t for all shifts s
    sum(x_i) = S
    0 <= x_i <= 1
    """
    n = len(y)
    S = TARGET_SUM
    
    # Define variables
    x = cp.Variable(n)
    t = cp.Variable(1)
    
    # Define constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(x) == S)
    
    # Box constraints
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Linear constraints from shifts
    for s in range(1 - n, n):
        # Compute the shifted index indices
        valid_indices = []
        shifted_indices = []
        for i in range(n):
            shifted_i = i + s
            if 0 <= shifted_i < n:
                valid_indices.append(i)
                shifted_indices.append(shifted_i)
        
        if len(valid_indices) > 0:
            # Sum over valid pairs (i, shifted_i)
            shift_sum = cp.sum(cp.multiply(x[valid_indices], y[shifted_indices]))
            constraints.append(shift_sum >= t)
    
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Try several solvers in order of preference
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
    except:
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                # If all fail, just use the initial feasible solution
                return y.copy(), np.min([np.sum(np.multiply(y, np.roll(y, s))) for s in range(1-n, n)])
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        x_sol = x.value
        t_sol = t.value
    else:
        x_sol = y.copy()
        t_sol = np.min([np.sum(np.multiply(y, np.roll(y, s))) for s in range(1-n, n)])
    
    return x_sol, t_sol

    except:

    # 2. Random initializations
    for i in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = project_onto_sum_box(h_rand, TARGET_SUM)
        seeds.append(("random", h_rand.copy()))
    
    # 3. Tent-like initialization
    x = np.linspace(0, 2, n_points)
    h_tent = np.zeros(n_points)
    h_tent[x <= 1] = x[x <= 1]
    h_tent[x >= 1] = 2 - x[x >= 1]
    h_tent = project_onto_sum_box(h_tent, TARGET_SUM)
    seeds.append(("tent", h_tent.copy()))
    
    # 4. Sine-shaped
    h_sine = 0.5 + 0.5 * np.sin(np.pi * x)
    h_sine = project_onto_sum_box(h_sine, TARGET_SUM)
    seeds.append(("sine", h_sine.copy()))
    
    # Alternate between seeds
    for seed_type, h_start in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_h = h_start.copy()
        current_t = -1.0
        last_improvement = 0
        
        for iteration in range(50):  # Max 50 outer iterations
            if time.time() - start_time > budget_s * 0.95:
                break
                
            x_next, t_next = solve_linearized_step(current_h)
            
            if t_next > current_t:
                current_h = x_next.copy()
                current_t = t_next
                last_improvement = iteration
                
                # Enforce symmetry
                symmetric_h = (current_h + current_h[::-1]) / 2.0
                current_h = project_onto_sum_box(symmetric_h, TARGET_SUM)
                
                # Evaluate the true C5
                c5_current = evaluate_c5(current_h)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = current_h.copy()
            else:
                # No improvement
                if iteration - last_improvement > 10:
                    break
                # Poison prevent premature exit
                current_h = project_onto_sum_box(current_h, TARGET_SUM)
        
        # Ensure final projection
        if best_h is not None:
            best_h = project_onto_sum_box(best_h, TARGET_SUM)
            c5_final = evaluate_c5(best_h)
            if c5_final < best_c5:
                best_c5 = c5_final
    
    # Final refinement with pairwise exchanges if time allows
    if best_h is not None and (time.time() - start_time) < budget_s * 0.90:
        h_refined = best_h.copy()
        current_c5 = evaluate_c5(h_refined)
        
        # Simulated annealing refinement
        temp = 0.1
        cooling_rate = 0.9995
        max_steps = 3000
        
        for step in range(max_steps):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Choose two random indices
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Propose move
            delta = np.random.uniform(-0.01, 0.01)
            new_i = max(0, min(1, h_refined[i] - delta))
            new_j = max(0, min(1, h_refined[j] + delta))
            
            # Compute new values
            h_new = h_refined.copy()
            h_new[i] = new_i
            h_new[j] = new_j
            
            # Re-project
            h_new = project_onto_sum_box(h_new, TARGET_SUM)
            
            # Evaluate new C5
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < current_c5:
                h_refined = h_new.copy()
                current_c5 = new_c5
            else:
                p_accept = np.exp(-(new_c5 - current_c5) / (temp + 1e-10))
                if np.random.rand() < p_accept:
                    h_refined = h_new.copy()
                    current_c5 = new_c5
            
            # Update temperature
            temp *= cooling_rate
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined.copy()
                
    if best_h is None:
        # Fallback if nothing was found
        try:
            h0 = np.array(initial_h_values)
            h0 = project_onto_sum_box(h0, TARGET_SUM)
            best_h = h0.copy()
            best_c5 = evaluate_c5(best_h)
        except:
            # Last resort: uniform distribution
            h_uniform = np.ones(n_points) * 0.5
            h_uniform = project_onto_sum_box(h_uniform, TARGET_SUM)
            best_h = h_uniform.copy()
            best_c5 = evaluate_c5(best_h)
            
    return (best_h.tolist(), float(best_c5), n_points)
