import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value from correlation"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial solution"""
    m = n // 2
    # Linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        h = np.concatenate([first, first[::-1]])
    else:
        h = np.concatenate([first, first[-2::-1]])
    # Adjust to match target sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def random_feasible_vector(n, target):
    """Create a random feasible vector"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target)
    return h

def grad_shift(h, s):
    """Compute gradient for shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using smooth approximation of max"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Parameters
    alphas = np.linspace(0.5, 30, 20)
    step = 0.3
    max_iter_per_alpha = 150
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for _ in range(max_iter_per_alpha):
            if time.time() - start_time > time_limit:
                break
            # Compute correlation
            c = compute_corr(h)
            # Compute weights with numerical stability
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n_points - 1)
                grad += weight * grad_shift(h, s)
            
            # Adaptive step size with backtracking
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
        
        # Gradually reduce step size
        step *= 0.9
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using subgradients"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Parameters
    max_iter = 300
    step = 0.3
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute correlation
        c = compute_corr(h)
        
        # Find worst shift(s)
        max_c = np.max(c)
        indices = np.where(np.abs(c - max_c) < 1e-8)[0]
        
        # Compute average gradient over worst shifts
        grad = np.zeros_like(h)
        for idx in indices:
            s = idx - (n_points - 1)
            grad += grad_shift(h, s)
        
        # Average if multiple shifts
        if len(indices) > 1:
            grad /= len(indices)
        
        # Adaptive step size with backtracking
        step_cur = step
        for _ in range(5):
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
                break
            else:
                step_cur *= 0.5
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    best_h = None
    best_c5 = np.inf
    best_n = None
    
    # Try different configurations
    for n in [120, 200]:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        target_sum = n / 2
        n_points = n
        
        # Try multiple random restarts
        for restart in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Initialize with triangular shape
            h0 = triangular_initial(n)
            
            # Smooth optimization
            h_smooth, val_smooth = smooth_optimize(h0, target_sum, dx, 
                                                budget_s * 0.7, start_time)
            
            # Subgradient optimization
            h_final, val_final = subgradient_optimize(h_smooth, target_sum, dx,
                                                    budget_s * 0.9, start_time)
            
            if val_final < best_c5:
                best_h = h_final.copy()
                best_c5 = val_final
                best_n = n
    
    # Ensure proper final projection
    if best_h is not None:
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = c5_value(best_h)
    
    return (best_h.tolist(), float(best_c5), best_n)
