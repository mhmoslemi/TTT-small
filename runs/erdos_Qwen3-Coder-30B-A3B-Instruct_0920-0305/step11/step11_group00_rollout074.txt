import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.sparse import eye as sparse_eye

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum using binary search."""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def corr_full(h):
    """Compute full correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute gradient of correlation term for shift k."""
    n = len(h)
    if k >= 0:
        grad = np.zeros(n)
        grad[k:] = 1 - h[:n-k]
        if k > 0:
            grad[:k] = -h[k:]
    else:
        grad = np.zeros(n)
        grad[:n+k] = 1 - h[-k:]
        if k < 0:
            grad[n+k:] = -h[:k]
    return grad

def max_overlap(h, dx):
    """Compute maximum overlap with dx scaling."""
    c = corr_full(h)
    return dx * np.max(c)

def build_sdp(n):
    """Build the SDP relaxation problem."""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Construct masks for each shift k
    masks = {}
    for k in range(-(n-1), n):
        if k >= 0:
            row_indices = np.arange(n - k)
            col_indices = np.arange(k, n)
        else:
            row_indices = np.arange(-k, n)
            col_indices = np.arange(n + k)
        if len(row_indices) > 0 and len(col_indices) > 0:
            masks[k] = sparse_eye(n, format='csr')
            masks[k] = masks[k][row_indices, :][:, col_indices]
    
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    for k in range(-(n-1), n):
        if k in masks:
            M = masks[k]
            constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau

def solve_sdp(prob):
    """Solve the SDP with ECOS."""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return prob.value, prob.variables()[0].value, prob.variables()[1].value
    except:
        return None, None, None
    return None, None, None

def round_from_sdp(h_sdp, X_sdp, n):
    """Try different rounding methods and return the best feasible vector."""
    # Method 1: Diagonal rounding
    h1 = h_sdp.copy()
    h1 = proj_sum_box(h1, n/2)
    c5_1 = max_overlap(h1, 2.0/n)
    
    # Method 2: Leading eigenvector rounding
    try:
        _, U = np.linalg.eigh(X_sdp)
        u = U[:, -1]
        if np.sum(u) < 0:
            u = -u
        h2 = np.sqrt(np.max(np.linalg.eigvals(X_sdp))) * np.abs(u)
        h2 = proj_sum_box(h2, n/2)
        c5_2 = max_overlap(h2, 2.0/n)
    except:
        c5_2 = float('inf')
    
    # Method 3: Gaussian rounding
    best_c5 = min(c5_1, c5_2)
    best_h = h1 if c5_1 <= c5_2 else h2
    
    for _ in range(5):
        try:
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, n/2)
            c5_3 = max_overlap(h3, 2.0/n)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3
        except:
            continue
    
    return best_h

def polish(h_start, n, dx, remaining_time):
    """Polish the solution using projected gradient descent with subgradients."""
    h = h_start.copy()
    h_prev = h.copy()
    step = 0.07
    beta = 0.9
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    
    start_time = time.time()
    
    for it in range(2000):
        if time.time() - start_time > remaining_time - 5:
            break
            
        c = corr_full(h)
        Cmax = np.max(c)
        Kmax = np.where(np.abs(c - Cmax) < 1e-12)[0]
        
        if len(Kmax) == 0:
            continue
            
        g = np.mean([grad_shift(h, k) for k in Kmax], axis=0)
        
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step*1.08, 0.2)
        else:
            step *= 0.5
            
        h_prev = h.copy()
        h = h_new.copy()
        
        if it % 250 == 0 or (it % 50 == 0 and it > 0):
            # Random swap moves to escape plateaus
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            old_h_i, old_h_j = h[i], h[j]
            h[i] = max(0, min(1, h[i] - delta))
            h[j] = max(0, min(1, h[j] + delta))
            h = proj_sum_box(h, n/2)
            c5_swap = max_overlap(h, dx)
            if c5_swap < best_c5:
                best_c5 = c5_swap
                best_h = h.copy()
            else:
                h[i], h[j] = old_h_i, old_h_j
                
    return best_h, best_c5

def triangular_initial(n):
    """Generate triangular initial shape."""
    h = np.zeros(n)
    mid = n // 2
    for i in range(n):
        if i < mid:
            h[i] = 2 * i / (n - 1)
        else:
            h[i] = 2 * (n - 1 - i) / (n - 1)
    h = proj_sum_box(h, n/2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    for n in [120, 140]:
        if time.time() - start_time > budget_s - 10:
            break
        dx = 2.0 / n
        
        # Build and solve SDP
        try:
            prob, h_vars, X_vars, tau_vars = build_sdp(n)
            tau_opt, h_sdp, X_sdp = solve_sdp(prob)
            if tau_opt is None:
                continue
                
            # Rounding
            h_rounded = round_from_sdp(h_sdp, X_sdp, n)
            c5_rounded = max_overlap(h_rounded, dx)
            
            if c5_rounded < best_c5:
                best_c5 = c5_rounded
                best_h = h_rounded.copy()
                best_n = n
                
            # Polishing with different starting points
            seeds = [h_rounded]
            for _ in range(5):
                initial = np.random.random(n)
                initial = proj_sum_box(initial, n/2)
                seeds.append(initial)
            
            for i, h_seed in enumerate(seeds):
                if time.time() - start_time > budget_s - 10:
                    break
                remaining_time = budget_s - (time.time() - start_time)
                h_p, c5_p = polish(h_seed, n, dx, remaining_time)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    best_n = n

        except Exception as e:
            continue
    
    return (best_h.tolist(), float(best_c5), best_n)
