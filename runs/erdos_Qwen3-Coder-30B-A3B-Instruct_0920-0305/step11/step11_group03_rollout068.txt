import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    try:
        # Variables
        r = cp.Variable(n)
        t = cp.Variable()
        
        # Build Toeplitz matrix
        T = cp.Variable((n, n), symmetric=True)
        for i in range(n):
            for j in range(n):
                T[i, j].value = r[abs(i - j)].value if i < n and j < n else 0
                
        # Constraints
        constraints = [
            T >> 0,  # PSD constraint
            cp.sum((n - cp.arange(n)) * r) == M**2,  # Mass constraint
            r >= t,  # Minimality constraint
            r >= 0,  # Non-negativity
            r <= M,  # Upper bound
            r <= n - cp.arange(n)  # Row-wise bound
        ]
        
        # Objective
        objective = cp.Maximize(t)
        
        # Solve SDP
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-6)
        
        if prob.status != cp.OPTIMAL:
            raise Exception("SDP did not converge")
        
        r_opt = r.value

        max_iter = 200
        eta = 0.1
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:

            # Gradient approximated via finite differences
            grad = np.zeros(n)
            eps = 1e-6
            current_val = np.min(correlate(h, 1 - h, mode='full')) * dx

                h_eps = h.copy()
                h_eps[j] += eps
                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps * (M / np.sum(h_eps))
                val_eps = np.min(correlate(h_eps, 1 - h_eps, mode='full')) * dx
                grad[j] = (val_eps - current_val) / eps
            
            # Projected gradient step
            h_new = h - eta * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (M / np.sum(h_new))
            
            # Check improvement
            new_val = np.min(correlate(h_new, 1 - h_new, mode='full')) * dx

                grad[i] += h[i - k]

            if eta0 < 1e-6:
                break
    
    return (h.tolist(), float(best_c5), n)
