```python
import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK

# Global variables for the problem
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """
    Project vector v onto the hyperplane sum(x_i) = target_sum with bounds 0 <= x_i <= 1.
    Uses O(n log n) water-filling algorithm.
    """
    n = len(v)
    # Sort indices by value (descending)
    sorted_indices = np.argsort(-v)
    # Compute partial sums
    cumulative_sum = 0.0
    i = 0
    while i < n and cumulative_sum + v[sorted_indices[i]] <= target_sum:
        cumulative_sum += v[sorted_indices[i]]
        i += 1

    # At this point, all v[sorted_indices[0..i-1]] are in [0,1],
    # and we set rest of the v[sorted_indices[i..n-1]] to the same value.
    # That value is (target_sum - cumulative_sum)/ (n - i)
    threshold = (target_sum - cumulative_sum) / (n - i) if i < n else 0.0
    result = np.zeros_like(v)
    for j in range(i):
        result[sorted_indices[j]] = v[sorted_indices[j]]
    for j in range(i, n):
        result[sorted_indices[j]] = threshold
    
    # Ensure final sum is correct
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        # Add the difference evenly to lowest entries
        # It should be small, so we just add to indices with small values
        sorted_low_indices = sorted(range(n), key=lambda x: result[x])
        for idx in sorted_low_indices:
            if diff < 1e-12:
                break
            to_add = min(diff, 1 - result[idx])
            if to_add <= 0:
                continue
            result[idx] += to_add
            diff -= to_add
    return np.clip(result, 0, 1)

def corr_fft(h):
    """Efficiently compute the correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Compute C5 using the full correlation from FFT"""
    corr = corr_fft(h)
    return dx * np.max(corr)

def solve_linearized_step(y):
    """
    Solve the LP relaxation:
    maximize t
    s.t. sum_i x_i * y_{i+s} >= t         for all s = -n+1 to n-1
         sum_i x_i == S
         0 <= x_i <= 1
    """
    n = len(y)
    # Decision variables
    x = cp.Variable(n)
    t = cp.Variable(1)

    # Build linear constraints for all shifts
    constraints = []
    # Add constraints one by one -- assume no shifting out
    for s in range(-n+1, n):  
        # Compute correlation term with wrap-around indices
        if s >= 0:
            # Do we need to consider positive shift?
            shift_indices = np.arange(n-s)
            coeffs = y[s:] 
        else: # s < 0
            # Negative shift  
            shift_indices = np.arange(-s, n)
            coeffs = y[:n+s]
        
        coeffs = coeffs[shift_indices]
        a = np.zeros(n)    # Coefficients such that sum_a[i]*x[i] >= t
        a[shift_indices] = coeffs
        constraints.append(cp.sum(cp.multiply(a, x)) >= t)

    # Sum constraint
    constraints.append(cp.sum(x) == TARGET_SUM)

    # Box constraints
    constraints.append(x >= 0)
    constraints.append(x <= 1)

    # Objective
    objective = cp.Maximize(t)

    # Solve program
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            x_opt = x.value
            return x_opt, t.value
        else:
            #Fallback to GLPK
            try:
                prob.solve(solver=GLPK, verbose=False)
                if prob.status == cp.OPTIMAL:
                    x_opt = x.value
                    return x_opt, t.value
            except:
                pass
    except Exception as e:
        pass
    # If problem not solved, return something safe
    return y.copy(), 0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # Define multiple initialization strategies
    initial_seeds = []

    # 1. Use provided initial_h_values if available
    try:
        initial_h_values = [float(x) for x in initial_h_values]
        h0 = np.array(initial_h_values)
        h0 = project_onto_sum_box(h0, TARGET_SUM)
        initial_seeds.append(h0.copy())
    except NameError:
        pass

    # 2. Random vectors with projection
    for _ in range(3):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_onto_sum_box(h_rand, TARGET_SUM)
        initial_seeds.append(h_rand.copy())

    # 3. Tent-like initial guess 
    # We build a piecewise linear function with maximum around middle and vanishing at ends
    tent = np.array([max(0, 1 - 2 * abs(i - n_points/2 + 0.5) / (n_points/2 + 0.5)) for i in range(n_points)])
    tent = project_onto_sum_box(tent, TARGET_SUM)
    initial_seeds.append(tent.copy())

    # Allocate time: 70% to LP iterations, 25% to restarts, 5% to refinement
    total_time = budget_s
    lp_time = 0.7 * total_time
    rest_time = 0.25 * total_time
    refine_time = 0.05 * total_time

    # Main iterative process
    restart_count = 0
    for h_seed in initial_seeds:
        if time.time() - start_time > (total_time - 5):
            break
        restart_count += 1
        
        h = h_seed.copy()
        last_t = -1.0
        tolerance = 1e-8
        convergence_threshold = 20  # Stop after no changes in t
        consecutive_no_improvement = 0

        # Run until time is up or convergence
        while (time.time() - start_time) < lp_time and consecutive_no_improvement < convergence_threshold:
            h_proj = project_onto_sum_box(h, TARGET_SUM)
            x_new, t_now = solve_linearized_step(h_proj)

            if abs(t_now - last_t) < tolerance:
                consecutive_no_improvement += 1
            else:
                consecutive_no_improvement = 0
            last_t = t_now

            h = x_new.copy()

        # Evaluate true C5
        c5_estimate = evaluate_c5(h)

        # Symmetry enforcement
        h_sym = (h + h[::-1]) / 2
        h_sym = project_onto_sum_box(h_sym, TARGET_SUM)
        c5_sym = evaluate_c5(h_sym)
        
        if c5_sym < c5_estimate:
            h = h_sym
            c5_estimate = c5_sym

        if c5_estimate < best_c5:
            best_c5 = c5_estimate
            best_h = h.copy()

    # Now perform refinement with pairwise exchanges
    if best_h is not None:
        h_fine = best_h.copy()
        c5_fine = evaluate_c5(h_fine)
        end_time = start_time + total_time * 0.95
        T = 0.1  # Initial temperature
        decay_rate = 0.999
        step_limit = int(refine_time * 3000)  # ~2000 steps per second roughly

        cnt = 0
        while cnt < step_limit and time.time() < end_time:
            cnt += 1
            # Choose two random indices
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if i == j:
                continue
            # Compute current overlap for this shift
            delta = 0.01 * (np.random.rand() - 0.5)
            # Check if move is valid
            new_h = h_fine.copy()
            if new_h[i] - delta >= 0 and new_h[j] + delta <= 1:
                new_h[i] -= delta
                new_h[j] += delta
                # Re-project onto sum and box
                new_h = project_onto_sum_box(new_h, TARGET_SUM)
                c5_new = evaluate_c5(new_h)
                
                if c5_new < c5_fine:
                    h_fine = new_h
                    c5_fine = c5_new
                elif np.random.rand() < np.exp(-(c5_new - c5_fine)/T):
                    h_fine = new_h
                    c5_fine = c5_new
                T *= decay_rate
        best_c5 = c5_fine
        best_h = h_fine

    if best_h is None:
        # Fallback to initial_h if nothing worked
        try:
            h_0 = np.array(initial_h_values)
            best_h = project_onto_sum_box(h_0, TARGET_SUM)
            best_c5 = evaluate_c5(best_h)
        except:
            best_h = np.ones(n_points) / 2  
            best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
```
