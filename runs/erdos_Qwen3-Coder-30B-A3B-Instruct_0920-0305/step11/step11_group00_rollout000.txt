import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n // 2
    # linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        second = first[::-1]
    else:
        second = first[-2::-1]
    h = np.concatenate([first, second])
    return h

def random_feasible_vector(n, target_sum):
    """Generate random feasible vector"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h

def grad_shift(h, s):
    """Compute gradient for a given shift s"""
    n = len(h)
    # Use padded array for easy indexing
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    
    # Compute the gradient components
    term1 = 1 - padded[base + s]  # 1 - h_{j+s} 
    term2 = padded[base - s]     # h_{j-s}
    
    return term1 - term2

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Smooth gradient descent optimizer"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    step_size = 0.5
    alpha_schedule = np.linspace(0.5, 30, 15)
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        for _ in range(200):
            if time.time() - start_time > time_limit:
                break
                
            # Compute correlation values
            c = compute_corr(h)
            
            # Compute weights using softmax
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute weighted gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n-1)
                shift_grad = grad_shift(h, s)
                grad += weight * shift_grad
            
            # Perform step with line search
            step_cur = step_size
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
            
            # Adjust step size
            if val_new < best_val - 1e-12:
                step_size = min(step_size * 1.1, 2.0)
            else:
                step_size *= 0.5
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Subgradient descent optimizer for fine-tuning"""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    step_size = 0.2
    
    # Determine max shifts to consider for gradient
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute correlations  
        c = compute_corr(h)
        c_max = np.max(c)
        
        # Find indices where correlation is near maximum
        tol = 1e-8
        max_indices = np.where(np.abs(c - c_max) < tol)[0]
        
        # Compute average gradient from the maximum shifts
        grad = np.zeros_like(h)
        for idx in max_indices:
            s = idx - (n-1)
            shift_grad = grad_shift(h, s)
            grad += shift_grad
            
        if len(max_indices) > 0:
            grad /= len(max_indices)
            
        # Perform step with line search
        step_cur = step_size
        for _ in range(5): 
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
                break
            else:
                step_cur *= 0.5
                
        # Adjust step size
        if val_new < best_val - 1e-12:
            step_size = min(step_size * 1.1, 2.0)
        else:
            step_size *= 0.5
    
    return best_h, best_val
