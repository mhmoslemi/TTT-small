import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import cvxpy as cp
import time

def project_onto_sum_box(v, target_sum):
    """Project vector v onto the box constraints [0,1] with sum = target_sum using water-filling"""
    n = len(v)
    # Sort in descending order
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # initially all are clipped to 1
    cumsum = np.cumsum(sorted_v)
    # find the threshold at which to clip
    for i in range(n):
        if cumsum[i] > target_sum:
            # interpolate between i-1 and i
            if i == 0:
                threshold = target_sum / (i+1)
            else:
                threshold = (target_sum - cumsum[i-1]) / (n - i)
            # clip all elements from index i onwards to threshold
            new_values = sorted_v[:i].copy()
            new_values = np.append(new_values, np.full(n-i, max(0.0, threshold)))
            # reconstruct the final array
            proj = v.copy()
            proj[sorted_indices] = new_values
            return np.clip(proj, 0, 1)
    
    # if we reach here, then all elements can be set to 1
    proj = v.copy()
    proj[sorted_indices] = np.ones_like(sorted_v) * (target_sum / n)
    return np.clip(proj, 0, 1)

def compute_correlation_fft(h):
    """Compute correlation using FFT"""
    h_rev = np.flip(h)
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation_fft(h)
    return np.max(corr)

def solve_linearized_step(y, target_sum):
    """Solve the linearized problem using CVXPY"""
    n = len(y)
    # decision variables
    x = cp.Variable(n)
    t = cp.Variable()
    
    # build constraint matrix for shifts
    constraints = []
    constraints.append(cp.sum(x) == target_sum)
    for s in range(-n+1, n):
        # build the shifted index list
        shifted_indices = np.arange(n) + s
        valid_mask = (shifted_indices >= 0) & (shifted_indices < n)
        shifted_y = np.zeros(n)
        shifted_y[shifted_indices[valid_mask]] = y[valid_mask]
        constraints.append(shifted_y @ x >= t)
    constraints.extend([x >= 0, x <= 1])
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        # fallback to GLPK if ECOS fails or is not available
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            pass  # If this also fails, use simple gradient ascent steps for fallback
    
    # get solution
    if x.value is not None and t.value is not None:
        new_x = np.array(x.value).clip(0, 1)
        new_x = project_onto_sum_box(new_x, target_sum)
        achieved_t = t.value
        return new_x, achieved_t
    else:
        # Return current y if optimization failed
        return y, np.min([np.sum(y[i] * y[i+s]) for s in range(-n+1, n) if s != 0])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 144  # slightly larger than the previous 120 to give more flexibility
    target_sum = n_points / 2  # sum of h should be exactly 1 (since dx=2/n_points, integral dx * sum h should be 1)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Multiple restarts from different seeds
    n_restarts = 10
    restarts = []
    # first restart: initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_onto_sum_box(h0, target_sum)
        restarts.append(h0)
    # second restart: random vector
    h_rand = np.random.rand(n_points)
    h_rand = project_onto_sum_box(h_rand, target_sum)
    restarts.append(h_rand)
    # third restart: piecewise linear tent piece
    x = np.linspace(0, 2, n_points)
    tent = np.minimum(x, 2 - x)
    tent = tent / np.sum(tent) * target_sum
    restarts.append(tent)
    # additional uniform random restarts
    for _ in range(n_restarts - 3):
        h_random = np.random.rand(n_points)
        h_random = project_onto_sum_box(h_random, target_sum)
        restarts.append(h_random)
    
    # Use preallocated weights matrix to improve efficiency
    weights_matrix = {}
    for s in range(-n_points+1, n_points):
        # compute indices
        shift_indices = np.arange(n_points) + s
        valid_mask = (shift_indices >= 0) & (shift_indices < n_points)
        weights = np.zeros(n_points)
        weights[shift_indices[valid_mask]] = 1
        weights_matrix[s] = weights
    
    # global outer loop
    for i, start_h in enumerate(restarts):
        # Set current iterate
        current_h = start_h.copy()
        
        # Outer alternating loop
        last_t = -float('inf')
        iter_count = 0
        max_inner_iters = 30  # limit outer steps
        while time.time() - start_time < 0.7 * budget_s and iter_count < max_inner_iters:
            old_h = current_h.copy()
            new_h, t_val = solve_linearized_step(current_h, target_sum)
            
            if t_val - last_t < 1e-8:
                break  # converged
            last_t = t_val
            current_h = new_h
            
            # apply symmetry constraint after each iteration if applicable
            symmetric_h = (current_h + np.flip(current_h)) / 2
            current_h = project_onto_sum_box(symmetric_h, target_sum)
            
            iter_count += 1
        
        # evaluate actual C5
        actual_c5 = dx * (1 - last_t)  # since C5 = dx * max_i (1 - sum_i h_i h_{i+s})
        
        if actual_c5 < best_c5:
            best_c5 = actual_c5
            best_h = current_h.copy()
        
        # early stopping if time is almost up
        if time.time() - start_time > 0.8 * budget_s:
            break
    
    # final refinement with iterative pairwise exchange
    if best_h is not None:
        refinement_iter = 0
        refinement_steps = 3000
        current_h = best_h.copy()  # work with a copy
        current_c5 = evaluate_c5(current_h)
        
        # compute initial full correlation
        corr = compute_correlation_fft(current_h)
        # get the worst overlap shifts
        max_corr_idx = np.argmax(corr)
        worst_shifts = [max_corr_idx - len(current_h) + 1, max_corr_idx - len(current_h) + 1]
        
        # try to make incremental improvement
        for _ in range(refinement_steps):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # try random pair-wise exchange
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # goal: try to reduce overlap with a given shift
            shift_s = np.random.randint(-n_points+1, n_points)
            
            # compute the contribution of moving mass from i to j for this shift
            # get contributions: h_ij = h[i]*h[j]; h_i_j = h[i]*h[j+s]
            # original output of correlation array
            old_val_i = current_h[i]
            old_val_j = current_h[j]
            
            # try a small delta
            delta = np.random.uniform(-0.01, 0.01)
            # ensure that change is valid (flag to track success)
            valid_move = True
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            # check box constraints
            if new_h[i] < 0 or new_h[i] > 1 or new_h[j] < 0 or new_h[j] > 1:
                valid_move = False
                
            if valid_move:
                # project to feasible set
                new_h = project_onto_sum_box(new_h, target_sum)
                new_c5 = evaluate_c5(new_h)
                if new_c5 < current_c5:
                    current_h = new_h.copy()
                    current_c5 = new_c5
        
        # recompute final C5
        best_c5 = evaluate_c5(current_h)
        best_h = current_h
        
    # One final check to ensure sum constraint is appropriately enforced
    best_h = project_onto_sum_box(best_h, target_sum)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
