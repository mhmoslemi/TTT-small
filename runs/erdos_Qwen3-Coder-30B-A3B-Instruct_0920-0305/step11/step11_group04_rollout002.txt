import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant using the exact formula"""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a modest discretization that keeps the MILP tractable but still gives good bounds
    n = 100
    dx = 2.0 / n
    target_sum = n // 2
    
    # Build the MILP model
    h = cp.Variable(n, boolean=True)
    z = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == target_sum]
    
    # Create auxiliary variables for products h[i] * h[i+k]
    # We only need to consider k >= 0 due to symmetry, but create for full range
    y = {}
    for k in range(-(n-1), n):
        y[k] = cp.Variable((n - abs(k)), nonneg=True)
    
    # Add McCormick constraints
    for k in range(-(n-1), n):
        if k >= 0:
            # For k >= 0, y[i,k] represents h[i] * h[i+k]
            for i in range(n - k):
                # y[i,k] <= h[i]
                constraints.append(y[k][i] <= h[i])
                # y[i,k] <= h[i+k] 
                constraints.append(y[k][i] <= h[i+k])
                # y[i,k] >= h[i] + h[i+k] - 1
                constraints.append(y[k][i] >= h[i] + h[i+k] - 1)
        else:
            # For k < 0, k = -j and we have y[i,-j] = h[i] * h[i-j] 
            # This is equivalent to y[i+j,j] = h[i] * h[i+j] so we use the same entries
            for i in range(n + k):
                # y[i,-k] <= h[i]
                constraints.append(y[k][i] <= h[i])
                # y[i,-k] <= h[i-k]
                constraints.append(y[k][i] <= h[i-k])
                # y[i,-k] >= h[i] + h[i-k] - 1
                constraints.append(y[k][i] >= h[i] + h[i-k] - 1)
    
    # Add correlation constraints
    # For each shift k, compute correlation value
    for k in range(-(n-1), n):
        if k >= 0:
            rho_k = dx * cp.sum(y[k])
        else:
            # Symmetric case
            rho_k = dx * cp.sum(y[k])
        constraints.append(z >= 1 - rho_k)
    
    # Solve the problem
    prob = cp.Problem(cp.Minimize(z), constraints)
    
    # Use a reasonable time limit
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        solver_time = remaining_time - 10
        try:
            prob.solve(solver=cp.CBC, TimeLimit=solver_time, warm_start=True)
        except:
            # Fallback to a different approach
            # Try using a gradient-based approach instead
            return fallback_approach(start_time, budget_s)
    else:
        return fallback_approach(start_time, budget_s)
    
    # Extract the solution
    h_opt = h.value + 0.5  # Convert to float and avoid exact 0/1 if needed
    h_opt = np.clip(h_opt, 0, 1)
    
    # Apply final polishing
    h_opt = polish_solution(h_opt, start_time, budget_s)
    
    c5_bound = evaluate_erdos_solution(h_opt)
    return (h_opt.tolist(), float(c5_bound), n)

def polish_solution(h_opt, start_time, budget_s):
    """Perform greedy swap local search to refine the solution"""
    # Calculate correlations using full convolution
    h = np.array(h_opt)
    corr = np.correlate(h, h, mode='full')
    dx = 2.0 / len(h)
    corr = corr * dx
    
    # Find the worst correlation
    min_idx = np.argmin(corr)
    k = min_idx - len(h) + 1  # -k corresponds to shift k
    # For min_idx=(len(h)-1) (the center index), k=0
    # For min_idx=0, k=1-2*N=-len(h)+1  
    # Adjust to proper shift value
    if min_idx < len(h):
        k_actual = 1 - min_idx 
    else:
        k_actual = min_idx - len(h) + 1

    # Try to improve by local swaps
    max_swaps = 100
    for _ in range(max_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Recompute correlations to get accurate current one
        corr = np.correlate(h, h, mode='full') 
        corr = corr * dx
        min_idx = np.argmin(corr)
        k = min_idx - len(h) + 1
        
        # Find best gain/loss indices
        best_gain = -np.inf
        swap_i = None
        swap_j = None
        
        for i in range(len(h)):
            if h[i] < 0.5:  # h[i] = 0
                gain = 0
                if i + k >= 0 and i + k < len(h):
                    gain += h[i+k]
                if i - k >= 0 and i - k < len(h):
                    gain += h[i-k]
                for j in range(len(h)):
                    if h[j] > 0.5:  # h[j] = 1
                        loss = 0
                        if j + k >= 0 and j + k < len(h):
                            loss += h[j+k]
                        if j - k >= 0 and j - k < len(h):
                            loss += h[j-k]
                        if gain - loss > best_gain:
                            best_gain = gain - loss
                            swap_i, swap_j = i, j
                        
        if best_gain <= 0:
            break
        else:
            h[swap_i] = 1
            h[swap_j] = 0
    return h
