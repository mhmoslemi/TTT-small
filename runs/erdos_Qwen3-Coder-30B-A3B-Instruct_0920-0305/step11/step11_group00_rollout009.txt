import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    dx = 2.0 / n
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def create_symmetric_spline(n_points, knots, knot_values):
    """Create symmetric cubic spline based on knot positions and values"""
    # Define the half interval knot positions (for x in [0, 1])
    # k = (n_knots-1)/2 
    # (x_j, y_j) with j=0,...,k; then reflect 
    k = (len(knots) - 1) // 2
    x_half = np.array(knots)
    y_half = np.array(knot_values)
    # Construct spline
    f = interp1d(x_half, y_half, kind='cubic', fill_value='extrapolate')
    # Build full domain evaluation points
    x_full = np.linspace(0, 2, n_points)
    y_full = f(x_full)
    # now reflect to get symmetric function
    y_left = y_full[:n_points//2+1]
    y_right = y_full[n_points//2:][::-1]
    y_full = np.append(y_left, y_right[1:])
    # ensure all values are within [0,1]
    y_full = np.clip(y_full, 0, 1)
    return y_full

def solve_with_cma_es(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Define spline knots (symmetric around center)
    n_knots = 7
    # Place knots such that they divide [0,1] into equal spacing
    x_knots = np.linspace(0, 1, n_knots)
    # We only parameterize top half (knots j=0,...,3) since we are symmetric
    n_params = (n_knots + 1) // 2  # 4 parameters
    
    # Sigmoid transform
    def sigmoid(x):
        return 1 / (1 + np.exp(-x))
    
    def eval_spline(z_params):
        # Map parameters to values in [0,1]
        y_params = sigmoid(z_params)
        # Create symmetric spline
        h = create_symmetric_spline(n_points, x_knots, y_params)
        # Scale to make sum = target_sum
        s = np.sum(h)
        if s > 0:
            # Scale and clip to ensure box constraints
            h_scaled = h * target_sum / s
            h_proj = project_to_sum_box(h_scaled, target_sum)
        else:
            h_proj = project_to_sum_box(h, target_sum)
        return h_proj
    
    # CMA-ES parameters
    lambda_ = 4 + int(3 * np.log(n_params))  # Population size  
    mu = lambda_ // 2                         # Number selected
    dim = n_params                          # Dimension of search space
    
    # Initial search distribution
    mean = np.zeros(dim)   # Biased initialization for flat h
    sigma = 2.0            # Initial step size
    C = np.eye(dim)        # Covariance matrix
    
    best_f = float('inf')
    best_z = mean.copy()
    
    max_iter = 300
    iter_count = 0
    
    # Try multiple evaluations first with seed initialization
    seed_z = np.zeros(dim)
    seed_h = eval_spline(seed_z)
    seed_c5 = evaluate_c5(seed_h)
    if seed_c5 < best_f:
        best_f = seed_c5
        best_z = seed_z.copy()
    
    # Some additional deterministic starting points:
    try:
        # Triangular U-shape
        triangular_h = np.array([min(1, i / (n_points//4)) for i in range(n_points//2)] + 
                               [min(1, (n_points - i) / (n_points//4)) for i in range(n_points//2, n_points)])
        triangular_c5 = evaluate_c5(triangular_h)
        if triangular_c5 < best_f:
            best_f = triangular_c5
            # we would do обратное to find mean, but keep current for now
    except Exception:
        pass

    # Main CMA-ES loop
    for iteration in range(max_iter):
        if time.time() - start_time > 0.75 * budget_s:
            break
            
        # Sample population
        offspring = []
        for _ in range(lambda_):
            z = np.random.multivariate_normal(mean, sigma**2 * C)
            offspring.append(z)
        
        # Evaluate
        fitness = []
        for z in offspring:
            h = eval_spline(z)
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Select best mu individuals
        idx = np.argsort(fitness)
        selected_indices = idx[:mu]
        
        # Update mean
        z_selected = [offspring[i] for i in selected_indices]
        z_mean = np.mean(z_selected, axis=0)
        mean = z_mean.copy()
        
        # Update covariance matrix
        if mu > 1:
            # Rank-mu update
            weights = np.log(mu + 1) - np.log(np.arange(1, mu + 1))
            weights = weights / weights.sum()
            # Compute (weighted) difference matrix
            diff = np.array(z_selected) - mean
            C = (1 - 1/lambda_) * C + (1/lambda_) * (diff.T @ diff) * np.eye(dim) / (weights @ weights)
        
        # Update step size sigma
        sigma = sigma * np.exp(1/2 * (1/lambda_) * np.sum(weights * np.ones(mu)) * (1 - np.sqrt(1/lambda_)))
        
        # Update best
        if fitness[selected_indices[0]] < best_f:
            best_f = fitness[selected_indices[0]]
            best_z = z_selected[0].copy()
    
    # Final refinement step around the best CMA-ES solution
    best_h = eval_spline(best_z)
    current_c5 = evaluate_c5(best_h)
    
    # Do a few steps of gradient-based refinement
    max_refine_steps = 1200
    for _ in range(max_refine_steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute worst correlations
        corr = compute_corr_fft(best_h)
        k_max = np.argmax(corr)
        # Compute gradient of correlation at k_max
        g = np.zeros(len(best_h))
        for i in range(len(best_h)):
            j = i + k_max
            if 0 <= j < len(best_h):
                g[i] += 1 - best_h[j]
            j = i - k_max
            if 0 <= j < len(best_h):
                g[i] -= best_h[j]
        
        # Update
        eta = 0.08
        new_h = best_h - eta * g
        new_h = np.clip(new_h, 0, 1)
        # Project onto sum = target_sum
        # Use binary search to find correct scaling factor
        def sigma_func(s):
            h_test = new_h * s
            return np.sum(h_test) - target_sum
        
        # Bisection to find correct scaling factor
        a, b = 0.0, 2.0
        for _ in range(50):
            m = (a + b) / 2
            if abs(sigma_func(m)) < 1e-12:
                break
            if sigma_func(m) * sigma_func(a) < 0:
                b = m
            else:
                a = m
                
        scale_factor = (a + b) / 2
        new_h = np.clip(new_h * scale_factor, 0, 1)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < current_c5:
            best_h = new_h
            current_c5 = new_c5
        else:
            eta *= 0.5  # backtracking on step size
            if eta < 1e-6:
                break
        
        # Periodic random perturbations to escape local minima
        if _ % 200 == 0:
            i1, i2 = np.random.randint(0, len(best_h), 2)
            delta = np.random.uniform(-0.015, 0.015)
            orig = best_h[i2]
            best_h[i2] = np.clip(best_h[i2] - delta, 0, 1)
            best_h[i1] = np.clip(best_h[i1] + delta, 0, 1)
            # Check if this improves:
            test_c5 = evaluate_c5(best_h)
            if test_c5 >= current_c5:
                best_h[i2] = orig  # revert if worse
                best_h[i1] = np.clip(best_h[i1] - delta, 0, 1)
    
    return best_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = solve_with_cma_es(seed, budget_s)
    return (best_h.tolist(), float(best_c5), len(best_h))
