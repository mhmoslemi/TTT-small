import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5_exact(h, dx):
    """Compute exact C5 using full correlation"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_box_sum(x, target_sum):
    """Project x onto [0,1]^n with sum = target_sum using water filling"""
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x_clipped
    
    # Binary search for the right lambda
    def f(lam):
        proj = np.clip(x_clipped + lam, 0, 1)
        return np.sum(proj) - target_sum
    
    a, b = -1.0, 1.0
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x_clipped + (a + b)/2, 0, 1)
    return result

def round_autocorr_to_signal(r, n):
    """Spectral factorization: extract signal from autocorrelation"""
    # Compute power spectrum (DFT of real symmetric sequence)
    S = np.fft.fft(r)
    # Ensure non-negative real values due to PSD property
    S = np.real(S)
    A = np.sqrt(np.maximum(S, 0))  # Element-wise square root
    
    # Try 4 different phases (including zero phase)
    best_h = None
    best_c5 = float('inf')
    
    for phase_seed in [0, 1, 2, 3]:
        np.random.seed(phase_seed)
        # Random phases for better results
        phase = 2 * np.pi * np.random.random(n)
        # For this implementation, we use deterministic approach
        phase = np.zeros(n) if phase_seed == 0 else (phase_seed * np.pi/4) * np.ones(n)
        
        # Inverse FFT with phase
        h_candidate = np.fft.ifft(A * np.exp(1j * phase)).real
        
        # Project to feasible set
        h_projected = project_to_box_sum(h_candidate, n/2)
        
        # Evaluate C5
        c5 = evaluate_c5_exact(h_projected, 2.0/n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_projected
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n = 80  # Initial size - can be extended if time allows
    dx = 2.0 / n
    M = n / 2  # Target sum
    
    # Build and solve SDP
    r = cp.Variable(n)
    t = cp.Variable()
    
    # Toeplitz matrix
    T = cp.bmat([[r[abs(i-j)] for j in range(n)] for i in range(n)])
    
    # Constraints
    constraints = [
        T >> 0,  # PSD constraint
        cp.sum((n - cp.arange(n)) * r) == M**2,  # Mass constraint
        r >= t,  # Lower bound on all autocorrelations
        r >= 0,  # Non-negativity
        r <= cp.minimum(M, n - cp.arange(n))  # Box constraints
    ]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS solver (free)
    try:
        prob.solve(solver=cp.SCS, max_iter=5000, verbose=False)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            r_opt = r.value
            t_opt = t.value
        else:
            # Fallback: use heuristic gradient descent from uniform
            h0 = np.full(n, 0.5)
            h0 = project_to_box_sum(h0, M)
            return (h0.tolist(), evaluate_c5_exact(h0, dx), n)
    except:
        # Fallback to heuristics
        h0 = np.full(n, 0.5)
        h0 = project_to_box_sum(h0, M)
        return (h0.tolist(), evaluate_c5_exact(h0, dx), n)
    
    # Round to extract signal
    h_candidate, c5_candidate = round_autocorr_to_signal(r_opt, n)
    
    # Local refinement with projected gradient ascent
    max_iter = 50
    eta0 = 0.2
    h_refined = h_candidate.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current minimal correlation
        corr = correlate(h_refined, 1-h_refined, mode="full")
        min_corr_idx = np.argmin(corr)
        min_corr_value = corr[min_corr_idx]
        
        # Compute gradient
        grad = np.zeros_like(h_refined)
        k = min_corr_idx - (n-1)  # shift index
        if k >= 0:
            grad[k:] += h_refined[:-k]
        if k <= 0:
            grad[:k] += h_refined[-k:]
            
        # Update with step size
        eta = eta0 / np.sqrt(i+1)
        h_new = h_refined + eta * grad
        h_new = project_to_box_sum(h_new, M)
        h_refined = h_new
        
        # Early stopping if improvement is small
        if i > 20:
            c5_new = evaluate_c5_exact(h_refined, dx)
            if abs(c5_new - evaluate_c5_exact(h_candidate, dx)) < 1e-6:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5_exact(h_refined, dx)
    
    # If we have time left, try larger n
    if (time.time() - start_time < budget_s - 200) and n == 80:
        # Try larger n
        n_large = 100
        dx_large = 2.0 / n_large
        M_large = n_large / 2
        
        # Re-solve SDP with larger n using same approach
        r_large = cp.Variable(n_large)
        t_large = cp.Variable()
        
        # Toeplitz matrix
        T_large = cp.bmat([[r_large[abs(i-j)] for j in range(n_large)] for i in range(n_large)])
        
        # Constraints
        constraints_large = [
            T_large >> 0,
            cp.sum((n_large - cp.arange(n_large)) * r_large) == M_large**2,
            r_large >= t_large,
            r_large >= 0,
            r_large <= cp.minimum(M_large, n_large - cp.arange(n_large))
        ]
        
        # Objective
        objective_large = cp.Maximize(t_large)
        
        # Problem
        prob_large = cp.Problem(objective_large, constraints_large)
        
        try:
            prob_large.solve(solver=cp.SCS, max_iter=5000, verbose=False)
            if prob_large.status == cp.OPTIMAL or prob_large.status == cp.OPTIMAL_INACCURATE:
                r_opt_large = r_large.value
                t_opt_large = t_large.value
                
                # Round and refine with larger n
                h_large, c5_large = round_autocorr_to_signal(r_opt_large, n_large)
                
                # Local refinement
                max_iter_large = 30
                eta0_large = 0.2
                h_refined_large = h_large.copy()
                
                for i in range(max_iter_large):
                    if time.time() - start_time > budget_s - 10:
                        break
                        
                    corr_large = correlate(h_refined_large, 1-h_refined_large, mode="full")
                    min_corr_idx_large = np.argmin(corr_large)
                    min_corr_value_large = corr_large[min_corr_idx_large]
                    
                    grad_large = np.zeros_like(h_refined_large)
                    k_large = min_corr_idx_large - (n_large-1)
                    if k_large >= 0:
                        grad_large[k_large:] += h_refined_large[:-k_large]
                    if k_large <= 0:
                        grad_large[:k_large] += h_refined_large[-k_large:]
                        
                    eta_large = eta0_large / np.sqrt(i+1)
                    h_new_large = h_refined_large + eta_large * grad_large
                    h_new_large = project_to_box_sum(h_new_large, M_large)
                    h_refined_large = h_new_large
                    
                    if i > 20:
                        c5_new_large = evaluate_c5_exact(h_refined_large, dx_large)
                        if abs(c5_new_large - evaluate_c5_exact(h_large, dx_large)) < 1e-6:
                            break
                
                final_c5_large = evaluate_c5_exact(h_refined_large, dx_large)
                
                if final_c5_large < final_c5:
                    final_c5 = final_c5_large
                    h_refined = h_refined_large
                    n = n_large
                    dx = dx_large
                    
        except:
            pass
    
    return (h_refined.tolist(), float(final_c5), int(n))
