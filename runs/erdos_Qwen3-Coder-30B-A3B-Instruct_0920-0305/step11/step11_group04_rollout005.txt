import numpy as np
import cvxpy as cp
from scipy.linalg import eigh
from scipy.sparse import diags
import time

def build_shift_matrices(n):
    """Build shift matrices S_k for all valid shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create Toeplitz matrix for shift k
        offsets = [k]
        data = [1]
        shape = (n, n)
        S_k = diags(data, offsets, shape=shape, format='csr')
        shift_matrices.append(S_k)
    return shift_matrices

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    # Ensure h is numpy array
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # Compute correlation with reversed h
    corr = np.correlate(h, 1 - h, mode="full")
    # Return maximum scaled by dx
    return dx * np.max(corr)

def solve_sdp(n, shift_matrices, target_sum, max_time=400):
    """Solve SDP relaxation"""
    # Variable definition
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD constraint
        cp.diag(X) == 1,  # Diagonal constraint
        cp.trace(X) == n,  # Trace constraint (balanced in terms of sum)
    ]
    
    # Add shift constraints
    for k, S_k in enumerate(shift_matrices):
        # Compute number of valid positions for this shift
        n_k = min(n, n + k) if k >= 0 else min(n, n - k)
        # Trace constraint
        constraints.append(cp.trace(S_k @ X) >= 2 * n * (1 - t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem definition
    prob = cp.Problem(objective, constraints)
    
    # Solve with SCS
    start_time = time.time()
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4, verbose=False)
        end_time = time.time()
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return X.value, t.value, end_time - start_time
    except Exception as e:
        pass
    return None, None, time.time() - start_time

def round_solution(X_val, n, target_sum, R=100):
    """Round SDP solution to binary vectors"""
    # Compute eigenvalue decomposition
    try:
        eigenvals, eigenvecs = eigh(X_val)
        # Keep only positive eigenvalues
        mask = eigenvals > 1e-8
        eigenvals = eigenvals[mask]
        eigenvecs = eigenvecs[:, mask]
        
        # Compute square root of eigenvalues
        sqrt_eigenvals = np.sqrt(eigenvals)
        V_sqrt = eigenvecs @ np.diag(sqrt_eigenvals)
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(R):
            # Sample random vector
            g = np.random.randn(V_sqrt.shape[1])
            # Compute signs
            signs = np.sign(V_sqrt.T @ g)
            # Convert to binary (0 or 1)
            h = (signs + 1) / 2
            
            # Adjust to satisfy sum constraint
            current_sum = np.sum(h)
            if current_sum < target_sum:
                # Need to flip some zeros to ones
                diff = int(target_sum - current_sum)
                indices = np.argsort(np.abs(V_sqrt.T @ g))[:diff]
                for idx in indices:
                    if h[idx] == 0:
                        h[idx] = 1
            elif current_sum > target_sum:
                # Need to flip some ones to zeros
                diff = int(current_sum - target_sum)
                indices = np.argsort(np.abs(V_sqrt.T @ g))[-diff:]
                for idx in indices:
                    if h[idx] == 1:
                        h[idx] = 0
            
            # Evaluate C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                
        return best_h, best_c5
    except Exception:
        return None, float('inf')

def local_improvement(h, max_iter=500):
    """Perform local improvement on binary vectors"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for i in range(max_iter):
        # Find k that gives maximum overlap
        _, k_optimal = find_best_shift(current_h)
        
        # Try to make improvements
        best_gain = 0
        best_pair = None
        
        # Consider flipping pairs
        for i in range(len(current_h)):
            if current_h[i] == 0:  # Try to flip to 1
                # Calculate change in overlap
                new_h = current_h.copy()
                new_h[i] = 1
                new_c5 = evaluate_c5(new_h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, 1)
            else:  # Try to flip to 0
                # Calculate change in overlap
                new_h = current_h.copy()
                new_h[i] = 0
                new_c5 = evaluate_c5(new_h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, 0)
        
        if best_pair and best_gain > 0:
            idx, val = best_pair
            current_h[idx] = val
            current_c5 = evaluate_c5(current_h)
        else:
            break
            
    return current_h, current_c5

def find_best_shift(h):
    """Find the shift that maximizes overlap"""
    h = np.array(h)
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr_idx = np.argmax(corr)
    k = max_corr_idx - (n - 1)
    return corr[max_corr_idx], k

                best_h = h.copy()
                
        # Selection (tournament selection)
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx].copy()
            
            # Crossover
            if np.random.rand() < 0.8:
                other_idx = np.random.choice(population_size)
                other_parent = population[other_idx]
                # Uniform crossover that preserves sum
                mask = np.random.rand(n) < 0.5
                child = np.where(mask, parent, other_parent)
                
                # Repair to ensure exactly target_sum ones
                current_sum = np.sum(child)
                if current_sum < target_sum:
                    diff = int(target_sum - current_sum)
                    zero_indices = np.where(child == 0)[0]
                    to_flip = np.random.choice(zero_indices, diff, replace=False)
                    child[to_flip] = 1
                elif current_sum > target_sum:
                    diff = int(current_sum - target_sum)
                    one_indices = np.where(child == 1)[0]
                    to_flip = np.random.choice(one_indices, diff, replace=False)
                    child[to_flip] = 0
                    
                parent = child
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Flip some bits
                to_flip = np.random.choice(n, int(mutation_rate * n), replace=False)
                parent[to_flip] = 1 - parent[to_flip]
                
            new_population.append(parent)
            
        population = new_population
        
        if time.time() - start_time >= budget_s * 0.8:
            break
            
    return best_h, best_c5
