def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max(corr, alpha):
    """Compute smooth max of correlations"""
    return (1/alpha) * np.log(np.sum(np.exp(alpha * corr)))

def grad_smooth_max(corr, alpha):

    w = np.exp(alpha * corr)

    n = len(corr) // 2 + 1  # length of original signal for auto-correlation

    L = 2 * n - 1  # full correlation length = 2*n - 1
    # We shift correlation indices back to match the -n+1 to n-1 range
    # For now, we'll take a different approach: 
    # correlation already ordered with shift=0 at center 
    # So we need:
    #   convolve(w, h) = (w * h)[j] = sum_k w[k] h[j-k] 
    #   convolve(w_rev, h) = (w_rev * h)[j] = sum_k w_rev[k] h[j-k] 
    # But w_rev in numpy means in reverse but we must be careful of indexing
    # It's easier to compute with the appropriate padding.
    # Pad w to same length as corr then reverse
    w_pad = np.zeros(L)
    w_pad[:len(w)] = w
    
    # We don't need to reverse w_pad since weights are symmetric by definition.
    # Instead construct an inverse permutation or just use the actual convolution.
    # A more direct way:
    # In the future we'll compute FFT of corr and whatnot. 
    # Simpler: reorder to be centered-ed, manually convolve with FFT
    # Use the original way: To be safe, just reconstruct the part useful for linear problem.
    # For large-scale problems like this, more directly:
    # The gradient approach relying on:
    # shift of i in correlation matrix.
    # Let's simplify it using the iterative derivative and not worry about trivial proof:
    # ∂ smooth_max / ∂ h[m] can be computed directly.
    # since f_k = sum_i h_i * (1 - h_{i+k}), 
    # ∂ f_k / ∂ h_i = (1 - h_{i+k}) - h_{i-k}, 
    # for k < 0 it's the convolution of w with (1-h) + w with h
    # Let me take a known fixed approach like convolution of sp. matrix
    
    return 1.0 - correlate(w, h, mode='full')[:len(h)] - correlate(w[::-1], h, mode='full')[:len(h)]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b, mode='full'):
    """Compute convolution with FFT, padding to a safe size"""
    L = len(a) + len(b) - 1
    L_fft = 1
    while L_fft < L:
        L_fft *= 2
    A = fft(a, L_fft)
    B = fft(b, L_fft)
    C = A * B
    c = ifft(C).real
    if mode == 'full':
        return c[:L]
    elif mode == 'same':
        return c[(len(b)-1)//2 : (len(b)-1)//2 + len(a)]
    else:  # 'valid'
        return c[len(b)-1 : len(a)]
    
def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)
