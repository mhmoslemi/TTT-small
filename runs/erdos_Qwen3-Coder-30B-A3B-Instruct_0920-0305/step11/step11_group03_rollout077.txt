import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def project_onto_feasible_set(h, target_sum):
    """Project h onto the feasible set: 0 <= h <= 1 and sum(h) = target_sum"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust via binary search for the right sum
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        h_test = np.clip(h + m, 0, 1)
        s_test = np.sum(h_test)
        if abs(s_test - target_sum) < 1e-12:
            return h_test
        if s_test < target_sum:
            a = m
        else:
            b = m
            
    return np.clip(h + (a + b)/2, 0, 1)

def softmin_weights(R, beta):
    """Compute soft-minimum weights for the given autocorrelations and temperature"""
    exp_R = np.exp(-beta * R)
    weights = exp_R / np.sum(exp_R)
    return weights

def gradient_via_fft(h, weights):
    """Compute gradient of softmin using FFT-based convolution trick"""
    # weights is the soft-weight vector for each shift k
    n = len(h)
    # sign convention: negative gradient
    weights_rev = weights[::-1]
    
    # Forward shift part (h_{i+k})
    conv_forward = np.convolve(weights, h, mode='full')
    # Backward shift part (h_{i-k})
    conv_backward = np.convolve(weights_rev, h, mode='full')
    
    grad = - (conv_forward[n-1:2*n-1] + conv_backward[0:n])
    return grad

def compute_autocorrelations(h):
    """Compute full autocorrelation via FFT"""
    n = len(h)
    H = fft(h, 2*n-1)
    R_full = ifft(H * np.conj(H)).real
    return R_full[:n]

    corr_values = corr[len(h)-1+len(valid_k):len(h)-1+len(valid_k)+len(valid_k)]
    return np.max(corr_values) * dx
