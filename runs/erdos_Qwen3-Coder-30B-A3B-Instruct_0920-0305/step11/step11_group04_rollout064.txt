import numpy as np
from scipy.linalg import eigh
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import trace, Variable, Minimize, Problem, Semidef
from scipy.sparse import diags
from scipy.sparse import csr_matrix
import random

def build_shift_matrices(n):
    """Build shift matrices for all valid shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create sparse diagonal matrix for shift k
        offsets = [k]
        data = [1.0]
        shift_mat = diags(data, offsets, shape=(n, n), format='csr')
        shift_matrices.append(shift_mat)
    return shift_matrices

def evaluate_c5(h):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def solve_sdp(n, shift_matrices, max_time=400):
    """Solve SDP relaxation to find optimal h"""
    # Define variables
    X = Semidef(n, name='X')
    t = Variable(name='t')
    
    # Constraints
    constraints = [
        cp.diag(X) == 1,
        trace(X) == n,  # Normalize trace to n
    ]
    
    # Add shift constraints for all shifts
    for k, S_k in enumerate(shift_matrices):
        # n_k is the effective size for this shift
        shift_idx = k - (n-1)
        n_k = n - abs(shift_idx)
        constraints.append(trace(S_k @ X) >= 2*n*(1 - t) - n_k)
    
    # Objective
    objective = Minimize(t)
    
    # Create and solve problem
    prob = Problem(objective, constraints)
    
    try:
        start_time = time.time()
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        elapsed = time.time() - start_time
        
        if prob.status == 'optimal' and elapsed < max_time:
            return X.value, t.value
    except:
        pass
    
    return None, None

def random_rounding(X, n, R=100):
    """Perform random hyperplane rounding on PSD matrix X"""
    # Eigenvalue decomposition
    try:
        eigenvals, eigenvecs = eigh(X)
        # Keep only positive eigenvalues
        positive_mask = eigenvals > 1e-8
        eigenvals = eigenvals[positive_mask]
        eigenvecs = eigenvecs[:, positive_mask]
        
        # Form L matrix
        L = eigenvecs @ np.diag(np.sqrt(eigenvals))
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(R):
            g = np.random.randn(L.shape[1])
            x = np.sign(L.T @ g)
            h = (x + 1) / 2
            
            # Adjust to ensure sum is n/2
            current_sum = np.sum(h)
            if current_sum > n / 2:
                # Find indices with smallest values to flip to 0
                flip_indices = np.argsort(h)[:int(current_sum - n/2)]
                h[flip_indices] = 0
            elif current_sum < n / 2:
                # Find indices with largest values to flip to 1
                flip_indices = np.argsort(h)[int(n/2 - current_sum):]
                h[flip_indices] = 1
                
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    except:
        return None, float('inf')

def local_swap_improvement(h, max_iter=1000):
    """Improve h using local swaps"""
    h = h.copy()
    current_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Find k that produces current max overlap
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_best = max_idx - len(h) + 1
        
        # Simulate potential swaps
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        for i in range(len(h)):
            if h[i] == 0:  # Try to set to 1
                # Compute potential change to max overlap for this shift
                change = 0
                for k in range(len(h)):
                    if i + k < len(h) and (k - k_best) >= 0:
                        change -= h[i + k] * (1 - h[i + k + k_best])
                if change < best_improvement:
                    best_improvement = change
                    best_i = i
            elif h[i] == 1:  # Try to set to 0
                # Compute potential change to max overlap for this shift
                change = 0
                for k in range(len(h)):
                    if i + k < len(h) and (k - k_best) >= 0:
                        change += h[i + k] * (1 - h[i + k + k_best])
                if change < best_improvement:
                    best_improvement = change
                    best_j = i
        
        if best_improvement < 0:
            if best_i != -1:
                h[best_i] = 1
            if best_j != -1:
                h[best_j] = 0
            current_c5 = evaluate_c5(h)
        else:
            break
            
    return h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n = 128
    dx = 2.0 / n
    target_sum = n / 2
    
    # Build shift matrices
    shift_matrices = build_shift_matrices(n)
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # First attempt: SDP approach
    print("Attempting SDP approach...")
    try:
        X_val, t_val = solve_sdp(n, shift_matrices, max_time=400)
        if X_val is not None:
            # Round to get candidate solution
            h_candidate, c5_candidate = random_rounding(X_val, n, R=100)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                print(f"SDP found C5 = {best_c5}")
    except Exception as e:
        print(f"SDP failed: {e}")
    
    # Second attempt: Local search from uniform random
    if time.time() - start_time >= budget_s * 0.9:
        return (best_h.tolist(), float(best_c5), n) if best_h is not None else (np.zeros(n).tolist(), float('inf'), n)
    
    print("Starting local search from random...")
    # Random initialization
    h0 = np.random.rand(n)
    h0 = np.clip(h0, 0, 1)
    h0 = project_to_sum_box(h0, target_sum)
    
    current_h = h0.copy()
    current_c5 = evaluate_c5(h0)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Local search
    iter_count = 0
    max_iter = 5000
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Try small random perturbations
        idx = np.random.randint(0, n)
        delta = np.random.uniform(-0.05, 0.05)
        h_test = current_h.copy()
        h_test[idx] += delta
        
        # Clamp to [0, 1]
        h_test = np.clip(h_test, 0, 1)
        h_test = project_to_sum_box(h_test, target_sum)
        
        test_c5 = evaluate_c5(h_test)
        
        if test_c5 < current_c5:
            current_h = h_test.copy()
            current_c5 = test_c5
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = current_h.copy()
        
        iter_count += 1
    
    # Final polishing
    if best_h is not None:
        final_h, final_c5 = local_swap_improvement(best_h, 1000)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    return (best_h.tolist(), float(best_c5), n) if best_h is not None else (np.zeros(n).tolist(), float('inf'), n)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result
