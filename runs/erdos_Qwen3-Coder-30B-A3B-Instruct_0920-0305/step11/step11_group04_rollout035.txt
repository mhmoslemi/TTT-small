def build_shift_matrices(n):
    """Build shift matrices for all possible shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create diagonal matrix with 1s shifted by k
        offsets = [k]
        data = [np.ones(max(0, n - abs(k)))]
        shift_matrices.append(diags(data, offsets, shape=(n, n), format='csr'))
    return shift_matrices

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

        eigenvecs = eigenvecs[:, mask]
        
        # Compute square root of eigenvalues
        sqrt_eigenvals = np.sqrt(eigenvals)
        L = eigenvecs * sqrt_eigenvals
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(R):
            # Sample random vector
            g = np.random.randn(L.shape[1])
            
            # Compute signs
            signs = np.sign(L.T @ g)
            signs = np.where(signs == 0, 1, signs)  # Avoid zeros
            
            # Transform to h values
            h = (signs + 1) / 2
            
            # Adjust to target sum
            current_sum = np.sum(h)
            if current_sum != n / 2:
                # Find indices with smallest absolute values (closest to 0.5)
                if current_sum > n / 2:
                    # Need to flip some 1s to 0s
                    flip_indices = np.argsort(np.abs(h - 0.5))[:int(current_sum - n/2)]
                    h[flip_indices] = 0
                else:
                    # Need to flip some 0s to 1s
                    flip_indices = np.argsort(np.abs(h - 0.5))[:int(n/2 - current_sum)]
                    h[flip_indices] = 1
            
            # Evaluate C5
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        return best_h, best_c5
    except:
        return None, float('inf')

def solve_sdp(n, shift_matrices, budget_s, start_time):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # PSD constraint
        cp.diag(X) == 1,  # Diagonal constraints
        cp.trace(cp.ones((n, n)) @ X) == 0  # Balanced constraint
    ]
    
    # Shift constraints
    for k, S_k in enumerate(shift_matrices):
        # The actual trace constraint depends on the shift size
        # This is: trace(S_k @ X) >= 2*n*(1-t) - n_k
        # For shift k, n_k = max(0, n - abs(k))
        n_k = max(0, n - abs(k - (len(shift_matrices) // 2)))
        constraints.append(cp.trace(S_k @ X) >= 2 * n * (1 - t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
    except:
        pass
    
    return None, None
