import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_onto_feasible_set(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum using water-filling method."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    
    def f(lam):
        projected = np.clip(h + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    for _ in range(30):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b)/2, 0, 1)
    return result

def compute_autocorrelations(h):
    """Compute all autocorrelations R_k using FFT."""
    n = len(h)
    H = np.fft.fft(h, 2*n-1)
    R_full = np.fft.ifft(H * np.conj(H)).real
    # Return only non-negative lags
    return R_full[n-1:]

def softmin_weights(R, beta):
    """Compute softmax weights for softmin."""
    exp_R = np.exp(-beta * R)
    return exp_R / np.sum(exp_R)

def compute_gradient_via_fft(h, w):
    """Compute gradient using convolution."""
    n = len(h)
    w_rev = w[::-1]
    # Forward shifts contribute to indices n-1 to 2*n-2
    # Backward shifts contribute to indices 0 to n-1
    grad = np.zeros(n)
    grad += np.convolve(w, h, mode='full')[n-1:2*n-1]  # Forward shifts
    grad += np.convolve(w_rev, h, mode='full')[0:n]    # Backward shifts
    return -grad

def softmin_objective(h, beta):
    """Compute softmin objective 1 - softmin(R)."""
    R = compute_autocorrelations(h)
    exp_R = np.exp(-beta * R)
    return np.log(np.sum(exp_R)) / beta
