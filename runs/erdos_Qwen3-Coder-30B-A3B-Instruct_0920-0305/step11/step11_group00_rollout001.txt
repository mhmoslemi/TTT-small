import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def grad_shift(h, s):
    """Compute gradient for shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n//2
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        h = np.concatenate([first, first[::-1]])
    else:
        h = np.concatenate([first, first[::-1][1:]])
    
    # Adjust to meet sum constraint exactly
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def random_feasible_vector(n, target_sum):
    """Generate random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h

    """Smooth gradient optimization"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Parameters
    alphas = np.linspace(0.5, 30, 15)
    step = 0.5
    max_step = 1.0
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for iter in range(200):
            if time.time() - start_time > time_limit:
                break
            c = compute_corr(h)
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (len(h) - 1)
                grad += weight * grad_shift(h, s)
            
            # Adaptive step size
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    step = min(step * 1.1, max_step)
                    break
                else:
                    step_cur *= 0.5
                    step = max(step * 0.5, 0.001)
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Subgradient optimization with multiple restarts"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Early stopping criteria
    patience = 50
    patience_counter = 0
    prev_val = best_val
    
    for iter in range(500):
        if time.time() - start_time > time_limit:
            break
            
        c = compute_corr(h)
        max_c = np.max(c)
        indices = np.where(np.abs(c - max_c) < 1e-8)[0]
        
        # Average gradient over all max indices
        grad = np.zeros_like(h)
        for idx in indices:
            s = idx - (len(h) - 1)
            grad += grad_shift(h, s)
        grad /= len(indices)
        
        # Adaptive step size
        step_size = 0.2
        h_new = h - step_size * grad
        h_new = project_to_sum_box(h_new, target_sum)
        val_new = c5_value(h_new)
        
        if val_new < best_val - 1e-12:
            h = h_new
            best_val = val_new
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Random perturbation to escape plateau
                noise = np.random.normal(0, 0.01, len(h))
                h_perturbed = h + noise
                h_perturbed = project_to_sum_box(h_perturbed, target_sum)
                val_perturbed = c5_value(h_perturbed)
                if val_perturbed < best_val - 1e-12:
                    h = h_perturbed
                    best_val = val_perturbed
                    best_h = h.copy()
                    patience_counter = 0
                    
        if abs(prev_val - best_val) < 1e-10:
            patience_counter += 1
        else:
            patience_counter = 0
        prev_val = best_val
    
    return best_h, best_val
