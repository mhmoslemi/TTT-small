import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (lam_low + lam_high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = m
        else:
            lam_low = m
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def compute_gradient_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def compute_smooth_max_grad(h, alpha):
    """Compute gradient of smooth max surrogate"""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    
    # Safe exponentiation
    exp_arg = alpha * (corr - max_corr)
    exp_vals = np.exp(exp_arg)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        if w > 1e-15:  # Avoid numerical issues
            k = i - (n_points - 1)
            grad_k = compute_gradient_shift(h, k)
            grad += w * grad_k
    
    return grad

def backtracking_line_search(h, grad, alpha, eta, tol=1e-6):
    """Simple backtracking line search"""
    c5_current = evaluate_c5(h)
    max_corr_current = np.max(compute_corr(h))

        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_current - tol:
            return h_new, eta
        eta *= 0.5
    return h, eta

        # Nesterov update

def subgradient_polish(h, max_iter=1500):

    eta = 0.08
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for i in range(max_iter):
        corr = compute_corr(h)
        max_corr = np.max(corr)
        K_max = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        if len(K_max) == 0:
            break
            
        # Average gradient of active shifts
        grad = np.zeros_like(h)
        for k in K_max:
            k_shift = k - (n_points - 1)
            grad_k = compute_gradient_shift(h, k_shift)
            grad += grad_k
        grad /= len(K_max)
        
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            if eta < 0.15:
                eta *= 1.05
        else:
            eta *= 0.5
        
        # Random swap move

            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = 0.015
            h_new[i1] -= delta
            h_new[i2] += delta
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        h = h_new
    
    return best_h, best_c5
