```python
import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Pre-imported global variables
initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 120
    dx = 2.0 / n_points
    TARGET_SUM = n_points / 2
    n_knots = 7  # must be odd for symmetry
    n_params = (n_knots + 1) // 2  # number of independent parameters
    
    # Knot positions (symmetric about 1)
    knot_positions = np.linspace(0, 1, (n_knots + 1) // 2)
    
    # Sigmoid function to map reals to [0,1]
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def project_to_sum_box(h, target_sum):
        """Project h to sum=target_sum, keeping in [0,1]"""
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            return h
        alpha = target_sum / s
        h_new = alpha * h
        # Clip to stay in [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Adjust if needed due to clipping
        s_new = np.sum(h_new)
        if abs(s_new - target_sum) > 1e-10:
            # Binary search for correct alpha
            left, right = 0.0, 2.0 * target_sum / s
            for _ in range(50):
                mid = (left + right) / 2
                h_test = np.clip(mid * h, 0, 1)
                s_test = np.sum(h_test)
                if abs(s_test - target_sum) < 1e-10:
                    return h_test
                elif s_test < target_sum:
                    left = mid
                else:
                    right = mid
            h_test = np.clip((left + right) / 2 * h, 0, 1)
            return h_test
        return h_new

    def eval_h_from_params(params):
        """Evaluate h from parameters using spline"""
        # Map parameters via sigmoid to [0,1]
        y_vals = sigmoid(params)
        # Build full h vector with symmetry
        # left half including middle point
        y_left = y_vals
        # right half (mirror, excluding middle)
        y_right = y_vals[-2::-1]  
        y_full = np.concatenate([y_left, y_right[1:]])
        
        # Generate spline
        spline = CubicSpline(knot_positions, y_full[:len(knot_positions)], bc_type='clamped')
        
        # Interpolate at uniform grid points
        x_eval = np.linspace(0, 2, n_points)
        h = spline(x_eval)
        
        # Enforce sum constraint
        h = project_to_sum_box(h, TARGET_SUM)
        
        return h
    
    def compute_corr_fft(h):
        """Compute correlation using FFT"""
        h_rev = h[::-1]
        b = 1 - h
        L = 2 * len(h) - 1
        corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
        return corr
    
    def evaluate_c5(h):
        """Fast evaluation of C5 using FFT"""
        corr = compute_corr_fft(h)
        return dx * np.max(corr)
    
    # CMA-ES implementation
    def cma_es_optimize():
        # Simple sub-optimal CMA-ES component for demonstration
        # Use a simplified version that explores parameter space with few iterations
        best_c5 = np.inf
        best_params = None
        best_h = None
        
        # Initialize
        population_size = 4 + int(3 * np.log(n_params))
        max_evals = 500
        time_limit = 0.7 * budget_s
        
        # Start with flat profile (middle of [0,1] for sigmoid)
        initial_params = np.zeros(n_params)
        
        current_best_params = initial_params.copy()
        current_best_h = eval_h_from_params(current_best_params)
        current_best_c5 = evaluate_c5(current_best_h)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_params = current_best_params.copy()
            best_h = current_best_h.copy()
        
        # Random exploration for few steps
        for _ in range(min(max_evals, 100)):
            if time.time() - start_time > time_limit:
                break
            params = initial_params + np.random.normal(0, 1.0, n_params)
            try:
                h = eval_h_from_params(params)
                c5 = evaluate_c5(h)
                if c5 < current_best_c5:
                    current_best_c5 = c5
                    current_best_params = params.copy()
                    current_best_h = h.copy()
                if c5 < best_c5:
                    best_c5 = c5
                    best_params = params.copy()
                    best_h = h.copy()
            except:
                continue
        
        return best_params, best_c5, best_h
    
    # Initial optimization
    params, c5_bound, h_values = cma_es_optimize()
    
    # Refinement step: local optimization on the full space
    def refine_solution(h_start):
        h_current = h_start.copy()
        iterations = min(1200, int(2000 * (budget_s - (time.time() - start_time)) / budget_s))
        
        for i in range(iterations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation to find worst k
            corr = compute_corr_fft(h_current)
            max_corr = np.max(corr)
            K_max = np.where(np.abs(corr - max_corr) < 1e-12)[0]
            
            # Compute average gradient for worst k
            g = np.zeros_like(h_current)
            for k in K_max:
                # Compute grad at k
                if k == 0:
                    tmp = h_current - h_current
                else:
                    shifted_h = np.roll(h_current, k)
                    g += (1 - shifted_h) - h_current
            
            g /= len(K_max)
            
            # gradient descent step
            eta = 0.08
            h_new = h_current - eta * g
            
            # Project onto feasible set
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            c5_new = evaluate_c5(h_new)
            
            if c5_new < c5_bound:
                c5_bound = c5_new
                h_current = h_new
                
                # Sell % increase in step size if improvement
                if c5_new < evaluate_c5(h_current):  # hack: recompute
                    eta *= 1.1
                    eta = min(eta, 0.1)
            
            # Line search if needed
            if c5_new > c5_bound:
                for _ in range(5):
                    eta /= 2
                    h_new = h_current - eta * g
                    h_new = project_to_sum_box(h_new, TARGET_SUM)
                    c5_new = evaluate_c5(h_new)
                    if c5_new < c5_bound:
                        c5_bound = c5_new
                        h_current = h_new
                        break
            
            # Occasionally do small random perturbation
            if i % 200 == 0 and np.random.rand() < 0.1:
                idx1, idx2 = np.random.randint(0, n_points, 2)
                delta = np.random.uniform(-0.015, 0.015)
                h_test = h_current.copy()
                h_test[idx1] += delta
                h_test[idx2] -= delta
                h_test = np.clip(h_test, 0, 1)
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                c5_test = evaluate_c5(h_test)
                if c5_test < c5_bound:
                    c5_bound = c5_test
                    h_current = h_test
                    
        return h_current, c5_bound
    
    # Refinement using best from CMA-ES
    h_values, c5_bound = refine_solution(h_values)
    
    # Try a few more known good seeds
    seeds = []
    # U-shape: triangular drop from 1 to 0
    u_shape = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    u_shape[x <= 1] = 1 - x[x <= 1]
    u_shape[x >= 1] = x[x >= 1] - 1
    u_shape = project_to_sum_box(u_shape, TARGET_SUM)
    seeds.append(('u_shape', u_shape))
    
    # Project initial h_values
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        h_init = project_to_sum_box(h_init, TARGET_SUM)
        seeds.append(('initial', h_init))
    
    for name, s in seeds:
        s, c5_s = refine_solution(s)
        if c5_s < c5_bound:
            h_values, c5_bound = s, c5_s
    
    return (h_values.tolist(), float(c5_bound), n_points)
```
