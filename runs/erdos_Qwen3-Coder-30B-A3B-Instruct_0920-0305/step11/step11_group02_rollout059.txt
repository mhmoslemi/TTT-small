import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    auto_corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return auto_corr

def gradient_from_weights(h, w):
    """Compute gradient from weights using FFT"""
    n = len(h)
    L = 2 * n - 1
    w_rev = w[::-1]
    
    # Convolution with h shifted by +j and -j
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract central part of the convolution results
    grad = -(conv_plus[n-1:] + conv_rev[n-1:])
    return grad

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    auto_corr = compute_autocorrelation(h)
    S = len(h) / 2
    g = S - auto_corr
    return np.max(g)
