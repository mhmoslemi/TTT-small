import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_C5(h):
    """Compute C5 value from h"""
    corr = compute_corr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort values in descending order
    sorted_x = np.sort(x)[::-1]
    
    # Binary search for the threshold lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    
    def sum_with_lambda(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected)
    
    # Binary search for lambda that satisfies sum = target_sum
    for _ in range(100):
        mid = (low + high) / 2
        s = sum_with_lambda(mid)
        if abs(s - target_sum) < tol:
            break
        if s > target_sum:
            low = mid
        else:
            high = mid
    
    # Apply final projection
    lam = (low + high) / 2
    result = np.clip(x + lam, 0, 1)
    return result

def thue_morse_binary(n):
    """Create a Thue-Morse binary pattern of length n"""
    # Generate Thue-Morse sequence up to n elements
    t = [0]
    while len(t) < n:
        t.extend(1 - np.array(t))
    t = t[:n]
    # Ensure exactly n//2 ones
    ones = np.where(t)[0]
    if len(ones) > n//2:
        # Remove extras
        to_remove = len(ones) - n//2
        idx = np.random.choice(ones, to_remove, replace=False)
        t[idx] = 0
    elif len(ones) < n//2:
        # Add more ones
        zeros = np.where(1 - np.array(t))[0]
        to_add = n//2 - len(ones)
        idx = np.random.choice(zeros, to_add, replace=False)
        t[idx] = 1
    return np.array(t, dtype=float)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Locally improve a binary solution with greedy swapping"""
    # Compute initial correlation
    corr = np.correlate(h, h, mode="full") * dx
    min_idx = np.argmin(corr)
    worst_k = min_idx - (len(h) - 1)
    max_iter = 10000
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Find worst shift
        min_idx = np.argmin(corr)
        worst_k = min_idx - (len(h) - 1)
        
        # Locate index pairs that can be swapped to improve worst shift
        best_gain = -np.inf
        best_pair = None
        
        for i in range(len(h)):
            if h[i] == 0:
                continue
            
            gain = 0
            # Check how much correlation increases at shift k by flipping i
            # Only check shift k when either i+k or i-k is within range
            if worst_k >= 0 and i + worst_k < len(h):
                if h[i + worst_k] == 1:
                    gain += 1
            elif worst_k < 0 and i - worst_k < len(h):
                if h[i - worst_k] == 1:
                    gain += 1
            
            if gain > best_gain:
                best_gain = gain
                best_pair = (i, None)  # We'll find a suitable zero to flip

        # Now find a zero to swap with best_pair[0]
        if best_pair and best_gain > 0:
            i_one = best_pair[0]
            best_gain = -np.inf
            best_zero = None
            
            # Find a zero that can improve the correlation
            for j in range(len(h)):
                if h[j] == 1:
                    continue
                    
                # Check gain/loss
                gain = 0
                loss = 0
                
                # If j can help improve the worst shift
                if worst_k >= 0 and j + worst_k < len(h) and h[j + worst_k] == 1:
                    gain += 1
                elif worst_k < 0 and j - worst_k < len(h) and h[j - worst_k] == 1:
                    gain += 1
                    
                # If i_one would lose when swapped
                if worst_k >= 0 and i_one + worst_k < len(h) and h[i_one + worst_k] == 1:
                    loss += 1
                elif worst_k < 0 and i_one - worst_k < len(h) and h[i_one - worst_k] == 1:
                    loss += 1
                
                if gain - loss > best_gain:
                    best_gain = gain - loss
                    best_zero = j
                    
            if best_zero is not None:
                # Perform the swap
                h[i_one] = 0
                h[best_zero] = 1
                # Recompute correlation after small change (simplified)
                corr = np.correlate(h, h, mode="full") * dx
            else:
                break
        else:
            break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    TARGET_SUM = n_points // 2
    best_C5 = float('inf')
    best_h = None
    
    seeds = []
    
    # Thue-Morse binary
    seeds.append(thue_morse_binary(n_points))
    
    # Random uniform
    r = np.random.rand(n_points)
    seeds.append(project_to_sum_box(r, TARGET_SUM))
    
    # Provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Rescale to correct length
            h_init_rescaled = np.interp(np.linspace(0, 1, n_points), 
                                       np.linspace(0, 1, len(h_init)), 
                                       h_init)
            seeds.append(project_to_sum_box(h_init_rescaled, TARGET_SUM))
        else:
            seeds.append(project_to_sum_box(h_init, TARGET_SUM))
    
    for h0 in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h = h0.copy()
        tau = 0.2
        alpha = 0.15
        last_improvement_iter = 0
        
        # Safe loop iteration count
        max_iter = min(3000, int((budget_s * 0.8) / 0.02))  # Estimate 0.02 sec/iter
        
        for it in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute correlation
            corr = compute_corr(h)
            C5 = 1.0 - np.min(corr)
            
            # Check for new best
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                last_improvement_iter = it
            
            # Gradient using smooth minimum
            w = np.exp(-corr/tau)
            w /= w.sum()
            g = np.zeros_like(h)
            
            # Efficiently compute weighted sum of shifted versions of h
            for shift in range(-(n_points - 1), n_points):
                wk = w[shift + (n_points - 1)]
                if wk < 1e-16: 
                    continue
                
                # Indices for h_{i+shift} and h_{i-shift}
                if shift >= 0:
                    lo = 0
                    hi = n_points - shift
                    g[lo:hi] += wk * h[lo + shift:hi + shift]
                    g[lo + shift:hi + shift] += wk * h[lo:hi]
                else:
                    lo = -shift
                    hi = n_points
                    g[lo:hi] += wk * h[lo + shift:hi + shift]
                    g[lo + shift:hi + shift] += wk * h[lo:hi]
            
            g *= dx
            
            # Update with step size
            h_candidate = project_to_sum_box(h + alpha * g, TARGET_SUM)
            
            # Evaluate new correlation
            corr_new = compute_corr(h_candidate)
            C5_new = 1.0 - np.min(corr_new)
            
            # Accept or reject
            if C5_new < C5:
                h = h_candidate
                alpha = min(alpha * 1.02, 0.5)  # Small increment
                last_improvement_iter = it
            else:
                alpha *= 0.5  # Shrink step size

            # Cool down
            tau *= 0.99
            
            # Early termination if no improvement in a while
            if it - last_improvement_iter > 200:
                break
    
    # If we found a good solution, polish it with binary swap search
    if best_h is not None:
        # Convert to binary with exactly TARGET_SUM ones
        idx = np.argsort(-best_h)[:TARGET_SUM]
        h_binary = np.zeros(n_points)
        h_binary[idx] = 1.0
        
        # Apply local search for binary improvement
        h_binary = greedy_swap_local_search(h_binary, dx, budget_s, start_time)
        final_C5 = compute_C5(h_binary)
        
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h_binary
    
    if best_h is None:
        # Return the last attempt
        best_h = project_to_sum_box(np.random.rand(n_points), TARGET_SUM)
        best_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(best_C5), n_points)
