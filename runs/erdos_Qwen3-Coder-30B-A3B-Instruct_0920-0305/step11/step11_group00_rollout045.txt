import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.sparse import diags

def proj_sum_box(v, target):
    """Project vector v onto box constraints [0,1] with sum = target"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    return np.clip(v + (a + b)/2, 0, 1)

def corr_full(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute gradient of correlation with shift k"""
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[:-k] = 1 - h[k:]
        g[-k:] = 1  # Dummy values that will be ignored
    elif k < 0:
        g[-k:] = 1 - h[:k]
        g[:-k] = 0
    else:
        g[:] = 1 - h
    return g

def max_overlap(h):
    """Computes C5 using FFT"""
    corr = corr_full(h)
    return np.max(corr)

def shift_mask(k, n):
    """Construct the sparse mask matrix for correlation shift k"""
    offset = -k
    if offset > 0:
        # upper diagonal
        indices = np.arange(n - offset)
        data = np.ones(len(indices))
        M = diags(data, offset, shape=(n, n))
    elif offset < 0:
        # lower diagonal
        indices = np.arange(-offset, n)
        data = np.ones(len(indices))
        M = diags(data, offset, shape=(n, n))
    else:
        # main diagonal
        M = diags(np.ones(n), 0, shape=(n, n))
    return M

def build_sdp(n):
    """Build SDP problem"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Create masks for all possible shifts
    constraints = []
    constraints.append(cp.diag(X) == h)
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == n/2)
    constraints.append(tau >= 0)
    
    for k in range(-(n-1), n):
        M = shift_mask(k, n).toarray()
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
        
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob

def solve_sdp(problem):
    """Solve SDP problem"""
    try:
        problem.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if problem.status == cp.OPTIMAL:
            return problem.value
    except:
        pass
    return None

def round_from_sdp(h_sdp, X_sdp, n):
    """Round SDP solution to nearest valid vector"""
    
    # Method 1: diagonal rounding
    h_diag = h_sdp.copy()
    
    # Method 2: leading eigenvector
    try:
        evals, evecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(evals)
        u = evecs[:, idx]
        h_eig = np.sqrt(max(0, evals[idx])) * np.abs(u)
        h_eig = proj_sum_box(h_eig, n/2)
    except:
        h_eig = h_diag.copy()
    
    # Method 3: Gaussian rounding
    h_gauss = None
    try:
        # Sample from the distribution defined by the matrix
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h_trial = np.abs(z)
            h_trial = proj_sum_box(h_trial, n/2)
            if h_gauss is None or max_overlap(h_trial) < max_overlap(h_gauss):
                h_gauss = h_trial.copy()
    except:
        pass
    
    # Choose the best among the three methods
    candidates = [h_diag, h_eig]
    if h_gauss is not None:
        candidates.append(h_gauss)
    
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        try:
            c5 = max_overlap(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except:
            continue

    return best_h, best_c5

def polish(h_start, n, dx, time_left):
    """Polish the solution using projected gradient descent"""
    max_iter = 2000
    step = 0.07
    
    h = h_start.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = max_overlap(h)
    
    for it in range(max_iter):
        if time.time() - t_start > time_left - 10:
            break
            
        # Compute gradients for active shifts
        c = corr_full(h)
        Cmax = np.max(c)
        Kmax = np.where(c >= Cmax - 1e-12)[0]
        
        # Average gradient
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov-style extrapolation
        y = h + 0.9 * (h - h_prev)
        
        # Perform update
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        # Evaluate
        c5_new = max_overlap(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.08, 0.2)
        else:
            step *= 0.5
            
        h_prev = h
        h = h_new

        # Occasionally do a swap move
        if it % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] - delta, 0, 1)
            h_new[j] = np.clip(h_new[j] + delta, 0, 1)
            h_new = proj_sum_box(h_new, n/2)
            c5_new = max_overlap(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
    return best_h, best_c5
