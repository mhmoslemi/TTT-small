import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorrelation(h):
    """Compute autocorrelation via FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, weights):
    """Compute gradient from soft-max weights using FFT"""
    n = len(h)
    L = 2 * n - 1
    # Shift weights to center
    w_rev = weights[::-1]
    # Convolution with h
    conv_plus = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    conv_rev = np.fft.ifft(np.fft.fft(weights, L) * np.fft.fft(h, L)).real
    # Extract central components
    grad = -(conv_plus[n-1:] + conv_rev[n-1:])
    return grad

def compute_c5(h):
    """Compute C5 value for given h"""
    O = autocorrelation(h)
    g = TARGET_SUM - O
    return dx * np.max(g)

    start_time = time.time()
    
    # Step 1: Initialize with two different starting points

    # Linear interpolation to desired resolution

    h_upsampled = np.interp(x_new, x_old, initial_h)
    h_start1 = project_to_sum_box(h_upsampled, TARGET_SUM)
    
    # Start 2: Random binary start
    h_start2 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(TARGET_SUM), replace=False)
    h_start2[ones_indices] = 1.0
    h_start2 = project_to_sum_box(h_start2, TARGET_SUM)
    
    # Choose better start
    c5_1 = compute_c5(h_start1)
    c5_2 = compute_c5(h_start2)
    if c5_1 < c5_2:
        current_h = h_start1
        current_c5 = c5_1
    else:
        current_h = h_start2
        current_c5 = c5_2
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Step 2: Main optimization loop using smooth max gradient descent
    alpha = 1.0
    eta = 0.05
    max_alpha = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute autocorrelation and g values
        O = autocorrelation(current_h)
        g = TARGET_SUM - O  # length 2*n-1
        
        # Compute smooth max and weights
        max_g = np.max(g)
        g_shifted = g - max_g
        exp_g = np.exp(alpha * g_shifted)
        w = exp_g / np.sum(exp_g)
        
        # Compute gradient
        grad = gradient_from_weights(current_h, w)
        
        # Update with projected gradient
        y = current_h - eta * grad
        new_h = project_to_sum_box(y, TARGET_SUM)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        
        # Adaptive scheduling
        if iter_count % 200 == 0:
            alpha = min(alpha * 1.3, max_alpha)
            eta = eta * 0.9
        
        # Occasionally shake out of flat regions
        if iter_count % 500 == 0 and iter_count > 1000:
            noise = np.random.normal(0, 0.02, n_points)
            h_noisy = current_h + noise
            h_noisy = project_to_sum_box(np.clip(h_noisy, 0, 1), TARGET_SUM)

                    best_h = current_h.copy()
    
    # Final polishing with SLSQP
    def obj_full(x):
        O = autocorrelation(x)
        return dx * np.max(TARGET_SUM - O)
    
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - TARGET_SUM},)
    bounds = [(0.0, 1.0)] * n_points
    
    try:
        res = minimize(obj_full,
                       x0=best_h,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        
        if res.success and obj_full(res.x) < best_c5:
            best_h = res.x
            best_c5 = obj_full(res.x)
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
