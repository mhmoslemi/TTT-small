import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def inverse_sigmoid(y):
    return np.log(y / (1.0 - y))

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def spline_to_h(z):
    """Convert spline control points to step function h"""
    # Fixed positions for spline knots on [0,1]
    # We use m=7 knots, i.e. 4 unique values
    knots = np.linspace(0, 1, 4)
    # Map z to y values via sigmoid
    y = sigmoid(z)
    # Interpolate
    cs = CubicSpline(knots, y, bc_type='clamped')
    # Evaluate at full grid
    x_full = np.linspace(0, 1, (n_points+1)//2)
    y_full = cs(x_full)
    # Mirror for symmetric profile
    h_half = np.concatenate([y_full, y_full[-2::-1]])
    # Scale to correct sum
    s = np.sum(h_half)
    if s > 0:
        alpha = TARGET_SUM / s
        h = np.clip(alpha * h_half, 0, 1)
    else:
        h = h_half.copy()
    return h

def cma_es_optimize():
    """CMA-ES optimization in low-dimensional parameter space"""
    # Parameters: 4 control points (symmetric cubic spline)
    d = 4
    # Initial mean
    m = np.zeros(d)
    # Initial step size
    sigma = 2.0
    # Initial covariance matrix
    C = np.eye(d)
    # Population size
    lambda_ = 4 + int(3 * np.log(d))
    mu = lambda_ // 2
    
    # Function to evaluate objective
    def objective(z):
        h = spline_to_h(z)
        return evaluate_c5(h)
    
    best_z = m.copy()
    best_c5 = objective(best_z)
    
    # Simple CMA-ES implementation
    for gen in range(300):
        # Sample offspring
        offspring = []
        for _ in range(lambda_):
            z = m + sigma * np.random.multivariate_normal(np.zeros(d), C)
            offspring.append(z)
        
        # Evaluate
        fitness = [objective(z) for z in offspring]
        # Sort by fitness
        indices = np.argsort(fitness)
        best_idx = indices[0]
        if fitness[best_idx] < best_c5:
            best_z = offspring[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Select parents
        parents = [offspring[i] for i in indices[:mu]]
        # Update mean
        m = np.mean(parents, axis=0)
        # Update covariance matrix (simplified)
        diff = np.array(parents) - m
        C = (1 - 0.1) * C + 0.1 * (diff.T @ diff) / mu
        
        # Update step size
        sigma *= np.exp(0.1 * (np.random.rand() - 0.5))
    
    return best_z, best_c5

def polish_h(h):
    """Polish the solution with projected gradient descent"""
    # Use small batch gradient steps
    max_iter = 1200
    eta = 0.08
    for i in range(max_iter):
        if i % 200 == 0:
            # Try random move to escape plateaus
            idx1, idx2 = np.random.randint(0, n_points, 2)
            delta = np.random.uniform(-0.015, 0.015)
            h_copy = h.copy()
            h_copy[idx1] += delta
            h_copy[idx2] -= delta
            h_copy = project_to_sum_box(h_copy, TARGET_SUM)
            c5_new = evaluate_c5(h_copy)
            if c5_new < evaluate_c5(h):
                h = h_copy.copy()
        
        # Compute correlation and find worst shifts
        corr = compute_corr_fft(h)
        max_val = np.max(corr)
        K_max = np.where(np.abs(corr - max_val) < 1e-12)[0]
        if len(K_max) == 0:
            K_max = [np.argmax(corr)]
        
        # Gradient direction
        g = np.zeros(n_points)
        for k in K_max:
            for i in range(n_points):
                if i + k >= 0 and i + k < n_points:
                    g[i] += 1 - h[i + k]
                if i - k >= 0 and i - k < n_points:
                    g[i] -= h[i - k]
        
        # Update
        h_new = h - eta * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Line search
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new
        else:
            eta *= 0.5
            if eta < 1e-6:
                eta = 0.08
    
    return h
