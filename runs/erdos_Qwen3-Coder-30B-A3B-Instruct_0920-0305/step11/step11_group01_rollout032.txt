TARGET_SUM = None

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Compute convolution of a and b using FFT. a,b are 1D arrays."""
    L = len(a) + len(b) - 1
    L_fft = 1
    while L_fft < L:
        L_fft *= 2
    A = np.fft.fft(a, L_fft)
    B = np.fft.fft(b, L_fft)
    C = np.fft.ifft(A * B).real[:L]
    return C

def smooth_max_corr(h, alpha):
    """Compute smooth maximum correlation using soft-max approximation"""
    corr = compute_corr_fft(h)
    exp_corr = np.exp(alpha * corr)
    w = exp_corr / np.sum(exp_corr)
    # Convolution for gradient: (w * h) and (w_rev * h)
    w_rev = w[::-1]
    conv1 = convolve_fft(w, h)
    conv2 = convolve_fft(w_rev, h)
    grad = 1.0 - conv1 - conv2
    smooth_max_val = np.sum(w * corr) / alpha
    return smooth_max_val, grad
