import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def project_onto_feasible_set(h, target_sum):
    """Project h onto the set {h : 0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    def f(lam):
        ph = np.clip(h + lam, 0, 1)
        return np.sum(ph) - target_sum
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(h + (a + b) / 2, 0, 1)
    return result

def softmin_weights(R, beta):
    """Compute softmin weights for given R values and temperature beta"""
    R = np.array(R)
    exp_neg_beta_R = np.exp(-beta * R)
    weights = exp_neg_beta_R / np.sum(exp_neg_beta_R)
    return weights

def compute_autocorrelation(h):
    """Compute full autocorrelation using FFT"""
    L = 2 * len(h) - 1
    H = np.fft.fft(h, L)
    R_full = np.fft.ifft(H * np.conj(H)).real
    return R_full

def gradient_from_weights(w, h):
    """Compute gradient using softmin weights"""
    n = len(h)
    w_rev = w[::-1]
    # Forward shifts contribute to gradient
    forward_shift = np.convolve(w, h, mode='full')
    # Backward shifts contribute to gradient  
    backward_shift = np.convolve(w_rev, h, mode='full')
    # Take appropriate slices
    grad = -(forward_shift[n-1:2*n-1] + backward_shift[0:n])
    return grad

def compute_correlation(h):
    """Compute exact C5 using correlation"""
    corr = correlate(h, 1-h, mode='full')
    return np.max(corr) * (2.0 / len(h))
