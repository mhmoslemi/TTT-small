import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.sparse import diags
import time
from scipy.linalg import eigh

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    a, b = -1.0, 1.0
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def corr_full(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx=1.0):
    """Compute maximum overlap"""
    corr = corr_full(h)
    return dx * np.max(corr)

def shift_mask(k, n):
    """Generate shift mask for shift k"""
    if k >= 0:
        return diags([1]* (n-k), k=k, shape=(n,n), format='csr')
    else:
        return diags([1]* (n+k), k=k, shape=(n,n), format='csr')

def build_sdp(n):
    """Build the SDP problem"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Overlap constraints for all shifts
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob

def solve_sdp(prob):
    """Solve the SDP"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False, eps=1e-6)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return True, prob.value, prob.variables()[0].value, prob.variables()[1].value
        else:
            return False, None, None, None
    except Exception:
        return False, None, None, None

def round_from_sdp(h_sdp, X_sdp, n):
    """Extract feasible h from SDP solution using three methods"""
    
    # Method 1: Diagonal rounding
    h1 = h_sdp.copy()
    h1 = proj_sum_box(h1, n/2)
    
    # Method 2: Leading eigenvector rounding
    try:
        eigenvals, eigenvecs = eigh(X_sdp)
        idx = np.argmax(eigenvals)
        lambda_max = eigenvals[idx]
        u_max = eigenvecs[:, idx]
        h2 = np.sqrt(lambda_max) * u_max
        h2 = proj_sum_box(h2, n/2)
    except:
        h2 = h1.copy()
    
    # Method 3: Gaussian randomization
    best_h = h1.copy()
    best_c5 = max_overlap(h1)
    for _ in range(5):
        z = np.random.multivariate_normal(np.zeros(n), X_sdp)
        h3 = np.abs(z)
        h3 = proj_sum_box(h3, n/2)
        c5 = max_overlap(h3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h3.copy()
    
    # Compare all methods - return best
    c5_1 = max_overlap(h1)
    c5_2 = max_overlap(h2)
    
    if c5_1 <= c5_2 and c5_1 <= best_c5:
        return h1
    elif c5_2 <= best_c5:
        return h2
    else:
        return best_h

def grad_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1 - h[:-k] if k < n else 1
        grad[:k] = 0
    else:
        grad[:k] = 1 - h[-k:] if -k < n else 1
        grad[k:] = 0
    return grad

def polish(h_start, n, dx, remaining_time):
    """Polish the solution using projected gradient descent"""
    h = h_start.copy()
    h_prev = h.copy()
    step = 0.07
    beta = 0.9
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    start_time = time.time()
    
    for i in range(2000):
        if time.time() - start_time > remaining_time - 5:
            break
            
        corr = corr_full(h)
        max_val = np.max(corr)
        k_indices = np.where(np.abs(corr - max_val) < 1e-12)[0]
        
        # Compute average gradient
        g = np.zeros(n)
        for k in k_indices:
            g += grad_shift(h, k)
        g /= len(k_indices)
        
        # Nesterov-style extrapolation
        y = h + beta * (h - h_prev)
        
        # Gradient step
        h_new = y - step * g
        
        # Project
        h_new = proj_sum_box(h_new, n/2)
        
        # Evaluate
        new_c5 = max_overlap(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            step = min(step * 1.08, 0.2)
        else:
            step *= 0.5
            
        # Swap move every 250 steps
        if i % 250 == 0:
            i1, i2 = np.random.randint(n, size=2)
            delta = np.random.uniform(-0.02, 0.02)
            h_temp = h_new.copy()
            h_temp[i1] += delta
            h_temp[i2] -= delta
            h_temp = proj_sum_box(h_temp, n/2)
            c5_temp = max_overlap(h_temp, dx)
            if c5_temp < new_c5:
                h_new = h_temp.copy()
        
        h_prev = h.copy()
        h = h_new.copy()
    
    return best_h, best_c5
