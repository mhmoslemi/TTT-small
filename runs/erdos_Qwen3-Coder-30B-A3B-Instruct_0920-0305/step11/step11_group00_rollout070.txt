import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(60):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def grad_shift(h, s):
    """Compute gradient for correlation shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n//2
    first = np.linspace(1, 0, m, endpoint=False)
    if n % 2 == 0:
        h = np.concatenate([first, first[::-1]])
    else:
        h = np.concatenate([first, first[-2::-1]])
    # Adjust to match target sum
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def random_feasible_vector(n, target):
    """Create random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target)
    return h

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using smooth approximation"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Smooth optimization parameters
    alphas = np.linspace(0.5, 30, 15)   # schedule
    step = 0.3  # base step size
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        # Inner iterations
        for iter in range(100):
            if time.time() - start_time > time_limit:
                break
            c = compute_corr(h)
            # Compute weights
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute weighted gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (len(h)-1)
                grad += weight * grad_shift(h, s)
            
            # Adaptive step size
            step_size = step
            for _ in range(5):  # Max 5 line search attempts
                h_new = h - step_size * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_size *= 0.5
            
            if val_new < best_val:
                best_val = val_new
                best_h = h.copy()
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Subgradient optimization parameters
    step = 0.2
    max_iter = 200
    
    for iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        c = compute_corr(h)
        c_max = np.max(c)
        # Find indices where correlation is approximately maximum
        tol = 1e-8
        max_indices = np.where(np.abs(c - c_max) < tol)[0]
        
        if len(max_indices) == 0:
            max_indices = [np.argmax(c)]
        
        # Average gradient over all maximizing shifts
        grad = np.zeros_like(h)
        for idx in max_indices:
            s = idx - (len(h) - 1)
            grad += grad_shift(h, s)
        
        if len(max_indices) > 0:
            grad /= len(max_indices)
        
        # Adaptive step size
        step_size = step
        for _ in range(5):  # Max 5 line search attempts
            h_new = h - step_size * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
                break
            else:
                step_size *= 0.5
        
        if val_new < best_val:
            best_val = val_new
            best_h = h.copy()
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    
    np.random.seed(seed)
    start_time = time.time()
    
    # Set parameters
    TARGET_SUM = n_points / 2
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = np.inf
    best_n = n_points
    
    # Try different discretizations
    for n in [120, 200]:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        TARGET_SUM = n / 2
        
        # Try multiple initializations
        for init_type in ['triangular', 'random']:
            if time.time() - start_time > budget_s * 0.95:
                break
            if init_type == 'triangular':
                h0 = triangular_initial(n)
            else:
                h0 = random_feasible_vector(n, TARGET_SUM)
            
            # Smooth optimization
            h_smooth, val_smooth = smooth_optimize(h0, TARGET_SUM, dx, budget_s * 0.9, start_time)
            if val_smooth < best_c5:
                best_h = h_smooth
                best_c5 = val_smooth
                best_n = n
                
            # Subgradient optimization
            h_sub, val_sub = subgradient_optimize(h_smooth, TARGET_SUM, dx, budget_s * 0.95, start_time)
            if val_sub < best_c5:
                best_h = h_sub
                best_c5 = val_sub
                best_n = n
    
    # Final evaluation
    final_c5 = c5_value(best_h)
    
    return (best_h.tolist(), float(final_c5), int(best_n))
