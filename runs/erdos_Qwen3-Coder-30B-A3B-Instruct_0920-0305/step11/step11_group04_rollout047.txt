from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h4[2*n_points//6:3*n_points//6] = 1.0
    h4[3*n_points//6:4*n_points//6] = 0.5

    iter_count = 0

            # Improved gradient descent with better convergence
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM
            
            cons = {'type': 'eq', 'fun': constraint_eq}
            
            try:

                for _ in range(3):

                            if test_c5 < best_c5:
                                best_c5 = test_c5
                                best_h = current_h.copy()
                            improvement_found = True
                            break

    # Final refinement with L-BFGS
    if time.time() - start_time < budget_s * 0.8 and best_h is not None:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_c5(h)
            
            def constraint_eq(h):
                return np.sum(h) - TARGET_SUM

        best_h = initial_configs[0].copy()
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
