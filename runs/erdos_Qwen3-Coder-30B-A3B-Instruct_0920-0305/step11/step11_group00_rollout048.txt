import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def cubic_spline_evaluation(knots, values, x_grid):
    """Evaluate cubic spline at grid points"""
    from scipy.interpolate import CubicSpline
    
    # Make symmetric
    x_left = knots
    x_right = 2 - knots[::-1]
    x_full = np.concatenate([x_left, x_right[1:]])
    
    y_left = values
    y_right = values[::-1]
    y_full = np.concatenate([y_left, y_right[1:]])
    
    cs = CubicSpline(x_full, y_full, bc_type='clamped')
    return cs(x_grid)
