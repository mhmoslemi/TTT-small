# Global constants
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_shift_matrices(n):
    """Compute shift matrices S_k for all valid shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create a Toeplitz matrix representing shift k
        # For shift k: M[i,j] = 1 if j = i + k
        data = [1] * (n - abs(k))
        offsets = [k]
        S_k = diags(data, offsets, shape=(n, n), format='csr')
        shift_matrices.append(S_k)
    return shift_matrices

def compute_overlap(h):
    """Compute C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def solve_sdp():
    """Solve SDP relaxation for the Erdos problem"""
    # Build shift matrices
    shift_matrices = compute_shift_matrices(n_points)
    
    # Define CVXPY variables
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == 1,  # Diagonal ones
        cp.trace(X) == n_points,  # Trace constraint
        t >= 0, 
        t <= 1
    ]
    
    # Shift constraints
    for k, S_k in enumerate(shift_matrices):
        # For shift k, length of shift is n_points - abs(k)
        n_k = n_points - abs(k - (n_points - 1))
        # Constraint: trace(S_k @ X) >= 2*n*(1 - t) - n_k
        constraints.append(cp.trace(S_k @ X) >= 2 * n_points * (1 - t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=1500, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
        else:
            return None, None
    except:
        return None, None

        # Compute square root of eigenvalues
        sqrt_eigenvals = np.sqrt(positive_vals)
        
        # Create L matrix
        L = positive_vecs @ np.diag(sqrt_eigenvals)
        
        # Number of rounds
        R = 100
        
        best_h = None
        best_c5 = float('inf')
        
        for r in range(R):
            # Sample random vector
            g = np.random.randn(L.shape[1])
            
            # Compute signs
            projections = L @ g
            signs = np.sign(projections)
            
            # Convert to binary (0/1)
            h = 0.5 * (signs + 1)
            
            # Adjust to ensure exact sum
            current_sum = np.sum(h)
            if abs(current_sum - TARGET_SUM) > 1e-6:
                # Sort by absolute margin to determine which to flip
                margins = np.abs(projections)
                sorted_indices = np.argsort(margins)
                
                if current_sum > TARGET_SUM:
                    # Need to flip some 1s to 0s
                    indices_to_flip = sorted_indices[:int(current_sum - TARGET_SUM)]
                    for i in indices_to_flip:
                        h[i] = 0
                else:
                    # Need to flip some 0s to 1s
                    indices_to_flip = sorted_indices[:int(TARGET_SUM - current_sum)]
                    for i in indices_to_flip:
                        h[i] = 1
            
            # Compute overlap
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        if best_h is not None:
            return best_h, best_c5
        else:
            return None, float('inf')
            
    except Exception as e:
        return None, float('inf')

    h = h.copy()
    current_c5 = compute_overlap(h)
    
    for _ in range(max_iter):
        # Find best swap
        best_gain = 0
        best_i, best_j = None, None
        
        # For each zero index, compute marginal gain
        for i in range(n_points):
            if h[i] < 0.5:  # h[i] == 0
                # Can flip this to 1
                old_corr = np.correlate(h, 1-h, mode='full')
                h_test = h.copy()
                h_test[i] = 1
                new_corr = np.correlate(h_test, 1-h_test, mode='full')
                
                gain = np.max(old_corr) - np.max(new_corr)
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
        
        # If no improvement possible, break
        if best_gain <= 0:
            break
            
        # Perform the swap
        if best_i is not None:
            h[best_i] = 1
            
        current_c5 = compute_overlap(h)
    
    return h, current_c5
