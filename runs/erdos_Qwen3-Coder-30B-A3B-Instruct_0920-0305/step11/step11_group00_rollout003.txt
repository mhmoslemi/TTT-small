import numpy as np
import time
from scipy.signal import correlate

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    a = -1.0 - np.max(x)
    b = 1.0 - np.min(x)
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_grad_shift(h, k):
    """Compute gradient of correlation shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def lse_surrogate(h, alpha):
    """Compute log-sum-exp surrogate"""
    c = compute_correlation(h)
    max_c = np.max(c)
    exp_terms = np.exp(alpha * (c - max_c))
    log_sum_exp = max_c + np.log(np.sum(exp_terms)) / alpha
    return log_sum_exp

def compute_lse_gradient(h, alpha):
    """Compute gradient of log-sum-exp surrogate"""
    c = compute_correlation(h)
    max_c = np.max(c)
    exp_terms = np.exp(alpha * (c - max_c))
    weights = exp_terms / np.sum(exp_terms)
    
    grad = np.zeros_like(h)
    n = len(h)
    
    # Compute weighted sum of gradients
    for k in range(-n+1, n):
        grad_k = compute_grad_shift(h, k)
        grad += weights[k+n-1] * grad_k
    
    return grad

def accelerated_projected_gradient(h, alpha, max_iter=150, eta=0.1, beta=0.9):
    """APG with backtracking line search"""
    h_prev = h.copy()
    y = h.copy()
    
    for iter in range(max_iter):
        # Compute gradient
        grad = compute_lse_gradient(h, alpha)
        
        # Nesterov step
        y = h + beta * (h - h_prev)
        h_tmp = y - eta * grad
        
        # Project
        h_new = project_to_sum_box(h_tmp, len(h) / 2)
        
        # Backtracking line search
        f_new = lse_surrogate(h_new, alpha)
        f_old = lse_surrogate(h, alpha)
        
        # Simple line search
        while f_new > f_old and eta > 1e-10:
            eta *= 0.5
            h_tmp = y - eta * grad
            h_new = project_to_sum_box(h_tmp, len(h) / 2)
            f_new = lse_surrogate(h_new, alpha)
        
        h_prev = h.copy()
        h = h_new
        
        # Adaptive step size
        if f_new < f_old:
            eta = min(eta * 1.1, 1.0)
        else:
            eta = max(eta * 0.5, 1e-6)
    
    return h

def subgradient_polish(h, max_iter=1500):

        max_val = np.max(c)
        
        # Find active shifts
        active = np.abs(c - max_val) < 1e-9
        active_indices = np.where(active)[0]
        
        if len(active_indices) == 0:
            break
            
        # Compute average gradient
        avg_grad = np.zeros_like(h)
        for k_idx in active_indices:
            k = k_idx - len(h) + 1
            avg_grad += compute_grad_shift(h, k)
        avg_grad /= len(active_indices)
        
        # Update
        h_new = h - eta * avg_grad
        h_new = project_to_sum_box(h_new, len(h) / 2)
        
        # Accept if improvement
        new_c5 = np.max(compute_correlation(h_new))
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            eta = min(eta * 1.05, 0.15)
        else:
            eta = max(eta * 0.5, 1e-6)
            
        h = h_new
        
        # Random perturbation escape every 200 iterations
        if iter % 200 == 0:
            i, j = np.random.choice(len(h), 2, replace=False)
            delta = 0.015
            h_new = h.copy()

            h_new = project_to_sum_box(h_new, len(h) / 2)
            new_c5 = np.max(compute_correlation(h_new))

                best_h = h_new.copy()
                best_c5 = new_c5
                
        if time.time() > time.time():
            break
    
    return best_h, best_c5

    h[:half] = np.linspace(1.0, 0.0, half)
    h[half:] = np.linspace(0.0, 1.0, n - half)
    return h
