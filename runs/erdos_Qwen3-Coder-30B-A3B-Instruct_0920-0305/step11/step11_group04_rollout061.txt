import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorr(h):
    """Compute linear autocorrelation using FFT for efficiency."""
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(h, L).conj()).real
    return corr

def compute_C5(h, dx):
    """Compute C5 bound for a normalized function h."""
    corr = autocorr(h)
    return 1.0 - np.min(corr) * dx

def thue_morse_binary(n):
    """Generate a Thue-Morse binary sequence of length n."""
    tm = []
    for i in range(n):
        # This is a simplified version of Thue-Morse generating sequence
        # We continue the pattern by avoiding three consecutive same bits
        # For exact binary Thue-Morse, we can also generate it recursively but for simplicity:
        # if i=0, then tm[0] = 0, else we generate the pattern directly
        # Properly generate Thue-Morse sequence for first n bits
        if i == 0:
            tm.append(0)
        elif i == 1:
            tm.append(1)
        else:
            # Generate from prior terms
            new_bit = (tm[i-1] + tm[i//2]) % 2
            tm.append(new_bit)
            
    # Ensure that we have exactly n/2 ones for even n
    if n % 2 == 0:
        ones_count = sum(tm)
        if ones_count != n // 2:
            # Find indices with 0s to make into 1s until we have exactly n/2
            zero_indices = [i for i in range(n) if tm[i] == 0]
            fill_count = (n // 2) - ones_count
            for i in range(fill_count):
                if i < len(zero_indices):
                    tm[zero_indices[i]] = 1
        return np.array(tm)
    else:
        # For odd n, we may need to approximate more carefully
        tm = np.array(tm[:n], dtype=int)
        ones_count = sum(tm)
        diff = (n // 2) - ones_count
        zero_indices = [i for i in range(n) if tm[i] == 0]
        for i in range(min(abs(diff), len(zero_indices))):
            if diff > 0:
                tm[zero_indices[i]] = 1
        return tm

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Apply greedy swaps to improve correlation for worst shift."""
    n = len(h)
    target_sum = n // 2

    # Get correlation and find worst shift
    corr = autocorr(h)
    k_star = np.argmin(corr)
    shift_val = k_star - (n - 1)
    
    # If shift_val is 0, this is zero shift, which we skip for practical reasons
    # unless no other shift can be worse.
    worst_shift = shift_val
    
    max_iter = 10000
    iter_count = 0
    while iter_count < max_iter:
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Swap indices optimization
        zero_indices = np.where(h == 0)[0]
        one_indices = np.where(h == 1)[0]
        
        best_gain = -1e9
        best_pair = None
        
        for i0 in zero_indices:
            for i1 in one_indices:
                # Check effects on correlation at worst shift
                gain = 0.0
                loss = 0.0
                
                # During swap: h[i0] becomes 1, h[i1] becomes 0
                # Effect on correlation value when i0 is shifted by worst_shift
                if 0 <= i0 + worst_shift < n:
                    if h[i0 + worst_shift] == 1:
                        gain += 1.0  # Added contribution in i0+shift
                
                # Effect on pos i1 and shift
                if 0 <= i1 + worst_shift < n:
                    if h[i1 + worst_shift] == 1:
                        loss += 1.0  # Lost contribution at i1+shift
                        
                # Effect on negative correlation at shift -worst_shift (i.e., i's position)
                if 0 <= i0 - worst_shift < n:
                    if h[i0 - worst_shift] == 1:
                        gain += 1.0
                        
                if 0 <= i1 - worst_shift < n:
                    if h[i1 - worst_shift] == 1:
                        loss += 1.0
                        
                net_gain = gain - loss
                if net_gain > best_gain:
                    best_gain = net_gain
                    best_pair = (i0, i1)
                    
        if best_pair is None or best_gain <= 0:
            break
            
        i0, i1 = best_pair
        h[i0] = 1
        h[i1] = 0
        
        # Recalculate now that swap has been done
        corr = autocorr(h)
        k_star = np.argmin(corr)
        shift_val = k_star - (n - 1)
        worst_shift = shift_val

        iter_count += 1
        
    return h

            # Gradient of smooth min            
            w = np.exp(-corr / τ)
            w /= w.sum()
            
            g = np.zeros_like(h)
            for shift in range(-(n - 1), n):
                wk = w[shift + n - 1]
                if wk == 0.0:
                    continue
                lo = max(0, -shift)
                hi = min(n, n - shift)
                if lo < hi:
                    g[lo:hi] += wk * h[lo + shift:hi + shift]
                    g[lo + shift:hi + shift] += wk * h[lo:hi]
                    
            g *= dx
            
            # Ascent step + projection
            h_candidate = project_to_sum_box(h + α * g, target_sum) 
            
            # Accept or reject
            corr_new = autocorr(h_candidate)
            C5_new = 1.0 - np.min(corr_new) * dx
            
            if C5_new < C5:
                h = h_candidate
                α = min(α * 1.02, 0.5)
            else:
                α *= 0.5

        h_bin[top_indices] = 1.0
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        final_C5 = compute_C5(h_bin)
        if final_C5 < best_C5:
            best_C5 = final_C5
