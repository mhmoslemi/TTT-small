def build_shift_matrices(n):
    """Build shift matrices S_k for all valid shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create Toeplitz matrix for shift k
        offsets = [k]
        data = [1]
        shape = (n, n)
        S_k = diags(data, offsets, shape=shape, format='csr')
        shift_matrices.append(S_k)
    return shift_matrices

def compute_c5(h):
    """Compute C5 value using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search for lambda
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp(n_points):
    """Solve SDP relaxation for the Erdős problem"""
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Build shift matrices
    shift_matrices = build_shift_matrices(n_points)
    
    # Variable definition
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # Positive semidefinite
        cp.diag(X) == 1,  # Diagonal constraints
        cp.trace(cp.ones((n_points, n_points)) @ X) == 0,  # Balanced constraint
        t >= 0, t <= 1  # Bounding
    ]
    
    # Add shift constraints
    for k, S_k in enumerate(shift_matrices):
        if k < n_points:  # Avoid redundant constraints
            n_k = 2 * n_points - 1 - 2 * abs(k)
            constraints.append(cp.trace(S_k @ X) >= 2 * n_points * (1 - t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=1500, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
            return X.value, t.value
    except:
        pass
    
    return None, None

        positive_indices = eigenvals > 1e-8
        sqrt_eigenvals = np.sqrt(eigenvals[positive_indices])
        V = eigenvectors[:, positive_indices]
        L = V @ np.diag(sqrt_eigenvals)
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(num_rounds):
            # Generate random vector
            g = np.random.randn(L.shape[1])
            # Compute signs of projections
            projections = L.T @ g
            x = np.sign(projections)
            # Convert to binary
            h = (x + 1) / 2
            
            # Balance the sum
            current_sum = np.sum(h)
            if current_sum != n_points / 2:
                # Adjust by flipping the smallest-margin bits
                margin = np.abs(projections)
                if current_sum > n_points / 2:
                    # Need to decrease ones
                    flip_indices = np.argsort(margin)[:int(current_sum - n_points / 2)]
                    h[flip_indices] = 0
                else:
                    # Need to increase ones
                    flip_indices = np.argsort(margin)[-int(n_points / 2 - current_sum):]
                    h[flip_indices] = 1
            
            # Evaluate
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    except:
        return None, None

def local_swap_improve(h, max_iter=100):
    """Improve solution with local swaps"""
    n_points = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for _ in range(max_iter):
        # Find maximum overlap shift
        corr = correlate(h, 1-h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        if max_val <= best_c5:
            break
            
        # Try flipping one bit at a time
        best_improvement = 0
        best_i = -1
        best_val = best_c5
        
        for i in range(n_points):
            # Compute change in overlap if we flip bit i
            old_h = best_h[i]
            new_h = 1 - old_h
            
            # Create test vector
            test_h = best_h.copy()
            test_h[i] = new_h
            
            # Compute new C5  
            new_c5 = compute_c5(test_h)
            improvement = best_c5 - new_c5
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_i = i
                best_val = new_c5
        
        if best_improvement <= 0:
            break
            
        best_h[best_i] = 1 - best_h[best_i]
        best_c5 = best_val
        
    return best_h, best_c5
