import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_c5_exact(h, dx):
    """Compute exact C5 using full correlation"""
    corr = correlate(h, 1-h, mode="full")
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def spectral_rounding(r_opt, n, dx, M):
    """Round the optimal autocorrelation to a feasible step function"""
    # Compute power spectrum
    S = np.fft.fft(r_opt)
    # Ensure realness and non-negativity
    S = np.real(S)
    S = np.maximum(S, 0.0)
    
    best_h = None
    best_c5 = float('inf')
    
    # Try 4 different phase configurations
    phase_seeds = [0, np.pi/4, np.pi/2, np.pi]
    
    for seed in phase_seeds:
        # Take square root of spectrum
        A = np.sqrt(S)
        # Generate phase vector
        phi = seed + np.random.random(n) * 2 * np.pi
        
        # Construct signal candidate
        h_tilde = np.fft.ifft(A * np.exp(1j * phi)).real
        
        # Project onto feasible set
        h_candidate = project_to_sum_box(h_tilde, M)
        
        # Evaluate exact C5
        c5_candidate = evaluate_c5_exact(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_values = [80, 100]
    max_time_per_solve = 250.0
    dx = 2.0 / n_values[0]
    M = n_values[0] / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Solve SDP for both sizes
    for n in n_values:
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n
        M = n / 2
        
        # Set up SDP
        r = cp.Variable(n)
        t = cp.Variable()
        
        # Build Toeplitz matrix
        T = cp.bmat([[r[abs(i-j)] for j in range(n)] for i in range(n)])
        
        constraints = [
            T >> 0,  # PSD constraint
            cp.sum((n - np.arange(n)) * r) == M**2,  # mass constraint
            r >= t,  # min autocorrelation lower bound
            r >= 0,  # non-negativity
            r <= np.minimum(M, n - np.arange(n))  # box constraints
        ]
        
        # Solve with SCS solver (free)
        prob = cp.Problem(cp.Maximize(t), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, time_limit=max_time_per_solve)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_OUT:
                t_opt = t.value
                r_opt = r.value
                
                # Round to binary function
                h_candidate, c5_candidate = spectral_rounding(r_opt, n, dx, M)
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    
                # If we solved the first SDP quickly, try the larger one
                if n == 80 and (time.time() - start_time) < 200:
                    max_time_per_solve = 200.0
                    continue
            else:
                # If SDP failed, fallback to simpler method with current h
                pass
        except:
            pass

    # If SDP didn't work, use simpler method
    if best_h is None:
        # Start with a symmetric construction
        h0 = np.ones(n_values[0]) * 0.5
        h0 = project_to_sum_box(h0, M)
        best_h = h0.copy()
        best_c5 = evaluate_c5_exact(best_h, dx)
    
    # If we have time, refine using gradient ascent
    if time.time() - start_time < budget_s - 5:
        # Perform a few rounds of projected gradient ascent
        h_current = best_h.copy()
        c5_current = best_c5
        max_iter = 50
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute current minimum
            corr = correlate(h_current, 1 - h_current, mode="full")
            min_corr_idx = np.argmin(corr)
            min_corr_val = corr[min_corr_idx]
            
            # Compute gradient (simple approximation for each component)
            grad = np.zeros(n)
            for k in range(len(corr)):
                if abs(corr[k] - min_corr_val) < 1e-8:
                    # Add contribution from this shift
                    for i in range(n):
                        if i + k < n:
                            grad[i] += h_current[i + k]
                        if i >= k:
                            grad[i] += h_current[i - k]
            
            # Gradient ascent step (projected)
            step_size = 0.1 / np.sqrt(i + 1)
            h_new = h_current + step_size * grad
            h_new = project_to_sum_box(h_new, M)
            
            # Evaluate new C5
            c5_new = evaluate_c5_exact(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Reduce step size
                step_size *= 0.5
                
            # Early termination if improvement is too small
            if abs(c5_current - best_c5) < 1e-6:
                break
                
        best_h = h_current.copy()
        best_c5 = c5_current
    
    return (best_h.tolist(), float(best_c5), len(best_h))
