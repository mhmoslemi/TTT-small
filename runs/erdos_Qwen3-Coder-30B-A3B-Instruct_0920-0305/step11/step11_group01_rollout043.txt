import numpy as np
from scipy.signal import correlate
from scipy.fftpack import fft, ifft
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_fft(h):
    """Compute the full correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max(corr, alpha):
    """Compute smooth maximum of correlation values with smoothing parameter alpha."""
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * corr)))

def convolve_fft(a, b):
    """Compute convolution of a and b using FFT."""
    L = len(a) + len(b) - 1
    L_fft = 2 ** int(np.ceil(np.log2(L)))
    return np.fft.irfft(np.fft.rfft(a, L_fft) * np.fft.rfft(b, L_fft), L)

def compute_c5(h):
    """Compute the C5 value using FFT for full correlation."""
    corr = correlation_fft(h)
    return dx * np.max(corr)

            # DE/rand/1 mutation
            r1, r2, r3 = np.random.choice(popsize, 3, replace=False)
            v = population[r1] + F * (population[r2] - population[r3])

            # Crossover
            u = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR or j == 0:
                    u[j] = v[j]

            # Repair
            u = project_to_sum_box(u, TARGET_SUM)

            # Selection
            u_c5 = compute_c5(u)
            if u_c5 < compute_c5(population[i]):
                population[i] = u

        for h in population:

            if c5 < best_c5:

            for i in flip_indices:
                rb[i] = 1 - rb[i]
            rb = project_to_sum_box(rb, TARGET_SUM)

            # Swap hill climbing
            for _ in range(max_swaps):
                if time.time() - start_time > (start_time + 0.1 * budget_s):
                    break

                i1 = np.random.randint(0, n_points)
                if rb[i1] != 1:
                    continue
                i0 = np.random.randint(0, n_points)
                if rb[i0] != 0:
                    continue

                # Estimate effect on C5
                old_corr = correlation_fft(rb)
                old_max = np.max(old_corr)

                # Try swap
                rb[i1], rb[i0] = rb[i0], rb[i1]
                new_corr = correlation_fft(rb)
                new_max = np.max(new_corr)

                if new_max < old_max:
                    # Accept swap
                    pass
                else:
                    # Reject swap
                    rb[i1], rb[i0] = rb[i0], rb[i1]

            # Evaluate final binary version
            final_c5 = compute_c5(rb)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = rb.copy()

    # Return
    if best_h is None:
        best_h = h0.copy()

    return (best_h.tolist(), float(best_c5), n_points)
