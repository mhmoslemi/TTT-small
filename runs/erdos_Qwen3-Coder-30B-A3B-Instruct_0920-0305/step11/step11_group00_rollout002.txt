import numpy as np
import time
from scipy.signal import correlate

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        fm = f(lam)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = lam
        else:
            lam_low = lam
    
    result = np.clip(x + lam, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_gradient_single_shift(h, k):
    """Compute gradient of a single shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def compute_gradient_full(h):
    """Compute weighted sum of gradients for all shifts"""
    corr = compute_correlation(h)
    n = len(h)
    max_c = np.max(corr)
    
    # Safe computation of weights
    exp_arg = 10 * (corr - max_c)  # Scale for numerical stability
    exp_arg = np.clip(exp_arg, -50, 50)  # Avoid overflow
    exp_vals = np.exp(exp_arg)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute weighted gradient
    grad = np.zeros_like(h)
    for i in range(-n+1, n):
        if i == 0:
            continue  # Skip zero shift
        grad_k = compute_gradient_single_shift(h, i)
        grad += weights[i+n-1] * grad_k  # Index mapping
    return grad

def accelerated_projection_gradient(h, alpha, eta, beta, max_iter):
    """Accelerated projected gradient descent for smooth surrogate"""
    h_prev = h.copy()
    y = h.copy()
    
    for _ in range(max_iter):
        # Compute gradient
        corr = compute_correlation(y)
        max_c = np.max(corr)
        
        # Safe weights
        exp_arg = alpha * (corr - max_c)
        exp_arg = np.clip(exp_arg, -50, 50)
        exp_vals = np.exp(exp_arg)
        weights = exp_vals / np.sum(exp_vals)
        
        # Compute gradient
        grad = np.zeros_like(y)
        n = len(y)
        for i in range(-n+1, n):
            if i == 0:
                continue
            grad_k = compute_gradient_single_shift(y, i)
            grad += weights[i+n-1] * grad_k
            
        # Line search
        eta_local = eta
        h_tmp = y - eta_local * grad
        h_new = project_to_sum_box(h_tmp, len(h) / 2)
        corr_new = compute_correlation(h_new)
        
        for _ in range(5):
            if np.max(corr_new) < np.max(corr) - 1e-12:
                break
            eta_local /= 2
            h_tmp = y - eta_local * grad
            h_new = project_to_sum_box(h_tmp, len(h) / 2)
            corr_new = compute_correlation(h_new)
        
        # Update
        if eta_local > 0:
            y = h_new + beta * (h_new - h_prev)
            h_prev = h_new.copy()
        else:
            y = h_new
            
    return h_new

def subgradient_polishing(h, max_iter):
    """Subgradient polishing for exact maximum"""
    eta = 0.08
    last_improvement = 0
    
    for i in range(max_iter):
        if i - last_improvement > 200:
            break
            
        corr = compute_correlation(h)
        max_val = np.max(corr)
        tol = 1e-12
        max_indices = np.where(np.abs(corr - max_val) < tol)[0]
        
        # Average gradient of all maximum shifts
        avg_grad = np.zeros_like(h)
        for idx in max_indices:
            k = idx - len(h) + 1
            avg_grad += compute_gradient_single_shift(h, k)
        avg_grad /= len(max_indices)
        
        # Update step
        h_tmp = h - eta * avg_grad
        h_new = project_to_sum_box(h_tmp, len(h) / 2)
        corr_new = compute_correlation(h_new)
        
        new_max = np.max(corr_new)
        if new_max < max_val - 1e-14:
            h = h_new
            last_improvement = i
            if eta < 0.15:
                eta *= 1.05
        else:
            eta /= 2
            if eta < 1e-6:
                break
                
    return h

def random_perturbation_escape(h, max_attempts=50):
    """Escape from local minima with random moves"""
    for _ in range(max_attempts):
        i, j = np.random.choice(len(h), 2, replace=False)
        delta = 0.02 * np.random.random()  # Small random move
        if h[i] >= delta:
            h[i] -= delta
            h[j] += delta
            h = project_to_sum_box(h, len(h) / 2)
            return h
    return h
