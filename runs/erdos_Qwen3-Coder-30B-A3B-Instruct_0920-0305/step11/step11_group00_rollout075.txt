import numpy as np
import cvxpy as cp
import time

from scipy.sparse import csr_matrix
import scipy.sparse as sp

def proj_sum_box(v, target):
    """Project vector v onto sum = target with box constraints [0,1]"""
    v = np.array(v)
    low, high = -np.max(v) - 1, 1 - np.min(v) + 1
    for _ in range(30):
        mid = (low + high) / 2
        proj = np.clip(v + mid, 0, 1)
        sum_proj = np.sum(proj)
        if sum_proj < target:
            high = mid
        else:
            low = mid
    return np.clip(v + (low + high) / 2, 0, 1)

def corr_full(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Gradient of correlation with shift k"""
    grad = np.zeros_like(h)
    if k > 0:
        grad[:-k] = 1 - h[k:]
        grad[-k:] = 1
    elif k < 0:
        grad[-k:] = 1 - h[:-k]
        grad[:k] = 1
    else:
        grad[:] = 1 - h
    return grad

def max_overlap(h):
    """Compute maximum overlap (C5)"""
    corr = corr_full(h)
    return np.max(corr)

def build_sdp(n):
    """Build SDP problem"""
    # Precompute shift masks
    shift_masks = []
    for k in range(-(n-1), n):
        mask = np.zeros((n, n))
        if k >= 0:
            mask[np.arange(n-k), np.arange(k, n)] = 1
        else:
            mask[np.arange(-k, n), np.arange(n+k)] = 1
        shift_masks.append(csr_matrix(mask))
    
    # Variables
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Overlap constraints
    for k, M in zip(range(-(n-1), n), shift_masks):
        constraints.append(cp.trace(M @ X) >= (n/2) * (1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau, shift_masks

def solve_sdp(prob, solver='SCS'):
    """Solve SDP"""
    try:
        prob.solve(solver=solver, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL:
            return True, prob.value
        else:
            return False, float('inf')
    except:
        return False, float('inf')

def round_from_sdp(h_sdp, X_sdp, n):
    """Three rounding methods"""
    candidates = []
    
    # 1. Diagonal rounding
    h1 = h_sdp.copy()
    candidates.append(h1)
    
    # 2. Leading eigenvector rounding
    try:
        w, v = np.linalg.eigh(X_sdp)
        if w[-1] > 1e-8:
            h2 = np.sqrt(w[-1]) * np.abs(v[:, -1])
            h2 = proj_sum_box(h2, n/2)
            candidates.append(h2)
    except:
        pass
    
    # 3. Gaussian randomization
    try:
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(n), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, n/2)
            candidates.append(h3)
    except:
        pass
    
    # Evaluate candidates and return best
    best_h = None
    best_c5 = float('inf')
    for h in candidates:
        c5 = max_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

    """Polish with projected gradient descent"""
    h = h_seed.copy()
    best_h = h.copy()
    best_c5 = max_overlap(h) * dx
    
    h_prev = h.copy()
    step = 0.07
    beta = 0.9
    
    start_time = time.time()
    
    for it in range(2000):
        if time.time() - start_time > budget_s - 10:
            break
            
        corr = corr_full(h)
        Cmax = np.max(corr)
        Kmax = np.where(corr >= Cmax - 1e-12)[0]
        
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov extrapolation
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new) * dx
        
        if c5_new < best_c5:
            step = min(step * 1.08, 0.2)
            best_h = h_new.copy()
            best_c5 = c5_new
            h_prev = h.copy()
            h = h_new.copy()
        else:
            step *= 0.5
            
        # Swap move every 250 iterations
        if it % 250 == 0:
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] - delta, 0, 1)
            h_new[j] = np.clip(h_new[j] + delta, 0, 1)
            h_new = proj_sum_box(h_new, n/2)
            
            c5_new = max_overlap(h_new) * dx
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new.copy()
    
    return best_h, best_c5
