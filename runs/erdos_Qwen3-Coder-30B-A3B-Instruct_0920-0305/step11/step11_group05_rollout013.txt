import numpy as np
from scipy.fft import fft, ifft
import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum_entries, Constraint, maximum

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """Project vector v onto box [0,1] with sum = target_sum using sorting method."""
    n = len(v)
    sorted_indices = np.argsort(-v)  # descending order
    sorted_v = v[sorted_indices]
    cumsum = np.cumsum(sorted_v)
    
    # Find threshold
    threshold = 0.0
    if cumsum[-1] < target_sum:
        # All values will be 1
        projected = np.clip(v, 0, 1)
    else:
        # Need to find where to clip
        for i in range(n):
            if cumsum[i] <= target_sum:
                continue
            # Compute the value to cap at for index i
            if i == 0:
                threshold = target_sum / (i + 1)
            else:
                threshold = (target_sum - cumsum[i-1]) / (n - i)
            break
        # Clip the sorted values
        sorted_projected = np.clip(sorted_v, 0, threshold)
        # Adjust for sum constraint
        current_sum = np.sum(sorted_projected)
        diff = target_sum - current_sum
        # Add uniformly to the lower elements
        if diff > 0 and n - i > 0:
            base_adjustment = diff / (n - i)
            sorted_projected[i:] = np.clip(sorted_projected[i:] + base_adjustment, 0, 1)
        # Reassemble
        projected = np.zeros_like(v)
        projected[sorted_indices] = sorted_projected
    return projected

def compute_corr_full(h):
    """Fast correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT correlation."""
    corr = compute_corr_full(h)
    return dx * np.max(corr)

def solve_linearized_step(y):
    """Given fixed y, solve the linearized LP to maximize lower bound on min correlation."""
    # Build LP
    n = len(y)
    x = Variable(n)
    t = Variable()
    
    constraints = [t <= sum_entries(x[i] * y[np.clip(i+s, 0, n-1)] 
                                    for i in range(n)) 
                   for s in range(-n+1, n)]
    constraints.append(sum_entries(x) == TARGET_SUM)
    constraints.extend([x[i] >= 0 for i in range(n)])
    constraints.extend([x[i] <= 1 for i in range(n)])
    
    prob = Problem(Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    if prob.status in ["optimal", "optimal_inaccurate"]:
        x_sol = np.array(x.value).reshape(-1)
    else:
        # Fallback to projection when optimization failed
        x_sol = project_onto_sum_box(y, TARGET_SUM)
    
    t_sol = prob.value if prob.status in ["optimal", "optimal_inaccurate"] else -1e6
    return x_sol, t_sol
