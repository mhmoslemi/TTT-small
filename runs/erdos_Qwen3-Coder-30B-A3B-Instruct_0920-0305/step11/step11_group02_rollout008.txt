n_points = 300

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def compute_gradient(h, weights):
    """Compute gradient from weights via FFT"""
    # Reversing weights for convolution with h
    w_rev = weights[::-1]
    
    L = 2 * len(h) - 1
    conv_plus  = np.fft.ifft(np.fft.fft(weights, L) * np.fft.fft(h, L)).real
    conv_rev   = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract central values
    n = len(h)
    grad = -(1/TARGET_SUM) * (conv_plus[n-1:] + conv_rev[n-1:])
    
    return grad
