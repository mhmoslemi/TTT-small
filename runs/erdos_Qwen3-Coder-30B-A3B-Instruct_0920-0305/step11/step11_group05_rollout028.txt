import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling projection
    x = np.array(x)
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x_clipped
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x - lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x - (a + b)/2, 0, 1)
    return result

def log_sum_exp(x, tau):
    """Compute log(sum(exp(x/tau))) with numerical stability"""
    x_max = np.max(x)
    return x_max + tau * np.log(np.sum(np.exp((x - x_max) / tau)))

def compute_gradient(h, tau):
    """Compute gradient of smoothed max using FFT"""
    # FFT-based correlation
    h_fft = np.fft.fft(h)
    h1_fft = np.fft.fft(1 - h)
    corr = np.fft.ifft(h_fft.conj() * h1_fft).real
    
    # Compute weights (soft-max probabilities)
    g_s = dx * (TARGET_SUM - corr)
    max_g = np.max(g_s)
    exp_g = np.exp((g_s - max_g) / tau)
    p_s = exp_g / np.sum(exp_g)
    
    # Compute gradient terms
    p_fft = np.fft.fft(p_s)
    term1 = np.fft.ifft(p_fft.conj() * h1_fft).real
    term2 = np.fft.ifft(p_fft.conj() * h_fft).real
    
    grad = dx * (term1 - term2)
    return grad

def apg_with_backtracking(h0, tau, max_iter=3000, eta=1e-4):
    """Accelerated projected gradient with backtracking line search"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    
    for k in range(max_iter):
        # Compute gradient
        grad = compute_gradient(y, tau)
        
        # Backtracking line search
        alpha = 1.0
        fk = log_sum_exp(dx * (TARGET_SUM - compute_corr_fft(y)), tau)
        
        while True:
            h_new = project_to_sum_box(y - alpha * grad, TARGET_SUM)
            fk_new = log_sum_exp(dx * (TARGET_SUM - compute_corr_fft(h_new)), tau)
            
            if fk_new <= fk - eta * alpha * np.sum(grad**2):
                break
            alpha *= 0.5
    
        # Update
        h = h_new
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y = h + ((t - 1) / t_new) * (h - h0)
        h0 = h
        t = t_new
        
        if k % 100 == 0:
            # Check convergence
            if np.sum(np.abs(grad)) < 1e-6:
                break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Generate initial vectors
    initial_vecs = generate_initial_vectors()
    
    best_c5 = float('inf')
    best_h = None
    
    # Temperature schedule
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    
    for i, h0 in enumerate(initial_vecs):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        current_h = h0.copy()
        
        # Continue with different temperatures
        for tau in tau_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Run APG with backtracking
            current_h = apg_with_backtracking(current_h, tau, max_iter=3000)
            # Time check after each temperature stage
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Ensure sum constraint
            current_h = project_to_sum_box(current_h, TARGET_SUM)
        
        # Final evaluation
        current_c5 = evaluate_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final refinement with pairwise exchanges
    if best_h is not None:
        current_h = best_h.copy()
        current_c5 = best_c5
        remaining_time = budget_s - (time.time() - start_time)
        
        if remaining_time > 10:
            # Pairwise exchange refinement
            num_exchanges = min(1000, int(remaining_time * 50))
            for _ in range(num_exchanges):
                if time.time() - start_time > budget_s - 5:
                    break
                
                i, j = np.random.choice(n_points, 2, replace=False)
                delta_max = min(current_h[i], 1 - current_h[j])
                if delta_max > 0:
                    delta = np.random.uniform(-delta_max, delta_max)
                    h_test = current_h.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    h_test = project_to_sum_box(h_test, TARGET_SUM)
                    
                    test_c5 = evaluate_c5(h_test)
                    if test_c5 < current_c5:
                        current_h = h_test
                        current_c5 = test_c5
    
    if best_h is None:
        # Fallback to original
        h0 = np.array(initial_h_values)
        current_h = project_to_sum_box(h0, TARGET_SUM)
        current_c5 = evaluate_c5(current_h)
        
    return (current_h.tolist(), float(current_c5), n_points)
