import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def soft_max_surrogate(h, tau=0.05):
    """Compute soft-max surrogate for max_k integral."""
    corr = compute_corr_fft(h)
    corr = corr * dx
    # Add a small epsilon to avoid log(0)
    corr = corr + 1e-12
    exp_terms = np.exp(corr / tau)
    sum_exp = np.sum(exp_terms)
    return tau * np.log(sum_exp)

def compute_gradient_soft_max(h, tau=0.05):
    """Compute exact gradient of soft-max surrogate using FFT."""
    n = len(h)
    
    # Compute needed FFTs
    H_fft = fft(h)
    H1_fft = fft(1 - h)
    
    # Compute all correlation terms (in log domain for numerical stability)
    corr = np.fft.ifft(H_fft.conj() * H1_fft).real * dx
    
    # Avoid numerical issues near zero
    corr = corr + 1e-12
    
    exp_corr = np.exp(corr / tau)
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp
    
    # Compute weighted gradients
    W_fft = fft(weights)
    # Gradient term for 1 - h_{i+s}
    A_fft = fft(weights * (1 - h))
    # Gradient term for h_{i-s}
    B_fft = fft(weights * h)
    
    # Real result
    grad = dx * (np.fft.ifft(A_fft).real - np.fft.ifft(B_fft).real)
    
    return grad

def backtracking_line_search(h, grad, tau, eta=1e-4, max_iterations=10):
    """Backtracking line search for step size."""
    alpha = 1.0
    phi_old = soft_max_surrogate(h, tau)
    for _ in range(max_iterations):
        h_new = project_to_sum_box(h - alpha * grad, TARGET_SUM)
        phi_new = soft_max_surrogate(h_new, tau)
        armijo_condition = phi_new <= phi_old - eta * alpha * np.dot(grad, grad)
        if armijo_condition:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new

def accelerated_projected_gradient(h0, tau, max_iter=3000, eta=1e-4):
    """Accelerated projected gradient with backtracking line search."""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    
    for k in range(max_iter):
        # Compute gradient
        grad = compute_gradient_soft_max(y, tau)
        
        # Backtracking line search
        alpha, h_new = backtracking_line_search(y, grad, tau, eta)
        
        # Project
        h = project_to_sum_box(h_new, TARGET_SUM)
        
        # Accelerated step
        t_old = t
        t = (1 + np.sqrt(1 + 4 * t_old**2)) / 2
        y = h + ((t_old - 1) / t) * (h - h0)
        h0 = h
        
        # Check convergence (very basic check)
        if k > 10 and np.abs(soft_max_surrogate(h, tau) - soft_max_surrogate(h0, tau)) < 1e-6:
            break
            
    return h

        tau0 = 0.05
        gamma = 0.7
        tau_min = 1e-4
        tau = tau0
        
        current_h = h0.copy()

            final_h = accelerated_projected_gradient(current_h, tau, max_iter=2000)

            current_c5 = evaluate_c5(final_h)

                best_h = final_h.copy()
            
            # Update current solution and reduce temperature
            current_h = final_h
            tau *= gamma
            
        if time.time() - seed_start_time >= seed_time_budget:

        h_ref = best_h.copy()
        current_c5 = evaluate_c5(h_ref)
        
        # Pairwise exchange refinement
        for _ in range(2000):
            if time.time() - ref_start > ref_budget * 0.9:
                break
                
            # Pick two random indices
            i, j = np.random.randint(0, n_points, 2)
            if i == j:

            if delta_max <= 0:

            h_new = h_ref.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Project to sum constraint
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            new_c5 = evaluate_c5(h_new)

                current_c5 = new_c5

                current_c5 = new_c5

        final_c5 = evaluate_c5(h_ref)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_ref.copy()
    
    # Final safety check
    if best_h is not None:
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        final_c5 = evaluate_c5(best_h)
        best_c5 = final_c5
    
    if best_h is None:
        # Fallback if nothing was found
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), float(best_c5), n_points)
