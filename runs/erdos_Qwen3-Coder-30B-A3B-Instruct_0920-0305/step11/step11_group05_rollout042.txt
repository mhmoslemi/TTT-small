# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_corr_fft(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using sorting"""
    # Water-filling projection
    x = np.array(x)
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    n = len(x_sorted)
    
    # Find threshold
    threshold = 0
    for i in range(n):
        if cumsum[i] <= target_sum:
            threshold = x_sorted[i]
        else:
            break
    
    # Project
    projected = np.clip(x, 0, threshold)
    delta = target_sum - np.sum(projected)
    
    # Adjust values greater than threshold
    below_thresh = np.where(projected < threshold)[0]
    if len(below_thresh) > 0:
        adjusted = projected.copy()
        diff = delta / len(below_thresh)
        for i in below_thresh:
            adjusted[i] = min(1.0, adjusted[i] + diff)
        return adjusted
    
    return projected

def build_and_solve_lp(y):
    """Solve LP that maximizes t subject to:
    sum_i x_i * y_{i+s} >= t for all shifts s, and sum_i x_i = target_sum, 0 <= x_i <= 1
    """
    n = len(y)
    t = cp.Variable()
    x = cp.Variable(n)
    
    constraints = [cp_sum(x) == TARGET_SUM]
    for s in range(-n+1, n):
        # Compute shifted indices and values
        if s >= 0:
            shifted_y = y[s:]
            original_indices = np.arange(0, len(shifted_y))
        else:
            shifted_y = y[:s]
            original_indices = np.arange(-s, n)
        
        if len(shifted_y) > 0:
            # Compute correlation sum for this shift
            expr = cp_sum(cp.multiply(x[original_indices], shifted_y))
            constraints.append(expr >= t)
    
    constraints.extend([x[i] >= 0 for i in range(n)])
    constraints.extend([x[i] <= 1 for i in range(n)])
    
    objective = cp.Maximize(t)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000, abstol=1e-8, reltol=1e-8)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(x.value).flat, t.value
        else:
            return None, None
    except Exception as e:
        return None, None

            new_h[j] += delta

            new_h = project_to_sum_box(new_h, TARGET_SUM)

    # Final evaluation to make sure everything is okay
    try:
        corr_final = compute_corr_fft(final_h)
        final_c5 = c5_from_corr(corr_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
        best_h = final_h.copy()
    except:
        pass
    
    return (best_h.tolist(), float(best_c5), n_points)
