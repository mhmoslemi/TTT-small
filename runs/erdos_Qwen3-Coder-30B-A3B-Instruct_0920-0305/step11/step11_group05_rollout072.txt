import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """Fast projection using water-filling algorithm"""
    n = len(v)
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    
    # Simple water-filling approach: iteratively decrease highest values
    # until the sum equals the target
    remaining_sum = target_sum
    for i in range(n):
        if i > 0 and sorted_v[i] >= sorted_v[i-1]:
            # Already at a higher value, no need to go further
            break
        # Compute how much we can reduce current value while keeping sum right
        left_sum = np.sum(sorted_v[i:]) if i < n else 0
        if left_sum <= remaining_sum:
            # We can keep all of this part
            # nothing to do, just continue
            pass
        else:
            # adjust current part to make sum equal
            # s_k = sum over coefficients required to make sum = target_sum
            bar_value = (remaining_sum - (left_sum - sorted_v[i])) / (n - i)
            # clip individual values
            for j in range(i, n):
                sorted_v[j] = max(0, min(1, bar_value))
            break
        # Current segment is all above the value that would just bring us to target
        # not feasible, go to next one
    # Final adjustment
    # Copy back the sorted values
    result = v.copy()
    result[sorted_indices] = sorted_v
    return np.clip(result, 0, 1)

def corr_fft(h):
    """FFt-based full correlation to compute overlap"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5"""
    corr = corr_fft(h)
    return dx * np.max(corr)

def find_worst_shift(h):
    """Find shift that gives max overlap"""
    corr = corr_fft(h)
    return np.argmax(corr)

def solve_lp_step(y):
    """Solve the LP problem with frozen y"""
    n = len(y)
    # By definition, we have 2*n-1 constraints, one for each shift
    # Then we'll do the same to each constraint.
    x = cp.Variable(n)
    t = cp.Variable()
    
    # constraints
    constraints = []
    for s in range(-n+1, n):
        # Since indices might go out of range, use a mask or zero out
        # we define the shifted version of y appropriately
        # indices go from 0 to n-1 for standard indexing
        # but we simulate circular or zero padding with shifting by s
        cumsum = 0.0
        for i in range(n):
            j = i + s
            if 0 <= j < n:
                cumsum += x[i] * y[j]
            else:
                cumsum += 0.0  # out of bounds
        # Use cp.Pos to avoid negative infeasibility issue
        constraints.append(cumsum >= t)
        
    constraints.append(cp.sum(x) == TARGET_SUM)
    for i in range(n):
        constraints.append(x[i] <= 1)
        constraints.append(x[i] >= 0)
        
    # Now solve
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            return np.array(x.value), t.value
        else:
            # fallback to GLPK if ECOS fails for some reason
            try:
                prob.solve(solver=GLPK, verbose=False)
                if prob.status == 'optimal':
                    return np.array(x.value), t.value
            except:
                pass
    except:
        pass
    
    # Continue with fallback or return current point, e.g., wait and use simpler method.
    # For now, use the given y as approximation if we get an error
    return y, np.min([sum(x * np.roll(y, s)) for s in range(-n+1, n)])

def alternating_lp(h0, max_iter=50, tol=1e-8):
    """Main alternating LP solver loop"""
    h_current = h0.copy()
    t_best = -np.inf
    for _ in range(max_iter):
        x_new, t_new = solve_lp_step(h_current)
        if t_new > t_best + tol:
            h_current = x_new
            t_best = t_new
        else:
            break
    return h_current, t_best

    corr = corr_fft(h)
    c5_current = dx * np.max(corr)
    
    temp = init_temp
    for i in range(maxiter):
        if temp < 1e-8:
            break
        # Move
        i0 = np.random.randint(0, n_points)
        j0 = (i0 + np.random.randint(1, n_points)) % n_points
        
        # How much to move
        max_step = min(1.0 - h[i0], h[j0])
        if max_step == 0:
            continue
        step = np.random.rand() * max_step
        
        # Change h[i0], h[j0], and updates
        delta = step
        old_corr = corr.copy()
        
        # update array mappings (simplified - only update affects effect on each shift)
        # recompute only large components
        corr_new = corr.copy()
        for s in range(1-n_points, n_points):
            if 0 <= i0+s < n_points and 0 <= j0+s < n_points:
                corr_new[s] += delta*(h[j0+s] - h[i0+s])
        
        c5_proposed = dx * np.max(corr_new)
        
        if c5_proposed < c5_current:
            h[i0] -= delta
            h[j0] += delta
            corr = corr_new.copy()
            c5_current = c5_proposed
        else:
            if np.random.rand() < np.exp(-(c5_proposed - c5_current)/temp):
                h[i0] -= delta
                h[j0] += delta
                corr = corr_new.copy()
                c5_current = c5_proposed
                
        # Reduce temperature
        temp *= 0.9995
        
    return h, c5_current
