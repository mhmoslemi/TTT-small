import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def logsumexp(x, axis=None, keepdims=False):
    """Numerically stable logsumexp"""
    xmax = np.max(x, axis=axis, keepdims=True)
    return xmax + np.log(np.sum(np.exp(x - xmax), axis=axis, keepdims=keepdims))

def smooth_max(h, tau=0.01):
    """Compute smooth approximation to max using log-sum-exp"""
    corr = compute_corr_fft(h)
    corr = corr[0:2*n_points-1]  # Only the valid correlations
    return tau * logsumexp(corr/tau)

def smooth_gradient(h, tau=0.01):
    """Compute gradient of smooth max using FFT-based correlation"""
    # Correlations
    h_fft = np.fft.fft(h)
    h_rev = h[::-1]
    b = 1 - h
    b_rev = b[::-1]
    b_fft = np.fft.fft(b)
    
    # Full correlation 
    corr = np.fft.ifft(h_fft.conj() * b_fft).real
    corr_full = np.concatenate([corr[-(n_points-1):], corr])
    
    # Probabilities
    corr_scaled = corr_full / tau
    max_corr = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - max_corr)
    prob = exp_corr / np.sum(exp_corr)
    
    # Compute gradients using FFTs
    p_fft = np.fft.fft(prob)
    grad1 = np.fft.ifft(p_fft * b_fft).real
    grad2 = np.fft.ifft(p_fft * h_fft).real
    
    return dx * (grad1 - grad2)

def apg_with_linesearch(h0, max_iter=3000, tol=1e-6, eta=1e-4):
    """Accelerated projected gradient with backtracking line search"""
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    prev_f = smooth_max(h)
    
    for k in range(max_iter):
        # Compute gradient
        grad = smooth_gradient(y)
        
        # Backtracking line search
        alpha = 1.0
        while True:
            h_new = project_to_sum_box(y - alpha * grad, TARGET_SUM)
            f_new = smooth_max(h_new)
            f_approx = prev_f - eta * alpha * np.dot(grad, grad)
            
            if f_new <= f_approx:
                break
            alpha /= 2.0
            
            # Break if too small
            if alpha < 1e-10:
                break
        
        if alpha < 1e-10:
            break
            
        h = h_new
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y = h + ((t - 1) / t_new) * (h - h0)
        h0 = h
        t = t_new
        
        # Check convergence
        f = smooth_max(h)
        if abs(prev_f - f) < tol:
            break
        prev_f = f
    
    return h

    np.random.seed(seed)
    start_time = time.time()

    TARGET_SUM = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Define initial solutions
    initial_solutions = []

    h0 = project_to_sum_box(h0, TARGET_SUM)
    initial_solutions.append(h0.copy())
    
    # 2. Tent shape
    tent = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x <= 1:
            tent[i] = 2 * x
        else:
            tent[i] = 2 - 2 * x
    tent = project_to_sum_box(tent, TARGET_SUM)
    initial_solutions.append(tent.copy())
    
    # 3. Random solutions
    for _ in range(3):
        random_h = np.random.random(n_points)
        random_h = project_to_sum_box(random_h, TARGET_SUM)
        initial_solutions.append(random_h.copy())
    
    # Continuation with different temperatures
    temperatures = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    
    for i, h_start in enumerate(initial_solutions):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Warm start with current solution
        current_h = h_start.copy()
        
        # Continuation from high temperature to low
        for tau in temperatures:
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # APG with this temperature
            current_h = apg_with_linesearch(current_h, max_iter=2000, tau=tau)
            
            # Project to ensure constraints
            current_h = project_to_sum_box(current_h, TARGET_SUM)
        
        # Final evaluation
        current_c5 = evaluate_c5(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final refinement with pairwise exchange
    if best_h is not None and(time.time() - start_time < 0.8 * budget_s):
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Simulated annealing refinement
        temp = 0.1
        temp_decay = 0.999
        max_refinements = 10000
        
        for _ in range(max_refinements):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Random index pairs
            i, j = np.random.choice(n_points, 2, replace=False)
            
            # Compute allowable change
            delta_max = min(current_h[i], 1 - current_h[j])
            if delta_max <= 1e-10:
                continue
                
            # Sample change
            delta = np.random.uniform(-delta_max, delta_max)
            
            # Apply change
            h_new = current_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            
            # Project to constraints
            h_new = project_to_sum_box(h_new, TARGET_SUM)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            
            # Accept or reject
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
            else:
                if np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                    current_h = h_new
                    current_c5 = c5_new
            
            temp *= temp_decay
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
