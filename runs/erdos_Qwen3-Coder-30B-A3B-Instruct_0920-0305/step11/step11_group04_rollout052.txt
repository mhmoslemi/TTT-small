import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant using the official method"""
    # Compute the correlation using full mode
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find the maximum
    dx = 2.0 / len(h)
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n=100 for manageable MILP size while maintaining good resolution
    n = 100
    dx = 2.0 / n
    target_sum = n // 2
    
    # Create variables
    h = cp.Variable(n, integer=True)
    z = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1
    ]
    
    # Generate all valid shifts for correlation
    # We'll consider only k >= 0 due to symmetry
    correlation_constraints = []
    
    # Add auxiliary variables for McCormick envelopes
    y = {}
    shift_range = range(-n+1, n)
    
    # For each valid shift k and index i such that i+k is valid
    for k in shift_range:
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                # Create auxiliary variable for h[i] * h[j]
                y_key = (i, k)
                y[y_key] = cp.Variable()
                constraints.extend([
                    y[y_key] <= h[i],
                    y[y_key] <= h[j],
                    y[y_key] >= h[i] + h[j] - 1,
                    y[y_key] >= 0
                ])
                correlation_constraints.append(y[y_key])
    
    # Constraints for each shift k
    for k in shift_range:
        # Get indices for this shift
        valid_y_vars = [y[(i, k)] for i in range(n) if 0 <= i + k < n]
        if valid_y_vars:
            # Compute correlation for this shift
            rho_k = dx * cp.sum(valid_y_vars)
            # Constraint: z >= 1 - rho_k
            constraints.append(z >= 1 - rho_k)
    
    # Objective
    prob = cp.Problem(cp.Minimize(z), constraints)
    
    # Solve the problem
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        prob.solve(solver=cp.CBC, TimeLimit=remaining_time - 10, verbose=False)
    else:
        # Fallback to the previous method if no time for MILP
        return None, None, None
    
    # Extract solution
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h.value
        z_opt = z.value
    else:
        # If MILP failed, fall back to previous method
        return None, None, None
    
    # Polishing with greedy swaps (fast)
    h_opt = np.round(h_opt).astype(int)
    
    def compute_correlation(h_arr):
        corr = correlate(h_arr, 1 - h_arr, mode='full')
        return corr * dx
    
    def polish_h(h_arr, max_iterations=100):
        # Compute correlations
        corr = compute_correlation(h_arr)
        for _ in range(max_iterations):
            best_k = np.argmin(corr)
            min_corr = corr[best_k]
            
            # Skip if optimal
            if min_corr < 1e-10:
                break
            
            # Try swaps
            best_gain = -1.0
            swap_i = None
            swap_j = None
            
            # Check all possible swaps
            for i in range(n):
                if h_arr[i] == 0:
                    gain = 0
                    # Check the two possible neighbors for this k
                    j1 = i - best_k + n
                    j2 = i + best_k - n
                    
                    if 0 <= j1 < n and best_k != 0:
                        gain += h_arr[j1]
                    if 0 <= j2 < n and best_k != 0:
                        gain += h_arr[j2]
                    
                    if gain > best_gain:
                        best_gain = gain
                        swap_i = i
                
                elif h_arr[i] == 1:
                    loss = 0
                    # Check the two possible neighbors for this k
                    j1 = i - best_k + n
                    j2 = i + best_k - n
                    
                    if 0 <= j1 < n and best_k != 0:
                        loss += h_arr[j1]
                    if 0 <= j2 < n and best_k != 0:
                        loss += h_arr[j2]
                    
                    if loss > best_gain:
                        best_gain = loss
                        swap_j = i
            
            if swap_i is not None and swap_j is not None:
                h_arr[swap_i] = 1
                h_arr[swap_j] = 0
                corr = compute_correlation(h_arr)  # Recompute
            else:
                break
                
        return h_arr
    
    h_opt = polish_h(h_opt)
    
    # Final evaluate
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # Convert to list
    h_final = h_opt.tolist()
    
    return (h_final, float(c5_bound), n)
