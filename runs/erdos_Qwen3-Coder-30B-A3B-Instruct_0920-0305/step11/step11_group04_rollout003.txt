import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant using the official evaluator"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    n = 100
    dx = 2.0 / n
    target_sum = n // 2
    
    # Create variables for the binary step function
    h = cp.Variable(n, boolean=True)
    z = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == target_sum]
    
    # Add McCormick envelopes for all pairs
    # We only consider k >= 0 because correlation is symmetric
    y_vars = {}
    for k in range(n):
        for i in range(n - k):
            j = i + k
            y = cp.Variable()
            y_vars[(i, k)] = y
            constraints.extend([
                y <= h[i],
                y <= h[j],
                y >= h[i] + h[j] - 1,
                y >= 0
            ])
    
    # Add constraints for the correlation bounds
    for k in range(n):
        if k == 0:
            # Special case for k=0: correlation with itself
            rho_k = dx * cp.sum([y_vars[(i, k)] for i in range(n - k)])
        else:
            # Use symmetric property of correlation
            rho_k = dx * cp.sum([y_vars[(i, k)] for i in range(n - k)])
        constraints.append(z >= 1 - rho_k)
    
    # Objective
    prob = cp.Problem(cp.Minimize(z), constraints)
    
    # Solve with time limit
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 30:
        return (np.zeros(n).tolist(), 1.0, n)
    
    try:
        prob.solve(solver=cp.CBC, TimeLimit=remaining_time - 10)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value

            # Fallback to gradient-based approach if MILP fails
            return fallback_gradient_approach(seed, budget_s)
    except Exception:

        return fallback_gradient_approach(seed, budget_s)

    h_opt = np.round(h_opt)

                break

            k_star = np.argmin(corr)

            best_i = -1
            best_j = -1

                    gain_i = 0
                    if i + k_star < n:
                        gain_i += h_current[i + k_star]
                    if i - k_star >= 0:
                        gain_i += h_current[i - k_star]
                    
                    if gain_i > best_gain:
                        best_gain = gain_i
                        best_i = i
            
            if best_i >= 0:

                        if j - k_star >= 0:
                            loss_j += h_current[j - k_star]

                            h_current[j] = 0
                            break

    return (h_polished.tolist(), float(c5_bound), n)
