import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_fft_correlations(h):
    """Compute all correlations using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_corr_full(h):
    """Compute correlation using full mode"""
    h_rev = h[::-1]
    b = 1 - h
    corr = correlate(h, b, mode='full')
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_fft_correlations(h)
    return c5_from_corr(corr)

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort the values in descending order to apply water-filling
    sorted_indices = np.argsort(-x)
    sorted_vals = x[sorted_indices]
    
    # Compute the cumulative sum and check
    cumsum = np.cumsum(sorted_vals)
    n = len(sorted_vals)
    
    # Find the threshold
    s = 0
    for i in range(n):
        s += sorted_vals[i]
        # Check if sum of top i+1 elements is <= target_sum
        if s + (n - 1 - i) * sorted_vals[i] >= target_sum:
            break
    
    # The threshold value
    if i == n - 1:
        threshold = sorted_vals[i]
    else:
        # Use linear interpolation to find exact lambda
        if i < n - 1:
            diff = s - sorted_vals[i] + target_sum
            threshold = sorted_vals[i] - diff / (n - 1 - i)
        else:
            threshold = 0
    
    # Adjust values and project
    result = np.clip(x, 0, 1)
    if np.sum(result) > target_sum:
        keep = result >= threshold
        total_excess = np.sum(result[keep]) - target_sum
        result[keep] = np.maximum(result[keep] - total_excess / np.sum(keep), 0)
    elif np.sum(result) < target_sum:
        keep = result <= threshold
        total_deficit = target_sum - np.sum(result[keep])
        result[keep] = np.minimum(result[keep] + total_deficit / np.sum(keep), 1)
    
    return result

def soft_max_gradient(h, tau=0.01):
    """Compute soft max gradient using FFT"""
    # Compute correlation values
    corr = compute_fft_correlations(h)
    
    # Weights based on soft max
    corr_scaled = corr / tau
    max_corr = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - max_corr)
    sum_exp_corr = np.sum(exp_corr)
    weights = exp_corr / sum_exp_corr
    
    # Compute weighted gradient term
    # Gradient: dx * [sum_{s} w_s * (1 - h_{i+s}) - sum_{s} w_s * h_{i-s}]
    # We need to apply this for each i in [0, n-1]
    reordered_weights = np.roll(weights, len(weights)//2)  # Align with convolution
    # First compute A = sum_s w_s * (1 - h_{i+s})
    h_conj = np.conj(fft(1 - h))
    w_conj = np.conj(fft(reordered_weights))
    A = ifft(w_conj * h_conj).real
    
    # Then compute B = sum_s w_s * h_{i-s}  
    h_conj = np.conj(fft(h))
    w_conj = np.conj(fft(reordered_weights))
    B = ifft(w_conj * h_conj).real
    
    # Final gradient is dx * (A - B)
    return dx * (A - B)

def backtracking_line_search(h, grad, tau, max_iter=10):
    """Backtracking line search"""
    alpha = 1.0
    phi0 = np.sum(np.log(np.exp(compute_fft_correlations(h)/tau) + 1e-10))
    phi0 = tau * phi0
    
    for _ in range(max_iter):
        h_new = project_to_box_sum(h - alpha * grad, TARGET_SUM)
        g_new = np.sum(np.log(np.exp(compute_fft_correlations(h_new)/tau) + 1e-10))
        g_new = tau * g_new
        if g_new <= phi0 - 1e-4 * alpha * np.sum(grad**2):
            return h_new, alpha
        alpha *= 0.5
    return h_new, alpha

def accelerated_projected_gradient(h0, tau, max_iter=3000):
    """Accelerated projected gradient descent with backtracking"""
    h = h0.copy()
    y = h0.copy()
    t = 1
    for k in range(max_iter):
        # Compute gradient
        grad = soft_max_gradient(y, tau)
        
        # Backtracking line search
        h_new, _ = backtracking_line_search(y, grad, tau)
        
        # Update with acceleration
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y_new = h_new + ((t-1)/t_new) * (h_new - h)
        
        h = h_new
        y = y_new
        t = t_new
        
        # Check convergence based on function decrease
        if k > 10 and abs(np.sum(np.log(np.exp(compute_fft_correlations(h)/tau) + 1e-10))) < 1e-6:
            break
            
    return h

    mid = n_points // 2

        if i <= mid:
            tent_h[i] = 2 * i / n_points
        else:
            tent_h[i] = 2 * (n_points - i) / n_points
    tent_h = project_to_box_sum(tent_h, TARGET_SUM)
    initial_strategies.append(("tent", tent_h))

    for i in range(3):
        random_h = np.random.random(n_points)
        random_h = project_to_box_sum(random_h, TARGET_SUM)

            # Accelerated projected gradient with backtracking
            current_h = accelerated_projected_gradient(current_h, tau, max_iter=2000)
            
            # Early stopping if no benefit
            c5_val = evaluate_c5(current_h)

    # Final refinement using short simulated annealing
    if best_h is not None:
        current_h = best_h.copy()

                if i == j:

                # Get the old C5
                old_c5 = evaluate_c5(current_h)

                h_new = current_h.copy()

                # Project back to feasible set
                h_new = project_to_box_sum(h_new, TARGET_SUM)

    if best_h is None:
        # Default fallback
        h_default = np.array(initial_h_values)
        h_default = project_to_box_sum(h_default, TARGET_SUM)
        best_h = h_default.copy()
