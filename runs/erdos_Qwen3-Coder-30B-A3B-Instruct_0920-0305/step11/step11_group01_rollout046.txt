def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

    def f(lam):

        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2

            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def tournament_select(population, tournament_size=3):
    """Select an individual using tournament selection"""
    tournament = np.random.choice(len(population), tournament_size)
    tournament_fitness = [population[i][1] for i in tournament]
    winner_idx = tournament[np.argmin(tournament_fitness)]
    return population[winner_idx][0]

def crossover(parent1, parent2):
    """Arithmetic crossover"""
    alpha = np.random.random()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(individual, mutation_rate):
    """Gaussian mutation"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return mutated
