import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, sum_expr, mul_elemwise
import cvxpy as cp

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_onto_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum using O(n log n) water-filling"""
    v = np.array(v)
    n = len(v)
    
    if np.isclose(np.sum(v), target_sum):
        return np.clip(v, 0, 1)
    
    # Sort in descending order
    sorted_indices = np.argsort(-v)
    sorted_v = v[sorted_indices]
    
    # Find the threshold
    cumulative_sum = 0.0
    threshold_idx = 0
    
    for i in range(n):
        cumulative_sum += sorted_v[i]
        if cumulative_sum >= target_sum:
            threshold_idx = i
            break
        elif cumulative_sum - (i + 1) * sorted_v[i] < target_sum - cumulative_sum:
            pass
        else:
            threshold_idx = i
            break
    
    if threshold_idx >= 0 and threshold_idx < n:
        gamma = (target_sum - cumulative_sum + sorted_v[threshold_idx]) / (threshold_idx + 1)
        if gamma <= 0:
            gamma = 0
        elif gamma >= 1:
            gamma = 1
    else:
        # This shouldn't happen in typical cases
        gamma = 0.0
    
    # Apply thresholding
    result = np.array(v)
    for i in range(threshold_idx + 1):
        result[sorted_indices[i]] = gamma
    for i in range(threshold_idx + 1, n):
        result[sorted_indices[i]] = min(1.0, max(0.0, result[sorted_indices[i]]))
    
    # Re-project to make sure target sum is really hit
    current_sum = np.sum(result)
    diff = target_sum - current_sum
    
    # Simple adjustment via small increments/decrements
    if abs(diff) > 1e-10:
        # Sort indices by value
        diffs = result - 0.5
        sorted_idx = np.argsort(diffs)
        # Increment allowed elements
        i = 0
        while i < len(sorted_idx) and np.abs(diff) > 1e-10:
            idx = sorted_idx[i]
            if diff > 0 and result[idx] < 1:
                inc = min(diff, 1 - result[idx])
                result[idx] += inc
                diff -= inc
            elif diff < 0 and result[idx] > 0:
                dec = min(-diff, result[idx])
                result[idx] -= dec
                diff += dec
            i += 1
    
    return np.clip(result, 0, 1)

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_h(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def build_qp_matrix(y):
    """Build and solve LP for the current y"""
    # y is fixed; we want to solve min -t subject to:
    #  sum_i x_i * y_{i+s} >= t for all s,
    #  sum_i x_i = TARGET_SUM
    #  0 <= x_i <= 1
    n = len(y)
    x = Variable(n)
    t = Variable(1)
    
    # Build constraints
    constraints = [sum_expr(mul_elemwise(x, y)) >= t for y in build_shifted_vectors(y)]
    constraints += [sum(x) == TARGET_SUM]
    constraints += [x >= 0]
    constraints += [x <= 1]
    
    # We maximize t by minimizing -t
    prob = Problem(Minimize(-t), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        x_vals = x.value
        t_val = -prob.value
        return x_vals, t_val
    except Exception:
        # fallback to simpler approach
        return None, -np.inf

def build_shifted_vectors(v):
    """Build all shifted versions of v"""
    n = len(v)
    shifted = []
    for i in range(1 - n, n):
        if i < 0:
            part = v[-i:]
            pad = np.zeros(-i)
            shifted.append(np.concatenate([pad, part]))
        elif i == 0:
            shifted.append(v)
        else:
            part = v[:-i]
            pad = np.zeros(i)
            shifted.append(np.concatenate([part, pad]))
    return shifted

def alternate_lp(y0, max_iter=30):
    """Main alternating LP solver for improving the solution"""
    y = np.array(y0)
    best_y = y.copy()
    best_t = -np.inf
    
    for iteration in range(max_iter):
        # Create shifted vectors
        shifted = build_shifted_vectors(y)
        n = len(y)
        M = len(shifted)
        
        # Build LP: maximize t subject to:
        # sum_i x_i * y_{i+s} >= t   for all s
        # sum_i x_i = TARGET_SUM
        # 0 <= x_i <= 1
        
        x = cp.Variable(n)
        t = cp.Variable(1)
        
        constraints = []
        for i in range(M):
            # Compute dot product of x with shifted y[i]
            y_shifted = shifted[i]
            constraints.append(cp.sum(cp.multiply(x, y_shifted)) >= t)
        
        # Set sum constraint
        constraints.append(cp.sum(x) == TARGET_SUM)
        # Box constraints
        constraints.append(x >= 0)
        constraints.append(x <= 1)
        
        # Objective
        objective = cp.Minimize(-t)  # Since we want to maximize t
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                x_val = x.value
                t_val = -prob.value
                
                if t_val > best_t:
                    best_t = t_val
                    best_y = x_val.copy()
                
                y = x_val
            else:
                break
        except:
            break
    
    return best_y, best_t

def local_refinement(h, iterations=3000):

    current_h = h.copy()
    current_c5 = evaluate_h(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Correlation cache for incremental updates
    corr_cache = compute_corr_fft(current_h)
    
    temp = 0.5
    alpha = 0.9998
    for i in range(iterations):
        if i % 1000 == 0:
            temp *= alpha
            
        # Choose two indices at random
        i1, i2 = np.random.randint(0, len(current_h), 2)
        
        # Calculate how much we can move
        delta = np.random.uniform(-0.01, 0.01)
        delta = max(-current_h[i1], delta)
        delta = min(1 - current_h[i2], delta)
        
        if abs(delta) < 1e-8:
            continue
            
        # Propose the move
        new_h = current_h.copy()
        new_h[i1] = current_h[i1] + delta
        new_h[i2] = current_h[i2] - delta
        
        # Ensure feasibility
        if new_h[i1] < 0:
            new_h[i1] = 0
        elif new_h[i1] > 1:
            new_h[i1] = 1
        if new_h[i2] < 0:
            new_h[i2] = 0
        elif new_h[i2] > 1:
            new_h[i2] = 1
            
        # Re-project to enforce sum
        new_h = project_onto_sum_box(new_h, TARGET_SUM)
        
        # Quick evaluation using FFT
        new_c5 = evaluate_h(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            prob = np.exp(-(new_c5 - current_c5) / (temp + 1e-10))
            if np.random.rand() < prob:
                current_h = new_h
                current_c5 = new_c5
        
        if i % 1000 == 0:
            temp = max(1e-6, temp * 0.99)
    
    return best_h, best_c5
