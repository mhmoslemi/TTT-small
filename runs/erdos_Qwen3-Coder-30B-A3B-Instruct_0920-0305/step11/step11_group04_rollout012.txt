import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant using the official evaluator"""
    # This would normally call the imported function
    # For now, we'll compute it directly
    h = np.array(h)
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller discretization for tractability
    n = 100
    dx = 2.0 / n
    target_sum = n // 2
    
    # Create the MILP model
    # Decision variables
    h = cp.Variable(n, boolean=True)
    y = cp.Variable((n, 2*n-1), nonneg=True)  # y[i,k] for k in [-n+1, n-1]
    z = cp.Variable(nonneg=True)
    
    # Constraints
    constraints = [
        cp.sum(h) == target_sum
    ]
    
    # McCormick envelopes for product h[i] * h[i+k]
    for k in range(-(n-1), n):
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                # y[i,k] represents h[i] * h[j]
                constraints.extend([
                    y[i, k + n - 1] <= h[i],
                    y[i, k + n - 1] <= h[j],
                    y[i, k + n - 1] >= h[i] + h[j] - 1,
                    y[i, k + n - 1] >= 0
                ])
    
    # Add correlation constraints
    for k in range(-(n-1), n):
        idx = k + n - 1
        if k >= 0:
            corr_term = cp.sum(y[:, idx]) * dx
        else:
            # For negative k, we need to shift appropriately
            corr_term = cp.sum(y[:-k, idx]) * dx
        constraints.append(z >= 1 - corr_term)
    
    # Objective: minimize z
    prob = cp.Problem(cp.Minimize(z), constraints)
    
    # Solve with time limit
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 30:
        # Not enough time for MILP solve, fallback to heuristic
        return run_heuristic(seed, budget_s, n, dx, target_sum)
    
    try:
        prob.solve(solver=cp.CBC, TimeLimit=remaining_time - 10, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            z_opt = z.value
        else:
            # If solver failed, return best heuristic solution
            return run_heuristic(seed, budget_s, n, dx, target_sum)
    except:
        # If solver fails completely, fallback to heuristic
        return run_heuristic(seed, budget_s, n, dx, target_sum)
    
    # Polish with local search if time permits
    if time.time() - start_time < budget_s - 10:
        h_opt = polish_solution(h_opt, n, dx)
    
    # Ensure h_opt is binary and has correct sum
    h_opt = np.round(h_opt).astype(int)
    current_sum = np.sum(h_opt)
    if current_sum != target_sum:
        # Adjust to maintain sum constraint
        diff = target_sum - current_sum
        if diff > 0:  # Need to turn some 0s to 1s
            zero_indices = np.where(h_opt == 0)[0]
            for idx in zero_indices[:diff]:
                h_opt[idx] = 1
        elif diff < 0:  # Need to turn some 1s to 0s
            one_indices = np.where(h_opt == 1)[0]
            for idx in one_indices[:abs(diff)]:
                h_opt[idx] = 0
    
    c5_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt.tolist(), float(c5_bound), n)

def project_to_sum(h, target_sum):
    """Project onto the sum constraint using water-filling approach"""
    h = np.array(h)
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Sort descending to apply water filling from the top
    indices = np.argsort(h)[::-1]
    h_sorted = h[indices]
    
    remaining = target_sum - current_sum
    
    if remaining > 0:
        # Need to increase values
        for i in range(len(h_sorted)):
            if remaining <= 0:
                break
            increase = min(1 - h_sorted[i], remaining)
            h_sorted[i] += increase
            remaining -= increase
    else:
        # Need to decrease values
        remaining = abs(remaining)
        for i in range(len(h_sorted)):
            if remaining <= 0:
                break
            decrease = min(h_sorted[i], remaining)
            h_sorted[i] -= decrease
            remaining -= decrease
    
    # Restore original ordering
    h_new = h.copy()
    for i, orig_idx in enumerate(indices):
        h_new[orig_idx] = h_sorted[i]
    
    return h_new

def polish_solution(h_opt, n, dx):
    """Heuristic local search to improve solution"""
    h = h_opt.copy()
    max_iterations = 500
    
    for _ in range(max_iterations):
        # Compute correlations
        corr_full = correlate(h, h, mode='full')
        # We want to minimize the maximum correlation, but we're computing 1 - rho_k
        # The original problem we want to solve is minimizing max_k 1-rho_k = min_k rho_k
        # But evaluating the actual integral is more direct
        
        # Convert to original definition of C5 
        # Equivalent to max_k int h(x)(1-h(x+k))dx = max_k (1 - int h(x)h(x+k)dx) = 1 - min_k int h(x)h(x+k)dx
        
        # Here, correlation is {i,k} -> h[i]*h[i+k] (where idx is shifted by k)
        # For max_k int h(x)(1-h(x+k))dx = max_k (1 - rho_k) = max_k (1 - dx*sum(h[i]*h[i+k]))
        # So we want to maximize min_k rho_k = min_k (dx * sum(h[i]*h[i+k]))
        
        # Therefore minimize max_k (1 - rho_k)
        # Which we're already doing by minimizing z in the program
        
        # Let's just compute final C5 and make sure it's correct here
        corr_full = np.correlate(h, 1 - h, mode='full')
        c5_current = dx * np.max(corr_full)
        
        # Make single small swaps to try to improve
        k_best = np.argmin(np.correlate(h, h, mode='full')) - (n-1)
        k_best = min(n-1, max(-n+1, k_best))
        
        # Try to improve the worst correlation by swapping elements
        changed = False
        for i in range(n):
            if h[i] == 0:
                # Try moving a 1 from far away to here
                j = (i + k_best) % n
                if h[j] == 1:
                    h_test = h.copy()
                    h_test[i] = 1
                    h_test[j] = 0
                    c5_test = evaluate_erdos_solution(h_test)
                    if c5_test < c5_current:
                        h = h_test.copy()
                        c5_current = c5_test
                        changed = True
                        break
            elif h[i] == 1:
                # Try moving a 0 to here (swap)
                j = (i + k_best) % n
                if j >= 0 and j < n and h[j] == 0:
                    h_test = h.copy()
                    h_test[i] = 0
                    h_test[j] = 1
                    c5_test = evaluate_erdos_solution(h_test)
                    if c5_test < c5_current:
                        h = h_test.copy()
                        c5_current = c5_test
                        changed = True
                        break
        
        if not changed:
            break
            
    return h
