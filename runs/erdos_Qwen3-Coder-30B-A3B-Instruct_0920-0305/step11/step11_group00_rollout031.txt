```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n // 2
    # Linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    # Symmetric second half
    if n % 2 == 0:
        second = first[::-1]
    else:
        second = first[::-1][1:]
    
    h = np.concatenate([first, second])
    # Adjust to match target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def random_feasible_vector(n, target_sum):
    """Create random feasible vector"""
    h = np.random.uniform(0, 1, n)
    h = project_to_sum_box(h, target_sum)
    return h

def smooth_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using smooth approximation gradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    # Parameters
    alphas = np.linspace(0.5, 30, 15)
    step = 0.5
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for iter in range(200):
            if time.time() - start_time > time_limit:
                break
            c = compute_corr(h)
            
            # Compute weights
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute gradient
            grad = np.zeros_like(h)
            n = len(h)
            base = np.arange(n) + (n-1)
            padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
            
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n-1)
                term1 = 1 - padded[base + s]
                term2 = padded[base - s]
                grad += weight * (term1 - term2)
            
            # Line search for step size
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
    
    return best_h, best_val

def subgradient_optimize(h0, target_sum, dx, time_limit, start_time):
    """Optimize using subgradient descent"""
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h)
    
    step = 0.3
    for iter in range(500):
        if time.time() - start_time > time_limit:
            break
        c = compute_corr(h)
        max_c = np.max(c)
        
        # Find all indices achieving maximum
        tol = 1e-10
        max_indices = np.where(np.abs(c - max_c) < tol)[0]
        
        # Compute average gradient of all maximizing shifts
        grad = np.zeros_like(h)
        for idx in max_indices:
            s = idx - (len(h) - 1)
            if s >= 0:
                # h_{j+s} defined for j from 0 to n-1-s
                term1 = np.empty(len(h))
                term1[:len(h)-s] = 1 - h[s:]
                term1[len(h)-s:] = 1
                # h_{j-s} defined for j >= s
                term2 = np.empty(len(h))
                term2[:s] = 0
                term2[s:] = h[:len(h)-s]
            else:
                s_abs = -s
                # h_{j+s}=h_{j-|s|}
                term1 = np.empty(len(h))
                term1[:len(h)-s_abs] = 1 - h[s_abs:]
                term1[len(h)-s_abs:] = 1
                # h_{j+s} = h_{j+|s|} 
                term2 = np.empty(len(h))
                term2[:s_abs] = 0
                term2[s_abs:] = h[:len(h)-s_abs]
            
            grad += term1 - term2
        
        grad = grad / len(max_indices)
        
        # Line search for step size
        step_cur = step
        for _ in range(5):
            h_new = h - step_cur * grad
            h_new = project_to_sum_box(h_new, target_sum)
            val_new = c5_value(h_new)
            if val_new < best_val - 1e-12:
                h = h_new
                best_val = val_new
                best_h = h.copy()
                break
            else:
                step_cur *= 0.5
    
    return best_h, best_val

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    best_h = None
    best_c5 = np.inf
    best_n = n_points
    
    # Try multiple configurations
    configs = [
        (120, 'triangular'),
        (200, 'triangular'),
        (120, 'random'),
        (200, 'random')
    ]
    
    for n, init_type in configs:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n
        target_sum = n / 2
        
        try:
            # Initialize
            if init_type == 'triangular':
                h0 = triangular_initial(n)
            else:
                h0 = random_feasible_vector(n, target_sum)
            
            # Smooth optimization
            h_smooth, val_smooth = smooth_optimize(h0, target_sum, dx, budget_s * 0.8, start_time)
            
            # Subgradient optimization
            h_sub, val_sub = subgradient_optimize(h_smooth, target_sum, dx, budget_s * 0.95, start_time)
            
            # Keep best solution
            if val_sub < best_c5:
                best_h = h_sub
                best_c5 = val_sub
                best_n = n
                
        except Exception as e:
            continue
    
    return (best_h.tolist(), float(best_c5), best_n)
```
