import numpy as np

import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def correlation_fft(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlation_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Convolve using FFT. Returns same length as b"""
    L = len(a) + len(b) - 1
    L_fft = 1
    while L_fft < L:
        L_fft <<= 1
    A = np.fft.fft(a, L_fft)
    B = np.fft.fft(b, L_fft)
    return np.fft.ifft(A * B).real[:len(b)]

def smooth_max(corr, alpha):
    """Smoothed max for gradient computation"""
    exp_corr = np.exp(alpha * corr)
    w = exp_corr / np.sum(exp_corr)
    return w

def compute_gradient(h, corr, alpha):
    """Compute gradient of smooth max using FFT convolution"""
    w = smooth_max(corr, alpha)
    # conv1 = w * h
    conv1 = convolve_fft(w, h)
    # conv2 = w_rev * h
    w_rev = w[::-1]
    conv2 = convolve_fft(w_rev, h)
    
    # Gradient is 1 - conv1 - conv2
    g = 1.0 - conv1 - conv2
    return g

def line_search(h, g, target_sum, alpha):
    """Line search to find suitable step size"""
    h_orig = h.copy()
    c5_orig = compute_c5(h_orig)
    
    eta_min = 1e-5
    eta = 0.5
    for _ in range(20):  # Max steps
        h_new = project_to_sum_box(h_orig - eta * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_orig:
            return h_new, eta
        eta /= 2.0
        if eta < eta_min:
            break
    return h_orig, eta
