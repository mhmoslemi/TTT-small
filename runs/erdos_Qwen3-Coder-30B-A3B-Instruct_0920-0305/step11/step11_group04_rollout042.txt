import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * np.max(corr)
