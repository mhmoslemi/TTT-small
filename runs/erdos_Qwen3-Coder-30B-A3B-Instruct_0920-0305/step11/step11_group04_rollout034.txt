import numpy as np
from scipy.linalg import eigh
from scipy.sparse import diags
import cvxpy as cp
import time
from scipy.signal import correlate

n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_shift_matrices(n):
    """Build shift matrices for all possible shifts"""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create diagonal matrix with ones on the k-th diagonal
        offsets = [k]
        data = [np.ones(max(0, n - abs(k)))]
        shift_matrix = diags(data, offsets, shape=(n, n), format='csr')
        shift_matrices.append(shift_matrix)
    return shift_matrices

def solve_sdp(shift_matrices, n):
    """Solve the SDP relaxation"""
    # Variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        X >> 0,           # Positive semidefinite
        cp.diag(X) == 1,  # Diagonal is 1
        cp.trace(X) == n, # Trace constraint for balancing
    ]

    # Add shift constraints
    for k, S_k in enumerate(shift_matrices):
        # The constraint: trace(S_k @ X) >= 2*n*(1 - t) - n_k
        # where n_k is the number of valid pairs for shift k
        n_k = n - abs(k - (n-1))
        constraints.append(cp.trace(S_k @ X) >= 2*n*(1 - t) - n_k)

    # Objective
    objective = cp.Minimize(t)

    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
    except:
        pass
    return None, None

def round_solution(X, n, R=100):
    """Round the PSD solution to binary vectors"""
    # Eigenvalue decomposition
    eigenvals, eigenvecs = eigh(X)
    # Keep only positive eigenvalues
    positive_mask = eigenvals > 1e-8
    eigenvals = eigenvals[positive_mask]
    eigenvecs = eigenvecs[:, positive_mask]
    
    # Compute root of eigenvalues
    sqrt_eigenvals = np.sqrt(eigenvals)
    V_sqrt = eigenvecs @ np.diag(sqrt_eigenvals)
    
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(R):
        # Sample random vector g
        g = np.random.randn(V_sqrt.shape[1])
        # Compute signs
        signs = np.sign(V_sqrt.T @ g)
        # Convert to binary
        h = (signs + 1) / 2
        # Adjust to have exactly n/2 ones
        current_sum = np.sum(h)
        if current_sum > TARGET_SUM:
            # Flip smallest values to zero
            indices_to_flip = np.argsort(h)[:int(current_sum - TARGET_SUM)]
            h[indices_to_flip] = 0
        elif current_sum < TARGET_SUM:
            # Flip largest values to one
            indices_to_flip = np.argsort(h)[-int(TARGET_SUM - current_sum):]
            h[indices_to_flip] = 1
            
        # Compute C5
        corr = correlate(h, 1-h, mode='full')
        c5 = dx * np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

                        if shift_idx == i:

                        elif shift_idx == i + k:
                            gain += 1  # 1-h[i] x h[j] term
                        
                # More accurately compute change in overlap at shift k
                if i + k >= 0 and i + k < len(h):
                    old_contribution = h[i] * (1-h[i+k])
                    new_contribution = 1 * (1-h[i+k])
                    gain = new_contribution - old_contribution

                    best_gain = gain

            # Apply swap

            h[best_j] = 1 - h[best_j]
            # Recompute correlation
            corr = correlate(h, 1-h, mode='full')
            current_c5 = dx * np.max(corr)
        else:
            break
            
    return h, current_c5

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlate(h, 1-h, mode='full')
    return dx * np.max(corr)
