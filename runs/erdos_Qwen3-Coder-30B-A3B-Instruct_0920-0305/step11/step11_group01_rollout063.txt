import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def correlation_fft(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = correlation_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def convolve_fft(a, b):
    """Convolve two arrays using FFT"""
    L = len(a) + len(b) - 1
    L = 1 << int(math.ceil(math.log2(L)))
    A = fft(a, L)
    B = fft(b, L)
    return np.real(ifft(A * B))

def smooth_max(corr, alpha):
    """Smooth maximum of correlation values"""
    exp_corr = np.exp(alpha * corr)
    return (1/alpha) * np.log(np.sum(exp_corr))

def smooth_max_gradient(corr, alpha, h):
    """Compute gradient of smooth max with respect to h"""
    exp_corr = np.exp(alpha * corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Compute convolutions needed: w * h and w_rev * h  
    conv1 = convolve_fft(weights, h)
    conv2 = convolve_fft(weights[::-1], h)
    
    # Gradient: 1 - (w * h) - (w_rev * h)
    grad = 1.0 - conv1 - conv2
    return grad
