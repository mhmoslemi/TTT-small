import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr(h):
    """Compute linear correlation using FFT"""
    # Use full mode to get correlation over all shifts
    corr = correlate(h, h, mode='full')
    return corr * dx

def compute_C5(h):
    """Evaluate C5 from h"""
    corr = compute_corr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water filling"""
    # Sort elements in descending order
    sorted_x = np.sort(x)[::-1]
    # Compute cumulative sums
    cumsum_x = np.cumsum(sorted_x)
    # Find threshold
    n = len(x)
    # We need sum of clipped values to be target_sum
    # We will set the first k elements to 1, and others to their clipped values
    # Using water-filling: find largest k such that sum of first k elements (clipped)
    # is <= target_sum (since we want to fill up with 1s first) then adjust remaining
    # We know 0 <= x_i <= 1 so the worst case is we take all 1s (sum = n)
    # But we want exactly target_sum
    # Method: binary search on the threshold lambda
    # We define a lambda such that when we add lambda to each x, then clip at [0,1],
    # the sum equals target_sum.
    
    # Actually, let's compute sorted x
    # Water filling: start with sorted x descending: x_sorted
    # Initially assume all are at 1. If not enough, reduce some.
    # The idea: send cumulative sum:
    # We want: sum_i min(1, x_i + lambda) == target_sum
    # This is equivalent to finding lambda such that
    # sum_i(max(0, min(1, x_i + lambda))) = target_sum
    # This can be computed via water-filling method:
    # Let lambda = (target_sum - last_cumsum) / n_remaining
    
    # Simpler approach: we can compute this inside a binary search range.
    
    # binary search for the best lambda:
    # Let f(lam) = sum_i(max(0, min(1, x_i + lam))) - target_sum
    
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    # find bracketing interval
    while f(a) * f(b) > 0:
        a -= 0.01
        b += 0.01
    
    # Binary search for zero of f
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-12:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    # The projection is now x + (a+b)/2 with clipping
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence with exactly n/2 ones."""
    # Generate Thue-Morse sequence up to length n, then truncate and make sure sum = n/2
    # Thue-Morse has the property that exactly half of the bits are 1 if length = 2^k
    # For anything else, we will round the appropriate number to get sum = n/2
    if n <= 0:
        return np.zeros(n)
    # Try with power of 2 at least n (power of 2 used for FFT efficiency)
    m = 1
    while m < n:
        m *= 2
    # Generate up to m
    tm = np.zeros(m, dtype=int)
    tm[0] = 0
    for i in range(1, m):
        tm[i] = 1 - tm[i//2]
    
    # Take first n, but adjust so sum is exactly n/2
    result = tm[:n].astype(float)
    current_sum = np.sum(result)
    
    if current_sum > n/2:
        # Need to flip some 1s to 0s
        idxs = np.where(result == 1)[0]
        # Figure out how many to flip by reducing sum by (current_sum - n/2)
        to_flip = int(current_sum - n/2)
        # flip elements with smallest indices to reduce sum
        flip_indices = idxs[:to_flip]
        result[flip_indices] = 0
    elif current_sum < n/2:
        # Need to flip some 0s to 1s
        idxs = np.where(result == 0)[0]
        to_flip = int(n/2 - current_sum)
        flip_indices = idxs[:to_flip]
        result[flip_indices] = 1
    
    return result

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()

    # Initialize
    TARGET_SUM = n_points // 2
    best_c5 = float('inf')
    best_h = None
    
    # Define helper functions at top level for simplicity
    def compute_corr(h):
        corr = correlate(h, h, mode='full')
        return corr * dx

    def compute_C5(h):
        corr = compute_corr(h)
        return 1.0 - np.min(corr)

    def project_to_sum_box(x, target_sum):
        # Sort elements in descending order
        sorted_x = np.sort(x)[::-1]
        # Binary search for lambda
        def f(lam):
            projected = np.clip(x + lam, 0, 1)
            return np.sum(projected) - target_sum
        
        a, b = -1.0, 1.0
        # find bracketing interval
        while f(a) * f(b) > 0:
            a -= 0.01
            b += 0.01
        
        # Binary search for zero of f
        for _ in range(50):
            m = (a + b) / 2
            fm = f(m)
            if abs(fm) < 1e-12:
                break
            if fm * f(a) < 0:
                b = m
            else:
                a = m
        
        # The projection is now x + (a+b)/2 with clipping
        result = np.clip(x + (a + b)/2, 0, 1)
        return result

    def thue_morse_binary(n):
        # Implementation from above
        if n <= 0:
            return np.zeros(n)
        m = 1
        while m < n:
            m *= 2
        tm = np.zeros(m, dtype=int)
        tm[0] = 0
        for i in range(1, m):
            tm[i] = 1 - tm[i//2]
        result = tm[:n].astype(float)
        current_sum = np.sum(result)
        if current_sum > n/2:
            idxs = np.where(result == 1)[0]
            to_flip = int(current_sum - n/2)
            flip_indices = idxs[:to_flip]
            result[flip_indices] = 0
        elif current_sum < n/2:
            idxs = np.where(result == 0)[0]
            to_flip = int(n/2 - current_sum)
            flip_indices = idxs[:to_flip]
            result[flip_indices] = 1
        return result

    def greedy_swap_local_search(h, dx, budget_s, start_time):
        # Implementation from above, embedded here because top-level required
        corr = correlate(h, h, mode='full')
        min_correlation_idx = np.argmin(corr)
        min_shift = min_correlation_idx - (n_points - 1)
        
        while time.time() - start_time < budget_s - 1.0:
            change_made = False
            min_corr_idx = np.argmin(corr)
            best_shift = min_corr_idx - (n_points - 1)
            best_net_gain = -1e-10
            pair_i0, pair_i1 = None, None
            
            indices_0 = np.where(h == 0)[0]
            indices_1 = np.where(h == 1)[0]
            if len(indices_0) == 0 or len(indices_1) == 0:
                break
                
            num_tests = min(1000, len(indices_0) * len(indices_1))
            test_pairs = []
            for i0 in indices_0:
                for i1 in indices_1:
                    test_pairs.append((i0, i1))
                    if len(test_pairs) >= num_tests:
                        break
                if len(test_pairs) >= num_tests:
                    break
                    
            for i0, i1 in test_pairs:
                gain = 0.0
                loss = 0.0
                
                # gain from increasing h[i0] = 1 (and esp. for the min shift)
                if 0 <= i0 + best_shift < n_points:
                    if h[i0 + best_shift] == 1:
                        gain += dx
                if 0 <= i0 - best_shift < n_points:
                    if h[i0 - best_shift] == 1:
                        gain += dx
                        
                # loss from decreasing h[i1] = 0  
                if 0 <= i1 + best_shift < n_points:
                    if h[i1 + best_shift] == 1:
                        loss += dx
                if 0 <= i1 - best_shift < n_points:
                    if h[i1 - best_shift] == 1:
                        loss += dx
                        
                net = gain - loss
                if net > best_net_gain and net > 1e-15:
                    best_net_gain = net
                    pair_i0, pair_i1 = i0, i1
            
            if pair_i0 is not None and pair_i1 is not None:
                h[pair_i0] = 1
                h[pair_i1] = 0
                change_made = True
            else:
                break

        return h

    # List of seeds
    seeds = []
    
    # Seed 1: Thue-Morse binary
    seeds.append(thue_morse_binary(n_points))
    
    # Seed 2: Random uniform start
    r = np.random.rand(n_points)
    seeds.append(project_to_sum_box(r, TARGET_SUM))
    
    # Seed 3: Provided initial_h_values
    if 'initial_h_values' in globals():
        # interpolate or truncate/extend
        orig = np.array(initial_h_values)
        if len(orig) > n_points:
            # truncate
            h_interp = orig[:n_points]
        elif len(orig) < n_points:
            # pad or interpolate
            h_interp = np.interp(np.linspace(0, len(orig), n_points), np.arange(len(orig)), orig)
        else:
            h_interp = orig.copy()
        seeds.append(project_to_sum_box(h_interp, TARGET_SUM))
        
    # Main optimisation loop across all seeds
    for h0 in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h = h0.copy()
        
        # Optimisation parameters
        tau = 0.2
        alpha = 0.15
        max_iter = 3000
        
        last_improvement = 0
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            corr = compute_corr(h)
            c5 = 1.0 - np.min(corr)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = it
            
            if it - last_improvement > 200:
                break
                
            # Gradient computation for smooth min
            w = np.exp(-corr / tau)
            w /= w.sum()
            
            g = np.zeros(n_points)
            for shift in range(-(n_points-1), n_points):
                wk = w[shift + n_points - 1]
                if wk == 0.0:
                    continue
                lo = max(0, -shift)
                hi = min(n_points, n_points-shift)
                if lo < hi:
                    g[lo:hi] += wk * h[lo+shift:hi+shift]
                    g[lo+shift:hi+shift] += wk * h[lo:hi]
                    
            g *= dx
            
            # Ascent step + projection
            h_candidate = project_to_sum_box(h + alpha * g, TARGET_SUM)
            
            corr_new = compute_corr(h_candidate)
            c5_new = 1.0 - np.min(corr_new)
            
            if c5_new < c5:
                h = h_candidate
                alpha = min(alpha * 1.02, 0.5)
            else:
                alpha *= 0.5
            tau *= 0.99
    
    # Binary polishing after gradient phase
    if best_h is not None:
        # Binarise by keeping the largest target_sum entries as 1
        idx = np.argsort(-best_h)[:TARGET_SUM]
        h_bin = np.zeros(n_points)
        h_bin[idx] = 1.0
        # Greedy swap local search
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        c5_bin = compute_C5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    return (best_h.tolist(), float(best_c5), n_points)
