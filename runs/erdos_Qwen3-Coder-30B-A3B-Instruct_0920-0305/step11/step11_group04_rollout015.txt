import numpy as np
from scipy.sparse import diags
from scipy.linalg import eigh
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return np.max(corr) * (2.0 / len(h))

def compute_shift_matrices(n):
    """Compute shift matrices efficiently"""
    shifts = []
    for k in range(-(n-1), n):
        # Create diagonal matrix for shift k
        offsets = [k]
        data = [1] * (n - abs(k))
        shift_mat = diags(data, offsets, shape=(n, n), format='csr')
        shifts.append(shift_mat)
    return shifts

def solve_sdp(n, shifts):
    """Solve SDP relaxation for the problem"""
    # Create variables
    X = cp.Variable((n, n), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == 1,  # diag(X) = 1
        t >= 0,
        t <= 1
    ]
    
    # Balanced constraint
    J = np.ones((n, n))
    constraints.append(cp.trace(J @ X) == 0)
    
    # Shift constraints
    for k, shift_mat in enumerate(shifts):
        # trace(S_k @ X) >= 2n*(1 - t) - n_k
        n_k = n - abs(k - (n-1))
        constraints.append(cp.trace(shift_mat @ X) >= 2 * n * (1 - t) - n_k)
        
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            return X.value, t.value
    except:
        pass
    return None, None

def round_solution(X, n):
    """Round PSD solution to binary vectors"""
    # Eigenvalue decomposition
    eigenvals, eigenvecs = eigh(X)
    # Keep only positive eigenvalues
    mask = eigenvals > 1e-6
    eigenvals = eigenvals[mask]

    # Compute V * sqrt(Lambda)
    V_sqrt = eigenvecs @ np.diag(np.sqrt(eigenvals))
    
    best_h = None
    best_c5 = float('inf')
    R = 100  # Number of random rounds
    
    for r in range(R):
        # Sample random vector
        g = np.random.randn(V_sqrt.shape[1])
        # Compute signs
        signs = np.sign(V_sqrt.T @ g)
        h = (signs + 1) / 2
        
        # Adjust to have exactly n/2 ones
        current_ones = int(np.sum(h))
        target_ones = n // 2
        
        if current_ones > target_ones:
            # Flip some ones to zeros
            diffs = np.abs(h - 0.5)
            flip_indices = np.argsort(diffs)[:current_ones - target_ones]
            h[flip_indices] = 0
        elif current_ones < target_ones:
            # Flip some zeros to ones
            diffs = np.abs(h - 0.5)
            flip_indices = np.argsort(diffs)[:target_ones - current_ones]
            h[flip_indices] = 1
            
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return best_h, best_c5

        best_j = -1

        for i in range(len(h)):
            if h[i] == 0:
                gain = 0

                for k in range(-(len(h)-1), len(h)):
                    new_corr = 0

                        new_corr += h[i+k] * (1 - h[i])  # h[i] was 0, now 1
                        new_corr += h[i] * (1 - h[i+k])  # h[i+k] was unchanged
                        gain += new_corr - h[i] * (1-h[i+k]) - h[i+k] * (1-h[i])
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    
        # For each one index, compute loss if flipped to zero
        for j in range(len(h)):
            if h[j] == 1:
                loss = 0
                for k in range(-(len(h)-1), len(h)):
                    if j+k >= 0 and j+k < len(h):
                        loss += (1-h[j]) * h[j+k]  # j was 1, now 0
                        loss += h[j] * (1-h[j+k])  # j+k unchanged
                if loss > best_gain:
                    best_gain = loss
                    best_j = j
                    
        if best_i >= 0 and best_gain > 0:
            # Perform swap
            h[best_i] = 1
            h[best_j] = 0
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            break
            
    return best_h, best_c5
