```python
import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_fft(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max(corr, alpha):
    """Compute smooth max with parameter alpha"""
    exp_corr = np.exp(alpha * corr)
    return (1.0 / alpha) * np.log(np.sum(exp_corr))

def compute_gradient(h, corr, alpha):
    """Compute gradient of smooth max using FFT-based convolution"""
    exp_corr = np.exp(alpha * corr)
    w = exp_corr / np.sum(exp_corr)
    
    # Convolution with w
    L = 2 * len(h) - 1
    w_extended = np.zeros(L)
    w_extended[:len(w)] = w
    h_extended = np.zeros(L)
    h_extended[:len(h)] = h
    
    conv1 = np.fft.ifft(np.fft.fft(w_extended) * np.fft.fft(h_extended)).real[:len(h)]
    
    # Convolution with w_rev
    w_rev = w[::-1]
    w_rev_extended = np.zeros(L)
    w_rev_extended[:len(w_rev)] = w_rev
    conv2 = np.fft.ifft(np.fft.fft(w_rev_extended) * np.fft.fft(h_extended)).real[:len(h)]
    
    return 1.0 - conv1 - conv2

def line_search(h, grad, eta_min=1e-5):
    """Simple backtracking line search"""
    eta = 0.5
    h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
    while True:
        try:
            corr_new = correlation_fft(h_new)
            c5_new = dx * np.max(corr_new)
            corr_old = correlation_fft(h)
            c5_old = dx * np.max(corr_old)
            if c5_new < c5_old:
                return h_new, eta
        except:
            pass
        eta /= 2
        if eta < eta_min:
            break
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
    return h, eta

def swap_hill_climb(h, max_swaps=5*n_points):
    """Perform hill-climb swaps on binary version of h"""
    # Convert to binary by median thresholding
    t = np.median(h)
    b = (h >= t).astype(float)
    
    # Fix sum to be exactly TARGET_SUM
    ones_count = np.sum(b)
    if ones_count > TARGET_SUM:
        indices = np.where(b == 1)[0]
        to_flip = indices[:int(ones_count - TARGET_SUM)]
        b[to_flip] = 0
    elif ones_count < TARGET_SUM:
        indices = np.where(b == 0)[0]
        to_flip = indices[:int(TARGET_SUM - ones_count)]
        b[to_flip] = 1
    
    # Now do swaps
    corr = correlation_fft(b)
    best_c5 = dx * np.max(corr)
    best_b = b.copy()
    
    np.random.seed(42)
    swaps_done = 0
    for _ in range(max_swaps):
        if swaps_done >= max_swaps:
            break
        # Pick a 1 and a 0 at random
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Make swap
        b_swapped = b.copy()
        b_swapped[i] = 0
        b_swapped[j] = 1
        
        # Update correlation efficiently 
        # For a swap between i (1) and j (0),
        # we need to update correlations at shifts:
        # k such that |k| = |i-j| or k such that a specific element 
        # in the correlation gets affected.
        # For simplicity just recompute full corr.
        try:
            corr_swapped = correlation_fft(b_swapped)
            c5_swapped = dx * np.max(corr_swapped)
            if c5_swapped < best_c5:
                best_c5 = c5_swapped
                best_b = b_swapped.copy()
                swaps_done = 0  # Reset counter on improvement
            else:
                swaps_done += 1
        except:
            swaps_done += 1
    return best_b, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global n_points, dx, TARGET_SUM
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize DE population
    popsize = 30
    population = [project_to_sum_box(np.random.rand(n_points), TARGET_SUM) for _ in range(popsize)]
    
    # Evaluate initial population
    fitness = [dx * np.max(correlation_fft(h)) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # DE parameters
    F = 0.5
    CR = 0.9
    alpha = 4.0
    
    # Stage 1: Differential Evolution (60% of time)
    iteration = 0
    de_end_time = start_time + 0.6 * budget_s
    while time.time() < de_end_time:
        iteration += 1
        for i in range(popsize):
            # Mutation
            r1, r2, r3 = np.random.choice(popsize, 3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            
            # Crossover
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR or j == (iteration % n_points):
                    trial[j] = mutant[j]
            
            # Repair
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Selection
            try:
                corr_trial = correlation_fft(trial)
                c5_trial = dx * np.max(corr_trial)
                if c5_trial < fitness[i]:
                    population[i] = trial
                    fitness[i] = c5_trial
                    if c5_trial < best_c5:
                        best_h = trial.copy()
                        best_c5 = c5_trial
            except:
                pass
                
            # Self-adaptation (every 10 generations)
            if iteration % 10 == 0 and iteration > 0:
                F = np.clip(F + np.random.normal(0, 0.05), 0.1, 1.5)
                CR = np.clip(CR + np.random.normal(0, 0.05), 0.1, 1.0)
    
    # Stage 2: Smoothed max gradient descent (30% of time)
    h = best_h.copy()
    grad_end_time = start_time + 0.9 * budget_s
    while time.time() < grad_end_time:
        try:
            corr = correlation_fft(h)
            c5 = dx * np.max(corr)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            
            grad = compute_gradient(h, corr, alpha)
            h, _ = line_search(h, grad, 1e-5)
            
            # Increase alpha for sharpness
            alpha = min(alpha * 1.5, 200)
        except Exception as e:
            break
    
    # Stage 3: Binary rounding and swap hill-climb (10% of time)
    binary_h, binary_c5 = swap_hill_climb(best_h, max_swaps=5*n_points)
    if binary_c5 < best_c5:
        best_h = binary_h
        best_c5 = binary_c5
    
    return (best_h.tolist(), float(best_c5), n_points)
```
