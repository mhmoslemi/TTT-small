import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Clip to [0,1]
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x.copy()
    
    # Water-filling algorithm to adjust to exact sum
    # Sort values in descending order
    sorted_indices = np.argsort(x)[::-1]
    sorted_x = x[sorted_indices]
    
    # Find the threshold value
    cumsum = np.cumsum(sorted_x)
    thresholds = (cumsum - target_sum) / np.arange(1, len(sorted_x) + 1)
    
    # Find where the threshold must be applied
    i = np.searchsorted(thresholds, 0, side='right')
    if i == 0:
        # All values are above threshold; not possible given our constraints
        threshold = 0
    else:
        threshold = thresholds[i-1]
    
    # Apply threshold
    new_x = np.clip(x - threshold, 0, 1)
    return new_x

def compute_soft_max_grad(h, tau):
    """Compute gradient of log-sum-exp surrogate"""
    corr = compute_corr_fft(h)
    # Compute softmax probabilities
    corr_scaled = corr / tau
    max_corr = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - max_corr)
    sum_exp = np.sum(exp_corr)
    p = exp_corr / sum_exp
    
    # Compute gradients for each shift
    # Gradient of g_s(h) = dx * (S - sum_i h_i * h_{i+s}) w.r.t h_i is dx * (1 - h_{i+s} - h_{i-s})
    # But since h_{i+s} is circular, we can compute: (1-h_{i+s})*p_s - h_{i-s}*p_s for each s
    # Let's compute it more efficiently using FFT
    
    # FFT approach:
    # double convolution using idft
    
    h_fft = np.fft.fft(h)
    one_minus_h_fft = np.fft.fft(1 - h)
    p_fft = np.fft.fft(p)
    
    A = np.fft.ifft(p_fft.conj() * one_minus_h_fft).real  # This includes the (1 - h_{i+s}) terms
    B = np.fft.ifft(p_fft.conj() * h_fft).real  # This includes the h_{i-s} terms
    
    grad = dx * (A - B)
    return grad

def accelerated_projected_gradient(h_start, tau, max_iter=3000, eta=1e-4):
    """Accelerated projected gradient for minimizing Phi_tau"""
    h = h_start.copy()
    y = h_start.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for k in range(max_iter):
        # Compute gradient
        grad = compute_soft_max_grad(y, tau)
        
        # Backtracking line search
        alpha = 1.0
        max_alpha = 10.0
        # Ensure we don't make too large steps
        while alpha > 1e-10:
            # Step forward
            h_candidate = y - alpha * grad
            h_candidate = project_to_sum_box(h_candidate, TARGET_SUM)
            
            # Check if the surrogate decreased
            corr = compute_corr_fft(h_candidate)
            phi_candidate = tau * np.log(np.sum(np.exp(corr / tau)))  # Approx for numerical stability
            
            # Now evaluate actual C5 (our real objective)
            c5_candidate = evaluate_c5(h_candidate)
            
            # If this is better in the surrogate sense, also check actual objective
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
            
            # Avoid the need to compute the full surrogate for every iteration
            if alpha == 1.0:
                if c5_candidate < evaluate_c5(h):
                    break
                else:
                    # For the first iteration, if the move didn't improve,
                    # suppress learning rate for the next try again
                    alpha *= 0.5
            else:
                alpha *= 0.5
                
        if alpha < 1e-10:
            break  # Safe exit
            
        # Update with accelerated step
        t_old = t
        t = (1 + np.sqrt(1 + 4 * t_old * t_old)) / 2
        h = h_candidate.copy()
        y = h + ((t_old - 1) / t) * (h - h_candidate)
        y = project_to_sum_box(y, TARGET_SUM)
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    
    # Initial vectors
    initial_seeds = []
    
    # Seed 1: the provided initial solution
    h0 = project_to_sum_box(INITIAL_H, TARGET_SUM)
    initial_seeds.append(h0.copy())
    
    # Seed 2: Tent-shaped
    tent_h = np.array([min(i, 2 * n_points - 1 - i) / (n_points - 1) for i in range(n_points)])
    tent_h = project_to_sum_box(tent_h, TARGET_SUM)
    initial_seeds.append(tent_h.copy())
    
    # Seed 3-5: random vectors
    for _ in range(3):
        rand_h = np.random.random(n_points)
        rand_h = project_to_sum_box(rand_h, TARGET_SUM)
        initial_seeds.append(rand_h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    remaining_budget = budget_s
    
    for i, seed_h in enumerate(initial_seeds):
        if time.time() - start_time >= budget_s - 2:  # Leave time for final refinement
            break
            
        local_start = time.time()
        local_budget = remaining_budget / len(initial_seeds) - 1  # Leave some buffer
        
        # Perform continuation with decreasing temperatures
        current_h = seed_h.copy()
        tau = 0.05
        
        while tau > 1e-4 and (time.time() - local_start) < local_budget:
            # Run accelerated projected gradient
            h_new, c5_new = accelerated_projected_gradient(current_h, tau)
            
            # Update best solution
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            current_h = h_new.copy()
            # Reduce temperature
            tau *= 0.7
            
        remaining_budget -= (time.time() - local_start)
    
    if best_h is None:
        best_h = INITIAL_H.copy()
        best_h = project_to_sum_box(best_h, TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
    
    # Final refinement using pairwise exchange
    if time.time() - start_time < budget_s - 2:
        current_h = best_h.copy()
        current_c5 = best_c5
        temp = 0.05
        
        for iteration in range(5000):
            if time.time() - start_time >= budget_s - 1:
                break
                
            # Pick two random indices
            i, j = np.random.randint(0, n_points, 2)
            if i == j:
                continue
                
            # Calculate allowable shift
            delta_max = min(current_h[i], 1 - current_h[j])
            if delta_max < 1e-6:
                continue
                
            # Sample small shift
            delta = np.random.uniform(-delta_max, delta_max)
            
            # Compute new values
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            
            # Evaluate new C5
            new_c5 = evaluate_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / (temp + 1e-6)):
                    current_h = new_h.copy()
                    current_c5 = new_c5
            
            # Cool down temperature
            temp *= 0.999
    
    if best_h is None:
        return (INITIAL_H.tolist(), evaluate_c5(INITIAL_H), n_points)
        
    # Final verification
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
