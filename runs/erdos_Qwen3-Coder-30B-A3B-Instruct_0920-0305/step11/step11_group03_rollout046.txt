def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute full autocorrelation using FFT"""
    # Zero pad for full correlation
    pad_len = 2 * len(h) - 1
    h_padded = np.pad(h, (0, pad_len - len(h)), mode='constant')
    h_fft = np.fft.fft(h_padded)
    autocorr = np.fft.ifft(h_fft * np.conj(h_fft)).real
    # Take the first half (positive lags)
    autocorr = autocorr[:len(h)]
    return autocorr * dx

def softmin_weights(R, beta):
    """Compute softmin weights for given autocorrelations and beta"""
    R = np.array(R)
    exp_neg_beta_R = np.exp(-beta * R)
    weights = exp_neg_beta_R / np.sum(exp_neg_beta_R)
    return weights

def gradient_from_weights(weights, h):
    """Compute gradient from softmin weights"""
    # Weights for each shift, compute gradient via convolution
    w_full = np.zeros(2 * len(h) - 1)
    w_full[len(h)-1:] = weights  # forward shifts
    w_full[:len(h)-1] = weights[:0:-1]  # reverse backward shifts
    
    # Convolution trick
    grad = -np.convolve(w_full, h, mode='valid')[len(h)-1:]
    return grad

def compute_c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_autocorrelation(h)
    # Compute the full correlation with 1-h in one go
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr_full = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    # Take the part of correlation that corresponds to k in [0,2]
    corr_positive = corr_full[len(h)-1:]
    return compute_c5_from_corr(corr_positive)

def gradient_of_objective(h, beta):
    """Compute gradient of smooth objective 1 - softmin(R)"""
    R = compute_autocorrelation(h)
    weights = softmin_weights(R, beta)
    return gradient_from_weights(weights, h)

def projected_gradient_descent(h0, beta, max_iter, eta0=0.1):
    """Projected gradient descent for fixed beta"""
    h = h0.copy()
    for i in range(max_iter):
        grad = gradient_of_objective(h, beta)
        eta = eta0 / np.sqrt(i + 1)
        h_new = h - eta * grad
        h = project_to_sum_box(h_new, n_points / 2)
        if i % 50 == 0:
            # Add small noise to escape plateaus
            noise = np.random.normal(0, 1e-3, len(h))
            h = project_to_sum_box(h + noise, n_points / 2)
    return h
