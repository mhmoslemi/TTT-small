import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse import diags

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum using binary search"""
    v = np.array(v)
    low, high = -1.0, 1.0
    tol = 1e-12
    
    def eval_sum(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    # Ensure bounds are correct
    while eval_sum(low) * eval_sum(high) > 0:
        low -= 0.1
        high += 0.1
        
    # Binary search for lambda
    for _ in range(50):
        mid = (low + high) / 2
        if abs(eval_sum(mid)) < tol:
            break
        if eval_sum(mid) > 0:
            high = mid
        else:
            low = mid
    
    result = np.clip(v + (low + high)/2, 0, 1)
    return result

def corr_full(h):
    """Fast evaluation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Return max overlap value (C5)"""
    corr = corr_full(h)
    return dx * np.max(corr)

def grad_shift(h, k):
    """Gradient of c_k = sum h_i (1 - h_{i+k})"""
    n = len(h)
    if k >= 0:
        g = np.zeros(n)
        g[:-k] = 1 - h[k:]  # gradient from h_i * (1 - h_{i+k})
        g[k:] -= h[:-k]     # gradient from -(h_{i-k} * h_i)
        return g
    else:
        g = np.zeros(n)
        g[-k:] = 1 - h[:k]   # gradient from h_i * (1 - h_{i+k})
        g[:k] -= h[-k:]      # gradient from -(h_{i-k} * h_i)
        return g

def polish(h_seed, n, dx, remaining_budget):
    """Polish with projected gradient descent on true max overlap"""
    start_time = time.time()
    
    h = h_seed.copy()
    h_prev = h_seed.copy()
    step = 0.07
    beta = 0.9
    
    best_h = h.copy()
    best_c5 = max_overlap(h, dx)
    
    for it in range(2000):
        if time.time() - start_time > remaining_budget - 5:
            break
            
        c = corr_full(h)
        Cmax = np.max(c)
        Kmax = np.where(c >= Cmax - 1e-12)[0]
        if len(Kmax) == 0:
            Kmax = [0]  # fallback
            
        # Gradient ascent: average gradients over argmax indices
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov step
        y = h + beta * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        c5_new = max_overlap(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.08, 0.2)
        else:
            step *= 0.5
            
        h_prev = h
        h = h_new
        
        # Stochastic swap move every 250 iterations
        if it % 250 == 0:
            i, j = np.random.randint(0, n, 2)
            delta = np.random.uniform(-0.02, 0.02)
            h_new[i] -= delta
            h_new[j] += delta
            h_new = proj_sum_box(h_new, n/2)
            c5_new = max_overlap(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return best_h, best_c5

def shift_mask(k, n):
    """Generate sparse shift mask matrix for shift k"""
    if k >= 0:
        rows = np.arange(n-k)
        cols = np.arange(k, n)
    else:
        rows = np.arange(-k, n)
        cols = np.arange(n+k)
    data = np.ones(len(rows))
    return csr_matrix((data, (rows, cols)), shape=(n,n))

def solve_sdp(n):
    """Solve SDP relaxation with cvxpy"""
    h = cp.Variable(n)
    X = cp.Variable((n,n), PSD=True)
    tau = cp.Variable()
    
    # Constraints
    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    # Add overlap constraints
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    
    # Solve with very high tolerance
    prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
    
    if prob.status != cp.OPTIMAL:
        # Try cheap fallback if infeasible
        h_sdp = np.clip(np.random.rand(n), 0, 1)
        h_sdp = proj_sum_box(h_sdp, n/2)
        return h_sdp, None, 1.0
    
    return h.value, X.value, tau.value
    
def round_from_sdp(h_sdp, X_sdp):
    """Try several rounding methods"""
    best_h = h_sdp
    best_c5 = max_overlap(h_sdp, 2.0/len(h_sdp))
    
    # Method 1: Diagonal rounding
    h1 = h_sdp.copy()
    c5_1 = max_overlap(h1, 2.0/len(h_sdp))
    if c5_1 < best_c5:
        best_h = h1
        best_c5 = c5_1
    
    # Method 2: Leading eigenvector rounding
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_sdp)
        u = eigenvecs[:, -1]
        lambda_max = eigenvals[-1]
        h2 = np.sqrt(max(lambda_max, 0)) * u
        h2 = proj_sum_box(h2, len(h_sdp)/2) 
        c5_2 = max_overlap(h2, 2.0/len(h_sdp))
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
    except:
        pass
    
    # Method 3: Gaussian rounding (stochastic)
    try:
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(len(h_sdp)), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, len(h_sdp)/2)
            c5_3 = max_overlap(h3, 2.0/len(h_sdp))
            if c5_3 < best_c5:
                best_h = h3
                best_c5 = c5_3
    except:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = None
    
    # Try 120-point discretization
    n = 120
    dx = 2.0 / n
    
    try:
        # Solve SDP
        h_sdp, X_sdp, tau_sdp = solve_sdp(n)
        if tau_sdp is not None and tau_sdp < best_c5:
            # Rounding and polishing
            h_rounded, c5_rounded = round_from_sdp(h_sdp, X_sdp)
            if c5_rounded < best_c5:
                remaining_time = budget_s - (time.time() - start_time)
                if remaining_time > 5:
                    polished_h, polished_c5 = polish(h_rounded, n, dx, remaining_time)
                    if polished_c5 < best_c5:
                        best_c5 = polished_c5
                        best_h = polished_h
                        best_n = n
                        
    except Exception as e:
        print(f"SDP failed: {e}")
    
    # If we didn't beat previous record, try a better starting strategy
    if best_h is None or best_c5 >= 0.38092:
        # Use triangular start
        h_triangular = np.zeros(n)
        for i in range(n):
            if i < n/2:
                h_triangular[i] = 2 * i / (n - 1)
            else:
                h_triangular[i] = 2 - 2 * i / (n - 1)
        h_triangular = np.clip(h_triangular, 0, 1)
        h_triangular = proj_sum_box(h_triangular, n/2)
        
        # Polish triangular
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            polished_h, polished_c5 = polish(h_triangular, n, dx, remaining_time)
            if polished_c5 < best_c5:
                best_c5 = polished_c5
                best_h = polished_h
                best_n = n
    
    # Try large discretization if time permits
    if budget_s - (time.time() - start_time) > 20:
        n = 140
        dx = 2.0 / n
        try:
            h_sdp, X_sdp, tau_sdp = solve_sdp(n)
            if tau_sdp is not None and tau_sdp < best_c5:
                h_rounded, c5_rounded = round_from_sdp(h_sdp, X_sdp)
                if c5_rounded < best_c5:
                    remaining_time = budget_s - (time.time() - start_time)
                    if remaining_time > 5:
                        polished_h, polished_c5 = polish(h_rounded, n, dx, remaining_time)
                        if polished_c5 < best_c5:
                            best_c5 = polished_c5
                            best_h = polished_h
                            best_n = n
        except: 
            pass
    
    # Fallback to a clean random start if no good solution found
    if best_h is None:
        h_random = np.random.rand(n)
        h_random = proj_sum_box(h_random, n/2)
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 5:
            polished_h, polished_c5 = polish(h_random, n, dx, remaining_time)
            best_c5 = polished_c5
            best_h = polished_h
            best_n = n
    
    return (best_h.tolist(), float(best_c5), best_n)
