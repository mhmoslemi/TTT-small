import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def spline_profile(z_values, n_points=120):
    """Generate a symmetric spline profile using the z_values as knot values"""
    # Ensure number of points is odd for symmetric structure
    m = 7  
    # Knot positions on [0, 1] - fixed, symmetric about 0.5
    x_knots = np.linspace(0, 1, (m+1)//2)
    
    # Map z values to [0,1] using sigmoid
    y_values = sigmoid(np.array(z_values))
    
    # Extend to full symmetric profile 
    y_full = np.concatenate([y_values, y_values[-2::-1]])
    
    # Create cubic spline interpolator
    spline = CubicSpline(x_knots, y_values, bc_type='clamped')
    
    # Generate the full profile
    x_full = np.linspace(0, 2, n_points)
    h_full = spline(x_full)
    
    # Scale to meet integral constraint
    integral = np.sum(h_full) * dx
    scale = TARGET_SUM / integral
    h_scaled = h_full * scale
    
    # Ensure it's feasible
    h_feasible = np.clip(h_scaled, 0, 1)
    
    # If clipping happened, recompute scaling
    if np.max(h_feasible) > 1.0:
        # Re-scale with clipping considered
        integral = np.sum(h_feasible) * dx
        scale = TARGET_SUM / integral
        h_feasible = h_feasible * scale
        h_feasible = np.clip(h_feasible, 0, 1)
    
    return h_feasible

def cma_es_optimize():
    """Simple CMA-ES implementation to minimize C5"""
    # Parameters
    d = 4  # dimensionality of z vector
    N = 120  # points
    global TARGET_SUM
    TARGET_SUM = N / 2
    
    # Initial settings
    m = np.zeros(d)
    sigma = 2.0
    C = np.eye(d)
    mu = d // 2
    lambda_ = 4 + int(3 * np.log(d))
    
    best_f = float('inf')
    best_z = m.copy()
    
    # Generate initial population
    population = []
    fitness = []
    
    for i in range(lambda_):
        z = m + sigma * np.random.multivariate_normal(np.zeros(d), C)
        h = spline_profile(z, N)
        f = evaluate_c5(h)
        
        if f < best_f:
            best_f = f
            best_z = z.copy()
            
        population.append(z)
        fitness.append(f)
        
    # Main loop
    maxiter = 300
    for iter_num in range(maxiter):
        # Sort by fitness
        idx = np.argsort(fitness)
        best_idx = idx[:mu]
        best_zs = [population[i] for i in best_idx]
        
        # Update mean
        z_mean = np.mean(best_zs, axis=0)
        m = z_mean
        
        # Update covariance matrix
        # Simple rank-μ update
        C = np.zeros((d, d))
        for i in range(mu):
            z_diff = best_zs[i] - z_mean
            C += np.outer(z_diff, z_diff) / (mu * sigma**2)
        
        # Update step size
        sigma *= np.exp(0.3 * (np.random.normal() / np.sqrt(d) - 0.5))
        
        # Generate new population
        population = []
        fitness = []
        
        for i in range(lambda_):
            z = m + sigma * np.random.multivariate_normal(np.zeros(d), C)
            h = spline_profile(z, N)
            f = evaluate_c5(h)
            
            if f < best_f:
                best_f = f
                best_z = z.copy()
                
            population.append(z)
            fitness.append(f)
            
        # Early stopping if no improvement in recent steps
        if iter_num > 200 and abs(np.diff(fitness[:10])).mean() < 1e-12:
            break
            
    return best_z, best_f

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    np.random.seed(seed)
    
    global TARGET_SUM, n_points, dx
    TARGET_SUM = n_points / 2
    
    # Initial profile (U-shape, already feasible)  
    h0 = np.array([1.0] * 20 + [0.0] * 80)  # Simple initial profile
    h0 = project_to_sum_box(h0, TARGET_SUM)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Run CMA-ES optimization on a low-dimensional parametric space
    best_z, best_c5_cma = cma_es_optimize()
    h_cma = spline_profile(best_z, n_points)
    h_cma = project_to_sum_box(h_cma, TARGET_SUM)
    c5_cma = evaluate_c5(h_cma)
    
    if c5_cma < best_c5:
        best_h = h_cma
        best_c5 = c5_cma

    # Polish with projected gradient descent
    def compute_gradient_at_shift(k, h):
        """Compute subgradient for a specific shift k"""
        # Use simplified version of gradient computation
        n = len(h)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1 - h[k:]) - h[:-k] if k > 0 else (1 - h) - h
        else:
            grad[:k] += (1 - h[:k]) - h[-k:] if k < 0 else (1 - h) - h
        return grad

    # Identify worst shifts
    corr = compute_corr_fft(best_h)
    max_val = np.max(corr)
    bad_shifts = np.where(corr >= max_val - 1e-12)[0]
    
    # Average sub-gradients for worst shifts
    g = np.zeros_like(best_h)
    for k in bad_shifts:
        # We'll approximate gradient using a simple symmetric shift manual
        grad_k = np.zeros_like(best_h)
        if k == 0:
            # For k = 0 we approximate gradient as -h (expecting that h*(1-h) is small)
            grad_k = -best_h
        else:
            # Simple 1-step approximations: shift+1 and shift-1
            grad_k[1:] += -best_h[:-1]  # first order derivative for shift
            grad_k[:-1] += -best_h[1:]
        g += grad_k
    
    eta = 0.08
    for _ in range(1200):
        if time.time() - start_time > budget_s - 5:
            break
        # Compute new h
        h_new = best_h - eta * g
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        else:
            eta *= 0.5
            if eta < 1e-5:
                break
            
        # Line search for better step size
        if abs(c5_new - best_c5) < 1e-10:
            # Try a tiny random swap to escape plateau (less than 1/100th of total points)
            i = np.random.randint(len(best_h))
            j = np.random.choice(list(set(range(len(best_h))) - {i}))
            delta = np.random.uniform(-0.015, 0.015)
            old_val = best_h[i] + best_h[j]
            new_h = best_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_h = project_to_sum_box(new_h, TARGET_SUM)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                
    return (best_h.tolist(), float(best_c5), n_points)
