def gradient_smooth_max(h, w):
    n = len(h)
    g = np.ones(n)  # derivative of sum_i h_i (constant) is 1
    # first conv w * h
    for s in range(-(n-1), n):
        wk = w[s + n - 1]
        if wk == 0.0:
            continue
        if s >= 0:
            g[:n-s] -= wk * h[s:]
        else:
            g[-s:] -= wk * h[:n+s]
    # second conv with w_rev (equiv to w[-s])
    for s in range(-(n-1), n):
        wk = w[-s + n - 1]
        if wk == 0.0:
            continue
        if s >= 0:
            g[:n-s] -= wk * h[s:]
        else:
            g[-s:] -= wk * h[:n+s]
    return g

def smooth_max_descent(h, target_sum, dx, time_budget):
    start = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    α = 4.0
    step = 0.2  # initial step size (in h space)
    while time.time() - start < time_budget:
        corr = correlation_fft(best_h)
        max_corr = np.max(corr)
        # compute softmax weights
        exp_vals = np.exp(α * (corr - max_corr))
        w = exp_vals / exp_vals.sum()
        # gradient
        g = gradient_smooth_max(best_h, w)
        # Try step sizes
        improved = False
        for η in (step, step/2, step/4):
            h_new = best_h - η * g  # descent direction
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum_box(h_new, target_sum)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
        # slowly increase α (sharpen max)
        if α < 200:
            α *= 1.05
    return best_h, best_c5

    target_ones = int(round(target_sum))

    idx_sorted = np.argsort(-h)

    start = time.time()
    attempts_without_improve = 0
    while time.time() - start < time_budget:

            attempts_without_improve = 0

def project_to_sum_box(x, target):
    # Clip values first
    x = np.clip(x, 0.0, 1.0)
    total = np.sum(x)
    if abs(total - target) < 1e-12:
        return x
    # binary search lam
    lam_low, lam_high = -1.0, 1.0
    def f(l):
        return np.sum(np.clip(x + l, 0.0, 1.0)) - target
    # Expand bounds until signs differ
    while f(lam_low) * f(lam_high) > 0:
        lam_low -= 1.0
        lam_high += 1.0
    for _ in range(50):
        lam_mid = (lam_low + lam_high) / 2.0
        fm = f(lam_mid)
        if abs(fm) < 1e-12:
            break
        if f(lam_low) * fm <= 0:
            lam_high = lam_mid
        else:
            lam_low = lam_mid
    lam = (lam_low + lam_high) / 2.0
    return np.clip(x + lam, 0.0, 1.0)
```

import numpy as np
import time
from scipy.fft import fft, ifft

# global used by evaluate_c5
DX = None  # will be set inside run()


def project_to_sum_box(x, target):
    """Clamp to [0,1] and adjust by a common shift so that the sum equals target."""
    x = np.clip(x, 0.0, 1.0)
    cur = np.sum(x)
    if abs(cur - target) < 1e-12:
        return x

    def f(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0)) - target

    lo, hi = -1.0, 1.0
    # enlarge interval until signs differ
    while f(lo) * f(hi) > 0:
        lo -= 1.0
        hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        fm = f(mid)
        if abs(fm) < 1e-12:
            break
        if f(lo) * fm <= 0:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    return np.clip(x + lam, 0.0, 1.0)


def correlation_fft(h):
    """Full correlation of h with (1-h) using FFT."""
    n = h.shape[0]
    L = 2 * n - 1
    h_rev = h[::-1]
    b = 1.0 - h
    H = fft(h_rev, L)
    B = fft(b, L)
    corr = ifft(H * B).real
    return corr[:L]  # length 2n-1


def evaluate_c5(h):
    """C5 value for a feasible h (global DX must be set)."""
    corr = correlation_fft(h)
    return DX * np.max(corr)


def gradient_softmax_max(h, w):
    """Gradient of the softmax‑max objective w.r.t. h."""
    n = h.shape[0]
    g = np.ones(n)          # derivative of Σ h_i is 1
    # first convolution with w
    for s in range(-(n - 1), n):
        wk = w[s + n - 1]
        if wk == 0.0:
            continue
        if s >= 0:
            g[: n - s] -= wk * h[s:]
        else:
            g[-s:] -= wk * h[: n + s]
    # second convolution with w_rev (w_{-s})
    for s in range(-(n - 1), n):
        wk = w[-s + n - 1]   # w_rev[s] = w[-s]
        if wk == 0.0:
            continue
        if s >= 0:
            g[: n - s] -= wk * h[s:]
        else:
            g[-s:] -= wk * h[: n + s]
    return g


def differential_evolution(n, target_sum, budget, init_vec):
    """Simple DE with projection onto the simplex‑box."""
    start = time.time()
    popsize = 30
    F = 0.5
    CR = 0.9

    # initialise population
    population = []
    fitness = []
    if init_vec is not None:
        population.append(init_vec.copy())
        fitness.append(evaluate_c5(init_vec))
    while len(population) < popsize:
        cand = np.random.rand(n)
        cand = project_to_sum_box(cand, target_sum)
        population.append(cand)
        fitness.append(evaluate_c5(cand))
    population = [np.asarray(p) for p in population]
    fitness = np.array(fitness)

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_f = fitness[best_idx]

    while time.time() - start < budget:
        for i in range(popsize):
            if time.time() - start >= budget:
                break
            # choose three distinct others
            idxs = np.random.choice([j for j in range(popsize) if j != i], 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = project_to_sum_box(mutant, target_sum)

            # crossover
            mask = np.random.rand(n) < CR
            if not np.any(mask):
                mask[np.random.randint(n)] = True
            trial = np.where(mask, mutant, population[i])
            trial = project_to_sum_box(trial, target_sum)

            val = evaluate_c5(trial)
            if val < fitness[i]:
                population[i] = trial
                fitness[i] = val
                if val < best_f:
                    best_f = val
                    best_h = trial.copy()
    return best_h, best_f


def smooth_max_descent(h0, target_sum, time_budget):
    """Projected gradient descent on a smoothed max (soft‑max) objective."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_f = evaluate_c5(h)

    alpha = 4.0
    step = 0.2

    while time.time() - start < time_budget:
        corr = correlation_fft(h)
        m = np.max(corr)
        # soft‑max weights, numerically stable
        exp_vals = np.exp(alpha * (corr - m))
        w = exp_vals / np.sum(exp_vals)

        g = gradient_softmax_max(h, w)

        improved = False
        for eta in (step, step * 0.5, step * 0.25):
            h_new = h - eta * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum_box(h_new, target_sum)
            f_new = evaluate_c5(h_new)
            if f_new < best_f:
                h = h_new
                best_h = h_new.copy()
                best_f = f_new
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break
        # slowly sharpen the soft‑max
        if alpha < 200:
            alpha *= 1.03
    return best_h, best_f


def binary_hill_climb(h_cont, target_sum, time_budget):
    """Round to a binary vector and improve it by random swaps."""
    n = h_cont.shape[0]
    target_ones = int(round(target_sum))
    # start by taking the largest entries as 1
    order = np.argsort(-h_cont)
    b = np.zeros(n)
    b[order[:target_ones]] = 1.0
    cur_f = evaluate_c5(b)
    start = time.time()
    stuck = 0
    while time.time() - start < time_budget:
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        b[i], b[j] = 0.0, 1.0
        new_f = evaluate_c5(b)
        if new_f < cur_f:
            cur_f = new_f
            stuck = 0
        else:
            b[i], b[j] = 1.0, 0.0
            stuck += 1
        if stuck > 2000:
            # re‑randomise a little to escape stagnation
            perm = np.random.permutation(n)[:target_ones]
            b[:] = 0.0
            b[perm] = 1.0
            cur_f = evaluate_c5(b)
            stuck = 0
    return b, cur_f
