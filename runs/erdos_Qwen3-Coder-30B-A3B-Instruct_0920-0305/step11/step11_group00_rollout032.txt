import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds for lambda
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        lam = (lam_low + lam_high) / 2
        fm = f(lam)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = lam
        else:
            lam_low = lam
    
    result = np.clip(x + lam, 0, 1)
    return result

def compute_full_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_gradient_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1 - h[:-k] if k < n else 1
        grad[:k] = 0
    else:
        grad[:k] = 1 - h[-k:] if -k < n else 1
        grad[k:] = 0
        
    if k <= 0:
        grad[:abs(k)] = 1 - h[abs(k):] if abs(k) < n else 1
        grad[abs(k):] = 0
    else:
        grad[abs(k):] = 1 - h[:-abs(k)] if abs(k) < n else 1
        grad[:abs(k)] = 0
        
    return grad

def compute_full_gradient(h):
    """Compute all gradients for shifts"""
    n = len(h)
    grads = []
    for k in range(-n+1, n):
        grad = compute_gradient_shift(h, k)
        grads.append(grad)
    return np.array(grads)

def lse_surrogate(h, alpha):
    """Compute log-sum-exp surrogate of maximum correlation"""
    corr = compute_full_correlation(h)
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    lse = (np.log(sum_exp) + max_corr * alpha) / alpha
    return lse

def compute_weights(h, alpha):
    """Compute weights for log-sum-exp surrogate"""
    corr = compute_full_correlation(h)
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    return weights

def accelerated_projected_gradient(h0, alpha, steps=150, eta=0.1, beta=0.9):
    """APG with continuation on alpha"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = np.max(compute_full_correlation(h))
    
    y = h.copy()
    h_prev = h.copy()
    
    for i in range(steps):
        corr = compute_full_correlation(h)
        weights = compute_weights(h, alpha)
        
        # Compute gradient of surrogate
        grad = np.zeros_like(h)
        for k in range(-len(h)+1, len(h)):
            if k >= 0:
                tmp = np.zeros_like(h)
                tmp[k:] = 1 - h[:-k] if k < len(h) else 1
                tmp[:k] = 0
            else:
                tmp = np.zeros_like(h)
                tmp[:k] = 1 - h[-k:] if -k < len(h) else 1
                tmp[k:] = 0
            grad += weights[k + len(h) - 1] * tmp
            
        # APG step
        y = h + beta * (h - h_prev)
        
        # Backtracking line search
        eta_trial = eta
        for _ in range(10):
            h_tmp = y - eta_trial * grad
            h_tmp = project_to_sum_box(h_tmp, len(h) / 2)
            c5_trial = np.max(compute_full_correlation(h_tmp))
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_tmp.copy()
            if lse_surrogate(h_tmp, alpha) < lse_surrogate(h, alpha) - 0.5 * eta_trial * np.linalg.norm(grad) ** 2:
                h = h_tmp
                break
            else:
                eta_trial *= 0.5
                
        h_prev = h.copy()
        
        if i % 30 == 0:
            corr = compute_full_correlation(h)
            max_corr = np.max(corr)
            if max_corr < best_c5:
                best_c5 = max_corr
                best_h = h.copy()
    
    return best_h, best_c5

def subgradient_polishing(h0, max_steps=1500, eta=0.08):
    """Exact max polishing with subgradients"""
    h = h0.copy()
    best_c5 = np.max(compute_full_correlation(h))
    
    for i in range(max_steps):
        corr = compute_full_correlation(h)
        max_corr = np.max(corr)
        k_max = np.where(np.abs(corr - max_corr) < 1e-12)[0]
        
        # Compute average gradient
        avg_grad = np.zeros_like(h)
        for k in k_max:
            grad = compute_gradient_shift(h, k - len(h) + 1)
            avg_grad += grad
        
        avg_grad /= len(k_max)
        
        # Update
        h_new = h - eta * avg_grad
        h_new = project_to_sum_box(h_new, len(h) / 2)
        
        new_c5 = np.max(compute_full_correlation(h_new))
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            if eta < 0.15:
                eta *= 1.05
        else:
            eta *= 0.5
            
        if i % 200 == 0:
            # Tiny random swap move to escape plateaus
            i1, i2 = np.random.choice(len(h), 2, replace=False)
            delta = np.random.uniform(-0.015, 0.015)
            h[i1] -= delta
            h[i2] += delta
            h = project_to_sum_box(h, len(h) / 2)
            new_c5 = np.max(compute_full_correlation(h))
            if new_c5 < best_c5:
                best_c5 = new_c5
                
    return h, best_c5
