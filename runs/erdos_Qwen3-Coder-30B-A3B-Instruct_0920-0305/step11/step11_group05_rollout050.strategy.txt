def lp_step(y, target_sum):
    n = len(y)
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    for s in range(-n+1, n):
        w = np.zeros(n)
        if s >= 0:
            w[:n-s] = y[s:]
        else:
            w[-s:] = y[:n+s]
        constraints.append(w @ x >= t)
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback to another solver
        prob.solve(solver=cp.SCS, max_iters=2000)
    x_opt = np.clip(x.value, 0, 1)
    t_opt = t.value

def local_refine(h0, target_sum, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    n = len(h)
    start = time.time()
    iter = 0
    while time.time() - start < time_limit:
        iter += 1
        # temperature schedule
        t = max(0.001, 0.02 * (1 - (time.time() - start) / time_limit))
        # Choose pair indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 2 * t  # in [-t,t]
        # adjust to stay within [0,1]
        hi = h[i]
        hj = h[j]
        # propose new hi' = clip(hi - delta, 0, 1), hj' = clip(hj + delta, 0, 1)
        # but we need to preserve sum exactly. We'll adjust delta to not exceed bounds.
        delta_i = -delta
        delta_j = delta
        # Clip:
        delta_i = max(-hi, min(1 - hi, delta_i))  # ensures hi+delta_i in [0,1]
        delta_j = max(-hj, min(1 - hj, delta_j))
        # To keep sum zero, we need delta_i + delta_j = 0. However clipping may break that.
        # Instead we can choose delta such that both stay in bounds. Simpler: move small amount
        max_delta = min(hi, 1 - hj)  # can move at most hi from i to j
        delta = np.random.uniform(-max_delta*0.5, max_delta*0.5)  # small random shift
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # No need to project as sum unchanged and bounds satisfied.
        new_c5 = evaluate_c5(new_h)
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            prob = np.exp(-(new_c5 - cur_c5) / (t + 1e-9))
            if np.random.rand() < prob:
                h = new_h
                cur_c5 = new_c5
    return best_h, best_c5
