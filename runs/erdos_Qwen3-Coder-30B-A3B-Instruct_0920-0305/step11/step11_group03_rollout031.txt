import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_autocorr(h):
    """Compute auto-correlation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    autocorr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return autocorr

def compute_corr(h):
    """Compute correlation of h with (1-h) using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_gradient(h, min_shifts):
    """Compute gradient for specified shifts"""
    grad = np.zeros_like(h)
    for k in min_shifts:
        if k >= 0:
            # left shift contribution: h[i+k] for i = 0 to N-1-k
            grad[:len(h)-k] += h[k:]
            # right shift contribution: h[i-k] for i = k to N-1  
            grad[k:] += h[:len(h)-k]
        else:
            # Negative shift: h[i+k] contribution where k < 0
            # This corresponds to right shift of |k|
            grad[-k:] += h[:-k]
            # h[i-k] contribution where k < 0
            grad[:-k] += h[-k:]
    return grad
