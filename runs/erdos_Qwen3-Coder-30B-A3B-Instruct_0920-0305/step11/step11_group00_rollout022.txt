import numpy as np

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Improve initial bounds
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (lam_low + lam_high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = m
        else:
            lam_low = m
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def grad_shift(h, k):
    """Compute gradient of correlation for shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def lse_surrogate(h, alpha):
    """Log-sum-exp surrogate for max correlation"""
    corr = compute_corr(h)
    max_c = np.max(corr)
    # Safe exponentiation with max subtraction
    exp_vals = np.exp(alpha * (corr - max_c))
    sum_exp = np.sum(exp_vals)
    if sum_exp == 0:
        sum_exp = 1e-16
    f_alpha = max_c + np.log(sum_exp) / alpha
    weights = exp_vals / sum_exp
    grad = np.zeros_like(h)
    for i, w in enumerate(weights):
        if w > 0:
            k = i - (n_points - 1)
            grad += w * grad_shift(h, k)
    return f_alpha, grad

def backtracking_line_search(h, grad, alpha, eta_init, f_prev):
    """Simple backtracking line search"""
    eta = eta_init
    h_tmp = h - eta * grad
    f_new = lse_surrogate(h_tmp, alpha)[0]
    for _ in range(5):
        if f_new <= f_prev:
            return eta
        eta /= 2
        h_tmp = h - eta * grad
        f_new = lse_surrogate(h_tmp, alpha)[0]
    return eta

def random_perturbation_escape(h, alpha, current_c5):
    """Perform random swap-move to escape local minima"""
    h_copy = h.copy()
    i, j = np.random.choice(n_points, 2, replace=False)
    delta = 0.02 * (np.random.random() - 0.5)
    h_copy[i] -= delta
    h_copy[j] += delta
    h_copy = project_to_sum_box(h_copy, TARGET_SUM)
    new_c5 = evaluate_c5(h_copy)
    if new_c5 < current_c5:
        return h_copy
    return h

                h_new = project_to_sum_box(h_tmp, TARGET_SUM)

                if iterations % 100 == 0 or no_improve_count > 200:

                    current_c5 = evaluate_c5(h)
                    no_improve_count = 0
            
            # Increase alpha
            alpha *= 1.4
            
            # Early stopping condition
            if prev_c5 - current_c5 < 1e-10:

                    current_c5 = new_c5

                    eta /= 2.0
                
                # Random perturbation every 200 iterations
                if sub_iter % 200 == 0:
                    h = random_perturbation_escape(h, alpha, current_c5)
                    current_c5 = evaluate_c5(h)
            
            # Final check
            current_c5 = evaluate_c5(h)

            best_c5 = current_c5

    if best_h is None:

        h = np.array(initial_h_values)
        h = project_to_sum_box(h, TARGET_SUM)
        best_c5 = evaluate_c5(h)
        best_h = h
    
    return (best_h.tolist(), float(best_c5), n_points)
