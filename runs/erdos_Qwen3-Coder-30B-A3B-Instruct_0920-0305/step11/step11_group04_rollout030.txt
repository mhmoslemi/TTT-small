import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorr(h):
    """Compute full correlation using FFT for efficiency"""
    h_rev = h[::-1]
    b = h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr * dx

def compute_C5(h):
    """Compute C5 = max of overlap integral"""
    rho = autocorr(h)
    return 1.0 - np.min(rho)

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n"""
    if n <= 0:
        return np.zeros(n)
    # Generate binary Thue-Morse sequence of length >= n using recurrence
    # Using the standard algorithm as described
    sequence = [0]
    while len(sequence) < n:
        # Extend by complementing: 0->1, 1->0
        complement = [1 - x for x in sequence]
        sequence.extend(complement)
    TM = np.array(sequence[:n], dtype=float)
    # Ensure exactly half ones
    ones_count = np.sum(TM)
    if ones_count < TARGET_SUM:
        # Set additional ones to 1
        zero_indices = np.where(TM == 0)[0]
        for i in range(TARGET_SUM - ones_count):
            TM[zero_indices[i]] = 1
    elif ones_count > TARGET_SUM:
        # Set some ones to 0
        one_indices = np.where(TM == 1)[0]
        for i in range(ones_count - TARGET_SUM):
            TM[one_indices[i]] = 0
    return TM

            if k_worst_shift > 0:
                if i + k_worst_shift < n_points:

                    if h[j] == 1:
                        # Gain = how many 1's will become 1 at this shift if we flip i to 1
                        gain = 1.0  # We turn h_i = 0 to h_i = 1, so the product becomes 1 at shift k
                        # Loss = how many 1's will be lost if we flip a 1 to 0
                        loss = 0
                        # Check for other 1s that will be lost
                        # Look at all positions around this shift to estimate loss
                        if k_worst_shift > 0:

                                loss += 1
                        net = gain - loss
                        if net > best_gain:
                            best_gain = net
                            best_i, best_j = i, j
            elif k_worst_shift < 0:

                    if h[j] == 1:
                        gain = 1.0
                        loss = 0

                                loss += 1

                                loss += 1
                        elif k_worst_shift < 0:

                                loss += 1
                            if j - k_worst_shift < n_points and h[j - k_worst_shift] == 1:
                                loss += 1
                        net = gain - loss
                        if net > best_gain:
                            best_gain = net
                            best_i, best_j = i, j
            else:
                # Should not happen unless k=0, we naturally ignore the zero-effect shift
                pass
                
        # Check if any good swap found
        if best_i != -1 and best_j != -1:
            # Check if other parts agree
            # Now perform the actual swap
            h_new = h.copy()
            h_new[best_i] = 1.0
            h_new[best_j] = 0.0
            # Re-evaluate the worst shift
            rho_new = autocorr(h_new)
            new_k_worst = np.argmin(rho_new)
            new_k_worst_shift = shifts[new_k_worst]
            if rho_new[new_k_worst] < rho[k_worst]:
                h = h_new
                k_worst = new_k_worst
                k_worst_shift = new_k_worst_shift
                rho = rho_new
            else:
                break  # No improvement
        else:
            break  # No beneficial swap found
            
    return h
