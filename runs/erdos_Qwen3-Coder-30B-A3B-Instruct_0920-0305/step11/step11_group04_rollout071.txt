import numpy as np
from scipy.signal import correlate
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort x in descending order to apply water-filling method
    x_sorted = np.sort(x)[::-1]
    cumsum = np.cumsum(x_sorted)
    
    # Find threshold index
    idx = np.searchsorted(cumsum, target_sum, side='right')
    if idx == 0:
        lambda_val = (target_sum - cumsum[0]) / (1 - x_sorted[0]) if x_sorted[0] < 1 else 0
    else:
        lambda_val = (target_sum - cumsum[idx-1]) / (idx + 1)
    
    # Compute the projection
    projected = np.clip(x - lambda_val, 0, 1)
    return projected

def compute_corr(h):
    """Compute full correlation via FFT for speed"""
    n = len(h)
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(h, L)).real
    return corr

def compute_C5(h, dx):
    """Compute C5 using the correlation of h with itself"""
    corr = compute_corr(h)
    return 1.0 - np.min(corr) * dx

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n (exactly n/2 ones)"""
    if n % 2 != 0:
        raise ValueError("n must be even for Thue-Morse to have equal 0s and 1s")
    t = [0]
    while len(t) < n:
        t.extend([1-x for x in t])
    return np.array(t[:n], dtype=float)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Improve h by swapping 0s and 1s that improve the worst shift correlation"""
    n = len(h)
    target_sum = n // 2
    
    # Compute initial overlap (1 - correlation) vector for all shifts
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * n - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    
    # Iterate until no improvement
    while time.time() - start_time < 0.95 * budget_s:
        # Find worst shift
        min_corr_idx = np.argmin(corr)
        worst_shift = min_corr_idx - (n - 1)
        
        # Compute potential gains from swaps
        best_swap = None
        best_net = -1e-10
        
        # Look for a 0 at position i and 1 at position j with beneficial swap
        for i in range(n):
            if h[i] == 0:
                # Potential gain from converting h[i] to 1
                gain = 0
                # Check if counterpart exists at shift +worst_shift
                j = i + worst_shift
                if 0 <= j < n and h[j] == 1:
                    gain += dx
                    
                # If shift is negative, check also the other direction
                if worst_shift < 0:
                    j = i - worst_shift
                    if 0 <= j < n and h[j] == 1:
                        gain += dx
                        
                # Now consider loss if we convert a 1 to 0
                for j in range(n):
                    if h[j] == 1:
                        loss = 0
                        # Check if pair would be lost
                        k = j + worst_shift
                        if 0 <= k < n and h[k] == 1:
                            loss += dx
                            
                        net = gain - loss
                        if net > best_net:
                            best_net = net
                            best_swap = (i, j)
                            
        # If no beneficial swap found, stop
        if best_net <= 0:
            break
            
        # Perform the swap
        i, j = best_swap
        h[i], h[j] = 1, 0
        
        # Update correlation in-place quickly
        # This is simplified, better to recompute both the arrays if needed
        # But for now we'll prioritize finding better swaps first
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Parameters
    n = 128
    dx = 2.0 / n
    target_sum = n // 2
    
    def autocorr(h):
        """Compute correlation efficiently"""
        _corr = compute_corr(h)
        return _corr * dx

    def compute_C5_for_h(h):
        """Compute C5 from h"""
        h_corr = autocorr(h)
        return 1.0 - np.min(h_corr)
    
    # Helper functions defined at top level
    def project_to_sum_box_fast(x, target_sum):
        return project_to_sum_box(x, target_sum)

        return compute_C5_for_h(h)

    seeds = []

    tm = thue_morse_binary(n)
    seeds.append(project_to_sum_box_fast(tm, target_sum))
    
    # 2. Random uniform with projection

    seeds.append(project_to_sum_box_fast(r, target_sum))

            h_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(initial_h_values)), initial_h_values)
            seeds.append(project_to_sum_box_fast(h_interp, target_sum))
    except:
        pass
    
    # Main gradient ascent loop
    best_h = None
    best_c5 = float('inf')
    
    for h0 in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h = h0.copy()
        tau = 0.2
        alpha = 0.15
        prev_best_c5 = float('inf')
        no_improve_count = 0
        
        for it in range(3000):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            rho = autocorr(h)
            c5 = 1.0 - np.min(rho)
            
            # Keep track of best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Early stopping on no improvement
            if c5 >= prev_best_c5 - 1e-7:
                no_improve_count += 1
            else:
                prev_best_c5 = c5
                no_improve_count = 0
                
            if no_improve_count > 200:
                break
                
            # Compute gradient of smooth min
            w = np.exp(-rho / tau)
            w /= w.sum()
            g = np.zeros_like(h)
            
            for shift in range(-(n-1), n):
                wk = w[shift + n - 1]
                if np.isclose(wk, 0.0):
                    continue
                lo = max(0, -shift)
                hi = min(n, n - shift)
                if lo < hi:
                    g[lo:hi] += wk * h[lo+shift:hi+shift]
                    g[lo+shift:hi+shift] += wk * h[lo:hi]
            g *= dx
            
            # Ascent step + projection
            h_new = project_to_sum_box_fast(h + alpha * g, target_sum)
            
            # Acceptance
            c5_new = 1.0 - np.min(autocorr(h_new))
            if c5_new < c5:
                h = h_new
                alpha = min(alpha * 1.02, 0.5)
            else:
                alpha *= 0.5
                
            tau *= 0.99
    
    # Final polishing with binary swap
    if best_h is not None:
        # Binarise
        idx = np.argsort(-best_h)[:target_sum]
        h_bin = np.zeros(n)
        h_bin[idx] = 1.0
        # Use local search
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        final_c5 = compute_C5_for_h(h_bin)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_bin.copy()
    
    return (best_h.tolist(), float(best_c5), n)
