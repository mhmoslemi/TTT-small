import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    h2 = np.zeros(n_points)

    h3[1::2] = 0.2
    h3 = project_to_sum_box(h3, TARGET_SUM)
    initial_configs.append(h3.copy())
    
    # Try all initial configurations
    for h_init in initial_configs:
        current_h = h_init.copy()
        current_c5 = evaluate_c5(h_init)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Main optimization loop with multiple strategies
    max_iter = 2000
    iter_count = 0
    strategies = ['gradient', 'simulated_annealing', 'local_search', 'Nelder-Mead']
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        strategy = strategies[iter_count % len(strategies)]
        
        if strategy == 'gradient':
            # Use trust-constr method which is more robust
            bounds = [(0, 1) for _ in range(n_points)]

                return evaluate_c5(h)

                return np.sum(h) - TARGET_SUM

                            current_c5 = test_c5

                                best_h = current_h.copy()

                            break

                    break
                    
        else:  # Nelder-Mead
            # Use a simpler direct search method
            try:

                h_perturbed = current_h + np.random.normal(0, 0.01, n_points)
