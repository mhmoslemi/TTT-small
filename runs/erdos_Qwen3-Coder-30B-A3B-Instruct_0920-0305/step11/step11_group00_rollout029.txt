def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def spline_profile(z, n_points=120):
    """Generate a symmetric spline profile from parameters z"""
    # Knot positions on [0,1]
    m = 7  # number of knots (must be odd for symmetry)
    x_knots = np.linspace(0, 1, (m+1)//2)
    
    # Map z to y using sigmoid
    y = sigmoid(z)
    
    # Extend to full interval [0,2] with symmetry
    x_full = np.concatenate([x_knots, 2 - x_knots[-2::-1]])
    y_full = np.concatenate([y, y[-2::-1]])
    
    # Create cubic spline
    cs = CubicSpline(x_full, y_full, bc_type='clamped')
    
    # Evaluate on uniform grid
    x_uniform = np.linspace(0, 2, n_points)
    h = cs(x_uniform)
    
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    
    return h

        offspring = []
