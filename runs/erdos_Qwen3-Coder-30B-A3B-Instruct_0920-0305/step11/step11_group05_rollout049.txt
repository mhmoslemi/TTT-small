def fft_correlation(h):
    """Compute full correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_C5(h):
    """Compute C5 using FFT correlation"""
    corr = fft_correlation(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""

    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

    weights = exp_vals / Z

    h_fft = np.fft.fft(h)
    ones_fft = np.fft.fft(np.ones_like(h))

    gradient = dx * (A_fft - B_fft).real
    return gradient

        t = t_new

        if k > 100 and np.abs(compute_C5(h) - compute_C5(h0)) < 1e-6:
            break
