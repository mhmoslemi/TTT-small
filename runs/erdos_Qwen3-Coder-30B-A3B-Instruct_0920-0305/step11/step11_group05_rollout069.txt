def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Sort values in descending order to find threshold
    sorted_x = np.sort(x)[::-1]
    
    # Find the threshold lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Find largest j such that sorted_x[j] - (target_sum - cumsum[j-1]) / (n-j) >= 0
    # This is equivalent to: sorted_x[j] >= (target_sum - cumsum[j-1]) / (n-j)
    # or: (n-j) * sorted_x[j] >= target_sum - cumsum[j-1]
    # with cumsum[0] = sorted_x[0]
    
    # Compute the threshold using a more direct approach
    # We solve the equation for lambda in terms of the sorted array
    threshold_idx = 0
    for i in range(n):
        # Check if lambda=key[i] works
        if i == 0:
            prefix_sum = 0
        else:
            prefix_sum = cumsum[i-1]
        if sorted_x[i] >= (target_sum - prefix_sum) / (n - i):
            threshold_idx = i
        else:
            break
    
    # The threshold is determined by the first element in index threshold_idx
    lambda_val = (target_sum - cumsum[threshold_idx-1]) / (n - threshold_idx) if threshold_idx > 0 else 0
    
    # Project: truncate at 0 and 1, and then adjust with lambda
    result = np.clip(x - lambda_val, 0, 1)
    
    # Normalization: should already hold due to the definition, but let's double-check
    if abs(np.sum(result) - target_sum) > 1e-10:
        # If not exactly met, correct using the remaining difference
        diff = target_sum - np.sum(result)
        # Reinsert the difference in a way that stays within bounds
        indices = np.argsort(result)
        for i in range(len(indices)):
            idx = indices[i]
            if diff > 0:
                add = min(diff, 1 - result[idx])
                result[idx] += add
                diff -= add
            else:
                break
    
    return result

def compute_correlation_fft(h):
    """Use FFT to compute all correlations efficiently"""

    b = 1.0 - h

def accelerated_projected_gradient(h_init, niter=3000):
    """Run APG with backtracking and projection"""
    h = h_init.copy()
    h_projected = project_to_box_sum(h, n / 2)
    
    y = h_projected.copy()
    t = 1.0
    
    for k in range(niter):
        # Compute surrogate objective and gradient  
        corr = compute_correlation_fft(h_projected)
        # Just estimate for final value
        g_s_values = []
        for i in range(-n+1, n): 
            g_s_values.append(2.0/n * (n/2 - sum(h_projected[j] * h_projected[(j+i)%n] for j in range(n))))
        g_s_values = np.array(g_s_values)
        
        g_s_max = np.max(g_s_values) 
        C5 = g_s_max
        # Calculate now gradient
        # USE FFT based gradient:
        grad = compute_gradient_via_fft(h_projected, np.exp(g_s_values/1e-3) / np.sum(np.exp(g_s_values/1e-3)))  
        # Backtracking line search
        alpha = 1.0
        eta = 1e-4
        while True: 
            h_test = project_to_box_sum(h_projected - alpha * grad, n / 2)
            corr_test = compute_correlation_fft(h_test)
            g_s_test_values = []
            for i in range(-n+1, n): 
                g_s_test_values.append(2.0/n * (n/2 - sum(h_test[j] * h_test[(j+i)%n] for j in range(n))))
            g_s_test_values = np.array(g_s_test_values)
            g_s_test_max = np.max(g_s_test_values)
            
            if g_s_test_max < C5 or alpha < 1e-10:
                break
            alpha *= 0.5
        
        # Update step
        h_next = project_to_box_sum(h_projected - alpha * grad, n / 2)
        t_new = (1 + np.sqrt(1 + 4 * t**2)) / 2
        y_next = h_next + ((t - 1) / t_new) * (h_next - h_projected)
        
        h_projected = h_next.copy()
        y = y_next.copy()
        t = t_new
        
        if k % 100 == 0:
            corr = compute_correlation_fft(h_projected)
            c5_current = np.max(corr) * 2.0 / n
            if c5_current < 0.3809:
                break
    
    return h_projected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants  
    n = 120
    dx = 2.0 / n
    target_sum = n / 2
    
    # Initializations
    h_start = np.array(initial_h_values)
    h_start = project_to_box_sum(h_start, target_sum)
    
    # Generate multiple restarts 
    restarts = [h_start.copy()]
    
    # Tent shaped
    tent = np.zeros(n)
    mid = n // 2
    for i in range(mid):
        tent[i] = 2.0 * i / (mid - 1)
        tent[n - 1 - i] = 2.0 * i / (mid - 1)
    tent = project_to_box_sum(tent, target_sum)
    restarts.append(tent.copy())
    
    # Random restarts  
    for _ in range(3):
        h_random = np.random.rand(n)
        h_random = project_to_box_sum(h_random, target_sum)
        restarts.append(h_random.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    for r_idx, h_seed in enumerate(restarts):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        h_opt = accelerated_projected_gradient(h_seed, niter=3000)
        corr_opt = compute_correlation_fft(h_opt)
        c5_opt = np.max(corr_opt) * 2.0 / n
        
        if best_c5 > c5_opt:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final improvement with SA-based local search
    if best_h is not None:
        # Perform simulated annealing style local search
        current_h = best_h.copy()
        current_c5 = best_c5  
        
        # Warmup with speedup searching  
        T = 10.0
        T_min = 1e-5
        decay_rate = 0.9995
        it = 0
        
        while T > T_min and (time.time() - start_time) < (budget_s * 0.99):
            i, j = np.random.randint(0, n), np.random.randint(0, n)
            if i == j:
                continue
                
            delta_max = min(current_h[i], 1 - current_h[j])
            if delta_max <= 1e-10:
                continue
                
            delta = np.random.uniform(-delta_max, delta_max)
            h_new = current_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            h_new = project_to_box_sum(h_new, n / 2)
            
            corr_new = compute_correlation_fft(h_new)
            c5_new = np.max(corr_new) * 2.0 / n
            
            if c5_new < current_c5:
                current_h = h_new.copy()
                current_c5 = c5_new
            else:
                if np.random.rand() < np.exp(-(c5_new - current_c5) / T):
                    current_h = h_new.copy()
                    current_c5 = c5_new
            
            T *= decay_rate
            it += 1
            
            if it % 100 == 0:
                corr = compute_correlation_fft(current_h)
                c5 = np.max(corr) * 2.0 / n
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current_h.copy()
    
    # Final re-check and true evaluation
    corr = compute_correlation_fft(best_h)
    final_c5 = np.max(corr) * 2.0 / n
    
    return (best_h.tolist(), float(final_c5), n)
