import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def autocorr(h):
    """Compute full linear autocorrelation of h using FFT."""
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h, L) * np.fft.fft(h, L).conj()).real
    return corr * dx

def compute_C5(h):
    """Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx."""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project x onto box constraints [0,1] with sum = target_sum."""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n."""
    if n <= 0:
        return np.array([])
    if n == 1:
        return np.array([0])
    
    # Generate TM sequence using bit counting
    tm = []
    for i in range(n):
        # Number of 1s in binary representation of i
        ones = bin(i).count('1')
        tm.append(ones % 2)
    
    h = np.array(tm)
    # Ensure exactly half ones
    current_sum = int(np.sum(h))
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h == 1)[0]
        flip_indices = ones_idx[:current_sum - target_sum]
        h[flip_indices] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        zeros_idx = np.where(h == 0)[0]
        flip_indices = zeros_idx[:target_sum - current_sum]
        h[flip_indices] = 1
    
    return h.astype(float)

    corr = autocorr(h)
    # Identify worst shift

    improvement_found = True
    while improvement_found and (time.time() - start < budget_s * 0.9):
        improvement_found = False
        best_net = 0
        best_pair = None
        # Look at all pairs that are 0 and 1
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        
        for i_zero in zeros:
            gain = 0
            loss = 0
            for shift in range(-n_points + 1, n_points):
                if shift == 0:
                    continue
                shifted_idx = i_zero + shift
                if 0 <= shifted_idx < n_points:
                    if h[shifted_idx] == 1:
                        gain += 1.0  # Gain of 1 for dx = 1 for this shift
                # Try to compute loss when we move the 1
            
            for j_one in ones:
                shifted_idx = j_one + (worst_shift)
                if 0 <= shifted_idx < n_points:
                    if h[shifted_idx] == 1:

                if shift == 0:
                    continue
                shifted_idx = j_one - (worst_shift)
                if 0 <= shifted_idx < n_points:
                    if h[shifted_idx] == 1:
                        loss += 1.0

            if net > best_net:
                best_net = net

            h[j_one] = 0
            improvement_found = True
            corr = autocorr(h)  # Recompute
            worst_shift = np.argmin(corr)
    
    return h

def run(seed=42, budget_s=600, **kwargs):

    target_sum = n // 2
    n_points = n

        corr = autocorr(h)
        return 1.0 - np.min(corr)

        pass

    # ---- main optimisation loop -----------------------------------------
    for h0 in seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
        h = h0.copy()
        τ = 0.2
        α = 0.15
        for it in range(3000):
            if time.time() - start_time > budget_s * 0.95:
                break
            ρ = autocorr(h)
            C5 = 1.0 - np.min(ρ)
            if C5 < best_val:
                best_val, best_h = C5, h.copy()

            # gradient of smooth min
            w = np.exp(-ρ/τ)
            w /= w.sum()
            g = np.zeros_like(h)
            for shift in range(-(n-1), n):
                wk = w[shift + n - 1]
                if wk == 0.0: continue
                lo = max(0, -shift)
                hi = min(n, n-shift)
                g[lo:hi]               += wk * h[lo+shift:hi+shift]
                g[lo+shift:hi+shift]   += wk * h[lo:hi]
            g *= dx

            # ascent step + projection
            h_candidate = project_to_sum_box(h + α * g, target_sum)

            # acceptance test
            C5_new = 1.0 - np.min(autocorr(h_candidate))
            if C5_new < C5:
                h = h_candidate
                α = min(α*1.02, 0.5)
            else:
                α *= 0.5

            τ *= 0.99
            # optional early stop if no improvement for many iters …
            if it > 200 and abs(C5 - C5_new) < 1e-7:
                break

    # ---- binary polishing ------------------------------------------------
    if best_h is not None:
        # binarise by keeping the largest target_sum entries as 1
        idx = np.argsort(-best_h)[:target_sum]
        h_bin = np.zeros(n)
        h_bin[idx] = 1.0
        # greedy swap local search focussed on the worst shift
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        C5_bin = compute_C5(h_bin)
        if C5_bin < best_val:
            best_val, best_h = C5_bin, h_bin.copy()

    # --------------------------------------------------------------------
    return (best_h.tolist(), float(best_val), n)
