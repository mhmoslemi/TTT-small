import numpy as np

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def compute_corr(h):
    """Compute full correlation using FFT for speed"""
    # Use correlate with full mode which matches the integral definition
    corr = correlate(h, h, mode='full')
    return corr * dx

def compute_C5(h):
    """Compute C5 value for given step function h"""
    corr = compute_corr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort values in descending order
    sorted_x = np.sort(x)[::-1]
    
    # Binary search for the threshold lambda
    low, high = -1.0, 1.0
    tol = 1e-12
    
    while high - low > tol:
        mid = (low + high) / 2
        projected = np.clip(x + mid, 0, 1)
        sum_proj = np.sum(projected)
        if sum_proj > target_sum:
            low = mid
        else:
            high = mid
            
    lambda_val = (low + high) / 2
    result = np.clip(x + lambda_val, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse binary pattern of length n"""
    # Generate TM sequence of length 2^k >= n, then truncate
    k = int(np.ceil(np.log2(n)))
    length = 2**k
    tm = np.zeros(length, dtype=int)
    tm[1] = 1
    
    for i in range(1, k):
        # Double the sequence: append complement
        start = 2**i
        for j in range(2**i):
            tm[start + j] = 1 - tm[j]
    
    tm = tm[:n]
    
    # Adjust to ensure exactly half ones
    ones_count = np.sum(tm)
    if ones_count > n // 2:
        # Find indices where we can flip 1 to 0
        ones_idx = np.where(tm == 1)[0]
        flip_indices = ones_idx[:ones_count - n//2]
        tm[flip_indices] = 0
    elif ones_count < n // 2:
        # Find indices where we can flip 0 to 1
        zeros_idx = np.where(tm == 0)[0]
        flip_indices = zeros_idx[:n//2 - ones_count]
        tm[flip_indices] = 1
    
    return tm.astype(float)

def greedy_swap_local_search(h, dx, budget_s, start_time):

    while iter_count < max_iter and time.time() - start_time < budget_s * 0.9:
        # Get indices of zeros and ones
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_gain = -1e10
        best_pair = None
        
        # Look for swaps that increase the correlation at worst shift
        shift = worst_shift - (n_points - 1)  # convert index to actual shift
        
        for i_zero in zeros:
            # Check which ones in the neighborhood contribute to this shift
            gain = 0.0
            
            # Test if swapping i_zero (0) to 1 and some j_one (1) to 0 helps
            for j_one in ones:
                # Estimate net gain from swap
                loss = 0.0
                gain = 0.0
                
                # Check contribution to shift: h[i_zero + shift] and h[i_zero - shift]  
                # Only add if in valid range
                if shift != 0:
                    pos1 = i_zero + shift
                    pos2 = i_zero - shift
                    if 0 <= pos1 < n_points and h[pos1] == 1:
                        gain += 1.0
                        
                    if 0 <= pos2 < n_points and h[pos2] == 1:
                        gain += 1.0
                
                # Loss from j_one being set to 0
                pos3 = j_one + shift
                pos4 = j_one - shift
                if 0 <= pos3 < n_points and h[pos3] == 1:
                    loss += 1.0
                    
                if 0 <= pos4 < n_points and h[pos4] == 1:
                    loss += 1.0
                    
                net = gain - loss
                if net > best_gain:
                    best_gain = net
                    best_pair = (i_zero, j_one)
                    
        if best_pair is None or best_gain <= 0:
            break
            
        # Perform the swap with positive net gain
        i_zero, j_one = best_pair
        h[i_zero] = 1.0
        h[j_one] = 0.0
        
        # Recompute correlation efficiently (only affected terms)
        corr = compute_corr(h)
        worst_shift = np.argmin(corr) 
        min_corr = corr[worst_shift]
        
        iter_count += 1
        
    return h
