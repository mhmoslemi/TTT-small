import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 180
dx = 2.0 / n_points
M = 12  # number of cosine modes

# Precompute cosine basis matrix
phi = np.ones((n_points, M))
for i in range(1, M):
    phi[:, i] = np.cos(np.pi * i * np.arange(n_points) / (n_points - 1))

def compute_autocorrelation(h):
    """Compute full autocorrelation using FFT"""
    h_padded = np.concatenate([h, np.zeros(n_points - 1)])
    H = np.fft.fft(h_padded)
    R_full = np.fft.ifft(H * np.conj(H)).real
    return R_full[:n_points] * dx

def softmin_weights(R, beta):
    """Compute soft-min weights for temperature beta"""
    exp_R = np.exp(-beta * R)
    return exp_R / np.sum(exp_R)

def gradient_via_fft(h, weights):
    """Compute gradient using FFT convolution trick"""
    w_rev = weights[::-1]
    # Forward shift part: h[i+k] terms
    forward_shift = np.convolve(weights, h, mode='full')[n_points-1:2*n_points-1]
    # Backward shift part: h[i-k] terms  
    backward_shift = np.convolve(w_rev, h, mode='full')[0:n_points]
    return -(forward_shift + backward_shift)

def project_onto_feasible_set(h):
    """Project h onto box constraints [0,1] and sum = n_points/2"""

        else:
            a = m
    return np.clip(h_clipped + (a + b)/2, 0, 1)

def evaluate_c5_exact(h):
    """Compute exact C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

    betas = [1.0]

        betas.append(betas[-1] * 2)
    return betas
