import numpy as np

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def lp_step(y, target_sum):
    """Solve LP to maximize lower bound t of minimal product sum"""
    n = len(y)
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Build constraints for all shifts
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    
    # For each shift s, compute weight vector w_s and add constraint
    for s in range(-n+1, n):
        w = np.zeros(n)
        if s >= 0:
            w[:n-s] = y[s:]
        else:
            w[-s:] = y[:n+s]
        constraints.append(w @ x >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            x_opt = np.clip(x.value, 0, 1)
            t_opt = t.value
        else:
            # Fallback to SCS solver
            prob.solve(solver=cp.SCS, max_iters=2000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                x_opt = np.clip(x.value, 0, 1)
                t_opt = t.value
            else:
                # Use current y as fallback
                x_opt = y.copy()
                t_opt = 0
    except:
        # In case of optimization issues, fallback to current point
        x_opt = y.copy()
        t_opt = 0
    
    return x_opt, t_opt

def alternating_lp(y0, target_sum, max_outer_iter=20, tol=1e-8, time_limit=None):
    """Perform alternating LP optimization for better C5 bound"""
    y = y0.copy()
    best_h = y.copy()
    best_c5 = evaluate_c5(y)
    
    start = time.time()
    for outer in range(max_outer_iter):
        if time_limit is not None and time.time() - start > time_limit:
            break
            
        x, t_opt = lp_step(y, target_sum)
        
        # Evaluate C5
        c5 = evaluate_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
            
        # Check convergence
        if np.linalg.norm(x - y) < tol:
            break
            
        y = x
        
    return best_h, best_c5

def local_refine(h0, target_sum, time_limit):

    h = h0.copy()
    best_h = h.copy()

    cur_c5 = best_c5
    n = len(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        # Temperature schedule
        t = max(0.001, 0.02 * (1 - (time.time() - start) / time_limit))
        
        # Choose pair indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
            
        # Random move
        delta = (np.random.rand() - 0.5) * 2 * t
        
        # Adjust to keep values in [0,1] and preserve sum
        hi = h[i]
        hj = h[j]
        delta_i = -delta
        delta_j = delta
        
        # Clip to respect bounds and maintain sum
        max_delta_i = min(hi, 1 - hj)  # can move at most hi from i to j  
        max_delta_j = min(hj, 1 - hi)
        actual_delta = np.random.uniform(-max_delta_i*0.5, max_delta_i*0.5)
        
        new_h = h.copy()
        new_h[i] -= actual_delta
        new_h[j] += actual_delta
        
        # No need to project as sum preserved and bounds satisfied
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < cur_c5:
            h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
        else:
            prob = np.exp(-(new_c5 - cur_c5) / (t + 1e-9))
            if np.random.rand() < prob:
                h = new_h
                cur_c5 = new_c5
                
    return best_h, best_c5
