def evaluate_c5_exact(h, dx):
    """Compute exact C5 using full correlation"""
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def spectral_round_autocorr(r, n, dx):

    S = np.fft.fft(r)

    A = np.sqrt(np.maximum(S, 0))
    
    # Try multiple phase choices

    best_c5 = float('inf')

    phases = [np.zeros(n), np.pi * np.random.rand(n), 
              np.pi * np.random.rand(n), np.pi * np.random.rand(n)]
    
    for phase in phases:
        # Construct signal in frequency domain
        h_freq = A * np.exp(1j * phase)
        # Inverse FFT
        h = np.real(np.fft.ifft(h_freq))
        # Ensure it's real and handle numerical errors
        h = np.real(h)
        # Project onto feasible set
        h_proj = project_to_box_sum(h, n/2)

        c5_val = evaluate_c5_exact(h_proj, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_proj
    
    return best_h, best_c5
