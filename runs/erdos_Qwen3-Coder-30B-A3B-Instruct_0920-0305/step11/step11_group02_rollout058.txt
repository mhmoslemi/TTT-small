import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    n = len(h)
    L = 2 * n - 1
    h_rev = h[::-1]
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient from soft-max weights using FFT"""
    n = len(h)
    L = 2 * n - 1
    
    # Reverse weights for convolution
    w_rev = w[::-1]
    
    # Convolve with h to get sums
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract central part
    grad = -(conv_plus[n-1:] + conv_rev[n-1:])
    return grad

def compute_c5(h):
    """Compute C5 value"""
    n = len(h)
    S = n / 2
    dx = 2.0 / n
    O = compute_autocorrelation(h)
    g = S - O
    return dx * np.max(g)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Discretization parameters
    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    best_c5 = float('inf')
    best_h = None
    
    # Two initialization strategies
    init_solutions = []
    
    # Strategy 1: Upsample initial solution
    h0 = np.array(initial_h_values)
    h0_upsampled = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h0)), h0)
    h0_upsampled = project_to_sum_box(h0_upsampled, target_sum)
    init_solutions.append(h0_upsampled)
    
    # Strategy 2: Random binary
    h1 = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    h1[ones_indices] = 1.0
    h1 = project_to_sum_box(h1, target_sum)
    init_solutions.append(h1)
    
    # Run optimization for each initialization
    for h_start in init_solutions:
        # Smooth max optimization parameters
        alpha = 1.0
        alpha_max = 100.0
        eta = 0.05
        eta_decay = 0.9
        eta_min = 1e-6
        
        h_current = h_start.copy()
        best_h_local = h_start.copy()
        best_c5_local = compute_c5(h_start)
        
        iteration = 0
        last_improvement = 0
        
        while time.time() - start_time < budget_s * 0.95 and iteration < 20000:
            iteration += 1
            
            # Compute autocorrelation and g values
            O = compute_autocorrelation(h_current)
            S = target_sum
            g = S - O  # length 2*n-1
            
            # Compute smooth max
            max_g = np.max(g)
            shifted_g = g - max_g
            exp_g = np.exp(alpha * shifted_g)
            w_raw = exp_g
            w = w_raw / np.sum(w_raw)
            
            # Compute gradient
            grad = gradient_from_weights(h_current, w)
            
            # Gradient step
            y = h_current - eta * grad
            h_current = project_to_sum_box(y, target_sum)
            
            # Evaluate true objective
            current_c5 = compute_c5(h_current)
            
            if current_c5 < best_c5_local:
                best_c5_local = current_c5
                best_h_local = h_current.copy()
                last_improvement = iteration
            
            # Update parameters
            if iteration % 200 == 0:
                alpha = min(alpha * 1.3, alpha_max)
                eta = max(eta * eta_decay, eta_min)
            
            # Random perturbation to escape local minima
            if iteration % 500 == 0 and iteration - last_improvement > 200:
                noise = np.random.normal(0, 0.02, n_points)
                y = h_current + noise
                h_current = project_to_sum_box(np.clip(y, 0, 1), target_sum)
                current_c5 = compute_c5(h_current)
                if current_c5 < best_c5_local:
                    best_c5_local = current_c5
                    best_h_local = h_current.copy()
                    last_improvement = iteration
        
        # Update global best
        if best_c5_local < best_c5:
            best_c5 = best_c5_local
            best_h = best_h_local.copy()
    
    # Final polishing with SLSQP
    if best_h is not None and time.time() - start_time < budget_s * 0.99:
        def obj_full(x):
            O = compute_autocorrelation(x)
            S = target_sum
            return dx * np.max(S - O)  # Exact C5 value

        constraints = ({'type':'eq', 'fun': lambda x: np.sum(x) - target_sum},)
        bounds = [(0.0, 1.0)] * n_points
        
        res = minimize(obj_full,
                       x0=best_h,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-12})
        
        if res.success and obj_full(res.x) < best_c5:
            best_h = res.x
            best_c5 = obj_full(res.x)
    
    if best_h is None:
        # Fallback to initial solution
        h0 = np.array(initial_h_values)
        h0_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h0)), h0)
        h0_upsampled = project_to_sum_box(h0_upsampled, target_sum)
        best_h = h0_upsampled
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
