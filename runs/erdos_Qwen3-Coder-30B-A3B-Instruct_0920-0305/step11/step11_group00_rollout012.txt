import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 200
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Initial bounds based on analytical estimates
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (lam_low + lam_high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = m
        else:
            lam_low = m
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def compute_grad_shift(h, k):
    """Compute gradient of c_k(h) for shift k"""
    n = len(h)
    pad = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    base = np.arange(n) + (n-1)
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def compute_lse_weights(c, alpha):
    """Compute softmax weights for log-sum-exp surrogate"""
    c_alpha = alpha * c
    c_max = np.max(c_alpha)
    exp terms = np.exp(c_alpha - c_max)
    weights = exp_terms / np.sum(exp_terms)
    return weights

def compute_surrogate_grad(h, c, alpha):
    """Compute gradient of log-sum-exp surrogate"""
    weights = compute_lse_weights(c, alpha)
    grad = np.zeros_like(h)
    n = len(h)
    
    # For each shift k, compute gradient of c_k and accumulate weighted
    for k in range(-n+1, n):
        if k == 0:
            continue  # skip the zeroshift for lse gradient (redundant)
        grad_k = compute_grad_shift(h, k)
        grad += weights[k + n - 1] * grad_k
    
    return grad

def backtracking_line_search(h, grad, alpha, F_alpha_prev, eta, max_iter=5):
    """Simple backtracking line search"""
    c = compute_corr(h)
    F_alpha_prev = np.sum(np.log(np.exp(alpha * c) + 1e-15)) / alpha
    
    for i in range(max_iter):
        h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
        c_new = compute_corr(h_new)
        F_alpha_new = np.sum(np.log(np.exp(alpha * c_new) + 1e-15)) / alpha
        
        if F_alpha_new <= F_alpha_prev - 0.5 * eta * np.dot(grad, grad):
            return h_new, eta
        else:
            eta *= 0.5
    
    return h, eta

def accelerated_projected_gradient(h, alpha, max_iter=150, eta_init=0.05, beta=0.9):
    """APG with backtracking line search"""
    h_prev = h.copy()
    y = h.copy()
    eta = eta_init
    
    for i in range(max_iter):
        # Nesterov momentum step
        y = h + beta * (h - h_prev)
        
        # Compute correlation and gradient
        c = compute_corr(y)
        grad = compute_surrogate_grad(y, c, alpha)
        
        # Line search
        h_new, eta = backtracking_line_search(y, grad, alpha, np.sum(np.log(np.exp(alpha * c) + 1e-15)) / alpha, eta)
        
        # Update for next iteration
        h_prev = h.copy()
        h = h_new
        
        # Random escape every 100 iterations or if no improvement
        if i % 100 == 0:
            # Random perturbation
            indices = np.random.choice(n_points, 2, replace=False)
            delta = 0.02
            if np.random.rand() > 0.5:
                h[indices[0]] -= delta
                h[indices[1]] += delta
            else:
                h[indices[0]] += delta
                h[indices[1]] -= delta
            h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def subgradient_polishing(h, max_iter=2000, eta_init=0.08):
    """Polish with exact subgradients"""
    eta = eta_init
    
    for i in range(max_iter):
        c = compute_corr(h)
        c_max = np.max(c)
        
        # Find indices where c_k ≈ max(c)
        K_max = np.where(np.abs(c - c_max) < 1e-9)[0]
        
        if len(K_max) == 0:
            K_max = np.argmax(c)
            K_max = [K_max]
        
        # Compute average gradient
        grad_avg = np.zeros_like(h)
        for k in K_max:
            grad_k = compute_grad_shift(h, k - n_points + 1)
            grad_avg += grad_k
        grad_avg /= len(K_max)
        
        # Update
        h_new = project_to_sum_box(h - eta * grad_avg, TARGET_SUM)
        
        # Accept if better
        c_new = compute_corr(h_new)
        if np.max(c_new) < np.max(c) - 1e-12:
            h = h_new
            # Increase step size if beneficial
            if eta < 0.15:
                eta *= 1.05
        else:
            # Decrease step size
            eta *= 0.5
            if eta < 1e-6:
                break
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate starting points
    starting_points = generate_starting_points()
    
    best_h = None
    best_c5 = float('inf')
    
    for i, h0 in enumerate(starting_points):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Continuation loop
        h = h0.copy()
        alpha = 0.5
        
        while alpha <= 40 and time.time() - start_time < 0.95 * budget_s:
            h = accelerated_projected_gradient(h, alpha, max_iter=150)
            alpha *= 1.4
            
            # During continuation, periodically check for the best solution
            corr = compute_corr(h)
            current_c5 = dx * np.max(corr)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Subgradient polishing
        if time.time() - start_time < 0.95 * budget_s:
            h_polished = subgradient_polishing(h, max_iter=1500)
            corr = compute_corr(h_polished)
            current_c5 = dx * np.max(corr)
            if current_c5 < best_c5:
                best_h = h_polished.copy()
                best_c5 = current_c5
    
    if best_h is None or best_c5 == float('inf'):
        # Fallback to initial construction
        best_h = project_to_sum_box(np.array(initial_h_values), TARGET_SUM)
        corr = compute_corr(best_h)
        best_c5 = dx * np.max(corr)
    
    return (best_h.tolist(), float(best_c5), n_points)
