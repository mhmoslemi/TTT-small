def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant using the official evaluation method"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap integral C5
    c5 = dx * np.max(corr)
    return c5
