import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global variables for consistent evaluation
TARGET_SUM = None

dx = 2.0 / n_points

def sigmoid(x):
    """Numerically stable sigmoid function"""
    return 1 / (1 + np.exp(-x))

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    # Reverse h for correlation computation
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
        
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_spline_profile(z, n_points, m=7):
    """Generate spline profile from parameter vector z"""
    # Positions of knots on [0,1]
    x_knots = np.linspace(0, 1, m)
    # Map z to y values in [0,1]
    y_values = sigmoid(z)
    # Ensure symmetry: y[j] = y[m-1-j] 
    half = (m-1)//2
    y_full = np.zeros(m)
    y_full[:half+1] = y_values
    y_full[half+1:] = y_values[:half][::-1]
    
    # Create spline with clamped boundary conditions
    cs = CubicSpline(x_knots, y_full, bc_type='clamped')
    
    # Evaluate at uniform grid
    x_grid = np.linspace(0, 1, n_points//2)
    y_vals = cs(x_grid)
    # Apply symmetry to full domain [0,2]
    y_full = np.concatenate([y_vals, y_vals[::-1]])
    return y_full

def cma_es_optimization(seed=42, max_time_s=300):
    """CMA-ES optimization loop with minimal dependencies"""
    np.random.seed(seed)
    
    # Parameters for CMA-ES
    d = 4  # Dimension of parameter space
    lambda_ = 4 + int(3 * np.log(d))  # Population size
    mu = lambda_ // 2
    sigma = 2.0
    C = np.eye(d)
    mean = np.zeros(d)
    
    best_f = float('inf')
    best_z = None
    start_time = time.time()
    
    n_generations = 300  # Max generations
    for gen in range(n_generations):
        if time.time() - start_time > max_time_s - 5:
            break
            
        # Sample offspring
        offspring = []
        for _ in range(lambda_):
            z = mean + sigma * np.random.multivariate_normal(np.zeros(d), C)
            offspring.append(z)
            
        # Evaluate in fitness space
        fitness_scores = []
        for z in offspring:
            h = compute_spline_profile(z, n_points)
            # Scale and clip to ensure integral = 1 and values in [0,1]
            s = np.sum(h)
            alpha = n_points / 2 / s if s > 0 else 1.0
            h_scaled = np.clip(alpha * h, 0, 1)
            # Ensure sum constraint
            h_scaled = project_to_sum_box(h_scaled, n_points/2)
            c5 = evaluate_c5(h_scaled)
            fitness_scores.append(c5)
            
        # Selection
        indices = np.argsort(fitness_scores)[:mu]
        selected = [offspring[i] for i in indices]
        selected_fitness = [fitness_scores[i] for i in indices]
        
        # Update parameters
        if len(selected) > 0:
            z_mean = np.mean(selected, axis=0)
            mean = z_mean
            
            # CMA-ES covariance update (simplified rank-mu update)
            rank_updates = np.zeros((d, d))
            for z in selected:
                v = z - mean
                rank_updates += np.outer(v, v) / mu
            
            # Update with learning rate
            C = 0.9 * C + 0.1 * rank_updates / (sigma ** 2)
            
            # Step size update (simplified)
            sigma = sigma * np.exp(1.0 / (d+10) * (np.mean(selected_fitness) - best_f) / best_f)
            
            # Update best
            if selected_fitness[0] < best_f:
                best_f = selected_fitness[0]
                best_z = selected[0].copy()
    
    # Refine with final scan in case of early stopping
    if best_z is not None:
        h = compute_spline_profile(best_z, n_points)
        s = np.sum(h)
        alpha = n_points / 2 / s if s > 0 else 1.0
        h_scaled = np.clip(alpha * h, 0, 1)
        h = project_to_sum_box(h_scaled, n_points/2)
        best_f = evaluate_c5(h)
        
    return best_f, best_z

def polish_solution(h_orig, max_iter=1200):
    """Polishing with projected gradient descent"""
    h = h_orig.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    eta = 0.08
    
    for _ in range(max_iter):
        if time.time() - start_time > 550:
            break
            
        # Identify worst shifts
        corr = np.correlate(h, 1-h, mode='full')
        max_val = np.max(corr)
        K_max = np.where(np.abs(corr - max_val) < 1e-12)[0]
        
        # Compute subgradient for these shifts
        g = np.zeros_like(h)
        for k in K_max:
            if k < len(h):
                # Index shifts for correlation direction
                c_idx = k - (len(h)-1) // 2
                if c_idx < 0:
                    # h[0] contributes to corr[-c_idx], so overestimate
                    g[:c_idx] += 1 - h[c_idx:]
                elif c_idx > 0:
                    g[c_idx:] += 1 - h[:-c_idx]
                else:
                    g[:] += 1 - h[:]
                    
        # Normalize by number of contributing shifts
        if len(K_max) > 0:
            g /= len(K_max)
            
        # Gradient descent step
        h_new = h - eta * g
        h_new = project_to_sum_box(h_new, n_points/2)
        new_c5 = evaluate_c5(h_new)
        
        # Line search
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            if eta < 0.2:
                eta *= 1.1
        else:
            eta *= 0.5
            
        # Save best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Plateau escape every 200 iterations
        if _ % 200 == 0 and _ > 0:
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            delta = np.random.uniform(-0.015, 0.015)
            temp_h = h.copy()
            temp_h[i1] += delta
            temp_h[i2] -= delta
            # Clamp values
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = project_to_sum_box(temp_h, n_points/2)
            temp_c5 = evaluate_c5(temp_h)
            if temp_c5 < best_c5:
                h = temp_h
                best_c5 = temp_c5
                best_h = h.copy()
    
    return best_h, best_c5
