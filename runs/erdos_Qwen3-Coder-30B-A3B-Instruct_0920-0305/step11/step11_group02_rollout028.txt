import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient of smooth max using FFT"""
    L = 2 * len(h) - 1
    w_rev = w[::-1]
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    # Only take the center part
    center = len(h) - 1
    grad = -(conv_plus[center:] + conv_rev[center:])
    return grad

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    O = autocorrelation(h)
    g = TARGET_SUM - O
    return dx * np.max(g)

def evaluate_c5_smooth(h, alpha=1.0):
    """Evaluate smooth surrogate of C5"""
    O = autocorrelation(h)
    g = TARGET_SUM - O
    # For numerical stability, subtract max
    g_shifted = g - np.max(g)
    exp_g = np.exp(alpha * g_shifted)
    w = exp_g / np.sum(exp_g)
    grad = gradient_from_weights(h, w)
    return -np.dot(w, g), grad
