import numpy as np
import cvxpy as cp
from scipy.signal import correlate
from scipy.sparse import diags
import time

def proj_sum_box(v, target_sum):
    """Project vector v onto box [0,1] with sum = target_sum using binary search."""
    low, high = -1.0, 1.0
    while True:
        mid = (low + high) / 2
        projected = np.clip(v + mid, 0, 1)
        current_sum = np.sum(projected)
        if abs(current_sum - target_sum) < 1e-12:
            return projected
        elif current_sum > target_sum:
            high = mid
        else:
            low = mid

def shift_mask(k, n):
    """Generate sparse matrix for shift k."""
    offsets = [k]
    data = [1]
    return diags(data, offsets, shape=(n, n)).toarray()

def corr_full(h):
    """Compute full correlation using FFT."""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def max_overlap(h, dx):
    """Compute max overlap using correlation."""
    c = corr_full(h)
    return dx * np.max(c)

def grad_shift(h, k):
    """Compute gradient of overlap for shift k."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] = 1 - h[:n-k]
        g[:k] = -h[:k]
    else:
        g[:n+k] = 1 - h[-k:]
        g[n+k:] = -h[n+k:]
    return g

def polish(h_start, n, dx, budget_remaining):

        c = corr_full(h)
        Cmax = np.max(c)
        Kmax = np.where(c >= Cmax - 1e-12)[0]
        
        g = np.zeros(n)
        for k in Kmax:
            g += grad_shift(h, k)
        g /= len(Kmax)
        
        # Nesterov-style extrapolation
        y = h + 0.9 * (h - h_prev)

        new_C5 = max_overlap(h_new, dx)
        old_C5 = max_overlap(h, dx)

            h = h_new.copy()
