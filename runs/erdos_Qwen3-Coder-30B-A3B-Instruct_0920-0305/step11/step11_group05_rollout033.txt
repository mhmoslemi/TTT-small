import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time
from cvxpy import *
import cvxpy as cp

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def corr_fft(h):
    """Compute full convolution of h with (1-h) using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.real(ifft(fft(h_rev, L) * fft(b, L)))
    return corr

def evaluate_c5(h):
    """Evaluate the C5 bound using FFT-based correlation"""
    corr = corr_fft(h)
    return dx * np.max(corr)

def project_onto_sum_box(v, target_sum):
    """Project vector v onto [0,1]^n with sum = target_sum using water-filling algorithm"""
    n = len(v)
    # Sort values in descending order
    sorted_indices = np.argsort(v)[::-1]
    sorted_v = v[sorted_indices]
    
    # Initial guess of lambda
    low, high = -1.0, 1.0
    while True:
        mid = (low + high) / 2
        projected = np.clip(sorted_v + mid, 0, 1)
        sum_proj = np.sum(projected)
        if np.abs(sum_proj - target_sum) < 1e-10:
            break
        elif sum_proj > target_sum:
            low = mid
        else:
            high = mid
            
    lambda_val = (low + high) / 2
    result = np.clip(sorted_v + lambda_val, 0, 1)
    # Restore original order
    final = np.empty_like(result)
    final[sorted_indices] = result
    
    return final

def lp_step(y):
    """Solve the linearized subproblem with fixed y"""
    n = len(y)
    # Variables
    x = Variable(n)
    t = Variable()
    
    # Constraints
    constraints = [x >= 0, x <= 1, sum(x) == TARGET_SUM]
    
    # Compute shifts for all s in [-n+1, n-1]
    for s in range(-n+1, n):
        # Build constraint for shift s
        if s >= 0:
            left = y[s:]
            right = x[:n-s]
        else:
            left = y[:n+s]
            right = x[-s:]
        
        # Sum of product
        constraints.append(sum(left * right) >= t)
    
    # Objective
    prob = Problem(Maximize(t), constraints)
    
    # Solve with ECOS
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            x_val = x.value
            t_val = t.value
            return x_val, t_val
        else:
            return None, None
    except Exception:
        return None, None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Initial vectors: try multiple restarts
    restarts = []
    
    # Include the provided initial solution
    if 'initial_h_values' in globals():
        restarts.append(np.array(initial_h_values))
    else:
        restarts.append(np.random.uniform(0, 1, n_points))
    
    # Add random starting points
    for _ in range(5):
        restarts.append(np.random.uniform(0, 1, n_points))
    
    # Add a simple tent-like shape
    tent = np.zeros(n_points)
    mid = n_points // 2
    for i in range(n_points):
        if i <= mid:
            tent[i] = 2 * i / (mid)
        else:
            tent[i] = 2 - 2 * (i - mid) / (n_points - mid)
    restarts.append(tent)
    
    # Docking time for final sanity checks
    final_time = budget_s * 0.95
    
    for i, y0 in enumerate(restarts):
        y = project_onto_sum_box(y0, TARGET_SUM)
        t_prev = -float('inf')
        max_iter = 50  # Max alternating steps
        
        for iter_count in range(max_iter):
            if time.time() - start_time > final_time:
                break
                
            # Solve LP with current y
            x, t = lp_step(y)
            if x is None:
                break
                
            # Update
            y = x
            
            # Check for convergence
            if t - t_prev < 1e-8:
                break
            t_prev = t
            
            # Evaluate
            tmp_h = project_onto_sum_box(y, TARGET_SUM)
            tmp_c5 = evaluate_c5(tmp_h)
            if tmp_c5 < best_c5:
                best_c5 = tmp_c5
                best_h = tmp_h.copy()
        
        if best_h is not None and time.time() - start_time > final_time:
            break
    
    # Early return if stuck
    if best_h is None:
        best_h = project_onto_sum_box(np.random.uniform(0, 1, n_points), TARGET_SUM)
        best_c5 = evaluate_c5(best_h)
        
    # Local refinement with pairwise exchange if time remains
    if time.time() - start_time < budget_s * 0.9:
        # Refine the best solution found using simulated annealing with pairwise exchanges
        current_h = best_h.copy()
        current_c5 = best_c5
        temp = 0.1
        cooling_rate = 0.999
        max_refinement_steps = 5000
        
        # Precompute left-to-right shifts
        shifts = {}
        shifts_list = range(-n_points+1, n_points)
        for s in shifts_list:
            shifts[s] = np.array([(i + s) % n_points if 0 <= i + s < n_points else -1 for i in range(n_points)])
        
        # Precompute initial correlation for fast updates
        def fast_corr_update(h, i, j, delta):
            """Update correlation efficiently in O(1) time"""
            # We'll account for the change in total correlation sum
            old_sum = 0.0
            new_sum = 0.0
            
            for s in shifts_list:
                # old contribution of indices i, j to shift s
                idx_i = i
                idx_j = j
                
                if s >= 0:
                    idx_j_s = (j + s) % n_points
                    idx_i_s = (i + s) % n_points
                else:
                    idx_j_s = (j - s) % n_points
                    idx_i_s = (i - s) % n_points
                
                if 0 <= idx_i_s < n_points and 0 <= idx_j_s < n_points:
                    old_sum += h[idx_i] * h[idx_j_s]
                    old_sum += h[idx_j] * h[idx_i_s]

                # new contribution
                new_idx_i = min(max(idx_i + delta, 0), 1)
                new_idx_j = min(max(idx_j - delta, 0), 1)
                
                new_sum += new_idx_i * h[idx_j_s] + new_idx_j * h[idx_i_s]
                
            # Return change in sum
            return new_sum - old_sum
                
        # Main refinement loop
        for step in range(max_refinement_steps):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Choose two random indices
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            
            # Choose move size (with decreasing amplitude)
            move_scale = 0.1/max(1.0, step / 100.0)  
            delta = np.random.uniform(-move_scale, move_scale)
            
            # Ensure we don't go outside bounds
            if current_h[i] + delta < 0 or current_h[i] + delta > 1:
                continue
            if current_h[j] - delta < 0 or current_h[j] - delta > 1:
                continue
            
            # Apply move and project
            h_trial = current_h.copy()
            h_trial[i] += delta
            h_trial[j] -= delta
            
            # Re-project to sum and unit box
            h_trial = project_onto_sum_box(h_trial, TARGET_SUM)
            
            # Evaluate new C5 value
            new_c5 = evaluate_c5(h_trial) 
            
            if new_c5 < current_c5:
                current_h = h_trial
                current_c5 = new_c5
            else:
                # Accept with probability that decreases with temperature
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                    current_h = h_trial
                    current_c5 = new_c5
                
            temp *= cooling_rate
            
        best_h = current_h
        best_c5 = current_c5

    # Final evaluation
    best_h = project_onto_sum_box(best_h, TARGET_SUM)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
