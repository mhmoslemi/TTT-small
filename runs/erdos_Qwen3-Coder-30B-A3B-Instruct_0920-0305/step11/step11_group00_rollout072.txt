import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Use analytical bounds to initialize search
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    
    # Ensure we have a valid interval
    if f(lam_low) * f(lam_high) > 0:
        # If the initial interval doesn't bracket the root, expand it
        while f(lam_low) * f(lam_high) > 0:
            if f(lam_low) > 0:
                lam_low -= 1.0
            else:
                lam_high += 1.0
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (lam_low + lam_high) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(lam_low) < 0:
            lam_high = m
        else:
            lam_low = m
    
    result = np.clip(x + (lam_low + lam_high)/2, 0, 1)
    return result

def get_shift_gradient(h, k):
    """Get gradient of correlation at shift k"""
    # Pad h with zeros
    pad_h = np.concatenate([np.zeros(n_points), h, np.zeros(n_points)])
    base_indices = np.arange(n_points) + n_points
    
    # Compute gradients
    grad_k = (1 - pad_h[base_indices + k]) - pad_h[base_indices - k]
    return grad_k

def soft_max_grad(c, alpha):
    """Compute the gradient of log-sum-exp approximation"""
    c_scaled = alpha * c
    c_max = np.max(c_scaled)
    exp_c = np.exp(c_scaled - c_max)  # Subtract max to avoid overflow
    weights = exp_c / np.sum(exp_c)
    
    # Compute weighted sum of gradients
    grad = np.zeros_like(c)
    for i in range(len(c)):
        # Get the gradient for shift i
        grad_i = get_shift_gradient(h, i - n_points + 1)
        grad += weights[i] * grad_i
    
    return grad
