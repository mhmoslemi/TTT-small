import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def project_to_sum_box(y, target_sum):
    """Project vector y onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(y+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(y + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(y + (a + b)/2, 0, 1)
    return result

def compute_autocorr_fft(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient using FFT-based convolution"""
    n = len(h)
    L = 2 * n - 1
    w_rev = w[::-1]
    
    # Convolution with w (shifted by +j)
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    
    # Convolution with w_rev (shifted by -j)  
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Only take the valid center portion
    grad = -(conv_plus[n-1:] + conv_rev[n-1:])
    return grad

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_autocorr_fft(h)
    # The correlation is of length 2n-1, we want the middle n values
    # C5 = dx * max_j g_j where g_j = S - O_j and S = n/2
    n = len(h)
    S = n / 2
    g = S - corr[:2*n-1]
    return np.max(g)

    n_points = 300
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize best solution

    best_h = None
    
    # Two initialization strategies
    initializations = []

    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = project_to_sum_box(h_upsampled, target_sum)
    initializations.append(h_upsampled)
    
    # 2. Random binary start
    h_random = np.zeros(n_points)
    ones_indices = np.random.choice(n_points, size=int(target_sum), replace=False)
    h_random[ones_indices] = 1.0
    h_random = project_to_sum_box(h_random, target_sum)
    initializations.append(h_random)
    
    # Run optimization for each initialization
    for h0 in initializations:
        # Restore h0 for each optimization round
        current_h = h0.copy()
        current_c5 = evaluate_c5(h0)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Optimization parameters
        alpha = 1.0
        eta = 0.05
        max_iter = 5000
        last_improvement = 0
        
        for iteration in range(max_iter):
            if time.time() - start_time >= budget_s * 0.95:
                break
                
            # Compute autocorrelation and gradient components
            autocorr = compute_autocorr_fft(current_h)
            # We need the values in the range [0, 2n-1] which corresponds to shifts [0, 2n-2]
            # We want the actual shifts from -n+1 to n-1, so we extract the middle n values
            # But for simplicity we just use all values and find max
            g = target_sum - autocorr
            max_g = np.max(g)
            
            # Compute smooth max and weights
            g_shifted = g - max_g
            exp_g = np.exp(alpha * g_shifted)
            w = exp_g / np.sum(exp_g)
            
            # Compute gradient
            grad = gradient_from_weights(current_h, w)
            
            # Gradient step
            y = current_h - eta * grad
            current_h = project_to_sum_box(y, target_sum)
            
            # Evaluate
            current_c5 = evaluate_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                last_improvement = iteration
            
            # Adaptive scheduling
            if iteration % 200 == 0:
                alpha = min(alpha * 1.3, 100.0)
                eta *= 0.9
            
            # Random shakeup
            if iteration % 500 == 0 and iteration - last_improvement > 200:
                noise = 0.02 * np.random.randn(n_points)
                current_h = project_to_sum_box(np.clip(current_h + noise, 0, 1), target_sum)
                last_improvement = iteration
        
        if time.time() - start_time >= budget_s * 0.95:
            break
    
    # Final polishing with SLSQP
    if best_h is not None and time.time() - start_time < budget_s * 0.9:
        def obj_full(x):
            corr = compute_autocorr_fft(x)
            return dx * np.max(target_sum - corr[:2*n_points-1])
        
        bounds = [(0.0, 1.0)] * n_points
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        res = minimize(obj_full,
                      x0=best_h,
                      method='SLSQP',
                      bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 200, 'ftol': 1e-12})
        
        if res.success:
            polished_c5 = obj_full(res.x)
            if polished_c5 < best_c5:
                best_c5 = polished_c5
                best_h = res.x.copy()
    
    if best_h is None:
        # Fall back to initial solution
        best_h = initial_h_values.copy()
        best_h = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, len(initial_h_values)), 
                          initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = project_to_sum_box(best_h, target_sum)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), float(best_c5), n_points)
