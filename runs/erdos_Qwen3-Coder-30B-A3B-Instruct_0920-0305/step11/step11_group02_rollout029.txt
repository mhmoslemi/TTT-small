import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 300
dx = 2.0 / n_points
target_sum = n_points / 2

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return corr

def gradient_from_weights(h, w):
    """Compute gradient using FFT-based convolution"""
    L = 2 * len(h) - 1
    w_rev = w[::-1]
    
    # Convolution with h shifted by +j
    conv_plus = np.fft.ifft(np.fft.fft(w, L) * np.fft.fft(h, L)).real
    
    # Convolution with h shifted by -j  
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract valid range (centered at i + (n-1))
    grad = -(1/target_sum) * (conv_plus[(n_points-1):] + conv_rev[(n_points-1):])
    return grad

def compute_c5(h):
    """Fast evaluation of C5 using FFT"""
    autocorr = compute_autocorrelation(h)
    g = target_sum - autocorr
    return dx * np.max(g)

    # Try both and take the better
    c5_0 = compute_c5(h0)

    best_c5 = current_c5
    
    # Smooth max optimization parameters
    alpha = 1.0
    alpha_max = 100.0
    eta = 0.05
    max_iters = 10000

        # Adaptive scheduling
        if iteration % 200 == 0:
            alpha = min(alpha * 1.3, alpha_max)

        # Occasional shake-up
        if iteration % 500 == 0 and iteration > 0:

            h_shaken = current_h + noise
