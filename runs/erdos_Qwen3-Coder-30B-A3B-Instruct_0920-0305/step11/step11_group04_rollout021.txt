import numpy as np
from scipy.signal import correlate
import time

# Global constants
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def autocorr(h):
    """Compute linear autocorrelation using full mode correlation"""
    # correlate(h, h, mode='full') gives correlation for all shifts
    # output length = 2*n -1, indexed from -(n-1) to n-1
    corr = correlate(h, h, mode='full')
    return corr * dx

def compute_C5(h):
    """Compute C5 value from h"""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto [0,1]^n with sum = target_sum using water-filling"""
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum
    
    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse binary sequence of length n"""
    # Generate Thue-Morse sequence for n=128 (power of 2)
    tm = [0]
    for _ in range(7):  # log2(128) = 7
        tm = tm + [(1-x) for x in tm]
    
    # Take first n elements
    tm = tm[:n]
    
    # Normalize sum to target_sum
    h = np.array(tm, dtype=float)
    h = project_to_sum_box(h, TARGET_SUM)
    return h

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Apply greedy local swaps to improve worst shift correlation"""

    max_iter = 1000
    iter_count = 0 
    while iter_count < max_iter and (time.time() - start_time < budget_s * 0.95):
        # Find all zeros and ones
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_swap = None
        best_delta = -1e-10
        
        # Check each zero-one pair to see if swapping improves worst shift
        for i_zero in zeros:
            for i_one in ones:
                # Compute change in correlation at worst shift
                delta = 0
                
                # Check if swapping i_zero and i_one changes correlation at worst shift
                # Compute contribution changes from h[i_zero] and h[i_one]
                
                # Check contribution from shifted positions
                if (i_zero + worst_shift >= 0) and (i_zero + worst_shift < n_points):
                    if h[i_zero + worst_shift] == 1:
                        delta -= dx  # losing one contribution 
                if (i_one - worst_shift >= 0) and (i_one - worst_shift < n_points):
                    if h[i_one - worst_shift] == 1:
                        delta -= dx  # losing one contribution
                
                if (i_one + worst_shift >= 0) and (i_one + worst_shift < n_points):
                    if h[i_one + worst_shift] == 0:
                        delta += dx  # gaining one contribution
                if (i_zero - worst_shift >= 0) and (i_zero - worst_shift < n_points):
                    if h[i_zero - worst_shift] == 0:
                        delta += dx  # gaining one contribution
                
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i_zero, i_one)
        
        # If found an improving swap, make it
        if best_swap and best_delta > 0:
            i_zero, i_one = best_swap
            h[i_zero] = 1
            h[i_one] = 0
            iter_count += 1
        else:
            break
            
    return h
