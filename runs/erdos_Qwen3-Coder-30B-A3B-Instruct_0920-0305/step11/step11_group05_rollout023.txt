import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import time
import cvxpy as cp

# Global constants
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def corr_fft(h):
    """Compute full cross-correlation of h with 1-h"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Evaluate C5 using FFT"""
    corr = corr_fft(h)
    return dx * np.max(corr)

def project_onto_sum_box(v, target_sum):
    """Project vector v to [0,1] with sum = target_sum (using waterfilling)"""
    v = np.array(v)
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    
    # Find the threshold lambda using binary search
    low, high = -1.0, 1.0
    tol = 1e-12
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(v + lam, 0, 1))
    
    while high - low > tol:
        mid = (low + high) / 2
        if sum_at_lambda(mid) > target_sum:
            high = mid
        else:
            low = mid
    
    lambda_val = (low + high) / 2
    result = np.clip(v + lambda_val, 0, 1)
    return result

def build_and_solve_lp(y):
    """Solve the LP for fixed vector y"""
    # Parameters
    n = len(y)
    # Build constraints: sum_i x_i * y_{i+s} >= t for each s, sum_i x_i = target_sum, 0 <= x_i <= 1
    # For the full correlation, s ranges from -(n-1) to (n-1)  
    
    # Build LHS matrix for all shift constraints
    A_eq = np.zeros((1, n))
    A_eq[0] = np.ones(n)
    b_eq = np.array([TARGET_SUM])
    
    # Constraints corresponding to shifts s = -(n-1), ..., n-1
    A_ub_rows = []
    b_ub = []
    
    for shift in range(-(n-1), n):
        row = np.zeros(n)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                row[i] = y[j]
        A_ub_rows.append(row)
        b_ub.append(0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)  # These are set to zero because we're modeling sum >= t as -sum <= -t
    
    # LP formulation: maximize t subject to constraints
    # Since we maximize t, and the index i of t is the last variable,
    # we define the objective vector as [0, 0, ..., 0, -1]
    # But the convex optimization library will prefer formulation in other way.
    # Let's define a similar formulation using cvxpy
    
    # Variables
    x = cp.Variable(n)
    t = cp.Variable(1)
    
    # Constraints
    constraints = [cp.sum(x) == TARGET_SUM,
                   x >= 0, 
                   x <= 1]
    
    # Add shift constraints
    for shift in range(-(n-1), n):
        expr = 0.0
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                expr += x[i] * y[j]
        constraints.append(expr >= t)
        
    # Objective
    objective = cp.Maximize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000, feastol=1e-8)
        if prob.status == cp.OPTIMAL:
            x_sol = x.value
            t_sol = t.value
            return x_sol, t_sol
        else:
            # fallback to scipy linprog if cvxpy fails
            return None, None
    except:
        return None, None


def alternating_LP(y, max_iter=30, tol=1e-8):
    """Run alternating LP iterations"""
    h = y.copy()
    best_t = -1e10
    best_h = h.copy()
    
    for i in range(max_iter):
        x, t = build_and_solve_lp(h)
        if x is None:
            break
        h = x
        if t > best_t:
            best_t = t
            best_h = h.copy()
        if i > 1 and abs(t - best_t) < tol:
            break
    return best_h, best_t

def symmetric_average(h):
    """Apply symmetry constraint: h = (h + reverse(h)) / 2, then project"""
    h_sym = (h + h[::-1]) / 2
    return project_onto_sum_box(h_sym, TARGET_SUM)

def refine_with_pairwise_exchange(h, n_trials=5000):
    """Perform fine-grained refinement by pairwise moves"""
    h = h.copy()
    old_c5 = evaluate_c5(h)
    
    # Precompute correlation changes due to updates
    corr = corr_fft(h)
    current_max_corr = np.max(corr)
    
    # Compute contribution of each element to each shift
    nshifts = 2 * n_points - 1
    shift_contrib = np.zeros((n_points, nshifts))
    for s in range(-(n_points-1), n_points):
        s_idx = s + n_points - 1
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                shift_contrib[i, s_idx] += h[j]
    
    for _ in range(n_trials):
        # Randomly choose indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
        
        # Compute maximal delta
        d = np.random.uniform(-0.01, 0.01)
        if h[i] + d < 0:
            d = -h[i]
        elif h[i] + d > 1:
            d = 1 - h[i]
        if h[j] - d < 0 or h[j] - d > 1:
            continue
        new_h = h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_h = project_onto_sum_box(new_h, TARGET_SUM)
        
        # Exact C5 update
        new_c5 = evaluate_c5(new_h)
        if new_c5 <= old_c5:
            h = new_h
            old_c5 = new_c5
        elif np.random.rand() < np.exp(-(new_c5 - old_c5) * 100):
            h = new_h
            old_c5 = new_c5
            
    return h
