import numpy as np
from scipy.fft import fft, ifft
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import ECOS, GLPK, CBC

TARGET_SUM = None
n_points = 144
dx = 2.0 / n_points

def project_onto_sum_box(v, target_sum):
    """
    Project v onto box constraints [0,1] with sum = target_sum.
    Uses fast O(n log n) water-filling approach.
    """
    n = len(v)
    # Sort indices by value (descending)
    idx_sorted = np.argsort(-v)
    v_sorted = v[idx_sorted]
    
    # Find the threshold lambda
    # s_i = max(0, v_i - lambda)
    # sum(s_i) = target_sum
    # So sum_i max(0, v_i - lambda) = target_sum
    # This can be solved by binary search or direct computation
    
    # Accumulate prefix sums
    prefix_sum = np.cumsum(v_sorted)
    
    # Find the largest k such that sum_{i=0}^{k} max(0, v_sorted[i] - lambda) >= target_sum
    # i.e., k*lambda <= sum_{i=0}^{k} v_sorted[i] - target_sum
    # lambda <= (sum_{i=0}^{k} v_sorted[i] - target_sum) / k  
    # Want largest k where this is >= 0
    k = 0
    for i in range(n):
        s = prefix_sum[i] - target_sum
        if s < 0:
            break
        k = i + 1
    if k == 0:
        # All entries need to be 0 or small values
        return np.zeros_like(v) if target_sum > 0 else v.copy()
    
    # lambda = (sum_{i=0}^{k-1} v_sorted[i] - target_sum) / k
    if k == n:
        # lambda = (sum(v_sorted) - target_sum) / n
        lambda_val = (np.sum(v_sorted) - target_sum) / n
        # All values clipped to [lambda_val, 1] or [0, lambda_val]
        result = np.clip(v_sorted, 0, lambda_val)
        # Reassign back to original indices
        final_result = v.copy()
        final_result[idx_sorted] = result
        return final_result
    else:
        # Lambda for first k entries
        s = prefix_sum[k-1] - target_sum
        lambda_val = s / k
        # First k entries: max(0, v_sorted[i] - lambda_val)
        # Remaining n-k entries: 0 (since v_sorted[i] - lambda_val < 0)
        result = np.zeros_like(v_sorted)
        result[:k] = np.maximum(0, v_sorted[:k] - lambda_val)
        # Reassign back to original indices
        final_result = v.copy()
        final_result[idx_sorted] = result
        return final_result

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def eval_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr)

def solve_linearized_step(y):
    """
    Solve the linearized LP for current y
    min t subject to: 
        sum_{i} x_i * y_{i+s} >= t for all s in [-n+1, ..., n-1]
        sum_{i} x_i = TARGET_SUM
        0 <= x_i <= 1
    """
    n = len(y)
    # For each shift s, we sample a row of the linear constraint matrix
    # Build constraints: sum_{i} x_i * y_{i+s} >= t, where indices are wrapped:
    # y_{i+s} = 0 if i+s < 0 or i+s >= n
    
    # Variables: x_i, t
    x = cp.Variable(n)
    t = cp.Variable()
    
    # Note that for shift s, the constraint is sum_{i=0}^{n-1} x_i * y_{i+s} - t >= 0
    # y_{i+s} = y[(i+s) % n] for i=0 to n-1
    constraints = []
    
    # Add all shift constraints
    for s in range(-n+1, n):
        if abs(s) >= n:
            # x_i * y_{i+s} = 0 for all i, so sum = 0
            # constraint 0 >= t, i.e. t <= 0
            constraints.append(t <= 0)
        else:
            # sum_{i=0}^{n-1} x_i * y_{i+s} >= t
            # y_{i+s} = y[(i+s) % n] where i >= 0 and i + s >= 0 and i + s < n
            # So we compute the shifted y vector, outside range gets replaced by zero
            # Build the column vector for y_{i+s}
            y_shifted = np.zeros(n)
            if s >= 0:
                y_shifted[s:] = y[:-s]
            else:
                y_shifted[:s] = y[-s:]
            constraints.append(cp.sum(cp.multiply(x, y_shifted)) >= t)

    # Add sum constraint and box constraints
    constraints.append(cp.sum(x) == TARGET_SUM)
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Maximize t
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try multiple solvers to avoid solver-specific issues
        solvers = [ECOS]

        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status == cp.OPTIMAL:
                    break
            except:
                continue

        if prob.status != cp.OPTIMAL:
            # fallback to a high tolerance, silly but works
            prob.solve(solver=GLPK, verbose=False)
            
        if prob.status != cp.OPTIMAL:
            # Last resort fallback to very simple method
            # Take a Hessian-based method or just make a warm start or maybe noise injection
            
            x_sol = np.array([1.0]*n)
            x_sol = project_onto_sum_box(x_sol, TARGET_SUM)
            t_val = np.min([np.sum(x_sol * np.roll(y, s)) for s in range(-n+1, n)])
            return x_sol, t_val
            
        x_sol = x.value
        t_val = t.value
        
        # Project back to feasible domain, just in case
        x_sol = project_onto_sum_box(x_sol, TARGET_SUM)
        
        return x_sol, t_val
    except Exception as e:
        # Fallback just in case
        x_sol = np.array([1.0]*n)
        x_sol = project_onto_sum_box(x_sol, TARGET_SUM)
        t_val = np.min([np.sum(x_sol * np.roll(y, s)) for s in range(-n+1, n)])
        return x_sol, t_val

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points / 2
    total_time = budget_s
    best_c5 = float('inf')
    best_h = None
    
    all_seeds = []
    # Use provided initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_onto_sum_box(h0, TARGET_SUM)
        all_seeds.append(('initial', h0))
        
    # Add some random seeds with different initialization strategies
    for i in range(5):
        # Random uniform vector
        rand_h = np.random.uniform(0, 1, n_points)
        rand_h = project_onto_sum_box(rand_h, TARGET_SUM)
        all_seeds.append((f'rand_{i}', rand_h))
        
    # Add a tent shaped vector
    tent = np.array([min(i, n_points - 1 - i) for i in range(n_points)])
    tent = tent / np.sum(tent) * TARGET_SUM
    all_seeds.append(('tent', tent))

    # Run the alternating algorithm with different seeds
    max_iterations = 30
    remaining_time = total_time * 0.7  # 70% of the time for main loop
    
    for seed_name, h0 in all_seeds:
        if time.time() - start_time > total_time * 0.95:
            break

        current_h = h0.copy()
        current_t = 0.0
        old_t = -1.0
        
        for iteration in range(max_iterations):
            if time.time() - start_time > total_time * 0.95:
                break
            # Solve the LP subproblem with fixed current_h
            new_h, new_t = solve_linearized_step(current_h)
            
            # Update the candidate solution
            current_h = new_h.copy()
            old_t = current_t
            current_t = new_t
            
            # Check for convergence
            if abs(current_t - old_t) < 1e-8:
                break
            
        c5_val = eval_c5(current_h)
        # Apply symmetry averaging to potentially improve it further
        if n_points > 1:
            sym_h = (current_h + current_h[::-1]) / 2
            sym_h = project_onto_sum_box(sym_h, TARGET_SUM)
            sym_c5 = eval_c5(sym_h)
            if sym_c5 < c5_val:
                current_h = sym_h
                c5_val = sym_c5
                
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = current_h.copy()

    # Refinement phase using simulated annealing and pairwise exchanges 
    if best_h is None:
        best_h = np.full(n_points, 0.5)
        best_h = project_onto_sum_box(best_h, TARGET_SUM)

    current_h = best_h.copy()
    current_c5 = eval_c5(current_h)
    refinement_time = total_time * 0.2
    step = 0
    temp = 1.0
    
    while step < 5000 and (time.time() - start_time) < total_time - 1.0:
        # Pairwise exchange
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.01, 0.01)  # Small variation
        
        # Ensure delta doesn't violate bounds
        h_copy = current_h.copy()
        h_copy[i] += delta
        h_copy[j] -= delta
        
        # Clip bounds
        if h_copy[i] < 0:
            temp_d = h_copy[i]  
            h_copy[i] = 0
            h_copy[j] -= temp_d
            
        elif h_copy[i] > 1:
            temp_d = h_copy[i] - 1
            h_copy[i] = 1
            h_copy[j] += temp_d
            
        if h_copy[j] < 0:
            temp_d = h_copy[j]
            h_copy[j] = 0
            h_copy[i] -= temp_d
            
        elif h_copy[j] > 1:
            temp_d = h_copy[j] - 1
            h_copy[j] = 1
            h_copy[i] += temp_d

        # Re-project to sum and box constraints
        h_copy = project_onto_sum_box(h_copy, TARGET_SUM)
        
        # Compute new C5 (just necessary for testing)
        new_c5 = eval_c5(h_copy)
        if new_c5 < current_c5:
            current_h = h_copy.copy()
            current_c5 = new_c5
        else:
            # Accept with probability
            exp_arg = (current_c5 - new_c5) / (temp + 1e-10)
            if np.random.rand() < np.exp(exp_arg):
                current_h = h_copy.copy()
                current_c5 = new_c5
                
        # Update temperature
        temp *= 0.9995
        step += 1
        
    best_c5 = current_c5
    best_h = current_h.copy()
    
    # Final verification and projection if needed
    best_h = project_onto_sum_box(best_h, TARGET_SUM)
    final_c5 = eval_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
