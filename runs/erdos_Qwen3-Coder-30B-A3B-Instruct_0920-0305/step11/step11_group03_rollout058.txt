import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 80
    dx = 2.0 / n_points
    M = n_points / 2
    
    # SDP solve time budget
    sdp_time_budget = min(250.0, budget_s * 0.4)
    
    # Solve SDP for autocorrelation
    r = cp.Variable(n_points)
    t = cp.Variable()
    
    # Toeplitz matrix
    T = cp.Variable((n_points, n_points), symmetric=True)
    
    # Construct Toeplitz matrix from r
    for i in range(n_points):
        for j in range(n_points):
            T[i,j] = r[abs(i-j)]
    
    # Constraints
    constraints = [
        T >> 0,  # PSD constraint
        cp.sum((n_points - np.arange(n_points)) * r) == M**2,  # Mass constraint
        r >= t,  # Lower bound constraint
        r >= 0,  # Non-negativity
        r <= np.minimum(M, n_points - np.arange(n_points))  # Box constraints
    ]
    
    # Objective
    objective = cp.Maximize(t)
    
    # Solve SDP
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, eps=1e-6, max_iters=10000, verbose=False, 
                   use_indirect=True, scaling=5)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            r_opt = r.value
            t_opt = t.value
        else:
            raise Exception("SDP not solved to optimality")
    except Exception as e:
        # Fallback to projected gradient descent from uniform
        h0 = np.ones(n_points) * 0.5
        h0 = project_to_sum_box(h0, M)
        return (h0.tolist(), evaluate_c5(h0, dx), n_points)
    
    # Extract solution
    r_sol = r_opt
    t_sol = t_opt
    
    # Convert to feasible signal using spectral factorization
    best_h = None
    best_C5 = float('inf')
    
    # Try multiple phase seeds
    phase_seeds = [0, np.pi/4, np.pi/2, np.pi]
    
    for phase_seed in phase_seeds:
        # Compute power spectrum
        S = np.fft.fft(r_sol)
        A = np.sqrt(np.maximum(S, 0))  # Non-negative real square root
        
        # Apply phase
        phase = phase_seed + np.random.random(n_points) * 0.01  # Small perturbation
        h_tilde = np.fft.ifft(A * np.exp(1j * phase)).real
        
        # Project to feasible set
        h_proj = project_to_sum_box(h_tilde, M)
        
        # Evaluate C5
        C5 = evaluate_c5(h_proj, dx)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h_proj.copy()
    
    # Local refinement with projected gradient descent
    max_iter = 50
    eta = 0.2
    epsilon = 1e-8
    current_h = best_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current objective
        current_C5 = evaluate_c5(current_h, dx)
        min_r = 1 - dx * current_C5
        
        # Compute gradient for all active shifts
        grad = np.zeros(n_points)
        for k in range(n_points):
            if abs(correlate(current_h, 1-current_h, mode='full')[k] - min_r) < epsilon:
                # Gradient contribution from shift k
                shift = k - (n_points - 1)
                if shift >= 0:
                    grad[:n_points-shift] += current_h[shift:]
                else:
                    grad[-shift:] += current_h[:n_points+shift]
        
        # Projected gradient step with step size decreasing as sqrt(iter)
        eta_step = eta / np.sqrt(i+1)
        new_h = current_h + eta_step * grad
        new_h = project_to_sum_box(new_h, M)
        
        # Evaluate new solution
        new_C5 = evaluate_c5(new_h, dx)
        
        if new_C5 < current_C5:
            current_h = new_h.copy()
        else:
            # Reduce step size
            eta *= 0.95
            
    # Final evaluation
    final_C5 = evaluate_c5(current_h, dx)
    
    return (current_h.tolist(), float(final_C5), n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result  

def evaluate_c5(h, dx):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)
