import numpy as np

import time

# Global variables
n_points = 300
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def compute_autocorrelation(h):
    """Compute autocorrelation using FFT"""
    h_rev = h[::-1]
    L = 2 * len(h) - 1
    autocorr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(h, L)).real
    return autocorr

def compute_gradient_from_weights(h, weights):
    """Compute gradient using FFT-based convolution"""
    L = 2 * len(h) - 1
    w_rev = weights[::-1]
    
    # Convolution with h shifted by +j (forward shift)
    conv_plus = np.fft.ifft(np.fft.fft(weights, L) * np.fft.fft(h, L)).real
    
    # Convolution with h shifted by -j (backward shift) 
    conv_rev = np.fft.ifft(np.fft.fft(w_rev, L) * np.fft.fft(h, L)).real
    
    # Extract the center portion corresponding to valid indices
    center = len(h) - 1
    grad = -(1/TARGET_SUM) * (conv_plus[center:] + conv_rev[center:])
    
    return grad[:len(h)]

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    autocorr = compute_autocorrelation(h)
    # The autocorrelation includes negative shifts; we only care about non-negative
    # We are looking for max of g_j = S - O_j, so we want min of O_j excluding center
    g = TARGET_SUM - autocorr
    # The first index corresponds to shift=-n+1, so we take elements  n-1 to 2*n-2
    # This gives us shifts from -n+1 to n-1. We want to exclude shift 0.
    g_2n_1 = g[len(h)-1:]  # This contains indices from 0 to 2*n-2
    max_g = np.max(g_2n_1)
    return dx * max_g
