import numpy as np
from scipy.signal import correlate
from scipy.fft import fft, ifft
import time

# Global variable
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def project_to_sum_box(x, target_sum):
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def correlation_fft(h):
    """Compute the full correlation of h with (1-h) using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def smooth_max(corr, alpha):
    """Compute the smooth max of correlation values"""
    exp_corr = np.exp(alpha * corr)
    return (1 / alpha) * np.log(np.sum(exp_corr))

def compute_gradient(corr, h, alpha):
    """Compute gradient of smooth max with respect to h"""
    # Compute weights
    exp_corr = np.exp(alpha * corr)
    weights = exp_corr / np.sum(exp_corr)
    
    # Two convolutions needed: w * h and w_rev * h
    L = len(h) * 2 - 1
    w = weights
    w_rev = w[::-1]
    
    conv1 = np.convolve(w, h, mode='full')[:len(h)]
    conv2 = np.convolve(w_rev, h, mode='full')[:len(h)]
    
    grad = 1.0 - conv1 - conv2
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    popsize = 30
    population = []
    for _ in range(popsize):
        h = np.random.rand(n_points)
        h = project_to_sum_box(h, TARGET_SUM)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        corr = correlation_fft(ind)
        c5 = dx * np.max(corr)
        fitness.append(c5)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # DE parameters
    F = 0.5
    CR = 0.9
    
    # DE evolutionary loop
    de_time_limit = 0.6 * budget_s
    t0 = time.time()
    gen = 0
    
    while time.time() - t0 < de_time_limit:
        gen += 1
        for i in range(popsize):
            # Select three distinct individuals
            idxs = np.random.choice(popsize, 3, replace=False)
            r1, r2, r3 = idxs
            
            # Mutation
            mutant = population[r1] + F * (population[r2] - population[r3])
            
            # Crossover
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR or j == 0:
                    trial[j] = mutant[j]
                    
            # Repair
            trial = project_to_sum_box(trial, TARGET_SUM)
            
            # Evaluate
            corr = correlation_fft(trial)
            trial_c5 = dx * np.max(corr)
            
            # Selection
            if trial_c5 < fitness[i]:
                population[i] = trial
                fitness[i] = trial_c5
                
                # Update best
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        
        # Adaptive parameters
        if gen % 10 == 0:
            F = np.clip(F + np.random.normal(0, 0.05), 0.1, 1.0)
            CR = np.clip(CR + np.random.normal(0, 0.05), 0.0, 1.0)
    
    # Smoothed max gradient descent
    t1 = time.time()
    t2 = time.time()
    h = best_h.copy()
    alpha = 4.0
    iterations = 0
    
    while time.time() - t1 < 0.3 * budget_s:
        iterations += 1
        
        # Compute correlation
        c = correlation_fft(h)
        c5_true = dx * np.max(c)
        
        # Compute smooth max
        smooth_c5 = smooth_max(c, alpha)
        
        # Compute gradient
        grad = compute_gradient(c, h, alpha)
        
        # Line search
        eta = 0.5
        for _ in range(10):
            h_new = project_to_sum_box(h - eta * grad, TARGET_SUM)
            c_new = correlation_fft(h_new)
            c5_new = dx * np.max(c_new)
            
            if c5_new < c5_true:
                h = h_new
                best_c5 = min(best_c5, c5_new)
                break
            eta /= 2.0
            
        # Adapt alpha
        if iterations % 30 == 0:
            alpha = min(alpha * 1.5, 200.0)
            
        # Update best
        c = correlation_fft(h)
        c5 = dx * np.max(c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Binary rounding and swap climbing
    t3 = time.time()
    
    # Convert to binary
    t = np.median(best_h)
    b = (best_h >= t).astype(float)
    ones_count = np.sum(b)
    
    if ones_count > TARGET_SUM:
        # Fix excess 1s: convert largest elements to 0s
        indices = np.argsort(b)[::-1]
        for i in range(int(ones_count - TARGET_SUM)):
            b[indices[i]] = 0
    elif ones_count < TARGET_SUM:
        # Fix deficit 1s: convert smallest elements to 1s
        indices = np.argsort(b)
        for i in range(int(TARGET_SUM - ones_count)):
            b[indices[i]] = 1
    
    # Now try several rounds with swaps
    max_tries = int(5 * n_points)
    current_b = b.copy()
    best_binary_c5 = best_c5
    
    def compute_binary_c5(b_vector):
        # Compute correlation for binary values
        corr = correlation_fft(b_vector)
        return dx * np.max(corr)
        
    # Simple swap hill climb
    for _ in range(max_tries):
        if time.time() - t3 > 0.1 * budget_s:
            break
            
        i = np.random.choice(np.where(b_vector == 1)[0])
        j = np.random.choice(np.where(b_vector == 0)[0])
        
        # Try swapping
        b_test = b_vector.copy()
        b_test[i] = 0
        b_test[j] = 1
        
        c5_test = compute_binary_c5(b_test)
        if c5_test < best_binary_c5:
            best_binary_c5 = c5_test
            current_b = b_test.copy()
    
    # Return the best among continuous and binary
    if best_binary_c5 < best_c5:
        return (current_b.tolist(), float(best_binary_c5), n_points)
    else:
        return (best_h.tolist(), float(best_c5), n_points)
