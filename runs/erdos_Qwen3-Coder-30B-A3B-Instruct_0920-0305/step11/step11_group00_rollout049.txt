import numpy as np
from scipy.signal import correlate
from scipy.interpolate import CubicSpline
from scipy.optimize import minimize
import time

# Global constants
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
n_knots = 7  # odd number for symmetric knots
half_knots = (n_knots - 1) // 2  # number of knots on [0,1]

# Precomputed knot positions on [0,1]
knot_positions = np.linspace(0, 1, n_knots)

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def project_to_box(z):
    """Map from real line to [0,1] using sigmoid transformation"""
    return sigmoid(z)

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(h, target_sum):
    """Fast projection of h onto sum=target_sum, clip to [0,1]"""
    # Simple scaling
    total = np.sum(h)
    if abs(total) < 1e-15:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * (target_sum / total)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Renormalize to keep sum close to target
    h = h * (target_sum / np.sum(h))
    
    return h

def spline_profile(z, n_knots, knots_x):
    """Build a symmetric spline profile with z as control points"""
    # Transform z to [0,1] using sigmoid
    y = project_to_box(z)
    
    # For symmetric profile: use knots on [0,1], mirror to [1,2]
    y_vals = np.concatenate([y, y[::-1][1:]])
    # Add endpoints if needed
    knots_x_full = np.concatenate([knots_x, 2 - knots_x[1:][::-1]])
    
    # Create cubic spline
    cs = CubicSpline(knots_x_full, y_vals, bc_type='clamped')
    
    # Generate the full profile
    x_full = np.linspace(0, 2, n_points)
    h_full = cs(x_full)
    
    # Ensure all values are in [0,1] and enforce sum = n_points/2
    # Determine the scaling factor
    h_unscaled = np.clip(h_full, 0, 1)
    total_unscaled = np.sum(h_unscaled)
    scale = TARGET_SUM / total_unscaled
    h_final = np.clip(h_unscaled * scale, 0, 1)
    
    # ensure final sum is exactly correct
    h_final = project_to_sum_box(h_final, TARGET_SUM)
    
    return h_final

def evaluate_spline_objective(z):
    """Objective function for CMA-ES: evaluates C5 for given z"""
    h = spline_profile(z, n_knots, knot_positions)
    return evaluate_c5(h)

def cma_es_optimize(func, dim, budget_s):
    """Simple implementation of CMA-ES to optimize func with dim dimensions"""
    # Parameters from CMA-ES
    lambda_ = 4 + int(3 * np.log(dim))  # population size
    mu = lambda_ // 2  # number of parents
    
    # Initial conditions
    mean = np.zeros(dim)
    sigma = 2.0  # Initial step size
    C = np.eye(dim)  # Covariance matrix
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Best solution seen so far
    best_z = None
    best_cost = float('inf')
    
    # Time tracking
    start_time = time.time()
    
    generation = 0
    while time.time() - start_time < budget_s * 0.75:
        generation += 1
        
        # Generate offspring
        offspring = []
        for i in range(lambda_):
            z = mean + sigma * np.random.multivariate_normal(np.zeros(dim), C)
            offspring.append(z)
        
        # Evaluate all offspring
        costs = []
        for z in offspring:
            try:
                cost = func(z)
                costs.append(cost)
            except:
                costs.append(float('inf'))
        
        # Sort by cost
        sorted_indices = np.argsort(costs)
        best_idx = sorted_indices[0]
        if costs[best_idx] < best_cost:
            best_cost = costs[best_idx]
            best_z = offspring[best_idx].copy()
        
        # Select top mu individuals
        top_mu = [offspring[i] for i in sorted_indices[:mu]]
        
        # Update mean
        z_mean = np.sum([weights[i] * top_mu[i] for i in range(mu)], axis=0)
        mean = z_mean
        
        # Update covariance matrix
        C = (1 - 0.1) * C + 0.1 * np.outer(z_mean, z_mean)
        
        # Update step size
        sigma *= np.exp(0.1 * (np.sqrt(np.sum(z_mean**2)) / dim - 1))
    
    return best_z, best_cost

def polish_h(h, max_iter=1200):
    """Polish the solution using projected gradient descent"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    eta = 0.08  # initial step size
    for i in range(max_iter):
        if time.time() - start_time > 0.95 * 600:
            break
            
        # compute correlation for current h
        corr = compute_corr_fft(best_h)
        max_k = np.where(np.abs(corr - np.max(corr)) < 1e-12)[0]
        
        # compute gradient
        g = np.zeros_like(best_h)
        for k in max_k:
            if k == 0:
                g += (1 - best_h) - best_h
            else:
                g[:-k] += (1 - best_h[k:]) - best_h[:-k]
                g[k:] += (1 - best_h[:-k]) - best_h[k:]
        
        g = g / (len(max_k) + 1e-12)
        
        # update step
        new_h = best_h - eta * g
        
        # project to [0,1] and sum constraint
        new_h = project_to_sum_box(new_h, TARGET_SUM)
        
        # accept if it improves
        new_c5 = evaluate_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
            # line search step size
            if eta < 0.01:
                eta = 0.01
            else:
                eta *= 1.1
        else:
            # try smaller step size
            eta *= 0.5
            if eta < 1e-6:
                break
        
        # periodic random perturbation
        if i % 200 == 0:
            # make small random swap
            i1, i2 = np.random.randint(0, len(best_h), size=2)
            delta = np.random.uniform(-0.015, 0.015)
            old_val = best_h[i1]
            best_h[i1] += delta
            best_h[i2] -= delta
            best_h = np.clip(best_h, 0, 1)
            best_h = project_to_sum_box(best_h, TARGET_SUM)
            new_c5 = evaluate_c5(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                best_h[i1] = old_val
    
    return best_h, best_c5
