import numpy as np
from scipy.linalg import eigh
from scipy.sparse import diags, csr_matrix
import cvxpy as cp
import time
from scipy.signal import correlate

# Constants
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def build_shift_matrices(n):
    """Build shift matrices for all possible shifts"""
    shift_matrices = []
    for k in range(-n+1, n):
        # Create diagonal matrix with ones on k-th diagonal
        offsets = [k]
        data = [1] * (n - abs(k))
        shift_matrix = diags(data, offsets, shape=(n, n), format='csr')
        shift_matrices.append(shift_matrix)
    return shift_matrices

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def solve_sdp():
    """Solve SDP relaxation to find a good starting point"""
    # Build shift matrices efficiently
    shift_matrices = build_shift_matrices(n_points)
    
    # Create the SDP problem
    X = cp.Variable((n_points, n_points), symmetric=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        X >> 0,  # X is positive semidefinite
        cp.diag(X) == 1,  # diagonal elements are 1
        cp.trace(cp.ones((n_points, n_points)) @ X) == 0  # balanced constraint
    ]
    
    # Add shift constraints
    for k, S_k in enumerate(shift_matrices):
        # Effective shift for indexing
        actual_k = k - (n_points - 1)
        if actual_k == 0:
            continue
        n_k = n_points - abs(actual_k)
        constraints.append(cp.trace(S_k @ X) >= 2 * n_points * (1 - t) - n_k)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=2000, eps=1e-4)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return X.value, t.value
    except:
        pass
    
    return None, None

        # Only keep positive eigenvalues
        mask = eigenvals > 1e-8
        eigenvals = eigenvals[mask]
        eigenvecs = eigenvecs[:, mask]
        
        # Form the square root matrix
        V_sqrt = eigenvecs @ np.diag(np.sqrt(eigenvals))
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(R):
            # Generate random vector
            g = np.random.randn(V_sqrt.shape[1])
            
            # Compute signs
            signs = np.sign(V_sqrt.T @ g)
            
            # Convert to {0,1} values
            h = (signs + 1) / 2
            
            # Project to correct sum
            current_sum = np.sum(h)
            if current_sum != TARGET_SUM:
                # Flip the smallest absolute value entries
                idxs = np.argsort(np.abs(h - 0.5))
                if current_sum > TARGET_SUM:
                    # Need to flip from 1 to 0
                    to_flip = int(current_sum - TARGET_SUM)
                    flip_idx = idxs[:to_flip]
                    h[flip_idx] = 0
                else:
                    # Need to flip from 0 to 1
                    to_flip = int(TARGET_SUM - current_sum)
                    flip_idx = idxs[:to_flip]
                    h[flip_idx] = 1
            
            # Evaluate C5
            c5 = evaluate_c5(h)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        return best_h, best_c5
    
    except:
        return None, None

def local_swap_improvement(h0, max_iter=1000):

    h = h0.copy()
    current_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        # Find k that maximizes overlap
        # This is an approximate approach using a precomputed correlation
        corr = correlate(h, 1-h, mode='full')
        max_corr_idx = np.argmax(corr)
        k_max = max_corr_idx - (len(h) - 1)
        
        # Try swapping pairs
        best_gain = 0
        best_pair = None
        
        # Check all pairs of positions
        for i in range(len(h)):
            for j in range(len(h)):
                if i == j:
                    continue
                
                # Compute gain from swapping
                # Swap elements i and j
                h_test = h.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                
                # Use projection with sum requirement
                h_test = project_to_sum_box(h_test, TARGET_SUM)
                
                test_c5 = evaluate_c5(h_test)
                gain = current_c5 - test_c5
                
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
        
        if best_pair and best_gain > 1e-10:
            i, j = best_pair
            h[i], h[j] = h[j], h[i]
            h = project_to_sum_box(h, TARGET_SUM)
            current_c5 = evaluate_c5(h)
        else:
            break
    
    return h, current_c5
