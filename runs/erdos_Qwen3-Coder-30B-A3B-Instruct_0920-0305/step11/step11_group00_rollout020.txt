import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 200
dx = 2.0 / n_points

def compute_corr(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_value(h):
    """Compute C5 value"""
    corr = compute_corr(h)
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def triangular_initial(n):
    """Create triangular initial shape"""
    m = n // 2
    # Linear decreasing from 1 to 0 over first half
    first = np.linspace(1, 0, m, endpoint=False)
    # Mirror for second half
    if n % 2 == 0:
        second = first[::-1]
    else:
        second = first[-2::-1]
    h = np.concatenate([first, second])
    # Adjust to match target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def random_feasible_vector(n, target_sum):
    """Generate random feasible vector"""
    h = np.random.random(n)
    h = project_to_sum_box(h, target_sum)
    return h

def grad_shift(h, s):
    """Compute gradient for a specific shift s"""
    n = len(h)
    base = np.arange(n) + (n-1)
    padded = np.concatenate([np.zeros(n-1), h, np.zeros(n-1)])
    term1 = 1 - padded[base + s]
    term2 = padded[base - s]
    return term1 - term2

            break

            if time.time() - start_time > time_limit:
                break
            # Compute correlation and weights
            c = compute_corr(h)
            max_c = np.max(c)
            expvals = np.exp(alpha * (c - max_c))
            w = expvals / expvals.sum()
            
            # Compute weighted gradient
            grad = np.zeros_like(h)
            for idx_shift, weight in enumerate(w):
                if weight == 0:
                    continue
                s = idx_shift - (n_points - 1)
                grad += weight * grad_shift(h, s)
            
            # Line search for step size
            step_cur = step
            for _ in range(5):
                h_new = h - step_cur * grad
                h_new = project_to_sum_box(h_new, target_sum)
                val_new = c5_value(h_new)
                if val_new < best_val - 1e-12:
                    h = h_new
                    best_val = val_new
                    best_h = h.copy()
                    break
                else:
                    step_cur *= 0.5
    return best_h, best_val

            grad += grad_shift(h, s)

        if len(max_indices) > 0:
            grad /= len(max_indices)

        h_new = h - step * grad

        # Evaluate and accept if improvement
        val_new = c5_value(h_new)
        if val_new < best_val - 1e-12:
            h = h_new
            best_val = val_new
            best_h = h.copy()
        
        # Reduce step size if no improvement
        if i % 20 == 0 and i > 0:
            step *= 0.9
    
    return best_h, best_val
