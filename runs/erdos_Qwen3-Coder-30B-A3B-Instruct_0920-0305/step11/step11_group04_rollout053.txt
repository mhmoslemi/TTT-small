def project_to_sum_box(x, target_sum):
    """Project onto box constraints with sum = target_sum"""
    # Simple approach: water filling
    x = np.clip(x, 0, 1)
    current_sum = np.sum(x)
    if abs(current_sum - target_sum) < 1e-8:
        return x
    diff = target_sum - current_sum
    if diff > 0:
        # Increase values proportionally
        x = np.minimum(1, x + diff / len(x))
    else:
        # Decrease values proportionally
        x = np.maximum(0, x + diff / len(x))
    return x
