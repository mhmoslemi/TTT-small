def build_shift_matrices(n):
    """Build all shift matrices S_k efficiently."""
    shift_matrices = []
    for k in range(-(n-1), n):
        # Create a sparse diagonal matrix for shift k
        offsets = np.arange(n)
        if k >= 0:
            rows = offsets[k:]
            cols = offsets[:-k]
        else:
            rows = offsets[-k:]
            cols = offsets[:k]
        data = np.ones(len(rows))
        S_k = csr_matrix((data, (rows, cols)), shape=(n, n))
        shift_matrices.append(S_k)
    return shift_matrices

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return dx * np.max(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def sdp_rounding(X, num_rounds=R_ROUNDS):
    """Round the SDP solution using random hyperplane rounding."""
    try:
        # Perform eigenvalue decomposition
        eigenvals, eigenvecs = eigh(X)
        # Keep only positive eigenvalues (with small tolerance)
        threshold = 1e-8
        positive_idx = eigenvals > threshold
        V = eigenvecs[:, positive_idx]
        D = np.diag(np.sqrt(eigenvals[positive_idx]))
        L = V @ D
        
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(num_rounds):
            g = np.random.randn(L.shape[1])
            # Project onto random vector
            projections = L @ g
            # Take signs
            h_candidate = (np.sign(projections) + 1) / 2
            
            # Ensure exactly half ones by adjusting
            current_sum = np.sum(h_candidate)
            if current_sum > TARGET_SUM:
                # Too many ones; flip smallest absolute values to zero
                flip_indices = np.argsort(np.abs(projections))[:int(current_sum - TARGET_SUM)]
                h_candidate[flip_indices] = 0
            elif current_sum < TARGET_SUM:
                # Too few ones; flip smallest absolute values to one
                flip_indices = np.argsort(np.abs(projections))[:int(TARGET_SUM - current_sum)]
                h_candidate[flip_indices] = 1
            
            c5 = evaluate_c5(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
        
        return best_h, best_c5
        
    except Exception as e:
        return None, float('inf')

                                delta += h[k] - (1 - h[i] - h[k])
                            else:
                                delta -= (1 - h[i] - h[k])
                    else:
                        delta += h[i] - (1 - h[i] - h[k])
                    
                if delta > best_delta:
                    best_delta = delta
                    best_i = i
        
        for j in range(len(h)):
            if h[j] == 1:
                delta = 0
                for k in range(len(h)):
                    if j != k:
                        if h[k] == 0:
                            if 1 - h[j] - h[k] >= 0:
                                delta += h[k] - (1 - h[j] - h[k])
                            else:
                                delta -= (1 - h[j] - h[k])
                        else:
                            if 1 - h[j] - h[k] >= 0:
                                delta += h[k] - (1 - h[j] - h[k])
                            else:
                                delta -= (1 - h[j] - h[k])
                    else:
                        delta += h[j] - (1 - h[j] - h[k])
                    
                if delta > best_delta:
                    best_delta = delta
                    best_j = j
                    
        if best_delta > 0:
            h[best_i] = 1
            h[best_j] = 0
            current_c5 = evaluate_c5(h)

            break
