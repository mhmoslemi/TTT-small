# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Use water-filling approach
    x = np.array(x)
    # Clip to [0,1]
    x_clipped = np.clip(x, 0, 1)
    current_sum = np.sum(x_clipped)
    
    if abs(current_sum - target_sum) < 1e-10:
        return x_clipped
    
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def smooth_max(h, tau=0.01):
    """Smooth approximation to max using log-sum-exp"""

    g_s = dx * (TARGET_SUM - np.correlate(h, h, mode='full'))

    # Compute log-sum-exp
    g_s = g_s / tau
    max_g = np.max(g_s)
    log_sum_exp = max_g + np.log(np.sum(np.exp(g_s - max_g)))
    return tau * log_sum_exp

def compute_smooth_grad(h, tau=0.01):
    """Compute gradient of smooth max using FFT"""
    corr = compute_corr_fft(h)
    g_s = dx * (TARGET_SUM - np.correlate(h, h, mode='full'))
    shift_ind = np.arange(-len(h)+1, len(h))
    
    # Compute soft probabilities
    g_s = g_s / tau
    max_g = np.max(g_s)
    exp_g = np.exp(g_s - max_g)
    sum_exp = np.sum(exp_g)
    p = exp_g / sum_exp
    
    # Compute gradient: dx * sum_s p_s * ( (1 - h_{i+s}) - h_{i-s} )
    # This is correlation: ifft(fft(p)*fft(1-h)) - ifft(fft(p)*fft(h))
    h_fft = np.fft.fft(h)
    one_minus_h = 1 - h
    one_minus_h_fft = np.fft.fft(one_minus_h)
    p_fft = np.fft.fft(p)
    
    # Gradient contribution from (1 - h_{i+s})
    grad1 = np.fft.ifft(p_fft * one_minus_h_fft).real
    # Gradient contribution from h_{i-s} 
    grad2 = np.fft.ifft(p_fft * h_fft).real
    
    gradient = dx * (grad1 - grad2)
    return gradient

def accelerated_projected_gradient(h0, max_iter=3000, eta=1e-4, tau=0.05, gamma=0.7):

    h = h0.copy()

        # Backtracking line search

                break

def pairwise_exchange_refinement(h0, max_iter=10000):
    """Refine solution with pairwise exchanges (simulated annealing)"""
    h = h0.copy()
    current_c5 = evaluate_c5(h)
    
    temp = 1.0
    temp_decay = 0.9999
    for i in range(max_iter):
        # Choose two random indices
        i1, i2 = np.random.randint(0, n_points, 2)
        
        # Compute allowable range
        delta_max = min(h[i1], 1 - h[i2])
        if delta_max <= 0:
            continue
            
        # Sample small step
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Update
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate new value
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / temp)
            if np.random.rand() < prob:
                h = h_new
                current_c5 = new_c5
                
        # Update temperature
        temp *= temp_decay
        
        if temp < 1e-6:
            break
            
    return h
