import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
n_points = 120
dx = 2.0 / n_points
TARGET_SUM = n_points / 2

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def inv_sigmoid(y):
    return np.log(y / (1.0 - y))

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def spline_interpolate(y_vals, x_vals, eval_points):
    """Simple cubic spline interpolation"""
    from scipy.interpolate import interp1d
    f = interp1d(x_vals, y_vals, kind='cubic', fill_value='extrapolate')
    return f(eval_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Binary search for lambda: find lambda such that sum(clamp(x+lambda, 0, 1)) = target_sum
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def build_h_from_params(z):
    """Build h vector from spline parameters using symmetric cubic spline"""
    # Parameters: 4 knot values on [0,1] due to symmetry
    # Using sigmoid to map to [0,1]
    y_vals = sigmoid(z[:4])  # 4 knot values
    
    # Define knot positions on [0,1]
    x = np.linspace(0, 1, 4)
    x_full = np.linspace(0, 2, n_points)
    
    # Interpolate to full grid
    h_left = spline_interpolate(y_vals, x, np.linspace(0, 1, n_points//2))
    h_right = h_left[::-1]  # symmetric
    
    h = np.concatenate([h_left, h_right[1:]])
    
    # Scale to sum = TARGET_SUM
    s = np.sum(h)
    alpha = TARGET_SUM / s
    h = h * alpha
    
    # Ensure [0,1] bounds
    h = np.clip(h, 0, 1)
    
    # Possibly re-scale slightly to maintain sum
    h = project_to_sum_box(h, TARGET_SUM)
    
    return h

def cma_es_optimization():
    """CMA-ES optimization on the reduced space"""
    d = 4  # number of parameters
    n_pop = 4 + int(3 * np.log(d))  # population size
    n_gen = 300  # max generations
    
    # Initial parameters
    m = np.zeros(d)
    sigma = 2.0
    
    # Covariance matrix
    C = np.eye(d)
    
    # Best solution
    best_z = m.copy()
    best_c5 = float('inf')
    
    # For time control
    start_time = time.time()
    budget_s = 500  # Leave 100s for post-processing
    
    for gen in range(n_gen):
        if time.time() - start_time > budget_s:
            break
            
        # Sample offspring
        offspring = []
        for _ in range(n_pop):
            z = m + sigma * np.random.multivariate_normal(np.zeros(d), C)
            offspring.append(z)
            
        # Evaluate
        fitness = []
        for z in offspring:
            try:
                h = build_h_from_params(z)
                c5 = evaluate_c5(h)
                fitness.append(c5)
                if c5 < best_c5:
                    best_c5 = c5
                    best_z = z.copy()
            except:
                fitness.append(float('inf'))
                
        # Sort and select best
        indices = np.argsort(fitness)
        mu = n_pop // 2
        selected = [offspring[i] for i in indices[:mu]]
        
        # Update distribution parameters  
        if len(selected) > 1:
            # Update mean
            y = np.array(selected)
            m = np.mean(y, axis=0)
            
            # Update covariance matrix using rank-mu update
            z = (y - m) / sigma
            C = (1 - 1/mu) * C + 1/mu * np.dot(z.T, z)
            
            # Update sigma (step size)
            sigma *= np.exp(0.3 * (np.sqrt(np.linalg.det(C)) - 1) / d)
            
    return build_h_from_params(best_z), best_c5

def gradient_descent_refinement(h):
    """Refine solution using gradient descent on the full space"""
    max_iter = 1200
    eta = 0.08
    start_time = time.time()
    budget_s = 50  # 50s for refinement
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute correlation
        corr = compute_corr_fft(h)
        k_max = np.argmax(corr)
        
        # Compute gradient w.r.t. each element
        g = np.zeros_like(h)
        for k in range(len(corr)):
            if abs(corr[k] - np.max(corr)) < 1e-12:
                for i in range(len(h)):
                    inc = 0.0
                    if i + k < len(h):
                        inc += 1 - h[i + k]
                    if i - k >= 0:
                        inc -= h[i - k]
                    g[i] += inc
                    
        # Normalize gradient
        if np.linalg.norm(g) > 0:
            g = g / np.linalg.norm(g)
            
        # Perform update
        h_new = h - eta * g
        
        # Project
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Check improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new
        else:
            eta *= 0.5
            if eta < 1e-6:
                break
                
        # Plateau escape
        if i % 200 == 0:
            i1, i2 = np.random.randint(0, len(h), 2)
            delta = np.random.uniform(-0.015, 0.015)
            h_copy = h.copy()
            h_copy[i1] += delta
            h_copy[i2] -= delta
            h_copy = np.clip(h_copy, 0, 1)
            h_copy = project_to_sum_box(h_copy, TARGET_SUM)
            
            c5_copy = evaluate_c5(h_copy)
            if c5_copy < evaluate_c5(h):
                h = h_copy
                
    return h
