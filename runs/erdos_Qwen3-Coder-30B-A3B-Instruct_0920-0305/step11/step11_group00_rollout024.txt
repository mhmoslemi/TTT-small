import numpy as np

def proj_sum_box(v, target_sum):
    """Project vector v onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(v + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1

    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m

    result = np.clip(v + (a + b)/2, 0, 1)
    return result

def corr_full(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def grad_shift(h, k):
    """Compute gradient of c_k(h) = sum_i h_i*(1-h_{i+k})"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] = 1 - h[:-k]
        g[:k] = 1 - h[:k] if k > 0 else 1 - h
    else:
        g[:k] = 1 - h[-k:]
        g[k:] = 1 - h[:k] if k < 0 else 1 - h
    return g

def max_overlap(h):
    """Computes C5 as max of overlap terms"""
    corr = corr_full(h)
    return np.max(corr)

def shift_mask(k, n):
    """Create sparse matrix mask for shift k"""
    if k >= 0:
        rows = np.arange(n - k)
        cols = rows + k
    else:
        rows = np.arange(-k, n)
        cols = rows - k
    data = np.ones(len(rows))
    M = csr_matrix((data, (rows, cols)), shape=(n, n))
    return M

def build_sdp(n):
    """Build SDP problem for estimating lower bound on C5"""
    h = cp.Variable(n)
    X = cp.Variable((n, n), PSD=True)
    tau = cp.Variable()

    constraints = [
        cp.diag(X) == h,
        h >= 0, h <= 1,
        cp.sum(h) == n/2,
        tau >= 0
    ]
    
    for k in range(-(n-1), n):
        M = shift_mask(k, n)
        constraints.append(cp.trace(M @ X) >= (n/2)*(1 - tau))
    
    prob = cp.Problem(cp.Minimize(tau), constraints)
    return prob, h, X, tau

def solve_sdp(prob):
    """Solve SDP and return solution"""
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        if prob.status == cp.OPTIMAL:
            return True, prob.value, prob.variables()[0].value, prob.variables()[1].value
        else:
            return False, np.inf, None, None
    except:
        return False, np.inf, None, None

def round_from_sdp(h_sdp, X_sdp):
    """Three rounding strategies for the SDP solution"""
    candidates = []
    
    # Strategy 1: Diagonal rounding
    h1 = h_sdp.copy()
    candidates.append((h1, max_overlap(h1)))
    
    # Strategy 2: Leading eigenvector
    try:
        eigenvals, eigenvecs = np.linalg.eigh(X_sdp)
        idx = np.argmax(eigenvals)
        v = eigenvecs[:, idx]
        h2 = np.sqrt(np.abs(eigenvals[idx])) * v
        h2 = proj_sum_box(h2, len(h2)/2)
        candidates.append((h2, max_overlap(h2)))
    except:
        pass
    
    # Strategy 3: Gaussian randomisation
    try:
        for _ in range(5):
            z = np.random.multivariate_normal(np.zeros(len(h_sdp)), X_sdp)
            h3 = np.abs(z)
            h3 = proj_sum_box(h3, len(h3)/2)
            candidates.append((h3, max_overlap(h3)))
    except:
        pass
    
    best_h, best_c5 = min(candidates, key=lambda x: x[1])
    return best_h, best_c5

    # Time tracking
    start_time = time.time()
    
    h = h_start.copy()

    step = 0.07
    
    C5_old = max_overlap(h)
    max_C5 = C5_old
    best_h = h.copy()
    best_C5 = C5_old
    
    iterations = 0
    no_improve_steps = 0
    
    while time.time() - start_time < budget_remaining - 5:
        iterations += 1
        corr = corr_full(h) * dx  # raw correlation values
        C5_max = np.max(corr)
        
        # indices of shifts that achieve the max
        K_max = np.where(np.abs(corr - C5_max) < 1e-12)[0]
        
        # Compute average gradient
        g = np.zeros(n)
        for k in K_max:
            g += grad_shift(h, k)
        g /= len(K_max) if len(K_max) > 0 else 1
        
        # Nesterov-style extrapolation
        y = h + 0.9 * (h - h_prev)
        h_new = y - step * g
        h_new = proj_sum_box(h_new, n/2)
        
        C5_new = max_overlap(h_new)
        if C5_new < C5_old:
            step = min(step * 1.08, 0.2)
            h_prev = h.copy()
            h = h_new.copy()
            C5_old = C5_new
            if C5_new < best_C5:
                best_h = h.copy()
                best_C5 = C5_new
            no_improve_steps = 0
        else:
            step *= 0.5
            no_improve_steps += 1
            
        # Swap moves to escape plateaus
        if iterations % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-0.02, 0.02)
            if 0 <= i < n and 0 <= j < n:
                h_old = h.copy()
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0, 1)
                h = proj_sum_box(h, n/2)
                C5_new = max_overlap(h)
                if C5_new < C5_old:
                    C5_old = C5_new
                    best_h = h.copy()
                    best_C5 = C5_new
                else:
                    h = h_old
            
        if no_improve_steps > 200:
            break
    
    return best_h, best_C5
