def compute_correlation(h):
    """Compute full correlation using FFT"""
    n = len(h)
    h_rev = h[::-1]
    b = 1.0 - h
    L = 2 * n - 1
    corr = ifft(fft(h_rev, L) * fft(b, L)).real
    return corr 

def evaluate_c5(h):
    """Evaluate C5 from a vector"""
    corr = compute_correlation(h)
    return np.max(corr) * 2.0 / len(h)

def solve_linearized_step(y):
    """
    Solve the linearized LP:
    maximize t subject to:
        sum_{i} x_i * y_{i+s} >= t for all s,
        sum_i x_i = target_sum,
        0 <= x_i <= 1 for all i
    """
    n = len(y)
    target_sum = n / 2
    t = cp.Variable()
    x = cp.Variable(n)
    
    # Constraints
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    
    # Add linear constraints for each shift s
    # Correlation with anti-corr (for computing max-min)
    for s in range(-n + 1, n):
        # Compute linear term based on shift s
        # This simulates: sum_i x_i * y_{i+s}
        # If i+s is outside bounds, then the term is 0
        expr = 0
        for i in range(n):
            if 0 <= i + s < n:
                expr += x[i] * y[i + s]
            else:
                continue
        constraints.append(expr >= t)
        
    # Objective
    objective = cp.Maximize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with ECOS first, fallback to CLP or GLPK if needed
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except:
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            pass
    
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        return x.value, t.value
    else:
        return None, None
