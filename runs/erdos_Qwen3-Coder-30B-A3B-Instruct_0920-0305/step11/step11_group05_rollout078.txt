import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

# Global variables
TARGET_SUM = None
n_points = 120
dx = 2.0 / n_points
tau_schedule = [0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.005, 0.001, 0.0005, 0.0001]
gamma = 0.6
eta = 1e-4
max_iter_apg = 3000
max_iter_exchange = 5000

def compute_corr_fft(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def c5_from_corr(corr, n):
    """Convert correlation result to C5 value"""
    return dx * np.max(corr)

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_corr_fft(h)
    return c5_from_corr(corr, n_points)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    while f(a) * f(b) > 0:
        a -= 0.1
        b += 0.1
    
    # Binary search
    tol = 1e-10
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def evaluate_phi_tau(h, tau):
    """Evaluate smooth surrogate Phi_tau(h) = tau * log(sum(exp(g_s(h)/tau)))"""
    corr = compute_corr_fft(h)
    corr = corr * dx  # scale by dx to match the original definition
    
    # Compute soft max with numerically stable exponentials
    max_g = np.max(corr)
    if max_g == -np.inf:
        return np.inf
    exp_corr = np.exp((corr - max_g) / tau)
    sum_exp = np.sum(exp_corr)
    
    if sum_exp == 0:
        return np.inf
                        
    return tau * np.log(sum_exp) + max_g

def compute_gradient(h, tau):
    """Compute gradient of Phi_tau using FFT"""
    # Compute the full correlation
    corr = compute_corr_fft(h)
    corr = corr * dx
    
    # Compute soft max probabilities
    max_g = np.max(corr)
    if max_g == -np.inf:
        return np.zeros_like(h)
    exp_corr = np.exp((corr - max_g) / tau)
    sum_exp = np.sum(exp_corr)
    
    if sum_exp == 0:
        return np.zeros_like(h)
        
    p = exp_corr / sum_exp
    
    # Compute the weighted gradient components via FFT
    # Gradient is dx * [ sum_s p_s * (1-h_{i+s}) - sum_s p_s * h_{i-s} ]
    # We compute:
    # A = ifft(fft(p*) * fft(1-h)), B = ifft(fft(p*) * fft(h))
    # G = dx * (A - B)
    
    # Using the convolution theorem:
    # Given the time-domain vectors:
    # v = h, w = 1-h, u = p
    A = np.fft.ifft(np.fft.fft(p.conj()) * np.fft.fft(1 - h)).real
    B = np.fft.ifft(np.fft.fft(p.conj()) * np.fft.fft(h)).real
    G = dx * (A - B)
    
    return G

def backtracking_line_search(h, gradient, tau, alpha_max=1.0):
    """Backtracking line search for step size"""
    alpha = alpha_max
    phi_old = evaluate_phi_tau(h, tau)
    
    while True:
        h_new = project_to_sum_box(h - alpha * gradient, TARGET_SUM)
        phi_new = evaluate_phi_tau(h_new, tau)
        
        # Armijo condition
        if phi_new <= phi_old - eta * alpha * np.dot(gradient, gradient):
            return alpha, h_new
        alpha *= 0.5
        if alpha < 1e-10:
            return alpha, h_new

def accelerated_projected_gradient(h_in, tau, start_time, budget_s):
    """APG algorithm for minimizing Phi_tau with backtracking line search"""
    
    h = h_in.copy()
    y = h.copy()
    t = 1.0
    
    for k in range(max_iter_apg):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        grad = compute_gradient(y, tau)
        alpha, h_new = backtracking_line_search(y, grad, tau)
        
        h = h_new
        t_old = t
        t = (1 + np.sqrt(1 + 4 * t_old * t_old)) / 2
        y = h + ((t_old - 1) / t) * (h - h_new)
        
    return h

def pairwise_exchange_refinement(h, start_time, budget_s):
    """Simulated annealing style pairwise exchanges for fine-tuning"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Cooling schedule
    temp_start = 0.01
    temp_end = 1e-6
    t = temp_start
    cooling_rate = (temp_end / temp_start) ** (1.0 / max_iter_exchange)
    
    for _ in range(max_iter_exchange):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Pick two indices
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute admissible range
        delta_max = min(current_h[i], 1 - current_h[j])
        if delta_max <= 0:
            continue
            
        # Sample shift
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Create new vector
        h_new = current_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        
        # Project to feasible set
        h_new = project_to_sum_box(h_new, TARGET_SUM)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = h_new.copy()
            current_c5 = new_c5
        else:
            prob = np.exp(-(new_c5 - current_c5) / t)
            if np.random.rand() < prob:
                current_h = h_new.copy()
                current_c5 = new_c5
                
        # Update temperature
        t *= cooling_rate
        
    return current_h, current_c5

    # Initialize
    TARGET_SUM = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial candidates
    initial_vectors = generate_initial_vectors()
    
    # Keep track of best solution
    best_final_h = None
    best_final_c5 = float('inf')
    
    # Try each initial vector
    for i, h0 in enumerate(initial_vectors):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Run continuation on temperature
        current_h = h0.copy()
        current_c5 = evaluate_c5(current_h)
        
        for tau in tau_schedule:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Accelerated projected gradient step
            current_h = accelerated_projected_gradient(current_h, tau, start_time, budget_s)
            
            # Project to satisfy constraints (possible tiny numerical drift)
            current_h = project_to_sum_box(current_h, TARGET_SUM)
            
            # Check current c5
            tmp_c5 = evaluate_c5(current_h)
            if tmp_c5 < current_c5:
                current_c5 = tmp_c5
                
        # Final refinement
        refined_h, refined_c5 = pairwise_exchange_refinement(current_h, start_time, budget_s)
        
        if refined_c5 < best_final_c5:
            best_final_c5 = refined_c5
            best_final_h = refined_h.copy()
    
    if best_final_h is None:
        best_final_h = initial_vectors[0].copy()
        
    return (best_final_h.tolist(), float(best_final_c5), n_points)
