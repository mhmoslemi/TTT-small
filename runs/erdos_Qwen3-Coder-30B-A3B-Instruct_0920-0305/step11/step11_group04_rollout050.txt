import numpy as np
from scipy.signal import correlate
import time

# Global variables
n_points = 128
dx = 2.0 / n_points
TARGET_SUM = n_points // 2

def autocorr(h):
    """Compute full correlation of h with itself, scaled by dx"""
    # Using correlate with full mode to get all shifts
    # The result is of length 2*n - 1. The middle index corresponds to shift 0
    corr = correlate(h, h, mode='full') * dx
    return corr

def compute_C5(h):
    """Compute the C5 value from a step function h"""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling method
    x = np.asarray(x)
    # Binary search for lambda
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    a, b = -1.0, 1.0
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < 1e-10:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def thue_morse_binary(n):
    """Generate Thue-Morse sequence as binary array of length n"""
    if n == 0:
        return np.array([])
    # Generate Thue-Morse in binary format using bitwise operations
    t = np.zeros(n, dtype=int)
    t[1::2] = 1  # initial pattern for AC
    for i in range(2, n):
        if i & (i - 1) == 0:  # if i is a power of 2
            # Mirror the previous pattern
            start_idx = i
            end_idx = min(2*i, n)
            ref_idx = 0
            for j in range(start_idx, end_idx):
                if t[ref_idx] == 0:
                    t[j] = 1
                else:
                    t[j] = 0
                ref_idx += 1
    return t.astype(float)
