import numpy as np
import time
from scipy.fft import fft, ifft
import cvxpy as cp

def project_onto_sum_box(v, S):
    """Project v onto [0,1]^n with sum = S using water filling algorithm."""
    n = len(v)
    sorted_idx = np.argsort(v)[::-1]
    v_sorted = v[sorted_idx]
    
    # Compute cumulative sums
    cumsum = np.cumsum(v_sorted)
    
    # Find threshold index
    # We want sum of clipped values to be S
    # If v_sorted[j] > (S - cumsum[j-1]) / (n-j) then we clip v_sorted[j] to (S - cumsum[j-1]) / (n-j)
    # If not, we can take its full value
    # So adjust such that sum is exactly S
    # Sort by descending order of (v_sorted[j] - (S - cumsum[j-1]) / (n-j))
    # But let's just compute it directly
    
    # Start with all at 1 and reduce where necessary
    # But it's more reliable to do the binary search separately on the actual clip value
    
    # Actually, water level algorithm:
    # Start from the highest, and consider how much of a value we can take
    # cumulative sum approach
    threshold_idx = 0
    for i in range(n):
        if cumsum[i] <= S:
            threshold_idx = i + 1
        else:
            break
            
    # Now we have a spanning set of values from 0 to threshold_idx-1 that we keep as is,
    # and need to reduce the rest to make sum = S
    # Those from threshold_idx onwards get reduced
    # But to compute final projection, we find value where that total sum matches S
    if threshold_idx == n and cumsum[n-1] <= S:
        # All values can be kept as they are
        # but we need to adjust the final value so that sum = S
        # (This shouldn't happen since sum of v is less than S, but let's be careful)
        scaled = v / cumsum[n-1] * S
        return np.clip(scaled, 0, 1)
    # Compute the amount we can cut from the threshold_idx:th value down to 0
    # if sum of first threshold_idx is S, put idx = 0
    # otherwise, value at threshold_index should be adjusted so:
    # sum_0^threshold_idx-1 + (n - threshold_idx) * new_val = S
    # sum_0^threshold_idx-1 + (n - threshold_idx) * new_val = S
    # new_val = (S - sum_0^threshold_idx-1) / (n - threshold_idx) 
    

    if threshold_idx == 0:  # All entries need to be scaled
        scaled = v * S / cumsum[n-1]
        return np.clip(scaled, 0, 1)
    else:
        # Threshold is at threshold_idx - 1, so size 'threshold_idx'
        sum_up_to = cumsum[threshold_idx - 1] if threshold_idx > 0 else 0
        remainder = S - sum_up_to
        if n - threshold_idx == 0: 
            # All values must be exactly at threshold, no remaining values
            # If sum_up_to > S then all should be 0? No, we are taking their positions. 
            # Let's recompute carefully:
            # If sum of first i values is > S, the threshold is at i-1. The above logic 
            # should work. If i values sum to > S that indicates issue, but likely safe without.
            # We'll use simple approach:
            scaled = np.zeros_like(v_sorted)
            scaled[:threshold_idx] = 1.0 # but fit sum
            # Find the right hand box sum  
            actual_sum = np.sum(scaled[:threshold_idx])
            if actual_sum > 0:
                scaled[:threshold_idx] *= S / actual_sum  
            else:
                # All scaled to 0 TODO handle edge case properly  
                scaled = np.zeros_like(v_sorted)
            return np.clip(scaled, 0, 1)  
        else:
            scalar = remainder / (n - threshold_idx)
            new_val = np.clip(scalar, 0, 1)
            # fill values from 0 to threshold_idx-1 with 1
            # and from threshold_idx to end with new_val
            result = np.zeros_like(v)
            result[sorted_idx[:threshold_idx]] = 1
            result[sorted_idx[threshold_idx:]] = new_val
            return result

def corr_fft(h):
    """Fast correlation using FFT. Returns full correlation."""
    n = len(h)
    d = 1 - h
    r = ifft(fft(h, 2*n-1) * np.conj(fft(d, 2*n-1))).real
    return r[n-1:n-1+2*n-1]

def compute_c5_from_corr(corr, dx):
    """Calculate C5 value from correlation."""
    return dx * np.max(corr)

def compute_c5_direct(h, dx):
    """Compute C5 direct from array using FFT."""
    corr = corr_fft(h)
    return compute_c5_from_corr(corr, dx)

def solve_lp_step(y, S, dx, n_points):
    """Solve an LP with current y frozen, returning new x and achieved lower bound."""
    
    # LP vars for x
    x = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints matrix construction here due to shifts
    # Build a constraint matrix per shift
    # For each k from -n+1 to n-1, store the linear form of sum_i x[i] * y[i+k]
    
    constraints = []
    constraints.append(cp.sum(x) == S)
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Now build the constraints for correlation forms using stretched shifts
    # We need sum_i x_i * y_{i+s} >= t for each s
    
    # For each shift s:
    for s in range(-n_points+1, n_points):
        # build vector for the shifted indices in s 
        lhs = 0
        # loop over all i in 0 .. n_points-1
        for i in range(n_points):
            j = i + s
            if 0 <= j < n_points:
                lhs += x[i] * y[j]
        # Add the constraint lhs >= t
        constraints.append(lhs >= t)
        
    # The objective is to maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000, feastol=1e-8)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return x.value, t.value
        else:
            # fallback to previous method for local solution?
            return x.value, -np.inf
    except:
        return x.value, -np.inf

def alternating_linearization_run(h_init, dx, S, max_iter=30, tol=1e-8):
    """Run alternating LP optimization using linearization."""
    h = h_init.copy()
    t_best = -np.inf
    best_h = h.copy()
    
    for k in range(max_iter):
        # Solve LP with current h frozen
        try:
            x, t = solve_lp_step(h, S, dx, len(h))
        except:
            x, t = h, -np.inf
            
        if t > t_best:
            t_best = t
            best_h = x.copy()
            
        if np.isnan(t) or np.isinf(t):
            break
            
        # Update iterate
        h = x.copy()
        
        # Convergence check
        if k > 0 and abs(t - t_prev) < tol:
            break
        t_prev = t
        
    return best_h, t_best

def run_initialization_seeds(source_h, n_points, S, dx, num_seeds=8):
    """Generate several starting points."""
    seeds = []
    # 1. Use the provided seed
    if source_h is not None and len(source_h) == n_points:
        initial = np.array(source_h)
        initial = project_onto_sum_box(initial, S)
        seeds.append(initial.copy())
    else:
        # Original seed is prepended if not right size
        initial = np.ones(n_points) * 0.5
        initial = project_onto_sum_box(initial, S)
        seeds.append(initial.copy())
    
    # 2. Random seeds (not sure of imports handled so small random vector).
    for i in range(num_seeds - 1):
        # sample from uniform distribution, then project to [0,1] with sum=S
        random_h = np.random.uniform(0, 1, n_points)
        random_h = project_onto_sum_box(random_h, S)
        seeds.append(random_h.copy())
    
    return seeds

def pairwise_exchange_refinement(initial_h, dx, num_trials=5000):
    """Perform a quick refinement with pairwise exchanges."""
    h = initial_h.copy()
    n = len(h)
    c5_old = compute_c5_direct(h, dx)
    
    # Precompute corr for efficiency
    corr = corr_fft(h)
    
    # a fast update scheme - avoid recomputing everything each time
    # but it's complex to do incrementally across all shifts with a full update
    # Simple approach: for each trial, pick two indices and swap small amounts
    
    # Check constant time update for a few drastic cases
    for i in range(num_trials):
        # quiet temperature schedule gives a better chance of escape
        T = 0.1 * np.exp(-i / 1000.0)  # exponentially decaying temp
        
        # pick two random points
        idx1 = np.random.randint(0, n)
        idx2 = np.random.randint(0, n)
        if idx1 == idx2:
            continue
            
        # random delta
        delta = np.random.uniform(-0.01, 0.01)
        
        # Clip delta to stay in [0,1] range for both h[idx1] and h[idx2]
        new_val1 = h[idx1] + delta
        new_val2 = h[idx2] - delta
        
        if new_val1 < 0 or new_val1 > 1 or new_val2 < 0 or new_val2 > 1:
            continue
            
        # Evaluate impact on loss after clipping
        # Full re-computation because pairwise is deterministic.
        
        h_new = h.copy()
        h_new[idx1] = new_val1
        h_new[idx2] = new_val2
        
        h_new = project_onto_sum_box(h_new, len(h_new)/2)
        c5_new = compute_c5_direct(h_new, dx)
        
        if c5_new < c5_old:
            h = h_new
            c5_old = c5_new
        elif np.random.rand() < np.exp(-(c5_new - c5_old) / (T + 1e-10)):
            h = h_new
            c5_old = c5_new
            
    return h
