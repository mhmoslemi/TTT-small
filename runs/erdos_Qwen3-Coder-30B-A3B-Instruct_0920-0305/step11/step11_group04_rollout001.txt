import numpy as np
from scipy.signal import correlate
import time

# Global variables
TARGET_SUM = None
n_points = 128
dx = 2.0 / n_points

def thue_morse_binary(n):
    """Generate Thue-Morse sequence of 0s and 1s of length n"""
    if n <= 0:
        return np.array([])
    # For simplicity, use a Thue-Morse-like pattern by building iteratively
    # But build based on lower bits to have exact half the values as 1
    t = np.array([0, 1])  # Start with TM sequence seed
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    t = t[:n]
    # Adjust if sum is not exactly half
    current_sum = np.sum(t)
    if current_sum > n // 2:
        idxs = np.where(t == 1)[0]
        for i in range(int(current_sum - n // 2)):
            t[idxs[i]] = 0
    elif current_sum < n // 2:
        idxs = np.where(t == 0)[0]
        for i in range(int(n // 2 - current_sum)):
            t[idxs[i]] = 1
    return t.astype(float)

def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    # Water-filling algorithm
    sorted_x = np.sort(x)[::-1]
    cumsum = np.cumsum(sorted_x)
    # Find threshold index
    idx = -1
    for i in range(len(sorted_x)):
        if sorted_x[i] * (i + 1) - cumsum[i] >= target_sum:
            idx = i
            break
    if idx == -1:
        # All values inside the simplex
        lambda_val = (cumsum[-1] - target_sum) / len(sorted_x)
        result = np.clip(x - lambda_val, 0, 1)
    else:
        lambda_val = (cumsum[idx-1] - target_sum) / idx if idx > 0 else (cumsum[-1] - target_sum) / len(sorted_x)
        result = np.clip(x - lambda_val, 0, 1)
    return result

def autocorr(h):
    """Compute full correlation of h with itself."""
    # Use correlate with mode-full to get full correlation
    corr = correlate(h, h, mode='full')
    # The full correlation includes 2*n-1 terms
    # We want 2*n-1 values for shifts from -(n-1) to n-1.
    # The central index (n-1) corresponds to shift 0.
    return corr * dx

def compute_C5(h):
    """Compute C5 value for the given h"""
    corr = autocorr(h)
    return 1.0 - np.min(corr)

def greedy_swap_local_search(h, dx, budget_s, start_time):
    """Apply greedy binary swap search on worst correlation shift"""
    if np.sum(h) != TARGET_SUM:
        h = project_to_sum_box(h, TARGET_SUM)
    
    corr = autocorr(h)
    worst_shift = np.argmin(corr)
    min_corr = np.min(corr)
    shift = worst_shift - (n_points - 1)
    best_h = h.copy()
    best_corr = min_corr
    best_C5 = 1.0 - best_corr
    
    # Try few passes
    for _ in range(200):
        if time.time() - start_time > budget_s * 0.98:
            break
            
        gain_best = 0
        best_pair = (0, 0)
        # Find best swapping pair
        for i in range(n_points):
            if h[i] == 0:
                # Calculate potential gain
                gain = 0
                if shift > 0 and i + shift < n_points:
                    gain += h[i + shift]
                if shift < 0 and i - shift >= 0:
                    gain += h[i - shift]
                # Determine loss if we swap h[i]=0 to 1
                loss = 0
                if shift > 0 and i + shift < n_points:
                    loss += h[i + shift]
                if shift < 0 and i - shift >= 0:
                    loss += h[i - shift]

                if gain - loss > gain_best:
                    gain_best = gain - loss
                    best_pair = (i, i)  # Hold current position if we swap

        # Do one step
        i_zero, i_one = best_pair
        if gain_best > 0 and i_zero >= 0:
            # We are swapping i_zero from 0 to 1, and i_one from 1 to 0
            # This requires finding another 1 to swap with
            # Simple approach: go find any 1 to swap with, adjust for the shift
            last_one_idx = -1
            for j in range(n_points):
                if h[j] == 1 and (j == i_one or abs(j - i_one) != abs(shift)):
                    last_one_idx = j
                    break
            if last_one_idx >= 0:
                # Perform the swap
                h[i_zero], h[last_one_idx] = 1., 0.
                # Update corr manually (incremental change)
                corr = autocorr(h)
                new_min_corr = np.min(corr)
                new_C5 = 1.0 - new_min_corr
                if new_C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = new_C5
                    best_corr = new_min_corr
                    worst_shift = np.argmin(corr)
                    shift = worst_shift - (n_points - 1)
                else:
                    h[i_zero], h[last_one_idx] = 0., 1.
            else:
                break
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    global TARGET_SUM, n_points, dx
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    TARGET_SUM = n_points // 2
    best_C5 = float('inf')
    best_h = None

    def compute_C5(h):
        corr = autocorr(h)
        return 1.0 - np.min(corr)
    
    # Define initial candidates
    initial_seeds = []

    # 1) Thue-Morse binary (n = 128)
    tm = thue_morse_binary(n_points)
    initial_seeds.append(project_to_sum_box(tm, TARGET_SUM))

    # 2) Random uniform vector
    r = np.random.rand(n_points)
    initial_seeds.append(project_to_sum_box(r, TARGET_SUM))

    # 3) Provided initial_h_values
    try:
        init_vals = np.array(initial_h_values)
        if len(init_vals) != n_points:
            if len(init_vals) > n_points:
                init_vals = init_vals[:n_points]
            else:
                init_vals = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(init_vals)), init_vals)
        initial_seeds.append(project_to_sum_box(init_vals, TARGET_SUM))
    except:
        pass  # Simply skip if unavailable

    for h0 in initial_seeds:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Gradient ascent with smooth minimum
        h = h0.copy()
        τ = 0.2  # Temperature for smooth min
        α = 0.15  # Step size

        for iter_count in range(3000):
            if time.time() - start_time > budget_s * 0.99:
                break
            
            # Compute correlation and C5
            corr = autocorr(h)
            C5 = 1.0 - np.min(corr)
            if C5 < best_C5:
                best_C5, best_h = C5, h.copy()

            # Compute gradient of smooth min
            w = np.exp(-corr / τ)
            w /= np.sum(w)
            g = np.zeros(n_points)
            # Refine gradient loop to only include valid shifts
            for shift in range(-(n_points - 1), n_points):
                wk = w[shift + (n_points - 1)]
                if wk == 0:
                    continue
                # Add contributions to gradient
                if shift < 0:
                    # shift is negative
                    lo = 0
                    hi = n_points + shift
                    g[lo:hi] += wk * h[lo-shift:hi-shift]
                elif shift > 0:
                    # shift is positive
                    lo = shift
                    hi = n_points
                    g[lo:hi] += wk * h[lo-shift:hi-shift]
                else:  # shift == 0
                    # for shift 0, we get 2*wk*h[i]
                    g += 2 * wk * h
                    
            g *= dx
            
            # Update with projected gradient ascent
            h_candidate = project_to_sum_box(h + α * g, TARGET_SUM)
            
            # Evaluate new C5
            corr_new = autocorr(h_candidate)
            C5_new = 1.0 - np.min(corr_new)
            
            # Accept or reject steps
            if C5_new < C5:
                h = h_candidate
                α = min(α * 1.02, 0.5)  # Increase step size
            else:
                α *= 0.5  # Reduce step size

            τ *= 0.99  # Cool down temperature

    # Binary polishing step
    if best_h is not None:
        # Make binary version of the best solution
        h_bin = np.zeros(n_points)
        idx = np.argsort(-best_h)[:TARGET_SUM]
        h_bin[idx] = 1.0
        h_bin = project_to_sum_box(h_bin, TARGET_SUM)
        h_bin = greedy_swap_local_search(h_bin, dx, budget_s, start_time)
        final_C5 = compute_C5(h_bin)
        if final_C5 < best_C5:
            best_C5 = final_C5
            best_h = h_bin.copy()
    
    return (best_h.tolist(), float(best_C5), n_points)
