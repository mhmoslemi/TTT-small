def project_to_sum_box(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum using water-filling"""
    # Sort in descending order
    sorted_indices = np.argsort(-x)
    sorted_x = x[sorted_indices]
    
    # Water-filling: find threshold lambda
    cumsum = np.cumsum(sorted_x)
    n = len(sorted_x)
    
    # Binary search for lambda
    a, b = -1.0, 1.0
    tol = 1e-12

        projected = np.clip(sorted_x + lam, 0, 1)
        return np.sum(projected)
    
    for _ in range(50):
        m = (a + b) / 2
        fm = sum_with_lambda(m) - target_sum
        if abs(fm) < tol:
            break
        if fm * sum_with_lambda(a) < 0:
            b = m
        else:
            a = m
            
    lambda_val = (a + b) / 2
    
    # Apply projection
    result = np.clip(x + lambda_val, 0, 1)
    return result

def autocorr(h):
    """Compute full correlation of h with itself"""
    # Using correlate with full mode for linear correlation
    corr = correlate(h, h, mode='full')
    # Scale by dx after computing
    return corr * dx

def compute_C5(h):
    """Compute C5 value using full correlation of h with (1-h)"""
    # Compute correlation between h and (1 - h)
    corr = correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def thue_morse_binary(n):
    """Generate Thue-Morse binary pattern of length n"""
    # Generate Thue-Morse sequence up to length n (use n=128 for power of 2)
    tm = [0]
    while len(tm) < n:
        # Extend by complementing current sequence
        tm += [1 - x for x in tm]
    return np.array(tm[:n], dtype=float)

    max_iter = 10000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Look for pairs to swap that improve worst shift correlation
        best_net = -1e-10  # Negative to accept any improvement
        best_pair = None
        
        for i in range(n_points):
            if h[i] == 0:  # Look for a zero to convert to one
                gain = 0.0
                
                # Check if i+worst_shift has a one
                j1 = i + worst_shift
                if 0 <= j1 < n_points:
                    gain += h[j1]
                    
                j2 = i - worst_shift
                if 0 <= j2 < n_points:
                    gain += h[j2]
                
                # Check if h[i] = 0 and we are swapping one 
                if gain > 0:
                    # Find lost contribution from making 1->0 to prevent invalidating sum
                    loss = 0.0
                    for j in range(n_points):
                        if h[j] == 1:  # We're about to zero this j
                            # Check impact of this swap on worst shift
                            k1 = j + worst_shift
                            k2 = j - worst_shift
                            if 0 <= k1 < n_points:
                                loss += h[k1]
                            if 0 <= k2 < n_points:
                                loss += h[k2]
                            break  # Just one candidate
                
                    net = gain - loss
                    if net > best_net:
                        best_net = net
                        best_pair = (i, j)
        
        if best_net > 0 and best_pair:
            i, j = best_pair
            h[i] = 1.0
            h[j] = 0.0
            iter_count += 1
        else:
            break
    
    return h
