import numpy as np
import time
from scipy.signal import correlate

def project_to_box_sum(x, target_sum):
    """Project vector x onto box constraints [0,1] with sum = target_sum"""
    def f(lam):
        projected = np.clip(x + lam, 0, 1)
        return np.sum(projected) - target_sum

    # Binary search with improved bounds
    a, b = -1.0, 1.0
    # Tighter initial bounds
    lam_low = -np.max(x) - 1
    lam_high = 1 - np.min(x) + 1
    a = max(a, lam_low)
    b = min(b, lam_high)
    
    # Ensure we have a valid bracket
    while f(a) * f(b) > 0:
        if f(a) > 0:
            a -= 0.1
        else:
            b += 0.1
        if abs(a - b) < 1e-12:
            break
    
    tol = 1e-12
    for _ in range(50):
        m = (a + b) / 2
        fm = f(m)
        if abs(fm) < tol:
            break
        if fm * f(a) < 0:
            b = m
        else:
            a = m
    
    result = np.clip(x + (a + b)/2, 0, 1)
    return result

def compute_correlation(h):
    """Compute correlation using FFT"""
    h_rev = h[::-1]
    b = 1 - h
    L = 2 * len(h) - 1
    corr = np.fft.ifft(np.fft.fft(h_rev, L) * np.fft.fft(b, L)).real
    return corr

def evaluate_c5(h):
    """Fast evaluation of C5 using FFT"""
    corr = compute_correlation(h)
    return np.max(corr)

def gradient_shift(h, k):
    """Compute gradient of shift k"""
    n = len(h)
    # Create padded array
    pad = np.zeros(2*n - 1)
    pad[n-1:n-1+n] = h
    base = np.arange(n) + (n-1)
    
    grad_k = (1 - pad[base + k]) - pad[base - k]
    return grad_k

def log_sum_exp_surrogate(h, alpha):
    """Compute smooth surrogate using log-sum-exp"""
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    # Safe computation with max subtraction
    exp_arg = alpha * (corr - max_corr)
    exp_arg = np.clip(exp_arg, -700, 700)  # Prevent overflow
    exp_values = np.exp(exp_arg)
    log_sum = max_corr + np.log(np.sum(exp_values)) / alpha
    return log_sum

def compute_gradient_surrogate(h, alpha):
    """Compute gradient of the log-sum-exp surrogate"""
    corr = compute_correlation(h)
    max_corr = np.max(corr)
    
    # Compute weights with numerical stability
    exp_arg = alpha * (corr - max_corr)
    exp_arg = np.clip(exp_arg, -700, 700)
    exp_values = np.exp(exp_arg)
    weights = exp_values / np.sum(exp_values)
    
    # Compute weighted sum of gradients
    grad = np.zeros_like(h)
    n = len(h)
    for k in range(-n+1, n):
        grad_k = gradient_shift(h, k)
        grad += weights[k + n - 1] * grad_k
    
    return grad

            improvement_count = 0

        alpha = min(alpha_max, alpha * 1.4)
        
        current_time = time.time()

    eta = 0.08

        if time.time() - t_start > 0.9 * 600:

        # Compute average gradient
        avg_grad = np.zeros_like(best_h)

            best_h = h_new

            eta *= 0.5
